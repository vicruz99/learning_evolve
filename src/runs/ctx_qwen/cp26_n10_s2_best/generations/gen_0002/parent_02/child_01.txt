import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Minimize negative sum of radii (equivalent to maximizing sum of radii)."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = []
    # Boundary constraints: circle must stay within [0,1]x[0,1]
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise non-overlap: squared distance >= squared sum of radii
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    d_sq = dx**2 + dy**2
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(d_sq[mask] - dr[mask]**2)
    
    return np.concatenate(c)

def run_single_opt(v0, bounds, cons):
    """Run SLSQP optimizer from a given initial state and return result if valid."""
    try:
        res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        curr_sum = -res.fun
        c_val = constraints(res.x)
        if np.min(c_val) >= -1e-6:
            return res.x, curr_sum
    except Exception:
        pass
    return None, -np.inf

def run_packing():

    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    inits = []
    np.random.seed(42)
    
    # 1. Hexagonal lattice variations with random shifts & rotations
    for _ in range(40):
        pts = []
        r0 = 0.095
        sy = np.random.uniform(-0.025, 0.025)
        sx = np.random.uniform(-0.025, 0.025)
        rot = np.random.uniform(-0.25, 0.25)
        y = r0 + sy
        row = 0
        while len(pts) < N:
            x_start = r0 + sx + (row % 2) * r0
            x = x_start
            while x <= 1 - r0 and len(pts) < N:
                pts.append([x, y])
                x += 2 * r0
            y += np.sqrt(3) * r0
            row += 1
        pts = np.array(pts[:N])
        if rot != 0:
            c = np.mean(pts, axis=0)
            pts = pts - c
            c_rot = np.array([[np.cos(rot), -np.sin(rot)], [np.sin(rot), np.cos(rot)]])
            pts = pts @ c_rot.T
            pts = pts + c
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(np.concatenate([pts.flatten(), np.full(N, 0.04)]))

    # 2. Staggered grid variations
    for _ in range(15):
        shift = np.random.uniform(0.0, 0.05)
        pts = []
        for i in range(6):
            for j in range(5):
                if len(pts) < N:
                    pts.append([0.1 + i*0.15 + shift, 0.1 + j*0.16 + shift])
        pts = np.array(pts[:N])
        pts += np.random.uniform(-0.01, 0.01, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(np.concatenate([pts.flatten(), np.full(N, 0.04)]))

    # 3. Random dense scatter
    for _ in range(10):
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        inits.append(np.concatenate([pts.flatten(), np.full(N, 0.03)]))

    # Phase 1: Multi-start exploration
    for v0 in inits:
        v_res, s_res = run_single_opt(v0, bounds, cons)
        if s_res > best_sum:
            best_sum = s_res
            best_v = v_res

    # Phase 2: Local perturbation to escape shallow minima
    if best_v is not None:
        for _ in range(25):
            v_pert = best_v.copy()
            v_pert[:2*N] += np.random.uniform(-0.004, 0.004, 2*N)
            v_pert[:2*N] = np.clip(v_pert[:2*N], 0.01, 0.99)
            v_pert[2*N:] *= 0.97  # Shrink radii to guarantee feasibility after perturbation

                best_sum = s_res
                best_v = v_res

            if best_v is None: break

    # Fallback initialization
    if best_v is None:

    cy = best_v[N:2*N]

    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))

    cr = np.maximum(cr, 0.0)

    for _ in range(30):

                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
