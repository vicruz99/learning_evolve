N = 26
# Precompute pair indices for efficient constraint evaluation
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Minimize negative sum of radii to maximize total radius."""
    return -np.sum(v[2*N:])

def constraints(v):
    """
    Compute inequality constraints: g(v) >= 0.
    1. Boundary: r <= x <= 1-r, r <= y <= 1-r
    2. Non-overlap: dist^2 >= (r_i + r_j)^2
    Returns a flat array of constraint values.
    """
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]

    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r

    dy = y[PAIR_I] - y[PAIR_J]

    return c
