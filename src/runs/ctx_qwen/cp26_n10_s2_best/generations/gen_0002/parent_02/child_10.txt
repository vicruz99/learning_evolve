import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v, p_i, p_j):
    """Vectorized inequality constraints: boundaries and non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(p_i))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[p_i] - x[p_j]
    dy = y[p_i] - y[p_j]
    c[4*N:] = dx**2 + dy**2 - (r[p_i] + r[p_j])**2
    
    return c

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on current center positions."""
    r = np.full(N, 0.15)
    for i in range(N):
        # Distance to walls
        wall_d = np.min([centers[i, 0], 1.0 - centers[i, 0], 
                         centers[i, 1], 1.0 - centers[i, 1]])
        r[i] = wall_d
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            val = d / 2.0
            if val < r[i]:
                r[i] = val
            if val < r[j]:
                r[j] = val
    return r * 0.82  # Leave slack for optimizer to expand

def generate_initial_configs():
    """Generate diverse starting configurations."""
    configs = []
    
    # 1. Hexagonal lattices with varying rotations
    for rot in [0.0, 0.05, -0.05, 0.12, -0.12, 0.2]:
        pts = []
        y = 0.075
        row = 0
        while len(pts) < N + 10:
            x = 0.075 + (row % 2) * 0.075
            while x < 0.95 and len(pts) < N + 10:
                pts.append([x, y])
                x += 0.15
            y += 0.13
            row += 1
        pts = np.array(pts[:N])
        
        # Apply rotation around center
        if rot != 0.0:
            c, s = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
        configs.append(np.clip(pts, 0.02, 0.98))
        
    # 2. Uniform grids
    for n_grid in [5, 6]:
        pts = np.array([[i/(n_grid-1)*0.86 + 0.07, j/(n_grid-1)*0.86 + 0.07] 
                        for i in range(n_grid) for j in range(n_grid)])[:N]
        configs.append(pts)
        
    # 3. Random scatter
    np.random.seed(123)
    for _ in range(6):
        configs.append(np.random.uniform(0.05, 0.95, (N, 2)))
        
    # 4. Boundary-hugging
    for _ in range(4):
        np.random.seed(_ + 500)
        pts = []
        # Place some near edges
        for _ in range(N//2):
            side = np.random.randint(4)
            if side == 0: pts.append([np.random.uniform(0.02, 0.98), 0.02])
            elif side == 1: pts.append([np.random.uniform(0.02, 0.98), 0.98])
            elif side == 2: pts.append([0.02, np.random.uniform(0.02, 0.98)])
            else: pts.append([0.98, np.random.uniform(0.02, 0.98)])
        # Fill rest randomly
        pts.extend(np.random.uniform(0.1, 0.9, (N - N//2, 2)))
        configs.append(np.array(pts[:N]))
        
    return configs

def run_packing():
    p_i, p_j = PAIR_I, PAIR_J
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints, 'args': (p_i, p_j)}
    
    best_sum = -1.0
    best_v = None
    
    # Phase 1: Multi-start exploration
    initial_configs = generate_initial_configs()
    
    for idx, centers in enumerate(initial_configs):
        np.random.seed(idx * 17)
        # Add controlled jitter
        centers_jit = centers + np.random.uniform(-0.015, 0.015, centers.shape)
        centers_jit = np.clip(centers_jit, 0.01, 0.99)
        
        r_init = get_feasible_radii(centers_jit)
        v0 = np.concatenate([centers_jit[:, 0], centers_jit[:, 1], r_init])
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            
            curr_sum = -res.fun
            cons_vals = constraints(res.x, p_i, p_j)
            if np.min(cons_vals) >= -1e-6 and curr_sum > best_sum:
                best_sum = curr_sum
                best_v = res.x.copy()
        except Exception:
            continue
            
    # Fallback if all fail
    if best_v is None:
        v0 = np.concatenate([np.random.uniform(0.1, 0.9, N), np.random.uniform(0.1, 0.9, N), np.full(N, 0.05)])
        res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-12})
        best_v = res.x
        best_sum = -res.fun
        
    # Phase 2: Iterative local search refinement
    no_improve_count = 0
    max_refinements = 20
    
    for _ in range(max_refinements):
        if no_improve_count >= 5:
            break
            
        np.random.seed(_ * 31 + 7)
        v_pert = best_v.copy()
        
        # Perturb centers slightly
        v_pert[:2*N] += np.random.uniform(-0.006, 0.006, 2*N)
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.01, 0.99)
        
        # Shrink radii to ensure feasibility after perturbation
        v_pert[2*N:] *= 0.95
        
        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            
            curr_sum = -res.fun
            cons_vals = constraints(res.x, p_i, p_j)
            if np.min(cons_vals) >= -1e-6 and curr_sum > best_sum:
                best_sum = curr_sum
                best_v = res.x.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        except Exception:
            no_improve_count += 1
            
    # Extract best configuration
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:]
    
    # Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    
    # 2. Enforce non-overlap constraints iteratively
    for _ in range(5):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                if cr[i] + cr[j] > d - 1e-9:
                    shrink = (cr[i] + cr[j] - d) / 2.0 + 1e-7
                    cr[i] -= shrink
                    cr[j] -= shrink
                    changed = True
        if not changed:
            break
        cr = np.maximum(cr, 0.0)
        
    # Final boundary clamp
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    
    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
