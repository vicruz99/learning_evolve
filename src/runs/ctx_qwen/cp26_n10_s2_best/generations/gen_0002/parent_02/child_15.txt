import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """
    Computes all inequality constraints: g(v) >= 0.
    1. Boundary: r <= x <= 1-r, r <= y <= 1-r
    2. Non-overlap: dist(i,j) >= r_i + r_j
    """
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    d = np.hypot(dx, dy)
    r_sum = r[i_idx] + r[j_idx]
    
    return np.concatenate([c, d - r_sum])

def generate_initial_configs(n_seeds):
    """Generate a diverse set of initial center configurations."""
    configs = []
    
    # 1. Hexagonal lattices with varied densities and rotations
    for dens in [0.08, 0.09, 0.10, 0.11, 0.12]:
        for rot in [0.0, 0.15, 0.3, 0.5, 1.0]:
            pts = []
            y = dens
            row = 0
            while len(pts) < N + 10:
                x_start = dens if row % 2 == 0 else 2 * dens
                x = x_start

                    pts.append([x, y])
                    x += 2 * dens
                y += dens * np.sqrt(3)
                row += 1
            pts = np.array(pts[:N])
            
            # Rotate around center (0.5, 0.5)
            pts_centered = pts - 0.5

            pts_rot = pts_centered @ np.array([[cos_r, -sin_r], [sin_r, cos_r]])
            pts = pts_rot + 0.5
            pts = np.clip(pts, 0.02, 0.98)
            configs.append(pts)
            
    # 2. Grid configurations
    for step in [0.12, 0.14, 0.16]:

        configs.append(grid[idx] * 0.9 + 0.05)
        
    # 3. Random scatters
    for _ in range(n_seeds):
        configs.append(np.random.uniform(0.05, 0.95, (N, 2)))
        
    # 4. Corner-focused
    corners = np.array([[0.12,0.12], [0.88,0.12], [0.12,0.88], [0.88,0.88]])
    rest = np.random.uniform(0.25, 0.75, (N-4, 2))
    configs.append(np.vstack([corners, rest]))
        
    return configs

def ensure_feasible_r(centers, r_vals):
    """Quickly compute a strictly feasible radius set for given centers."""
    r = r_vals.copy()
    for i in range(N):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(N):
            if i == j: continue
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            if d/2.0 < mx:
                mx = d/2.0
        r[i] = mx * 0.85  # Leave slack for optimizer
    return r

def run_packing():
    np.random.seed(42)
    
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    # Phase 1: Multi-start optimization
    init_configs = generate_initial_configs(n_seeds=8)
    
    for centers in init_configs:

        v0 = np.concatenate([centers[:,0], centers[:,1], r_init])
        
        try:

    radii = np.minimum(radii, 1.0 - centers[:,0])

    radii = np.minimum(radii, centers[:,1])
    radii = np.minimum(radii, 1.0 - centers[:,1])
    
    return centers, radii, float(np.sum(radii))
