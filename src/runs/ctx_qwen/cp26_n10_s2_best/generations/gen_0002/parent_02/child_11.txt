

<strategy>
The current approach uses SLSQP to jointly optimize centers and radii. While effective, SLSQP can get stuck in local optima, and the radii subproblem is actually a linear program (LP) for fixed centers. By decoupling the problem, we can significantly improve the solution:

1. **LP for Radii**: For any fixed set of centers, finding the maximum sum of radii is a linear programming problem. Constraints are linear: $r_i \le \text{dist_to_walls}_i$ and $r_i + r_j \le \text{dist}_{ij}$. This can be solved exactly and efficiently using `scipy.optimize.linprog` with the 'highs' solver.
2. **Hybrid Optimization**:
   - **Phase 1**: Use multiple diverse initial configurations (hexagonal, grid, random, corner-biased) and run SLSQP to find high-quality basins. Initialize radii in SLSQP using the LP solution to guarantee feasibility and a strong starting point.
   - **Phase 2 (Polish)**: Perform a local search on the centers only. At each step, perturb a random center, solve the LP to get the optimal radii and their sum, and accept the move if it improves the objective. This avoids the non-convexity trap of jointly optimizing radii and centers. Occasional larger jumps help escape shallow local minima.
   - **Phase 3**: Final strict post-processing to guarantee validator tolerance compliance.
3. This strategy leverages the exact convex structure of the radii subproblem while using stochastic search to navigate the non-convex center landscape, reliably surpassing previous SLSQP-only results.

import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the linear program to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Maximize sum(r) -> Minimize -sum(r)
    
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[idx, i] = 1.0
            b_ub[idx] = b
            idx += 1
            
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def objective_slsqp(v, n):
    return -np.sum(v[2 * n:])

def constraints_slsqp(v, n, pi, pj):
    x = v[:n]
    y = v[n:2 * n]
    r = v[2 * n:]
    
    cons = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    dx = x[pi] - x[pj]
    dy = y[pi] - y[pj]
    dist_sq = dx**2 + dy**2
    r_sum_sq = (r[pi] + r[pj])**2
    cons.append(dist_sq - r_sum_sq)
    
    return np.concatenate(cons)

def run_packing():
    n = 26
    pi, pj = np.triu_indices(n, k=1)
    bounds_xy = [(0.0, 1.0)] * (2 * n)
    bounds_r = [(0.0, 0.5)] * n
    bounds = bounds_xy + bounds_r
    
    best_sum = -1.0
    best_centers = None
    
    # Generate diverse initial configurations
    configs = []
    
    # 1. Hexagonal lattice
    r_hex = 0.095
    pts = []
    y = r_hex
    row = 0
    while len(pts) < n + 10:
        x_start = r_hex if row % 2 == 0 else 2 * r_hex
        x = x_start
        while x <= 1.0 - r_hex and len(pts) < n + 10:
            pts.append([x, y])
            x += 2 * r_hex
        y += r_hex * np.sqrt(3)
        row += 1
    configs.append(np.array(pts[:n]))
    
    # 2. Uniform grid
    pts_grid = []
    for i in range(6):
        for j in range(5):
            if len(pts_grid) < n:
                pts_grid.append([0.1 + i * 0.16, 0.1 + j * 0.2])
    configs.append(np.array(pts_grid))
    
    # 3. Random scatter
    np.random.seed(42)
    for seed in range(6):
        np.random.seed(seed)
        configs.append(np.random.uniform(0.1, 0.9, (n, 2)))
        
    # 4. Corner/edge biased
    pts_custom = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85], [0.5, 0.5]]
    for seed in range(5):
        np.random.seed(100 + seed)
        pts_c = pts_custom + list(np.random.uniform(0.2, 0.8, (n - 5, 2)))
        configs.append(np.array(pts_c))

    # Phase 1: SLSQP optimization from diverse starts
    for cfg in configs:
        centers_p = cfg + np.random.uniform(-0.02, 0.02, cfg.shape)
        centers_p = np.clip(centers_p, 0.02, 0.98)
        
        # Initialize radii via LP for guaranteed feasibility and strong start
        r_init, _ = solve_lp_radii(centers_p)
        r_init *= 0.92  # Slight shrink to give SLSQP breathing room
        
        v0 = np.concatenate([centers_p[:, 0], centers_p[:, 1], r_init])
        
        try:
            res = minimize(objective_slsqp, v0, args=(n,), method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraints_slsqp, 'args': (n, pi, pj)},
                           options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            
            curr_sum = -res.fun
            if curr_sum > best_sum:
                c_vals = constraints_slsqp(res.x, n, pi, pj)
                if np.min(c_vals) >= -1e-5:
                    best_sum = curr_sum
                    best_centers = res.x[:2 * n].reshape(n, 2).copy()
        except Exception:
            pass
            
    if best_centers is None:
        best_centers = configs[0]
        
    # Phase 2: Local search on centers using LP for exact radii evaluation
    np.random.seed(123)
    current_sum, current_radii = solve_lp_radii(best_centers)
    
    # Hill climbing with occasional large jumps to escape local minima
    for step in range(2000):
        idx = np.random.randint(n)
        new_c = best_centers.copy()
        new_c[idx] += np.random.uniform(-0.004, 0.004, 2)
        new_c[idx] = np.clip(new_c[idx], 0.0, 1.0)
        
        new_r, new_s = solve_lp_radii(new_c)
        
        if new_s > current_sum:
            best_centers = new_c
            current_sum = new_s
            current_radii = new_r
            
        # Occasional large perturbation
        if np.random.random() < 0.04:
            jump_c = best_centers.copy()
            jump_c += np.random.uniform(-0.015, 0.015, (n, 2))
            jump_c = np.clip(jump_c, 0.0, 1.0)
            jr, js = solve_lp_radii(jump_c)
            if js > current_sum:
                best_centers = jump_c
                current_sum = js
                current_radii = jr
                
    # Phase 3: Final LP evaluation and strict post-processing
    final_radii, final_sum = solve_lp_radii(best_centers)
    
    # Enforce boundary constraints strictly
    for i in range(n):
        cx, cy = best_centers[i]
        mr = min(cx, 1.0 - cx, cy, 1.0 - cy)
        if final_radii[i] > mr:
            final_radii[i] = mr
            
    # Enforce non-overlap strictly with safety margin
    for _ in range(5):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                if d < final_radii[i] + final_radii[j] - 1e-10:
                    shrink = (final_radii[i] + final_radii[j] - d) / 2.0 + 1e-9
                    final_radii[i] -= shrink
                    final_radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    final_radii = np.maximum(final_radii, 0.0)
    
    return best_centers, final_radii, float(np.sum(final_radii))
