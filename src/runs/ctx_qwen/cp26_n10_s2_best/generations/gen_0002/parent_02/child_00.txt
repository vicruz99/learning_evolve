import numpy as np
from scipy.optimize import minimize

N = 26
# Precompute mask for unique pairwise constraints to avoid redundant calculations
TRI_MASK = np.triu(np.ones((N, N), dtype=bool), k=1)

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Vectorized inequality constraints: boundaries and non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: circle must stay inside [0,1]x[0,1]
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    d_sq = dx**2 + dy**2
    c.append(d_sq[TRI_MASK] - dr[TRI_MASK]**2)
    
    return np.concatenate(c)

def get_max_feasible_radii(centers):
    """Compute the exact maximum feasible radii for a given set of centers."""
    n = centers.shape[0]
    r = np.full(n, 0.5)
    for i in range(n):
        # Boundary limits
        r[i] = min(r[i], centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        # Pairwise limits
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            half_d = d * 0.5
            if half_d < r[i]:
                r[i] = half_d
            if half_d < r[j]:
                r[j] = half_d
    return r

def run_packing():
    # Variable bounds: x,y in [0,1], r in [0,0.5]
    bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    inits = []
    
    # 1. Rotated Hexagonal Lattices with varying densities
    for seed in range(8):
        np.random.seed(seed)
        r_est = 0.085 + seed * 0.005
        pts = []
        y = r_est
        row = 0
        while len(pts) < N + 5:
            x_start = r_est + (row % 2) * r_est
            x = x_start
            while x <= 1 - r_est:
                pts.append([x, y])
                x += 2 * r_est
            y += r_est * np.sqrt(3)
            row += 1
            
        pts = np.array(pts[:N])
        # Random rotation and translation to escape symmetry
        angle = np.random.uniform(-0.3, 0.3)
        shift = np.random.uniform(-0.05, 0.05, 2)
        c_val, s_val = np.cos(angle), np.sin(angle)
        pts = pts @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5 + shift
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    # 2. Random Uniform Distributions
    for seed in range(6):
        np.random.seed(seed + 100)
        inits.append(np.random.uniform(0.05, 0.95, (N, 2)))
        
    # 3. Structured Grids with Jitter
    for seed in range(4):
        np.random.seed(seed + 200)
        pts = np.array([[i*0.17 + 0.1, j*0.17 + 0.1] for i in range(6) for j in range(5)])
        idx = np.random.choice(len(pts), N, replace=False)
        pts = pts[idx] + np.random.uniform(-0.015, 0.015, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    # 4. Concentric Ring Patterns
    for seed in range(4):
        np.random.seed(seed + 300)
        pts = [np.array([0.5, 0.5])]
        for ring in range(1, 6):
            n_in_ring = 8 * ring
            angles = np.linspace(0, 2*np.pi, n_in_ring, endpoint=False) + np.random.uniform(0, 0.3)
            rad = ring * 0.14
            for a in angles:
                if len(pts) >= N: break
                pts.append([0.5 + rad*np.cos(a), 0.5 + rad*np.sin(a)])
        pts = np.array(pts[:N])
        pts += np.random.uniform(-0.01, 0.01, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)

    # Phase 1: Multi-start Optimization
    for centers in inits:
        # Start close to the feasible boundary to guide optimizer efficiently
        r_init = get_max_feasible_radii(centers) * 0.88
        v0 = np.concatenate([centers[:,0], centers[:,1], r_init])
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            
            curr_sum = -res.fun
            # Accept if significantly feasible and better
            cons_vals = constraints(res.x)
            if np.min(cons_vals) >= -1e-5 and curr_sum > best_sum:
                best_sum = curr_sum
                best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Basin Hopping / Local Refinement
    if best_v is not None:
        for trial in range(20):
            np.random.seed(trial * 777 + 42)
            v_pert = best_v.copy()
            # Perturb centers to explore neighboring basins
            v_pert[:2*N] += np.random.uniform(-0.008, 0.008, 2*N)
            v_pert[:2*N] = np.clip(v_pert[:2*N], 0.01, 0.99)
            # Shrink radii slightly to ensure strict feasibility after perturbation
            v_pert[2*N:] *= 0.93
            
            try:
                res = minimize(objective, v_pert, method='SLSQP', bounds=bounds,
                               constraints=cons_dict,
                               options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                
                curr_sum = -res.fun
                cons_vals = constraints(res.x)
                if np.min(cons_vals) >= -1e-5 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_v = res.x.copy()
            except Exception:
                pass
                
    # Fallback safety net
    if best_v is None:
        centers = np.random.uniform(0.1, 0.9, (N, 2))
        r = get_max_feasible_radii(centers) * 0.5
        best_v = np.concatenate([centers[:,0], centers[:,1], r])
        
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()
    
    # Phase 3: Strict Projection to Guarantee Validator Compliance
    # Iteratively project onto feasible set with minimal radius loss
    for _ in range(10):
        # Enforce boundary constraints strictly
        cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
        cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
        
        # Enforce non-overlap constraints strictly
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(cx[i]-cx[j], cy[i]-cy[j])
                if d < cr[i] + cr[j] - 1e-9:
                    # Shared shrinkage to maintain balance
                    shrink = (cr[i] + cr[j] - d) / 2.0 + 1e-8
                    cr[i] -= shrink
                    cr[j] -= shrink
                    changed = True
        if not changed:
            break
        cr = np.maximum(cr, 0.0)
        
    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
