Key improvements in this approach:
1. **Feasible Initialization**: Before passing configurations to the optimizer, we explicitly compute strictly feasible radii based on inter-circle distances and boundary distances. This prevents `SLSQP` from wasting iterations searching for feasibility.
2. **Diverse Multi-Start Configurations**: We generate initial center layouts from hexagonal lattices, square grids, and random scatters, each scaled and jittered. This covers multiple basins of attraction.
3. **Two-Phase Optimization**:

   - Phase 2 performs iterative local perturbation: we shrink radii slightly, perturb centers, re-enforce feasibility, and re-optimize. This helps escape local optima by temporarily creating "breathing room".
4. **Squared Distance Constraints**: Using `dist^2 >= (r_i + r_j)^2` avoids square roots in constraints, improving numerical stability and gradient approximation for `SLSQP`.
5. **Strict Post-Processing**: After optimization, we iteratively shrink overlapping circles and clamp to boundaries to guarantee the validator passes with floating-point safety margins.

import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    d_sq = dx**2 + dy**2
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, d_sq[mask] - dr[mask]**2])
    return c

def make_feasible_radii(centers, base_r=0.10, shrink_factor=0.85):
    """Adjust radii to guarantee strict feasibility for given centers."""
    r = np.full(N, base_r)
    for _ in range(5):
        for i in range(N):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0], 
                        centers[i, 1], 1.0 - centers[i, 1])
            for j in range(N):
                if i != j:
                    d = np.hypot(centers[i, 0] - centers[j, 0], 
                                 centers[i, 1] - centers[j, 1])
                    max_r = min(max_r, d / 2.0)
            r[i] = min(r[i], max_r)
    return r * shrink_factor

def get_initial_configs():
    """Generate diverse initial configurations."""
    configs = []
    
    # 1. Hexagonal lattice variations
    for scale in [0.95, 1.0, 1.05]:
        pts = []
        y = 0.05
        row = 0
        while len(pts) < N + 5:
            x = 0.05 if row % 2 == 0 else 0.12
            while x <= 0.95 and len(pts) < N + 5:
                pts.append([x, y])
                x += 0.18
            y += 0.15
            row += 1
        pts = np.array(pts[:N]) * scale
        pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts)
        
    # 2. Square grid variations
    for scale in [0.95, 1.0, 1.05]:
        pts = np.array([[i*0.18+0.09, j*0.18+0.09] for i in range(6) for j in range(5)])[:N]
        pts = pts * scale
        pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts)
        
    # 3. Random scatter
    for seed in range(10):
        np.random.seed(seed)
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        configs.append(pts)
        
    return configs

def run_packing():
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    np.random.seed(42)
    
    inits = get_initial_configs()
    
    # Phase 1: Multi-start optimization
    for idx, base_centers in enumerate(inits):
        # Add jitter to break symmetry
        centers = base_centers + np.random.uniform(-0.015, 0.015, base_centers.shape)
        centers = np.clip(centers, 0.03, 0.97)
        
        # Ensure initial feasibility
        r_init = make_feasible_radii(centers, base_r=0.10, shrink_factor=0.85)
        v0 = np.concatenate([centers[:, 0], centers[:, 1], r_init])
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            
            curr_sum = -res.fun
            c_vals = constraints(res.x)
            if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                best_sum = curr_sum
                best_v = res.x.copy()
        except Exception:
            continue
            
    if best_v is None:
        # Fallback
        centers = np.random.uniform(0.2, 0.8, (N, 2))
        r_init = make_feasible_radii(centers, base_r=0.08, shrink_factor=0.9)
        best_v = np.concatenate([centers[:, 0], centers[:, 1], r_init])
        
    # Phase 2: Local perturbation refinement
    current_v = best_v.copy()
    for step in range(12):
        scale = 0.97 - step * 0.008
        if scale < 0.85:
            break
            
        v_pert = current_v.copy()
        v_pert[:2*N] += np.random.uniform(-0.006, 0.006, 2*N)
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.02, 0.98)
        v_pert[2*N:] *= scale
        
        # Re-ensure feasibility before optimization
        pts_pert = v_pert[:2*N].reshape(N, 2)
        r_pert = make_feasible_radii(pts_pert, base_r=np.max(v_pert[2*N:]), shrink_factor=0.95)
        v_pert[2*N:] = r_pert
        
        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
            
            curr_sum = -res.fun
            if curr_sum > best_sum and np.min(constraints(res.x)) >= -1e-6:
                best_sum = curr_sum
                best_v = res.x.copy()
                current_v = best_v
        except Exception:
            continue
            
    # Extract results
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:]
    
    # Phase 3: Strict post-processing for validator compliance
    # Enforce boundaries
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    
    # Enforce non-overlap iteratively
    for _ in range(10):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                if cr[i] + cr[j] > d + 1e-9:
                    shrink = (cr[i] + cr[j] - d) / 2.0 + 1e-8
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
