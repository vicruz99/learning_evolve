import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and non-overlap using squared distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = []
    # Boundary constraints
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    dist_sq = dx**2 + dy**2
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(dist_sq[mask] - dr[mask]**2)
    
    return np.concatenate(c)

def get_feasible_radii(centers):
    """Compute strictly feasible radii for a given set of centers."""
    x = centers[:, 0]
    y = centers[:, 1]
    r = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    for _ in range(3):
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(x[i] - x[j], y[i] - y[j])
                if d < r[i] + r[j]:
                    shrink = (r[i] + r[j] - d) / 2.0
                    r[i] -= shrink
                    r[j] -= shrink
    return np.maximum(r, 1e-6)

                    if 0.02 <= px <= 0.98 and 0.02 <= py <= 0.98:

                    x += 2 * r0

                row += 1

    radii = np.minimum(radii, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
