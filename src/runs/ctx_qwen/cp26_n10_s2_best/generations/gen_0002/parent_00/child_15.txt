import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v, i_idx, j_idx):
    """Compute inequality constraints: boundaries and pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    cons = []
    # Boundary constraints: circles inside [0,1]x[0,1]
    cons.append(x - r)
    cons.append(1.0 - x - r)
    cons.append(y - r)
    cons.append(1.0 - y - r)
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dists = np.hypot(dx, dy)
    r_sum = r[i_idx] + r[j_idx]
    cons.append(dists - r_sum)
    
    return np.concatenate(cons)

def generate_hex_lattice(rho):
    """Generate points on a hexagonal lattice."""
    pts = []
    y = rho
    row = 0
    while len(pts) < N + 5:
        x_start = rho if row % 2 == 0 else 2 * rho
        x = x_start
        while x <= 1.0 - rho:
            pts.append([x, y])
            x += 2 * rho
        y += np.sqrt(3) * rho
        row += 1
    return np.array(pts[:N])

def generate_grid(s):
    """Generate points on a shifted square grid."""
    pts = []
    for i in range(6):
        for j in range(5):
            if len(pts) < N:
                pts.append([s + i * 0.15, s + j * 0.18])
    return np.array(pts[:N])
