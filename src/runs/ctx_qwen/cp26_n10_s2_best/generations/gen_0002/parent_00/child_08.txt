import numpy as np
from scipy.optimize import minimize

N = 26
# Precompute pair indices for vectorized constraint evaluation
PI, PJ = np.triu_indices(N, k=1)

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and pairwise non-overlap."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    cons = [
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r,
        np.sqrt(np.sum((c[PI] - c[PJ])**2, axis=1)) - (r[PI] + r[PJ])
    ]
    return np.concatenate(cons)

def smooth_penalty(v, lam):
    """Smooth penalty objective for global exploration."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    val = -np.sum(r)
    
    # Boundary penalties
    val += lam * (np.sum(np.maximum(0, r - c[:,0])**2) +
                  np.sum(np.maximum(0, r - (1.0 - c[:,0]))**2) +
                  np.sum(np.maximum(0, r - c[:,1])**2) +
                  np.sum(np.maximum(0, r - (1.0 - c[:,1]))**2))
                  
    # Overlap penalties
    d = np.sqrt(np.sum((c[PI] - c[PJ])**2, axis=1))
    ov = np.maximum(0, r[PI] + r[PJ] - d)
    val += lam * np.sum(ov**2)
    return val

def run_packing():

    best_sum = -1.0
    best_v = None

    starts = []

    for s in range(12):
        np.random.seed(s)
        r0 = 0.095
        pts = []
        y = r0
        row = 0
        while len(pts) < N:
            x_s = r0 if row % 2 == 0 else 2 * r0
            x = x_s
            while x <= 1 - r0 and len(pts) < N:
                pts.append([x, y])
                x += 2 * r0
            y += np.sqrt(3) * r0
            row += 1
        pts = np.array(pts[:N])
        pts += np.random.uniform(-0.02, 0.02, pts.shape)
        pts = np.clip(pts, 0.05, 0.95)
        starts.append(pts)

    # 2. Random dense clusters
    for s in range(13):
        np.random.seed(100 + s)
        starts.append(np.random.uniform(0.1, 0.9, (N, 2)))

    # Optimization pipeline
    for centers in starts:
        r_init = np.full(N, 0.04)
        v0 = np.concatenate([centers.flatten(), r_init])

        curr = v0.copy()

            curr = res.x

                       constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': 5000, 'ftol': 1e-12})

        # Accept if better and sufficiently feasible
        if -res.fun > best_sum:

            if np.min(c_val) >= -1e-7:
                best_sum = -res.fun
                best_v = res.x.copy()

    # Extract final configuration
    centers = best_v[:2*N].reshape(N, 2)
    radii = best_v[2*N:].copy()

    # Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
