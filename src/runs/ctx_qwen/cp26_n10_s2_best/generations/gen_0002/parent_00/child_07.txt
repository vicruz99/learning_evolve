def objective_func(v, n):

def constraint_func(v, n, pi, pj):
    """Compute inequality constraints: boundaries and non-overlap."""
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    bc = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])

    dx = c[pi, 0] - c[pj, 0]
    dy = c[pi, 1] - c[pj, 1]
    dist_sq = dx**2 + dy**2
    r_sum_sq = (r[pi] + r[pj])**2
    
    return np.concatenate([bc, dist_sq - r_sum_sq])

def compute_feasible_radii(centers, n):

    r = np.full(n, 0.5)
    for i in range(n):
