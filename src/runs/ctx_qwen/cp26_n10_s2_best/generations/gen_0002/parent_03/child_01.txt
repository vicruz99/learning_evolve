2. **Diverse Multi-Start**: We seed optimizations from hexagonal lattices, square grids, and random placements, covering different structural basins of attraction.

4. **Squared Distance Constraints**: Using `dist^2 >= (r_i + r_j)^2` avoids square roots in constraints, improving numerical stability and gradient quality for SLSQP.

import numpy as np
from scipy.optimize import minimize

N = 26
# Precompute pairwise indices for efficient constraint evaluation
p_i, p_j = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and non-overlap (squared distances)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(p_i))
    # Boundary: r <= x <= 1-r, r <= y <= 1-r
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[p_i] - x[p_j]
    dy = y[p_i] - y[p_j]
    c[4*N:] = dx**2 + dy**2 - (r[p_i] + r[p_j])**2
    return c

def get_initial_guess(seed, layout='hex'):
    """Generates a strictly feasible initial configuration."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    
    if layout == 'hex':
        r0 = 0.090 + seed * 0.0015
        pts = []
        y = r0
        row = 0
        while len(pts) < N + 10:
            xs = r0 if row % 2 == 0 else 2 * r0
            x = xs
            while x <= 1.0 - r0 and len(pts) < N + 10:
                pts.append([x, y])
                x += 2 * r0
            y += r0 * np.sqrt(3)
            row += 1
        pts = np.array(pts)
        idx = np.random.choice(len(pts), N, replace=False)
        centers = pts[idx]
    elif layout == 'grid':
        pts = []
        for i in range(6):
            for j in range(5):
                pts.append([0.08 + i * 0.16, 0.08 + j * 0.18])
        pts = np.array(pts)
        idx = np.random.choice(len(pts), N, replace=False)
        centers = pts[idx]
    else:  # random
        centers = np.random.uniform(0.12, 0.88, (N, 2))
        
    # Jitter to break symmetry
    centers += np.random.uniform(-0.012, 0.012, centers.shape)
    centers = np.clip(centers, 0.025, 0.975)
    
    # Compute maximum feasible radius for each circle given centers
    r_max = np.full(N, 0.45)
    for i in range(N):
        # Boundary distance
        mx = min(centers[i, 0], 1.0 - centers[i, 0], 
                 centers[i, 1], 1.0 - centers[i, 1])
        if mx < r_max[i]:
            r_max[i] = mx
        # Pairwise distances
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            val = d / 2.0
            if val < r_max[i]: r_max[i] = val
            if val < r_max[j]: r_max[j] = val
            
    v = np.zeros(3 * N)
    v[:N] = centers[:, 0]
    v[N:2*N] = centers[:, 1]
    # Start at 65% of feasible max to guarantee strict feasibility + slack for expansion
    v[2*N:] = r_max * 0.65
    return v

def run_packing():
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.45)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_val = -np.inf
    best_v = None
    
    # Phase 1: Multi-start exploration from structured & random layouts
    layouts = ['hex', 'grid', 'rand']
    for seed in range(45):
        layout = layouts[seed % 3]
        v0 = get_initial_guess(seed, layout)
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, 
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_val:
                # Accept if sufficiently feasible
                c_val = constraints(res.x)
                if np.all(c_val >= -1e-5):
                    best_val = -res.fun
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Fallback if optimization fails
    if best_v is None:
        best_v = get_initial_guess(0, 'hex')
        
    # Phase 2: Iterative perturbation & radius expansion to escape local minima
    current_v = best_v
    for step in range(30):
        # Perturb centers
        c_pert = current_v[:2*N] + np.random.uniform(-0.008, 0.008, 2*N)
        c_pert = np.clip(c_pert, 0.015, 0.985)
        
        # Gradually push radii larger to force tighter packing
        r_pert = current_v[2*N:] * (1.0 + 0.003 * step)
        
        # Enforce rough feasibility to guide optimizer back into basin
        for i in range(N):
            mx = min(c_pert[i], 1.0 - c_pert[i], c_pert[N+i], 1.0 - c_pert[N+i])
            if r_pert[i] > mx: r_pert[i] = mx
            for j in range(i + 1, N):
                d = np.hypot(c_pert[i] - c_pert[j], c_pert[N+i] - c_pert[N+j])
                if r_pert[i] + r_pert[j] > d:
                    shrink = (r_pert[i] + r_pert[j] - d) / 2.0 + 1e-7
                    r_pert[i] = max(0.0, r_pert[i] - shrink)
                    r_pert[j] = max(0.0, r_pert[j] - shrink)
                    
        v0 = np.concatenate([c_pert, r_pert])
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, 
                           options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_val:
                c_val = constraints(res.x)
                if np.all(c_val >= -1e-5):
                    best_val = -res.fun
                    best_v = res.x.copy()
                    current_v = best_v
        except Exception:
            pass
            
    # Extract final configuration
    centers = best_v[:2*N].reshape(N, 2)
    radii = best_v[2*N:].copy()
    
    # Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints strictly
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], 
                 centers[i, 1], 1.0 - centers[i, 1])
        if radii[i] > mx:
            radii[i] = mx
            
    # 2. Enforce non-overlap constraints iteratively with safety margin
    for _ in range(15):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-7
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
