import numpy as np
from scipy.optimize import minimize

def objective(v, N):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v, N, pi, pj):
    """
    Computes inequality constraints: boundaries and pairwise non-overlap.
    Returns array where each element >= 0 indicates satisfied constraint.
    Uses squared distances for numerical stability.
    """
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r,
        (x[pi] - x[pj])**2 + (y[pi] - y[pj])**2 - (r[pi] + r[pj])**2
    ])
    return c

def rotate_points(centers, angle, cx=0.5, cy=0.5):
    """Rotate a set of 2D points around a center."""
    cos_a, sin_a = np.cos(angle), np.sin(angle)
    xc, yc = centers[:, 0] - cx, centers[:, 1] - cy
    xr = xc * cos_a - yc * sin_a + cx
    yr = xc * sin_a + yc * cos_a + cy
    return np.column_stack((xr, yr))

def get_initial_guess(N, method='hex', seed=0, r0=0.09, angle=0.0):

    return np.concatenate([centers[:, 0], centers[:, 1], r])
