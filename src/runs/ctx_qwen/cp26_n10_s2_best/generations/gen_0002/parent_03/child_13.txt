import numpy as np
from scipy.optimize import minimize

def objective_func(v, N):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2*N:])

def constraint_func(v, N, pair_i, pair_j):
    """Computes all inequality constraints. Returns array where elements >= 0 are satisfied."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    cons = np.empty(4*N + len(pair_i))
    cons[:N] = x - r
    cons[N:2*N] = 1.0 - x - r
    cons[2*N:3*N] = y - r
    cons[3*N:4*N] = 1.0 - y - r
    dx = x[pair_i] - x[pair_j]
    dy = y[pair_i] - y[pair_j]
    cons[4*N:] = dx**2 + dy**2 - (r[pair_i] + r[pair_j])**2
    return cons

def make_feasible(v, N):
    """Projects an infeasible configuration back to the feasible set by shrinking radii."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:].copy()
    # Boundary constraints
    for i in range(N):
        r[i] = min(r[i], x[i], 1.0-x[i], y[i], 1.0-y[i])
    # Pairwise non-overlap constraints
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(x[i]-x[j], y[i]-y[j])
            if r[i] + r[j] > d:
                shr = (r[i] + r[j] - d)/2.0
                r[i] -= shr
                r[j] -= shr
    r = np.maximum(r, 0.0)
    v[2*N:] = r
    return v

    N = 26

    best_sum = -1.0
    best_v = None

    # Hexagonal lattices with varying densities
    for r0 in np.linspace(0.085, 0.115, 6):
        pts = []
        y = r0
        row = 0
        while len(pts) < N + 10:

                pts.append([x, y])
                x += 2*r0
            y += r0 * np.sqrt(3)
            row += 1
        configs.append(np.array(pts[:N]))
        
    # Grid layouts
    for s in [0.05, 0.08, 0.10, 0.13]:
        pts = []
        for i in range(6):

                if len(pts) < N:
                    pts.append([s + i*0.15, s + j*0.18])
        configs.append(np.array(pts[:N]))
        
    # Random scatter
    np.random.seed(42)
    for _ in range(10):
        configs.append(np.random.uniform(0.1, 0.9, (N, 2)))

                curr_sum = -res.fun

                        best_v = res.x.copy()

    return centers, radii, float(np.sum(radii))
