import numpy as np
from scipy.optimize import minimize

def objective(v, N):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v, N, pi, pj):
    """
    Computes inequality constraints: boundaries and non-overlap.
    Returns array where all elements >= 0 indicates satisfaction.
    Uses squared distances for numerical stability.
    """
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    num_pairs = len(pi)
    c = np.empty(4*N + num_pairs)
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[pi] - x[pj]
    dy = y[pi] - y[pj]
    c[4*N:] = dx**2 + dy**2 - (r[pi] + r[pj])**2
    
    return c

def generate_initial_configs(N, num_configs):
    """Generates diverse initial configurations to explore the solution space."""
    configs = []
    
    # 1. Rotated Hexagonal Lattices
    for rot in np.linspace(0.0, np.pi/6, 6):
        for seed in range(3):
            np.random.seed(seed)
            r_est = 0.10
            centers = []
            y = r_est
            row = 0
            while len(centers) < N + 5:
                x_start = r_est + (row % 2) * r_est
                x = x_start
                while x <= 1 - r_est:
                    centers.append([x, y])
                    x += 2 * r_est
                y += r_est * np.sqrt(3)
                row += 1
            centers = np.array(centers[:N])
            
            # Rotate around center
            c_centered = centers - 0.5
            rot_mat = np.array([[np.cos(rot), -np.sin(rot)], [np.sin(rot), np.cos(rot)]])
            c_rot = (c_centered @ rot_mat.T) + 0.5
            
            # Clip and add jitter
            c_rot = np.clip(c_rot + np.random.uniform(-0.01, 0.01, c_rot.shape), 0.02, 0.98)
            r_init = np.full(N, 0.08) + np.random.uniform(-0.005, 0.005, N)
            configs.append(np.concatenate([c_rot[:,0], c_rot[:,1], r_init]))
            
    # 2. Boundary-Hugging Configurations
    for seed in range(5):
        np.random.seed(seed + 1000)
        pts = []
        # Corners
        pts.extend([[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]])
        # Mid-edges
        for k in range(4):
            pts.append([0.5, 0.12 + k*0.01])
            pts.append([0.12 + k*0.01, 0.5])
            pts.append([0.88 - k*0.01, 0.5])
            pts.append([0.5, 0.88 - k*0.01])
        # Fill remainder randomly
        while len(pts) < N:
            pts.append(np.random.uniform(0.2, 0.8, 2).tolist())
            
        pts = np.array(pts[:N]) + np.random.uniform(-0.01, 0.01, (N, 2))
        pts = np.clip(pts, 0.02, 0.98)
        r_init = np.full(N, 0.09) + np.random.uniform(-0.005, 0.005, N)
        configs.append(np.concatenate([pts[:,0], pts[:,1], r_init]))
        
    # 3. Random Dense Scatter
    for seed in range(15):
        np.random.seed(seed + 2000)
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        r_init = np.full(N, 0.085) + np.random.uniform(-0.005, 0.005, N)
        configs.append(np.concatenate([pts[:,0], pts[:,1], r_init]))
        
    # 4. Perturbed Grids
    for seed in range(5):
        np.random.seed(seed + 3000)
        pts = np.array([[i*0.18+0.1, j*0.18+0.1] for i in range(6) for j in range(5)])
        np.random.shuffle(pts)
        pts = pts[:N] + np.random.uniform(-0.008, 0.008, (N, 2))
        pts = np.clip(pts, 0.02, 0.98)
        r_init = np.full(N, 0.09) + np.random.uniform(-0.005, 0.005, N)
        configs.append(np.concatenate([pts[:,0], pts[:,1], r_init]))
        
    return configs[:num_configs]

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    N = 26
    pi, pj = np.triu_indices(N, k=1)
    
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints, 'args': (N, pi, pj)}
    
    best_v = None
    best_val = -np.inf
    
    # Phase 1: Multi-start exploration
    configs = generate_initial_configs(N, 60)
    for v0 in configs:
        try:
            res = minimize(objective, v0, args=(N,), method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2500, 'ftol': 1e-13, 'disp': False})
            curr_val = -res.fun
            if curr_val > best_val:
                # Verify strict feasibility tolerance
                c_val = constraints(res.x, N, pi, pj)
                if np.min(c_val) >= -1e-7:
                    best_val = curr_val
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        # Fallback
        centers = np.random.uniform(0.2, 0.8, (N, 2))
        radii = np.full(N, 0.05)
        return centers, radii, float(np.sum(radii))
        
    # Phase 2: Decaying perturbation refinement to escape local minima
    current_v = best_v
    for step in range(25):
        # Decay perturbation scale
        scale = 0.004 * (1.0 - step/25.0)
        pert = np.random.uniform(-scale, scale, current_v.shape)
        pert[2*N:] *= 0.3  # Smaller noise for radii
        
        v_pert = current_v + pert
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.01, 0.99)
        v_pert[2*N:] = np.clip(v_pert[2*N:], 0.01, 0.49)
        
        try:
            res = minimize(objective, v_pert, args=(N,), method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
            curr_val = -res.fun
            if curr_val > best_val:
                c_val = constraints(res.x, N, pi, pj)
                if np.min(c_val) >= -1e-7:
                    best_val = curr_val
                    best_v = res.x.copy()
                    current_v = best_v
        except Exception:
            pass
            
    # Extract best configuration
    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    radii = best_v[2*N:].copy()
    
    # Phase 3: Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints
    for i in range(N):
        mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = min(radii[i], mr)
        
    # 2. Enforce non-overlap constraints iteratively
    for _ in range(10):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d:
                    shr = (radii[i] + radii[j] - d) * 0.5 + 1e-8
                    radii[i] = max(0.0, radii[i] - shr)
                    radii[j] = max(0.0, radii[j] - shr)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
