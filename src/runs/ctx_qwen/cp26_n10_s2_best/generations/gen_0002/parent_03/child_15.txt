import numpy as np
from scipy.optimize import minimize


def objective(v, N):
    """Minimize negative sum of radii."""
    return -np.sum(v[2 * N :])


def constraints(v, N, pair_i, pair_j):
    """Inequality constraints: boundaries and pairwise non-overlap (squared)."""
    x = v[:N]
    y = v[N : 2 * N]
    r = v[2 * N :]

    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])

    dx = x[pair_i] - x[pair_j]
    dy = y[pair_i] - y[pair_j]
    r_sum = r[pair_i] + r[pair_j]

    c = np.concatenate([c, dx ** 2 + dy ** 2 - r_sum ** 2])
    return c


def make_hex_centers(N, r0, angle=0.0, shift_x=0.0, shift_y=0.0):
    """Create hexagonal lattice centers."""
    centers = []
    y = r0 + shift_y
    row = 0
    while len(centers) < N + 15:
        x_start = r0 + (row % 2) * r0
        x = x_start + shift_x
        while x <= 1 - r0 and len(centers) < N + 15:
            centers.append([x, y])
            x += 2 * r0
        y += r0 * np.sqrt(3)
        row += 1

    centers = np.array(centers[:N])

    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        c = centers - 0.5
        centers = c @ np.array([[ca, -sa], [sa, ca]]) + 0.5
        centers = np.clip(centers, 0.01, 0.99)

    return centers


def make_grid_centers(N, n_rows, n_cols):
    """Create grid-based centers."""
    centers = []
    for i in range(n_rows):
        for j in range(n_cols):
            if len(centers) >= N:
                break
            x = (j + 0.5) / n_cols
            y = (i + 0.5) / n_rows
            centers.append([x, y])
        if len(centers) >= N:
            break
    return np.array(centers[:N])


def make_corner_centers(N, seed):
    """Create corner-exploiting centers with larger corner circles."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    corners = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.87]]
    for i, c in enumerate(corners):
        centers[i] = c
    centers[4:] = np.random.uniform(0.15, 0.85, (N - 4, 2))
    return centers


def run_packing():

    # Grid configurations

                    radii[j] = max(0.0, radii[j] - ov)
                    changed = True
        if not changed:
            break

    return centers, radii, float(np.sum(radii))
