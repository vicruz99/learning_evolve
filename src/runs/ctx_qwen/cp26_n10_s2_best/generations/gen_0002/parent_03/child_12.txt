from scipy.optimize import minimize

N = 26
# Precompute upper triangular mask for pairwise constraints
MASK = np.triu(np.ones((N, N), dtype=bool), k=1)

def objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute all inequality constraints: boundaries and non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dist_sq[MASK] - r_sum[MASK]**2])
    return c

def get_hex_init(r0, seed):
    """Generate a hexagonal lattice configuration."""
    np.random.seed(seed)
    centers = []
    y = r0
    row = 0
    while len(centers) < N + 10:
        x_start = r0 if row % 2 == 0 else 2*r0
        x = x_start
        while x <= 1.0 - r0 and len(centers) < N + 10:
            centers.append([x, y])
            x += 2*r0
        y += np.sqrt(3)*r0
        row += 1
    centers = np.array(centers[:N])
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    return np.clip(centers, 0.02, 0.98)

def get_rotated_init(r0, angle, seed):

    cos_a, sin_a = np.cos(angle), np.sin(angle)
    rot = rel @ np.array([[cos_a, -sin_a], [sin_a, cos_a]]) + c

    if len(valid) >= N:
        idx = np.random.choice(len(valid), N, replace=False)
        return valid[idx]
    return None

    centers = np.random.uniform(0.08, 0.92, (N, 2))
    corners = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
