import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_pair_indices():
    i_idx, j_idx = [], []
    for i in range(N):
        for j in range(i+1, N):
            i_idx.append(i)
            j_idx.append(j)
    return np.array(i_idx), np.array(j_idx)

PAIR_I, PAIR_J = compute_pair_indices()

def get_optimal_radii_lp(centers):
    """Solves an LP to find the maximum sum of radii for fixed centers."""
    x = centers[:, 0]
    y = centers[:, 1]
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    A_ub.append(np.eye(N)); b_ub.append(x)
    A_ub.append(np.eye(N)); b_ub.append(1.0 - x)
    A_ub.append(np.eye(N)); b_ub.append(y)
    A_ub.append(np.eye(N)); b_ub.append(1.0 - y)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    pair_rows = np.zeros((len(PAIR_I), N))
    for k in range(len(PAIR_I)):
        pair_rows[k, PAIR_I[k]] = 1.0
        pair_rows[k, PAIR_J[k]] = 1.0
    A_ub.append(pair_rows)
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dists = np.sqrt(dx**2 + dy**2)
    b_ub.append(dists)
    
    A_ub = np.vstack(A_ub)
    b_ub = np.concatenate(b_ub)
    
    # Maximize sum(r) <=> Minimize -sum(r)
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*N, method='highs')
    return res.x if res.success else np.full(N, 0.02)

def objective_func(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraint_func(v):
    """Compute inequality constraints: boundaries and non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    cons = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist = np.sqrt(dx**2 + dy**2)
    cons = np.concatenate([cons, dist - r[PAIR_I] - r[PAIR_J]])
    
    return cons

def fps_init(n, seed, n_cand=1500):
    """Farthest Point Sampling initialization for well-spaced centers."""
    np.random.seed(seed)
    pts = [[0.5, 0.5]]
    for _ in range(n - 1):
        cand = np.random.uniform(0, 1, size=(n_cand, 2))
        dists = np.inf
        for p in pts:
            dists = np.minimum(dists, np.linalg.norm(cand - p, axis=1))
        # Distance to boundaries
        db = np.minimum(np.minimum(cand[:, 0], 1 - cand[:, 0]), 
                        np.minimum(cand[:, 1], 1 - cand[:, 1]))
        dists = np.minimum(dists, db)
        pts.append(cand[np.argmax(dists)])
    return np.array(pts)

def hex_init(n, r0, seed):
    """Hexagonal lattice initialization with jitter."""
    np.random.seed(seed)
    centers = []
    y = r0
    row = 0
    while len(centers) < n + 5:
        xs = r0 if row % 2 == 0 else 2 * r0
        x = xs
        while x <= 1 - r0:
            centers.append([x, y])
            x += 2 * r0
        y += r0 * np.sqrt(3)
        row += 1
    centers = np.array(centers[:n])
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    return np.clip(centers, 0.01, 0.99)

def run_packing():
    bounds = [(0, 1)] * (2*N) + [(0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_v = None
    
    # Generate diverse initial center configurations
    starts = []
    for s in range(10):
        starts.append(fps_init(N, s))
    for s in range(10):
        starts.append(hex_init(N, 0.09, s))
    for s in range(10):
        np.random.seed(s)
        starts.append(np.random.uniform(0.1, 0.9, (N, 2)))
        
    for centers in starts:
        # Initialize radii optimally for these centers using LP
        r_lp = get_optimal_radii_lp(centers)
        v0 = np.concatenate([centers[:, 0], centers[:, 1], r_lp])
        
        # Perturb to break symmetries and ensure strict initial feasibility
        v0[:2*N] += np.random.uniform(-0.003, 0.003, 2*N)
        v0[:2*N] = np.clip(v0[:2*N], 0.005, 0.995)
        v0[2*N:] *= 0.995 
        
        try:
            res = minimize(objective_func, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2500, 'ftol': 1e-13})
            if -res.fun > best_sum:
                # Verify feasibility before accepting
                c_val = constraint_func(res.x)
                if np.all(c_val >= -1e-7):
                    best_sum = -res.fun
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Fallback if optimization fails
    if best_v is None:
        centers = np.random.rand(N, 2) * 0.8 + 0.1
        radii = np.full(N, 0.05)
        best_v = np.concatenate([centers[:,0], centers[:,1], radii])
        
    centers = best_v[:2*N].reshape(N, 2)
    radii = best_v[2*N:]
    
    # Strict post-processing to guarantee validator compliance
    for _ in range(5):
        changed = False
        # Enforce boundary constraints
        for i in range(N):
            mr = min(centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1])
            if radii[i] > mr:
                radii[i] = mr
                changed = True
        # Enforce non-overlap constraints
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if radii[i] + radii[j] > d:
                    ov = radii[i] + radii[j] - d
                    shr = ov * 0.5 + 1e-9
                    radii[i] = max(0.0, radii[i] - shr)
                    radii[j] = max(0.0, radii[j] - shr)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
