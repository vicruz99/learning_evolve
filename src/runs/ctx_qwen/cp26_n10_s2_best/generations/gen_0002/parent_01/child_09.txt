

<strategy>
To improve the packing sum of radii beyond ~2.628, we need a more robust global search strategy to escape local minima that plague standard gradient-based optimizers like SLSQP when applied directly from structured grids. The hybrid approach combines:
1. **Penalty-based Global Search**: Use `differential_evolution` on an unconstrained penalty function that heavily penalizes boundary violations and overlaps. This explores the configuration space broadly and finds high-quality basins of attraction.
2. **Exact Constraint Refinement**: Take the best candidates from the global search and refine them using SLSQP with exact inequality constraints. This guarantees strict feasibility and maximizes the objective locally.
3. **Diverse Multi-start & Perturbation**: Combine the global search results with optimized starts from hexagonal/grid lattices, followed by iterative perturbation (shrinking radii, jittering centers, re-optimizing) to squeeze out the last 0.1-0.2% of potential.
4. **Squared Distance Constraints**: Use `dist^2 >= (r_i + r_j)^2` in the exact constraints for better numerical stability and smoothness near tangency points.
5. **Strict Post-processing**: Iteratively enforce boundaries and pairwise non-overlap with a safety margin to guarantee the validator passes without false negatives due to floating-point drift.

import numpy as np
from scipy.optimize import minimize, differential_evolution

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def exact_objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def exact_constraints(v):
    """Compute exact inequality constraints: boundaries and non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c_bound = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist_sq = dx**2 + dy**2
    r_sum_sq = (r[PAIR_I] + r[PAIR_J])**2
    c_pair = dist_sq - r_sum_sq
    
    return np.concatenate([c_bound, c_pair])

def penalty_objective(v):
    """Penalty-based objective for global search."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary violation penalties
    pen += np.sum(np.maximum(0.0, r - x)**2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - x))**2)
    pen += np.sum(np.maximum(0.0, r - y)**2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - y))**2)
    
    # Overlap violation penalties
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist = np.sqrt(dx*dx + dy*dy)
    pen += np.sum(np.maximum(0.0, r[PAIR_I] + r[PAIR_J] - dist)**2)
    
    return obj + 5000.0 * pen

def generate_lattice_init(seed, layout='hex'):
    """Generate a feasible initial configuration from a lattice."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    
    if layout == 'hex':
        pts = []
        r_est = 0.095
        y = r_est
        row = 0
        while len(pts) < N + 20:
            x = r_est + (row % 2) * r_est
            while x <= 1 - r_est:
                pts.append([x, y])
                x += 2 * r_est
            y += np.sqrt(3) * r_est
            row += 1
        pts = np.array(pts)
        idx = np.random.choice(len(pts), N, replace=False)
        centers = pts[idx]
    else: # grid
        pts = []
        for i in range(6):
            for j in range(5):
                pts.append([0.09 + i*0.17, 0.09 + j*0.19])
        pts = np.array(pts)
        idx = np.random.choice(len(pts), N, replace=False)
        centers = pts[idx]
        
    centers += np.random.uniform(-0.01, 0.01, size=centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    # Compute strictly feasible initial radii
    dists = np.sqrt(((centers[:, None, :] - centers[None, :, :])**2).sum(axis=2))
    np.fill_diagonal(dists, np.inf)
    wall_dists = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    r_init = np.minimum(dists.min(axis=1) / 2.0, wall_dists) * 0.85
    r_init = np.maximum(r_init, 0.01)
    
    return np.concatenate([centers[:, 0], centers[:, 1], r_init])

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': exact_constraints}
    
    best_sum = -1.0
    best_v = None
    
    # Phase 1: Global search via Differential Evolution on penalty function
    for seed in range(4):
        np.random.seed(seed)
        try:
            res_de = differential_evolution(penalty_objective, bounds, maxiter=500, 
                                            popsize=15, mutation=0.9, recombination=0.9,
                                            tol=1e-8, polish=False, workers=1, seed=seed)
            v0 = res_de.x.copy()
            v0[:2*N] = np.clip(v0[:2*N], 0.0, 1.0)
            v0[2*N:] = np.clip(v0[2*N:], 0.0, 0.5)
            
            res_loc = minimize(exact_objective, v0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
            if -res_loc.fun > best_sum and np.min(exact_constraints(res_loc.x)) >= -1e-7:
                best_sum = -res_loc.fun
                best_v = res_loc.x.copy()
        except Exception:
            pass
            
    # Phase 2: Multi-start from structured lattices
    for layout in ['hex', 'grid']:
        for seed in range(10):
            v0 = generate_lattice_init(seed, layout)
            try:
                res = minimize(exact_objective, v0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12})
                if -res.fun > best_sum and np.min(exact_constraints(res.x)) >= -1e-7:
                    best_sum = -res.fun
                    best_v = res.x.copy()
            except Exception:
                pass
                
    if best_v is None:
        best_v = np.concatenate([np.random.rand(2*N)*0.8+0.1, np.full(N, 0.05)])
        
    # Phase 3: Iterative perturbation to escape local minima
    for _ in range(8):
        v_pert = best_v.copy()
        v_pert[2*N:] *= 0.97  # Shrink radii to create breathing room
        v_pert[:2*N] += np.random.uniform(-0.008, 0.008, 2*N)
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.02, 0.98)
        
        try:
            res = minimize(exact_objective, v_pert, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12})
            if -res.fun > best_sum and np.min(exact_constraints(res.x)) >= -1e-7:
                best_sum = -res.fun
                best_v = res.x.copy()
        except Exception:
            pass
            
    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    radii = best_v[2*N:]
    
    # Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints strictly
    radii = np.minimum(radii, centers[:, 0])
    radii = np.minimum(radii, 1.0 - centers[:, 0])
    radii = np.minimum(radii, centers[:, 1])
    radii = np.minimum(radii, 1.0 - centers[:, 1])
    
    # 2. Enforce non-overlap strictly with iterative shrinkage
    for _ in range(15):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if radii[i] + radii[j] > d:
                    shrink = (radii[i] + radii[j] - d) * 0.5 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
