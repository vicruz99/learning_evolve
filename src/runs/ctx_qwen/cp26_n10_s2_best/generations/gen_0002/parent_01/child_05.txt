import numpy as np
from scipy.optimize import minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)

def objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and non-overlap (squared distances)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints (squared distances for stability)
    dx = x[IDX_I] - x[IDX_J]
    dy = y[IDX_I] - y[IDX_J]
    r_sum = r[IDX_I] + r[IDX_J]
    c = np.concatenate([c, dx**2 + dy**2 - r_sum**2])
    
    return c

def get_hex_init(angle, shift, r0):
    """Generate a rotated and shifted hexagonal lattice initialization."""
    centers = []
    y = r0
    row = 0
    while len(centers) < N + 20:
        x_start = r0 if row % 2 == 0 else 2 * r0
        x = x_start
        while x <= 1.0 - r0 and len(centers) < N + 20:
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    centers = np.array(centers[:N])
    
    # Rotation
    cos_a, sin_a = np.cos(angle), np.sin(angle)
    rot = np.array([[cos_a, -sin_a], [sin_a, cos_a]])
    centers = centers @ rot.T
    
    # Shift and clip
    centers += [shift, shift]
    return np.clip(centers, 0.01, 0.99)

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_val = -np.inf
    rng = np.random.default_rng(42)
    
    # Phase 1: Multi-start exploration from diverse configurations
    configs = []
    
    # Rotated hexagonal lattices
    for angle in np.linspace(0.0, np.pi/4, 12):
        for shift in np.linspace(-0.08, 0.08, 6):
            configs.append(get_hex_init(angle, shift, 0.095))
            
    # Random configurations
    for _ in range(40):
        configs.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    for centers in configs:
        r_init = np.full(N, 0.04)
        v0 = np.concatenate([centers.flatten(), r_init])
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_val:
                best_val = -res.fun
                best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Local perturbation search to escape local minima
    if best_v is not None:
        for i in range(20):
            v_pert = best_v.copy()
            # Perturb centers and radii
            v_pert += rng.uniform(-0.003, 0.003, v_pert.shape)
            # Shrink radii slightly to create feasible slack
            v_pert[2*N:] *= 0.995
            # Enforce bounds
            v_pert = np.clip(v_pert, 0.0, 1.0)
            v_pert[2*N:] = np.clip(v_pert[2*N:], 0.0, 0.5)
            
            try:
                res = minimize(objective, v_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                if -res.fun > best_val:
                    best_val = -res.fun
                    best_v = res.x.copy()
            except Exception:
                pass
                
    # Fallback initialization if optimization fails completely
    if best_v is None:
        best_v = np.zeros(3*N)
        best_v[:2*N] = rng.uniform(0.2, 0.8, 2*N)
        best_v[2*N:] = 0.03
        
    # Extract results
    centers = best_v[:2*N].reshape(N, 2)
    radii = best_v[2*N:].copy()
    
    # Phase 3: Strict validity enforcement to satisfy validator tolerances
    eps = 1e-9
    
    # Enforce boundary constraints
    for i in range(N):
        max_r = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = min(radii[i], max_r - eps)
        
    # Enforce non-overlap constraints iteratively
    for _ in range(15):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d - eps:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + eps
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    # Final clamp to ensure no negative radii or boundary violations
    radii = np.maximum(radii, 0.0)
    for i in range(N):
        radii[i] = min(radii[i], centers[i, 0], 1.0 - centers[i, 0], 
                       centers[i, 1], 1.0 - centers[i, 1])
        
    return centers, radii, float(np.sum(radii))
