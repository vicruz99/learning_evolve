N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):

    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist_sq = dx**2 + dy**2
    r_sum = r[PAIR_I] + r[PAIR_J]
    c = np.concatenate([c, dist_sq - r_sum**2])
    
    return c

    r_sum = r[PAIR_I] + r[PAIR_J]
    c = np.concatenate([c, dist - r_sum])

    return np.clip(centers, 0.01, 0.99)

def compute_feasible_radii(centers, scale=0.85):
    """Compute feasible radii for given centers."""
    n = len(centers)
    r = np.full(n, 0.5)
    
    # Boundary limits
    for i in range(n):
        r[i] = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
    
    # Pairwise limits
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            r[i] = min(r[i], d)
            r[j] = min(r[j], d)
    
    return r * scale

def run_optimization(x0, use_linear=False, max_iter=4000):
    """Run single optimization and return result."""
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_linear if use_linear else constraints}

                       constraints=cons,

        return res
    except Exception:
        return None

def enforce_validity(centers, radii, safety=1e-8):

    radii = radii.copy()
    centers = centers.copy()
    
    # Boundary constraints

    return centers, radii
