import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    # Using squared distances improves numerical smoothness for the optimizer
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    c_pair = dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx]**2
    
    return np.concatenate([c, c_pair])

def get_feasible_init(centers, r_base=0.03):
    """Compute strictly feasible initial radii for a given set of centers."""
    r = np.full(N, r_base)
    for i in range(N):
        # Max radius allowed by boundaries
        mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        # Max radius allowed by other centers
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                mr = min(mr, d/2.0)
        r[i] = min(r[i], mr * 0.85)
    return r

            row = 0

                row += 1
            configs.append(np.array(pts[:N]))

                    radii[j] = max(0.0, radii[j] - shrink)

        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
