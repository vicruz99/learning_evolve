import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """
    Inequality constraints: boundaries and non-overlap.
    Uses squared distances for better numerical stability and smooth gradients.
    Returns a flat array where each element >= 0 indicates a satisfied constraint.
    """
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - dr**2
    
    return c

def compute_feasible_radii(centers, scale_factor=0.85):
    """Computes a set of radii that strictly satisfies all constraints for given centers."""
    r = np.full(N, 0.5)
    for i in range(N):
        cx, cy = centers[i]
        # Distance to boundaries
        r[i] = min(cx, 1.0 - cx, cy, 1.0 - cy)
        # Distance to other centers
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d / 2.0 < r[i]:
                    r[i] = d / 2.0
    return r * scale_factor

def get_initial_configs():
    """Generates a list of diverse base configurations."""
    configs = []
    
    # 1. Hexagonal lattices with varying parameters
    for scale in [0.08, 0.09, 0.10, 0.11]:
        for shift in [0.0, 0.02, -0.02, 0.04]:
            centers = []
            y = scale + shift
            row = 0
            while len(centers) < N + 10:
                x_start = scale if row % 2 == 0 else 2 * scale
                x = x_start
                while x <= 1 - scale:
                    centers.append([x, y])
                    x += 2 * scale
                y += scale * np.sqrt(3)
                row += 1
            configs.append(np.array(centers[:N]))
            
    # 2. Grid lattices
    for sx in [0.18, 0.19, 0.20]:
        for sy in [0.18, 0.19, 0.20]:
            pts = []
            for i in range(6):
                for j in range(5):
                    if len(pts) < N:
                        pts.append([0.1 + i * sx, 0.1 + j * sy])
            configs.append(np.array(pts))
            
    # 3. Random scatter
    np.random.seed(42)
    for _ in range(15):
        configs.append(np.random.uniform(0.1, 0.9, (N, 2)))
        
    return configs

def run_packing():
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    base_configs = get_initial_configs()
    
    # Phase 1: Multi-start optimization
    for cfg in base_configs:
        for pert_idx in range(4):
            np.random.seed(pert_idx * 1000 + len(base_configs))
            c_pert = cfg + np.random.uniform(-0.015, 0.015, cfg.shape)
            c_pert = np.clip(c_pert, 0.05, 0.95)
            
            r_init = compute_feasible_radii(c_pert, scale_factor=0.80)
            v0 = np.concatenate([c_pert[:, 0], c_pert[:, 1], r_init])
            
            try:
                res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                               constraints=cons,

                if curr_sum > best_sum:

                    if np.min(constraints(res.x)) >= -1e-7:
                        best_sum = curr_sum
                        best_v = res.x.copy()
            except Exception:
                continue

    if best_v is not None:

            np.random.seed(i * 777)
            v_pert = best_v.copy()

            v_pert[:2*N] = np.clip(v_pert[:2*N], 0.02, 0.98)

                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
