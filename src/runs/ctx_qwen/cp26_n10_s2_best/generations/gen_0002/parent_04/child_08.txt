import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints: dist >= r_i + r_j
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist = np.hypot(dx, dy)
    r_sum = r[PAIR_I] + r[PAIR_J]
    c = np.concatenate([c, dist - r_sum])
    return c

def get_init_config(seed, layout='hex'):
    """Generates a feasible initial configuration based on lattice or grid."""
    np.random.seed(seed)
    if layout == 'hex':
        r0 = 0.095
        pts = []
        y = r0
        row = 0
        while len(pts) < N + 10:
            x_start = r0 if row % 2 == 0 else 2 * r0
            x = x_start
            while x <= 1 - r0 and len(pts) < N + 10:
                pts.append([x, y])
                x += 2 * r0
            y += r0 * np.sqrt(3)
            row += 1
        pts = np.array(pts[:N])
    else:
        pts = []

            for j in range(5):

    pts += np.random.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    
    # Compute strictly feasible initial radii
    r_init = np.full(N, 0.05)
    for i in range(N):
        mr = min(pts[i,0], 1-pts[i,0], pts[i,1], 1-pts[i,1])

                mr = min(mr, d/2.0)
        r_init[i] = mr * 0.85
        
    return np.concatenate([pts[:,0], pts[:,1], r_init])

def fix_feasibility(v):
    """Analytically shrinks radii to guarantee all constraints are satisfied."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:].copy()
    for _ in range(5):
        for i in range(N):
            r[i] = min(r[i], c[i,0], 1-c[i,0], c[i,1], 1-c[i,1])

            for j in range(i+1, N):
                d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
                if r[i] + r[j] > d:
                    shrink = (r[i] + r[j] - d)/2.0 + 1e-7
                    r[i] = max(0.0, r[i] - shrink)
                    r[j] = max(0.0, r[j] - shrink)
    v[2*N:] = r
    return v

def run_packing():

    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).

    bounds = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N

    best_v = None

    for x0 in starts:

    return centers, radii, float(np.sum(radii))
