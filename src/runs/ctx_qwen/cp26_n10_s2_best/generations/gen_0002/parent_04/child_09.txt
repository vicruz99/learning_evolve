import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints: dist >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (dist - r_sum)[mask]])
    return c

def fix_overlaps(x, y, r, iterations=10):
    """Adjusts radii to guarantee the configuration satisfies all constraints strictly."""
    r = r.copy()
    for _ in range(iterations):
        # Enforce boundary constraints
        r = np.minimum(r, np.minimum(x, 1.0 - x))
        r = np.minimum(r, np.minimum(y, 1.0 - y))
        r = np.maximum(r, 0.0)
        
        # Enforce non-overlap constraints
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(x[i] - x[j], y[i] - y[j])
                if d < r[i] + r[j]:
                    shrink = (r[i] + r[j] - d) / 2.0 + 1e-9
                    r[i] = max(0.0, r[i] - shrink)
                    r[j] = max(0.0, r[j] - shrink)
    return r

def generate_initial_guesses():
    """Generates a diverse set of feasible initial configurations."""
    guesses = []
    np.random.seed(123)
    
    # 1. Hexagonal lattices with varying densities and jitter
    for _ in range(8):
        r0 = 0.08 + np.random.uniform(0.0, 0.03)
        centers = []
        y = r0
        row = 0
        while len(centers) < N + 10:
            x = r0 + (row % 2) * r0
            while x < 1.0 - r0:
                centers.append([x, y])
                x += 2 * r0
            y += r0 * np.sqrt(3)
            row += 1
        c = np.array(centers[:N]) + np.random.uniform(-0.02, 0.02, (N, 2))
        c = np.clip(c, 0.02, 0.98)
        r = np.full(N, 0.04)
        r = fix_overlaps(c[:, 0], c[:, 1], r)
        guesses.append(np.concatenate([c[:, 0], c[:, 1], r]))
        
    # 2. Grid-based configurations
    for _ in range(4):
        c = np.array([[0.05 + i*0.17, 0.05 + j*0.18] for i in range(6) for j in range(5)])[:N]
        c += np.random.uniform(-0.015, 0.015, c.shape)
        c = np.clip(c, 0.02, 0.98)
        r = np.full(N, 0.04)
        r = fix_overlaps(c[:, 0], c[:, 1], r)
        guesses.append(np.concatenate([c[:, 0], c[:, 1], r]))
        
    # 3. Random dense placements
    for _ in range(4):
        c = np.random.uniform(0.05, 0.95, (N, 2))
        r = np.full(N, 0.035)
        r = fix_overlaps(c[:, 0], c[:, 1], r)
        guesses.append(np.concatenate([c[:, 0], c[:, 1], r]))
        
    return guesses

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_val = -np.inf
    
    # Stage 1: Multi-start optimization from diverse layouts
    guesses = generate_initial_guesses()
    
    for v0 in guesses:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_val:
                if np.all(constraints(res.x) >= -1e-7):
                    best_val = -res.fun
                    best_v = res.x.copy()
        except Exception:
            continue
            
    if best_v is None:
        best_v = guesses[0]
        
    # Stage 2: Iterative refinement with controlled scaling and perturbation
    np.random.seed(999)
    for step in range(12):
        x_curr = best_v[:N].copy()
        y_curr = best_v[N:2*N].copy()
        r_curr = best_v[2*N:].copy()
        
        # Perturb positions to escape local minima
        x_curr += np.random.uniform(-0.003, 0.003, N)
        y_curr += np.random.uniform(-0.003, 0.003, N)
        x_curr = np.clip(x_curr, 0.01, 0.99)
        y_curr = np.clip(y_curr, 0.01, 0.99)
        
        # Scale up radii to push against constraints
        r_curr *= 1.012
        
        # Restore feasibility for the optimizer
        r_curr = fix_overlaps(x_curr, y_curr, r_curr)
        
        v0 = np.concatenate([x_curr, y_curr, r_curr])
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_val:
                if np.all(constraints(res.x) >= -1e-7):
                    best_val = -res.fun
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Extract final configuration
    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    radii = best_v[2*N:].copy()
    
    # Stage 3: Strict post-processing to guarantee validation passes
    radii = fix_overlaps(centers[:, 0], centers[:, 1], radii, iterations=30)
    
    return centers, radii, float(np.sum(radii))
