import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """
    Inequality constraints: boundaries and non-overlap.
    Uses squared distances for better numerical conditioning.
    Returns array where all elements >= 0 indicate feasibility.
    """
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    # Boundary constraints: circle inside [0,1]^2
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - dr**2
    
    return c

def relax_centers(centers, radii, steps=120):
    """
    Repulsion-based relaxation to spread circles apart and resolve overlaps.
    Serves as a high-quality initialization step for the optimizer.
    """
    centers = centers.copy()
    for _ in range(steps):
        forces = np.zeros_like(centers)
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d2 = dx * dx + dy * dy
                if d2 < 0.04 and d2 > 1e-8:
                    d = np.sqrt(d2)
                    target = radii[i] + radii[j] + 0.005
                    if d < target:
                        rep = (target - d) / d
                        fx = dx * rep
                        fy = dy * rep
                        forces[i, 0] += fx
                        forces[i, 1] += fy
                        forces[j, 0] -= fx
                        forces[j, 1] -= fy
                        
            # Boundary repulsion
            x, y = centers[i]
            r = radii[i]
            if x < r + 0.015: forces[i, 0] += 0.3
            if x > 1.0 - r - 0.015: forces[i, 0] -= 0.3
            if y < r + 0.015: forces[i, 1] += 0.3
            if y > 1.0 - r - 0.015: forces[i, 1] -= 0.3
            
        centers += forces * 0.008
        centers = np.clip(centers, 0.005, 0.995)
    return centers

def run_packing():
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    # Phase 1: Generate diverse initial configurations
    initial_configs = []
    
    # 1. Hexagonal lattices with varied spacing & jitter
    for seed in range(8):
        np.random.seed(seed)
        centers = []
        r0 = 0.09 + np.random.uniform(-0.005, 0.005)
        y = r0
        row = 0
        while len(centers) < N:
            x_start = r0 if row % 2 == 0 else 2 * r0
            x = x_start
            while x <= 1.0 - r0 and len(centers) < N:
                centers.append([x, y])
                x += 2 * r0
            y += r0 * np.sqrt(3)
            row += 1
        centers = np.array(centers[:N])
        centers += np.random.uniform(-0.01, 0.01, centers.shape)
        initial_configs.append(centers)
        
    # 2. Square grids & rotated variants
    for seed in range(6):
        np.random.seed(seed + 50)
        pts = []
        for i in range(5):
            for j in range(5):
                pts.append([0.1 + i * 0.2, 0.1 + j * 0.2])
        pts.append([0.5, 0.5])
        centers = np.array(pts[:N])
        centers += np.random.uniform(-0.015, 0.015, centers.shape)
        initial_configs.append(centers)
        
    # 3. Random dense packs
    for seed in range(6):
        np.random.seed(seed + 100)
        initial_configs.append(np.random.uniform(0.08, 0.92, (N, 2)))
        
    # Phase 2: Optimize from each configuration
    radii_init = np.full(N, 0.08)
    for centers in initial_configs:
        centers_relaxed = relax_centers(centers, radii_init, steps=100)
        v0 = np.concatenate([centers_relaxed[:, 0], centers_relaxed[:, 1], radii_init])
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 20000, 'ftol': 1e-14})

            # Accept if better and sufficiently feasible

                best_sum = curr_sum
                best_v = res.x.copy()
        except Exception:
            continue
            
    # Phase 3: Basin hopping refinement to escape local minima
    if best_v is not None:
        current_v = best_v.copy()
        for _ in range(8):
            # Perturb: shrink radii to create breathing room, jitter centers
            pert = current_v.copy()
            pert[:2*N] += np.random.uniform(-0.004, 0.004, 2*N)

            pert[2*N:] *= 0.95

            c_p = pert[:2*N].reshape(N, 2)

            c_rel = relax_centers(c_p, r_p, steps=60)
            pert[:2*N] = c_rel.flatten()

    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()
    centers = np.column_stack((cx, cy))
    
    # Phase 4: Strict post-processing to guarantee validator compliance
    # Enforce boundary constraints strictly
    for i in range(N):
        max_r = min(cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
        if cr[i] > max_r:
            cr[i] = max_r
            
    # Enforce non-overlap constraints iteratively with safety margin
    for _ in range(10):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                sum_r = cr[i] + cr[j]
                if d < sum_r - 1e-9:
                    shrink = (sum_r - d) / 2.0 + 1e-9
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, cr, float(np.sum(cr))
