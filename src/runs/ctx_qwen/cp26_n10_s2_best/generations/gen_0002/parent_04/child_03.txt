import numpy as np
from scipy.optimize import minimize

N = 26

def get_pair_indices(n):
    """Precompute indices for all unique circle pairs."""
    return np.triu_indices(n, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v, i_idx, j_idx):
    """
    Inequality constraints: boundaries and non-overlap.
    Uses squared distances for numerical stability.
    Returns array where all elements must be >= 0.
    """
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def generate_hex_init(r0, shift=0.0, angle=0.0):
    """Generate a hexagonal lattice configuration with optional shift and rotation."""
    centers = []
    y = r0 + shift
    row = 0
    while len(centers) < N + 5:
        x_start = r0 + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(centers) < N + 5:
            centers.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
        row += 1
        
    pts = np.array(centers[:N])
    if angle != 0.0:
        c = np.array([0.5, 0.5])
        pts = pts - c
        rot = np.array([[np.cos(angle), -np.sin(angle)],
                        [np.sin(angle), np.cos(angle)]])
        pts = pts @ rot.T + c
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def compute_initial_radii(centers):
    """Compute strictly feasible initial radii based on geometry."""
    r = np.full(N, 0.04)
    for i in range(N):
        d_wall = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        for j in range(N):
            if i != j:
                d_neigh = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                d_wall = min(d_wall, d_neigh / 2.0)
        r[i] = min(r[i], d_wall * 0.85)
    return r

def run_packing():

        starts.append(np.random.uniform(0.05, 0.95, (N, 2)))
        
    for pts in starts:
        r_init = compute_initial_radii(pts)
        x0 = np.concatenate([pts[:, 0], pts[:, 1], r_init])
        
        current_v = x0.copy()
        # Continuation & perturbation loop
        for step in range(20):
            # Gradually push radii larger to encourage expansion
            if step > 0:
                current_v[2*N:] *= 1.003
                
            # Perturb centers to escape local minima

                break

    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
