from scipy.optimize import minimize

N = 26
# Precompute pair indices for vectorized constraint evaluation
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def constraints(v):
    """Compute all inequality constraints: boundaries and non-overlap."""
    x = v[:N]
    y = v[N:2 * N]
    r = v[2 * N:]
    
    # Pre-allocate constraint array for efficiency
    c = np.empty(4 * N + NUM_PAIRS)
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c[:N] = x - r
    c[N:2 * N] = 1.0 - x - r
    c[2 * N:3 * N] = y - r
    c[3 * N:4 * N] = 1.0 - y - r
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    r_sum = r[PAIR_I] + r[PAIR_J]
    c[4 * N:] = dx**2 + dy**2 - r_sum**2
    
    return c

def project_to_feasible(v):
    """Adjusts radii to guarantee the configuration satisfies all constraints strictly."""
    x = v[:N].copy()
    y = v[N:2 * N].copy()
    r = v[2 * N:].copy()
    tol = 1e-7
    
    # Enforce boundary constraints strictly
    for i in range(N):
        max_r = min(x[i], 1.0 - x[i], y[i], 1.0 - y[i])
        if r[i] > max_r:
            r[i] = max_r

        changed = False

            for j in range(i + 1, N):

                    r[j] = max(0.0, r[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return np.concatenate([x, y, r])

    np.random.seed(seed)

    y = r0
    row = 0

            x += 2 * r0

        row += 1

    centers -= 0.5

    centers = centers @ rot_mat.T + 0.5

    jitter = 0.015
    centers += np.random.uniform(-jitter, jitter, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    
    # Start with slightly smaller radii to guarantee initial feasibility
    r_init = np.full(N, r0 * 0.75)
    return np.concatenate([centers[:, 0], centers[:, 1], r_init])
