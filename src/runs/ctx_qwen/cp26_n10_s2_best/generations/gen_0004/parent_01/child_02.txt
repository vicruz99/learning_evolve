from scipy.optimize import NonlinearConstraint, Bounds
import math

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)


def objective(v):
    return -np.sum(v[2 * N:])


def constraints_fn(v):
    x = v[:N]
    y = v[N:2 * N]
    r = v[2 * N:]
    c = np.empty(4 * N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2 * N] = 1.0 - x - r
    c[2 * N:3 * N] = y - r
    c[3 * N:4 * N] = 1.0 - y - r
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    r_sum = r[PAIR_I] + r[PAIR_J]
    c[4 * N:] = dx ** 2 + dy ** 2 - r_sum ** 2
    return c


def get_initial_radii(centers, shrink=0.88):
    r = np.full(N, 0.25)
    wall_dists = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    r = np.minimum(r, wall_dists)
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    return np.clip(r * shrink, 1e-5, 0.25)


def generate_hex_pattern(row_pattern, r0, rotation=0, shift=(0, 0)):

    y = r0 + shift[1]
    row_idx = 0
    for count in row_pattern:
        x_start = r0 + shift[0]
        if row_idx % 2 == 1:
            x_start += r0
        for k in range(count):
            x = x_start + k * 2.0 * r0
            pts.append([x, y])
        y += r0 * math.sqrt(3)
        row_idx += 1
    pts = np.array(pts[:N])
    if rotation != 0:
        c, s = np.cos(rotation), np.sin(rotation)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    return np.clip(pts, 0.02, 0.98)


def force_relax(centers, steps=300, target=0.22):
    pts = centers.copy()
    lr = 0.008
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[i] - pts[j]
                d = np.linalg.norm(diff)
                if d < target and d > 1e-6:
                    f = (target - d) / d * lr
                    forces[i] += f * diff
                    forces[j] -= f * diff
        pts += forces
        pts = np.clip(pts, 0.02, 0.98)
        lr *= 0.998
    return pts
