import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints
    c_bound = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    r_sum = r[PAIR_I] + r[PAIR_J]
    c_pair = dx**2 + dy**2 - r_sum**2
    
    return np.concatenate([c_bound, c_pair])

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.25)
    
    # Distance to square boundaries
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    r = np.minimum(r, wall_dists)
    
    # Distance to other centers
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    r = np.minimum(r, min_dists / 2.0)
    
    # Scale down to guarantee strict feasibility for optimizer start
    return np.clip(r * 0.85, 1e-4, 0.25)

def force_directed_init(seed, steps=250):
    """Generate a well-spread initial configuration using repulsive forces."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    lr = 0.025
    target = 0.22
    
    for _ in range(steps):
        # Vectorized pairwise repulsion
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.linalg.norm(diff, axis=2)
        np.fill_diagonal(dists, 1e9)
        
        mask = (dists < target)
        forces_mag = np.where(mask, (target - dists) / dists, 0.0)
        forces_vec = diff * forces_mag[:, :, None]
        forces = np.sum(forces_vec / (dists[:, :, None] + 1e-9), axis=1)
        
        # Wall repulsion to keep points inside
        forces[:, 0] += np.where(pts[:, 0] < 0.08, 0.15, 0.0) - np.where(pts[:, 0] > 0.92, 0.15, 0.0)
        forces[:, 1] += np.where(pts[:, 1] < 0.08, 0.15, 0.0) - np.where(pts[:, 1] > 0.92, 0.15, 0.0)
        
        pts += forces * lr
        pts = np.clip(pts, 0.03, 0.97)
        lr *= 0.992
        
    return pts

def generate_initial_configs():

    configs = []
    
    # 1. Force-relaxed random layouts
    for seed in range(15):
        pts = force_directed_init(seed)
        r = get_feasible_radii(pts)
        configs.append(np.concatenate([pts[:, 0], pts[:, 1], r]))

        pts = []
        y = r0 + sy
        row = 0
        while len(pts) < N + 8:
            x_start = r0 + sx + (row % 2) * r0
            x = x_start
            while x <= 1.0 - r0 and len(pts) < N + 8:
                pts.append([x, y])
                x += 2 * r0
            y += np.sqrt(3) * r0
            row += 1
            
        pts = np.array(pts[:N])
        
        # Rotate around center

            c, s = np.cos(angle), np.sin(angle)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5

def run_packing():

    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).

    bounds = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.5)] * N

    cx = best_v[:N]
    cy = best_v[N:2*N]

                sum_r = cr[i] + cr[j]

                if sum_r > d + 1e-11:
                    shrink = (sum_r - d - 1e-11) / 2.0
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, cr, float(np.sum(cr))
