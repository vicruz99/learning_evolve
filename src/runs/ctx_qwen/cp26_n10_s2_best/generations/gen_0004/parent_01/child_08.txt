import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def penalty_obj(v, P):
    """Objective with penalty for constraint violations."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    loss = -np.sum(r)
    
    # Boundary penalties (squared for smooth gradients)
    loss += P * np.sum(np.maximum(0.0, r - x)**2)
    loss += P * np.sum(np.maximum(0.0, x + r - 1.0)**2)
    loss += P * np.sum(np.maximum(0.0, r - y)**2)
    loss += P * np.sum(np.maximum(0.0, y + r - 1.0)**2)
    
    # Pairwise non-overlap penalties
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist = np.sqrt(dx*dx + dy*dy + 1e-12)
    r_sum = r[PAIR_I] + r[PAIR_J]
    loss += P * np.sum(np.maximum(0.0, r_sum - dist)**2)
    
    return loss

def compute_violation(v):
    """Compute sum of linear constraint violations."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    viol = 0.0
    viol += np.sum(np.maximum(0.0, r - x))
    viol += np.sum(np.maximum(0.0, x + r - 1.0))
    viol += np.sum(np.maximum(0.0, r - y))
    viol += np.sum(np.maximum(0.0, y + r - 1.0))
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist = np.sqrt(dx*dx + dy*dy + 1e-12)
    r_sum = r[PAIR_I] + r[PAIR_J]
    viol += np.sum(np.maximum(0.0, r_sum - dist))
    
    return viol

def generate_inits():
    """Generate diverse initial configurations."""
    configs = []
    
    # 1. Hexagonal lattices with rotations
    for seed in range(12):
        np.random.seed(seed)
        r0 = 0.095 + np.random.uniform(-0.01, 0.01)
        angle = np.random.uniform(-0.3, 0.3)
        pts = []
        y = r0
        row = 0
        while len(pts) < N + 10:
            x_start = r0 + (row % 2) * r0
            x = x_start
            while x <= 1.0 - r0 and len(pts) < N + 10:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
        pts = np.array(pts[:N])
        
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        pts += np.random.uniform(-0.01, 0.01, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        r = np.full(N, 0.04)
        configs.append(np.concatenate([pts[:,0], pts[:,1], r]))
        
    # 2. Force-relaxed random starts
    for seed in range(15):
        np.random.seed(seed + 100)
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        for _ in range(200):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    d = np.linalg.norm(pts[i]-pts[j])
                    if d < 0.2 and d > 1e-4:
                        f = (0.2 - d)/d
                        forces[i] += f*(pts[i]-pts[j])
                        forces[j] -= f*(pts[i]-pts[j])
            pts += forces * 0.02
            pts = np.clip(pts, 0.02, 0.98)
        r = np.full(N, 0.03)
        configs.append(np.concatenate([pts[:,0], pts[:,1], r]))
        
    # 3. Staggered grids
    for seed in range(5):
        np.random.seed(seed + 200)
        pts = []
        for i in range(6):
            for j in range(5):
                pts.append([0.08 + i*0.15 + np.random.uniform(-0.02, 0.02),
                            0.08 + j*0.18 + np.random.uniform(-0.02, 0.02)])
        pts = np.array(pts[:N])
        pts = np.clip(pts, 0.02, 0.98)
        r = np.full(N, 0.035)
        configs.append(np.concatenate([pts[:,0], pts[:,1], r]))
        
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    bounds = [(0.0, 1.0)]*(2*N) + [(0.0, 0.45)]*N
    inits = generate_inits()
    
    best_v = None
    best_score = -np.inf
    
    # Penalty annealing schedule
    P_steps = np.linspace(50, 5000, 10)
    
    for v0 in inits:
        v_curr = v0.copy()
        v_curr[:2*N] = np.clip(v_curr[:2*N], 0.01, 0.99)
        
        for P in P_steps:
            try:
                res = minimize(penalty_obj, v_curr, args=(P,), method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': 1200, 'ftol': 1e-14, 'gtol': 1e-8})
                v_curr = res.x
                
                # Controlled perturbation to escape local minima
                noise = np.random.uniform(-0.003, 0.003, 2*N)
                v_curr[:2*N] += noise
                v_curr[:2*N] = np.clip(v_curr[:2*N], 0.01, 0.99)
            except Exception:
                pass
                
        viol = compute_violation(v_curr)
        s = np.sum(v_curr[2*N:])
        # Score prioritizes sum, but penalizes violations heavily to guide towards feasibility
        score = s - 200.0 * viol
        
        if score > best_score:
            best_score = score
            best_v = v_curr.copy()
            
    if best_v is None:
        return np.zeros((N,2)), np.zeros(N), 0.0
        
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()
    
    # Strict post-processing for validator compliance
    # 1. Boundary enforcement
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    cr = np.maximum(cr, 0.0)
    
    # 2. Pairwise non-overlap enforcement (iterative shrinkage)
    for _ in range(15):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(cx[i]-cx[j], cy[i]-cy[j])
                sum_r = cr[i] + cr[j]
                if sum_r > d - 1e-11:
                    shrink = (sum_r - d + 1e-11) / 2.0
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
