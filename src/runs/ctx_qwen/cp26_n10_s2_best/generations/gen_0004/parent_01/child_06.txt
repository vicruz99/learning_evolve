import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)


def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])


def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    
    # Boundary constraints
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c


def get_feasible_radii(centers, scale=0.85):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.25)
    
    # Distance to square boundaries
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    r = np.minimum(r, wall_dists)
    
    # Distance to other centers
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    r = np.minimum(r, min_dists / 2.0)
    
    return np.clip(r * scale, 1e-5, 0.25)


def generate_hex_lattice(r0, angle=0.0, shift_x=0.0, shift_y=0.0):

        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    
    pts = np.clip(pts, 0.02, 0.98)

def force_relax_init(seed, steps=300):
    """Generate well-spaced configuration using force-directed layout."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[i] - pts[j]
                d = np.linalg.norm(diff)
                if d < 0.25 and d > 1e-5:
                    f = (0.25 - d) * 2.0 / d
                    forces[i] += f * diff
                    forces[j] -= f * diff
        
        # Push away from boundaries
        for i in range(N):
            if pts[i, 0] < 0.1:
                forces[i, 0] += 0.3
            if pts[i, 0] > 0.9:
                forces[i, 0] -= 0.3

                forces[i, 1] += 0.3
            if pts[i, 1] > 0.9:
                forces[i, 1] -= 0.3
        
        pts += forces * 0.02
        pts = np.clip(pts, 0.03, 0.97)
    
    return pts

def run_optimization(v0, method='SLSQP', maxiter=5000):
    """Run optimization with given initial point."""
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.25)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    try:
        if method == 'trust-constr':
            from scipy.optimize import Bounds, NonlinearConstraint
            b = Bounds(lb=[0.0]*2*N + [1e-5]*N, ub=[1.0]*2*N + [0.25]*N)
            res = minimize(objective, v0, method='trust-constr', bounds=b,
                          constraints=cons, options={'maxiter': maxiter, 'verbose': 0})
        else:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        return res
    except Exception:
        return None


def fix_feasibility(v, iterations=30):
    """Make configuration strictly feasible by shrinking radii."""
    cx = v[:N].copy()
    cy = v[N:2*N].copy()
    cr = v[2*N:].copy()
    
    # Boundary constraints
    cr = np.minimum(cr, cx)
    cr = np.minimum(cr, 1.0 - cx)
    cr = np.minimum(cr, cy)
    cr = np.minimum(cr, 1.0 - cy)
    
    # Pairwise non-overlap
    for _ in range(iterations):
        changed = False
        for idx in range(len(PAIR_I)):
            i, j = PAIR_I[idx], PAIR_J[idx]
            d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
            overlap = cr[i] + cr[j] - d
            if overlap > 1e-10:
                shrink = overlap * 0.5 + 1e-10
                cr[i] = max(0.0, cr[i] - shrink)
                cr[j] = max(0.0, cr[j] - shrink)
                changed = True
        if not changed:
            break
    
    return np.concatenate([cx, cy, cr])
