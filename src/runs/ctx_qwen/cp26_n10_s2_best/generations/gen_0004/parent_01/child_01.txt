import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.25)
    
    # Distance to square boundaries
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    r = np.minimum(r, wall_dists)
    
    # Distance to other centers
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    r = np.minimum(r, min_dists / 2.0)
    
    # Scale down slightly to guarantee strict feasibility for optimizer start
    return np.clip(r * 0.85, 1e-5, 0.25)

def generate_initial_configs():
    """Generate diverse initial configurations."""
    configs = []
    
    # 1. Tilted Hexagonal Lattices
    for ang in np.linspace(-0.35, 0.35, 11):
        for r0 in [0.090, 0.095, 0.100, 0.105, 0.110]:
            pts = []
            y = r0
            row = 0
            while len(pts) < N + 10:
                x_start = r0 + (row % 2) * r0
                x = x_start
                while x <= 1.0 - r0 and len(pts) < N + 10:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3.0) * r0
                row += 1
                
            pts = np.array(pts[:N])
            
            # Rotate around center
            c, s = np.cos(ang), np.sin(ang)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pts = np.clip(pts, 0.04, 0.96)
            
            if len(pts) >= N:
                r = get_feasible_radii(pts)
                configs.append(np.concatenate([pts[:, 0], pts[:, 1], r]))
                
    # 2. Perturbed Square Grids
    for sx in np.linspace(-0.02, 0.02, 5):
        for sy in np.linspace(-0.02, 0.02, 5):
            pts = np.array([[0.10 + i*0.16 + sx + np.random.uniform(-0.008, 0.008), 
                             0.10 + j*0.19 + sy + np.random.uniform(-0.008, 0.008)]
                            for i in range(6) for j in range(5)])[:N]
            pts = np.clip(pts, 0.04, 0.96)
            r = get_feasible_radii(pts)
            configs.append(np.concatenate([pts[:, 0], pts[:, 1], r]))

    # 3. Force-Relaxed Random Scatter
    for seed in range(12):
        np.random.seed(seed + 500)
        pts = np.random.uniform(0.15, 0.85, size=(N, 2))
        # Quick repulsion relaxation
        for _ in range(150):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.linalg.norm(pts[i] - pts[j])
                    if d < 0.24 and d > 1e-4:
                        f = (0.24 - d) / d
                        diff = pts[i] - pts[j]
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.012
            pts = np.clip(pts, 0.04, 0.96)
            
        r = get_feasible_radii(pts)
        configs.append(np.concatenate([pts[:, 0], pts[:, 1], r]))
        
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    inits = generate_initial_configs()
    
    # Phase 1: Multi-start optimization
    for v0 in inits:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            
            curr_sum = -res.fun
            if curr_sum > best_sum:
                c_val = constraints(res.x)
                # Accept if sufficiently feasible
                if np.min(c_val) >= -1e-7:
                    best_sum = curr_sum
                    best_v = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization fails completely
    if best_v is None:
        best_v = inits[0]
        best_sum = -np.sum(best_v[2*N:])

    current_v = best_v.copy()
    for step in range(80):
        np.random.seed(step + 2000)
        v_pert = current_v.copy()
        
        # Perturb centers with annealing noise scale
        noise_scale = 0.007 * (1.0 - step / 80.0)
        v_pert[:2*N] += np.random.uniform(-noise_scale, noise_scale, 2*N)
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.01, 0.99)
        
        # Shrink radii to create breathing room, then recompute feasible radii
        centers_pert = v_pert[:2*N].reshape(N, 2)
        v_pert[2*N:] = get_feasible_radii(centers_pert) * 0.90

                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            
            curr_sum = -res.fun
            if curr_sum > best_sum:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    best_sum = curr_sum
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            continue
            
    # Extract final configuration
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()
    centers = np.column_stack((cx, cy))
    
    # Phase 3: Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints strictly
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    cr = np.maximum(cr, 0.0)

    # Validator allows: dist >= r1 + r2 - 1e-12

        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                sum_r = cr[i] + cr[j]
                # Add tiny buffer to ensure strict compliance after floating point ops
                if sum_r > d + 1e-11:
                    shrink = (sum_r - d - 1e-11) / 2.0
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, cr, float(np.sum(cr))
