import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)


def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])


def constraints(v):
    """Inequality constraints: boundaries and non-overlap using direct distance."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]

    c = np.empty(4*N + NUM_PAIRS)

    # Boundary constraints
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r

    # Pairwise non-overlap: dist >= r_i + r_j
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist = np.sqrt(dx**2 + dy**2 + 1e-20)
    c[4*N:] = dist - (r[PAIR_I] + r[PAIR_J])

    return c


def compute_feasible_radii(centers, shrink=0.90):
    """Compute strictly feasible radii for given centers."""
    r = np.full(N, 0.25)

    # Distance to walls
    wall_dists = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    r = np.minimum(r, wall_dists)

    # Distance to nearest neighbor
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)

    return np.clip(r * shrink, 1e-5, 0.25)


def generate_hex_pattern(r0, row_counts, shift_x=0.0, shift_y=0.0):
    """Generate hexagonal pattern with specified row counts."""
    pts = []
    y = r0 + shift_y
    for row_idx, count in enumerate(row_counts):
        x_start = r0 + shift_x + (row_idx % 2) * r0
        for k in range(count):
            if len(pts) >= N:
                break
            x = x_start + k * 2.0 * r0
            pts.append([x, y])
        y += r0 * np.sqrt(3.0)
    return np.array(pts[:N])


def force_relax(centers, steps=300):
    """Relax centers using repulsive forces."""
    pts = centers.copy()
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[i] - pts[j]
                d = np.linalg.norm(diff)
                if d < 0.20 and d > 1e-6:
                    f = (0.20 - d) / d * 0.05
                    forces[i] += f * diff
                    forces[j] -= f * diff

        # Wall repulsion
        forces[:, 0] += np.where(pts[:, 0] < 0.1, 0.1 - pts[:, 0], 0.0) * 2.0
        forces[:, 0] -= np.where(pts[:, 0] > 0.9, pts[:, 0] - 0.9, 0.0) * 2.0
        forces[:, 1] += np.where(pts[:, 1] < 0.1, 0.1 - pts[:, 1], 0.0) * 2.0
        forces[:, 1] -= np.where(pts[:, 1] > 0.9, pts[:, 1] - 0.9, 0.0) * 2.0

        pts += forces
        pts = np.clip(pts, 0.04, 0.96)
    return pts

def run_packing():

    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).

    bounds = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.25)] * N

            curr_sum = -res.fun
            if curr_sum > best_sum:
                c_val = constraints(res.x)

                    best_sum = curr_sum
                    best_v = res.x.copy()
        except Exception:
            continue
