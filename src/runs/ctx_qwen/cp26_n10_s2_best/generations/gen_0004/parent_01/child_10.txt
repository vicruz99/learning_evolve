import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    
    # Boundary constraints
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def make_feasible(v):
    """Adjusts radii to guarantee the configuration satisfies all constraints strictly."""
    cx = v[:N].copy()
    cy = v[N:2*N].copy()
    cr = v[2*N:].copy()
    
    # Enforce boundaries
    cr = np.minimum(cr, cx)
    cr = np.minimum(cr, 1.0 - cx)
    cr = np.minimum(cr, cy)
    cr = np.minimum(cr, 1.0 - cy)
    cr = np.maximum(cr, 0.0)
    
    # Iteratively resolve overlaps
    for _ in range(15):
        dx = cx[PAIR_I] - cx[PAIR_J]
        dy = cy[PAIR_I] - cy[PAIR_J]
        dist = np.hypot(dx, dy)
        overlap = (cr[PAIR_I] + cr[PAIR_J]) - dist
        
        if np.max(overlap) < 1e-10:
            break
            
        shrink = np.maximum(0.0, overlap) / 2.0 + 1e-9
        cr[PAIR_I] = np.maximum(0.0, cr[PAIR_I] - shrink)
        cr[PAIR_J] = np.maximum(0.0, cr[PAIR_J] - shrink)
        
    return np.concatenate([cx, cy, cr])

def generate_initial_configs():
    """Generate diverse, high-quality initial configurations."""
    configs = []
    
    # 1. Hexagonal lattices with specific row counts for N=26
    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 6, 6, 4],
        [7, 5, 5, 5, 4], [6, 6, 5, 5, 4], [5, 5, 6, 6, 4]
    ]
    
    for pat in row_patterns:
        if sum(pat) < N:
            continue
        r0 = 0.105  # Initial guess radius
        for angle in np.linspace(-0.3, 0.3, 9):
            for sx in np.linspace(-0.03, 0.03, 5):
                for sy in np.linspace(-0.03, 0.03, 5):
                    pts = []
                    y = r0 + sy
                    row_idx = 0
                    for count in pat:
                        x_start = r0 + sx + (row_idx % 2) * r0
                        for k in range(count):
                            if len(pts) >= N:
                                break
                            x = x_start + k * 2.0 * r0
                            pts.append([x, y])
                        y += r0 * np.sqrt(3.0)
                        row_idx += 1
                        
                    pts = np.array(pts[:N])
                    
                    # Apply rotation
                    if abs(angle) > 1e-6:
                        c, s = np.cos(angle), np.sin(angle)
                        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                        
                    pts = np.clip(pts, 0.02, 0.98)
                    
                    # Compute feasible radii for this layout
                    r = np.full(N, 0.05)
                    dists = np.linalg.norm(pts[:, None, :] - pts[None, :, :], axis=2)
                    np.fill_diagonal(dists, np.inf)
                    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
                    r = np.minimum(r, np.minimum(pts[:, 0], 1.0 - pts[:, 0]))
                    r = np.minimum(r, np.minimum(pts[:, 1], 1.0 - pts[:, 1]))
                    r = np.clip(r * 0.85, 0.01, 0.2)
                    
                    configs.append(np.concatenate([pts[:, 0], pts[:, 1], r]))

    # 2. Force-directed random starts
    for seed in range(20):
        np.random.seed(seed + 1000)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(200):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    diff = pts[i] - pts[j]
                    d = np.linalg.norm(diff)
                    if d < 0.2 and d > 1e-5:
                        f = (0.2 - d) / d * 0.5
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.02
            pts = np.clip(pts, 0.05, 0.95)
            
        r = np.full(N, 0.04)
        dists = np.linalg.norm(pts[:, None, :] - pts[None, :, :], axis=2)
        np.fill_diagonal(dists, np.inf)
        r = np.minimum(r, np.min(dists, axis=1) / 2.0 * 0.8)
        configs.append(np.concatenate([pts[:, 0], pts[:, 1], r]))

    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    inits = generate_initial_configs()
    
    # Phase 1: Multi-start exploration
    for v0 in inits:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            
            s = -res.fun
            c_val = constraints(res.x)
            if np.min(c_val) >= -1e-7 and s > best_sum:
                best_sum = s
                best_v = res.x.copy()
        except Exception:
            continue
            
    if best_v is None:
        best_v = make_feasible(inits[0])
        
    # Phase 2: Homotopy & Perturbation Refinement
    current_v = best_v.copy()
    for step in range(25):
        np.random.seed(step + 5000)
        v_pert = current_v.copy()
        
        # Shrink radii to create space for rearrangement
        v_pert[2*N:] *= max(0.75, 0.95 - step * 0.005)
        
        # Perturb centers
        noise = np.random.uniform(-0.004, 0.004, 2*N)
        v_pert[:2*N] += noise
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.01, 0.99)
        
        # Randomly swap radii to break symmetry traps
        idx = np.random.choice(N, 4, replace=False)
        v_pert[2*N + idx] = np.roll(v_pert[2*N + idx], 2)
        
        # Ensure feasibility before optimization
        v_pert = make_feasible(v_pert)
        
        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            
            s = -res.fun
            c_val = constraints(res.x)
            if np.min(c_val) >= -1e-7 and s > best_sum:
                best_sum = s
                best_v = res.x.copy()
                current_v = best_v.copy()
        except Exception:
            continue
            
    # Extract final configuration
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()
    centers = np.column_stack((cx, cy))
    
    # Phase 3: Strict Post-Processing for Validator Compliance
    eps = 1e-9
    
    # 1. Enforce boundary constraints strictly
    for i in range(N):
        mr = min(cx[i] - eps, 1.0 - cx[i] - eps, cy[i] - eps, 1.0 - cy[i] - eps)
        cr[i] = min(cr[i], mr)
        cr[i] = max(cr[i], 0.0)
        
    # 2. Enforce non-overlap constraints iteratively
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                sum_r = cr[i] + cr[j]
                # Validator allows: dist >= r1 + r2 - 1e-12
                # We enforce: r1 + r2 <= dist - 2*eps to be safe
                if sum_r > d - 2.0 * eps:
                    shrink = (sum_r - (d - 2.0 * eps)) / 2.0
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    final_sum = float(np.sum(cr))
    return centers, cr, final_sum
