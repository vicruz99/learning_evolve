from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    # Distance to square boundaries
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Distance to other centers
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    
    # Scale down slightly to guarantee strict feasibility for optimizer start
    return np.clip(r * 0.88, 1e-4, 0.25)

def hex_pattern(row_counts, r0, shift_x=0.0, shift_y=0.0):
    """Generate a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r0 + shift_y
    for idx, count in enumerate(row_counts):
        x_start = r0 + shift_x + (idx % 2) * r0
        for _ in range(count):
            if len(pts) >= N:
                break
            pts.append([x_start, y])
            x_start += 2.0 * r0
        y += np.sqrt(3.0) * r0
    return np.array(pts[:N])

def force_relax(centers, steps=200):
    """Relax configuration using repulsive forces to find dense local packing."""
    pts = centers.copy()
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(pts[i, 0] - pts[j, 0], pts[i, 1] - pts[j, 1])
                if d < 0.25 and d > 1e-4:
                    f = (0.25 - d) / d * 0.5
                    diff = pts[i] - pts[j]
                    forces[i] += f * diff
                    forces[j] -= f * diff
        pts += forces * 0.02
        pts = np.clip(pts, 0.03, 0.97)
    return pts
