import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def get_feasible_radii(centers, scale=0.92):
    """Compute strictly feasible initial radii based on local geometry."""
    wall_dists = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]),
                            np.minimum(centers[:,1], 1.0-centers[:,1]))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    np.fill_diagonal(dists, np.inf)
    neighbor_dists = np.min(dists, axis=1)
    r = np.minimum(wall_dists, neighbor_dists / 2.0)
    return r * scale

def force_relax(centers, steps=200):
    """Generate a well-spread initial configuration using repulsion forces."""
    pts = centers.copy()
    lr = 0.03
    for _ in range(steps):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
        mask = (dist < 0.25) & (dist > 1e-5)
        forces = np.zeros_like(pts)
        f_mag = np.zeros_like(dist)
        f_mag[mask] = (0.25 - dist[mask]) / dist[mask]
        f_vec = diff * f_mag[:, :, np.newaxis]
        forces += np.sum(f_vec, axis=1)
        pts += forces * lr
        pts = np.clip(pts, 0.05, 0.95)
        lr *= 0.995
    return pts

    inits = []
    np.random.seed(42)

                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        pts = np.array(pts[:N])

                    p += [sx, sy]
                    p = np.clip(p, 0.03, 0.97)
                    r = get_feasible_radii(p, 0.9)
                    inits.append(np.concatenate([p[:,0], p[:,1], r]))
                    
    # 2. Force-relaxed random starts

        np.random.seed(seed + 100)
        pts = np.random.uniform(0.15, 0.85, (N, 2))

        r = get_feasible_radii(pts, 0.9)
        inits.append(np.concatenate([pts[:,0], pts[:,1], r]))

def run_packing():

    bounds = [(0.0, 1.0)] * (2*N) + [(1e-4, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    inits = generate_inits()
    
    # Primary optimization pass
    for v0 in inits:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,

            s = -res.fun

                    best_v = res.x.copy()

    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
