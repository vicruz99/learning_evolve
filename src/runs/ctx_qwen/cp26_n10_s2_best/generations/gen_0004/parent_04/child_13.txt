import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    
    # Boundary constraints: circles must be inside [0, 1]x[0, 1]
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def compute_init_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    
    # Scale down to guarantee strict feasibility for optimizer start
    return np.clip(r * 0.85, 1e-4, 0.25)

def generate_configs():
    """Generate a diverse set of initial configurations."""
    configs = []
    
    # 1. Hexagonal lattices with various shifts and rotations
    for r0 in np.linspace(0.085, 0.108, 8):
        for angle in np.linspace(-0.2, 0.2, 7):

                for sy in [-0.03, 0.0, 0.03]:
                    pts = []
                    y = r0 + sy
                    row = 0
                    while len(pts) < N + 10:
                        x_start = r0 + sx + (row % 2) * r0
                        x = x_start

                            pts.append([x, y])
                            x += 2.0 * r0
                        y += np.sqrt(3) * r0
                        row += 1
                    pts = np.array(pts[:N])
                    
                    if angle != 0.0:
                        c, s = np.cos(angle), np.sin(angle)
                        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                    
                    pts = np.clip(pts, 0.02, 0.98)
                    r = compute_init_radii(pts)
                    configs.append(np.concatenate([pts[:, 0], pts[:, 1], r]))

    # 2. Perturbed grid configurations

        for sp_y in np.linspace(0.15, 0.19, 5):
            pts = []

                for j in range(5):
                    if len(pts) < N:
                        pts.append([0.05 + i * sp_x, 0.05 + j * sp_y])
            pts = np.array(pts[:N])
            pts += np.random.uniform(-0.015, 0.015, pts.shape)
            pts = np.clip(pts, 0.02, 0.98)
            r = compute_init_radii(pts)
            configs.append(np.concatenate([pts[:, 0], pts[:, 1], r]))

    # 3. Force-relaxed random scatter
    for seed in range(15):
        np.random.seed(seed)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(300):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.linalg.norm(pts[i] - pts[j])

        r = compute_init_radii(pts)
        configs.append(np.concatenate([pts[:, 0], pts[:, 1], r]))

    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-4, 0.25)] * N
    cons = {'type': 'ineq', 'fun': constraints}

    configs = generate_configs()
    best_sum = -1.0
    best_v = None

    # Phase 1: Multi-start optimization
    for v0 in configs:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,

            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass

    if best_v is None:
        best_v = configs[0]

    # Phase 2: Iterative perturbation refinement to escape local minima
    curr = best_v.copy()
    for step in range(25):
        np.random.seed(step + 2000)
        pert = curr.copy()

        pert[:2 * N] += np.random.uniform(-noise, noise, 2 * N)
        pert[:2 * N] = np.clip(pert[:2 * N], 0.01, 0.99)
        
        c_pts = pert[:2 * N].reshape(N, 2)

            pass

    try:
        res = minimize(objective, best_v, method='SLSQP', bounds=bounds,

        s = -res.fun
        if s > best_sum and np.min(constraints(res.x)) >= -1e-7:
            best_v = res.x.copy()
    except Exception:
        pass

    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()

    # Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints strictly
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    cr = np.maximum(cr, 0.0)

    # 2. Enforce non-overlap constraints iteratively with safety margin
    for _ in range(15):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                if cr[i] + cr[j] > d - 1e-11:
                    shrink = (cr[i] + cr[j] - d) / 2.0 + 1e-11
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break

    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
