import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def compute_feasible_radii(centers, factor=0.90):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.5)
    # Distance to boundaries
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    r = np.minimum(r, wall_dists)
    
    # Distance to other centers
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    r = np.minimum(r, min_dists / 2.0)
    
    return np.clip(r * factor, 1e-4, 0.25)

def force_relaxed_config(seed, steps=500):
    """Generate a well-spread configuration using repulsive forces."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    lr = 0.05
    target = 0.22

        dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        mask = (dists < target) & (~np.eye(N, dtype=bool))
        
        forces = np.zeros_like(pts)
        repulsion = np.zeros_like(dists)

        forces += np.sum(f_vec, axis=1)
        
        pts += forces * lr
        pts = np.clip(pts, 0.05, 0.95)
        lr *= 0.995
    return pts

            break
            
    return centers, radii, float(np.sum(radii))
