from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap constraints (squared for smooth gradients)
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def compute_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.25)
    
    # Distance to square boundaries
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    r = np.minimum(r, wall_dists)
    
    # Distance to other centers
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    r = np.minimum(r, min_dists / 2.0)
    
    # Scale down to guarantee strict feasibility
    return np.clip(r * 0.85, 1e-5, 0.25)

    return np.clip(pts, 0.02, 0.98)

def generate_grid_pattern(sx, sy, ox=0.0, oy=0.0, rows=5, cols=6):

    pts = []
    for j in range(rows):
        for i in range(cols):
            x = ox + i * sx
            y = oy + j * sy
            pts.append([x, y])
    pts = np.array(pts[:N])
    return np.clip(pts, 0.02, 0.98)

def force_relaxed_init(seed, steps=400):
    """Generate a well-spread configuration using repulsive forces."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    target = 0.18
    lr = 0.04

        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i+1, N):
                diff = pts[i] - pts[j]
                d = np.hypot(diff[0], diff[1])
                if d < target and d > 1e-6:
                    f = (target - d) / d * 0.5
                    forces[i] += f * diff
                    forces[j] -= f * diff

            for coord in range(2):

                    forces[i, coord] -= 0.1 * (pts[i, coord] - 0.95)
        
        pts += forces * lr
        pts = np.clip(pts, 0.03, 0.97)
        lr *= 0.995
        
    return pts

def ensure_strict_feasibility(centers, radii):
    """Adjust radii to guarantee constraints are strictly satisfied."""
    r = radii.copy()
    
    # Enforce boundary constraints
    r = np.minimum(r, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    r = np.minimum(r, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))

                    r[j] = max(0.0, r[j] - shrink)
                    changed = True
        if not changed:
