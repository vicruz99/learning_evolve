import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d_sq = dx**2 + dy**2
    r_sum_sq = (r[:, None] + r[None, :])**2
    
    triu_idx = np.triu_indices(N, k=1)
    c_pairs = d_sq[triu_idx] - r_sum_sq[triu_idx]
    
    return np.concatenate([c, c_pairs])

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_d = np.min(dists, axis=1) / 2.0
    
    # Scale down slightly to guarantee strict feasibility for optimizer start
    return np.clip(np.minimum(wall, min_d) * 0.88, 1e-4, 0.5)

def sa_find_centers(seed):

    temp = 0.06
    best_centers = centers.copy()

            best_centers = new_centers.copy()
            curr_score = new_score

            if np.random.rand() > prob:
                continue
                
        centers = new_centers
        curr_score = new_score
        temp *= 0.9985
        
    return best_centers

def generate_initial_configs():

    configs = []
    
    # 1. SA-optimized starts

        configs.append(sa_find_centers(s))

                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, cr, float(np.sum(cr))
