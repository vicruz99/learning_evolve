from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap (squared for stability)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    return np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r,
        (x[PAIR_I] - x[PAIR_J])**2 + (y[PAIR_I] - y[PAIR_J])**2 - (r[PAIR_I] + r[PAIR_J])**2
    ])

def compute_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.5)
    
    # Distance to square boundaries
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    r = np.minimum(r, wall_dists)
    
    # Distance to other centers
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            val = d / 2.0
            if val < r[i]:
                r[i] = val
            if val < r[j]:
                r[j] = val
    
    return r * 0.85

def generate_hex_config(r0, angle=0.0, shift=(0.0, 0.0)):

    centers = []
    y = r0 + shift[1]

            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    centers = np.array(centers[:N])
    
    if angle != 0.0:
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        centers = centers - 0.5
        centers = centers @ np.array([[cos_a, -sin_a], [sin_a, cos_a]])
        centers = centers + 0.5
        
    return np.clip(centers, 0.02, 0.98)

def force_relax(centers, steps=400):
    """Relax configuration using repulsive forces to find dense local packing."""
    c = centers.copy()
    for _ in range(steps):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                diff = c[i] - c[j]
                d = np.hypot(diff[0], diff[1])

                    f = (0.25 - d) / d * 0.5
                    forces[i] += f * diff
                    forces[j] -= f * diff
        c += forces * 0.03
        c = np.clip(c, 0.02, 0.98)
    return c
