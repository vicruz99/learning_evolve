from scipy.optimize import minimize, NonlinearConstraint

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraint_ineq(v):
    """Compute inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.empty(4*N + NUM_PAIRS)
    
    # Boundary constraints
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap constraints (squared for smooth gradients)
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def get_feasible_radii(centers, shrink=0.85):
    """Compute strictly feasible initial radii based on local geometry."""
    # Distance to square boundaries
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Distance to other centers
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    
    return np.clip(r * shrink, 1e-5, 0.5)

def force_relax(centers, steps=300):
    """Relax configuration using repulsive forces to find dense local packing."""
    pts = centers.copy()
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[i] - pts[j]
                d = np.hypot(diff[0], diff[1])
                if d < 0.25 and d > 1e-6:
                    f = (0.25 - d) / d * 0.5
                    forces[i] += f * diff
                    forces[j] -= f * diff
        pts += forces * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

        c += np.random.uniform(-0.005, 0.005, c.shape)

        inits.append(c)

        np.random.seed(seed)
        c = np.random.uniform(0.15, 0.85, (N, 2))
        c = force_relax(c)
        inits.append(c)

    return centers, cr, float(np.sum(cr))
