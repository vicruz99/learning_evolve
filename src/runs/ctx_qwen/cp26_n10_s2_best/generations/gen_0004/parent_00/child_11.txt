import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def get_max_radii_lp(centers):
    """
    Computes the maximal feasible radii for fixed centers using Linear Programming.
    Maximizes sum(r) subject to r_i + r_j <= dist_ij and r_i <= wall_dist_i.
    """
    walls = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]),
                       np.minimum(centers[:,1], 1.0-centers[:,1]))
    
    dx = centers[PAIR_I, 0] - centers[PAIR_J, 0]
    dy = centers[PAIR_I, 1] - centers[PAIR_J, 1]
    dists = np.hypot(dx, dy)
    
    # A_ub x <= b_ub
    A_ub = np.zeros((N + len(PAIR_I), N))
    b_ub = np.empty(N + len(PAIR_I))
    
    # Boundary constraints: r_i <= walls[i]
    A_ub[:N, :] = np.eye(N)
    b_ub[:N] = walls
    
    # Pairwise constraints: r_i + r_j <= dists + tolerance
    pair_rows = np.arange(N, N+len(PAIR_I))
    A_ub[pair_rows, PAIR_I] = 1.0
    A_ub[pair_rows, PAIR_J] = 1.0
    b_ub[N:] = dists + 1e-11  # Tolerance ensures strict validator compliance
    
    c_obj = -np.ones(N)  # Minimize -sum(r) => Maximize sum(r)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*N)
    
    if res.success:
        return res.x
    return np.full(N, 0.01)  # Fallback

def generate_configs():
    """Generate diverse initial center configurations."""
    configs = []
    np.random.seed(42)
    
    # 1. Hexagonal lattice patterns with varying densities and rotations
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,6,6,4], [6,6,5,5,4], [7,5,5,5,4]]
    for pat in patterns:
        for r0 in np.linspace(0.08, 0.11, 6):
            for ang in [0.0, 0.05, -0.05, 0.1, -0.1]:
                pts = []
                y = r0
                for row_idx, count in enumerate(pat):
                    x_start = r0 if row_idx % 2 == 0 else 2*r0
                    for k in range(count):
                        if len(pts) >= N: break
                        x = x_start + k * 2.0 * r0
                        pts.append([x, y])
                    y += r0 * np.sqrt(3.0)
                pts = np.array(pts[:N])
                if ang != 0.0:
                    c, s = np.cos(ang), np.sin(ang)
                    pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                pts += np.random.uniform(-0.005, 0.005, pts.shape)
                configs.append(np.clip(pts, 0.02, 0.98))
                
    # 2. Square grid configurations
    for s in np.linspace(0.14, 0.20, 6):
        pts = np.array([[i*s+0.05, j*s+0.05] for i in range(6) for j in range(5)])[:N]
        pts += np.random.uniform(-0.005, 0.005, pts.shape)
        configs.append(np.clip(pts, 0.02, 0.98))
        
    # 3. Force-relaxed random configurations
    for seed in range(20):
        np.random.seed(seed)
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        for _ in range(80):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    diff = pts[i] - pts[j]
                    d = np.linalg.norm(diff)
                    if d < 0.25 and d > 1e-5:
                        f = (0.25 - d) / d
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.03
            pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts)
        
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    configs = generate_configs()
    
    # Phase 1: Multi-start optimization with LP refinement
    for c_init in configs:
        r_init = get_max_radii_lp(c_init)
        v0 = np.concatenate([c_init[:,0], c_init[:,1], r_init])
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            s = -res.fun
            if s > best_sum:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Iterative perturbation & LP recovery to escape local minima
    if best_v is not None:
        curr_v = best_v.copy()
        for step in range(40):
            np.random.seed(step + 100)
            v_p = curr_v.copy()
            noise = 0.006 * (1.0 - step/40.0)
            v_p[:2*N] += np.random.uniform(-noise, noise, 2*N)
            v_p[:2*N] = np.clip(v_p[:2*N], 0.01, 0.99)
            
            # Shrink radii to create free volume for rearrangement
            v_p[2*N:] *= 0.88
            c_pts = v_p[:2*N].reshape(N, 2)
            v_p[2*N:] = get_max_radii_lp(c_pts) * 0.95
            
            try:
                res = minimize(objective, v_p, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                s = -res.fun
                if s > best_sum:
                    c_val = constraints(res.x)
                    if np.min(c_val) >= -1e-7:
                        best_sum = s
                        best_v = res.x.copy()
                        curr_v = best_v.copy()
            except Exception:
                pass
                
    # Extract final centers
    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    
    # Final LP pass to guarantee radii are maximized for the final geometry
    radii = get_max_radii_lp(centers)
    
    return centers, radii, float(np.sum(radii))
