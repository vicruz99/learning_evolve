import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints_and_jac(v):
    """Compute constraints and their exact Jacobian w.r.t variables."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    r_sum = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    
    # Exact Jacobian
    jac = np.zeros((4*N + NUM_PAIRS, 3*N))
    
    # Boundary constraints Jacobian
    jac[:N, :N] = 1.0
    jac[:N, 2*N:] = -1.0
    jac[N:2*N, :N] = -1.0
    jac[N:2*N, 2*N:] = -1.0
    jac[2*N:3*N, N:2*N] = 1.0
    jac[2*N:3*N, 2*N:] = -1.0
    jac[3*N:4*N, N:2*N] = -1.0
    jac[3*N:4*N, 2*N:] = -1.0
    
    # Pairwise constraints Jacobian
    idx_base = 4*N
    for k in range(NUM_PAIRS):

        row = idx_base + k
        
        dx_k = 2.0 * dx[k]
        dy_k = 2.0 * dy[k]
        rs_k = -2.0 * r_sum[k]
        
        jac[row, i] = dx_k
        jac[row, j] = -dx_k
        jac[row, N+i] = dy_k
        jac[row, N+j] = -dy_k
        jac[row, 2*N+i] = rs_k
        jac[row, 2*N+j] = rs_k
        
    return c, jac

def get_feasible_radii(centers, scale=0.92):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    
    return np.clip(r * scale, 1e-6, 0.5)

def make_strictly_feasible(v, tol=1e-9):
    """Adjust radii to guarantee all constraints are strictly satisfied."""
    cx = v[:N].copy()
    cy = v[N:2*N].copy()
    cr = v[2*N:].copy()
    
    # Boundary
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    
    # Pairwise overlaps
    for _ in range(15):
        dx = cx[PAIR_I] - cx[PAIR_J]
        dy = cy[PAIR_I] - cy[PAIR_J]
        dist = np.hypot(dx, dy)
        overlap = cr[PAIR_I] + cr[PAIR_J] - dist
        mask = overlap > tol
        if not np.any(mask):
            break
        shrink = overlap[mask] / 2.0 + tol
        cr[PAIR_I[mask]] -= shrink
        cr[PAIR_J[mask]] -= shrink
        cr = np.maximum(cr, 0.0)
        
    return np.concatenate([cx, cy, cr])

def generate_initial_configs():
