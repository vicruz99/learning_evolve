N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def compute_feasible_radii(centers, margin=0.0):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0 - margin)
    return np.clip(r, 0.001, 0.5)

def repulsion_relax(centers, radii, steps=150, lr=0.015):
    """Quickly resolve overlaps using repulsive forces before gradient optimization."""
    pts = centers.copy()
    r = radii.copy()
    
    for _ in range(steps):
        # Vectorized pairwise repulsion
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.linalg.norm(diff, axis=2)
        np.fill_diagonal(dist, np.inf)
        
        overlap = r[:, None] + r[None, :] - dist
        force_mag = np.maximum(0, overlap)**2 / (dist + 1e-8)
        force_vec = diff * force_mag[:, :, None]
        forces = np.sum(force_vec, axis=1)
        
        # Wall repulsion
        wall_ov = np.maximum(0, r - pts) + np.maximum(0, r - (1.0 - pts))
        forces -= wall_ov * 2.0
        
        pts += forces * lr
        pts = np.clip(pts, 0.005, 0.995)
        lr *= 0.995
    return pts, r
