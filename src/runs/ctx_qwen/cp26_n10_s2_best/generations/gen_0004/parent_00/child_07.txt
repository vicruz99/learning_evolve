import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.empty(4*N + NUM_PAIRS)
    
    # Boundary constraints
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def get_initial_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.5)
    dists = np.linalg.norm(centers[:, None] - centers[None, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    
    # Wall distances
    r = np.minimum(r, centers[:, 0])
    r = np.minimum(r, 1.0 - centers[:, 0])
    r = np.minimum(r, centers[:, 1])
    r = np.minimum(r, 1.0 - centers[:, 1])
    
    return np.clip(r * 0.85, 1e-4, 0.5)

def compute_scaling_factor(centers, radii):
    """Compute exact scaling factor to guarantee strict feasibility."""
    s = 1.0
    # Check boundaries
    for i in range(N):
        x, y = centers[i]
        ri = radii[i]
        if ri > 1e-12:
            s = min(s, x / ri, (1.0 - x) / ri, y / ri, (1.0 - y) / ri)
            
    # Check pairwise distances
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            r_sum = radii[i] + radii[j]
            if r_sum > 1e-12:
                s = min(s, d / r_sum)
                
    return s

def generate_configs():
    """Generate diverse initial configurations."""
    configs = []
    
    # 1. Hexagonal lattice patterns with systematic variations
    patterns = [[6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5]]
    for pat in patterns:

            continue
        r0 = 0.095
        pts = []
        y = r0
        row_idx = 0
        for count in pat:

            for k in range(count):
                if len(pts) >= N:
                    break
                pts.append([x_start + k * 2.0 * r0, y])
            y += r0 * np.sqrt(3.0)
            row_idx += 1
            
        base_pts = np.array(pts[:N])
        
        # Systematic rotations and shifts to find optimal alignment
        for angle in np.linspace(-0.15, 0.15, 7):
            for sx in np.linspace(-0.03, 0.03, 3):
                for sy in np.linspace(-0.03, 0.03, 3):
                    p = base_pts.copy()
                    if angle != 0.0:
                        c, s_val = np.cos(angle), np.sin(angle)
                        rot = p - 0.5
                        p = np.column_stack([rot[:, 0] * c_val - rot[:, 1] * s_val, 
                                             rot[:, 0] * s_val + rot[:, 1] * c_val]) + 0.5
                    p += [sx, sy]
                    p = np.clip(p, 0.02, 0.98)
                    configs.append(p)
                    
    # 2. Force-relaxed random starts to cover non-lattice optima
    for seed in range(25):
        np.random.seed(seed)
        pts = np.random.uniform(0.15, 0.85, (N, 2))

            forces = np.zeros_like(pts)
            for i in range(N):

                        forces[i] += f * diff

            pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts)
        
    return configs

def run_packing():
    """Optimizes packing of 26 circles in a unit square to maximize sum of radii."""
    np.random.seed(42)
    configs = generate_configs()
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None

    for c_init in configs:

        try:

                           options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})

                    best_sum = s

            pass

    radii *= 0.9999999

    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
