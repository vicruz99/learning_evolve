import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap constraints (squared for smooth gradients)
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    r_sum = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    
    return c

def compute_feasible_radii(centers, scale=0.90):
    """Compute strictly feasible initial radii based on local geometry."""
    # Distance to walls
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    r = wall_dists.copy()
    
    # Distance to other centers
    dists = squareform(pdist(centers))
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    r = np.minimum(r, min_dists / 2.0)
    
    return np.clip(r * scale, 1e-6, 0.5)

def generate_initial_configs():
    """Generate diverse initial configurations."""
    configs = []
    
    # 1. Hexagonal lattices with various rotations and densities
    for r0 in np.linspace(0.085, 0.105, 8):
        pts = []
        y = r0
        row = 0
        while len(pts) < N + 6:
            xs = r0 if row % 2 == 0 else 2.0 * r0
            x = xs
            while x <= 1.0 - r0 and len(pts) < N + 6:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
            
        pts = np.array(pts[:N])
        base = pts.copy()
        
        for ang in [0.0, 0.04, -0.04, 0.08, -0.08, 0.12, -0.12]:
            c_a, s_a = np.cos(ang), np.sin(ang)
            rot = base - 0.5
            rot = rot @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
            valid = (rot[:, 0] >= 0.03) & (rot[:, 0] <= 0.97) & \
                    (rot[:, 1] >= 0.03) & (rot[:, 1] <= 0.97)
            if np.sum(valid) >= N:
                configs.append(rot[valid][:N])
                
    # 2. Perturbed square/staggered grids
    for s in np.linspace(0.15, 0.20, 6):
        pts = np.array([[i * s + 0.05, j * s + 0.05] for i in range(6) for j in range(5)])[:N]
        configs.append(pts)
        
    # 3. Force-relaxed random configurations
    for seed in range(20):
        np.random.seed(seed)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(200):
            forces = np.zeros_like(pts)
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
            mask = (dists < 0.25) & (dists > 1e-6) & (~np.eye(N, dtype=bool))
            
            f_mag = np.zeros_like(dists)
            f_mag[mask] = (0.25 - dists[mask]) * 0.5
            f_vec = diff * f_mag[:, :, np.newaxis] / dists[:, :, np.newaxis]
            forces += np.sum(f_vec, axis=1)
            
            pts += forces * 0.05
            pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts)
        
    return configs

def run_packing():
    np.random.seed(42)
    configs = generate_initial_configs()
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    # Phase 1: Multi-start optimization
    for c_init in configs:
        r_init = compute_feasible_radii(c_init, scale=0.88)
        v0 = np.concatenate([c_init[:, 0], c_init[:, 1], r_init])
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        # Fallback to first config optimized once
        v0 = np.concatenate([configs[0][:, 0], configs[0][:, 1], compute_feasible_radii(configs[0])])
        best_v = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 10000, 'ftol': 1e-12}).x
        best_sum = -objective(best_v)
        
    # Phase 2: Adaptive perturbation & refinement to escape local minima
    current_v = best_v.copy()
    for step in range(80):
        np.random.seed(step + 5000)
        v_pert = current_v.copy()
        
        # Decaying noise schedule
        noise_scale = 0.006 * max(0.1, 1.0 - step / 60.0)
        v_pert[:2*N] += np.random.uniform(-noise_scale, noise_scale, 2*N)
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.02, 0.98)
        
        # Shrink radii to guarantee feasibility after perturbation
        c_pts = v_pert[:2*N].reshape(N, 2)
        shrink_factor = max(0.80, 0.95 - step * 0.001)
        v_pert[2*N:] = compute_feasible_radii(c_pts, scale=shrink_factor)
        
        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            pass
            
    # Phase 3: High-precision refinement with trust-constr
    try:
        cons_tc = {'type': 'ineq', 'fun': constraints}
        res_tc = minimize(objective, best_v, method='trust-constr', bounds=bounds, 
                          constraints=cons_tc, options={'maxiter': 5000, 'gtol': 1e-12, 'verbose': 0})
        s_tc = -res_tc.fun
        if s_tc > best_sum and np.min(constraints(res_tc.x)) >= -1e-7:
            best_sum = s_tc
            best_v = res_tc.x.copy()
    except Exception:
        pass

    # Extract results
    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    radii = best_v[2*N:].copy()
    
    # Phase 4: Strict post-processing to guarantee validator compliance
    # Enforce boundary constraints strictly
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    radii = np.minimum(radii, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    radii = np.maximum(radii, 0.0)
    
    # Enforce non-overlap constraints iteratively with minimal safety margin
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-10
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
