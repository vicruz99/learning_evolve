import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints_ineq(v):
    """Inequality constraints: boundaries and non-overlap using actual distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist >= r_i + r_j
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist = np.sqrt(dx**2 + dy**2)
    c[4*N:] = dist - (r[PAIR_I] + r[PAIR_J])
    
    return c

def constraints_squared(v):
    """Constraints using squared distances for numerical stability."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def compute_feasible_radii(centers, scale=0.90):
    """Compute strictly feasible initial radii."""
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    r = np.minimum(r, min_dists / 2.0)
    
    return np.clip(r * scale, 0.001, 0.25)

def make_strictly_feasible(centers, radii, max_iter=30):
    """Adjust radii to guarantee strict feasibility with minimal shrinkage."""
    r = radii.copy()
    cx = centers[:, 0]
    cy = centers[:, 1]
    
    for _ in range(max_iter):
        # Boundary constraints
        r = np.minimum(r, cx)
        r = np.minimum(r, 1.0 - cx)
        r = np.minimum(r, cy)
        r = np.minimum(r, 1.0 - cy)
        
        # Pairwise constraints
        dx = centers[PAIR_I, 0] - centers[PAIR_J, 0]
        dy = centers[PAIR_I, 1] - centers[PAIR_J, 1]
        dist = np.hypot(dx, dy)
        overlap = (r[PAIR_I] + r[PAIR_J]) - dist
        
        if np.max(overlap) < 1e-12:
            break
        
        shrink = np.maximum(0.0, overlap) / 2.0 + 1e-8
        r[PAIR_I] = np.maximum(0.0, r[PAIR_I] - shrink)
        r[PAIR_J] = np.maximum(0.0, r[PAIR_J] - shrink)
    
    return r

def generate_hex_lattice(row_counts, r0=0.095, angle=0.0, shift_x=0.0, shift_y=0.0):
    """Generate hexagonal lattice with specified row counts."""
    pts = []
    y = r0 + shift_y
    for row_idx, count in enumerate(row_counts):
        x_start = r0 + shift_x
        if row_idx % 2 == 1:
            x_start += r0
        for k in range(count):
            x = x_start + k * 2.0 * r0
            pts.append([x, y])
        y += r0 * np.sqrt(3.0)
    
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        pts = pts - 0.5
        pts = pts @ np.array([[cos_a, -sin_a], [sin_a, cos_a]]) + 0.5
    
    pts = np.clip(pts, 0.01, 0.99)
    return pts

def run_optimization(v0, method='SLSQP', cons_func=constraints_squared, maxiter=4000):
    """Run a single optimization from initial guess."""
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': cons_func}
    
    try:
        res = minimize(objective, v0, method=method, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        return res
    except Exception:
        return None
