from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def make_feasible(centers, radii):
    """Iteratively adjust radii to guarantee strict feasibility for given centers."""
    radii = radii.copy()
    # Enforce boundary constraints
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    radii = np.minimum(radii, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Enforce pairwise non-overlap constraints
    for _ in range(15):
        dx = centers[PAIR_I, 0] - centers[PAIR_J, 0]
        dy = centers[PAIR_I, 1] - centers[PAIR_J, 1]
        dist = np.hypot(dx, dy)
        overlap = radii[PAIR_I] + radii[PAIR_J] - dist
        
        if np.max(overlap) < 1e-9:
            break
            
        shrink = np.maximum(0.0, overlap) / 2.0 + 1e-7
        radii[PAIR_I] = np.maximum(0.0, radii[PAIR_I] - shrink)
        radii[PAIR_J] = np.maximum(0.0, radii[PAIR_J] - shrink)
        
    return radii

def run_packing():

    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).

    np.random.seed(42)

    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0

                    pts += np.random.uniform(-0.005, 0.005, pts.shape)
                    pts = np.clip(pts, 0.02, 0.98)
                    r = make_feasible(pts, np.full(N, 0.04))
                    inits.append(np.concatenate([pts[:, 0], pts[:, 1], r]))

        np.random.seed(seed + 1000)
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        r = make_feasible(pts, np.full(N, 0.035))

                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
