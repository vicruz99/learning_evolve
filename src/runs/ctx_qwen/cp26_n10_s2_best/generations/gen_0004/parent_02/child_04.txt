import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def make_feasible(centers, r_init):
    """Iteratively adjust radii to guarantee strict feasibility."""
    r = r_init.copy()
    for _ in range(20):
        r = np.minimum(r, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
        r = np.minimum(r, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
        
        dx = centers[PAIR_I, 0] - centers[PAIR_J, 0]
        dy = centers[PAIR_I, 1] - centers[PAIR_J, 1]
        dist = np.hypot(dx, dy)
        overlap = (r[PAIR_I] + r[PAIR_J]) - dist
        
        if np.max(overlap) < 1e-9:
            break
            
        shrink = np.maximum(0.0, overlap) / 2.0 + 1e-7
        r[PAIR_I] = np.maximum(0.0, r[PAIR_I] - shrink)
        r[PAIR_J] = np.maximum(0.0, r[PAIR_J] - shrink)
    return r

def generate_hex_pattern(row_counts, r_est, shift_x, shift_y, angle):
    """Generates a hexagonal lattice configuration with specified row counts."""
    centers = []
    y = r_est + shift_y
    for idx, count in enumerate(row_counts):
        x_start = r_est + shift_x + (idx % 2) * r_est
        for k in range(count):
            centers.append([x_start + k * 2.0 * r_est, y])
        y += r_est * np.sqrt(3.0)
        
    pts = np.array(centers[:N])
    if angle != 0.0:
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        pts = pts - 0.5
        rot = np.array([[cos_a, -sin_a], [sin_a, cos_a]])
        pts = pts @ rot.T + 0.5
    return np.clip(pts, 0.02, 0.98)

def generate_configs():

    configs = []

    ]
    
    for pat in row_patterns:
        if sum(pat) != N:
            continue
        for r_est in [0.085, 0.095, 0.105]:
            for sx in [-0.03, 0.0, 0.03]:
                for sy in [-0.03, 0.0, 0.03]:
                    for ang in [0.0, 0.05, -0.05]:
                        pts = generate_hex_pattern(pat, r_est, sx, sy, ang)
                        # Add slight jitter to break exact symmetry
                        pts += np.random.uniform(-0.002, 0.002, pts.shape)
                        r_init = make_feasible(pts, np.full(N, 0.04))
                        configs.append(np.concatenate([pts[:, 0], pts[:, 1], r_init]))
                        
    # Random repelled starts
    for seed in range(15):
        np.random.seed(seed + 500)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        r_init = make_feasible(pts, np.full(N, 0.03))
        configs.append(np.concatenate([pts[:, 0], pts[:, 1], r_init]))
        
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    inits = generate_configs()
    
    # Phase 1: Multi-start optimization
    for v0 in inits:

            res = minimize(objective, v0, method='SLSQP', bounds=bounds,

            s = -res.fun
            cv = constraints(res.x)

                best_v = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Perturbation refinement to escape local minima

    # 1. Enforce boundary constraints strictly
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))

    radii = np.maximum(radii, 0.0)

    for _ in range(30):

                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
