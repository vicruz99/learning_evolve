import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def compute_lp_radii(centers):
    """Compute optimal radii for fixed centers using Linear Programming."""
    c = -np.ones(N)
    A_ub = np.zeros((4 * N + NUM_PAIRS, N))
    b_ub = np.zeros(4 * N + NUM_PAIRS)
    
    # Wall constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(N):
        w = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[i, i] = 1.0
        b_ub[i] = w
        
    idx = N
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i, j in zip(PAIR_I, PAIR_J):
        d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        b_ub[idx] = d
        idx += 1
        
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return np.full(N, 0.01)

def objective_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def constraints_joint(v):
    """Inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N:2 * N]
    r = v[2 * N:]
    
    c = np.empty(4 * N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2 * N] = 1.0 - x - r
    c[2 * N:3 * N] = y - r
    c[3 * N:4 * N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    r_sum = r[PAIR_I] + r[PAIR_J]
    c[4 * N:] = dx**2 + dy**2 - r_sum**2
    
    return c

def force_relax_init(seed, steps=150):
    """Generate a well-spaced initial configuration using repulsive forces."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    for _ in range(steps):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        dist = np.sqrt(dist_sq + 1e-12)
        mask = (dist < 0.25) & (~np.eye(N, dtype=bool))
        
        f_mag = np.zeros_like(dist)
        f_mag[mask] = (0.25 - dist[mask]) * 0.5
        f_vec = diff * f_mag[:, :, np.newaxis] / (dist[:, :, np.newaxis] + 1e-12)
        forces = np.sum(f_vec, axis=1)
        
        pts += forces * 0.05
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def generate_configs():
    """Generate diverse initial configurations."""
    configs = []
    shifts = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    
    # 1. Rotated Hexagonal Lattices
    for r0 in np.linspace(0.085, 0.115, 9):
        for angle in np.linspace(-0.3, 0.3, 7):
            for sx, sy in shifts:
                pts = []
                y = r0 + sy * 0.015
                row = 0
                while len(pts) < N + 5:
                    x_start = r0 + sx * 0.015 + (row % 2) * r0
                    x = x_start
                    while x <= 1.0 - r0 and len(pts) < N + 5:
                        pts.append([x, y])
                        x += 2.0 * r0
                    y += r0 * np.sqrt(3.0)
                    row += 1
                    
                pts = np.array(pts[:N])
                if angle != 0.0:
                    c, s = np.cos(angle), np.sin(angle)
                    pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                    
                pts += np.random.uniform(-0.003, 0.003, pts.shape)
                pts = np.clip(pts, 0.02, 0.98)
                configs.append(pts)
                
    # 2. Force-relaxed random starts
    for seed in range(15):
        configs.append(force_relax_init(seed))
        
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_joint}
    
    configs = generate_configs()
    best_v = None
    best_sum = -1.0
    
    # Phase 1: Multi-start SLSQP optimization
    for c_init in configs:
        r_init = np.full(N, 0.09)
        v0 = np.concatenate([c_init[:, 0], c_init[:, 1], r_init])
        try:
            res = minimize(objective_joint, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
            s = -res.fun
            if s > best_sum:
                if np.min(constraints_joint(res.x)) >= -1e-5:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = np.concatenate([configs[0][:, 0], configs[0][:, 1], np.full(N, 0.05)])
        
    # Phase 2: Iterative perturbation & LP refinement to escape local minima
    current_v = best_v.copy()
    for step in range(20):
        scale = 0.97 - step * 0.002
        v_p = current_v.copy()
        noise = 0.004 * (1.0 - step / 20.0)
        v_p[:2 * N] += np.random.uniform(-noise, noise, 2 * N)
        v_p[:2 * N] = np.clip(v_p[:2 * N], 0.01, 0.99)
        
        c_p = v_p[:2 * N].reshape(N, 2)
        r_p = compute_lp_radii(c_p) * scale
        v_p[2 * N:] = r_p
        
        try:
            res = minimize(objective_joint, v_p, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13})
            s = -res.fun
            if s > best_sum:
                if np.min(constraints_joint(res.x)) >= -1e-5:
                    best_sum = s
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            pass
            
    # Phase 3: Final extraction & strict LP polishing
    centers = np.column_stack((best_v[:N], best_v[N:2 * N]))
    radii = compute_lp_radii(centers)
    
    # Minimal safety shrink to guarantee validator compliance against float precision
    radii *= 0.9999999
    radii = np.maximum(radii, 1e-9)
    
    return centers, radii, float(np.sum(radii))
