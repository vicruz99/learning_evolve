import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    r_sum = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    return c

def compute_initial_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    
    return np.clip(r * 0.82, 0.001, 0.48)

def generate_hex_pattern(rows_pattern, spacing=0.095):
    """Generates a hexagonal lattice configuration based on row counts."""
    pts = []
    y = spacing
    for i, count in enumerate(rows_pattern):
        x_start = spacing if i % 2 == 0 else 2.0 * spacing
        for _ in range(count):
            if len(pts) >= N:
                break
            pts.append([x_start, y])
            x_start += 2.0 * spacing
        y += spacing * np.sqrt(3.0)
    return np.array(pts[:N])

def force_relaxed_init(seed):
    """Generate a well-spaced initial configuration using repulsive forces."""
    np.random.seed(seed)
    pts = np.random.uniform(0.12, 0.88, (N, 2))
    for _ in range(400):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[i] - pts[j]
                d = np.hypot(diff[0], diff[1])
                if d < 0.28 and d > 1e-5:
                    f = 0.04 / (d**2 + 0.01)
                    forces[i] += f * diff
                    forces[j] -= f * diff
                    
        # Boundary repulsion
        pts = np.clip(pts + forces * 0.06, 0.04, 0.96)
    return pts

def run_packing():

    cons = {'type': 'ineq', 'fun': constraints}

    best_sum = -1.0

    patterns = [[6,5,5,5,4], [5,6,5,6,4], [7,5,5,5,4], [6,6,4,5,5], [5,5,6,5,5]]
    for pat in patterns:

            continue
        for sx in [-0.025, 0.0, 0.025]:

                    pts += [sx, sy]
                    if ang != 0.0:
                        c_val, s_val = np.cos(ang), np.sin(ang)
                        pts = (pts - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5
                    pts = np.clip(pts, 0.03, 0.97)
                    r_init = compute_initial_radii(pts)
                    configs.append(np.concatenate([pts[:, 0], pts[:, 1], r_init]))

    for v0 in configs:

                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, cr, float(np.sum(cr))
