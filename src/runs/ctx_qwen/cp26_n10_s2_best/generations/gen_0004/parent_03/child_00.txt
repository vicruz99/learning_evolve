import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    # Boundary constraints
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap (squared distance for smoother gradients)
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    r_sum = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    return c

def compute_initial_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    # Distance to walls
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    # Distance to other centers
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    return np.clip(r * 0.95, 0.001, 0.5)

def generate_hex_configs():
    """Generate high-quality initial configurations from rotated hexagonal lattices."""
    configs = []
    # Row patterns that sum to 26
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 6, 5, 4], [4, 6, 6, 6, 4],
        [6, 6, 5, 5, 4], [5, 5, 6, 6, 4], [7, 5, 5, 5, 4],
        [4, 5, 6, 6, 5], [5, 6, 5, 6, 4]
    ]
    
    for pat in patterns:
        if sum(pat) != N:
            continue
            
        pts = []
        y = 0.0
        row = 0
        for count in pat:
            x_start = 0.5 if row % 2 == 0 else 0.0
            for k in range(count):
                x = x_start + k * 1.0
                pts.append([x, y])
            y += np.sqrt(3.0) / 2.0
            row += 1
        pts = np.array(pts)
        
        # Try a range of rotation angles
        for ang in np.linspace(0.0, 25.0, 21):
            theta = np.radians(ang)
            c_val, s_val = np.cos(theta), np.sin(theta)
            
            # Rotate around centroid
            pts_rot = pts - pts.mean(axis=0)
            pts_rot = pts_rot @ np.array([[c_val, -s_val], [s_val, c_val]])
            
            # Normalize to unit bounding box and center
            min_coords = pts_rot.min(axis=0)
            max_coords = pts_rot.max(axis=0)
            span = max_coords - min_coords
            max_dim = max(span[0], span[1])
            
            if max_dim > 1e-6:
                pts_norm = (pts_rot - min_coords) / max_dim
                # Center in [0, 1]
                pts_final = pts_norm - pts_norm.mean(axis=0) + 0.5
                pts_final = np.clip(pts_final, 0.02, 0.98)
                configs.append(pts_final)
                
    return configs

def run_packing():
    """Optimizes packing of 26 circles in a unit square to maximize sum of radii."""
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    # Generate diverse structured starts
    hex_configs = generate_hex_configs()
    inits = hex_configs
    
    # Phase 1: Multi-start exploration from structured lattices
    for centers in inits:
        r_init = compute_initial_radii(centers)
        v0 = np.concatenate([centers[:, 0], centers[:, 1], r_init])
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, 
                           constraints=cons, 
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            
            s = -res.fun
            if s > best_sum:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-6:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Fallback if optimization fails completely
    if best_v is None and len(inits) > 0:
        centers_fallback = inits[0]
        r_fallback = compute_initial_radii(centers_fallback)

            # Shrink radii to create slack, then recompute feasible radii

            pert[2*N:] = compute_initial_radii(c_pts) * 0.96

                pass
                
    # Phase 3: Extract and strictly enforce validation constraints
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()

    # 1. Enforce boundary constraints strictly
    for i in range(N):
        mr = min(cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
        cr[i] = min(cr[i], mr)
        
    # 2. Enforce non-overlap constraints iteratively with safety margin
    # Validator requires: dist >= r1 + r2 - 1e-12

        for i in range(N):

                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    # Final safety clamp
    cr = np.maximum(cr, 0.0)
    
    return centers, cr, float(np.sum(cr))
