import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

# Precompute constraint matrix structure for LP (static)
# Rows 0..NUM_PAIRS-1: pairwise constraints r_i + r_j <= d_ij
# Rows NUM_PAIRS..2*NUM_PAIRS-1: wall constraints r_i <= w_i
A_ub_pairs = np.zeros((NUM_PAIRS, N))
A_ub_pairs[np.arange(NUM_PAIRS), PAIR_I] = 1.0
A_ub_pairs[np.arange(NUM_PAIRS), PAIR_J] = 1.0
A_ub_walls = np.eye(N)
A_ub = np.vstack([A_ub_pairs, A_ub_walls])

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def compute_max_radii_lp(centers, wall_dists):
    """Solve LP to find maximum sum of radii for fixed centers."""
    dx = centers[PAIR_I, 0] - centers[PAIR_J, 0]
    dy = centers[PAIR_I, 1] - centers[PAIR_J, 1]
    d = np.hypot(dx, dy)
    
    # b_ub: pairwise distances followed by wall distances
    b_ub = np.concatenate([d, wall_dists])
    
    try:
        # Minimize -sum(r) subject to A_ub @ r <= b_ub and 0 <= r <= 0.5
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0, 0.5), method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    # Fallback heuristic if LP fails
    r = wall_dists.copy()

            i, j = PAIR_I[k], PAIR_J[k]

                shrink = (r[i] + r[j] - d[k]) / 2.0
                r[i] -= shrink
                r[j] -= shrink
    return np.maximum(r, 0.0)

                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3.0)
            row += 1
            
        pts = np.array(pts[:N])
        cx, cy = 0.5, 0.5
        pts[:, 0] -= cx
        pts[:, 1] -= cy
        c_val, s_val = np.cos(angle), np.sin(angle)
        pts = pts @ np.array([[c_val, -s_val], [s_val, c_val]])
        pts[:, 0] += cx
        pts[:, 1] += cy
        configs.append(np.clip(pts, 0.02, 0.98))
