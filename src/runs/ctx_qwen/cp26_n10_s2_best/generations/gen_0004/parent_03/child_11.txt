import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    return -np.sum(v[2*N:])

def constraints(v):
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def compute_initial_radii(centers):
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    
    return np.clip(r * 0.88, 0.001, 0.5)

def generate_hex_pattern(r0, angle=0.0, sx=0.0, sy=0.0):
    pts = []
    y = r0 + sy
    row = 0
    while len(pts) < N + 15:
        x_start = r0 + sx + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 15:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
        row += 1
    
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = pts - 0.5
        pts = pts @ np.array([[c, -s], [s, c]])
        pts = pts + 0.5
    
    return np.clip(pts, 0.02, 0.98)

def generate_boundary_hugging_config(seed):
    """Generate configuration with circles hugging boundaries."""
    np.random.seed(seed)
    pts = []
    
    # Corner circles
    corner_margin = 0.08 + np.random.uniform(-0.01, 0.01)
    pts.append([corner_margin, corner_margin])
    pts.append([1.0 - corner_margin, corner_margin])
    pts.append([corner_margin, 1.0 - corner_margin])
    pts.append([1.0 - corner_margin, 1.0 - corner_margin])
    
    # Edge circles
    edge_positions = np.sort(np.random.uniform(0.2, 0.8, 12))
    margin = 0.07 + np.random.uniform(-0.005, 0.005)
    
    for p in edge_positions[:3]:
        pts.append([margin, p])
        pts.append([1.0 - margin, p])
    for p in edge_positions[3:6]:
        pts.append([p, margin])
        pts.append([p, 1.0 - margin])
    
    # Interior circles
    interior_count = N - len(pts)
    if interior_count > 0:
        interior = np.random.uniform(0.2, 0.8, (interior_count, 2))
        pts.extend(interior.tolist())
    
    pts = np.array(pts[:N])
    return np.clip(pts, 0.03, 0.97)

def force_relax(centers, steps=800):
    pts = centers.copy()
    lr = 0.03
    target_dist = 0.28
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[i, 0] - pts[j, 0]
                dy = pts[i, 1] - pts[j, 1]
                d = np.hypot(dx, dy)
                if d < target_dist and d > 1e-4:
                    f = (target_dist - d) / d * 0.8
                    forces[i, 0] += f * dx
                    forces[i, 1] += f * dy
                    forces[j, 0] -= f * dx
                    forces[j, 1] -= f * dy
        
        for i in range(N):
            if pts[i, 0] < 0.12:
                forces[i, 0] += 0.12
            if pts[i, 0] > 0.88:
                forces[i, 0] -= 0.12
            if pts[i, 1] < 0.12:
                forces[i, 1] += 0.12
            if pts[i, 1] > 0.88:
                forces[i, 1] -= 0.12
        
        pts += forces * lr
        pts = np.clip(pts, 0.05, 0.95)
        lr *= 0.997
    
    return pts

def optimize_config(v0, bounds, cons):
    """Try optimizing with multiple methods."""
    best_res = None
    best_obj = np.inf
    
    # Try SLSQP
    try:
        res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
        if -res.fun < best_obj:
            cv = constraints(res.x)
            if np.min(cv) >= -1e-6:
                best_obj = -res.fun
                best_res = res
    except Exception:
        pass
    
    # Try COBYLA (gradient-free)
    try:
        bounds_list = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        cons_eq = []
        cons_ineq = [{'type': 'ineq', 'fun': constraints}]
        
        res_cobyla = minimize(objective, v0, method='COBYLA',
                              bounds=bounds_list, constraints=cons_ineq,
                              options={'maxiter': 15000, 'rhobeg': 0.001, 'rhofinal': 1e-10})
        if -res_cobyla.fun < best_obj:
            cv = constraints(res_cobyla.x)
            if np.min(cv) >= -1e-5:
                best_obj = -res_cobyla.fun
                best_res = res_cobyla
    except Exception:
        pass
    
    return best_res
