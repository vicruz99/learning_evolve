import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints_sq(v):
    """Squared distance constraints for SLSQP."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    r_sum = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    return c

def compute_feasible_radii(centers, shrink_factor=0.82):
    """Compute feasible initial radii based on geometry."""
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    
    return np.clip(r * shrink_factor, 0.001, 0.25)

def hex_lattice_config(r0, angle=0.0, shift=(0.0, 0.0)):
    """Generate hexagonal lattice configuration."""
    pts = []
    y = r0 + shift[1]
    row = 0
    while len(pts) < N + 10:
        x_start = r0 + shift[0] + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
        row += 1
    
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        pts[:, 0] -= cx
        pts[:, 1] -= cy
        c_val, s_val = np.cos(angle), np.sin(angle)
        pts = pts @ np.array([[c_val, -s_val], [s_val, c_val]])
        pts[:, 0] += cx
        pts[:, 1] += cy
    
    return np.clip(pts, 0.01, 0.99)

def repulsion_layout(seed, steps=600):
    """Generate layout using repulsive forces."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    
    for step in range(steps):
        lr = 0.04 * (1.0 - step / steps)
        forces = np.zeros_like(pts)
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[i, 0] - pts[j, 0]
                dy = pts[i, 1] - pts[j, 1]
                d = np.hypot(dx, dy)
                if d < 0.22 and d > 1e-6:
                    f = (0.22 - d) / (d * d + 1e-10) * 0.3
                    forces[i, 0] += f * dx
                    forces[i, 1] += f * dy
                    forces[j, 0] -= f * dx
                    forces[j, 1] -= f * dy
        
        # Wall repulsion
        pts[:, 0] = np.clip(pts[:, 0] + forces[:, 0] * lr, 0.05, 0.95)
        pts[:, 1] = np.clip(pts[:, 1] + forces[:, 1] * lr, 0.05, 0.95)
    
    return pts

def grid_config(rows, cols, jitter=0.005):
    """Generate staggered grid configuration."""
    pts = []
    total = 0
    for row_idx in range(rows):
        count = cols + (1 if row_idx % 2 == 1 else 0)
        x_start = 0.07 if row_idx % 2 == 0 else 0.13
        for c in range(count):
            if total >= N:
                break
            x = x_start + c * 0.15
            y = 0.07 + row_idx * 0.17
            pts.append([x, y])
            total += 1
    
    pts = np.array(pts[:N])
    pts += np.random.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def optimize_from(v0, maxiter=8000, ftol=1e-13):
    """Run SLSQP optimization from initial point."""
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.25)] * N
    cons = {'type': 'ineq', 'fun': constraints_sq}
    
    try:
        res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return res
    except Exception:
        return None

def enforce_constraints(centers, radii, tol=1e-10):
    """Enforce all constraints strictly."""
    cx = centers[:, 0].copy()
    cy = centers[:, 1].copy()
    cr = radii.copy()
    
    # Boundary constraints
    for i in range(N):
        cr[i] = min(cr[i], cx[i] - tol, 1.0 - cx[i] - tol, 
                    cy[i] - tol, 1.0 - cy[i] - tol)
        cr[i] = max(cr[i], 0.0)
    
    # Non-overlap constraints
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                if cr[i] + cr[j] > d + tol:
                    shrink = (cr[i] + cr[j] - d) / 2.0 + tol
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
    
    return np.column_stack((cx, cy)), cr
