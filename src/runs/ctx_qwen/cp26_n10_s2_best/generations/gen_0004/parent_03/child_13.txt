import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def compute_initial_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    # Distance to walls
    r = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    r = np.minimum(r, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Distance to other centers
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    
    # Scale down to guarantee strict feasibility for the optimizer start
    return np.clip(r * 0.85, 0.001, 0.5)

                    x += 2.0 * r0

                row += 1
            pts = np.array(pts[:N])

                pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)

                    diff = pts[i] - pts[j]
                    d = np.linalg.norm(diff)
                    if d < 0.25 and d > 1e-5:
                        f = (0.25 - d) / d
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.03
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)

        pts = np.array([[0.08 + i * 0.16 + np.random.uniform(-0.015, 0.015),

                        for i in range(6) for j in range(5)])[:N]
        inits.append(np.clip(pts, 0.02, 0.98))
        
    return inits

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None

        v0 = np.concatenate([centers[:, 0], centers[:, 1], r_init])

    # 1. Enforce boundary constraints strictly

        if not changed:

    return centers, cr, float(np.sum(cr))
