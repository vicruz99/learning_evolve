import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    r_sum = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    return c

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    return np.clip(r * 0.92, 0.005, 0.5)

def generate_configs():
    """Generate a diverse set of initial configurations."""
    configs = []
    np.random.seed(42)
    
    # 1. Hexagonal lattices with varying densities and rotations
    for r0 in np.linspace(0.085, 0.115, 8):
        for ang in np.linspace(-0.25, 0.25, 7):
            pts = []
            y = r0
            row = 0
            while len(pts) < N + 8:
                x = r0 + (row % 2) * r0
                while x <= 1.0 - r0 and len(pts) < N + 8:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
                row += 1
            pts = np.array(pts[:N])
            c_val, s_val = np.cos(ang), np.sin(ang)
            pts = (pts - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5
            pts += np.random.uniform(-0.008, 0.008, pts.shape)
            configs.append(np.clip(pts, 0.02, 0.98))

    # 2. Structured row patterns (known to yield dense packings)
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [7,5,6,8], [4,6,6,6,4], [5,5,6,5,5]]
    for pat in patterns:
        if sum(pat) < N: continue
        pts = []
        y = 0.11
        for idx, cnt in enumerate(pat):
            x_start = 0.11 + (idx % 2) * 0.055
            for k in range(cnt):
                if len(pts) >= N: break
                pts.append([x_start + k * 0.155, y])
            y += 0.175
        pts = np.array(pts[:N])
        pts += np.random.uniform(-0.012, 0.012, pts.shape)
        configs.append(np.clip(pts, 0.02, 0.98))

    # 3. Force-directed relaxation from random starts
    for seed in range(20):
        np.random.seed(seed)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(600):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    dx = pts[i,0] - pts[j,0]
                    dy = pts[i,1] - pts[j,1]
                    d = np.hypot(dx, dy)
                    if d < 0.22 and d > 1e-4:
                        f = (0.22 - d) / d * 0.06
                        forces[i, 0] += f * dx
                        forces[i, 1] += f * dy
                        forces[j, 0] -= f * dx
                        forces[j, 1] -= f * dy
            pts += forces * 0.025
            pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts)
        
    return configs

def run_packing():
    """Optimizes packing of 26 circles in a unit square to maximize sum of radii."""
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2*N) + [(0.001, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    configs = generate_configs()
    
    # Phase 1: Multi-start optimization
    for centers in configs:
        r_init = get_feasible_radii(centers)
        v0 = np.concatenate([centers[:, 0], centers[:, 1], r_init])
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,

            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-6:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Fallback initialization
    if best_v is None:
        centers_f = configs[0]
        r_f = get_feasible_radii(centers_f)
        best_v = np.concatenate([centers_f[:, 0], centers_f[:, 1], r_f])
        best_sum = np.sum(r_f)

    # Phase 2: Iterative refinement & perturbation to escape local minima
    current_v = best_v.copy()

        np.random.seed(step + 1000)
        pert = current_v.copy()

        centers_p = pert[:2*N].reshape(N, 2)

                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            pass
            
    # Extract results
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()
    centers = np.column_stack((cx, cy))
    
    # Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints strictly
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    cr = np.maximum(cr, 0.0)
    
    # 2. Enforce non-overlap constraints iteratively with minimal safety margin
    for _ in range(40):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                if cr[i] + cr[j] > d - 1e-9:
                    shrink = (cr[i] + cr[j] - d) / 2.0 + 1e-9
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, cr, float(np.sum(cr))
