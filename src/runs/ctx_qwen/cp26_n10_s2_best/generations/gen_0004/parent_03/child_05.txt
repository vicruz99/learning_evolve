import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    # Boundary constraints: circles inside [0,1]x[0,1]
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    return r

def generate_hex_configs():
    """Generate hexagonal lattice configurations with rotations and shifts."""
    configs = []
    for seed in range(35):
        np.random.seed(seed)
        r0 = 0.085 + np.random.uniform(-0.01, 0.01)
        pts = []
        y = r0
        row = 0
        while len(pts) < N + 8:
            x_start = r0 + (row % 2) * r0
            x = x_start
            while x <= 1.0 - r0 and len(pts) < N + 8:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
            
        pts = np.array(pts[:N])
        angle = np.random.uniform(-0.3, 0.3)
        shift = np.random.uniform(-0.03, 0.03, 2)
        c_val, s_val = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5 + shift
        pts = np.clip(pts, 0.02, 0.98)
        configs.append(pts)
    return configs

def generate_force_configs():
    """Generate configurations using repulsive forces with boundary pushing."""
    configs = []
    for seed in range(25):
        np.random.seed(seed)
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        lr = 0.05
        for _ in range(800):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    dx = pts[i, 0] - pts[j, 0]
                    dy = pts[i, 1] - pts[j, 1]
                    d = np.hypot(dx, dy)
                    if d < 0.22 and d > 1e-5:
                        f = (0.22 - d) / d
                        forces[i, 0] += f * dx
                        forces[i, 1] += f * dy
                        forces[j, 0] -= f * dx
                        forces[j, 1] -= f * dy
                        
            # Push towards boundaries to exploit edge slack
            for i in range(N):
                if pts[i, 0] < 0.15: forces[i, 0] += 0.2
                if pts[i, 0] > 0.85: forces[i, 0] -= 0.2
                if pts[i, 1] < 0.15: forces[i, 1] += 0.2
                if pts[i, 1] > 0.85: forces[i, 1] -= 0.2
                
            pts += forces * lr
            pts = np.clip(pts, 0.03, 0.97)
            lr *= 0.994
        configs.append(pts)
    return configs

def run_packing():
    """Optimizes packing of 26 circles in a unit square to maximize sum of radii."""
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    # Combine diverse initialization strategies
    all_configs = generate_hex_configs() + generate_force_configs()

        v0 = np.concatenate([cfg[:, 0], cfg[:, 1], r_init])
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        # Fallback to first config with safe radii
        cfg = all_configs[0]
        r_safe = get_feasible_radii(cfg) * 0.5
        best_v = np.concatenate([cfg[:, 0], cfg[:, 1], r_safe])
        
    # Phase 2: Intensive local refinement to escape local minima
    curr_v = best_v.copy()
    for step in range(60):
        np.random.seed(step + 100)
        pert = curr_v.copy()
        
        # Decaying noise for exploration -> exploitation
        noise_scale = 0.006 * (1.0 - step / 70.0)
        pert[:2 * N] += np.random.uniform(-noise_scale, noise_scale, 2 * N)
        pert[:2 * N] = np.clip(pert[:2 * N], 0.01, 0.99)
        
        # Shrink radii to create slack, then recompute strictly feasible radii
        c_pts = pert[:2 * N].reshape(N, 2)
        r_feas = get_feasible_radii(c_pts)
        shrink_factor = 0.82 + 0.12 * np.exp(-step / 15.0)
        pert[2 * N:] = r_feas * shrink_factor
        
        try:

            s = -res.fun

        except Exception:
            pass
            
    # Extract optimal configuration
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()
    centers = np.column_stack((cx, cy))
    
    # Phase 3: Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    cr = np.maximum(cr, 0.0)

    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):

                if cr[i] + cr[j] > d - 1e-10:
                    shrink = (cr[i] + cr[j] - d) / 2.0 + 1e-10
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, cr, float(np.sum(cr))
