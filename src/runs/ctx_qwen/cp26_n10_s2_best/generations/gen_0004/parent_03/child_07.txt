import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    r_sum = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    return c

def compute_init_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    
    return np.clip(r * 0.82, 0.001, 0.5)

def generate_hex_configs():
    """Generate hexagonal lattice configurations with various parameters."""
    configs = []
    spacings = np.linspace(0.14, 0.18, 5)
    for s in spacings:
        for rot in np.linspace(-0.35, 0.35, 7):
            for dx_shift in np.linspace(-0.06, 0.06, 4):
                for dy_shift in np.linspace(-0.06, 0.06, 4):
                    pts = []
                    y = s + dy_shift
                    row = 0
                    while len(pts) < N + 6:
                        x_start = s + dx_shift + (row % 2) * s
                        x = x_start
                        while x <= 1.0 - s and len(pts) < N + 6:
                            pts.append([x, y])
                            x += 2.0 * s
                        y += s * np.sqrt(3.0)
                        row += 1
                    pts = np.array(pts[:N])
                    if rot != 0.0:
                        c_val, s_val = np.cos(rot), np.sin(rot)
                        pts = (pts - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5
                    pts = np.clip(pts, 0.02, 0.98)
                    configs.append(pts)
    return configs

def generate_force_configs():
    """Generate configurations using repulsive force relaxation."""
    configs = []
    for seed in range(25):
        np.random.seed(seed + 500)
        pts = np.random.uniform(0.12, 0.88, (N, 2))
        for _ in range(500):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(pts[i, 0] - pts[j, 0], pts[i, 1] - pts[j, 1])
                    if d < 0.22 and d > 1e-4:
                        f = 0.06 / d**2
                        diff = pts[i] - pts[j]
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.04
            pts = np.clip(pts, 0.04, 0.96)
        configs.append(pts)
    return configs

def run_packing():
    """Optimizes packing of 26 circles in a unit square to maximize sum of radii."""
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    # Combine diverse initial configurations
    inits = generate_hex_configs() + generate_force_configs()
    
    # Phase 1: Multi-start exploration
    for centers in inits:
        r_init = compute_init_radii(centers)
        v0 = np.concatenate([centers[:, 0], centers[:, 1], r_init])
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            s = -res.fun
            if s > best_sum and np.min(constraints(res.x)) >= -1e-6:
                best_sum = s
                best_v = res.x.copy()
        except Exception:
            pass
            
    # Fallback initialization if optimization fails completely
    if best_v is None:
        centers_f = inits[0]
        r_f = compute_init_radii(centers_f)
        best_v = np.concatenate([centers_f[:, 0], centers_f[:, 1], r_f])
        best_sum = -np.sum(r_f)

    curr_v = best_v.copy()
    for step in range(50):

        pert = curr_v.copy()

        noise = 0.006 * (1.0 - step / 50.0)
        pert[:2*N] += np.random.uniform(-noise, noise, 2*N)
        pert[:2*N] = np.clip(pert[:2*N], 0.01, 0.99)
        
        # Shrink radii progressively to allow centers to rearrange
        shrink_factor = 0.88 - step * 0.003
        pert[2*N:] *= max(0.65, shrink_factor)
        pert[2*N:] = compute_init_radii(pert[:2*N].reshape(N, 2)) * 0.92
        
        try:
            res = minimize(objective, pert, method='SLSQP', bounds=bounds, constraints=cons,

                best_sum = s
                best_v = res.x.copy()
                curr_v = best_v.copy()
        except Exception:
            pass
            
    # Extract optimal configuration
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()

    # 1. Enforce boundary constraints strictly
    for i in range(N):
        cr[i] = min(cr[i], cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
        
    # 2. Enforce non-overlap constraints iteratively with safety margin
    for _ in range(25):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                if cr[i] + cr[j] > d - 1e-9:
                    shrink = (cr[i] + cr[j] - d) / 2.0 + 1e-9
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    # Ensure non-negativity
    cr = np.maximum(cr, 0.0)
    centers = np.column_stack((cx, cy))
    
    return centers, cr, float(np.sum(cr))
