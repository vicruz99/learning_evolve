import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def get_pair_indices(n):
    """Precompute indices for all unique circle pairs (i < j)."""
    i_idx, j_idx = np.triu_indices(n, k=1)
    return i_idx, j_idx

def objective(v, n):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def constraints(v, n, pair_i, pair_j):
    """Compute inequality constraints: boundaries and non-overlap."""
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    
    num_pair = len(pair_i)
    cons = np.empty(4*n + num_pair)
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons[:n] = c[:, 0] - r
    cons[n:2*n] = 1.0 - c[:, 0] - r
    cons[2*n:3*n] = c[:, 1] - r
    cons[3*n:4*n] = 1.0 - c[:, 1] - r
    
    # Pairwise non-overlap constraints: dist(i,j) >= r_i + r_j
    ci = c[pair_i]
    cj = c[pair_j]
    ri = r[pair_i]
    rj = r[pair_j]
    
    dx = ci[:, 0] - cj[:, 0]
    dy = ci[:, 1] - cj[:, 1]
    dists = np.sqrt(dx*dx + dy*dy)
    
    cons[4*n:] = dists - ri - rj
    return cons

def hex_init(n, seed):
    """Generate a perturbed hexagonal lattice initial configuration."""
    np.random.seed(seed)
    pts = []
    r = 0.085
    y = r
    row = 0
    while len(pts) < n:
        x = r if row % 2 == 0 else 2*r
        while x <= 1 - r:
            pts.append([x, y])
            x += 2*r
        y += np.sqrt(3)*r
        row += 1
    pts = pts[:n]
    # Add controlled jitter to break symmetry and avoid degeneracy
    pts = np.array(pts) + np.random.uniform(-0.015, 0.015, (n, 2))
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = N_CIRCLES
    pair_i, pair_j = get_pair_indices(n)
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    best_v = None
    best_sum = -1.0
    
    # 1. Multiple hexagonal starts
    for seed in range(10):
        pts = hex_init(n, seed)
        r0 = np.full(n, 0.07)
        v0 = np.concatenate([pts.flatten(), r0])
        
        res = minimize(objective, v0, args=(n,), method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constraints, 'args': (n, pair_i, pair_j)},
                       options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                       
        if -res.fun > best_sum:
            c_opt = res.x[:2*n].reshape(n, 2)
            r_opt = res.x[2*n:]
            # Basic feasibility check before accepting
            if np.all(r_opt >= 0) and np.all(c_opt >= 0) and np.all(c_opt <= 1):
                best_sum = -res.fun
                best_v = res.x.copy()
                
    # 2. Iterative perturbation to escape local minima
    if best_v is not None:
        for _ in range(40):
            v_curr = best_v.copy()
            idx = np.random.randint(n)
            v_curr[2*idx] += np.random.uniform(-0.02, 0.02)
            v_curr[2*idx+1] += np.random.uniform(-0.02, 0.02)
            v_curr[:2*n] = np.clip(v_curr[:2*n], 0.0, 1.0)
            
            res = minimize(objective, v_curr, args=(n,), method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraints, 'args': (n, pair_i, pair_j)},
                           options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_v = res.x.copy()
                
    centers = best_v[:2*n].reshape(n, 2)
    radii = best_v[2*n:]
    
    # 3. Post-processing for strict validity
    # Enforce boundary constraints strictly
    for i in range(n):
        radii[i] = min(radii[i], centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        
    # Iteratively fix any remaining overlaps
    for _ in range(5):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-7:
                    shrink = (radii[i] + radii[j] - dist) / 2.0 + 1e-7
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    
    return centers, radii, float(np.sum(radii))
