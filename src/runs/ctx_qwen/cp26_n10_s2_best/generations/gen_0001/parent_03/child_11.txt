import numpy as np
from scipy.optimize import minimize

# Precompute pair indices for 26 circles to speed up constraint evaluation
PAIR_I = []
PAIR_J = []
for i in range(26):
    for j in range(i + 1, 26):
        PAIR_I.append(i)
        PAIR_J.append(j)
PAIR_I = np.array(PAIR_I)
PAIR_J = np.array(PAIR_J)

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[52:])

def constraint_fun(v):
    """Compute all inequality constraints: boundaries and non-overlap."""
    centers = v[:52].reshape(26, 2)
    radii = v[52:]
    
    cons = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons.append(centers[:, 0] - radii)
    cons.append(1.0 - centers[:, 0] - radii)
    cons.append(centers[:, 1] - radii)
    cons.append(1.0 - centers[:, 1] - radii)
    
    # Pairwise non-overlap constraints: dist >= r_i + r_j
    c_i = centers[PAIR_I]
    c_j = centers[PAIR_J]
    r_i = radii[PAIR_I]
    r_j = radii[PAIR_J]
    
    dx = c_i[:, 0] - c_j[:, 0]
    dy = c_i[:, 1] - c_j[:, 1]
    dists = np.sqrt(dx*dx + dy*dy)
    cons.append(dists - r_i - r_j)
    
    return np.concatenate(cons)

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).

    n = 26
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons_dict = {'type': 'ineq', 'fun': constraint_fun}
    
    best_sum = -1.0
    best_x = None
    
    init_configs = []
    
    # Strategy 1: Hexagonal lattice with perturbations (dense packing prior)
    np.random.seed(42)
    for trial in range(20):
        r0 = 0.03
        c = []
        y = r0 + 0.02
        row = 0
        while len(c) < n:
            x_start = r0 + 0.02 if row % 2 == 0 else 2*r0 + 0.02
            x = x_start
            while x < 1.0 - r0 - 0.02 and len(c) < n:
                c.append([x, y])
                x += 2*r0
            y += r0 * np.sqrt(3)
            row += 1
        c = np.array(c[:n])
        c += np.random.uniform(-0.025, 0.025, size=c.shape)
        c = np.clip(c, 0.05, 0.95)
        init_configs.append(np.concatenate([c.flatten(), np.full(n, r0)]))
        
    # Strategy 2: Pure random starts to capture non-lattice optima
    for trial in range(15):
        c = np.random.uniform(0.1, 0.9, size=(n, 2))
        init_configs.append(np.concatenate([c.flatten(), np.full(n, 0.02)]))

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,

            if np.isfinite(res.fun) and -res.fun > best_sum:
                best_sum = -res.fun
                best_x = res.x.copy()
        except Exception:
            pass
            
    if best_x is None:
        # Fallback to trivial valid packing
        centers = np.random.uniform(0.2, 0.8, size=(n, 2))
        radii = np.full(n, 0.05)
        return centers, radii, n * 0.05
        
    centers = best_x[:2*n].reshape(n, 2)
    radii = best_x[2*n:]

    # Enforce boundary constraints strictly
    for i in range(n):
        radii[i] = min(radii[i], centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1])

    for _ in range(20):

                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
