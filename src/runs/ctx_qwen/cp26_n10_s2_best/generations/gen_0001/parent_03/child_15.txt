def get_pair_indices(n):
    """Precompute indices for all unique circle pairs."""
    i_idx, j_idx = [], []
    for i in range(n):
        for j in range(i + 1, n):
            i_idx.append(i)
            j_idx.append(j)
    return np.array(i_idx), np.array(j_idx)

def objective_func(vars_vec, n):

def constraint_func(vars_vec, n, pair_i, pair_j):

    cons = np.concatenate([
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
