import numpy as np
from scipy.optimize import minimize

def compute_objective(vars_vec, n):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(vars_vec[2*n:])

def compute_constraints(vars_vec, n, pair_i, pair_j):
    """Compute all inequality constraints: boundaries and non-overlap."""
    centers = vars_vec[:2*n].reshape(n, 2)
    radii = vars_vec[2*n:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons = np.concatenate([
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    c_i = centers[pair_i]
    c_j = centers[pair_j]
    r_i = radii[pair_i]
    r_j = radii[pair_j]
    
    dist_sq = np.sum((c_i - c_j)**2, axis=1)
    r_sum = r_i + r_j
    cons = np.concatenate([cons, dist_sq - r_sum**2])
    
    return cons
