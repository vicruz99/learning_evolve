import numpy as np
from scipy.optimize import minimize

def get_pair_indices(n):
    """Precompute indices for all unique circle pairs."""
    i_idx = []
    j_idx = []
    for i in range(n):
        for j in range(i + 1, n):
            i_idx.append(i)
            j_idx.append(j)
    return np.array(i_idx), np.array(j_idx)

def objective(v, n):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def constraints(v, n, pi, pj):
    """Compute inequality constraints: boundaries and non-overlap."""
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    c_i = centers[pi]
    c_j = centers[pj]
    r_i = radii[pi]
    r_j = radii[pj]
    
    dist_sq = np.sum((c_i - c_j)**2, axis=1)
    r_sum = r_i + r_j
    
    c = np.concatenate([c, dist_sq - r_sum**2])
    return c

def run_packing():

    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).

    n = 26
    pi, pj = get_pair_indices(n)
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    best_sum = 0.0
    best_v = None

    # Place each circle at the location maximizing min distance to boundaries and existing circles
    np.random.seed(42)
    centers_g = np.zeros((n, 2))
    radii_g = np.zeros(n)
    
    for i in range(n):
        samples = np.random.random((20000, 2))
        best_d = 0.0
        best_pt = None
        
        # Vectorized distance evaluation for speed
        d_bounds = np.min(np.stack([samples, 1.0 - samples], axis=1), axis=1)
        
        if i > 0:
            # dist to existing centers
            diff = centers_g[:i] - samples[:, np.newaxis, :]
            dist_to_cen = np.sqrt(np.sum(diff**2, axis=2))
            min_dist_to_cen = np.min(dist_to_cen, axis=1)
            d_all = np.minimum(d_bounds, min_dist_to_cen)
        else:
            d_all = d_bounds
            
        idx = np.argmax(d_all)
        best_d = d_all[idx]
        best_pt = samples[idx]
        
        centers_g[i] = best_pt
        radii_g[i] = best_d
        
    # Shrink slightly to guarantee strict initial feasibility for optimizer
    v0 = np.concatenate([centers_g.flatten(), radii_g * 0.94])
    
    # 2. Optimization Loop with Adaptive Perturbations
    num_trials = 25
    for trial in range(num_trials):
        np.random.seed(trial * 13 + 7)
        
        if trial == 0:
            curr_v = v0.copy()
        else:
            # Perturb best known configuration to escape local minima
            curr_v = best_v.copy()
            centers_curr = curr_v[:2*n].reshape(n, 2)
            radii_curr = curr_v[2*n:]
            
            # Adaptive jitter scale: starts wide, narrows down
            scale = 0.04 * (1.0 - trial / (num_trials + 5.0))
            jitter = np.random.uniform(-scale, scale, size=centers_curr.shape)
            centers_curr = np.clip(centers_curr + jitter, 0.01, 0.99)
            
            # Random shrinkage to ensure constraints are satisfied after jitter
            radii_curr *= (0.90 + 0.05 * np.random.rand(n))
            
            curr_v = np.concatenate([centers_curr.flatten(), radii_curr])
            
        try:

            curr_sum = -res.fun
            if curr_sum > best_sum:

    final_centers = best_v[:2*n].reshape(n, 2)
    final_radii = best_v[2*n:]

    for i in range(n):
        final_radii[i] = min(final_radii[i], 
                             final_centers[i,0], 1.0 - final_centers[i,0],

        for j in range(i + 1, n):

                shrink = (s_r - d) / 2.0 + 1e-7
                final_radii[i] = max(0.0, final_radii[i] - shrink)
                final_radii[j] = max(0.0, final_radii[j] - shrink)
                
    return final_centers, final_radii, float(np.sum(final_radii))
