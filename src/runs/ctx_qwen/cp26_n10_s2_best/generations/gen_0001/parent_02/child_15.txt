from scipy.optimize import minimize

def objective(v, n):

    return -np.sum(v[2*n:])

def constraints(v, n):
    """Compute all inequality constraints: boundaries and non-overlap."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    
    cons = []
    # Boundary constraints
    cons.append(cx - r)
    cons.append(1 - cx - r)
    cons.append(cy - r)
    cons.append(1 - cy - r)
    
    # Pairwise non-overlap constraints (vectorized)
    c1 = cx[:, np.newaxis] - cx[np.newaxis, :]
    c2 = cy[:, np.newaxis] - cy[np.newaxis, :]

    r_sum = r[:, np.newaxis] + r[np.newaxis, :]

    cons.append((dists - r_sum)[mask])
    
    return np.concatenate(cons)
