from scipy.optimize import minimize

def compute_objective(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def compute_constraints(v, n, pair_i, pair_j):
    """Inequality constraints: boundaries and non-overlap (squared)."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    
    cons = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons.append(cx - r)
    cons.append(1.0 - cx - r)
    cons.append(cy - r)
    cons.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[pair_i] - cx[pair_j]
    dy = cy[pair_i] - cy[pair_j]
    d2 = dx*dx + dy*dy
    rs = r[pair_i] + r[pair_j]
    cons.append(d2 - rs*rs)
    
    return np.concatenate(cons)

def create_initial_state(n, seed):

    np.random.seed(seed)
    centers = np.zeros((n, 2))

    row = 0
    idx = 0

            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * base_r

        row += 1

    radii = np.zeros(n)
