import numpy as np
from scipy.optimize import minimize

def obj(v, n, i_idx, j_idx):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def constraints_fun(v, n, i_idx, j_idx):
    """
    Vectorized inequality constraints.
    Returns array where each element must be >= 0.
    Structure: [x-r, 1-x-r, y-r, 1-y-r, dist-(r_i+r_j)]
    """
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    
    num_pair = len(i_idx)
    num_cons = 4*n + num_pair
    c = np.empty(num_cons)
    idx = 0
    
    # Boundary constraints
    c[idx:idx+n] = cx - r; idx += n
    c[idx:idx+n] = 1.0 - cx - r; idx += n
    c[idx:idx+n] = cy - r; idx += n
    c[idx:idx+n] = 1.0 - cy - r; idx += n
    
    # Pairwise non-overlap constraints
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dist = np.sqrt(dx**2 + dy**2)
    c[idx:] = dist - (r[i_idx] + r[j_idx])
    
    return c

def generate_initial_config(seed, n):
    """Generates diverse initial configurations to help escape local minima."""
    np.random.seed(seed)
    mode = seed % 3
    
    if mode == 0:  # Hexagonal lattice
        pts = []
        r_est = 0.09 + (seed // 3) * 0.003
        y = r_est
        row = 0
        while len(pts) < n + 5:
            x_start = r_est + (row % 2) * r_est
            x = x_start
            while x <= 1.0 - r_est:
                pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
        pts = np.array(pts[:n])
        
    elif mode == 1:  # Staggered grid
        pts = []
        x = 0.08
        while len(pts) < n + 5:
            y = 0.08
            while y <= 0.92 and len(pts) < n + 5:
                pts.append([x, y])
                y += 0.16
            x += 0.16
        pts = np.array(pts[:n])
        
    else:  # Random uniform with jitter
        pts = np.random.uniform(0.1, 0.9, size=(n + 5, 2))
        pts = pts[:n]
        
    # Add small random perturbation and clip
    pts += np.random.uniform(-0.004, 0.004, pts.shape)
    pts = np.clip(pts, 0.03, 0.97)
    
    # Start with small valid radii
    r_init = np.full(n, 0.02)
    return np.concatenate([pts[:, 0], pts[:, 1], r_init])

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    i_idx, j_idx = np.triu_indices(n, k=1)
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    best_val = -1e9
    best_v = None
    
    # Multiple restarts to explore configuration space
    for s in range(40):
        v0 = generate_initial_config(s, n)
        try:
            res = minimize(
                obj, 
                v0, 
                method='SLSQP', 
                bounds=bounds, 
                args=(n, i_idx, j_idx),
                constraints={'type': 'ineq', 'fun': constraints_fun, 'args': (n, i_idx, j_idx)},
                options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False}
            )
            
            current_sum = -res.fun
            if current_sum > best_val:
                # Verify feasibility with tolerance
                c_vals = constraints_fun(res.x, n, i_idx, j_idx)
                if np.all(c_vals >= -1e-6):
                    best_val = current_sum
                    best_v = res.x.copy()
        except Exception:
            continue
            
    if best_v is not None:
        cx = best_v[:n]
        cy = best_v[n:2*n]
        r = best_v[2*n:]
        
        # Strict post-processing to guarantee validity per validator rules
        # 1. Enforce boundary constraints strictly
        for i in range(n):
            margin = min(cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
            if r[i] > margin:
                r[i] = margin
                
        # 2. Enforce non-overlap strictly with safety margin
        for _ in range(5):  # Iterate to propagate constraints
            changed = False
            for i in range(n):
                for j in range(i + 1, n):
                    dx = cx[i] - cx[j]
                    dy = cy[i] - cy[j]
                    d = np.hypot(dx, dy)
                    sum_r = r[i] + r[j]
                    if sum_r > d - 1e-9:
                        shrink = (sum_r - d + 1e-9) / 2.0 + 1e-7
                        r[i] = max(0.0, r[i] - shrink)
                        r[j] = max(0.0, r[j] - shrink)
                        changed = True
            if not changed:
                break
                
        centers = np.column_stack((cx, cy))
        return centers, r, float(np.sum(r))
    else:
        # Fallback to a guaranteed valid configuration (5x6 grid subset)
        cx, cy, r = np.zeros(n), np.zeros(n), np.full(n, 0.08)
        count = 0
        for i in range(6):
            for j in range(5):
                if count < n:
                    cx[count] = 0.1 + i * 0.16
                    cy[count] = 0.1 + j * 0.18
                    count += 1
        return np.column_stack((cx, cy)), r, float(np.sum(r))
