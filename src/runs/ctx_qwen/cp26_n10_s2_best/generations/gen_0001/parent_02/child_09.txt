import numpy as np
from scipy.optimize import minimize

def objective(v, n):
    """Objective function to minimize: negative sum of radii."""
    return -np.sum(v[2*n:])

def constraint_func(v, n):
    """
    Computes all inequality constraints that must be >= 0.
    v layout: [x1..xn, y1..yn, r1..rn]
    """
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    
    cons = [
        cx - r,                # x - r >= 0
        1.0 - cx - r,          # 1 - x - r >= 0
        cy - r,                # y - r >= 0
        1.0 - cy - r           # 1 - y - r >= 0
    ]
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dist = np.sqrt(dx**2 + dy**2 + 1e-16)
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pair_dist = dist[mask]
    pair_r_sum = (r[:, np.newaxis] + r[np.newaxis, :])[mask]
    
    cons.append(pair_dist - pair_r_sum)
    return np.concatenate(cons)

def force_relax_init(n, steps=400, seed=0):
    """
    Initializes centers using a repulsive force simulation to spread points evenly.
    This avoids clustered starts that trap gradient-based optimizers.
    """
    np.random.seed(seed)
    x = np.random.uniform(0.15, 0.85, n)
    y = np.random.uniform(0.15, 0.85, n)
    
    for _ in range(steps):
        dx = x[:, np.newaxis] - x[np.newaxis, :]
        dy = y[:, np.newaxis] - y[np.newaxis, :]
        d2 = dx**2 + dy**2
        np.fill_diagonal(d2, 1.0)
        d = np.sqrt(d2)
        f = 1.0 / (d**3 + 1e-6)
        
        fx = np.sum(dx * f, axis=1)
        fy = np.sum(dy * f, axis=1)
        
        # Wall repulsion
        fx += np.where(x < 0.2, -2.0, np.where(x > 0.8, 2.0, 0.0))
        fy += np.where(y < 0.2, -2.0, np.where(y > 0.8, 2.0, 0.0))
        
        x += 0.002 * fx
        y += 0.002 * fy
        x = np.clip(x, 0.02, 0.98)
        y = np.clip(y, 0.02, 0.98)
        
    return np.column_stack((x, y))

def run_packing():
    n = 26

    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}

    best_x = None

    while len(pts) < n + 10:

            pts.append([x, yh]); x += 2 * rh
        yh += np.sqrt(3) * rh; row += 1
    starts.append(np.array(pts[:n]))
    
    # 2. Grid initialization
    pts_g = []
    for i in range(6):
        for j in range(6):
            pts_g.append([0.1 + i*0.15, 0.1 + j*0.15])
    starts.append(np.array(pts_g[:n]))

    for s in range(8):
        starts.append(force_relax_init(n, steps=300, seed=s*10+1))
        
    # Run optimization from each start
    for c_init in starts:
        x0 = np.concatenate([c_init.flatten(), np.full(n, 0.02)])
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, args=(n,),
