from scipy.optimize import minimize

N_CIRCLES = 26

def objective(v):

def constraint_fun(v):

    2. Pairwise: dist^2 >= (r_i + r_j)^2

    ys = v[N_CIRCLES:2*N_CIRCLES]
    rs = v[2*N_CIRCLES:]

    cons.append(xs - rs)
    cons.append(1 - xs - rs)
    cons.append(ys - rs)
    cons.append(1 - ys - rs)
    
    # Pairwise non-overlap constraints (squared distance formulation for smoothness)
    diff_x = xs[:, np.newaxis] - xs[np.newaxis, :]
    diff_y = ys[:, np.newaxis] - ys[np.newaxis, :]
    dist_sq = diff_x**2 + diff_y**2
    
    sum_r = rs[:, np.newaxis] + rs[np.newaxis, :]
    sum_r_sq = sum_r**2

    cons.append((dist_sq - sum_r_sq)[mask])
    
    return np.concatenate(cons)
