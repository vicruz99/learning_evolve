from scipy.optimize import minimize

def objective_func(v, n):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def constraint_func(v, n):
    """
    Returns inequality constraints that must be >= 0.
    1. Boundary: 0 <= center - r,  center + r <= 1
    2. Non-overlap: dist^2 >= (r_i + r_j)^2
    """
    xs = v[:n]
    ys = v[n:2*n]
    rs = v[2*n:]
    
    cons = []
    # Boundary constraints
    cons.append(xs - rs)
    cons.append(1.0 - xs - rs)
    cons.append(ys - rs)
    cons.append(1.0 - ys - rs)
    
    # Pairwise non-overlap constraints (vectorized)
    dx = xs[:, np.newaxis] - xs[np.newaxis, :]
    dy = ys[:, np.newaxis] - ys[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    sum_r = rs[:, np.newaxis] + rs[np.newaxis, :]
    
    # Only upper triangle needed
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cons.append(dist_sq[mask] - sum_r[mask]**2)
    
    return np.concatenate(cons)
