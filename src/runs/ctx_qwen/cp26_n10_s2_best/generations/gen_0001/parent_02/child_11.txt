

Looking at this problem, I need to maximize the sum of radii for 26 circles in a unit square. The key insights from analyzing past solutions:

1. **Hexagonal lattice initialization** tends to work better than random
2. **Distance squared constraints** are numerically more stable than sqrt-based ones
3. **Multiple restarts** with diverse initializations help escape local minima
4. **Post-processing** is essential to guarantee strict feasibility
5. **Higher iteration counts** and tighter tolerances help converge better

<strategy>
My strategy:
1. Use hexagonal lattice initialization with multiple row configurations to get diverse starting points
2. Use squared distance constraints to avoid sqrt singularities at optimization
3. Run many restarts (20+) with different seeds and initialization patterns
4. Post-process to guarantee strict feasibility with safety margins
5. Use a two-stage approach: grow radii from small feasible start
</strategy>

```python
import numpy as np
from scipy.optimize import minimize


def objective(v, n):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*n:])


def constraint_function(v, n, pair_i, pair_j):
    """Compute all inequality constraints. All must be >= 0."""
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    
    cons = []
    
    # Boundary constraints: center +/- radius within [0, 1]
    cons.append(centers[:, 0] - radii)           # x - r >= 0
    cons.append(1 - centers[:, 0] - radii)      # x + r <= 1
    cons.append(centers[:, 1] - radii)           # y - r >= 0
    cons.append(1 - centers[:, 1] - radii)      # y + r <= 1
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    c_i = centers[pair_i]
    c_j = centers[pair_j]
    r_i = radii[pair_i]
    r_j = radii[pair_j]
    
    diff = c_i - c_j
    dist_sq = np.sum(diff**2, axis=1)
    sum_r = r_i + r_j
    sum_r_sq = sum_r**2
    
    cons.append(dist_sq - sum_r_sq)
    
    return np.concatenate(cons)


def get_pair_indices(n):
    """Precompute indices for all unique circle pairs."""
    i_idx = []
    j_idx = []
    for i in range(n):
        for j in range(i + 1, n):
            i_idx.append(i)
            j_idx.append(j)
    return np.array(i_idx), np.array(j_idx)


def hex_initialization(n, seed, variant=0):
    """Generate hexagonal lattice initialization with multiple variants."""
    np.random.seed(seed)
    
    centers = []
    
    if variant == 0:
        # Standard hexagonal packing
        r_init = 0.095
        y = r_init
        row = 0
        while len(centers) < n + 5:
            x_start = r_init if row % 2 == 0 else 2 * r_init
            x = x_start
            while x <= 1 - r_init and len(centers) < n + 5:
                centers.append([x, y])
                x += 2 * r_init
            y += np.sqrt(3) * r_init
            row += 1
    
    elif variant == 1:
        # Offset hexagonal (shifted)
        r_init = 0.09
        y = r_init + 0.03
        row = 0
        while len(centers) < n + 5:
            x_start = r_init + 0.02 if row % 2 == 0 else 2 * r_init + 0.02
            x = x_start
            while x <= 1 - r_init and len(centers) < n + 5:
                centers.append([x, y])
                x += 2 * r_init
            y += np.sqrt(3) * r_init
            row += 1
    
    elif variant == 2:
        # Grid-based
        cols = 6
        rows = 5
        x_vals = np.linspace(0.1, 0.9, cols)
        y_vals = np.linspace(0.1, 0.9, rows)
        for y in y_vals:
            for x in x_vals:
                if len(centers) < n + 5:
                    centers.append([x, y])
    
    elif variant == 3:
        # Random perturbed hexagonal
        r_init = 0.085
        y = r_init
        row = 0
        while len(centers) < n + 5:
            x_start = r_init if row % 2 == 0 else 2 * r_init
            x = x_start
            while x <= 1 - r_init and len(centers) < n + 5:
                centers.append([x, y])
                x += 2 * r_init
            y += np.sqrt(3) * r_init * (0.95 + 0.1 * np.random.rand())
            row += 1
    
    centers = np.array(centers[:n])
    
    # Add small perturbation
    perturbation = np.random.uniform(-0.005, 0.005, size=centers.shape)
    centers = centers + perturbation
    centers = np.clip(centers, 0.02, 0.98)
    
    return centers


def run_packing():
    n = 26
    pair_i, pair_j = get_pair_indices(n)
    
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    cons_dict = {'type': 'ineq', 'fun': constraint_function, 'args': (n, pair_i, pair_j)}
    
    # Multiple restarts with diverse initializations
    num_restarts = 20
    for restart in range(num_restarts):
        variant = restart % 4
        seed = restart * 37 + 7
        
        centers_init = hex_initialization(n, seed, variant)
        x_init = centers_init[:, 0]
        y_init = centers_init[:, 1]
        
        # Small initial radius for feasibility
        r_init = np.full(n, 0.02)
        
        x0 = np.concatenate([x_init, y_init, r_init])
        
        # Check initial feasibility
        init_cons = constraint_function(x0, n, pair_i, pair_j)
        if np.any(init_cons < -0.001):
            # Reduce radius if not feasible
            r_init = np.full(n, 0.01)
            x0 = np.concatenate([x_init, y_init, r_init])
        
        try:
            res = minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                args=(n,),
                constraints=cons_dict,
                options={
                    'maxiter': 3000,
                    'ftol': 1e-12,
                    'disp': False
                }
            )
            
            if res.success or -res.fun > best_sum + 0.001:
                # Check constraint satisfaction
                final_cons = constraint_function(res.x, n, pair_i, pair_j)
                min_constraint = np.min(final_cons)
                
                current_sum = -res.fun
                
                # Only accept if reasonably feasible
                if min_constraint > -1e-8:
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_centers = res.x[:2*n].reshape(n, 2).copy()
                        best_radii = res.x[2*n:].copy()
        except Exception:
            pass
    
    if best_centers is None:
        # Fallback: simple grid
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.08)
        idx = 0
        for i in range(6):
            for j in range(5):
                if idx < n:
                    centers[idx] = [0.1 + i * 0.16, 0.1 + j * 0.2]
                    idx += 1
        return centers, radii, float(np.sum(radii))
    
    # Post-processing to guarantee strict validity
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    # Ensure radii are non-negative
    radii = np.maximum(radii, 0.0)
    
    # Enforce boundary constraints strictly
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        if radii[i] > max_r:
            radii[i] = max_r
    
    # Enforce non-overlap strictly with safety margin
    max_iterations = 50
    for _ in range(max_iterations):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                sum_r = radii[i] + radii[j]
                
                if dist < sum_r - 1e-10:
                    # Need to shrink
                    excess = (sum_r - dist) / 2.0 + 1e-8
                    # Shrink proportionally
                    if radii[i] > radii[j]:
                        shrink_i = excess
                        shrink_j = 0
                    elif radii[j] > radii[i]:
                        shrink_i = 0
                        shrink_j = excess
                    else:
                        shrink_i = excess / 2.0
                        shrink_j = excess / 2.0
                    
                    radii[i] = max(0.0, radii[i] - shrink_i)
                    radii[j] = max(0.0, radii[j] - shrink_j)
                    changed = True
        
        if not changed:
            break
    
    # Final boundary check
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        radii[i] = min(radii[i], max_r)
    
    # Tiny safety scale to ensure 1e-12 tolerance is met
    if np.sum(radii) > 0:
        radii *= 0.9999999
    
    return centers, radii, float(np.sum(radii))
```
