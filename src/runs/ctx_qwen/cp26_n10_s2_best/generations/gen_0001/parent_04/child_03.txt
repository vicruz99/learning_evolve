import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N_CIRCLES:])

def constraints(v):
    """Compute inequality constraints: boundaries and non-overlap."""
    n = N_CIRCLES
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    
    # Boundary constraints: center +/- radius within [0, 1]
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    
    # Pairwise non-overlap constraints (squared form to avoid sqrt singularity)
    # dist^2 >= (r_i + r_j)^2  <=>  dist^2 - (r_i + r_j)^2 >= 0
    X = x[:, np.newaxis] - x[np.newaxis, :]
    Y = y[:, np.newaxis] - y[np.newaxis, :]
    R = r[:, np.newaxis] + r[np.newaxis, :]
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    c5 = X[i_idx, j_idx]**2 + Y[i_idx, j_idx]**2 - R[i_idx, j_idx]**2
    
    return np.concatenate([c1, c2, c3, c4, c5])

def run_packing():
    n = N_CIRCLES
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_val = -np.inf
    
    # Generate diverse initial guesses to escape poor local minima
    guesses = []
    
    # 1. Hexagonal lattice patterns with varying densities and seeds
    for seed in range(10):
        np.random.seed(seed)
        pts = []
        rows = np.random.randint(5, 8)
        r_est = 0.08
        y = r_est
        row = 0
        while len(pts) < n and row < rows:
            x = r_est if row % 2 == 0 else 2 * r_est
            while x <= 1.0 - r_est and len(pts) < n:
                pts.append([x, y])
                x += 2 * r_est
            y += np.sqrt(3) * r_est
            row += 1
            
        if len(pts) < n:
            while len(pts) < n:
                pts.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
                
        centers = np.array(pts[:n]) + np.random.uniform(-0.01, 0.01, size=(n, 2))
        centers = np.clip(centers, 0.05, 0.95)
        guesses.append(np.concatenate([centers.ravel(), np.full(n, 0.05)]))
        
    # 2. Random uniform distributions
    for seed in range(10):
        np.random.seed(100 + seed)
        centers = np.random.uniform(0.1, 0.9, (n, 2))
        guesses.append(np.concatenate([centers.ravel(), np.full(n, 0.05)]))
        
    # 3. Structured grid patterns
    for seed in range(5):
        np.random.seed(200 + seed)
        pts = []
        for i in range(6):
            for j in range(5):
                if len(pts) < n:
                    pts.append([0.1 + i * 0.15, 0.1 + j * 0.18])
        if len(pts) < n:
            while len(pts) < n:
                pts.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
        centers = np.array(pts[:n]) + np.random.uniform(-0.02, 0.02, (n, 2))
        centers = np.clip(centers, 0.05, 0.95)
        guesses.append(np.concatenate([centers.ravel(), np.full(n, 0.05)]))
        
    # Main optimization loop
    for v0 in guesses:
        res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-12})
        # Accept if successful or significantly better, and strictly feasible
        if res.success or -res.fun > best_val - 0.005:
            c = constraints(res.x)
            if np.all(c >= -1e-7):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
                    
    # Local refinement loop: perturb and re-optimize from the best found configuration
    if best_v is not None:
        for _ in range(15):
            vp = best_v.copy()
            # Perturb centers to escape symmetry locks
            vp[:2*n] += np.random.uniform(-0.005, 0.005, 2*n)
            vp[:2*n] = np.clip(vp[:2*n], 0.05, 0.95)
            # Slightly shrink radii to give optimizer room to reposition and regrow
            vp[2*n:] *= 0.98 
            
            res = minimize(objective, vp, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 1000, 'ftol': 1e-12})
            
            if res.success or -res.fun > best_val - 0.001:
                if np.all(constraints(res.x) >= -1e-7) and -res.fun > best_val:
                    best_val = -res.fun
                    best_v = res.x.copy()
                    
    x = best_v[:n]
    y = best_v[n:2*n]
    r = best_v[2*n:]
    
    # Safety scale to strictly satisfy validator tolerance (handles 1e-12 float edge cases)
    r *= 0.9999999
    
    return np.column_stack((x, y)), r, float(np.sum(r))
