import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2*N_CIRCLES:])

def constraint_fun(vars):
    """Inequality constraints: boundaries and non-overlap (vectorized)."""
    n = N_CIRCLES
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap: (xi-xj)^2 + (yi-yj)^2 >= (ri+rj)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    dist_sq = dx**2 + dy**2
    r_sum_sq = dr**2
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c = np.concatenate([c, (dist_sq - r_sum_sq)[mask]])
    return c

def generate_init(n, seed, strategy):

    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.04)  # Small radius ensures initial feasibility
    
    if strategy == 0:  # Hexagonal pattern
        y = 0.05
        row = 0
        idx = 0
        while idx < n:
            x = 0.05 + (row % 2) * 0.09

                centers[idx] = [x, y]
                idx += 1
                x += 0.18
            y += 0.15
            row += 1

        idx = 0
        for r in range(5):
            for c in range(5):

                    idx += 1

            idx += 1

    centers += np.random.uniform(-0.02, 0.02, (n, 2))

    return np.concatenate([centers.flatten(), radii])
