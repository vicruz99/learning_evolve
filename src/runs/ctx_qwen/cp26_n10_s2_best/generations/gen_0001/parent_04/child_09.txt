from scipy.optimize import minimize

def objective(v, n):

    return -np.sum(v[2*n:])

def constraints(v, n):
    """Compute all inequality constraints: boundaries and non-overlap."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    
    # Boundary constraints: x >= r, x + r <= 1, y >= r, y + r <= 1
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints: dist(i,j) >= r_i + r_j
    centers = np.column_stack((x, y))

    dists = np.sqrt(np.sum(diffs**2, axis=2))

    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c = np.concatenate([c, (dists - r_sums)[mask]])
    
    return c
