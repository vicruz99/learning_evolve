import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(v):
    """Objective function: minimize negative sum of radii."""
    n = N_CIRCLES
    return -np.sum(v[2*n:3*n])

def compute_constraints(v):
    """Vectorized inequality constraints: boundaries and non-overlap."""
    n = N_CIRCLES
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:3*n]
    
    # Boundary constraints: circles inside [0, 1]^2
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    
    # Pairwise non-overlap constraints: dist >= r_i + r_j
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Extract upper triangle (i < j) to avoid duplicates and self-constraints
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c_pair = (dist - r_sum)[mask]
    
    return np.concatenate([c1, c2, c3, c4, c_pair])

def run_packing():

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    for trial in range(15):

            x = x_start
            while x <= 1 - r_init and len(centers) < n + 5:
                centers.append([x, y])
                x += 2 * r_init
            y += r_init * np.sqrt(3)
            row += 1
            
        centers = np.array(centers[:n])

        centers += rng.uniform(-0.02, 0.02, size=centers.shape)

        v0 = np.concatenate([centers.flatten(), np.full(n, 0.025)])

                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 10000, 'ftol': 1e-12, 'disp': False})

    for i in range(n):
