import numpy as np
from scipy.optimize import differential_evolution, linprog, minimize

N_CIRCLES = 26
i_idx, j_idx = np.triu_indices(N_CIRCLES, k=1)

# Precompute LP constraint matrix structure (constant for fixed N)
A_ub = np.zeros((len(i_idx), N_CIRCLES))
for k in range(len(i_idx)):
    A_ub[k, i_idx[k]] = 1.0
    A_ub[k, j_idx[k]] = 1.0

def compute_lp_objective(centers_flat):
    """
    Given center coordinates, solve LP to find optimal radii maximizing sum.
    Returns negative sum (for minimization).
    """
    centers = centers_flat.reshape(N_CIRCLES, 2)
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    b_i = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                     np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
                     
    c_obj = -np.ones(N_CIRCLES)
    rad_bounds = [(0.0, b_i[i]) for i in range(N_CIRCLES)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=dists, bounds=rad_bounds, method='highs')
        if res.success:
            return -res.fun
    except Exception:
        pass
    return 100.0

def obj_refine(v):
    """Objective for SLSQP refinement: maximize sum of radii."""
    return -np.sum(v[2*N_CIRCLES:])
    
def cons_refine(v):
    """Inequality constraints for SLSQP: boundaries and non-overlap (squared)."""
    X = v[:N_CIRCLES]
    Y = v[N_CIRCLES:2*N_CIRCLES]
    R = v[2*N_CIRCLES:]
    
    c = [X - R, 1.0 - X - R, Y - R, 1.0 - Y - R]
    
    diffX = X[i_idx] - X[j_idx]
    diffY = Y[i_idx] - Y[j_idx]
    c.append((diffX**2 + diffY**2) - (R[i_idx] + R[j_idx])**2)
    
    return np.concatenate(c)

def run_packing():
    # 1. Global optimization of centers using Differential Evolution
    bounds_centers = [(0.0, 1.0)] * (2 * N_CIRCLES)
    
    de_res = differential_evolution(compute_lp_objective, bounds_centers, 
                                    maxiter=250, popsize=12, tol=1e-9, 
                                    seed=42, workers=1, disp=False)
    
    best_centers = de_res.x.reshape(N_CIRCLES, 2)
    
    # 2. Compute exact optimal radii for the DE-found centers via LP
    dx = best_centers[i_idx, 0] - best_centers[j_idx, 0]
    dy = best_centers[i_idx, 1] - best_centers[j_idx, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    b_i = np.minimum(np.minimum(best_centers[:, 0], 1.0 - best_centers[:, 0]),
                     np.minimum(best_centers[:, 1], 1.0 - best_centers[:, 1]))
                     
    c_obj = -np.ones(N_CIRCLES)
    rad_bounds = [(0.0, b_i[i]) for i in range(N_CIRCLES)]
    lp_res = linprog(c_obj, A_ub=A_ub, b_ub=dists, bounds=rad_bounds, method='highs')
    best_radii = lp_res.x if lp_res.success else np.full(N_CIRCLES, 0.04)
    
    # 3. Smooth local refinement of both centers and radii using SLSQP
    x0 = np.concatenate([best_centers.flatten(), best_radii])
    bounds_all = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    
    res = minimize(obj_refine, x0, method='SLSQP',
                   constraints={'type': 'ineq', 'fun': cons_refine},
                   bounds=bounds_all,
                   options={'maxiter': 3000, 'ftol': 1e-12})
                   
    final_centers = res.x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    final_radii = res.x[2*N_CIRCLES:]
    
    # 4. Strict validity enforcement per validator tolerances
    final_radii = np.maximum(final_radii, 0.0)
    final_centers = np.clip(final_centers, 1e-9, 1.0 - 1e-9)
    
    return final_centers, final_radii, float(np.sum(final_radii))
