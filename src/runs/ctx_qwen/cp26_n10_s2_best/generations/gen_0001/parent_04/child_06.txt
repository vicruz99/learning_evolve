import numpy as np
from scipy.optimize import minimize

def objective_func(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def constraint_func(v, n, triu_mask):
    """Inequality constraints: boundaries and non-overlap."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c_bound = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints: dist(i,j) >= r_i + r_j
    # Use broadcasting for vectorized distance computation
    X = x[:, np.newaxis]
    Y = y[:, np.newaxis]
    R = r[:, np.newaxis]
    
    dist_sq = (X - X.T)**2 + (Y - Y.T)**2
    # Clamp to 0 to avoid NaNs from floating point errors when points coincide
    dist_sq = np.maximum(dist_sq, 0.0)
    
    r_sum = R + R.T
    
    # Extract only upper triangle constraints (i < j)
    c_overlap = np.sqrt(dist_sq[triu_mask]) - r_sum[triu_mask]
    
    return np.concatenate([c_bound, c_overlap])

def run_packing():
    N = 26
    # Precompute upper triangle mask for constraints
    triu_mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Prepare diverse initial configurations
    starts = []
    
    # 1. Hexagonal lattice initialization
    r_hex = 0.095
    pts_hex = []
    y_c = r_hex
    for row in range(6):
        x_s = r_hex if row % 2 == 0 else 2 * r_hex
        x_c = x_s
        cnt = 6 if row % 2 == 0 else 5
        for _ in range(cnt):
            if len(pts_hex) < N:
                pts_hex.append([x_c, y_c])
                x_c += 2 * r_hex
        y_c += r_hex * np.sqrt(3)
        if len(pts_hex) >= N: break
    starts.append(np.array(pts_hex[:N]))
    
    # 2. Grid lattice initialization
    pts_grid = []
    for i in range(5):
        for j in range(6):
            if len(pts_grid) < N:
                pts_grid.append([0.05 + j * 0.18, 0.05 + i * 0.20])
    starts.append(np.array(pts_grid[:N]))
    
    # 3. Random initializations
    for seed in range(8):
        np.random.seed(seed + 100)
        starts.append(np.random.uniform(0.05, 0.95, size=(N, 2)))
        
    # Optimize from each start
    for idx, centers_init in enumerate(starts):
        np.random.seed(idx)
        jitter = np.random.uniform(-0.015, 0.015, size=centers_init.shape)
        c_init = np.clip(centers_init + jitter, 0.02, 0.98)
        v0 = np.concatenate([c_init.flatten(), np.full(N, 0.04)])
        
        res = minimize(
            objective_func, v0, args=(N, triu_mask),
            method='SLSQP', bounds=bounds,
            constraints={'type': 'ineq', 'fun': constraint_func, 'args': (N, triu_mask)},
            options={'maxiter': 10000, 'ftol': 1e-12, 'disp': False}
        )
        
        if not np.isnan(res.x).any():
            c_opt = res.x[:2*N].reshape(N, 2)
            r_opt = res.x[2*N:]
            current_sum = np.sum(r_opt)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
                
    centers = best_centers
    radii = best_radii
    
    # Strict post-processing to guarantee validity per validation rules
    # 1. Enforce boundary constraints
    for i in range(N):
        x, y = centers[i]
        radii[i] = min(radii[i], x, 1.0 - x, y, 1.0 - y)
        
    # 2. Iteratively resolve overlaps
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                needed = radii[i] + radii[j]
                if dist < needed - 1e-12:
                    shrink = (needed - dist) / 2.0 + 1e-8
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
