To improve the packing sum of radii beyond the current ~2.595, we need to:
1. **Use squared distance constraints**: `dist^2 >= (r_i + r_j)^2` avoids square roots in constraints, providing smoother gradients and better numerical stability for SLSQP.
2. **Vectorize constraint evaluation**: Precompute upper-triangle indices and use array operations to evaluate all 325 pairwise constraints at once, drastically reducing overhead per iteration.
3. **Diverse & Structured Initializations**: Run multiple restarts from varied configurations (hexagonal lattice, perturbed grid, corner/edge-focused layouts, and random). This helps escape shallow local minima where SLSQP often gets stuck.
4. **Strict Feasibility Enforcement**: After optimization, apply a deterministic shrinking routine to guarantee all boundary and non-overlap constraints satisfy the `1e-12` tolerance of the validator.
5. **Robust Optimization Loop**: Increase iterations, tighten tolerance, and filter results by actual constraint satisfaction rather than just optimizer success flags.

These changes collectively allow the optimizer to explore the feasible region more efficiently and converge to denser packings closer to the theoretical optimum.

import numpy as np
from scipy.optimize import minimize

N = 26
# Precompute indices for upper triangular pairs to speed up constraint evaluation
UTRIU_IDX = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute all inequality constraints: boundaries and non-overlap (squared)."""
    cx = v[:N]
    cy = v[N:2*N]
    cr = v[2*N:]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b_cons = np.concatenate([
        cx - cr,
        1.0 - cx - cr,
        cy - cr,
        1.0 - cy - cr
    ])
    
    # Pairwise non-overlap: (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    dx = cx[UTRIU_IDX[0]] - cx[UTRIU_IDX[1]]
    dy = cy[UTRIU_IDX[0]] - cy[UTRIU_IDX[1]]
    dist_sq = dx**2 + dy**2
    
    sr = cr[UTRIU_IDX[0]] + cr[UTRIU_IDX[1]]
    sr_sq = sr**2
    
    return np.concatenate([b_cons, dist_sq - sr_sq])

def generate_initial_configs(seed):
    """Generate diverse, feasible initial configurations."""
    np.random.seed(seed)
    configs = []
    base_r = 0.05
    
    # 1. Hexagonal lattice perturbation
    pts = []
    y = base_r
    row = 0
    while len(pts) < N + 10:
        x_start = base_r if row % 2 == 0 else 2 * base_r
        x = x_start
        while x <= 1 - base_r:
            pts.append([x, y])
            x += 2 * base_r
        y += np.sqrt(3) * base_r
        row += 1
    hex_pts = np.array(pts[:N])
    hex_pts += np.random.uniform(-0.02, 0.02, hex_pts.shape)
    hex_pts = np.clip(hex_pts, 0.06, 0.94)
    configs.append(np.concatenate([hex_pts.ravel(), np.full(N, base_r * 0.8)]))
    
    # 2. Uniform grid with jitter
    pts = []
    for i in range(6):
        for j in range(5):
            x = 0.12 + i * 0.14
            y = 0.12 + j * 0.17
            if x <= 0.9 and y <= 0.9:
                pts.append([x, y])
    grid_pts = np.array(pts[:N])
    grid_pts += np.random.uniform(-0.015, 0.015, grid_pts.shape)
    grid_pts = np.clip(grid_pts, 0.06, 0.94)
    configs.append(np.concatenate([grid_pts.ravel(), np.full(N, base_r * 0.8)]))
    
    # 3. Corner/Edge heavy layout
    pts = [
        [0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85],
        [0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5],
        [0.35, 0.35], [0.65, 0.35], [0.35, 0.65], [0.65, 0.65],
        [0.5, 0.5], [0.5, 0.3], [0.5, 0.7], [0.3, 0.5], [0.7, 0.5],
        [0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75]
    ]
    while len(pts) < N:
        pts.append([np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)])
    edge_pts = np.array(pts[:N])
    edge_pts += np.random.uniform(-0.01, 0.01, edge_pts.shape)
    edge_pts = np.clip(edge_pts, 0.06, 0.94)
    configs.append(np.concatenate([edge_pts.ravel(), np.full(N, base_r * 0.7)]))
    
    # 4. Pure random valid start
    rand_pts = np.random.uniform(0.1, 0.9, (N, 2))
    configs.append(np.concatenate([rand_pts.ravel(), np.full(N, base_r * 0.6)]))
    
    return configs

def run_packing():
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    # Multi-start optimization
    for seed in range(20):
        for cfg in generate_initial_configs(seed):
            try:
                res = minimize(objective, cfg, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
                
                # Accept only if strictly feasible enough (allow tiny numerical slack)
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-7:
                    current_sum = -res.fun
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_v = res.x.copy()
            except Exception:
                continue
                
    # Fallback to a safe configuration if optimization fails completely
    if best_v is None:
        best_v = generate_initial_configs(0)[0]
        
    cx = best_v[:N].copy()
    cy = best_v[N:2*N].copy()
    cr = best_v[2*N:].copy()
    
    # --- Strict Post-Processing to Guarantee Validator Compliance ---
    
    # 1. Enforce boundary constraints strictly
    for i in range(N):
        cr[i] = min(cr[i], cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
        
    # 2. Enforce non-overlap strictly with iterative shrinking
    # Repeatedly check pairs and shrink the larger radius slightly if overlapping
    for _ in range(10):
        max_shrink = 0.0
        for i in range(N):
            for j in range(i + 1, N):
                dx = cx[i] - cx[j]
                dy = cy[i] - cy[j]
                dist = np.sqrt(dx*dx + dy*dy)
                overlap = cr[i] + cr[j] - dist
                if overlap > 1e-10:
                    # Shrink proportionally to radius, or equally if one is near zero
                    if cr[i] + cr[j] > 1e-9:
                        shrink_i = overlap * (cr[i] / (cr[i] + cr[j]))
                        shrink_j = overlap - shrink_i
                    else:
                        shrink_i = overlap * 0.5
                        shrink_j = overlap * 0.5
                    
                    cr[i] = max(0.0, cr[i] - shrink_i)
                    cr[j] = max(0.0, cr[j] - shrink_j)
                    max_shrink = max(max_shrink, overlap)
        if max_shrink < 1e-12:
            break
            
    # Final boundary clamp to be absolutely safe against float drift
    for i in range(N):
        cr[i] = min(cr[i], cx[i] - 1e-9, 1.0 - cx[i] - 1e-9, 
                    cy[i] - 1e-9, 1.0 - cy[i] - 1e-9)
        cr[i] = max(0.0, cr[i])
        
    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
