import numpy as np
from scipy.optimize import minimize

def objective(v, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def constraints(v, n, i_idx, j_idx):
    """
    Computes all inequality constraints:
    1. Boundary: x >= r, x + r <= 1, y >= r, y + r <= 1
    2. Pairwise: dist^2 >= (r_i + r_j)^2
    Returns a flat array where each element >= 0 indicates a satisfied constraint.
    """
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    
    # Boundary constraints
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints (vectorized)
    xi, xj = x[i_idx], x[j_idx]
    yi, yj = y[i_idx], y[j_idx]
    ri, rj = r[i_idx], r[j_idx]
    
    dx = xi - xj
    dy = yi - yj
    dist_sq = dx*dx + dy*dy
    r_sum = ri + rj
    
    c = np.concatenate([c, dist_sq - r_sum*r_sum])
    return c

def get_feasible_radii(centers, n):
    """
    Projects an initial radius guess to a strictly feasible configuration
    for the given centers by iteratively shrinking overlapping circles.
    """
    r = np.full(n, 0.12)
    for _ in range(80):
        for i in range(n):
            # Wall constraints
            r[i] = min(r[i], centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            # Neighbor constraints
            for j in range(n):
                if i != j:
                    d = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
                    r[i] = min(r[i], d - r[j])
        r = np.maximum(r, 0.0)
    return r

    init_centers_list = []

    init_centers_list.append(np.array(pts[:n]))

            grid_pts.append([x_val, y_val])
    init_centers_list.append(np.array(grid_pts[:n]))
    
    # 3. Perturbed hexagonal lattices

        hex_pts = init_centers_list[0].copy()
        hex_pts += np.random.uniform(-0.03, 0.03, hex_pts.shape)
