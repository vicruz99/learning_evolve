import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(v, n):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def compute_constraints(v, n, ii, jj):
    """
    Computes all inequality constraints as a flat array.
    Constraints must be >= 0.
    Structure: [x-r, 1-x-r, y-r, 1-y-r, squared_dist - (r_i+r_j)^2]
    """
    xs = v[:n]
    ys = v[n:2*n]
    rs = v[2*n:]
    
    n_pairs = len(ii)
    c = np.empty(4*n + n_pairs)
    k = 0
    
    # Boundary constraints
    c[k:k+n] = xs - rs; k += n
    c[k:k+n] = 1.0 - xs - rs; k += n
    c[k:k+n] = ys - rs; k += n
    c[k:k+n] = 1.0 - ys - rs; k += n
    
    # Pairwise non-overlap constraints (squared distances)
    dx = xs[ii] - xs[jj]
    dy = ys[ii] - ys[jj]
    dr = rs[ii] + rs[jj]
    c[k:] = dx*dx + dy*dy - dr*dr
    
    return c

def run_packing():
    n = N_CIRCLES
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Bounds: coordinates in [0, 1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    # Precompute unique pair indices for vectorized pairwise constraints
    ii, jj = np.triu_indices(n, k=1)
    
    # Multi-restart optimization to escape local minima
    for seed in range(20):
        np.random.seed(seed)
        
        # 1. Initialization: Hexagonal lattice with perturbation
        pts = []
        r_init = 0.095
        y = r_init
        row = 0
        while len(pts) < n + 5:
            x_start = r_init + (row % 2) * r_init
            x = x_start
            while x <= 1.0 - r_init:
                pts.append([x, y])
                x += 2 * r_init
            y += np.sqrt(3) * r_init
            row += 1
            
        pts = np.array(pts[:n])
        # Add controlled noise to break symmetry
        pts += np.random.uniform(-0.01, 0.01, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        
        # Vary initial radii slightly to encourage unequal packing exploration
        r0 = np.full(n, 0.06) * (1.0 + np.random.uniform(-0.2, 0.2, n))
        v0 = np.concatenate([pts.ravel(), r0])
        
        # 2. Optimization
        try:
            res = minimize(compute_objective, v0, args=(n,),
                           method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': compute_constraints, 'args': (n, ii, jj)},
                           options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            
            current_sum = -res.fun
            if current_sum > best_sum:
                cs = res.x[:2*n].reshape(n, 2)
                rs = res.x[2*n:]
                
                # 3. Strict validity verification
                valid = True
                if np.any(rs < -1e-6): valid = False
                if np.any(cs[:, 0] - rs < -1e-6) or np.any(1 - cs[:, 0] - rs < -1e-6): valid = False
                if np.any(cs[:, 1] - rs < -1e-6) or np.any(1 - cs[:, 1] - rs < -1e-6): valid = False
                
                if valid:
                    dx = cs[ii, 0] - cs[jj, 0]
                    dy = cs[ii, 1] - cs[jj, 1]
                    dists = np.sqrt(dx*dx + dy*dy)
                    if np.any(dists < rs[ii] + rs[jj] - 1e-6):
                        valid = False
                        
                if valid:
                    best_sum = current_sum
                    best_centers = cs.copy()
                    best_radii = rs.copy()
        except Exception:
            continue
            
    # 4. Deterministic post-processing to guarantee strict feasibility
    centers = best_centers
    radii = best_radii
    
    # Clamp to boundaries
    for i in range(n):
        radii[i] = min(radii[i], centers[i,0], 1.0 - centers[i,0], 
                       centers[i,1], 1.0 - centers[i,1])

    for _ in range(15):
        changed = False
        for i in range(n):
            for j in range(i+1, n):

                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
