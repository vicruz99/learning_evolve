import numpy as np
from scipy.optimize import minimize

N = 26

# Precompute upper-triangle pair indices for efficient constraint evaluation
_PAIR_INDICES = np.array([(i, j) for i in range(N) for j in range(i + 1, N)])

def _objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def _constraints(v):
    """
    Computes inequality constraints:
    1. Boundary: center +/- radius within [0, 1]
    2. Pairwise: distance >= sum of radii
    Returns a 1D array where all elements must be >= 0.
    """
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]

    c = np.column_stack((cx, cy))
    # Vectorized pairwise distances
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

    # Boundary constraints (4 per circle)
    cons = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])

    # Pairwise non-overlap constraints
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    pair_cons = dists[_PAIR_INDICES[:, 0], _PAIR_INDICES[:, 1]] - r_sum[_PAIR_INDICES[:, 0], _PAIR_INDICES[:, 1]]
    cons = np.concatenate([cons, pair_cons])
    
    return cons

def run_packing():
    # Variable bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': _constraints}

    best_val = -np.inf
    best_v = None

    # Multi-start optimization with diverse hexagonal initializations
    for seed in range(20):
        np.random.seed(seed)
        pts = []
        y = 0.06
        row = 0
        # Generate more points than needed to allow selection/perturbation
        while len(pts) < N + 15:
            x_start = 0.06 if row % 2 == 0 else 0.12
            x = x_start
            while x <= 0.94:
                pts.append([x, y])
                x += 0.12
            y += 0.10
            row += 1
        pts = np.array(pts[:N])

        # Perturb to break symmetry and explore different contact graphs
        pts += np.random.uniform(-0.015, 0.015, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)

        # Start with small feasible radii to guarantee initial validity
        r_init = np.full(N, 0.03)
        v0 = np.concatenate([pts.flatten(), r_init])

        try:
            res = minimize(_objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons_dict, options={'maxiter': 4000, 'ftol': 1e-12})
            
            # Accept if better and sufficiently feasible
            if -res.fun > best_val:
                cvals = _constraints(res.x)
                if np.min(cvals) >= -1e-7:
                    best_val = -res.fun
                    best_v = res.x.copy()
        except Exception:
            pass

    if best_v is not None:
        cx = best_v[:N]
        cy = best_v[N:2*N]
        radii = best_v[2*N:].copy()
        centers = np.column_stack((cx, cy))

        # Strict validity enforcement to satisfy validator's 1e-12 tolerance
        # 1. Enforce boundary constraints
        for i in range(N):
            x, y = centers[i]
            radii[i] = min(radii[i], x, 1.0 - x, y, 1.0 - y)

        # 2. Enforce pairwise non-overlap with safety margin
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx*dx + dy*dy)
                if radii[i] + radii[j] > d:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)

        return centers, radii, float(np.sum(radii))
    else:
        # Fallback valid configuration (should rarely be reached)
        centers = np.random.uniform(0.15, 0.85, (N, 2))
        radii = np.full(N, 0.03)
        return centers, radii, float(np.sum(radii))
