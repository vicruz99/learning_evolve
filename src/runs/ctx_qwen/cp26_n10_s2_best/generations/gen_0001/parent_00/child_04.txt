import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2::3])

def constraints(v):
    """
    Computes all inequality constraints:
    1. Boundary: center +/- radius within [0, 1]
    2. Pairwise: squared distance >= squared sum of radii
    Returns a flat array where each element >= 0 indicates a satisfied constraint.
    """
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    # Boundary constraints
    cons = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Pairwise squared distance constraints (smoother optimization landscape)
    pair_cons = []
    for i in range(N):
        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
        for j in range(i + 1, N):
            dx = xi - v[3*j]
            dy = yi - v[3*j+1]
            sum_r = ri + v[3*j+2]
            pair_cons.append(dx*dx + dy*dy - sum_r**2)
            
    return np.concatenate([cons, pair_cons])

def get_initial_guess(seed):

    np.random.seed(seed)
    r_est = 0.10
    centers = []
    y = r_est
    row = 0

def run_packing():

    cons = {'type': 'ineq', 'fun': constraints}

    for seed in range(15):
        v0 = get_initial_guess(seed)
