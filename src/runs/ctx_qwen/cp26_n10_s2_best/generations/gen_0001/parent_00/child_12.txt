import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(v):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2*N_CIRCLES:])

def constraint_func(v):
    """
    Computes all inequality constraints:
    1. Boundary: center +/- radius within [0, 1]
    2. Pairwise: distance >= sum of radii
    Returns a flat array where each element >= 0 indicates a satisfied constraint.
    """
    cx = v[:N_CIRCLES]
    cy = v[N_CIRCLES:2*N_CIRCLES]
    r = v[2*N_CIRCLES:]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Pairwise non-overlap constraints (vectorized)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    # Extract upper triangle (i < j) to avoid duplicates and self-comparisons
    mask = np.triu(np.ones((N_CIRCLES, N_CIRCLES), dtype=bool), k=1)
    c = np.concatenate([c, (dist - r_sum)[mask]])
    return c

def get_initial_guess(seed, r_start):
    """Generate a perturbed hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    y = r_start
    row = 0
    while len(pts) < N_CIRCLES + 10:
        x = r_start if row % 2 == 0 else 2.0 * r_start
        while x <= 1.0 - r_start and len(pts) < N_CIRCLES + 10:
            pts.append([x, y])
            x += 2.0 * r_start
        y += r_start * np.sqrt(3.0)
        row += 1
        
    pts = np.array(pts[:N_CIRCLES])
    # Add controlled jitter to break symmetry and aid optimization
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    
    r = np.full(N_CIRCLES, r_start)
    return np.concatenate([pts[:, 0], pts[:, 1], r])

def run_packing():

    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n

    for s in scales:
