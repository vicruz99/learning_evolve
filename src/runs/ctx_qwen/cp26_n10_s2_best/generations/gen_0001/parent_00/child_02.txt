import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """
    Computes inequality constraints.
    Returns a 1D array where each element >= 0 means the constraint is satisfied.
    Uses squared distances for better numerical stability.
    """
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    cx_col = cx[:, np.newaxis]
    cy_col = cy[:, np.newaxis]
    r_col = r[:, np.newaxis]
    
    dx = cx_col - cx_col.T
    dy = cy_col - cy_col.T
    dist_sq = dx*dx + dy*dy
    r_sum = r_col + r_col.T
    
    # Extract only upper triangle (i < j)
    i_idx, j_idx = np.triu_indices(N, k=1)
    pair_cons = dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx]**2
    
    return np.concatenate([cons, pair_cons])

def generate_initial_guess(seed):

    np.random.seed(seed)

    counts = [6, 5, 6, 5, 4]
    pts = []

    spacing_x = 0.16
    spacing_y = spacing_x * np.sqrt(3) / 2.0

            if len(pts) >= N:
                break
            pts.append([x, y])
            x += spacing_x
        y += spacing_y
        
    pts = np.array(pts[:N])
    
    # Add controlled jitter to break symmetry and help escape local minima
    jitter = np.random.uniform(-0.015, 0.015, size=pts.shape)
    pts = np.clip(pts + jitter, 0.05, 0.95)
    
    # Initial radii small enough to guarantee feasibility (min dist ~0.18, so 0.08 is safe)
    r_init = np.full(N, 0.06)
    
    return np.concatenate([pts.ravel(), r_init])

def run_packing():

    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}

    best_x = None

    for seed in range(30):

            x0 = generate_initial_guess(seed)
            
            # SLSQP is robust for this problem size
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,

        return centers, radii, float(np.sum(radii))

    radii = np.maximum(radii, 0.0)

    margin_x = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
