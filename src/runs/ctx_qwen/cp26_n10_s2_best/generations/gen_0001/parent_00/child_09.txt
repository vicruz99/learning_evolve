import numpy as np
from scipy.optimize import minimize

N = 26
INDICES = np.triu_indices(N, k=1)

def obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons(v):
    """Inequality constraints: boundaries and non-overlap."""
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Pairwise non-overlap: dist >= r_i + r_j
    i, j = INDICES
    dx = cx[i] - cx[j]
    dy = cy[i] - cy[j]
    dist = np.sqrt(dx*dx + dy*dy)
    c = np.concatenate([c, dist - (r[i] + r[j])])
    
    return c

def run_packing():
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_sum = 0.0
    best_v = None
    candidates = []
    
    # 1. Generate Hexagonal Lattice Candidates
    for seed in range(8):

                x += 2 * r0

            row += 1

                    pts.append([x, y])

        pts += np.random.uniform(-0.01, 0.01, pts.shape)
        pts = np.clip(pts, 0.01, 0.99)
        candidates.append(np.concatenate([pts.ravel(), np.full(N, 0.04)]))

    for v0 in candidates:
