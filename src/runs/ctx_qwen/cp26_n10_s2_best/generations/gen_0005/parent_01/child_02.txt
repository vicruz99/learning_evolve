import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """
    Compute inequality constraints: boundaries and pairwise non-overlap.
    All elements must be >= 0 for SLSQP.
    """
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    # Boundary constraints
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap constraints (squared for smooth gradients)
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def solve_lp_radii(centers):
    """
    Solve LP to maximize sum of radii for fixed centers.
    Constraints: r_i <= wall_dist, r_i + r_j <= dist(i,j), r_i >= 0
    """
    c_obj = -np.ones(N)
    A_ub = np.zeros((4*N + NUM_PAIRS, N))
    b_ub = np.zeros(4*N + NUM_PAIRS)
    
    # Wall constraints
    for i in range(N):
        x, y = centers[i]
        idx = 4*i
        A_ub[idx, i] = 1.0; b_ub[idx] = x
        A_ub[idx+1, i] = 1.0; b_ub[idx+1] = 1.0 - x
        A_ub[idx+2, i] = 1.0; b_ub[idx+2] = y
        A_ub[idx+3, i] = 1.0; b_ub[idx+3] = 1.0 - y
        
    # Pairwise distance constraints
    dx = centers[PAIR_I, 0] - centers[PAIR_J, 0]
    dy = centers[PAIR_I, 1] - centers[PAIR_J, 1]
    dists = np.hypot(dx, dy)
    
    for k in range(NUM_PAIRS):
        idx = 4*N + k
        A_ub[idx, PAIR_I[k]] = 1.0
        A_ub[idx, PAIR_J[k]] = 1.0
        b_ub[idx] = dists[k]
        
    bounds = [(0.0, None)] * N
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success and not res.unbounded:
        return res.x
    # Fallback if LP fails (e.g., centers too close)
    return np.full(N, 0.03)
