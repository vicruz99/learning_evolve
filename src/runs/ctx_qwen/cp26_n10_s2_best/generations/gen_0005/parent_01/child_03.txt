import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """
    Compute inequality constraints: boundaries and pairwise non-overlap.
    All elements must be >= 0 for feasibility.
    """
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    
    # Boundary constraints: circles inside [0,1]x[0,1]
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def make_feasible(centers, radii):
    """Adjusts radii to guarantee strict constraint satisfaction."""
    x, y = centers[:, 0], centers[:, 1]
    r = radii.copy()
    
    # Enforce boundaries
    r = np.minimum(r, np.minimum(x, 1.0 - x))
    r = np.minimum(r, np.minimum(y, 1.0 - y))
    
    # Iteratively resolve pairwise overlaps
    for _ in range(15):
        changed = False
        for k in range(NUM_PAIRS):
            i, j = PAIR_I[k], PAIR_J[k]
            d = np.hypot(x[i] - x[j], y[i] - y[j])
            if r[i] + r[j] > d + 1e-10:
                shrink = (r[i] + r[j] - d) * 0.5 + 1e-9
                r[i] = max(0.0, r[i] - shrink)
                r[j] = max(0.0, r[j] - shrink)
                changed = True
        if not changed:
            break
    return r

def generate_configs():
    """Generates a diverse set of high-quality initial configurations."""
    configs = []
    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 6, 6, 4],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [7, 5, 5, 5, 4]
    ]

        if sum(pat) < N:
            continue

        y = r0
        row_idx = 0
        for count in pat:
            x_start = r0 if row_idx % 2 == 0 else 2.0 * r0
            for k in range(count):
                if len(pts) >= N:
                    break
                x = x_start + k * 2.0 * r0
                pts.append([x, y])
            y += r0 * np.sqrt(3.0)
            row_idx += 1
        pts = np.array(pts[:N])
        
        for angle in [0.0, 0.06, -0.06, 0.12]:

                    p = pts.copy()

                        c, s = np.cos(angle), np.sin(angle)
                        p = (p - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                    p += [sx, sy]
                    p = np.clip(p, 0.02, 0.98)
                    
                    r_init = make_feasible(p, np.full(N, 0.04))
                    v = np.concatenate([p[:, 0], p[:, 1], r_init])
                    configs.append(v)
                    
    # 2. Force-Relaxed Random Starts
    for seed in range(20):
        np.random.seed(seed)
        pts = np.random.uniform(0.1, 0.9, (N, 2))

    return configs

def run_packing():

    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).

                cr[j] = max(0.0, cr[j] - shrink)
                changed = True
        if not changed:
            break
            
    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
