import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def make_feasible(v):
    """Adjusts radii to guarantee the configuration satisfies all constraints strictly."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:].copy()
    
    # Enforce boundary constraints
    r = np.minimum(r, np.minimum(x, 1.0 - x))
    r = np.minimum(r, np.minimum(y, 1.0 - y))
    
    # Enforce non-overlap constraints iteratively
    for _ in range(25):
        dx = x[PAIR_I] - x[PAIR_J]
        dy = y[PAIR_I] - y[PAIR_J]
        dist = np.hypot(dx, dy)
        overlap = (r[PAIR_I] + r[PAIR_J]) - dist
        if np.max(overlap) < 1e-11:
            break
        shrink = np.maximum(0.0, overlap) / 2.0 + 1e-9
        r[PAIR_I] = np.maximum(0.0, r[PAIR_I] - shrink)
        r[PAIR_J] = np.maximum(0.0, r[PAIR_J] - shrink)
        
    return np.concatenate([x, y, r])

def generate_hex_configs():

    configs = []
    row_patterns = [
        [5, 5, 5, 5, 6], [6, 5, 6, 5, 4], [4, 6, 6, 6, 4],
        [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], [7, 5, 5, 5, 4],

    ]

            continue
        r0 = 0.095
        pts = []
        y = r0
        row_idx = 0
        for count in pat:
            x_start = r0 if row_idx % 2 == 0 else 2.0 * r0
            for k in range(count):
                if len(pts) >= N:
                    break
                x = x_start + k * 2.0 * r0
                pts.append([x, y])
            y += r0 * np.sqrt(3.0)
            row_idx += 1
            
        pts = np.array(pts[:N])
        
        # Apply variations
        for angle in np.linspace(-0.15, 0.15, 5):
            for sx in np.linspace(-0.03, 0.03, 3):
                for sy in np.linspace(-0.03, 0.03, 3):
                    p = pts.copy()
                    if angle != 0.0:
                        c_val, s_val = np.cos(angle), np.sin(angle)
                        rot = np.array([[c_val, -s_val], [s_val, c_val]])
                        p = (p - 0.5) @ rot.T + 0.5
                    p += [sx, sy]
                    p = np.clip(p, 0.02, 0.98)
                    
                    r_init = 0.04 + np.random.uniform(-0.002, 0.002, N)
                    v = np.concatenate([p[:,0], p[:,1], r_init])
                    configs.append(make_feasible(v))
    return configs

def generate_force_configs():
    """Generate configurations using repulsive force relaxation."""
    configs = []
    for seed in range(20):
        np.random.seed(seed + 1000)
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        
        # Repulsion relaxation

def run_packing():

    Returns (centers, radii, sum_radii).

            c = constraints(res.x)

                best_sum = s
                best_v = res.x.copy()
        except Exception:
            pass

        best_v = configs[0]

        # Shrink radii to create slack for rearrangement
        pert[2*N:] *= 0.97

        pert = make_feasible(pert)

            s = -res.fun
            c = constraints(res.x)

                best_sum = s
                best_v = res.x.copy()
                current_v = best_v.copy()
        except Exception:
            pass

    # Extract final configuration
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()
    
    # Phase 3: Strict Post-Processing for Validator Compliance
    eps = 1e-9
    # 1. Enforce boundary constraints strictly
    for i in range(N):
        mr = min(cx[i] - eps, 1.0 - cx[i] - eps,

        cr[i] = min(cr[i], max(mr, 0.0))

    for _ in range(30):
        changed = False
        for i in range(N):

                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
