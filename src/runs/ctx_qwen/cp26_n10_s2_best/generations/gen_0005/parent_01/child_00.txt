import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """
    Compute inequality constraints: boundaries and pairwise non-overlap.
    Uses squared distances for smoother gradients and numerical stability.
    All elements must be >= 0.
    """
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    r = np.minimum(r, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    
    # Scale down slightly to guarantee strict feasibility for optimizer
    return np.clip(r * 0.98, 1e-5, 0.5)

def generate_configs():
    """Generates a diverse set of initial configurations."""
    configs = []
    
    # 1. Hexagonal patterns with variations
    for r0 in np.linspace(0.075, 0.115, 7):
        for angle in np.linspace(-0.25, 0.25, 5):
            pts = []
            y = r0
            row = 0
            while len(pts) < N + 5:
                x_start = r0 if row % 2 == 0 else 2.0 * r0
                x = x_start
                while x <= 1.0 - r0 and len(pts) < N + 5:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
                row += 1
            pts = np.array(pts[:N])
            if angle != 0.0:
                c, s = np.cos(angle), np.sin(angle)
                pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pts = np.clip(pts, 0.02, 0.98)
            r_init = get_feasible_radii(pts)
            configs.append(np.concatenate([pts[:, 0], pts[:, 1], r_init]))
            
    # 2. Force-directed repulsion layouts
    for seed in range(20):
        np.random.seed(seed)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(200):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    d = np.linalg.norm(pts[i] - pts[j])
                    if d < 0.22 and d > 1e-6:
                        f = (0.22 - d) / d * 0.5
                        diff = pts[i] - pts[j]
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.02
            pts = np.clip(pts, 0.05, 0.95)
        r_init = get_feasible_radii(pts)
        configs.append(np.concatenate([pts[:, 0], pts[:, 1], r_init]))
        
    # 3. Largest gap heuristic initialization
    for seed in range(10):
        np.random.seed(seed)
        pts = np.zeros((N, 2))
        pts[0] = np.random.uniform(0.3, 0.7, 2)
        for k in range(1, N):
            grid = np.linspace(0.05, 0.95, 35)
            gx, gy = np.meshgrid(grid, grid)
            pts_grid = np.column_stack([gx.ravel(), gy.ravel()])
            
            min_d = np.min(np.linalg.norm(pts_grid[:, None, :] - pts[:k, None, :], axis=2), axis=1)
            wall_d = np.minimum(np.minimum(pts_grid[:, 0], 1.0 - pts_grid[:, 0]), 
                                np.minimum(pts_grid[:, 1], 1.0 - pts_grid[:, 1]))
            score = np.minimum(min_d, wall_d)
            best_idx = np.argmax(score)
            pts[k] = pts_grid[best_idx]
        r_init = get_feasible_radii(pts)
        configs.append(np.concatenate([pts[:, 0], pts[:, 1], r_init]))
        
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    configs = generate_configs()
    
    # Phase 1: Multi-start exploration
    for v0 in configs:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,

            s = -res.fun
            if s > best_sum:
                c = constraints(res.x)
                if np.min(c) >= -1e-6:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = configs[0]
        
    # Phase 2: Iterative shrink & perturb to escape local minima

        v_p = current_v.copy()

        v_p[:2*N] += np.random.uniform(-noise, noise, 2*N)
        v_p[:2*N] = np.clip(v_p[:2*N], 0.01, 0.99)
        
        # Shrink radii to create room for center rearrangement
        v_p[2*N:] *= 0.90
        v_p[2*N:] = get_feasible_radii(v_p[:2*N].reshape(N, 2))
        
        try:
            res = minimize(objective, v_p, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                c = constraints(res.x)
                if np.min(c) >= -1e-6:
                    best_sum = s
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            pass
            
    # Extract results
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()
    
    # Phase 3: Strict Post-Processing for Validator Compliance
    eps = 1e-9
    # 1. Enforce boundary constraints strictly
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    cr = np.maximum(cr, 0.0)
    
    # 2. Enforce non-overlap constraints iteratively with safety margin
    for _ in range(20):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                if cr[i] + cr[j] > d - eps:
                    shrink = (cr[i] + cr[j] - d) / 2.0 + eps
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
