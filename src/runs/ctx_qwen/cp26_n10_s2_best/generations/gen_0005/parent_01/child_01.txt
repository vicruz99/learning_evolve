import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    return np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r,
        (x[PAIR_I] - x[PAIR_J])**2 + (y[PAIR_I] - y[PAIR_J])**2 - (r[PAIR_I] + r[PAIR_J])**2
    ])

def get_feasible_radii(centers, scale=0.7):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.25)
    r = np.minimum(r, centers[:, 0])
    r = np.minimum(r, 1.0 - centers[:, 0])
    r = np.minimum(r, centers[:, 1])
    r = np.minimum(r, 1.0 - centers[:, 1])
    
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            val = d / 2.0
            r[i] = min(r[i], val)
            r[j] = min(r[j], val)
    
    return np.clip(r * scale, 1e-5, 0.25)

def force_relax(centers, iters=120, target=0.24):
    """Relax configuration using repulsive forces."""
    pts = centers.copy()
    lr = 0.02
    for _ in range(iters):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[i] - pts[j]
                d = np.linalg.norm(diff)
                if d < target and d > 1e-6:
                    f = (target - d) * 0.4 / d
                    forces[i] += f * diff
                    forces[j] -= f * diff
        # Wall repulsion
        for i in range(N):
            if pts[i, 0] < 0.08:
                forces[i, 0] += 0.08
            if pts[i, 0] > 0.92:
                forces[i, 0] -= 0.08
            if pts[i, 1] < 0.08:
                forces[i, 1] += 0.08
            if pts[i, 1] > 0.92:
                forces[i, 1] -= 0.08
        pts += forces * lr
        pts = np.clip(pts, 0.03, 0.97)
        lr *= 0.995
    return pts

def generate_hex_configs():

    configs = []
    r0_values = np.linspace(0.075, 0.125, 11)

    shifts = [(-0.015, -0.015), (-0.015, 0.0), (-0.015, 0.015),
              (0.0, -0.015), (0.0, 0.0), (0.0, 0.015),
              (0.015, -0.015), (0.015, 0.0), (0.015, 0.015)]
    
    for r0 in r0_values:
        for angle in angles:
            for sx, sy in shifts:
                pts = []
                y = r0 + sy
                row = 0
                while len(pts) < N + 12:
                    x_start = r0 + sx + (row % 2) * r0
                    x = x_start
                    while x <= 1.0 - r0 and len(pts) < N + 12:
                        pts.append([x, y])
                        x += 2 * r0
                    y += np.sqrt(3) * r0
                    row += 1
                
                pts = np.array(pts[:N])
                if angle != 0.0:
                    c, s = np.cos(angle), np.sin(angle)
                    pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                
                pts = np.clip(pts, 0.02, 0.98)
                valid = ((pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) &
                         (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99))
                if np.sum(valid) >= N:
                    cfg = pts[valid][:N]
                    r = get_feasible_radii(cfg, scale=0.65)
                    configs.append(np.concatenate([cfg[:, 0], cfg[:, 1], r]))
    
    return configs

def generate_random_configs(n_configs=20):
    """Generate force-relaxed random configurations."""
    configs = []
    for seed in range(n_configs):
        np.random.seed(seed + 1000)
        pts = force_relax(np.random.uniform(0.1, 0.9, (N, 2)))
        r = get_feasible_radii(pts, scale=0.65)
        configs.append(np.concatenate([pts[:, 0], pts[:, 1], r]))
    return configs

def run_optimization(v0, bounds, cons, maxiter=8000):
    """Run single optimization, return result or None."""
    try:
        res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        return res
    except Exception:
        return None
