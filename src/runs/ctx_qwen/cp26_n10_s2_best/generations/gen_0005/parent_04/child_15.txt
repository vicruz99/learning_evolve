import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def compute_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.5)
    wall_dists = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    r = np.minimum(r, wall_dists)
    
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    r = np.minimum(r, min_dists / 2.0)
    
    return np.clip(r * 0.5, 1e-6, 0.25)

def generate_hex_centers(r0, shift_x=0.0, shift_y=0.0, angle=0.0):
    """Generate hexagonal lattice centers."""
    pts = []
    y = r0 + shift_y
    row = 0
    while len(pts) < N + 10:
        x_start = r0 + shift_x + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
        row += 1
    
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        pts = pts - 0.5
        c, s = np.cos(angle), np.sin(angle)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts = pts + 0.5
    
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def generate_configs():
    """Generate diverse initial configurations."""
    configs = []
    
    # 1. Extensive hexagonal lattice exploration
    for r0 in np.linspace(0.075, 0.110, 14):
        for shift_x in [-0.02, -0.01, 0.0, 0.01, 0.02]:
            for shift_y in [-0.02, -0.01, 0.0, 0.01, 0.02]:
                for ang in np.linspace(-0.15, 0.15, 9):
                    pts = generate_hex_centers(r0, shift_x, shift_y, ang)
                    valid = (pts[:, 0] >= 0.03) & (pts[:, 0] <= 0.97) & (pts[:, 1] >= 0.03) & (pts[:, 1] <= 0.97)
                    if np.sum(valid) >= N:
                        configs.append(pts[valid][:N])
    
    # 2. Force-relaxed random configurations
    for seed in range(30):
        np.random.seed(seed)
        pts = np.random.uniform(0.12, 0.88, (N, 2))
        for _ in range(300):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    diff = pts[i] - pts[j]
                    d = np.linalg.norm(diff)
                    if d < 0.28 and d > 1e-6:
                        f = (0.28 - d) * 0.8 / d
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.03
            pts = np.clip(pts, 0.04, 0.96)
        configs.append(pts)
    
    # 3. Grid configurations with variations
    for s_x in np.linspace(0.15, 0.20, 6):
        for s_y in np.linspace(0.18, 0.22, 4):
            pts_list = []
            for i in range(6):
                for j in range(5):
                    pts_list.append([0.06 + i * s_x, 0.06 + j * s_y])
            if len(pts_list) >= N:
                configs.append(np.array(pts_list[:N]))
    
    # 4. Corner-focused configurations
    for seed in range(15):
        np.random.seed(seed + 500)
        # Place circles near corners first, then fill middle
        corners = np.array([[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]])
        mid_points = np.random.uniform(0.25, 0.75, (N - 4, 2))
        pts = np.vstack([corners, mid_points])
        # Relax
        for _ in range(200):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    diff = pts[i] - pts[j]
                    d = np.linalg.norm(diff)
                    if d < 0.22 and d > 1e-6:
                        f = (0.22 - d) / d
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.02
            pts = np.clip(pts, 0.03, 0.97)
        configs.append(pts)
    
    # 5. Edge-focused configurations

        n_per_edge = 6

            edge_pts.append([0.95, 0.05 + i * 0.9 / (n_per_edge - 1)])
        used = np.array(edge_pts[:N-10])
        mid_pts = np.random.uniform(0.3, 0.7, (N - len(used), 2))
        pts = np.vstack([used, mid_pts])
        pts += np.random.uniform(-0.02, 0.02, pts.shape)
        pts = np.clip(pts, 0.03, 0.97)
        for _ in range(100):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    diff = pts[i] - pts[j]
                    d = np.linalg.norm(diff)
                    if d < 0.20 and d > 1e-6:
                        f = (0.20 - d) / d
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.015
            pts = np.clip(pts, 0.03, 0.97)
        configs.append(pts)
    
    return configs

def run_packing():
    np.random.seed(42)
    configs = generate_configs()
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    # Phase 1: Multi-start optimization
    for c_init in configs:
        r_init = compute_feasible_radii(c_init)
        v0 = np.concatenate([c_init[:, 0], c_init[:, 1], r_init])
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
    
    if best_v is None:
        return np.zeros((N, 2)), np.zeros(N), 0.0
    
    # Phase 2: Iterative perturbation & refinement
    current_v = best_v.copy()
    for step in range(50):
        np.random.seed(step + 3000)
        v_pert = current_v.copy()
        noise_scale = 0.008 - step * 0.0001
        noise_scale = max(noise_scale, 0.001)
        v_pert[:2*N] += np.random.uniform(-noise_scale, noise_scale, 2*N)
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.01, 0.99)
        
        # Shrink radii progressively
        shrink_factor = max(0.65, 0.85 - step * 0.003)
        v_pert[2*N:] *= shrink_factor
        
        # Recompute feasible radii from perturbed centers
        c_pts = v_pert[:2*N].reshape(N, 2)
        v_pert[2*N:] = compute_feasible_radii(c_pts) * 0.88
        
        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            pass
    
    # Phase 3: More aggressive perturbation with larger noise
    current_v = best_v.copy()
    for step in range(25):
        np.random.seed(step + 7000)
        v_pert = current_v.copy()
        noise_scale = 0.015 - step * 0.0005
        noise_scale = max(noise_scale, 0.003)
        v_pert[:2*N] += np.random.uniform(-noise_scale, noise_scale, 2*N)
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.01, 0.99)
        
        c_pts = v_pert[:2*N].reshape(N, 2)
        v_pert[2*N:] = compute_feasible_radii(c_pts) * 0.82
        
        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            pass
    
    # Extract results
    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    radii = best_v[2*N:].copy()
    
    # Strict post-processing to guarantee validator compliance
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    radii = np.minimum(radii, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    for _ in range(25):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
    
    return centers, radii, float(np.sum(radii))
