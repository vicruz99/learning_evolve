import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def compute_feasible_radii(centers):
    """Compute strictly feasible radii based on local geometry."""
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    return r * 0.95

def generate_starts():
    """Generate diverse initial configurations."""
    starts = []
    # 1. Hexagonal lattices with variations
    for r0 in np.linspace(0.085, 0.105, 5):
        for ang in np.linspace(-0.15, 0.15, 5):
            pts = []
            y = r0
            row = 0
            while len(pts) < N + 5:
                x = r0 + (row % 2) * r0
                while x <= 1.0 - r0 and len(pts) < N + 5:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
                row += 1
            pts = np.array(pts[:N])
            if ang != 0.0:
                pts -= 0.5
                c, s = np.cos(ang), np.sin(ang)
                pts = pts @ np.array([[c, -s], [s, c]])
                pts += 0.5
            pts = np.clip(pts, 0.02, 0.98)
            starts.append(np.concatenate([pts[:, 0], pts[:, 1], compute_feasible_radii(pts)]))

    # 2. Force-directed random layouts
    for seed in range(15):
        np.random.seed(seed)
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        for _ in range(300):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    d = np.linalg.norm(pts[i] - pts[j])
                    if d < 0.25 and d > 1e-5:
                        f = (0.25 - d) / d * 0.5
                        diff = pts[i] - pts[j]
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.02
            pts = np.clip(pts, 0.05, 0.95)
        starts.append(np.concatenate([pts[:, 0], pts[:, 1], compute_feasible_radii(pts)]))

    # 3. Perturbed grid variations
    for sx in [0.15, 0.17, 0.18]:
        for sy in [0.18, 0.20, 0.22]:
            pts = []
            for i in range(6):
                for j in range(5):
                    pts.append([0.05 + i*sx, 0.05 + j*sy])
            pts = np.array(pts[:N])
            pts += np.random.uniform(-0.01, 0.01, pts.shape)
            pts = np.clip(pts, 0.02, 0.98)
            starts.append(np.concatenate([pts[:, 0], pts[:, 1], compute_feasible_radii(pts)]))
            
    return starts

def run_packing():
    """Optimizes packing of 26 circles in a unit square to maximize sum of radii."""
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    # Phase 1: Multi-start optimization
    starts = generate_starts()
    for v0 in starts:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        return np.zeros((N,2)), np.zeros(N), 0.0

    # Phase 2: Iterative local search refinement
    current_v = best_v.copy()
    for step in range(50):
        pert = current_v.copy()
        noise_scale = 0.008 * (1.0 - step/50.0)
        pert[:2*N] += np.random.uniform(-noise_scale, noise_scale, 2*N)
        pert[:2*N] = np.clip(pert[:2*N], 0.01, 0.99)
        
        centers_p = pert[:2*N].reshape(N, 2)
        pert[2*N:] = compute_feasible_radii(centers_p) * 0.92
        
        try:
            res = minimize(objective, pert, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            pass
            
    # Phase 3: Inflation escape to push boundaries
    for _ in range(5):
        v_inf = best_v.copy()
        v_inf[2*N:] *= 1.02
        v_inf[2*N:] = np.minimum(v_inf[2*N:], 0.49)
        try:
            res = minimize(objective, v_inf, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass

    # Extract final configuration
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()
    
    # Phase 4: Strict post-processing for validator compliance
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    cr = np.maximum(cr, 0.0)
    
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(cx[i]-cx[j], cy[i]-cy[j])
                if cr[i] + cr[j] > d - 1e-9:
                    shrink = (cr[i] + cr[j] - d) / 2.0 + 1e-9
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    centers = np.column_stack([cx, cy])
    return centers, cr, float(np.sum(cr))
