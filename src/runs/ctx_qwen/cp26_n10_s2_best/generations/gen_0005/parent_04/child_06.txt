import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    
    # Boundary constraints
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def get_feasible_radii(centers, shrink=0.85):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.5)
    for i in range(N):
        mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d/2.0 < mr:
                    mr = d/2.0
        r[i] = mr
    return r * shrink

def generate_starts():
    """Generate diverse initial configurations."""
    starts = []
    
    # 1. Hexagonal lattices with variations
    for r0 in np.linspace(0.088, 0.105, 6):
        for ang in np.linspace(-0.2, 0.2, 5):
            for sy in np.linspace(-0.02, 0.02, 3):
                pts = []
                y = r0 + sy
                row = 0
                while len(pts) < N + 5:
                    x = r0 + (row % 2) * r0
                    while x <= 1.0 - r0 and len(pts) < N + 5:
                        pts.append([x, y])
                        x += 2.0 * r0
                    y += r0 * np.sqrt(3.0)
                    row += 1
                pts = np.array(pts[:N])
                if ang != 0.0:
                    pts -= 0.5
                    c, s = np.cos(ang), np.sin(ang)
                    pts = pts @ np.array([[c, -s], [s, c]])
                    pts += 0.5
                pts = np.clip(pts, 0.02, 0.98)
                starts.append(np.concatenate([pts[:, 0], pts[:, 1], get_feasible_radii(pts, 0.80)]))

    # 2. Force-relaxed random configurations
    for seed in range(25):
        np.random.seed(seed)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(600):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    d = np.hypot(pts[i,0]-pts[j,0], pts[i,1]-pts[j,1])
                    if d < 0.25 and d > 1e-5:
                        f = (0.25 - d) / d * 0.5
                        diff = pts[i] - pts[j]
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.03
            pts = np.clip(pts, 0.05, 0.95)
        starts.append(np.concatenate([pts[:, 0], pts[:, 1], get_feasible_radii(pts, 0.80)]))
        
    # 3. Perturbed grid configurations
    for s in np.linspace(0.0, 0.04, 3):
        pts = np.array([[0.08 + i*0.16 + s, 0.08 + j*0.18 + s] 
                        for i in range(6) for j in range(5)][:N])
        pts += np.random.uniform(-0.01, 0.01, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        starts.append(np.concatenate([pts[:, 0], pts[:, 1], get_feasible_radii(pts, 0.80)]))
        
    return starts

def run_packing():
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    # Phase 1: Multi-start optimization
    starts = generate_starts()
    for v0 in starts:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-6:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = starts[0]
        
    # Phase 2: Iterative refinement to escape local minima
    current_v = best_v.copy()
    for step in range(40):
        pert = current_v.copy()

        pert[:2*N] += np.random.uniform(-noise_scale, noise_scale, 2*N)
        pert[:2*N] = np.clip(pert[:2*N], 0.02, 0.98)
        
        # Shrink radii progressively to allow major rearrangement
        shrink_factor = max(0.60, 0.85 - step * 0.006)
        centers_pert = np.column_stack([pert[:N], pert[N:2*N]])

            s = -res.fun
            if s > best_sum:

                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            pass
            
    # Extract final configuration
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()

    # 1. Enforce boundary constraints strictly
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    cr = np.maximum(cr, 0.0)
    
    # 2. Enforce non-overlap constraints iteratively with safety margin
    for _ in range(15):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                if cr[i] + cr[j] > d - 1e-9:
                    shrink = (cr[i] + cr[j] - d) / 2.0 + 1e-9
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    centers = np.column_stack([cx, cy])
    return centers, cr, float(np.sum(cr))
