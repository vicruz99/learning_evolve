import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.empty(4*N + NUM_PAIRS)
    
    # Boundary constraints
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def compute_feasible_radii(centers):
    """Compute strictly feasible radii based on local geometry."""
    r = np.full(N, 0.5)
    for i in range(N):
        mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                val = d / 2.0
                if val < mr:
                    mr = val
        r[i] = mr
    # Keep a safe margin to guarantee strict feasibility for the optimizer
    return r * 0.88

def generate_hex_config(spacing, margin=0.05):
    """Generates a hexagonal lattice pattern."""
    pts = []
    y = margin + spacing/2.0
    row = 0
    while len(pts) < N + 5:
        x_start = margin + spacing/2.0 + (row % 2) * spacing
        x = x_start
        while x <= 1.0 - margin - spacing/2.0 and len(pts) < N + 5:
            pts.append([x, y])
            x += spacing
        y += spacing * np.sqrt(3.0)/2.0
        row += 1
    pts = np.array(pts[:N])
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def force_relax(centers, iters=400):
    """Relax configuration using repulsive forces to spread circles evenly."""
    pts = centers.copy()
    for _ in range(iters):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i+1, N):
                diff = pts[i] - pts[j]
                d = np.linalg.norm(diff)
                if d < 0.25 and d > 1e-6:
                    f = (0.25 - d) / d * 0.5
                    forces[i] += f * diff
                    forces[j] -= f * diff
        pts += forces * 0.02
        pts = np.clip(pts, 0.05, 0.95)
    return pts

    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    inits = []

            if ang != 0.0:

                c_pts -= 0.5
                c_pts = c_pts @ np.array([[cos_a, -sin_a], [sin_a, cos_a]]) + 0.5
                c_pts = np.clip(c_pts, 0.02, 0.98)
            r = compute_feasible_radii(c_pts)
            inits.append(np.concatenate([c_pts[:, 0], c_pts[:, 1], r]))
            
    # 2. Force-relaxed random starts
    for seed in range(20):
        np.random.seed(seed + 50)
        c = np.random.uniform(0.15, 0.85, (N, 2))
        c = force_relax(c)
        r = compute_feasible_radii(c)
        inits.append(np.concatenate([c[:, 0], c[:, 1], r]))
        
    # 3. Boundary-pulled starts (circles near edges often allow larger radii)

        np.random.seed(seed + 200)
        c = np.random.uniform(0.05, 0.95, (N, 2))

            pass

    cy = best_v[N:2*N]

    # 1. Enforce boundary constraints strictly
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    cr = np.maximum(cr, 0.0)

                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    centers = np.column_stack([cx, cy])
    return centers, cr, float(np.sum(cr))
