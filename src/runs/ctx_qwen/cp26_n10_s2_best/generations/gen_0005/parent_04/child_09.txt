import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    
    return np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r,
        dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    ])

def get_max_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.5)
    cx, cy = centers[:, 0], centers[:, 1]
    for i in range(N):
        mr = min(cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
        for j in range(N):
            if i != j:
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                if d / 2.0 < mr:
                    mr = d / 2.0
        r[i] = mr
    # Scale down slightly to guarantee strict feasibility for the optimizer
    return r * 0.88

def hex_pack(rows, r0, angle=0.0, shift=(0.0, 0.0)):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0 + shift[1]
    for row_idx, count in enumerate(rows):
        x_start = r0 + shift[0] + (row_idx % 2) * r0
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r0
        y += r0 * np.sqrt(3.0)
        
    pts = np.array(pts[:N])
    if angle != 0.0:
        pts -= 0.5
        c, s = np.cos(angle), np.sin(angle)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts += 0.5
        
    return np.clip(pts, 0.01, 0.99)

def generate_inits():

                inits.append(np.concatenate([c[:, 0], c[:, 1], r]))

            c = np.clip(c, 0.02, 0.98)

            inits.append(np.concatenate([c[:, 0], c[:, 1], r]))

        r = get_max_radii(c)
        inits.append(np.concatenate([c[:, 0], c[:, 1], r]))

def run_packing():

    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
