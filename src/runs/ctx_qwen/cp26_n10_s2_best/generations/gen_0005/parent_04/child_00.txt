import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    r_sum = r[PAIR_I] + r[PAIR_J]
    c_pair = dx**2 + dy**2 - r_sum**2
    
    return np.concatenate([c, c_pair])

def get_feasible_radii(centers):
    """Compute strictly feasible radii for given centers."""
    r = np.full(N, 0.5)
    for i in range(N):
        mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                half_d = d * 0.5
                if half_d < mr:
                    mr = half_d
        r[i] = mr
    return r * 0.98

def repulsion_layout(seed, iters=400):
    """Generate well-spaced initial configuration using repulsive forces."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.15, 0.85, (N, 2))
    lr = 0.05
    target_d = 0.24
    for _ in range(iters):
        forces = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-8)
        np.fill_diagonal(dists, 1e9)
        
        mask = dists < target_d
        f_mag = np.zeros_like(dists)
        f_mag[mask] = (target_d - dists[mask]) / dists[mask]
        forces += np.sum(diff * f_mag[:,:,None], axis=1)
        
        pts = np.clip(pts + forces * lr, 0.04, 0.96)
        lr *= 0.995
    return pts

def generate_starts():

                    row += 1
                pts = np.array(pts[:N])

                    pts -= 0.5

                    pts = pts @ np.array([[c, -s], [s, c]]) + 0.5
                pts = np.clip(pts, 0.02, 0.98)
                starts.append(np.concatenate([pts[:,0], pts[:,1], get_feasible_radii(pts)]))

        pts = repulsion_layout(seed)
        starts.append(np.concatenate([pts[:,0], pts[:,1], get_feasible_radii(pts)]))

def run_packing():
    """Optimizes packing of 26 circles in a unit square to maximize sum of radii."""
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    # Phase 1: Multi-start exploration
    starts = generate_starts()
    for v0 in starts:
        try:

            s = -res.fun
            if s > best_sum:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass

        best_v = starts[0]

    # Enforce boundary constraints

            cr[i] = mr

    for _ in range(20):

        if not changed:
            break
            
    centers = np.column_stack([cx, cy])
    return centers, cr, float(np.sum(cr))
