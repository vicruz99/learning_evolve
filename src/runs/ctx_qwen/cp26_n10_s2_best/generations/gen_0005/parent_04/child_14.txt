import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r,
        (x[PAIR_I] - x[PAIR_J])**2 + (y[PAIR_I] - y[PAIR_J])**2 - (r[PAIR_I] + r[PAIR_J])**2
    ])
    return c

def compute_initial_radii(centers, scale=0.85):
    """Compute strictly feasible radii for given centers."""
    r = np.full(N, 0.5)
    r = np.minimum(r, centers[:, 0])
    r = np.minimum(r, 1.0 - centers[:, 0])
    r = np.minimum(r, centers[:, 1])
    r = np.minimum(r, 1.0 - centers[:, 1])
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    
    return r * scale

def generate_hex_centers(r0, angle=0.0, shift=(0.0, 0.0)):
    """Generate hexagonal lattice centers with rotation and shift."""
    pts = []
    y = r0 + shift[1]
    row = 0
    while len(pts) < N + 15:
        x_start = r0 + shift[0] + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 15:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
        row += 1
    
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        pts -= 0.5
        c, s = np.cos(angle), np.sin(angle)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts += 0.5
    
    return np.clip(pts, 0.01, 0.99)

def force_relax(pts, n_iter=500):
    """Force-directed relaxation with wall repulsion."""
    pts = pts.copy()
    lr = 0.04
    for it in range(n_iter):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(pts[i] - pts[j])
                if d < 0.28 and d > 1e-5:
                    f = (0.28 - d) / (d + 1e-10) * 0.35
                    diff = pts[i] - pts[j]
                    forces[i] += f * diff
                    forces[j] -= f * diff
        
        for i in range(N):
            if pts[i, 0] < 0.12:
                forces[i, 0] += 0.08
            if pts[i, 0] > 0.88:
                forces[i, 0] -= 0.08
            if pts[i, 1] < 0.12:
                forces[i, 1] += 0.08
            if pts[i, 1] > 0.88:
                forces[i, 1] -= 0.08
        
        pts += forces * lr
        pts = np.clip(pts, 0.02, 0.98)
        lr *= 0.997
    
    return pts
