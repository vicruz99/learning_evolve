import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints (squared for smooth gradients)
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist_sq = dx**2 + dy**2
    r_sum_sq = (r[PAIR_I] + r[PAIR_J])**2
    
    return np.concatenate([c, dist_sq - r_sum_sq])

def get_feasible_radii(centers):
    """Compute strictly feasible radii for given centers."""
    r = np.full(N, 0.5)
    for i in range(N):
        mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d/2.0 < mr:
                    mr = d/2.0
        r[i] = mr
    # Scale down to guarantee strict interior feasibility for SLSQP
    return r * 0.82

def hex_grid_init(r0):
    """Generate a hexagonal lattice configuration."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 + (row % 2) * r0
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
        row += 1
    return np.array(pts[:N])

def force_relax(centers, radii, steps=400):

    pts = centers.copy()
    r = radii.copy()
    dt = 0.03
    
    for _ in range(steps):

        forces[:, 1] += np.where(pts[:, 1] < r, 80.0*(r - pts[:, 1]), 0.0)
        forces[:, 1] -= np.where(pts[:, 1] > 1-r, 80.0*(pts[:, 1] - (1-r)), 0.0)
        
        # Update positions
        pts += forces * dt
        pts = np.clip(pts, 1e-6, 1.0-1e-6)
        
        # Grow radii uniformly, shrink if significant overlap occurs
        r += 0.00015
        r -= np.max(overlap, axis=1) * 0.4
        r = np.clip(r, 0.01, 0.3)
        
    return pts, r
