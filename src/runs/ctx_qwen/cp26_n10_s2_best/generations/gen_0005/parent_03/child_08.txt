import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)


def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])


def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]

    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r

    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - dr**2
    return c


def get_initial_radii(centers, scale=0.50):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            val = d / 2.0
            if val < r[i]:
                r[i] = val
            if val < r[j]:
                r[j] = val
    return np.clip(r * scale, 1e-6, 0.5)


def generate_hex_config(r0, angle, sx, sy):
    """Generate a hexagonal lattice configuration."""
    pts = []
    y = r0 + sy
    row = 0
    while len(pts) < N + 10:
        x_start = r0 + sx + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    pts = np.array(pts[:N])

    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5

    pts = np.clip(pts, 0.02, 0.98)
    return pts


def force_relax(centers, iters=150):
    """Relax configuration using repulsive forces."""
    pts = centers.copy()
    lr = 0.05
    for _ in range(iters):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[i] - pts[j]
                d = np.linalg.norm(diff)
                if d < 0.25 and d > 1e-6:
                    f = (0.25 - d) / d
                    forces[i] += f * diff
                    forces[j] -= f * diff
        pts += forces * lr
        pts = np.clip(pts, 0.03, 0.97)
        lr *= 0.99
    return pts


def generate_configs():
    """Generate diverse initial configurations."""
    configs = []

    # 1. Hexagonal lattices with various rotations and densities
    for r0 in np.linspace(0.07, 0.12, 11):
        pts = []
        y = r0
        row = 0
        while len(pts) < N + 10:
            xs = r0 if row % 2 == 0 else 2 * r0
            x = xs
            while x <= 1.0 - r0 and len(pts) < N + 10:
                pts.append([x, y])
                x += 2 * r0
            y += np.sqrt(3) * r0
            row += 1
        pts = np.array(pts[:N])
        for ang in [0, 0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.2, -0.2]:
            c, s = np.cos(ang), np.sin(ang)
            rot = pts - 0.5
            rot = rot @ np.array([[c, -s], [s, c]]) + 0.5
            valid = (rot[:, 0] >= 0.02) & (rot[:, 0] <= 0.98) & (rot[:, 1] >= 0.02) & (rot[:, 1] <= 0.98)
            if np.sum(valid) >= N:
                configs.append(rot[valid][:N])

    # 2. Grid configurations
    for s in np.linspace(0.15, 0.22, 10):
        pts = np.array([[i * s + 0.05, j * s + 0.05] for i in range(6) for j in range(5)])[:N]
        configs.append(pts)

    # 3. Force-relaxed random configurations
    for seed in range(30):
        np.random.seed(seed)
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        for _ in range(120):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    diff = pts[i] - pts[j]
                    d = np.linalg.norm(diff)
                    if d < 0.3 and d > 1e-6:
                        f = (0.3 - d) * 0.5 / d
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.05
            pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts)

    # 4. Staggered grid with jitter
    for seed in range(15):
        np.random.seed(seed + 5000)
        pts = []
        for i in range(6):
            for j in range(5):
                pts.append([0.08 + i * 0.16 + np.random.uniform(-0.02, 0.02),
                            0.08 + j * 0.19 + np.random.uniform(-0.02, 0.02)])
        if len(pts) >= N:
            configs.append(np.array(pts[:N]))

    # 5. Corner-optimized configurations
    for seed in range(10):
        np.random.seed(seed + 8000)
        pts = np.array([
            [0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85],
        ])
        rest = np.random.uniform(0.2, 0.8, (N - 4, 2))
        pts = np.vstack([pts, rest])
        pts = force_relax(pts, 100)
        configs.append(pts)

    return configs


def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    configs = generate_configs()
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}

    best_sum = -1.0
    best_v = None

    # Phase 1: Multi-start optimization
    for c_init in configs:
        r_init = get_initial_radii(c_init, 0.50)
        v0 = np.concatenate([c_init[:, 0], c_init[:, 1], r_init])
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 18000, 'ftol': 1e-14, 'disp': False})
            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-8:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass

    if best_v is None:
        return np.zeros((N, 2)), np.zeros(N), 0.0

    # Phase 2: Iterative perturbation & refinement
    current_v = best_v.copy()
    for step in range(50):
        np.random.seed(step + 3000)
        v_pert = current_v.copy()

        # Larger noise early, smaller later
        noise_scale = max(0.003, 0.010 - step * 0.00013)
        v_pert[:2 * N] += np.random.uniform(-noise_scale, noise_scale, 2 * N)
        v_pert[:2 * N] = np.clip(v_pert[:2 * N], 0.01, 0.99)

        # Progressively shrink radii to allow circles to rearrange
        shrink_factor = max(0.6, 0.88 - step * 0.003)
        v_pert[2 * N:] *= shrink_factor

        # Recompute feasible radii from perturbed centers
        c_pts = v_pert[:2 * N].reshape(N, 2)
        v_pert[2 * N:] = get_initial_radii(c_pts, 0.85)

        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-8:
                    best_sum = s
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            pass

    # Phase 3: Coarse perturbation for escaping deeper local minima
    for step in range(20):
        np.random.seed(step + 7000)
        v_pert = best_v.copy()

        # Larger jumps
        noise_scale = max(0.005, 0.015 - step * 0.0005)
        v_pert[:2 * N] += np.random.uniform(-noise_scale, noise_scale, 2 * N)
        v_pert[:2 * N] = np.clip(v_pert[:2 * N], 0.01, 0.99)

        # More aggressive shrinking
        v_pert[2 * N:] *= 0.70

        c_pts = v_pert[:2 * N].reshape(N, 2)
        v_pert[2 * N:] = get_initial_radii(c_pts, 0.80)

        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-8:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass

    # Phase 4: Fine-grained perturbation near best solution
    for step in range(30):
        np.random.seed(step + 10000)
        v_pert = best_v.copy()

        noise_scale = 0.003 * (1.0 - step / 30.0)
        v_pert[:2 * N] += np.random.uniform(-noise_scale, noise_scale, 2 * N)
        v_pert[:2 * N] = np.clip(v_pert[:2 * N], 0.01, 0.99)
        v_pert[2 * N:] *= 0.96

        c_pts = v_pert[:2 * N].reshape(N, 2)
        v_pert[2 * N:] = get_initial_radii(c_pts, 0.90)

        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-8:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass

    # Extract results
    centers = np.column_stack((best_v[:N], best_v[N:2 * N]))
    radii = best_v[2 * N:].copy()

    # Strict post-processing to guarantee validator compliance
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    radii = np.minimum(radii, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    radii = np.maximum(radii, 0.0)

    for _ in range(25):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break

    return centers, radii, float(np.sum(radii))
