import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    return -np.sum(v[2*N:])

def constraints(v):
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - dr**2
    return c

def get_feasible_radii(centers, scale=0.85):
    x = centers[:, 0]
    y = centers[:, 1]
    wall = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    r = np.minimum(wall, min_dists / 2.0)
    return np.clip(r * scale, 1e-6, 0.5)

def force_relax(pts, iters=200):
    pts = pts.copy()
    for _ in range(iters):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i+1, N):
                diff = pts[i] - pts[j]
                d = np.linalg.norm(diff)
                if d < 0.25 and d > 1e-5:
                    f = (0.25 - d) / (d + 0.001)
                    forces[i] += f * diff
                    forces[j] -= f * diff
        pts += forces * 0.02
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def generate_hex_lattice(r0, angle=0.0, shift=(0.0, 0.0)):
    pts = []
    y = r0 + shift[1]
    row = 0
    while len(pts) < N + 15:
        x_start = r0 + shift[0] + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 15:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    
    pts = np.array(pts[:N])
    
    if abs(angle) > 1e-8:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    
    return np.clip(pts, 0.02, 0.98)

def run_single_optimization(v0, bounds, cons, max_iter=8000, method='SLSQP'):
    try:
        res = minimize(objective, v0, method=method, bounds=bounds,
                       constraints=cons, 
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False})
        cv = constraints(res.x)
        if np.min(cv) >= -1e-7:
            return res.x.copy(), -res.fun
    except:
        pass
    return None, -1.0

def run_packing():

                    inits.append(pts)
    
    # Force-relaxed random starts

        pts = np.random.uniform(0.12, 0.88, (N, 2))
        pts = force_relax(pts, 200)
        inits.append(pts)

        ])
