import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: Minimize negative sum of radii (maximize sum)."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - dr**2
    
    return c

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    # Distance to square boundaries
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Distance to nearest neighbor
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    # Theoretical max radius for each center
    r_max = np.minimum(wall, min_dists / 2.0)
    
    # Scale down slightly to guarantee strict feasibility for optimizer start
    return np.clip(r_max * 0.92, 1e-5, 0.5)

def generate_hex_lattice(r0, shift_x, shift_y, angle):
    """Generate a hexagonal lattice configuration."""
    pts = []
    y = r0 + shift_y
    row = 0
    while len(pts) < N + 5:
        x_start = r0 + shift_x + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
        
    pts = np.array(pts[:N])
    
    if abs(angle) > 1e-5:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    return np.clip(pts, 0.01, 0.99)

def force_directed_init(seed, iters=150):
    """Generate a well-spread configuration using repulsive forces."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    lr = 0.06
    for _ in range(iters):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[i] - pts[j]
                d = np.linalg.norm(diff)
                if d < 0.24 and d > 1e-4:
                    f = (0.24 - d) / d * lr
                    forces[i] += f * diff
                    forces[j] -= f * diff
        pts += forces
        pts = np.clip(pts, 0.03, 0.97)
        lr *= 0.995
    return pts

def generate_initial_configs():
    """Generate diverse initial configurations."""
    configs = []
    
    # 1. Hexagonal lattices with systematic parameter sweep
    for r0 in np.linspace(0.075, 0.115, 8):
        for sx in np.linspace(-0.02, 0.02, 3):
            for sy in np.linspace(-0.02, 0.02, 3):
                for ang in np.linspace(-0.25, 0.25, 5):
                    configs.append(generate_hex_lattice(r0, sx, sy, ang))
                    
    # 2. Force-directed layouts
    for s in range(15):
        configs.append(force_directed_init(s))
        
    # 3. Perturbed grids

            pts = np.array([[0.08 + i*0.16 + sx, 0.08 + j*0.19 + sy]

            configs.append(pts)
            
    return configs

def run_packing():

    Returns (centers, radii, sum_radii).

    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None

                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})

            if curr_sum > best_sum:
                c_val = constraints(res.x)

                    best_v = res.x.copy()
        except Exception:
            continue

        r_fb = get_feasible_radii(cfg_fb)

                    shrink = (sum_r - d - 1e-11) / 2.0
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
