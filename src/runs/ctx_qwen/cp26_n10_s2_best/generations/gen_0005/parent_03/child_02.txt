import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - dr**2
    return c

def compute_initial_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    x, y = centers[:, 0], centers[:, 1]
    wall = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    r = np.minimum(wall, min_dists / 2.0)
    return np.clip(r * 0.85, 1e-5, 0.5)

def generate_hex_config(r0, angle, sx, sy):
    """Generates a rotated/shifted hexagonal lattice configuration."""
    pts = []
    y = r0 + sy
    row = 0
    while len(pts) < N + 10:
        x_start = r0 + sx + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        c_val, s_val = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5
    return np.clip(pts, 0.02, 0.98)

def generate_inits():
    """Generate diverse initial configurations."""
    inits = []
    
    # 1. Hexagonal lattices with varying densities, rotations, and shifts
    for r0 in np.linspace(0.085, 0.115, 5):
        for ang in np.linspace(-0.25, 0.25, 5):
            for sx in np.linspace(-0.02, 0.02, 3):
                for sy in np.linspace(-0.02, 0.02, 3):
                    inits.append(generate_hex_config(r0, ang, sx, sy))
                    
    # 2. Force-directed repelled random starts
    for seed in range(15):
        np.random.seed(seed)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(80):
            forces = np.zeros_like(pts)
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            np.fill_diagonal(dists, np.inf)
            mask = dists < 0.22
            f = np.where(mask, (0.22 - dists) / dists, 0.0)
            fvec = diff * f[:, :, None]
            forces = np.sum(fvec, axis=1)
            pts += forces * 0.05
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
    return inits

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    inits = generate_inits()
    
    # Phase 1: Multi-start optimization
    for centers in inits:
        r_init = compute_initial_radii(centers)
        v0 = np.concatenate([centers[:, 0], centers[:, 1], r_init])
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if -res.fun > best_sum:
                if np.min(constraints(res.x)) >= -1e-7:
                    best_sum = -res.fun
                    best_v = res.x.copy()
        except Exception:
            continue
            
    if best_v is None:
        centers_fallback = np.random.uniform(0.2, 0.8, (N, 2))
        radii_fallback = compute_initial_radii(centers_fallback)
        best_v = np.concatenate([centers_fallback[:, 0], centers_fallback[:, 1], radii_fallback])
        best_sum = -np.sum(radii_fallback)
        
    # Phase 2: Iterative refinement to escape local minima
    current_v = best_v.copy()
    for step in range(25):
        scale = 0.995 - step * 0.003
        v_pert = current_v.copy()
        v_pert[2*N:] *= max(0.75, scale)
        
        noise = np.random.uniform(-0.004, 0.004, 2*N)
        v_pert[:2*N] += noise
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.02, 0.98)
        
        v_pert[2*N:] = compute_initial_radii(v_pert[:2*N].reshape(N, 2)) * 0.9
        
        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if -res.fun > best_sum:
                if np.min(constraints(res.x)) >= -1e-7:
                    best_sum = -res.fun
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            continue
            
    # Extract final configuration
    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    radii = best_v[2*N:].copy()
    
    # Phase 3: Strict post-processing for validator compliance
    # 1. Enforce boundary constraints strictly
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    radii = np.minimum(radii, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    radii = np.maximum(radii, 0.0)
    
    # 2. Enforce non-overlap strictly with iterative shrinkage
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d - 1e-10:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-10
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
