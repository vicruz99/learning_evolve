1. **Linear Programming for Radii**: For any fixed set of centers, the problem of maximizing $\sum r_i$ subject to $r_i + r_j \le \|c_i - c_j\|$ and $r_i \le \text{boundary}(c_i)$ is a Linear Program. This can be solved exactly and efficiently using `scipy.optimize.linprog` (with the `highs` solver). This guarantees that for any center configuration, we get the mathematically optimal radii.
2. **Global Optimization for Centers**: With radii computed exactly via LP, the objective becomes a function of centers only: $f(C) = \sum r_i(C)$. This function is continuous but non-smooth. I will use `differential_evolution` for robust global exploration of the 52-dimensional center space, followed by `Nelder-Mead` local refinement. 
3. **Diverse Initializations**: I will seed the optimization with deterministic hexagonal and grid patterns to ensure the search starts near physically plausible dense packings, combined with DE's random exploration.
4. **Strict Post-processing**: After finding the optimal centers and radii, I will apply iterative shrinkage to guarantee compliance with the validator's tolerance margins.

import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

N = 26

def compute_lp(centers_flat):
    """Solves the LP to find optimal radii for fixed centers."""
    centers = centers_flat.reshape(N, 2)
    
    # Boundary constraints: r_i <= dist to wall
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # LP constraints: r_i + r_j <= dists[i,j]
    m = N * (N - 1) // 2
    A_ub = np.zeros((m, N))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = [(0.0, b) for b in ub]
    # Minimize -sum(r) <=> Maximize sum(r)
    res = linprog(c=-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def objective(centers_flat):
    """Objective for optimizers: minimize negative sum of radii."""
    s, _ = compute_lp(centers_flat)
    return -s

def run_packing():
    np.random.seed(42)
    bounds_c = [(0.01, 0.99)] * (2 * N)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Global search with Differential Evolution
    de_res = differential_evolution(
        objective, bounds=bounds_c, popsize=10, maxiter=150,
        mutation=(0.5, 1.5), recombination=0.8, seed=42,
        polish=False, disp=False
    )
    candidates = [de_res.x]
    
    # Phase 2: Deterministic starts (Hexagonal & Grid patterns)
    # Hexagonal lattices with varying densities
    for r0 in [0.085, 0.095, 0.105]:
        for shift in [0.0, 0.02]:
            pts = []
            y = r0 + shift
            row = 0
            while len(pts) < N:
                x_start = r0 if row % 2 == 0 else 2 * r0
                x = x_start
                while x <= 1.0 - r0 and len(pts) < N:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
                row += 1
            if len(pts) >= N:
                candidates.append(np.array(pts[:N]).flatten())
                
    # Rectangular grids with varying spacings
    for dx in np.linspace(0.12, 0.18, 4):
        for dy in np.linspace(0.12, 0.18, 4):
            pts = []
            y = dy / 2.0
            while len(pts) < N:
                x = dx / 2.0
                while x <= 1.0 - dx / 2.0 and len(pts) < N:
                    pts.append([x, y])
                    x += dx
                y += dy
            if len(pts) >= N:
                candidates.append(np.array(pts[:N]).flatten())

    # Phase 3: Local refinement using Nelder-Mead
    for cand in candidates:
        try:
            res = minimize(objective, cand, method='Nelder-Mead', 
                           bounds=bounds_c, 
                           options={'maxiter': 10000, 'xatol': 1e-7, 'fatol': 1e-9})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Compute final optimal radii for the best centers found
    _, best_radii = compute_lp(best_centers.flatten())
    
    # Phase 4: Strict post-processing for validator compliance
    # 1. Enforce boundary constraints strictly
    best_radii = np.minimum(best_radii, np.minimum(best_centers[:, 0], 1.0 - best_centers[:, 0]))
    best_radii = np.minimum(best_radii, np.minimum(best_centers[:, 1], 1.0 - best_centers[:, 1]))
    best_radii = np.maximum(best_radii, 0.0)
    
    # 2. Iterative shrinkage to guarantee non-overlap with safety margin
    for _ in range(15):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                if best_radii[i] + best_radii[j] > d - 1e-10:
                    shrink = (best_radii[i] + best_radii[j] - d) / 2.0 + 1e-10
                    best_radii[i] = max(0.0, best_radii[i] - shrink)
                    best_radii[j] = max(0.0, best_radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return best_centers, best_radii, float(np.sum(best_radii))
