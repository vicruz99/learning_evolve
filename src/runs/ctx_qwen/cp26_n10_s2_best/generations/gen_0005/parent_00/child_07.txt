The core difficulty in packing circles to maximize the sum of radii is the tight coupling between center positions and radii sizes in non-linear constraint solvers, which often leads to premature convergence in local minima. 

To overcome this, we decouple the problem:
1. **Linear Programming for Radii**: For any fixed set of circle centers, the problem of maximizing the sum of radii subject to non-overlap and boundary constraints is a Linear Program. Solving this LP exactly guarantees we extract the maximum possible sum of radii for that specific center configuration, removing the radius optimization bottleneck.
2. **Derivative-Free Center Optimization**: With radii computed optimally via LP, we define an objective function `f(centers) = -sum(optimal_radii)`. We then optimize the 52-dimensional center space using the `Powell` method, which is robust for non-smooth objective surfaces typical of LP value functions.
3. **Diverse Initialization & Escape**: We generate high-quality starting points using rotated/shifted hexagonal lattices (known to approach optimal packing densities) and force-relaxed random scatters. An iterative perturbation phase with decaying noise is applied to escape shallow local minima.
4. **Strict Post-Processing**: Finally, we apply boundary and overlap enforcement with a tight safety margin to guarantee 100% compliance with the validator's numerical tolerances.

This strategy mathematically separates the combinatorial geometry of centers from the radial sizing, leading to more stable and higher-quality packings.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_max_radii_lp(centers):
    """Computes optimal radii for fixed centers using Linear Programming."""
    # Boundary constraints: r_i <= dist to walls
    b = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    b = np.maximum(b, 1e-9)
    
    # Build LP constraints: r_i + r_j <= d_ij
    n_pair = N * (N - 1) // 2
    A_ub = np.zeros((N + n_pair, N))
    b_ub = np.zeros(N + n_pair)
    
    idx = 0
    for i in range(N):
        A_ub[idx, i] = 1.0
        b_ub[idx] = b[i]
        idx += 1
        
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    c_obj = -np.ones(N)
    # highs solver is extremely fast for this size
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * N, method='highs')
    if res.success:
        return res.x
    return b * 0.5

def objective_centers(x):
    """Objective function for center optimization: negative sum of optimal radii."""
    centers = x.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    r = compute_max_radii_lp(centers)
    return -np.sum(r)

def generate_initial_configs():
    """Generates diverse, high-quality initial center configurations."""
    configs = []
    
    # 1. Rotated & shifted hexagonal lattices
    for r0 in np.linspace(0.08, 0.115, 8):
        for angle in np.linspace(-0.25, 0.25, 7):
            for sx in np.linspace(-0.02, 0.02, 3):
                for sy in np.linspace(-0.02, 0.02, 3):
                    pts = []
                    y = r0 + sy
                    row = 0
                    while len(pts) < N + 5:
                        x_start = r0 + sx + (row % 2) * r0
                        x = x_start
                        while x <= 1.0 - r0 and len(pts) < N + 5:
                            pts.append([x, y])
                            x += 2.0 * r0
                        y += r0 * np.sqrt(3.0)
                        row += 1
                    pts = np.array(pts[:N])
                    if angle != 0.0:
                        c, s = np.cos(angle), np.sin(angle)
                        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                    pts = np.clip(pts, 0.01, 0.99)
                    configs.append(pts.flatten())
                    
    # 2. Force-relaxed random starts
    for seed in range(12):
        np.random.seed(seed)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(80):
            forces = np.zeros_like(pts)
            diff = pts[:, None] - pts[None, :]
            dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
            mask = dists < 0.25
            mag = np.zeros_like(dists)
            mag[mask] = (0.25 - dists[mask]) / dists[mask]
            vec = diff * mag[:, :, None]
            forces = np.sum(vec, axis=1)
            pts += forces * 0.02
            pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts.flatten())
        
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    configs = generate_initial_configs()
    
    best_sum = -1.0
    best_centers = None
    
    # Phase 1: Multi-start local optimization on centers using Powell's method
    for x0 in configs:
        try:
            res = minimize(objective_centers, x0, method='Powell',
                           options={'maxiter': 1500, 'xtol': 1e-6, 'ftol': 1e-8})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    if best_centers is None:
        best_centers = configs[0].reshape(N, 2)
        
    # Phase 2: Iterative perturbation to escape local minima
    curr_centers = best_centers.copy()
    for step in range(15):
        np.random.seed(step + 500)
        noise_scale = 0.004 * (1.0 - step / 15.0)
        x_pert = curr_centers.flatten() + np.random.uniform(-noise_scale, noise_scale, 2 * N)
        x_pert = np.clip(x_pert, 0.01, 0.99)
        
        try:
            res = minimize(objective_centers, x_pert, method='Powell',
                           options={'maxiter': 1200, 'xtol': 1e-6, 'ftol': 1e-8})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
                curr_centers = best_centers.copy()
        except Exception:
            pass
            
    # Compute final mathematically optimal radii for the best centers
    radii = compute_max_radii_lp(best_centers)
    
    # Phase 3: Strict post-processing for validator compliance
    # 1. Enforce boundary constraints strictly
    radii = np.minimum(radii, np.minimum(best_centers[:, 0], 1.0 - best_centers[:, 0]))
    radii = np.minimum(radii, np.minimum(best_centers[:, 1], 1.0 - best_centers[:, 1]))
    radii = np.maximum(radii, 0.0)
    
    # 2. Enforce non-overlap constraints iteratively with safety margin
    for _ in range(10):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                if radii[i] + radii[j] > d - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return best_centers, radii, float(np.sum(radii))
