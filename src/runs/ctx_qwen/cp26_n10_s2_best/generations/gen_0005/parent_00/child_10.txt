N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def constraints(v):
    """Compute inequality constraints: boundaries and non-overlap (linear dist form)."""
    x = v[:N]
    y = v[N: 2 * N]
    r = v[2 * N:]
    
    c = np.empty(4 * N + NUM_PAIRS)
    
    # Boundary constraints
    c[:N] = x - r
    c[N: 2 * N] = 1.0 - x - r
    c[2 * N: 3 * N] = y - r
    c[3 * N: 4 * N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist >= r_i + r_j (using actual distance for better gradient)
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist = np.sqrt(dx ** 2 + dy ** 2 + 1e-20)
    c[4 * N:] = dist - (r[PAIR_I] + r[PAIR_J])
    
    return c

def compute_feasible_radii(centers):
    """Compute strictly feasible radii from centers."""
    r = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    r = np.minimum(r, min_dists / 2.0)
    
    return np.clip(r * 0.85, 1e-4, 0.5)

            x = x_start + c * 2.0 * r0
            pts.append([x, y])

    pts = np.array(pts[:N])
