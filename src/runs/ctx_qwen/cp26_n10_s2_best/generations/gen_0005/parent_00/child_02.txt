import numpy as np
from scipy.optimize import minimize, differential_evolution

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)


def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])


def constraints_vec(v):
    """Compute inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    
    # Boundary constraints
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c


def get_max_radii_for_centers(centers):

    dy = centers[PAIR_I, 1] - centers[PAIR_J, 1]
    dists = np.hypot(dx, dy)
    
    # Each pair constrains r[i] + r[j] <= dist[i,j]

    # This is a linear program!

    # Variables: r[0], ..., r[N-1]

    c_obj = np.ones(N)

def solve_radii_for_fixed_centers(centers):

    from scipy.optimize import linprog
    
    c_obj = np.ones(N)
    A_ub = np.zeros((NUM_PAIRS, N))
    b_ub = dists.copy()
    for k in range(NUM_PAIRS):
        i, j = PAIR_I[k], PAIR_J[k]
        A_ub[k, i] = 1.0

def force_directed_layout(seed, target_dist=0.15, iters=500):
    """Generate a well-spread configuration using force-directed layout."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    
    for iteration in range(iters):
        forces = np.zeros_like(pts)
        lr = 0.02 * (1.0 - iteration / iters)
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[i, 0] - pts[j, 0]
                dy = pts[i, 1] - pts[j, 1]
                d = np.hypot(dx, dy)
                if d < target_dist and d > 1e-8:
                    f = (target_dist - d) / d
                    forces[i, 0] += f * dx
                    forces[i, 1] += f * dy
                    forces[j, 0] -= f * dx
                    forces[j, 1] -= f * dy
        
        # Boundary repulsion
        for i in range(N):
            if pts[i, 0] < 0.05:
                forces[i, 0] += 0.5
            if pts[i, 0] > 0.95:
                forces[i, 0] -= 0.5
            if pts[i, 1] < 0.05:
                forces[i, 1] += 0.5
            if pts[i, 1] > 0.95:
                forces[i, 1] -= 0.5
        
        pts += forces * lr
        pts = np.clip(pts, 0.02, 0.98)
    
    return pts


def generate_hex_lattice(r0, angle=0.0, shift_x=0.0, shift_y=0.0):
    """Generate hexagonal lattice pattern."""
    pts = []
    y = r0 + shift_y
    row = 0
    while len(pts) < N + 10:
        x_start = r0 + shift_x + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
        row += 1
    
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        pts = pts - 0.5
        pts = pts @ np.array([[cos_a, -sin_a], [sin_a, cos_a]]) + 0.5
    
    return np.clip(pts, 0.02, 0.98)

    dists = np.hypot(dx, dy)

        i, j = PAIR_I[k], PAIR_J[k]

    v0 = np.concatenate([centers[:, 0], centers[:, 1], r_init])
