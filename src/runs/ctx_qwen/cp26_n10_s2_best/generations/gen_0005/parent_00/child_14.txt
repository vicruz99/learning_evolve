import numpy as np
from scipy.optimize import minimize

N = 26

def compute_penalty_vec(v):
    """Vectorized quadratic penalty for boundary and overlap constraints."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    pen = 0.0
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    pen += np.sum(np.maximum(0.0, r - x)**2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - x))**2)
    pen += np.sum(np.maximum(0.0, r - y)**2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - y))**2)
    
    # Pairwise non-overlap: dist >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    
    r_sum = r[:, None] + r[None, :]
    overlap = np.maximum(0.0, r_sum - dist)
    pen += np.sum(overlap**2)
    return pen

def obj(v, lam):
    """Objective: minimize negative sum of radii + penalty."""
    return -np.sum(v[2*N:]) + lam * compute_penalty_vec(v)

bounds = [(0.01, 0.99)] * (2*N) + [(0.001, 0.5)] * N

    """Sequential penalty optimization with increasing lambda."""
    v = v0.copy()
    lambdas = [1.0, 5.0, 20.0, 100.0, 500.0, 2500.0, 10000.0]
    for lam in lambdas:
        res = minimize(obj, v, args=(lam,), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
        v = res.x
    return v
