

<strategy>
The previous approaches relied heavily on joint optimization of centers and radii using SLSQP, which struggles with the non-convex squared-distance constraints and often gets trapped in shallow local minima. 

A more robust strategy for circle packing is to decouple the problem: optimize centers first to maximize the sum of feasible radii, then refine jointly. Specifically:
1. Define a function `compute_sum_radii(centers)` that calculates the maximum possible radius for each circle given fixed centers (constrained by boundaries and pairwise distances). This transforms the constrained optimization problem into a bound-constrained, non-smooth objective maximization over centers only.
2. Use `Nelder-Mead` (a derivative-free optimizer robust to non-smoothness) to maximize this objective from many diverse starting configurations (hexagonal lattices with rotations/shifts, and force-relaxed random layouts). Clipping centers inside the objective function effectively enforces bounds.
3. Take the best center configuration and compute its optimal radii.
4. Perform a final joint refinement using `SLSQP` starting from this high-quality feasible point to polish the arrangement and squeeze out remaining slack.
5. Apply strict post-processing to guarantee validator compliance with appropriate safety margins.

This two-phase approach bypasses the constraint handling difficulties of direct joint optimization and typically finds significantly better packings for moderate N.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def compute_sum_radii(centers):
    """Compute the sum of maximum feasible radii for fixed centers."""
    r = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.sqrt(dx**2 + dy**2 + 1e-16)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    return np.sum(r)

def objective(v):
    """Objective for center optimization: maximize sum of radii."""
    centers = np.clip(v.reshape(N, 2), 1e-7, 1.0 - 1e-7)
    return -compute_sum_radii(centers)

def generate_starts():
    """Generate diverse initial center configurations."""
    starts = []
    
    # 1. Hexagonal lattices with various densities, rotations, and shifts
    for r0 in [0.085, 0.095, 0.105, 0.115]:
        for ang in np.linspace(-0.15, 0.15, 5):
            for sx in [-0.02, 0.0, 0.02]:
                for sy in [-0.02, 0.0, 0.02]:
                    pts = []
                    y = r0 + sy
                    row = 0
                    while len(pts) < N + 5:
                        x_start = r0 + sx + (row % 2) * r0
                        x = x_start
                        while x <= 1.0 - r0 and len(pts) < N + 5:
                            pts.append([x, y])
                            x += 2.0 * r0
                        y += r0 * np.sqrt(3.0)
                        row += 1
                    pts = np.array(pts[:N])
                    if ang != 0.0:
                        c, s = np.cos(ang), np.sin(ang)
                        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                    pts = np.clip(pts, 0.02, 0.98)
                    starts.append(pts.flatten())
                    
    # 2. Force-relaxed random configurations
    for seed in range(15):
        np.random.seed(seed)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(150):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    d = np.hypot(pts[i,0]-pts[j,0], pts[i,1]-pts[j,1])
                    if d < 0.25 and d > 1e-4:
                        f = (0.25 - d) / d
                        diff = pts[i] - pts[j]
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.05
            pts = np.clip(pts, 0.02, 0.98)
        starts.append(pts.flatten())
        
    return starts

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2 * N)
    best_val = -np.inf
    best_centers = None
    
    starts = generate_starts()
    
    # Phase 1: Optimize centers using Nelder-Mead
    for v0 in starts:
        try:
            res = minimize(objective, v0, method='Nelder-Mead', 
                           options={'maxiter': 40000, 'xatol': 1e-9, 'fatol': 1e-12})
            if -res.fun > best_val:
                best_val = -res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    if best_centers is None:
        best_centers = np.random.uniform(0.1, 0.9, (N, 2))
        
    best_centers = np.clip(best_centers, 1e-6, 1.0 - 1e-6)
    
    # Compute optimal radii for best centers
    radii = np.minimum(
        np.minimum(best_centers[:, 0], 1.0 - best_centers[:, 0]),
        np.minimum(best_centers[:, 1], 1.0 - best_centers[:, 1])
    )
    dx = best_centers[:, 0, None] - best_centers[:, 0]
    dy = best_centers[:, 1, None] - best_centers[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    radii = np.minimum(radii, np.min(dists, axis=1) / 2.0)
    
    # Phase 2: Joint refinement with SLSQP to polish the solution
    v0_full = np.concatenate([best_centers.flatten(), radii * 0.99])
    
    def obj_full(v):
        return -np.sum(v[2*N:])
        
    def cons_full(v):
        x = v[:N]
        y = v[N:2*N]
        r = v[2*N:]
        c = np.concatenate([
            x - r,
            1.0 - x - r,
            y - r,
            1.0 - y - r,
            (x[PAIR_I] - x[PAIR_J])**2 + (y[PAIR_I] - y[PAIR_J])**2 - (r[PAIR_I] + r[PAIR_J])**2
        ])
        return c
        
    try:
        res_full = minimize(obj_full, v0_full, method='SLSQP', 
                            bounds=[(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N,
                            constraints={'type': 'ineq', 'fun': cons_full},
                            options={'maxiter': 5000, 'ftol': 1e-13})
        if -res_full.fun > np.sum(radii):
            best_centers = res_full.x[:2*N].reshape(N, 2)
            radii = res_full.x[2*N:]
    except Exception:
        pass
        
    # Phase 3: Strict post-processing for validator compliance
    radii = np.minimum(radii, np.minimum(best_centers[:, 0], 1.0 - best_centers[:, 0]))
    radii = np.minimum(radii, np.minimum(best_centers[:, 1], 1.0 - best_centers[:, 1]))
    radii = np.maximum(radii, 0.0)
    
    for _ in range(15):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                if radii[i] + radii[j] > d - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return best_centers, radii, float(np.sum(radii))
```
