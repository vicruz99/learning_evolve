import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def compute_feasible_radii(centers, scale=0.85):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.25)
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    r = np.minimum(r, wall_dists)
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    return np.clip(r * scale, 0.005, 0.25)

def force_relax_init(seed, iters=300):
    """Generate a well-spread initial configuration using repulsive forces."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    target_d = 0.22
    lr = 0.02
    for _ in range(iters):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i+1, N):
                diff = pts[i] - pts[j]
                d = np.hypot(diff[0], diff[1])
                if d < target_d and d > 1e-6:
                    f = (target_d - d) / d
                    forces[i] += f * diff
                    forces[j] -= f * diff
        pts += forces * lr
        pts = np.clip(pts, 0.02, 0.98)
        lr *= 0.99
    return pts

def generate_inits():

    inits = []

    for r0 in np.linspace(0.08, 0.11, 6):

                            pts.append([x, y])
                            x += 2.0 * r0
                        y += np.sqrt(3) * r0
                        row += 1
                    pts = np.array(pts[:N])

                    pts = np.clip(pts, 0.02, 0.98)

                    inits.append(np.concatenate([pts[:, 0], pts[:, 1], r_init]))
                    
    # 2. Perturbed Grid patterns

        pts = []

        pts = np.array(pts[:N])
        r_init = compute_feasible_radii(pts)

        r_init = compute_feasible_radii(pts)

def run_packing():

                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
