import numpy as np
from scipy.optimize import minimize, differential_evolution

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints_direct(v):
    """Compute inequality constraints using direct distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist >= r_i + r_j
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist = np.sqrt(dx**2 + dy**2 + 1e-20)
    c[4*N:] = dist - (r[PAIR_I] + r[PAIR_J])
    
    return c

def constraints_squared(v):
    """Compute inequality constraints using squared distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def compute_initial_radii(centers, safety=0.88):
    """Compute strictly feasible initial radii."""
    r = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    r = np.minimum(r, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    
    return np.clip(r * safety, 0.001, 0.25)

def make_strictly_feasible(centers, radii):
    """Ensure the configuration is strictly feasible."""
    r = radii.copy()
    
    # Boundary constraints
    r = np.minimum(r, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    r = np.minimum(r, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Pairwise constraints - iterative adjustment
    for iteration in range(50):
        changed = False

            i, j = PAIR_I[k], PAIR_J[k]
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            sum_r = r[i] + r[j]
            if sum_r > d - 1e-10:
                shrink = (sum_r - d + 1e-10) / 2.0 + 1e-12
                r[i] = max(1e-8, r[i] - shrink)
                r[j] = max(1e-8, r[j] - shrink)
                changed = True
        if not changed:
            break
    
    return np.maximum(r, 0.0)

def generate_hex_lattice(r0, angle=0.0, shift_x=0.0, shift_y=0.0, n_target=None):
    """Generate hexagonal lattice points."""
    if n_target is None:
        n_target = N
    pts = []
    y = r0 + shift_y
    row = 0
    while len(pts) < n_target + 10:
        x_start = r0 + shift_x + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < n_target + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
        row += 1
    
    pts = np.array(pts[:n_target])
    
    if abs(angle) > 1e-8:
        c_val, s_val = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5
    
    return np.clip(pts, 0.01, 0.99)

def force_directed_layout(seed, n_iters=300):
    """Generate well-spread configuration using repulsive forces."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    
    for it in range(n_iters):
        forces = np.zeros_like(pts)
        lr = 0.03 * (1.0 - it / n_iters)
        
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[i] - pts[j]
                d = np.linalg.norm(diff)
                if d < 0.2 and d > 1e-6:
                    f = max(0.05, 0.15 / d**2)
                    forces[i] += f * diff / d
                    forces[j] -= f * diff / d
        
        # Boundary repulsion
        for i in range(N):
            if pts[i, 0] < 0.15:
                forces[i, 0] += 0.5
            if pts[i, 0] > 0.85:
                forces[i, 0] -= 0.5
            if pts[i, 1] < 0.15:
                forces[i, 1] += 0.5
            if pts[i, 1] > 0.85:
                forces[i, 1] -= 0.5
        
        pts += forces * lr
        pts = np.clip(pts, 0.05, 0.95)
    
    return pts

def optimize_from_init(v0, method='SLSQP', cons_func=constraints_direct, maxiter=20000):
    """Run optimization from a given initial point."""
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-8, 0.25)] * N
    
    if method == 'SLSQP':
        cons = {'type': 'ineq', 'fun': cons_func}
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False})
            return res
        except Exception:
            return None
    elif method == 'trust_constr':
        from scipy.optimize import NonlinearConstraint
        cons = NonlinearConstraint(cons_func, 0, np.inf)
        try:
            res = minimize(objective, v0, method='trust-constr', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter, 'verbose': 0})
            return res
        except Exception:
            return None
    return None
