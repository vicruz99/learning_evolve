

Looking at this problem, I notice we're consistently stuck around 2.630-2.636. The key insight is that equal circles for N=26 in a unit square have radius ~0.1004 (sum ≈ 2.61), so we're getting about 2.5% improvement with unequal circles. To push further, I need to:

1. Use structurally diverse starting configurations (especially corner-heavy arrangements)
2. Try non-uniform initial radii (large corner circles, small interior ones)
3. More aggressive perturbation to escape deep local minima
4. Multiple optimization methods

<strategy>
My strategy focuses on three improvements:
1. Better structural initial configurations - hexagonal patterns with various row distributions, force-directed layouts that naturally create space, and corner-heavy arrangements where corner circles can be larger
2. Non-uniform initial radii based on position (corners get larger radii) to break symmetry and guide optimizer toward unequal-circle solutions
3. More aggressive multi-phase optimization with varying perturbation scales to escape deep local minima
4. Post-optimization radius inflation phase to squeeze out every bit of sum
</strategy>

import numpy as np
from scipy.optimize import minimize
from scipy.optimize import dual_annealing

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def compute_feasible_radii(centers, factor=0.85):
    """Compute feasible initial radii based on local geometry."""
    r = np.full(N, 0.5)
    # Distance to boundaries
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    r = np.minimum(r, wall)
    # Distance to other centers
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    return np.clip(r * factor, 0.001, 0.5)

def compute_position_based_radii(centers, factor=0.85):
    """Compute radii that are larger for corner/edge positions."""
    r = compute_feasible_radii(centers, 1.0)
    # Boost corner/edge circles
    for i in range(N):
        x, y = centers[i]
        corner_dist = min(x, 1-x) + min(y, 1-y)
        if corner_dist < 0.3:
            r[i] = min(r[i] * 1.3, 0.45)
        elif corner_dist < 0.5:
            r[i] = min(r[i] * 1.1, 0.40)
    return np.clip(r * factor, 0.001, 0.5)

def generate_hex_config(r0, row_pattern, angle=0.0, sx=0.0, sy=0.0):
    """Generate hexagonal lattice with given row pattern."""
    pts = []
    y = r0 + sy
    for row_idx, count in enumerate(row_pattern):
        x_start = r0 + sx if row_idx % 2 == 0 else 2.0 * r0 + sx
        for k in range(count):
            if len(pts) >= N:
                break
            x = x_start + k * 2.0 * r0
            pts.append([x, y])
        y += r0 * np.sqrt(3.0)
    
    pts = np.array(pts[:N])
    
    if abs(angle) > 1e-8:
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[cos_a, -sin_a], [sin_a, cos_a]]) + 0.5
    
    return np.clip(pts, 0.02, 0.98)

def force_relax_init(seed, iterations=400):
    """Generate configuration using force-directed layout."""
    np.random.seed(seed)
    pts = np.random.uniform(0.12, 0.88, (N, 2))
    
    for it in range(iterations):
        forces = np.zeros_like(pts)
        lr = 0.04 * (1.0 - it / iterations)
        
        # Repulsion between circles
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[i] - pts[j]
                d = np.linalg.norm(diff)
                if d < 0.20 and d > 1e-8:
                    f = 0.04 / d
                    forces[i] += f * diff
                    forces[j] -= f * diff
        
        # Wall forces (keep away from edges)
        for i in range(N):
            forces[i, 0] += 0.15 * (pts[i, 0] - 0.08) if pts[i, 0] < 0.10 else (-0.15 if pts[i, 0] > 0.90 else 0)
            forces[i, 1] += 0.15 * (pts[i, 1] - 0.08) if pts[i, 1] < 0.10 else (-0.15 if pts[i, 1] > 0.90 else 0)
        
        pts += forces * lr
        pts = np.clip(pts, 0.04, 0.96)
    
    return pts

def corner_heavy_config(seed):
    """Generate a configuration with larger circles in corners."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    
    # Place 4 corner circles
    corner_positions = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]
    for i, pos in enumerate(corner_positions):
        pts[i] = [pos[0] + np.random.uniform(-0.03, 0.03), 
                  pos[1] + np.random.uniform(-0.03, 0.03)]
    
    # Place 8 edge circles
    edge_positions = [
        [0.35, 0.12], [0.65, 0.12],
        [0.35, 0.88], [0.65, 0.88],
        [0.12, 0.35], [0.12, 0.65],
        [0.88, 0.35], [0.88, 0.65]
    ]
    for i, pos in enumerate(edge_positions):
        pts[4 + i] = [pos[0] + np.random.uniform(-0.02, 0.02),
                      pos[1] + np.random.uniform(-0.02, 0.02)]
    
    # Fill remaining with scattered points
    remaining = N - 12
    center_pts = np.random.uniform(0.25, 0.75, (remaining, 2))
    pts[12:12 + remaining] = center_pts
    
    # Force relax a bit
    pts = force_relax_init.__wrapped__(seed + 1000, 100) if hasattr(force_relax_init, '__wrapped__') else pts
    
    return np.clip(pts, 0.05, 0.95)

def generate_configs():
    """Generate diverse initial configurations."""
    configs = []
    
    # 1. Hexagonal patterns with various row distributions
    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [7, 5, 5, 5, 4],
        [4, 6, 6, 6, 4], [6, 6, 6, 6, 2], [5, 5, 5, 5, 5, 1],
        [8, 6, 6, 6], [6, 7, 7, 6], [5, 6, 7, 5, 3]
    ]
    
    for r0 in [0.085, 0.090, 0.095, 0.100, 0.105]:
        for pat in row_patterns:
            if sum(pat) < N:
                continue
            base = generate_hex_config(r0, pat)
            configs.append(base)
            # Rotate variations
            for ang in [-0.15, -0.08, 0.0, 0.08, 0.15]:
                for sx in [-0.02, 0.0, 0.02]:
                    for sy in [-0.02, 0.0, 0.02]:
                        p = generate_hex_config(r0, pat, ang, sx, sy)
                        valid = np.all((p[:, 0] > 0.03) & (p[:, 0] < 0.97) & 
                                      (p[:, 1] > 0.03) & (p[:, 1] < 0.97))
                        if valid:
                            configs.append(p)
    
    # 2. Staggered grids
    for s in np.linspace(0.14, 0.18, 7):
        for offset in [0.0, 0.02, -0.02]:
            pts = []
            y = s + offset
            row = 0
            while len(pts) < N:
                x_start = s + (row % 2) * s / 2 + offset
                x = x_start
                while x < 1.0 - s and len(pts) < N:
                    pts.append([x, y])
                    x += s
                y += s * np.sqrt(3) * 0.5 + 0.02
                row += 1
            if len(pts) >= N:
                p = np.array(pts[:N]) + np.random.uniform(-0.005, 0.005, (N, 2))
                configs.append(np.clip(p, 0.03, 0.97))
    
    # 3. Force-directed layouts
    for seed in range(20):
        configs.append(force_relax_init(seed))
    
    # 4. Random starts with good spread
    for seed in range(15):
        np.random.seed(seed + 500)
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        # Simple repulsion pass
        for _ in range(200):
            for i in range(N):
                for j in range(i+1, N):
                    d = np.linalg.norm(pts[i] - pts[j])
                    if d < 0.15 and d > 1e-6:
                        diff = pts[i] - pts[j]
                        pts[i] += diff / d * 0.003
                        pts[j] -= diff / d * 0.003
        configs.append(np.clip(pts, 0.05, 0.95))
    
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    configs = generate_configs()
    best_sum = -1.0
    best_v = None
    
    # Phase 1: Multi-start optimization with position-based radii
    for c_init in configs:
        # Try both uniform and position-based initial radii
        for r_type in ['position', 'uniform']:
            if r_type == 'position':
                r_init = compute_position_based_radii(c_init, 0.85)
            else:
                r_init = compute_feasible_radii(c_init, 0.85)
            
            v0 = np.concatenate([c_init[:, 0], c_init[:, 1], r_init])
            
            try:
                res = minimize(objective, v0, method='SLSQP', bounds=bounds, 
                             constraints=cons,
                             options={'maxiter': 25000, 'ftol': 1e-13, 'disp': False})
                s = -res.fun
                if s > best_sum:
                    cv = constraints(res.x)
                    if np.min(cv) >= -1e-7:
                        best_sum = s
                        best_v = res.x.copy()
            except Exception:
                pass
    
    # Fallback
    if best_v is None:
        c_fb = force_relax_init(42)
        r_fb = compute_feasible_radii(c_fb, 0.80)
        best_v = np.concatenate([c_fb[:, 0], c_fb[:, 1], r_fb])
        best_sum = -np.sum(r_fb)
    
    # Phase 2: Iterative perturbation with varying strategies
    current_v = best_v.copy()
    
    for step in range(80):
        np.random.seed(step + 7000)
        v_pert = current_v.copy()
        
        # Vary perturbation strategy based on step
        if step < 30:
            # Small local search
            noise_scale = 0.002 * (1.0 - step / 30.0)
            v_pert[:2*N] += np.random.normal(0, noise_scale, 2*N)
            v_pert[2*N:] *= 0.97
        elif step < 50:
            # Medium perturbation
            noise_scale = 0.005
            v_pert[:2*N] += np.random.uniform(-noise_scale, noise_scale, 2*N)
            v_pert[2*N:] *= 0.93
        else:
            # Larger perturbation with position-based recovery
            noise_scale = 0.008
            v_pert[:2*N] += np.random.uniform(-noise_scale, noise_scale, 2*N)
            centers_pert = v_pert[:2*N].reshape(N, 2)
            centers_pert = np.clip(centers_pert, 0.02, 0.98)
            v_pert[:2*N] = centers_pert.ravel()
            v_pert[2*N:] = compute_position_based_radii(centers_pert, 0.80)
        
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.02, 0.98)
        
        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds,
                         constraints=cons,
                         options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                cv = constraints(res.x)
                if np.min(cv) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            pass
    
    # Phase 3: Try radius inflation from current best
    for inflate_step in range(10):
        v_inf = best_v.copy()
        # Slightly inflate radii and re-optimize
        v_inf[2*N:] *= 1.003
        v_inf[:2*N] += np.random.uniform(-0.001, 0.001, 2*N)
        v_inf[:2*N] = np.clip(v_inf[:2*N], 0.02, 0.98)
        
        try:
            res = minimize(objective, v_inf, method='SLSQP', bounds=bounds,
                         constraints=cons,
                         options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                cv = constraints(res.x)
                if np.min(cv) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
    
    # Extract final configuration
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()
    centers = np.column_stack((cx, cy))
    
    # Phase 4: Strict Post-Processing for Validator Compliance
    # 1. Enforce boundary constraints
    cr = np.minimum(cr, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    cr = np.minimum(cr, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    cr = np.maximum(cr, 0.0)
    
    # 2. Enforce non-overlap with minimal shrinkage
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                sum_r = cr[i] + cr[j]
                if sum_r > d + 1e-13:
                    shrink = (sum_r - d - 1e-13) / 2.0
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
    
    # 3. Try to recover some radius by careful redistribution
    for _ in range(5):
        for i in range(N):
            # Compute max possible radius for circle i
            max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                       centers[i, 1], 1.0 - centers[i, 1])
            for j in range(N):
                if i != j:
                    d = np.hypot(centers[i, 0] - centers[j, 0], 
                               centers[i, 1] - centers[j, 1])
                    max_r = min(max_r, d - cr[j])
            if max_r > cr[i] + 1e-12:
                cr[i] = max_r
    
    # Final overlap check and fix
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if cr[i] + cr[j] > d + 1e-13:
                    shrink = (cr[i] + cr[j] - d - 1e-13) / 2.0
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
    
    cr = np.maximum(cr, 0.0)
    return centers, cr, float(np.sum(cr))
