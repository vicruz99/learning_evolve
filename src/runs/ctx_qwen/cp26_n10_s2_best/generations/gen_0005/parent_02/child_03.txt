import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.empty(4*N + len(PAIR_I))
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    return np.clip(r * 0.92, 1e-4, 0.25)

def generate_hex_configs():
    """Generate hexagonal lattice patterns with various rotations and shifts."""
    configs = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 6, 4], [7, 5, 6, 5, 3], [4, 6, 6, 6, 4]
    ]
    for pat in patterns:
        if sum(pat) < N: 
            continue
        pts = []
        y = 0.1
        row_idx = 0
        for count in pat:
            x_start = 0.1 if row_idx % 2 == 0 else 0.15
            for k in range(count):
                if len(pts) >= N: 
                    break
                x = x_start + k * 0.16
                pts.append([x, y])
            y += 0.15 * np.sqrt(3.0)
            row_idx += 1
        base_pts = np.array(pts[:N])
        
        for ang in np.linspace(-0.25, 0.25, 11):
            for sx in np.linspace(-0.04, 0.04, 5):
                for sy in np.linspace(-0.04, 0.04, 5):
                    p = base_pts.copy()
                    c_val, s_val = np.cos(ang), np.sin(ang)
                    p = (p - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5
                    p[:, 0] += sx
                    p[:, 1] += sy
                    if np.all((p[:,0] > 0.02) & (p[:,0] < 0.98) & (p[:,1] > 0.02) & (p[:,1] < 0.98)):
                        configs.append(p.copy())
    return configs

def generate_random_configs():
    """Generate force-relaxed random configurations."""
    configs = []
    for seed in range(40):
        np.random.seed(seed)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(250):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    diff = pts[i] - pts[j]
                    d = np.linalg.norm(diff)
                    if d < 0.24 and d > 1e-5:
                        f = (0.24 - d) / d
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.04
            pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts)
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    all_configs = generate_hex_configs() + generate_random_configs()
    
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.3)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    # Phase 1: Multi-start optimization
    for cfg in all_configs:
        r_init = get_feasible_radii(cfg)
        v0 = np.concatenate([cfg[:, 0], cfg[:, 1], r_init])
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            cv = constraints(res.x)
            if s > best_sum and np.min(cv) >= -1e-7:
                best_sum = s
                best_v = res.x.copy()
        except Exception:
            pass
            
    # Fallback if all optimizations fail
    if best_v is None:
        cfg = all_configs[0]
        best_v = np.concatenate([cfg[:, 0], cfg[:, 1], get_feasible_radii(cfg)])
        best_sum = -np.sum(best_v[2*N:])
        
    # Phase 2: Iterative perturbation & refinement to escape local minima
    current_v = best_v.copy()
    for step in range(80):
        np.random.seed(step + 9999)
        v_pert = current_v.copy()
        
        # Adaptive noise scale (simulated annealing style)
        noise_scale = 0.008 * (1.0 - step/80.0)**0.5
        v_pert[:2*N] += np.random.normal(0, noise_scale, 2*N)
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.02, 0.98)
        
        # Adaptive shrink to create constraint slack for rearrangement
        shrink = 0.75 + 0.20 * (1.0 - step/80.0)
        v_pert[2*N:] *= shrink
        
        centers_p = v_pert[:2*N].reshape(N, 2)
        v_pert[2*N:] = get_feasible_radii(centers_p)
        
        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            cv = constraints(res.x)
            if s > best_sum and np.min(cv) >= -1e-7:
                best_sum = s
                best_v = res.x.copy()
                current_v = best_v.copy()
        except Exception:
            pass
            
    # Extract results
    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    radii = best_v[2*N:].copy()
    
    # Phase 3: Strict Post-Processing for Validator Compliance
    # 1. Enforce boundary constraints strictly
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    radii = np.minimum(radii, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    radii = np.maximum(radii, 0.0)
    
    # 2. Enforce non-overlap constraints iteratively with minimal safety margin
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d + 1e-12:
                    shrink = (radii[i] + radii[j] - d - 1e-12) / 2.0
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
