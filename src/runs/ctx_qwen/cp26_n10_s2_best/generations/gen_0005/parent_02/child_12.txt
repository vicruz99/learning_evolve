import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap distances."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def compute_safe_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]), 
                   np.minimum(centers[:,1], 1.0-centers[:,1]))
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    return np.clip(r * 0.82, 0.002, 0.22)

def force_relaxation(seed, iters=350):
    """Generate a well-spread configuration using repulsive forces."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    target_dist = 0.21
    
    for i in range(iters):
        forces = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        mask = dists < target_dist
        f_mag = np.zeros_like(dists)
        f_mag[mask] = (target_dist - dists[mask]) / dists[mask] * 0.6
        f_vec = diff * f_mag[:, :, None]
        forces = np.sum(f_vec, axis=1)
        
        # Strong wall repulsion
        forces[pts[:, 0] < 0.12, 0] += 3.0
        forces[pts[:, 0] > 0.88, 0] -= 3.0
        forces[pts[:, 1] < 0.12, 1] += 3.0
        forces[pts[:, 1] > 0.88, 1] -= 3.0
        
        step_size = 0.04 * (1.0 - i/iters)
        pts += forces * step_size
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def generate_hex_configs():
    """Generate diverse hexagonal lattice configurations with rotations/shifts."""
    configs = []
    for r0 in [0.092, 0.098, 0.104, 0.108]:
        for ang in np.linspace(-0.18, 0.18, 9):
            for sx in np.linspace(-0.025, 0.025, 5):
                for sy in np.linspace(-0.025, 0.025, 5):
                    pts = []
                    y = r0 + sy
                    row = 0
                    while len(pts) < N + 6:
                        x_start = r0 + sx + (row % 2) * r0
                        x = x_start
                        while x <= 1.0 - r0 and len(pts) < N + 6:
                            pts.append([x, y])
                            x += 2.0 * r0
                        y += r0 * np.sqrt(3.0)
                        row += 1
                    
                    p = np.array(pts[:N])
                    if abs(ang) > 1e-5:
                        c, s = np.cos(ang), np.sin(ang)
                        p = (p - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                    
                    p = np.clip(p, 0.02, 0.98)
                    if (np.all(p[:, 0] > 0.015) and np.all(p[:, 0] < 0.985) and 
                        np.all(p[:, 1] > 0.015) and np.all(p[:, 1] < 0.985)):
                        configs.append(p)
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.25)] * N
    cons = {'type': 'ineq', 'fun': constraints}

    best_sum = -1.0
    best_v = None

    # Collect initial configurations
    inits = []
    inits.extend(generate_hex_configs())
    for s in range(40):
        inits.append(force_relaxation(s, iters=300))

    # Phase 1: Multi-start optimization
    for c_init in inits:
        r_init = compute_safe_radii(c_init)
        v0 = np.concatenate([c_init[:,0], c_init[:,1], r_init])
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if -res.fun > best_sum:
                cv = constraints(res.x)
                if np.min(cv) >= -1e-8:
                    best_sum = -res.fun
                    best_v = res.x.copy()
        except Exception:
            pass

    # Fallback if optimization completely fails
    if best_v is None:
        c_f = np.random.uniform(0.1, 0.9, (N, 2))
        r_f = compute_safe_radii(c_f)
        best_v = np.concatenate([c_f[:,0], c_f[:,1], r_f])
        best_sum = -np.sum(r_f)

    # Phase 2: Iterative perturbation & refinement to escape local minima
    current_v = best_v.copy()
    for step in range(100):
        np.random.seed(step + 2000)
        v_p = current_v.copy()
        
        # Decaying noise schedule
        noise_scale = 0.0045 * (1.0 - step / 100.0)
        v_p[:2*N] += np.random.normal(0, noise_scale, 2*N)
        v_p[:2*N] = np.clip(v_p[:2*N], 0.02, 0.98)
        
        # Shrink radii to create slack for rearrangement
        v_p[2*N:] *= 0.965
        
        # Recompute strictly feasible radii for perturbed centers
        c_p = v_p[:2*N].reshape(N, 2)
        v_p[2*N:] = compute_safe_radii(c_p)

        try:
            res = minimize(objective, v_p, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            if -res.fun > best_sum:
                cv = constraints(res.x)
                if np.min(cv) >= -1e-8:
                    best_sum = -res.fun
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            pass

    # Extract final configuration
    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    radii = best_v[2*N:].copy()

    # Phase 3: Strict Post-Processing for Validator Compliance
    # 1. Enforce boundary constraints strictly
    radii = np.minimum(radii, np.minimum(centers[:,0], 1.0-centers[:,0]))
    radii = np.minimum(radii, np.minimum(centers[:,1], 1.0-centers[:,1]))
    radii = np.maximum(radii, 0.0)

    # 2. Enforce non-overlap constraints iteratively with minimal safety margin
    # Validator allows dist >= r1 + r2 - 1e-12. We use 1e-13 buffer.
    for _ in range(25):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if radii[i] + radii[j] > d + 1e-13:
                    shrink = (radii[i] + radii[j] - d - 1e-13) / 2.0
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break

    return centers, radii, float(np.sum(radii))
