import numpy as np
from scipy.optimize import minimize, basinhopping

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)


def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])


def constraints(v):
    """Compute inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N: 2 * N]
    r = v[2 * N:]
    c = np.empty(4 * N + len(PAIR_I))
    c[:N] = x - r
    c[N: 2 * N] = 1.0 - x - r
    c[2 * N: 3 * N] = y - r
    c[3 * N: 4 * N] = 1.0 - y - r
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4 * N:] = dx ** 2 + dy ** 2 - (r[PAIR_I] + r[PAIR_J]) ** 2
    return c


def get_initial_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1]),
    )
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, np.min(dists, axis=1) / 2.0)
    return np.clip(r * 0.75, 0.001, 0.25)


def generate_hex_centers(r0, angle=0.0, sx=0.0, sy=0.0):
    """Generate hexagonal lattice centers."""
    pts = []
    y = r0 + sy
    row = 0
    while len(pts) < N + 8:
        x_start = r0 + sx + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 8:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
        row += 1
    pts = np.array(pts[:N])
    if abs(angle) > 1e-8:
        c_val, s_val = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5
    pts = np.clip(pts, 0.02, 0.98)
    return pts


def force_relaxed_init(seed, iters=200):
    """Generate well-spread configuration using repulsion forces."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(iters):
        forces = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff ** 2, axis=2) + 1e-12)
        mask = (dists < 0.25) & (dists > 1e-5)
        f_mag = np.zeros_like(dists)
        f_mag[mask] = (0.25 - dists[mask]) / dists[mask]
        f_vec = diff * f_mag[:, :, np.newaxis]
        forces = np.sum(f_vec, axis=1)
        pts += forces * 0.03
        pts = np.clip(pts, 0.03, 0.97)
    return pts


def corner_packed_init(seed):
    """Place larger circles in corners and fill the rest."""
    np.random.seed(seed)
    # 4 corners + center + random fill
    base = np.array(
        [
            [0.12, 0.12],
            [0.88, 0.12],
            [0.12, 0.88],
            [0.88, 0.88],
            [0.50, 0.50],
        ]
    )
    rest = np.random.uniform(0.15, 0.85, (N - 5, 2))
    pts = np.vstack([base, rest])
    # Quick repulsion
    for _ in range(100):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(pts[i] - pts[j])
                if d < 0.20 and d > 1e-5:
                    f = (0.20 - d) / d
                    diff = pts[i] - pts[j]
                    forces[i] += f * diff
                    forces[j] -= f * diff
        pts += forces * 0.02
        pts = np.clip(pts, 0.05, 0.95)
    return pts


def staggered_grid_init(sx=0.0, sy=0.0, spacing=0.17):
    """Generate staggered grid configuration."""
    pts = []
    for row in range(6):
        for col in range(5):
            x = 0.08 + sx + col * spacing + (row % 2) * spacing * 0.5
            y = 0.08 + sy + row * spacing * 0.85
            if x < 0.95 and y < 0.95:
                pts.append([x, y])
    pts = np.array(pts[:N])
    return np.clip(pts, 0.03, 0.97)


def generate_configs():
    """Generate diverse initial configurations."""
    configs = []

    # 1. Hexagonal lattices with various parameters
    for r0 in np.linspace(0.080, 0.115, 12):
        for ang in np.linspace(-0.2, 0.2, 7):
            for sx in np.linspace(-0.03, 0.03, 3):
                for sy in np.linspace(-0.03, 0.03, 3):
                    try:
                        pts = generate_hex_centers(r0, ang, sx, sy)
                        configs.append(pts)
                    except Exception:
                        pass

    # 2. Force-relaxed random configurations
    for seed in range(30):
        configs.append(force_relaxed_init(seed, iters=250))

    # 3. Corner-packed configurations
    for seed in range(20):
        configs.append(corner_packed_init(seed))

    # 4. Staggered grids
    for sx in np.linspace(-0.03, 0.03, 5):

                try:

                except Exception:
                    pass

    # 5. Mixed row patterns
    row_patterns = [[6, 5, 6, 5, 4], [7, 5, 5, 5, 4], [5, 6, 6, 6, 3], [6, 6, 5, 5, 4], [4, 6, 6, 5, 5]]
    for pat in row_patterns:
        if sum(pat) < N:
            continue
        for r0 in [0.09, 0.095, 0.10, 0.105]:
            pts = []
            y = r0
            row_idx = 0
            for count in pat:
                x_start = r0 if row_idx % 2 == 0 else 2.0 * r0
                for k in range(count):
                    if len(pts) >= N:
                        break
                    x = x_start + k * 2.0 * r0
                    pts.append([x, y])
                y += r0 * np.sqrt(3.0)
                row_idx += 1
            pts = np.array(pts[:N])
            configs.append(np.clip(pts, 0.02, 0.98))

    return configs


def basinhopping_step(v):
    """Step function for basinhopping."""
    v_new = v.copy()
    # Perturb centers
    v_new[: 2 * N] += np.random.normal(0, 0.008, 2 * N)
    v_new[: 2 * N] = np.clip(v_new[: 2 * N], 0.01, 0.99)
    # Perturb radii multiplicatively
    v_new[2 * N:] *= np.exp(np.random.normal(0, 0.03, N))
    v_new[2 * N:] = np.clip(v_new[2 * N:], 1e-5, 0.3)
    return v_new


def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    configs = generate_configs()

    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.3)] * N
    cons = {"type": "ineq", "fun": constraints}

    best_sum = -1.0
    best_v = None

    # Phase 1: Multi-start SLSQP optimization
    for c_init in configs:
        r_init = get_initial_radii(c_init)
        v0 = np.concatenate([c_init[:, 0], c_init[:, 1], r_init])

        try:
            res = minimize(
                objective,
                v0,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 50000, "ftol": 1e-14, "disp": False},
            )
            s = -res.fun
            if s > best_sum:
                cv = constraints(res.x)
                if np.min(cv) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass

    # Fallback
    if best_v is None:
        c_fb = np.random.uniform(0.1, 0.9, (N, 2))
        r_fb = get_initial_radii(c_fb)
        best_v = np.concatenate([c_fb[:, 0], c_fb[:, 1], r_fb])
        best_sum = -np.sum(r_fb)

    # Phase 2: Basinhopping on the best solution to escape local minima
    minimizer_kwargs = {
        "method": "SLSQP",
        "bounds": bounds,
        "constraints": cons,
        "options": {"maxiter": 20000, "ftol": 1e-13, "disp": False},
    }

    try:
        rh = basinhopping(
            objective,
            best_v.copy(),
            minimizer_kwargs=minimizer_kwargs,
            niter=80,
            stepsize=0.008,
            step=basinhopping_step,
            interval=5,
            accept_test=None,
            callback=None,
        )
        s = -rh.fun
        cv = constraints(rh.x)
        if np.min(cv) >= -1e-7 and s > best_sum:
            best_sum = s
            best_v = rh.x.copy()
    except Exception:
        pass

    # Phase 3: Iterative perturbation and refinement
    current_v = best_v.copy()
    for step in range(80):
        np.random.seed(step + 10000)
        v_pert = current_v.copy()

        # Decaying noise
        noise_scale = 0.005 * np.exp(-step / 30.0)
        v_pert[: 2 * N] += np.random.normal(0, noise_scale, 2 * N)
        v_pert[: 2 * N] = np.clip(v_pert[: 2 * N], 0.015, 0.985)

                objective,
                v_pert,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 40000, "ftol": 1e-14, "disp": False},
            )
            s = -res.fun
            if s > best_sum:
                cv = constraints(res.x)
                if np.min(cv) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            pass

    # Phase 4: Second round of basinhopping from the improved solution
    try:
        rh2 = basinhopping(
            objective,
            best_v.copy(),
            minimizer_kwargs=minimizer_kwargs,
            niter=50,
            stepsize=0.004,
            step=basinhopping_step,
            interval=10,
        )
        s = -rh2.fun
        cv = constraints(rh2.x)
        if np.min(cv) >= -1e-7 and s > best_sum:
            best_sum = s
            best_v = rh2.x.copy()
    except Exception:
        pass

    # Extract results
    centers = np.column_stack((best_v[:N], best_v[N: 2 * N]))
    radii = best_v[2 * N:].copy()

    # Phase 5: Strict Post-Processing for Validator Compliance
    # 1. Enforce boundary constraints strictly
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    radii = np.minimum(radii, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))

    # 2. Enforce non-overlap constraints iteratively with minimal safety margin
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d + 1e-13:
                    shrink = (radii[i] + radii[j] - d - 1e-13) / 2.0
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break

    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
