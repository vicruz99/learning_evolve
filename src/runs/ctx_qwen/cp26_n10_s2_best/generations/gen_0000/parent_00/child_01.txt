                centers[idx, 1] = 0.1 + row * y_spacing

            idx += 1

def objective_penalty(params, n, alpha=1e6):
    """Objective with penalty for constraints - maximize sum of radii."""
    centers = params[:2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n:], 1e-10)
    
    sum_r = np.sum(radii)
    penalty = 0.0
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0
        v = x - r
        if v < 0:
            penalty += alpha * v * v
        # x + r <= 1
        v = 1.0 - x - r
        if v < 0:
            penalty += alpha * v * v
        # y - r >= 0
        v = y - r
        if v < 0:
            penalty += alpha * v * v
        # y + r <= 1
        v = 1.0 - y - r
        if v < 0:
            penalty += alpha * v * v
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += alpha * overlap * overlap
    
    return -(sum_r - penalty)


def boundary_constraints(params, n):
    """Boundary constraints: circles inside [0,1]x[0,1]."""

        c.append(1.0 - centers[i, 0] - radii[i])
        c.append(centers[i, 1] - radii[i])
        c.append(1.0 - centers[i, 1] - radii[i])
        c.append(radii[i])
    return np.array(c)


def overlap_constraints(params, n):
    """Non-overlap constraints."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    c = []
    for i in range(n):

            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)

    return np.array(c)

        jac[5 * i + 3, 2 * n + i] = -1.0

def overlap_con_func(params, n):
    """Overlap constraint values."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    c = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            c.append(dist - radii[i] - radii[j])
    return np.array(c)


def boundary_con_jac_func(params, n):
    """Jacobian of boundary constraints."""
    jac = np.zeros((5 * n, 3 * n))
    for i in range(n):
        jac[5 * i, 2 * i] = 1.0
        jac[5 * i, 2 * n + i] = -1.0
        jac[5 * i + 1, 2 * i] = -1.0
        jac[5 * i + 1, 2 * n + i] = -1.0
        jac[5 * i + 2, 2 * i + 1] = 1.0
        jac[5 * i + 2, 2 * n + i] = -1.0
        jac[5 * i + 3, 2 * i + 1] = -1.0
        jac[5 * i + 3, 2 * n + i] = -1.0
        jac[5 * i + 4, 2 * n + i] = 1.0
    return jac


def overlap_con_jac_func(params, n):
    """Jacobian of overlap constraints."""
    centers = params[:2 * n].reshape(n, 2)
    m = n * (n - 1) // 2
    jac = np.zeros((m, 3 * n))
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist > 1e-12:
                jac[k, 2 * i] = dx / dist
                jac[k, 2 * i + 1] = dy / dist
                jac[k, 2 * j] = -dx / dist
                jac[k, 2 * j + 1] = -dy / dist
            jac[k, 2 * n + i] = -1.0
            jac[k, 2 * n + j] = -1.0
            k += 1
    return jac


def check_validity(centers, radii, n):
    """Quick check if packing is valid."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0:
            return False
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True

    centers = np.zeros((n, 2))
    idx = 0
    n_rows = len(row_config)
    
    if n_rows <= 1:
        grid_size = int(math.ceil(math.sqrt(n)))
        x_spacing = 0.8 / (grid_size - 1) if grid_size > 1 else 1
        y_spacing = 0.8 / (grid_size - 1) if grid_size > 1 else 1

            for col in range(grid_size):
                if idx >= n:
                    break
                centers[idx, 0] = 0.1 + col * x_spacing
                centers[idx, 1] = 0.1 + row * y_spacing
                idx += 1
            if idx >= n:
                break
        return centers

def objective_penalty_wrapper(params, n, alpha):
    """Objective with penalty for constraints."""
    centers = params[:2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n:], 1e-10)
    
    sum_r = np.sum(radii)
    penalty = 0.0

        x, y = centers[i]
        r = radii[i]
        v = x - r
        if v < 0:
            penalty += alpha * v * v
        v = 1.0 - x - r
        if v < 0:
            penalty += alpha * v * v
        v = y - r
        if v < 0:
            penalty += alpha * v * v
        v = 1.0 - y - r
        if v < 0:

            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += alpha * overlap * overlap
    
    return -(sum_r - penalty)

    centers = params[:2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n:], 1e-10)
    
    sum_r = np.sum(radii)
    penalty = 0.0
    
    # Boundary penalties
    left_viol = np.minimum(0, centers[:, 0] - radii)
    right_viol = np.maximum(0, centers[:, 0] + radii - 1)
    bottom_viol = np.minimum(0, centers[:, 1] - radii)
    top_viol = np.maximum(0, centers[:, 1] + radii - 1)
    
    penalty += alpha * (np.sum(left_viol ** 2) + np.sum(right_viol ** 2) + 
                        np.sum(bottom_viol ** 2) + np.sum(top_viol ** 2))

    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
