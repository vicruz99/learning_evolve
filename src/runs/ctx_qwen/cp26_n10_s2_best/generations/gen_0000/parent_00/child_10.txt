import numpy as np
from scipy.optimize import minimize


def objective(x, n):
    """Minimize negative sum of radii (maximize sum of radii)."""
    radii = x[2 * n:]
    return -np.sum(radii)


def constraint_boundary_x_lower(x, args):
    """x_i >= r_i"""
    i, n = args
    return x[2 * i] - x[2 * n + i]


def constraint_boundary_x_upper(x, args):
    """1 - x_i >= r_i"""
    i, n = args
    return 1 - x[2 * i] - x[2 * n + i]


def constraint_boundary_y_lower(x, args):
    """y_i >= r_i"""
    i, n = args
    return x[2 * i + 1] - x[2 * n + i]


def constraint_boundary_y_upper(x, args):
    """1 - y_i >= r_i"""
    i, n = args
    return 1 - x[2 * i + 1] - x[2 * n + i]


def constraint_overlap(x, args):
    """dist(i,j) >= r_i + r_j"""
    i, j, n = args
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist_sq = dx * dx + dy * dy
    r_sum = x[2 * n + i] + x[2 * n + j]
    return dist_sq - r_sum * r_sum


def initialize_hexagonal(n):
    """Initialize centers in a hexagonal lattice pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    row_counts = [4, 5, 4, 5, 4, 4]
    
    for row in range(6):
        n_in_row = row_counts[row]
        for col in range(n_in_row):
            x = 0.12 + col * 0.17 + (0.085 if row % 2 == 1 else 0)
            y = 0.12 + row * 0.14
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
    
    return centers


def physics_optimize(centers, radii, n, iterations=30000):
    """Physics-based optimization: grow radii while resolving overlaps with repulsive forces."""
    for iteration in range(iterations):
        # Gradually grow radii
        growth = 2e-6 * (1 + 0.5 * np.exp(-iteration / 5000))
        radii += growth
        
        # Compute repulsive forces
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(max(dist_sq, 1e-20))
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    force_mag = overlap / dist * 200
                    forces[i, 0] -= force_mag * dx
                    forces[i, 1] -= force_mag * dy
                    forces[j, 0] += force_mag * dx
                    forces[j, 1] += force_mag * dy
            
            # Boundary repulsion
            for dim in range(2):
                if centers[i, dim] < radii[i]:
                    forces[i, dim] += 200 * (radii[i] - centers[i, dim])
                if centers[i, dim] > 1 - radii[i]:
                    forces[i, dim] -= 200 * (centers[i, dim] - (1 - radii[i]))
        
        # Update positions with damping
        damping = 0.003 / (1 + iteration * 0.0001)
        centers += damping * forces
        
        # Enforce constraints
        for i in range(n):
            radii[i] = max(radii[i], 1e-8)
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
    
    return centers, radii


def alternating_optimize(centers, radii, n, iterations=500):
    """Alternating optimization: fix centers->optimize radii, fix radii->optimize centers."""
    for it in range(iterations):
        # Update radii given centers (each circle grows to fill available space)
        for i in range(n):
            max_r = min(
                centers[i, 0],
                1 - centers[i, 0],
                centers[i, 1],
                1 - centers[i, 1]
            )
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    max_r = min(max_r, dist - radii[j])
            radii[i] = max(max_r, 1e-8)
        
        # Update centers given radii (gradient descent to reduce overlaps)
        lr = 0.005 / (1 + it * 0.002)
        for step in range(50):
            forces = np.zeros_like(centers)
            
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[j, 0] - centers[i, 0]
                    dy = centers[j, 1] - centers[i, 1]
                    dist = np.sqrt(max(dx * dx + dy * dy, 1e-20))
                    min_dist = radii[i] + radii[j]
                    
                    if dist < min_dist:
                        overlap = min_dist - dist
                        f = overlap / dist * 50
                        forces[i, 0] -= f * dx
                        forces[i, 1] -= f * dy
                        forces[j, 0] += f * dx
                        forces[j, 1] += f * dy
                
                for dim in range(2):
                    if centers[i, dim] < radii[i]:
                        forces[i, dim] += 50 * (radii[i] - centers[i, dim])
                    if centers[i, dim] > 1 - radii[i]:
                        forces[i, dim] -= 50 * (centers[i, dim] - (1 - radii[i]))
            
            centers += lr * forces
            
            for i in range(n):
                centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
                centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
    
    return centers, radii


def random_restart_optimize(n, best_centers, best_radii, n_restarts=5):
    """Try random restarts to escape local minima."""
    for restart in range(n_restarts):
        # Generate random initial configuration
        centers = np.random.uniform(0.1, 0.9, (n, 2))
        radii = np.ones(n) * 0.02
        
        # Physics optimization
        centers, radii = physics_optimize(centers, radii, n, iterations=15000)
        
        if np.sum(radii) > np.sum(best_radii):
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    return best_centers, best_radii


def build_constraints(n):
    """Build constraint dictionary list for scipy optimizer."""
    cons = []
    
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': constraint_boundary_x_lower, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': constraint_boundary_x_upper, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': constraint_boundary_y_lower, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': constraint_boundary_y_upper, 'args': (i, n)})
    
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': constraint_overlap, 'args': (i, j, n)})
    
    return cons


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    np.random.seed(123)
    n = 26
    
    # Step 1: Initialize with hexagonal lattice
    centers = initialize_hexagonal(n)
    radii = np.ones(n) * 0.02
    
    # Step 2: Physics-based pre-optimization
    centers, radii = physics_optimize(centers, radii, n, iterations=30000)
    
    # Step 3: Alternating optimization for refinement
    centers, radii = alternating_optimize(centers, radii, n, iterations=800)
    
    # Step 4: Try random restarts
    centers, radii = random_restart_optimize(n, centers, radii, n_restarts=8)
    
    # Step 5: SLSQP final refinement
    x0 = np.concatenate([centers.flatten(), radii])
    cons = build_constraints(n)
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    result = minimize(
        objective, x0, args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 8000, 'ftol': 1e-15, 'disp': False}
    )
    
    centers = result.x[:2 * n].reshape(n, 2)
    radii = result.x[2 * n:]
    
    # Step 6: Final alternating optimization after SLSQP
    centers, radii = alternating_optimize(centers, radii, n, iterations=300)
    
    # Fix numerical issues
    for i in range(n):
        radii[i] = max(radii[i], 1e-10)
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
    
    return centers, radii, float(np.sum(radii))
