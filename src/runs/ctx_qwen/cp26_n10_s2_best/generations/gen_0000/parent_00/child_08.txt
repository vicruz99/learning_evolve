import numpy as np
from scipy.optimize import minimize

def compute_overlap_loss(centers_flat, r, N):
    """Compute sum of squared overlaps for a given radius and center positions."""
    centers = centers_flat.reshape(N, 2)
    # Compute pairwise distance matrix efficiently
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Set diagonal to 2r so self-comparison yields 0 overlap
    np.fill_diagonal(dists, 2.0 * r)
    
    # Overlap is positive only when distance < 2r
    overlaps = np.maximum(0.0, 2.0 * r - dists)
    return np.sum(overlaps**2)
