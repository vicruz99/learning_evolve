6. **Validation**: Ensure the final solution passes the validation checks.

Key insight: For maximizing sum of radii with fixed number of circles, equal circles tend to be near-optimal (they minimize total area for given sum of radii by convexity), so the hexagonal lattice provides an excellent starting point. def init_hexagonal(n, r, seed=0):
    """Initialize circles in hexagonal lattice pattern."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    
    h = r * np.sqrt(3)  # vertical spacing
    w = 2 * r  # horizontal spacing
    
    idx = 0
    row = 0
    while idx < n:
        # Alternate between 6 and 5 columns
        num_cols = 6 if row % 2 == 0 else 5
        y = r + row * h
        x_start = r if row % 2 == 0 else r + w / 2
        for col in range(min(num_cols, n - idx)):
            x = x_start + col * w
            centers[idx] = [x, y]
            idx += 1
        row += 1
    
    # Small random perturbation to break symmetry
    if seed > 0:
        centers += np.random.uniform(-0.005, 0.005, size=centers.shape)
        # Clip to valid range
        centers[:, 0] = np.clip(centers[:, 0], r + 1e-6, 1 - r - 1e-6)
        centers[:, 1] = np.clip(centers[:, 1], r + 1e-6, 1 - r - 1e-6)
    
    return centers


def init_grid(n, seed=0):
    """Initialize circles in a grid pattern."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    
    # Try 5x5 grid with one extra
    rows = 5
    cols = 5
    spacing = 0.2
    
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx >= n:
                break
            x = 0.1 + j * spacing
            y = 0.1 + i * spacing
            centers[idx] = [x, y]
            idx += 1
    
    # Small perturbation
    if seed > 0:
        centers += np.random.uniform(-0.01, 0.01, size=centers.shape)
    
    return centers


def optimize_positions_and_radii(centers, radii, n):

    params = centers.flatten().copy()
    r = radii[0]
    
    def obj(p):
        return 0  # We'll just find feasible solution
    
    def constr(p):
        c = []
        cc = p.reshape(n, 2)
        for i in range(n):
            x, y = cc[i]
            c.append(x - r)
            c.append(1 - x - r)
            c.append(y - r)
            c.append(1 - y - r)
        for i in range(n):
            for j in range(i + 1, n):
                dx = cc[i, 0] - cc[j, 0]
                dy = cc[i, 1] - cc[j, 1]
                c.append(dx * dx + dy * dy - 4 * r * r)
        return np.array(c)
    
    cons = [{'type': 'ineq', 'fun': constr}]
    bounds = [(1e-6, 1 - 1e-6)] * (2 * n)
    
    try:
        result = minimize(obj, params, method='SLSQP', bounds=bounds,

            centers_opt = result.x.reshape(n, 2)
        else:
            centers_opt = centers.copy()
    except Exception:
        centers_opt = centers.copy()
    
    return centers_opt, radii


def optimize_full(centers, radii, n):
    """Full optimization of positions and radii."""
    params = np.zeros(3 * n)
    for i in range(n):
        params[3 * i] = centers[i, 0]
        params[3 * i + 1] = centers[i, 1]
        params[3 * i + 2] = radii[i]
    
    def obj(p):
        return -np.sum(p[2::3])
    
    def constr(p):
        c = []
        for i in range(n):
            x, y, r = p[3 * i], p[3 * i + 1], p[3 * i + 2]
            c.append(x - r)
            c.append(1 - x - r)
            c.append(y - r)
            c.append(1 - y - r)
            c.append(r - 1e-8)
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi, ri = p[3 * i], p[3 * i + 1], p[3 * i + 2]
                xj, yj, rj = p[3 * j], p[3 * j + 1], p[3 * j + 2]
                dx = xi - xj
                dy = yi - yj
                c.append(dx * dx + dy * dy - (ri + rj) * (ri + rj))
        return np.array(c)
    
    cons = [{'type': 'ineq', 'fun': constr}]
    bounds = [(0, 1)] * (2 * n) + [(1e-8, 0.5)] * n
    
    try:
        result = minimize(obj, params, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 3000, 'ftol': 1e-15})
        
        centers_opt = np.zeros((n, 2))
        radii_opt = np.zeros(n)
        for i in range(n):
            centers_opt[i] = [result.x[3 * i], result.x[3 * i + 1]]
            radii_opt[i] = result.x[3 * i + 2]
        return centers_opt, radii_opt
    except Exception:
        return centers, radii


def final_refine(centers, radii, n):
    """Iterative refinement: expand radii and adjust positions."""
    for iteration in range(20):
        # Find minimum slack
        min_slack = float('inf')
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                slack = dist - radii[i] - radii[j]
                if slack < min_slack:
                    min_slack = slack
        
        for i in range(n):
            slack = centers[i, 0] - radii[i]
            if slack < min_slack:
                min_slack = slack
            slack = 1 - centers[i, 0] - radii[i]
            if slack < min_slack:
                min_slack = slack
            slack = centers[i, 1] - radii[i]
            if slack < min_slack:
                min_slack = slack
            slack = 1 - centers[i, 1] - radii[i]
            if slack < min_slack:
                min_slack = slack
        
        if min_slack > 1e-10:
            # Scale up radii slightly
            scale = 1 + 0.5 * min_slack / (1 + np.mean(radii))
            radii *= min(scale, 1.02)
            # Re-optimize
            centers, radii = optimize_full(centers, radii, n)
        else:
            break
    
    return centers, radii


def validate_local(centers, radii):
    """Quick local validation."""
    n = centers.shape[0]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True
