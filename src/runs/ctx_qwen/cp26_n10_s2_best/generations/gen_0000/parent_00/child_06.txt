import numpy as np
from scipy.optimize import linprog
import math

def get_A_ub_matrix(n):
    """Constructs the constant coefficient matrix A_ub for the LP constraints."""
    # Constraints structure:
    # 1. Pairwise: r_i + r_j <= dist_ij  (Rows 0 to N_pairs-1)
    # 2. Boundary: r_i <= x_i, 1-x_i, y_i, 1-y_i (Rows N_pairs to end)
    
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    
    num_pairs = len(pairs)
    num_constraints = num_pairs + 4 * n
    A = np.zeros((num_constraints, n))
    
    # Fill pairwise constraint rows
    for k, (i, j) in enumerate(pairs):
        A[k, i] = 1.0
        A[k, j] = 1.0
        
    # Fill boundary constraint rows
    for i in range(n):
        base = num_pairs + 4 * i
        # r_i <= x_i
        A[base, i] = 1.0
        # r_i <= 1-x_i
        A[base + 1, i] = 1.0
        # r_i <= y_i
        A[base + 2, i] = 1.0
        # r_i <= 1-y_i
        A[base + 3, i] = 1.0
        
    return A, pairs

def solve_radii_lp(centers, A, pairs, n):
    """
    Solves the LP to find optimal radii for fixed centers.
    Returns (radii_array, sum_radii).
    """
    num_pairs = len(pairs)
    num_constraints = A.shape[0]
    b = np.zeros(num_constraints)
    
    # Compute pairwise distances for RHS
    # Optimized distance calculation
    # centers is (n, 2)
    # Using broadcasting for speed
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
    dists = np.sqrt(np.sum(diff**2, axis=2)) # (n, n)
    
    for k, (i, j) in enumerate(pairs):
        b[k] = dists[i, j]
        
    # Compute boundary distances for RHS
    for i in range(n):
        x, y = centers[i]
        base = num_pairs + 4 * i
        b[base] = x
        b[base + 1] = 1.0 - x
        b[base + 2] = y
        b[base + 3] = 1.0 - y
        
    # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    
    try:
        # 'highs' is generally faster and more robust
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
        else:
            # Fallback or return zeros if failed
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0

def compute_forces(centers, radii, pairs, n):
    """
    Computes forces on centers to relieve tight constraints.
    Returns force vectors.
    """
    forces = np.zeros((n, 2))
    
    # Pairwise repulsion for tight constraints
    # To optimize, we can compute distances once
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for k, (i, j) in enumerate(pairs):
        dist = dists[i, j]
        r_sum = radii[i] + radii[j]
        slack = dist - r_sum
        
        # If circles are touching or overlapping (numerically)
        if slack < 1e-5:
            # Repulsive force
            # Magnitude can be tuned. 
            # If dist is very small, force should be large to separate
            if dist > 1e-9:
                # Force proportional to how tight? Or constant?
                # Constant unit repulsion is stable
                mag = 1.0 
                # Direction vector from j to i
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                ux = dx / dist
                uy = dy / dist
                
                forces[i, 0] += mag * ux
                forces[i, 1] += mag * uy
                forces[j, 0] -= mag * ux
                forces[j, 1] -= mag * uy
    
    # Boundary forces
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Check proximity to boundaries
        # Left
        if x - r < 1e-5:
            forces[i, 0] += 1.5 * r # Push right
        # Right
        if (1.0 - x) - r < 1e-5:
            forces[i, 0] -= 1.5 * r # Push left
        # Bottom
        if y - r < 1e-5:
            forces[i, 1] += 1.5 * r # Push up
        # Top
        if (1.0 - y) - r < 1e-5:
            forces[i, 1] -= 1.5 * r # Push down
            
    return forces

def run_packing():
    n = 26
    A, pairs = get_A_ub_matrix(n)
    
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum = 0.0
    
    # Configuration parameters
    num_restarts = 8
    iterations = 150
    base_lr = 0.05
    
    # Pre-generate some good starting patterns
    # Pattern 1: Grid 5x5 + 1
    # Pattern 2: Hexagonal-ish
    # We will randomize slightly
    
    for restart in range(num_restarts):
        np.random.seed(restart * 100 + 7)
        
        # Initialize centers
        # Try a perturbed grid
        xs = np.linspace(0.1, 0.9, 5)
        ys = np.linspace(0.1, 0.9, 5)
        centers = []
        count = 0
        for y in ys:
            for x in xs:
                if count < n:
                    centers.append([x, y])
                    count += 1
        
        # If we have fewer than n (e.g. 25), add one in center or random
        while len(centers) < n:
            centers.append([np.random.rand(), np.random.rand()])
        
        centers = np.array(centers)
        
        # Initial repulsion to spread if needed (optional, but LP handles radius 0)
        # Just proceed to optimization
        
        current_centers = centers.copy()
        
        for step in range(iterations):
            # 1. Solve LP for radii
            radii, current_sum = solve_radii_lp(current_centers, A, pairs, n)
            
            if radii is None:
                break
            
            # Track best
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = current_centers.copy()
                best_radii = radii.copy()
            
            # 2. Compute forces
            forces = compute_forces(current_centers, radii, pairs, n)
            
            # 3. Update centers
            # Adaptive learning rate
            lr = base_lr / (1 + step * 0.05)
            current_centers += lr * forces
            
            # Clip to valid range (keep small margin to avoid numerical issues with boundaries)
            # However, LP handles boundary constraints strictly. 
            # We just need centers to be within [0,1] for distance calc.
            # Clipping helps prevent explosion.
            current_centers = np.clip(current_centers, 0.0001, 0.9999)
            
            # Optional: If sum of radii stops improving, we could jitter?
            # But force method usually converges.

    return best_centers, best_radii, best_sum
