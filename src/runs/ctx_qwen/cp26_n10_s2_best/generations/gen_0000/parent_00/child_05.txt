import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Strategy:
    1. Initialize circle centers using a hexagonal lattice pattern to ensure high density.
    2. Use scipy.optimize.minimize (SLSQP) to simultaneously optimize centers and radii.
    3. Maximize sum(radii) subject to boundary and non-overlap constraints.
    """
    n_circles = 26
    
    # --- 1. Initialization: Hexagonal Lattice ---
    # We generate a hexagonal grid and pick the first 26 points that fit in the square.
    # Hexagonal packing density is optimal for infinite planes.
    # Spacing 'd' chosen to fit roughly 26 circles. 
    # Estimate: Area ~ 1, density ~ 0.9. 26 * pi * r^2 ~ 0.9 => r ~ 0.1. 
    # Center spacing d ~ 2r ~ 0.2.
    
    points = []
    row = 0
    # Vertical spacing for hexagonal grid: sqrt(3)/2 * d
    d = 0.20 
    h = np.sqrt(3)/2 * d
    
    # Generate rows until we have enough points
    while len(points) < n_circles:
        y = d/2 + row * h
        if y + d/2 > 1.0: # Check if row fits vertically
            break
            
        # X-coordinates for this row
        # Alternate starting x by d/2 for hexagonal shift
        start_x = d/2 if row % 2 == 0 else d
        x = start_x
        while x + d/2 <= 1.0:
            points.append([x, y])
            x += d
        row += 1
        
    # Take exactly 26 points
    init_centers = np.array(points[:n_circles])
    
    # Initialize radii. 
    # A safe initial radius for the lattice is roughly half the spacing, 
    # but we let the optimizer find the best sizes. 
    # Starting slightly smaller ensures valid initial state.
    init_radii = np.full(n_circles, d/2 * 0.9)
    
    # Flatten variables for optimizer: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.hstack([init_centers.flatten(), init_radii])
    
    # --- 2. Objective Function ---
    def objective(vars_flat):
        # Maximize sum of radii => Minimize negative sum
        radii = vars_flat[2::3]
        return -np.sum(radii)

    # --- 3. Constraints ---
    def constraint_funcs(vars_flat):
        centers = np.zeros((n_circles, 2))
        radii = np.zeros(n_circles)
        
        for i in range(n_circles):
            centers[i, 0] = vars_flat[3*i]
            centers[i, 1] = vars_flat[3*i + 1]
            radii[i] = vars_flat[3*i + 2]
            
        constraints = []
        
        # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
        # Equivalent to: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
        for i in range(n_circles):
            x, y, r = vars_flat[3*i], vars_flat[3*i + 1], vars_flat[3*i + 2]
            constraints.append(x - r)
            constraints.append(1.0 - x - r)
            constraints.append(y - r)
            constraints.append(1.0 - y - r)
            
        # Non-overlap constraints: dist(i, j) >= r_i + r_j
        # We use a linearized approximation or just raw distance?
        # Raw distance is non-convex but SLSQP handles it.
        # dist^2 >= (r_i + r_j)^2 is safer? 
        # Let's stick to dist >= r_i + r_j. 
        # However, distance is non-differentiable at 0 (unlikely here).
        # Better: dist - (r_i + r_j) >= 0.
        
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii[i] + radii[j]))
                
        return constraints

    # --- 4. Optimization Setup ---
    # We define bounds for variables.
    # x, y in [0, 1], r >= 0.
    bounds = []
    for _ in range(n_circles):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r (max radius 0.5)

    # Define constraints for SLSQP
    # SLSQP expects constraints as a list of dicts
    # But defining a function that returns values and jacobian is more efficient
    # However, scipy's `NonlinearConstraint` is easier for modern versions, 
    # but `minimize` with `constraints` list is standard.
    
    # Let's implement a custom constraint function that returns values and jacobian
    # to help convergence.
    
    def constraints_vals(vars_flat):
        vals = []
        # Boundary constraints
        for i in range(n_circles):
            x, y, r = vars_flat[3*i], vars_flat[3*i + 1], vars_flat[3*i + 2]
            vals.append(x - r)
            vals.append(1.0 - x - r)
            vals.append(y - r)
            vals.append(1.0 - y - r)
        
        # Overlap constraints
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                idx_i = 3*i
                idx_j = 3*j
                xi, yi = vars_flat[idx_i], vars_flat[idx_i+1]
                xj, yj = vars_flat[idx_j], vars_flat[idx_j+1]
                ri, rj = vars_flat[idx_i+2], vars_flat[idx_j+2]
                
                dx = xi - xj
                dy = yi - yj
                dist = np.sqrt(dx**2 + dy**2)
                # Avoid division by zero in gradient
                if dist < 1e-9:
                    dist = 1e-9
                    
                vals.append(dist - (ri + rj))
        return np.array(vals)

    def constraints_jac(vars_flat):
        # Number of constraints
        n_bound = 4 * n_circles
        n_overlap = n_circles * (n_circles - 1) // 2
        n_constraints = n_bound + n_overlap
        n_vars = 3 * n_circles
        jac = np.zeros((n_constraints, n_vars))
        
        idx_c = 0
        
        # Boundary Jacobians
        for i in range(n_circles):
            base = 3*i
            # x - r >= 0 -> [0..0, 1, 0..0, -1] at base
            jac[idx_c, base] = 1
            jac[idx_c, base+2] = -1
            idx_c += 1
            
            # 1 - x - r >= 0 -> [0..0, -1, 0..0, -1]
            jac[idx_c, base] = -1
            jac[idx_c, base+2] = -1
            idx_c += 1
            
            # y - r >= 0
            jac[idx_c, base+1] = 1
            jac[idx_c, base+2] = -1
            idx_c += 1
            
            # 1 - y - r >= 0
            jac[idx_c, base+1] = -1
            jac[idx_c, base+2] = -1
            idx_c += 1
            
        # Overlap Jacobians
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                idx_i = 3*i
                idx_j = 3*j
                
                xi, yi = vars_flat[idx_i], vars_flat[idx_i+1]
                xj, yj = vars_flat[idx_j], vars_flat[idx_j+1]
                ri, rj = vars_flat[idx_i+2], vars_flat[idx_j+2]
                
                dx = xi - xj
                dy = yi - yj
                dist = np.sqrt(dx**2 + dy**2)
                if dist < 1e-9: dist = 1e-9
                
                # Constraint: dist - (ri + rj) >= 0
                # Gradient w.r.t xi: dx/dist
                # Gradient w.r.t yi: dy/dist
                # Gradient w.r.t xj: -dx/dist
                # Gradient w.r.t yj: -dy/dist
                # Gradient w.r.t ri: -1
                # Gradient w.r.t rj: -1
                
                jac[idx_c, idx_i] = dx / dist
                jac[idx_c, idx_i+1] = dy / dist
                jac[idx_c, idx_j] = -dx / dist
                jac[idx_c, idx_j+1] = -dy / dist
                jac[idx_c, idx_i+2] = -1
                jac[idx_c, idx_j+2] = -1
                idx_c += 1
                
        return jac

    # Define constraint dict for minimize
    cons = {
        'type': 'ineq',
        'fun': constraints_vals,
        'jac': constraints_jac
    }
    
    # --- 5. Run Optimization ---
    # We might need to run this a few times or with different seeds, 
    # but given the hex init, it should work.
    # SLSQP is a good choice for constrained nonlinear optimization.
    
    try:
        result = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=[cons],
            options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
        )
        
        if result.success:
            best_vars = result.x
        else:
            # Fallback if optimization fails, return initial
            best_vars = x0
            
    except Exception:
        best_vars = x0

    # Extract results
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    for i in range(n_circles):
        centers[i, 0] = best_vars[3*i]
        centers[i, 1] = best_vars[3*i + 1]
        radii[i] = best_vars[3*i + 2]
        
    sum_radii = np.sum(radii)
    
    # Clip radii to be non-negative just in case
    radii = np.maximum(radii, 0)
    
    return centers, radii, sum_radii
