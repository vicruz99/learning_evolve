import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers_flat, r_target, n):

    Computes the sum of squared overlap violations for circles of radius r_target.
    """
    centers = centers_flat.reshape(n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]

    np.fill_diagonal(dists, np.inf)
    violations = np.maximum(0, 2 * r_target - dists)
    return np.sum(violations**2)

    for k, count in enumerate(row_counts):

        x_start = (1.0 - width) / 2.0 + r_target
        for j in range(count):
            x = x_start + j * 2 * r_target
            # Offset odd rows horizontally by r_target to form hexagonal packing
            if k % 2 == 1:
                x += r_target
            centers.append([x, y])
        y += r_target * np.sqrt(3)
        
    centers = np.array(centers)
