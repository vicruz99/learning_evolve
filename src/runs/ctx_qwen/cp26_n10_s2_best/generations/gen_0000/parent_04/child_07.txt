import numpy as np
import scipy.optimize as opt

def generate_hexagonal_initial_guess(n):
    """
    Generates an initial guess for circle centers using a hexagonal lattice.
    """
    # Estimate spacing for n circles in unit square
    # Area of hexagon per circle ~ sqrt(3)/2 * s^2
    # n * area ~ 1 => s ~ sqrt(2 / (n * sqrt(3)))
    s = np.sqrt(2 / (n * np.sqrt(3)))
    
    # We want to fill the square. Let's generate a grid of points
    # and pick the best n.
    points = []
    # Max rows needed
    max_rows = int(2 / (s * np.sqrt(3)/2)) + 2
    
    for row in range(max_rows):
        y = row * s * np.sqrt(3) / 2
        # Offset for odd rows
        x_offset = (s / 2) if row % 2 == 1 else 0
        
        # Max cols needed
        max_cols = int(1 / s) + 2
        for col in range(max_cols):
            x = col * s + x_offset
            # Check if point is reasonably inside (with some margin for radius)
            # Radius approx s/2
            r_approx = s / 2
            if r_approx <= x <= 1 - r_approx and r_approx <= y <= 1 - r_approx:
                points.append([x, y])
    
    # If we have more than n points, select n points that are most spread out?
    # Or just take the first n. For optimization, taking a subset is fine.
    # However, taking first n might leave gaps.
    # Let's just take the first n valid points generated.
    # To make it robust, let's just create a fixed grid and truncate.
    
    centers = np.array(points[:n])
    if len(points) < n:
        # Fallback to random if grid failed (unlikely for n=26)
        np.random.seed(42)
        centers = np.random.rand(n, 2) * 0.8 + 0.1
        
    # Initial radii small to ensure feasibility
    radii = np.full(n, 0.01)
    
    return centers, radii

def get_constraints(centers, radii, n):
    """
    Generates the constraints for the optimization problem.
    """
    constraints = []
    
    # Boundary constraints
    # x >= r, x <= 1-r => x - r >= 0, 1 - x - r >= 0
    # y >= r, y <= 1-r => y - r >= 0, 1 - y - r >= 0
    # In scipy, constraints must be >= 0.
    
    for i in range(n):
        # x_i >= r_i
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
        # 1 - x_i >= r_i
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]})
        # y_i >= r_i
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
        # 1 - y_i >= r_i
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]})
        
    # Non-overlap constraints
    # dist(i, j) >= r_i + r_j
    # sqrt((xi-xj)^2 + (yi-yj)^2) - ri - rj >= 0
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: np.sqrt((v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2) - v[3*i+2] - v[3*j+2]
            })
            
    return constraints

def run_packing():
    n = 26
    
    # Objective function to minimize: -sum(radii)
    def objective(v):
        # v is flattened [x0, y0, r0, x1, y1, r1, ...]
        radii = v[2::3]
        return -np.sum(radii)

    # Bounds: x, y in [0, 1], r >= 0
    # Actually r is constrained by geometry, but lower bound 0 is safe.
    # Upper bound for r is 0.5.
    bounds = []
    for i in range(n):
        bounds.extend([
            (0.0, 1.0), # x
            (0.0, 1.0), # y
            (0.0, 0.5)  # r
        ])

    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None

    # Try multiple restarts to avoid local minima
    # Restart 1: Hexagonal grid
    centers, radii = generate_hexagonal_initial_guess(n)
    v0 = np.array([centers[i][0], centers[i][1], radii[i] for i in range(n)])
    
    # Restart 2: Perturbed grid
    centers2 = centers + np.random.normal(0, 0.01, centers.shape)
    centers2 = np.clip(centers2, 0.05, 0.95)
    v1 = np.array([centers2[i][0], centers2[i][1], radii[i] for i in range(n)])

    # Restart 3: Random points (with small radii)
    np.random.seed(123)

    v2 = np.array([centers3[i][0], centers3[i][1], radii3[i] for i in range(n)])
    
    initial_guesses = [v0, v1, v2]
    
    constraints = get_constraints(centers, radii, n)

    for i, v_init in enumerate(initial_guesses):
        try:
            # Use SLSQP which supports bounds and constraints
            res = opt.minimize(
                objective, 
                v_init, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9}
            )
            
            current_sum = -res.fun
            if current_sum > best_sum_radii:
                best_sum_radii = current_sum

            print(f"Optimization failed for restart {i}: {e}")
            continue

    if best_centers is None:
        # Fallback if optimization failed completely (unlikely)
        best_centers, best_radii = generate_hexagonal_initial_guess(n)
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, best_sum_radii
