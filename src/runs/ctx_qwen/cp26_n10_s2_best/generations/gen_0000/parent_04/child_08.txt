import numpy as np
from scipy.optimize import minimize, linprog
from functools import partial


def compute_all_constraints(x, n, radii):
    """
    Compute boundary and non-overlap constraints.
    Returns array where each element >= 0 for feasibility.
    """
    pos = x.reshape(n, 2)
    constraints = []
    
    for i in range(n):
        xi, yi = pos[i]
        ri = radii[i]
        
        constraints.append(xi - ri)
        constraints.append(1.0 - xi - ri)
        constraints.append(yi - ri)
        constraints.append(1.0 - yi - ri)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = pos[i, 0] - pos[j, 0]
            dy = pos[i, 1] - pos[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            constraints.append(dist - min_dist)
    
    return np.array(constraints)


def dummy_objective(x):
    return 0.0


def init_hexagonal(n):
    """Initialize circles in a hexagonal-like pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    
    row_counts = [5, 5, 5, 5, 6]
    n_rows = len(row_counts)
    y_spacing = 0.8 / (n_rows - 1)
    
    for row_idx, count in enumerate(row_counts):
        if idx >= n:
            break
        y = 0.1 + row_idx * y_spacing
        x_spacing = 0.8 / (count - 1) if count > 1 else 0
        x_start = 0.1 + (row_idx % 2) * x_spacing / 2
        
        for col_idx in range(count):
            if idx >= n:
                break
            x = x_start + col_idx * x_spacing
            centers[idx] = [x, y]
            idx += 1
    
    return centers[:n]


def force_directed_init(centers, n, radii, num_iters=3000):
    """Force-directed simulation to improve initial configuration."""
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(num_iters):
        forces = np.zeros_like(centers)
        
        # Pairwise repulsive forces
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist > 1e-12:
                    min_dist = radii[i] + radii[j]
                    if dist < min_dist:
                        force_magnitude = (min_dist - dist) ** 2 * 500
                        unit_vec = diff / dist
                        forces[i] += force_magnitude * unit_vec
                        forces[j] -= force_magnitude * unit_vec
        
        # Boundary forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            if x - r < 0:
                forces[i, 0] += (r - x) ** 2 * 500
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) ** 2 * 500
            if y - r < 0:
                forces[i, 1] += (r - y) ** 2 * 500
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) ** 2 * 500
        
        dt = 0.008 / (1.0 + iteration * 0.001)
        centers = centers + dt * forces
        centers = np.clip(centers, 1e-8, 1.0 - 1e-8)
        
        # Gradually grow radii
        if iteration % 50 == 0 and iteration > 0:
            min_clearance = 1.0
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.linalg.norm(centers[i] - centers[j])
                    clearance = dist - radii[i] - radii[j]
                    if clearance < min_clearance:
                        min_clearance = clearance
                
                x, y = centers[i]
                r = radii[i]
                for val in [x - r, 1.0 - x - r, y - r, 1.0 - y - r]:
                    if val < min_clearance:
                        min_clearance = val
            
            if min_clearance > 1e-8:
                radii = radii + min_clearance * 0.3
    
    return centers, radii


def solve_lp_for_radii(centers, n):
    """Given fixed centers, solve LP to maximize sum of radii."""
    c = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        xi, yi = centers[i]
        
        row = np.zeros(n)
        row[i] = -1.0
        A_ub.append(row)
        b_ub.append(xi)
        
        row = np.zeros(n)
        row[i] = -1.0
        A_ub.append(row)
        b_ub.append(1.0 - xi)
        
        row = np.zeros(n)
        row[i] = -1.0
        A_ub.append(row)
        b_ub.append(yi)
        
        row = np.zeros(n)
        row[i] = -1.0
        A_ub.append(row)
        b_ub.append(1.0 - yi)
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
    
    bounds = [(0.0, None) for _ in range(n)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None


def optimize_positions_slsqp(centers, radii, n):
    """Optimize positions given fixed radii using SLSQP."""
    x0 = centers.ravel()
    
    cons = ({
        'type': 'ineq',
        'fun': partial(compute_all_constraints, n=n, radii=radii)
    })
    
    bounds = [(0.0, 1.0) for _ in range(2 * n)]
    
    result = minimize(dummy_objective, x0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14})
    
    if result.success:
        return result.x.reshape(n, 2)
    return centers


def full_optimization(n, seed=0):
    """Run full optimization pipeline."""
    np.random.seed(seed)
    
    centers = init_hexagonal(n)
    radii = np.full(n, 0.04)
    
    # Add random perturbation
    centers = centers + np.random.randn(n, 2) * 0.01
    centers = np.clip(centers, 0.05, 0.95)
    
    # Force-directed initialization
    centers, radii = force_directed_init(centers, n, radii, num_iters=2000)
    
    # Alternating optimization
    for iteration in range(30):
        # Solve LP for optimal radii
        new_radii = solve_lp_for_radii(centers, n)
        if new_radii is not None:
            radii = new_radii
        else:
            break
        
        # Optimize positions
        centers = optimize_positions_slsqp(centers, radii, n)
        
        current_sum = np.sum(radii)
        if iteration % 5 == 0:
            print(f"  Seed {seed}, Iter {iteration}: sum_radii = {current_sum:.6f}")
    
    return centers, radii, np.sum(radii)


def repair_packing(centers, radii, n):
    """Repair any minor violations in the packing."""
    centers = centers.copy()
    radii = radii.copy()
    
    # Fix overlaps
    for _ in range(10):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-13:
                    overlap = radii[i] + radii[j] - dist
                    radii[i] -= overlap / 2.0
                    radii[j] -= overlap / 2.0
    
    # Fix boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-13:
            radii[i] = min(radii[i], max(0.0, x + 1e-13))
        if x + r > 1.0 + 1e-13:
            radii[i] = min(radii[i], max(0.0, 1.0 - x + 1e-13))
        if y - r < -1e-13:
            radii[i] = min(radii[i], max(0.0, y + 1e-13))
        if y + r > 1.0 + 1e-13:
            radii[i] = min(radii[i], max(0.0, 1.0 - y + 1e-13))
    
    return centers, radii
