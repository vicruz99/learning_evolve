from scipy.optimize import minimize

    def get_constraints(vars, n):

        # x - r >= 0

            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[3*i] - v[3*i + 2]
            })

            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1 - v[3*i] - v[3*i + 2]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[3*i + 1] - v[3*i + 2]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1 - v[3*i + 1] - v[3*i + 2]
            })

        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                idx_i, idx_j = 3*i, 3*j
                constraints.append({
                    'type': 'ineq',
                    'fun': lambda v, i=idx_i, j=idx_j: 
                        (v[i] - v[j])**2 + (v[i+1] - v[j+1])**2 - (v[i+2] + v[j+2])**2
                })
        return constraints

        return -np.sum(radii)

                col += 1
