from scipy.optimize import minimize

def objective(vars):
    """Objective function to minimize negative sum of radii."""
    n = 26
    radii = vars[n*2:]
    return -np.sum(radii)

def compute_constraints(vars):

    n = 26
    centers = vars[:n*2].reshape(n, 2)
    radii = vars[n*2:]
    
    # Boundary constraints: 5 per circle
    # x - r >= 0
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    # r >= 0

    bnd[0::5] = centers[:, 0] - radii
    bnd[1::5] = 1 - centers[:, 0] - radii
    bnd[2::5] = centers[:, 1] - radii
    bnd[3::5] = 1 - centers[:, 1] - radii
    bnd[4::5] = radii

    dist_sq = np.sum((centers[i] - centers[j])**2, axis=1)
    r_sum_sq = (radii[i] + radii[j])**2
    dist_constraints = dist_sq - r_sum_sq
    
    return np.concatenate([bnd, dist_constraints])
