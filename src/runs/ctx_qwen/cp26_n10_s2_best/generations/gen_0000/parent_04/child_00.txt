import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # 1. Initialization: Hexagonal packing
    # Estimate a loose radius to start
    # Area of square = 1. 26 circles. 
    # If density ~0.5, 26 * pi * r^2 ~ 0.5 => r ~ 0.08
    r_init = 0.07
    
    # Create a hexagonal grid
    # Row spacing: r * sqrt(3) ~ 0.12
    # Col spacing: 2r ~ 0.14
    # We want to fit 26 points. 
    # Let's try to generate a grid and pick 26 points, or just place them.
    # A 5x5 grid is 25 points. Let's add one more or adjust.
    # Or just a random valid placement is safer for general N, but hexagonal is better for sum of radii.
    
    # Let's generate points on a grid and perturb slightly
    # 6 rows, 5 cols = 30 points. We take first 26.
    # Spacing in x: 1/5 = 0.2
    # Spacing in y: 1/6 ~ 0.166
    # But hexagonal shift.
    
    # Let's use a simple grid for initialization to ensure feasibility
    # Grid 5x6 (30 points) centered?
    # Actually, just placing them in a valid configuration is enough.
    
    # Let's use a 5x5 grid (25) + center
    # Grid points: 0.1, 0.3, 0.5, 0.7, 0.9
    xs = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    ys = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    
    centers_list = []
    for y in ys:
        for x in xs:
            centers_list.append([x, y])
    
    # We have 25 points. Add one more at (0.5, 0.25) or similar?
    # (0.5, 0.25) is between 0.1 and 0.3 in y, at 0.5 in x.
    # Distance to (0.5, 0.1) is 0.15. r_init=0.07 -> 2r=0.14 < 0.15. OK.
    centers_list.append([0.5, 0.25])
    
    # Take first 26
    centers_init = np.array(centers_list[:n])
    radii_init = np.full(n, r_init)
    
    # Convert to optimization variables: r, u, v
    # x = r + u(1-2r) => u = (x-r)/(1-2r)
    # y = r + v(1-2r) => v = (y-r)/(1-2r)
    
    r_vec = radii_init
    x_vec = centers_init[:, 0]
    y_vec = centers_init[:, 1]
    
    # Avoid division by zero if r=0.5, but r_init is small.
    denom = 1 - 2*r_vec
    u_vec = (x_vec - r_vec) / denom
    v_vec = (y_vec - r_vec) / denom
    
    # Clip to [0, 1] to ensure valid start
    u_vec = np.clip(u_vec, 0.0, 1.0)
    v_vec = np.clip(v_vec, 0.0, 1.0)
    
    x0 = np.concatenate([r_vec, u_vec, v_vec])
    
    # Bounds
    # r in [1e-6, 0.5] (lower bound > 0 helps numerics)
    # u, v in [0, 1]
    lb_r = 1e-6
    ub_r = 0.5
    lb_uv = 0.0
    ub_uv = 1.0
    
    bounds = [(lb_r, ub_r)] * n + [(lb_uv, ub_uv)] * n + [(lb_uv, ub_uv)] * n
    
    # Precompute indices for constraints
    # Pairs (i, j) with i < j
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    num_pairs = len(pairs)
    
    # Function to compute constraints and Jacobian
    def compute_constraints_and_jac(vars):
        r = vars[:n]
        u = vars[n:2*n]
        v = vars[2*n:3*n]
        
        # Reconstruct centers
        # x = r + u * (1 - 2r) = r(1-2u) + u
        # y = r + v * (1 - 2r) = r(1-2v) + v
        # Derivatives:
        # dx/dr = 1 - 2u
        # dx/du = 1 - 2r
        # dx/dv = 0
        # dy/dr = 1 - 2v
        # dy/du = 0
        # dy/dv = 1 - 2r
        
        # Compute centers
        factor = 1.0 - 2.0 * r
        x = r + u * factor
        y = r + v * factor
        
        # Compute derivatives of x, y wrt vars
        # vars structure: [r_0...r_n, u_0...u_n, v_0...v_n]
        # Index mapping:
        # r_i is at i
        # u_i is at n + i
        # v_i is at 2*n + i
        
        # Precompute gradients for x and y
        # grad_x[k] = dx_i/dvars[k]
        # grad_y[k] = dy_i/dvars[k]
        # But we only need them for specific i, j pairs.
        
        # Let's compute constraint values and Jacobian matrix
        # J shape: (num_pairs, 3*n)
        # J[k, :] corresponds to constraint k (pair i, j)
        
        con_vals = np.zeros(num_pairs)
        J = np.zeros((num_pairs, 3 * n))
        
        # dx/dr_i = 1-2u_i, dx/du_i = 1-2r_i
        # dy/dr_i = 1-2v_i, dy/dv_i = 1-2r_i
        
        # We can vectorize the loop over pairs
        
        # Indices for r, u, v
        # r_idx: 0..n-1
        # u_idx: n..2n-1
        # v_idx: 2n..3n-1
        
        # Extract needed derivatives
        # For circle i:
        # dr_i contribution to x: (1 - 2*u_i)
        # du_i contribution to x: (1 - 2*r_i)
        # dr_i contribution to y: (1 - 2*v_i)
        # dv_i contribution to y: (1 - 2*r_i)
        
        # Let's compute these vectors
        # d_x_dr[i] = 1 - 2*u[i]
        # d_x_du[i] = 1 - 2*r[i]
        # d_y_dr[i] = 1 - 2*v[i]
        # d_y_dv[i] = 1 - 2*r[i]
        
        d_x_dr = 1.0 - 2.0 * u
        d_x_du = 1.0 - 2.0 * r
        d_y_dr = 1.0 - 2.0 * v
        d_y_dv = 1.0 - 2.0 * r
        
        # We need to iterate over pairs to fill J
        # To speed up, we can use broadcasting if we construct arrays for i and j
        
        pair_i = np.array([p[0] for p in pairs])
        pair_j = np.array([p[1] for p in pairs])
        
        # Coordinates
        x_i = x[pair_i]
        y_i = y[pair_i]
        x_j = x[pair_j]
        y_j = y[pair_j]
        
        # Differences
        dx = x_i - x_j
        dy = y_i - y_j
        dist = np.sqrt(dx**2 + dy**2)
        
        # Avoid division by zero if dist is 0 (shouldn't happen with valid radii)
        # But numerically possible? Add small epsilon?
        # If dist=0, gradient is undefined. But r > 0 implies dist >= 0?
        # If r_i+r_j > 0, dist must be >= r_i+r_j > 0.
        # So dist > 0.
        
        # Constraint value: dist - (r_i + r_j)
        r_i = r[pair_i]
        r_j = r[pair_j]
        con_vals = dist - (r_i + r_j)
        
        # Jacobian calculation
        # For constraint k (pair i, j):
        # g_k = sqrt(dx^2+dy^2) - r_i - r_j
        # dg_k/dx_i = dx/dist, dg_k/dy_i = dy/dist
        # dg_k/dx_j = -dx/dist, dg_k/dy_j = -dy/dist
        # dg_k/dr_i = -1
        # dg_k/dr_j = -1
        
        # Chain rule:
        # dg_k/dvars = dg_k/dx * dx/dvars + dg_k/dy * dy/dvars + dg_k/dr
        
        # dx/dvars for circle i:
        #   dr_i: d_x_dr[i]
        #   du_i: d_x_du[i]
        #   others 0
        # dy/dvars for circle i:
        #   dr_i: d_y_dr[i]
        #   dv_i: d_y_dv[i]
        #   others 0
        
        # Similarly for circle j.
        
        # Let's compute components
        inv_dist = 1.0 / dist
        
        # Gradient parts from x differences
        # Term A = dx/dist
        term_A = dx * inv_dist # shape (num_pairs,)
        # Term B = dy/dist
        term_B = dy * inv_dist # shape (num_pairs,)
        
        # Contribution to J from circle i's variables
        # J[k, r_idx[i]] += term_A * d_x_dr[i] + term_B * d_y_dr[i] - 1
        # J[k, u_idx[i]] += term_A * d_x_du[i]
        # J[k, v_idx[i]] += term_B * d_y_dv[i]
        
        # Contribution to J from circle j's variables
        # J[k, r_idx[j]] += -term_A * d_x_dr[j] - term_B * d_y_dr[j] - 1
        # J[k, u_idx[j]] += -term_A * d_x_du[j]
        # J[k, v_idx[j]] += -term_B * d_y_dv[j]
        
        # Initialize J is already 0. We add to it.
        
        # Indices for J columns
        # r cols: 0..n-1
        # u cols: n..2n-1
        # v cols: 2n..3n-1
        
        # Update J for i
        # r contribution
        J[np.arange(num_pairs), pair_i] += term_A * d_x_dr[pair_i] + term_B * d_y_dr[pair_i] - 1.0
        # u contribution
        J[np.arange(num_pairs), n + pair_i] += term_A * d_x_du[pair_i]
        # v contribution
        J[np.arange(num_pairs), 2*n + pair_i] += term_B * d_y_dv[pair_i]
        
        # Update J for j
        # r contribution
        J[np.arange(num_pairs), pair_j] += -term_A * d_x_dr[pair_j] - term_B * d_y_dr[pair_j] - 1.0
        # u contribution
        J[np.arange(num_pairs), n + pair_j] += -term_A * d_x_du[pair_j]
        # v contribution
        J[np.arange(num_pairs), 2*n + pair_j] += -term_B * d_y_dv[pair_j]
        
        return con_vals, J

    def constraint_callback(vars):
        vals, jac = compute_constraints_and_jac(vars)
        return vals, jac

    # Define constraint dict for SLSQP
    # SLSQP accepts fun returning array and jac returning matrix
    cons = {'type': 'ineq', 'fun': constraint_callback}
    
    # Objective
    def objective(vars):
        return -np.sum(vars[:n])
    
    # Optimization
    # maxiter might need to be high
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
    
    # Extract results
    r_opt = res.x[:n]
    u_opt = res.x[n:2*n]
    v_opt = res.x[2*n:3*n]
    
    x_opt = r_opt + u_opt * (1 - 2*r_opt)
    y_opt = r_opt + v_opt * (1 - 2*r_opt)
    
    centers_opt = np.column_stack([x_opt, y_opt])
    radii_opt = r_opt
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
