import numpy as np
from scipy.optimize import minimize


def compute_objective(x, n, alpha):
    """Compute objective: negative sum of radii plus penalty for constraint violations."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    # Ensure non-negative radii
    radii = np.maximum(radii, 1e-10)
    
    obj = -np.sum(radii)
    penalty = 0.0
    
    # Boundary violations
    for i in range(n):
        r = radii[i]
        x_coord, y_coord = centers[i]
        # Left
        val = x_coord - r
        if val < 0:
            penalty += alpha * val * val
        # Right
        val = 1.0 - x_coord - r
        if val < 0:
            penalty += alpha * val * val
        # Bottom
        val = y_coord - r
        if val < 0:
            penalty += alpha * val * val
        # Top
        val = 1.0 - y_coord - r
        if val < 0:
            penalty += alpha * val * val
    
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            gap = dist - radii[i] - radii[j]
            if gap < 0:
                penalty += alpha * gap * gap
    
    return obj + penalty
