import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

# Constants
N_CIRCLES = 26
N_VARS = N_CIRCLES * 3

def get_initial_guess():
    """
    Generates an initial feasible configuration of 26 circles 
    arranged in a hexagonal grid pattern.
    """
    centers = []
    radii = []
    
    # Start with a small radius to ensure feasibility
    r_init = 0.04 
    # Hexagonal vertical spacing
    dy = r_init * np.sqrt(3)
    
    row = 0
    while len(centers) < N_CIRCLES:
        y = r_init + row * dy
        
        # If the row is too high, stop (should not happen with small r_init)
        if y + r_init > 1.0 + 1e-9:
            break
            
        # Horizontal start position depends on row parity for hex packing
        # Even rows start at r_init, odd rows shifted by r_init
        # Actually, for hex packing, shift is r_init horizontally?
        # Distance between (r, r) and (2r, r + r*sqrt(3)) is sqrt(r^2 + 3r^2) = 2r.
        # So yes, shift is r_init.
        x_start = r_init if row % 2 == 0 else 2 * r_init
        
        x = x_start
        while x + r_init <= 1.0 + 1e-9 and len(centers) < N_CIRCLES:
            centers.append([x, y])
            radii.append(r_init)
            x += 2 * r_init # Spacing 2r horizontally
        
        row += 1
        
    # If for some reason we didn't get 26 (unlikely), pad with small circles
    while len(centers) < N_CIRCLES:
        centers.append([0.5, 0.5])
        radii.append(0.001)
        
    # Flatten to optimization variable vector
    # Order: x1, y1, r1, x2, y2, r2, ...
    vars_init = []
    for i in range(N_CIRCLES):
        vars_init.append(centers[i][0])
        vars_init.append(centers[i][1])
        vars_init.append(radii[i])
        
    return np.array(vars_init)

def objective(vars):
    """
    Objective function: Minimize negative sum of radii (maximize sum of radii).
    vars shape: (N_VARS,)
    Radii are at indices 2, 5, 8, ...
    """
    radii = vars[2::3]
    return -np.sum(radii)

def constraint_function(vars):
    """
    Computes all inequality constraints as a vector >= 0.
    Constraints:
    1. Boundary: x >= r, y >= r, x <= 1-r, y <= 1-r
    2. Non-overlap: dist_ij >= r_i + r_j
    """
    # Reshape variables
    # vars is flat [x0, y0, r0, x1, y1, r1, ...]
    # Extract arrays
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    constraints = []
    
    # 1. Boundary constraints
    # x - r >= 0
    constraints.extend(xs - rs)
    # y - r >= 0
    constraints.extend(ys - rs)
    # 1 - x - r >= 0  =>  (1-x) - r >= 0
    constraints.extend(1.0 - xs - rs)
    # 1 - y - r >= 0
    constraints.extend(1.0 - ys - rs)
    
    # 2. Non-overlap constraints
    # For all i < j: (xi-xj)^2 + (yi-yj)^2 >= (ri+rj)^2
    # We can vectorize this
    
    # Create matrices for differences
    # xs is (N,), xs[:, None] is (N, 1)
    dx = xs[:, None] - xs[None, :] # Shape (N, N)
    dy = ys[:, None] - ys[None, :]
    dr_sum = rs[:, None] + rs[None, :]
    
    dist_sq = dx**2 + dy**2
    radii_sum_sq = dr_sum**2
    
    # Constraint value: dist_sq - radii_sum_sq
    # We only need upper triangle (i < j)
    # Create a mask for upper triangle
    mask = np.triu(np.ones((N_CIRCLES, N_CIRCLES), dtype=bool), k=1)
    
    constraints.extend((dist_sq - radii_sum_sq)[mask])
    
    return np.array(constraints)

def run_packing():
    """
    Solves the circle packing problem.
    Returns (centers, radii, sum_radii).
    """
    # 1. Initial guess
    x0 = get_initial_guess()
    
    # 2. Bounds
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    # 3. Constraints
    # Using trust-constr which accepts a single function returning a vector of constraints
    # The function returns values that must be >= 0
    cons = NonlinearConstraint(constraint_function, 0, np.inf)
    
    # 4. Optimization
    # Method 'trust-constr' is robust for nonlinear constraints
    # 'SLSQP' is also an option but might struggle with many constraints or vectorization
    # We try trust-constr first. If it fails, we might fallback, but let's keep it simple.
    
    try:
        res = minimize(
            objective,
            x0,
            method='trust-constr',
            bounds=bounds,
            constraints=[cons],
            options={
                'verbose': 0,
                'maxiter': 500,
                'gtol': 1e-8
            }
        )
        
        if not res.success:
            # If trust-constr fails, try SLSQP with a simpler setup?
            # But SLSQP doesn't take vector constraints easily without lambda closures.
            # Let's stick with the result if it's valid, or try a restart.
            pass
            
        # Extract solution
        x_opt = res.x
        
    except Exception as e:
        # Fallback to a valid grid if optimization crashes
        # Just use the initial guess
        x_opt = x0

    # Post-processing to ensure validity and format output
    centers = np.array([x_opt[i*3 : i*3+2] for i in range(N_CIRCLES)])
    radii = np.array([x_opt[i*3+2] for i in range(N_CIRCLES)])
    
    # Sanity check / repair if needed
    # Ensure non-negative radii
    radii = np.maximum(radii, 0.0)
    
    # Clamp centers to [0,1] just in case (though constraints should prevent)
    centers = np.clip(centers, 0.0, 1.0)
    
    # Re-check constraints strictly? 
    # The optimizer satisfies them within tolerance.
    # The validation function uses 1e-12 tolerance.
    
    sum_radii = np.sum(radii)
    
    # If the sum is very low or invalid, we might want to return the initial guess
    # But the initial guess has sum 26*0.04 = 1.04.
    # Optimizer should improve this significantly.
    
    return centers, radii, sum_radii
