        y = v[1::3]
        r = v[2::3]
        return x, y, r

        v = np.empty(n * 3)
        v[0::3] = x
        v[1::3] = y
        v[2::3] = r
        return v

        _, _, r = unpack(v)
        return -np.sum(r)

    def get_constraints(v):
        constraints = []
        x, y, r = unpack(v)

        # We can implement these as bounds or inequality constraints.
        # Bounds are handled by 'bounds' argument in minimize, 
        # but x >= r is coupled. So we use inequality constraints.
        # Actually, simpler to enforce r <= 0.5 via bounds, and check boundaries via constraints.
        # But for efficiency, let's just use the inequality constraints.
        
        # Boundary: x - r >= 0
        for i in range(n):
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[3*i] - v[3*i + 2]
            })
