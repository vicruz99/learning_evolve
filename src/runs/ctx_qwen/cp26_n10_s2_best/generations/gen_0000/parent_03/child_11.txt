import numpy as np
from scipy.optimize import minimize

def compute_constraints(v):
    """Compute inequality constraints for circle packing."""
    n = 26
    cx = v[:n]
    cy = v[n:2*n]
    cr = v[2*n:3*n]
    
    # 4 boundary constraints per circle + pairwise non-overlap constraints
    # 26 * 4 = 104 boundary constraints
    # 26 * 25 / 2 = 325 pairwise constraints
    # Total 429 constraints
    c = np.empty(429)
    idx = 0
    
    for i in range(n):
        c[idx] = cx[i] - cr[i]
        idx += 1
        c[idx] = 1.0 - cx[i] - cr[i]
        idx += 1
        c[idx] = cy[i] - cr[i]
        idx += 1
        c[idx] = 1.0 - cy[i] - cr[i]
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            dist = np.sqrt(dx*dx + dy*dy)
            c[idx] = dist - cr[i] - cr[j]
            idx += 1
            
    return c

def objective_func(v):
    """Negative sum of radii to be minimized."""
    return -np.sum(v[52:])

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds: coordinates in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    constraints = {'type': 'ineq', 'fun': compute_constraints}
    
    # Run optimization from multiple random starting points to escape local minima
    for seed in range(20):

        # Initial guess: random positions in [0,1], small initial radii to start feasible

        v0[2*n:] = 0.03
        
        try:

        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
