import numpy as np
from scipy.optimize import minimize
import warnings

warnings.filterwarnings('ignore')

def compute_penalty(centers_flat, r, n=26):
    """
    Computes a smooth penalty for constraint violations.
    Lower penalty means better packing. 0 means perfectly valid.
    """
    centers = centers_flat.reshape((n, 2))
    penalty = 0.0
    
    # Boundary penalties
    x = centers[:, 0]
    y = centers[:, 1]
    penalty += np.sum(np.maximum(0, r - x)**2)
    penalty += np.sum(np.maximum(0, x + r - 1)**2)
    penalty += np.sum(np.maximum(0, r - y)**2)
    penalty += np.sum(np.maximum(0, y + r - 1)**2)
    
    # Pairwise overlap penalties (vectorized)
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Ignore self-distance by setting diagonal to a large value
    np.fill_diagonal(dists, 2.0)
    
    overlaps = 2.0 * r - dists
    penalty += np.sum(np.maximum(0, overlaps)**2)
    
    return penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_centers = None
    best_r = 0.05
    
    # Prepare diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice initialization
    pts = []
    row = 0
    while len(pts) < n:
        col = 0
        while len(pts) < n:
            x = 0.08 + col * 0.14 + (row % 2) * 0.07
            y = 0.08 + row * 0.12
            if x <= 0.92 and y <= 0.92:
                pts.append([x, y])
            col += 1
        row += 1
    inits.append(np.array(pts[:n]).flatten())
    
    # 2 & 3. Random uniform initializations
    inits.append(np.random.uniform(0.1, 0.9, (n, 2)).flatten())
    inits.append(np.random.uniform(0.1, 0.9, (n, 2)).flatten())
    
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # Binary search for the maximum feasible radius
    low, high = 0.05, 0.105
    for _ in range(30):
        if high - low < 1e-5:
            break
            
        mid = (low + high) / 2
        success = False
        
        for init in inits:
            # Add slight random perturbation to avoid symmetric traps
            perturbed = init + np.random.normal(0, 0.003, size=init.shape)
            perturbed = np.clip(perturbed, 0.01, 0.99)
            
            res = minimize(
                compute_penalty, 
                perturbed, 
                args=(mid, n), 
                method='L-BFGS-B', 
                bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-14}
            )
            
            if res.fun < 1e-7:
                best_centers = res.x.reshape((n, 2))
                best_r = mid
                success = True
                low = mid
                break
                
        if not success:
            high = mid
            
    # Fallback if optimization fails (should not happen with this setup)
    if best_centers is None:
        best_r = 0.05
        best_centers = np.random.uniform(0.2, 0.8, (n, 2))
        
    radii = np.full(n, best_r)
    total_sum = float(np.sum(radii))
    
    return best_centers, radii, total_sum
