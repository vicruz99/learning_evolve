import numpy as np
from scipy.optimize import minimize


def compute_objective_and_penalty(vars, n, penalty_weight):
    centers = vars[:2*n].reshape(n, 2)
    radii = np.maximum(vars[2*n:], 1e-10)
    
    obj = -np.sum(radii)
    penalty = 0.0
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        penalty += max(0, r - x)**2
        penalty += max(0, r - (1 - x))**2
        penalty += max(0, r - y)**2
        penalty += max(0, r - (1 - y))**2
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += (min_dist - dist)**2
    
    return obj + penalty_weight * penalty


def hexagonal_init(n, seed):
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.06
    
    idx = 0
    row_spacing = 0.155
    col_spacing = 0.17
    
    for row in range(6):
        for col in range(5):
            if idx >= n:
                break
            x = 0.11 + col * col_spacing + (row % 2) * col_spacing / 2
            y = 0.11 + row * row_spacing
            x += np.random.uniform(-0.008, 0.008)
            y += np.random.uniform(-0.008, 0.008)
            centers[idx] = [x, y]
            radii[idx] = 0.055
            idx += 1
        if idx >= n:
            break
    
    return centers, radii


def force_based_simulation(n, centers, radii, num_iterations):
    dt = 0.002
    growth_rate = 0.00004
    
    for iteration in range(num_iterations):
        forces = np.zeros((n, 2))
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff * diff))
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    force_mag = overlap * 800 / (dist + 1e-10)
                    direction = diff / (dist + 1e-10)
                    forces[i] += direction * force_mag
                    forces[j] -= direction * force_mag
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            if x - r < 0.001:
                forces[i, 0] += (r - x + 0.001) * 500
            if x + r > 0.999:
                forces[i, 0] -= (x + r - 0.999) * 500
            if y - r < 0.001:
                forces[i, 1] += (r - y + 0.001) * 500
            if y + r > 0.999:
                forces[i, 1] -= (y + r - 0.999) * 500
        
        centers += forces * dt
        centers = np.clip(centers, 0.005, 0.995)
        
        radii += growth_rate
        radii = np.clip(radii, 1e-10, 0.35)
        
        if iteration % 5000 == 0 and iteration > 0:
            growth_rate *= 0.95
    
    return centers, radii


def optimize_with_penalty(n, centers_init, radii_init, weights):
    vars_init = np.concatenate([centers_init.flatten(), radii_init])
    
    bounds = []
    for i in range(n):
        bounds.extend([(0.001, 0.999), (0.001, 0.999)])
        bounds.append((1e-10, 0.4))
    
    current_vars = vars_init.copy()
    
    for weight in weights:
        def objective(vars, w=weight):
            return compute_objective_and_penalty(vars, n, w)
        
        result = minimize(objective, current_vars, bounds=bounds, 
                        method='L-BFGS-B',
                        options={'maxiter': 4000, 'ftol': 1e-15, 'gtol': 1e-10})
        current_vars = result.x
    
    centers_opt = current_vars[:2*n].reshape(n, 2)
    radii_opt = np.maximum(current_vars[2*n:], 1e-10)
    
    return centers_opt, radii_opt


def ensure_feasibility(centers, radii, n):
    for i in range(n):
        r = radii[i]
        centers[i, 0] = max(r + 1e-12, min(1 - r - 1e-12, centers[i, 0]))
        centers[i, 1] = max(r + 1e-12, min(1 - r - 1e-12, centers[i, 1]))
    
    for _ in range(50):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist - 1e-12 and dist > 1e-10:
                    overlap = min_dist - dist
                    direction = np.array([dx, dy]) / dist
                    centers[i] += direction * overlap * 0.51
                    centers[j] -= direction * overlap * 0.51
                    improved = True
        
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r + 1e-12, min(1 - r - 1e-12, centers[i, 0]))
            centers[i, 1] = max(r + 1e-12, min(1 - r - 1e-12, centers[i, 1]))
        
        if not improved:
            break
    
    return centers


def run_packing():
    n = 26
    
    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None
    
    for restart in range(15):
        centers_init, radii_init = hexagonal_init(n, seed=restart * 7 + 3)
        
        centers_sim, radii_sim = force_based_simulation(n, centers_init.copy(), radii_init.copy(), 30000)
        
        centers_opt, radii_opt = optimize_with_penalty(n, centers_sim, radii_sim, 
                                                       [50, 200, 1000, 5000, 20000, 50000])
        
        centers_opt = ensure_feasibility(centers_opt, radii_opt, n)
        
        current_sum = np.sum(radii_opt)
        
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    return best_centers, best_radii, best_sum_radii
