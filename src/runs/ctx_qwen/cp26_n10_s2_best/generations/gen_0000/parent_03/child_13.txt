def compute_hex_grid_points(n, target_radius=0.1):

    Generates a list of (x, y) coordinates for a hexagonal grid.
    The grid is designed to fit within the unit square [0, 1]x[0, 1].
    """
    points = []
    # Diameter approximation for spacing
    d = 2 * target_radius

    dy = d * np.sqrt(3) / 2.0

    return points[:n]

    violation_left = np.maximum(0.0, r - x)
    violation_right = np.maximum(0.0, r - (1.0 - x))
    violation_bottom = np.maximum(0.0, r - y)
    violation_top = np.maximum(0.0, r - (1.0 - y))
    
    # Squared penalties
    penalty += np.sum(violation_left**2)
    penalty += np.sum(violation_right**2)
    penalty += np.sum(violation_bottom**2)
    penalty += np.sum(violation_top**2)
    
    # 2. Overlap Violations
    # For each pair (i, j), distance >= r_i + r_j
    # Violation = max(0, r_i + r_j - distance)

    # Squared penalty for overlaps
    penalty += np.sum(positive_overlaps**2)
    
    # Total objective
