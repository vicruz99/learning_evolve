import numpy as np
from scipy.optimize import minimize
from functools import partial
import math


def compute_objective(x, n):
    """Compute negative sum of radii (to minimize)."""
    total = 0.0
    for i in range(n):
        total += x[3 * i + 2]
    return -total


def boundary_constraint_x_min(x, i):
    """Constraint: x_i >= r_i"""
    return x[3 * i] - x[3 * i + 2]


def boundary_constraint_x_max(x, i):
    """Constraint: x_i + r_i <= 1"""
    return 1.0 - x[3 * i] - x[3 * i + 2]


def boundary_constraint_y_min(x, i):
    """Constraint: y_i >= r_i"""
    return x[3 * i + 1] - x[3 * i + 2]


def boundary_constraint_y_max(x, i):
    """Constraint: y_i + r_i <= 1"""
    return 1.0 - x[3 * i + 1] - x[3 * i + 2]


def radius_nonnegative(x, i):
    """Constraint: r_i >= 0"""
    return x[3 * i + 2]


def nonoverlap_constraint(x, i, j):
    """Constraint: distance(i,j)^2 >= (r_i + r_j)^2"""
    dx = x[3 * i] - x[3 * j]
    dy = x[3 * i + 1] - x[3 * j + 1]
    dist_sq = dx * dx + dy * dy
    sum_r = x[3 * i + 2] + x[3 * j + 2]
    return dist_sq - sum_r * sum_r


def build_constraints(n):
    """Build list of constraint dicts."""
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': partial(boundary_constraint_x_min, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(boundary_constraint_x_max, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(boundary_constraint_y_min, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(boundary_constraint_y_max, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(radius_nonnegative, i=i)})
    
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': partial(nonoverlap_constraint, i=i, j=j)})
    
    return constraints


def build_bounds(n):
    """Build bounds for optimization variables."""
    bounds = []
    for k in range(3 * n):
        if k % 3 == 2:
            bounds.append((1e-12, 0.5))
        else:
            bounds.append((0.0, 1.0))
    return bounds

def extract_solution(x, n):
    """Extract centers and radii from optimization variable."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = max(0.0, min(1.0, x[3 * i]))
        centers[i, 1] = max(0.0, min(1.0, x[3 * i + 1]))
        radii[i] = max(0.0, x[3 * i + 2])
    return centers, radii


def pack_with_optimization(centers_init, radii_init, n, constraints, bounds, maxiter=3000):
    """Run optimization from given initial configuration."""
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers_init[i, 0]
        x0[3 * i + 1] = centers_init[i, 1]
        x0[3 * i + 2] = radii_init[i]
    
    def objective(x):
        return compute_objective(x, n)
    
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False}
    )
    
    centers_opt, radii_opt = extract_solution(result.x, n)
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii, result.success


def perturb_solution(centers, radii, n, scale=0.02):
    """Add random perturbation to solution."""
    centers_pert = centers.copy()
    radii_pert = radii.copy()
    
    for i in range(n):
        dx = np.random.uniform(-scale, scale)
        dy = np.random.uniform(-scale, scale)
        centers_pert[i, 0] = max(0.0, min(1.0, centers_pert[i, 0] + dx))
        centers_pert[i, 1] = max(0.0, min(1.0, centers_pert[i, 1] + dy))
        radii_pert[i] = max(1e-6, radii_pert[i] * (1 + np.random.uniform(-0.1, 0.1)))
    
    return centers_pert, radii_pert
