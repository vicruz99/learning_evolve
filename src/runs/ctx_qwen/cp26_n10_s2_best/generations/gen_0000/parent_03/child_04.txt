import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    radii = x[2*N_CIRCLES:]
    return -np.sum(radii)

def constr_boundary(x):
    xc = x[0::2]
    yc = x[1::2]
    r = x[2*N_CIRCLES:]
    c = np.zeros(4*N_CIRCLES)
    c[0::4] = r - xc
    c[1::4] = r + xc - 1.0
    c[2::4] = r - yc
    c[3::4] = r + yc - 1.0
    return c

def neg_constr_boundary(x):
    return -constr_boundary(x)

def constr_overlap(x):
    xc = x[0::2]
    yc = x[1::2]
    r = x[2*N_CIRCLES:]
    n_cons = N_CIRCLES * (N_CIRCLES - 1) // 2
    con = np.empty(n_cons)
    idx = 0
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            dist_sq = (xc[i] - xc[j])**2 + (yc[i] - yc[j])**2
            con[idx] = (r[i] + r[j])**2 - dist_sq
            idx += 1
    return con

def neg_constr_overlap(x):
    return -constr_overlap(x)

def get_initial_guess(method='hex'):
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, 0.05)
    
    if method == 'hex':
        pts = []
        dx = 0.22
        dy = dx * np.sqrt(3) / 2
        y_curr = dy
        row = 0
        while len(pts) < N_CIRCLES:
            x_offset = dx/2 if row % 2 == 1 else 0
            x_curr = x_offset + dx/2
            while x_curr < 1 - dx/2:
                pts.append([x_curr, y_curr])
                x_curr += dx
            y_curr += dy
            row += 1
        pts = np.array(pts[:N_CIRCLES])
        centers = pts
    elif method == 'random':
        centers = np.random.rand(N_CIRCLES, 2)
        
    centers += np.random.normal(0, 0.01, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    return centers, radii

def run_packing():
    bounds = [(0.0, 1.0), (0.0, 1.0)] * N_CIRCLES + [(0.0, 0.5)] * N_CIRCLES
    cons = [
        {'type': 'ineq', 'fun': neg_constr_boundary},
        {'type': 'ineq', 'fun': neg_constr_overlap}
    ]
    
    best_x = None
    best_sum = 0.0
    
    np.random.seed(123)
    # Hexagonal starts
    for _ in range(3):
        centers, radii = get_initial_guess('hex')
        x0 = np.concatenate([centers.ravel(), radii])
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 2000, 'ftol': 1e-9})
        if -res.fun > best_sum:
            best_x = res.x.copy()
            best_sum = -res.fun
            
    # Random starts
    for _ in range(3):
        centers, radii = get_initial_guess('random')
        x0 = np.concatenate([centers.ravel(), radii])
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-9})
        if -res.fun > best_sum:
            best_x = res.x.copy()
            best_sum = -res.fun
            
    if best_x is None:
        centers, radii = get_initial_guess('hex')
        best_x = np.concatenate([centers.ravel(), radii])
        
    centers = np.reshape(best_x[:2*N_CIRCLES], (N_CIRCLES, 2))
    radii = best_x[2*N_CIRCLES:]
    
    # Projection step to guarantee strict feasibility within tolerance
    for _ in range(20):
        updated = False
        for i in range(N_CIRCLES):
            for dim in range(2):
                if centers[i, dim] - radii[i] < -1e-10:
                    radii[i] = centers[i, dim] + 1e-10
                    updated = True
                if centers[i, dim] + radii[i] > 1 + 1e-10:
                    radii[i] = 1 - centers[i, dim] + 1e-10
                    updated = True
            for j in range(i+1, N_CIRCLES):

                    scale = (dist + 1e-10) / min_sum
                    radii[i] *= scale
                    radii[j] *= scale
                    updated = True
        if not updated:
            break
            
    return centers, radii, np.sum(radii)
