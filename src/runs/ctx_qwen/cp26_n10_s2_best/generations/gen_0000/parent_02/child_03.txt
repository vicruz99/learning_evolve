import numpy as np
from scipy.optimize import minimize

def run_packing():

    Pack 26 circles in a unit square to maximize the sum of radii.

    best_radii = None

    def generate_hexagonal_packing(n, r):
        """
        Generate centers for n circles of radius r in a hexagonal arrangement.
        Returns centers (n, 2) or None if not all fit.
        """
        # Vertical spacing
        dy = np.sqrt(3) * r
        # Horizontal spacing
        dx = 2 * r
        
        rows = []
        y = r
        row_idx = 0
        
        while True:
            # Determine x offset for this row (0 for even, r for odd)
            offset = r if row_idx % 2 == 1 else 0
            # If offset makes first center < r, adjust or skip?
            # Standard hex packing: row 0 starts at r, row 1 starts at 2r (offset r from r)
            # Actually, centers at x = r + k*2r + offset?
            # Let's stick to: row 0 centers at r, r+2r, ...
            # row 1 centers at r+r, r+r+2r, ... i.e. 2r, 4r, ...
            
            start_x = r + offset
            centers_in_row = []
            x = start_x
            while x <= 1 - r + 1e-9: # Allow tiny epsilon for floating point
                centers_in_row.append((x, y))
                x += 2 * r
            
            rows.append(centers_in_row)
            
            # Check if we have enough circles
            total = sum(len(r) for r in rows)
            if total >= n:
                break
            
            y += dy
            row_idx += 1
            
            # Safety break
            if y + r > 1 + 1e-9:
                break
        
        # Flatten and truncate
        flat_centers = []
        for row in rows:
            for c in row:
                flat_centers.append(c)
                if len(flat_centers) == n:
                    break
            if len(flat_centers) == n:
                break
        
        if len(flat_centers) < n:
            return None
            
        return np.array(flat_centers[:n])

        # we used the lattice. But now we do a general optimization.
        
        r = params[0]
        centers = params[1:].reshape((n, 2))
        
        penalty = 0.0
        sum_radii = n * r # Assuming equal radii for the optimization variable r

        # Circle i center (cx, cy), radius r.
        # r <= cx <= 1-r  =>  0 <= cx - r  and  cx + r <= 1

        for i in range(n):
            cx, cy = centers[i]
            # Lower bounds
            if cx - r < 0:
                penalty += 1000 * (cx - r) ** 2
            if cy - r < 0:
                penalty += 1000 * (cy - r) ** 2
            # Upper bounds
            if cx + r > 1:
                penalty += 1000 * (cx + r - 1) ** 2
            if cy + r > 1:
                penalty += 1000 * (cy + r - 1) ** 2

        # Overlap constraints

        for i in range(n):
            for j in range(i + 1, n):
                d2 = (centers[i] - centers[j])**2
                dist = np.sqrt(d2.sum())
                if dist < 2 * r:
                    penalty += 1000 * (2 * r - dist) ** 2
        
        return -sum_radii + penalty # Minimize negative sum + penalty

                            break
