import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def obj_func(vars):

    return -np.sum(vars[2::3])

def constr_func(vars):
    """Inequality constraints: non-overlap and boundary containment"""
    n = N_CIRCLES
    n_constraints = n * (n - 1) // 2 + 4 * n
    cons = np.empty(n_constraints)
    idx = 0
    
    # Non-overlap constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = vars[3 * i], vars[3 * i + 1], vars[3 * i + 2]
            xj, yj, rj = vars[3 * j], vars[3 * j + 1], vars[3 * j + 2]
            cons[idx] = (xi - xj) ** 2 + (yi - yj) ** 2 - (ri + rj) ** 2
            idx += 1
            
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    for i in range(n):
        xi, yi, ri = vars[3 * i], vars[3 * i + 1], vars[3 * i + 2]
        cons[idx] = xi - ri
        cons[idx + 1] = 1.0 - xi - ri
        cons[idx + 2] = yi - ri
        cons[idx + 3] = 1.0 - yi - ri
        idx += 4
        
    return cons

def compute_forces(centers, radii, k_rep, k_bound):
    """Compute repulsive forces between circles and boundaries"""
    n = len(centers)
    forces = np.zeros_like(centers)
    
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff)
            sum_r = radii[i] + radii[j]
            
            if dist < sum_r + 1e-8:
                if dist > 1e-8:
                    overlap = sum_r - dist
                    dir_vec = diff / dist
                    forces[i] += dir_vec * k_rep * overlap
                    forces[j] -= dir_vec * k_rep * overlap
                else:
                    # Prevent division by zero with random push
                    forces[i] += np.random.randn(2) * 0.05
                    forces[j] -= np.random.randn(2) * 0.05
                    
    # Boundary repulsion
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0: forces[i, 0] += k_bound * (r - x)
        if x + r > 1: forces[i, 0] -= k_bound * (x + r - 1)
        if y - r < 0: forces[i, 1] += k_bound * (r - y)
        if y + r > 1: forces[i, 1] -= k_bound * (y + r - 1)
        
    return forces

def run_packing():

    n = N_CIRCLES

    dy = np.sqrt(3) * r_init

                idx += 1

        centers += velocities * dt
