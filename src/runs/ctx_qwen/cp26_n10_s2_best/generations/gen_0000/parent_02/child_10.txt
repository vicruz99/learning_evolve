import numpy as np
from scipy.optimize import minimize

def compute_loss(v, n, penalty):
    """Compute penalized objective for circle packing."""
    cx = v[:n]
    cy = v[n:2*n]
    cr = v[2*n:]
    
    # Primary objective: maximize sum of radii
    loss = -np.sum(cr)
    
    # Boundary penalties: circles must stay within [0,1]x[0,1]
    loss += penalty * np.sum(np.maximum(0, cr - cx)**2)
    loss += penalty * np.sum(np.maximum(0, cx + cr - 1)**2)
    loss += penalty * np.sum(np.maximum(0, cr - cy)**2)
    loss += penalty * np.sum(np.maximum(0, cy + cr - 1)**2)
    
    # Overlap penalties: pairwise distance must be >= sum of radii
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dist = np.sqrt(dx**2 + dy**2)
    overlap = cr[:, np.newaxis] + cr[np.newaxis, :] - dist
    np.fill_diagonal(overlap, 0.0)
    loss += penalty * np.sum(np.maximum(0, overlap)**2)
    
    return loss
