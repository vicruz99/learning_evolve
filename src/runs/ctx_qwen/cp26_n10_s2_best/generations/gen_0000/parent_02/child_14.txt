import numpy as np
from scipy.optimize import minimize


def compute_max_radii(centers, radii, n):
    """Compute maximum possible radius for each circle given current configuration."""
    new_radii = np.zeros(n)
    for i in range(n):
        # Distance to boundaries
        r_max = min(centers[i][0], 1.0 - centers[i][0],
                    centers[i][1], 1.0 - centers[i][1])
        # Distance to other circles minus their radii
        for j in range(n):
            if i != j:
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx * dx + dy * dy)
                available = dist - radii[j]
                if available < r_max:
                    r_max = available
        new_radii[i] = max(r_max, 0.0)
    return new_radii


def objective_single_position(params, circle_idx, all_centers, all_radii, n):
    """Objective function for optimizing a single circle's position."""
    temp_centers = all_centers.copy()
    temp_centers[circle_idx] = params
    temp_radii = compute_max_radii(temp_centers, all_radii, n)
    return -np.sum(temp_radii)


def initialize_hexagonal_grid(n, spacing, offset_x, offset_y):
    """Initialize circle centers in a hexagonal grid pattern."""
    centers = np.zeros((n, 2))
    count = 0
    for row in range(10):
        for col in range(10):
            if count >= n:
                return centers
            x = offset_x + col * spacing + (row % 2) * spacing * 0.5
            y = offset_y + row * spacing * np.sqrt(3) / 2.0
            if 0.05 <= x <= 0.95 and 0.05 <= y <= 0.95:
                centers[count] = [x, y]
                count += 1
    return centers


def optimize_from_config(initial_centers, n, max_outer_iter=150, max_inner_iter=80,
                          max_pos_opt_iter=150):
    """Run local optimization from a given configuration."""
    centers = initial_centers.copy()
    radii = np.full(n, 0.005)

    for outer in range(max_outer_iter):
        # Update all radii to maximum possible
        radii = compute_max_radii(centers, radii, n)

        # Optimize each circle's position
        for i in range(n):
            result = minimize(
                objective_single_position,
                centers[i],
                args=(i, centers, radii, n),
                method='Nelder-Mead',
                options={
                    'maxiter': max_pos_opt_iter,
                    'xatol': 1e-9,
                    'fatol': 1e-10,
                    'adaptive': True
                }
            )
            if result.success or result.fun < -np.sum(radii) - 1e-12:
                centers[i] = result.x
                # Recompute radii after position change
                radii = compute_max_radii(centers, radii, n)

        # Small random perturbations to escape local minima
        if outer % 20 == 19:
            perturbation = 0.003 * np.random.randn(n, 2)
            centers += perturbation
            centers = np.clip(centers, 0.02, 0.98)

    return centers, radii


def run_packing():

    n = 26
    np.random.seed(12345)

    best_centers = None

    best_sum = 0.0

    # Try various hexagonal grid configurations
    configs_to_try = []

    for spacing in [0.16, 0.17, 0.18, 0.19, 0.20, 0.21]:
        for ox in [0.05, 0.07, 0.08, 0.10]:
            for oy in [0.05, 0.07, 0.08, 0.10]:
                cfg = initialize_hexagonal_grid(n, spacing, ox, oy)
                # Check if we got all n circles
                valid = True
                for i in range(n):
                    if cfg[i][0] == 0 and cfg[i][1] == 0 and i > 0:
                        valid = False
                        break
                if valid and np.all(cfg[:n] > 0):
                    configs_to_try.append(cfg.copy())

    # Also try a simple grid
    for spacing in [0.15, 0.18, 0.20, 0.22]:

            for oy in [0.05, 0.10]:
                cfg = np.zeros((n, 2))
                count = 0

                    for col in range(8):
                        if count >= n:
                            break
                        x = ox + col * spacing
                        y = oy + row * spacing
                        if x <= 0.95 and y <= 0.95:
                            cfg[count] = [x, y]
                            count += 1
                    if count >= n:
                        break
                if count >= n:
                    configs_to_try.append(cfg.copy())

    # Add randomized perturbations of good configs
    for cfg in configs_to_try[:20]:
        for _ in range(5):
            perturbed = cfg.copy()
            perturbed += np.random.randn(n, 2) * 0.02
            perturbed = np.clip(perturbed, 0.03, 0.97)
            configs_to_try.append(perturbed)

    # Recompute radii one final time
    final_radii = compute_max_radii(best_centers, final_radii, n)

    return best_centers, final_radii, np.sum(final_radii)
