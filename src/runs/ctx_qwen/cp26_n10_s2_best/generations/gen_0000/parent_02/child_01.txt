import numpy as np
from scipy.optimize import minimize

def compute_constraints(vars):
    """Compute inequality constraints for boundary and non-overlap conditions."""
    n = 26
    con_size = 4 * n + n * (n - 1) // 2
    con = np.empty(con_size)
    idx = 0
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    for i in range(n):
        x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
        con[idx] = x - r
        con[idx+1] = 1.0 - x - r
        con[idx+2] = y - r
        con[idx+3] = 1.0 - y - r
        idx += 4
        
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = vars[3*i] - vars[3*j]
            dy = vars[3*i+1] - vars[3*j+1]
            r_sum = vars[3*i+2] + vars[3*j+2]
            con[idx] = dx**2 + dy**2 - r_sum**2
            idx += 1
            
    return con

def objective(vars):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(vars[2::3])

def run_packing():
    n = 26
    r_init = 0.08  # Start strictly inside feasible region
    
    # Initialize with hexagonal lattice pattern
    centers = np.zeros((n, 2))
    idx = 0
    y = r_init
    row = 0
    while idx < n:
        x = r_init + (row % 2) * r_init
        while x + r_init <= 1.0 and idx < n:
            centers[idx] = [x, y]
            idx += 1
            x += 2 * r_init
        y += r_init * np.sqrt(3)
        row += 1
        
    # Flatten to optimization variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = r_init
        
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    constraints = {'type': 'ineq', 'fun': compute_constraints}
    
    # Optimize
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                   
    centers_opt = res.x[:2*n].reshape(-1, 2)
    radii_opt = res.x[2*n:]
    
    # Post-processing to guarantee strict validity within numerical tolerance
    eps = 1e-9
    radii_opt = np.maximum(radii_opt, eps)
    centers_opt[:, 0] = np.clip(centers_opt[:, 0], radii_opt, 1.0 - radii_opt)
    centers_opt[:, 1] = np.clip(centers_opt[:, 1], radii_opt, 1.0 - radii_opt)
    
    # Resolve microscopic overlaps by slight radius reduction
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
            required = radii_opt[i] + radii_opt[j] + eps
            if dist < required:
                scale = dist / (radii_opt[i] + radii_opt[j])
                radii_opt[i] *= scale
                radii_opt[j] *= scale
                
    return centers_opt, radii_opt, np.sum(radii_opt)
