import numpy as np
from scipy.optimize import minimize

def boundary_constraint_x_lower(vars, i):
    """Constraint: x_i - r_i >= 0"""
    return vars[3*i] - vars[3*i + 2]

def boundary_constraint_x_upper(vars, i):
    """Constraint: 1 - x_i - r_i >= 0"""
    return 1.0 - vars[3*i] - vars[3*i + 2]

def boundary_constraint_y_lower(vars, i):
    """Constraint: y_i - r_i >= 0"""
    return vars[3*i + 1] - vars[3*i + 2]

def boundary_constraint_y_upper(vars, i):
    """Constraint: 1 - y_i - r_i >= 0"""
    return 1.0 - vars[3*i + 1] - vars[3*i + 2]

def pairwise_constraint(vars, i, j):
    """Constraint: dist^2 - (ri + rj)^2 >= 0"""
    xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
    xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
    dx = xi - xj
    dy = yi - yj
    return dx*dx + dy*dy - (ri + rj)*(ri + rj)

def objective_func(vars, n):
    """Objective: Minimize -sum(radii)"""
    s = 0.0
    for i in range(n):
        s += vars[3*i + 2]
    return -s
