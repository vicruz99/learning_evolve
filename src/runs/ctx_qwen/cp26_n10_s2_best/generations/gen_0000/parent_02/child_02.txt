import numpy as np
from scipy.optimize import minimize

def obj(vars, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def con_bnd(vars, n):
    """Boundary constraints: circles must stay inside [0,1]x[0,1]."""
    cons = []
    for i in range(n):
        idx = 3 * i
        x, y, r = vars[idx], vars[idx+1], vars[idx+2]
        cons.append(x - r)
        cons.append(1.0 - x - r)
        cons.append(y - r)
        cons.append(1.0 - y - r)
    return np.array(cons)

def con_ovl(vars, n):
    """Overlap constraints: distance between centers >= sum of radii."""
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            idx_i, idx_j = 3 * i, 3 * j
            dx = vars[idx_i] - vars[idx_j]
            dy = vars[idx_i+1] - vars[idx_j+1]
            dr = vars[idx_i+2] + vars[idx_j+2]
            cons.append(dx*dx + dy*dy - dr*dr)
    return np.array(cons)

def run_packing():
    n = 26
    
    # 1. Generate initial hexagonal lattice positions
    pts = []
    r_init = 0.09
    y = r_init
    row = 0
    while len(pts) < n:
        x_start = r_init if row % 2 == 0 else 2.0 * r_init
        x = x_start
        while x <= 1.0 - r_init:
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r_init
        y += np.sqrt(3.0) * r_init
        row += 1
    pts = pts[:n]
    
    # 2. Construct initial variable vector: [x1, y1, r1, x2, y2, r2, ...]
    x0 = []
    for p in pts:
        x0.extend(p)
        x0.append(0.05)  # Start with safe small radii
    x0 = np.array(x0)
    
    # Bounds for variables: x, y in [0,1], r in [0,1]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 1.0)] * n
    
    constraints = [
        {'type': 'ineq', 'fun': con_bnd, 'args': (n,)},
        {'type': 'ineq', 'fun': con_ovl, 'args': (n,)}
    ]
    
    best_sum_r = -1.0
    best_res = None
    
    np.random.seed(42)
    # 3. Run optimization with multiple perturbations to escape local minima
    for _ in range(5):
        curr_x0 = x0.copy()
        # Perturb centers slightly
        curr_x0[:2*n] += np.random.normal(0, 0.02, size=2*n)
        curr_x0[2*n:] = 0.05  # Reset radii to safe start
        
        # Clip centers to valid range
        curr_x0[0::3] = np.clip(curr_x0[0::3], 0.005, 0.995)
        curr_x0[1::3] = np.clip(curr_x0[1::3], 0.005, 0.995)
        
        try:
            res = minimize(obj, curr_x0, args=(n,), method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 1500, 'ftol': 1e-9})
            current_sum_r = -res.fun
            if current_sum_r > best_sum_r:
                best_sum_r = current_sum_r
                best_res = res
        except Exception:
            continue
            
    if best_res is None:
        # Fallback to initial configuration if optimization fails entirely
        centers = np.array(pts)
        radii = np.full(n, 0.05)
        return centers, radii, float(np.sum(radii))
        
    centers = best_res.x[:2*n].reshape(n, 2)
    radii = best_res.x[2*n:]
    
    # 4. Minor post-processing to guarantee strict validity against floating-point noise
    # Shrink radii by a negligible amount if any constraint is borderline
    max_violation = 0.0
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        # Check boundaries
        margin_x = min(x, 1.0 - x)
        margin_y = min(y, 1.0 - y)
        if margin_x - r < -1e-9: max_violation = max(max_violation, r - margin_x + 1e-9)
        if margin_y - r < -1e-9: max_violation = max(max_violation, r - margin_y + 1e-9)
        
        # Check overlaps
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist - (radii[i] + radii[j]) < -1e-9:
                max_violation = max(max_violation, (radii[i] + radii[j]) - dist + 1e-9)
                
    if max_violation > 1e-10:
        radii = radii * (1.0 - max_violation / (radii.max() + 1e-12))
        
    return centers, radii, float(np.sum(radii))
