import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective_penalized(v, mu):
    """Objective function: maximize sum of radii with penalty for constraint violations."""
    X = v[0::3]
    Y = v[1::3]
    R = v[2::3]
    
    # Boundary violations: x-r >= 0, x+r <= 1, y-r >= 0, y+r <= 1
    v_bound = (np.maximum(0, R - X)**2 + 
               np.maximum(0, R + X - 1)**2 + 
               np.maximum(0, R - Y)**2 + 
               np.maximum(0, R + Y - 1)**2)
               
    # Pairwise overlap violations: dist >= r_i + r_j
    diff_X = X[:, None] - X[None, :]
    diff_Y = Y[:, None] - Y[None, :]
    D = np.sqrt(diff_X**2 + diff_Y**2)
    RS = R[:, None] + R[None, :]
    
    viol = np.maximum(0, RS - D)
    np.fill_diagonal(viol, 0.0)
    
    pen = np.sum(v_bound) + np.sum(viol**2)
    return -np.sum(R) + mu * pen

def compute_penalty(v):
    """Helper to compute only the penalty value."""
    X = v[0::3]
    Y = v[1::3]
    R = v[2::3]
    
    v_bound = (np.maximum(0, R - X)**2 + 
               np.maximum(0, R + X - 1)**2 + 
               np.maximum(0, R - Y)**2 + 
               np.maximum(0, R + Y - 1)**2)
               
    diff_X = X[:, None] - X[None, :]
    diff_Y = Y[:, None] - Y[None, :]
    D = np.sqrt(diff_X**2 + diff_Y**2)
    RS = R[:, None] + R[None, :]
    
    viol = np.maximum(0, RS - D)
    np.fill_diagonal(viol, 0.0)
    
    return np.sum(v_bound) + np.sum(viol**2)

def get_constraints(v):
    """Returns array of inequality constraints fun(x) >= 0 for SLSQP."""
    X = v[0::3]
    Y = v[1::3]
    R = v[2::3]
    
    c1 = X - R
    c2 = 1.0 - X - R
    c3 = Y - R
    c4 = 1.0 - Y - R
    
    diff_X = X[:, None] - X[None, :]
    diff_Y = Y[:, None] - Y[None, :]
    D = np.sqrt(diff_X**2 + diff_Y**2)
    RS = R[:, None] + R[None, :]
    
    idx = np.triu_indices(N_CIRCLES, k=1)
    c5 = D[idx] - RS[idx]
    
    return np.concatenate([c1, c2, c3, c4, c5])

def run_packing():
    best_v = None
    best_sum_r = -np.inf
    
    for restart in range(8):
        # Initialize centers on a grid with sinusoidal perturbation
        centers = np.zeros((N_CIRCLES, 2))
        idx = 0
        for r in range(5):
            for c in range(5):
                if idx >= N_CIRCLES: break
                cx = 0.1 + c * 0.2 + 0.03 * np.sin(idx + restart)
                cy = 0.1 + r * 0.2 + 0.03 * np.cos(idx + restart)
                centers[idx] = [cx, cy]
                idx += 1
        if idx < N_CIRCLES:
            centers[idx] = [0.5, 0.5]
            
        v0 = np.zeros(3 * N_CIRCLES)
        v0[0::3] = centers[:, 0]
        v0[1::3] = centers[:, 1]
        v0[2::3] = 0.02
        
        bounds = [(0, 1)] * N_CIRCLES + [(0, 1)] * N_CIRCLES + [(0, 0.5)] * N_CIRCLES
        
        # Phase 1: Fast expansion with penalty method
        v_curr = v0.copy()
        mu = 100.0
        for step in range(20):
            res = minimize(objective_penalized, v_curr, args=(mu,), 
                           method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 1500, 'ftol': 1e-12})
            v_curr = res.x
            pen = compute_penalty(v_curr)
            if pen < 1e-10:
                break
            mu *= 2.0
            
        # Phase 2: Strict constraint satisfaction with SLSQP
        cons = {'type': 'ineq', 'fun': get_constraints}
        try:
            res2 = minimize(objective_penalized, v_curr, args=(1e6,), 
                            method='SLSQP', bounds=bounds, constraints=cons,
                            options={'maxiter': 2000, 'ftol': 1e-12})
            v_curr = res2.x
        except Exception:
            pass 
            
        sum_r = np.sum(v_curr[2::3])
        if sum_r > best_sum_r:
            best_sum_r = sum_r
            best_v = v_curr.copy()
            
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3]
    return centers, radii, np.sum(radii)
