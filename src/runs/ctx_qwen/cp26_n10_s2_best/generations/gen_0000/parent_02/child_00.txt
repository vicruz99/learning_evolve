import numpy as np
from scipy.optimize import minimize
from functools import partial
import math

def objective_func(x, n):
    return -np.sum(x[2::3])

def penalized_objective(x, n, penalty):
    obj = -np.sum(x[2::3])
    
    for i in range(n):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        if ri - xi > 0:
            obj += penalty * (ri - xi) ** 2
        if xi + ri - 1 > 0:
            obj += penalty * (xi + ri - 1) ** 2
        if ri - yi > 0:
            obj += penalty * (ri - yi) ** 2
        if yi + ri - 1 > 0:
            obj += penalty * (yi + ri - 1) ** 2
    
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
            xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dist = math.sqrt(dx*dx + dy*dy)
            overlap = ri + rj - dist
            if overlap > 0:
                obj += penalty * overlap ** 2
    
    return obj

def flatten(centers, radii):
    n = centers.shape[0]
    x = np.zeros(3 * n)
    for i in range(n):
        x[3*i] = centers[i, 0]
        x[3*i+1] = centers[i, 1]
        x[3*i+2] = radii[i]
    return x

def unflatten(x, n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = x[3*i]
        centers[i, 1] = x[3*i+1]
        radii[i] = x[3*i+2]
    return centers, radii

def initialize_hexagonal(n):
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.06)
    
    r = 0.06
    col_spacing = 2 * r * 1.15
    row_spacing = col_spacing * math.sqrt(3) / 2
    
    idx = 0
    for row in range(10):
        y = r + row * row_spacing
        n_cols = 6 if row % 2 == 0 else 5
        offset = 0 if row % 2 == 0 else col_spacing / 2
        for col in range(n_cols):
            if idx >= n:
                break
            x = r + col * col_spacing + offset
            if x + r <= 1 and y + r <= 1:
                centers[idx] = [x, y]
                radii[idx] = r
                idx += 1
        if idx >= n:
            break
    
    return centers, radii

def initialize_corner_edge_center(n):

    radii = np.zeros(n)
    
    idx = 0
    
    # 4 corner circles
    r_corner = 0.12
    corners = [[r_corner, r_corner], [1-r_corner, r_corner], [r_corner, 1-r_corner], [1-r_corner, 1-r_corner]]

        centers[idx] = c
        radii[idx] = r_corner
        idx += 1

    edges = []

        edges.append([t, r_edge])
        edges.append([t, 1-r_edge])
        edges.append([r_edge, t])
        edges.append([1-r_edge, t])
    for e in edges:
        centers[idx] = e
        radii[idx] = r_edge
        idx += 1
    
    # Remaining circles in center
    r_center = 0.06
    grid_positions = []
    for i in range(4):
        for j in range(4):
            x = 0.25 + i * 0.15
            y = 0.25 + j * 0.15
            grid_positions.append([x, y])
    
    for gp in grid_positions:
        if idx >= n:
            break
        centers[idx] = gp
        radii[idx] = r_center
        idx += 1
    
    # Fill remaining if any
    while idx < n:
        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.01
        idx += 1
    
    return centers, radii

def optimize_with_penalty(x0, n):
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    
    x = x0.copy()
    penalty = 50
    
    for iteration in range(30):
        result = minimize(penalized_objective, x, args=(n, penalty), method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-15})
        x = result.x
        penalty = min(penalty * 1.5, 1000000)
    
    return x

def constraint_x_min_r(x, i):
    return x[3*i] - x[3*i+2]

def constraint_x_max_r(x, i):
    return 1 - x[3*i] - x[3*i+2]

def constraint_y_min_r(x, i):
    return x[3*i+1] - x[3*i+2]

def constraint_y_max_r(x, i):
    return 1 - x[3*i+1] - x[3*i+2]

def constraint_nonoverlap(x, i, j):
    xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
    xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
    return (xi-xj)**2 + (yi-yj)**2 - (ri+rj)**2

def get_constraints(n):
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': partial(constraint_x_min_r, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_x_max_r, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_y_min_r, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_y_max_r, i=i)})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': partial(constraint_nonoverlap, i=i, j=j)})
    return constraints

def refine_with_slsqp(x, n):
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    constraints = get_constraints(n)
    
    result = minimize(objective_func, x, args=(n,), method='SLSQP', bounds=bounds,
                     constraints=constraints, options={'maxiter': 3000, 'ftol': 1e-15})
    return result.x

def run_packing():

        centers, radii = unflatten(x, n)
        s = np.sum(radii)
        
        if s > best_sum:
            best_sum = s
            best_x = x.copy()
    
    centers, radii = unflatten(best_x, n)
    radii = np.maximum(radii, 0)
    
    return centers, radii, float(np.sum(radii))
