import numpy as np
from scipy.optimize import linprog

def get_pair_indices(n):
    """Returns list of (i, j) tuples for all unique pairs with i < j."""
    return [(i, j) for i in range(n) for j in range(i + 1, n)]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    N = 26
    pair_indices = get_pair_indices(N)
    M = len(pair_indices)
    
    # 1. Initialization: Grid layout with slight random jitter
    centers = np.zeros((N, 2))
    xs = np.linspace(0.15, 0.85, 5)
    ys = np.linspace(0.15, 0.85, 5)
    idx = 0
    for y in ys:
        for x in xs:
            if idx < N:
                centers[idx] = [x, y]
                idx += 1
    while idx < N:
        centers[idx] = rng.uniform(0.2, 0.8, 2)
        idx += 1
        
    # Preallocate LP structures
    c_obj = -np.ones(N)
    A_ub = np.zeros((M + N, N))
    b_ub = np.zeros(M + N)
    for k, (i, j) in enumerate(pair_indices):
        A_ub[k, i] = 1
        A_ub[k, j] = 1
        
    best_sum = -1.0
    best_centers = centers.copy()
    best_radii = np.zeros(N)
    
    lr = 0.03
    # 2. Optimization Loop
    for step in range(6000):
        # Update LP bounds based on current geometry
        for k, (i, j) in enumerate(pair_indices):
            b_ub[k] = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1]) + 1e-12
            
        for i in range(N):
            x, y = centers[i]
            b_ub[M + i] = min(x, 1.0 - x, y, 1.0 - y) + 1e-12
            
        # Solve LP: Maximize sum of radii for fixed positions
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * N, method='highs')
        
        if not res.success:
            continue
            
        radii = res.x
        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
            
        # 3. Compute Gradient Forces from LP Duals
        forces = np.zeros_like(centers)
        marginals = res.ineqlin.marginals
        
        if marginals is not None:
            for k, (i, j) in enumerate(pair_indices):
                mu = marginals[k]
                if mu > 1e-7:
                    d = b_ub[k] - 1e-12 + 1e-9
                    dx = (centers[i,0] - centers[j,0]) / d
                    dy = (centers[i,1] - centers[j,1]) / d
                    forces[i] += mu * np.array([dx, dy])
                    forces[j] -= mu * np.array([dx, dy])
                    
            # Wall repulsion heuristic to push circles away from boundaries
            for i in range(N):
                x, y = centers[i]
                r = radii[i]
                mx = min(x, 1.0 - x) - r
                my = min(y, 1.0 - y) - r
                if mx < 0.02:
                    forces[i, 0] += 1.0 * (0.5 - x)
                if my < 0.02:
                    forces[i, 1] += 1.0 * (0.5 - y)
        
        # Update positions
        centers += lr * forces
        centers = np.clip(centers, 0.005, 0.995)
        lr *= 0.998
        
        # Periodic noise to escape local minima
        if step % 1000 == 500:
            centers += rng.normal(0, 0.008, centers.shape)
            centers = np.clip(centers, 0.01, 0.99)
            lr = max(lr, 0.002)
            
    return best_centers, best_radii, best_sum
