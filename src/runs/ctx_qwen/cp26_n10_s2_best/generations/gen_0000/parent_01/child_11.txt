import numpy as np
import math
from scipy.optimize import minimize, basinhopping

def get_penalty(centers, radii, weight=1000.0):
    """
    Calculates the penalty for constraint violations.
    Constraints:
    1. Circles must not overlap: dist(c_i, c_j) >= r_i + r_j
    2. Circles must be inside [0,1]x[0,1]: r_i <= x_i <= 1-r_i, r_i <= y_i <= 1-r_i
    """
    n = len(radii)
    penalty = 0.0
    
    # 1. Circle-Circle Overlap Penalty
    # Compute pairwise distances using broadcasting
    # centers shape: (n, 2)
    # diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Compute sum of radii
    # r_sum shape: (n, n)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Violation: r_i + r_j - dist_ij > 0
    violation = r_sum - dists
    
    # Zero out diagonal (self-distance is 0, self-overlap check is not needed, 
    # but mathematically r_i + r_i > 0, so we must ignore i==j)
    np.fill_diagonal(violation, 0)
    
    # Apply max(0, violation) to only penalize overlaps
    overlap = np.maximum(0, violation)
    penalty += np.sum(overlap**2)
    
    # 2. Boundary Penalty
    # x_i >= r_i  => r_i - x_i <= 0
    # x_i <= 1 - r_i => x_i + r_i - 1 <= 0
    # Same for y
    
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    # Violations
    v_left = r - x
    v_right = x + r - 1
    v_bottom = r - y
    v_top = y + r - 1
    
    # Stack and apply max(0, v)
    wall_violations = np.maximum(0, np.stack([v_left, v_right, v_bottom, v_top], axis=1))
    penalty += np.sum(wall_violations**2)
    
    return penalty * weight

def objective_function(params, weight):
    """
    Objective function to minimize: -sum(radii) + penalty
    """
    n = 26
    # Reshape params from flat array to (26, 3)
    # Structure: [x1, y1, r1, x2, y2, r2, ...] or similar?
    # Let's use structure: centers (26, 2) then radii (26,)
    # Total 78 params.
    
    # Actually, L-BFGS-B works better if bounds are simple.
    # Let's keep the structure: x coords, y coords, radii.
    # Or interleaved? Let's do interleaved for easier indexing if needed, 
    # but separate blocks is easier for slicing.
    # params[0:52] -> centers (26, 2)
    # params[52:78] -> radii (26,)
    
    centers = params[:52].reshape(26, 2)
    radii = params[52:]
    
    # Ensure radii are non-negative (handled by bounds, but good to be safe)
    radii = np.maximum(radii, 1e-9)
    
    pen = get_penalty(centers, radii, weight)
    
    # We want to maximize sum of radii, so minimize negative sum
    return -np.sum(radii) + pen

def generate_hexagonal_initial_guess(n=26):
    """
    Generates an initial guess for centers and radii based on a hexagonal lattice.
    """
    centers = []
    radii = []
    
    # Estimate a starting radius. 
    # For N=26 in unit square, average radius ~ 0.1.
    # Start small to allow optimizer to expand.
    r_start = 0.05
    
    # Hexagonal grid parameters
    # Spacing s = 2 * r_approx
    # Let's try to fit points with spacing ~ 0.2
    s = 0.21
    
    row = 0
    y = 0.15
    while y <= 0.90: # Limit y to fit within square roughly
        # Offset for alternating rows
        offset = (row % 2) * (s / 2.0)
        
        x = 0.15 + offset
        while x <= 0.90:
            if len(centers) < n:
                centers.append([x, y])
                radii.append(r_start)
            else:
                break
            x += s
        
        # Vertical step for hexagonal packing: s * sqrt(3) / 2
        y += s * math.sqrt(3) / 2
        row += 1
        
    # If we didn't get enough points (unlikely with these bounds), add random ones
    while len(centers) < n:
        centers.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
        radii.append(r_start)
    
    # If we got too many (shouldn't happen with the break), trim
    centers = centers[:n]
    radii = radii[:n]
    
    return np.array(centers), np.array(radii)

def run_packing():
    n = 26
    
    # Generate initial guess
    init_centers, init_radii = generate_hexagonal_initial_guess(n)
    
    # Flatten to a single vector for scipy
    # Order: x1, y1, x2, y2, ..., r1, r2, ...
    # Wait, my objective function expects [centers flattened, radii flattened]
    # Let's stick to that.
    x0 = np.concatenate([init_centers.flatten(), init_radii])
    
    # Define bounds
    # For each circle i:
    # x_i in [0, 1]
    # y_i in [0, 1]
    # r_i in [0, 0.5] (can't be larger than 0.5)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
    for i in range(n):
        bounds.append((1e-6, 0.5)) # r (small lower bound to avoid 0 radius issues)
        
    # Set penalty weight. Needs to be high enough to enforce constraints.
    # If overlap is 0.01, penalty contribution ~ 1e-4.
    # Gradient of -sum(r) is -1.
    # We need penalty gradient to be comparable or larger when active.
    # d(Penalty)/dr ~ 2 * overlap * weight.
    # If overlap ~ 0.01, weight=1000 -> grad ~ 20. Strong enough.
    penalty_weight = 5000.0
    
    # Use basinhopping to explore the landscape and escape local minima
    # Minimzer inside basinhopping will be L-BFGS-B
    minimizer_kwargs = {
        "method": "L-BFGS-B",
        "bounds": bounds,
        "options": {"maxiter": 200} # Iterations per local minimization
    }
    
    # Define a callback to track progress? Not needed for return.
    
    # Run basinhopping
    # niter controls number of kicks. 50 is a reasonable number for this dimensionality.
    # Higher might be better but slower.
    optimizer = basinhopping(
        lambda p: objective_function(p, penalty_weight),
        x0,
        minimizer_kwargs=minimizer_kwargs,
        niter=80,
        stepsize=0.05, # Step size for perturbation
        interval=50,   # Print interval (ignored here)
        disp=False
    )
    
    # Extract result
    best_params = optimizer.x
    
    # Reshape back
    best_centers = best_params[:52].reshape(n, 2)
    best_radii = best_params[52:]
    
    # Validate and clean up tiny radii if any
    best_radii = np.maximum(best_radii, 1e-9)
    
    # Ensure centers are strictly within bounds [0,1] for safety against float errors
    # Clamping might slightly increase overlap, but validation handles 1e-12 tolerance.
    # Actually, clamping is safe if we ensure radii are small enough.
    # Let's trust the optimizer bounds.
    
    sum_radii = np.sum(best_radii)
    
    return best_centers, best_radii, sum_radii
