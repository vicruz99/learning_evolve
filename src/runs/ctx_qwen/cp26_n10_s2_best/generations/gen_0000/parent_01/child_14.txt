

Looking at this problem, I need to pack 26 circles in a unit square to maximize the sum of radii, with a target of 2.636.

**Key insights:**
1. A 5×5 grid of equal circles gives radius 0.1 and sum 2.5, leaving room for one more circle
2. Hexagonal packing is more area-efficient than square grids
3. I need to use optimization to find the optimal configuration
4. Multiple restarts with different initializations will help escape local minima

**Strategy:**
- Use scipy.optimize with SLSQP to maximize sum of radii
- Initialize with multiple configurations (hexagonal, perturbed grid)
- Use proper constraint formulation for boundary and non-overlap
- Add post-processing to validate and repair the solution

```python
import numpy as np
from scipy.optimize import minimize
import warnings

N = 26

def objective(v):
    return -np.sum(v[2*N:])

def constraint_x_lower_factory(i):
    def f(v):
        return v[i] - v[2*N + i]
    return f

def constraint_x_upper_factory(i):
    def f(v):
        return 1.0 - v[i] - v[2*N + i]
    return f

def constraint_y_lower_factory(i):
    def f(v):
        return v[N + i] - v[2*N + i]
    return f

def constraint_y_upper_factory(i):
    def f(v):
        return 1.0 - v[N + i] - v[2*N + i]
    return f

def constraint_no_overlap_factory(i, j):
    def f(v):
        dx = v[i] - v[j]
        dy = v[N + i] - v[N + j]
        return np.sqrt(dx*dx + dy*dy) - v[2*N + i] - v[2*N + j]
    return f

def constraint_radius_nonneg_factory(i):
    def f(v):
        return v[2*N + i]
    return f

def init_hexagonal_grid(n):
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    while idx < n:
        if row % 2 == 0:
            n_in_row = 5
            x_offset = 0.12
        else:
            n_in_row = 5 if row < 4 else 6
            x_offset = 0.18
        
        for col in range(n_in_row):
            if idx >= n:
                break
            centers[idx, 0] = x_offset + col * 0.17
            centers[idx, 1] = row * 0.13 + 0.07
            idx += 1
        row += 1
    return centers

def init_perturbed_grid(n):
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx >= n:
                break
            centers[idx, 0] = 0.1 + i * 0.2
            centers[idx, 1] = 0.1 + j * 0.2
            idx += 1
    
    if idx < n:
        centers[idx, 0] = 0.05
        centers[idx, 1] = 0.05
        idx += 1
    
    np.random.seed(42)
    centers += np.random.randn(n, 2) * 0.02
    return centers

def init_compact_hexagonal(n):
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    while idx < n:
        if row % 2 == 0:
            n_in_row = 5
        else:
            n_in_row = 5 if row < 4 else 6
        
        for col in range(n_in_row):
            if idx >= n:
                break
            if row % 2 == 1:
                centers[idx, 0] = 0.06 + col * 0.175
            else:
                centers[idx, 0] = 0.11 + col * 0.175
            centers[idx, 1] = 0.07 + row * 0.155
            idx += 1
        row += 1
    return centers

def build_constraints(n):
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': constraint_x_lower_factory(i)})
        constraints.append({'type': 'ineq', 'fun': constraint_x_upper_factory(i)})
        constraints.append({'type': 'ineq', 'fun': constraint_y_lower_factory(i)})
        constraints.append({'type': 'ineq', 'fun': constraint_y_upper_factory(i)})
    
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_no_overlap_factory(i, j)})
    
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': constraint_radius_nonneg_factory(i)})
    
    return constraints

def optimize_from_init(centers, radii, constraints, verbose=False):
    n = N
    x0 = np.zeros(3 * n)
    x0[:n] = centers[:, 0]
    x0[n:2*n] = centers[:, 1]
    x0[2*n:] = radii
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        result = minimize(
            objective,
            x0,
            constraints=constraints,
            bounds=bounds,
            method='SLSQP',
            options={'maxiter': 8000, 'ftol': 1e-14, 'disp': verbose}
        )
    
    return result

def repair_solution(centers, radii):
    n = N
    centers = centers.copy()
    radii = radii.copy()
    
    for i in range(n):
        r = radii[i]
        centers[i, 0] = max(r, min(1.0 - r, centers[i, 0]))
        centers[i, 1] = max(r, min(1.0 - r, centers[i, 1]))
        radii[i] = max(0.0, radii[i])
    
    return centers, radii

def run_packing():
    n = N
    constraints = build_constraints(n)
    
    initializations = []
    
    hex_centers = init_hexagonal_grid(n)
    hex_radii = np.ones(n) * 0.06
    initializations.append((hex_centers.copy(), hex_radii.copy()))
    
    grid_centers = init_perturbed_grid(n)
    grid_radii = np.ones(n) * 0.06
    initializations.append((grid_centers.copy(), grid_radii.copy()))
    
    compact_centers = init_compact_hexagonal(n)
    compact_radii = np.ones(n) * 0.06
    initializations.append((compact_centers.copy(), compact_radii.copy()))
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    for idx, (centers, radii) in enumerate(initializations):
        result = optimize_from_init(centers, radii, constraints)
        
        centers_opt = np.column_stack([result.x[:n], result.x[n:2*n]])
        radii_opt = result.x[2*n:]
        
        centers_opt, radii_opt = repair_solution(centers_opt, radii_opt)
        
        current_sum = np.sum(radii_opt)
        print(f"Init {idx}: sum_radii = {current_sum:.6f}, success={result.success}, fun={result.fun}")
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    if best_centers is not None:
        return best_centers, best_radii, best_sum
    
    return np.zeros((n, 2)), np.ones(n) * 0.01, 0.0
```
