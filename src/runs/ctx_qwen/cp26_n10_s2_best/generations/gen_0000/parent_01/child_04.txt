

To maximize the sum of radii for 26 circles in a unit square, we treat this as a non-linear programming problem. We initialize the circles in a structured grid, then use a penalty-based optimization strategy with `scipy.optimize.minimize`. This allows the circles to expand and find optimal positions without strict constraints that might cause solver failures.

### Strategy
1.  **Initialization**: Start with a small radius ($0.01$) and place the 26 circles in a regular grid. This ensures a valid starting point.
2.  **Penalty-Based Objective**: Define an objective function that minimizes the negative sum of radii (equivalent to maximizing the sum) plus a penalty term. The penalty term squares any violations of boundary and non-overlap constraints.
3.  **Optimization**: Use the L-BFGS-B algorithm, which supports bounds (keeping coordinates and radii non-negative). This helps the optimizer find a configuration where circles are as large as possible while respecting the geometry of the square.
4.  **Validation**: The final output is validated against the provided function to ensure it is physically valid before returning.

```python
import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def run_packing():
    N = 26
    # 1. Initialization: Start with a small radius and grid layout
    # A 6x5 grid provides a good distribution
    rows = 5
    cols = 6
    x_coords = np.linspace(0.15, 0.85, cols)
    y_coords = np.linspace(0.15, 0.85, rows)
    
    initial_centers = []
    initial_radii = []
    for r in range(rows):
        for c in range(cols):
            if len(initial_centers) < N:
                initial_centers.append([x_coords[c], y_coords[r]])
                initial_radii.append(0.01)
            else:
                break

    initial_centers = np.array(initial_centers)
    initial_radii = np.array(initial_radii)

    # Reshape to a single vector for the optimizer: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.empty(3 * N)
    for i in range(N):
        x0[3*i] = initial_centers[i, 0]
        x0[3*i+1] = initial_centers[i, 1]
        x0[3*i+2] = initial_radii[i]

    # 2. Bounds: [x_min, x_max], [y_min, y_max], [r_min, r_max]
    bounds = []
    for _ in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    # 3. Objective and Penalty Function
    def objective(vars):
        centers = vars[0::3].reshape(N, 2)
        radii = vars[2::3]

        # Negative sum of radii (to be minimized)
        obj = -np.sum(radii)

        # Penalty for boundary violations
        penalty = 0.0
        # Left/Right boundaries
        penalty += np.sum(np.maximum(0, radii - centers[:, 0])**2)
        penalty += np.sum(np.maximum(0, radii - (1.0 - centers[:, 0]))**2)
        # Top/Bottom boundaries
        penalty += np.sum(np.maximum(0, radii - centers[:, 1])**2)
        penalty += np.sum(np.maximum(0, radii - (1.0 - centers[:, 1]))**2)

        # Penalty for overlap violations
        for i in range(N):
            for j in range(i + 1, N):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    penalty += overlap**2

        return obj + 10000.0 * penalty

    # 4. Optimization
    # Using L-BFGS-B because it handles bounds and is efficient for medium-sized problems
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds)

    # 5. Extract and Validate
    final_vars = result.x
    final_centers = np.array([[final_vars[3*i], final_vars[3*i+1]] for i in range(N)])
    final_radii = final_vars[2::3]

    # Clean up tiny negative radii due to numerical precision
    final_radii = np.maximum(final_radii, 1e-12)
    sum_radii = np.sum(final_radii)

    # Validation check
    if validate_packing(final_centers, final_radii):
        return final_centers, final_radii, sum_radii
    else:
        # Fallback if optimization drifted out of bounds (highly unlikely with penalty)
        return initial_centers, initial_radii, 0.0

if __name__ == "__main__":
    centers, radii, s = run_packing()
    print(f"Sum of radii: {s}")
```
