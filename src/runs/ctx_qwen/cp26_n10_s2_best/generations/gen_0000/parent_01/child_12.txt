import numpy as np
from scipy.optimize import minimize

# Number of circles to pack
N_CIRCLES = 26

def objective_func(x):
    """
    Objective function to minimize.
    We want to maximize sum of radii, so we minimize negative sum.
    x is a flat array of shape (3*N,) containing [x0, y0, r0, x1, y1, r1, ...]
    """
    radii = x[2::3]
    return -np.sum(radii)

def compute_constraints(x, N):
    """
    Computes inequality constraints g(x) >= 0.
    Includes boundary constraints and non-overlap constraints.
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    # 1. x >= r  => x - r >= 0
    # 2. x <= 1 - r => 1 - x - r >= 0
    # 3. y >= r  => y - r >= 0
    # 4. y <= 1 - r => 1 - y - r >= 0
    c_bound_x1 = cx - r
    c_bound_x2 = 1.0 - cx - r
    c_bound_y1 = cy - r
    c_bound_y2 = 1.0 - cy - r
    
    # Collect boundary constraints
    all_ineq = np.concatenate([c_bound_x1, c_bound_x2, c_bound_y1, c_bound_y2])

    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # Vectorized computation for efficiency
    # Compute pairwise squared distances between centers
    # cx_diff shape (N, N)
    cx_diff = cx[:, None] - cx[None, :]
    cy_diff = cy[:, None] - cy[None, :]
    dist_sq = cx_diff**2 + cy_diff**2
    
    # Compute pairwise sum of radii squared
    r_sum = r[:, None] + r[None, :]
    r_sum_sq = r_sum**2

    overlaps = margin[mask]
    
    all_ineq = np.concatenate([all_ineq, overlaps])
    
    return all_ineq

        bounds.append((0.0, 1.0)) # y
