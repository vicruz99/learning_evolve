import numpy as np
import math
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    # Check for NaN values
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False
    # Check if radii are nonnegative
    for i in range(n):
        if radii[i] < 0:
            return False
        elif np.isnan(radii[i]):
            return False
    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True

def get_constraints_and_bounds(n):
    """
    Prepare bounds and initial guess structure for scipy optimization.
    Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    """
    bounds = []
    for i in range(n):
        # x in [0, 1]
        bounds.append((0.0, 1.0))
        # y in [0, 1]
        bounds.append((0.0, 1.0))
        # r >= 0. We can cap it at 0.5
        bounds.append((0.0, 0.5))
    return bounds

def objective_neg_sum_radii(vars, n):
    """
    Objective function to minimize: Negative sum of radii.
    vars is a flat array of size 3*n: [x1, y1, r1, ...]
    """
    radii = vars[2::3] # Every 3rd element starting from index 2
    return -np.sum(radii)

def penalty_function(vars, n):
    """
    Calculates a penalty based on constraint violations.
    This is used for the force-based approach or as a constraint in minimization.
    However, for SLSQP, we usually define constraints separately.
    Here we return the violation magnitude to be used in a custom loop if needed,
    but we will use explicit constraints in the optimizer.
    """
    penalty = 0.0
    for i in range(n):
        xi = vars[3*i]
        yi = vars[3*i+1]
        ri = vars[3*i+2]
        
        # Boundary penalties (soft)
        # We want xi - ri >= 0 => ri - xi <= 0
        if ri - xi > 0: penalty += (ri - xi)**2
        # xi + ri <= 1 => xi + ri - 1 <= 0
        if xi + ri - 1 > 0: penalty += (xi + ri - 1)**2
        # yi - ri >= 0
        if ri - yi > 0: penalty += (ri - yi)**2
        # yi + ri <= 1
        if yi + ri - 1 > 0: penalty += (yi + ri - 1)**2

    # Overlap penalties
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        for j in range(i+1, n):
            xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
            
            dist = math.sqrt((xi - xj)**2 + (yi - yj)**2)
            min_dist = ri + rj
            if dist < min_dist:
                violation = min_dist - dist
                penalty += violation**2
                
    return penalty

def optimization_constraints(vars, n):
    """
    Returns a list of constraint violations for SLSQP.
    SLSQP expects constraints of the form fun(x) >= 0 (ineq) or fun(x) == 0 (eq).
    
    Constraints:
    1. xi - ri >= 0
    2. 1 - xi - ri >= 0
    3. yi - ri >= 0
    4. 1 - yi - ri >= 0
    5. dist(i,j) - (ri + rj) >= 0
    """
    cons = []
    
    # Boundary constraints
    for i in range(n):
        xi = vars[3*i]
        yi = vars[3*i+1]
        ri = vars[3*i+2]
        
        # x >= r  => x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx] - v[3*idx+2]})
        # 1 - x >= r => 1 - x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1.0 - v[3*idx] - v[3*idx+2]})
        # y >= r
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx+1] - v[3*idx+2]})
        # 1 - y >= r
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1.0 - v[3*idx+1] - v[3*idx+2]})

    # Overlap constraints
    # Note: Passing index pairs in lambda can be tricky with closures, 
    # but since n is small (26), we can generate them.
    # To avoid closure issues, we use default arguments.
    for i in range(n):
        for j in range(i+1, n):

                xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
                xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
                d = math.sqrt((xi - xj)**2 + (yi - yj)**2)
                return d - (ri + rj)
            cons.append({'type': 'ineq', 'fun': dist_constraint})
            
    return cons

def initialize_hexagonal(n):
    """
    Initialize centers in a hexagonal lattice pattern.
    """
    centers = []
    radii = []
    
    # Estimate radius for equal circles to fit n in square
    # Area ~ 1, density ~ 0.9 -> n * pi * r^2 ~ 0.9 -> r ~ sqrt(0.9 / (n*pi))
    r_est = math.sqrt(0.85 / (n * math.pi)) 
    
    # Hexagonal grid parameters
    # Row height: r * sqrt(3)
    # Col width: 2 * r
    
    # We will try to fit points
    # Let's just place them in a grid and let optimizer move them
    # A simple grid is safer to start than a packed hex grid which might be invalid immediately
    
    # Let's try a 5x6 grid (30 points) and remove 4? 
    # Or just random valid placement.
    
    # Better: Spiral placement or just dense grid
    side = int(math.ceil(math.sqrt(n)))
    step_x = 1.0 / (side + 1)
    step_y = 1.0 / (side + 1)
    
    count = 0
    for i in range(1, side + 1):
        for j in range(1, side + 1):
            if count < n:
                x = i * step_x
                y = j * step_y
                centers.append([x, y])
                radii.append(r_est) # Start with a reasonable guess
                count += 1
                
    # Pad if needed (though loop covers it)
    while len(centers) < n:
        centers.append([0.5, 0.5])
        radii.append(r_est)
        
    return np.array(centers), np.array(radii)
