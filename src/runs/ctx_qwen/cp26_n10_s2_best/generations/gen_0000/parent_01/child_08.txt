import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26


def objective(vars_flat):
    radii = vars_flat[2 * N_CIRCLES:]
    return -np.sum(radii)


def boundary_constraints(vars_flat):
    constraints = np.zeros(4 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x = vars_flat[3 * i]
        y = vars_flat[3 * i + 1]
        r = vars_flat[3 * i + 2]
        constraints[4 * i] = x - r
        constraints[4 * i + 1] = 1.0 - x - r
        constraints[4 * i + 2] = y - r
        constraints[4 * i + 3] = 1.0 - y - r
    return constraints


def non_overlap_constraints(vars_flat):
    n_constraints = N_CIRCLES * (N_CIRCLES - 1) // 2
    constraints = np.zeros(n_constraints)
    idx = 0
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = vars_flat[3 * i] - vars_flat[3 * j]
            dy = vars_flat[3 * i + 1] - vars_flat[3 * j + 1]
            dist = np.sqrt(dx * dx + dy * dy)
            constraints[idx] = dist - vars_flat[3 * i + 2] - vars_flat[3 * j + 2]
            idx += 1
    return constraints


def create_hexagonal_lattice(r_init):
    centers = []
    row = 0
    while len(centers) < N_CIRCLES:
        y = r_init + row * np.sqrt(3) * r_init
        if y + r_init > 1.0:
            break
        col = 0
        x_offset = (row % 2) * r_init
        while len(centers) < N_CIRCLES:
            x = r_init + x_offset + col * 2 * r_init
            if x + r_init > 1.0:
                break
            centers.append([x, y])
            col += 1
        row += 1
    while len(centers) < N_CIRCLES:
        cx = 0.5 + (len(centers) - N_CIRCLES + 1) * 0.01
        cy = 0.5 + (len(centers) - N_CIRCLES + 1) * 0.01
        centers.append([cx, cy])
    return centers[:N_CIRCLES]


def create_grid_lattice(r_init):
    centers = []
    rows = 5
    cols = 6
    row_idx = 0
    col_idx = 0
    while len(centers) < N_CIRCLES:
        x = r_init + col_idx * (1.0 - 2 * r_init) / max(cols - 1, 1)
        y = r_init + row_idx * (1.0 - 2 * r_init) / max(rows - 1, 1)
        centers.append([x, y])
        col_idx += 1
        if col_idx >= cols:
            col_idx = 0
            row_idx += 1
    return centers[:N_CIRCLES]


def create_strategic_initial(r_init):
    centers = []
    centers.append([r_init, r_init])
    centers.append([1.0 - r_init, r_init])
    centers.append([r_init, 1.0 - r_init])
    centers.append([1.0 - r_init, 1.0 - r_init])
    centers.append([0.5, 0.5])
    
    for angle in np.linspace(0, 2 * np.pi, 8, endpoint=False):
        cx = 0.5 + 0.25 * np.cos(angle)
        cy = 0.5 + 0.25 * np.sin(angle)
        cx = np.clip(cx, r_init, 1.0 - r_init)
        cy = np.clip(cy, r_init, 1.0 - r_init)
        centers.append([cx, cy])
    
    for i in range(3):
        for j in range(3):
            if len(centers) >= N_CIRCLES:
                break
            cx = 0.25 + i * 0.25
            cy = 0.25 + j * 0.25
            centers.append([cx, cy])
        if len(centers) >= N_CIRCLES:
            break
    
    while len(centers) < N_CIRCLES:
        cx = 0.2 + len(centers) * 0.03
        cy = 0.7 + len(centers) * 0.02
        cx = np.clip(cx, r_init, 1.0 - r_init)
        cy = np.clip(cy, r_init, 1.0 - r_init)
        centers.append([cx, cy])
    
    return centers[:N_CIRCLES]


def build_bounds():
    bounds = []
    for i in range(N_CIRCLES):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))
    return bounds


def run_single_optimization(initial_centers, initial_radii, max_iter=3000):
    n = N_CIRCLES
    vars_flat = np.concatenate([np.array(initial_centers).flatten(), np.array(initial_radii)])
    
    cons_list = []
    cons_list.append({'type': 'ineq', 'fun': boundary_constraints})
    cons_list.append({'type': 'ineq', 'fun': non_overlap_constraints})
    
    bounds = build_bounds()
    
    result = minimize(
        objective,
        vars_flat,
        method='SLSQP',
        bounds=bounds,
        constraints=cons_list,
        options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False}
    )
    
    if result.success or result.fun < 0:
        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = result.x[2 * n:]
    else:
        centers_opt = np.array(initial_centers)
        radii_opt = np.array(initial_radii)
    
    return centers_opt, radii_opt


def validate_and_fix(centers, radii):
    n = N_CIRCLES
    radii = np.maximum(radii, 0.0)
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        centers[i, 0] = np.clip(x, r, 1.0 - r)
        centers[i, 1] = np.clip(y, r, 1.0 - r)
    
    return centers, radii


def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    initial_configs = []
    
    for r_init in [0.06, 0.07, 0.075, 0.08, 0.085, 0.09, 0.095, 0.10]:
        hex_centers = create_hexagonal_lattice(r_init)
        initial_configs.append((hex_centers, [r_init] * N_CIRCLES))
    
    for r_init in [0.06, 0.07, 0.08, 0.09, 0.10]:
        grid_centers = create_grid_lattice(r_init)
        initial_configs.append((grid_centers, [r_init] * N_CIRCLES))
    
    for r_init in [0.07, 0.08, 0.09, 0.10]:
        strat_centers = create_strategic_initial(r_init)
        initial_configs.append((strat_centers, [r_init] * N_CIRCLES))
    
    for centers_init, radii_init in initial_configs:
        centers_opt, radii_opt = run_single_optimization(centers_init, radii_init, max_iter=5000)
        centers_opt, radii_opt = validate_and_fix(centers_opt, radii_opt)
        current_sum = np.sum(radii_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    for iteration in range(3):
        centers_opt, radii_opt = run_single_optimization(
            best_centers.tolist(), best_radii.tolist(), max_iter=8000
        )
        centers_opt, radii_opt = validate_and_fix(centers_opt, radii_opt)
        current_sum = np.sum(radii_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    best_radii = np.maximum(best_radii, 0.0)
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, final_sum
