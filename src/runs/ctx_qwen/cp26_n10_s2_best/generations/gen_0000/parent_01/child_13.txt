To maximize the sum of radii for 26 circles in a unit square, we formulate the problem as a nonlinear programming (NLP) task. The objective is to maximize $\sum r_i$, which is equivalent to minimizing $-\sum r_i$. The constraints ensure each circle stays within $[0,1]^2$ and does not overlap with any other circle:
1. Boundary: $x_i - r_i \ge 0$, $1 - x_i - r_i \ge 0$, $y_i - r_i \ge 0$, $1 - y_i - r_i \ge 0$
2. Non-overlap: $\sqrt{(x_i-x_j)^2 + (y_i-y_j)^2} \ge r_i + r_j$ for all $i < j$

We use `scipy.optimize.minimize` with the `SLSQP` method, which handles nonlinear inequality constraints efficiently. To avoid local minima (common in packing problems), we employ a multi-start strategy with 5 different random initial configurations derived from a hexagonal lattice with jitter. Analytical Jacobians are provided for both constraint sets to accelerate convergence and improve numerical stability. A small numerical safeguard clips negative radii and ensures strict compliance with the validator's tolerance.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    num_vars = 3 * n  # x, y, r for each circle
    
    def objective(z):
        # Maximize sum of radii => minimize negative sum
        return -np.sum(z[2::3])

    def boundary_constraints(z):
        x = z[0::3]
        y = z[1::3]
        r = z[2::3]
        # Returns vector of size 4*n: [x-r, 1-x-r, y-r, 1-y-r]
        return np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])

    def boundary_jac(z):
        J = np.zeros((4 * n, num_vars))
        for i in range(n):
            col_base = 3 * i
            J[i, col_base] = 1.0;         J[i, col_base + 2] = -1.0
            J[n + i, col_base] = -1.0;    J[n + i, col_base + 2] = -1.0
            J[2*n + i, col_base + 1] = 1.0; J[2*n + i, col_base + 2] = -1.0
            J[3*n + i, col_base + 1] = -1.0; J[3*n + i, col_base + 2] = -1.0
        return J

    def pair_constraints(z):
        vals = []
        for i in range(n):
            xi, yi, ri = z[3*i], z[3*i+1], z[3*i+2]
            for j in range(i + 1, n):
                xj, yj, rj = z[3*j], z[3*j+1], z[3*j+2]
                dx, dy = xi - xj, yi - yj
                dist = np.sqrt(dx*dx + dy*dy)
                vals.append(dist - ri - rj)
        return np.array(vals)

    def pair_jac(z):
        m = n * (n - 1) // 2
        J = np.zeros((m, num_vars))
        idx = 0
        for i in range(n):
            xi, yi, ri = z[3*i], z[3*i+1], z[3*i+2]
            base_i = 3 * i
            for j in range(i + 1, n):
                xj, yj, rj = z[3*j], z[3*j+1], z[3*j+2]
                dx, dy = xi - xj, yi - yj
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < 1e-12:
                    dist = 1e-12
                inv = 1.0 / dist
                base_j = 3 * j
                
                J[idx, base_i] = dx * inv
                J[idx, base_i + 1] = dy * inv
                J[idx, base_i + 2] = -1.0
                
                J[idx, base_j] = -dx * inv
                J[idx, base_j + 1] = -dy * inv
                J[idx, base_j + 2] = -1.0
                
                idx += 1
        return J

    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints, 'jac': boundary_jac},
        {'type': 'ineq', 'fun': pair_constraints, 'jac': pair_jac}
    ]
    
    best_z = None
    best_sum_r = -np.inf
    
    # Multi-start optimization to escape local minima
    for seed in range(6):
        np.random.seed(seed)
        x0 = np.zeros(num_vars)
        
        # Hexagonal lattice initialization with jitter
        positions = []
        cols, rows = 6, 5
        for r in range(rows):
            for c in range(cols):
                if len(positions) >= n:
                    break
                x = (c + 0.5 + (0.5 if r % 2 else 0)) * (1.0 / cols)
                y = (r + 0.5) * (1.0 / rows)
                x += np.random.uniform(-0.03, 0.03)
                y += np.random.uniform(-0.03, 0.03)
                x = np.clip(x, 0.05, 0.95)
                y = np.clip(y, 0.05, 0.95)
                positions.append((x, y))
                
        for i in range(n):
            x0[3*i] = positions[i][0]
            x0[3*i+1] = positions[i][1]
            x0[3*i+2] = 0.015  # Small initial radius allows room to grow
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 1500, 'ftol': 1e-10, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_sum_r:
                z = res.x
                if np.all(z[2::3] >= -1e-7):
                    best_z = z
                    best_sum_r = -res.fun
        except Exception:
            continue
            
    if best_z is None:
        # Fallback valid configuration (should not trigger with optimizer)
        centers = np.tile([0.5, 0.5], (n, 1))
        radii = np.zeros(n)
        best_z = np.zeros(num_vars)
        for i in range(n):
            best_z[3*i] = centers[i][0]
            best_z[3*i+1] = centers[i][1]
            best_z[3*i+2] = radii[i]
            
    centers = np.column_stack((best_z[0::3], best_z[1::3]))
    radii = np.maximum(best_z[2::3], 0.0)
    
    return centers, radii, float(np.sum(radii))
```
