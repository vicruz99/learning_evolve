import numpy as np
from scipy.optimize import minimize

N = 26
# Precompute pairwise indices for efficient constraint evaluation
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Allocate constraint array
    c = np.empty(4*N + len(PAIR_I))
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist >= r_i + r_j
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = np.sqrt(dx**2 + dy**2) - (r[PAIR_I] + r[PAIR_J])
    
    return c

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.25)
    for i in range(N):
        # Distance to square boundaries
        wall_d = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
        r[i] = wall_d
        
        # Distance to other centers
        for j in range(N):
            if i == j:
                continue
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            if d / 2.0 < r[i]:
                r[i] = d / 2.0
                
    # Keep 92% to leave strict slack for the optimizer to expand
    return r * 0.92

    configs = []

    patterns = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4]]

                row += 1

                            p += 0.5
                        p += [sx, sy]
                        p = np.clip(p, 0.02, 0.98)
                        
                        r_init = get_feasible_radii(p)
                        configs.append(np.concatenate([p[:, 0], p[:, 1], r_init]))
                        
    # Add purely random dense starts
    for seed in range(25):
        np.random.seed(seed)
        p = np.random.uniform(0.1, 0.9, (N, 2))
        r_init = get_feasible_radii(p)
        configs.append(np.concatenate([p[:, 0], p[:, 1], r_init]))
        
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0

                    radii[j] = max(0.0, radii[j] - shrink)

            break
            
    return centers, radii, float(np.sum(radii))
