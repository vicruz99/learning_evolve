import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints (squared for smooth gradients)
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist_sq = dx**2 + dy**2
    r_sum_sq = (r[PAIR_I] + r[PAIR_J])**2
    
    return np.concatenate([c, dist_sq - r_sum_sq])

def make_feasible(v):
    """Adjust radii to guarantee strict feasibility."""
    x = v[:N].copy()
    y = v[N:2*N].copy()
    r = v[2*N:].copy()
    
    # Enforce boundary constraints
    r = np.minimum(r, x)
    r = np.minimum(r, 1.0 - x)
    r = np.minimum(r, y)
    r = np.minimum(r, 1.0 - y)
    
    # Enforce pairwise constraints iteratively
    for _ in range(50):
        dx = x[PAIR_I] - x[PAIR_J]
        dy = y[PAIR_I] - y[PAIR_J]
        dist = np.sqrt(dx**2 + dy**2)
        overlap = (r[PAIR_I] + r[PAIR_J]) - dist
        
        if np.max(overlap) < 1e-9:
            break
            
        shrink = np.maximum(0.0, overlap) / 2.0 + 1e-9
        r[PAIR_I] = np.maximum(0.0, r[PAIR_I] - shrink)
        r[PAIR_J] = np.maximum(0.0, r[PAIR_J] - shrink)
        
    return np.concatenate([x, y, r])

def generate_hex_config(r0, row_counts, angle=0.0):
    """Generate a hexagonal lattice configuration."""
    pts = []
    y = r0
    row_idx = 0
    for cnt in row_counts:
        x_start = r0 if row_idx % 2 == 0 else 2.0 * r0
        for k in range(cnt):
            pts.append([x_start + k * 2.0 * r0, y])
        y += np.sqrt(3) * r0
        row_idx += 1
        
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        pts = pts - 0.5
        c, s = np.cos(angle), np.sin(angle)
        pts = pts @ np.array([[c, -s], [s, c]]) + 0.5
        
    return pts

def generate_configs():

    configs = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],

                pts = generate_hex_config(r0, pat, ang)
                pts = np.clip(pts, 0.05, 0.95)
                r_init = np.full(N, r0 * 0.85)
                v = np.concatenate([pts[:, 0], pts[:, 1], r_init])
                configs.append(make_feasible(v))
                
    # Add random repelled configurations
    np.random.seed(42)
    for seed in range(10):
        np.random.seed(seed + 1000)
        pts = np.random.uniform(0.15, 0.85, (N, 2))

def run_packing():

    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).

    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-5, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}

    cy = best_v[N:2*N]

    # 1. Boundary constraints
    cr = np.minimum(cr, cx)
    cr = np.minimum(cr, 1.0 - cx)
    cr = np.minimum(cr, cy)

                    changed = True
        if not changed:
            break
            
    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
