import numpy as np
from scipy.optimize import minimize

N = 26
PI, PJ = np.triu_indices(N, k=1)

def objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def constraints(v):
    """Squared distance constraints for numerical stability."""
    x = v[:N]
    y = v[N:2 * N]
    r = v[2 * N:]
    
    c = np.empty(4 * N + len(PI))
    c[:N] = x - r
    c[N:2 * N] = 1.0 - x - r
    c[2 * N:3 * N] = y - r
    c[3 * N:4 * N] = 1.0 - y - r
    
    dx = x[PI] - x[PJ]
    dy = y[PI] - y[PJ]
    c[4 * N:] = dx ** 2 + dy ** 2 - (r[PI] + r[PJ]) ** 2
    
    return c

def compute_feasible_radii(centers, scale=0.90):
    """Compute strictly feasible initial radii based on local geometry."""
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    wall_dists = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    
    r = scale * np.minimum(min_dists / 2.0, wall_dists)
    return np.clip(r, 0.005, 0.25)

def generate_hex_lattice(r0, shift=0.0):
    """Generate hexagonal lattice centers."""
    pts = []
    y = r0 + shift
    row = 0
    while len(pts) < N + 15:
        x_start = r0 if row % 2 == 0 else 2.0 * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 15:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    return np.array(pts[:N])

def rotate_points(points, angle, center=(0.5, 0.5)):
    """Rotate points around a center."""
    c, s = np.cos(angle), np.sin(angle)
    pts = points - center
    rot_x = pts[:, 0] * c - pts[:, 1] * s
    rot_y = pts[:, 0] * s + pts[:, 1] * c
    result = np.column_stack([rot_x + center[0], rot_y + center[1]])
    return result

def generate_grid_lattice(r0, shift=0.0):
    """Generate square grid lattice centers."""
    pts = []
    y = r0 + shift
    while len(pts) < N + 10:
        x = r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += 2.0 * r0
    return np.array(pts[:N])

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-5, 0.25)] * N
    cons = {'type': 'ineq', 'fun': constraints}

    best_sum = -1.0
    best_v = None

    inits = []

    # 1. Dense hexagonal lattice sweep with fine parameter grid
    for r0 in np.arange(0.080, 0.108, 0.003):
        for shift in np.arange(-0.025, 0.028, 0.005):
            try:
                pts = generate_hex_lattice(r0, shift)
                mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & \
                       (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
                pts_valid = pts[mask]
                if len(pts_valid) >= N:
                    # Add small random jitter to break symmetry
                    for jitter_seed in range(3):
                        np.random.seed(jitter_seed)
                        p = pts_valid[:N].copy()
                        p += np.random.uniform(-0.003, 0.003, p.shape)
                        p = np.clip(p, 0.02, 0.98)
                        r = compute_feasible_radii(p, 0.90)
                        inits.append(np.concatenate([p[:, 0], p[:, 1], r]))
            except Exception:
                pass

    # 2. Rotated hexagonal lattices
    for seed in range(25):
        np.random.seed(seed)
        r0 = 0.090 + np.random.uniform(-0.008, 0.012)
        angle = np.random.uniform(-0.20, 0.20)
        shift = np.random.uniform(-0.02, 0.02, 2)

        pts = generate_hex_lattice(r0)
        pts = rotate_points(pts, angle) + shift
        mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & \
               (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
        pts_valid = pts[mask]
        if len(pts_valid) >= N:
            p = pts_valid[:N].copy()
            r = compute_feasible_radii(p, 0.88)
            inits.append(np.concatenate([p[:, 0], p[:, 1], r]))

    # 3. Square grid lattices
    for r0 in np.arange(0.085, 0.105, 0.005):
        for shift in np.arange(-0.02, 0.025, 0.005):
            pts = generate_grid_lattice(r0, shift)
            mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & \
                   (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
            if mask.sum() >= N:
                p = pts[mask][:N].copy()
                p += np.random.uniform(-0.002, 0.002, p.shape)
                p = np.clip(p, 0.02, 0.98)
                r = compute_feasible_radii(p, 0.85)
                inits.append(np.concatenate([p[:, 0], p[:, 1], r]))

    # 4. Random dense scatter
    for seed in range(20):
        np.random.seed(seed + 500)
        p = np.random.uniform(0.08, 0.92, (N, 2))
        r = compute_feasible_radii(p, 0.82)
        inits.append(np.concatenate([p[:, 0], p[:, 1], r]))

    # 5. Corner-focused configurations
    for seed in range(10):
        np.random.seed(seed + 700)
        corners = np.array([[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]])
        rest = np.random.uniform(0.15, 0.85, (N - 4, 2))
        p = np.vstack([corners, rest])
        r = compute_feasible_radii(p, 0.85)
        inits.append(np.concatenate([p[:, 0], p[:, 1], r]))

    # Phase 1: Multi-start optimization
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            curr_sum = -res.fun
            c_val = constraints(res.x)
            if np.min(c_val) >= -1e-7 and curr_sum > best_sum:
                best_sum = curr_sum
                best_v = res.x.copy()
        except Exception:
            pass

    # Phase 2: Aggressive perturbation refinement
    if best_v is not None:
        current_v = best_v.copy()
        for step in range(35):
            np.random.seed(step + 1000)
            pert = current_v.copy()
            
            # Perturb centers
            pert[:2 * N] += np.random.uniform(-0.004, 0.004, 2 * N)
            pert[:2 * N] = np.clip(pert[:2 * N], 0.02, 0.98)
            
            # Recompute feasible radii with high scale factor
            centers_pert = pert[:2 * N].reshape(N, 2)
            pert[2 * N:] = compute_feasible_radii(centers_pert, 0.93)

            try:
                res = minimize(objective, pert, method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                curr_sum = -res.fun
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_v = res.x.copy()
                    current_v = best_v.copy()
            except Exception:
                pass

    # Extract final configuration
    cx = best_v[:N]
    cy = best_v[N:2 * N]
    cr = best_v[2 * N:].copy()
    centers = np.column_stack((cx, cy))

    # Phase 3: Strict post-processing for validator compliance
    # 1. Enforce boundary constraints strictly
    for i in range(N):
        max_r = min(cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
        cr[i] = min(cr[i], max_r)
        cr[i] = max(cr[i], 0.0)

    # 2. Enforce non-overlap strictly with iterative shrinkage
    for _ in range(25):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                if cr[i] + cr[j] > d - 1e-9:
                    shrink = (cr[i] + cr[j] - d) / 2.0 + 1e-9
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break

    return centers, cr, float(np.sum(cr))
