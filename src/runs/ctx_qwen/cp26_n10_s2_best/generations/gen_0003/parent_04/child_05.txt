import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and squared pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap (squared for smoother gradients): dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c_pair = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return np.concatenate([c, c_pair])

def compute_initial_r(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.5)
    for i in range(N):
        # Distance to boundaries
        r[i] = min(centers[i, 0], 1.0 - centers[i, 0], 
                   centers[i, 1], 1.0 - centers[i, 1])
        # Distance to other centers
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            val = d / 2.0
            if val < r[i]: r[i] = val
            if val < r[j]: r[j] = val
    return r * 0.85

def force_relax(centers, steps=2000):
    """Physics-based relaxation to find a dense jammed packing."""
    r = np.full(N, 0.05)
    centers = centers.copy()
    dt = 0.025
    k_rep = 150.0
    k_wall = 120.0
    r_growth = 0.00005
    
    for _ in range(steps):
        forces = np.zeros_like(centers)
        
        # Wall forces
        for i in range(N):
            if centers[i, 0] < r[i] + 1e-7: forces[i, 0] += k_wall * (r[i] + 1e-7 - centers[i, 0])
            if centers[i, 0] > 1.0 - r[i] - 1e-7: forces[i, 0] -= k_wall * (centers[i, 0] - (1.0 - r[i] - 1e-7))
            if centers[i, 1] < r[i] + 1e-7: forces[i, 1] += k_wall * (r[i] + 1e-7 - centers[i, 1])
            if centers[i, 1] > 1.0 - r[i] - 1e-7: forces[i, 1] -= k_wall * (centers[i, 1] - (1.0 - r[i] - 1e-7))
            
        # Pairwise repulsion forces
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)
                if d < r[i] + r[j] and d > 1e-8:
                    f = k_rep * (r[i] + r[j] - d)
                    fx, fy = f * dx / d, f * dy / d
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
                    
        centers += forces * dt
        centers = np.clip(centers, r + 1e-7, 1.0 - r - 1e-7)
        r += r_growth
        
    return centers, r

def generate_starts():

    patterns = [[6,5,6,5,4], [5,6,5,6,4], [7,5,5,5,4], [5,5,6,5,5], [4,6,6,6,4], [6,6,5,5,4]]

        pts = np.random.uniform(0.15, 0.85, (N, 2))

        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
