import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and non-overlap (squared distances)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: circles must stay inside [0,1]^2
    c_bound = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c_pair = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return np.concatenate([c_bound, c_pair])

def generate_initial_configs():
    """Generate a diverse set of feasible initial configurations."""
    configs = []
    np.random.seed(42)
    
    # 1. Hexagonal lattices with rotations and shifts
    for r0 in [0.085, 0.090, 0.095, 0.100]:

                for sy in [-0.03, 0.0, 0.03]:

                    y = r0 + sy
                    row = 0

                        x = r0 + sx + (row % 2) * r0
                        while x <= 1 - r0:
                            pts.append([x, y])
                            x += 2 * r0
                        y += np.sqrt(3) * r0
                        row += 1
                    pts = np.array(pts[:N+10])
                    
                    if angle_deg != 0:
                        rad = np.deg2rad(angle_deg)
                        pts = pts - 0.5
                        rot = np.array([[np.cos(rad), -np.sin(rad)], [np.sin(rad), np.cos(rad)]])
                        pts = pts @ rot.T + 0.5
                        
                    mask = (pts[:,0] > 0.02) & (pts[:,0] < 0.98) & \
                           (pts[:,1] > 0.02) & (pts[:,1] < 0.98)
                    if np.sum(mask) >= N:
                        sel = pts[mask][:N]
                        sel += np.random.uniform(-0.004, 0.004, sel.shape)
                        configs.append(np.concatenate([sel[:,0], sel[:,1], np.full(N, 0.04)]))
                        
    # 2. Square grid variations
    for r0 in [0.095, 0.100, 0.105]:
        pts = []
        y = r0
        while len(pts) < N + 5:
            x = r0
            while x <= 1 - r0:
                pts.append([x, y])
                x += 2 * r0
            y += 2 * r0
        pts = np.array(pts[:N]) + np.random.uniform(-0.004, 0.004, (N, 2))
        configs.append(np.concatenate([pts[:,0], pts[:,1], np.full(N, 0.04)]))
        
    # 3. Force-relaxed random starts (spreads points evenly)
    for seed in range(15):
        np.random.seed(seed)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(200):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(pts[i,0] - pts[j,0], pts[i,1] - pts[j,1])
                    if d < 0.22 and d > 1e-4:
                        f = (0.22 - d) / d
                        diff = pts[i] - pts[j]
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.015
            pts = np.clip(pts, 0.05, 0.95)
        configs.append(np.concatenate([pts[:,0], pts[:,1], np.full(N, 0.03)]))
        
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    # Phase 1: Multi-start optimization from diverse basins
    inits = generate_initial_configs()
    
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-8:
                    best_sum = curr_sum
                    best_v = res.x.copy()
        except Exception:
            continue
            
    if best_v is None:
        best_v = inits[0]

    current_v = best_v

    centers = np.column_stack((best_v[:N], best_v[N:2*N]))

    # 1. Enforce boundary constraints strictly
    radii = np.minimum(radii, np.minimum(centers[:,0], 1.0 - centers[:,0]))
    radii = np.minimum(radii, np.minimum(centers[:,1], 1.0 - centers[:,1]))
    radii = np.maximum(radii, 0.0)

    for _ in range(30):
        changed = False

                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if radii[i] + radii[j] > d:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
