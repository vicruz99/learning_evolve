import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: Minimize negative sum of radii (maximize sum)."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + N*(N-1)//2)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def get_feasible_r(centers):
    """Compute strictly feasible initial radii for given centers."""
    r = np.full(N, 0.5)
    for i in range(N):
        r[i] = min(centers[i,0], 1.0 - centers[i,0], 
                   centers[i,1], 1.0 - centers[i,1])
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i,0] - centers[j,0], 
                             centers[i,1] - centers[j,1])
                r[i] = min(r[i], d / 2.0)
    return r * 0.95

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.25)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    configs = []
    
    # 1. Hexagonal lattices with various row patterns
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,6,6,4], [5,5,6,5,5]]
    for r0 in np.linspace(0.092, 0.108, 4):
        for pat in patterns:
            pts = []
            y = r0
            for row_idx, count in enumerate(pat):
                x_start = r0 if row_idx % 2 == 0 else 2.0 * r0
                for j in range(count):
                    x = x_start + j * 2.0 * r0
                    pts.append([x, y])
                y += np.sqrt(3) * r0
            pts = np.array(pts[:N])
            
            if (pts[:,0] >= 0.02).all() and (pts[:,0] <= 0.98).all() and \
               (pts[:,1] >= 0.02).all() and (pts[:,1] <= 0.98).all():
                r_init = get_feasible_r(pts)
                configs.append(np.concatenate([pts[:,0], pts[:,1], r_init]))
                
            # Rotated versions
            for angle in [0.05, -0.05, 0.1, -0.1]:
                cos_a, sin_a = np.cos(angle), np.sin(angle)
                centered = pts - 0.5
                rot_pts = np.column_stack([centered[:,0]*cos_a - centered[:,1]*sin_a,
                                           centered[:,0]*sin_a + centered[:,1]*cos_a]) + 0.5
                mask = (rot_pts[:,0]>=0.03) & (rot_pts[:,0]<=0.97) & \
                       (rot_pts[:,1]>=0.03) & (rot_pts[:,1]<=0.97)
                if np.sum(mask) >= N:
                    valid_pts = rot_pts[mask][:N]
                    r_init = get_feasible_r(valid_pts)
                    configs.append(np.concatenate([valid_pts[:,0], valid_pts[:,1], r_init]))
                    
    # 2. Perturbed lattice configs
    for i in range(8):
        base = configs[i % len(configs)].copy()
        base[:2*N] += np.random.uniform(-0.005, 0.005, 2*N)
        base[:2*N] = np.clip(base[:2*N], 0.02, 0.98)
        configs.append(base)
        
    # 3. Random repelled starts
    for seed in range(12):
        np.random.seed(seed)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(150):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    d = np.linalg.norm(pts[i] - pts[j])
                    if d < 0.22 and d > 1e-4:
                        f = (0.22 - d) * 3.0 / d
                        diff = pts[i] - pts[j]
                        forces[i] += f * diff
                        forces[j] -= f * diff
            pts += forces * 0.005
            pts = np.clip(pts, 0.05, 0.95)
        r_init = get_feasible_r(pts)
        configs.append(np.concatenate([pts[:,0], pts[:,1], r_init]))

    # Primary optimization
    for x0 in configs:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = configs[0]
        
    # Adaptive refinement to escape local minima
    current_v = best_v
    for step in range(15):
        pert = current_v.copy()
        pert[2*N:] *= 0.97
        pert[:2*N] += np.random.uniform(-0.004, 0.004, 2*N)
        pert[:2*N] = np.clip(pert[:2*N], 0.01, 0.99)
        
        try:
            res = minimize(objective, pert, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
                    current_v = best_v
        except Exception:
            pass

    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    radii = best_v[2*N:].copy()
    
    # Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints strictly
    for i in range(N):
        mr = min(centers[i,0], 1.0 - centers[i,0], 
                 centers[i,1], 1.0 - centers[i,1])
        if radii[i] > mr + 1e-9:
            radii[i] = mr
            
    # 2. Enforce non-overlap strictly with minimal necessary shrinkage
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0] - centers[j,0], 
                             centers[i,1] - centers[j,1])
                if radii[i] + radii[j] > d + 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
