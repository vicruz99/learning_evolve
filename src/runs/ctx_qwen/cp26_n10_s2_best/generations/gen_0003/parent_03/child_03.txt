import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
BOUNDS = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

def penalty_objective(v, lam):
    """Smooth penalty objective for L-BFGS-B. Encourages large radii while punishing overlaps."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Pairwise overlaps
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist = np.sqrt(dx**2 + dy**2)
    overlap = np.maximum(0.0, r[PAIR_I] + r[PAIR_J] - dist)
    
    # Boundary violations
    b1 = np.maximum(0.0, r - x)
    b2 = np.maximum(0.0, r - (1.0 - x))
    b3 = np.maximum(0.0, r - y)
    b4 = np.maximum(0.0, r - (1.0 - y))
    
    # Minimize negative sum of radii + penalty
    return -np.sum(r) + lam * (np.sum(overlap**2) + np.sum(b1**2) + np.sum(b2**2) + np.sum(b3**2) + np.sum(b4**2))

def hard_constraints(v):
    """Inequality constraints for SLSQP. Returns array where each element >= 0."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundaries: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c = np.concatenate([c, dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2])
    
    return c

def generate_initial_config(seed):

    np.random.seed(seed)
    config_type = seed % 3
    centers = np.zeros((N, 2))
    
    if config_type == 0:
        # Hexagonal lattice with random rotation and shift
        r0 = 0.092 + np.random.uniform(-0.008, 0.008)
        pts = []

        row = 0

            x = x_start

                pts.append([x, y])
                x += 2.0 * r0

            row += 1
        centers = np.array(pts[:N])

        c, s = np.cos(ang), np.sin(ang)
        centers[:, 0] -= 0.5
        centers[:, 1] -= 0.5
        centers = centers @ np.array([[c, -s], [s, c]])
        centers[:, 0] += 0.5
        centers[:, 1] += 0.5
        
    elif config_type == 1:
        # Perturbed square grid
        pts = []
        for i in range(6):
            for j in range(5):
                pts.append([0.1 + i * 0.16 + np.random.uniform(-0.02, 0.02),
                            0.1 + j * 0.20 + np.random.uniform(-0.02, 0.02)])
        centers = np.array(pts[:N])
    else:
        # Dense random scatter
        centers = np.random.uniform(0.08, 0.92, (N, 2))
        
    centers = np.clip(centers, 0.03, 0.97)
    r_init = 0.045 + np.random.uniform(-0.005, 0.005, N)
    return np.concatenate([centers[:, 0], centers[:, 1], r_init])

def run_packing():
    best_v = None
    best_sum = -1.0

    # Gradually increasing lambda allows the solver to slide out of local minima
    lam_schedule = [5.0, 30.0, 150.0, 800.0, 4000.0, 20000.0, 80000.0]
    
    for seed in range(80):
        v = generate_initial_config(seed)
        for lam in lam_schedule:
            res = minimize(penalty_objective, v, args=(lam,), method='L-BFGS-B', 
                           bounds=BOUNDS, options={'maxiter': 2500, 'ftol': 1e-13})
            v = res.x
            
        curr_sum = -np.sum(v[2*N:])
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_v = v.copy()

    if best_v is not None:
        current_v = best_v.copy()

    cx = best_v[:N]
    cy = best_v[N:2*N]

                    shrink = (cr[i] + cr[j] - d) / 2.0 + 1e-10
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    # Final safety clamp to guarantee no boundary violations
    cr = np.maximum(cr, 0.0)
    
    return centers, cr, float(np.sum(cr))
