from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + len(PAIR_I))
    # Boundary constraints
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap constraints (squared distance for stability)
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def generate_hex_centers(row_counts, r0, sx=0.0, sy=0.0, rot=0.0):
    """Generate a hexagonal lattice configuration with specified parameters."""
    pts = []
    y = r0 + sy
    for i, cnt in enumerate(row_counts):
        xs = r0 + sx + (i % 2) * r0
        for _ in range(cnt):
            if len(pts) >= N: 
                break
            pts.append([xs, y])
            xs += 2 * r0
        y += r0 * np.sqrt(3)
        
    pts = np.array(pts[:N])
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts -= 0.5
        pts = pts @ np.array([[c, -s], [s, c]])
        pts += 0.5
