import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)
NUM_CONS = 4 * N + NUM_PAIRS
IDX_P = np.arange(NUM_PAIRS)

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.empty(NUM_CONS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - dr**2
    return c

def constraint_jac(v):
    """Compute analytical Jacobian of constraints w.r.t variables."""
    J = np.zeros((NUM_CONS, 3*N))
    # Boundary constraint gradients
    J[:N, :N] = np.eye(N)
    J[:N, 2*N:] = -np.eye(N)
    J[N:2*N, :N] = -np.eye(N)
    J[N:2*N, 2*N:] = -np.eye(N)
    J[2*N:3*N, N:2*N] = np.eye(N)
    J[2*N:3*N, 2*N:] = -np.eye(N)
    J[3*N:4*N, N:2*N] = -np.eye(N)
    J[3*N:4*N, 2*N:] = -np.eye(N)

    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]

    base_k = 4*N
    rows = base_k + IDX_P
    # Pairwise constraint gradients (vectorized assignment)
    J[rows, PAIR_I] = 2.0 * dx
    J[rows, PAIR_J] = -2.0 * dx
    J[rows, N + PAIR_I] = 2.0 * dy
    J[rows, N + PAIR_J] = -2.0 * dy
    J[rows, 2*N + PAIR_I] = -2.0 * dr
    J[rows, 2*N + PAIR_J] = -2.0 * dr
    return J

def compute_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    # 50% factor guarantees strict feasibility initially
    r = 0.50 * np.minimum(min_dists / 2.0, wall_dists)
    return np.clip(r, 0.01, 0.25)

def generate_hex_config(seed):
    """Generate a perturbed hexagonal lattice configuration."""
    np.random.seed(seed)
    r0 = 0.085 + np.random.uniform(-0.005, 0.015)
    angle = np.random.uniform(-0.25, 0.25)
    shift = np.random.uniform(-0.05, 0.05, 2)

    pts = []
    y = r0
    row = 0
    while len(pts) < N + 15:
        x_start = r0 + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 15:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
        row += 1

    pts = np.array(pts)
    # Rotate and shift around center
    cx, cy = 0.5, 0.5
    pts[:, 0] -= cx
    pts[:, 1] -= cy
    c, s = np.cos(angle), np.sin(angle)
    pts = pts @ np.array([[c, -s], [s, c]])
    pts[:, 0] += cx + shift[0]
    pts[:, 1] += cy + shift[1]

    # Filter points safely inside square
    mask = (pts[:, 0] >= 0.05) & (pts[:, 0] <= 0.95) & \
           (pts[:, 1] >= 0.05) & (pts[:, 1] <= 0.95)
    pts_valid = pts[mask]
    
    # Fallback fill if rotation cut too many points
    if len(pts_valid) < N:
        extra = np.random.uniform(0.1, 0.9, (N - len(pts_valid), 2))
        pts_valid = np.vstack([pts_valid, extra])
    pts_valid = pts_valid[:N]

    r = compute_feasible_radii(pts_valid)
    return np.concatenate([pts_valid[:, 0], pts_valid[:, 1], r])

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints, 'jac': constraint_jac}
    
    best_v = None
    best_sum = -1.0
    
    # Phase 1: Multi-start optimization from diverse basins
    for seed in range(50):
        v0 = generate_hex_config(seed)
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            s = -res.fun
            # Accept if better and sufficiently feasible
            c_val = constraints(res.x)
            if np.min(c_val) >= -1e-6 and s > best_sum:
                best_sum = s
                best_v = res.x.copy()
        except Exception:
            pass
            
    # Fallback initialization
    if best_v is None:
        best_v = generate_hex_config(0)
        
    # Phase 2: Iterative refinement to escape local minima
    current_v = best_v
    for step in range(20):
        pert = current_v.copy()
        # Perturb centers to explore configuration space
        pert[:2*N] += np.random.uniform(-0.004, 0.004, 2*N)
        pert[:2*N] = np.clip(pert[:2*N], 0.01, 0.99)
        # Shrink radii to create breathing room for rearrangement
        pert[2*N:] *= 0.96
        
        # Recompute safe radii to guarantee feasibility before next opt step
        centers_p = pert[:2*N].reshape(N, 2)
        pert[2*N:] = compute_feasible_radii(centers_p)
        
        try:
            res = minimize(objective, pert, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            s = -res.fun
            c_val = constraints(res.x)
            if np.min(c_val) >= -1e-6 and s > best_sum:
                best_sum = s
                best_v = res.x.copy()
                current_v = best_v
        except Exception:
            pass
            
    # Extract optimal configuration
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()
    centers = np.column_stack((cx, cy))
    
    # Phase 3: Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints strictly
    for i in range(N):
        cr[i] = min(cr[i], centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        
    # 2. Enforce non-overlap constraints iteratively with safety margin
    for _ in range(15):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < cr[i] + cr[j] - 1e-9:
                    shrink = (cr[i] + cr[j] - d) / 2.0 + 1e-9
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, cr, float(np.sum(cr))
