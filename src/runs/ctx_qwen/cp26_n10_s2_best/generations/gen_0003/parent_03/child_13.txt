from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints (squared distance for better gradients)
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist_sq = dx**2 + dy**2
    r_sum = r[PAIR_I] + r[PAIR_J]
    c = np.concatenate([c, dist_sq - r_sum**2])
    
    return c

def compute_feasible_r(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.5)
    for i in range(N):
        # Distance to boundaries
        r[i] = min(centers[i, 0], 1.0 - centers[i, 0], 
                   centers[i, 1], 1.0 - centers[i, 1])
        # Distance to other centers
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            r[i] = min(r[i], d / 2.0)
            r[j] = min(r[j], d / 2.0)
    return r * 0.85  # Leave 15% slack for optimizer

def hex_lattice_init(row_counts, r_est, shift_x=0.0, shift_y=0.0, rot=0.0):

    pts = []
    y = r_est + shift_y
    row_idx = 0
    for count in row_counts:

        for k in range(count):
            x = x_start + k * 2 * r_est
            pts.append([x, y])
        y += r_est * np.sqrt(3)
        row_idx += 1
    pts = np.array(pts[:N])
    if rot != 0.0:
        pts = pts - 0.5
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts += 0.5
    return np.clip(pts, 0.01, 0.99)

def force_directed_init(seed, n_iter=400):

    np.random.seed(seed)

        # Wall repulsion

                if pts[i, k] > 0.95: forces[i, k] -= 0.2
        pts += forces * 0.005
        pts = np.clip(pts, 0.02, 0.98)
    return pts
