import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: Minimize negative sum of radii (equivalent to maximizing sum)."""
    return -np.sum(v[2*N:])

def constraints(v):
    """
    Compute inequality constraints: boundaries and pairwise non-overlap.
    Returns a flat array where each element >= 0 indicates a satisfied constraint.
    Uses squared distances for smoother gradients and numerical stability.
    """
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def repair_radii(centers, r_init=None):
    """
    Iteratively shrinks radii to guarantee strict feasibility for given centers.
    Ensures circles are inside the square and do not overlap.
    """
    if r_init is None:
        r = np.full(N, 0.5)
    else:
        r = r_init.copy()
        
    # Enforce boundary constraints
    for i in range(N):
        r[i] = min(r[i], centers[i,0], 1.0 - centers[i,0], 
                   centers[i,1], 1.0 - centers[i,1])
        
    # Enforce non-overlap constraints iteratively
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if r[i] + r[j] > d:
                    shrink = (r[i] + r[j] - d) / 2.0 + 1e-9
                    r[i] = max(0.001, r[i] - shrink)
                    r[j] = max(0.001, r[j] - shrink)
                    changed = True
        if not changed:
            break
    return r

def generate_hex_lattice(r0, shift_x, shift_y, rot):
    """Generate a hexagonal lattice configuration with optional rotation and shifts."""
    pts = []
    y = r0 + shift_y
    row = 0
    while len(pts) < N + 5:
        x_start = r0 + shift_x + (row % 2) * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
        row += 1
        
    pts = np.array(pts[:N])
    
    # Apply rotation around center (0.5, 0.5)
    if rot != 0.0:
        cx, cy = 0.5, 0.5
        pts[:, 0] -= cx
        pts[:, 1] -= cy
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts[:, 0] += cx
        pts[:, 1] += cy
        
    return np.clip(pts, 0.01, 0.99)

    cons = {'type': 'ineq', 'fun': constraints}

    best_sum = -1.0
