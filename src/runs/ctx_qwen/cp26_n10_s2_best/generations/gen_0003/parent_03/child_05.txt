

<strategy>
1. **Topology Exploration**: Hexagonal lattices are the densest 2D circle packings, but square boundaries create edge effects. I will generate dozens of initial configurations by varying lattice densities, shifts, and rotation angles (0° to 70°) to align the lattice optimally with the square's geometry.
2. **Repulsion-Based Initialization**: Lattices can get stuck in symmetric local minima. I'll add force-directed repulsion simulations to generate asymmetric, organic starting points that often lead to better basins of attraction.
3. **Joint Optimization with SLSQP**: Centers and radii will be optimized simultaneously. Constraints are formulated linearly for boundaries (`x >= r`, etc.) and as Euclidean distance inequalities (`dist >= r_i + r_j`). This avoids singularity issues of squared constraints near zero distance.
4. **Multi-Start & Refinement**: A broad multi-start phase explores diverse basins. A second phase perturbs the best solution, shrinks radii to create "breathing room," and re-optimizes to escape shallow local minima.
5. **Rigorous Post-Processing**: After optimization, boundary and overlap constraints are strictly enforced with iterative shrinkage and a safety margin to guarantee compliance with the validator's `1e-12` tolerance.
</strategy>

import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii (maximize sum)."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: circles must be inside [0, 1]x[0, 1]
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints: dist >= r_i + r_j
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist = np.hypot(dx, dy)
    c = np.concatenate([c, dist - r[PAIR_I] - r[PAIR_J]])
    
    return c

def compute_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.5)
    for i in range(N):
        # Distance to square boundaries
        r[i] = min(centers[i,0], 1.0 - centers[i,0], 
                   centers[i,1], 1.0 - centers[i,1])
        # Distance to other centers
        for j in range(N):
            if i == j: 
                continue
            d = np.hypot(centers[i,0] - centers[j,0], 
                         centers[i,1] - centers[j,1])
            r[i] = min(r[i], d * 0.5)
    return r * 0.88  # 88% leaves controlled slack for optimizer

def generate_hex_lattice(r0, shift_x, shift_y, angle):
    """Generate a hexagonal lattice configuration with optional rotation."""
    pts = []
    y = r0 + shift_y
    row = 0
    while len(pts) < N + 10:
        x_start = r0 + shift_x + (row % 2) * r0
        x = x_start
        while x <= 1 - r0:
            pts.append([x, y])
            x += 2 * r0
        y += r0 * np.sqrt(3)
        row += 1
        
    pts = np.array(pts[:N])
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        pts[:, 0] -= cx
        pts[:, 1] -= cy
        c, s = np.cos(angle), np.sin(angle)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts[:, 0] += cx
        pts[:, 1] += cy
        
    return np.clip(pts, 0.02, 0.98)

def generate_repelled_random(seed):
    """Generate diverse initial configurations via force-directed repulsion."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(60):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(pts[i,0] - pts[j,0], pts[i,1] - pts[j,1])
                if d < 0.25 and d > 1e-5:
                    f = (0.25 - d) * 0.5 / d
                    diff = pts[i] - pts[j]
                    forces[i] += f * diff
                    forces[j] -= f * diff
        pts += forces * 0.1
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-4, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    initial_configs = []
    
    # 1. Diverse Hexagonal Lattices
    for r0 in [0.090, 0.095, 0.100, 0.105]:
        for angle in [0.0, 0.15, 0.30, 0.45, 0.70]:
            for sx, sy in [(0.0, 0.0), (0.02, 0.0), (-0.02, 0.0), (0.0, 0.02)]:
                pts = generate_hex_lattice(r0, sx, sy, angle)
                r_init = compute_feasible_radii(pts)
                initial_configs.append(np.concatenate([pts[:,0], pts[:,1], r_init]))
                
    # 2. Repelled Random Configurations
    for seed in range(15):
        pts = generate_repelled_random(seed)
        r_init = compute_feasible_radii(pts)
        initial_configs.append(np.concatenate([pts[:,0], pts[:,1], r_init]))
        
    # Phase 1: Multi-start optimization
    for v0 in initial_configs:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                # Accept if sufficiently feasible
                if np.min(constraints(res.x)) >= -1e-5:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = initial_configs[0]
        
    # Phase 2: Local escape & refinement to escape shallow local minima
    current_v = best_v
    for step in range(20):
        pert = current_v.copy()
        # Perturb centers to explore neighborhood
        noise = np.random.uniform(-0.006, 0.006, 2*N)
        pert[:2*N] += noise
        pert[:2*N] = np.clip(pert[:2*N], 0.02, 0.98)
        # Shrink radii slightly to ensure the perturbed state is feasible
        pert[2*N:] *= 0.94
        
        try:
            res = minimize(objective, pert, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-5:
                    best_sum = s
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            pass
            
    # Extract optimal configuration
    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    radii = best_v[2*N:].copy()
    
    # Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints strictly
    radii = np.minimum(radii, np.minimum(centers[:,0], 1.0 - centers[:,0]))
    radii = np.minimum(radii, np.minimum(centers[:,1], 1.0 - centers[:,1]))
    radii = np.maximum(radii, 0.0)
    
    # 2. Enforce non-overlap constraints iteratively with safety margin
    for _ in range(20):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if radii[i] + radii[j] > d:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
