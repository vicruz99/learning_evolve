from scipy.optimize import minimize

N = 26
# Precompute mask for upper triangular pairwise constraints
MASK = np.triu(np.ones((N, N), dtype=bool), k=1)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + N*(N-1)//2)
    # Boundary constraints
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    # Using squared distances avoids sqrt and provides smoother gradients
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    d_sq = dx**2 + dy**2
    
    c[4*N:] = d_sq[MASK] - dr[MASK]**2
    return c

def compute_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    n = centers.shape[0]
    r = np.full(n, 0.5)
    # Distance to square boundaries
    r = np.minimum(r, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    r = np.minimum(r, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    # Distance to other centers
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            half_d = d * 0.5
            if half_d < r[i]: r[i] = half_d
            if half_d < r[j]: r[j] = half_d
    return r * 0.92

    return centers, cr, float(np.sum(cr))
