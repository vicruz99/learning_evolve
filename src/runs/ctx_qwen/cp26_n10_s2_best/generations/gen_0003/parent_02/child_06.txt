import numpy as np
from scipy.optimize import minimize, differential_evolution, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

# Precompute constant parts of the LP constraint matrix
# Structure: [I; A_pair; -I] corresponding to [r <= b; r_i+r_j <= d_ij; -r <= 0]
A_PAIR = np.zeros((len(PAIR_I), N))
A_PAIR[np.arange(len(PAIR_I)), PAIR_I] = 1.0
A_PAIR[np.arange(len(PAIR_I)), PAIR_J] = 1.0
A_UB_CONST = np.vstack([np.eye(N), A_PAIR, -np.eye(N)])
C_OBJ = -np.ones(N)  # Minimize negative sum => maximize sum

def compute_optimal_radii(centers):
    """Compute the exact maximum sum of radii for fixed centers via Linear Programming."""
    # Distance to boundaries
    b = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    b = np.maximum(b, 1e-9)  # Prevent degenerate bounds
    
    # Pairwise Euclidean distances
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    
    # Construct RHS vector for LP: [b; d_ij; 0]
    b_ub = np.concatenate([b, dists[PAIR_I, PAIR_J], np.zeros(N)])
    
    # Solve LP
    res = linprog(C_OBJ, A_ub=A_UB_CONST, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    # Fallback to greedy assignment if LP fails (rare)
    return np.minimum(b, np.min(dists, axis=1) / 2.0)

def objective_centers(centers_flat):
    """Objective function for center optimization: returns negative max sum of radii."""
    centers = centers_flat.reshape(-1, 2)
    # Keep centers strictly inside to avoid boundary singularities
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    r = compute_optimal_radii(centers)
    return -np.sum(r)

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(1e-4, 1.0 - 1e-4)] * (2 * N)
    
    # Phase 1: Global exploration using Differential Evolution
    # Efficiently navigates the non-smooth landscape of the LP-based objective
    res_de = differential_evolution(
        objective_centers, 
        bounds, 
        popsize=12, 
        maxiter=500, 
        tol=1e-7, 
        seed=42, 
        workers=1
    )
    best_x = res_de.x
    
    # Phase 2: High-precision local refinement using Nelder-Mead
    res_nm = minimize(
        objective_centers, 
        best_x, 
        method='Nelder-Mead', 
        tol=1e-9, 
        options={'maxiter': 3000, 'adaptive': True}
    )
    if res_nm.success:
        best_x = res_nm.x
        
    centers = best_x.reshape(N, 2)
    radii = compute_optimal_radii(centers)
    
    # Phase 3: Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints strictly
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    radii = np.minimum(radii, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # 2. Enforce non-overlap constraints iteratively with safety margin
    for _ in range(10):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
