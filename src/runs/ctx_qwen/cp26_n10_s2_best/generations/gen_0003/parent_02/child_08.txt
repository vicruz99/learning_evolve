import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints (squared for smoother gradients)
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist_sq = dx**2 + dy**2
    r_sum = r[PAIR_I] + r[PAIR_J]
    c = np.concatenate([c, dist_sq - r_sum**2])
    return c

def compute_initial_v(centers):
    """Compute strictly feasible initial radii for given centers."""
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    # Start at 92% of max possible to be close to optimum but strictly feasible
    r = 0.92 * np.minimum(min_dists / 2.0, wall_dists)
    r = np.clip(r, 1e-4, 0.25)
    return np.concatenate([centers[:, 0], centers[:, 1], r])

def generate_inits():
    """Generate diverse initial configurations."""
    inits = []
    np.random.seed(123)
    
    # 1. Rotated & shifted hexagonal lattices
    for seed in range(30):
        rng = np.random.RandomState(seed)
        r0 = rng.uniform(0.085, 0.105)
        angle = rng.uniform(-0.3, 0.3)
        shift = rng.uniform(-0.05, 0.05, 2)
        
        pts = []
        y = r0
        row = 0
        while len(pts) < N + 10:
            x_start = r0 + (row % 2) * r0
            x = x_start
            while x <= 1.0 - r0:
                p = np.array([x, y])
                p -= 0.5
                c, s = np.cos(angle), np.sin(angle)
                p = np.array([c*p[0] - s*p[1], s*p[0] + c*p[1]]) + 0.5
                p += shift
                if 0.02 < p[0] < 0.98 and 0.02 < p[1] < 0.98:
                    pts.append(p)
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
            row += 1
        pts = np.array(pts[:N]) + rng.uniform(-0.005, 0.005, (N, 2))
        pts = np.clip(pts, 0.01, 0.99)
        inits.append(compute_initial_v(pts))
        
    # 2. Perturbed square grids
    for seed in range(15):
        rng = np.random.RandomState(seed + 100)
        pts = np.array([[i*0.17 + 0.08, j*0.17 + 0.08] for i in range(6) for j in range(5)])[:N]
        pts += rng.uniform(-0.01, 0.01, pts.shape)
        pts = np.clip(pts, 0.01, 0.99)
        inits.append(compute_initial_v(pts))
        
    # 3. Repelled random placements
    for seed in range(15):
        rng = np.random.RandomState(seed + 200)
        pts = []
        while len(pts) < N:
            p = rng.uniform(0.1, 0.9, 2)
            if len(pts) == 0 or np.min(np.linalg.norm(np.array(pts) - p, axis=1)) > 0.13:
                pts.append(p)
        pts = np.array(pts)
        inits.append(compute_initial_v(pts))
        
    return inits

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-4, 0.25)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    inits = generate_inits()
    
    # Phase 1: Multi-start exploration
    for v0 in inits:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if np.min(constraints(res.x)) >= -1e-7 and s > best_sum:
                best_sum = s
                best_v = res.x.copy()
        except:
            pass
            
    if best_v is None:
        best_v = inits[0]
        best_sum = -np.sum(best_v[2*N:])
        
    # Phase 2: Adaptive perturbation & refinement to escape local minima
    np.random.seed(999)
    current_v = best_v
    for step in range(25):
        # Gradually shrink radii to create breathing room for center movement
        shrink = 0.92 - step * 0.006
        v_pert = current_v.copy()
        v_pert[2*N:] *= max(0.5, shrink)
        
        # Perturb centers
        v_pert[:2*N] += np.random.uniform(-0.006, 0.006, 2*N)
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.01, 0.99)
        
        # Guarantee strict feasibility after perturbation
        v_pert = compute_initial_v(v_pert[:2*N].reshape(N, 2))
        
        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if np.min(constraints(res.x)) >= -1e-7 and s > best_sum:
                best_sum = s
                best_v = res.x.copy()
                current_v = best_v.copy()
        except:
            pass
            
    # Phase 3: Final polishing pass with tighter tolerances
    v_polish = best_v.copy()
    v_polish[2*N:] *= 0.98
    try:
        res = minimize(objective, v_polish, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        s = -res.fun
        if np.min(constraints(res.x)) >= -1e-7 and s > best_sum:
            best_v = res.x.copy()
    except:
        pass
        
    # Extract results
    cx = best_v[:N]
    cy = best_v[N:2*N]
    cr = best_v[2*N:].copy()
    
    # Strict post-processing for validator compliance
    # 1. Enforce boundary constraints strictly
    cr = np.minimum(cr, np.minimum(cx, 1.0 - cx))
    cr = np.minimum(cr, np.minimum(cy, 1.0 - cy))
    cr = np.maximum(cr, 0.0)
    
    # 2. Iteratively resolve any residual numerical overlaps
    for _ in range(20):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(cx[i]-cx[j], cy[i]-cy[j])
                if cr[i] + cr[j] > d:
                    shrink = (cr[i] + cr[j] - d) / 2.0 + 1e-9
                    cr[i] = max(0.0, cr[i] - shrink)
                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    centers = np.column_stack((cx, cy))
    return centers, cr, float(np.sum(cr))
