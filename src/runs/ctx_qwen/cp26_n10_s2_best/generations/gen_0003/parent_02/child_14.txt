from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: circles must be inside [0, 1]x[0, 1]
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist_sq = dx**2 + dy**2
    r_sum = r[PAIR_I] + r[PAIR_J]
    c = np.concatenate([c, dist_sq - r_sum**2])
    
    return c

def get_feasible_radii(centers, safety=0.85):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.full(N, 0.5)
    for i in range(N):
        # Distance to boundaries
        mr = min(centers[i,0], 1.0 - centers[i,0], 
                 centers[i,1], 1.0 - centers[i,1])
        r[i] = mr
        # Distance to other centers
        for j in range(N):
            if i == j: 
                continue
            d = np.hypot(centers[i,0] - centers[j,0], 
                         centers[i,1] - centers[j,1])
            if d / 2.0 < r[i]:
                r[i] = d / 2.0
    return r * safety

def repulsion_relax(centers, steps=150, alpha=0.05):
    """Spread out centers using a simple repulsive force model."""
    c = centers.copy()
    for _ in range(steps):
        forces = np.zeros_like(c)
        for i in range(N):

                diff = c[i] - c[j]
                d = np.linalg.norm(diff)
                if d < 0.25 and d > 1e-5:
                    f = (0.25 - d) / d
                    forces[i] += f * diff
                    forces[j] -= f * diff
        c += alpha * forces
        c = np.clip(c, 0.02, 0.98)
    return c

    pts = []
    y = r0 + shift[1]

            pts.append([x, y])

        row += 1

        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    return np.clip(pts, 0.01, 0.99)
