import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and squared non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    r_sum = r[PAIR_I] + r[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    return c

def make_feasible(v):
    """Adjusts radii to guarantee strict feasibility."""
    x = v[:N].copy()
    y = v[N:2*N].copy()
    r = v[2*N:].copy()
    
    r = np.minimum(r, np.minimum(x, 1.0 - x))
    r = np.minimum(r, np.minimum(y, 1.0 - y))
    
    for _ in range(20):
        dx = x[PAIR_I] - x[PAIR_J]
        dy = y[PAIR_I] - y[PAIR_J]
        d = np.sqrt(dx**2 + dy**2)
        overlap = (r[PAIR_I] + r[PAIR_J]) - d
        if np.max(overlap) < 1e-10:
            break
        shrink = np.maximum(0.0, overlap) / 2.0 + 1e-10
        r[PAIR_I] = np.maximum(0.0, r[PAIR_I] - shrink)
        r[PAIR_J] = np.maximum(0.0, r[PAIR_J] - shrink)
    return np.concatenate([x, y, r])

def generate_hex_config(r0, shift_x, shift_y, angle):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0 + shift_y
    row = 0
    while len(pts) < N + 10:
        x_start = r0 + shift_x + (row % 2) * r0 * 0.5
        x = x_start
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += r0
        y += r0 * np.sqrt(3) * 0.5
        row += 1
    pts = np.array(pts[:N+10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = pts - 0.5
        pts = pts @ np.array([[c, -s], [s, c]])
        pts = pts + 0.5
    return pts
