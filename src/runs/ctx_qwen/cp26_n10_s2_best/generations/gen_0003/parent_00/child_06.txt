import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def ensure_feasible(v):
    """Adjust radii to guarantee strict feasibility for a given center configuration."""
    x = v[:N].copy()
    y = v[N:2*N].copy()
    r = v[2*N:].copy()
    
    # Enforce boundaries
    r = np.minimum(r, np.minimum(x, 1.0 - x))
    r = np.minimum(r, np.minimum(y, 1.0 - y))
    
    # Enforce non-overlap iteratively
    for _ in range(10):
        dx = x[PAIR_I] - x[PAIR_J]
        dy = y[PAIR_I] - y[PAIR_J]
        d = np.sqrt(dx**2 + dy**2)
        overlap = (r[PAIR_I] + r[PAIR_J]) - d
        if np.max(overlap) < 1e-9:
            break
        shrink = np.maximum(0.0, overlap) / 2.0 + 1e-9
        r[PAIR_I] = np.maximum(1e-6, r[PAIR_I] - shrink)
        r[PAIR_J] = np.maximum(1e-6, r[PAIR_J] - shrink)
        
    return np.concatenate([x, y, r])

def generate_initial_configs():

    configs = []
    np.random.seed(42)
    
    # 1. Hexagonal lattices with rotations
    for seed in range(8):

            while x <= 1.0 - r0 and len(pts) < N + 8:
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3.0)
            row += 1
            
        pts = np.array(pts) @ rot.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        pts += np.random.uniform(-0.015, 0.015, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        r_init = np.full(N, 0.06)
        configs.append(np.concatenate([pts[:, 0], pts[:, 1], r_init]))
        
    # 2. Perturbed grids

        np.random.seed(seed + 100)
        pts = np.array([[0.08 + i*0.17, 0.08 + j*0.20] for i in range(6) for j in range(5)])
        pts = pts[:N] + np.random.uniform(-0.02, 0.02, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        r_init = np.full(N, 0.06)
        configs.append(np.concatenate([pts[:, 0], pts[:, 1], r_init]))
        
    # 3. Random dense scatter

        np.random.seed(seed + 200)
        pts = np.random.uniform(0.12, 0.88, (N, 2))
        r_init = np.full(N, 0.04)
        configs.append(np.concatenate([pts[:, 0], pts[:, 1], r_init]))
        
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0

    inits = generate_initial_configs()

                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
