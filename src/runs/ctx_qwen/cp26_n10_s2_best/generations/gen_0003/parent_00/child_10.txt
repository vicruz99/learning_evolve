import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraint_fun(v):
    """Compute inequality constraints: boundaries and non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def make_v(centers, radii):
    return np.concatenate([centers[:, 0], centers[:, 1], radii])

def unpack_v(v):
    return np.column_stack((v[:N], v[N:2*N])), v[2*N:]

def compute_safe_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    return np.clip(np.minimum(min_dists / 2.0, wall_dists) * 0.9, 1e-6, 0.4)

def gen_hex_lattice(r0, angle, shift):
    """Generate a rotated and shifted hexagonal lattice."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 20:
        x_start = r0 if row % 2 == 0 else 2 * r0
        x = x_start
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2 * r0
        y += r0 * np.sqrt(3)
        row += 1
        
    pts = np.array(pts)
    c, s = np.cos(angle), np.sin(angle)
    rot_mat = np.array([[c, -s], [s, c]])
    pts = (pts - 0.5) @ rot_mat.T + 0.5 + shift
    
    mask = (pts[:, 0] > 0.02) & (pts[:, 0] < 0.98) & (pts[:, 1] > 0.02) & (pts[:, 1] < 0.98)
    pts = pts[mask]
    if len(pts) < N:
        return None
    np.random.shuffle(pts)
    return pts[:N]

    best_sum = -1.0

                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
