import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: Minimize negative sum of radii (maximize sum)."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and non-overlap (squared for stability)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def compute_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    r = 0.85 * np.minimum(min_dists / 2.0, wall_dists)
    return np.clip(r, 0.005, 0.25)

def optimize_radii_lp(centers):
    """
    Solve for maximum sum of radii given fixed centers using Linear Programming.
    This is exact and removes any slack left by gradient-based optimizers.
    """
    n = len(centers)
    c_obj = -np.ones(n)
    
    A_ub = np.empty((0, n))
    b_ub = np.empty(0)
    
    # Non-overlap: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub = np.vstack([A_ub, row])
            b_ub = np.append(b_ub, d)
            
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        for b in [centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1]]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub = np.vstack([A_ub, row])
            b_ub = np.append(b_ub, b)
            
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    return compute_feasible_radii(centers)

def generate_initial_configs():
    """Generate diverse initial configurations."""
    configs = []
    rng = np.random.default_rng(42)
    
    # 1. Hexagonal lattices with rotations and shifts
    for seed in range(40):
        angle = rng.uniform(-np.pi/6, np.pi/6)
        cx_shift, cy_shift = rng.uniform(-0.05, 0.05, 2)
        r0 = 0.10 + rng.uniform(-0.02, 0.02)
        
        pts = []
        for i in range(-6, 12):
            for j in range(-6, 12):
                x = i * r0 + (j % 2) * r0 * 0.5
                y = j * r0 * np.sqrt(3) * 0.5
                pts.append([x, y])
        pts = np.array(pts)
        
        c, s = np.cos(angle), np.sin(angle)
        rot = np.array([[c, -s], [s, c]])
        pts = pts @ rot.T
        pts -= pts.mean(axis=0)
        pts += [0.5 + cx_shift, 0.5 + cy_shift]
        
        mask = (pts[:, 0] > 0.05) & (pts[:, 0] < 0.95) & (pts[:, 1] > 0.05) & (pts[:, 1] < 0.95)
        pts_valid = pts[mask]
        
        if len(pts_valid) >= N:
            idx = rng.choice(len(pts_valid), N, replace=False)
            centers = pts_valid[idx]
            centers += rng.uniform(-0.005, 0.005, centers.shape)
            centers = np.clip(centers, 0.02, 0.98)
            configs.append(centers)
            
    # 2. Perturbed square grids
    for seed in range(20):
        rng2 = np.random.default_rng(seed + 100)
        pts = []
        for i in range(6):
            for j in range(5):
                pts.append([0.1 + i*0.16 + rng2.uniform(-0.03, 0.03), 
                            0.1 + j*0.20 + rng2.uniform(-0.03, 0.03)])
        if len(pts) >= N:
            configs.append(np.array(pts[:N]))

    # 3. Random dense scatter
    for seed in range(15):
        configs.append(np.random.default_rng(seed + 200).uniform(0.15, 0.85, size=(N, 2)))
        
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-4, 0.25)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    best_centers = None
    best_radii = None
    
    inits = generate_initial_configs()
    
    # Phase 1: Multi-start SLSQP optimization
    for centers in inits:
        r_init = compute_feasible_radii(centers)
        v0 = np.concatenate([centers[:, 0], centers[:, 1], r_init])
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
            
            if -res.fun > best_sum:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    best_sum = -res.fun
                    best_v = res.x.copy()
                    best_centers = best_v[:2*N].reshape(N, 2)
                    best_radii = best_v[2*N:].copy()
        except Exception:
            continue
            
    # Phase 2: Hybrid Refinement (Center Perturbation + LP Radius Maximization)
    # SLSQP finds good basins, LP extracts exact radii. Perturbing centers explores locally.
    if best_centers is not None:
        for step in range(50):
            rng = np.random.default_rng(step + 500)
            
            # Gradually decrease perturbation magnitude
            scale = max(0.05, 1.0 - step * 0.018)
            centers_pert = best_centers + rng.uniform(-0.008 * scale, 0.008 * scale, size=(N, 2))
            centers_pert = np.clip(centers_pert, 0.01, 0.99)
            
            # Exact radius optimization for fixed centers
            radii_opt = optimize_radii_lp(centers_pert)
            current_sum = np.sum(radii_opt)
            
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_pert.copy()
                best_radii = radii_opt.copy()
                
    # Extract final configuration
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    # Phase 3: Strict post-processing to guarantee validator compliance
    # LP ensures feasibility theoretically, but numerical precision requires clipping
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    radii = np.minimum(radii, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    radii = np.maximum(radii, 0.0)
    
    # Resolve any remaining microscopic overlaps
    for _ in range(10):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
