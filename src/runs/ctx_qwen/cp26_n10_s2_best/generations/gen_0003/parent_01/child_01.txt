import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and pairwise non-overlap."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Pre-allocate constraint array
    c = np.empty(4*N + len(PAIR_I))
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap constraints: dist >= r_i + r_j
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = np.sqrt(dx**2 + dy**2) - (r[PAIR_I] + r[PAIR_J])
    
    return c

def get_feasible_r(centers):
    """Compute strictly feasible initial radii for given centers."""
    # Distance to boundaries
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Vectorized pairwise distances
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    
    # r_i <= min_j(dist(i,j) / 2)
    r = np.minimum(r, np.min(dist, axis=1) / 2.0)
    return r * 0.88  # Keep slack for optimizer to expand

def generate_configs():
    """Generate diverse initial configurations."""
    configs = []
    np.random.seed(42)
    
    # 1. Hexagonal lattices with rotations and shifts
    for seed in range(50):
        np.random.seed(seed)
        r0 = 0.095 + np.random.uniform(-0.005, 0.010)
        angle = np.random.uniform(-np.pi/6, np.pi/6)
        shift = np.random.uniform(-0.04, 0.04, 2)
        
        pts = []
        y = r0
        row = 0
        while len(pts) < N + 10:
            x_start = r0 + (row % 2) * r0
            x = x_start
            while x <= 1.0 - r0:
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
            row += 1
            
        pts = np.array(pts[:N+10])
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        pts = pts - 0.5
        pts = pts @ np.array([[cos_a, -sin_a], [sin_a, cos_a]]) + 0.5 + shift
        
        valid = (pts[:,0] >= 0.02) & (pts[:,0] <= 0.98) & \
                (pts[:,1] >= 0.02) & (pts[:,1] <= 0.98)
        if np.sum(valid) >= N:
            pts = pts[valid][:N]
            pts += np.random.uniform(-0.003, 0.003, pts.shape)
            r_init = get_feasible_r(pts)
            configs.append(np.concatenate([pts[:,0], pts[:,1], r_init]))
            
    # 2. Force-based relaxation starts
    for seed in range(15):
        np.random.seed(2000 + seed)
        centers = np.random.uniform(0.2, 0.8, (N, 2))
        radii = np.full(N, 0.05)
        for _ in range(1500):
            radii *= 1.0005
            for i in range(N):
                for j in range(i+1, N):
                    dx = centers[i,0] - centers[j,0]
                    dy = centers[i,1] - centers[j,1]
                    d = np.hypot(dx, dy)
                    if d < radii[i] + radii[j]:
                        overlap = radii[i] + radii[j] - d
                        if d > 1e-5:
                            f = overlap / d * 0.5
                            centers[i,0] += dx * f
                            centers[i,1] += dy * f
                            centers[j,0] -= dx * f
                            centers[j,1] -= dy * f
            centers = np.clip(centers, 0.0, 1.0)
        
        r_init = get_feasible_r(centers)
        configs.append(np.concatenate([centers[:,0], centers[:,1], r_init]))
        
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    configs = generate_configs()
    
    # Phase 1: Multi-start optimization
    for x0 in configs:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-7:
                    best_sum = curr_sum
                    best_v = res.x.copy()
        except Exception:
            continue
            
    if best_v is None:
        best_v = configs[0]
        
    # Phase 2: Perturbation & Refinement to escape local minima
    current_v = best_v.copy()
    for step in range(25):
        pert = current_v.copy()
        # Perturb centers to explore neighborhood
        pert[:2*N] += np.random.uniform(-0.006, 0.006, 2*N)
        pert[:2*N] = np.clip(pert[:2*N], 0.01, 0.99)
        # Shrink radii to unstick circles from jammed local minima
        pert[2*N:] *= 0.92
        
        # Recompute feasible radii to guarantee strict feasibility at restart
        centers_pert = pert[:2*N].reshape(N, 2)
        pert[2*N:] = get_feasible_r(centers_pert) * 0.95
        
        try:
            res = minimize(objective, pert, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-7:
                    best_sum = curr_sum
                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            continue
            
    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    radii = best_v[2*N:].copy()
    
    # Phase 3: Strict Post-Processing to guarantee validator compliance
    # 1. Enforce boundary constraints strictly
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    radii = np.minimum(radii, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    radii = np.maximum(radii, 0.0)
    
    # 2. Enforce non-overlap constraints iteratively with safety margin
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if radii[i] + radii[j] > d + 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
