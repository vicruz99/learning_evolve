import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and non-overlap (squared distances)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r,
        (x[PAIR_I] - x[PAIR_J])**2 + (y[PAIR_I] - y[PAIR_J])**2 - (r[PAIR_I] + r[PAIR_J])**2
    ])
    return c

def compute_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    wall_dists = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    
    r = 0.85 * np.minimum(min_dists / 2.0, wall_dists)
    return np.clip(r, 0.005, 0.25)

def force_directed_relax(centers, max_iter=500):
    """Force-directed layout to spread circles evenly."""
    centers = centers.copy()
    for step in range(max_iter):
        forces = np.zeros_like(centers)
        cooling = max(0.01, 1.0 - step / max_iter)
        
        # Repulsive forces between circles
        for i in range(N):
            for j in range(i + 1, N):
                diff = centers[i] - centers[j]
                d = np.linalg.norm(diff)
                if d < 0.15 and d > 1e-6:
                    f = 0.001 * (0.15 - d) / d * cooling
                    forces[i] += f * diff
                    forces[j] -= f * diff
        
        # Attract to boundaries if too far
        for i in range(N):
            for coord in range(2):
                if centers[i, coord] < 0.15:
                    forces[i, coord] -= 0.0005 * (0.15 - centers[i, coord])
                elif centers[i, coord] > 0.85:
                    forces[i, coord] += 0.0005 * (centers[i, coord] - 0.85)
        
        centers += forces
        centers = np.clip(centers, 0.02, 0.98)
    
    return centers

def generate_hex_pattern(row_counts, r_base, angle_deg=0.0, shift=(0.0, 0.0)):

    pts = []
    y = r_base + shift[1]
    angle = np.deg2rad(angle_deg)
    cos_a, sin_a = np.cos(angle), np.sin(angle)
    
    row_idx = 0
    for count in row_counts:
        x_start = r_base + shift[0] + (row_idx % 2) * r_base
        for k in range(count):
            x = x_start + k * 2.0 * r_base
            pts.append([x, y])
        y += np.sqrt(3) * r_base
        row_idx += 1
    
    pts = np.array(pts[:N])
    
    if angle_deg != 0.0:
        pts = pts - 0.5
        rot_x = pts[:, 0] * cos_a - pts[:, 1] * sin_a
        rot_y = pts[:, 0] * sin_a + pts[:, 1] * cos_a
        pts[:, 0] = rot_x + 0.5
        pts[:, 1] = rot_y + 0.5
    
    return pts
