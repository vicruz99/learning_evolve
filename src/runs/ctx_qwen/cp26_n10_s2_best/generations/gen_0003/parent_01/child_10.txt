import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Compute inequality constraints: boundaries and non-overlap (squared for stability)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def make_feasible(v):
    """Adjusts radii to guarantee strict feasibility for a given configuration."""
    x = v[:N].copy()
    y = v[N:2*N].copy()
    r = v[2*N:].copy()
    
    # Enforce boundary constraints
    r = np.minimum(r, np.minimum(x, 1.0 - x))
    r = np.minimum(r, np.minimum(y, 1.0 - y))
    
    # Enforce non-overlap constraints iteratively
    for _ in range(10):
        changed = False
        for idx in range(NUM_PAIRS):
            i, j = PAIR_I[idx], PAIR_J[idx]
            d = np.hypot(x[i] - x[j], y[i] - y[j])
            if d < r[i] + r[j] - 1e-9:
                shrink = (r[i] + r[j] - d) * 0.5 + 1e-9
                r[i] = max(0.0, r[i] - shrink)
                r[j] = max(0.0, r[j] - shrink)
                changed = True
        if not changed:
            break
    return np.concatenate([x, y, r])

def generate_hex_config(row_counts, r_base, angle, seed):
    """Generates a hexagonal lattice configuration with optional rotation."""
    np.random.seed(seed)
    pts = []
    y = r_base
    for row_idx, count in enumerate(row_counts):
        x_start = r_base if row_idx % 2 == 0 else 2 * r_base
        x = x_start
        for _ in range(count):
            pts.append([x, y])
            x += 2 * r_base
        y += np.sqrt(3) * r_base
        
    pts = np.array(pts[:N])
    
    # Apply rotation around center (0.5, 0.5)
    if angle != 0.0:
        pts = pts - 0.5
        c, s = np.cos(angle), np.sin(angle)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts = pts + 0.5
        
    # Add jitter and clip to safe margins
    pts += np.random.uniform(-0.004, 0.004, pts.shape)
    return np.clip(pts, 0.015, 0.985)

    ]
    
    for pat in patterns:

            continue

                    # Compute feasible initial radii based on geometry

                    r_init *= 0.75 # Leave slack for optimizer to expand

                    configs.append(v0)

    for _ in range(20):

                radii[j] = max(0.0, radii[j] - shrink)
                changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
