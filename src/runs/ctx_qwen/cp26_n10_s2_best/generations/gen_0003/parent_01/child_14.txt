import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Inequality constraints: boundaries and non-overlap (squared distances for stability)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c_bound = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dist_sq = dx**2 + dy**2
    r_sum = r[PAIR_I] + r[PAIR_J]
    c_pair = dist_sq - r_sum**2
    
    return np.concatenate([c_bound, c_pair])

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    r = np.zeros(N)
    for i in range(N):
        d_wall = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        r[i] = d_wall
        for j in range(i+1, N):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            r[i] = min(r[i], d/2.0)
    return r * 0.98

def generate_initial_configs():
    """Generate diverse initial configurations."""
    configs = []
    np.random.seed(42)
    
    # 1. Random uniform placements
    for _ in range(30):
        c = np.random.uniform(0.15, 0.85, (N, 2))
        configs.append(c)
        
    # 2. Hexagonal lattices with varying densities and rotations
    for r0 in [0.085, 0.090, 0.095, 0.100, 0.105]:
        for angle in [0.0, 0.1, -0.1, np.pi/12, -np.pi/12, np.pi/6]:
            pts = []
            y = r0
            row = 0
            while len(pts) < N + 10:
                x = r0 if row % 2 == 0 else 2*r0
                while x <= 1.0 - r0:
                    pts.append([x, y])
                    x += 2*r0
                y += np.sqrt(3)*r0
                row += 1
            pts = np.array(pts[:N+10])
            
            # Rotate around center
            if angle != 0.0:
                pts = pts - 0.5
                c, s = np.cos(angle), np.sin(angle)
                pts = pts @ np.array([[c, -s], [s, c]]) + 0.5
                
            # Filter valid points and add jitter
            mask = (pts[:,0]>=0.05) & (pts[:,0]<=0.95) & (pts[:,1]>=0.05) & (pts[:,1]<=0.95)
            if np.sum(mask) >= N:
                valid = pts[mask]
                idx = np.random.choice(len(valid), N, replace=False)
                c = valid[idx] + np.random.uniform(-0.012, 0.012, (N, 2))
                configs.append(np.clip(c, 0.02, 0.98))
                
    # 3. Grid configurations
    for scale in [0.14, 0.16, 0.18, 0.20]:
        gx = np.linspace(0.05, 0.95, 6)
        gy = np.linspace(0.05, 0.95, 5)
        cx, cy = np.meshgrid(gx, gy)
        pts = np.column_stack([cx.flatten(), cy.flatten()])[:N]
        pts += np.random.uniform(-0.025, 0.025, pts.shape)
        configs.append(np.clip(pts, 0.02, 0.98))
        
    return configs

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    configs = generate_initial_configs()
    
    # Phase 1: Multi-start optimization
    for c_init in configs:
        r_init = get_feasible_radii(c_init)
        v0 = np.concatenate([c_init[:,0], c_init[:,1], r_init])
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-7:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            continue
            
    # Fallback initialization if all starts fail
    if best_v is None:
        c_fb = np.random.uniform(0.2, 0.8, (N, 2))
        r_fb = get_feasible_radii(c_fb)
        best_v = np.concatenate([c_fb[:,0], c_fb[:,1], r_fb])
        best_sum = -np.sum(r_fb)

    current_v = best_v.copy()
    for step in range(45):
        # Gradually shrink radii to force exploration of new basins
        shrink_factor = 0.88 - step * 0.005
        v_pert = current_v.copy()
        v_pert[2*N:] *= max(0.60, shrink_factor)
        
        # Perturb centers to break symmetry
        v_pert[:2*N] += np.random.uniform(-0.007, 0.007, 2*N)
        v_pert[:2*N] = np.clip(v_pert[:2*N], 0.01, 0.99)
        
        # Guarantee strict feasibility before optimization restart
        c_pert = v_pert[:2*N].reshape(N, 2)
        v_pert[2*N:] = get_feasible_radii(c_pert)
        
        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds,

            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-7:

                    best_v = res.x.copy()
                    current_v = best_v.copy()
        except Exception:
            continue
            
    # Extract final configuration
    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    radii = best_v[2*N:].copy()
    
    # Phase 3: Strict post-processing to guarantee validator compliance
    # 1. Enforce boundary constraints strictly
    for i in range(N):
        radii[i] = min(radii[i], centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        
    # 2. Enforce non-overlap strictly with iterative shrinkage
    for _ in range(25):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
