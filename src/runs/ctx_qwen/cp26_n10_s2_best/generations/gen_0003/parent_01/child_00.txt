import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints(v):
    """Vectorized inequality constraints: boundaries and non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + N*(N-1)//2)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    return c

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on local geometry."""
    x = centers[:, 0]
    y = centers[:, 1]
    r = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(x[i]-x[j], y[i]-y[j])
            half_d = d * 0.5
            if half_d < r[i]:
                r[i] = half_d
            if half_d < r[j]:
                r[j] = half_d
    return r * 0.85

def generate_inits():

    inits = []
    np.random.seed(123)
    
    # 1. Hexagonal lattices with varying densities and shifts
    for r0 in [0.085, 0.090, 0.095, 0.100, 0.105]:

            y = r0 + shift
            row = 0

                x = r0 + (row % 2) * r0
                while x <= 1.0 - r0 and len(pts) < N + 10:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
                row += 1
            pts = np.array(pts[:N])
            pts += np.random.uniform(-0.005, 0.005, pts.shape)
            pts = np.clip(pts, 0.02, 0.98)
            r_init = get_feasible_radii(pts)
            inits.append(np.concatenate([pts[:,0], pts[:,1], r_init]))

        rot_pts = base_grid - 0.5
        rot_pts = rot_pts @ np.array([[cos_a, -sin_a], [sin_a, cos_a]]) + 0.5
        mask = (rot_pts[:,0] >= 0.05) & (rot_pts[:,0] <= 0.95) & \
               (rot_pts[:,1] >= 0.05) & (rot_pts[:,1] <= 0.95)
        valid_pts = rot_pts[mask]
        if len(valid_pts) >= N:
            pts = valid_pts[:N] + np.random.uniform(-0.005, 0.005, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            r_init = get_feasible_radii(pts)
            inits.append(np.concatenate([pts[:,0], pts[:,1], r_init]))
            
    # 3. Random dense scatters
    for _ in range(25):
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        r_init = get_feasible_radii(pts)
        inits.append(np.concatenate([pts[:,0], pts[:,1], r_init]))
        
    return inits

def run_packing():
    """Optimizes packing of 26 circles in a unit square to maximize sum of radii."""
    np.random.seed(42)
    bounds = [(0.0, 1.0)]*(2*N) + [(0.001, 0.5)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    inits = generate_inits()
    
    # Phase 1: Multi-start exploration
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,

            s = -res.fun
            if s > best_sum:
                if np.min(constraints(res.x)) >= -1e-7:

                    best_v = res.x.copy()
        except Exception:
            pass

                    cr[j] = max(0.0, cr[j] - shrink)
                    changed = True
        if not changed:
            break
            
    return centers, cr, float(np.sum(cr))
