1. **LP-based radii optimization**: For any fixed center positions, we can solve an LP to find the EXACT optimal radii that maximize the sum. This is much more accurate than SLSQP which jointly optimizes centers and radii (and can get stuck).

2. **Differential Evolution for global center search**: Use DE to find good center positions, evaluating each configuration using LP for radii. DE explores the space globally and is less likely to get stuck in local minima.

3. **LP evaluation of many hexagonal lattice variations**: Generate diverse initial configurations from hexagonal lattices (the theoretically densest packing pattern) with various parameters (density, rotation, shift), compute optimal radii via LP for each, and use these as starting points.

4. **SLSQP local refinement**: After finding good configurations via LP+DE, refine with SLSQP which jointly optimizes centers and radii.

5. **Multi-start refinement with perturbation**: After the best configuration is found, perturb and re-optimize to escape shallow local minima.

The LP approach guarantees optimal radii for any center configuration, and DE guarantees good exploration of the center space. Combining them should push us above 2.636.

import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

# Precompute LP constraint matrix structure (static for all evaluations)
A_ub_lp = np.zeros((4*N + NUM_PAIRS, N))
for i in range(N):
    A_ub_lp[4*i, i] = 1
    A_ub_lp[4*i+1, i] = 1
    A_ub_lp[4*i+2, i] = 1
    A_ub_lp[4*i+3, i] = 1
for k in range(NUM_PAIRS):
    A_ub_lp[4*N + k, PAIR_I[k]] = 1
    A_ub_lp[4*N + k, PAIR_J[k]] = 1

c_obj_lp = -np.ones(N)
r_bounds_lp = [(0.0, 0.5)] * N


def compute_optimal_radii_lp(centers):
    """Given fixed centers, solve LP to find optimal radii maximizing sum."""
    b_ub = np.empty(4*N + NUM_PAIRS)
    for i in range(N):
        b_ub[4*i] = centers[i, 0]
        b_ub[4*i+1] = 1.0 - centers[i, 0]
        b_ub[4*i+2] = centers[i, 1]
        b_ub[4*i+3] = 1.0 - centers[i, 1]
    for k in range(NUM_PAIRS):
        dx = centers[PAIR_I[k], 0] - centers[PAIR_J[k], 0]
        dy = centers[PAIR_I[k], 1] - centers[PAIR_J[k], 1]
        b_ub[4*N + k] = np.sqrt(dx*dx + dy*dy)
    
    res = linprog(c_obj_lp, A_ub=A_ub_lp, b_ub=b_ub, bounds=r_bounds_lp, method='highs')
    if res.success:
        return -res.fun, -res.x
    return 0.0, np.zeros(N)


def de_objective(centers_flat):
    """Objective for DE: negative of LP-optimized sum of radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 0.001, 0.999)
    s, _ = compute_optimal_radii_lp(centers)
    return -s


def s_objective(v):
    """SLSQP objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])


def s_constraints(v):
    """SLSQP inequality constraints: boundaries and non-overlap (squared)."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r,
        (x[PAIR_I] - x[PAIR_J])**2 + (y[PAIR_I] - y[PAIR_J])**2 - (r[PAIR_I] + r[PAIR_J])**2
    ])
    return c


def generate_hex_centers(r_est, shift_x=0.0, shift_y=0.0, angle=0.0):
    """Generate hexagonal lattice centers with various parameters."""
    centers = []
    y = r_est + shift_y
    row = 0
    while len(centers) < N + 10:
        x_start = r_est + shift_x + (row % 2) * r_est
        x = x_start
        while x <= 1.0 - r_est and len(centers) < N + 10:
            centers.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3) * r_est
        row += 1
    
    pts = np.array(centers[:N])
    if angle != 0:
        cx, cy = 0.5, 0.5
        pts = pts - [cx, cy]
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        pts = pts @ np.array([[cos_a, -sin_a], [sin_a, cos_a]]) + [cx, cy]
    return np.clip(pts, 0.001, 0.999)


def ensure_feasible(v):
    """Adjust radii to guarantee strict feasibility."""
    centers = v[:2*N].reshape(N, 2).copy()
    r = v[2*N:].copy()
    
    r = np.minimum(r, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    r = np.minimum(r, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    for _ in range(10):
        changed = False
        for idx in range(NUM_PAIRS):
            i, j = PAIR_I[idx], PAIR_J[idx]
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            if d < r[i] + r[j] - 1e-9:
                shrink = (r[i] + r[j] - d) / 2.0 + 1e-9
                r[i] = max(0.0, r[i] - shrink)
                r[j] = max(0.0, r[j] - shrink)
                changed = True
        if not changed:
            break
    
    return np.concatenate([centers.flatten(), r])


def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Uses LP for optimal radii computation + DE for center search + SLSQP refinement.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Evaluate diverse hexagonal lattice configurations via LP
    center_configs = []
    
    for r_est in [0.085, 0.090, 0.095, 0.100, 0.105, 0.110, 0.115]:
        for shift_x in np.linspace(-0.05, 0.05, 9):
            for shift_y in np.linspace(-0.05, 0.05, 7):
                for angle in [0.0, 0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.2, -0.2]:
                    c = generate_hex_centers(r_est, shift_x, shift_y, angle)
                    c += np.random.uniform(-0.003, 0.003, c.shape)
                    center_configs.append(c.copy())
    
    # Grid-based configurations
    for base_x in [0.08, 0.09, 0.10]:
        for base_y in [0.08, 0.09, 0.10]:
            for spacing in [0.15, 0.16, 0.17, 0.18]:
                pts = []
                for i in range(7):
                    for j in range(4):
                        if len(pts) < N:
                            pts.append([base_x + i * spacing, base_y + j * spacing * 0.9])
                if len(pts) >= N:
                    pts = np.array(pts[:N])
                    pts += np.random.uniform(-0.005, 0.005, pts.shape)
                    center_configs.append(np.clip(pts, 0.001, 0.999))
    
    # Random configurations with good spacing
    for seed in range(30):
        np.random.seed(seed + 3000)
        c = np.random.uniform(0.10, 0.90, (N, 2))
        center_configs.append(c)
    
    # Evaluate all configs with LP
    for centers in center_configs:
        centers = np.clip(centers, 0.001, 0.999)
        s, radii = compute_optimal_radii_lp(centers)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    # Phase 2: Differential Evolution for global center search (with LP evaluation)
    de_bounds = [(0.02, 0.98)] * (2*N)
    
    try:
        res_de = differential_evolution(
            de_objective, de_bounds,
            popsize=20, maxiter=250,
            mutation=(0.3, 0.8), recombination=0.9,
            seed=42, polish=True, tol=1e-8
        )
        de_sum = -res_de.fun
        if de_sum > best_sum:
            de_centers = res_de.x.reshape(N, 2)
            de_sum2, de_radii = compute_optimal_radii_lp(de_centers)
            if de_sum2 > best_sum:
                best_sum = de_sum2
                best_centers = de_centers.copy()
                best_radii = de_radii.copy()
    except Exception:
        pass
    
    # Phase 3: SLSQP refinement from LP-optimized starting point
    if best_centers is not None:
        bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        cons = {'type': 'ineq', 'fun': s_constraints}
        
        v0 = np.concatenate([best_centers[:, 0], best_centers[:, 1], best_radii])
        
        try:
            res = minimize(s_objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            
            curr_sum = -res.fun
            if curr_sum > best_sum:
                c_val = s_constraints(res.x)
                if np.min(c_val) >= -1e-8:
                    best_sum = curr_sum
                    best_centers = res.x[:2*N].reshape(N, 2)
                    best_radii = res.x[2*N:].copy()
        except Exception:
            pass
        
        # Phase 4: Multi-start perturbation refinement
        current_v = np.concatenate([best_centers[:, 0], best_centers[:, 1], best_radii])
        
        for step in range(40):
            np.random.seed(step + 7000)
            v_pert = current_v.copy()
            
            # Vary perturbation magnitude
            pert_mag = 0.008 * (1.0 - step * 0.01)
            v_pert[:2*N] += np.random.uniform(-pert_mag, pert_mag, 2*N)
            v_pert[:2*N] = np.clip(v_pert[:2*N], 0.01, 0.99)
            v_pert[2*N:] *= 0.92
            
            # Ensure feasibility
            v_pert = ensure_feasible(v_pert)
            
            try:
                res = minimize(s_objective, v_pert, method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    c_val = s_constraints(res.x)
                    if np.min(c_val) >= -1e-8:
                        best_sum = curr_sum
                        best_centers = res.x[:2*N].reshape(N, 2)
                        best_radii = res.x[2*N:].copy()
                        current_v = res.x.copy()
            except Exception:
                pass
        
        # Phase 5: LP-guided refinement
        # After SLSQP, recompute radii via LP for potentially better result
        lp_sum, lp_radii = compute_optimal_radii_lp(best_centers)
        if lp_sum > best_sum:
            best_sum = lp_sum
            best_radii = lp_radii.copy()
    
    # Phase 6: Additional random center optimization with LP
    for seed in range(20):
        np.random.seed(seed + 9000)
        # Generate random centers, compute LP radii, then SLSQP refine
        centers_rand = np.random.uniform(0.05, 0.95, (N, 2))
        _, radii_rand = compute_optimal_radii_lp(centers_rand)
        
        v0 = np.concatenate([centers_rand[:, 0], centers_rand[:, 1], radii_rand])
        bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        cons = {'type': 'ineq', 'fun': s_constraints}
        
        try:
            res = minimize(s_objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                c_val = s_constraints(res.x)
                if np.min(c_val) >= -1e-8:
                    best_sum = curr_sum
                    best_centers = res.x[:2*N].reshape(N, 2)
                    best_radii = res.x[2*N:].copy()
        except Exception:
            pass
    
    # Extract and post-process
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    # Strict boundary enforcement
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    radii = np.minimum(radii, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    radii = np.maximum(radii, 0.0)
    
    # Strict non-overlap enforcement
    for _ in range(20):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-10
                    radii[i] = max(0.0, radii[i] - shrink)
                    radii[j] = max(0.0, radii[j] - shrink)
                    changed = True
        if not changed:
            break
    
    return centers, radii, float(np.sum(radii))
