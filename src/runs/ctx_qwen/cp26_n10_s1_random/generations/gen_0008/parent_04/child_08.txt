To maximize the sum of radii for 26 circles in a unit square, we recognize that for any fixed set of center positions, the optimal radii can be found exactly by solving a Linear Program (LP). This transforms the problem into a non-linear optimization over center coordinates: `maximize LP_Sum(Centers)`.

My strategy employs a multi-phase hybrid optimizer:
1. **Diverse Initialization**: Generate configurations from hexagonal lattices (various row distributions and densities), plus purely random placements. This ensures we explore both structured dense packings and irregular optimal configurations.
2. **LP-Based Hill Climbing**: For each initialization, perform a multi-scale random perturbation search. The step size decays exponentially, allowing broad exploration initially followed by fine-tuning.
3. **Coordinate-Wise Local Search**: Perform greedy line searches along x and y axes for each circle. This efficiently escapes shallow local minima that random perturbations might miss.
4. **Nelder-Mead Refinement**: Use a derivative-free simplex method on the best centers found so far. Since the LP objective is non-differentiable but well-behaved, Nelder-Mead converges reliably to high-quality basins.
5. **Precomputed LP Structure**: The constraint matrix `A_ub` for the LP has a fixed structure. Precomputing it avoids redundant allocations inside the tight optimization loops.
6. **Strict Safety Scaling**: Before returning, compute the maximum feasible scaling factor for all radii to guarantee strict satisfaction of the `1e-12` numerical tolerance in the validation function.

This approach decouples center placement from radius sizing, leveraging the speed and exactness of `scipy.optimize.linprog` to guide a robust spatial search.

import numpy as np
from scipy.optimize import linprog, minimize

def get_lp_sum_and_radii(centers, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed center positions."""
    n = centers.shape[0]
    
    # Maximum radius for each circle constrained by square boundaries
    lims = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    # Pairwise distances
    diffs = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_func_for_nm(x_flat, n, A_ub, idx_i, idx_j):
    """Objective for Nelder-Mead: negative of LP sum of radii."""
    centers = x_flat.reshape(n, 2)
    # Strictly keep centers inside to prevent LP bounds from collapsing
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, s = get_lp_sum_and_radii(centers, A_ub, idx_i, idx_j)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    # Precompute constant LP structure
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # 1. Generate diverse initial configurations
    inits = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [6, 6, 4, 6, 4], [5, 5, 6, 5, 5], [6, 4, 6, 4, 6],
        [7, 6, 7, 6], [6, 7, 6, 7], [8, 6, 6, 6], [6, 8, 6, 6],
        [5, 7, 6, 8], [7, 5, 7, 7]
    ]
    
    for pat in patterns:
        if sum(pat) != 26: 
            continue
        pts = []
        y = 0.08
        for idx, count in enumerate(pat):
            shift = 0.08 if idx % 2 == 1 else 0.0
            x = 0.08 + shift
            for _ in range(count):
                if len(pts) >= n: 
                    break
                pts.append([x, y])
                x += 0.16
            y += 0.08 * np.sqrt(3)
        if len(pts) == n:
            inits.append(np.array(pts))
            
    # Add random starts to break lattice symmetry
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # 2. Local optimization for each configuration
    for cfg in inits:
        curr_centers = cfg.copy()
        curr_radii, curr_sum = get_lp_sum_and_radii(curr_centers, A_ub, idx_i, idx_j)
        if curr_radii is None: 
            continue
            
        # Multi-scale random hill climbing
        for step_mult in np.logspace(1.0, -4.0, 20):
            step = 0.025 * step_mult
            improved = False
            for _ in range(30):
                pert = curr_centers + rng.uniform(-step, step, (n, 2))
                pert = np.clip(pert, 0.02, 0.98)
                r_p, s_p = get_lp_sum_and_radii(pert, A_ub, idx_i, idx_j)
                if r_p is not None and s_p > curr_sum:
                    curr_centers = pert.copy()
                    curr_radii = r_p.copy()
                    curr_sum = s_p
                    improved = True
            if not improved:
                break
                
        # Coordinate-wise greedy local search
        for _ in range(40):
            for i in range(n):
                old_x, old_y = curr_centers[i]
                best_improved = False
                
                # Search along X
                for dx in np.linspace(-0.03, 0.03, 7):
                    curr_centers[i, 0] = np.clip(old_x + dx, 0.02, 0.98)
                    r_c, s_c = get_lp_sum_and_radii(curr_centers, A_ub, idx_i, idx_j)
                    if r_c is not None and s_c > curr_sum:
                        curr_radii = r_c.copy()
                        curr_sum = s_c
                        best_improved = True
                        
                # Search along Y
                for dy in np.linspace(-0.03, 0.03, 7):
                    curr_centers[i, 1] = np.clip(old_y + dy, 0.02, 0.98)
                    r_c, s_c = get_lp_sum_and_radii(curr_centers, A_ub, idx_i, idx_j)
                    if r_c is not None and s_c > curr_sum:
                        curr_radii = r_c.copy()
                        curr_sum = s_c
                        best_improved = True
                        
                if not best_improved:
                    curr_centers[i] = [old_x, old_y]
                    
        if curr_sum > best_sum:
            best_centers = curr_centers.copy()
            best_radii = curr_radii.copy()
            best_sum = curr_sum
            
    # 3. Nelder-Mead refinement on the best found configuration
    if best_centers is not None:
        res = minimize(
            obj_func_for_nm, 
            best_centers.flatten(), 
            args=(n, A_ub, idx_i, idx_j),
            method='Nelder-Mead',
            options={'maxiter': 2500, 'xatol': 1e-7, 'fatol': 1e-9}
        )
        opt_centers = np.clip(res.x.reshape(n, 2), 1e-4, 1.0 - 1e-4)
        opt_radii, opt_sum = get_lp_sum_and_radii(opt_centers, A_ub, idx_i, idx_j)
        if opt_radii is not None and opt_sum > best_sum:
            best_centers = opt_centers
            best_radii = opt_radii
            best_sum = opt_sum
            
    # 4. Final strict safety scaling to guarantee validation tolerance
    if best_radii is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-9:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
                
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-9:
                    scale = min(scale, d / rs)
                    
        # Apply scaling with tiny buffer for float precision safety
        best_radii *= scale * 0.999999
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
