import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_for_centers(centers, n, triu_i, triu_j, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 0.0)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in lims]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[triu_i, triu_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 0.0), 0.0

def lp_wrapper(x, n, triu_i, triu_j, A_ub):
    """Wrapper to optimize centers by maximizing the LP radii sum."""
    c = np.clip(x.reshape(n, 2), 0.005, 0.995)
    _, s = solve_lp_for_centers(c, n, triu_i, triu_j, A_ub)
    return -s

def joint_obj(v, n):
    """Objective for joint center/radius optimization."""
    return -np.sum(v[2*n:])

def joint_cons(v, n):
    """Constraints for joint center/radius optimization."""
    cx, cy, cr = v[:n], v[n:2*n], v[2*n:]
    con = [cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr]
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(d, np.inf)
    rs = cr[:, None] + cr[None, :]
    idx = np.triu_indices(n, k=1)
    con.append(d[idx] - rs[idx])
    return np.concatenate(con)

def run_packing():
    n = 26
    triu_i, triu_j = np.triu_indices(n, k=1)
    A_ub = np.zeros((len(triu_i), n))
    A_ub[np.arange(len(triu_i)), triu_i] = 1.0
    A_ub[np.arange(len(triu_i)), triu_j] = 1.0
    
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [6,6,4,6,4],
        [5,5,6,5,5], [6,4,6,4,6], [7,6,7,6], [6,7,6,7],
        [8,5,7,6], [5,8,6,7], [4,5,6,6,5], [6,6,6,4,4]
    ]
    
    for pat in patterns:
        if sum(pat) != n: continue
        for r0 in [0.08, 0.09, 0.10]:
            for angle in [0.0, 0.05, -0.05, 0.1, -0.1, 0.15]:
                centers = []
                y = r0
                for ri, cnt in enumerate(pat):
                    shift = r0 if ri % 2 == 1 else 0.0
                    x = r0 + shift
                    for k in range(cnt):
                        if len(centers) >= n: break
                        cx = x + y * np.tan(angle)
                        cy = y
                        centers.append([cx, cy])
                    y += r0 * np.sqrt(3) * np.cos(angle)
                if len(centers) == n:
                    c_arr = np.array(centers)
                    c_arr -= c_arr.min(axis=0)
                    scale = 0.85 / c_arr.max(axis=0).max()
                    c_arr = c_arr * scale + (1.0 - scale) / 2.0
                    configs.append(c_arr)
                    
    for _ in range(15):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    for cfg in configs:
        cfg = cfg.copy()
        
        # Phase 1: Optimize centers to maximize LP value using Powell's method
        try:
            res = minimize(lp_wrapper, cfg.flatten(), args=(n, triu_i, triu_j, A_ub),

            opt_centers = np.clip(res.x.reshape(n, 2), 0.001, 0.999)

                    curr_s = s_try
                    curr_r = r_try.copy()

                        best_centers = curr_c.copy()
                        best_radii = curr_r.copy()
                else:
                    curr_c[idx] = old
                step *= 0.995
                
    # Final strict safety scaling to guarantee numerical validity
    if best_radii is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r < 1e-9: continue
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-9:
                    scale = min(scale, d / rs)
                    
        best_radii *= scale * 0.999999
        best_sum = float(np.sum(best_radii))
        
    # Fallback configuration
    if best_centers is None:
        best_centers = rng.uniform(0.2, 0.8, (n, 2))
        best_radii, best_sum = solve_lp_for_centers(best_centers, n, triu_i, triu_j, A_ub)
        best_radii *= 0.99
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
