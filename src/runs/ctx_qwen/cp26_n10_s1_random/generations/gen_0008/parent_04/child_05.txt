from scipy.optimize import minimize, linprog


def solve_lp(centers, n, m, idx_i, idx_j, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    d = np.min(
        np.stack([
            centers[:, 0], 1.0 - centers[:, 0],
            centers[:, 1], 1.0 - centers[:, 1]
        ], axis=1), axis=1
    )
    d = np.maximum(d, 1e-12)
    
    b_ub = np.sqrt(
        (centers[idx_i, 0] - centers[idx_j, 0]) ** 2 +
        (centers[idx_i, 1] - centers[idx_j, 1]) ** 2
    )
    
    bounds = [(0.0, di) for di in d]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0


def force_sim(seed, steps=8000, n=26):

        dists = np.sqrt(dists_sq + 1e-12)

        dirs = diff * inv_dists[:, :, None]

def generate_hex_config(row_counts, r0, rng, n=26, perturb_std=0.0):
    """Generates hexagonal lattice configuration."""
    pts = []
    y_curr = r0
    row = 0
    for count in row_counts:
        shift = r0 if row % 2 == 1 else 0.0
        x_start = r0 + shift
        for k in range(count):
            if len(pts) >= n:
                break
            pts.append([x_start + k * 2.0 * r0, y_curr])
        y_curr += r0 * np.sqrt(3)
        row += 1
    
    pts = np.array(pts[:n])
    
    if perturb_std > 0:
        pts += rng.normal(0, perturb_std, pts.shape)
        pts = np.clip(pts, 0.05, 0.95)
    
    mx, my = pts.min(0)
    Mx, My = pts.max(0)
    denom = np.array([Mx - mx, My - my])
    denom = np.where(denom < 1e-10, 1.0, denom)
    pts = (pts - np.array([mx, my])) / denom * 0.82 + 0.09
    
    return pts


def slsqp_objective(x, n):
    return -np.sum(x[2::3])


def slsqp_constraint(x, n):
    c = np.zeros(4 * n + n * (n - 1) // 2)
    c[:n] = x[0::3] - x[2::3]
    c[n:2 * n] = 1.0 - x[0::3] - x[2::3]
    c[2 * n:3 * n] = x[1::3] - x[2::3]
    c[3 * n:4 * n] = 1.0 - x[1::3] - x[2::3]
    
    off = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            di = off + (i * n + j - i * (i + 1) // 2 - i)
            dx = x[3 * i] - x[3 * j]
            dy = x[3 * i + 1] - x[3 * j + 1]
            c[di] = np.sqrt(dx * dx + dy * dy) - x[2 * n + i] - x[2 * n + j]
    
    return c
