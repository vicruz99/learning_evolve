

<strategy>
The problem requires packing 26 circles in a unit square to maximize the sum of radii. This is a non-convex optimization problem where centers and radii are coupled. A highly effective strategy for this class of problems is:
1. **LP-based Radius Optimization**: For any fixed set of centers, the maximum sum of radii is found exactly via a Linear Program. Constraints are $r_i + r_j \le \text{dist}(i,j)$ and $r_i \le \min(x_i, 1-x_i, y_i, 1-y_i)$.
2. **Stochastic Hill-Climbing on Centers**: Since the LP objective over centers is non-smooth but well-defined, we can optimize centers using adaptive random perturbations. Moving centers apart generally relaxes pairwise constraints, allowing larger radii. We adapt the step size based on success/failure to balance exploration and exploitation.
3. **Diverse Initialization**: We generate multiple starting configurations from hexagonal lattice patterns (known to be dense) with varying row distributions and initial radii, plus random starts. Each is polished via the hill-climbing routine.
4. **Safety Scaling**: Finally, we compute the tightest valid scaling factor for the best configuration and apply a tiny buffer to guarantee strict numerical feasibility under the $10^{-12}$ tolerance in the validator.

This approach avoids gradient-based pitfalls with non-convex constraints, leverages fast LP solves, and thoroughly explores the configuration space.

import numpy as np
from scipy.optimize import linprog

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Max radius limited by distance to boundaries
    wall_dists = np.min([centers[:, 0], 1.0 - centers[:, 0], 
                         centers[:, 1], 1.0 - centers[:, 1]], axis=0)
    wall_dists = np.maximum(wall_dists, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in wall_dists]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def hill_climbing_lp(centers, n_iter=300, seed=0):
    """Optimizes centers by hill-climbing with LP evaluation of radii sum."""
    rng = np.random.default_rng(seed)
    n = centers.shape[0]
    best_centers = centers.copy()
    radii, best_sum = solve_lp_radii(best_centers)
    best_radii = radii
    
    sigma = 0.015
    for step in range(n_iter):
        # Occasional large jump to escape local minima
        if rng.random() < 0.03:
            sigma = 0.05
        else:
            sigma = np.clip(sigma, 0.0005, 0.04)
            
        # Pick 1 or 2 circles to move
        k = 2 if rng.random() < 0.3 else 1
        idxs = rng.choice(n, k, replace=False)
            
        c_try = best_centers.copy()
        for idx in idxs:
            c_try[idx] += rng.normal(0, sigma, 2)
            c_try[idx] = np.clip(c_try[idx], 1e-4, 1.0 - 1e-4)
            
        r_try, s_try = solve_lp_radii(c_try)
        
        if s_try > best_sum:
            best_centers = c_try
            best_radii = r_try
            best_sum = s_try
            sigma *= 1.08
        else:
            sigma *= 0.94
            
    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Generate initial configs
    configs = []
    
    # Hexagonal patterns that sum to 26
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [6, 6, 4, 6, 4], [5, 5, 6, 5, 5], [6, 4, 6, 4, 6],
        [7, 6, 7, 6], [6, 7, 6, 7], [5, 7, 5, 7], [7, 5, 7, 5],
        [5, 6, 6, 5, 4], [6, 5, 4, 6, 5], [6, 6, 5, 5, 4]
    ]
    
    for pat in patterns:
        if sum(pat) != 26: continue
        for r_init in [0.092, 0.096, 0.100]:
            pts = []
            y = r_init
            row = 0
            for cnt in pat:
                shift = r_init if row % 2 == 1 else 0.0
                x = r_init + shift
                for _ in range(cnt):
                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2.0 * r_init
                y += r_init * np.sqrt(3)
                row += 1
            if len(pts) == n:
                pts = np.array(pts)
                # Normalize to fit comfortably in [0.05, 0.95]
                min_pts = pts.min(axis=0)
                max_pts = pts.max(axis=0)
                range_pts = np.maximum(max_pts - min_pts, 1e-9)
                pts_norm = (pts - min_pts) / range_pts
                pts_final = pts_norm * 0.85 + 0.075
                configs.append(pts_final)
                # Add perturbed version
                configs.append(pts_final + rng.uniform(-0.02, 0.02, pts_final.shape))
                
    # Random starts
    for _ in range(6):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Optimize each config
    for i, cfg in enumerate(configs):
        cfg = np.clip(cfg, 0.01, 0.99)
        c, r, s = hill_climbing_lp(cfg, n_iter=300, seed=42 + i)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r
            
    # Final strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y, ri = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if ri > 1e-9:
            scale = min(scale, x/ri, (1.0-x)/ri, y/ri, (1.0-y)/ri)
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
