import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_for_centers(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    bounds = [(0.0, lim) for lim in ub]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def slsqp_obj(v, n):
    """Objective: maximize sum of radii."""
    return -np.sum(v[:n])

def slsqp_cons(v, n):
    """Constraints: pairwise squared distances >= squared sum of radii."""
    r, u, v_ = v[:n], v[n:2*n], v[2*n:3*n]
    d = 1.0 - 2.0 * r
    x = r + d * u
    y = r + d * v_
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dsq = dx**2 + dy**2
    rsq = (r[:, None] + r[None, :])**2
    triu_i, triu_j = np.triu_indices(n, k=1)
    return dsq[triu_i, triu_j] - rsq[triu_i, triu_j]

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Generate diverse initial configurations
    configs = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 6, 6, 4],
        [6, 6, 4, 6, 4], [5, 5, 6, 5, 5], [6, 4, 6, 4, 6],
        [6, 7, 6, 7], [7, 6, 7, 6], [5, 7, 5, 6, 3], [6, 6, 6, 5, 3]
    ]
    
    for pat in patterns:
        if sum(pat) != n: continue
        for r_init in [0.085, 0.090, 0.095, 0.100]:
            pts = []
            y = r_init
            for idx, cnt in enumerate(pat):
                shift = r_init if idx % 2 == 1 else 0.0
                x = r_init + shift
                for _ in range(cnt):
                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2.0 * r_init
                y += np.sqrt(3) * r_init
            if len(pts) == n:
                configs.append(np.array(pts))
                
    for _ in range(8):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Phase 2: SLSQP optimization on parameterized variables
    bounds_vars = [(1e-5, 0.5)] * n + [(0.0, 1.0)] * n + [(0.0, 1.0)] * n
    cons_dict = {'type': 'ineq', 'fun': slsqp_cons, 'args': (n,)}
    
    for cfg in configs:
        r0 = np.full(n, 0.09)
        denom = 1.0 - 2.0 * r0
        u0 = np.clip((cfg[:, 0] - r0) / denom, 0.0, 1.0)
        v0 = np.clip((cfg[:, 1] - r0) / denom, 0.0, 1.0)
        x0 = np.concatenate([r0, u0, v0])
        
        try:
            res = minimize(slsqp_obj, x0, args=(n,), method='SLSQP', bounds=bounds_vars, 
                           constraints=cons_dict, options={'maxiter': 6000, 'ftol': 1e-12})
            if np.isfinite(res.fun):
                r_opt = res.x[:n]
                u_opt = res.x[n:2*n]
                v_opt = res.x[2*n:3*n]
                d_opt = 1.0 - 2.0 * r_opt
                c_opt = np.column_stack((r_opt + d_opt * u_opt, r_opt + d_opt * v_opt))
                rad, sm = solve_lp_for_centers(c_opt)
                if sm > best_sum:
                    best_sum = sm
                    best_centers = c_opt.copy()
                    best_radii = rad.copy()
        except Exception:
            pass
            
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum = solve_lp_for_centers(best_centers)

    # Phase 3: Iterative Hill Climbing with LP refinement
    current_sum = best_sum
    current_centers = best_centers.copy()
    current_radii = best_radii.copy()
    
    step_size = 0.025
    for step in range(8000):
        i = rng.integers(n)
        old_c = current_centers[i].copy()
        
        noise = rng.normal(0, step_size, 2)
        current_centers[i] += noise
        current_centers[i] = np.clip(current_centers[i], 1e-4, 1.0 - 1e-4)
        
        rad, sm = solve_lp_for_centers(current_centers)
        if sm > current_sum + 1e-7:
            current_sum = sm
            current_radii = rad.copy()
        else:
            current_centers[i] = old_c
            
        if step % 1500 == 0 and step > 0:
            step_size *= 0.75

    best_centers = current_centers
    best_radii = current_radii
    best_sum = current_sum

    # Phase 4: Strict numerical safety scaling
    scale = 1.0
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
