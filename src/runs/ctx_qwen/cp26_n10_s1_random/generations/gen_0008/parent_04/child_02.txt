import numpy as np
from scipy.optimize import minimize, linprog

def objective_func(vars, n):
    """Objective: minimize negative sum of radii => Maximize sum of radii"""
    return -np.sum(vars[:n])

def compute_constraints_slscp(vars, n, i_idx, j_idx):
    """Constraints for SLSQP: pairwise squared distances >= squared sum of radii"""
    r = vars[:n]
    u = vars[n:2*n]
    v = vars[2*n:3*n]
    
    # Parameterization guarantees x in [r, 1-r] and y in [r, 1-r]
    denom = 1.0 - 2.0 * r
    x = r + denom * u
    y = r + denom * v
    
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[i_idx] + r[j_idx]
    return dist_sq - r_sum**2

def solve_lp_radii(centers, A_ub, n, i_idx, j_idx):
    """Solves LP to maximize sum of radii for fixed centers."""
    wall_dists = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    wall_dists = np.maximum(wall_dists, 0.0)
    
    dx = centers[:, np.newaxis, 0] - centers[np.newaxis, :, 0]
    dy = centers[:, np.newaxis, 1] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    b_ub = dists[i_idx, j_idx]
    
    bounds = [(0.0, lim) for lim in wall_dists]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    m = len(i_idx)

        [5, 5, 6, 5, 5], [6, 5, 6, 5, 4], [4, 6, 6, 6, 4],

        [7, 6, 7, 6], [6, 7, 6, 7], [5, 6, 6, 5, 4], [6, 5, 4, 6, 5]
    ]
    
    configs = []
    for pat in patterns:
        if sum(pat) != n: 
            continue
        for r_init in [0.085, 0.090, 0.095, 0.100]:
            centers = []
            y_curr = r_init
            for row_idx, count in enumerate(pat):
                shift = r_init if row_idx % 2 == 1 else 0.0
                x_start = r_init + shift
                for k in range(count):
                    if len(centers) >= n: 
                        break
                    centers.append([x_start + k * 2.0 * r_init, y_curr])
                y_curr += r_init * np.sqrt(3)
            centers = np.array(centers[:n])

            v = (centers[:, 1] - r_init) / denom

    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
