import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    wall_dists = np.min([centers[:, 0], 1.0 - centers[:, 0], 
                         centers[:, 1], 1.0 - centers[:, 1]], axis=0)
    wall_dists = np.maximum(wall_dists, 1e-9)
    
    bounds = [(0.0, lim) for lim in wall_dists]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def obj_joint(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_joint(v, n):
    """Constraints: boundary inclusion and pairwise non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    c.append(d2[idx_i, idx_j] - rs[idx_i, idx_j]**2)
    return np.concatenate(c)
