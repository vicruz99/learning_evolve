import numpy as np
from scipy.optimize import linprog

def solve_lp_fast(centers, A_ub, i_idx, j_idx):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-9)) for lim in limits]
    
    diffs = centers[i_idx] - centers[j_idx]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun
    return None, 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    i_idx, j_idx = np.triu_indices(n, k=1)
    A_ub = np.zeros((len(i_idx), n))
    A_ub[np.arange(len(i_idx)), i_idx] = 1.0
    A_ub[np.arange(len(i_idx)), j_idx] = 1.0
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    np.random.seed(42)
    
    # Generate diverse initial configurations
    inits = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
        [5, 5, 6, 6, 4], [7, 5, 5, 5, 4], [5, 6, 4, 6, 5], 
        [4, 6, 6, 6, 4], [6, 5, 5, 6, 4]
    ]
    
    for pat in patterns:
        if sum(pat) != n: 
            continue
        pts = []
        y = 0.09
        dy = 0.155
        dx = 0.17
        for ri, cnt in enumerate(pat):
            shift = dx * 0.5 if ri % 2 == 1 else 0.0
            x_start = (1.0 - (cnt - 1) * dx) / 2.0 + shift
            for _ in range(cnt):
                pts.append([x_start, y])
                x_start += dx
            y += dy
        inits.append(np.array(pts))
        
    # Grid + center
    gx = np.linspace(0.12, 0.88, 5)
    gy = np.linspace(0.12, 0.88, 5)
    grid = np.array([(x, y) for y in gy for x in gx])
    inits.append(np.vstack([grid, [0.5, 0.5]]))
    
    # Random starts
    for _ in range(3):
        inits.append(np.random.uniform(0.1, 0.9, (n, 2)))
        
    for cfg in inits:
        c = cfg.copy()
        _, curr_sum = solve_lp_fast(c, A_ub, i_idx, j_idx)
        if curr_sum <= best_sum: 
            continue
        
        best_c = c.copy()
        best_s = curr_sum
        stagnation = 0
        
        for step in range(10000):
            step_size = 0.04 * max(0.05, 1.0 - step/10000.0)
            i = np.random.randint(n)
            old_pos = best_c[i].copy()
            
            best_move_s = best_s
            best_move_pos = old_pos
            current_r = None
            
            # Try small random perturbations
            for _ in range(6):
                cand = best_c[i] + np.random.uniform(-step_size, step_size, 2)
                cand = np.clip(cand, 0.02, 0.98)
                best_c[i] = cand
                current_r, s = solve_lp_fast(best_c, A_ub, i_idx, j_idx)
                if current_r is not None and s > best_move_s + 1e-9:
                    best_move_s = s
                    best_move_pos = cand.copy()
                    
            # Recovery for crushed circles
            if current_r is not None and current_r[i] < 0.015 and np.random.rand() < 0.2:
                for _ in range(3):
                    cand = np.random.uniform(0.1, 0.9, 2)
                    best_c[i] = cand
                    r2, s2 = solve_lp_fast(best_c, A_ub, i_idx, j_idx)
                    if r2 is not None and s2 > best_move_s + 1e-9:
                        best_move_s = s2
                        best_move_pos = cand.copy()

                best_s = best_move_s
                best_c[i] = best_move_pos
                stagnation = 0

                stagnation += 1

                _, best_s = solve_lp_fast(best_c, A_ub, i_idx, j_idx)
                stagnation = 0

            best_centers = best_c.copy()
            best_radii, _ = solve_lp_fast(best_c, A_ub, i_idx, j_idx)

    if best_centers is None or best_radii is None:
        best_centers = np.ones((n, 2)) * 0.5
        best_radii = np.full(n, 0.05)
        best_sum = float(np.sum(best_radii))

    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

    dists = np.sqrt(np.sum(diffs**2, axis=1))
    rs = best_radii[i_idx] + best_radii[j_idx]
    valid = rs > 1e-12
    if np.any(valid):
        scale = min(scale, np.min(dists[valid] / rs[valid]))
        
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
