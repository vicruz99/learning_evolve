1. **LP-Based Initialization**: For any fixed set of centers, the optimal radii can be found exactly via a Linear Program (LP). This gives a tight, feasible starting point for nonlinear optimization.
2. **Diverse Structured Starts**: Generate multiple hexagonal lattice configurations and random dense configurations to cover different basin attractors.
3. **Joint SLSQP Optimization**: Use Sequential Least Squares Programming to simultaneously adjust centers and radii. This handles the non-convex constraints (distances >= sum of radii) effectively when started near feasibility.
4. **Iterative Refinement Loop**: Alternate between LP radius extraction and SLSQP center/radius polishing with small random perturbations to escape shallow local minima.
5. **Strict Safety Scaling**: Before returning, compute the exact feasible scaling factor to guarantee the `1e-12` tolerance in the validator is strictly satisfied.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def obj_func(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def cons_func(x):
    """Inequality constraints >= 0 for valid packing."""
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    
    # Preallocate constraint array
    m = 4*N + N*(N-1)//2
    c = np.empty(m)
    
    # Boundary constraints
    idx = 0
    c[idx:idx+N] = xc - r
    idx += N
    c[idx:idx+N] = 1.0 - xc - r
    idx += N
    c[idx:idx+N] = yc - r
    idx += N
    c[idx:idx+N] = 1.0 - yc - r
    idx += N
    
    # Pairwise non-overlap constraints
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    c[idx:] = dists[PAIR_I, PAIR_J] - (r[PAIR_I] + r[PAIR_J])
    
    return c

def get_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-9)) for lim in limits]
    
    m = N * (N - 1) // 2
    A = np.zeros((m, N))
    b = np.sqrt((centers[PAIR_I, 0] - centers[PAIR_J, 0])**2 + 
                (centers[PAIR_I, 1] - centers[PAIR_J, 1])**2)
    
    for k, (u, v) in enumerate(zip(PAIR_I, PAIR_J)):
        A[k, u] = 1.0
        A[k, v] = 1.0
        
    try:
        res = linprog(-np.ones(N), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-6), 1e-6 * N

def generate_hex_configs(rng, num_patterns=15):
    """Generates multiple hexagonal lattice configurations."""
    configs = []
    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [6, 6, 5, 5, 4], [5, 4, 6, 6, 5], [5, 6, 4, 6, 5], [6, 5, 5, 6, 4],
        [7, 5, 5, 5, 4], [5, 6, 5, 5, 5], [6, 6, 6, 4, 4], [5, 5, 7, 5, 4]
    ]
    
    for pat in row_patterns:
        if sum(pat) < N: continue
        pts = []
        y = 0.095
        row = 0
        for cnt in pat:
            shift = 0.095 if row % 2 == 1 else 0.0
            x = 0.095 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 0.19
            y += 0.165
            row += 1
        base = np.array(pts[:N])
        configs.append(base)
        
        # Add perturbations
        for _ in range(num_patterns // len(row_patterns) + 1):
            p = base + rng.uniform(-0.025, 0.025, base.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_c = None
    best_r = None
    
    bounds_vars = [(0.0, 1.0)] * (2*N) + [(1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': cons_func}
    
    # Collect diverse starts
    starts = generate_hex_configs(rng, num_patterns=15)
    for _ in range(10):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    for cfg in starts:
        # LP to get tight initial radii
        r_lp, _ = get_lp_radii(cfg)
        
        # Create initial vector for SLSQP
        x0 = np.zeros(3*N)
        x0[0::3] = cfg[:, 0]
        x0[1::3] = cfg[:, 1]
        x0[2::3] = r_lp * 0.95  # Slight shrink for strict initial feasibility
        
        # Perturb radii to strongly encourage unequal packing exploration
        x0[2::3] *= (0.7 + 0.6 * rng.random(N))
        x0[2::3] = np.clip(x0[2::3], 1e-6, 0.45)
        
        try:
            res = minimize(obj_func, x0, method='SLSQP', bounds=bounds_vars,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13})
            
            if np.isfinite(res.fun):
                c_opt = res.x[:2*N].reshape(N, 2)
                r_opt = res.x[2*N:]
                
                # Quick strict validity check
                valid = True
                if np.any(r_opt < 1e-8): valid = False
                if np.any(c_opt[:, 0] - r_opt < -1e-9) or np.any(c_opt[:, 0] + r_opt > 1.0 + 1e-9): valid = False
                if np.any(c_opt[:, 1] - r_opt < -1e-9) or np.any(c_opt[:, 1] + r_opt > 1.0 + 1e-9): valid = False
                if valid:
                    dists = np.sqrt((c_opt[:, None, 0] - c_opt[None, :, 0])**2 + 
                                    (c_opt[:, None, 1] - c_opt[None, :, 1])**2)
                    if np.any(dists[PAIR_I, PAIR_J] < r_opt[PAIR_I] + r_opt[PAIR_J] - 1e-9):
                        valid = False
                        
                if valid:
                    s = np.sum(r_opt)
                    if s > best_sum:
                        best_sum = s
                        best_c = c_opt.copy()
                        best_r = r_opt.copy()
        except Exception:
            continue
            
    # Iterative refinement: LP squeeze + SLSQP restart with micro-perturbations
    if best_c is not None:
        for _ in range(12):
            r_lp, s_lp = get_lp_radii(best_c)
            if s_lp > best_sum:
                best_r = r_lp.copy()
                best_sum = s_lp
                
            x0 = np.zeros(3*N)
            x0[0::3] = best_c[:, 0]
            x0[1::3] = best_c[:, 1]
            x0[2::3] = best_r * 0.96
            
            # Micro-perturbation to escape flat plateaus
            pert = rng.uniform(-0.004, 0.004, best_c.shape)
            x0[:2*N] = np.clip(best_c + pert, 0.02, 0.98).flatten()
            
            try:
                res = minimize(obj_func, x0, method='SLSQP', bounds=bounds_vars,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
                if np.isfinite(res.fun):
                    c_new = res.x[:2*N].reshape(N, 2)
                    r_new = res.x[2*N:]
                    
                    valid = True
                    if np.any(c_new[:, 0] - r_new < -1e-8) or np.any(c_new[:, 0] + r_new > 1.0 + 1e-8): valid = False
                    if np.any(c_new[:, 1] - r_new < -1e-8) or np.any(c_new[:, 1] + r_new > 1.0 + 1e-8): valid = False
                    if valid:
                        dists = np.sqrt((c_new[:, None, 0] - c_new[None, :, 0])**2 + 
                                        (c_new[:, None, 1] - c_new[None, :, 1])**2)
                        if np.any(dists[PAIR_I, PAIR_J] < r_new[PAIR_I] + r_new[PAIR_J] - 1e-8):
                            valid = False
                            
                    if valid:
                        s_new = np.sum(r_new)
                        if s_new > best_sum:
                            best_sum = s_new
                            best_c = c_new.copy()
                            best_r = r_new.copy()
            except Exception:
                pass
                
    # Final LP extraction to guarantee maximal radii for converged centers
    if best_c is not None:
        r_lp, s_lp = get_lp_radii(best_c)
        if s_lp > best_sum:
            best_r = r_lp
            best_sum = s_lp
            
    # Strict safety scaling to guarantee 1e-12 validator tolerance
    if best_c is not None:
        scale = 1.0
        for i in range(N):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
                
        dists = np.sqrt((best_c[:, None, 0] - best_c[None, :, 0])**2 + 
                        (best_c[:, None, 1] - best_c[None, :, 1])**2)
        for i in range(N):
            for j in range(i+1, N):
                d = dists[i, j]
                rs = best_r[i] + best_r[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)
                    
        best_r *= scale * 0.9999994
        best_sum = float(np.sum(best_r))
    else:
        # Guaranteed fallback
        best_c = np.ones((N, 2)) * 0.5
        best_r = np.full(N, 0.01)
        best_sum = float(np.sum(best_r))
        
    return best_c, best_r, best_sum
