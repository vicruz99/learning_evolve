import numpy as np
from scipy.optimize import linprog, fmin_l_bfgs_b

def compute_distances(centers):
    """Computes the pairwise Euclidean distance matrix."""
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.sqrt(np.sum(diffs**2, axis=2))

def lp_solve(centers, dists, limits, A_ub, pair_i, pair_j):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns the scipy linprog result object.
    """
    bounds = [(0.0, max(l, 1e-9)) for l in limits]
    return linprog(
        -np.ones(len(limits)),
        A_ub=A_ub,
        b_ub=dists[pair_i, pair_j],
        bounds=bounds,
        method='highs'
    )

def obj_and_grad(center_flat, A_ub, pair_i, pair_j):
    """
    Objective and gradient for fmin_l_bfgs_b.
    Minimizes negative sum of radii. Returns (-sum_r, grad).
    """
    n = len(pair_i) + 1  # Approximate, but we can infer from array shape
    n = len(center_flat) // 2
    centers = center_flat.reshape(n, 2)
    
    dists = compute_distances(centers)
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    res = lp_solve(centers, dists, limits, A_ub, pair_i, pair_j)
    
    if not res.success:
        return 0.0, np.zeros(2 * n)
        
    sum_r = -res.fun
    grad_c = np.zeros((n, 2))
    
    # Pairwise duals: push centers apart along contact directions
    if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
        marginals = res.ineqlin.marginals
        for k, (i, j) in enumerate(zip(pair_i, pair_j)):
            lam = marginals[k]
            if lam > 1e-10:
                d = dists[i, j]
                if d > 1e-10:
                    diff = centers[i] - centers[j]
                    grad_c[i] += lam * diff / d
                    grad_c[j] -= lam * diff / d
                    
    # Boundary duals: push centers away from active walls
    if hasattr(res, 'marginals') and hasattr(res.marginals, 'upper'):
        for i in range(n):
            mu = res.marginals.upper[i]
            if mu > 1e-10:
                x, y = centers[i]
                lim = limits[i]
                if abs(lim - x) < 1e-6:
                    grad_c[i, 0] += mu
                elif abs(lim - (1.0 - x)) < 1e-6:
                    grad_c[i, 0] -= mu
                elif abs(lim - y) < 1e-6:
                    grad_c[i, 1] += mu
                elif abs(lim - (1.0 - y)) < 1e-6:
                    grad_c[i, 1] -= mu
                    
    # Return objective (to minimize) and negative gradient (since we maximize sum_r)
    return -sum_r, -grad_c.flatten()

        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
