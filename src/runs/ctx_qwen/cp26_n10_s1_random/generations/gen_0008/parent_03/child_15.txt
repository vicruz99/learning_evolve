import numpy as np
from scipy.optimize import linprog

def solve_lp_fast(n, pair_i, pair_j, A_ub, centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    b_ub = np.hypot(centers[pair_i, 0] - centers[pair_j, 0],
                     centers[pair_i, 1] - centers[pair_j, 1])
    wall_limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                             np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    wall_limits = np.maximum(wall_limits, 1e-10)
    bounds = [(0.0, lim) for lim in wall_limits]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun
    return None, 0.0

def make_hex(n, pat, rng):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = 0.09
    dy = 0.155
    dx = 0.175
    for i, cnt in enumerate(pat):
        shift = dx * 0.5 if i % 2 == 1 else 0.0
        x_start = (1.0 - (cnt - 1) * dx) / 2.0 + shift
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += dx
        y += dy
    res = np.array(pts[:n])
    res += rng.uniform(-0.015, 0.015, res.shape)
    return np.clip(res, 0.05, 0.95)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    pair_i, pair_j = np.triu_indices(n, k=1)
    num_pairs = len(pair_i)
    A_ub = np.zeros((num_pairs, n))
    A_ub[np.arange(num_pairs), pair_i] = 1.0
    A_ub[np.arange(num_pairs), pair_j] = 1.0
    
    rng = np.random.default_rng(123)
    configs = []
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], 
        [6,6,5,5,4], [5,4,6,6,5], [5,6,4,6,5], [6,5,5,6,4],
        [5,5,5,5,6], [4,5,6,5,6], [6,6,4,5,5]
    ]
    
    for pat in patterns:
        if sum(pat) >= n:
            configs.append(make_hex(n, pat, rng))
    for _ in range(12):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    max_iter = 5000
    init_step = 0.035
    
    # Phase 1: Simulated Annealing on centers
    for cfg in configs:
        curr_c = cfg.copy()
        _, curr_sum = solve_lp_fast(n, pair_i, pair_j, A_ub, curr_c)
        if curr_sum == 0.0: continue
        
        best_c_local = curr_c.copy()
        best_sum_local = curr_sum
        
        step = init_step
        T = 0.02
        for it in range(max_iter):
            step *= 0.9993
            T = max(T * 0.999, 1e-5)
            k = 1 if rng.random() < 0.85 else 2
            idx = rng.choice(n, k, replace=False)
            delta = rng.uniform(-step, step, (k, 2))
            
            new_c = curr_c.copy()
            new_c[idx] += delta
            new_c = np.clip(new_c, 0.005, 0.995)
            
            _, new_sum = solve_lp_fast(n, pair_i, pair_j, A_ub, new_c)
            if new_sum == 0.0: continue
            
            diff = new_sum - curr_sum
            if diff > 0 or rng.random() < np.exp(np.clip(diff / T, -100, 100)):
                curr_c = new_c
                curr_sum = new_sum
                if curr_sum > best_sum_local:
                    best_sum_local = curr_sum
                    best_c_local = curr_c.copy()
                    
        r_loc, s_loc = solve_lp_fast(n, pair_i, pair_j, A_ub, best_c_local)
        if s_loc > best_sum:
            best_sum = s_loc
            best_centers = best_c_local.copy()
            best_radii = r_loc.copy()
            
    # Phase 2: Coordinate Descent Polish
    if best_centers is not None:
        for _ in range(5):
            improved = False
            for i in range(n):
                best_i_sum = best_sum
                best_i_c = best_centers.copy()
                for dx, dy in [(0.004, 0), (-0.004, 0), (0, 0.004), (0, -0.004),
                               (0.003, 0.003), (-0.003, 0.003), (0.003, -0.003), (-0.003, -0.003),
                               (0.001, 0.002), (-0.001, -0.002)]:
                    test_c = best_centers.copy()
                    test_c[i, 0] = np.clip(test_c[i, 0] + dx, 0.001, 0.999)
                    test_c[i, 1] = np.clip(test_c[i, 1] + dy, 0.001, 0.999)
                    _, s = solve_lp_fast(n, pair_i, pair_j, A_ub, test_c)
                    if s > best_i_sum + 1e-7:
                        best_i_sum = s
                        best_i_c = test_c.copy()
                best_centers = best_i_c
                if best_i_sum > best_sum:
                    best_sum = best_i_sum
                    _, best_radii = solve_lp_fast(n, pair_i, pair_j, A_ub, best_centers)
                    improved = True
            if not improved:
                break
                
    # Phase 3: Strict Safety Scaling
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
                
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0],
                             best_centers[i, 1] - best_centers[j, 1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)
                    
        best_radii *= scale * 0.9999995
        best_sum = float(np.sum(best_radii))
    else:
        best_centers = np.ones((n, 2)) * 0.5
        best_radii = np.full(n, 0.01)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
