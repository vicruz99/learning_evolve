import numpy as np
from scipy.optimize import linprog, minimize

def compute_distances(centers):
    """Computes the pairwise Euclidean distance matrix."""
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.sqrt(np.sum(diffs**2, axis=2))

def solve_radii_lp_fast(centers, n, i_idx, j_idx, A_ub):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns (radii, sum_radii).
    """
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-12)) for lim in limits]
    
    diffs = centers[i_idx] - centers[j_idx]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun
    return np.full(n, 1e-6), 0.0

def obj_func(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def constr_func(x, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.column_stack((cx, cy))
    dists = compute_distances(c)
    
    cons = []
    cons.append(cx - r)
    cons.append(1.0 - cx - r)
    cons.append(cy - r)
    cons.append(1.0 - cy - r)
    
    r_sum = r[:, None] + r[None, :]
    cons.append(dists[triu_i, triu_j] - r_sum[triu_i, triu_j])
    return np.concatenate(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    triu_i, triu_j = np.triu_indices(n, k=1)
    m = len(triu_i)
    
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), triu_i] = 1.0
    A_ub[np.arange(m), triu_j] = 1.0
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)
    configs = []
    
    # Generate diverse hexagonal patterns
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], 
        [6,6,5,5,4], [5,4,6,6,5], [5,6,4,6,5], [6,5,5,6,4],
        [7,5,5,5,4], [5,7,5,5,4], [4,5,7,5,5], [5,5,7,5,4]
    ]
    
    for pat in patterns:
        if sum(pat) < n: continue
        r0 = 0.10
        pts = []
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) < n: pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        configs.append(np.array(pts[:n]))
        
    # Grid patterns
    for s in [5, 6]:
        if s*s >= n:
            sp = 1.0 / (s + 1)
            pts = []
            for i in range(s):
                for j in range(s):
                    if len(pts) < n:
                        pts.append([sp * (i + 1), sp * (j + 1)])
            configs.append(np.array(pts[:n]))
            
    # Random starts
    for _ in range(5):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    def lp_eval(c):
        return solve_radii_lp_fast(c, n, triu_i, triu_j, A_ub)
        
    # Phase 1: Targeted Hill Climbing on Centers
    for cfg in configs:
        c = cfg + rng.uniform(-0.01, 0.01, cfg.shape)
        c = np.clip(c, 0.05, 0.95)
        
        r_curr, s_curr = lp_eval(c)
        if r_curr is None: continue
        
        best_local_c = c.copy()
        best_local_r = r_curr.copy()
        best_local_s = s_curr
        
        # Simulated annealing schedule
        temp = 0.04
        for step in range(1200):
            i = rng.integers(n)
            step_size = 0.015 * (0.996 ** (step // 5))
            
            # Compute distances from circle i to all others
            dx = best_local_c[i, 0] - best_local_c[:, 0]
            dy = best_local_c[i, 1] - best_local_c[:, 1]
            dx[i] = dy[i] = 1.0 
            d = np.sqrt(dx**2 + dy**2)
            
            # Direction away from nearest neighbor
            closest = np.argmin(d)
            dir_rep = (best_local_c[i] - best_local_c[closest]) / d[closest]
            
            # Try repulsion move
            trial_c = best_local_c.copy()
            trial_c[i] = np.clip(best_local_c[i] + dir_rep * step_size, 0.02, 0.98)
            r_try, s_try = lp_eval(trial_c)
            
            delta = s_try - best_local_s if r_try is not None else -np.inf
            if delta > 0 or (r_try is not None and rng.random() < np.exp(delta / max(temp, 1e-9))):
                best_local_c = trial_c
                best_local_r = r_try
                best_local_s = s_try
                temp *= 0.998
            else:
                # Fallback: random perturbation
                trial_c2 = best_local_c.copy()
                trial_c2[i] += rng.uniform(-step_size, step_size, 2)
                trial_c2[i] = np.clip(trial_c2[i], 0.02, 0.98)
                r_try2, s_try2 = lp_eval(trial_c2)
                if r_try2 is not None and s_try2 > best_local_s:
                    best_local_c = trial_c2
                    best_local_r = r_try2
                    best_local_s = s_try2
                    
        if best_local_s > best_sum:
            best_sum = best_local_s
            best_centers = best_local_c.copy()
            best_radii = best_local_r.copy()
            
    # Phase 2: SLSQP Joint Optimization Polish
    if best_centers is not None:
        bounds_opt = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
        cons_slqp = {'type': 'ineq', 'fun': constr_func, 'args': (n, triu_i, triu_j)}
        
        # Run from best found, and a slightly perturbed version
        starts_slqp = [best_centers.copy()]
        starts_slqp.append(np.clip(best_centers + rng.uniform(-0.005, 0.005, best_centers.shape), 0.02, 0.98))
        
        for cfg_slqp in starts_slqp:
            x0 = np.zeros(3*n)
            x0[0::3] = cfg_slqp[:, 0]
            x0[1::3] = cfg_slqp[:, 1]
            x0[2::3] = best_radii * 0.985
            
            try:
                res = minimize(obj_func, x0, args=(n,), method='SLSQP', bounds=bounds_opt,
                               constraints=cons_slqp,
                               options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
                if np.isfinite(res.fun):
                    c_slqp = res.x[:2*n].reshape(n, 2)
                    r_slqp, s_slqp = lp_eval(c_slqp)
                    if r_slqp is not None and s_slqp > best_sum:
                        best_sum = s_slqp
                        best_centers = c_slqp.copy()
                        best_radii = r_slqp.copy()
            except Exception:
                pass
                
        # Final LP squeeze on the polished centers
        r_lp, s_lp = lp_eval(best_centers)
        if r_lp is not None and s_lp > best_sum:
            best_sum = s_lp
            best_radii = r_lp.copy()
            
    # Phase 3: Strict Safety Scaling for Validation Tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    c_diffs = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]
    c_dists = np.sqrt(np.sum(c_diffs**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            d = c_dists[i, j]
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
