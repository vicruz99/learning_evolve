import numpy as np
from scipy.optimize import linprog, minimize

def compute_dists(centers):
    """Computes the pairwise Euclidean distance matrix."""
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.sqrt(np.sum(diff**2, axis=2))

def get_lp_results(centers, n, p_i, p_j, dists, ub):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns optimal radii, sum of radii, and analytical gradient w.r.t centers.
    """
    m = len(p_i)
    A = np.zeros((m, n))
    A[np.arange(m), p_i] = 1.0
    A[np.arange(m), p_j] = 1.0
    b = dists[p_i, p_j]
    
    # Solve LP: max sum(r_i) s.t. r_i + r_j <= d_ij, 0 <= r_i <= ub_i
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success or not np.isfinite(res.fun):
        return None, 0.0, None
        
    r = res.x
    s = -res.fun
    
    grad = np.zeros((n, 2))
    # Extract dual variables (marginals) for gradient computation
    if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals') and res.ineqlin.marginals is not None:
        mu = res.ineqlin.marginals
        active = mu > 1e-8
        if np.any(active):
            ia, ja, ma = p_i[active], p_j[active], mu[active]
            da = dists[ia, ja]
            inv_da = np.where(da > 1e-9, 1.0/da, 0.0)
            dirs = (centers[ia] - centers[ja]) * inv_da[:, np.newaxis]
            grad[ia] += ma[:, np.newaxis] * dirs
            grad[ja] -= ma[:, np.newaxis] * dirs
    return r, s, grad

def obj_func(x, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def con_func(x, n, p_i, p_j):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    cx = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    d = compute_dists(cx)
    vals = np.concatenate([cx[:,0]-r, 1-cx[:,0]-r, cx[:,1]-r, 1-cx[:,1]-r, 
                           d[p_i, p_j] - (r[p_i] + r[p_j])])
    return vals
