import numpy as np
from scipy.optimize import linprog

def compute_pairwise_dists(centers):
    """Computes the pairwise Euclidean distance matrix."""
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.sqrt(np.sum(diffs**2, axis=2))

def solve_lp(centers, n, pair_i, pair_j, dists):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns optimal radii and sum of radii.
    """
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-12)) for lim in limits]
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[pair_i, pair_j]
    for k, (i, j) in enumerate(zip(pair_i, pair_j)):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success or not np.isfinite(res.fun):
        return None, None
        
    return res.x, -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    pair_i, pair_j = np.triu_indices(n, k=1)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    configs = []
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], 
        [6,6,5,5,4], [5,4,6,6,5], [5,6,4,6,5], [6,5,5,6,4],
        [7,5,5,5,4], [5,7,5,5,4], [5,5,7,5,4], [5,5,5,7,4]
    ]
    for pat in patterns:
        if sum(pat) < n: continue
        r0 = 0.095
        pts = []
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) < n: pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        configs.append(np.array(pts[:n]))
        
    np.random.seed(42)
    for _ in range(20):
        configs.append(np.random.uniform(0.15, 0.85, (n, 2)))
        
    # Local optimization for each configuration
    for cfg in configs:
        dists = compute_pairwise_dists(cfg)
        r, s = solve_lp(cfg, n, pair_i, pair_j, dists)
        if r is None: continue
        
        curr_c = cfg.copy()
        curr_s = s
        
        # Phase 1: Coordinate-wise local search
        step = 0.03
        for _ in range(800):
            improved = False
            for i in range(n):
                for _ in range(5):
                    old_c = curr_c[i].copy()
                    curr_c[i] += np.random.uniform(-step, step, 2)
                    curr_c[i] = np.clip(curr_c[i], 0.02, 0.98)
                    
                    dists_new = compute_pairwise_dists(curr_c)
                    r_new, s_new = solve_lp(curr_c, n, pair_i, pair_j, dists_new)
                    if r_new is not None and s_new > curr_s + 1e-7:
                        curr_s = s_new
                        improved = True
                        break
                    curr_c[i] = old_c
                if improved:
                    break
                    
            if improved:
                step = min(step * 1.02, 0.08)
            else:
                step *= 0.95
                if step < 1e-5:
                    break
                    
        # Phase 2: Random perturbation hill climbing
        step = 0.02
        for _ in range(500):
            best_improvement = 0.0
            best_neighbor = None
            for _ in range(8):
                neighbor = curr_c + np.random.uniform(-step, step, (n, 2))
                neighbor = np.clip(neighbor, 0.02, 0.98)
                dists_n = compute_pairwise_dists(neighbor)
                r_n, s_n = solve_lp(neighbor, n, pair_i, pair_j, dists_n)
                if r_n is not None and s_n > curr_s + 1e-7:
                    imp = s_n - curr_s
                    if imp > best_improvement:
                        best_improvement = imp
                        best_neighbor = (neighbor, r_n, s_n)
            if best_neighbor:
                curr_c, _, curr_s = best_neighbor
                step = min(step * 1.05, 0.06)
            else:
                step *= 0.9
                if step < 1e-5:
                    break
                    
        if curr_s > best_sum:
            best_sum = curr_s
            best_centers = curr_c.copy()
            dists_f = compute_pairwise_dists(best_centers)
            best_radii, _ = solve_lp(best_centers, n, pair_i, pair_j, dists_f)
            
    # Fallback
    if best_centers is None:
        best_centers = np.ones((n, 2)) * 0.5
        best_radii = np.full(n, 0.01)
        best_sum = float(np.sum(best_radii))
    else:
        # Phase 3: Strict Safety Scaling
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
        dists_f = compute_pairwise_dists(best_centers)
        for i in range(n):
            for j in range(i+1, n):
                d = dists_f[i, j]
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)
                    
        best_radii *= scale * 0.9999999
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
