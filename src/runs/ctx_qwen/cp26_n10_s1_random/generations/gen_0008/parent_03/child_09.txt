import numpy as np
from scipy.optimize import linprog, minimize

def compute_pairwise_distances(centers):
    """Efficiently computes pairwise Euclidean distances."""
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.sqrt(np.sum(diffs**2, axis=2))

def solve_radii_lp(centers, n, A_ub_pairs, i_idx, j_idx):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns radii, sum_radii, and dual marginals for pairwise constraints.
    """
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-9)) for lim in limits]
    
    dists = compute_pairwise_distances(centers)
    b_ub = dists[i_idx, j_idx]
    
    res = linprog(-np.ones(n), A_ub=A_ub_pairs, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        marginals = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else np.zeros(len(i_idx))
        return res.x, -res.fun, marginals
    return None, 0.0, None

def apply_gradient_step(centers, marginals, i_idx, j_idx, step_size, n):
    """Moves centers along dual gradient directions to increase sum of radii."""
    dists = compute_pairwise_distances(centers)
    grad = np.zeros_like(centers)
    
    active_mask = marginals > 1e-7
    if np.any(active_mask):
        for k in np.where(active_mask)[0]:
            i, j = i_idx[k], j_idx[k]
            lam = marginals[k]
            d = dists[i, j]
            if d > 1e-8:
                diff = centers[i] - centers[j]
                grad[i] += lam * diff / d
                grad[j] -= lam * diff / d
                
    centers_new = centers + step_size * grad
    return np.clip(centers_new, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    i_idx, j_idx = np.triu_indices(n, k=1)
    
    A_ub_pairs = np.zeros((len(i_idx), n))

    A_ub_pairs[np.arange(len(j_idx)), j_idx] = 1.0

    best_centers = None
    best_radii = None

            x = r0 + shift
            for _ in range(cnt):
                if len(pts) < n:
                    pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        inits.append(np.array(pts[:n]))
        
    # Add random dense starts
    for _ in range(6):
        inits.append(np.random.uniform(0.12, 0.88, (n, 2)))

        curr_c = np.clip(curr_c, 0.05, 0.95)
        
        r, s, marg = solve_radii_lp(curr_c, n, A_ub_pairs, i_idx, j_idx)
        if r is None: continue

        curr_s = s

                    best_radii = r_lp

                rs = best_radii[i] + best_radii[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)

        best_sum = float(np.sum(best_radii))
    else:
        best_centers = np.ones((n, 2)) * 0.5
        best_radii = np.full(n, 0.01)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
