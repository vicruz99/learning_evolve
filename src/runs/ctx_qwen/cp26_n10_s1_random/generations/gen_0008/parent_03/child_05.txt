import numpy as np
from scipy.optimize import linprog, minimize

def compute_dists(centers):
    """Computes the pairwise Euclidean distance matrix."""
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.sqrt(np.sum(diffs**2, axis=2))

def solve_radii_lp(centers, n, pair_i, pair_j, dists):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns optimal radii and their sum.
    """
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-9)) for lim in limits]
    
    A_ub = np.zeros((len(pair_i), n))
    b_ub = dists[pair_i, pair_j]
    A_ub[np.arange(len(pair_i)), pair_i] = 1.0
    A_ub[np.arange(len(pair_i)), pair_j] = 1.0
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-7), 0.0

def hill_climb_optimize(centers, n, pair_i, pair_j, rng):

    Stochastic hill-climbing that optimizes centers by evaluating max radii via LP at each step.
    """

        inits.append(np.array(pts[:n]))
        
    for _ in range(18):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return inits
