import numpy as np
from scipy.optimize import linprog

def solve_lp_and_gradient(centers, n, pair_i, pair_j):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns (radii, sum_radii, gradient_matrix) or (None, None, None) on failure.
    """
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    total_constraints = m_bound + m_pair
    
    A_ub = np.zeros((total_constraints, n))
    b_ub = np.zeros(total_constraints)
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        A_ub[4*i, i] = 1.0; b_ub[4*i] = x
        A_ub[4*i+1, i] = 1.0; b_ub[4*i+1] = 1.0 - x
        A_ub[4*i+2, i] = 1.0; b_ub[4*i+2] = y
        A_ub[4*i+3, i] = 1.0; b_ub[4*i+3] = 1.0 - y
        
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    k = m_bound
    for i, j in zip(pair_i, pair_j):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b_ub[k] = np.hypot(dx, dy)
        k += 1
        
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None
            
        radii = res.x
        sum_r = -res.fun
        grad = np.zeros((n, 2))
        
        # Extract gradient using dual variables (marginals)
        # Sensitivity of optimal value w.r.t RHS b is lambda.
        # We maximize S = sum(r), so dS/db = lambda (since linprog minimizes -S)
        if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            lam = res.ineqlin.marginals
            
            # Boundary gradients
            for i in range(n):
                grad[i, 0] += lam[4*i] - lam[4*i+1]
                grad[i, 1] += lam[4*i+2] - lam[4*i+3]
                
            # Pairwise gradients
            k = m_bound
            for i, j in zip(pair_i, pair_j):
                l = lam[k]
                if l > 1e-10:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = b_ub[k]
                    if d > 1e-12:
                        fx = l * dx / d
                        fy = l * dy / d
                        grad[i, 0] += fx
                        grad[i, 1] += fy
                        grad[j, 0] -= fx
                        grad[j, 1] -= fy
                k += 1
                
        return radii, sum_r, grad
    except Exception:
        return None, None, None

def optimize_trajectory(start_centers, n, pair_i, pair_j, rng):
    """Runs gradient ascent + local random search from a starting configuration."""
    centers = start_centers.copy()
    radii, curr_s, grad = solve_lp_and_gradient(centers, n, pair_i, pair_j)
    if radii is None:
        return None, None, None, -1.0
        
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_s = curr_s
    
    # Phase 1: Gradient Ascent
    step = 0.015
    for _ in range(250):
        if grad is None: break
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-8:
            break
            
        g_dir = grad / g_norm * step
        c_new = np.clip(centers + g_dir, 0.02, 0.98)
        r_new, s_new, g_new = solve_lp_and_gradient(c_new, n, pair_i, pair_j)
        
        if r_new is not None and s_new > best_s + 1e-9:
            centers, radii, curr_s, grad = c_new, r_new, s_new, g_new
            if curr_s > best_s:
                best_centers, best_radii, best_s = centers.copy(), radii.copy(), curr_s
            step = min(step * 1.04, 0.045)
        else:
            step *= 0.90
            
    # Phase 2: Local Random Search (Hill Climbing)
    shake_step = 0.025
    stagnation = 0
    for _ in range(400):
        i = rng.integers(n)
        old_c = centers[i].copy()
        centers[i] += rng.uniform(-shake_step, shake_step, 2)
        centers[i] = np.clip(centers[i], 0.02, 0.98)
        
        r_new, s_new, g_new = solve_lp_and_gradient(centers, n, pair_i, pair_j)
        if r_new is not None and s_new > best_s + 1e-9:
            centers, radii, curr_s, grad = centers.copy(), r_new, s_new, g_new
            best_centers, best_radii, best_s = centers.copy(), radii.copy(), curr_s
            stagnation = 0
        else:
            centers[i] = old_c
            stagnation += 1
            
        if stagnation > 50:
            # Escape local minimum: shake all circles slightly
            centers += rng.uniform(-0.015, 0.015, centers.shape)
            centers = np.clip(centers, 0.05, 0.95)
            r_new, s_new, g_new = solve_lp_and_gradient(centers, n, pair_i, pair_j)
            if r_new is not None:
                radii, curr_s, grad = r_new, s_new, g_new
            stagnation = 0
            shake_step *= 0.95
            if shake_step < 0.005: break
            
    return best_centers, best_radii, best_s, best_s

def generate_hex_init(row_pattern, r0, rng):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    for ri, cnt in enumerate(row_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= 26: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < 26:
        pts.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])
    return np.array(pts[:26])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    pair_i, pair_j = np.triu_indices(n, k=1)
    rng = np.random.default_rng(42)
    
    best_sum = -1.0

    best_radii = None
    
    # Diverse row distributions summing to 26

    ]

            p = cfg + rng.uniform(-0.02, 0.02, cfg.shape)
            starts.append(np.clip(p, 0.06, 0.94))
            
    # Grid and random starts
    gx = np.linspace(0.12, 0.88, 5)
    gy = np.linspace(0.12, 0.88, 5)
    grid = np.array([(x, y) for y in gy for x in gx])
    grid = np.vstack([grid, [0.5, 0.5]])
    starts.append(grid)
    for _ in range(5):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Evaluate each trajectory
    for cfg in starts:
        c_opt, r_opt, s_opt, _ = optimize_trajectory(cfg, n, pair_i, pair_j, rng)
        if c_opt is not None and s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            best_radii = r_opt
            
    # Fallback if all fail (unlikely)
    if best_centers is None:
        best_centers = starts[0]
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    # Final LP squeeze on best centers to guarantee maximal radii
    r_final, s_final, _ = solve_lp_and_gradient(best_centers, n, pair_i, pair_j)
    if r_final is not None and s_final > best_sum:
        best_radii = r_final
        best_sum = s_final
        
    # Strict safety scaling to satisfy 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i, j in zip(pair_i, pair_j):
        d = np.hypot(best_centers[i, 0]-best_centers[j, 0], best_centers[i, 1]-best_centers[j, 1])
        rs = best_radii[i] + best_radii[j]
        if rs > 1e-12:
            scale = min(scale, d / rs)
            
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
