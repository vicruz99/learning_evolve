def compute_obj(vars, n):
    """Objective: minimize negative sum of radii => maximize sum of radii"""
    return -np.sum(vars[2*n:])

def compute_cons(vars, n, i_idx, j_idx):
    """Inequality constraints >= 0 for valid packing"""
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]

    return c

        A[k, i] = 1.0
        A[k, j] = 1.0

        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun

        pass
    return np.full(n, 1e-6), 0.0
