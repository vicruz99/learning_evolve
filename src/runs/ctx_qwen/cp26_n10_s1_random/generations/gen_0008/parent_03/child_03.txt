import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers, n, pair_i, pair_j, A_ub):
    """Solves the LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    b_ub = np.hypot(centers[pair_i, 0] - centers[pair_j, 0],
                    centers[pair_i, 1] - centers[pair_j, 1])
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-6), 0.0

def coordinate_ascent(centers, n, pair_i, pair_j, A_ub, rng):
    """Optimizes centers via coordinate ascent guided by LP radius maximization."""
    best_c = centers.copy()
    best_r, best_s = solve_lp(best_c, n, pair_i, pair_j, A_ub)
    
    step = 0.025
    for epoch in range(80):
        improved = False
        for idx in range(n):
            old_pos = best_c[idx].copy()

                new_c = best_c.copy()
                new_c[idx] = np.clip(old_pos + trial, 1e-5, 1.0 - 1e-5)
                _, s_new = solve_lp(new_c, n, pair_i, pair_j, A_ub)
                if s_new > best_s + 1e-8:
                    best_c = new_c
                    best_r, best_s = solve_lp(best_c, n, pair_i, pair_j, A_ub)
                    improved = True
                    break
        if not improved:
            step *= 0.95
            if step < 1e-5:
                break
        else:
            step = min(step * 1.05, 0.04)
    return best_c, best_r, best_s

def make_hex_layout(rows, n):
    """Generates a hexagonal lattice initialization with specified row counts."""
    pts = []
    r = 0.10
    y = r
    for ri, cnt in enumerate(rows):
        shift = r if ri % 2 == 1 else 0.0
        x = r + shift
        for _ in range(cnt):
            if len(pts) < n:
                pts.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3) * r
    return np.array(pts[:n])

def objective_func(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def constraint_func(v, n, pair_i, pair_j):
    """Inequality constraints for SLSQP: >= 0 means valid."""
    cx, cy, rr = v[:n], v[n:2*n], v[2*n:]
    i, j = pair_i, pair_j
    d = np.hypot(cx[i]-cx[j], cy[i]-cy[j])
    return np.concatenate([
        cx - rr, 1.0 - cx - rr, cy - rr, 1.0 - cy - rr,
        d - (rr[i] + rr[j])
    ])
