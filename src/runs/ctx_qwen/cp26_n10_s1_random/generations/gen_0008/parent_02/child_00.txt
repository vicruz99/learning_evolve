import numpy as np
from scipy.optimize import minimize, linprog

def compute_slsqp_constraints(vars_arr, n, triu_i, triu_j):
    """Computes inequality constraints >= 0 for pairwise non-overlap."""
    r = vars_arr[:n]
    u = vars_arr[n:2*n]
    v = vars_arr[2*n:3*n]
    # Parameterization automatically satisfies boundary constraints
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return d2[triu_i, triu_j] - rs[triu_i, triu_j]**2

def slsqp_objective(vars_arr, n):
    """Objective: maximize sum of radii -> minimize negative sum"""
    return -np.sum(vars_arr[:n])

def solve_lp_radii(centers, n, triu_i, triu_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds_lp = [(0.0, max(l, 1e-9)) for l in limits]
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = len(triu_i)
    A = np.zeros((m, n))
    A[np.arange(m), triu_i] = 1.0
    A[np.arange(m), triu_j] = 1.0
    b = dists[triu_i, triu_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds_lp, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def centers_to_params(centers):
    """Converts centers to parameterization variables."""
    n = centers.shape[0]
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    r = np.maximum(r * 0.95, 1e-5)
    denom = 1.0 - 2.0 * r
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def params_to_centers(params):
    """Converts parameterization variables back to centers."""
    n = len(params) // 3
    r = params[:n]
    u = params[n:2*n]
    v = params[2*n:3*n]
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    return np.column_stack((x, y))

def run_packing():
    n = 26
    triu_i, triu_j = np.triu_indices(n, k=1)
    bounds_v = [(1e-6, 0.5)] * n + [(0.0, 1.0)] * (2 * n)
    cons = {'type': 'ineq', 'fun': compute_slsqp_constraints, 'args': (n, triu_i, triu_j)}
    obj_fun = slsqp_objective
    
    rng = np.random.default_rng(1234)
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Generate diverse initial configurations
    configs = []
    # Perturbed hexagonal lattices
    for _ in range(15):
        pts = []
        r0 = 0.09 + rng.uniform(-0.01, 0.01)
        dy = np.sqrt(3) * r0
        y = r0
        row = 0
        while len(pts) < n:
            shift = r0 if row % 2 == 1 else 0.0
            x = r0 + shift
            while x <= 1.0 - r0 and len(pts) < n:
                pts.append([x + rng.uniform(-0.005, 0.005), y + rng.uniform(-0.005, 0.005)])
                x += 2 * r0 + rng.uniform(-0.005, 0.005)
            y += dy
            row += 1
        while len(pts) < n:
            pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
        configs.append(np.array(pts[:n]))

    # Perturbed grids
    for _ in range(8):
        pts = []
        for i in range(6):
            for j in range(5):
                if len(pts) < n:
                    pts.append([0.08 + i*0.15 + rng.uniform(-0.01, 0.01),
                                0.08 + j*0.16 + rng.uniform(-0.01, 0.01)])
        configs.append(np.array(pts[:n]))

    # Phase 1: SLSQP joint optimization from diverse starts
    for cfg in configs:
        x0 = centers_to_params(cfg)
        res = minimize(obj_fun, x0, args=(n,), method='SLSQP', bounds=bounds_v,
                       constraints=cons, options={'maxiter': 6000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            cvals = compute_slsqp_constraints(res.x, n, triu_i, triu_j)
            if np.min(cvals) > -1e-7:
                centers_opt = params_to_centers(res.x)
                r_lp, s_lp = solve_lp_radii(centers_opt, n, triu_i, triu_j)
                if r_lp is not None and s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = centers_opt.copy()
                    best_radii = r_lp.copy()

    # Phase 2: Local search on centers with LP evaluation + SLSQP restart
    if best_centers is not None:
        cur_c = best_centers.copy()
        step = 0.015
        for _ in range(800):
            i = rng.integers(n)
            old = cur_c[i].copy()
            cur_c[i] += rng.uniform(-step, step, 2)
            cur_c[i] = np.clip(cur_c[i], 1e-4, 0.999)
            
            r_new, s_new = solve_lp_radii(cur_c, n, triu_i, triu_j)
            if r_new is not None and s_new > best_sum + 1e-8:
                best_sum = s_new
                best_centers = cur_c.copy()
                best_radii = r_new.copy()
                
                # Restart SLSQP from improved configuration to polish jointly
                x0 = centers_to_params(cur_c)
                res = minimize(obj_fun, x0, args=(n,), method='SLSQP', bounds=bounds_v,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
                if np.isfinite(res.fun):
                    if np.min(compute_slsqp_constraints(res.x, n, triu_i, triu_j)) > -1e-7:
                        c_opt = params_to_centers(res.x)
                        r_lp, s_lp = solve_lp_radii(c_opt, n, triu_i, triu_j)
                        if r_lp is not None and s_lp > best_sum:
                            best_sum = s_lp
                            best_centers = c_opt.copy()
                            best_radii = r_lp.copy()
            else:
                cur_c[i] = old
            step *= 0.997

    # Phase 3: Final strict safety scaling to guarantee validation tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    dx = best_centers[triu_i, 0] - best_centers[triu_j, 0]
    dy = best_centers[triu_i, 1] - best_centers[triu_j, 1]
    d = np.sqrt(dx**2 + dy**2)
    rs = best_radii[triu_i] + best_radii[triu_j]
    if np.any(rs > 1e-12):
        scale = min(scale, np.min(d / np.maximum(rs, 1e-12)))
        
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
