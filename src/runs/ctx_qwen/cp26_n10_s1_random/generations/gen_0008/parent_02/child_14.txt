2. **LP Refinement**: For any fixed set of centers, finding the optimal radii is a linear programming problem. Solving this LP extracts the maximum possible sum of radii for a given layout, often improving upon the SLSQP result by exploiting local slack.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIUI, TRIUJ = np.triu_indices(N, k=1)
M_PAIRS = len(TRIUI)

# Precompute constant LP constraint matrix structure
A_ub_pairs = np.zeros((M_PAIRS, N))
A_ub_pairs[np.arange(M_PAIRS), TRIUI] = 1.0
A_ub_pairs[np.arange(M_PAIRS), TRIUJ] = 1.0

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[TRIUI, TRIUJ]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_pairs, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-6), 1e-6 * N

def get_constraints(vars_arr):
    """Computes inequality constraints >= 0 for valid packing."""
    xs = vars_arr[:N]
    ys = vars_arr[N:2*N]
    rs = vars_arr[2*N:]
    
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[TRIUI] - xs[TRIUJ]
    dy = ys[TRIUI] - ys[TRIUJ]
    dr = rs[TRIUI] + rs[TRIUJ]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def objective_func(vars_arr):
    """Objective: maximize sum of radii -> minimize negative sum"""
    return -np.sum(vars_arr[2*N:])

def generate_hex_init(rows, rng):
    """Generates a hexagonal lattice initialization for given row counts."""
    pts = []
    y = 0.08
    dy = 0.14
    for ri, cnt in enumerate(rows):
        shift = 0.07 if ri % 2 == 1 else 0.0
        x_start = 0.08 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x_start, y])
            x_start += 0.14
        y += dy
    res = np.array(pts[:N])
    res += rng.uniform(-0.01, 0.01, res.shape)
    return np.clip(res, 0.02, 0.98)

def run_packing():
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_vars = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    # Diverse row distributions summing to 26
    row_dists = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 5, 5, 6], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4],
        [5, 5, 6, 5, 5], [6, 5, 5, 6, 4], [7, 5, 5, 5, 4],
        [5, 6, 4, 6, 5]
    ]
    
    configs = []
    for rd in row_dists:
        configs.append(generate_hex_init(rd, rng))
        
    for _ in range(10):
        configs.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    for cfg in configs:
        # Initial LP to get strictly feasible radii
        r_init, _ = solve_lp_radii(cfg)
        
        v0 = np.zeros(3 * N)
        v0[:N] = cfg[:, 0]
        v0[N:2*N] = cfg[:, 1]
        v0[2*N:] = r_init * 0.95  # Slight shrink guarantees initial feasibility
        
        # Phase 1: SLSQP Joint Optimization
        try:
            res = minimize(objective_func, v0, method='SLSQP', bounds=bounds_vars,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                centers_tmp = res.x[:2*N].reshape(N, 2)
                r_lp, s_lp = solve_lp_radii(centers_tmp)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = centers_tmp.copy()
                    best_radii = r_lp.copy()
        except Exception:
            continue

        if best_centers is None:
            continue
            
        # Phase 2: Hill-climb on centers using LP objective
        curr_c = best_centers.copy()
        curr_r, curr_s = solve_lp_radii(curr_c)
        
        step = 0.008
        for _ in range(80):
            improved = False
            for _ in range(15):
                trial_c = curr_c.copy()
                idx = rng.integers(N)
                trial_c[idx] += rng.uniform(-step, step, 2)
                trial_c = np.clip(trial_c, 0.005, 0.995)
                r_t, s_t = solve_lp_radii(trial_c)
                if s_t > curr_s + 1e-7:
                    curr_c = trial_c
                    curr_r = r_t
                    curr_s = s_t
                    improved = True
            if not improved:
                step *= 0.9
            if step < 1e-7:
                break
                
        if curr_s > best_sum:
            best_sum = curr_s
            best_centers = curr_c.copy()
            best_radii = curr_r.copy()
            
        # Phase 3: Local SLSQP restart from hill-climb result to polish
        v0_re = np.zeros(3 * N)
        v0_re[:N] = best_centers[:, 0]
        v0_re[N:2*N] = best_centers[:, 1]
        v0_re[2*N:] = best_radii * 0.95
        
        try:
            res_re = minimize(objective_func, v0_re, method='SLSQP', bounds=bounds_vars,
                              constraints=cons, options={'maxiter': 4000, 'ftol': 1e-14})
            if np.isfinite(res_re.fun):
                centers_re = res_re.x[:2*N].reshape(N, 2)
                r_re, s_re = solve_lp_radii(centers_re)
                if s_re > best_sum:
                    best_sum = s_re
                    best_centers = centers_re.copy()
                    best_radii = r_re.copy()
        except Exception:
            continue

    # Fallback configuration
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    dx = best_centers[TRIUI, 0] - best_centers[TRIUJ, 0]
    dy = best_centers[TRIUI, 1] - best_centers[TRIUJ, 1]
    d = np.sqrt(dx**2 + dy**2)
    rs = best_radii[TRIUI] + best_radii[TRIUJ]
    if np.any(rs > 1e-12):
        scale = min(scale, np.min(d / np.maximum(rs, 1e-12)))
        
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
