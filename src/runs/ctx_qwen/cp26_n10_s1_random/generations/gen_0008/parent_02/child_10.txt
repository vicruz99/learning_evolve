import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers, n, A_ub, i_idx, j_idx):
    """Solves LP to maximize sum of radii for fixed centers."""
    bounds = []
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds.append((0.0, max(mx - 1e-9, 1e-9)))
        
    dists = np.sqrt(np.sum((centers[i_idx] - centers[j_idx])**2, axis=1))
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def hill_climb(centers_init, n, A_ub, i_idx, j_idx, rng):
    """Hill climbing on centers optimizing the LP sum of radii."""
    centers = centers_init.copy()
    cur_r, cur_sum = solve_lp(centers, n, A_ub, i_idx, j_idx)
    
    step = 0.02
    for _ in range(3000):
        i = rng.integers(n)
        old_c = centers[i].copy()
        centers[i] += rng.uniform(-step, step, 2)
        centers[i] = np.clip(centers[i], 1e-4, 1 - 1e-4)
        
        new_r, new_sum = solve_lp(centers, n, A_ub, i_idx, j_idx)
        if new_sum > cur_sum + 1e-8:
            cur_sum = new_sum
            cur_r = new_r
        else:
            centers[i] = old_c
        step *= 0.9985
    return centers, cur_r, cur_sum

def get_constraints(vars_arr, n, i_idx, j_idx):
    """Inequality constraints for joint center/radius optimization."""
    xs = vars_arr[:n]
    ys = vars_arr[n:2*n]
    rs = vars_arr[2*n:]
    
    # Boundary constraints
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Pairwise non-overlap constraints (squared form for differentiability)
    dx = xs[i_idx] - xs[j_idx]
    dy = ys[i_idx] - ys[j_idx]
    dr = rs[i_idx] + rs[j_idx]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def obj_func(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def run_packing():
    n = 26
    m = n * (n - 1) // 2
    
    # Precompute constant constraint matrix structure for LP
    A_ub = np.zeros((m, n))
    i_idx = np.empty(m, dtype=int)
    j_idx = np.empty(m, dtype=int)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            i_idx[k] = i
            j_idx[k] = j
            k += 1
            
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    starts = []
    hex_patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], [5,6,4,6,5]]
    for rd in hex_patterns:
        pts = []
        y = 0.1
        for idx, cnt in enumerate(rd):
            shift = 0.1 if idx % 2 == 1 else 0.0
            x = 0.1 + shift
            for _ in range(cnt):
                if len(pts) < n:
                    pts.append([x, y])
                x += 0.2
            y += 0.17
        starts.append(np.array(pts[:n]))
        
    # Add random starts to escape structural biases
    for _ in range(8):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Phase 1: Hill Climbing on Centers
    for cfg in starts:
        c, r, s = hill_climb(cfg, n, A_ub, i_idx, j_idx, rng)
        if s > best_sum:
            best_sum = s
            best_centers = c.copy()
            best_radii = r.copy()
            
    # Phase 2: SLSQP Joint Refinement
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    x0 = np.zeros(3 * n)
    x0[:n] = best_centers[:, 0]
    x0[n:2*n] = best_centers[:, 1]
    x0[2*n:] = best_radii
    
    try:
        res = minimize(obj_func, x0, args=(n,), method='SLSQP', 
                       bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': get_constraints, 'args': (n, i_idx, j_idx)},
                       options={'maxiter': 2000, 'ftol': 1e-10})
        if np.isfinite(res.fun):
            cx, cy, rr = res.x[:n], res.x[n:2*n], res.x[2*n:]
            centers_tmp = np.column_stack((cx, cy))
            r_lp, s_lp = solve_lp(centers_tmp, n, A_ub, i_idx, j_idx)
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = centers_tmp
                best_radii = r_lp
    except Exception:
        pass
        
    # Phase 3: Strict Numerical Safety Scaling
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    dx = best_centers[i_idx, 0] - best_centers[j_idx, 0]
    dy = best_centers[i_idx, 1] - best_centers[j_idx, 1]
    d = np.sqrt(dx**2 + dy**2)
    rs = best_radii[i_idx] + best_radii[j_idx]
    if np.any(rs > 1e-12):
        scale = min(scale, np.min(d / np.maximum(rs, 1e-12)))
        
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
