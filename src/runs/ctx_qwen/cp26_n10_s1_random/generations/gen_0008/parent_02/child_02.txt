from scipy.optimize import linprog, minimize

def solve_lp_radii(centers, n, triu_i, triu_j):
    """Solves the LP to maximize sum of radii for fixed centers."""
    # Compute upper bounds for each radius from boundaries
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 0.0)
    bounds = [(0.0, lim) for lim in limits]
    
    # Pairwise distance constraints: r_i + r_j <= dist(i, j)
    diffs = centers[triu_i] - centers[triu_j]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    
    m = len(triu_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), triu_i] = 1.0
    A_ub[np.arange(m), triu_j] = 1.0
    b_ub = dists
    
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective_joint(vars_arr, n):

    return -np.sum(vars_arr[2*n:])

    xs = vars_arr[:n]
    ys = vars_arr[n:2*n]
    rs = vars_arr[2*n:]

    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])

    dx = xs[triu_i] - xs[triu_j]
    dy = ys[triu_i] - ys[triu_j]
    dr = rs[triu_i] + rs[triu_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
