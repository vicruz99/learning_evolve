import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in limits]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = len(idx_i)
    A = np.zeros((m, n))
    b = dists[idx_i, idx_j]
    for k, (u, v) in enumerate(zip(idx_i, idx_j)):
        A[k, u] = 1.0
        A[k, v] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    r_safe = np.min(limits) * 0.9
    return np.full(n, max(r_safe, 1e-7)), max(r_safe, 1e-7) * n

def get_constraints_joint(v, n):
    """Inequality constraints for joint center/radius optimization."""
    xs, ys, rs = v[:n], v[n:2*n], v[2*n:]
    c = np.concatenate([xs-rs, 1-xs-rs, ys-rs, 1-ys-rs])
    i, j = np.triu_indices(n, k=1)
    dx, dy = xs[i]-xs[j], ys[i]-ys[j]
    c = np.concatenate([c, dx**2 + dy**2 - (rs[i]+rs[j])**2])
    return c

def objective_func(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])
