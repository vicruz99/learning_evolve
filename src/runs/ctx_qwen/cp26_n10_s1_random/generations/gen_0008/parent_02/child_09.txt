The problem asks to maximize the sum of radii of 26 circles in a unit square. This is a non-smooth, non-convex optimization problem. The most robust approach for this specific formulation is a two-stage method:
1. **LP Refinement for Radii**: For any fixed set of centers, finding the optimal radii that maximize the sum subject to non-overlap and boundary constraints is a Linear Programming problem. This LP is very fast and exact.
2. **Center Optimization**: We treat the optimal LP value as the objective function over the center coordinates. Since the LP value function is non-smooth but locally well-behaved, derivative-free optimizers like Powell's method work exceptionally well.
3. **Diverse Initialization**: Hexagonal lattice patterns are theoretically optimal for dense packing. We generate multiple row-distribution variants and perturbations to ensure we cover different local basins.
4. **Safety Scaling**: After optimization, we compute a strict scaling factor to guarantee the solution passes the `1e-12` tolerance check in the validator without overlaps or boundary violations.

This approach avoids the pitfalls of constraint-based NLP solvers (like SLSQP getting stuck in infeasible regions) and leverages the speed of `scipy.optimize.linprog` to rapidly evaluate and improve configurations.

```python
import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(TRIU_I)

# Precompute the constant structure of the pairwise constraint matrix
A_ub = np.zeros((NUM_PAIRS, N))
for k, (i, j) in enumerate(zip(TRIU_I, TRIU_J)):
    A_ub[k, i] = 1.0
    A_ub[k, j] = 1.0

def get_lp_sum_and_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Boundary limits: distance to nearest wall
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, l) for l in np.maximum(lims, 1e-9)]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    b_ub = dists[TRIU_I, TRIU_J]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.full(N, 1e-7)

def objective(centers_flat):
    """Objective for center optimization: negative optimal LP sum."""
    centers = centers_flat.reshape(N, 2)
    s, _ = get_lp_sum_and_radii(centers)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    inits = []
    
    # Diverse hexagonal row patterns summing to 26
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4],
        [5, 5, 5, 5, 6], [7, 5, 5, 5, 4]
    ]
    
    for pat in patterns:
        if sum(pat) != N: continue
        r_init = 0.095
        pts = []
        y = r_init
        for idx, cnt in enumerate(pat):
            shift = r_init if idx % 2 == 1 else 0.0
            x = r_init + shift
            for _ in range(cnt):
                pts.append([x, y])
                x += 2.0 * r_init
            y += np.sqrt(3) * r_init
        inits.append(np.array(pts[:N]))
        
    # Perturb the first pattern multiple times to escape symmetry traps
    base = inits[0]
    for _ in range(5):
        p = base + rng.uniform(-0.015, 0.015, (N, 2))
        inits.append(np.clip(p, 0.05, 0.95))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_cent = [(0.02, 0.98)] * (2 * N)
    
    for cfg in inits:
        try:
            # Powell's method handles the non-smooth LP objective landscape very well
            res = minimize(objective, cfg.flatten(), method='Powell',
                           bounds=bounds_cent,
                           options={'maxiter': 1200, 'ftol': 1e-12, 'xatol': 1e-10})
            if np.isfinite(res.fun):
                curr_centers = res.x.reshape(N, 2)
                curr_sum, curr_radii = get_lp_sum_and_radii(curr_centers)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = curr_centers
                    best_radii = curr_radii
        except Exception:
            continue
            
    # Fallback if optimization unexpectedly fails
    if best_centers is None:
        best_centers = inits[0]
        _, best_radii = get_lp_sum_and_radii(best_centers)
        best_sum = np.sum(best_radii)
        
    # Final safety scaling to guarantee strict numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    dx = best_centers[TRIU_I, 0] - best_centers[TRIU_J, 0]
    dy = best_centers[TRIU_I, 1] - best_centers[TRIU_J, 1]
    d = np.sqrt(dx**2 + dy**2)
    rs = best_radii[TRIU_I] + best_radii[TRIU_J]
    if np.any(rs > 1e-12):
        scale = min(scale, np.min(d / np.maximum(rs, 1e-12)))
        
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
```
