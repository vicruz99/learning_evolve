import numpy as np
from scipy.optimize import minimize, linprog

def compute_constraints(vars_arr, n, triu_i, triu_j):
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    # Boundary constraints
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise constraints: dist^2 >= (r_i + r_j)^2
    dx = x[triu_i] - x[triu_j]
    dy = y[triu_i] - y[triu_j]
    dr = r[triu_i] + r[triu_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def objective_func(vars_arr, n):
    return -np.sum(vars_arr[2*n:])

def penalty_objective(vars_arr, n, triu_i, triu_j, weight):
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    obj = -np.sum(r)
    
    # Boundary penalties
    pen = np.sum(np.maximum(0.0, r - x)**2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - x))**2)
    pen += np.sum(np.maximum(0.0, r - y)**2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - y))**2)
    
    # Overlap penalties
    dx = x[triu_i] - x[triu_j]
    dy = y[triu_i] - y[triu_j]
    dist = np.sqrt(dx**2 + dy**2)
    rs = r[triu_i] + r[triu_j]
    pen += np.sum(np.maximum(0.0, rs - dist)**2)
    
    return obj + weight * pen

def solve_lp_radii(centers, n, triu_i, triu_j):
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-9)) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    m = len(triu_i)
    A_ub = np.zeros((m, n))
    b_ub = dists[triu_i, triu_j]
    
    for k, (u, v) in enumerate(zip(triu_i, triu_j)):
        A_ub[k, u] = 1.0
        A_ub[k, v] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def check_validity(centers, radii, n, triu_i, triu_j):
    if np.any(centers[:, 0] < radii - 1e-9) or np.any(centers[:, 0] > 1.0 - radii + 1e-9):
        return False
    if np.any(centers[:, 1] < radii - 1e-9) or np.any(centers[:, 1] > 1.0 - radii + 1e-9):
        return False
    dx = centers[triu_i, 0] - centers[triu_j, 0]
    dy = centers[triu_i, 1] - centers[triu_j, 1]
    if np.any(dx**2 + dy**2 < (radii[triu_i] + radii[triu_j])**2 - 1e-9):
        return False
    return True

def generate_hex_init(rows, n, rng):
    pts = []
    y = 0.075
    for idx, cnt in enumerate(rows):
        shift = 0.09 if idx % 2 == 1 else 0.0
        width = (cnt - 1) * 0.17
        x_start = 0.5 - width / 2.0 + shift
        for c in range(cnt):
            if len(pts) < n:
                pts.append([x_start + c * 0.17, y])
        y += 0.147
        if len(pts) >= n:
            break
    arr = np.array(pts[:n])
    arr += rng.uniform(-0.004, 0.004, arr.shape)
    return np.clip(arr, 0.05, 0.95)
