The main challenge is optimizing the centers. Gradient-based methods like SLSQP often get trapped in local minima due to the non-smooth nature of the active constraints. A highly effective strategy for this class of problems is:

3. **Intensive Refinement**: After scanning all initial basins, perform multiple intensive hill-climbing passes on the best configuration found, with occasional small random perturbations to escape shallow local minima.
4. **Strict Safety Scaling**: Before returning, compute a global scaling factor that guarantees all boundary and pairwise constraints strictly satisfy the $10^{-12}$ tolerance required by the validator.

import numpy as np
from scipy.optimize import linprog

N = 26

def solve_radii_lp(centers, A_ub, triu_idx):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Distance to walls for each circle
    walls = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    # Bounds: 0 <= r_i <= wall_i. Ensure valid interval even if very close to boundary.
    bounds = [(0.0, max(w - 1e-9, 1e-9)) for w in walls]
    
    # Pairwise Euclidean distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    b_ub = dists[triu_idx]
    
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    # Fallback to tiny radii if LP fails
    return np.full(n, 1e-8), 0.0

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    # Precompute constant parts for the LP constraint matrix
    triu_idx = np.triu_indices(n, k=1)
    m = len(triu_idx[0])
    A_ub = np.zeros((m, n))
    for k, (i, j) in enumerate(zip(triu_idx[0], triu_idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    # Generate diverse hexagonal initial configurations
    starts = []
    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4],
        [5, 5, 5, 5, 6], [5, 6, 4, 6, 5], [7, 5, 5, 5, 4],
        [6, 5, 5, 6, 4], [5, 5, 7, 5, 4]
    ]
    
    for pat in row_patterns:
        if sum(pat) < n: 
            continue
        pts = []
        y = 0.08
        for row_idx, cnt in enumerate(pat):
            shift = 0.085 if row_idx % 2 == 1 else 0.0
            x = 0.085 + shift
            for _ in range(cnt):
                if len(pts) < n:
                    pts.append([x, y])
                x += 0.17
            y += 0.147
        starts.append(np.array(pts[:n]))
        
    # Add controlled perturbations to break symmetry
    for base in starts[:6]:
        for _ in range(5):
            p = base + rng.uniform(-0.025, 0.025, (n, 2))
            starts.append(np.clip(p, 0.05, 0.95))
            
    # Add purely random starts for diversity
    for _ in range(8):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    global_best_sum = -1.0
    global_best_centers = None
    global_best_radii = None
    
    # Phase 1: Hill climbing from each diverse start
    for cfg in starts:
        centers = cfg.copy()
        radii, curr_sum = solve_radii_lp(centers, A_ub, triu_idx)
        
        step = 0.04
        for _ in range(600):
            i = rng.integers(n)
            old_c = centers[i].copy()
            
            centers[i] += rng.uniform(-step, step, 2)
            centers[i] = np.clip(centers[i], 0.02, 0.98)
            
            r_new, s_new = solve_radii_lp(centers, A_ub, triu_idx)
            if s_new > curr_sum + 1e-8:
                curr_sum = s_new
                radii = r_new
            else:
                centers[i] = old_c
                
            step *= 0.996
            
        if curr_sum > global_best_sum:
            global_best_sum = curr_sum
            global_best_centers = centers.copy()
            global_best_radii = radii.copy()
            
    # Phase 2: Intensive refinement on the best configuration found
    if global_best_centers is not None:
        centers = global_best_centers.copy()
        radii, curr_sum = solve_radii_lp(centers, A_ub, triu_idx)
        
        for restart in range(3):
            step = 0.025
            for _ in range(1200):
                i = rng.integers(n)
                old_c = centers[i].copy()
                
                centers[i] += rng.uniform(-step, step, 2)
                centers[i] = np.clip(centers[i], 0.02, 0.98)
                
                r_new, s_new = solve_radii_lp(centers, A_ub, triu_idx)
                if s_new > curr_sum + 1e-8:
                    curr_sum = s_new
                    radii = r_new
                else:
                    centers[i] = old_c
                    
                step *= 0.997
                
            if curr_sum > global_best_sum:
                global_best_sum = curr_sum
                global_best_centers = centers.copy()
                global_best_radii = radii.copy()
                
            # Small random perturbation to escape local minima if progress stalls
            centers += rng.uniform(-0.008, 0.008, (n, 2))
            centers = np.clip(centers, 0.05, 0.95)
            radii, curr_sum = solve_radii_lp(centers, A_ub, triu_idx)
            if curr_sum > global_best_sum:
                global_best_sum = curr_sum
                global_best_centers = centers.copy()
                global_best_radii = radii.copy()

    best_centers = global_best_centers
    best_radii = global_best_radii
    
    # Fallback configuration if something went wrong
    if best_centers is None:
        best_centers = starts[0]
        best_radii, _ = solve_radii_lp(best_centers, A_ub, triu_idx)
        global_best_sum = float(np.sum(best_radii))
        
    # Phase 3: Strict safety scaling to guarantee validation tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    dx = best_centers[triu_idx[0], 0] - best_centers[triu_idx[1], 0]
    dy = best_centers[triu_idx[0], 1] - best_centers[triu_idx[1], 1]
    d = np.sqrt(dx**2 + dy**2)
    rs = best_radii[triu_idx[0]] + best_radii[triu_idx[1]]
    mask = rs > 1e-12
    if np.any(mask):
        scale = min(scale, np.min(d[mask] / rs[mask]))
        
    best_radii *= scale * 0.9999999
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
