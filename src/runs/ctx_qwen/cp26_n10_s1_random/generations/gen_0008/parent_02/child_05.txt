import numpy as np
from scipy.optimize import minimize, linprog

def get_init_configs(n):
    configs = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4],
        [5, 5, 5, 5, 6], [6, 5, 5, 6, 4], [5, 6, 4, 6, 5],
        [7, 5, 5, 5, 4], [5, 6, 6, 5, 4], [4, 5, 7, 5, 5]
    ]
    for pat in patterns:
        if sum(pat) < n: continue
        pts = []
        y = 0.09
        for idx, cnt in enumerate(pat):
            shift = 0.095 if idx % 2 == 1 else 0.0
            width = (cnt - 1) * 0.19
            x_start = 0.5 - width / 2.0 + shift
            for c in range(cnt):
                if len(pts) < n:
                    pts.append([x_start + c * 0.19, y])
            y += 0.1645
            if len(pts) >= n: break
        configs.append(np.array(pts[:n]))
        
    rng = np.random.default_rng(42)
    for base in configs[:6]:
        for _ in range(3):
            p = base + rng.uniform(-0.02, 0.02, base.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    return configs

def solve_lp_fast(centers, n, triu_i, triu_j, A_ub):
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-9)) for lim in limits]
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[triu_i, triu_j]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-7), 0.0

def hill_climb(centers, n, triu_i, triu_j, A_ub, steps=2000, seed=0):
    rng = np.random.default_rng(seed)
    best_c = centers.copy()
    best_r, best_s = solve_lp_fast(best_c, n, triu_i, triu_j, A_ub)
    step = 0.03
    
    for _ in range(steps):
        i = rng.integers(n)
        trial = best_c.copy()
        trial[i] += rng.uniform(-step, step, 2)
        trial[i] = np.clip(trial[i], 1e-4, 1.0 - 1e-4)
        
        r_new, s_new = solve_lp_fast(trial, n, triu_i, triu_j, A_ub)
        if s_new > best_s + 1e-9:
            best_c = trial
            best_r = r_new
            best_s = s_new
            step *= 1.02
        else:
            step *= 0.98
    return best_c, best_r, best_s

def obj_joint(v, n):
    return -np.sum(v[2*n:])

def cons_joint(v, n):
    cx, cy, cr = v[:n], v[n:2*n], v[2*n:]
    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    rs = cr[:, None] + cr[None, :]
    idx = np.triu_indices(n, k=1)
    c = np.concatenate([c, d2[idx] - rs[idx]**2])
    return c

def joint_slsqp(centers, radii, n):
    x0 = np.zeros(3 * n)
    x0[:n] = centers[:, 0]
    x0[n:2*n] = centers[:, 1]
    x0[2*n:] = radii
    
    bnds = [(0.0, 1.0)]*(2*n) + [(1e-6, 0.5)]*n
    cons = {'type': 'ineq', 'fun': cons_joint, 'args': (n,)}
    
    try:
        res = minimize(obj_joint, x0, args=(n,), method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-13})
        if res.success and np.isfinite(res.fun):
            return res.x[:2*n].reshape(n, 2), res.x[2*n:]
    except Exception:
        pass
    return centers, radii

def run_packing():
    n = 26
    triu_i, triu_j = np.triu_indices(n, k=1)
    m = len(triu_i)
    
    A_ub = np.zeros((m, n))
    for k, (i, j) in enumerate(zip(triu_i, triu_j)):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    configs = get_init_configs(n)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    for idx, cfg in enumerate(configs):
        c1, r1, s1 = hill_climb(cfg, n, triu_i, triu_j, A_ub, steps=2000, seed=idx)
        c2, r2 = joint_slsqp(c1, r1, n)
        _, s2 = solve_lp_fast(c2, n, triu_i, triu_j, A_ub)
        
        if s2 > best_sum:
            best_sum = s2
            best_centers = c2.copy()
            best_radii, _ = solve_lp_fast(best_centers, n, triu_i, triu_j, A_ub)
            
    if best_centers is not None:
        rng_final = np.random.default_rng(123)
        for _ in range(15):
            pert = best_centers + rng_final.uniform(-0.015, 0.015, best_centers.shape)
            pert = np.clip(pert, 0.02, 0.98)
            c3, r3, s3 = hill_climb(pert, n, triu_i, triu_j, A_ub, steps=1500, seed=999)
            if s3 > best_sum:
                best_sum = s3
                best_centers = c3.copy()
                best_radii = r3.copy()
                c4, r4 = joint_slsqp(best_centers, best_radii, n)
                _, s4 = solve_lp_fast(c4, n, triu_i, triu_j, A_ub)
                if s4 > best_sum:
                    best_sum = s4
                    best_centers = c4.copy()
                    best_radii, _ = solve_lp_fast(best_centers, n, triu_i, triu_j, A_ub)

    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    dx = best_centers[triu_i, 0] - best_centers[triu_j, 0]
    dy = best_centers[triu_i, 1] - best_centers[triu_j, 1]
    d = np.sqrt(dx**2 + dy**2)
    rs = best_radii[triu_i] + best_radii[triu_j]
    if np.any(rs > 1e-12):
        scale = min(scale, np.min(d / np.maximum(rs, 1e-12)))
        
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
