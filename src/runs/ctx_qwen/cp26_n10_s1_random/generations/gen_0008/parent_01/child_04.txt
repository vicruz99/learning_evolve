from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lim = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                     np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lim = np.maximum(lim, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim[i]) for i in range(n)]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-20)
    np.fill_diagonal(dists, np.inf)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def joint_objective(x):

    n = 26
    return -np.sum(x[2*n:])

def joint_constraints(x):

    c = x[:2*n].reshape(n, 2)

    # Pairwise non-overlap constraints: dist(i,j) >= r_i + r_j
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    dists = np.sqrt(dx**2 + dy**2 + 1e-15)

    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    return np.concatenate([bc, dists[mask] - r_sum[mask]])

def generate_hex_configs(n):

    configs = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 4, 6, 4], [5, 6, 4, 6, 5],

        [7, 5, 6, 5, 3], [4, 5, 6, 6, 5], [5, 7, 4, 6, 4]
    ]
    
    for pat in patterns:
        if sum(pat) < n: continue
        for r0 in [0.095, 0.10, 0.105, 0.102]:
            pts = []
            y = r0
            for i, cnt in enumerate(pat):
                shift = r0 if i % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) < n:
                        pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            configs.append(np.clip(np.array(pts[:n]), 0.05, 0.95))
    return configs

def hill_climb_lp(centers, n, steps=1000, rng=None):
    """Derivative-free hill climbing optimizing centers by LP-evaluated radii sum."""
    if rng is None:
        rng = np.random.default_rng()
    best_c = centers.copy()

        if val_trial > val_best + 1e-8:
            best_c = trial_c
            r_best = r_trial
            val_best = val_trial
            
    return best_c, r_best, val_best
