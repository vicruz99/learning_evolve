import numpy as np
from scipy.optimize import minimize, linprog

def get_clearances(centers, n):
    """Computes distances to boundaries and half-pairwise distances."""
    dx = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    dy = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    b_dist = np.minimum(dx, dy)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    p_dist = dists / 2.0
    
    triu_idx = np.triu_indices(n, k=1)
    return np.concatenate([b_dist, p_dist[triu_idx]])

def neg_smooth_min(vars_flat, n, k):
    """Negative smooth minimum clearance for L-BFGS-B optimization."""
    centers = vars_flat.reshape(n, 2)
    c = get_clearances(centers, n)
    max_c = np.max(c)
    val = -np.log(np.sum(np.exp(-k * (c - max_c)))) / k - max_c
    return -val

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lim = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                     np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lim = np.maximum(lim, 1e-10)
    bounds = [(0.0, lim[i]) for i in range(n)]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    A_ub = np.zeros((len(idx_i), n))
    A_ub[np.arange(len(idx_i)), idx_i] = 1.0
    A_ub[np.arange(len(idx_i)), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0
