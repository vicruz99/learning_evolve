import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(w, 1e-9)) for w in wall_dists]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def get_equal_obj(x, n):
    """Objective for equal-radius phase: minimize -r"""
    return -x[2*n]

def get_equal_cons(x, n):
    """Inequality constraints for equal-radius packing: dist >= 2r and wall dist >= r"""
    c = x[:2*n].reshape(n, 2)
    r = x[2*n]
    bc = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    d2 = dx**2 + dy**2
    np.fill_diagonal(d2, 1.0)
    mask = np.triu_indices(n, k=1)
    return np.concatenate([bc, d2[mask] - 4.0 * r**2])

def make_hex_config(n, rows, r_init=0.10):
    """Generates a hexagonal lattice initialization with specified row counts."""
    pts = []
    y = r_init
    for i, cnt in enumerate(rows):
        shift = r_init if i % 2 == 1 else 0.0
        x = r_init + shift
        for _ in range(cnt):
            if len(pts) < n:
                pts.append([x, y])
            x += 2.0 * r_init
        y += np.sqrt(3) * r_init
    return np.array(pts[:n])

def refine_centers(centers, n, rng, steps=1200):
    """LP-guided hill climbing to optimize centers for maximum sum of radii."""
    best_c = centers.copy()
    _, val_best = solve_lp_radii(best_c)
    step = 0.015
    
    for _ in range(steps):
        i = rng.integers(n)
        best_local_val = val_best
        best_local_c = best_c.copy()
        
        # Try multiple candidate moves
        for _ in range(6):
            delta = rng.uniform(-step, step, 2)
            trial_c = best_c.copy()
            trial_c[i] = np.clip(trial_c[i] + delta, 1e-4, 1.0 - 1e-4)
            _, val_trial = solve_lp_radii(trial_c)
            if val_trial > best_local_val:
                best_local_val = val_trial
                best_local_c = trial_c
                
        if best_local_val > val_best + 1e-9:
            best_c = best_local_c
            val_best = best_local_val
            
        step = max(1e-4, step * 0.998)
    return best_c, val_best

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    # Diverse row distributions that sum to 26
    patterns = [

    ]
    
    configs = []

                r_lp, s_lp = solve_lp_radii(c_eq)

            best_centers = c_ref
