import numpy as np
from scipy.optimize import minimize, linprog

def compute_centers_from_params(params, n):
    r = params[:n]
    u = params[n:2*n]
    v = params[2*n:3*n]
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    return np.column_stack((x, y))

def lp_maximize_sum_radii(centers):
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    c_obj = -np.ones(n)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = [(0.0, ub[i]) for i in range(n)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def objective_param(params, n):
    return -np.sum(params[:n])

def constraints_param(params, n, triu_i, triu_j):
    r = params[:n]
    u = params[n:2*n]
    v = params[2*n:3*n]
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[triu_i, triu_j] - rs[triu_i, triu_j]**2

def hill_climb(centers, n, steps=3000, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    best_c = centers.copy()
    r_best, val_best = lp_maximize_sum_radii(best_c)
    
    step_size = 0.02
    for step in range(steps):
        i = rng.integers(n)
        trial_c = best_c.copy()
        trial_c[i] += rng.uniform(-step_size, step_size, 2)
        trial_c[i] = np.clip(trial_c[i], 1e-4, 1.0 - 1e-4)
        
        r_try, val_try = lp_maximize_sum_radii(trial_c)
        if val_try > val_best + 1e-8:
            best_c = trial_c
            r_best = r_try
            val_best = val_try
            
        step_size = max(0.0005, step_size * 0.999)
        
    return best_c, r_best, val_best
