import numpy as np
from scipy.optimize import linprog

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lim = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                     np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lim = np.maximum(lim, 0.0)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim[i]) for i in range(n)]
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def force_expand(centers, radii, steps=400):
    """Force-directed layout to spread circles and grow radii until tightly packed."""
    n = centers.shape[0]
    dt = 0.005
    for _ in range(steps):
        radii += 0.0004
        forces = np.zeros_like(centers)
        # Boundary repulsion
        for d in range(2):
            viol1 = np.maximum(0, radii - centers[:, d])
            viol2 = np.maximum(0, centers[:, d] + radii - 1.0)
            forces[:, d] += (viol2 - viol1) * 200.0
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                req = radii[i] + radii[j]
                if dist < req and dist > 1e-7:
                    f = (req - dist) * 150.0 / dist
                    forces[i] += diff * f
                    forces[j] -= diff * f
        centers += forces * dt
        centers = np.clip(centers, 0.001, 0.999)
    return centers, radii

def local_opt(centers, rng, max_iter=1200):
    """Optimizes centers to maximize LP-evaluated sum of radii."""
    n = centers.shape[0]
    best_c = centers.copy()
    _, current_sum = solve_radii_lp(best_c)
    step = 0.015
    
    for it in range(max_iter):
        step = 0.015 * (0.996 ** it)
        if step < 1e-5: step = 1e-5
        
        # Single move
        i = rng.integers(n)
        improved = False
        for _ in range(5):
            trial = best_c.copy()
            trial[i] += rng.uniform(-step, step, 2)
            trial[i] = np.clip(trial[i], 0.002, 0.998)
            _, new_sum = solve_radii_lp(trial)
            if new_sum > current_sum + 1e-7:
                best_c = trial
                current_sum = new_sum
                improved = True
                break
                
        # Swap move (helps with unequal radii arrangements)
        if not improved:
            i, j = rng.choice(n, 2, replace=False)
            trial = best_c.copy()
            trial[i], trial[j] = trial[j], trial[i]
            _, new_sum = solve_radii_lp(trial)
            if new_sum > current_sum + 1e-7:
                best_c = trial
                current_sum = new_sum
                
        # Occasional global perturbation to escape local minima
        if it % 250 == 0 and it > 0 and rng.random() < 0.25:
            trial = best_c + rng.uniform(-step*1.5, step*1.5, best_c.shape)
            trial = np.clip(trial, 0.01, 0.99)
            _, new_sum = solve_radii_lp(trial)
            if new_sum > current_sum + 1e-7:
                best_c = trial
                current_sum = new_sum
                
    radii, final_sum = solve_radii_lp(best_c)
    return best_c, radii, final_sum
