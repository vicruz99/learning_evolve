import numpy as np
from scipy.optimize import linprog

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Boundary constraints: r_i <= distance to each wall
    lim = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                     np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lim = np.maximum(lim, 1e-10)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim[i]) for i in range(n)]
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0:1] - centers[:, 0:1].T
    dy = centers[:, 1:2] - centers[:, 1:2].T
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, 1e9)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def force_directed_init(n, rng):
    """Generates initial positions via force-directed relaxation."""
    centers = rng.uniform(0.2, 0.8, (n, 2))
    dt = 0.01
    repulsion = 0.05
    for step in range(1500):
        forces = np.zeros((n, 2))
        # Wall repulsion (push towards center, hard boundary near edges)
        for i in range(n):
            forces[i, 0] += (0.5 - centers[i, 0]) * 2.0
            forces[i, 1] += (0.5 - centers[i, 1]) * 2.0
            forces[i, 0] += max(0, 0.05 - centers[i, 0]) * 100.0
            forces[i, 0] -= max(0, centers[i, 0] - 0.95) * 100.0
            forces[i, 1] += max(0, 0.05 - centers[i, 1]) * 100.0
            forces[i, 1] -= max(0, centers[i, 1] - 0.95) * 100.0
            
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist < 0.001: dist = 0.001
                f = repulsion / (dist**2)
                forces[i] += diff / dist * f
                forces[j] -= diff / dist * f
                
        centers += forces * dt
        centers = np.clip(centers, 0.01, 0.99)
        if step > 1000:
            dt *= 0.995
        repulsion *= 0.998
    return centers

def hill_climb(centers, n, rng):
    """Derivative-free hill climbing optimizing centers by LP-evaluated radii sum."""
    best_c = centers.copy()
    r_best, val_best = solve_radii_lp(best_c)
    
    step_size = 0.012
    for _ in range(4000):
        i = rng.integers(n)
        direction = rng.uniform(-1.0, 1.0, 2)
        norm = np.linalg.norm(direction)
        if norm > 0:
            direction /= norm
        trial_c = best_c.copy()
        trial_c[i] += direction * step_size
        trial_c[i] = np.clip(trial_c[i], 0.001, 0.999)
        
        r_trial, val_trial = solve_radii_lp(trial_c)
        if val_trial > val_best + 1e-8:
            best_c = trial_c
            r_best = r_trial
            val_best = val_trial
            
        step_size *= 0.9995
    return best_c, r_best, val_best

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)

    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],

    for cfg in inits:
        cfg = np.clip(cfg, 0.02, 0.98)

        c_opt, r_opt, val_opt = hill_climb(cfg, n, rng)

            best_sum = val_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()

            # Randomly select half the circles to perturb
            idx = rng.choice(n, size=n//2, replace=False)
            pert[idx] += rng.uniform(-0.006, 0.006, (n//2, 2))

            c_pert, r_pert, val_pert = hill_climb(pert, n, rng)

    scale = 1.0
    c = best_centers

    for i in range(n):

            scale = min(scale, c[i,0]/r[i], (1.0-c[i,0])/r[i], c[i,1]/r[i], (1.0-c[i,1])/r[i])

            r_sum = r[i] + r[j]
            if r_sum > 1e-12:
                scale = min(scale, d / r_sum)

    best_sum = float(np.sum(r))
    best_radii = r
    
    return best_centers, best_radii, best_sum
