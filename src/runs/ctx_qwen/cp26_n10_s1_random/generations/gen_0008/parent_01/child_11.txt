import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Maximum radius allowed by boundaries
    lim = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                     np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lim = np.maximum(lim, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim[i]) for i in range(n)]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def joint_obj(vars_arr, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def joint_cons(vars_arr, n):
    """Returns inequality constraints >= 0 for valid packing."""
    c = vars_arr[:2*n].reshape(n, 2)
    r = vars_arr[2*n:]
    
    # Boundary constraints
    bc = np.concatenate([c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r])
    
    # Pairwise separation constraints
    dx = c[:,0:1] - c[:,0:1].T
    dy = c[:,1:2] - c[:,1:2].T
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, 1.0)
    
    r_sum = r[:,None] + r[None,:]
    mask = np.triu(np.ones((n,n), dtype=bool), k=1)
    return np.concatenate([bc, dists[mask] - r_sum[mask]])

def sa_search(centers, n, rng, steps=1500):
    """Simulated annealing to optimize centers by evaluating radii sum via LP."""
    best_c = centers.copy()
    _, best_val = solve_radii_lp(best_c)
    r_best = np.zeros(n)
    
    temp = 0.006
    step_sz = 0.025
    
    for _ in range(steps):
        i = rng.integers(n)
        trial_c = best_c.copy()
        trial_c[i] += rng.uniform(-step_sz, step_sz, 2)
        trial_c[i] = np.clip(trial_c[i], 1e-4, 1.0 - 1e-4)
        
        r_try, val_try = solve_radii_lp(trial_c)
        delta = val_try - best_val
        
        # Accept better solutions or worse solutions with probability exp(delta/temp)
        if delta > 0 or rng.random() < np.exp(delta / max(temp, 1e-7)):
            best_c = trial_c
            r_best = r_try
            best_val = val_try
            
        step_sz *= 0.9995
        temp *= 0.9995
        
    return best_c, r_best, best_val

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    best_c, best_r, best_sum = None, None, -1.0
    
    # Generate diverse initial configurations
    starts = []
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], [6,6,5,5,4]]
    for pat in patterns:
        pts = []
        y = 0.1
        for ri, cnt in enumerate(pat):
            shift = 0.1 if ri % 2 == 1 else 0.0
            x = 0.1 + shift
            for _ in range(cnt):
                if len(pts) < n:
                    pts.append([x, y])
                x += 0.2
            y += 0.1732
        starts.append(np.clip(np.array(pts[:n]), 0.05, 0.95))
        
    for _ in range(8):
        starts.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    bounds_vars = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    cons_dict = {'type': 'ineq', 'fun': joint_cons, 'args': (n,)}
    
    # Optimization loop over all starts
    for cfg in starts:
        # Phase 1: Simulated Annealing on centers
        c_sa, r_sa, v_sa = sa_search(cfg, n, rng, steps=1200)
        if v_sa > best_sum:
            best_c, best_r, best_sum = c_sa, r_sa, v_sa
            
        # Phase 2: Joint SLSQP polish to tighten constraints
        x0 = np.concatenate([best_c.flatten(), best_r * 0.99])
        try:
            res = minimize(joint_obj, x0, args=(n,), method='SLSQP',
                          bounds=bounds_vars, constraints=cons_dict,
                          options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if np.isfinite(res.fun):
                c_opt = res.x[:2*n].reshape(n, 2)
                r_lp, v_lp = solve_radii_lp(c_opt)
                if v_lp > best_sum:
                    best_c, best_r, best_sum = c_opt, r_lp, v_lp
        except Exception:
            pass

    # Final safety scaling to strictly satisfy 1e-12 validator tolerance
    scale = 1.0
    for i in range(n):
        if best_r[i] > 1e-12:
            scale = min(scale, best_c[i,0]/best_r[i], (1.0-best_c[i,0])/best_r[i],
                        best_c[i,1]/best_r[i], (1.0-best_c[i,1])/best_r[i])
    for i in range(n):

            d = np.linalg.norm(best_c[i]-best_c[j])
            rs = best_r[i]+best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
