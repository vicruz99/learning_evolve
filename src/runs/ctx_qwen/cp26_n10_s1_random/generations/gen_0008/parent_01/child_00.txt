import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lim = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                     np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lim = np.maximum(lim, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim[i]) for i in range(n)]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def slsqp_objective(vars_arr, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def slsqp_constraints(vars_arr, n):
    """Inequality constraints for SLSQP: must be >= 0."""
    c = vars_arr[:2*n].reshape(n, 2)
    r = vars_arr[2*n:]
    bc = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    d2 = dx**2 + dy**2
    np.fill_diagonal(d2, 1.0)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    return np.concatenate([bc, d2[mask] - r_sum**2[mask]])

def get_inits(n):
    """Generates diverse initial configurations."""
    configs = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4],
        [5, 5, 5, 5, 6], [7, 5, 5, 5, 4], [4, 5, 6, 5, 6]
    ]
    for pat in patterns:
        if sum(pat) < n: continue
        pts = []
        y = 0.09
        row = 0
        for cnt in pat:
            shift = 0.09 if row % 2 == 1 else 0.0
            x = 0.09 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 0.18
            y += 0.155
            row += 1
        configs.append(np.array(pts[:n]))
        
    rng = np.random.default_rng(123)
    for _ in range(8):
        pts = rng.uniform(0.1, 0.9, (n, 2))
        configs.append(pts)
        
    return configs

def sa_optimize(centers, n, rng, steps=1500):
    """Simulated Annealing on centers evaluated via LP."""
    best_c = centers.copy()
    _, best_val = solve_lp(best_c)
    
    temp = 0.015
    for i in range(steps):
        temp = max(1e-6, 0.015 * np.exp(-i / 400.0))
        step_size = max(0.0005, 0.025 * np.exp(-i / 800.0))
        
        idx = rng.integers(n)
        trial_c = best_c.copy()
        trial_c[idx] += rng.normal(0, step_size, 2)
        trial_c[idx] = np.clip(trial_c[idx], 0.001, 0.999)
        
        _, trial_val = solve_lp(trial_c)
        
        diff = trial_val - best_val
        if diff > 0:
            best_c = trial_c
            best_val = trial_val
        elif rng.random() < np.exp(diff / max(temp, 1e-9)):
            best_c = trial_c
            best_val = trial_val
            
    return best_c, best_val

def slsqp_refine(centers, n, rng):
    """Local refinement using SLSQP from perturbed starting points."""
    best_c = centers.copy()
    _, best_val = solve_lp(best_c)
    
    r0 = np.full(n, 0.09)
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-5, 0.25)] * n
    
    for _ in range(3):
        pert = best_c.copy()
        pert += rng.uniform(-0.008, 0.008, pert.shape)
        pert = np.clip(pert, 0.05, 0.95)
        x0 = np.concatenate([pert.flatten(), r0 * 0.95])
        try:
            res = minimize(slsqp_objective, x0, args=(n,), method='SLSQP', bounds=bounds,
                          constraints={'type': 'ineq', 'fun': slsqp_constraints, 'args': (n,)},
                          options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            if np.isfinite(res.fun):
                c_opt = res.x[:2*n].reshape(n, 2)
                r_lp, val_lp = solve_lp(c_opt)
                if val_lp > best_val:
                    best_c = c_opt
                    best_val = val_lp
        except Exception:
            pass
            
    return best_c, best_val

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    best_c = None
    best_val = 0.0
    best_r = None
    
    inits = get_inits(n)
    
    # Phase 1: SA exploration from diverse starts
    for cfg in inits:
        c_sa, val_sa = sa_optimize(cfg, n, rng, steps=1200)
        if val_sa > best_val:
            best_c = c_sa
            best_val = val_sa
            best_r, _ = solve_lp(best_c)
            
    # Phase 2: SLSQP local polishing
    if best_c is not None:
        best_c, best_val = slsqp_refine(best_c, n, rng)
        best_r, _ = solve_lp(best_c)
        
    # Phase 3: Deep SA search to escape any remaining traps
    if best_c is not None:
        best_c, best_val = sa_optimize(best_c, n, rng, steps=2000)
        best_r, _ = solve_lp(best_c)
        
    # Fallback safety net
    if best_c is None:
        best_c = inits[0]
        best_r, best_val = solve_lp(best_c)
        
    # Strict numerical safety scaling to guarantee 1e-12 validator tolerance
    scale = 1.0
    for i in range(n):
        if best_r[i] > 1e-12:
            scale = min(scale, best_c[i,0]/best_r[i], (1.0-best_c[i,0])/best_r[i], 
                       best_c[i,1]/best_r[i], (1.0-best_c[i,1])/best_r[i])
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(best_c[i]-best_c[j])
            r_sum = best_r[i] + best_r[j]
            if r_sum > 1e-12:
                scale = min(scale, d / r_sum)
                
    best_r *= scale * 0.9999995
    best_val = float(np.sum(best_r))
    
    return best_c, best_r, best_val
