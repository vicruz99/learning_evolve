import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def decode_params(params, n):
    """Decodes parameters to centers and radii using automatic boundary parameterization."""
    r = params[:n]
    u = params[n:2*n]
    v = params[2*n:3*n]
    factor = 1.0 - 2.0 * r
    x = r + factor * u
    y = r + factor * v
    return np.column_stack((x, y)), r

def compute_constraints_slsqp(params, n):
    """Computes pairwise non-overlap constraints >= 0 for SLSQP."""
    centers, r = decode_params(params, n)
    dx = centers[:, 0:1] - centers[:, 0:1].T
    dy = centers[:, 1:2] - centers[:, 1:2].T
    d2 = dx**2 + dy**2
    np.fill_diagonal(d2, 1.0)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    return d2[mask] - r_sum_sq[mask]

def objective_slsqp(params, n):
    """Objective: maximize sum of radii => minimize negative sum."""
    _, r = decode_params(params, n)
    return -np.sum(r)

def solve_radii_lp(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    lim = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                     np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lim = np.maximum(lim, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim[i]) for i in range(n)]

    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def force_directed(centers, radii, n, steps=2000):

    centers = centers.copy()
    radii = radii.copy()
    for step in range(steps):
        radii += 2e-5
        forces = np.zeros_like(centers)
        
        # Boundary repulsion
        for d in range(2):
            viol1 = np.maximum(0, radii - centers[:, d])
            viol2 = np.maximum(0, centers[:, d] + radii - 1.0)
            forces[:, d] += (viol2 - viol1) * 150.0
            
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                min_d = radii[i] + radii[j]
                if dist < min_d and dist > 1e-7:
                    overlap = min_d - dist
                    dir_ = diff / dist
                    f = overlap * 300.0
                    forces[i] += dir_ * f
                    forces[j] -= dir_ * f
                    
        centers += forces * 0.004
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    return centers, radii
