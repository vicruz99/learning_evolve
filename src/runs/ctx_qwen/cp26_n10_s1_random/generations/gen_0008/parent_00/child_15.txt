import numpy as np
from scipy.optimize import linprog, minimize

def precompute_A_ub(n):
    """Precomputes the constant A_ub matrix for the LP."""
    m_boundary = 4 * n
    m_pairwise = n * (n - 1) // 2
    m = m_boundary + m_pairwise
    A = np.zeros((m, n))
    
    for i in range(n):
        A[i, i] = 1.0          # r_i <= x_i
        A[n + i, i] = 1.0      # r_i <= 1 - x_i
        A[2*n + i, i] = 1.0    # r_i <= y_i
        A[3*n + i, i] = 1.0    # r_i <= 1 - y_i
        
    idx = m_boundary
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
    return A

def solve_lp_radii(centers, A_ub):
    """Solves LP to maximize sum of radii given fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    b = np.empty(4 * n + n * (n - 1) // 2)
    
    for i in range(n):
        x, y = centers[i]
        b[i] = x
        b[n + i] = 1.0 - x
        b[2*n + i] = y
        b[3*n + i] = 1.0 - y
        
    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b[idx] = np.hypot(dx, dy)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-9)

def hill_climbing_centers(centers, A_ub, n_steps, init_step, rng):
    """Optimizes centers by hill climbing, evaluating radii via LP."""
    n = centers.shape[0]
    curr = centers.copy()
    rad = solve_lp_radii(curr, A_ub)
    best_s = np.sum(rad)
    step = init_step
    
    for _ in range(n_steps):
        step *= 0.999
        i = rng.integers(n)
        old = curr[i].copy()
        
        prop = old + rng.uniform(-step, step, 2)
        prop = np.clip(prop, 1e-6, 1.0 - 1e-6)
        
        curr[i] = prop
        new_rad = solve_lp_radii(curr, A_ub)
        new_s = np.sum(new_rad)
        
        if new_s > best_s + 1e-10:
            best_s = new_s
            rad = new_rad
        else:
            curr[i] = old
            
    return curr, rad, best_s

def make_hex(rows, r0):
    """Generates initial positions on a hexagonal lattice."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def joint_obj(x, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2 * n:])

def joint_cons(x, n):
    """Inequality constraints for SLSQP."""
    cx = x[:n]
    cy = x[n:2 * n]
    r = x[2 * n:]
    con = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    d2 = dx**2 + dy**2
    r_sum_sq = (r[idx_i] + r[idx_j])**2
    return np.concatenate([con, d2 - r_sum_sq])
