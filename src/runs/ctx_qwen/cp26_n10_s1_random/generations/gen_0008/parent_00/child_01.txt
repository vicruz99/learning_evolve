import numpy as np
from scipy.optimize import linprog

N = 26
M = 4 * N + N * (N - 1) // 2

def get_A_ub():
    A = np.zeros((M, N))
    for i in range(N):
        A[i, i] = 1.0
        A[N + i, i] = 1.0
        A[2 * N + i, i] = 1.0
        A[3 * N + i, i] = 1.0
    k = 4 * N
    for i in range(N):
        for j in range(i + 1, N):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

A_UB = get_A_ub()
C_OBJ = -np.ones(N)
BOUNDS_LP = [(0, None)] * N

def compute_b(centers):
    b = np.zeros(M)
    x = centers[:, 0]
    y = centers[:, 1]
    b[:N] = x
    b[N:2*N] = 1.0 - x
    b[2*N:3*N] = y
    b[3*N:4*N] = 1.0 - y
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    b[4*N:] = np.sqrt(dx**2 + dy**2)
    return b

def solve_lp(centers):
    b = compute_b(centers)
    try:
        res = linprog(C_OBJ, A_ub=A_UB, b_ub=b, bounds=BOUNDS_LP, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def run_local_search(centers0, rng, steps=1500):
    centers = centers0.copy()
    best_r, best_sum = solve_lp(centers)
    if best_r is None:
        return centers0, 0.0, np.full(N, 1e-6)

    current_sum = best_sum
    best_centers = centers.copy()
    best_radii = best_r.copy()

                    best_centers = current_centers.copy()

            else:
                current_centers[i] = old_c
        except Exception:
            current_centers[i] = old_c
            
        no_improve += 1
        if no_improve > 300:
            break
        step_size = max(step_size * 0.9995, 0.0002)
        
    return best_centers, best_sum, best_radii

def make_hex(rows, r0=0.09):
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
