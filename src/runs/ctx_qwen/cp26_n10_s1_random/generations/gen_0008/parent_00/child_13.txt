import numpy as np
from scipy.optimize import minimize, linprog

def precompute_A(n):
    """Precomputes the constant constraint matrix for the radius LP."""
    m_b = 4 * n
    m_p = n * (n - 1) // 2
    A = np.zeros((m_b + m_p, n))
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        A[i, i] = 1.0
        A[n + i, i] = 1.0
        A[2*n + i, i] = 1.0
        A[3*n + i, i] = 1.0
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    k = m_b
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp(centers, n, A):
    """Solves the LP to maximize sum of radii for fixed centers."""
    m = A.shape[0]
    b = np.zeros(m)
    # Fill boundary RHS
    for i in range(n):
        x, y = centers[i]
        b[i] = x
        b[n + i] = 1.0 - x
        b[2*n + i] = y
        b[3*n + i] = 1.0 - y
    # Fill pairwise RHS
    k = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b[k] = np.hypot(dx, dy)
            k += 1
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x
    except Exception:
        pass
    return None

def hill_climb(centers, n, A, rng, steps=300):
    """Greedy coordinate ascent on centers maximizing LP sum of radii."""
    cur = centers.copy()
    r = solve_lp(cur, n, A)
    if r is None: return cur, None
    best_s = np.sum(r)
    step_size = 0.015
    for _ in range(steps):
        i = rng.integers(n)
        old = cur[i].copy()
        cur[i] += rng.uniform(-step_size, step_size, 2)
        np.clip(cur, 1e-4, 1.0 - 1e-4, out=cur)
        
        r_new = solve_lp(cur, n, A)
        if r_new is not None:
            s_new = np.sum(r_new)
            if s_new > best_s:
                best_s = s_new
            else:
                cur[i] = old
        else:
            cur[i] = old
        step_size *= 0.998
    return cur, solve_lp(cur, n, A)

def joint_obj(x, *args):
    """Objective: minimize negative sum of radii."""
    n = x.size // 3
    return -np.sum(x[2*n:])

def joint_cons(x, *args):
    """Inequality constraints >= 0 for valid packing."""
    n = x.size // 3
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    
    con = np.empty(4*n + n*(n-1)//2)
    con[:n] = cx - r
    con[n:2*n] = 1.0 - cx - r
    con[2*n:3*n] = cy - r
    con[3*n:4*n] = 1.0 - cy - r
    
    # Vectorized pairwise squared distance constraints
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    idx_i, idx_j = np.triu_indices(n, k=1)
    con[4*n:] = cx_m[idx_i, idx_j]**2 + cy_m[idx_i, idx_j]**2 - r_m[idx_i, idx_j]**2
    return con

def make_hex(pattern, n=26, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    for _ in range(12):
        starts.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    best_sum = 0.0

        best_centers = make_hex(patterns[0])
        best_radii = solve_lp(best_centers, n, A)
        best_sum = float(np.sum(best_radii))
