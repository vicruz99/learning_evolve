import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_radii_lp(A, centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    b = np.empty(4*n + n*(n-1)//2)
    for i in range(n):
        x, y = centers[i]
        b[i] = max(x, 1e-9)
        b[n + i] = max(1.0 - x, 1e-9)
        b[2*n + i] = max(y, 1e-9)
        b[3*n + i] = max(1.0 - y, 1e-9)
        
    idx = 4*n
    for i in range(n):
        for j in range(i+1, n):
            b[idx] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            idx += 1
            
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def objective_lp(centers_flat, A):
    """Objective for center optimization: negative LP-optimal sum of radii."""
    centers = centers_flat.reshape(N, 2)
    r = solve_radii_lp(A, centers)
    if r is None:
        return -1.0
    return -np.sum(r)

def joint_obj(x):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def joint_cons(x):
    """Inequality constraints >= 0 for joint valid packing."""
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    con = []
    # Boundary constraints
    con.append(cx - r)
    con.append(1.0 - cx - r)
    con.append(cy - r)
    con.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    d2 = dx**2 + dy**2
    rs = r[i_idx] + r[j_idx]
    con.append(d2 - rs**2)
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        A[N + i, i] = 1.0
        A[2*N + i, i] = 1.0
        A[3*N + i, i] = 1.0
    idx = 4*N
    for i in range(N):
        for j in range(i+1, N):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
            
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse starting configurations

    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5]
    ]
    for p in patterns:
        pts = []
        y = 0.09
        for ri, cnt in enumerate(p):
            shift = 0.09 if ri % 2 == 1 else 0.0
            x = 0.09 + shift

                if len(pts) >= N: break
                pts.append([x, y])
                x += 0.18
            y += 0.155
        starts.append(np.array(pts[:N]))
        
    # Random starts to explore diverse basins
    for _ in range(20):
        starts.append(rng.uniform(0.05, 0.95, (N, 2)))

    # Nelder-Mead is robust for the non-smooth, piecewise-linear landscape of the LP value function
    for cfg in starts:

                           options={'maxiter': 2000, 'xatol': 1e-7, 'fatol': 1e-8})

                mask = np.triu(np.ones((N, N), dtype=bool), k=1)
                if np.any(d2_mat[mask] < rs_mat[mask] - 1e-8):
                    valid = False
                    
            if valid and s_j > best_sum:
                best_sum = s_j
                best_centers = centers_j
                best_radii = radii_j
    except Exception:
        pass
        
    # Final safety scaling to guarantee strict numerical validity within 1e-12 tolerance
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
