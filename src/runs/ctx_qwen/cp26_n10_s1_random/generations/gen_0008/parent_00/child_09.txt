import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    m = n + n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[idx, i] = 1.0
        b_ub[idx] = lim
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-8)

def obj_slsqp(vars, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2 * n:])

def cons_slsqp(vars, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = vars[:n]
    cy = vars[n:2 * n]
    r = vars[2 * n:]
    
    c = np.empty(4 * n + n * (n - 1) // 2)
    c[:n] = cx - r
    c[n:2 * n] = 1.0 - cx - r
    c[2 * n:3 * n] = cy - r
    c[3 * n:4 * n] = 1.0 - cy - r
    
    k = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            dr = r[i] + r[j]
            c[k] = dx * dx + dy * dy - dr * dr
            k += 1
    return c

def make_hex(rows_pattern, r0=0.095):
    """Generates initial positions on a hexagonal lattice."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):

        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

    r_init = solve_lp_radii(cur_c)
    best_sum = np.sum(r_init)
    
    for _ in range(steps):
        i = rng.integers(n)
        old = cur_c[i].copy()
        
        cur_c[i] += rng.uniform(-step, step, 2)
        cur_c[i] = np.clip(cur_c[i], 0.005, 0.995)
        
        r_try = solve_lp_radii(cur_c)
        s_try = np.sum(r_try)
        
        if s_try > best_sum + 1e-9:
            best_sum = s_try
            best_c = cur_c.copy()
        else:
            cur_c[i] = old
            
        step *= 0.998
        
    return best_c, best_sum
