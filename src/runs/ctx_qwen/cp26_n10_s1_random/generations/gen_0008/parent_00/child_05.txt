import numpy as np
from scipy.optimize import linprog


def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(lim, 1e-12)))
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    diffs = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    b_ub = dists
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-10), 1e-10 * n


def make_hex(rows_pattern, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        [5, 5, 5, 6, 5], [4, 6, 5, 6, 5], [5, 6, 6, 6, 3], [6, 5, 6, 6, 3],
        [6, 6, 4, 6, 4], [5, 6, 5, 6, 4], [5, 6, 6, 4, 5], [6, 5, 4, 6, 5],
        [5, 5, 6, 6, 4], [6, 4, 6, 5, 5], [4, 6, 6, 5, 5], [6, 6, 5, 4, 5],
        [5, 7, 4, 5, 5], [6, 5, 7, 4, 4], [4, 7, 5, 5, 5], [7, 4, 6, 5, 4]
    ]
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    inits = []
    for p in patterns:
        if sum(p) >= n:
            for r0 in [0.085, 0.09, 0.095, 0.10, 0.105]:
                pts = make_hex(p, r0)
                inits.append(pts)
                # Add perturbed variants

                    p2 = np.clip(pts + rng.uniform(-0.012, 0.012, pts.shape), 0.05, 0.95)
                    inits.append(p2)
    
    # Random initializations
    for _ in range(40):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
    
    # Phase 1: LP evaluation of all initial configurations
    for cfg in inits:
        r_lp, s_lp = solve_lp_radii(cfg)
        if s_lp > best_sum:
            best_sum = s_lp

            best_radii = r_lp.copy()

    # Safety scaling to guarantee strict validity within numerical tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-9:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
    
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
    
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
