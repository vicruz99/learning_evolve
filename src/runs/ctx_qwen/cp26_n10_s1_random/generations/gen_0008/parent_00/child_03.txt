import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute constraint matrix structure for the LP
pair_rows, pair_cols = np.triu_indices(N, k=1)
n_pairs = len(pair_rows)
A_ub_pairs = np.zeros((n_pairs, N))
A_ub_pairs[np.arange(n_pairs), pair_rows] = 1.0
A_ub_pairs[np.arange(n_pairs), pair_cols] = 1.0

A_ub_bounds = np.zeros((4*N, N))
for i in range(N):
    A_ub_bounds[4*i + 0, i] = 1.0  # x_i
    A_ub_bounds[4*i + 1, i] = 1.0  # 1-x_i
    A_ub_bounds[4*i + 2, i] = 1.0  # y_i
    A_ub_bounds[4*i + 3, i] = 1.0  # 1-y_i

A_ub_global = np.vstack([A_ub_pairs, A_ub_bounds])
n_constraints_global = n_pairs + 4*N

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    b_ub = np.zeros(n_constraints_global)
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.hypot(dx, dy)
    b_ub[:n_pairs] = dists[pair_rows, pair_cols]
    
    x = centers[:, 0]
    y = centers[:, 1]
    b_ub[n_pairs:n_pairs+N] = x
    b_ub[n_pairs+N:n_pairs+2*N] = 1.0 - x
    b_ub[n_pairs+2*N:n_pairs+3*N] = y
    b_ub[n_pairs+3*N:] = 1.0 - y

    c = -np.ones(N)
    bounds = [(0, None)] * N
    try:
        res = linprog(c, A_ub=A_ub_global, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(N, 1e-5)

def objective_centroids(centroid_flat):
    """Objective for center optimization: negative max sum of radii."""
    centers = centroid_flat.reshape(N, 2)
    r = solve_lp_radii(centers)
    return -np.sum(r)

def joint_obj(x):
    """Objective for joint optimization: negative sum of radii."""
    return -np.sum(x[2*N:])

def joint_cons(x):
    """Inequality constraints >= 0 for valid joint packing."""
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    c_list = []
    c_list.append(cx - r)
    c_list.append(1.0 - cx - r)
    c_list.append(cy - r)
    c_list.append(1.0 - cy - r)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    idx_i, idx_j = np.triu_indices(N, k=1)
    c_list.append(d2[idx_i, idx_j] - rs[idx_i, idx_j]**2)
    return np.concatenate(c_list)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Generate diverse initial configurations
    inits = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5], [5, 5, 5, 5, 6]
    ]
    for p in patterns:
        pts = []
        y = 0.08
        r0 = 0.085
        for ri, cnt in enumerate(p):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        pts = np.array(pts[:N])
        inits.append(pts)
        for _ in range(4):
            p_pts = pts + rng.uniform(-0.02, 0.02, pts.shape)
            inits.append(np.clip(p_pts, 0.05, 0.95))

    for _ in range(10):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))

    bounds_cent = [(0.02, 0.98)] * (2 * N)

    # Phase 1: Optimize centers using Powell's method with LP objective
    for init in inits:
        try:
            res = minimize(objective_centroids, init.flatten(), method='Powell',
                           bounds=bounds_cent, options={'maxiter': 800, 'ftol': 1e-9})
            if np.isfinite(res.fun):
                curr_centers = res.x.reshape(N, 2)
                curr_radii = solve_lp_radii(curr_centers)
                s = np.sum(curr_radii)
                if s > best_sum:
                    best_sum = s
                    best_centers = curr_centers.copy()
                    best_radii = curr_radii.copy()
        except Exception:
            pass

    if best_centers is None:
        best_centers = inits[0]
        best_radii = solve_lp_radii(best_centers)
        best_sum = np.sum(best_radii)

    # Phase 2: Joint SLSQP refinement
    x0 = np.concatenate([best_centers.flatten(), best_radii])
    bounds_joint = [(0.02, 0.98)] * (2*N) + [(1e-6, 0.5)] * N

    try:
        res_sl = minimize(joint_obj, x0, method='SLSQP', bounds=bounds_joint,
                          constraints={'type': 'ineq', 'fun': joint_cons},
                          options={'maxiter': 5000, 'ftol': 1e-14})
        if np.isfinite(res_sl.fun):
            fc = res_sl.x[:2*N].reshape(N, 2)
            fr = res_sl.x[2*N:]
            # Verify and shrink slightly for safety
            scale = 1.0
            for i in range(N):
                x, y, r = fc[i,0], fc[i,1], fr[i]
                if r > 1e-9:
                    scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
            for i in range(N):
                for j in range(i+1, N):
                    d = np.hypot(fc[i,0]-fc[j,0], fc[i,1]-fc[j,1])
                    rs = fr[i]+fr[j]
                    if rs > 1e-9:
                        scale = min(scale, d/rs)
            fr *= scale * 0.9999999
            s = np.sum(fr)
            if s > best_sum:
                best_sum = s
                best_centers = fc.copy()
                best_radii = fr.copy()
    except Exception:
        pass

    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i,0], best_centers[i,1], best_radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i]+best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
    best_radii *= scale * 0.9999999
    best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
