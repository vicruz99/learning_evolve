import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    bounds = [(0, None)] * n
    A = np.zeros((4*n + n*(n-1)//2, n))
    b = np.zeros(4*n + n*(n-1)//2)
    idx = 0
    for i in range(n):
        x, y = centers[i]
        A[idx, i] = 1.0; b[idx] = x; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - x; idx += 1
        A[idx, i] = 1.0; b[idx] = y; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - y; idx += 1
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            A[idx, i] = 1.0; A[idx, j] = 1.0; b[idx] = d; idx += 1
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-7)

def slack_obj(vars, n):
    """Objective: maximize minimum slack s."""
    return -vars[-1]

def slack_cons(vars, n, radii):
    """Constraints: ensure clearance >= s for all circles and pairs."""
    cx = vars[0:2*n:2]
    cy = vars[1:2*n:2]
    s = vars[-1]
    
    c = np.empty(4*n + n*(n-1)//2)
    c[:n] = cx - radii - s
    c[n:2*n] = 1.0 - cx - radii - s
    c[2*n:3*n] = cy - radii - s
    c[3*n:4*n] = 1.0 - cy - radii - s
    
    idx = 4*n
    for i in range(n):
        for j in range(i+1, n):
            c[idx] = np.hypot(cx[i]-cx[j], cy[i]-cy[j]) - radii[i] - radii[j] - s
            idx += 1
    return c

def joint_obj(vars, n):
    """Objective: maximize sum of radii."""
    return -np.sum(vars[2*n:])

def joint_cons(vars, n):
    """Constraints: valid packing with no overlaps or boundary violations."""
    cx = vars[0:n]
    cy = vars[n:2*n]
    r = vars[2*n:]
    
    c = np.empty(4*n + n*(n-1)//2)
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    
    idx = 4*n
    for i in range(n):
        for j in range(i+1, n):
            c[idx] = np.hypot(cx[i]-cx[j], cy[i]-cy[j]) - r[i] - r[j]
            idx += 1
    return c

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    # Structured hexagonal row distributions
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4],
        [5,7,5,5,4], [6,6,5,5,4], [5,6,4,6,5], [5,4,6,5,6]
    ]
    
    inits = []
    for p in patterns:
        if sum(p) < n: continue
        pts = []
        r0 = 0.095
        y = r0
        for ri, cnt in enumerate(p):
            shift = r0 if ri%2==1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts)>=n: break
                pts.append([x,y])
                x += 2.0*r0
            y += np.sqrt(3)*r0
        while len(pts)<n: pts.append([0.5, 0.5])
        cfg = np.array(pts[:n])
        inits.append(cfg)
        inits.append(np.clip(cfg + rng.uniform(-0.02, 0.02, cfg.shape), 0.05, 0.95))
        
    # Random dense starts
    for _ in range(4):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds_slack = [(0.0, 1.0)]*(2*n) + [(-0.1, 0.2)]
    
    for init_cfg in inits:
        centers = init_cfg.copy()
        for step in range(15):
            # 1. Solve LP for optimal radii given centers
            radii = solve_lp(centers)
            curr_sum = np.sum(radii)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                
            # 2. Optimize centers to maximize minimum slack
            x0 = np.concatenate([centers.flatten(), [0.0]])
            cons_slack = {'type': 'ineq', 'fun': slack_cons, 'args': (n, radii)}
            try:
                res = minimize(slack_obj, x0, args=(n,), method='SLSQP',
                               bounds=bounds_slack, constraints=cons_slack,
                               options={'maxiter': 1000, 'ftol': 1e-12})
                if res.success and res.x[-1] > -1e-5:
                    centers = res.x[:2*n].reshape(n, 2)
                else:
                    centers = np.clip(centers + rng.uniform(-0.005, 0.005, centers.shape), 0.05, 0.95)
            except Exception:
                centers = np.clip(centers + rng.uniform(-0.005, 0.005, centers.shape), 0.05, 0.95)

    # Final joint SLSQP refinement on best configuration
    if best_centers is not None:
        r_start = best_radii * 0.995
        x0_joint = np.concatenate([best_centers.flatten(), r_start])
        bounds_joint = [(0.0, 1.0)]*(2*n) + [(1e-6, 0.5)]*n
        cons_joint = {'type': 'ineq', 'fun': joint_cons, 'args': (n,)}
        
        try:
            res_j = minimize(joint_obj, x0_joint, args=(n,), method='SLSQP',
                             bounds=bounds_joint, constraints=cons_joint,
                             options={'maxiter': 5000, 'ftol': 1e-14})
            if np.isfinite(res_j.fun):
                fc = res_j.x[:2*n].reshape(n, 2)
                fr = np.maximum(res_j.x[2*n:], 1e-9)
                # Quick validity check
                valid = True
                if np.any(fc[:,0] < fr - 1e-9) or np.any(fc[:,0] > 1 - fr + 1e-9): valid = False
                if np.any(fc[:,1] < fr - 1e-9) or np.any(fc[:,1] > 1 - fr + 1e-9): valid = False
                if valid:
                    idx_i, idx_j = np.triu_indices(n, k=1)
                    d2 = (fc[idx_i,0]-fc[idx_j,0])**2 + (fc[idx_i,1]-fc[idx_j,1])**2
                    rs2 = (fr[idx_i]+fr[idx_j])**2
                    if np.any(d2 < rs2 - 1e-9): valid = False
                if valid:
                    s = np.sum(fr)
                    if s > best_sum:
                        best_sum = s
                        best_centers = fc
                        best_radii = fr
        except Exception:
            pass

    # Final LP squeeze & safety scaling
    if best_centers is not None:
        best_radii = solve_lp(best_centers)
        best_sum = np.sum(best_radii)
        
        scale = 1.0
        for i in range(n):
            x,y,r = best_centers[i,0], best_centers[i,1], best_radii[i]
            if r > 1e-9:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-9:
                    scale = min(scale, d/rs)
        best_radii *= scale * 0.9999999
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
