def solve_lp_radii(centers, n):

    m_bound = 4 * n

        x, y = centers[i]
        mx = max(0.0, min(x, 1.0 - x, y, 1.0 - y))
        bounds.append((0.0, max(mx, 1e-9)))
        for lim in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1

        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])

            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1

    except Exception:

    return None, 0.0

def get_hex_centers(n, row_counts, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):

        x = r0 + shift

            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def obj_joint(x, n):
    """Objective function for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def cons_joint(x, n):

    cy = x[n:2*n]
    r = x[2*n:]

        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])

    cy_m = cy[:, None] - cy[None, :]

    rs2 = r_m**2
    
    idx = np.triu_indices(n, k=1)
    c_pairwise = d2[idx] - rs2[idx]
    
    return np.concatenate([c_boundary, c_pairwise])

def neg_clearance(x, n):
    """Objective for Nelder-Mead: maximize minimum clearance."""
    c = x.reshape(n, 2)
    d_w = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                     np.minimum(c[:, 1], 1.0 - c[:, 1]))
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    return -min(np.min(d_w), np.min(dist) / 2.0)
