from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    A_ub = np.zeros((m_bound + m_pair, n))
    b_ub = np.zeros(m_bound + m_pair)
    bounds = []
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        mx = max(0.0, min(x, 1.0 - x, y, 1.0 - y))
        bounds.append((0.0, max(mx, 1e-9)))
        for lim in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def obj_slsqp(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_slsqp(v, n):
    """Inequality constraints >= 0 for valid packing."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    r2 = r_sum**2
    
    idx = np.triu_indices(n, k=1)
    c.append(d2[idx] - r2[idx])
    
    return np.concatenate(c)

def relax_packing(centers, n, steps=350, dt=0.007, k=700.0):
    """Force-directed relaxation to tightly pack circles."""
    radii = np.full(n, 0.035)
    for _ in range(steps):
        radii *= 1.00025
        forces = np.zeros_like(centers)
        
        # Boundary repulsion
        for d in range(2):
            v1 = np.maximum(0, radii - centers[:, d])
            v2 = np.maximum(0, centers[:, d] + radii - 1.0)
            forces[:, d] += (v2 - v1) * k
            
        # Pairwise repulsion (vectorized)
        diff = centers[:, None, :] - centers[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist, 1e9)
        
        min_d = radii[:, None] + radii[None, :]
        overlap = np.maximum(0, min_d - dist)
        safe_dist = np.maximum(dist, 1e-7)
        dir_mat = diff / safe_dist[:, :, None]
        force_mag = overlap * k / safe_dist
        
        forces += np.sum(dir_mat * force_mag[:, :, None], axis=1)
        
        centers += forces * dt
        centers = np.clip(centers, 0.001, 0.999)
    return centers, radii
