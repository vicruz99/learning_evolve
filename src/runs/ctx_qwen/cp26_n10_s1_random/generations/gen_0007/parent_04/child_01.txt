import numpy as np
from scipy.optimize import minimize, linprog

def build_A_ub(n):
    """Precompute the constraint matrix structure for the LP."""
    m_pair = n * (n - 1) // 2
    m_bound = 4 * n
    A = np.zeros((m_bound + m_pair, n))
    idx = 0
    for i in range(n):
        for _ in range(4):
            A[idx, i] = 1.0
            idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
    return A

def solve_lp_radii(centers, n, A_ub):
    """Solves the LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    b_ub = np.zeros(m_bound + m_pair)
    bounds = []
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lim = max(1e-9, min(x, 1.0 - x, y, 1.0 - y))
        bounds.append((0.0, lim))
        b_ub[idx] = x; idx += 1
        b_ub[idx] = 1.0 - x; idx += 1
        b_ub[idx] = y; idx += 1
        b_ub[idx] = 1.0 - y; idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def neg_clearance(x, n):
    """Objective for Nelder-Mead: maximize minimum clearance."""
    c = x.reshape(n, 2)
    d_w = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                     np.minimum(c[:, 1], 1.0 - c[:, 1]))
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    return -min(np.min(d_w), np.min(dist) / 2.0)

def hex_init(n, row_counts, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def objective_slsqp(p, n):
    return -np.sum(p[2*n:])

def constraints_slsqp(p, n):
    cx, cy, r = p[:n], p[n:2*n], p[2*n:]
    c = []
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, 1e9)
    idx = np.triu_indices(n, k=1)
    c.extend(dist[idx] - (r[:, None] + r[None, :])[idx])
    return np.array(c)

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    A_ub = build_A_ub(n)
    
    row_configs = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [6, 5, 5, 6, 4],
        [5, 6, 6, 4, 5], [5, 5, 5, 5, 6], [7, 6, 6, 7],
        [6, 7, 6, 7], [5, 6, 5, 6, 5, 1], [4, 5, 6, 6, 5],
        [5, 7, 5, 6, 3], [7, 5, 6, 5, 3], [6, 6, 6, 5, 3]
    ]
    
    inits = []
    for rc in row_configs:
        if sum(rc) < n: 
            continue
        base = hex_init(n, rc)
        mn = base.min(axis=0)
        mx = base.max(axis=0)
        span = mx - mn
        if span[0] > 0 and span[1] > 0:
            scale = 0.85 / max(span[0], span[1])
            base = (base - mn) * scale + 0.5 * (1.0 - scale)
        inits.append(base)
        
        for _ in range(4):
            pert = base + rng.uniform(-0.015, 0.015, (n, 2))
            inits.append(np.clip(pert, 0.05, 0.95))
            
    for _ in range(10):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Nelder-Mead + LP
    for cfg in inits:
        try:
            res = minimize(neg_clearance, cfg.flatten(), args=(n,), 
                          method='Nelder-Mead',
                          options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10})
            c_opt = res.x.reshape(n, 2)
            c_opt = np.clip(c_opt, 0.002, 0.998)
            
            r_lp = solve_lp_radii(c_opt, n, A_ub)
            if r_lp is not None:
                s = np.sum(r_lp)
                if s > best_sum:
                    best_sum = s
                    best_centers = c_opt.copy()
                    best_radii = r_lp.copy()
        except Exception:
            continue
            
    # Phase 2: Hill Climbing with LP
    if best_centers is not None:
        centers = best_centers.copy()
        radii = best_radii.copy()
        curr_sum = best_sum
        
        step = 0.02
        for _ in range(20000):
            i = rng.integers(n)
            old_pos = centers[i].copy()
            centers[i] += rng.uniform(-step, step, 2)
            centers[i] = np.clip(centers[i], 1e-5, 1.0 - 1e-5)
            
            r_new = solve_lp_radii(centers, n, A_ub)
            if r_new is not None:
                s_new = np.sum(r_new)
                if s_new > curr_sum + 1e-9:
                    curr_sum = s_new
                    radii = r_new.copy()
                    step *= 1.0003
                else:
                    centers[i] = old_pos
                    step *= 0.9997
            else:
                centers[i] = old_pos
                step *= 0.9997
                
            if step < 1e-7:
                step = 1e-7
                
        best_centers = centers.copy()
        best_radii = radii.copy()
        best_sum = curr_sum

    # Phase 3: SLSQP Joint Refinement
    if best_centers is not None:
        x0 = np.zeros(3 * n)
        x0[:n] = best_centers[:, 0]
        x0[n:2*n] = best_centers[:, 1]
        x0[2*n:] = best_radii * 0.98
        
        bounds_opt = [(0.001, 0.999)] * (2 * n) + [(1e-7, 0.5)] * n
        
        try:
            res = minimize(objective_slsqp, x0, args=(n,), method='SLSQP', bounds=bounds_opt,
                          constraints={'type': 'ineq', 'fun': constraints_slsqp, 'args': (n,)},
                          options={'maxiter': 6000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_ref = np.column_stack((res.x[:n], res.x[n:2*n]))
                r_ref = solve_lp_radii(c_ref, n, A_ub)
                if r_ref is not None:
                    s = np.sum(r_ref)
                    if s > best_sum:
                        best_sum = s
                        best_centers = c_ref.copy()
                        best_radii = r_ref.copy()
        except Exception:
            pass

    # Safety Scaling
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
