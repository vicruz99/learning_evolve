import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(lim, 1e-9)))
        
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = d
            k += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 1e-6)

def max_clearance_obj(x, n):
    """Objective for Nelder-Mead: maximize minimum clearance (equal radii proxy)."""
    c = x.reshape(n, 2)
    d_w = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                     np.minimum(c[:, 1], 1.0 - c[:, 1]))
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    min_dist = np.min(dist) / 2.0
    return -min(min(np.min(d_w), min_dist))

def hex_init(n, row_counts, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def obj_slsqp(p, n):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(p[2*n:])

def cons_slsqp(p, n):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    cx = p[:n]
    cy = p[n:2*n]
    r = p[2*n:]
    c = np.empty(4 * n + n * (n - 1) // 2)
    
    # Boundary constraints
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    
    # Pairwise non-overlap: d^2 >= (r_i + r_j)^2
    triu_idx = np.triu_indices(n, k=1)
    cx_m = cx[:, np.newaxis] - cx[np.newaxis, :]
    cy_m = cy[:, np.newaxis] - cy[np.newaxis, :]
    d2 = cx_m**2 + cy_m**2
    
    r_m = r[:, np.newaxis] + r[np.newaxis, :]
    rs2 = r_m**2
    
    c[4*n:] = d2[triu_idx] - rs2[triu_idx]
    return c
