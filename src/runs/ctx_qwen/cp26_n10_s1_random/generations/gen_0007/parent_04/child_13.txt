import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_IDX = np.triu_indices(N, k=1)

def solve_lp_fixed_centers(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    A_ub = np.zeros((m_bound + m_pair, n))
    b_ub = np.zeros(m_bound + m_pair)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i, j in zip(PAIR_IDX[0], PAIR_IDX[1]):
        d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        b_ub[idx] = d
        idx += 1
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-9)

def lp_neg_sum(x_flat):
    """Objective for center optimization: negative sum of LP-optimized radii."""
    centers = np.clip(x_flat.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    radii = solve_lp_fixed_centers(centers)
    return -np.sum(radii)

def obj_slsqp(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def get_constraints_joint(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = []
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    for i, j in zip(PAIR_IDX[0], PAIR_IDX[1]):
        dist = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
        c.append(dist - (r[i] + r[j]))
    return np.array(c)

def repulse(centers, n, steps=500):
    """Quick force-directed spread to resolve initial overlaps."""
    c = centers.copy()
    for _ in range(steps):
        f = np.zeros_like(c)
        for i in range(n):
            for j in range(i + 1, n):
                diff = c[i] - c[j]
                d = np.linalg.norm(diff) + 1e-6
                if d < 0.2:
                    force = diff / (d ** 2)
                    f[i] += force
                    f[j] -= force
        c += f * 0.0005
        c = np.clip(c, 0.02, 0.98)
    return c

def run_packing():

    rng = np.random.default_rng(42)
    
    inits = []

    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],

        [5, 6, 6, 4, 5], [6, 7, 6, 7], [5, 5, 5, 5, 6],
        [5, 6, 5, 5, 5], [6, 5, 5, 6, 4]
    ]
    
    for pattern in row_patterns:

    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
