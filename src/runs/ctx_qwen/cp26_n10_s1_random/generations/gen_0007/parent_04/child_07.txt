import numpy as np
from scipy.optimize import minimize, linprog

def get_triu_indices(n):
    return np.triu_indices(n, k=1)

def solve_lp_radii(centers, n, idx_i, idx_j):
    """Solves the LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    m_bound = 4 * n
    m_pair = len(idx_i)
    
    A_ub = np.zeros((m_bound + m_pair, n))
    b_ub = np.zeros(m_bound + m_pair)
    bounds = [(0.0, 0.5)] * n
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lims = [x, 1.0 - x, y, 1.0 - y]
        max_lim = max(lims)
        bounds[i] = (0.0, max(0.0, max_lim + 1e-9))
        
        for lim in lims:
            A_ub[idx, i] = 1.0
            b_ub[idx] = max(lim, 1e-12)
            idx += 1
            
    for k in range(m_pair):
        i, j = idx_i[k], idx_j[k]
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b_ub[idx] = np.hypot(dx, dy)
        idx += 1
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def obj_centers(vars_flat, n):
    """Objective for center optimization: minimize negative clearance t."""
    return -vars_flat[2 * n]

def cons_centers(vars_flat, n, radii, idx_i, idx_j):
    """Constraints for center optimization: clearances >= t."""
    xs = vars_flat[:n]
    ys = vars_flat[n:2 * n]
    t = vars_flat[2 * n]
    
    c = np.empty(4 * n + len(idx_i))
    c[:n] = xs - radii - t
    c[n:2 * n] = 1.0 - xs - radii - t
    c[2 * n:3 * n] = ys - radii - t
    c[3 * n:4 * n] = 1.0 - ys - radii - t
    
    xs_m = xs[:, None] - xs[None, :]
    ys_m = ys[:, None] - ys[None, :]
    dists = np.sqrt(xs_m**2 + ys_m**2)
    np.fill_diagonal(dists, 1e9)
    r_sum = radii[:, None] + radii[None, :]
    
    base_idx = 4 * n
    c[base_idx:] = dists[idx_i, idx_j] - r_sum[idx_i, idx_j] - t
    return c

def obj_joint(vars_flat, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(vars_flat[2 * n:])

def cons_joint(vars_flat, n, idx_i, idx_j):
    """Constraints for joint optimization: valid packing."""
    xs = vars_flat[:n]
    ys = vars_flat[n:2 * n]
    rs = vars_flat[2 * n:]
    
    c = np.empty(4 * n + len(idx_i))
    c[:n] = xs - rs
    c[n:2 * n] = 1.0 - xs - rs
    c[2 * n:3 * n] = ys - rs
    c[3 * n:4 * n] = 1.0 - ys - rs
    
    xs_m = xs[:, None] - xs[None, :]
    ys_m = ys[:, None] - ys[None, :]
    dists = np.sqrt(xs_m**2 + ys_m**2)
    np.fill_diagonal(dists, 1e9)
    r_sum = rs[:, None] + rs[None, :]
    
    base_idx = 4 * n
    c[base_idx:] = dists[idx_i, idx_j] - r_sum[idx_i, idx_j]
    return c

def generate_hex_init(n, row_counts, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
