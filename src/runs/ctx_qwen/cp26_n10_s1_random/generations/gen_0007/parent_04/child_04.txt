from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Boundary limits: 0 <= r_i <= min(x, 1-x, y, 1-y)
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    idx_i, idx_j = np.triu_indices(n, k=1)
    diffs = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    
    m = len(dists)
    A_ub = np.zeros((m, n))
    A_ub[:, idx_i] = 1.0
    A_ub[:, idx_j] = 1.0
    b_ub = dists
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def obj_slsqp(x, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def cons_slsqp(x, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    
    c = np.empty(4*n + len(idx_i))
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    dists = np.sqrt(cx_m**2 + cy_m**2)
    r_m = r[:, None] + r[None, :]
    c[4*n:] = dists[idx_i, idx_j] - r_m[idx_i, idx_j]
    return c
