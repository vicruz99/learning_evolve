import numpy as np
from scipy.optimize import minimize, linprog

def setup_lp_matrix(n):
    """Precompute the constant structure of the LP inequality matrix."""
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    A_ub = np.zeros((m_bound + m_pair, n))
    for i in range(n):
        A_ub[0*n + i, i] = 1.0
        A_ub[1*n + i, i] = 1.0
        A_ub[2*n + i, i] = 1.0
        A_ub[3*n + i, i] = 1.0
    idx = m_bound
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
    return A_ub

def solve_lp_radii(centers, A_ub, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    b_ub = np.zeros(m_bound + m_pair)
    bounds = []
    
    # Vectorized pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0-x, y, 1.0-y)
        bounds.append((0.0, max(mx, 1e-9)))
        b_ub[0*n + i] = x
        b_ub[1*n + i] = 1.0 - x
        b_ub[2*n + i] = y
        b_ub[3*n + i] = 1.0 - y
        
    idx = m_bound
    for i in range(n):
        for j in range(i+1, n):
            b_ub[idx] = dists[i, j]
            idx += 1
            
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return np.maximum(res.x, 0.0)
    return None

def lp_sum_objective(x, A_ub, n):
    """Objective for center optimization: maximize sum of radii via LP."""
    c = x.reshape(n, 2)
    r = solve_lp_radii(c, A_ub, n)
    return -np.sum(r) if r is not None else 1e6

            x += 2.0 * sp

            grid.append([0.5, 0.5])
        starts.append(np.array(grid[:n]))

def obj_joint(v, n):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(v[2*n:])

def cons_joint(v, n):

    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)

    c.extend(dists[idx] - r_m[idx])
    return np.array(c)
