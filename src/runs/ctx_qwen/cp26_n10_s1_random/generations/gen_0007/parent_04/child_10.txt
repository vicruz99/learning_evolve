import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def solve_lp(centers):
    n = centers.shape[0]
    m = n * (n - 1) // 2 + 4 * n
    A = np.zeros((m, n))
    b = np.zeros(m)
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lims = [x, 1.0 - x, y, 1.0 - y]
        for l in lims:
            A[idx, i] = 1.0
            b[idx] = l
            idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def force_directed(centers, r_init=0.05, steps=3000):
    n = centers.shape[0]
    r = np.full(n, r_init)
    dt = 0.005
    grow = 0.0005
    for step in range(steps):
        r += grow
        forces = np.zeros((n, 2))
        for d in range(2):
            for i in range(n):
                if centers[i, d] < r[i]:
                    forces[i, d] += (r[i] - centers[i, d]) * 200.0
                if centers[i, d] > 1.0 - r[i]:
                    forces[i, d] -= (centers[i, d] - (1.0 - r[i])) * 200.0
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                min_d = r[i] + r[j]
                if dist < min_d and dist > 1e-8:
                    overlap = min_d - dist
                    dir_ = diff / dist
                    f = overlap * 300.0
                    forces[i] += dir_ * f
                    forces[j] -= dir_ * f
        centers += forces * dt
        centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
        if step > 1500:
            grow *= 0.9995
            dt *= 0.9995
    return centers, r

def obj_slsqp(x):
    n = len(x) // 3
    return -np.sum(x[2 * n:])

def cons_slsqp(x):
    n = len(x) // 3
    cx = x[:n]
    cy = x[n:2 * n]
    r = x[2 * n:]
    c = []
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
            c.append(d - (r[i] + r[j]))
    return np.array(c)

def run_packing():
    n = N_CIRCLES
    rng = np.random.default_rng(123)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = []
    for scale in [0.85, 0.92, 1.0, 1.08, 1.15]:
        pts = []
        y = 0.1
        row = 0
        while len(pts) < n:
            shift = 0.1 if row % 2 == 1 else 0.0
            x = 0.1 + shift
            while x <= 0.9 and len(pts) < n:
                cx = x * scale + 0.5 * (1.0 - scale)
                cy = y * scale + 0.5 * (1.0 - scale)
                pts.append([cx, cy])
                x += 0.2
            y += 0.173205
            row += 1
        inits.append(np.array(pts[:n]))
        
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    bounds_slsqp = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    cons = {'type': 'ineq', 'fun': cons_slsqp}
    
    for cfg in inits:
        for _ in range(3):
            pert_cfg = cfg.copy()
            pert_cfg += rng.uniform(-0.02, 0.02, (n, 2))
            pert_cfg = np.clip(pert_cfg, 0.05, 0.95)
            
            c_fd, r_fd = force_directed(pert_cfg, r_init=0.04, steps=3000)
            
            x0 = np.zeros(3 * n)
            x0[:n] = c_fd[:, 0]
            x0[n:2 * n] = c_fd[:, 1]
            x0[2 * n:] = r_fd * 0.95
            
            try:
                res = minimize(obj_slsqp, x0, method='SLSQP', bounds=bounds_slsqp,
                              constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14})
                if np.isfinite(res.fun):
                    c_opt = np.column_stack((res.x[:n], res.x[n:2 * n]))
                    r_opt, s_opt = solve_lp(c_opt)
                    if r_opt is not None and s_opt > best_sum:
                        best_sum = s_opt
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
            except Exception:
                pass
                
    if best_centers is not None:
        curr_c = best_centers.copy()
        curr_r, curr_s = solve_lp(curr_c)
        best_sum = curr_s
        best_radii = curr_r
        best_centers = curr_c
        
        temp = 0.020
        for step in range(4000):
            i = rng.integers(n)
            old = curr_c[i].copy()
            move = rng.normal(0, temp, 2)
            curr_c[i] = np.clip(old + move, 0.005, 0.995)
            
            new_r, new_s = solve_lp(curr_c)
            if new_r is not None:
                if new_s > curr_s:
                    curr_s = new_s
                    best_sum = new_s
                    best_centers = curr_c.copy()
                    best_radii = new_r.copy()
                    temp *= 1.002
                else:
                    curr_c[i] = old
                temp = max(0.001, temp * 0.997)
            else:
                curr_c[i] = old
                
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
