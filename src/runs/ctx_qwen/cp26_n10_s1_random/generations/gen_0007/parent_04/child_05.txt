import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    m_pair = n * (n - 1) // 2
    A_ub = np.zeros((m_pair, n))
    b_ub = np.zeros(m_pair)
    bounds = []
    
    # Boundary limits
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(lim, 1e-9)))
        
    # Pairwise constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            r = np.maximum(res.x, 1e-9)
            return r, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def jitter_search(centers, n, rng, max_iter=2500):
    """Local search on centers optimizing the LP sum of radii objective."""
    best_centers = centers.copy()
    r_lp, best_sum = solve_lp(best_centers, n)
    best_radii = r_lp
    
    step = 0.06
    for _ in range(max_iter):
        i = rng.integers(n)
        old_pos = best_centers[i].copy()
        move = rng.uniform(-step, step, 2)
        new_pos = np.clip(best_centers[i] + move, 1e-4, 1.0 - 1e-4)
        
        best_centers[i] = new_pos
        r_new, s_new = solve_lp(best_centers, n)
        
        if s_new > best_sum + 1e-8:
            best_sum = s_new
            best_radii = r_new
            step = min(step * 1.015, 0.12)
        else:
            best_centers[i] = old_pos
            step *= 0.997
            
    return best_centers, best_radii, best_sum

def joint_obj(vars, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def joint_cons(vars, n):
    """Inequality constraints >= 0 for valid packing."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    c = np.empty(4 * n + n * (n - 1) // 2)
    c[:n] = x - r
    c[n:2*n] = 1.0 - x - r
    c[2*n:3*n] = y - r
    c[3*n:4*n] = 1.0 - y - r
    
    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            c[idx] = dx*dx + dy*dy - (r[i] + r[j])**2
            idx += 1
    return c

def hex_init(n, row_counts, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
