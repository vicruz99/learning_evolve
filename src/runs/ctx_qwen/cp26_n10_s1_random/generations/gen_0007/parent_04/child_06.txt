from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    m_b = 4 * n
    m_p = n * (n - 1) // 2
    A = np.zeros((m_b + m_p, n))
    b = np.zeros(m_b + m_p)
    bounds = []
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0-x, y, 1.0-y)
        bounds.append((0.0, max(mx, 1e-9)))
        for lim in (x, 1.0-x, y, 1.0-y):
            A[k, i] = 1.0
            b[k] = lim
            k += 1
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            A[k, i] = 1.0
            A[k, j] = 1.0
            b[k] = d
            k += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_slsqp(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_slsqp(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing (polynomial form)."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.empty(4*n + n*(n-1)//2)
    
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    
    cx_diff = cx[triu_i] - cx[triu_j]
    cy_diff = cy[triu_i] - cy[triu_j]
    r_sum = r[triu_i] + r[triu_j]
    c[4*n:] = cx_diff**2 + cy_diff**2 - r_sum**2
    return c

def get_hex_init(n, rows, r0):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for i, cnt in enumerate(rows):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
