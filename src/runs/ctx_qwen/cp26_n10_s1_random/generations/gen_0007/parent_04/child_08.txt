import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_fast(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    A_ub = np.zeros((m_bound + m_pair, n))
    b_ub = np.zeros(m_bound + m_pair)
    bounds = []
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mx, 1e-9)))
        for lim in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'disp': False})
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def force_expand(centers, radii, n, steps=1500):
    """Force-directed relaxation to find a dense feasible packing."""
    dt = 0.004
    k_rep = 150.0
    k_spr = 50.0
    
    for _ in range(steps):
        forces = np.zeros_like(centers)
        radii += 1.5e-5
        
        # Boundary repulsion
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0: forces[i, 0] += k_spr * (r - x)
            if x + r > 1: forces[i, 0] -= k_spr * (x + r - 1)
            if y - r < 0: forces[i, 1] += k_spr * (r - y)
            if y + r > 1: forces[i, 1] -= k_spr * (y + r - 1)
            
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)
                min_d = radii[i] + radii[j]
                if d < min_d and d > 1e-6:
                    f = k_rep * (min_d - d) / d
                    fx, fy = f * dx, f * dy
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
                    
        centers += forces * dt
        centers = np.clip(centers, 0.001, 0.999)
        
    return centers, radii

def hill_climb_centers(centers, n, rng, iters=2000):
    """Local search on centers using LP objective."""
    curr_centers = centers.copy()
    r_lp, curr_sum = solve_lp_fast(curr_centers)
    if r_lp is None: return curr_centers, np.zeros(n)
    
    best_centers = curr_centers.copy()
    best_radii = r_lp.copy()
    best_sum = curr_sum
    
    step = 0.02
    for it in range(iters):
        i = rng.integers(n)
        old_pos = curr_centers[i].copy()
        
        # Perturb
        curr_centers[i] += rng.uniform(-step, step, 2)
        curr_centers[i] = np.clip(curr_centers[i], 0.002, 0.998)
        
        r_try, s_try = solve_lp_fast(curr_centers)
        if r_try is not None and s_try > best_sum + 1e-8:
            best_sum = s_try
            best_centers = curr_centers.copy()
            best_radii = r_try.copy()
            step *= 0.9995
        else:
            curr_centers[i] = old_pos
            if rng.random() < 0.1:
                step *= 0.95
                
    return best_centers, best_radii

def hex_init(n, row_counts, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def slesqp_joint(x_flat, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x_flat[2*n:])

def slesqp_constraints(x_flat, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = x_flat[:n]
    cy = x_flat[n:2*n]
    r = x_flat[2*n:]
    
    c = []
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, 1e9)
    
    r_sum = r[:, None] + r[None, :]
    idx = np.triu_indices(n, k=1)
    c.extend(dists[idx] - r_sum[idx])
    
    return np.array(c)
