import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    A_ub = np.zeros((m_bound + m_pair, n))
    b_ub = np.zeros(m_bound + m_pair)
    bounds = [(1e-12, None)] * n
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[idx, i] = 1.0
            b_ub[idx] = val
            idx += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def lp_objective(centers_flat, n):
    """Objective for Nelder-Mead: minimize negative sum of LP radii."""
    centers = centers_flat.reshape(n, 2)
    r = solve_lp_radii(centers, n)
    if r is None:
        return 0.0
    return -np.sum(r)

    ]
    
    for pat in row_patterns:

            continue
        pts = []

                    break
                pts.append([x, y])
                x += 0.18
            y += 0.155
            row_idx += 1
        while len(pts) < n:
            pts.append([0.5, 0.5])
            
        cfg = np.array(pts[:n])
        configs.append(cfg)
        
        # Scaled variants to explore different densities
        for scale in [0.8, 1.0, 1.2]:
            mn = cfg.min(axis=0)
            mx = cfg.max(axis=0)
            span = mx - mn
            if span.max() > 1e-6:
                scaled = (cfg - mn) / span * (0.7 * scale) + 0.5 * (1.0 - 0.7 * scale)
                configs.append(np.clip(scaled, 0.05, 0.95))
                
        # Perturbed variants

            pert = cfg + rng.uniform(-0.02, 0.02, (n, 2))
            configs.append(np.clip(pert, 0.05, 0.95))
            
    # Random dense configurations
    for _ in range(8):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return configs
