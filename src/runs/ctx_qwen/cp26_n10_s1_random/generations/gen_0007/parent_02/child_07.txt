import numpy as np
from scipy.optimize import minimize, linprog

def _get_pair_indices(n):
    """Returns indices for the upper triangular part of the distance matrix."""
    return np.triu_indices(n, k=1)

def _solve_lp(centers, n, pi, pj, A_ub_struct):
    """Solves the LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    b_ub = dists[pi, pj]
    
    try:
        res = linprog(c_obj, A_ub=A_ub_struct, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    # Fallback iterative assignment if LP fails
    radii = np.full(n, 1e-6)
    for i in range(n):
        d_min = limits[i]
        for j in range(n):
            if i != j:
                d_min = min(d_min, dists[i, j] - radii[j])
        radii[i] = max(d_min, 1e-6)
    return radii, np.sum(radii)

def _objective(vars_arr, n):
    """Objective: minimize negative sum of radii => maximize sum of radii."""
    return -np.sum(vars_arr[2*n:])

def _constraints(vars_arr, n, pi, pj):
    """Computes inequality constraints >= 0 for valid packing using squared distances."""
    c = vars_arr[:2*n].reshape(n, 2)
    r = vars_arr[2*n:]
    
    # Boundary constraints
    cons = np.concatenate([
        c[:, 0] - r, 1.0 - c[:, 0] - r,
        c[:, 1] - r, 1.0 - c[:, 1] - r
    ])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    cons = np.concatenate([cons, dist_sq[pi, pj] - r_sum[pi, pj]**2])
    return cons

def _make_hex(rows, r0, n):
    """Generates a hexagonal lattice initialization with specified row counts."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) < n:
                pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(pts[:n])
