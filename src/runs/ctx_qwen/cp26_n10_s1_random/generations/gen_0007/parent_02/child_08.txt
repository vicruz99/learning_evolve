import numpy as np
from scipy.optimize import minimize, linprog

def _build_A_ub(n):
    """Precomputes the constant inequality matrix for the radii LP."""
    m = n * (n - 1) // 2
    A = np.zeros((m, n))
    idx = np.triu_indices(n, k=1)
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A[k, i] = 1.0
        A[k, j] = 1.0
    return A, idx

def _solve_lp(centers, A_ub, idx, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    b_ub = dists[idx]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    # Fallback greedy assignment if LP fails
    radii = np.zeros(n)
    for i in range(n):
        r = lims[i]
        for j in range(n):
            if i != j:
                r = min(r, dists[i, j] - radii[j])
        radii[i] = max(r, 1e-6)
    return radii, np.sum(radii)

def _lp_objective(c_flat, A_ub, idx, n):
    """Objective for center optimization: negative LP sum of radii."""
    c = c_flat.reshape(n, 2)
    _, s = _solve_lp(c, A_ub, idx, n)
    return -s

def _joint_objective(v, n):
    """Objective for joint optimization: negative sum of radii."""
    return -np.sum(v[2*n:])

def _joint_constraints(v, n, idx):
    """Inequality constraints >= 0 for joint optimization."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    
    c = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    rs2 = rs**2
    c = np.concatenate([c, d2[idx] - rs2[idx]])
    return c

def _make_hex(rows, r0, n):
    """Generates hexagonal lattice initialization."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) < n:
                pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    A_ub, idx = _build_A_ub(n)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)
    row_pats = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 4, 6, 5, 6], [6, 5, 5, 6, 4], [5, 5, 5, 5, 6]
    ]
    
    configs = []
    for pat in row_pats:
        if sum(pat) < n: continue
        for r0 in [0.090, 0.095, 0.100, 0.105]:
            cfg = _make_hex(pat, r0, n)
            cfg = np.clip(cfg, 0.05, 0.95)
            configs.append(cfg)
            
        # Rotated variants to break symmetry
        cfg_base = _make_hex(pat, 0.10, n)
        for angle in [0.05, -0.05, 0.15, -0.15]:
            cos_t, sin_t = np.cos(angle), np.sin(angle)
            rot = np.array([[cos_t, -sin_t], [sin_t, cos_t]])
            cfg_rot = cfg_base @ rot
            cfg_rot = (cfg_rot - cfg_rot.min(axis=0)) / (cfg_rot.max(axis=0) - cfg_rot.min(axis=0))
            cfg_rot = cfg_rot * 0.88 + 0.06
            configs.append(np.clip(cfg_rot, 0.05, 0.95))
            
    # Random dense starts
    for _ in range(10):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Phase 1 & 2: Adaptive Hill-Climbing on Centers evaluated by LP
    for cfg in configs:
        curr_centers = cfg.copy()
        curr_radii, curr_sum = _solve_lp(curr_centers, A_ub, idx, n)
        
        step = 0.025
        for it in range(1500):
            i = rng.integers(n)
            old_pos = curr_centers[i].copy()
            
            # Perturb

                curr_sum = new_sum

                step = min(step * 1.005, 0.05)
            else:
                curr_centers[i] = old_pos
                step *= 0.997

                c_opt = np.column_stack((res.x[:n], res.x[n:2*n]))
                r_opt, s_opt = _solve_lp(c_opt, A_ub, idx, n)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt

            pass
            
    # Fallback safety net
    if best_centers is None:
        best_centers = np.clip(_make_hex([5, 6, 5, 6, 4], 0.095, n), 0.1, 0.9)

    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.99999999
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
