import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, pair_indices):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    m = len(pair_indices[0])
    A_ub = np.zeros((m, n))
    b_ub = dists[pair_indices]
    A_ub[np.arange(m), pair_indices[0]] = 1.0
    A_ub[np.arange(m), pair_indices[1]] = 1.0
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-6

def get_objective(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def get_constraints(vars_arr, n, pair_indices):
    """Computes inequality constraints >= 0 for valid packing."""
    c = vars_arr[:2*n].reshape(n, 2)
    r = vars_arr[2*n:]
    
    b_cons = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    dists = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    return np.concatenate([b_cons, dists[pair_indices] - r_sum[pair_indices]])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    pair_indices = np.triu_indices(n, k=1)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    configs = []

    for pat in row_pats:

                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            while len(pts) < n:
                pts.append([0.5, 0.5])
            cfg = np.array(pts[:n])
            configs.append(cfg)
            # Perturbed hex to break symmetry

            configs.append(np.clip(cfg_p, 0.05, 0.95))

    g = np.linspace(0.12, 0.88, 5)
    grid = np.array([[x, y] for y in g for x in g])

    configs.append(grid[:n])

        configs.append(rng.uniform(0.15, 0.85, (n, 2)))

    for cfg in configs:

                rs = r[i] + r[j]
                if rs > 1e-12:
                    scale = min(scale, d/rs)
        best_radii *= scale * 0.9999995
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
