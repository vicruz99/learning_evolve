Key improvements:

2. **LP-Driven Center Optimization**: Instead of optimizing centers and radii jointly with SLSQP throughout, we use a two-phase approach:

   - Phase 2: Nelder-Mead optimization on centers *only*, where the objective is the negative sum of radii obtained by solving the exact LP for fixed centers. This effectively uses the LP as a black-box evaluator that naturally handles radius inequalities optimally for any given layout.
3. **Multiple Restarts**: Run Nelder-Mead from several slightly perturbed versions of the best centers found in Phase 1 to escape local optima in the center-space landscape.
4. **Final Polish**: Run one last SLSQP joint refinement from the best Nelder-Mead result to tighten constraints and maximize radii simultaneously.
5. **Strict Numerical Safety**: Apply a precise scaling factor at the end to guarantee all constraints are satisfied within the `1e-12` tolerance of the validator, while losing minimal sum.

import numpy as np
from scipy.optimize import minimize, linprog

def compute_lp_radii(centers, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    dx = centers[:, np.newaxis, 0] - centers[np.newaxis, :, 0]
    dy = centers[:, np.newaxis, 1] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def lp_objective(centers_flat, n, A_ub, idx_i, idx_j):
    """Objective for center optimization: negative max sum of radii."""
    centers = centers_flat.reshape(n, 2)
    centers = np.clip(centers, 0.05, 0.95)
    _, s = compute_lp_radii(centers, A_ub, idx_i, idx_j)
    return -s

def joint_constraints(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing."""
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    cons = np.concatenate([
        c[:,0] - r, 1.0 - c[:,0] - r,
        c[:,1] - r, 1.0 - c[:,1] - r
    ])
    dx = c[:,0:1] - c[:,0:1].T
    dy = c[:,1:2] - c[:,1:2].T
    dist = np.sqrt(dx**2 + dy**2)
    rs = r[:, None] + r[None, :]
    cons = np.concatenate([cons, dist[triu_i, triu_j] - rs[triu_i, triu_j]])
    return cons

def joint_objective(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    triu_i, triu_j = np.triu_indices(n, k=1)
    
    # Precompute constant LP constraint matrix
    A_ub = np.zeros((len(triu_i), n))
    A_ub[np.arange(len(triu_i)), triu_i] = 1.0
    A_ub[np.arange(len(triu_i)), triu_j] = 1.0
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # 1. Generate diverse hexagonal initial configurations
    configs = []
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], [6,6,5,5,4]]
    for pat in patterns:
        if sum(pat) < n: continue
        for r0 in [0.09, 0.095, 0.10]:
            pts = []
            y = r0
            for ri, cnt in enumerate(pat):
                shift = r0 if ri % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) < n: pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            configs.append(np.array(pts[:n]))
            
    np.random.seed(42)
    for cfg in configs[:4]:
        for _ in range(3):
            p = cfg + np.random.uniform(-0.02, 0.02, cfg.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    bounds_vars = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    
    # Phase 1: Joint SLSQP optimization from diverse starts
    for cfg in configs:
        x0 = np.concatenate([cfg.flatten(), np.full(n, 0.08)])
        try:
            res = minimize(joint_objective, x0, args=(n,),
                          method='SLSQP', bounds=bounds_vars,
                          constraints={'type': 'ineq', 'fun': joint_constraints, 'args': (n, triu_i, triu_j)},
                          options={'maxiter': 3000, 'ftol': 1e-12})
            if np.isfinite(res.fun):
                c_opt = res.x[:2*n].reshape(n, 2)
                r_opt, s_opt = compute_lp_radii(c_opt, A_ub, triu_i, triu_j)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Nelder-Mead on centers to explore layout space using LP objective
    if best_centers is not None:
        for _ in range(6):
            start_c = best_centers + np.random.uniform(-0.012, 0.012, best_centers.shape)
            start_c = np.clip(start_c, 0.05, 0.95)
            try:
                nm_res = minimize(lp_objective, start_c.flatten(), 
                                  args=(n, A_ub, triu_i, triu_j),
                                  method='Nelder-Mead', 
                                  options={'maxiter': 2500, 'xatol': 1e-6, 'fatol': 1e-8})
                c_nm = np.clip(nm_res.x.reshape(n, 2), 0.05, 0.95)
                r_nm, s_nm = compute_lp_radii(c_nm, A_ub, triu_i, triu_j)
                if s_nm > best_sum:
                    best_sum = s_nm
                    best_centers = c_nm.copy()
                    best_radii = r_nm.copy()
            except Exception:
                pass
                
    # Phase 3: Final SLSQP polish from refined centers
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), np.full(n, 0.08)])
        try:
            res = minimize(joint_objective, x0, args=(n,),
                          method='SLSQP', bounds=bounds_vars,
                          constraints={'type': 'ineq', 'fun': joint_constraints, 'args': (n, triu_i, triu_j)},
                          options={'maxiter': 4000, 'ftol': 1e-12})
            if np.isfinite(res.fun):
                c_opt = res.x[:2*n].reshape(n, 2)
                r_opt, s_opt = compute_lp_radii(c_opt, A_ub, triu_i, triu_j)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass

    # Fallback safety net
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum = compute_lp_radii(best_centers, A_ub, triu_i, triu_j)
        
    # Final safety scaling to strictly satisfy 1e-12 validator tolerance
    c = best_centers
    r = best_radii
    scale = 1.0
    for i in range(n):
        if r[i] > 1e-12:
            scale = min(scale, c[i,0]/r[i], (1.0-c[i,0])/r[i], c[i,1]/r[i], (1.0-c[i,1])/r[i])
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(c[i]-c[j])
            rs = r[i] + r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    r *= scale * 0.999999
    best_sum = float(np.sum(r))
    best_radii = r
    
    return best_centers, best_radii, best_sum
