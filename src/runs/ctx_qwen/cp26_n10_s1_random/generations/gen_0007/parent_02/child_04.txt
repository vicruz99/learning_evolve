1. **Geometric Layout (Centers)**: Find optimal positions for circle centers.
2. **Radial Assignment (LP)**: For fixed centers, solve a Linear Program to assign optimal radii that maximize $\sum r_i$ while respecting boundary and pairwise distance constraints.

The LP objective $f(C) = \max \sum r_i$ subject to $r_i \le \text{dist}(c_i, \partial \Omega)$ and $r_i + r_j \le \|c_i - c_j\|$ is piecewise linear and non-smooth. Traditional gradient-based optimizers struggle with it. Instead, we use:
- **Diverse Initializations**: Hexagonal lattices (known to be dense), grids, and random placements.
- **Derivative-Free Local Search**: `Nelder-Mead` for coarse alignment, followed by an adaptive **Random Walk / Hill-Climbing** that perturbs individual circles and accepts moves that improve the LP objective. This is highly effective for packing landscapes.
- **Exact LP Solver**: `scipy.optimize.linprog` with HiGHS solves the radial assignment in milliseconds, providing an exact evaluation of the geometry's quality.
- **Safety Scaling**: Finally, we apply a strict multiplicative scale factor to guarantee numerical validity against the $10^{-12}$ tolerance in the validator.

import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_for_centers(centers, pair_i, pair_j, n):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns (sum_radii, radii_array)
    """
    # Distance to boundaries
    d_b = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                     np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    d_b = np.maximum(d_b, 1e-9)
    
    # Pairwise distances
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[pair_i, pair_j]
    
    c_obj = -np.ones(n)
    bounds = [(0.0, db) for db in d_b]
    
    # Constraint matrix: r_i + r_j <= d_ij
    A_ub = np.zeros((len(pair_i), n))
    A_ub[np.arange(len(pair_i)), pair_i] = 1.0
    A_ub[np.arange(len(pair_i)), pair_j] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return -res.fun, res.x
    except Exception:
        pass
        
    # Fallback greedy assignment if LP fails
    r = np.zeros(n)
    for i in range(n):
        r[i] = d_b[i]
        for j in range(i):
            r[i] = min(r[i], dists[i, j] - r[j])
        r[i] = max(r[i], 1e-9)
    return np.sum(r), r

def run_packing():
    n = 26
    pair_i, pair_j = np.triu_indices(n, k=1)
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    configs = []
    
    # 1. Hexagonal patterns with various row distributions and scales
    row_pats = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 4, 6, 5, 6]
    ]
    for pat in row_pats:
        if sum(pat) < n: continue
        for r0 in [0.085, 0.09, 0.095, 0.10, 0.105]:
            pts = []
            y = r0
            for ri, cnt in enumerate(pat):
                shift = r0 if ri % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) < n:
                        pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            if len(pts) >= n:
                cfg = np.array(pts[:n])
                cfg = np.clip(cfg, 0.05, 0.95)
                configs.append(cfg)
                
    # 2. Grid patterns
    for sp in [0.15, 0.18, 0.20, 0.22]:
        pts = []
        for i in range(7):
            for j in range(7):
                if len(pts) < n:
                    pts.append([0.05 + i*sp, 0.05 + j*sp])
        if len(pts) >= n:
            configs.append(np.array(pts[:n]))
            
    # 3. Random dense placements
    for _ in range(10):
        configs.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    # Optimization loop over diverse starts
    for cfg in configs:
        curr_c = cfg.copy()
        
        # Phase 1: Nelder-Mead coarse optimization
        def obj(x):
            c = np.clip(x.reshape(n, 2), 1e-4, 1.0 - 1e-4)
            s, _ = solve_lp_for_centers(c, pair_i, pair_j, n)
            return -s
            
        try:
            res = minimize(obj, curr_c.flatten(), method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-6, 'fatol': 1e-7})
            curr_c = np.clip(res.x.reshape(n, 2), 1e-4, 1.0 - 1e-4)
        except Exception:
            pass
            
        # Phase 2: Adaptive Random Walk refinement
        curr_s, _ = solve_lp_for_centers(curr_c, pair_i, pair_j, n)
        step = 0.018
        for _ in range(2500):
            idx = rng.integers(n)
            old_pos = curr_c[idx].copy()
            move = rng.uniform(-step, step, 2)
            curr_c[idx] = np.clip(old_pos + move, 1e-4, 1.0 - 1e-4)
            
            s, _ = solve_lp_for_centers(curr_c, pair_i, pair_j, n)
            if s > curr_s + 1e-9:
                curr_s = s
                step *= 0.9985
            else:
                curr_c[idx] = old_pos
                if rng.random() < 0.03:
                    step *= 0.94
                    
        if curr_s > best_sum:
            best_sum = curr_s
            best_centers = curr_c.copy()
            _, best_radii = solve_lp_for_centers(best_centers, pair_i, pair_j, n)
            
    # Fallback safety net
    if best_centers is None:
        best_centers = configs[0]
        _, best_radii = solve_lp_for_centers(best_centers, pair_i, pair_j, n)
        best_sum = np.sum(best_radii)
        
    # Final strict safety scaling to guarantee validation tolerance
    scale = 1.0
    c = best_centers
    r = best_radii
    for i in range(n):
        if r[i] > 1e-12:
            scale = min(scale, c[i,0]/r[i], (1-c[i,0])/r[i], c[i,1]/r[i], (1-c[i,1])/r[i])
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(c[i]-c[j])
            r_sum = r[i] + r[j]
            if r_sum > 1e-12:
                scale = min(scale, d / r_sum)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
