import numpy as np
from scipy.optimize import linprog

def solve_lp_radii(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: 
      0 <= r_i <= distance to boundary
      r_i + r_j <= dist(i, j)
    """
    n = centers.shape[0]
    # Upper bounds for radii based on distance to square boundaries
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    # Pairwise distance constraints: r_i + r_j <= d_ij
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    c_obj = -np.ones(n)  # Maximize sum(r) => minimize -sum(r)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    # Fallback if LP fails
    return np.full(n, 1e-5), 0.0

    ]
    
    for pat in patterns:
        if sum(pat) < n: 
            continue
        for r0 in [0.085, 0.095, 0.105, 0.115]:
            pts = []
            y = r0

                x = r0 + shift

                    if len(pts) < n:
                        pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            configs.append(np.array(pts[:n]))
            
    # 2. Rotated hexagonal patterns to break axis-aligned symmetry
    for pat in patterns[:5]:
        if sum(pat) < n: continue
        base = []
        y = 0.1
        for ri, cnt in enumerate(pat):
            shift = 0.1 if ri % 2 == 1 else 0.0
            x = 0.1 + shift
            for _ in range(cnt):
                if len(base) < n: base.append([x, y])
                x += 0.2
            y += 0.173

            c, s = np.cos(ang), np.sin(ang)

            best_radii = curr_radii.copy()

                    scale = min(scale, d / r_sum)

        r *= scale * 0.9999995
        best_sum = float(np.sum(r))
        best_radii = r
    else:
        # Absolute fallback
        best_centers = np.clip(configs[0], 0.1, 0.9)
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    return best_centers, best_radii, best_sum
