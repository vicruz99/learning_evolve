from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRIU_I)
A_PAIRWISE = np.zeros((M_PAIRS, N))
for k, (i, j) in enumerate(zip(TRIU_I, TRIU_J)):
    A_PAIRWISE[k, i] = 1.0
    A_PAIRWISE[k, j] = 1.0

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    lims = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[TRIU_I, TRIU_J]
    
    try:
        res = linprog(np.ones(N)*-1.0, A_ub=A_PAIRWISE, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(N, 1e-5), 0.0

def neg_lp_sum(centers_flat):
    """Objective for center optimization: minimize negative LP sum of radii."""
    return -solve_radii_lp(centers_flat.reshape(-1, 2))[1]

def obj_joint(v, n):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_joint(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid joint packing."""
    cx, cy, r = v[:n], v[n:2*n], v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c = np.concatenate([c, d2[triu_i, triu_j] - rs[triu_i, triu_j]**2])
    return c
