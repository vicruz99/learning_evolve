        res = linprog(-np.ones(n), A_ub=A_ub_struct, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def _local_search(centers, n, triu_i, triu_j, A_ub_struct, rng):

def _obj_slsqp(x, n):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(x[2*n:])

def _cons_slsqp(x, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing in SLSQP."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    c = np.empty(4*n + len(triu_i))
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    dists = np.sqrt(cx_m**2 + cy_m**2)
    r_sum = r[:, None] + r[None, :]
    c[4*n:] = dists[triu_i, triu_j] - r_sum[triu_i, triu_j]
    return c

def _slsqp_opt(centers, n, triu_i, triu_j, A_ub_struct, rng):
    """Jointly optimizes centers and radii using SLSQP."""
    x0 = np.zeros(3*n)
    x0[:n] = centers[:, 0]
    x0[n:2*n] = centers[:, 1]
    x0[2*n:] = np.full(n, 0.08)
    
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    cons = {'type': 'ineq', 'fun': _cons_slsqp, 'args': (n, triu_i, triu_j)}
    
    try:
        res = minimize(_obj_slsqp, x0, args=(n,), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13})
        if np.isfinite(res.fun):
            c_opt = res.x[:2*n].reshape(n, 2)
            return c_opt
    except Exception:
        pass
    return centers

def _make_hex(pat, r0, n):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    for ri, cnt in enumerate(pat):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) < n:
                pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(pts[:n])
