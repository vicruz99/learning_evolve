Strategy:

2. **Derivative-Free Optimization**: The LP objective is non-smooth with respect to centers. We use `Nelder-Mead` (a derivative-free method) to navigate the center-space landscape effectively, avoiding gradient approximation issues.
3. **Greedy Hill-Climbing Refinement**: After Nelder-Mead converges, we apply a local perturbation search with adaptive step sizes. Randomly moving one circle at a time and accepting improvements allows fine-tuning past the coarse NM steps.

5. **Strict Validation Scaling**: Finally, we compute the exact scaling factor that brings all constraints strictly within the `1e-12` tolerance of the validator and apply it, guaranteeing a valid submission.

import numpy as np
from scipy.optimize import minimize, linprog

def _get_pair_indices(n):
    return np.triu_indices(n, k=1)

def _build_pair_matrix(n, pair_i, pair_j):
    A = np.zeros((len(pair_i), n))
    for k, (i, j) in enumerate(zip(pair_i, pair_j)):
        A[k, i] = 1.0
        A[k, j] = 1.0
    return A

def _solve_lp(centers, n, pair_i, pair_j, A_pair):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Upper bounds on radii from distance to boundaries
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[pair_i, pair_j]
    
    # Objective: maximize sum(r) <=> minimize -sum(r)
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_pair, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    
    # Fallback: uniform small radius if LP fails
    return np.full(n, 1e-6), 0.0

def _eval_packing(x_flat, n, pair_i, pair_j, A_pair):
    """Objective for optimizer: negative max sum of radii."""
    centers = x_flat.reshape(n, 2)
    # Keep centers strictly inside to avoid degenerate LP bounds
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, s = _solve_lp(centers, n, pair_i, pair_j, A_pair)
    return -s

def _make_hex(rows, r0, n):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) < n:
                pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    pair_i, pair_j = _get_pair_indices(n)
    A_pair = _build_pair_matrix(n, pair_i, pair_j)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # 1. Generate diverse initial configurations
    configs = []
    row_pats = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5]
    ]
    for pat in row_pats:
        for r0 in [0.095, 0.10, 0.105]:
            cfg = _make_hex(pat, r0, n)
            configs.append(np.clip(cfg, 0.05, 0.95))
            
    # Random dense starts
    for _ in range(6):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # 2. Optimize each configuration
    for cfg in configs:
        x0 = cfg.flatten()
        
        # Phase A: Nelder-Mead global search on centers
        try:
            res = minimize(_eval_packing, x0, args=(n, pair_i, pair_j, A_pair),
                           method='Nelder-Mead', options={'maxiter': 8000, 'xatol': 1e-7, 'fatol': 1e-8})
            curr_centers = np.clip(res.x.reshape(n, 2), 1e-4, 1.0 - 1e-4)
        except Exception:
            curr_centers = cfg.copy()
            
        curr_radii, curr_sum = _solve_lp(curr_centers, n, pair_i, pair_j, A_pair)
        
        # Phase B: Adaptive Hill-Climbing refinement
        step = 0.015
        for _ in range(3000):
            i = rng.integers(n)
            old_pos = curr_centers[i].copy()
            
            # Perturb
            curr_centers[i] += rng.uniform(-step, step, 2)
            curr_centers[i] = np.clip(curr_centers[i], 0.01, 0.99)
            
            r_new, s_new = _solve_lp(curr_centers, n, pair_i, pair_j, A_pair)
            
            if s_new > curr_sum:
                curr_sum = s_new
                curr_radii = r_new
                step *= 1.001  # Slightly increase step on success
            else:
                curr_centers[i] = old_pos
                step *= 0.997  # Decay step on failure
                
            if s_new > best_sum:
                best_sum = s_new
                best_centers = curr_centers.copy()
                best_radii = curr_radii.copy()
                
    # 3. Iterative Escape: Perturb best and re-optimize to escape local minima
    if best_centers is not None:
        for _ in range(3):
            pert_centers = best_centers + rng.normal(0, 0.03, best_centers.shape)
            pert_centers = np.clip(pert_centers, 0.1, 0.9)
            
            try:
                res = minimize(_eval_packing, pert_centers.flatten(), args=(n, pair_i, pair_j, A_pair),
                               method='Nelder-Mead', options={'maxiter': 5000, 'xatol': 1e-7, 'fatol': 1e-8})
                c_opt = np.clip(res.x.reshape(n, 2), 1e-4, 1.0 - 1e-4)
            except Exception:
                c_opt = pert_centers.copy()
                
            r_opt, s_opt = _solve_lp(c_opt, n, pair_i, pair_j, A_pair)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
                
    # 4. Final Safety Scaling to strictly satisfy 1e-12 validator tolerance
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            if best_radii[i] > 1e-12:
                scale = min(scale, 
                            best_centers[i, 0] / best_radii[i], 
                            (1.0 - best_centers[i, 0]) / best_radii[i],
                            best_centers[i, 1] / best_radii[i], 
                            (1.0 - best_centers[i, 1]) / best_radii[i])
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(best_centers[i] - best_centers[j])
                r_sum = best_radii[i] + best_radii[j]
                if r_sum > 1e-12:
                    scale = min(scale, d / r_sum)
                    
        best_radii *= scale * 0.9999995
        best_sum = float(np.sum(best_radii))
    else:
        best_sum = 0.0
        
    return best_centers, best_radii, best_sum
