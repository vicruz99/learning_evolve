from scipy.optimize import linprog, minimize

def _get_pair_indices(n):
    """Returns indices for upper triangle of distance matrix."""
    return np.triu_indices(n, k=1)

def _solve_lp(centers, n, pi, pj, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Boundary limits for each circle
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    d_ij = dists[pi, pj]
    
    bounds = [(0.0, lim) for lim in lims]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=d_ij, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def _lp_obj(centers_flat, n, pi, pj, A_ub):

    _, s = _solve_lp(c, n, pi, pj, A_ub)
    return -s

            if len(pts) < n:
                pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(pts[:n])
