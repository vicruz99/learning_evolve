import numpy as np
from scipy.optimize import linprog, minimize

def compute_distances(centers):
    """Computes the pairwise Euclidean distance matrix."""
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.sqrt(np.sum(diffs**2, axis=2))

def solve_lp(centers, n, pair_i, pair_j, dists):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns optimal radii, sum of radii, and gradient of sum w.r.t centers.
    """
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-9)) for lim in limits]
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[pair_i, pair_j]
    for k, (i, j) in enumerate(zip(pair_i, pair_j)):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success or not np.isfinite(res.fun):
        return None, None, None
        
    radii = res.x
    sum_r = -res.fun
    
    # Compute gradient using dual variables (marginals)
    grad_c = np.zeros((n, 2))
    if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
        marginals = res.ineqlin.marginals
        for k, (i, j) in enumerate(zip(pair_i, pair_j)):
            lam = marginals[k]
            if lam > 1e-9:
                d = dists[i, j]
                if d > 1e-9:
                    diff = centers[i] - centers[j]
                    grad_c[i] += lam * diff / d
                    grad_c[j] -= lam * diff / d
                    
    return radii, sum_r, grad_c

def objective_func(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def constraint_func(x, n, pair_i, pair_j):
    """Inequality constraints >= 0 for valid packing."""
    c = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    dists = compute_distances(c)
    
    cons = []
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    # Pairwise non-overlap
    cons.append(dists[pair_i, pair_j] - (r[pair_i] + r[pair_j]))
    return np.concatenate(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_radii = None

        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], 
        [6,6,5,5,4], [5,4,6,6,5], [5,6,4,6,5], [6,5,5,6,4]
    ]
    
    for pat in patterns:
        if sum(pat) < n: continue
        r0 = 0.10
        pts = []
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) < n: pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        configs.append(np.array(pts[:n]))
        
    np.random.seed(42)
    for _ in range(10):
        configs.append(np.random.uniform(0.1, 0.9, (n, 2)))

                r_lp, s_lp, _ = solve_lp(c_slqp, n, pair_i, pair_j, dists_f)

                    best_sum = s_lp

            pass

                d = dists_f[i, j]
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)

        best_sum = float(np.sum(best_radii))

        best_centers = np.ones((n, 2)) * 0.5
        best_radii = np.full(n, 0.01)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
