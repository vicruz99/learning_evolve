import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_fast(centers, pair_i, pair_j, A_ub):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    diffs = centers[pair_i] - centers[pair_j]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    
    # Bounds: 0 <= r_i <= lims[i]
    bounds_arr = np.column_stack((np.zeros(n), lims))
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=dists, bounds=bounds_arr, method='highs')
        if res.success and np.isfinite(res.fun):
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def constr_joint(vars_arr, n, pair_i, pair_j):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    c = vars_arr[:2*n].reshape(n, 2)
    r = vars_arr[2*n:]
    
    lims = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), np.minimum(c[:, 1], 1.0 - c[:, 1]))
    c_list = lims - r
    
    diffs = c[pair_i] - c[pair_j]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    r_sum = r[pair_i] + r[pair_j]
    
    return np.concatenate([c_list, dists - r_sum])

def neg_lp_sum_wrapper(centers_flat, n, pair_i, pair_j, A_ub):
    """Objective for derivative-free optimization: negative LP sum of radii."""
    c = centers_flat.reshape(n, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    return -solve_lp_fast(c, pair_i, pair_j, A_ub)[0]

def obj_slsqp(vars_arr, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    pair_i, pair_j = np.triu_indices(n, k=1)
    m_pairs = len(pair_i)
    
    # Precompute constant constraint matrix for pairwise distances
    A_ub = np.zeros((m_pairs, n))
    A_ub[np.arange(m_pairs), pair_i] = 1.0
    A_ub[np.arange(m_pairs), pair_j] = 1.0
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    configs = []
    np.random.seed(42)
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [6, 6, 5, 5, 4], [5, 6, 4, 6, 5], [5, 4, 6, 5, 6], [6, 5, 5, 6, 4],
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 7, 5, 5, 4],
        [7, 5, 5, 5, 4], [5, 5, 7, 5, 4], [4, 5, 7, 5, 5], [5, 6, 6, 5, 4]
    ]
    
    for pat in patterns:
        if sum(pat) < n: continue
        for r0 in np.linspace(0.085, 0.115, 5):
            pts = []
            y = r0
            for ri, cnt in enumerate(pat):
                shift = r0 if ri % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) < n:
                        pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            if len(pts) >= n:
                cfg = np.array(pts[:n])
                mn, mx = cfg.min(axis=0), cfg.max(axis=0)
                span = mx - mn
                if span > 0:
                    cfg = (cfg - mn) / span * 0.9 + 0.05
                configs.append(cfg)
                
    for _ in range(15):
        configs.append(np.random.uniform(0.1, 0.9, (n, 2)))
        
    # Phase 1: Local search + Nelder-Mead polish for each config
    for cfg in configs:
        centers = cfg.copy()
        curr_sum, curr_r = solve_lp_fast(centers, pair_i, pair_j, A_ub)
        
        # Stochastic hill-climbing on centers
        for step in range(2000):
            i = np.random.randint(n)
            old_pos = centers[i].copy()
            step_size = 0.012 * (0.996 ** step)
            dx = np.random.uniform(-step_size, step_size)
            dy = np.random.uniform(-step_size, step_size)
            new_pos = old_pos + np.array([dx, dy])
            new_pos = np.clip(new_pos, 1e-5, 1.0 - 1e-5)
            
            centers[i] = new_pos
            new_sum, _ = solve_lp_fast(centers, pair_i, pair_j, A_ub)
            
            if new_sum > curr_sum:
                curr_sum = new_sum
            else:
                centers[i] = old_pos
                
        # Derivative-free polish
        try:
            res = minimize(neg_lp_sum_wrapper, centers.flatten(), args=(n, pair_i, pair_j, A_ub),
                           method='Nelder-Mead', options={'maxiter': 1500, 'xatol': 1e-8, 'fatol': 1e-9})
            c_opt = res.x.reshape(n, 2)
            c_opt = np.clip(c_opt, 1e-6, 1.0 - 1e-6)
            s_opt, r_opt = solve_lp_fast(c_opt, pair_i, pair_j, A_ub)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt
                best_radii = r_opt
        except Exception:
            pass
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            _, best_radii = solve_lp_fast(best_centers, pair_i, pair_j, A_ub)

    # Phase 2: Joint SLSQP refinement from the best found configuration
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii * 0.9])
        bounds_vars = [(0.0, 1.0)] * (2 * n) + [(1e-5, 0.5)] * n
        cons = {'type': 'ineq', 'fun': constr_joint, 'args': (n, pair_i, pair_j)}
        try:
            res = minimize(obj_slsqp, x0, args=(n,), method='SLSQP', bounds=bounds_vars,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-12})
            if np.isfinite(res.fun):
                c_s = res.x[:2*n].reshape(n, 2)
                c_s = np.clip(c_s, 1e-6, 1.0 - 1e-6)
                s_s, r_s = solve_lp_fast(c_s, pair_i, pair_j, A_ub)
                if s_s > best_sum:
                    best_sum = s_s
                    best_centers = c_s
                    best_radii = r_s
        except Exception:
            pass

    # Final safety scaling to strictly satisfy 1e-12 validator tolerance
    if best_centers is not None:
        scale = 1.0
        c = best_centers
        r = best_radii
        for i in range(n):
            if r[i] > 1e-12:
                scale = min(scale, c[i,0]/r[i], (1-c[i,0])/r[i], c[i,1]/r[i], (1-c[i,1])/r[i])
        for i in range(n):
            for j in range(i+1, n):
                d = np.linalg.norm(c[i]-c[j])
                r_sum = r[i] + r[j]
                if r_sum > 1e-12:
                    scale = min(scale, d / r_sum)
        r *= scale * 0.9999995
        best_radii = r
        best_sum = float(np.sum(r))
        
    return best_centers, best_radii, best_sum
