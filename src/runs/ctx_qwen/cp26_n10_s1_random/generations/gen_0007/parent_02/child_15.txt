from scipy.optimize import minimize, linprog

def get_pair_indices(n):
    """Returns indices for upper triangular pairs (i, j) with i < j."""
    return np.triu_indices(n, k=1)

def solve_lp(centers, n, pair_i, pair_j):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n_pairs = len(pair_i)
    
    # Compute pairwise distances efficiently
    dx = centers[:, 0:1] - centers[:, 0:1].T
    dy = centers[:, 1:2] - centers[:, 1:2].T
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, 1e9)
    
    # Boundary limits for each circle
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in limits]
    
    # Construct constraint matrix: r_i + r_j <= dist_ij
    A_ub = np.zeros((n_pairs, n))
    A_ub[np.arange(n_pairs), pair_i] = 1.0
    A_ub[np.arange(n_pairs), pair_j] = 1.0
    b_ub = dists[pair_i, pair_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback if LP fails
    radii = np.full(n, 1e-6)
    return radii, np.sum(radii)

def objective_joint(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def constraints_joint(x, n, pair_i, pair_j):
    """Inequality constraints >= 0 for valid packing."""
    c = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    
    # Boundary constraints
    con = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    con = np.concatenate([con, dist[pair_i, pair_j] - r_sum[pair_i, pair_j]])
    return con

def make_hex(rows, r0, n):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) < n:
                pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(pts[:n])
