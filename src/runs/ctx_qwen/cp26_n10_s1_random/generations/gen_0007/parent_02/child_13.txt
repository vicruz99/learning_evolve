import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_pairwise_indices():
    return np.triu_indices(N, k=1)

def compute_clearances(centers):
    """Computes boundary distances and half-pairwise distances."""
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Boundary clearances: distance to each wall
    c_b = np.concatenate([x, 1.0 - x, y, 1.0 - y])
    
    # Pairwise clearances: half distance between centers
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    c_p = dists[np.triu_indices(N, k=1)] / 2.0
    
    return np.concatenate([c_b, c_p])

def smooth_min_clearance(centers, k=50.0):
    """Smooth approximation of min clearance using log-sum-exp."""
    c = compute_clearances(centers)
    # Numerically stable log-sum-exp
    max_c = np.max(c)
    val = -np.log(np.sum(np.exp(-k * (c - max_c)))) / k + max_c
    return val

def neg_smooth_clearance(vars_flat, k):
    """Objective for center optimization: maximize smooth min clearance."""
    centers = vars_flat.reshape(N, 2)
    return -smooth_min_clearance(centers, k)

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Boundary limits for each circle
    lims = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    idx_i, idx_j = np.triu_indices(N, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, N))
    b_ub = dists[idx_i, idx_j]
    
    for k_idx in range(m):
        A_ub[k_idx, idx_i[k_idx]] = 1.0
        A_ub[k_idx, idx_j[k_idx]] = 1.0
        
    c_obj = -np.ones(N)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    # Fallback: iterative assignment
    radii = np.zeros(N)
    for i in range(N):
        d_min = lims[i]
        for j in range(N):
            if i != j:
                d_min = min(d_min, dists[i, j] - radii[j])
        radii[i] = max(d_min, 1e-6)
    return radii, np.sum(radii)

def joint_objective(vars_flat):
    return -np.sum(vars_flat[2*N:])

def joint_constraints(vars_flat, idx_i, idx_j):
    c = vars_flat[:2*N].reshape(N, 2)
    r = vars_flat[2*N:]
    
    # Boundary
    cons = np.concatenate([
        c[:, 0] - r, 1.0 - c[:, 0] - r,
        c[:, 1] - r, 1.0 - c[:, 1] - r
    ])
    
    # Pairwise
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    cons = np.concatenate([cons, dist[idx_i, idx_j] - r_sum[idx_i, idx_j]])
    return cons

    c = best_centers

            d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
            rs = r[i] + r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    r *= scale * 0.9999999
    final_sum = float(np.sum(r))
    
    return c, r, final_sum
