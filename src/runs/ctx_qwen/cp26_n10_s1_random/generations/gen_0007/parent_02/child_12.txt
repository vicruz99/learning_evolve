import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, A_ub, idx):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    b_ub = dists[idx]
    bounds = [(0.0, lim) for lim in lims]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def _obj_joint(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def _cons_joint(v, n, idx):
    """Inequality constraints >= 0 for valid packing."""
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    res = np.concatenate([c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r])
    dx = c[:,0:1] - c[:,0:1].T
    dy = c[:,1:2] - c[:,1:2].T
    dist = np.sqrt(dx**2 + dy**2)
    rs = r[:, None] + r[None, :]
    res = np.concatenate([res, dist[idx] - rs[idx]])
    return res

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    idx = np.triu_indices(n, k=1)
    m = len(idx[0])
    A_ub = np.zeros((m, n))
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = []
    row_pats = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [6, 6, 5, 5, 4], [5, 6, 4, 6, 5], [5, 5, 5, 5, 6], [6, 4, 6, 5, 5],
        [5, 7, 5, 5, 4], [4, 5, 6, 6, 5], [6, 6, 6, 4, 4]
    ]
    
    for pat in row_pats:
        if sum(pat) < n: continue
        for r0 in [0.09, 0.095, 0.10, 0.105]:
            pts = []
            y = r0
            for ri, cnt in enumerate(pat):
                shift = r0 if ri % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) < n:
                        pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            configs.append(np.array(pts[:n]))
            
            for angle in [0.1, -0.1, 0.2, -0.2, 0.3]:
                cos_t, sin_t = np.cos(angle), np.sin(angle)
                rot = np.array([[cos_t, -sin_t], [sin_t, cos_t]])
                cfg_rot = np.array(pts[:n]) @ rot
                mn, mx = cfg_rot.min(axis=0), cfg_rot.max(axis=0)
                cfg_rot = (cfg_rot - mn) / (mx - mn + 1e-9)
                cfg_rot = cfg_rot * 0.86 + 0.07
                configs.append(np.clip(cfg_rot, 0.05, 0.95))

    np.random.seed(42)
    for _ in range(5):
        configs.append(np.random.uniform(0.15, 0.85, (n, 2)))

    # Local search on centers evaluated via LP
    for cfg in configs:

            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
