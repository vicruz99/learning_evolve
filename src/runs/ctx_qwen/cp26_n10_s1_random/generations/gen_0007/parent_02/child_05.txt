import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_IDX = None
A_MAT = None

def _setup_global():
    """Precompute constant matrices for LP and constraints."""
    global TRIU_IDX, A_MAT
    TRIU_IDX = np.triu_indices(N, k=1)
    m = len(TRIU_IDX[0])
    A_MAT = np.zeros((m, N))
    for k, (i, j) in enumerate(zip(TRIU_IDX[0], TRIU_IDX[1])):
        A_MAT[k, i] = 1.0
        A_MAT[k, j] = 1.0

def _solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(limits, 1e-9)
    bounds_lp = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    b_ub = dists[TRIU_IDX]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_MAT, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(N, 1e-5), 0.0

def _joint_obj(x):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def _joint_con(x):
    """Inequality constraints >= 0 for joint optimization."""
    c = x[:2*N].reshape(N, 2)
    r = x[2*N:]
    
    cons = np.concatenate([
        c[:, 0] - r, 1.0 - c[:, 0] - r,
        c[:, 1] - r, 1.0 - c[:, 1] - r
    ])
    
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    cons = np.concatenate([cons, dist[TRIU_IDX] - r_sum[TRIU_IDX]])
    return cons

def _pos_obj(pos, data):
    """Objective for coordinate descent: minimize negative LP sum of radii."""
    idx, fixed_c = data
    temp_c = fixed_c.copy()
    temp_c[idx] = np.clip(pos, 0.0, 1.0)
    _, s = _solve_lp(temp_c)
    return -s

def _gen_hex(rows, r0):
    """Generates hexagonal lattice points."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def run_packing():
    _setup_global()
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_vars = [(0.0, 1.0)] * (2 * N) + [(1e-5, 0.5)] * N

        [5,6,4,6,5], [5,4,6,5,6], [6,5,5,6,4], [5,5,5,5,6], [7,5,5,5,4],
        [4,5,6,6,5], [6,4,6,5,5], [5,7,5,5,4]
    ]
    
    for pat in pats:
        if sum(pat) < N: continue
        for r0 in [0.085, 0.095, 0.105, 0.115]:
            cfg = _gen_hex(pat, r0)
            configs.append(np.clip(cfg, 0.05, 0.95))
            
        cfg_base = _gen_hex(pat, 0.10)
        for ang in [0.15, -0.15, 0.3, -0.3]:
            ct, st = np.cos(ang), np.sin(ang)
            rot = np.array([[ct, -st], [st, ct]])
            cr = cfg_base @ rot

            rng_span = mx - mn
            rng_span[rng_span < 1e-9] = 1.0
            cr = (cr - mn) / rng_span * 0.86 + 0.07
            configs.append(np.clip(cr, 0.05, 0.95))
            
    np.random.seed(42)

        configs.append(np.random.uniform(0.15, 0.85, (N, 2)))

        try:

            if np.isfinite(res.fun):
                c_opt = res.x[:2*N].reshape(N, 2)
                r_lp, s_lp = _solve_lp(c_opt)

                    improved = True
            except Exception:
                pass
                
        if not improved:
            # Escape local minima with controlled perturbation
            best_centers += np.random.uniform(-0.025, 0.025, best_centers.shape)

            best_radii, best_sum = _solve_lp(best_centers)

    scale = 1.0
    for i in range(N):
        if best_radii[i] > 1e-12:
            x, y = best_centers[i]
            scale = min(scale, x/best_radii[i], (1.0-x)/best_radii[i],

            r_sum = best_radii[i] + best_radii[j]
            if r_sum > 1e-12:
                scale = min(scale, d / r_sum)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
