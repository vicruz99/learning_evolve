import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, A_ub_static):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns (radii_array, sum_of_radii)
    """
    n = centers.shape[0]
    # Boundary limits for each circle
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    idx = np.triu_indices(n, k=1)
    b_ub = dists[idx]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_static, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def constraints_joint(vars_arr, n):
    """
    Computes inequality constraints >= 0 for valid packing.
    Layout: [x0, y0, ..., x25, y25, r0, r1, ..., r25]
    """
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    cx_m = cx[:, None]
    cy_m = cy[:, None]
    r_m = r[:, None]
    
    dx = cx_m - cx_m.T
    dy = cy_m - cy_m.T
    dr = r_m + r_m.T
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    d2 = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    rs2 = dr[i_idx, j_idx]**2
    
    return np.concatenate([c, d2 - rs2])

def obj_slqp(vars_arr, n):
    """Objective for SLSQP: maximize sum of radii"""
    return -np.sum(vars_arr[2*n:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    # Precompute static constraint matrix for LP (each row has exactly two 1s)
    m = n * (n - 1) // 2
    A_ub_static = np.zeros((m, n))
    idx_0, idx_1 = np.triu_indices(n, k=1)
    for k, (i, j) in enumerate(zip(idx_0, idx_1)):
        A_ub_static[k, i] = 1.0
        A_ub_static[k, j] = 1.0
        
    # Generate diverse initial configurations
    inits = []
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], 
        [5,7,5,5,4], [5,5,5,6,5], [6,6,5,5,4], [5,6,4,6,5]
    ]
    
    for p in patterns:
        for r0 in [0.085, 0.095, 0.105]:
            pts = []
            y = r0
            for ri, cnt in enumerate(p):
                shift = r0 if ri % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2 * r0
                y += np.sqrt(3) * r0
            while len(pts) < n:
                pts.append(rng.uniform(r0, 1-r0, 2).tolist())
            inits.append(np.array(pts[:n]))
            
    # Add fully random dense placements
    for _ in range(8):
        inits.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_slqp = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    cons_slqp = {'type': 'ineq', 'fun': constraints_joint, 'args': (n,)}
    
    for cfg in inits:
        centers = cfg.copy()
        radii, current_sum = solve_lp_radii(centers, A_ub_static)
        
        # Phase 1: Simulated Annealing on centers
        T = 0.04
        step = 0.035
        for it in range(6000):
            i = rng.integers(n)
            old_c = centers[i].copy()
            new_c = old_c + rng.uniform(-step, step, 2)
            new_c = np.clip(new_c, 0.01, 0.99)
            centers[i] = new_c
            
            new_radii, new_sum = solve_lp_radii(centers, A_ub_static)
            delta = new_sum - current_sum
            
            # Acceptance criterion
            accept_prob = 1.0 if delta > 0 else np.exp(np.clip(delta / max(T, 1e-9), -20, 20))
            if rng.random() < accept_prob:
                current_sum = new_sum
                radii = new_radii
            else:
                centers[i] = old_c
                
            # Cooling schedule
            T *= 0.998
            step *= 0.9992
            
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
        # Phase 2: SLSQP Joint Polish from SA result
        x0 = np.zeros(3 * n)
        x0[:n] = best_centers[:, 0]
        x0[n:2*n] = best_centers[:, 1]
        x0[2*n:] = best_radii * 0.995
        
        # Small random perturbation to help SLSQP escape flat regions
        x0[:2*n] += rng.uniform(-0.001, 0.001, 2*n)
        x0[:2*n] = np.clip(x0[:2*n], 0.01, 0.99)
        
        try:
            res = minimize(obj_slqp, x0, args=(n,), method='SLSQP', 
                           bounds=bounds_slqp, constraints=cons_slqp,
                           options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
            if np.isfinite(res.fun):
                cx_f, cy_f, r_f = res.x[:n], res.x[n:2*n], res.x[2*n:]
                centers_f = np.column_stack((cx_f, cy_f))
                r_lp_f, s_lp_f = solve_lp_radii(centers_f, A_ub_static)
                if s_lp_f > best_sum:
                    best_sum = s_lp_f
                    best_centers = centers_f
                    best_radii = r_lp_f
        except Exception:
            pass

    # Fallback configuration
    if best_centers is None:
        fallback = inits[0]
        best_radii, best_sum = solve_lp_radii(fallback, A_ub_static)
        best_centers = fallback

    # Phase 3: Strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
