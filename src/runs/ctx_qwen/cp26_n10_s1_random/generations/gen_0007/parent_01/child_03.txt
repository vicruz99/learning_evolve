import numpy as np
from scipy.optimize import linprog

def solve_lp_fast(centers, A_ub, idx, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[idx]
    
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = list(zip(np.zeros(n), lims))
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    rng = np.random.default_rng(42)
    
    idx = np.triu_indices(n, k=1)
    A_ub = np.zeros((n * (n - 1) // 2, n))

        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    inits = []
    
    # Hexagonal lattices with different scales
    for r0 in [0.090, 0.095, 0.100, 0.105]:
        pts = []

        row = 0

            shift = r0 if row % 2 == 1 else 0.0
            x = r0 + shift

                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        while len(pts) < n:
            pts.append([0.5, 0.5])
        inits.append(np.array(pts[:n]))
        
        for _ in range(2):
            p = np.array(pts[:n]) + rng.uniform(-0.015, 0.015, (n, 2))
            inits.append(np.clip(p, 0.05, 0.95))

    # Grid patterns
    g = np.linspace(0.14, 0.86, 5)

    # Random dense
    for _ in range(4):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))

    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    for init_cfg in inits:

        r_curr, sum_curr = solve_lp_fast(centers, A_ub, idx, n)

            r_lim = current_radii[i]
            new_pos[0] = np.clip(new_pos[0], r_lim + 1e-7, 1.0 - r_lim - 1e-7)
            new_pos[1] = np.clip(new_pos[1], r_lim + 1e-7, 1.0 - r_lim - 1e-7)

            r_new, sum_new = solve_lp_fast(centers, A_ub, idx, n)

    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
