import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Precomputes the constant structure matrix for pairwise LP constraints."""
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A = np.zeros((m, n))
    for k, (i, j) in enumerate(zip(idx_i, idx_j)):
        A[k, i] = 1.0
        A[k, j] = 1.0
    return A, idx_i, idx_j

def solve_lp_radii(centers, n, A_ub, idx_i, idx_j):
    """Solves the LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    diffs = centers[idx_i] - centers[idx_j]
    b_ub = np.sqrt(np.sum(diffs**2, axis=1))
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def objective_joint(x, n):
    """Objective: minimize negative sum of radii => maximize sum of radii"""
    return -np.sum(x[2*n:])

def constraints_joint(x, n):
    """Inequality constraints >= 0 for valid joint packing."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    idx = np.triu_indices(n, k=1)
    d2 = dx[idx]**2 + dy[idx]**2
    rs2 = dr[idx]**2
    
    return np.concatenate([c, d2 - rs2])

def refine_centers_lp(centers, n, A_ub, idx_i, idx_j, rng, iters=2000):
    """Stochastic coordinate descent on centers maximizing the LP sum of radii."""
    curr = centers.copy()
    _, best_sum = solve_lp_radii(curr, n, A_ub, idx_i, idx_j)

        step = 0.045 * (0.999 ** t)
        idx = rng.integers(n)
        old = curr[idx].copy()
        
        improved = False
        # Try multiple directions per step

            delta = rng.uniform(-step, step, 2)
            new_pos = np.clip(old + delta, 0.015, 0.985)
            curr[idx] = new_pos
            r, s = solve_lp_radii(curr, n, A_ub, idx_i, idx_j)
            if r is not None and s > best_sum + 1e-9:
                best_sum = s
                best = curr.copy()
                improved = True
                break
        if not improved:
            curr[idx] = old
    return best, best_sum
