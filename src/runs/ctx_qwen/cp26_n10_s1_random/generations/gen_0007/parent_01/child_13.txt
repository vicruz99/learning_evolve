import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n, triu_i, triu_j):
    """Solves the LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[triu_i, triu_j]
    
    A_ub[np.arange(m), triu_i] = 1.0
    A_ub[np.arange(m), triu_j] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def objective_centers_wrapper(x_flat, n, triu_i, triu_j):
    """Objective for center optimization: negative sum of LP-optimized radii."""
    return -solve_lp_radii(x_flat.reshape(n, 2), n, triu_i, triu_j)[1]

def constraints_joint(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    
    d2 = dx[triu_i, triu_j]**2 + dy[triu_i, triu_j]**2
    rs2 = dr[triu_i, triu_j]**2
    
    return np.concatenate([c, d2 - rs2])

def objective_joint(v, n):
    """Objective for joint optimization: maximize sum of radii."""
    return -np.sum(v[2*n:])

    ]
    for p in patterns:

            y = r0

                    pts.append([x, y])
                    x += 2 * r0

            pass

    scale = 1.0

        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
