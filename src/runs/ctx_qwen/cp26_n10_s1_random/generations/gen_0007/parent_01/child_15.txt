import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def objective_joint(vars_arr, n):
    """Objective: minimize negative sum of radii => maximize sum of radii"""
    return -np.sum(vars_arr[2 * n:])

def constraints_joint(vars_arr, n):
    """Computes inequality constraints >= 0 for valid packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    cx_m = cx[:, None]
    cy_m = cy[:, None]
    r_m = r[:, None]
    
    dx = cx_m - cx_m.T
    dy = cy_m - cy_m.T
    dr = r_m + r_m.T
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    d2 = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    rs2 = dr[i_idx, j_idx]**2
    
    return np.concatenate([c, d2 - rs2])

def coord_obj(p, idx, centers, n):
    """Objective for coordinate-wise optimization: negative LP sum."""
    c_tmp = centers.copy()
    c_tmp[idx] = np.clip(p, 1e-4, 1.0 - 1e-4)
    _, s = solve_lp_radii(c_tmp, n)
    return -s

def make_hex_init(n, rows_pattern, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append(rng.uniform(r0, 1-r0, 2).tolist())
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(2024)
    bounds_vars = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    cons_dict = {'type': 'ineq', 'fun': constraints_joint, 'args': (n,)}
    
    # Diverse row distributions known to yield dense packings
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 5, 5, 6, 5], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 5, 6, 4, 6],
        [4, 5, 6, 5, 6], [5, 5, 5, 5, 4, 2], [6, 5, 5, 6, 4], [5, 5, 5, 5, 5, 1]
    ]
    
    inits = []
    for p in patterns:
        if sum(p) < n: continue
        for r0 in [0.08, 0.09, 0.10, 0.11]:
            inits.append(make_hex_init(n, p, r0, rng))
            
    # Grid variations
    g = np.linspace(0.12, 0.88, 5)
    grid = np.array([[x, y] for y in g for x in g])
    for extra in [[0.5, 0.5], [0.08, 0.92], [0.92, 0.08], [0.5, 0.92], [0.08, 0.08]]:
        inits.append(np.vstack([grid, extra]))
        
    for _ in range(15):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Multi-start SLSQP to find strong baselines
    for cfg in inits:
        x0 = np.zeros(3 * n)
        x0[:n] = cfg[:, 0]
        x0[n:2*n] = cfg[:, 1]
        x0[2*n:] = 0.05
        
        try:
            res = minimize(objective_joint, x0, args=(n,), method='SLSQP', bounds=bounds_vars,
                           constraints=cons_dict, options={'maxiter': 4000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                cx, cy, r = res.x[:n], res.x[n:2*n], res.x[2*n:]
                centers = np.column_stack((cx, cy))
                r_lp, s_lp = solve_lp_radii(centers, n)
                if r_lp is not None and s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = centers.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass
            
    # Phase 2: Coordinate-wise optimization using Nelder-Mead
    # Highly effective for navigating the non-smooth LP objective landscape
    if best_centers is not None:
        current_c = best_centers.copy()
        current_r = best_radii.copy()
        current_sum = best_sum
        
        for epoch in range(4):
            for i in range(n):
                old_pos = current_c[i].copy()
                res_nm = minimize(coord_obj, old_pos, args=(i, current_c, n), method='Nelder-Mead', 
                                  options={'xatol': 1e-6, 'fatol': 1e-8, 'maxiter': 150})
                if -res_nm.fun > current_sum + 1e-8:
                    current_c[i] = res_nm.x
                    current_r, current_sum = solve_lp_radii(current_c, n)
                    
        if current_sum > best_sum:
            best_centers = current_c
            best_radii = current_r
            best_sum = current_sum

        # Phase 3: Joint SLSQP polish from refined centers
        x0_final = np.zeros(3 * n)
        x0_final[:n] = best_centers[:, 0]
        x0_final[n:2*n] = best_centers[:, 1]
        x0_final[2*n:] = best_radii * 0.99
        
        try:
            res_final = minimize(objective_joint, x0_final, args=(n,), method='SLSQP', bounds=bounds_vars,
                                 constraints=cons_dict, options={'maxiter': 6000, 'ftol': 1e-15})
            if np.isfinite(res_final.fun):
                cx_f, cy_f, r_f = res_final.x[:n], res_final.x[n:2*n], res_final.x[2*n:]
                centers_f = np.column_stack((cx_f, cy_f))
                r_lp_f, s_lp_f = solve_lp_radii(centers_f, n)
                if r_lp_f is not None and s_lp_f > best_sum:
                    best_centers = centers_f
                    best_radii = r_lp_f
                    best_sum = s_lp_f
        except Exception:
            pass

    # Fallback configuration if optimization unexpectedly fails
    if best_centers is None:
        fallback = make_hex_init(n, [5, 6, 5, 6, 4], 0.09, rng)
        r_fb, s_fb = solve_lp_radii(fallback, n)
        best_centers = fallback
        best_radii = r_fb if r_fb is not None else np.full(n, 0.08)
        best_sum = s_fb if s_fb > 0 else np.sum(best_radii)

    # Final safety scaling to guarantee strict numerical validity for the checker
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
