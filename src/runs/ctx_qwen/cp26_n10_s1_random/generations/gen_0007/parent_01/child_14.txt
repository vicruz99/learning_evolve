import numpy as np
from scipy.optimize import minimize, linprog

def min_clearance(centers):
    """Computes the minimum clearance to boundaries and other circles."""
    n = centers.shape[0]
    b = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    return min(np.min(b), np.min(dists) / 2.0)

def neg_min_clearance(vars_arr, n):
    """Objective for Nelder-Mead: maximize min clearance."""
    return -min_clearance(vars_arr.reshape(n, 2))

def solve_lp_radii(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def objective_joint(vars_arr, n):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(vars_arr[2*n:])

def constraints_joint(vars_arr, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    idx = np.triu_indices(n, k=1)
    d2 = dx[idx]**2 + dy[idx]**2
    rs2 = dr[idx]**2
    return np.concatenate([c, d2 - rs2])

        [4,6,6,5,5], [5,4,6,6,5], [6,4,6,5,5], [4,5,6,6,5], [5,6,4,6,5],
        [5,5,5,5,4,2], [5,5,5,4,5,2], [4,6,5,6,5], [5,4,5,6,6], [6,5,4,6,5]
    ]
    
    for p in patterns:
        if sum(p) < n: continue
        for r0 in [0.075, 0.085, 0.095, 0.105]:
            pts = []
            y = r0

                x = r0 + shift

                    pts.append([x, y])
                    x += 2 * r0

    scale = 1.0

        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
