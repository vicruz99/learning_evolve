import numpy as np
from scipy.optimize import minimize, linprog

def obj_slsqp(vars_arr, n):
    """Objective: minimize negative sum of radii => maximize sum of radii"""
    return -np.sum(vars_arr[2 * n:])

def constraints_slsqp(vars_arr, n):
    """
    Computes inequality constraints >= 0 for valid packing.
    Layout: [x0, y0, ..., x25, y25, r0, r1, ..., r25]
    """
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    cx_m = cx[:, None]
    cy_m = cy[:, None]
    r_m = r[:, None]
    
    dx = cx_m - cx_m.T
    dy = cy_m - cy_m.T
    dr = r_m + r_m.T
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    d2 = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    rs2 = dr[i_idx, j_idx]**2
    
    return np.concatenate([c, d2 - rs2])

def solve_lp_sum(centers, n, A_ub, idx):
    """Solves the LP to maximize sum of radii for fixed centers."""
    # Maximum radius allowed by boundaries for each circle
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    b_ub = dists[idx[0], idx[1]]
    
    try:
        # Maximize sum(r) <=> Minimize -sum(r)
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0
