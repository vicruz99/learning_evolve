5. **Safety Scaling & Validation**: A final deterministic scaling ensures strict compliance with the $10^{-12}$ tolerance in the validation function.

import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j) and 0 <= r_i <= dist(i, boundary)
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Boundary limits for each circle
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    # Pairwise distance constraints: r_i + r_j <= d_ij
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx_i, idx_j]
    
    for k, (i, j) in enumerate(zip(idx_i, idx_j)):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            radii = np.maximum(res.x, 0.0)
            return radii, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def sa_optimize(centers, n, rng, steps=2500):
    """
    Simulated Annealing to optimize center positions.
    Evaluates LP for radii at each step.
    """
    curr_centers = centers.copy()
    curr_radii, curr_sum = solve_lp_radii(curr_centers)
    
    best_centers = curr_centers.copy()
    best_sum = curr_sum
    
    step_size = 0.045
    temp = 0.045
    
    for _ in range(steps):
        # Pick a random circle to perturb
        i = rng.integers(n)
        old_pos = curr_centers[i].copy()
        
        # Propose new position
        move = rng.uniform(-step_size, step_size, 2)
        new_pos = np.clip(old_pos + move, 0.02, 0.98)
        curr_centers[i] = new_pos
        
        new_radii, new_sum = solve_lp_radii(curr_centers)
        
        # Acceptance criterion
        delta = new_sum - curr_sum
        if delta > 0 or rng.random() < np.exp(delta / (temp + 1e-12)):
            curr_sum = new_sum
            curr_radii = new_radii
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = curr_centers.copy()
        else:
            curr_centers[i] = old_pos # Revert
            
        # Cool down
        step_size *= 0.996
        temp *= 0.996
        
    return best_centers, best_sum

def objective_joint(vars_arr, n):
    """Objective: minimize negative sum of radii"""
    return -np.sum(vars_arr[2 * n:])

def constraints_joint(vars_arr, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    
    d2 = dx**2 + dy**2
    rs2 = dr**2
    return np.concatenate([c, d2 - rs2])

def slsqp_polish(centers, radii, n, rng):
    """Joint optimization of centers and radii using SLSQP."""
    x0 = np.zeros(3 * n)
    x0[:n] = centers[:, 0]
    x0[n:2*n] = centers[:, 1]
    x0[2*n:] = radii * 0.98
    
    # Add small perturbation to avoid stagnation
    x0 += rng.uniform(-0.002, 0.002, 3 * n)
    x0[2*n:] = np.maximum(x0[2*n:], 1e-6) # Ensure positive radii
    
    bounds_vars = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    cons_dict = {'type': 'ineq', 'fun': constraints_joint, 'args': (n,)}
    
    try:
        res = minimize(objective_joint, x0, args=(n,), method='SLSQP', 
                       bounds=bounds_vars, constraints=cons_dict,
                       options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
        if np.isfinite(res.fun):
            cx = res.x[:n]
            cy = res.x[n:2*n]
            r = res.x[2*n:]
            return np.column_stack((cx, cy)), r, -res.fun
    except Exception:
        pass
    return centers, radii, 0.0

def make_hex_init(n, rows_pattern, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append(rng.uniform(r0, 1-r0, 2).tolist())
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(2024)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Diverse row distributions
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 5, 5, 6, 5], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 5, 6, 4, 6]
    ]
    
    inits = []
    for p in patterns:
        if sum(p) < n: continue
        for r0 in [0.085, 0.095, 0.105]:
            inits.append(make_hex_init(n, p, r0, rng))
            
    # Grid and corner variations
    g = np.linspace(0.14, 0.86, 5)
    grid = np.array([[x, y] for y in g for x in g])
    for extra in [[0.5, 0.5], [0.08, 0.92], [0.92, 0.08], [0.5, 0.92], [0.92, 0.92], [0.08, 0.08]]:
        inits.append(np.vstack([grid, extra])[:n])
        
    for _ in range(5):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Phase 1: Simulated Annealing on Centers
    for cfg in inits:
        opt_centers, opt_sum = sa_optimize(cfg, n, rng, steps=3000)
        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centers = opt_centers.copy()
            
    # Solve LP for best SA centers
    if best_centers is not None:
        best_radii, best_sum = solve_lp_radii(best_centers)
        
        # Phase 2: SLSQP Joint Polish
        polished_c, polished_r, polished_s = slsqp_polish(best_centers, best_radii, n, rng)
        if polished_s > best_sum:
            best_centers = polished_c
            best_radii = polished_r
            best_sum = polished_s
            
        # Phase 3: Final LP extraction to maximize radii strictly for fixed optimal centers
        final_r, final_s = solve_lp_radii(best_centers)
        if final_s > best_sum:
            best_radii = final_r
            best_sum = final_s
            
    # Fallback (should not be reached)
    if best_centers is None:
        fallback = make_hex_init(n, [5, 6, 5, 6, 4], 0.09, rng)
        best_centers = fallback
        best_radii, best_sum = solve_lp_radii(fallback)

    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
