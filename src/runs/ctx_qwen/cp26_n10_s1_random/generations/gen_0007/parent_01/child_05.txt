import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_structure(n):
    """Precomputes the constant constraint matrix for the radius LP."""
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    for k, (i, j) in enumerate(zip(idx_i, idx_j)):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
    return A_ub, idx_i, idx_j

def solve_lp_radii(centers, n, A_ub, idx_i, idx_j):
    """Solves the LP to maximize sum of radii for fixed centers."""
    # Maximum radius allowed by boundaries for each circle
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        # Maximize sum(r) <=> Minimize -sum(r)
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def coordinate_descent(centers, n, rng, A_ub, idx_i, idx_j, max_iter=3000, init_step=0.04):
    """Refines center positions by moving one circle at a time, re-optimizing radii via LP."""
    radii, curr_sum = solve_lp_radii(centers, n, A_ub, idx_i, idx_j)
    if radii is None:
        return centers, np.full(n, 1e-6), 0.0
        
    best_c = centers.copy()
    best_r = radii.copy()
    best_s = curr_sum
    step = init_step
    
    for it in range(max_iter):
        idx = rng.integers(n)
        old_pos = best_c[idx].copy()
        
        # Try multiple perturbations per iteration to increase acceptance rate
        improved = False
        for _ in range(4):
            new_pos = old_pos + rng.uniform(-step, step, 2)
            new_pos = np.clip(new_pos, 1e-5, 1.0 - 1e-5)
            best_c[idx] = new_pos
            
            r_new, s_new = solve_lp_radii(best_c, n, A_ub, idx_i, idx_j)
            if r_new is not None and s_new > best_s + 1e-9:
                best_r = r_new
                best_s = s_new
                improved = True
                step = min(step * 1.02, 0.15)  # slightly increase step on success
                break
            else:
                best_c[idx] = old_pos
                
        if not improved:
            step *= 0.996  # decay step on failure
            
        # Occasional larger jump to escape local minima
        if it > 0 and it % 400 == 0:
            step = max(step, 0.03)
            
    return best_c, best_r, best_s

def generate_inits(n, rng):
    """Generates diverse starting configurations."""
    inits = []
    
    # Hexagonal lattice patterns
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 5, 5, 6, 5], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 5, 6, 4, 6]
    ]
    for p in patterns:
        if sum(p) < n: continue
        for r0 in [0.085, 0.095, 0.105]:
            pts = []
            y = r0
            for ri, cnt in enumerate(p):
                shift = r0 if ri % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2 * r0
                y += np.sqrt(3) * r0
            while len(pts) < n:
                pts.append([0.5, 0.5])
            inits.append(np.array(pts[:n]))
            
    # Grid patterns
    for sp in [0.18, 0.20, 0.22]:
        g = np.linspace(0.5 - 2*sp, 0.5 + 2*sp, 5)
        grid = np.array([[x, y] for y in g for x in g])
        inits.append(np.vstack([grid, [0.5, 0.5]]))
        
    # Random dense placements
    for _ in range(8):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return inits

def objective_joint(vars_arr, n):
    """Objective: minimize negative sum of radii => maximize sum of radii"""
    return -np.sum(vars_arr[2 * n:])

def constraints_joint(vars_arr, n):
    """Inequality constraints >= 0 for valid joint packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    cx_m = cx[:, None]
    cy_m = cy[:, None]
    r_m = r[:, None]
    
    dx = cx_m - cx_m.T
    dy = cy_m - cy_m.T
    dr = r_m + r_m.T
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    d2 = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    rs2 = dr[i_idx, j_idx]**2
    
    return np.concatenate([c, d2 - rs2])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(123)
    A_ub, idx_i, idx_j = build_lp_structure(n)
    
    inits = generate_inits(n, rng)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Local Search from diverse starts
    for cfg in inits:
        c, r, s = coordinate_descent(cfg, n, rng, A_ub, idx_i, idx_j, max_iter=3000, init_step=0.04)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r
            
    # Phase 2: SLSQP Joint Refinement
    if best_centers is not None:
        x0 = np.zeros(3 * n)
        x0[:n] = best_centers[:, 0]
        x0[n:2*n] = best_centers[:, 1]
        x0[2*n:] = best_radii * 0.99
        
        bounds_vars = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
        cons_dict = {'type': 'ineq', 'fun': constraints_joint, 'args': (n,)}
        
        try:
            res = minimize(objective_joint, x0, args=(n,), method='SLSQP', bounds=bounds_vars,
                           constraints=cons_dict, options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            if np.isfinite(res.fun):
                cx, cy, r = res.x[:n], res.x[n:2*n], res.x[2*n:]
                centers = np.column_stack((cx, cy))
                r_lp, s_lp = solve_lp_radii(centers, n, A_ub, idx_i, idx_j)
                if r_lp is not None and s_lp > best_sum:
                    best_centers = centers
                    best_radii = r_lp
                    best_sum = s_lp
        except Exception:
            pass
            
        # Phase 3: Final Local Search on SLSQP result
        if best_centers is not None:
            c, r, s = coordinate_descent(best_centers, n, rng, A_ub, idx_i, idx_j, max_iter=1500, init_step=0.015)
            if s > best_sum:
                best_centers = c
                best_radii = r
                best_sum = s

    # Fallback
    if best_centers is None:
        best_centers = inits[0]
        best_radii, best_sum = solve_lp_radii(best_centers, n, A_ub, idx_i, idx_j)
        if best_radii is None: best_radii = np.full(n, 0.08)

    # Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
