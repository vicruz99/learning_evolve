import numpy as np
from scipy.optimize import linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    
    # Boundary limits for each circle
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    # Pairwise distance constraints: r_i + r_j <= d_ij
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        # Maximize sum(r) <=> Minimize -sum(r)
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                      method='highs', options={'presolve': True})
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def optimize_packing(centers, n, rng):
    """Local search combining coordinate descent and simulated annealing."""
    r_curr, s_curr = solve_lp_radii(centers)
    if r_curr is None:
        return centers, np.zeros(n), 0.0
        
    best_c = centers.copy()
    best_r = r_curr.copy()
    best_s = s_curr
    
    c_curr = centers.copy()
    
    # Phase 1: Coordinate Descent with adaptive step size
    step = 0.04
    for _ in range(25000):
        i = int(rng.random() * n)
        old = c_curr[i].copy()
        c_curr[i] += rng.uniform(-step, step, 2)
        c_curr[i] = np.clip(c_curr[i], 1e-5, 1.0 - 1e-5)
        
        r_new, s_new = solve_lp_radii(c_curr)
        if r_new is not None and s_new > s_curr:
            s_curr = s_new
            if s_new > best_s:
                best_s = s_new
                best_c = c_curr.copy()
                best_r = r_new.copy()
        else:
            c_curr[i] = old
            
        step *= 0.9996
        
    # Phase 2: Simulated Annealing to escape local minima
    temp = 0.025
    for _ in range(10000):
        # Perturb a random subset of circles
        k = int(rng.random() * (n // 3)) + 1
        idxs = rng.integers(0, n, size=k)
        old = c_curr[idxs].copy()
        c_curr[idxs] += rng.normal(0, temp, (k, 2))
        c_curr[idxs] = np.clip(c_curr[idxs], 1e-5, 1.0 - 1e-5)
        
        r_new, s_new = solve_lp_radii(c_curr)
        if r_new is not None:
            if s_new > s_curr or rng.random() < np.exp((s_new - s_curr) / (temp + 1e-6)):
                s_curr = s_new
                if s_new > best_s:
                    best_s = s_new
                    best_c = c_curr.copy()
                    best_r = r_new.copy()
            else:
                c_curr[idxs] = old
        else:
            c_curr[idxs] = old
            
        temp *= 0.998
        
    return best_c, best_r, best_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    # Generate diverse starting configurations
    inits = []
    
    # 1. Hexagonal lattices with various row distributions
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [5, 6, 4, 6, 5], [6, 6, 5, 5, 4]
    ]
    for p in patterns:
        for r0 in [0.085, 0.095, 0.105]:
            pts = []
            y = r0
            for ri, cnt in enumerate(p):
                shift = r0 if ri % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2 * r0
                y += np.sqrt(3) * r0
            while len(pts) < n:
                pts.append(rng.uniform(r0, 1-r0, 2).tolist())
            inits.append(np.array(pts[:n]))
            
    # 2. Random dense configurations
    for _ in range(6):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    for cfg in inits:
        # Small random perturbation to break exact symmetries before optimization
        cfg = np.clip(cfg + rng.uniform(-0.005, 0.005, (n, 2)), 0.05, 0.95)
        
        c_opt, r_opt, s_opt = optimize_packing(cfg, n, rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            best_radii = r_opt
            
    # Fallback if all optimizations fail unexpectedly
    if best_centers is None:
        best_centers = rng.uniform(0.2, 0.8, (n, 2))
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    # Final safety scaling to guarantee strict numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
