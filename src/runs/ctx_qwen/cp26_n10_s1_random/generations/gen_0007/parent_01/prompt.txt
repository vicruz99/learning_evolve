import numpy as np
from scipy.optimize import minimize

def get_clearances(centers, n):
    """Compute clearances to boundaries and half-pairwise distances."""
    # Distance to each boundary
    b_left = centers[:, 0]
    b_right = 1.0 - centers[:, 0]
    b_bottom = centers[:, 1]
    b_top = 1.0 - centers[:, 1]
    b = np.minimum(np.minimum(b_left, b_right), np.minimum(b_bottom, b_top))
    
    # Half distance to all other circles (upper triangle only)
    i_idx, j_idx = np.triu_indices(n, k=1)
    diffs = centers[i_idx] - centers[j_idx]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    p = dists / 2.0
    
    return np.concatenate([b, p])

def compute_smooth_min(centers, n, k):
    """Smooth approximation of the minimum clearance using log-sum-exp."""
    c = get_clearances(centers, n)
    max_c = np.max(c)
    # Stable computation: min(c) approx -1/k * log(sum(exp(-k*(c - max_c)))) - max_c
    exp_vals = np.exp(-k * (c - max_c))
    return -np.log(np.sum(exp_vals)) / k - max_c

    obj = -np.sum(r)
    
    # Boundary penalties: circle must be inside [0,1]x[0,1]
    # Violation occurs if r > x, r > 1-x, r > y, or r > 1-y
    b_pen = np.sum(np.maximum(0, r - x)**2)
    b_pen += np.sum(np.maximum(0, r - (1.0 - x))**2)
    b_pen += np.sum(np.maximum(0, r - y)**2)
    b_pen += np.sum(np.maximum(0, r - (1.0 - y))**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)

    r_sum = r[:, None] + r[None, :]

    triu_idx = np.triu_indices(n, k=1)
    o_pen = np.sum(overlap[triu_idx]**2)
