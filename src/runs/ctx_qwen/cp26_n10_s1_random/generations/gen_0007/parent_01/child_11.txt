

<strategy>
1. Decouple center and radius optimization: For any fixed set of centers, the optimal radii can be found exactly by solving a linear program (LP) that maximizes the sum of radii subject to non-overlap ($r_i + r_j \le d_{ij}$) and boundary constraints ($r_i \le \min(x_i, 1-x_i, y_i, 1-y_i)$).
2. Generate diverse initial center configurations using hexagonal lattice patterns with varying row distributions and densities, plus grid and random starts. This covers different topological arrangements.
3. Perform a robust local search on the centers using coordinate descent with Simulated Annealing (SA). This allows the algorithm to escape local minima by occasionally accepting worse configurations, while a decaying step size ensures convergence to a fine-tuned arrangement.
4. Include periodic random swaps of circle positions to break symmetries and explore disconnected regions of the configuration space.
5. After the search, apply a strict safety scaling factor to guarantee numerical validity against the $10^{-12}$ tolerance in the validator.
6. Run multiple restarts and track the best sum of radii found. The LP solver (`scipy.optimize.linprog` with `method='highs'`) is highly efficient for $N=26$, allowing thousands of evaluations within time limits.

import numpy as np
from scipy.optimize import linprog

def solve_lp_radii(centers, n, A_ub_template, triu_i, triu_j):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j) and 0 <= r_i <= dist(i, boundary)
    """
    # Maximum radius allowed by boundaries for each circle
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    # Pairwise distances
    c0 = centers[:, 0]
    c1 = centers[:, 1]
    d2 = (c0[:, None] - c0[None, :])**2 + (c1[:, None] - c1[None, :])**2
    np.fill_diagonal(d2, 1e18)
    dists = np.sqrt(d2)
    
    b_ub = dists[triu_i, triu_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_template, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def get_init_configs(n, rng):
    """Generates diverse initial center configurations."""
    configs = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5], [5, 5, 5, 5, 6],
        [5, 5, 6, 4, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [4, 5, 6, 5, 6]
    ]
    
    for p in patterns:
        if sum(p) < n: continue
        for r0 in [0.08, 0.09, 0.10]:
            pts = []
            y = r0
            for ri, cnt in enumerate(p):
                shift = r0 if ri % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2 * r0
                y += np.sqrt(3) * r0
            while len(pts) < n:
                pts.append(rng.uniform(r0, 1 - r0, 2).tolist())
            configs.append(np.array(pts[:n]))
            
    # Grid variations
    for sp in [0.15, 0.18, 0.20]:
        g = np.linspace(sp, 1.0 - sp, 5)
        grid = np.array([[x, y] for y in g for x in g])
        configs.append(grid)
        
    # Random dense placements
    for _ in range(5):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return configs

def local_search(centers, n, rng, A_ub_template, triu_i, triu_j, iters=10000):
    """Performs coordinate descent with simulated annealing to optimize centers."""
    best_c = centers.copy()
    best_r, best_s = solve_lp_radii(best_c, n, A_ub_template, triu_i, triu_j)
    if best_r is None:
        return best_c, np.full(n, 0.01), 0.0
        
    step = 0.04
    for it in range(iters):
        i = rng.integers(n)
        old = best_c[i].copy()
        
        # Perturb position
        delta = rng.uniform(-step, step, 2)
        new = old + delta
        new = np.clip(new, 0.02, 0.98)
        best_c[i] = new
        
        r_new, s_new = solve_lp_radii(best_c, n, A_ub_template, triu_i, triu_j)
        
        # Simulated Annealing acceptance criterion
        temp = 0.008 * (1.0 - it / iters)
        accepted = False
        if r_new is not None:
            if s_new > best_s + 1e-9:
                accepted = True
            elif temp > 1e-7 and rng.random() < np.exp((s_new - best_s) / temp):
                accepted = True
                
        if accepted:
            best_s = s_new
            best_r = r_new
            # Slightly increase step on success to explore
            if it % 500 == 0:
                step = min(0.06, step * 1.015)
        else:
            best_c[i] = old # Revert
            # Decay step on failure
            step *= 0.997
            
        # Periodic swap to escape symmetries
        if rng.random() < 0.015:
            i1, i2 = rng.choice(n, 2, replace=False)
            best_c[[i1, i2]] = best_c[[i2, i1]]
            r_swap, s_swap = solve_lp_radii(best_c, n, A_ub_template, triu_i, triu_j)
            if r_swap is not None and s_swap > best_s + 1e-9:
                best_s = s_swap
                best_r = r_swap
            else:
                best_c[[i1, i2]] = best_c[[i2, i1]] # Revert swap
                
    return best_c, best_r, best_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    # Precompute LP constraint matrix structure
    triu_i, triu_j = np.triu_indices(n, k=1)
    m = len(triu_i)
    A_ub_template = np.zeros((m, n))
    A_ub_template[np.arange(m), triu_i] = 1.0
    A_ub_template[np.arange(m), triu_j] = 1.0
    
    inits = get_init_configs(n, rng)
    
    best_c = None
    best_r = None
    best_s = -1.0
    
    # Run local search from each initialization
    for cfg in inits:
        c, r, s = local_search(cfg, n, rng, A_ub_template, triu_i, triu_j, iters=8000)
        if s > best_s:
            best_c, best_r, best_s = c, r, s
            
    # Fallback if all searches failed
    if best_c is None:
        best_c = rng.uniform(0.2, 0.8, (n, 2))
        best_r = np.full(n, 0.08)
        best_s = np.sum(best_r)
        
    # Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_r *= scale * 0.9999995
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
