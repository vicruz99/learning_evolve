import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, idx_i, idx_j, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    b_ub = dists[idx_i, idx_j]
    
    A_ub = np.zeros((len(idx_i), n))
    for k, (i, j) in enumerate(zip(idx_i, idx_j)):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def constraints_joint(vars_arr, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    cx_m = cx[:, None]
    cy_m = cy[:, None]
    r_m = r[:, None]
    
    dx = cx_m - cx_m.T
    dy = cy_m - cy_m.T
    dr = r_m + r_m.T
    
    d2 = dx[idx_i, idx_j]**2 + dy[idx_i, idx_j]**2
    rs2 = dr[idx_i, idx_j]**2
    
    return np.concatenate([c, d2 - rs2])

def objective_joint(vars_arr, n):
    """Objective: minimize negative sum of radii => maximize sum of radii"""
    return -np.sum(vars_arr[2 * n:])

def force_directed_pack(centers, n, steps=1500, dt=0.02, stiff=150.0, growth=0.00015):
    """Physics-based relaxation to jam circles densely."""
    r = np.full(n, 0.04)
    for _ in range(steps):
        r += growth
        forces = np.zeros((n, 2))
        
        diffs = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        
        min_dists = r[:, None] + r[None, :]
        overlap = np.maximum(0, min_dists - dists)
        mask = overlap > 0
        
        dirs = diffs / (dists[:, :, None] + 1e-10)
        dirs[~mask[:, :, None]] = 0.0
        
        f_mags = overlap * stiff
        f_mags[~mask] = 0.0
        
        forces += np.sum(f_mags[:, :, None] * dirs, axis=1)
        
        x, y = centers[:, 0], centers[:, 1]
        forces[:, 0] += np.maximum(0, r - x) * stiff
        forces[:, 0] -= np.maximum(0, x + r - 1.0) * stiff
        forces[:, 1] += np.maximum(0, r - y) * stiff
        forces[:, 1] -= np.maximum(0, y + r - 1.0) * stiff
        
        centers += forces * dt
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
        
    return centers

def make_hex_init(n, rows_pattern, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append(rng.uniform(r0, 1-r0, 2).tolist())
    return np.array(pts[:n])
