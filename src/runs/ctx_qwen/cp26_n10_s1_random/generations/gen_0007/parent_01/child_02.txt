import numpy as np
from scipy.optimize import minimize, linprog

# Precompute constant matrices for performance
N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = N * (N - 1) // 2
A_LP = np.zeros((M_PAIRS, N))
A_LP[np.arange(M_PAIRS), TRIU_I] = 1.0
A_LP[np.arange(M_PAIRS), TRIU_J] = 1.0

def get_constraints(vars_arr, n):
    """Computes inequality constraints >= 0 for valid packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    cx_m = cx[:, None]
    cy_m = cy[:, None]
    r_m = r[:, None]
    
    dx = cx_m - cx_m.T
    dy = cy_m - cy_m.T
    dr = r_m + r_m.T
    
    d2 = dx[TRIU_I, TRIU_J]**2 + dy[TRIU_I, TRIU_J]**2
    rs2 = dr[TRIU_I, TRIU_J]**2
    
    return np.concatenate([c, d2 - rs2])

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    b_ub = dists[TRIU_I, TRIU_J]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def make_hex_init(n, rows_pattern, r0, rng):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append(rng.uniform(r0, 1-r0, 2).tolist())
    return np.array(pts[:n])

def obj_slsqp(vars_arr):
    """Objective: minimize negative sum of radii => maximize sum of radii"""
    return -np.sum(vars_arr[2*N:])

def cons_slsqp(vars_arr):
    """Inequality constraints for SLSQP"""
    return get_constraints(vars_arr, N)

    return best_centers, best_radii, best_sum
