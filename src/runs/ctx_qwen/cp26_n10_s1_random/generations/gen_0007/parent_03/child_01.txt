1. **Center Placement (Non-convex)**: The positions of the circles are optimized using an adaptive stochastic hill-climbing algorithm. Instead of optimizing centers and radii jointly (which leads to difficult non-convex constraints), we treat centers as the primary variables.
2. **Radius Allocation (Convex)**: For any fixed set of centers, finding the optimal radii that maximize the sum is a linear programming (LP) problem. The constraints are linear: `r_i <= min(x_i, 1-x_i, y_i, 1-y_i)` and `r_i + r_j <= dist(i, j)`. This LP is solved efficiently using `scipy.optimize.linprog` with the `highs` solver.

**Strategy Details**:
- **Precomputation**: The sparse structure of the LP constraint matrix `A_ub` is precomputed once, as it only depends on the number of circles and pairwise relationships, not their positions.
- **Initialization**: We generate diverse starting configurations including hexagonal lattices at varying densities, a regular grid with a center point, and random dense placements. This helps cover the search space and avoid poor local minima.
- **Adaptive Hill-Climbing**: We iteratively pick a random circle, try several random displacements, and accept the move if the LP-sum-of-radii improves. The step size decays over time. If no improvement occurs after a threshold of steps, we perturb all circles globally to escape local optima.
- **Refinement & Safety**: After the stochastic search converges, we run a final refinement pass. We then compute a strict safety scaling factor to guarantee all constraints satisfy the validator's `1e-12` tolerance, shrinking radii slightly if necessary.

import numpy as np
from scipy.optimize import linprog

def compute_lp_objective(centers, A_ub, n, triu_idx):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns (optimal_radii, sum_radii) or (None, 0.0) if infeasible.
    """
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Boundary constraints RHS: r_i <= x, 1-x, y, 1-y
    b_bdry = np.empty(4 * n)
    for i in range(n):
        b_bdry[4*i]     = x[i]
        b_bdry[4*i + 1] = 1.0 - x[i]
        b_bdry[4*i + 2] = y[i]
        b_bdry[4*i + 3] = 1.0 - y[i]
        
    # Pairwise distance constraints RHS: r_i + r_j <= dist(i,j)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_pairs = dists[triu_idx]
    
    b_ub = np.concatenate([b_bdry, b_pairs])
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def optimize_centers(centers_init, A_ub, n, triu_idx, rng, max_iter=600):
    """
    Adaptive stochastic hill-climbing to optimize center positions.
    Uses LP evaluation for the true objective value at each step.
    """
    centers = centers_init.copy()
    radii, best_obj = compute_lp_objective(centers, A_ub, n, triu_idx)
    if radii is None:
        return centers, 0.0
        
    current_obj = best_obj
    step = 0.012
    stagnation = 0
    
    for _ in range(max_iter):
        idx = rng.integers(n)
        old_pos = centers[idx].copy()
        best_local_obj = current_obj
        best_local_pos = None
        
        # Try multiple perturbations for this circle
        for _ in range(6):
            delta = rng.uniform(-step, step, 2)
            new_pos = np.clip(old_pos + delta, 1e-4, 1.0 - 1e-4)
            centers[idx] = new_pos
            
            _, obj = compute_lp_objective(centers, A_ub, n, triu_idx)
            if obj > best_local_obj + 1e-9:
                best_local_obj = obj
                best_local_pos = new_pos.copy()
                
        if best_local_pos is not None:
            centers[idx] = best_local_pos
            radii, current_obj = compute_lp_objective(centers, A_ub, n, triu_idx)
            stagnation = 0
            step = max(0.0005, step * 0.998)
        else:
            centers[idx] = old_pos
            stagnation += 1
            if stagnation > 50:
                # Global perturbation to escape local minima
                centers += rng.uniform(-0.006, 0.006, (n, 2))
                centers = np.clip(centers, 0.02, 0.98)
                radii, current_obj = compute_lp_objective(centers, A_ub, n, triu_idx)
                if radii is None:
                    centers = centers_init.copy()
                    radii, current_obj = compute_lp_objective(centers, A_ub, n, triu_idx)
                stagnation = 0
                
    return centers, current_obj

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    triu_idx = np.triu_indices(n, k=1)
    
    # Precompute A_ub structure for LP: r_i <= boundary, r_i + r_j <= dist
    A_bdry = np.zeros((4 * n, n))
    for i in range(n):
        A_bdry[4*i, i] = 1.0
        A_bdry[4*i+1, i] = 1.0
        A_bdry[4*i+2, i] = 1.0
        A_bdry[4*i+3, i] = 1.0
        
    A_pairs = np.zeros((len(triu_idx[0]), n))
    for k in range(len(triu_idx[0])):
        i, j = triu_idx[0][k], triu_idx[1][k]
        A_pairs[k, i] = 1.0
        A_pairs[k, j] = 1.0
        
    A_ub = np.vstack([A_bdry, A_pairs])
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattices at varying densities
    for r0 in [0.085, 0.095, 0.105, 0.110]:
        pts = []
        y = r0
        row = 0
        while y + r0 < 1.0 and len(pts) < n:
            shift = r0 if row % 2 == 1 else 0.0
            x = r0 + shift
            while x + r0 < 1.0 and len(pts) < n:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        while len(pts) < n:
            pts.append([0.5, 0.5])
        inits.append(np.array(pts[:n]))
        
        # Perturbed hex to break symmetry
        pert = np.array(pts[:n]) + rng.uniform(-0.012, 0.012, (n, 2))
        inits.append(np.clip(pert, 0.05, 0.95))
        
    # 2. Grid + center
    grid = np.array([[0.1 + i*0.18, 0.1 + j*0.18] for j in range(5) for i in range(5)])
    grid = np.vstack([grid, [0.5, 0.5]])
    inits.append(grid)
    
    # 3. Random dense placements
    for _ in range(4):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Optimize each configuration
    for cfg in inits:
        centers_opt, obj = optimize_centers(cfg, A_ub, n, triu_idx, rng, max_iter=600)
        if obj > best_sum:
            best_sum = obj
            best_centers = centers_opt.copy()
            best_radii, _ = compute_lp_objective(best_centers, A_ub, n, triu_idx)
            
    # Refine the best found configuration with a fresh RNG seed and longer run
    if best_centers is not None:
        rng2 = np.random.default_rng(99)
        centers_opt, obj = optimize_centers(best_centers, A_ub, n, triu_idx, rng2, max_iter=1000)
        if obj > best_sum:
            best_sum = obj
            best_centers = centers_opt.copy()
            best_radii, _ = compute_lp_objective(best_centers, A_ub, n, triu_idx)
            
    # Final safety scaling to guarantee strict numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for k in range(len(triu_idx[0])):
        i, j = triu_idx[0][k], triu_idx[1][k]
        d = np.linalg.norm(best_centers[i] - best_centers[j])
        rs = best_radii[i] + best_radii[j]
        if rs > 1e-12:
            scale = min(scale, d/rs)
            
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
