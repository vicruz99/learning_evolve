from scipy.optimize import minimize, linprog

def compute_constraints(vars_arr, n, idx_i, idx_j):
    """Computes inequality constraints >= 0 for valid packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    c = []
    # Boundary constraints
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    # Pairwise non-overlap constraints (squared distance >= squared sum of radii)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c.extend(dx**2 + dy**2 - dr**2)
    
    return np.array(c)

def objective_func(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def solve_lp_radii(centers, n, idx_i, idx_j):
    """Solve LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    m = 4 * n + len(idx_i)
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    bounds = []
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        mx = min(x, 1.0-x, y, 1.0-y)
        bounds.append((0.0, max(mx, 1e-9)))
        
    for idx in range(len(idx_i)):
        i, j = idx_i[idx], idx_j[idx]
        d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = d
        k += 1
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, -np.inf
