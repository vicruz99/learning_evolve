

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we treat the problem as a two-stage optimization:
1. **LP Oracle for Radii**: For any fixed set of centers, finding the optimal radii is a linear programming problem. We maximize $\sum r_i$ subject to boundary constraints ($r_i \le \text{dist to wall}$) and pairwise non-overlap constraints ($r_i + r_j \le \text{dist}(i,j)$). Using `scipy.optimize.linprog` with the `highs` solver provides an exact, fast oracle.
2. **Stochastic Center Search**: We optimize the 52 center coordinates using hill-climbing. In each iteration, we randomly perturb 1 or 2 circles, query the LP oracle, and accept the move if the sum of radii improves. An exponentially decaying step size ensures coarse exploration early on and fine-grained polishing later.
3. **Diverse Initializations**: We seed the search with multiple hexagonal lattice patterns (different row counts, densities, and offsets) scaled to the square, along with random dense configurations. This prevents getting trapped in poor local minima.
4. **Safety Scaling**: A final pass computes the maximum valid scaling factor for the best found configuration, ensuring strict compliance with the $10^{-12}$ numerical tolerance.

This combination of exact radius resolution and robust center-space exploration is highly effective for dense circle packing problems.

import numpy as np
from scipy.optimize import linprog

def solve_lp(n, A_ub, bounds_lp, c_obj, b_ub, triu_idx, centers):
    """Compute optimal radii and their sum for fixed centers using LP."""
    # Boundary constraints
    b_ub[:n] = centers[:, 0]
    b_ub[n:2*n] = 1.0 - centers[:, 0]
    b_ub[2*n:3*n] = centers[:, 1]
    b_ub[3*n:4*n] = 1.0 - centers[:, 1]
    
    # Pairwise distance constraints
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub[4*n:] = dists[triu_idx]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def hill_climbing(init_centers, rng, max_iter, init_step, n, A_ub, bounds_lp, c_obj, b_ub, triu_idx):
    """Local search on centers to maximize LP-optimized sum of radii."""
    centers = init_centers.copy()
    radii, best_sum = solve_lp(n, A_ub, bounds_lp, c_obj, b_ub, triu_idx, centers)
    best_radii = radii.copy()
    step = init_step
    
    for _ in range(max_iter):
        # Perturb 1 or 2 circles randomly
        k = 1 if rng.random() < 0.8 else 2
        idxs = rng.choice(n, k, replace=False)
        old_pos = centers[idxs].copy()
        
        centers[idxs] += rng.uniform(-step, step, (k, 2))
        centers[idxs] = np.clip(centers[idxs], 0.02, 0.98)
        
        radii_new, new_sum = solve_lp(n, A_ub, bounds_lp, c_obj, b_ub, triu_idx, centers)
        
        if new_sum > best_sum + 1e-12:
            best_sum = new_sum
            best_radii = radii_new
        else:
            centers[idxs] = old_pos
            
        step *= 0.9985
        
    return centers, best_radii, best_sum

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    # Precompute constant LP constraint structure
    triu_idx = np.triu_indices(n, k=1)
    n_pairs = len(triu_idx[0])
    m_total = 4 * n + n_pairs
    
    A_ub = np.zeros((m_total, n))
    for i in range(n):
        A_ub[i, i] = 1.0          # r_i <= x_i
        A_ub[n+i, i] = 1.0        # r_i <= 1-x_i
        A_ub[2*n+i, i] = 1.0      # r_i <= y_i
        A_ub[3*n+i, i] = 1.0      # r_i <= 1-y_i
    for k, (i, j) in enumerate(zip(triu_idx[0], triu_idx[1])):
        A_ub[4*n+k, i] = 1.0      # r_i + r_j <= d_ij
        A_ub[4*n+k, j] = 1.0
        
    bounds_lp = [(0.0, None)] * n
    c_obj = -np.ones(n)
    b_ub = np.zeros(m_total)
    
    # Generate diverse initial configurations
    inits = []
    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [5, 6, 4, 6, 5], [6, 6, 5, 5, 4],
        [5, 7, 4, 5, 5], [5, 5, 5, 5, 6]
    ]
    
    for pat in row_patterns:
        if sum(pat) != n: continue
        for r0 in [0.09, 0.10, 0.115]:
            pts = []
            y = r0
            for idx, cnt in enumerate(pat):
                shift = r0 if idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            while len(pts) < n:
                pts.append([0.5, 0.5])
            base = np.array(pts[:n])
            # Normalize to fit comfortably inside [0.05, 0.95]
            mn = base.min(axis=0)
            mx = base.max(axis=0)
            span = mx - mn
            if span[0] > 0: base[:, 0] = (base[:, 0] - mn[0]) / span[0] * 0.85 + 0.075
            if span[1] > 0: base[:, 1] = (base[:, 1] - mn[1]) / span[1] * 0.85 + 0.075
            inits.append(base)
            inits.append(np.clip(base + rng.uniform(-0.025, 0.025, (n, 2)), 0.05, 0.95))
            
    for _ in range(6):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Run local search from each initialization
    for cfg in inits:
        c_opt, r_opt, s_opt = hill_climbing(cfg, rng, 2500, 0.03, n, A_ub, bounds_lp, c_obj, b_ub, triu_idx)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
