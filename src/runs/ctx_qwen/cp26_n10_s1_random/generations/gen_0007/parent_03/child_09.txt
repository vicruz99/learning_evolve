import numpy as np
from scipy.optimize import linprog, minimize

def compute_lp_sum(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j), r_i <= distance to boundaries.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    n_pair = n * (n - 1) // 2
    n_bound = 4 * n
    A_ub = np.zeros((n_pair + n_bound, n))
    b_ub = np.zeros(n_pair + n_bound)
    
    idx = 0
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        lims = (x, 1.0 - x, y, 1.0 - y)
        for lim in lims:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-7), 0.0

def circle_opt_obj(pos, centers, idx):
    """
    Objective for optimizing a single circle's center.
    Minimizes negative sum of radii (thus maximizing it).
    Includes boundary penalties to keep centers inside [0.005, 0.995].
    """
    if pos[0] < 0.005 or pos[0] > 0.995 or pos[1] < 0.005 or pos[1] > 0.995:
        return 1e6
        
    c = centers.copy()
    c[idx] = pos
    _, s = compute_lp_sum(c)
    return -s

    rng = np.random.default_rng(42)
    
    best_sum = -1.0

    return best_centers, best_radii, best_sum
