import numpy as np
from scipy.optimize import linprog

def solve_lp(n, A_ub, num_boundary, centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    num_pairwise = n * (n - 1) // 2
    b_ub = np.empty(num_boundary + num_pairwise)
    bounds = []
    
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mx, 1e-9)))
        b_ub[i] = x
        b_ub[n + i] = 1.0 - x
        b_ub[2*n + i] = y
        b_ub[3*n + i] = 1.0 - y
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    idx = num_boundary
    for i in range(n):
        for j in range(i + 1, n):
            b_ub[idx] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 1e-9

def run_packing():
    n = 26
    rng = np.random.default_rng(42)

    for i in range(n):
        A_ub[i, i] = 1.0
        A_ub[n + i, i] = 1.0
        A_ub[2*n + i, i] = 1.0
        A_ub[3*n + i, i] = 1.0
        
    idx = num_boundary
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
            
    # Generate diverse initial configurations
    inits = []
    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [5, 6, 4, 6, 5], [6, 4, 5, 6, 5],
        [5, 5, 5, 5, 6], [5, 7, 4, 5, 5], [6, 6, 5, 5, 4], [5, 6, 6, 5, 4]
    ]
    
    for pat in row_patterns:
        if sum(pat) != n: continue

            y = scale
            row_idx = 0
            for cnt in pat:

                x = scale + shift

                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2.0 * scale
                y += np.sqrt(3) * scale
                row_idx += 1
            while len(pts) < n:
                pts.append([0.5, 0.5])
            inits.append(np.array(pts[:n]))
            
    for _ in range(8):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Coordinate descent optimization on centers evaluated via LP
    for cfg in inits:
        # Small perturbation to break symmetry
        cfg = cfg + rng.uniform(-0.005, 0.005, (n, 2))
        cfg = np.clip(cfg, 0.02, 0.98)
        
        curr_c = cfg.copy()
        _, curr_s = solve_lp(n, A_ub, num_boundary, curr_c)

    else:
        final_sum = 0.0
        
    return best_centers, best_radii, final_sum
