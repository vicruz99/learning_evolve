import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        A[4*i, i] = 1.0; b[4*i] = x
        A[4*i+1, i] = 1.0; b[4*i+1] = 1.0 - x
        A[4*i+2, i] = 1.0; b[4*i+2] = y
        A[4*i+3, i] = 1.0; b[4*i+3] = 1.0 - y
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[4*n + idx, i] = 1.0
            A[4*n + idx, j] = 1.0
            b[4*n + idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-8), 0.0

def obj_joint(vars_arr):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars_arr[52:])

def cons_joint(vars_arr):
    """Inequality constraints >= 0 for SLSQP."""
    n = 26
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    triu = np.triu_indices(n, k=1)
    c = np.concatenate([c, d2[triu] - rs[triu]**2])
    return c

def generate_hex_configs(rng):
    """Generates diverse hexagonal lattice configurations."""
    configs = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [5, 6, 4, 6, 5], [6, 4, 6, 5, 5],
        [5, 7, 5, 4, 5], [4, 5, 6, 6, 5], [6, 5, 5, 6, 4],
        [5, 5, 5, 5, 6], [6, 6, 5, 5, 4]
    ]
    
    for pat in patterns:
        if sum(pat) != 26:
            continue
        for r0 in [0.075, 0.085, 0.095, 0.105]:
            pts = []
            y = r0
            row = 0
            for cnt in pat:
                shift = r0 if row % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= 26:
                        break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
                row += 1
                
            if len(pts) >= 26:
                c = np.array(pts[:26])
                # Center and scale to fit comfortably inside [0.1, 0.9]
                mn = c.min(axis=0)
                mx = c.max(axis=0)
                span = mx - mn
                if span[0] > 0.001:
                    c[:, 0] = (c[:, 0] - mn[0]) / span[0] * 0.8 + 0.1
                if span[1] > 0.001:
                    c[:, 1] = (c[:, 1] - mn[1]) / span[1] * 0.8 + 0.1
                    
                configs.append(c)
                # Add perturbed variants to break symmetry
                p = c + rng.uniform(-0.025, 0.025, (26, 2))
                configs.append(np.clip(p, 0.05, 0.95))
                
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    rng = np.random.default_rng(42)
    
    configs = generate_hex_configs(rng)

    for _ in range(6):

    cons = {'type': 'ineq', 'fun': cons_joint}

        dists_wall = np.minimum(np.minimum(cfg[:, 0], 1.0 - cfg[:, 0]),

        dists_nn = np.sqrt(np.sum(diffs**2, axis=2))

        r_init = np.maximum(r_init, 0.02)

        try:
            res = minimize(obj_joint, x0, method='SLSQP', bounds=bounds, constraints=cons,

                c_opt = np.column_stack((res.x[:n], res.x[n:2*n]))

    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:

            d = np.hypot(best_centers[i, 0] - best_centers[j, 0],

            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
