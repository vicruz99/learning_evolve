import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    bounds = []
    
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        lims = (x, 1.0 - x, y, 1.0 - y)
        mx = min(lims)
        bounds.append((0.0, max(mx, 1e-9)))
        for lim in lims:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(lim)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    for i, j in zip(idx_i, idx_j):
        d = dists[i, j]
        row = np.zeros(n)
        row[i] = 1.0
        row[j] = 1.0
        A_ub.append(row)
        b_ub.append(d)
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def generate_starts(n, rng):
    """Generates diverse initial center configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [4, 6, 6, 6, 4],
        [5, 5, 6, 5, 5], [5, 6, 4, 6, 5], [6, 4, 5, 6, 5],
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 7, 4, 5, 5],
        [4, 5, 6, 5, 6], [5, 4, 6, 5, 6]
    ]
    
    for pat in patterns:
        if sum(pat) != n: continue
        pts = []
        y = 0.1
        row_idx = 0
        for cnt in pat:
            shift = 0.1 if row_idx % 2 == 1 else 0.0
            x = 0.1 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 0.2
            y += 0.1732 # sqrt(3)/2 * 0.2
            row_idx += 1
        pts = np.array(pts[:n])
        starts.append(pts)
        
        # Perturbed variants to break symmetry
        p = pts + rng.uniform(-0.02, 0.02, (n, 2))
        starts.append(np.clip(p, 0.05, 0.95))
        
    # Random dense placements
    for _ in range(10):
        starts.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return starts

def simulated_annealing(centers, n, rng, steps=3500):
    """Simulated annealing to optimize centers, evaluating quality via LP."""
    best_centers = centers.copy()
    radii, best_sum = solve_lp(centers)
    if radii is None:
        return centers, np.zeros(n), 0.0
        
    curr_centers = centers.copy()
    curr_sum = best_sum
    t_init = 0.005
    
    for k in range(steps):
        temp = t_init * (0.02 ** (k / steps))
        step_size = 0.008 * (0.15 ** (k / steps))
        
        i = rng.integers(n)
        old_pos = curr_centers[i].copy()
        curr_centers[i] = old_pos + rng.uniform(-step_size, step_size, 2)
        curr_centers[i] = np.clip(curr_centers[i], 0.02, 0.98)
        
        new_radii, new_sum = solve_lp(curr_centers)
        if new_radii is None:
            curr_centers[i] = old_pos
            continue
            
        delta = new_sum - curr_sum
        # Metropolis criterion
        if delta > 0 or rng.random() < np.exp(min(delta / temp, 500.0)):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = curr_centers.copy()
        else:
            curr_centers[i] = old_pos
            
    radii, _ = solve_lp(best_centers)
    return best_centers, radii, best_sum

            return -np.sum(v[2*n:])

            dy = cy[idx_i] - cy[idx_j]
            dr = r[idx_i] + r[idx_j]

            return c
            
        bounds_opt = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
