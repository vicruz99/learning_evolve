import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mx, 1e-9)))
        
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def neg_lp_objective(centers_flat, n):
    """Objective for derivative-free optimization: minimize negative LP sum."""
    centers = centers_flat.reshape(n, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    _, s = solve_lp_radii(centers)
    return -s

def generate_configs(n, rng):
    """Generate diverse starting configurations for optimization."""
    configs = []
    hex_pats = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4]]
    for r0 in [0.09, 0.10, 0.11]:
        for pat in hex_pats:
            pts = []
            y = r0
            for idx, cnt in enumerate(pat):
                shift = r0 if idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) < n:
                        pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            while len(pts) < n:
                pts.append([0.5, 0.5])
            configs.append(np.array(pts[:n]))
            configs.append(np.clip(pts[:n] + rng.uniform(-0.015, 0.015, (n, 2)), 0.05, 0.95))
            
    # Corner and edge focused pattern
    base = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85],
            [0.5, 0.12], [0.5, 0.88], [0.12, 0.5], [0.88, 0.5], [0.5, 0.5]]
    gx = np.linspace(0.25, 0.75, 4)
    gy = np.linspace(0.25, 0.75, 4)
    for y in gy:
        for x in gx:
            if len(base) < n:
                base.append([x, y])
    configs.append(np.array(base[:n]))
    configs.append(np.clip(base[:n] + rng.uniform(-0.02, 0.02, (n, 2)), 0.05, 0.95))
    
    for _ in range(4):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return configs

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    configs = generate_configs(n, rng)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Derivative-free optimization on centers using Powell
    # Powell efficiently handles the piecewise-linear LP objective landscape
    for cfg in configs:
        try:
            res = minimize(neg_lp_objective, cfg.flatten(), args=(n,), 
                          method='Powell', 
                          options={'maxiter': 600, 'ftol': 1e-13, 'disp': False})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = res.x.reshape(n, 2).copy()
                _, best_radii = solve_lp_radii(best_centers)
        except Exception:
            continue
            
    # Phase 2: Joint SLSQP refinement to exploit smooth constraint gradients
    if best_centers is not None:
        x0 = np.zeros(3 * n)
        x0[:n] = best_centers[:, 0]
        x0[n:2*n] = best_centers[:, 1]
        x0[2*n:] = best_radii * 0.99
        
        bounds_opt = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
        
        def obj_joint(v):
            return -np.sum(v[2*n:])
            
        def cons_joint(v):
            cx = v[:n]
            cy = v[n:2*n]
            r = v[2*n:]
            c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
            dx = cx[:, None] - cx[None, :]
            dy = cy[:, None] - cy[None, :]
            d2 = dx**2 + dy**2
            rs = r[:, None] + r[None, :]
            idx = np.triu_indices(n, k=1)
            c = np.concatenate([c, d2[idx] - rs[idx]**2])
            return c
            
        cons_dict = {'type': 'ineq', 'fun': cons_joint}
        try:
            res_j = minimize(obj_joint, x0, method='SLSQP', bounds=bounds_opt,
                            constraints=cons_dict, options={'maxiter': 6000, 'ftol': 1e-13})
            if np.isfinite(res_j.fun):
                c_j = np.column_stack((res_j.x[:n], res_j.x[n:2*n]))
                r_j, s_j = solve_lp_radii(c_j)
                if s_j > best_sum:
                    best_sum = s_j
                    best_centers = c_j.copy()
                    best_radii = r_j.copy()
        except Exception:
            pass
            
    # Phase 3: Stochastic hill climbing on centers for fine-grained adjustment
    if best_centers is not None:
        step = 0.004
        for _ in range(800):
            i = rng.integers(n)
            old = best_centers[i].copy()
            best_centers[i] += rng.uniform(-step, step, 2)
            best_centers[i] = np.clip(best_centers[i], 0.02, 0.98)
            
            r_new, s_new = solve_lp_radii(best_centers)
            if s_new > best_sum + 1e-9:
                best_sum = s_new
                best_radii = r_new.copy()
            else:
                best_centers[i] = old
            step *= 0.994
            
    # Final safety scaling to guarantee strict numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
