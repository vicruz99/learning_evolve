import numpy as np
from scipy.optimize import minimize, linprog

def obj_sum_radii(v):
    """Objective: minimize negative sum of radii => Maximize sum of radii"""
    n = len(v) // 3
    return -np.sum(v[2*n:])

def cons_joint(v):
    """Inequality constraints >= 0 for valid joint packing"""
    n = len(v) // 3
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    
    idx = np.triu_indices(n, k=1)
    dx = cx[idx[0]] - cx[idx[1]]
    dy = cy[idx[0]] - cy[idx[1]]
    dr = r[idx[0]] + r[idx[1]]
    
    return np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r,
        dx**2 + dy**2 - dr**2
    ])

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    
    # Boundary constraints handled via LP bounds
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mx, 1e-9)))
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    n_pairs = len(idx[0])
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    
    for k in range(n_pairs):
        i, j = idx[0][k], idx[1][k]
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 1e-12)
    except Exception:
        pass
    return None

def generate_hex_configs(n, r0, patterns, rng):
    """Generates hexagonal lattice configurations with specific row distributions."""
    configs = []
    for pat in patterns:
        if sum(pat) != n: 
            continue
        pts = []
        y = r0
        row = 0
        for cnt in pat:
            shift = r0 if row % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: 
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
            
        if len(pts) == n:
            cfg = np.array(pts)
            configs.append(cfg)
            # Add perturbed variants to break symmetry
            for _ in range(3):
                p = cfg + rng.uniform(-0.015, 0.015, (n, 2))
                configs.append(np.clip(p, 0.05, 0.95))
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    # Competitive row distributions for N=26 hexagonal packing
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [4, 5, 6, 6, 5]
    ]
    
    configs = []
    for r0 in [0.085, 0.095, 0.105]:
        configs.extend(generate_hex_configs(n, r0, patterns, rng))
        
    # Grid configurations
    for scale in [0.8, 0.9]:
        grid = np.array([[ (i+0.5)/n*scale, (j+0.5)/n*scale ] for j in range(5) for i in range(5)])
        grid = np.vstack([grid, [0.5, 0.5]])
        configs.append(grid * scale + (1-scale)/2)
        
    # Random dense placements
    for _ in range(10):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_opt = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    cons_dict = {'type': 'ineq', 'fun': cons_joint}
    
    # Phase 1: Multi-start SLSQP optimization
    for cfg in configs:
        x0 = np.zeros(3 * n)
        x0[:n] = cfg[:, 0]
        x0[n:2*n] = cfg[:, 1]

        dists_to_wall = np.minimum(np.minimum(cfg[:, 0], 1.0 - cfg[:, 0]),
                                   np.minimum(cfg[:, 1], 1.0 - cfg[:, 1]))
        diffs = cfg[:, np.newaxis, :] - cfg[np.newaxis, :, :]
        dists_nn = np.sqrt(np.sum(diffs**2, axis=2))

        min_dists_nn = np.min(dists_nn, axis=1)

        r_init = np.maximum(r_init, 1e-4)

        if r_new is not None:

            if s_new > best_sum + 1e-9:

                best_radii = r_new.copy()

                patience += 1
        else:
            best_centers[idx] = old_c
            patience += 1

                best_centers[i] = np.clip(best_centers[i], 0.05, 0.95)
            patience = 0
            step = 0.015

    x0_final = np.zeros(3*n)
    x0_final[:n] = best_centers[:, 0]
    x0_final[n:2*n] = best_centers[:, 1]
    x0_final[2*n:] = best_radii * 0.95
    
    try:
        res_final = minimize(obj_sum_radii, x0_final, method='SLSQP',

            c_polish = np.column_stack((res_final.x[:n], res_final.x[n:2*n]))
            r_polish = solve_lp_radii(c_polish)
            if r_polish is not None:
                s_polish = np.sum(r_polish)
                if s_polish > best_sum:
                    best_sum = s_polish
                    best_centers = c_polish
                    best_radii = r_polish
    except Exception: 
        pass

    # Phase 4: Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    idx_i, idx_j = np.triu_indices(n, k=1)
    for k in range(len(idx_i)):
        i, j = idx_i[k], idx_j[k]
        d = np.linalg.norm(best_centers[i] - best_centers[j])
        rs = best_radii[i] + best_radii[j]
        if rs > 1e-12:
            scale = min(scale, d / rs)
            
    best_radii *= scale * 0.9999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
