import numpy as np
from scipy.optimize import minimize, linprog

def compute_lp_radii(centers, n, idx_i, idx_j, A_pair):
    """Solves LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    b_ub = np.zeros(len(idx_i))
    bounds = []
    
    # Compute upper bounds from boundaries and fill pairwise distances
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mx, 1e-9)))
        
    for k in range(len(idx_i)):
        i, j = idx_i[k], idx_j[k]
        b_ub[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        
    try:
        res = linprog(c_obj, A_ub=A_pair, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def get_constraints(vars, n, idx_i, idx_j):
    """Computes inequality constraints >= 0 for valid packing."""
    cx = vars[:n]
    cy = vars[n:2*n]
    r = vars[2*n:]
    
    c = np.empty(4*n + len(idx_i))
    # Boundary constraints
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    
    # Pairwise non-overlap constraints (squared distance >= squared sum of radii)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c[4*n:] = dx**2 + dy**2 - dr**2
    
    return c

def obj_func(vars, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2*n:])

def generate_initializations(n, rng):
    """Generate diverse starting configurations."""
    configs = []
    
    # Hexagonal patterns with various row distributions summing to 26
    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [5, 6, 4, 6, 5], [6, 4, 5, 6, 5],
        [5, 7, 4, 5, 5], [4, 5, 6, 6, 5], [5, 5, 5, 5, 6]
    ]
    
    for pat in row_patterns:
        if sum(pat) != n:
            continue
        for r0 in [0.085, 0.095, 0.105]:
            pts = []
            y = r0
            for idx, cnt in enumerate(pat):
                shift = r0 if idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= n:
                        break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            pts = np.array(pts[:n])
            configs.append(pts)
            
            # Perturbed variants
            p = pts + rng.uniform(-0.02, 0.02, (n, 2))
            configs.append(np.clip(p, 0.04, 0.96))
            
    # Grid + center fallback
    gx = np.linspace(0.12, 0.88, 5)
    gy = np.linspace(0.12, 0.88, 5)
    grid = np.array([(x, y) for y in gy for x in gx])
    grid = np.vstack([grid, [0.5, 0.5]])
    configs.append(grid)
    configs.append(grid + rng.uniform(-0.015, 0.015, (n, 2)))
    
    # Random dense placements
    for _ in range(6):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return configs

def run_packing():

    rng = np.random.default_rng(42)
    idx_i, idx_j = np.triu_indices(n, k=1)
    
    # Precompute constant part of LP matrix for speed
    A_pair = np.zeros((len(idx_i), n))

        A_pair[k, i] = 1.0

        d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
        rs = best_radii[i] + best_radii[j]
        if rs > 1e-12:
            scale = min(scale, d / rs)
            
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
