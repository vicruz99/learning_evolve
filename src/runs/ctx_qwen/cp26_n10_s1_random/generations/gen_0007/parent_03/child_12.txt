def compute_objective(vars_arr, n):

    return -np.sum(vars_arr[2*n:])

def compute_constraints(vars_arr, n, idx_i, idx_j):
    """Compute inequality constraints >= 0 for valid packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    c = []
    # Boundary constraints
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    # Pairwise non-overlap constraints (squared distance >= squared sum of radii)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c.extend(dx**2 + dy**2 - dr**2)
    
    return np.array(c)

def solve_lp_radii(centers, n, idx_i, idx_j):
    """Solve LP to maximize sum of radii for fixed centers."""
    # Boundary limits for each circle
    mx = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))

    # Pairwise distances
    diffs = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diffs**2, axis=1))

    A_ub = np.zeros((len(idx_i), n))
    A_ub[np.arange(len(idx_i)), idx_i] = 1.0
    A_ub[np.arange(len(idx_i)), idx_j] = 1.0

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0
