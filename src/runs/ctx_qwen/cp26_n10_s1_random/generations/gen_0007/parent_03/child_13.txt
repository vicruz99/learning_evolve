from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n, idx_i, idx_j):
    """Solves the LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    bounds = []
    A_ub_rows = []
    b_ub_vals = []
    
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mx, 1e-9)))
        for lim in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(lim)
            
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    for i, j in zip(idx_i, idx_j):
        row = np.zeros(n)
        row[i] = 1.0
        row[j] = 1.0
        A_ub_rows.append(row)
        b_ub_vals.append(dists[i, j])
        
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub_rows), b_ub=np.array(b_ub_vals), 
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def force_directed_expansion(cfg, n, rng):

        centers += forces * dt
        centers = np.clip(centers, 1e-6, 1.0 - 1e-6)

def slsqp_obj(v, n):
    return -np.sum(v[2*n:])

def slsqp_cons(v, n, idx_i, idx_j):
    cx, cy, rad = v[:n], v[n:2*n], v[2*n:]
    c = [cx - rad, 1.0 - cx - rad, cy - rad, 1.0 - cy - rad]
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    rs = rad[:, None] + rad[None, :]
    c.append(d2[idx_i, idx_j] - rs[idx_i, idx_j]**2)
    return np.concatenate(c)
