import numpy as np
from scipy.optimize import linprog

def solve_lp_radii(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i <= dist(i, boundary) and r_i + r_j <= dist(i, j)
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    A_ub = np.zeros((m_bound + m_pair, n))
    b_ub = np.zeros(m_bound + m_pair)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0; A_ub[idx, j] = 1.0; b_ub[idx] = d; idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def get_lp_sum(centers):
    r = solve_lp_radii(centers)
    return np.sum(r) if r is not None else 0.0

def force_directed_init(n, seed, steps=1500):
    rng = np.random.default_rng(seed)
    centers = rng.uniform(0.25, 0.75, (n, 2))
    
    for _ in range(steps):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist < 1e-5: dist = 1e-5
                f = 0.0005 / (dist**2)
                forces[i] += f * diff / dist
                forces[j] -= f * diff / dist
                
        for i in range(n):
            x, y = centers[i]
            forces[i, 0] += -0.05 if x < 0.15 else (0.05 if x > 0.85 else 0)
            forces[i, 1] += -0.05 if y < 0.15 else (0.05 if y > 0.85 else 0)
            
        centers += forces * 0.08
        centers = np.clip(centers, 0.05, 0.95)
        
    return centers

    best_sum = 0.0

    ]
    for rc in row_configs:
        pts = []
        r0 = 0.092
        y = r0
        for i, cnt in enumerate(rc):
            shift = r0 if i % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        while len(pts) < n:
            pts.append([0.5, 0.5])
        pts = np.array(pts[:n])
        cx_m, cy_m = pts.mean(axis=0)
        pts -= np.array([cx_m - 0.5, cy_m - 0.5])
        inits.append(np.clip(pts, 0.02, 0.98))

                step *= 0.997

            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
