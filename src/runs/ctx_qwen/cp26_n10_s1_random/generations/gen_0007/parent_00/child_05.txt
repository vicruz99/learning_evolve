This allows us to decouple center optimization from radius optimization, making the problem much more tractable and avoiding non-convex constraint issues common in joint SLSQP approaches.

Strategy:
1. **LP Formulation**: For any set of centers, the maximum sum of radii is found by solving `max sum(r_i)` subject to `r_i + r_j <= dist(i,j)` and boundary constraints `r_i <= min(x_i, 1-x_i, y_i, 1-y_i)`. This LP is solved efficiently using `scipy.optimize.linprog` with the `highs` solver.
2. **Diverse Initialization**: Generate hexagonal lattices (optimal for density), grid patterns, and random placements. Perturb them to break symmetry and cover different regions of the configuration space.
3. **Force-Driven Relaxation**: For each initialization, run a molecular dynamics-like phase. This quickly pushes configurations into locally dense packings.
4. **Coordinate Ascent**: Fine-tune centers by randomly perturbing one circle at a time. If the LP sum of radii increases, accept the move. This gradient-free method navigates the piecewise-linear objective landscape effectively.
5. **Polishing**: Apply `Powell` method on the best found configuration to squeeze out remaining slack.
6. **Safety Scaling**: Finally, compute the minimum violation ratio across all constraints and scale radii down slightly to guarantee strict numerical validity against the `1e-12` tolerance in the validator.

import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers, A_ub, pair_indices, c_obj):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    M = len(pair_indices) + 4 * n
    b_ub = np.empty(M)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b_ub[k] = x; k += 1
        b_ub[k] = 1.0 - x; k += 1
        b_ub[k] = y; k += 1
        b_ub[k] = 1.0 - y; k += 1
    for i, j in pair_indices:
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b_ub[k] = np.sqrt(dx * dx + dy * dy)
        k += 1
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0.0, None), method='highs')
        if res.success and np.isfinite(res.fun):
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    M = 4 * N + N * (N - 1) // 2
    A_ub = np.zeros((M, N))
    pair_indices = []
    idx = 0
    for i in range(N):
        A_ub[idx, i] = 1.0; idx += 1
        A_ub[idx, i] = 1.0; idx += 1
        A_ub[idx, i] = 1.0; idx += 1
        A_ub[idx, i] = 1.0; idx += 1
    for i in range(N):
        for j in range(i + 1, N):
            pair_indices.append((i, j))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
    c_obj = -np.ones(N)
    
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = []
    # 1. Hexagonal patterns
    for r0 in [0.085, 0.090, 0.095, 0.100]:
        for row_pat in [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [6, 6, 4, 5, 5]]:
            if sum(row_pat) < N: continue
            pts = []
            y = r0
            for ri, cnt in enumerate(row_pat):
                shift = r0 if ri % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= N: break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            while len(pts) < N: pts.append([0.5, 0.5])
            inits.append(np.array(pts[:N]))
            inits.append(np.clip(np.array(pts[:N]) + rng.uniform(-0.015, 0.015, (N, 2)), 0.05, 0.95))
            
    # 2. Grid patterns
    for sp in [0.185, 0.200, 0.215]:
        pts = []
        for i in range(6):
            for j in range(5):
                if len(pts) >= N: break
                pts.append([0.1 + i * sp, 0.1 + j * sp])
        while len(pts) < N: pts.append([0.5, 0.5])
        inits.append(np.array(pts[:N]))
        inits.append(np.clip(np.array(pts[:N]) + rng.uniform(-0.01, 0.01, (N, 2)), 0.05, 0.95))
        
    # 3. Random dense
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # Optimization loop over all initializations
    for cfg in inits:
        centers = cfg.copy()
        
        # Phase 1: Force-driven relaxation
        for step_num in range(500):
            s, r = solve_lp(centers, A_ub, pair_indices, c_obj)
            if s == 0.0: break
            
            forces = np.zeros_like(centers)
            strength = 30.0 / (step_num + 15.0)
            
            for i in range(N):
                for j in range(i + 1, N):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = np.sqrt(dx * dx + dy * dy)
                    if d < 1e-8: d = 1e-8
                    min_d = r[i] + r[j]
                    if d < min_d:
                        f = (min_d - d) * strength / d
                        forces[i, 0] += f * dx
                        forces[i, 1] += f * dy
                        forces[j, 0] -= f * dx
                        forces[j, 1] -= f * dy
                        
            for i in range(N):
                for dim in range(2):
                    if centers[i, dim] < r[i]:
                        forces[i, dim] += (r[i] - centers[i, dim]) * strength * 3
                    elif centers[i, dim] > 1.0 - r[i]:
                        forces[i, dim] -= (centers[i, dim] + r[i] - 1.0) * strength * 3
                        
            centers += forces
            centers = np.clip(centers, 0.005, 0.995)
            
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = r.copy()
                
        # Phase 2: Coordinate Ascent
        step_size = 0.006
        for _ in range(1800):
            idx = rng.integers(N)
            old = centers[idx].copy()
            move = rng.uniform(-step_size, step_size, 2)
            centers[idx] = np.clip(centers[idx] + move, 0.01, 0.99)
            s_new, r_new = solve_lp(centers, A_ub, pair_indices, c_obj)
            if s_new > best_sum + 1e-9:
                best_sum = s_new
                best_centers = centers.copy()
                best_radii = r_new.copy()
                if rng.random() < 0.05: step_size *= 0.99
            else:
                centers[idx] = old
                if rng.random() < 0.1: step_size *= 0.995
                
    # Phase 3: Powell polish on best configuration
    if best_centers is not None:
        def obj_polish(x):
            c = x.reshape(N, 2)
            s, _ = solve_lp(c, A_ub, pair_indices, c_obj)
            return -s
            
        try:
            res = minimize(obj_polish, best_centers.flatten(), method='Powell', 
                           options={'maxiter': 400, 'ftol': 1e-13, 'xtol': 1e-13})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
                _, best_radii = solve_lp(best_centers, A_ub, pair_indices, c_obj)
        except Exception:
            pass
            
    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = rng.uniform(0.1, 0.9, (N, 2))
        best_radii = np.full(N, 0.08)
        best_sum = float(np.sum(best_radii))
        
    # Phase 4: Strict safety scaling for numerical validation tolerance
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
