import numpy as np
from scipy.optimize import linprog

def solve_lp_and_duals(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers and returns dual variables.
    Returns: radii, sum_radii, duals (marginals for pairwise constraints)
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= d_ij
    m_pair = n * (n - 1) // 2
    A_ub = np.zeros((m_pair, n))
    b_ub = np.zeros(m_pair)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.hypot(dx, dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    # Boundary constraints handled via bounds: 0 <= r_i <= min(dist to walls)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mx, 1e-9)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            # Dual variables for inequality constraints
            duals = getattr(getattr(res, 'marginals', None), 'ineqlin', None)
            if duals is None:
                # Fallback for older scipy versions or missing attribute
                duals = np.zeros(m_pair)
            return np.maximum(res.x, 0.0), -res.fun, duals
    except Exception:
        pass
    return None, 0.0, None

def compute_forces_from_duals(centers, duals, n):
    """Computes repulsive forces on centers based on LP dual variables."""
    forces = np.zeros((n, 2))
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]
            if lam > 1e-6:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)
                if d > 1e-8:
                    # Force magnitude proportional to pressure and inverse distance
                    f = lam / d
                    forces[i, 0] += f * dx
                    forces[i, 1] += f * dy
                    forces[j, 0] -= f * dx
                    forces[j, 1] -= f * dy
            idx += 1
    return forces

def generate_hex_start(n, row_counts, rng):
    """Generates a centered hexagonal lattice initialization."""
    pts = []
    r0 = 0.095
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        
    while len(pts) < n:
        pts.append([0.5, 0.5])
        
    pts = np.array(pts[:n])
    # Center configuration
    cx_mean, cy_mean = pts.mean(axis=0)
    pts -= np.array([cx_mean - 0.5, cy_mean - 0.5])
    return np.clip(pts, 0.01, 0.99)

def optimize_packing(start_centers, n, rng):
    """Runs LP + Dual Ascent optimization from a given starting configuration."""
    centers = start_centers.copy()
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    lr = 0.012
    max_iter = 4000
    
    for step in range(max_iter):
        radii, s, duals = solve_lp_and_duals(centers)
        if radii is None:
            centers = best_centers.copy() if best_centers is not None else centers
            continue
            
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
            
        if duals is not None:
            forces = compute_forces_from_duals(centers, duals, n)
            centers += lr * forces
            
        # Decay learning rate
        lr *= 0.9992
        
        # Clip to valid region
        centers = np.clip(centers, 0.005, 0.995)
        
        # Occasional shake to escape local minima
        if step % 600 == 0 and step > 0:
            centers += rng.uniform(-0.008, 0.008, centers.shape)
            centers = np.clip(centers, 0.005, 0.995)
            
    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Diverse topological seeds
    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [7, 6, 6, 7]
    ]
    
    seeds = []
    for pat in row_patterns:
        if sum(pat) < n:
            continue
        base = generate_hex_start(n, pat, rng)
        seeds.append(base)
        # Add controlled perturbations
        for _ in range(3):
            pert = base + rng.uniform(-0.025, 0.025, (n, 2))
            seeds.append(np.clip(pert, 0.02, 0.98))
            
    # Random dense seeds
    for _ in range(5):
        seeds.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    for seed_cfg in seeds:
        c_opt, r_opt, s_opt = optimize_packing(seed_cfg, n, rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            best_radii = r_opt
            
    # Fallback if all optimizations fail (highly unlikely)
    if best_centers is None:
        best_centers = generate_hex_start(n, [6, 5, 6, 5, 4], rng)
        best_radii, best_sum, _ = solve_lp_and_duals(best_centers)
        
    # Strict safety scaling to guarantee validation tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999999
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
