import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    A_ub = np.zeros((m_bound + m_pair, n))
    b_ub = np.zeros(m_bound + m_pair)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lims = (x, 1.0 - x, y, 1.0 - y)
        for lim in lims:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def get_lp_sum(centers):
    """Returns the maximum sum of radii for given centers."""
    r = solve_lp_radii(centers)
    if r is not None:
        return np.sum(r)
    return -1e9

def obj_for_opt(x, n):
    """Objective for center optimization: negative LP sum of radii."""
    return -get_lp_sum(x.reshape(n, 2))

def generate_starts(n, rng):
    """Generates diverse initial center configurations."""
    starts = []
    
    # Hexagonal patterns with various row distributions
    row_counts_options = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [6, 5, 5, 6, 4],
        [5, 6, 6, 4, 5], [5, 5, 5, 5, 6], [7, 6, 6, 7], [5, 5, 5, 5, 5, 1]
    ]
    
    for rc in row_counts_options:
        if sum(rc) < n: 
            continue
        pts = []
        r0 = 0.09
        y = r0
        for i, cnt in enumerate(rc):
            shift = r0 if i % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: 
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        while len(pts) < n:
            pts.append([0.5, 0.5])
        pts = np.array(pts[:n])
        starts.append(pts)
        
        # Add perturbed variants to break symmetry
        p = pts + rng.uniform(-0.015, 0.015, (n, 2))
        starts.append(np.clip(p, 0.02, 0.98))
        
    # Grid patterns
    for sp in [0.18, 0.20, 0.22]:
        pts = []
        for i in range(7):
            for j in range(7):
                if len(pts) >= n: 
                    break
                pts.append([0.5 - 3*sp + i*sp, 0.5 - 3*sp + j*sp])
        starts.append(np.array(pts[:n]))
        
    # Random dense placements
    for _ in range(5):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    starts = generate_starts(n, rng)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Nelder-Mead optimization on centers
    # Maximizes the LP-evaluated sum of radii
    for init_c in starts:
        try:
            res = minimize(obj_for_opt, init_c.flatten(), args=(n,), method='Nelder-Mead',
                           options={'maxiter': 20000, 'xatol': 1e-8, 'fatol': 1e-10})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(n, 2).copy()
                best_radii = solve_lp_radii(best_centers)
        except Exception:
            continue
            
    # Phase 2: Multi-circle Local Search on centers
    if best_centers is not None:
        centers = best_centers.copy()
        step = 0.005

            # Randomly choose 1-4 circles to move simultaneously

            old_positions = centers[idxs].copy()

            norms[norms < 1e-12] = 1.0
            dirs /= norms
            
            # Apply perturbation and clip

            r_new = solve_lp_radii(centers)
            if r_new is not None:
                s_new = np.sum(r_new)
                if s_new > best_sum + 1e-12:
                    best_sum = s_new
                    best_radii = r_new.copy()
                    best_centers = centers.copy()
                    step *= 0.999  # Shrink step slowly on success
                else:
                    centers[idxs] = old_positions  # Revert
                    step *= 0.9995  # Shrink step slightly on failure
            else:
                centers[idxs] = old_positions
                
    # Phase 3: Joint SLSQP refinement for precise polishing
    if best_centers is not None:
        x0 = np.zeros(3 * n)
        x0[:n] = best_centers[:, 0]
        x0[n:2*n] = best_centers[:, 1]

            return -np.sum(x[2*n:])
            
        def cons_joint(x):
            cx = x[:n]

            r = x[2*n:]

            dy = cy[:, None] - cy[None, :]
            d2 = dx**2 + dy**2
            rs = r[:, None] + r[None, :]
            rs2 = rs**2
            idx = np.triu_indices(n, k=1)
            c = np.concatenate([c, d2[idx] - rs2[idx]])
            return c
            
        try:
            res = minimize(obj_joint, x0, method='SLSQP', bounds=bounds_opt,
                           constraints={'type': 'ineq', 'fun': cons_joint},
                           options={'maxiter': 8000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_opt = np.column_stack((res.x[:n], res.x[n:2*n]))
                r_opt = solve_lp_radii(c_opt)
                if r_opt is not None and np.sum(r_opt) > best_sum:
                    best_sum = np.sum(r_opt)
                    best_centers = c_opt
                    best_radii = r_opt
        except Exception:
            pass
            
    # Fallback configuration
    if best_centers is None:
        best_centers = starts[0]
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    # Safety scaling to guarantee strict numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
