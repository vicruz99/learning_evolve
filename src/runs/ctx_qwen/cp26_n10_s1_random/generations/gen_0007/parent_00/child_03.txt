import numpy as np
from scipy.optimize import linprog

def solve_lp_fast(centers, A_ub, pair_indices, num_pairs, c_obj, bounds_lp):
    """Solves the LP to maximize sum of radii for fixed centers efficiently."""
    n = centers.shape[0]
    m = A_ub.shape[0]
    b_ub = np.empty(m)
    idx = 0
    for (i, j) in pair_indices:
        d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        b_ub[idx] = d
        idx += 1
    for i in range(n):
        x, y = centers[i]
        b_ub[idx] = x; idx += 1
        b_ub[idx] = 1.0 - x; idx += 1
        b_ub[idx] = y; idx += 1
        b_ub[idx] = 1.0 - y; idx += 1
        
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
    if res.success:
        return res.x, res.marginals.ineqlin[:num_pairs], -res.fun
    return None, None, 0.0

def compute_forces(centers, duals, pair_indices):
    """Computes repulsion forces based on LP dual variables."""
    forces = np.zeros_like(centers)
    for (i, j), lam in zip(pair_indices, duals):
        if lam > 1e-12:
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.hypot(dx, dy)
            if d > 1e-12:
                inv_d = 1.0 / d
                fx = lam * dx * inv_d
                fy = lam * dy * inv_d
                forces[i, 0] += fx
                forces[i, 1] += fy
                forces[j, 0] -= fx
                forces[j, 1] -= fy
    return forces

def generate_initial_configs(n, rng):
    """Generates diverse starting configurations (hexagonal, perturbed, random)."""
    configs = []
    row_counts_options = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [6, 5, 5, 6, 4],
        [5, 6, 6, 4, 5], [5, 5, 5, 5, 6]
    ]
    
    for rc in row_counts_options:
        if sum(rc) < n: continue
        pts = []
        r0 = 0.09
        y = r0
        for i, cnt in enumerate(rc):
            shift = r0 if i % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        while len(pts) < n: pts.append([0.5, 0.5])
        pts = np.array(pts[:n])
        pts -= pts.mean(axis=0)
        pts += 0.5
        configs.append(np.clip(pts, 0.05, 0.95))
        
        for _ in range(2):
            p = pts + rng.uniform(-0.02, 0.02, (n, 2))
            configs.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(4):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    m = num_pairs + num_bound
    A_ub = np.zeros((m, n))
    pair_indices = []
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            pair_indices.append((i, j))
            idx += 1
    for i in range(n):
        for _ in range(4):
            A_ub[idx, i] = 1.0
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    c_obj = -np.ones(n)
    
    inits = generate_initial_configs(n, rng)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Subgradient ascent on centers guided by LP duals
    for cfg in inits:
        centers = cfg.copy()
        step = 0.015
        for it in range(1000):
            radii, duals, s = solve_lp_fast(centers, A_ub, pair_indices, num_pairs, c_obj, bounds_lp)
            if radii is None: break
            
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()
                
            forces = compute_forces(centers, duals, pair_indices)
            centers += step * forces
            centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
            
            # Occasional jitter to escape plateaus
            if rng.random() < 0.12:
                centers += rng.uniform(-step*0.3, step*0.3, centers.shape)
                centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
                
            step *= 0.996

    # Fallback if optimization fails unexpectedly
    if best_centers is None:
        best_centers = inits[0]
        radii, _, s = solve_lp_fast(best_centers, A_ub, pair_indices, num_pairs, c_obj, bounds_lp)
        best_radii = radii
        best_sum = s
        
    # Final LP refinement on the best centers found
    final_radii, _, final_s = solve_lp_fast(best_centers, A_ub, pair_indices, num_pairs, c_obj, bounds_lp)
    if final_radii is not None and final_s > best_sum:
        best_radii = final_radii
        best_sum = final_s
        
    # Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
