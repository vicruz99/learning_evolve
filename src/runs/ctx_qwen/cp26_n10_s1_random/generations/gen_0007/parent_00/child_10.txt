def objective_func(v, n):

def constraint_func(v, n):

    cy = v[n:2*n]
    r = v[2*n:]
    c = []
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)

    rs2 = r_m**2
    idx = np.triu_indices(n, k=1)
    c.append(d2[idx] - rs2[idx])
    return np.concatenate(c)

def solve_lp_fast(centers, A_ub, n):

    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    m_total = m_bound + m_pair
    b_ub = np.empty(m_total)
    bounds = []
    idx = 0
    
    for i in range(n):
        x, y = centers[i]
        mx = max(1e-9, min(x, 1.0-x, y, 1.0-y))
        bounds.append((0.0, mx))
        b_ub[idx] = x; idx += 1
        b_ub[idx] = 1.0-x; idx += 1
        b_ub[idx] = y; idx += 1
        b_ub[idx] = 1.0-y; idx += 1

            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
