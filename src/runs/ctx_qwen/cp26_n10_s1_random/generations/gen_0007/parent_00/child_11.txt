

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
idx_i, idx_j = np.triu_indices(N, k=1)
A_PAIR = np.zeros((len(idx_i), N))
A_PAIR[np.arange(len(idx_i)), idx_i] = 1.0
A_PAIR[np.arange(len(idx_i)), idx_j] = 1.0

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    diffs = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    
    res = linprog(-np.ones(N), A_ub=A_PAIR, b_ub=dists, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return np.maximum(res.x, 0.0)
    return None

def obj_2d(pos, centers, idx):
    """Objective for 2D optimization: maximize sum of radii."""
    pos = np.clip(pos, 1e-5, 1.0 - 1e-5)
    centers[idx] = pos
    r = solve_lp(centers)
    return -np.sum(r) if r is not None else 1000.0

def generate_inits():
    """Generates diverse initial configurations."""
    inits = []
    rng = np.random.default_rng(42)
    
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [7, 5, 5, 6], 
        [5, 5, 5, 5, 6], [6, 6, 4, 5, 5]
    ]
    
    for pat in patterns:
        pts = []
        r0 = 0.09
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        while len(pts) < N:
            pts.append([0.5, 0.5])
        pts = np.array(pts[:N])
        pts = (pts - pts.min(axis=0)) / (pts.max(axis=0) - pts.min(axis=0) + 1e-9)
        pts = pts * 0.9 + 0.05
        inits.append(pts)
        
    for _ in range(5):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return inits

def run_packing():
    np.random.seed(42)
    inits = generate_inits()
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Cyclic Coordinate Descent on diverse starts
    for cfg in inits:
        c = cfg.copy()
        c += np.random.uniform(-1e-4, 1e-4, c.shape)
        
        for i in range(N):
            res = minimize(obj_2d, c[i], args=(c, i), method='Nelder-Mead',
                           options={'maxiter': 150, 'xatol': 1e-8, 'fatol': 1e-10})
            c[i] = np.clip(res.x, 1e-5, 1.0 - 1e-5)
            
        r_opt = solve_lp(c)
        if r_opt is not None:
            s = np.sum(r_opt)
            if s > best_sum:
                best_sum = s
                best_centers = c.copy()
                best_radii = r_opt.copy()
                
    # Phase 2: Simulated Annealing refinement
    if best_centers is not None:
        curr_c = best_centers.copy()
        curr_s = best_sum
        rng = np.random.default_rng(888)
        T = 0.008
        
        for step in range(6000):
            i = rng.integers(N)
            old_pos = curr_c[i].copy()
            
            step_sz = 0.015 * (1.0 - step/6000)
            new_pos = curr_c[i] + rng.uniform(-step_sz, step_sz, 2)
            new_pos = np.clip(new_pos, 1e-4, 1.0 - 1e-4)
            curr_c[i] = new_pos
            
            new_r = solve_lp(curr_c)
            new_s = np.sum(new_r) if new_r is not None else 0.0
            
            delta = new_s - curr_s
            if delta > 0 or rng.random() < np.exp(delta / max(T, 1e-6)):
                curr_s = new_s
                if new_s > best_sum:
                    best_sum = new_s
                    best_centers = curr_c.copy()
                    best_radii = new_r.copy()
            else:
                curr_c[i] = old_pos
                
            T *= 0.9995
            
    # Safety scaling to guarantee strict numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
