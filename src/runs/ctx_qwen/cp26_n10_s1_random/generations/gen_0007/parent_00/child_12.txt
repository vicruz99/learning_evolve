import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j) and 0 <= r_i <= dist(i, boundary)
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    m = m_bound + m_pair
    
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    bounds = []
    
    idx = 0
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(lim, 1e-9)))
        for val in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[idx, i] = 1.0
            b_ub[idx] = max(val, 1e-9)
            idx += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', 
                      options={'presolve': True, 'time_limit': 2.0})
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def neg_lp_sum(centers_flat):
    """Objective function for center optimization: negative sum of optimal radii."""
    centers = centers_flat.reshape(-1, 2)
    # Clip to strictly inside to maintain LP feasibility
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    radii = solve_lp_radii(centers)
    if radii is not None:
        return -np.sum(radii)
    return 1e6  # Heavy penalty for infeasible LP

def generate_hex_configs(n, rng):
    """Generates diverse hexagonal lattice starting configurations."""
    configs = []
    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [7, 6, 6, 7],
        [5, 7, 5, 6], [6, 4, 6, 5, 5], [5, 5, 5, 5, 6],
        [6, 5, 5, 5, 5], [5, 6, 4, 5, 6]
    ]
    
    for pat in row_patterns:
        if sum(pat) < n: 
            continue
        pts = []
        r0 = 0.092
        y = r0
        for i, cnt in enumerate(pat):
            shift = r0 if i % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: 
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            
        while len(pts) < n:
            pts.append([0.5, 0.5])
            
        arr = np.array(pts[:n])
        # Normalize and center in the unit square
        arr = (arr - arr.min(axis=0)) / (arr.max(axis=0) - arr.min(axis=0) + 1e-9)
        arr = arr * 0.72 + 0.14
        configs.append(arr)
        
    # Add fully random dense starts
    for _ in range(6):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return configs
