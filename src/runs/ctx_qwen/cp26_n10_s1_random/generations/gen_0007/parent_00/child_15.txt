import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist

def compute_lp_sum(centers, A_ub_struct, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    b_ub = pdist(centers)
    lims = np.min(np.column_stack([centers, 1.0 - centers]), axis=1)
    bounds = [(1e-9, max(l, 1e-9)) for l in lims]
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_struct, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 1e-9), -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_slsqp(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def cons_slsqp(vars_arr, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    d2 = cx_m**2 + cy_m**2
    rs2 = r_m**2
    idx = np.triu_indices(n, k=1)
    c = np.concatenate([c, d2[idx] - rs2[idx]])
    return c

def generate_hex_starts(n, rng):
    """Generates diverse hexagonal and random initial configurations."""
    starts = []
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], 
                [6,6,5,5,4], [5,6,6,4,5], [5,7,5,5,4], [7,5,5,5,4]]
    for pat in patterns:
        pts = []
        r0 = 0.095
        y = r0
        for i, cnt in enumerate(pat):
            shift = r0 if i % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        while len(pts) < n:
            pts.append([0.5, 0.5])
        cfg = np.array(pts[:n])
        cfg -= cfg.mean(axis=0)
        cfg += 0.5
        cfg = np.clip(cfg, 0.05, 0.95)
        starts.append(cfg)
        starts.append(np.clip(cfg + rng.uniform(-0.015, 0.015, (n, 2)), 0.02, 0.98))
        
    for _ in range(8):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)))
    return starts

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    # Precompute constant LP constraint matrix structure

    A_ub_struct = np.zeros((m, n))
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub_struct[idx, i] = 1.0
            A_ub_struct[idx, j] = 1.0
            idx += 1
            
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    bounds_slsqp = [(0.001, 0.999)] * (2*n) + [(1e-6, 0.25)] * n
    starts = generate_hex_starts(n, rng)
    
    # Phase 1: Joint SLSQP optimization from diverse starts
    for cfg in starts:
        x0 = np.zeros(3*n)
        x0[:n] = cfg[:, 0]
        x0[n:2*n] = cfg[:, 1]
        x0[2*n:] = 0.085
        
        try:
            res = minimize(obj_slsqp, x0, args=(n,), method='SLSQP', bounds=bounds_slsqp,
                           constraints={'type': 'ineq', 'fun': cons_slsqp, 'args': (n,)},
                           options={'maxiter': 4000, 'ftol': 1e-14})
            if not np.isfinite(res.fun): continue
            c_opt = np.column_stack((res.x[:n], res.x[n:2*n]))
            r_lp, s_lp = compute_lp_sum(c_opt, A_ub_struct, n)
            if r_lp is not None and s_lp > best_sum:
                best_sum = s_lp
                best_centers = c_opt.copy()
                best_radii = r_lp.copy()
        except Exception:
            pass

    if best_centers is not None:

            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
