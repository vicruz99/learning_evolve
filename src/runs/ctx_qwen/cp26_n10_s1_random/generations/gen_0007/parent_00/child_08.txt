import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_radii(centers):
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(lim, 1e-9)))
        
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b_ub[idx] = np.hypot(dx, dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def compute_slsqp_constraints(vars_arr, n):
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    c = []
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.append(d2[mask] - rs[mask]**2)
    
    return np.concatenate(c)

def objective_slsqp(vars_arr, n):
    return -np.sum(vars_arr[2*n:])

def force_directed_init(n, seed):
    rng = np.random.default_rng(seed)
    centers = rng.uniform(0.2, 0.8, (n, 2))
    r = 0.05
    for _ in range(800):
        forces = np.zeros_like(centers)
        for i in range(n):
            for d in range(2):
                if centers[i, d] < r:
                    forces[i, d] += (r - centers[i, d]) * 200.0
                elif centers[i, d] > 1.0 - r:
                    forces[i, d] -= (centers[i, d] - (1.0 - r)) * 200.0
                    
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist < 2.0 * r and dist > 1e-7:
                    overlap = 2.0 * r - dist
                    f = overlap * 100.0 / dist
                    forces[i] += diff * f
                    forces[j] -= diff * f
                    
        centers += forces * 0.005
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    return centers
