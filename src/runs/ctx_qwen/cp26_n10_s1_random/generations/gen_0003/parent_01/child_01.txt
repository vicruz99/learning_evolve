import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_hex_configs():
    """Generates diverse hexagonal lattice configurations as starting points."""
    configs = []
    patterns = [
        [5, 6, 5, 6, 4],
        [6, 5, 6, 5, 4],
        [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4],
        [6, 6, 5, 5, 4],
        [5, 6, 4, 6, 5],
        [7, 6, 6, 5, 2],
        [5, 5, 5, 5, 6]
    ]
    r_init = 0.105
    for pat in patterns:
        pts = []
        y = r_init
        for row_idx, cnt in enumerate(pat):
            shift = r_init if row_idx % 2 == 1 else 0.0
            x = r_init + shift
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2 * r_init
            y += np.sqrt(3) * r_init
        if len(pts) < N:
            pts.extend([[0.5, 0.5]] * (N - len(pts)))
        configs.append(np.array(pts[:N]))
    return configs

def solve_radii_lp(centers):
    """Solves for optimal radii given fixed centers using Linear Programming."""
    n = centers.shape[0]
    # Max radius for each circle is limited by distance to boundaries
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    wall_dists = np.maximum(wall_dists, 0.0)
    bounds = [(0.0, w) for w in wall_dists]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    b_ub = dists[idx_i, idx_j]
    
    for k, (i, j) in enumerate(zip(idx_i, idx_j)):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def objective_joint(vars):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2*N:])

def constraints_joint(vars):
    """Inequality constraints >= 0 for boundary and non-overlap."""
    cx = vars[:N]
    cy = vars[N:2*N]
    r = vars[2*N:]
    
    con = []
    # Boundary constraints
    con.extend(cx - r)
    con.extend(1.0 - cx - r)
    con.extend(cy - r)
    con.extend(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    d2 = cx_m**2 + cy_m**2
    rs2 = r_m**2
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    con.extend((d2 - rs2)[mask])
    return np.array(con)

def run_packing():
    np.random.seed(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate base and perturbed initial configurations
    inits = get_hex_configs()
    for base in inits[:4]:
        for _ in range(4):
            pert = base + np.random.uniform(-0.03, 0.03, base.shape)
            inits.append(np.clip(pert, 0.02, 0.98))
            
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    
    for cfg in inits:
        x0 = np.zeros(3*N)
        x0[:N] = cfg[:, 0]
        x0[N:2*N] = cfg[:, 1]
        x0[2*N:] = 0.095
        
        try:
            res = minimize(objective_joint, x0, method='SLSQP', bounds=bounds,
                          constraints={'type': 'ineq', 'fun': constraints_joint},
                          options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            
            cx = res.x[:N]
            cy = res.x[N:2*N]
            r_opt = np.maximum(res.x[2*N:], 1e-9)
            centers = np.column_stack((cx, cy))
            
            # LP refinement maximizes radii sum for these exact centers
            r_lp, sum_lp = solve_radii_lp(centers)
            if r_lp is not None and sum_lp > np.sum(r_opt):
                r_opt = r_lp
                
            s = np.sum(r_opt)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = r_opt.copy()
                
            # Iterative shaking to escape local minima when high quality is found
            if s > 2.60:
                for _ in range(3):
                    c_pert = best_centers + np.random.uniform(-0.01, 0.01, best_centers.shape)
                    c_pert = np.clip(c_pert, 0.01, 0.99)
                    x0_p = np.zeros(3*N)
                    x0_p[:N] = c_pert[:, 0]
                    x0_p[N:2*N] = c_pert[:, 1]
                    x0_p[2*N:] = best_radii * 0.98
                    
                    try:
                        res2 = minimize(objective_joint, x0_p, method='SLSQP', bounds=bounds,
                                      constraints={'type': 'ineq', 'fun': constraints_joint},
                                      options={'maxiter': 8000, 'ftol': 1e-14})
                        c2 = np.column_stack((res2.x[:N], res2.x[N:2*N]))
                        r2_lp, s2_lp = solve_radii_lp(c2)
                        if r2_lp is not None and s2_lp > best_sum:
                            best_sum = s2_lp
                            best_centers = c2.copy()
                            best_radii = r2_lp.copy()
                    except Exception:
                        pass
        except Exception:
            continue

    # Fallback configuration
    if best_centers is None:
        best_centers = inits[0]
        best_radii = np.full(N, 0.09)
        best_sum = np.sum(best_radii)

    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
