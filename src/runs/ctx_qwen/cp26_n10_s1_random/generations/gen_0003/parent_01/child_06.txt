import numpy as np
from scipy.optimize import minimize, linprog

def neg_min_clearance(centers_flat):
    """Objective: maximize the minimum clearance between circles and boundaries."""
    n = 26
    centers = centers_flat.reshape(n, 2)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pair = np.min(dists)
    
    # Distances to walls
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    min_wall = np.min(wall)
    
    # Maximize min(min_pair/2, min_wall) <=> maximize min(min_pair, 2*min_wall)
    return -min(min_pair, 2.0 * min_wall)

def generate_starts():
    """Generates diverse initial configurations for center optimization."""
    n = 26
    starts = []
    
    # Base hexagonal lattice
    pts = []
    r0 = 0.102
    y = r0
    row = 0
    dy = np.sqrt(3) * r0
    while len(pts) < n and y + r0 < 1.0:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x + r0 < 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r0
        y += dy
        row += 1
    while len(pts) < n:
        pts.append([0.5, 0.5])
    base = np.array(pts[:n])
    starts.append(base)
    
    np.random.seed(42)
    # Perturbed versions to escape local minima
    for _ in range(5):
        starts.append(np.clip(base + np.random.uniform(-0.03, 0.03, (n, 2)), 0.05, 0.95))
        
    # Center-scaled version
    sc = (base - base.min(axis=0)) / (base.max(axis=0) - base.min(axis=0))
    sc = sc * 0.8 + 0.1
    starts.append(sc)
    starts.append(np.random.uniform(0.1, 0.9, (n, 2)))
    
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    starts = generate_starts()
    best_val = -np.inf
    best_centers = None
    
    # Phase 1: Optimize centers to maximize minimum clearance
    for init_pts in starts:
        x0 = init_pts.flatten()
        res = minimize(neg_min_clearance, x0, method='Nelder-Mead', 
                       options={'maxiter': 4000, 'fatol': 1e-12, 'xatol': 1e-12})
        if res.fun < best_val:
            best_val = res.fun
            best_centers = res.x.reshape(n, 2)
            
    # Phase 2: Solve LP to maximize sum of radii for fixed centers
    # Constraints: r_i + r_j <= dist(i,j), r_i <= wall_dist(i), r_i >= 0
    wall_dists = np.minimum(np.minimum(best_centers[:, 0], 1.0 - best_centers[:, 0]),
                            np.minimum(best_centers[:, 1], 1.0 - best_centers[:, 1]))
                            
    diff = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m + n, n))
    b_ub = np.zeros(m + n)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
    for i in range(n):
        A_ub[m + i, i] = 1.0
        b_ub[m + i] = wall_dists[i]
        
    c = -np.ones(n)  # Maximize sum(r) -> minimize -sum(r)
    bounds = [(0, None)] * n
    
    radii = None
    try:
        res_lp = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res_lp.success:
            radii = res_lp.x
    except Exception:
        pass
        
    # Fallback to equal radii if LP fails
    if radii is None:
        r_eq = min(np.min(dists)/2.0, np.min(wall_dists))

    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

            rs = radii[i] + radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    radii *= max(scale * 0.999999, 0.0)
    sum_r = np.sum(radii)
    
    return best_centers, radii, float(sum_r)
