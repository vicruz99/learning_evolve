from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns (radii, sum_radii).
    """
    n = centers.shape[0]
    # Distance to nearest boundary for each circle
    walls = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    walls = np.maximum(walls, 0.0)
    
    # Pairwise distances
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    
    # LP setup: Maximize sum(r) <=> Minimize -sum(r)
    c_obj = np.ones(n) * -1.0
    bounds = [(0.0, w) for w in walls]
    
    # Pairwise constraints: r_i + r_j <= d_ij
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def joint_objective(vars_array, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_array[2 * n:])

def joint_constraints(vars_array, n):
    """
    Computes inequality constraints >= 0 for valid packing.
    """
    cx = vars_array[:n]
    cy = vars_array[n:2 * n]
    r = vars_array[2 * n:]
    
    c_list = []
    # Boundary constraints
    c_list.extend(cx - r)
    c_list.extend(1.0 - cx - r)
    c_list.extend(cy - r)
    c_list.extend(1.0 - cy - r)
    
    # Pairwise non-overlap constraints (vectorized)
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    
    d2 = cx_m**2 + cy_m**2
    rs2 = r_m**2
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c_list.extend((d2 - rs2)[mask])
    
    return np.array(c_list)

def get_hex_init(r0, n):
    """Generates an initial hexagonal grid."""
    pts = []
    y = r0
    row = 0
    dy = np.sqrt(3) * r0
    while len(pts) < n and y + r0 < 1.0:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x + r0 < 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2 * r0
        y += dy
        row += 1
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
