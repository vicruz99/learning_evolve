import numpy as np
from scipy.optimize import minimize, linprog

def objective_smooth(params, n, mu):
    """Smooth penalty objective for center optimization."""
    cx = params[:n]
    cy = params[n:2*n]
    r = params[2*n:]
    
    # Primary goal: maximize sum of radii
    obj = -np.sum(r)
    
    # Boundary penalties: circles must stay within [0, 1]
    v1 = np.maximum(0.0, r - cx)
    v2 = np.maximum(0.0, cx + r - 1.0)
    v3 = np.maximum(0.0, r - cy)
    v4 = np.maximum(0.0, cy + r - 1.0)
    obj += mu * np.sum(v1**2 + v2**2 + v3**2 + v4**2)
    
    # Overlap penalties: distance must be >= r_i + r_j
    diff_x = cx[:, None] - cx[None, :]
    diff_y = cy[:, None] - cy[None, :]
    dist = np.sqrt(diff_x**2 + diff_y**2) + 1e-12
    np.fill_diagonal(dist, 1.0)
    
    r_sum = r[:, None] + r[None, :]
    viol = np.maximum(0.0, r_sum - dist)
    
    # Apply only to upper triangle to avoid double-counting
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    obj += mu * np.sum((viol * mask)**2)
    
    return obj

def get_init_configs(n):
    """Generates diverse initial center configurations."""
    configs = []
    
    # 1. Hexagonal lattice pattern (5,6,5,6,4 rows)
    r0 = 0.10
    pts = []
    rows = [5, 6, 5, 6, 4]
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    configs.append(np.array(pts[:n]))
    
    # 2. Centered and scaled version of hex lattice
    c = configs[0].copy()
    mn = c.min(axis=0)
    mx = c.max(axis=0)
    c = (c - mn) / (mx - mn) * 0.8 + 0.1
    configs.append(c)
    
    # 3. Uniform grid 5x5 + center
    g = np.linspace(0.15, 0.85, 5)
    pts2 = [[x, y] for y in g for x in g]
    pts2.append([0.5, 0.5])
    configs.append(np.array(pts2[:n]))
    
    # 4-12. Perturbed versions to escape local minima
    np.random.seed(42)
    for _ in range(10):
        base = configs[1].copy()
        base += np.random.uniform(-0.03, 0.03, base.shape)
        configs.append(np.clip(base, 0.05, 0.95))
        
    return configs

def optimize_centers(centers_init, n, mu_steps):
    """Runs homotopy penalty optimization on centers and radii."""
    # Initialize radii to a safe feasible value
    params = np.concatenate([centers_init[:, 0], centers_init[:, 1], np.full(n, 0.08)])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    for mu in mu_steps:
        res = minimize(
            objective_smooth, 
            params, 
            args=(n, mu), 
            method='L-BFGS-B', 
            bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-12}
        )
        params = res.x
        
    return params[:2*n].reshape(n, 2), params[2*n:]

def solve_radii_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c = np.ones(n) * -1.0  # Minimize -sum(r) => Maximize sum(r)
    
    # Upper bounds for each radius based on distance to square boundaries
    max_r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    max_r = np.maximum(max_r, 1e-9)  # Prevent zero bounds
    bounds = [(0.0, mr) for mr in max_r]
    
    # Pairwise distance constraints: r_i + r_j <= d_ij
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'disp': False})
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    # Fallback if LP fails
    return np.full(n, 0.01), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = get_init_configs(n)
    mu_steps = [10.0, 100.0, 1000.0, 5000.0]
    
    for cfg in configs:
        centers_opt, _ = optimize_centers(cfg, n, mu_steps)
        
        # Extract optimal radii for these centers via LP
        radii_lp, sum_lp = solve_radii_lp(centers_opt)
        
        # Quick validity verification
        valid = True
        for i in range(n):
            x, y, r = centers_opt[i, 0], centers_opt[i, 1], radii_lp[i]
            if r < 0 or x - r < -1e-12 or x + r > 1.0 + 1e-12 or y - r < -1e-12 or y + r > 1.0 + 1e-12:
                valid = False
                break
                
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(centers_opt[i, 0] - centers_opt[j, 0], 
                                 centers_opt[i, 1] - centers_opt[j, 1])
                    if d < radii_lp[i] + radii_lp[j] - 1e-12:
                        valid = False
                        break
                if not valid:
                    break
                    
        if valid and sum_lp > best_sum:
            best_sum = sum_lp
            best_centers = centers_opt.copy()
            best_radii = radii_lp.copy()
            
    # Safety scaling to guarantee strict numerical validity
    if best_centers is not None:
        best_radii *= 0.999999
        best_sum = float(np.sum(best_radii))
    else:
        # Absolute fallback
        best_centers = configs[1]
        best_radii = np.full(n, 0.09)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
