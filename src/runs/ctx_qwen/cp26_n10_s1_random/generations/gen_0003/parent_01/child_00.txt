import numpy as np
from scipy.optimize import minimize, linprog

def objective(vars_array, n):
    """Objective: minimize negative common radius R."""
    return -vars_array[2 * n]

def constraint_func(vars_array, n):
    """
    Computes inequality constraints >= 0 for valid equal-radius packing.
    Constraints:
    1. Boundary: x - R >= 0, 1 - x - R >= 0, y - R >= 0, 1 - y - R >= 0
    2. Non-overlap: dist_ij^2 - (2R)^2 >= 0
    """
    cx = vars_array[:n]
    cy = vars_array[n:2 * n]
    R = vars_array[2 * n]
    
    # Boundary constraints
    c = np.concatenate([
        cx - R,
        1.0 - cx - R,
        cy - R,
        1.0 - cy - R
    ])
    
    # Pairwise non-overlap constraints (vectorized)
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    d2 = cx_m**2 + cy_m**2
    
    mask = np.triu_indices(n, k=1)
    c = np.concatenate([c, d2[mask] - 4.0 * R**2])
    
    return c

                pts.append([x + jitter[0], y + jitter[1]])
                x += 2 * r0
            y += dy
            row += 1

            pts.append([0.5 + jitter[0], 0.5 + jitter[1]])

            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    radii *= scale * 0.999999
    sum_radii = float(np.sum(radii))
    
    return best_centers, radii, sum_radii
