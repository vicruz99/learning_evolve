import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars_arr):
    """Minimize negative sum of radii."""
    return -np.sum(vars_arr[2 * N:])

def constraints(vars_arr):
    """Returns array of constraint values >= 0 for valid packing."""
    cx = vars_arr[:N]
    cy = vars_arr[N:2 * N]
    r = vars_arr[2 * N:]
    
    cons = []
    # Boundary constraints: circle inside [0,1]x[0,1]
    cons.extend(cx - r)
    cons.extend(1.0 - cx - r)
    cons.extend(cy - r)
    cons.extend(1.0 - cy - r)
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    
    d2 = cx_m**2 + cy_m**2
    rs2 = r_m**2
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    cons.extend((d2 - rs2)[mask])
    
    return np.array(cons)

def get_initial_radii(centers):
    """Compute a safe, feasible initial radius for each center."""
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        # Distance to nearest boundary
        min_d = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        # Distance to nearest other center
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                if d < min_d:
                    min_d = d / 2.0
        # Use 40% to guarantee strict feasibility initially
        radii[i] = min_d * 0.40
    return radii

def generate_configs(rng):
    """Generate diverse initial center configurations."""
    configs = []
    
    # 1. Hexagonal lattice
    hex_pts = []
    r0 = 0.10
    y = r0
    row = 0
    dy = np.sqrt(3) * r0
    while len(hex_pts) < N and y + r0 < 1.0:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x + r0 < 1.0 and len(hex_pts) < N:
            hex_pts.append([x, y])
            x += 2.0 * r0
        y += dy
        row += 1
    while len(hex_pts) < N:
        hex_pts.append([0.5, 0.5])
    configs.append(np.array(hex_pts[:N]))
    
    # 2. Perturbed hex
    configs.append(np.clip(configs[0] + rng.uniform(-0.03, 0.03, (N, 2)), 0.05, 0.95))
    
    # 3. Uniform random
    configs.append(rng.uniform(0.15, 0.85, (N, 2)))
    
    # 4. Corner-heavy layout
    corners = np.array([[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]])
    rest = rng.uniform(0.3, 0.7, (N - 4, 2))
    configs.append(np.vstack([corners, rest]))
    
    # 5. Grid-like layout
    grid_x = np.linspace(0.15, 0.85, 5)
    grid_y = np.linspace(0.15, 0.85, 5)
    pts = np.array([(x, y) for y in grid_y for x in grid_x])
    configs.append(pts[:N])
    
    # 6-15. Additional random variants for diversity
    for _ in range(10):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    configs = generate_configs(rng)
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    # Main optimization loop over diverse starts
    for cfg in configs:
        r_init = get_initial_radii(cfg)
        # Break symmetry: vary initial radii while maintaining feasibility
        variation = rng.uniform(0.85, 1.15, N)
        r_init = np.minimum(r_init * variation, r_init)
        
        x0 = np.zeros(3 * N)
        x0[:N] = cfg[:, 0]
        x0[N:2 * N] = cfg[:, 1]
        x0[2 * N:] = r_init
        
        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds,
                constraints=cons_dict,
                options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False}
            )
            
            if np.isfinite(res.fun):
                cx = res.x[:N]
                cy = res.x[N:2 * N]
                r = res.x[2 * N:]
                
                centers = np.column_stack((cx, cy))
                radii = np.maximum(r, 1e-9)
                
                # Fast validity check
                valid = True
                if np.any(centers[:, 0] < radii - 1e-11) or np.any(centers[:, 0] > 1.0 - radii + 1e-11) or \
                   np.any(centers[:, 1] < radii - 1e-11) or np.any(centers[:, 1] > 1.0 - radii + 1e-11):
                    valid = False
                    
                if valid:
                    cx_m = centers[:, 0][:, None] - centers[:, 0][None, :]
                    cy_m = centers[:, 1][:, None] - centers[:, 1][None, :]
                    d2 = cx_m**2 + cy_m**2
                    rs = radii[:, None] + radii[None, :]
                    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
                    if np.any(d2[mask] < rs[mask]**2 - 1e-10):
                        valid = False
                        
                if valid:
                    s = np.sum(radii)
                    if s > best_sum:
                        best_sum = s
                        best_centers = centers.copy()
                        best_radii = radii.copy()
        except Exception:
            continue
            
    # Jitter phase: escape local minima near the best found solution
    if best_centers is not None:
        for _ in range(15):
            jitter = rng.normal(0, 0.004, best_centers.shape)
            jittered_c = np.clip(best_centers + jitter, 0.02, 0.98)
            jittered_r = best_radii * 0.92  # Shrink slightly to guarantee feasible start
            
            x0_jit = np.zeros(3 * N)
            x0_jit[:N] = jittered_c[:, 0]
            x0_jit[N:2 * N] = jittered_c[:, 1]
            x0_jit[2 * N:] = jittered_r
            
            try:
                res = minimize(
                    objective, x0_jit, method='SLSQP', bounds=bounds,
                    constraints=cons_dict, options={'maxiter': 2000, 'ftol': 1e-13}
                )
                if np.isfinite(res.fun):
                    cx = res.x[:N]
                    cy = res.x[N:2 * N]
                    r = res.x[2 * N:]
                    c_opt = np.column_stack((cx, cy))
                    r_opt = np.maximum(r, 1e-9)
                    
                    valid = True
                    if np.any(c_opt[:, 0] < r_opt - 1e-11) or np.any(c_opt[:, 0] > 1.0 - r_opt + 1e-11) or \
                       np.any(c_opt[:, 1] < r_opt - 1e-11) or np.any(c_opt[:, 1] > 1.0 - r_opt + 1e-11):
                        valid = False
                    if valid:
                        cx_m = c_opt[:, 0][:, None] - c_opt[:, 0][None, :]
                        cy_m = c_opt[:, 1][:, None] - c_opt[:, 1][None, :]
                        d2 = cx_m**2 + cy_m**2
                        rs = r_opt[:, None] + r_opt[None, :]
                        mask = np.triu(np.ones((N, N), dtype=bool), k=1)
                        if np.any(d2[mask] < rs[mask]**2 - 1e-10):
                            valid = False
                            
                    if valid:
                        s = np.sum(r_opt)
                        if s > best_sum:
                            best_sum = s
                            best_centers = c_opt.copy()
                            best_radii = r_opt.copy()
            except Exception:
                continue

    # Fallback if optimization fails completely
    if best_centers is None:
        best_centers = configs[0]
        best_radii = get_initial_radii(best_centers)
        best_sum = np.sum(best_radii)

    # Final strict safety scaling to guarantee validity against 1e-12 tolerance
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= max(scale * 0.99999, 0.0)
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
