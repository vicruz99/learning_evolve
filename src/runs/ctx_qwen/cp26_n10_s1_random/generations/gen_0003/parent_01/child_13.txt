import numpy as np
from scipy.optimize import minimize, linprog

def penalty_function(x_vars, n, mu):
    cx = x_vars[:n]
    cy = x_vars[n:2*n]
    r = x_vars[2*n:3*n]
    
    # Objective: maximize sum of radii => minimize negative sum
    obj = -np.sum(r)
    
    # Boundary violation penalties
    obj += mu * np.sum(np.maximum(0, r - cx)**2)
    obj += mu * np.sum(np.maximum(0, cx + r - 1.0)**2)
    obj += mu * np.sum(np.maximum(0, r - cy)**2)
    obj += mu * np.sum(np.maximum(0, cy + r - 1.0)**2)
    
    # Pairwise overlap violation penalties
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    dists = np.sqrt(cx_m**2 + cy_m**2)
    np.fill_diagonal(dists, 1.0)  # Avoid self-interaction zeros
    
    r_sum = r[:, None] + r[None, :]
    viol = np.maximum(0, r_sum - dists)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    obj += mu * np.sum((viol * mask)**2)
    
    return obj

def run_lp_for_radii(centers):
    n = centers.shape[0]
    # Distance to nearest boundary for each circle
    limits = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1 - centers[:, 1]))
    limits = np.maximum(limits, 0.0)
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # LP: maximize sum(r) => minimize -sum(r)
    c_lp = -np.ones(n)
    bounds_lp = [(0.0, lim) for lim in limits]
    
    # Constraints: r_i + r_j <= dist(i, j)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattices with different starting radii
    for r_init in [0.09, 0.10, 0.11]:
        pts = []
        y = r_init
        row = 0
        dy = np.sqrt(3) * r_init
        while y + r_init < 1.0 and len(pts) < n:
            shift = r_init if row % 2 == 1 else 0.0
            x = r_init + shift
            while x + r_init < 1.0 and len(pts) < n:
                pts.append([x, y])
                x += 2.0 * r_init
            y += dy
            row += 1
        # Fill remaining spots if any
        while len(pts) < n:
            pts.append([0.5, 0.5])
        inits.append(np.array(pts[:n]))
        
    # 2. Random perturbations of hex lattice
    rng = np.random.RandomState(42)
    for _ in range(8):
        base = inits[0].copy()
        base += rng.uniform(-0.03, 0.03, base.shape)
        base = np.clip(base, 0.05, 0.95)
        inits.append(base)
        
    # 3. Fully random distributions
    for _ in range(5):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    bounds_vars = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    for init_cfg in inits:
        cx0 = init_cfg[:, 0]
        cy0 = init_cfg[:, 1]
        r0 = np.ones(n) * 0.08
        
        current_x = np.concatenate([cx0, cy0, r0])
        
        # Continuation method: gradually increase penalty weight to enforce constraints
        for mu in [10.0, 100.0, 1000.0]:
            try:
                res = minimize(penalty_function, current_x, args=(n, mu),
                               method='L-BFGS-B', bounds=bounds_vars,
                               options={'maxiter': 3000, 'ftol': 1e-12})
                current_x = res.x
            except Exception:
                break
                
        # Alternating optimization: LP for radii -> Re-optimize positions -> LP
        for _ in range(3):
            centers = current_x[:2*n].reshape(n, 2)
            radii = np.maximum(current_x[2*n:], 1e-9)
            
            # Exact radius maximization for fixed centers
            lp_radii = run_lp_for_radii(centers)
            if lp_radii is not None:
                radii = lp_radii
                current_x[2*n:] = radii
                
            # Refine centers with high penalty to accommodate radii
            try:
                res = minimize(penalty_function, current_x, args=(n, 2000.0),
                               method='L-BFGS-B', bounds=bounds_vars,
                               options={'maxiter': 2000, 'ftol': 1e-12})
                current_x = res.x
            except Exception:
                break
                
        cx = current_x[:n]
        cy = current_x[n:2*n]
        r = np.maximum(current_x[2*n:], 1e-9)
        centers = np.column_stack((cx, cy))
        radii = r
        
        # Strict validity check to filter out numerically unstable results
        valid = True
        if np.any(centers[:, 0] < radii - 1e-5) or np.any(centers[:, 0] > 1 - radii + 1e-5) or \
           np.any(centers[:, 1] < radii - 1e-5) or np.any(centers[:, 1] > 1 - radii + 1e-5):
            valid = False
        if valid:
            dists = np.sqrt(((centers[:, 0][:, None] - centers[:, 0][None, :])**2) + 
                            ((centers[:, 1][:, None] - centers[:, 1][None, :])**2))
            np.fill_diagonal(dists, np.inf)
            r_sum = radii[:, None] + radii[None, :]
            if np.any(dists < r_sum - 1e-5):
                valid = False
                
        if valid:
            s = np.sum(radii)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()
                
    # Final LP pass on the best configuration found
    if best_centers is not None:
        final_radii = run_lp_for_radii(best_centers)
        if final_radii is not None:
            best_radii = final_radii
            best_sum = np.sum(best_radii)
            
    # Safety scaling to guarantee strict validity within checker tolerance
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-9:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
                
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(best_centers[i, 0]-best_centers[j, 0], 
                             best_centers[i, 1]-best_centers[j, 1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-9:
                    scale = min(scale, d/rs)
                    
        best_radii *= scale * 0.999999
        best_sum = np.sum(best_radii)
    else:
        # Fallback to a safe hex grid
        r_fb = 0.095
        pts = []
        y = r_fb
        row = 0
        while len(pts) < n:
            shift = r_fb if row % 2 else 0
            x = r_fb + shift
            while x + r_fb <= 1.0 and len(pts) < n:
                pts.append([x, y])
                x += 2 * r_fb
            y += np.sqrt(3) * r_fb
            row += 1
        best_centers = np.array(pts[:n])
        best_radii = np.full(n, r_fb)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
