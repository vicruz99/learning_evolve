import numpy as np
from scipy.optimize import minimize, linprog

def objective(vars_arr):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_arr[2::3])

def constraint_func(vars_arr, n):
    """Compute all inequality constraints >= 0 for valid packing."""
    xs = vars_arr[0::3]
    ys = vars_arr[1::3]
    rs = vars_arr[2::3]
    
    # Boundary constraints
    b_cons = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Pairwise non-overlap constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    dist_sq = dx**2 + dy**2
    r_sum_sq = dr**2
    
    idx = np.triu_indices(n, k=1)
    p_cons = dist_sq[idx] - r_sum_sq[idx]
    
    return np.concatenate([b_cons, p_cons])

def solve_radii_lp(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    c_obj = np.ones(n) * -1
    bounds = [(0.0, lim) for lim in limits]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def generate_initial_configs(n, seed=42):
    """Generates diverse initial configurations for the optimizer."""
    configs = []
    rng = np.random.default_rng(seed)
    
    # Hexagonal lattice variations
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [5,5,5,6,5], 
        [5,6,6,5,4], [4,6,6,6,4], [5,5,5,5,6], [6,5,5,5,5]
    ]
    for pat in patterns:
        pts = []
        y = 0.09
        r_init = 0.09
        for row_idx, cnt in enumerate(pat):
            shift = r_init if row_idx % 2 else 0.0
            width = (cnt - 1) * 2 * r_init
            x_start = 0.5 - width/2.0 + shift
            for k in range(cnt):
                if len(pts) >= n: break
                pts.append([x_start + k * 2 * r_init, y])
            y += np.sqrt(3) * r_init
        while len(pts) < n:
            pts.append([0.5, 0.5])
        configs.append(np.array(pts[:n]))
        
    # Grid + center fallback
    gx = np.linspace(0.12, 0.88, 5)
    gy = np.linspace(0.12, 0.88, 5)
    pts_grid = np.array([[x, y] for y in gy for x in gx])
    pts_grid = np.vstack([pts_grid, [0.5, 0.5]])
    configs.append(pts_grid)
    
    # Random perturbations to break symmetry
    for _ in range(10):
        base = configs[0].copy()
        base += rng.uniform(-0.025, 0.025, base.shape)
        base = np.clip(base, 0.05, 0.95)
        configs.append(base)
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * n
    cons_dict = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = generate_initial_configs(n)
    
    for cfg in configs:
        v0 = np.zeros(3 * n)
        v0[0::3] = cfg[:, 0]
        v0[1::3] = cfg[:, 1]
        v0[2::3] = 0.09
        
        # Phase 1: Joint SLSQP optimization
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                          constraints=cons_dict, options={'maxiter': 15000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                cx = res.x[0::3]
                cy = res.x[1::3]
                r_slqp = res.x[2::3]
                
                # Phase 2: LP refinement for optimal radii given centers
                centers_temp = np.column_stack((cx, cy))
                r_lp = solve_radii_lp(centers_temp, n)
                
                if r_lp is not None:
                    s = np.sum(r_lp)
                    if s > best_sum:
                        best_sum = s
                        best_centers = centers_temp.copy()
                        best_radii = r_lp.copy()
                else:
                    s = np.sum(r_slqp)
                    if s > best_sum:
                        best_sum = s
                        best_centers = np.column_stack((cx, cy)).copy()
                        best_radii = r_slqp.copy()
                        
                # Phase 3: Re-optimize centers using LP-optimized radii as initialization
                v0_new = np.zeros(3*n)
                v0_new[0::3] = cx
                v0_new[1::3] = cy
                v0_new[2::3] = r_lp if r_lp is not None else r_slqp
                
                res2 = minimize(objective, v0_new, method='SLSQP', bounds=bounds,
                               constraints=cons_dict, options={'maxiter': 10000, 'ftol': 1e-14})
                if np.isfinite(res2.fun):
                    cx2 = res2.x[0::3]
                    cy2 = res2.x[1::3]
                    r_slqp2 = res2.x[2::3]
                    s2 = np.sum(r_slqp2)
                    if s2 > best_sum:
                        best_sum = s2
                        best_centers = np.column_stack((cx2, cy2)).copy()
                        best_radii = r_slqp2.copy()
                        
        except Exception:
            continue
            
    # Fallback configuration
    if best_centers is None:
        best_centers = configs[0]
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    # Strict safety scaling to guarantee validity within 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
