import numpy as np
from scipy.optimize import minimize

N = 26

def con_eq(vars):
    """Inequality constraints >= 0 for equal-radii packing."""
    cx = vars[:2*N].reshape(N, 2)
    r = vars[2*N]
    
    con = []
    # Boundary constraints
    con.append(cx[:, 0] - r)
    con.append(1.0 - cx[:, 0] - r)
    con.append(cx[:, 1] - r)
    con.append(1.0 - cx[:, 1] - r)
    
    # Pairwise non-overlap: ||c_i - c_j||^2 >= 4r^2
    diff = cx[:, None, :] - cx[None, :, :]
    d2 = np.sum(diff**2, axis=2)
    np.fill_diagonal(d2, 1.0)  # Ignore self-distance
    idx = np.triu_indices(N, k=1)
    con.append(d2[idx] - 4.0 * r**2)
    
    return np.concatenate(con)

def obj_eq(vars):
    """Objective: minimize negative radius => maximize radius."""
    return -vars[2*N]

def get_init_hex(row_counts, r0):
    """Generate initial hexagonal lattice positions."""
    pts = []
    y = r0
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 else 0.0
        row_w = (cnt - 1) * 2 * r0
        x_start = 0.5 - row_w / 2.0 + shift
        for c in range(cnt):
            pts.append([x_start + c * 2 * r0, y])
        y += np.sqrt(3) * r0
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds_c = [(0.0, 1.0)] * (2*N) + [(0.05, 0.2)]
    con_dict = {'type': 'ineq', 'fun': con_eq}
    
    best_r = 0.0
    best_cx = None
    
    # Hexagonal row patterns that yield ~26 circles
    patterns = [
        [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 5, 6, 5, 5],
        [5, 5, 5, 6, 5], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4],
        [4, 6, 6, 6, 4], [5, 5, 5, 5, 5, 1], [5, 5, 6, 5, 5]
    ]
    
    configs = []
    rng = np.random.default_rng(42)
    
    for pat in patterns:
        init = get_init_hex(pat, 0.09)
        configs.append(init)
        # Add controlled perturbations to escape symmetry traps
        for _ in range(2):
            p = init + rng.uniform(-0.015, 0.015, init.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    for cfg in configs:
        # Compute a feasible starting radius
        dists = np.linalg.norm(cfg[:, None] - cfg[None, :], axis=2)
        np.fill_diagonal(dists, 1.0)
        min_pair = np.min(dists)
        
        wall_d = np.minimum(
            np.minimum(cfg[:, 0], 1.0 - cfg[:, 0]),
            np.minimum(cfg[:, 1], 1.0 - cfg[:, 1])
        )
        min_wall = np.min(wall_d)
        
        # Start slightly smaller than the theoretical max to ensure strict feasibility
        r_start = min(min_pair, min_wall) / 2.0 * 0.95
        
        v0 = np.concatenate([cfg.flatten(), [r_start]])
        
        try:
            res = minimize(
                obj_eq, v0, method='SLSQP', bounds=bounds_c,
                constraints=con_dict,
                options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False}
            )
            
            if np.isfinite(res.fun):
                r_opt = res.x[2*N]
                cx_opt = res.x[:2*N].reshape(N, 2)
                
                # Strict validation
                valid = True
                if r_opt < 0.01:
                    valid = False
                else:
                    # Check boundaries
                    if np.any(cx_opt[:, 0] < r_opt - 1e-10) or np.any(cx_opt[:, 0] > 1.0 - r_opt + 1e-10):
                        valid = False
                    if np.any(cx_opt[:, 1] < r_opt - 1e-10) or np.any(cx_opt[:, 1] > 1.0 - r_opt + 1e-10):
                        valid = False
                    
                    # Check overlaps
                    d2 = np.sum((cx_opt[:, None] - cx_opt[None, :])**2, axis=2)
                    np.fill_diagonal(d2, 1.0)
                    idx = np.triu_indices(N, k=1)
                    if np.any(d2[idx] < 4.0 * r_opt**2 - 1e-10):
                        valid = False
                        
                if valid and r_opt > best_r:
                    best_r = r_opt
                    best_cx = cx_opt.copy()
        except Exception:
            continue
            
    # Fallback if optimization fails
    if best_cx is None:
        best_cx = configs[0]
        best_r = 0.09
        
    # Final safety scaling to guarantee strict numerical validity
    cx_final = best_cx.copy()
    radii = np.full(N, best_r)
    
    scale = 1.0
    for i in range(N):
        x, y = cx_final[i]
        r = radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    dists = np.linalg.norm(cx_final[:, None] - cx_final[None, :], axis=2)
    np.fill_diagonal(dists, 1.0)
    idx = np.triu_indices(N, k=1)
    for k in range(len(idx[0])):
        i, j = idx[0][k], idx[1][k]
        d = dists[i, j]
        rs = radii[i] + radii[j]
        if rs > 1e-12:
            scale = min(scale, d / rs)
            
    # Apply scale with minimal margin for floating point comparisons
    radii *= scale * 0.999999
    sum_r = float(np.sum(radii))
    
    return cx_final, radii, sum_r
