from scipy.optimize import minimize, linprog

def penalty_objective(vars_arr, n, mu):
    """
    Computes the penalized objective function for circle packing.
    Minimizes negative sum of radii while penalizing boundary and overlap violations.
    """
    cx = vars_arr[0::3]
    cy = vars_arr[1::3]
    r = vars_arr[2::3]

    # Primary objective: maximize sum of radii => minimize -sum(r)
    obj = -np.sum(r)

    # Boundary violation penalties: circle must be inside [0,1]x[0,1]
    # r - x <= 0, r + x - 1 <= 0, r - y <= 0, r + y - 1 <= 0
    b_viol = np.concatenate([
        np.maximum(0.0, r - cx),
        np.maximum(0.0, r + cx - 1.0),
        np.maximum(0.0, r - cy),
        np.maximum(0.0, r + cy - 1.0)
    ])

    # Pairwise overlap penalties: dist(i,j) >= r_i + r_j
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, 1.0)  # Avoid zero division or self-comparison
    r_sum = r[:, None] + r[None, :]
    o_viol = np.maximum(0.0, r_sum - dist)
    
    # Only count each pair once (upper triangle)
    triu_idx = np.triu_indices(n, k=1)
    o_viol = o_viol[triu_idx]

    # Add quadratic penalty terms scaled by mu
    obj += mu * (np.sum(b_viol**2) + np.sum(o_viol**2))
    return obj

def solve_radii_lp(centers):
    """
    Given fixed centers, solves the LP to maximize sum of radii.
    Constraints: r_i <= dist(i, boundary) and r_i + r_j <= dist(i, j)
    """
    n = centers.shape[0]
    
    # Max radius allowed by boundaries for each circle
    limits = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    limits = np.maximum(limits, 0.0)

    # Objective: maximize sum(r) => minimize -sum(r)
    c = -np.ones(n)
    bounds = [(0.0, lim) for lim in limits]

    # Pairwise distance constraints: r_i + r_j <= dist(i,j)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)

    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    b_ub = dists[idx_i, idx_j]

    for k in range(m):
        A_ub[k, idx_i[k]] = 1.0
        A_ub[k, idx_j[k]] = 1.0

    try:
        # Use HiGHS solver for speed and robustness
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0
