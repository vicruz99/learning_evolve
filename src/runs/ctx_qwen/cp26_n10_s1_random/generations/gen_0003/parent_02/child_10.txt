def objective(vars_arr):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_arr[2::3])

def get_constraints(vars_arr, n):
    """Compute all inequality constraints >= 0 for valid packing."""
    xs = vars_arr[0::3]
    ys = vars_arr[1::3]
    rs = vars_arr[2::3]
    
    # Boundary constraints: circles must stay inside [0,1]x[0,1]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])

    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]

    c = np.concatenate([c, (dx[idx]**2 + dy[idx]**2) - dr[idx]**2])

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    
    # Max radius for each circle limited by boundaries
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))

    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in limits]

    b_ub = dists[idx]

        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
