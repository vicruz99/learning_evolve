import numpy as np
from scipy.optimize import minimize, linprog

def objective_equal_r(params, n):
    """Objective function: minimize negative radius (maximize r)."""
    return -params[-1]

def constraints_equal_r(params, n):
    """Compute inequality constraints >= 0 for valid packing with equal radius r."""
    xs = params[0:n]
    ys = params[n:2*n]
    r = params[2*n]
    
    con = []
    # Boundary constraints
    con.extend(xs - r)
    con.extend(1.0 - xs - r)
    con.extend(ys - r)
    con.extend(1.0 - ys - r)
    
    # Pairwise non-overlap constraints: squared distance >= 4r^2
    xs_m = xs[:, None] - xs[None, :]
    ys_m = ys[:, None] - ys[None, :]
    dist_sq = xs_m**2 + ys_m**2
    idx = np.triu_indices(n, k=1)
    con.extend(dist_sq[idx] - 4.0 * r**2)
    
    return np.array(con)

def generate_hex_config(row_counts, r_init=0.09, n=26):
    """Generate initial positions on a hexagonal lattice with specified row counts."""
    pts = []
    y = r_init
    for idx, cnt in enumerate(row_counts):
        if cnt <= 0:
            y += np.sqrt(3) * r_init
            continue
            
        shift = r_init if idx % 2 == 1 else 0.0
        width = (cnt - 1) * 2 * r_init
        x_start = 0.5 - width / 2.0 + shift
        
        for k in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x_start + k * 2 * r_init, y])
        y += np.sqrt(3) * r_init
        
    return np.array(pts[:n])

def solve_lp_radii(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-12)
    
    c = -np.ones(n)
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 0.08), 0.0

    for rc in row_configs:
