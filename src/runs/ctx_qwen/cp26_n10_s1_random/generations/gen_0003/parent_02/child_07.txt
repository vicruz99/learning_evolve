import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: 
      0 <= r_i <= min distance to wall
      r_i + r_j <= distance(i, j)
    """
    n = centers.shape[0]
    
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Boundary limits for each circle
    b_limits = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    
    # LP: minimize -sum(r) => maximize sum(r)
    c_obj = -np.ones(n)
    bounds = [(0.0, b_limits[i]) for i in range(n)]
    
    # Pairwise non-overlap constraints: r_i + r_j <= d_ij
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return None, 0.0

def lp_objective(vars, n):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = vars.reshape(n, 2)
    # Strictly keep centers inside the square to avoid degenerate LP bounds
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    radii, s = solve_radii_lp(centers)
    if radii is None:
        return 1e6
    return -s

def generate_initial_configs(n):
    """Generates diverse initial center configurations."""
    configs = []
    rng = np.random.default_rng(42)
    
    # 1. Hexagonal lattice patterns with various row distributions
    patterns = [
        [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 5, 6, 5, 5],
        [5, 5, 5, 6, 5], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4],
        [5, 6, 6, 5, 4], [4, 6, 6, 6, 4], [5, 5, 5, 5, 5, 1]
    ]
    
    for pat in patterns:
        pts = []
        y = 0.1
        row_idx = 0
        for cnt in pat:
            shift = 0.05 if row_idx % 2 == 1 else 0.0
            row_width = (cnt - 1) * 0.2
            x_start = 0.5 - row_width / 2.0 + shift
            for c in range(cnt):
                if len(pts) < n:
                    pts.append([x_start + c * 0.2, y])
            y += 0.175
            row_idx += 1
        configs.append(np.array(pts[:n]))
        
    # 2. Grid layout
    gx = np.linspace(0.12, 0.88, 5)
    gy = np.linspace(0.12, 0.88, 5)
    pts_grid = [[x, y] for y in gy for x in gx]
    pts_grid.append([0.5, 0.5])
    configs.append(np.array(pts_grid[:n]))
    
    # 3. Random dense packings
    for _ in range(5):
        pts_rand = rng.uniform(0.1, 0.9, (n, 2))
        configs.append(pts_rand)
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = generate_initial_configs(n)
    rng = np.random.default_rng(123)

    for cfg in configs:
        # Try a few perturbations to break symmetry and escape local minima

            pert = cfg + rng.uniform(-0.04, 0.04, cfg.shape)
            pert = np.clip(pert, 0.05, 0.95)
            
            try:
                # Nelder-Mead is robust for the non-smooth LP objective
                res = minimize(
                    lp_objective, 
                    pert.flatten(), 
                    args=(n,),
                    method='Nelder-Mead', 
                    options={'maxiter': 4000, 'xatol': 1e-5, 'fatol': 1e-6}
                )
                
                if np.isfinite(res.fun):
                    centers_opt = res.x.reshape(n, 2)
                    radii_opt, s_exact = solve_radii_lp(centers_opt)
                    
                    if radii_opt is not None and s_exact > best_sum:
                        best_sum = s_exact
                        best_centers = centers_opt.copy()
                        best_radii = radii_opt.copy()
            except Exception:
                continue
                
    # Fallback if optimization fails
    if best_centers is None:
        best_centers = configs[0]
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)

    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)

        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    # Apply scaling with a tiny buffer for floating point safety
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
