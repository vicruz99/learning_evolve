import numpy as np
from scipy.optimize import minimize, linprog

def objective(vars_arr):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_arr[2::3])

def get_constraints(vars_arr, n):
    """Compute all inequality constraints >= 0 for valid packing."""
    xs = vars_arr[0::3]
    ys = vars_arr[1::3]
    rs = vars_arr[2::3]
    
    con = []
    # Boundary constraints
    con.extend(xs - rs)
    con.extend(1.0 - xs - rs)
    con.extend(ys - rs)
    con.extend(1.0 - ys - rs)
    
    # Pairwise non-overlap constraints: squared distance >= squared sum of radii
    xs_m = xs[:, None] - xs[None, :]
    ys_m = ys[:, None] - ys[None, :]
    rs_m = rs[:, None] + rs[None, :]
    
    dist_sq = xs_m**2 + ys_m**2
    r_sum_sq = rs_m**2
    
    idx = np.triu_indices(n, k=1)
    con.extend(dist_sq[idx] - r_sum_sq[idx])
    
    return np.array(con)

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Max radius for each circle based on boundaries
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 0.0)
    
    c = -np.ones(n)  # Maximize sum(r) -> minimize -sum(r)
    bounds = [(0.0, lim) for lim in limits]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Constraint matrix: r_i + r_j <= dist(i, j)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def generate_inits(n, rng):
    """Generates a diverse set of initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattices with varying base radii
    for r0 in [0.085, 0.09, 0.095, 0.10]:
        pts = []
        y = r0
        row = 0
        while len(pts) < n:
            shift = r0 if row % 2 == 1 else 0.0
            x = r0 + shift
            while x + r0 <= 1.0 and len(pts) < n:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        if len(pts) < n:
            while len(pts) < n:
                pts.append([0.5, 0.5])
        inits.append(np.array(pts[:n]))
        
    # 2. Grid patterns
    for k in range(4, 7):
        g = np.linspace(0.12, 0.88, k)
        pts = []
        for y in g:
            for x in g:
                if len(pts) < n:
                    pts.append([x, y])
        while len(pts) < n:
            pts.append([0.5, 0.5])
        inits.append(np.array(pts[:n]))
        
    # 3. Corner-focused dense packing
    corner_pts = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]
    center_grid = [(0.5, 0.5)]
    ring = [(0.5, 0.2), (0.5, 0.8), (0.2, 0.5), (0.8, 0.5)]
    pts = corner_pts + center_grid + ring + [rng.uniform(0.2, 0.8, 2) for _ in range(n - 10)]
    inits.append(np.array(pts[:n]))
        
    # 4. Random dense configurations
    for _ in range(5):
        pts = rng.uniform(0.08, 0.92, (n, 2))
        inits.append(pts)
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    rng = np.random.default_rng(42)

        v0[2::3] = 0.09

                objective, 
                v0, 
                method='SLSQP', 
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': get_constraints, 'args': (n,)},
                options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False}
            )
            
            if np.isfinite(res.fun):
                con_vals = get_constraints(res.x, n)
                if np.min(con_vals) >= -1e-9:
                    cx = res.x[0::3]
                    cy = res.x[1::3]
                    r = res.x[2::3]

        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
