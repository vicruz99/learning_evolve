import numpy as np
from scipy.optimize import minimize

def compute_constraints(vars_flat, n):
    """
    Computes inequality constraints for equal-radius packing.
    Constraints: boundary distances >= t, pairwise squared distances >= 4*t^2
    Returns an array of constraint values (all must be >= 0).
    """
    t = vars_flat[-1]
    xs = vars_flat[0::2]
    ys = vars_flat[1::2]
    
    # Boundary constraints: x >= t, 1-x >= t, y >= t, 1-y >= t
    c = np.concatenate([
        xs - t,
        1.0 - xs - t,
        ys - t,
        1.0 - ys - t
    ])
    
    # Pairwise squared distance constraints: ||c_i - c_j||^2 >= 4*t^2
    i, j = np.triu_indices(n, k=1)
    dx = xs[i] - xs[j]
    dy = ys[i] - ys[j]
    dist_sq = dx**2 + dy**2
    
    # Concatenate pairwise constraints
    pc = dist_sq - 4.0 * t * t
    return np.concatenate([c, pc])

def objective(vars_flat):
    """Objective: maximize t => minimize -t"""
    return -vars_flat[-1]

def get_hex_init(r, n, pattern):
    """Generates an initial hexagonal grid of n circles with estimated radius r."""
    pts = []
    y = r
    row_idx = 0
    for cnt in pattern:
        shift = r if row_idx % 2 == 1 else 0.0
        x = r + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r
        y += r * np.sqrt(3.0)
        row_idx += 1
    return np.array(pts[:n])

def run_packing():
    n = 26
    # Hexagonal row distributions that sum to 26
    patterns = [

        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5]
    ]
    
    # Bounds: centers in [0, 1], radius t in [0.06, 0.15]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.06, 0.15)]
    cons = {'type': 'ineq', 'fun': compute_constraints, 'args': (n,)}
    
    best_t = 0.0
    best_centers = None
    
    np.random.seed(42)
    r_init = 0.09
    
    # Run multiple optimization trials with perturbed initializations
    for pat in patterns:
        base_pts = get_hex_init(r_init, n, pat)
        # Center the lattice in the unit square
        cx, cy = base_pts.mean(axis=0)
        base_pts -= np.array([cx - 0.5, cy - 0.5])
        
        for _ in range(8):
            # Perturb to escape local minima
            pts = base_pts + np.random.uniform(-0.025, 0.025, base_pts.shape)
            pts = np.clip(pts, 0.05, 0.95)
            
            # Variables: [x0, y0, x1, y1, ..., x25, y25, t]
            x0 = np.concatenate([pts.flatten(), [0.095]])
            
            try:
                res = minimize(
                    objective, 
                    x0, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons, 
                    options={'maxiter': 5000, 'ftol': 1e-12}
                )
                # Track the best radius found
                if res.x[-1] > best_t:
                    best_t = res.x[-1]
                    best_centers = res.x[:2 * n].reshape(n, 2)
            except Exception:
                continue

    # Fallback if optimization didn't find a valid configuration
    if best_centers is None:
        best_centers = get_hex_init(0.09, n, [5, 5, 5, 5, 6])
        best_t = 0.09

    # Final geometric refinement: compute exact maximal valid radii for the optimized centers
    # This guarantees strict validity against the checker's tolerance
    min_dist = 1.0
    
    # Check distance to boundaries
    for i in range(n):
        x, y = best_centers[i]
        min_dist = min(min_dist, x, 1.0 - x, y, 1.0 - y)

    for i in range(n):

            min_dist = min(min_dist, d / 2.0)

    best_radii = np.full(n, final_t)
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
