To maximize the sum of radii for 26 circles in a unit square, we need to find a configuration that closely approximates the known optimal hexagonal packing density. The theoretical maximum equal radius is approximately $r \approx 0.10139$, yielding a sum of $\approx 2.636$. 

The strategy involves three key phases:
1. **Diverse Initialization**: Generate multiple starting configurations, including a carefully tuned hexagonal lattice pattern (rows of 5, 6, 5, 6, 4 circles), perturbed versions of it, grid layouts, and random valid packings. This ensures we cover different topological basins of attraction.
2. **Multi-Stage Penalty Optimization**: Use `L-BFGS-B` to optimize both centers and radii simultaneously. Instead of hard constraints which can cause solvers to stall, we use a smooth quadratic penalty function. We run the optimization in stages with increasingly large penalty weights (`mu`). This "homotopy" approach allows the solver to first find a feasible basin, then strictly enforce non-overlap and boundary constraints while aggressively expanding radii.
3. **Exact Linear Programming Refinement**: Once centers are optimized, the radius assignment problem becomes a convex Linear Program (LP). We fix the centers and solve $\max \sum r_i$ subject to $r_i + r_j \leq d_{ij}$ and $r_i \leq \text{dist to walls}$. This guarantees the mathematically maximal radii for the given centers. A tiny safety scaling factor ($0.999999$) is applied to satisfy the strict $10^{-12}$ tolerance in the validator.

This hybrid approach separates the non-convex center placement from the convex radius assignment, leveraging the strengths of both gradient-based optimization and exact solvers.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def penalty_objective(vars_flat, n, mu):
    """
    Smooth penalty objective for circle packing.
    Minimizes negative sum of radii while penalizing boundary and overlap violations.
    """
    cx = vars_flat[:2*n].reshape(n, 2)
    r = vars_flat[2*n:]
    
    # Base objective: maximize sum of radii
    obj = -np.sum(r)
    
    # Boundary penalty: circles must stay within [0, 1] x [0, 1]
    obj += mu * np.sum(np.maximum(0, r - cx[:, 0])**2)
    obj += mu * np.sum(np.maximum(0, cx[:, 0] + r - 1.0)**2)
    obj += mu * np.sum(np.maximum(0, r - cx[:, 1])**2)
    obj += mu * np.sum(np.maximum(0, cx[:, 1] + r - 1.0)**2)
    
    # Pairwise overlap penalty: dist(i, j) >= r_i + r_j
    dx = cx[:, 0][:, None] - cx[:, 0][None, :]
    dy = cx[:, 1][:, None] - cx[:, 1][None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)  # Ignore self-distance
    
    rs = r[:, None] + r[None, :]
    viol = np.maximum(0, rs - dist)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    obj += mu * np.sum((viol * mask)**2)
    
    return obj

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # --- Phase 1: Generate Diverse Initial Configurations ---
    inits = []
    
    # 1. Structured Hexagonal Lattice (optimal topology for dense packing)
    # Rows distribution summing to 26: 5, 6, 5, 6, 4
    r_h = 0.101
    pts_hex = []
    rows_dist = [5, 6, 5, 6, 4]
    y = r_h
    for i, count in enumerate(rows_dist):
        shift = r_h if i % 2 == 1 else 0.0
        total_width = (count - 1) * 2 * r_h
        x_start = (1.0 - total_width) / 2.0 + shift
        for k in range(count):
            pts_hex.append([x_start + k * 2 * r_h, y])
        y += r_h * np.sqrt(3)
    inits.append(np.array(pts_hex[:n]))
    
    # 2. Uniform Grid Layout (alternative topology)
    g = np.linspace(0.1, 0.9, 5)
    grid_pts = np.array([[x, y] for y in g for x in g])
    grid_pts = np.vstack([grid_pts, [[0.5, 0.05]]])  # 25 + 1 = 26
    inits.append(grid_pts)
    
    # 3-12. Perturbed Hexagonal & Random Starts
    np.random.seed(42)
    for _ in range(5):
        p = inits[0] + np.random.uniform(-0.03, 0.03, (n, 2))
        inits.append(np.clip(p, 0.05, 0.95))
        
    for _ in range(5):
        inits.append(np.random.uniform(0.1, 0.9, (n, 2)))
        
    # Variable bounds for L-BFGS-B
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    # --- Phase 2: Multi-Stage Penalty Optimization ---
    for cfg in inits:
        x0 = np.zeros(3*n)
        x0[:2*n] = cfg.flatten()
        x0[2*n:] = 0.09  # Initial feasible radius
        
        current_vars = x0.copy()
        
        # Homotopy schedule: gradually increase penalty weight to enforce strict feasibility
        for mu in [500, 2000, 10000, 30000]:
            res = minimize(
                penalty_objective, 
                current_vars, 
                args=(n, mu),
                method='L-BFGS-B', 
                bounds=bounds,
                options={'maxiter': 1500, 'ftol': 1e-13}
            )
            current_vars = res.x
            
        cx_opt = current_vars[:2*n].reshape(n, 2)
        r_opt = current_vars[2*n:]
        
        # --- Phase 3: Exact LP Refinement for Radii ---
        # Maximize sum(r) subject to r_i + r_j <= dist(i,j) and r_i <= wall_dist
        pairs = []
        for i in range(n):
            for j in range(i + 1, n):
                pairs.append((i, j))
                
        num_pairs = len(pairs)
        A_ub = np.zeros((num_pairs + 4*n, n))
        b_ub = np.zeros(num_pairs + 4*n)
        
        idx = 0
        for i, j in pairs:
            d = np.sqrt(np.sum((cx_opt[i] - cx_opt[j])**2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
        for i in range(n):
            x, y = cx_opt[i]
            A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
            A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
            A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
            A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
            
        try:
            lp_res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
            if lp_res.success and np.isfinite(lp_res.fun):
                # Apply microscopic scaling to guarantee strict satisfaction of 1e-12 tolerance
                radii_lp = lp_res.x * 0.999999
                s = np.sum(radii_lp)
                if s > best_sum:
                    best_sum = s
                    best_centers = cx_opt.copy()
                    best_radii = radii_lp.copy()
        except Exception:
            continue
            
    # Fallback safety net (should rarely trigger with this setup)
    if best_centers is None:
        best_centers = inits[0]
        best_radii = np.full(n, 0.09)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
