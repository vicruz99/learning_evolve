import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective_func(vars_flat):
    """Minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def constraint_func(vars_flat):
    """Compute inequality constraints: returns array where each element must be >= 0."""
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    c = []
    # Boundary constraints: circle inside [0,1]x[0,1]
    c.extend(x - r)
    c.extend(1.0 - x - r)
    c.extend(y - r)
    c.extend(1.0 - y - r)
    
    # Pairwise non-overlap constraints using squared distances for smooth gradients
    i, j = np.triu_indices(N, k=1)
    dx = x[i] - x[j]
    dy = y[i] - y[j]
    dr = r[i] + r[j]
    c.extend(dx**2 + dy**2 - dr**2)
    
    return np.array(c)

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    A_rows = []
    b_vals = []
    
    # Pairwise distance constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_rows.append(row)
            b_vals.append(d)
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_rows.append(row)
            b_vals.append(val)
            
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.zeros(n)

def run_packing():
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    np.random.seed(42)
    
    inits = []
    # 1. Hexagonal lattice pattern
    pts = []
    y = 0.05
    row = 0
    while len(pts) < N + 10:
        x = 0.05
        shift = 0.05 if row % 2 else 0.0
        while x < 0.95:
            pts.append([x + shift, y])
            x += 0.10
        y += 0.0866
        row += 1
    inits.append(np.array(pts[:N]))
    
    # 2. Uniform grid pattern
    g = np.linspace(0.1, 0.9, 5)
    pts = [[xx, yy] for yy in g for xx in g]
    pts.append([0.5, 0.95])
    inits.append(np.array(pts[:N]))
    
    # 3. Randomized starts for diversity
    for _ in range(12):
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))
        
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-4, 0.5)] * N
    
    # Stage 1: Optimize centers and radii jointly
    for cfg in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = cfg[:, 0]
        x0[1::3] = cfg[:, 1]
        x0[2::3] = 0.08
        
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraint_func},
                           options={'maxiter': 3000, 'ftol': 1e-12})
            
            if res.success or res.fun < -2.5:
                c_vals = constraint_func(res.x)
                if np.min(c_vals) >= -1e-6:
                    c_opt = res.x[:2*N].reshape(N, 2)
                    r_opt = res.x[2*N:]
                    s = np.sum(r_opt)
                    if s > best_sum:
                        best_sum = s
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
        except Exception:
            continue
            
    # Stage 2: Refine using LP on fixed centers (exploits local spacing variations)
    if best_centers is not None:
        # Try perturbations of the best found layout
        for _ in range(8):
            pert_c = best_centers + np.random.uniform(-0.015, 0.015, (N, 2))
            pert_c = np.clip(pert_c, 0.05, 0.95)
            r_pert = solve_lp_radii(pert_c)
            s_pert = np.sum(r_pert)
            if s_pert > best_sum:
                best_sum = s_pert
                best_centers = pert_c
                best_radii = r_pert
                
    # Fallback safe configuration if optimization unexpectedly fails
    if best_centers is None:
        best_centers = inits[0]
        best_radii = np.full(N, 0.085)
        best_sum = np.sum(best_radii)
        
    # Stage 3: Strict numerical validation and safety scaling
    scale = 1.0
    for _ in range(100):
        ok = True
        for i in range(N):
            x, y = best_centers[i]
            r = best_radii[i]
            if x < r - 1e-9 or x > 1.0-r + 1e-9 or y < r - 1e-9 or y > 1.0-r + 1e-9:
                ok = False; break
        if ok:
            for i in range(N):
                for j in range(i+1, N):
                    d = np.linalg.norm(best_centers[i] - best_centers[j])
                    if d < best_radii[i] + best_radii[j] - 1e-9:
                        ok = False; break
                if not ok: break
        if ok: break
        scale *= 0.99995
        best_radii *= scale
        
    best_sum = float(np.sum(best_radii))
    return best_centers, best_radii, best_sum
