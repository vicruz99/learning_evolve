import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def obj_equal(vars):
    """Objective: maximize t => minimize -t"""
    return -vars[-1]

def cons_equal(vars):
    """Constraints: boundaries >= t and pairwise squared distances >= 4t^2"""
    t = vars[-1]
    x = vars[0::2]
    y = vars[1::2]
    
    # Boundary constraints
    b = np.concatenate([x - t, 1.0 - x - t, y - t, 1.0 - y - t])
    
    # Pairwise squared distance constraints
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    idx = np.triu_indices(N, k=1)
    p = dist_sq[idx] - 4.0 * t * t
    
    return np.concatenate([b, p])

def get_hex_init(row_counts, r_init):
    """Generate a hexagonal lattice configuration for given row counts."""
    pts = []
    y = r_init
    for ri, cnt in enumerate(row_counts):
        shift = r_init if ri % 2 == 1 else 0.0
        width = (cnt - 1) * 2 * r_init
        x_start = (1.0 - width) / 2.0 + shift
        for k in range(cnt):
            pts.append([x_start + k * 2 * r_init, y])
        y += r_init * np.sqrt(3)
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_t = 0.0
    best_centers = None
    
    # Row distributions that sum to 26
    row_configs = [
        [5, 6, 5, 6, 4],
        [6, 5, 6, 5, 4],
        [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4]
    ]
    
    configs = []
    r_init = 0.05  # Safe initial radius to guarantee feasibility
    
    for rc in row_configs:
        pts = get_hex_init(rc, r_init)
        if len(pts) == N:
            configs.append(pts)
            
    # Add perturbations to escape symmetric minima
    for cfg in configs[:2]:
        for _ in range(4):
            p = cfg + np.random.uniform(-0.012, 0.012, cfg.shape)
            p = np.clip(p, 0.04, 0.96)
            configs.append(p)
            
    # Dense random start
    configs.append(np.random.uniform(0.1, 0.9, size=(N, 2)))
    
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.05, 0.12)]
    
    # Optimize centers for equal radius packing
    for cfg in configs:
        x0 = np.zeros(2 * N + 1)
        x0[0::2] = cfg[:, 0]
        x0[1::2] = cfg[:, 1]
        x0[-1] = r_init
        
        try:
            res = minimize(obj_equal, x0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': cons_equal},
                           options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            
            if res.x[-1] > best_t:
                best_t = res.x[-1]
                best_centers = res.x[:2 * N].reshape(N, 2).copy()
        except Exception:
            continue
            
    # Fallback if optimization yields poor results
    if best_centers is None or best_t < 0.095:
        gs = np.linspace(0.12, 0.88, 5)
        pts = np.array([(x, y) for y in gs for x in gs])
        best_centers = np.vstack([pts, [[0.5, 0.95]]])
        best_t = 0.09
        
    # Phase 2: LP to maximize sum of radii for fixed optimal centers
    diffs = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= d_ij
    for i in range(N):
        for j in range(i + 1, N):
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(N):
        x, y = best_centers[i]
        for lim in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(N)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(lim)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        lp_res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if lp_res.success:
            radii = lp_res.x
            # Tiny shrink to guarantee strict numerical validity against 1e-12 tolerance
            radii *= 0.99999999
            return best_centers, radii, float(np.sum(radii))
    except Exception:
        pass
        
    # Fallback radii
    radii = np.full(N, best_t * 0.999999)
    return best_centers, radii, float(np.sum(radii))
