import numpy as np
from scipy.optimize import minimize, linprog

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)

def objective(vars_flat):
    """Minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def constraints(vars_flat):
    """Compute inequality constraints: returns array where each element must be >= 0."""
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints using squared distances for smooth gradients
    dx = x[IDX_I] - x[IDX_J]
    dy = y[IDX_I] - y[IDX_J]
    ri = r[IDX_I]
    rj = r[IDX_J]
    c = np.concatenate([c, dx**2 + dy**2 - (ri + rj)**2])
    
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    configs = []
    hex_rows = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5]
    ]
    
    np.random.seed(42)
    
    # Generate diverse hexagonal lattice initializations
    for rows in hex_rows:
        pts = []
        y = 0.08
        for ri_idx, cnt in enumerate(rows):
            shift = 0.08 if ri_idx % 2 == 1 else 0.0
            x = 0.08 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 0.16
            y += 0.14
        pts = np.array(pts[:N])
        p_min = pts.min(axis=0)
        p_max = pts.max(axis=0)
        pts = (pts - p_min) / (p_max - p_min + 1e-9) * 0.9 + 0.05
        configs.append(pts)
        
        for _ in range(4):
            p = pts + np.random.uniform(-0.025, 0.025, (N, 2))
            configs.append(np.clip(p, 0.05, 0.95))
            
    # Add grid-based starts
    g = np.linspace(0.1, 0.9, 6)
    grid_pts = np.array([[x, y] for y in g for x in g])
    configs.append(grid_pts[:N])
    configs.append(grid_pts[N-1:2*N])
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    r_init = 0.04
    
    # Stage 1: Multi-start SLSQP optimization
    for cfg in configs:
        x0 = np.zeros(3 * N)
        x0[0::3] = cfg[:, 0]
        x0[1::3] = cfg[:, 1]
        x0[2::3] = r_init
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            
            if res.success or res.fun < -2.4:
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-4:
                    centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    radii = res.x[2::3]
                    s = np.sum(radii)
                    if s > best_sum:
                        best_sum = s
                        best_centers = centers.copy()
                        best_radii = radii.copy()
        except Exception:
            continue

    if best_centers is None:
        best_centers = configs[0]
        best_radii = np.full(N, 0.09)
        best_sum = np.sum(best_radii)

    # Stage 2: Iterative refinement to escape local minima
    for _ in range(8):
        perturbed = best_centers + np.random.uniform(-0.005, 0.005, (N, 2))
        perturbed = np.clip(perturbed, 0.01, 0.99)
        x0 = np.zeros(3 * N)
        x0[0::3] = perturbed[:, 0]
        x0[1::3] = perturbed[:, 1]
        x0[2::3] = best_radii
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            if res.success or res.fun < best_sum * -1:
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-4:
                    centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    radii = res.x[2::3]
                    s = np.sum(radii)
                    if s > best_sum:
                        best_sum = s
                        best_centers = centers.copy()
                        best_radii = radii.copy()
        except Exception:
            pass

    # Stage 3: Exact Linear Programming for radii given fixed optimal centers
    num_pairs = len(IDX_I)
    A_ub = np.zeros((num_pairs + 4 * N, N))
    b_ub = np.zeros(num_pairs + 4 * N)
    
    idx = 0
    for i, j in zip(IDX_I, IDX_J):
        d = np.sqrt(np.sum((best_centers[i] - best_centers[j])**2))
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        b_ub[idx] = d
        idx += 1
        
    for i in range(N):
        x, y = best_centers[i]
        lims = [x, 1.0 - x, y, 1.0 - y]
        for lim in lims:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    try:
        lp_res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if lp_res.success:
            # Apply tiny shrink to guarantee strict validity against 1e-12 tolerance
            best_radii = lp_res.x * 0.9999995
            best_sum = np.sum(best_radii)
    except Exception:
        pass
        
    return best_centers, best_radii, float(best_sum)
