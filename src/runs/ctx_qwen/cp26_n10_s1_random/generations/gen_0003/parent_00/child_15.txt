import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """Solves the LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    
    # Maximum radius allowed by boundaries for each circle
    w = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    h = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    upper = np.minimum(w, h)
    
    # Pairwise distances between centers
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # LP Formulation:
    # Objective: minimize -sum(r)  =>  maximize sum(r)
    # Constraints:
    #   0 <= r_i <= upper[i]
    #   r_i + r_j <= dists[i, j]  for all i < j
    c = -np.ones(n)
    bounds = [(0.0, max(0.0, u)) for u in upper]
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        # 'highs' is a modern, highly efficient LP solver
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective_centers(centers_flat):
    """Objective for center optimization: negative sum of optimal radii."""
    centers = centers_flat.reshape(-1, 2)
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    _, s = solve_radii_lp(centers)
    return -s

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    inits = []
    
    # 1. Hexagonal lattice initialization
    r0 = 0.101
    pts = []
    y = r0
    row = 0
    while len(pts) < n:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x + r0 <= 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
        row += 1
    inits.append(np.array(pts[:n]))
    
    # 2. Grid initialization (5x5 + 1)
    g = np.linspace(0.12, 0.88, 5)
    pts2 = [[x, y] for y in g for x in g]
    pts2.append([0.5, 0.5])
    inits.append(np.array(pts2[:n]))
    
    # 3. Perturbed configurations to explore nearby basins
    for _ in range(20):
        base = inits[0] if rng.random() > 0.4 else inits[1]
        pert = base.copy() + rng.uniform(-0.025, 0.025, (n, 2))
        inits.append(np.clip(pert, 0.02, 0.98))
        
    best_sum = -np.inf
    best_centers = None
    
    # Optimize centers using Powell's method
    # Powell is robust for non-smooth objectives and moderate dimensions
    for x0 in inits:
        try:
            res = minimize(objective_centers, x0.flatten(), method='Powell',
                           options={'maxiter': 1000, 'ftol': 1e-10})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(n, 2)
        except Exception:
            continue
            
    if best_centers is None:
        best_centers = inits[0]
        
    # Final LP solve to get exact optimal radii for the best centers found
    radii, final_sum = solve_radii_lp(best_centers)
    
    # Apply a minimal safety scale to guarantee strict validity against float tolerances
    scale = 0.99999
    radii *= scale
    final_sum *= scale
    
    return best_centers, radii, float(final_sum)
