import numpy as np
from scipy.optimize import minimize, linprog

def get_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c = np.ones(n) * -1.0
    
    # Boundary constraints: r_i <= dist to nearest wall
    min_walls = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                           np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(0.0, m)) for m in min_walls]
    
    # Pairwise constraints: r_i + r_j <= ||c_i - c_j||
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 0.01), 0.26

def fps_init(n, seed):
    """Farthest Point Sampling initialization."""
    rng = np.random.default_rng(seed)
    centers = [rng.uniform(0.1, 0.9, 2)]
    for _ in range(n - 1):
        candidates = rng.uniform(0, 1, (1500, 2))
        dists_to_centers = np.linalg.norm(candidates[:, np.newaxis, :] - np.array(centers)[np.newaxis, :, :], axis=2).min(axis=1)
        dists_to_boundaries = np.minimum(np.minimum(candidates, 1.0 - candidates), axis=1)
        all_dists = np.minimum(dists_to_centers, dists_to_boundaries)
        best_idx = np.argmax(all_dists)
        centers.append(candidates[best_idx])
    return np.array(centers)

def hill_climb(centers, n, seed, steps=600):
    """Adaptive hill-climbing on centers using LP radii evaluation."""
    rng = np.random.default_rng(seed)
    best_centers = centers.copy()
    radii, best_sum = get_lp_radii(best_centers)
    step = 0.025
    
    for _ in range(steps):
        pert = rng.uniform(-step, step, (n, 2))
        new_centers = np.clip(best_centers + pert, 0.0, 1.0)
        new_radii, new_sum = get_lp_radii(new_centers)
        
        if new_sum > best_sum:
            best_centers = new_centers
            radii = new_radii
            best_sum = new_sum
            step = min(step * 1.003, 0.08)
        else:
            step *= 0.996
            if step < 1e-6:
                step = 1e-6
    return best_centers, radii, best_sum

def objective(vars_array, n):
    """Minimize negative sum of radii."""
    return -np.sum(vars_array[2*n:3*n])

def constraints(vars_array, n, triu_idx):
    """Inequality constraints >= 0 for valid packing."""
    x = vars_array[:n]
    y = vars_array[n:2*n]
    r = vars_array[2*n:3*n]
    
    b_cons = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    p_cons = dist_sq[triu_idx] - r_sum[triu_idx]**2
    
    return np.concatenate([b_cons, p_cons])

def run_packing() -> tuple:
    n = 26
    triu_idx = np.triu_indices(n, k=1)
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints, 'args': (n, triu_idx)}

            while x + r_h <= 1.0 and len(pts) < n:
                pts.append([x, y])
                x += dx
            y += dy
            row += 1
        configs.append(np.array(pts[:n]))
        
    # Uniform grid + center
    g = np.linspace(0.1, 0.9, 5)
    pts_g = []
    for yi in g:

            pts_g.append([xi, yi])
    pts_g.append([0.5, 0.5])
    configs.append(np.array(pts_g[:n]))
    
    # Random configurations
    np.random.seed(99)
    for _ in range(4):
        configs.append(np.random.uniform(0.1, 0.9, (n, 2)))
        
    # Optimization pipeline
    for i, cfg in enumerate(configs):
        # Stage 1: Hill climb centers
        hc_c, hc_r, hc_sum = hill_climb(cfg, n, seed=100 + i)
        
        # Stage 2: SLSQP joint refinement
        x0 = np.concatenate([hc_c.flatten(), hc_r])

                if s > best_sum:

                s2 = -res2.fun

                    break

    return best_centers, best_radii, float(best_sum)
