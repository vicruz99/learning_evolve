import numpy as np
from scipy.optimize import minimize, linprog

def compute_objective(vars_array, n):
    """Objective: minimize negative sum of radii"""
    return -np.sum(vars_array[:n])

def compute_constraints(vars_array, n, triu_idx):
    """Constraints: pairwise non-overlap dist_sq >= (r_i + r_j)^2"""
    r = vars_array[:n]
    u = vars_array[n:2*n]
    v = vars_array[2*n:3*n]
    
    # Parameterization guarantees boundary satisfaction: x = r + (1-2r)u, u in [0,1] => x in [r, 1-r]
    denom = np.maximum(1.0 - 2.0 * r, 1e-9)
    x = r + denom * u
    y = r + denom * v
    
    diff_x = x[:, np.newaxis] - x[np.newaxis, :]
    diff_y = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = diff_x**2 + diff_y**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    return dist_sq[triu_idx] - r_sum_sq[triu_idx]

def solve_radii_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Max distance to any boundary
    limits = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1 - centers[:, 1]))
    limits = np.maximum(limits, 0.0)
    
    c = -np.ones(n)  # Maximize sum(r) -> minimize -sum(r)
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x
    except Exception:
        pass
    return None

def generate_hex_configs(n, r_init):
    """Generates diverse hexagonal lattice initial configurations."""
    configs = []
    # Row distributions that sum to 26, mimicking known dense packings
    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 4, 6, 4], [5, 7, 4, 6, 4],
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5]
    ]
    
    for i, pattern in enumerate(row_patterns):
        if sum(pattern) != n: 
            continue
            
        dy = r_init * np.sqrt(3)
        dx = 2.0 * r_init
        pts = []
        y = r_init
        row = 0
        for count in pattern:
            shift = r_init if row % 2 == 1 else 0.0
            x = r_init + shift
            for _ in range(count):
                pts.append([x, y])
                x += dx
            y += dy
            row += 1
            
        pts = np.array(pts[:n])
        
        # Center the configuration in [0,1]^2
        cx, cy = np.mean(pts, axis=0)
        pts -= np.array([cx - 0.5, cy - 0.5])
        
        # Convert to (r, u, v) parameterization
        denom = 1.0 - 2.0 * r_init
        u = (pts[:, 0] - r_init) / denom
        v = (pts[:, 1] - r_init) / denom
        
        u = np.clip(u, 0.0, 1.0)
        v = np.clip(v, 0.0, 1.0)
        
        cfg = np.concatenate([np.full(n, r_init), u, v])
        configs.append(cfg)
        
        # Add controlled perturbations to break symmetries
        for k in range(4):
            rng_p = np.random.default_rng(1000 * i + k)
            pert = cfg.copy()
            pert[n:] += rng_p.uniform(-0.04, 0.04, 2*n)
            pert[n:] = np.clip(pert[n:], 0.0, 1.0)
            configs.append(pert)
            
    # Add a few fully random starts
    rng_rand = np.random.default_rng(42)
    for _ in range(5):
        r_rand = 0.08
        u_rand = rng_rand.uniform(0.1, 0.9, n)
        v_rand = rng_rand.uniform(0.1, 0.9, n)
        cfg = np.concatenate([np.full(n, r_rand), u_rand, v_rand])
        configs.append(cfg)
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    triu_idx = np.triu_indices(n, k=1)
    bounds = [(1e-5, 0.49)] * n + [(0.0, 1.0)] * n + [(0.0, 1.0)] * n
    cons = {'type': 'ineq', 'fun': compute_constraints, 'args': (n, triu_idx)}
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    configs = generate_hex_configs(n, 0.101)
    
    for x0 in configs:
        # Phase 1: Joint SLSQP optimization
        try:
            res = minimize(
                compute_objective, x0, args=(n,), method='SLSQP', bounds=bounds,
                constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13}
            )
            if not np.isfinite(res.fun): 
                continue
        except Exception:
            continue
            
        r_opt = res.x[:n]
        u_opt = res.x[n:2*n]
        v_opt = res.x[2*n:3*n]
        denom = np.maximum(1.0 - 2.0 * r_opt, 1e-9)
        x_opt = r_opt + denom * u_opt
        y_opt = r_opt + denom * v_opt
        centers = np.column_stack((x_opt, y_opt))
        
        # Phase 2: LP refinement to push radii to theoretical limits for fixed centers
        lp_radii = solve_radii_lp(centers, n)
        if lp_radii is not None:
            s_lp = np.sum(lp_radii)
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = centers.copy()
                best_radii = lp_radii.copy()
                
            # Phase 3: Warm restart SLSQP with LP-optimized radii
            denom_lp = np.maximum(1.0 - 2.0 * lp_radii, 1e-9)
            u_lp = (x_opt - lp_radii) / denom_lp
            v_lp = (y_opt - lp_radii) / denom_lp
            u_lp = np.clip(u_lp, 0.0, 1.0)
            v_lp = np.clip(v_lp, 0.0, 1.0)
            
            x0_restart = np.concatenate([lp_radii, u_lp, v_lp])
            try:
                res2 = minimize(
                    compute_objective, x0_restart, args=(n,), method='SLSQP', bounds=bounds,
                    constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13}
                )
                if np.isfinite(res2.fun):
                    r_final = res2.x[:n]
                    u_final = res2.x[n:2*n]
                    v_final = res2.x[2*n:3*n]
                    denom_f = np.maximum(1.0 - 2.0 * r_final, 1e-9)
                    x_f = r_final + denom_f * u_final
                    y_f = r_final + denom_f * v_final
                    s_final = np.sum(r_final)
                    if s_final > best_sum:
                        best_sum = s_final
                        best_centers = np.column_stack((x_f, y_f))
                        best_radii = r_final.copy()
            except Exception:
                pass
                
    if best_centers is None:
        # Fallback configuration
        best_radii = np.full(n, 0.09)
        best_centers = np.column_stack([np.tile(np.linspace(0.15, 0.85, 5), 6)[:n],
                                        np.repeat(np.linspace(0.15, 0.85, 6), 5)[:n]])
        best_sum = np.sum(best_radii)
        
    # Strict validity safeguard to handle floating-point precision limits
    scale = 1.0
    for _ in range(100):
        valid = True
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if x - r < -1e-11 or x + r > 1 + 1e-11 or y - r < -1e-11 or y + r > 1 + 1e-11:
                valid = False
                break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt((best_centers[i,0]-best_centers[j,0])**2 + (best_centers[i,1]-best_centers[j,1])**2)
                    if d < best_radii[i] + best_radii[j] - 1e-11:
                        valid = False
                        break
                if not valid: 
                    break
        if valid:
            break
        scale *= 0.99995
        best_radii *= scale
        
    best_sum = float(np.sum(best_radii))
    return best_centers, best_radii, best_sum
