from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective_fn(vars_array):

    return -np.sum(vars_array[2 * N_CIRCLES:3 * N_CIRCLES])

def constraint_fn(vars_array):

    x = vars_array[:N_CIRCLES]
    y = vars_array[N_CIRCLES:2 * N_CIRCLES]
    r = vars_array[2 * N_CIRCLES:3 * N_CIRCLES]

    b_cons = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])

    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2

    p_cons = dist_sq[triu_idx] - r_sum_sq[triu_idx]
    
    return np.concatenate([b_cons, p_cons])

def solve_lp_radii(centers):

    n = centers.shape[0]

        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),

    b_ub = np.zeros(m)
    idx = 0

            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0

            idx += 1

    return np.full(n, 0.05)
