from scipy.optimize import minimize, linprog

def compute_pairwise_dists(centers):
    """Computes Euclidean distance matrix between all pairs of centers."""
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    return dists

def solve_radii_lp(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i, j) and 0 <= r_i <= dist_to_wall(i)
    """
    n = centers.shape[0]
    # Distance to nearest boundary for each circle
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 0.0)
    
    # Objective: minimize -sum(r) <=> maximize sum(r)
    c = np.ones(n) * -1.0
    bounds = [(0.0, lim) for lim in limits]
    
    dists = compute_pairwise_dists(centers)
    np.fill_diagonal(dists, np.inf)
    
    # Pairwise constraints: r_i + r_j <= d_ij
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def objective(vars_array, n):

    return -np.sum(vars_array[2*n:3*n])

def constraints(vars_array, n):

    Inequality constraints >= 0 for valid packing.
    Variables layout: [x_0..x_{n-1}, y_0..y_{n-1}, r_0..r_{n-1}]

    y = vars_array[n:2*n]

    idx = np.triu_indices(n, k=1)

    return np.concatenate([b_cons, p_cons])
