import numpy as np
from scipy.optimize import minimize, linprog

def objective(vars_array, n):
    """Objective: minimize negative sum of radii => maximize sum of radii"""
    return -np.sum(vars_array[:n])

def constraints(vars_array, n, triu_idx):
    """Non-overlap constraints: squared distance >= (r_i + r_j)^2"""
    r = vars_array[:n]
    u = vars_array[n:2*n]
    v = vars_array[2*n:3*n]
    
    # Parameterization ensures boundary constraints are automatically satisfied:
    # x = r + (1-2r)*u  => when u in [0,1], x in [r, 1-r]
    denom = 1.0 - 2.0 * r
    x = r + denom * u
    y = r + denom * v
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Return only upper triangle pairwise constraints
    return dist_sq[triu_idx] - r_sum_sq[triu_idx]

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii"""
    n = centers.shape[0]
    b = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    b = np.maximum(b, 0.0)
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    
    # Construct constraint matrix: r_i + r_j <= dist(i, j)
    A_ub = np.zeros((m, n))
    for k in range(m):
        A_ub[k, idx_i[k]] = 1.0
        A_ub[k, idx_j[k]] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    c_obj = -np.ones(n)  # Maximize sum(r) -> minimize -sum(r)
    bounds = [(0.0, blim) for blim in b]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def make_hex_config(n, r_init, rows_counts, seed=None):
    """Generate initial hexagonal lattice configuration"""
    pts = []
    y = r_init
    row = 0
    for cnt in rows_counts:
        shift = r_init if row % 2 == 1 else 0.0
        x = r_init + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r_init
        if len(pts) >= n: break
        y += r_init * np.sqrt(3)
        row += 1
        
    pts = np.array(pts[:n])
    denom = 1.0 - 2.0 * r_init
    u = (pts[:, 0] - r_init) / denom
    v = (pts[:, 1] - r_init) / denom
    
    if seed is not None:
        rng_temp = np.random.default_rng(seed)
        u = np.clip(u + rng_temp.uniform(-0.15, 0.15, n), 0.0, 1.0)
        v = np.clip(v + rng_temp.uniform(-0.15, 0.15, n), 0.0, 1.0)
        
    return np.concatenate([np.full(n, r_init), u, v])

def optimize_step(vars0, n, triu_idx, bounds_opt):
    """Run SLSQP optimization from a given starting point"""
    cons = {'type': 'ineq', 'fun': constraints, 'args': (n, triu_idx)}
    try:
        res = minimize(objective, vars0, args=(n,), method='SLSQP', 
                      bounds=bounds_opt, constraints=cons,
                      options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
        if np.isfinite(res.fun):
            c_vals = constraints(res.x, n, triu_idx)
            # Accept if constraints are satisfied within numerical tolerance
            if np.min(c_vals) >= -1e-5:
                return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

    # Generate diverse initial configurations
    configs = []

        if sum(rd) >= n:
            configs.append(make_hex_config(n, 0.095, rd))
            for s in range(1, 8):
                configs.append(make_hex_config(n, 0.095, rd, seed=s))
                
    # Add randomized valid starts
    for _ in range(10):
        r0 = 0.08 + rng.random(n) * 0.02

    for cfg in configs:
        vars_opt, s = optimize_step(cfg, n, triu_idx, bounds_opt)

            best_sum = s
            best_vars = vars_opt.copy()

    v = best_vars[2*n:3*n]

        radii = r
