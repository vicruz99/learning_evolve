import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    Constraints: r_i + r_j <= dist(i,j), r_i <= dist(i, boundary)
    """
    n = centers.shape[0]
    
    # Distance to nearest boundary for each circle
    walls = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    walls = np.maximum(walls, 1e-9)  # Prevent non-positive bounds
    
    # Pairwise Euclidean distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # Ignore self-distance
    
    # Build LP constraint matrix: r_i + r_j <= dist(i, j)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Bounds: 0 <= r_i <= wall_distance
    bounds = [(0.0, w) for w in walls]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    try:
        res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0
