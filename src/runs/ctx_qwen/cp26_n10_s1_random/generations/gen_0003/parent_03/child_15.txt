import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints:
      0 <= r_i <= distance to boundary
      r_i + r_j <= ||c_i - c_j||
    """
    n = centers.shape[0]
    
    # Maximum radius allowed by boundaries for each circle
    lim = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    lim = np.maximum(lim, 0.0)
    bounds = [(0.0, lim[i]) for i in range(n)]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Pairwise distance constraints: r_i + r_j <= d_ij
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)  # Self-distance irrelevant
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    b_ub = dists[idx_i, idx_j]
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
        # Fallback to simplex if highs fails
    return np.zeros(n), 0.0

def neg_lp_sum(centers_flat):
    """Objective function for center optimization: negative LP sum of radii."""
    centers = np.clip(centers_flat.reshape(-1, 2), 1e-5, 1.0 - 1e-5)
    _, s = solve_lp_radii(centers)
    return -s
