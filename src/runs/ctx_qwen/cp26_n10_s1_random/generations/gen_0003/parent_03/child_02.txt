import numpy as np
from scipy.optimize import minimize, linprog

def compute_obj(vars_array, n, mu):
    """
    Objective function with smooth penalty terms.
    Minimizes negative sum of radii while penalizing overlaps and boundary violations.
    """
    x = vars_array[:n]
    y = vars_array[n:2*n]
    r = vars_array[2*n:3*n]

    # Boundary penalties: ensure r <= x <= 1-r and r <= y <= 1-r
    b_pen = np.sum(np.maximum(0, r - x)**2)
    b_pen += np.sum(np.maximum(0, r + x - 1.0)**2)
    b_pen += np.sum(np.maximum(0, r - y)**2)
    b_pen += np.sum(np.maximum(0, r + y - 1.0)**2)

    # Overlap penalties: ensure dist(i,j) >= r_i + r_j
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(d, np.inf)  # Ignore self-distances

    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    viol = np.maximum(0, r_sum - d)
    o_pen = 0.5 * np.sum(viol**2)  # Divide by 2 to count each pair once

    return -np.sum(r) + mu * (b_pen + o_pen)

def solve_lp_radii(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j) and 0 <= r_i <= dist_to_boundary_i

    n = centers.shape[0]

    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 0.0)

    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in lims]

    # Pairwise distances
    dx = centers[:, np.newaxis, 0] - centers[np.newaxis, :, 0]
    dy = centers[:, np.newaxis, 1] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)

    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0
