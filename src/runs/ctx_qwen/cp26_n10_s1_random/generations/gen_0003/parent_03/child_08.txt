from scipy.optimize import minimize, linprog, differential_evolution

def compute_lp_radii(centers):
    """Solves the LP to find the maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    bounds = [(0, None)] * n
    A = []
    b = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A.extend([row, row, row, row])
        b.extend([centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1]])

    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A.append(row)
            b.append(d)
            
    try:
        res = linprog(c, A_ub=np.array(A), b_ub=np.array(b), bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def obj_de(centers_flat):
    """Objective for DE: negative of the maximum achievable sum of radii."""
    centers = centers_flat.reshape(-1, 2)
    _, s = compute_lp_radii(centers)
    return -s

def obj_joint(vars_array, n):

    return -np.sum(vars_array[2::3])

def con_joint(vars_array, n):
    """Constraints for SLSQP: boundaries and pairwise non-overlap."""
    x = vars_array[0::3]
    y = vars_array[1::3]
    r = vars_array[2::3]
    
    # Boundary constraints
    cons = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise squared distance constraints
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2

    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cons = np.concatenate([cons, d2[mask] - rs[mask]**2])
    return cons

def get_hex_init(n, r_init, rot=0.0):

    pts = []
    y = r_init
    row = 0
    while len(pts) < n + 10:

        x = r_init + shift

            pts.append([x, y])
            x += 2.0 * r_init
        y += np.sqrt(3) * r_init
        row += 1
    pts = np.array(pts[:n])
    
    if rot != 0.0:
        cos_t, sin_t = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[cos_t, -sin_t], [sin_t, cos_t]])
        # Re-center and scale to fit safely within [0,1]^2
        pts -= pts.min(axis=0)
        pts /= pts.max(axis=0)
        pts = pts * 0.8 + 0.1
    return pts
