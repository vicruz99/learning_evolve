

<strategy>
The key insight is to decouple the problem into two steps that can be optimized robustly:
1. **LP for Radii**: For any fixed set of centers, finding the optimal radii that maximize the sum while satisfying non-overlap and boundary constraints is a linear programming problem. This LP is fast, exact, and avoids the non-convex distance-squared constraints that plague direct optimizers.
2. **Center Optimization**: We treat the maximum sum of radii as an objective function of the centers alone. We then use a derivative-free optimizer (Powell's method) to search the space of center configurations. Since the LP guarantees feasibility for any centers, the outer optimizer only needs to respect the simple box constraints $[0,1]^2$.

This approach bypasses the local minima traps and constraint coupling issues of previous attempts. We will initialize with multiple high-quality layouts (hexagonal, grid, perturbed, row-pattern) and run local optimizations. Finally, we apply a strict safety scaling to guarantee validation against the grader's tolerance.
</strategy>

import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints:
      0 <= r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
      r_i + r_j <= ||c_i - c_j||
    """
    n = centers.shape[0]
    
    # Distance to nearest boundary for each circle
    limits = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    limits = np.maximum(limits, 1e-9)
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c = -np.ones(n)
    bounds = [(0.0, lim) for lim in limits]
    
    # Pairwise distances
    dx = centers[:, 0:1] - centers[:, 0:1].T
    dy = centers[:, 1:2] - centers[:, 1:2].T
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    
    # Construct constraint matrix A_ub * r <= b_ub
    # Each row corresponds to r_i + r_j <= dist(i,j)
    i_idx, j_idx = np.triu_indices(n, k=1)
    m = len(i_idx)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), i_idx] = 1.0
    A_ub[np.arange(m), j_idx] = 1.0
    b_ub = dists[i_idx, j_idx]
    
    try:
        # HiGHS solver is fast and robust for this size
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback to zero radii if LP fails
    return np.zeros(n), 0.0

def objective(centers_flat):
    """Objective function for center optimization: negative max sum of radii."""
    centers = centers_flat.reshape(-1, 2)
    # Strictly enforce boundaries to keep centers valid
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    _, s = solve_radii_lp(centers)
    return -s

def generate_initial_configs(n):
    """Generates diverse initial center configurations."""
    configs = []
    rng = np.random.RandomState(42)
    
    # 1. Hexagonal lattice
    r_h = 0.101
    pts = []
    row = 0
    y = r_h
    while y + r_h <= 1.0 and len(pts) < n + 5:
        shift = r_h if row % 2 == 1 else 0.0
        x = r_h + shift
        while x + r_h <= 1.0 and len(pts) < n + 5:
            pts.append([x, y])
            x += 2.0 * r_h
        y += np.sqrt(3) * r_h
        row += 1
    configs.append(np.array(pts[:n]))
    
    # 2. Uniform grid with center fill
    g = np.linspace(0.1, 0.9, 5)
    pts_grid = [[x, y] for y in g for x in g]
    pts_grid.append([0.5, 0.5])
    configs.append(np.array(pts_grid[:n]))
    
    # 3. Known dense row pattern for n=26: 5,6,5,6,4
    rows = [5, 6, 5, 6, 4]
    pts_row = []
    y = 0.101
    for i, cnt in enumerate(rows):
        shift = 0.0505 if i % 2 == 1 else 0.0
        x_start = 0.101 + shift
        if cnt > 1:
            spacing = (1.0 - 2.0 * 0.101 - shift) / (cnt - 1)
        else:
            spacing = 0.0
        for k in range(cnt):
            if len(pts_row) < n:
                pts_row.append([x_start + k * spacing, y])
        y += 0.101 * np.sqrt(3)
    configs.append(np.array(pts_row[:n]))
    
    # 4-12. Perturbed versions of best base configs
    for base in [configs[0], configs[2]]:
        for _ in range(4):
            pert = base + rng.uniform(-0.04, 0.04, base.shape)
            configs.append(np.clip(pert, 0.05, 0.95))
            
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds_c = [(0.0, 1.0)] * (2 * n)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = generate_initial_configs(n)
    
    # Phase 1: Local optimization from multiple starts
    for cfg in configs:
        x0 = cfg.flatten()
        try:
            res = minimize(objective, x0, method='Powell', bounds=bounds_c,
                           options={'maxiter': 3000, 'xtol': 1e-8, 'ftol': 1e-11})
            if np.isfinite(res.fun):
                cur_sum = -res.fun
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    opt_c = res.x.reshape(n, 2)
                    best_radii, _ = solve_radii_lp(opt_c)
                    best_centers = opt_c.copy()
        except Exception:
            pass
            
    # Phase 2: Iterative refinement (perturb & optimize)
    if best_centers is not None:
        rng = np.random.RandomState(123)
        for trial in range(8):
            pert_scale = 0.008 * (1.0 - trial / 8.0)
            x0 = best_centers.flatten() + rng.normal(0, pert_scale, 2 * n)
            x0 = np.clip(x0, 0.0, 1.0)
            
            try:
                res = minimize(objective, x0, method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-9, 'fatol': 1e-11})
                if np.isfinite(res.fun):
                    cur_sum = -res.fun
                    if cur_sum > best_sum:
                        best_sum = cur_sum
                        opt_c = res.x.reshape(n, 2)
                        best_radii, _ = solve_radii_lp(opt_c)
                        best_centers = opt_c.copy()
            except Exception:
                pass

    # Fallback safety
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum = solve_radii_lp(best_centers)
        
    # Final safety scaling to strictly satisfy validator's 1e-12 tolerance
    # LP is exact but floating point accumulation can cause micro-violations
    scale = 0.999999
    best_radii *= scale
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
