from scipy.optimize import minimize, linprog
import math

def compute_pairwise_dists(centers):
    """Compute pairwise distances matrix."""
    n = centers.shape[0]
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    return dists

def solve_radii_lp(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Distance to nearest boundary
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    wall_dists = np.maximum(wall_dists, 1e-9)
    
    c = -np.ones(n)  # Maximize sum(r) -> minimize -sum(r)
    bounds = [(0.0, w) for w in wall_dists]
    
    dists = compute_pairwise_dists(centers)
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):

    except Exception:
        pass
    return None

            if x > 1.0 - r: forces[i, 0] -= repulsion * (x - (1.0 - r))
            if y < r: forces[i, 1] += repulsion * (r - y)

                    forces[i] += f * np.array([dx, dy])
                    forces[j] -= f * np.array([dx, dy])

def obj_uneq(vars_array, n):
    return -np.sum(vars_array[:n])

def con_uneq(vars_array, n, triu_idx):
    r = vars_array[:n]
    u = vars_array[n:2*n]
    v = vars_array[2*n:3*n]
    
    # Parameterized coords guarantee boundary constraints
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    rs2 = rs**2
    
    return d2[triu_idx] - rs2[triu_idx]
