import numpy as np
from scipy.optimize import minimize

def objective(vars):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars, n, triu_i, triu_j):
    """
    Returns array of constraint values >= 0 for valid packing.
    Constraints:
    1. Boundary: x >= r, x <= 1-r, y >= r, y <= 1-r
    2. Non-overlap: (xi-xj)^2 + (yi-yj)^2 >= (ri+rj)^2
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints
    b_cons = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints (vectorized)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    rs2 = rs**2
    
    p_cons = d2[triu_i, triu_j] - rs2[triu_i, triu_j]
    
    return np.concatenate([b_cons, p_cons])

def get_hex_init(n, r_init, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization, optionally rotated and scaled."""
    pts = []
    y = r_init
    row = 0
    while len(pts) < n + 10:  # Generate extra points to fill square after rotation
        shift = r_init if row % 2 == 1 else 0.0
        x = r_init + shift
        while x + r_init <= 1.0:
            pts.append([x, y])
            x += 2.0 * r_init
        y += np.sqrt(3) * r_init
        row += 1
    pts = np.array(pts[:n])
    
    if rot != 0.0:
        cos_t, sin_t = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[cos_t, -sin_t], [sin_t, cos_t]])
        # Re-center and scale to fit within [0, 1] safely
        pts -= pts.min(axis=0)
        pts /= pts.max(axis=0)
        pts = pts * 0.8 * scale + 0.1 * (1 - 0.8 * scale)
        
    return pts
