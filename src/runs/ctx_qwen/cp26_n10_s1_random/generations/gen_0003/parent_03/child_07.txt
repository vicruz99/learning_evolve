import numpy as np
from scipy.optimize import minimize, linprog

def objective(vars_array):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_array[2 * 26:])

def constraints(vars_array):
    """Constraints: boundaries and pairwise non-overlap >= 0."""
    n = 26
    centers = vars_array[:2 * n].reshape(n, 2)
    radii = vars_array[2 * n:]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b_cons = np.concatenate([
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
    ])
    
    # Pairwise non-overlap constraints: ||c_i - c_j||^2 - (r_i + r_j)^2 >= 0
    dx = centers[:, 0:1] - centers[:, 0:1].T
    dy = centers[:, 1:2] - centers[:, 1:2].T
    d2 = dx**2 + dy**2
    np.fill_diagonal(d2, 1.0)  # Self-distance irrelevant
    
    rs = radii[:, np.newaxis] + radii[np.newaxis, :]
    p_cons = d2[np.triu_indices(n, k=1)] - rs[np.triu_indices(n, k=1)]**2
    
    return np.concatenate([b_cons, p_cons])

def solve_radii_lp(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    lims = np.maximum(lims, 1e-9)
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m_pairs = n * (n - 1) // 2
    A_ub = np.zeros((m_pairs + n, n))
    b_ub = np.zeros(m_pairs + n)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        A_ub[m_pairs + i, i] = 1.0
        b_ub[m_pairs + i] = lims[i]
        
    try:
        res = linprog(np.ones(n) * -1, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def get_initial_configs(n):
    """Generate diverse initial configurations."""
    configs = []
    rng = np.random.default_rng(42)
    
    # 1. Hexagonal lattice
    r_h = 0.10
    pts = []
    row = 0
    y = r_h
    while y + r_h <= 1.0 and len(pts) < n + 10:
        shift = r_h if row % 2 == 1 else 0.0
        x = r_h + shift
        while x + r_h <= 1.0:
            pts.append([x, y])
            x += 2.0 * r_h
        y += np.sqrt(3) * r_h
        row += 1
    configs.append(np.array(pts[:n]))
    
    # 2. Rotated hexagonal lattices
    for angle in [0.1, 0.2, 0.35, 0.5, 0.7]:
        cfg = configs[0].copy()
        cfg -= 0.5
        c, s = np.cos(angle), np.sin(angle)
        cfg = cfg @ np.array([[c, -s], [s, c]])
        cfg += 0.5
        cfg = np.clip(cfg, 0.05, 0.95)
        configs.append(cfg)
        
    # 3. Grid layouts
    for scale in [0.9, 1.0, 1.1]:
        g = np.linspace(0.05, 0.95, 5) * scale + 0.5 * (1.0 - scale)
        grid_pts = np.array([[x, y] for y in g for x in g])
        grid_pts = np.vstack([grid_pts, [[0.24, 0.24], [0.76, 0.76]]])
        configs.append(grid_pts[:n])
        
    # 4. Random perturbations
    for _ in range(8):
        base = configs[rng.integers(0, len(configs))]
        pert = base + rng.uniform(-0.04, 0.04, (n, 2))
        configs.append(np.clip(pert, 0.05, 0.95))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = get_initial_configs(n)
    
    for cfg in configs:
        x0 = np.concatenate([cfg.flatten(), np.full(n, 0.09)])
        
        # Try a few perturbations per config to escape local minima
        for trial in range(3):
            x0_trial = x0.copy()
            if trial > 0:
                x0_trial[:2*n] += np.random.uniform(-0.015, 0.015, 2*n)
                x0_trial[:2*n] = np.clip(x0_trial[:2*n], 0.02, 0.98)
                x0_trial[2*n:] *= np.random.uniform(0.95, 1.05, n)
                
            try:
                res = minimize(objective, x0_trial, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12})
                
                if np.isfinite(res.fun):
                    c_opt = res.x[:2*n].reshape(n, 2)
                    
                    # Refine radii using LP for exact maximum given centers
                    r_lp, s_lp = solve_radii_lp(c_opt)
                    if r_lp is not None and s_lp > best_sum:
                        best_sum = s_lp
                        best_centers = c_opt.copy()
                        best_radii = r_lp.copy()
            except Exception:
                continue
                
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum = solve_radii_lp(best_centers)
        if best_radii is None:
            best_radii = np.full(n, 0.09)
            best_sum = np.sum(best_radii)
            
    # Final safety validation & scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= max(scale * 0.999999, 0.0)
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
