import numpy as np
from scipy.optimize import linprog, minimize

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIRS_I)

# Precompute LP constraint matrix structure (constant across calls)
A_LP = np.zeros((M_PAIRS + 4 * N, N))
for k, (i, j) in enumerate(zip(PAIRS_I, PAIRS_J)):
    A_LP[k, i] = 1.0
    A_LP[k, j] = 1.0
for i in range(N):
    A_LP[M_PAIRS + i, i] = 1.0          # r_i <= x_i
    A_LP[M_PAIRS + N + i, i] = 1.0      # r_i <= 1 - x_i
    A_LP[M_PAIRS + 2*N + i, i] = 1.0    # r_i <= y_i
    A_LP[M_PAIRS + 3*N + i, i] = 1.0    # r_i <= 1 - y_i

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, and dual marginals."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    b_ub = np.zeros(M_PAIRS + 4 * N)
    diffs = centers[PAIRS_I] - centers[PAIRS_J]
    b_ub[:M_PAIRS] = np.hypot(diffs[:, 0], diffs[:, 1])
    
    for i in range(N):
        b_ub[M_PAIRS + i] = centers[i, 0]
        b_ub[M_PAIRS + N + i] = 1.0 - centers[i, 0]
        b_ub[M_PAIRS + 2*N + i] = centers[i, 1]
        b_ub[M_PAIRS + 3*N + i] = 1.0 - centers[i, 1]
        
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            marg = None
            if hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
                try:
                    marg = np.asarray(res.marginals.ineqlin)
                except Exception:
                    pass
            return res.x, -res.fun, marg
    except Exception:
        pass
    return None, 0.0, None

def obj_grad(c_flat):
    """Objective and gradient for center optimization via LP duals."""
    c = c_flat.reshape(N, 2)
    r, s, marg = solve_lp(c)
    if r is None:
        return 1e6, np.zeros_like(c_flat)
        
    grad = np.zeros((N, 2))
    if marg is not None:
        # Pairwise constraints contribution
        lams = marg[:M_PAIRS]
        mask = lams > 1e-9
        if np.any(mask):
            idx = np.where(mask)[0]
            vals = lams[idx]
            ii = PAIRS_I[idx]
            jj = PAIRS_J[idx]
            dx = c[ii, 0] - c[jj, 0]
            dy = c[ii, 1] - c[jj, 1]
            d = np.hypot(dx, dy)
            d = np.where(d < 1e-12, 1e-12, d)
            fx = vals * dx / d
            fy = vals * dy / d
            
            np.add.at(grad[:, 0], ii, fx)
            np.add.at(grad[:, 1], ii, fy)
            np.add.at(grad[:, 0], jj, -fx)
            np.add.at(grad[:, 1], jj, -fy)
            
        # Boundary constraints contribution
        for i in range(N):
            grad[i, 0] += marg[M_PAIRS + i] - marg[M_PAIRS + N + i]
            grad[i, 1] += marg[M_PAIRS + 2*N + i] - marg[M_PAIRS + 3*N + i]
            
    return -s, -grad.flatten()

def slsqp_obj(x, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def slsqp_cons(x, n):
    """Inequality constraints >= 0 for SLSQP."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    
    bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    pc = d2[np.triu_indices(n, k=1)] - rs[np.triu_indices(n, k=1)]**2
    
    return np.concatenate([bc, pc])
