N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

def solve_lp_for_centers(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, and solver result."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    A_ub = np.zeros((M_PAIRS + 4 * N, N))
    for k, (i, j) in enumerate(zip(PAIR_I, PAIR_J)):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
    for i in range(N):
        A_ub[M_PAIRS + i, i] = 1.0
        A_ub[M_PAIRS + N + i, i] = 1.0
        A_ub[M_PAIRS + 2 * N + i, i] = 1.0
        A_ub[M_PAIRS + 3 * N + i, i] = 1.0

        b_ub[M_PAIRS + N + i] = 1.0 - x
        b_ub[M_PAIRS + 2 * N + i] = y
        b_ub[M_PAIRS + 3 * N + i] = 1.0 - y
    
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        pass

def get_lp_gradient(centers, res):

    grad = np.zeros((n, 2))
    
    try:
        marg = None

            return grad

        return grad

        d = np.where(d < 1e-12, 1e-12, d)
        fx = lam_vals * dx / d
        fy = lam_vals * dy / d
        np.add.at(grad[:, 0], ii, fx)
        np.add.at(grad[:, 1], ii, fy)
        np.add.at(grad[:, 0], jj, -fx)
        np.add.at(grad[:, 1], jj, -fy)

        grad[i, 1] += marg[M_PAIRS + 2 * N + i] - marg[M_PAIRS + 3 * N + i]
    
    return grad

def lp_gradient_objective(c_flat):

    c = c_flat.reshape(N, 2)
    r, s, res = solve_lp_for_centers(c)
    if r is None:
        return 1e6, np.zeros_like(c_flat)
    grad = get_lp_gradient(c, res)
    return -s, -grad.flatten()

def joint_constraints(v, n):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])

    dy = cy[PAIR_I] - cy[PAIR_J]
    d2 = dx**2 + dy**2
    rs = r[PAIR_I] + r[PAIR_J]
    pc = d2 - rs**2
    
    return np.concatenate([bc, pc])

def joint_objective(v):

                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0

        pts.append([0.5, 0.5])
    return np.array(pts[:N])
