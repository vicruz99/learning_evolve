import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_and_grad(centers, n, pair_i, pair_j):
    """
    Solves LP to maximize sum of radii for fixed centers and computes gradient via duals.
    Returns (sum_radii, gradient_of_negative_sum)
    """
    m_pairs = len(pair_i)
    m_bounds = 4 * n
    m_con = m_pairs + m_bounds
    
    # Precompute distances for pairwise constraints
    dx = centers[pair_i, 0] - centers[pair_j, 0]
    dy = centers[pair_i, 1] - centers[pair_j, 1]
    d = np.hypot(dx, dy)
    
    # Build LP matrices
    A_ub = np.zeros((m_con, n))
    b_ub = np.zeros(m_con)
    
    # Pairwise: r_i + r_j <= d_ij
    A_ub[:m_pairs, pair_i] = 1.0
    A_ub[:m_pairs, pair_j] = 1.0
    b_ub[:m_pairs] = d
    
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        A_ub[m_pairs + i, i] = 1.0
        b_ub[m_pairs + i] = centers[i, 0]
        
        A_ub[m_pairs + n + i, i] = 1.0
        b_ub[m_pairs + n + i] = 1.0 - centers[i, 0]
        
        A_ub[m_pairs + 2*n + i, i] = 1.0
        b_ub[m_pairs + 2*n + i] = centers[i, 1]
        
        A_ub[m_pairs + 3*n + i, i] = 1.0
        b_ub[m_pairs + 3*n + i] = 1.0 - centers[i, 1]
        
    # Maximize sum(r) => Minimize -sum(r)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return 1e6, np.zeros(2 * n)
            
        s = -res.fun
        lams = np.zeros(m_con)
        
        # Extract dual variables (marginals) safely
        try:
            if hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
                lams = np.asarray(res.marginals.ineqlin)
            elif hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
                lams = np.asarray(res.ineqlin.marginals)
        except Exception:
            pass
            
        # Compute gradient of -sum(r) w.r.t centers
        grad = np.zeros((n, 2))
        
        # Pairwise contribution: grad_i -= sum_j lam_ij * (C_i - C_j) / d_ij
        lam_vals = lams[:m_pairs]
        d_safe = np.where(d < 1e-12, 1e-12, d)
        fx = lam_vals * dx / d_safe
        fy = lam_vals * dy / d_safe
        
        np.add.at(grad[:, 0], pair_i, -fx)
        np.add.at(grad[:, 1], pair_i, -fy)
        np.add.at(grad[:, 0], pair_j, fx)
        np.add.at(grad[:, 1], pair_j, fy)
        
        # Boundary contribution
        # Left: r <= x, Right: r <= 1-x, Bottom: r <= y, Top: r <= 1-y
        # grad_x -= (lam_R - lam_L), grad_y -= (lam_T - lam_B)
        for i in range(n):
            lam_L = lams[m_pairs + i]
            lam_R = lams[m_pairs + n + i]
            lam_B = lams[m_pairs + 2*n + i]
            lam_T = lams[m_pairs + 3*n + i]
            grad[i, 0] -= (lam_R - lam_L)
            grad[i, 1] -= (lam_T - lam_B)
            
        return s, grad.flatten()
        
    except Exception:
        return 1e6, np.zeros(2 * n)

def objective_function(x, n, pair_i, pair_j):
    """Objective and gradient for scipy minimize."""
    centers = x.reshape(n, 2)
    return solve_lp_and_grad(centers, n, pair_i, pair_j)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

            p = norm_base + rng.uniform(-0.02, 0.02, norm_base.shape)

    if best_c is None:

    try:
        lims = np.minimum(np.minimum(best_c[:, 0], 1.0 - best_c[:, 0]),

        A_r[np.arange(len(pair_i)), pair_j] = 1.0
        b_r = dists[pair_i, pair_j]
        
        res_final = linprog(-np.ones(n), A_ub=A_r, b_ub=b_r, bounds=bounds_r, method='highs')
