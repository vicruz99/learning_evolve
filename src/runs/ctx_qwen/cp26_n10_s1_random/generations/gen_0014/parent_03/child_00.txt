import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Max radius limited by boundaries
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    # Pairwise distances
    dx = centers[PAIR_I, 0] - centers[PAIR_J, 0]
    dy = centers[PAIR_I, 1] - centers[PAIR_J, 1]
    dists = np.hypot(dx, dy)
    
    # LP constraint matrix: r_i + r_j <= dist_ij
    A_ub = np.zeros((M_PAIRS, N))
    A_ub[np.arange(M_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(M_PAIRS), PAIR_J] = 1.0
    
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def make_hex(rows, r0=0.09):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0
    for r_idx, cnt in enumerate(rows):
        shift = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_s = -1.0
    best_c = None
    best_r = None

    # Diverse hexagonal row patterns summing to >= 26
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [4, 6, 6, 6, 4],
        [5, 5, 6, 5, 5], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4],
        [6, 5, 5, 6, 4], [5, 6, 4, 6, 5], [4, 5, 6, 5, 6],
        [6, 4, 5, 6, 5], [5, 6, 5, 5, 5]
    ]

    inits = []
    for pat in patterns:
        if sum(pat) < N:
            continue
        cfg = make_hex(pat, 0.09)
        # Normalize to fit comfortably inside [0,1]^2
        mn, mx = cfg.min(axis=0), cfg.max(axis=0)
        span = mx - mn + 1e-9
        norm_cfg = (cfg - mn) / span * 0.85 + 0.075
        inits.append(norm_cfg)
        
        # Add controlled perturbations to break symmetry
        for _ in range(3):
            p = norm_cfg + rng.uniform(-0.02, 0.02, norm_cfg.shape)
            inits.append(np.clip(p, 0.02, 0.98))
            
    # Add dense random starts
    for _ in range(5):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))

    bounds_c = [(1e-4, 1.0 - 1e-4)] * (2 * N)

    # Phase 1: L-BFGS-B optimization from diverse starts
    for cfg in inits:
        c0 = np.clip(cfg, 1e-4, 1.0 - 1e-4)
        try:
            res = minimize(lambda c: -solve_lp(c.reshape(N, 2))[1], c0.flatten(),
                           method='L-BFGS-B', bounds=bounds_c,
                           options={'maxiter': 800, 'ftol': 1e-12, 'gtol': 1e-9})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(N, 2)
                r_opt, s_opt = solve_lp(c_opt)
                if r_opt is not None and s_opt > best_s:
                    best_s = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            pass

    # Phase 2: Adaptive Jiggle Search to escape local minima & polish
    if best_c is not None:
        curr_c = best_c.copy()
        step = 0.015
        
        # Coarse jiggle
        for _ in range(2500):
            i = rng.integers(N)
            old = curr_c[i].copy()
            curr_c[i] += rng.uniform(-step, step, 2)
            curr_c[i] = np.clip(curr_c[i], 1e-4, 1.0 - 1e-4)
            
            r_try, s_try = solve_lp(curr_c)
            if r_try is not None and s_try > best_s + 1e-9:
                best_s = s_try
                best_c = curr_c.copy()
                best_r = r_try.copy()
                step *= 0.9995
            else:
                curr_c[i] = old
                if rng.random() < 0.02:
                    step *= 0.95
                    
        # Fine jiggle
        curr_c = best_c.copy()
        step = 0.002
        for _ in range(2000):
            i = rng.integers(N)
            old = curr_c[i].copy()
            curr_c[i] += rng.uniform(-step, step, 2)
            curr_c[i] = np.clip(curr_c[i], 1e-4, 1.0 - 1e-4)
            
            r_try, s_try = solve_lp(curr_c)
            if r_try is not None and s_try > best_s + 1e-9:
                best_s = s_try
                best_c = curr_c.copy()
                best_r = r_try.copy()
            else:
                curr_c[i] = old

    # Fallback if optimization unexpectedly fails
    if best_c is None:
        best_c = make_hex([5, 6, 5, 6, 4], 0.09)
        best_r, best_s = solve_lp(best_c)

    # Phase 3: Strict safety scaling to guarantee validator tolerance
    scale = 1.0
    for i in range(N):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_r *= scale * 0.9999995
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
