import numpy as np
from scipy.optimize import minimize, linprog


def compute_constraints_param(vars_arr, n, triu_i, triu_j):
    """
    Pairwise non-overlap constraints with automatic boundary handling.
    Uses parameterization x = r + (1-2r)*u, y = r + (1-2r)*v which
    ensures r <= x <= 1-r and r <= y <= 1-r when 0 <= u,v <= 1.
    """
    r = vars_arr[:n]
    u = vars_arr[n:2*n]
    v = vars_arr[2*n:3*n]

    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v

    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2

    rs = r[:, None] + r[None, :]
    return d2[triu_i, triu_j] - rs[triu_i, triu_j]**2


def objective_param(vars_arr, n):
    """Objective: minimize negative sum of radii => maximize sum of radii."""
    return -np.sum(vars_arr[:n])


def solve_lp_radii(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    bounds_lp = [(0.0, l) for l in limits]

    m = n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1

    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds_lp, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0


def centers_to_params(centers):
    """Converts centers to parameterization variables."""
    n = centers.shape[0]
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    r = np.maximum(r * 0.88, 1e-5)
    denom = 1.0 - 2.0 * r
    denom = np.where(denom < 1e-10, 1e-10, denom)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])


def params_to_centers(params):
    """Converts parameterization variables back to centers."""
    n = len(params) // 3
    r = params[:n]
    u = params[n:2*n]
    v = params[2*n:3*n]
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    return np.column_stack((x, y))


def generate_hex_pattern(row_counts, r0):
    """Generates hexagonal lattice configuration."""
    pts = []
    y = r0
    for ri, cnt in enumerate(row_counts):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= 26:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    return np.array(pts[:26])


def normalize_config(cfg, rng):
    """Normalize configuration to fit comfortably in unit square."""
    mn = cfg.min(axis=0)
    mx = cfg.max(axis=0)
    span = mx - mn + 1e-9
    cfg_norm = (cfg - mn) / span * 0.84 + 0.08
    return cfg_norm


def run_packing():
    n = 26
    triu_i, triu_j = np.triu_indices(n, k=1)

    bounds_v = [(1e-6, 0.5)] * n + [(0.0, 1.0)] * (2 * n)
    cons = {'type': 'ineq', 'fun': compute_constraints_param, 'args': (n, triu_i, triu_j)}
    obj_fun = objective_param

    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Comprehensive set of row patterns for hexagonal packing
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 6, 4, 5],
        [5, 7, 5, 5, 4], [7, 5, 6, 5, 3], [5, 5, 5, 6, 5],
        [4, 5, 6, 6, 5], [5, 6, 5, 5, 5], [6, 5, 5, 6, 4],
        [5, 4, 6, 6, 5], [6, 4, 6, 5, 5], [5, 5, 6, 6, 4],
        [4, 6, 5, 6, 5], [6, 5, 4, 6, 5], [5, 6, 4, 5, 6],
        [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], [6, 4, 5, 5, 6],
        [5, 5, 4, 6, 6], [6, 6, 4, 5, 5], [4, 5, 5, 6, 6]
    ]

    configs = []
    for pat in patterns:
        if sum(pat) < 26:
            continue
        for r0 in [0.078, 0.08, 0.083, 0.085, 0.087, 0.089, 0.091, 0.093, 0.095, 0.097, 0.099, 0.101, 0.103, 0.105]:
            cfg = generate_hex_pattern(pat, r0)
            cfg_norm = normalize_config(cfg, rng)
            configs.append(cfg_norm)

            # Perturbations to break symmetry
            for _ in range(6):
                p = cfg_norm + rng.uniform(-0.03, 0.03, cfg_norm.shape)
                configs.append(np.clip(p, 0.03, 0.97))

    # Random starts
    for _ in range(25):
        configs.append(rng.uniform(0.05, 0.95, (n, 2)))

    # Phase 1: SLSQP with parameterization from diverse starts
    for cfg in configs:
        x0 = centers_to_params(cfg)
        try:
            res = minimize(obj_fun, x0, args=(n,), method='SLSQP', bounds=bounds_v,
                          constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if np.isfinite(res.fun):
                cvals = compute_constraints_param(res.x, n, triu_i, triu_j)
                if np.min(cvals) > -1e-4:
                    centers_opt = params_to_centers(res.x)
                    r_lp, s_lp = solve_lp_radii(centers_opt, n)
                    if r_lp is not None and s_lp > best_sum:
                        best_sum = s_lp
                        best_centers = centers_opt.copy()
                        best_radii = r_lp.copy()
        except Exception:
            continue

    # Phase 2: Simulated annealing on centers with LP evaluation
    if best_centers is not None:
        cur_c = best_centers.copy()
        cur_s = best_sum
        temp = 0.025

        for it in range(10000):
            i = rng.integers(n)
            old = cur_c[i].copy()
            cur_c[i] += rng.uniform(-temp, temp, 2)
            cur_c[i] = np.clip(cur_c[i], 0.001, 0.999)

            r_new, s_new = solve_lp_radii(cur_c, n)
            if r_new is not None:
                if s_new > cur_s or rng.random() < np.exp((s_new - cur_s) / max(temp, 1e-6)):
                    # Accept
                    cur_s = s_new
                    if s_new > best_sum:
                        best_sum = s_new
                        best_centers = cur_c.copy()
                        best_radii = r_new.copy()
                    temp = min(temp * 1.002, 0.06)
                else:
                    # Reject
                    cur_c[i] = old
                    temp *= 0.9993
            else:
                cur_c[i] = old
                temp *= 0.9993
            temp = max(temp, 0.0005)

    # Phase 3: Iterative SLSQP polish with small perturbations
    if best_centers is not None:
        for round_ in range(20):
            pert = best_centers + rng.uniform(-0.004, 0.004, best_centers.shape)
            pert = np.clip(pert, 0.001, 0.999)
            x0 = centers_to_params(pert)
            try:
                res = minimize(obj_fun, x0, args=(n,), method='SLSQP', bounds=bounds_v,
                              constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                if np.isfinite(res.fun):
                    if np.min(compute_constraints_param(res.x, n, triu_i, triu_j)) > -1e-4:
                        c_opt = params_to_centers(res.x)
                        r_lp, s_lp = solve_lp_radii(c_opt, n)
                        if r_lp is not None and s_lp > best_sum:
                            best_sum = s_lp
                            best_centers = c_opt.copy()
                            best_radii = r_lp.copy()
            except Exception:
                continue

    # Phase 4: Final local coordinate search with decreasing step
    if best_centers is not None:
        cur_c = best_centers.copy()
        step = 0.01
        for it in range(3000):
            i = rng.integers(n)
            old = cur_c[i].copy()
            cur_c[i] += rng.uniform(-step, step, 2)
            cur_c[i] = np.clip(cur_c[i], 0.001, 0.999)

            r_new, s_new = solve_lp_radii(cur_c, n)
            if r_new is not None and s_new > best_sum + 1e-9:
                best_sum = s_new
                best_centers = cur_c.copy()
                best_radii = r_new.copy()
                step = min(step * 1.002, 0.02)
            else:
                cur_c[i] = old
                step *= 0.9995
            step = max(step, 1e-5)

    # Phase 5: Additional multi-start SLSQP from best solution
    if best_centers is not None:
        for _ in range(10):
            pert = best_centers + rng.uniform(-0.002, 0.002, best_centers.shape)
            pert = np.clip(pert, 0.001, 0.999)
            r_init, _ = solve_lp_radii(pert, n)
            if r_init is None:
                r_init = best_radii * 0.98

            # Direct SLSQP with x, y, r variables
            bounds_opt = [(0.001, 0.999)] * (2*n) + [(1e-6, 0.5)] * n

            def obj_direct(x, n_circles):
                return -np.sum(x[2*n_circles:])

            def cons_direct(x, n_circles):
                cx = x[:n_circles]
                cy = x[n_circles:2*n_circles]
                r = x[2*n_circles:]
                c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
                dx = cx[:, None] - cx[None, :]
                dy = cy[:, None] - cy[None, :]
                d2 = dx**2 + dy**2
                rs = r[:, None] + r[None, :]
                mask = np.triu(np.ones((n_circles, n_circles), dtype=bool), k=1)
                c = np.concatenate([c, d2[mask] - rs[mask]**2])
                return c

            x0_direct = np.concatenate([pert.flatten(), r_init])
            try:
                res_direct = minimize(obj_direct, x0_direct, args=(n,), method='SLSQP',
                                     bounds=bounds_opt,
                                     constraints={'type': 'ineq', 'fun': cons_direct, 'args': (n,)},
                                     options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                if np.isfinite(res_direct.fun):
                    c_opt_direct = res_direct.x[:2*n].reshape(n, 2)
                    r_lp, s_lp = solve_lp_radii(c_opt_direct, n)
                    if r_lp is not None and s_lp > best_sum:
                        best_sum = s_lp
                        best_centers = c_opt_direct.copy()
                        best_radii = r_lp.copy()
            except Exception:
                continue

    # Safety scaling to guarantee strict numerical validity against 1e-12 tolerance
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-12:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)

        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0],
                             best_centers[i, 1] - best_centers[j, 1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)

        best_radii *= scale * 0.999999
        best_sum = float(np.sum(best_radii))

    # Fallback if optimization fails
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.full(n, 0.08)
        k = 0
        for i in range(6):
            for j in range(5):
                if k >= n:
                    break
                best_centers[k] = [0.1 + j * 0.16, 0.1 + i * 0.16]
                k += 1
        best_sum = float(np.sum(best_radii))

    return best_centers, best_radii, best_sum
