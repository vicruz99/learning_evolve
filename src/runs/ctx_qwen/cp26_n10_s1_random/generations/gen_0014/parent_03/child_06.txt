1. **LP-Radii Solver**: For any fixed set of centers, the optimal radii that maximize the sum while satisfying non-overlap and boundary constraints form a standard LP problem. We solve this efficiently using `scipy.optimize.linprog` with the HiGHS solver.
2. **Dual Gradient Ascent**: The sensitivity of the LP objective to center positions is given by the dual variables (shadow prices/marginals). We compute $\nabla_c \sum r_i$ directly from these marginals. This gives an exact gradient direction to move centers apart or towards free space to increase total radius sum.
3. **L-BFGS-B Optimization**: We use the computed gradients to drive a derivative-based optimizer (L-BFGS-B) over many diverse initial configurations.

5. **Local Perturbation Search**: After gradient ascent converges, we perform a stochastic local search (jiggle + LP evaluation) to escape local minima, followed by a final gradient polish.
6. **Strict Validation & Scaling**: A final safety pass scales radii down microscopically to guarantee strict compliance with the $10^{-12}$ tolerance in the validator, while preserving near-optimality.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def get_triu_indices(n):
    """Returns indices for the upper triangle of an n x n matrix."""
    return np.triu_indices(n, k=1)

def solve_lp(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns radii, sum_radii, and solver result object.
    """
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-10)
    
    pairs_i, pairs_j = get_triu_indices(n)
    m_pairs = len(pairs_i)
    
    diffs = centers[pairs_i] - centers[pairs_j]
    dists = np.hypot(diffs[:, 0], diffs[:, 1])
    
    m_total = m_pairs + 4 * n
    A_ub = np.zeros((m_total, n))
    A_ub[np.arange(m_pairs), pairs_i] = 1.0
    A_ub[np.arange(m_pairs), pairs_j] = 1.0
    b_ub = np.zeros(m_total)
    b_ub[:m_pairs] = dists
    
    for i in range(n):
        b_ub[m_pairs + i] = centers[i, 0]
        b_ub[m_pairs + n + i] = 1.0 - centers[i, 0]
        b_ub[m_pairs + 2*n + i] = centers[i, 1]
        b_ub[m_pairs + 3*n + i] = 1.0 - centers[i, 1]
        
    bounds = [(0.0, lim) for lim in lims]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = np.maximum(res.x, 0.0)
            return radii, -res.fun, res
    except Exception:
        pass
    return None, 0.0, None

def obj_grad(c_flat):
    """Objective and gradient for center optimization via LP duals."""
    c = c_flat.reshape(N, 2)
    radii, val, res = solve_lp(c)
    if radii is None:
        return 1e6, np.zeros_like(c_flat)
        
    grad = np.zeros((N, 2))
    pairs_i, pairs_j = get_triu_indices(N)
    m_pairs = len(pairs_i)
    m_total = m_pairs + 4 * N
    
    marg = np.zeros(m_total)
    try:
        if hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
            marg = np.asarray(res.marginals.ineqlin)
        elif hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            marg = np.asarray(res.ineqlin.marginals)
    except Exception:
        pass
        
    mask = marg[:m_pairs] > 1e-10
    if np.any(mask):
        idx = np.where(mask)[0]
        lams = marg[:m_pairs][idx]
        ii = pairs_i[idx]
        jj = pairs_j[idx]
        dx = c[ii, 0] - c[jj, 0]
        dy = c[ii, 1] - c[jj, 1]
        d = np.sqrt(dx**2 + dy**2)
        d = np.maximum(d, 1e-12)
        
        fx = lams * dx / d
        fy = lams * dy / d
        
        np.add.at(grad[:, 0], ii, fx)
        np.add.at(grad[:, 1], ii, fy)
        np.add.at(grad[:, 0], jj, -fx)
        np.add.at(grad[:, 1], jj, -fy)
        
    for i in range(N):
        grad[i, 0] += marg[m_pairs + i] - marg[m_pairs + N + i]
        grad[i, 1] += marg[m_pairs + 2*N + i] - marg[m_pairs + 3*N + i]
        
    # We minimize negative sum of radii
    return -val, -grad.flatten()

def generate_configs(rng):
    """Generates diverse high-quality initial configurations."""
    configs = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [5, 7, 5, 5, 4], [5, 6, 6, 5, 4],
        [6, 4, 6, 6, 4], [5, 5, 5, 6, 5], [5, 6, 5, 5, 5],
        [6, 5, 5, 6, 4], [5, 6, 4, 6, 5]
    ]
    
    for pat in patterns:
        if sum(pat) != N: continue
        pts = []
        y = 0.09
        row_idx = 0
        for cnt in pat:
            shift = 0.09 if row_idx % 2 == 1 else 0.0
            x = 0.09 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 0.18
            y += 0.09 * np.sqrt(3)
            row_idx += 1
            
        base = np.array(pts[:N])
        # Normalize to comfortably fit in [0.05, 0.95]
        mn = base.min(axis=0)
        mx = base.max(axis=0)
        span = mx - mn + 1e-9
        norm = (base - mn) / span * 0.85 + 0.075
        configs.append(norm)
        
        # Add perturbed versions to break symmetry
        for _ in range(4):
            pert = norm + rng.uniform(-0.02, 0.02, norm.shape)
            configs.append(np.clip(pert, 0.05, 0.95))
            
    # Random dense starts
    for _ in range(12):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return configs

def run_packing():
    rng = np.random.default_rng(42)
    configs = generate_configs(rng)
    
    best_s = -1.0
    best_c = None
    best_r = None
    
    bounds_c = [(1e-4, 1.0 - 1e-4)] * (2 * N)
    
    # Phase 1: Gradient ascent via L-BFGS-B from diverse starts
    for cfg in configs:
        c0 = np.clip(cfg, 1e-4, 1.0 - 1e-4)
        try:
            res = minimize(obj_grad, c0.flatten(), method='L-BFGS-B', jac=True, 
                           bounds=bounds_c, options={'maxiter': 6000, 'ftol': 1e-15, 'gtol': 1e-12})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(N, 2)
                r_opt, s_opt, _ = solve_lp(c_opt)
                if r_opt is not None and s_opt > best_s:
                    best_s = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            continue
            
    # Phase 2: Stochastic local search to escape local minima
    if best_c is not None:
        cur_c = best_c.copy()
        step = 0.012
        for it in range(4000):
            idx = rng.integers(N)
            old = cur_c[idx].copy()
            
            move = rng.uniform(-step, step, 2)
            cur_c[idx] = np.clip(cur_c[idx] + move, 1e-4, 1.0 - 1e-4)
            
            r_try, s_try, _ = solve_lp(cur_c)
            if r_try is not None and s_try > best_s + 1e-9:
                best_s = s_try
                best_c = cur_c.copy()
                best_r = r_try.copy()
                step *= 1.002  # Slightly increase step on success
            else:
                cur_c[idx] = old
                if rng.random() < 0.04:
                    step *= 0.96  # Decay step occasionally
                    
        # Phase 3: Final L-BFGS-B polish from improved configuration
        c0 = best_c.copy()
        res = minimize(obj_grad, c0.flatten(), method='L-BFGS-B', jac=True,
                       bounds=bounds_c, options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12})
        if np.isfinite(res.fun):
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt, _ = solve_lp(c_opt)
            if r_opt is not None and s_opt > best_s:
                best_s = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
                
    # Fallback safety net
    if best_c is None:
        best_c = configs[0]
        best_r, best_s, _ = solve_lp(best_c)
        
    # Final strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(N):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    pairs_i, pairs_j = get_triu_indices(N)
    dx = best_c[pairs_i, 0] - best_c[pairs_j, 0]
    dy = best_c[pairs_i, 1] - best_c[pairs_j, 1]
    d = np.hypot(dx, dy)
    rs = best_r[pairs_i] + best_r[pairs_j]
    valid_pairs = rs > 1e-12
    if np.any(valid_pairs):
        scale = min(scale, np.min(d[valid_pairs] / rs[valid_pairs]))
        
    best_r *= scale * 0.999999
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
