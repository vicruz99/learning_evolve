import numpy as np
from scipy.optimize import linprog, minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

# Precompute constant LP constraint matrix for pairwise non-overlap
A_LP = np.zeros((M_PAIRS, N))
for k, (i, j) in enumerate(zip(PAIR_I, PAIR_J)):
    A_LP[k, i] = 1.0
    A_LP[k, j] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    dx = centers[PAIR_I, 0] - centers[PAIR_J, 0]
    dy = centers[PAIR_I, 1] - centers[PAIR_J, 1]
    b_ub = np.hypot(dx, dy)
    
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_slqp(v):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[PAIR_I] - cx[PAIR_J]
    dy = cy[PAIR_I] - cy[PAIR_J]
    d2 = dx**2 + dy**2
    rs = r[PAIR_I] + r[PAIR_J]
    pc = d2 - rs**2
    
    return np.concatenate([bc, pc])

def generate_hex_configs(rng):
    """Generates diverse hexagonal lattice initial configurations."""
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], 
        [4,6,6,6,4], [5,5,5,5,6], [6,5,5,6,4], [5,6,6,4,5],
        [7,6,6,7], [8,6,6,6], [5,7,5,5,4], [6,4,5,6,5],
        [5, 6, 5, 5, 5], [6, 5, 5, 5, 5]
    ]
    
    for pat in patterns:
        if sum(pat) < N: 
            continue
        r0 = 0.09
        pts = []
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: 
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        
        base = np.array(pts[:N])
        mn = base.min(axis=0)
        mx = base.max(axis=0)
        span = mx - mn + 1e-9
        norm_base = (base - mn) / span * 0.8 + 0.1
        
        configs.append(norm_base)
        for _ in range(3):
            p = norm_base + rng.uniform(-0.025, 0.025, norm_base.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(5):
        configs.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return configs

def run_packing():

    best_s = -1.0
    best_c = None
    best_r = None
    
    configs = generate_hex_configs(rng)
    
    # Phase 1: Hill climbing on centers with LP evaluation
    for cfg in configs:

        r_curr, s_curr = solve_lp(curr_c)
        if r_curr is None: 
            continue
        
        step = 0.015
        for it in range(1200):
            idx = rng.integers(N)
            old = curr_c[idx].copy()

            curr_c[idx] = np.clip(curr_c[idx], 1e-4, 0.999)
            
            r_try, s_try = solve_lp(curr_c)
            if r_try is not None and s_try > s_curr + 1e-9:
                s_curr = s_try
                r_curr = r_try
                if s_curr > best_s:
                    best_s = s_curr
                    best_c = curr_c.copy()
                    best_r = r_curr.copy()
                step = min(step * 1.02, 0.05)
            else:
                curr_c[idx] = old
                step *= 0.998

    if best_c is not None:

                        best_c = c_p.copy()

            except Exception:
                pass

        best_c = np.tile(np.linspace(0.12, 0.88, 5), 5).reshape(25, 1)

        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    dx = best_c[PAIR_I, 0] - best_c[PAIR_J, 0]
    dy = best_c[PAIR_I, 1] - best_c[PAIR_J, 1]
    d = np.hypot(dx, dy)
    rs = best_r[PAIR_I] + best_r[PAIR_J]
    if np.any(rs > 1e-12):
        scale = min(scale, np.min(d / np.maximum(rs, 1e-12)))
        
    best_r *= scale * 0.999999
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
