The LP gradient works by: for each active pairwise constraint (r_i + r_j = d_ij), the dual variable tells us how much the objective changes per unit change in d_ij. Moving circle i in the direction away from j increases d_ij, so we add the dual-weighted direction.

import numpy as np
from scipy.optimize import minimize, linprog

def compute_slsqp_constraints(vars_arr, n, triu_i, triu_j):
    r = vars_arr[:n]
    u = vars_arr[n:2*n]
    v = vars_arr[2*n:3*n]
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    return d2[triu_i, triu_j] - rs[triu_i, triu_j]**2

def slsqp_objective(vars_arr, n):
    return -np.sum(vars_arr[:n])

def solve_lp_radii(centers, n, triu_i, triu_j):
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds_lp = [(0.0, max(l, 1e-9)) for l in limits]
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    m = len(triu_i)
    A = np.zeros((m, n))
    A[np.arange(m), triu_i] = 1.0
    A[np.arange(m), triu_j] = 1.0
    b = dists[triu_i, triu_j]
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds_lp, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return None, 0.0, None

def get_lp_dual_gradient(centers, n, triu_i, triu_j):
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds_lp = [(0.0, max(l, 1e-9)) for l in limits]
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    m = len(triu_i)
    A = np.zeros((m, n))
    A[np.arange(m), triu_i] = 1.0
    A[np.arange(m), triu_j] = 1.0
    b = dists[triu_i, triu_j]
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds_lp, method='highs')
        if res.success and np.isfinite(res.fun):
            duals = np.asarray(res.ineqlin.marginals)
            grad = np.zeros((n, 2))
            lams = duals[:m]
            mask = lams > 1e-10
            if np.any(mask):
                idx = np.where(mask)[0]
                lam_vals = lams[idx]
                ii = triu_i[idx]
                jj = triu_j[idx]
                dx = centers[ii, 0] - centers[jj, 0]
                dy = centers[ii, 1] - centers[jj, 1]
                d = np.sqrt(dx**2 + dy**2)
                d = np.maximum(d, 1e-12)
                np.add.at(grad[:, 0], ii, lam_vals * dx / d)
                np.add.at(grad[:, 1], ii, lam_vals * dy / d)
                np.add.at(grad[:, 0], jj, -lam_vals * dx / d)
                np.add.at(grad[:, 1], jj, -lam_vals * dy / d)
            for i in range(n):
                r = res.x[i]
                if r > limits[i] - 1e-10:
                    if abs(centers[i, 0] - r) < abs(1.0 - centers[i, 0] - r) + 1e-10:
                        grad[i, 0] += 1.0
                    else:
                        grad[i, 0] -= 1.0
                    if abs(centers[i, 1] - r) < abs(1.0 - centers[i, 1] - r) + 1e-10:
                        grad[i, 1] += 1.0
                    else:
                        grad[i, 1] -= 1.0
            return grad, res.x, -res.fun
    except Exception:
        pass
    return None, None, 0.0

def centers_to_params(centers):
    n = centers.shape[0]
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    r = np.maximum(r * 0.95, 1e-5)
    denom = 1.0 - 2.0 * r
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def params_to_centers(params):
    n = len(params) // 3
    r = params[:n]
    u = params[n:2*n]
    v = params[2*n:3*n]
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    return np.column_stack((x, y))

def run_packing():
