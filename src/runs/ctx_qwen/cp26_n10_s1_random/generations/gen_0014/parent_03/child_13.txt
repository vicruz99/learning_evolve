import numpy as np
from scipy.optimize import linprog

def solve_lp_radii(centers, n, triu_i, triu_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Boundary limits for each circle
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    m = len(triu_i)
    A = np.zeros((m, n))
    A[np.arange(m), triu_i] = 1.0
    A[np.arange(m), triu_j] = 1.0
    b = dists[triu_i, triu_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def jiggle_sa(centers, n, triu_i, triu_j, rng, steps=2000):
    """Simulated annealing coordinate-wise jiggle search to optimize centers."""
    best_c = centers.copy()
    best_r, best_s = solve_lp_radii(best_c, n, triu_i, triu_j)
    if best_r is None:
        return best_c, np.zeros(n), 0.0
        
    cur_c = best_c.copy()
    cur_r, cur_s = best_r.copy(), best_s
    step = 0.02
    T = 0.01
    
    for _ in range(steps):
        i = rng.integers(n)
        old = cur_c[i].copy()
        cur_c[i] += rng.uniform(-step, step, 2)
        cur_c[i] = np.clip(cur_c[i], 1e-4, 1.0 - 1e-4)
        
        r_new, s_new = solve_lp_radii(cur_c, n, triu_i, triu_j)
        if r_new is None:
            cur_c[i] = old
            continue
            
        delta = s_new - cur_s
        accept = False
        if delta > 0:
            accept = True
        else:
            if T > 1e-6 and rng.random() < np.exp(delta / T):
                accept = True
                
        if accept:
            cur_r, cur_s = r_new.copy(), s_new
            if s_new > best_s:
                best_s = s_new
                best_r = r_new.copy()
                best_c = cur_c.copy()
            T *= 0.9995
            step = min(step * 1.002, 0.05)
        else:
            cur_c[i] = old
            step *= 0.9995
            T *= 0.9998
    return best_c, best_r, best_s

def run_packing():
    n = 26
    triu_i, triu_j = np.triu_indices(n, k=1)
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    configs = []
    # Hexagonal patterns with varied row distributions
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [5, 7, 5, 5, 4], [6, 4, 6, 6, 4],
        [5, 6, 4, 6, 5], [4, 5, 6, 5, 6]
    ]
    for pat in patterns:
        if sum(pat) < n: continue
        pts = []
        r0 = 0.1
        y = r0
        for row_idx, cnt in enumerate(pat):
            shift = r0 if row_idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        base = np.array(pts[:n])
        mn = base.min(axis=0)
        mx = base.max(axis=0)
        span = mx - mn + 1e-9
        norm = (base - mn) / span * 0.85 + 0.075
        configs.append(norm)
        for _ in range(2):
            pert = norm + rng.uniform(-0.025, 0.025, norm.shape)
            configs.append(np.clip(pert, 0.05, 0.95))
            
    # Perturbed grid patterns
    for _ in range(5):
        pts = []
        for i in range(6):
            for j in range(5):
                if len(pts) >= n: break
                pts.append([0.08 + i*0.15 + rng.uniform(-0.01, 0.01),

        configs.append(np.array(pts[:n]))
        
    # Random dense starts
    for _ in range(5):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # Multi-start optimization
    for cfg in configs:
        c_opt, r_opt, s_opt = jiggle_sa(cfg, n, triu_i, triu_j, rng, steps=2000)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    # Final strict safety scaling to guarantee validation tolerance
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
                
        dx = best_centers[triu_i, 0] - best_centers[triu_j, 0]
        dy = best_centers[triu_i, 1] - best_centers[triu_j, 1]

        rs = best_radii[triu_i] + best_radii[triu_j]
        if np.any(rs > 1e-12):
            scale = min(scale, np.min(d / np.maximum(rs, 1e-12)))
            
        best_radii *= scale * 0.9999995
        best_sum = float(np.sum(best_radii))
    else:
        # Fallback (should not be reached)
        best_centers = rng.uniform(0.2, 0.8, (n, 2))
        best_radii = np.full(n, 0.08)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
