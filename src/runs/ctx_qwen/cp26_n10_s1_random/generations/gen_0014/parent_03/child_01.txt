import numpy as np
from scipy.optimize import linprog, minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
M_PAIRS = len(IDX_I)

# Precompute constant pair constraint matrix for LP
A_PAIRS = np.zeros((M_PAIRS, N))
A_PAIRS[np.arange(M_PAIRS), IDX_I] = 1.0
A_PAIRS[np.arange(M_PAIRS), IDX_J] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    b = dists[IDX_I, IDX_J]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_PAIRS, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0

def obj_cent(c_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    c = c_flat.reshape(N, 2)
    _, s = solve_lp(c)
    return -s

def joint_obj(v):
    """Objective for joint SLSQP optimization."""
    return -np.sum(v[2 * N :])

def joint_con(v):
    """Inequality constraints >= 0 for joint optimization."""
    cx = v[:N]
    cy = v[N : 2 * N]
    r = v[2 * N :]
    
    bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    pc = d2[IDX_I, IDX_J] - rs[IDX_I, IDX_J]**2
    
    return np.concatenate([bc, pc])

def init_configs():

    configs = []
    rng = np.random.default_rng(42)
    
    # Known effective row distributions for N=26 hex packing
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [5, 7, 5, 5, 4], [6, 6, 5, 5, 4],
        [5, 6, 4, 6, 5], [6, 4, 5, 6, 5]
    ]
    
    for pat in patterns:
        pts = []
        y = 0.09
        for r_idx, cnt in enumerate(pat):
            shift = 0.09 if r_idx % 2 == 1 else 0.0
            x = 0.09 + shift

                if len(pts) >= N:
                    break
                pts.append([x, y])
                x += 0.18
            y += 0.155884 # sqrt(3)*0.09
        base = np.array(pts[:N])
        
        # Normalize to fit comfortably in [0.05, 0.95]
        mn = base.min(axis=0)
        mx = base.max(axis=0)
        span = mx - mn + 1e-9
        norm_base = (base - mn) / span * 0.9 + 0.05
        configs.append(norm_base)
        
        # Add controlled perturbations
        for _ in range(3):
            p = norm_base + rng.uniform(-0.02, 0.02, norm_base.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    # Random dense starts
    for _ in range(10):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds_c = [(1e-4, 1.0 - 1e-4)] * (2 * N)
    best_s = -1.0
    best_c = None
    best_r = None
    rng = np.random.default_rng(1234)
    
    configs = init_configs()
    
    # Phase 1: L-BFGS-B center optimization from diverse starts
    for cfg in configs:
        x0 = cfg.ravel()
        try:
            res = minimize(obj_cent, x0, method='L-BFGS-B', bounds=bounds_c,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-10})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(N, 2)
                r_opt, s_opt = solve_lp(c_opt)
                if s_opt > best_s:
                    best_s = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            continue

    scale = 1.0

        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    dx = best_c[IDX_I, 0] - best_c[IDX_J, 0]
    dy = best_c[IDX_I, 1] - best_c[IDX_J, 1]
    d = np.sqrt(dx**2 + dy**2)
    rs = best_r[IDX_I] + best_r[IDX_J]
    if np.any(rs > 1e-12):
        scale = min(scale, np.min(d / np.maximum(rs, 1e-12)))
        
    best_r *= scale * 0.9999995
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
