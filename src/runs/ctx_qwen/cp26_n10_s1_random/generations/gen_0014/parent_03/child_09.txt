import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n, triu_i, triu_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = len(triu_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), triu_i] = 1.0
    A_ub[np.arange(m), triu_j] = 1.0
    b_ub = dists[triu_i, triu_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_slsqp(v, n):
    """Objective: maximize sum of radii -> minimize negative sum"""
    return -np.sum(v[2*n:])

def cons_slsqp(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap: d^2 - (r_i + r_j)^2 >= 0
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    pc = d2[triu_i, triu_j] - rs[triu_i, triu_j]**2
    return np.concatenate([c, pc])

                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        cfg = np.array(pts[:n])

        cfg = cfg * 0.92 + 0.04
        configs.append(cfg)

            configs.append(np.clip(p, 0.05, 0.95))

        configs.append(rng.uniform(0.1, 0.9, (n, 2)))

    best_s = float(np.sum(best_r))
