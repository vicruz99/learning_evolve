import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[PAIR_I] - centers[PAIR_J]
    dists = np.hypot(diffs[:, 0], diffs[:, 1])
    
    A_ub = np.zeros((M_PAIRS, N))
    A_ub[np.arange(M_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(M_PAIRS), PAIR_J] = 1.0
    
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_slsqp(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_slsqp(v):
    """Inequality constraints >= 0 for SLSQP."""
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    
    bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[PAIR_I] - cx[PAIR_J]
    dy = cy[PAIR_I] - cy[PAIR_J]
    d2 = dx**2 + dy**2
    rs = r[PAIR_I] + r[PAIR_J]
    pc = d2 - rs**2
    
    return np.concatenate([bc, pc])

def generate_hex_config(row_counts, r0, rng):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r0
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x + rng.uniform(-0.005, 0.005), y + rng.uniform(-0.005, 0.005)])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < N:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # 1. Generate diverse initial configurations
    configs = []
    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [5, 6, 6, 5, 4], [6, 4, 5, 6, 5]
    ]
    
    for pat in row_patterns:
        base = generate_hex_config(pat, 0.095, rng)
        # Normalize to fit comfortably inside [0,1]
        mn = base.min(axis=0)
        mx = base.max(axis=0)
        span = mx - mn + 1e-9
        norm_base = (base - mn) / span * 0.92 + 0.04
        configs.append(norm_base)
        
        # Add perturbations
        for _ in range(3):
            p = norm_base + rng.uniform(-0.02, 0.02, norm_base.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    # Random dense starts
    for _ in range(10):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    bounds_slqp = [(1e-4, 1.0 - 1e-4)] * (2 * N) + [(1e-6, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slsqp}
    
    # 2. Optimize each configuration
    for cfg in configs:
        c_cur = cfg.copy()
        r_cur, s_cur = solve_lp(c_cur)
        if r_cur is None:
            continue
            
        # SLSQP Phase 1: Joint polish
        v0 = np.concatenate([c_cur.ravel(), r_cur])
        try:
            res = minimize(obj_slsqp, v0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_dict, options={'maxiter': 2000, 'ftol': 1e-12})
            if np.isfinite(res.fun):
                v = res.x
                c_opt = v[:2*N].reshape(N, 2)
                c_opt = np.clip(c_opt, 1e-4, 1.0 - 1e-4)
                r_opt, s_opt = solve_lp(c_opt)
                if r_opt is not None and s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass
            
        # Jiggle Search Phase 2: Stochastic coordinate descent
        if best_centers is not None:
            c_jig = best_centers.copy()
            r_jig = best_radii.copy()
            s_jig = best_sum
            step = 0.015
            
            for it in range(2500):
                # Occasionally shake all points to escape local minima
                if rng.random() < 0.03:
                    shake = rng.uniform(-step*0.2, step*0.2, (N, 2))
                    c_jig = np.clip(c_jig + shake, 1e-4, 0.999)
                    r_tmp, s_tmp = solve_lp(c_jig)
                    if r_tmp is not None and s_tmp > s_jig:
                        r_jig, s_jig = r_tmp, s_tmp
                    continue
                    
                idx = rng.integers(N)
                old = c_jig[idx].copy()
                move = rng.uniform(-step, step, 2)
                c_jig[idx] = np.clip(c_jig[idx] + move, 1e-4, 0.999)
                
                r_tmp, s_tmp = solve_lp(c_jig)
                if r_tmp is not None and s_tmp > s_jig + 1e-10:
                    r_jig, s_jig = r_tmp, s_tmp
                    step = min(step * 1.015, 0.04)
                else:
                    c_jig[idx] = old
                    if rng.random() < 0.04:
                        step *= 0.94
                        
            if s_jig > best_sum:
                best_sum = s_jig
                best_centers = c_jig.copy()
                best_radii = r_jig.copy()
                
            # SLSQP Phase 3: Final polish after jiggle
            v0 = np.concatenate([best_centers.ravel(), best_radii])
            try:
                res = minimize(obj_slsqp, v0, method='SLSQP', bounds=bounds_slqp,
                               constraints=cons_dict, options={'maxiter': 1500, 'ftol': 1e-13})
                if np.isfinite(res.fun):
                    c_pol = res.x[:2*N].reshape(N, 2)
                    c_pol = np.clip(c_pol, 1e-4, 0.999)
                    r_pol, s_pol = solve_lp(c_pol)
                    if r_pol is not None and s_pol > best_sum:
                        best_sum = s_pol
                        best_centers = c_pol.copy()
                        best_radii = r_pol.copy()
            except Exception:
                pass
                
    # Fallback if optimization fails
    if best_centers is None:
        best_centers = np.tile(np.linspace(0.12, 0.88, 6), 5).reshape(30, 1)
        best_centers = np.hstack([best_centers, np.repeat(np.linspace(0.12, 0.88, 6), 5).reshape(30, 1)][:N])
        best_radii = np.full(N, 0.08)
        best_sum = np.sum(best_radii)
        
    # Final strict safety scaling to guarantee 1e-12 tolerance
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    dx = best_centers[PAIR_I, 0] - best_centers[PAIR_J, 0]
    dy = best_centers[PAIR_I, 1] - best_centers[PAIR_J, 1]
    d = np.hypot(dx, dy)
    rs = best_radii[PAIR_I] + best_radii[PAIR_J]
    valid_mask = rs > 1e-12
    if np.any(valid_mask):
        scale = min(scale, np.min(d[valid_mask] / rs[valid_mask]))
        
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
