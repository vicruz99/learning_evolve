import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRIU_I)

# Precompute LP constraint structure for speed
A_LP_PAIR = np.zeros((M_PAIRS, N))
A_LP_PAIR[np.arange(M_PAIRS), TRIU_I] = 1.0
A_LP_PAIR[np.arange(M_PAIRS), TRIU_J] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    c_obj = -np.ones(N)
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[TRIU_I, TRIU_J]
    
    try:
        res = linprog(c_obj, A_ub=A_LP_PAIR, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0, None

def local_search(centers, rng, max_iter=5000):
    """Iterative coordinate descent with adaptive step size and re-annealing."""
    curr_c = centers.copy()
    curr_r, curr_s, _ = solve_lp(curr_c)
    
    best_c = curr_c.copy()
    best_r = curr_r.copy()
    best_s = curr_s
    
    step = 0.03
    stagnation = 0
    
    for it in range(max_iter):
        idx = rng.integers(N)
        old_pos = curr_c[idx].copy()
        
        # Try multiple random directions for this circle
        candidates = []
        for _ in range(6):
            d = rng.uniform(-step, step, 2)
            new_pos = np.clip(curr_c[idx] + d, 1e-4, 1.0 - 1e-4)
            curr_c[idx] = new_pos
            r_try, s_try, _ = solve_lp(curr_c)
            candidates.append((s_try, new_pos.copy()))
            curr_c[idx] = old_pos
            
        # Select best candidate manually to avoid lambda
        best_cand_idx = 0
        best_cand_val = candidates[0][0]
        for k in range(1, len(candidates)):
            if candidates[k][0] > best_cand_val:
                best_cand_val = candidates[k][0]
                best_cand_idx = k
                
        best_candidate_s, best_candidate_pos = candidates[best_cand_idx]
        
        if best_candidate_s > curr_s + 1e-9:
            curr_c[idx] = best_candidate_pos
            curr_r, curr_s, _ = solve_lp(curr_c)
            step = min(step * 1.02, 0.05)
            stagnation = 0
            
            if curr_s > best_s:
                best_s = curr_s
                best_c = curr_c.copy()
                best_r = curr_r.copy()
        else:
            step *= 0.994
            stagnation += 1
            
        # Re-anneal if stuck for too long
        if stagnation > 400:
            perturb_idx = rng.choice(N, 3, replace=False)
            curr_c[perturb_idx] += rng.uniform(-0.05, 0.05, (3, 2))
            curr_c = np.clip(curr_c, 1e-4, 1.0 - 1e-4)
            curr_r, curr_s, _ = solve_lp(curr_c)
            step = 0.04
            stagnation = 0
            
    return best_c, best_r, best_s

def slsqp_polish(centers, radii, rng):
    """Polish configuration using SLSQP on joint variables."""
    x0 = np.concatenate([centers[:, 0], centers[:, 1], radii * 0.995])
    
    def obj(v):
        return -np.sum(v[2*N:])
        
    def cons(v):
        cx = v[:N]
        cy = v[N:2*N]
        r = v[2*N:]
        c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
        dx = cx[TRIU_I] - cx[TRIU_J]
        dy = cy[TRIU_I] - cy[TRIU_J]
        dr = r[TRIU_I] + r[TRIU_J]
        c = np.concatenate([c, dx**2 + dy**2 - dr**2])
        return c
        
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 5000, 'ftol': 1e-13})
        if np.isfinite(res.fun):
            c_opt = np.column_stack((res.x[:N], res.x[N:2*N]))
            r_opt, s_opt, _ = solve_lp(c_opt)
            return c_opt, r_opt, s_opt
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def generate_inits(rng):

    # Hexagonal lattice patterns
    patterns = [

    ]

                pts.append([x, y])
                x += 2.0 * r0

        mn, mx = pts.min(axis=0), pts.max(axis=0)
