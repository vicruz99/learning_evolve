from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRIU_I)

def solve_lp(centers):
    n = centers.shape[0]
    lims = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    lims = np.minimum(lims, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    A_ub = np.zeros((M_PAIRS, n))
    A_ub[np.arange(M_PAIRS), TRIU_I] = 1.0
    A_ub[np.arange(M_PAIRS), TRIU_J] = 1.0
    b_ub = dists[TRIU_I, TRIU_J]
    
    bounds = [(0.0, l) for l in lims]
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun, res
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0, None

def objective_and_grad(v):

    try:
        marg = res.ineqlin.marginals
    except Exception:

        except Exception:

            j = TRIU_J[idx]

            np.add.at(grad[:, 0], j, -f[:, 0])
            np.add.at(grad[:, 1], j, -f[:, 1])
            
    return -s, -grad.flatten()

    y = r0

            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    return np.array(pts[:N])
