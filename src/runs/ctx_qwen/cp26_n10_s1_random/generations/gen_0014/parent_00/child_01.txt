import numpy as np
from scipy.optimize import linprog, minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

# Precompute constant parts of LP constraint matrix
A_LP = np.zeros((M_PAIRS + 4*N, N))
A_LP[:M_PAIRS, PAIR_I] = 1.0
A_LP[:M_PAIRS, PAIR_J] = 1.0
for i in range(N):
    A_LP[M_PAIRS + i, i] = 1.0
    A_LP[M_PAIRS + N + i, i] = 1.0
    A_LP[M_PAIRS + 2*N + i, i] = 1.0
    A_LP[M_PAIRS + 3*N + i, i] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = np.zeros(M_PAIRS + 4*N)
    b_ub[:M_PAIRS] = dists[PAIR_I, PAIR_J]
    
    for i in range(n):
        x, y = centers[i]
        b_ub[M_PAIRS + i] = x
        b_ub[M_PAIRS + N + i] = 1.0 - x
        b_ub[M_PAIRS + 2*N + i] = y
        b_ub[M_PAIRS + 3*N + i] = 1.0 - y
        
    try:
        res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun, res
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0, None

def obj_grad(c_flat):
    """Objective and gradient for center optimization via LP duals."""
    c = c_flat.reshape(N, 2)
    r, s, res = solve_lp(c)
    if r is None:
        return 1e6, np.zeros_like(c_flat)
        
    grad = np.zeros((N, 2))
    marg = None
    try:
        marg = getattr(getattr(res, 'marginals', None), 'ineqlin', None)
        if marg is None:
            marg = getattr(getattr(res, 'ineqlin', None), 'marginals', None)
    except Exception:
        pass
        
    if marg is not None:
        lams = np.maximum(marg[:M_PAIRS], 0.0)
        mask = lams > 1e-9

            lam_vals = lams[idx]
            ii = PAIR_I[idx]
            jj = PAIR_J[idx]

            dy = c[ii, 1] - c[jj, 1]
            d = np.sqrt(dx**2 + dy**2)
            d = np.where(d < 1e-12, 1e-12, d)
            fx = lam_vals * dx / d
            fy = lam_vals * dy / d
            np.add.at(grad[:, 0], ii, fx)
            np.add.at(grad[:, 1], ii, fy)
            np.add.at(grad[:, 0], jj, -fx)
            np.add.at(grad[:, 1], jj, -fy)
            
        for i in range(N):
            grad[i, 0] += marg[M_PAIRS + i] - marg[M_PAIRS + N + i]
            grad[i, 1] += marg[M_PAIRS + 2*N + i] - marg[M_PAIRS + 3*N + i]
            
    return -s, -grad.flatten()

def joint_obj(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def joint_cons(v):
    """Inequality constraints >= 0 for SLSQP."""
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[PAIR_I] - cx[PAIR_J]
    dy = cy[PAIR_I] - cy[PAIR_J]
    rs = r[PAIR_I] + r[PAIR_J]
    pc = dx**2 + dy**2 - rs**2
    return np.concatenate([bc, pc])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
