from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRIU_I)

# Preallocate LP constraint matrix for pairwise distances: r_i + r_j <= d_ij
A_LP_PAIRS = np.zeros((M_PAIRS, N))
A_LP_PAIRS[np.arange(M_PAIRS), TRIU_I] = 1.0
A_LP_PAIRS[np.arange(M_PAIRS), TRIU_J] = 1.0

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[TRIU_I, TRIU_J]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_LP_PAIRS, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def obj_centers(c_flat):

    c = c_flat.reshape(N, 2)
    _, s = solve_lp_radii(c)
