import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRIU_I)

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    c_obj = -np.ones(n)
    A_ub = np.zeros((M_PAIRS, n))
    A_ub[np.arange(M_PAIRS), TRIU_I] = 1.0
    A_ub[np.arange(M_PAIRS), TRIU_J] = 1.0
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[TRIU_I, TRIU_J]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0, None

def compute_lp_gradient(centers, lp_res):
    """Computes gradient of LP objective w.r.t centers using dual variables."""
    n = centers.shape[0]
    grad = np.zeros((n, 2))
    if lp_res is None:
        return grad
        
    try:
        marginals = None
        if hasattr(lp_res, 'marginals') and hasattr(lp_res.marginals, 'ineqlin'):
            marginals = np.asarray(lp_res.marginals.ineqlin)
        elif hasattr(lp_res, 'ineqlin') and hasattr(lp_res.ineqlin, 'marginals'):
            marginals = np.asarray(lp_res.ineqlin.marginals)
            
        if marginals is not None:
            gains = np.maximum(marginals[:M_PAIRS], 0.0)
            if np.any(gains > 1e-8):
                valid = gains > 1e-8
                ii = TRIU_I[valid]
                jj = TRIU_J[valid]
                g = gains[valid]
                dx = centers[ii, 0] - centers[jj, 0]
                dy = centers[ii, 1] - centers[jj, 1]
                d = np.sqrt(dx**2 + dy**2)
                d_safe = np.where(d > 1e-8, d, 1.0)
                
                fx = g * dx / d_safe
                fy = g * dy / d_safe
                
                np.add.at(grad[:, 0], ii, fx)
                np.add.at(grad[:, 1], ii, fy)
                np.add.at(grad[:, 0], jj, -fx)
                np.add.at(grad[:, 1], jj, -fy)
    except Exception:
        pass
    return grad

def joint_objective(v):
    return -np.sum(v[2*N:])

def joint_constraints(v):
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[TRIU_I] - x[TRIU_J]
    dy = y[TRIU_I] - y[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def generate_initial_configs(rng):

    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [6,6,4,6,4],

    ]

        pts = np.array(pts[:N])

    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
