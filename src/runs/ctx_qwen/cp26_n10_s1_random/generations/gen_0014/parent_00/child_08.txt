import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRIU_I)
M_CON = M_PAIRS + 4 * N

# Precompute constant part of LP constraint matrix
A_LP_PAIRS = np.zeros((M_PAIRS, N))
A_LP_PAIRS[np.arange(M_PAIRS), TRIU_I] = 1.0
A_LP_PAIRS[np.arange(M_PAIRS), TRIU_J] = 1.0

A_LP_BOUND = np.zeros((4 * N, N))
for i in range(N):
    A_LP_BOUND[i, i] = 1.0
    A_LP_BOUND[N + i, i] = 1.0
    A_LP_BOUND[2 * N + i, i] = 1.0
    A_LP_BOUND[3 * N + i, i] = 1.0

A_LP = np.vstack([A_LP_PAIRS, A_LP_BOUND])

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    bounds = [(0.0, None)] * N
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    b_ub = np.zeros(M_CON)
    b_ub[:M_PAIRS] = dists[TRIU_I, TRIU_J]
    for i in range(N):
        b_ub[M_PAIRS + i] = centers[i, 0]
        b_ub[M_PAIRS + N + i] = 1.0 - centers[i, 0]
        b_ub[M_PAIRS + 2 * N + i] = centers[i, 1]
        b_ub[M_PAIRS + 3 * N + i] = 1.0 - centers[i, 1]
        
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0, None

def get_lp_gradient(centers, lp_res):
    """Computes gradient of LP objective w.r.t centers using dual variables."""
    grad = np.zeros((N, 2))
    if lp_res is None:
        return grad
        
    try:
        marg = None
        if hasattr(lp_res, 'marginals') and hasattr(lp_res.marginals, 'ineqlin'):
            marg = np.asarray(lp_res.marginals.ineqlin)
        elif hasattr(lp_res, 'ineqlin') and hasattr(lp_res.ineqlin, 'marginals'):
            marg = np.asarray(lp_res.ineqlin.marginals)

            return grad
            
        # Pairwise constraints contribution
        lams = np.maximum(marg[:M_PAIRS], 0.0)

            mask = lams > 1e-9
            idx = np.where(mask)[0]
            ii = TRIU_I[idx]
            jj = TRIU_J[idx]
            g = lams[idx]
            dx = centers[ii, 0] - centers[jj, 0]
            dy = centers[ii, 1] - centers[jj, 1]
            d = np.sqrt(dx**2 + dy**2)
            d = np.where(d < 1e-9, 1e-9, d)
            fx = g * dx / d
            fy = g * dy / d
            np.add.at(grad[:, 0], ii, fx)
            np.add.at(grad[:, 1], ii, fy)
            np.add.at(grad[:, 0], jj, -fx)
            np.add.at(grad[:, 1], jj, -fy)
            
        # Boundary constraints contribution
        for i in range(N):
            grad[i, 0] += marg[M_PAIRS + i] - marg[M_PAIRS + N + i]
            grad[i, 1] += marg[M_PAIRS + 2 * N + i] - marg[M_PAIRS + 3 * N + i]
    except Exception:
        pass
    return grad

def obj_and_grad(c_flat):
    """Objective and gradient for center optimization via LP duals."""
    c = c_flat.reshape(N, 2)
    r, s, res = solve_lp(c)
    if r is None or not np.isfinite(s):
        return 1e6, np.zeros_like(c_flat)
    grad = get_lp_gradient(c, res)
    return -s, -grad.flatten()

def joint_obj(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def joint_cons(v):
    """Inequality constraints >= 0 for SLSQP."""
    cx = v[:N]
    cy = v[N:2 * N]
    r = v[2 * N:]
    
    bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[TRIU_I] - cx[TRIU_J]
    dy = cy[TRIU_I] - cy[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    pc = dx**2 + dy**2 - dr**2
    
    return np.concatenate([bc, pc])

def generate_hex_config(row_counts, r0):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])
