import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(vars_arr):
    """Objective: minimize negative sum of radii => Maximize sum of radii"""
    return -np.sum(vars_arr[2*N:])

def constraint_func(vars_arr):
    """Computes inequality constraints >= 0 for valid packing"""
    cx, cy, r = vars_arr[:N], vars_arr[N:2*N], vars_arr[2*N:]
    bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    pc = np.empty(N*(N-1)//2)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            pc[idx] = (cx[i] - cx[j])**2 + (cy[i] - cy[j])**2 - (r[i] + r[j])**2
            idx += 1
    return np.concatenate([bc, pc])

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def generate_hex(row_counts, n):
    """Generates a hexagonal lattice configuration for N circles"""
    pts = []
    r0 = 0.095
    dy = np.sqrt(3) * r0
    y = r0
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += dy
    pts = np.array(pts[:n])
    # Center and scale to comfortably fit inside [0.05, 0.95]
    mn = pts.min(axis=0)
    mx = pts.max(axis=0)
    span = mx - mn
    if span[0] > 1e-8: pts[:, 0] = (pts[:, 0] - mn[0]) / span[0] * 0.85 + 0.075
    if span[1] > 1e-8: pts[:, 1] = (pts[:, 1] - mn[1]) / span[1] * 0.85 + 0.075
    return pts

    ]

    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
