import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Upper bounds for each radius based on distance to walls
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i,j)
    diffs = centers[PAIR_I] - centers[PAIR_J]
    dists = np.hypot(diffs[:, 0], diffs[:, 1])
    
    A_ub = np.zeros((M_PAIRS, n))
    A_ub[np.arange(M_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(M_PAIRS), PAIR_J] = 1.0
    b_ub = dists
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def center_penalty(c_flat, radii):
    """Smooth penalty function for optimizing centers given fixed radii."""
    c = c_flat.reshape(N, 2)
    x, y = c[:, 0], c[:, 1]
    r = radii
    pen = 0.0
    
    # Boundary violations
    pen += np.sum(np.maximum(0.0, r - x)**2)
    pen += np.sum(np.maximum(0.0, x + r - 1.0)**2)
    pen += np.sum(np.maximum(0.0, r - y)**2)
    pen += np.sum(np.maximum(0.0, y + r - 1.0)**2)
    
    # Overlap violations
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dists = np.hypot(dx, dy)
    r_sums = r[PAIR_I] + r[PAIR_J]
    pen += np.sum(np.maximum(0.0, r_sums - dists)**2)
    
    return pen

def cons_func(v):
    """Inequality constraints >= 0 for SLSQP."""
    cx, cy, r = v[:N], v[N:2*N], v[2*N:]
    c = np.empty(4*N + M_PAIRS)
    
    # Boundary constraints
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[PAIR_I] - cx[PAIR_J]
    dy = cy[PAIR_I] - cy[PAIR_J]
    c[4*N:] = dx**2 + dy**2 - (r[PAIR_I] + r[PAIR_J])**2
    
    return c

def obj_func(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def generate_hex(row_counts, r0=0.09):
    """Generates a hexagonal lattice configuration for N circles."""
    pts = []
    y = r0
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        
    pts = np.array(pts[:N])
    # Normalize to fit comfortably inside [0.05, 0.95]
    mn = pts.min(axis=0)
    mx = pts.max(axis=0)
    span = mx - mn
    if span[0] < 1e-8 or span[1] < 1e-8: return pts
    pts = (pts - mn) / span * 0.85 + 0.075
    return np.clip(pts, 0.05, 0.95)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_s = -1.0
    best_c = None
    best_r = None
    
    # Diverse row distributions for hexagonal initialization
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], [4,6,6,6,4],
        [5,7,5,5,4], [6,5,5,6,4], [5,6,6,4,5], [7,6,6,7], [8,6,6,6],
        [5,5,5,5,6], [6,5,5,5,5], [5,5,5,6,5], [5,5,6,6,4], [5,6,5,5,5],
        [5,5,6,4,6], [6,4,6,5,5], [4,5,6,6,5], [6,5,4,6,5]
    ]
    
    configs = []
    for pat in patterns:
        if sum(pat) < N: continue
        cfg = generate_hex(pat)
        configs.append(cfg)
        for _ in range(4):
            p = cfg + rng.uniform(-0.025, 0.025, cfg.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    # Add purely random dense starts
    for _ in range(8):
        configs.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    bounds_xy = [(0.0, 1.0)] * (2 * N)
    
    # Phase 1: SLSQP Joint Optimization
    for cfg in configs:
        r0 = np.full(N, 0.095)
        # Quick LP estimate to warm-start radii
        c_est, s_est = solve_lp(cfg)
        if c_est is not None:
            r0 = c_est * 0.95
            
        x0 = np.concatenate([cfg.flatten(), r0])
        bounds_full = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.5)] * N
        
        try:
            res = minimize(obj_func, x0, method='SLSQP', bounds=bounds_full,
                           constraints={'type': 'ineq', 'fun': cons_func},
                           options={'maxiter': 5000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_try = res.x[:2*N].reshape(N, 2)
                c_try = np.clip(c_try, 1e-5, 1.0 - 1e-5)

                if r_try is not None and s_try > best_s:
                    best_s = s_try
                    best_c = c_try.copy()
                    best_r = r_try.copy()
        except Exception:
            continue
            
    if best_c is None:
        best_c = configs[0]
        best_r, best_s = solve_lp(best_c)
        
    # Phase 2: Alternating Refinement (Centers -> LP Radii)
    if best_c is not None:
        curr_c = best_c.copy()
        curr_r = best_r.copy()
        curr_s = best_s
        
        for it in range(20):
            # Optimize centers for current radii using L-BFGS-B
            res_c = minimize(center_penalty, curr_c.flatten(), args=(curr_r,),
                             method='L-BFGS-B', bounds=bounds_xy,
                             options={'maxiter': 2000, 'ftol': 1e-14})
            new_c = res_c.x.reshape(N, 2)
            
            # Project to exact maximal radii
            r_new, s_new = solve_lp(new_c)
            if r_new is not None and s_new > curr_s + 1e-8:
                curr_s = s_new
                curr_c = new_c.copy()
                curr_r = r_new.copy()

            if r_pert is not None and s_pert > curr_s:
                curr_s = s_pert
                curr_r = r_pert.copy()
                
        best_c, best_r, best_s = curr_c, curr_r, curr_s
        
    # Phase 3: Coordinated Jiggle Search
    if best_c is not None:
        curr_c = best_c.copy()
        curr_r = best_r.copy()
        curr_s = best_s
        step = 0.012
        
        for it in range(2500):
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            
            best_move = None
            best_val = curr_s
            
            # Sample multiple directions per iteration
            for _ in range(6):
                move = rng.normal(0, step, 2)
                nc = np.clip(curr_c[idx] + move, 1e-4, 0.9999)
                curr_c[idx] = nc
                rt, st = solve_lp(curr_c)
                if rt is not None and st > best_val:
                    best_val = st
                    best_move = nc.copy()
                curr_c[idx] = old
                
            if best_move is not None:
                curr_c[idx] = best_move
                curr_s = best_val
                curr_r, _ = solve_lp(curr_c)
                if curr_s > best_s:
                    best_s = curr_s
                    best_c = curr_c.copy()
                    best_r = curr_r.copy()
                step *= 0.9985
            else:
                if rng.random() < 0.04:
                    step *= 0.95
                    
    # Final strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    if best_c is not None:
        scale = 1.0
        for i in range(N):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
                
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                rs = best_r[i] + best_r[j]
                if rs > 1e-12:
                    scale = min(scale, d/rs)
                    
        best_r *= scale * 0.9999995
        best_s = float(np.sum(best_r))
        
    return best_c, best_r, best_s
