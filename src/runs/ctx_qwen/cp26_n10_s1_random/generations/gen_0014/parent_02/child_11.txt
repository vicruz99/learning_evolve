import numpy as np
from scipy.optimize import linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)
M_BOUND = 4 * N
M_CON = M_PAIRS + M_BOUND

# Precompute constant constraint matrix structure for LP
A_ub = np.zeros((M_CON, N))
for k, (i, j) in enumerate(zip(PAIR_I, PAIR_J)):
    A_ub[k, i] = 1.0
    A_ub[k, j] = 1.0
for i in range(N):
    A_ub[M_PAIRS + i, i] = 1.0
    A_ub[M_PAIRS + N + i, i] = 1.0
    A_ub[M_PAIRS + 2*N + i, i] = 1.0
    A_ub[M_PAIRS + 3*N + i, i] = 1.0

def solve_lp_fast(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    b_ub = np.empty(M_CON)
    diffs = centers[PAIR_I] - centers[PAIR_J]
    b_ub[:M_PAIRS] = np.hypot(diffs[:, 0], diffs[:, 1])
    
    for i in range(N):
        x, y = centers[i]
        b_ub[M_PAIRS + i] = x
        b_ub[M_PAIRS + N + i] = 1.0 - x
        b_ub[M_PAIRS + 2*N + i] = y
        b_ub[M_PAIRS + 3*N + i] = 1.0 - y
        
    bounds = [(0.0, lim) for lim in lims]
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def hill_climb(centers, rng, max_iters=2500):
    """Coordinate descent on centers with LP radii optimization."""
    c = centers.copy()
    r_best, s_best = solve_lp_fast(c)
    if r_best is None:
        r_best = np.full(N, 1e-5)
        s_best = np.sum(r_best)
        
    step = 0.015
    stagnation = 0
    for _ in range(max_iters):
        idx = rng.integers(N)
        old = c[idx].copy()
        move = rng.uniform(-step, step, 2)
        c[idx] = np.clip(old + move, 1e-4, 1.0 - 1e-4)
        
        r_new, s_new = solve_lp_fast(c)
        if r_new is not None and s_new > s_best + 1e-10:
            r_best = r_new
            s_best = s_new
            step = min(0.035, step * 1.002)
            stagnation = 0
        else:
            c[idx] = old
            step *= 0.9985
            stagnation += 1
            
        # Random jump if stagnant to escape deep local minima
        if stagnation > 600:
            c[idx] = rng.uniform(0.15, 0.85, 2)
            stagnation = 0
            step = 0.015
            
    return c, r_best, s_best

def generate_configs(n, rng):
    """Generates diverse hexagonal and random initial configurations."""
    configs = []
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4],
        [5,6,6,4,5], [6,4,5,6,5], [5,7,5,5,4], [7,6,6,7],
        [8,6,6,6], [9,8,9], [5,5,5,5,6], [6,5,5,5,5]
    ]
    for pat in patterns:
        if sum(pat) < n: continue
        r0 = 0.09
        pts = []
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        base = np.array(pts[:n])
        
        # Normalize to fit comfortably inside [0.08, 0.92]
        mn = base.min(axis=0)
        mx = base.max(axis=0)
        span = mx - mn
        if span[0] > 0: base[:, 0] = (base[:, 0] - mn[0]) / span[0] * 0.84 + 0.08
        if span[1] > 0: base[:, 1] = (base[:, 1] - mn[1]) / span[1] * 0.84 + 0.08
        configs.append(base)
        
        # Add perturbed versions
        for _ in range(3):
            p = base + rng.uniform(-0.03, 0.03, base.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    # Add random dense starts
    for _ in range(8):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_s = -1.0
    best_c = None
    best_r = None

            best_s = s
            best_c = c.copy()
            best_r = r.copy()

            pert = best_c + rng.uniform(-0.005, 0.005, best_c.shape)

                best_s = s
                best_c = c.copy()
                best_r = r.copy()
                
    # Fallback safety net
    if best_c is None:
        best_c = generate_configs(N, rng)[0]
        best_r, best_s = solve_lp_fast(best_c)

    scale = 1.0
    for i in range(N):

            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

        for j in range(i+1, N):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999995
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
