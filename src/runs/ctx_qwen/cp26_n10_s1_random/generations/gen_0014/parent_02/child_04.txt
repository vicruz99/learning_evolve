from scipy.optimize import minimize, linprog


def solve_lp(centers, N):
    """Solve LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    c_obj = -np.ones(N)
    m_pairs = N * (N - 1) // 2
    A_ub = np.zeros((m_pairs, N))
    b_ub = np.zeros(m_pairs)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun, res
    except Exception:
        pass
    return None, 0.0, None

def generate_hex_init(row_counts, N, scale=0.9):
    """Generate hexagonal lattice configuration."""
    r0 = 0.1
    pts = []
    y = r0
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        row_width = (cnt - 1) * 2 * r0
        x_start = (1.0 - row_width) / 2.0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x_start, y])
            x_start += 2 * r0
        y += np.sqrt(3) * r0
    pts = np.array(pts[:N])
    mn = pts.min(axis=0)
    mx = pts.max(axis=0)
    span = mx - mn
    max_span = max(span[0], span[1])
    if max_span > 1e-8:
        pts = (pts - mn) / max_span * scale + (1.0 - scale) / 2.0
    return np.clip(pts, 0.02, 0.98)


def constraint_func(vars_arr, n):
    """Computes inequality constraints >= 0 for valid packing."""
    cx, cy, r = vars_arr[:n], vars_arr[n:2 * n], vars_arr[2 * n:]
    c = np.empty(4 * n + n * (n - 1) // 2)
    c[:n] = cx - r
    c[n:2 * n] = 1.0 - cx - r
    c[2 * n:3 * n] = cy - r
    c[3 * n:4 * n] = 1.0 - cy - r
    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            c[idx] = (cx[i] - cx[j]) ** 2 + (cy[i] - cy[j]) ** 2 - (r[i] + r[j]) ** 2
            idx += 1
    return c


def joint_obj(v, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])
