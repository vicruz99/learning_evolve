1. **LP Radii Refinement**: Given fixed centers, solve an LP to find the exact maximal radii that satisfy non-overlap and boundary constraints. This is fast and guarantees feasibility for the radii.
2. **Center Optimization**: With radii fixed, optimize centers to minimize constraint violations using a smooth penalty function and L-BFGS-B. This pushes circles apart and toward optimal positions.
3. **Jiggle Search**: A local coordinate descent/random perturbation phase is highly effective for escaping local minima in packing problems. We randomly nudge individual circles, accept moves that increase the LP-sum-of-radii, and gradually reduce step size.
4. **Diverse Initializations**: We seed the process with multiple hexagonal lattice patterns (commonly optimal for circle packing), grid layouts, and random starts. Perturbations break symmetry.
5. **Strict Safety Scaling**: Finally, we compute the exact bottleneck scaling factor to guarantee the configuration satisfies the validator's `1e-12` tolerance strictly, then apply a microscopic shrink to be safe.

import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mx, 1e-9)))
        
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-7), 0.0

def center_penalty(centers_flat, radii, n):
    """Smooth penalty function for center optimization given fixed radii."""
    c = centers_flat.reshape(n, 2)
    x, y = c[:, 0], c[:, 1]
    
    # Boundary violations
    pen = np.sum(np.maximum(0.0, radii - x)**2)
    pen += np.sum(np.maximum(0.0, x - (1.0 - radii))**2)
    pen += np.sum(np.maximum(0.0, radii - y)**2)
    pen += np.sum(np.maximum(0.0, y - (1.0 - radii))**2)
    
    # Overlap violations
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, 1e9)
    rs = radii[:, None] + radii[None, :]
    
    # Quadratic penalty + linear term for better gradients near violation boundary
    overlap = np.maximum(0.0, rs - dists)
    pen += np.sum(overlap**2) + 5.0 * np.sum(overlap)
    
    return pen

def generate_hex(row_counts, n):
    """Generates a hexagonal lattice configuration for N circles."""
    r0 = 0.09
    dy = np.sqrt(3.0) * r0
    y = r0
    pts = []
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        row_width = (cnt - 1) * 2 * r0
        x_start = (1.0 - row_width) / 2.0 + shift
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2 * r0
        y += dy
        
    pts = np.array(pts[:n])
    return np.clip(pts, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds_c = [(0.001, 0.999)] * (2 * n)
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    # Diverse hexagonal row distributions
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 6, 4], [4, 6, 6, 5, 5],
        [6, 6, 5, 5, 4], [5, 6, 5, 5, 5], [5, 5, 5, 6, 5], [5, 7, 4, 5, 5],
        [6, 4, 6, 5, 5], [5, 6, 6, 5, 4], [5, 5, 6, 5, 5], [6, 5, 5, 6, 4]
    ]
    
    configs = []
    for pat in patterns:
        if sum(pat) >= n:
            cfg = generate_hex(pat, n)
            configs.append(cfg)
            for _ in range(4):
                p = cfg + rng.uniform(-0.035, 0.035, cfg.shape)
                configs.append(np.clip(p, 0.02, 0.98))
                
    # Grid and random starts
    gx = np.linspace(0.12, 0.88, 5)
    gy = np.linspace(0.12, 0.88, 5)
    grid = np.array([[x, y] for y in gy for x in gx])
    grid = np.vstack([grid, [0.5, 0.5]])
    configs.append(grid)
    
    for _ in range(6):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Phase 1: Alternating Optimization (Centers <-> LP Radii)
    for init_c in configs:
        curr_c = init_c.copy()
        curr_r, curr_s = solve_lp_radii(curr_c, n)
        
        for _ in range(12):
            # Optimize centers given radii
            res_c = minimize(center_penalty, curr_c.flatten(), args=(curr_r, n),
                             method='L-BFGS-B', bounds=bounds_c,
                             options={'maxiter': 2000, 'ftol': 1e-13})
            new_c = np.clip(res_c.x.reshape(n, 2), 0.001, 0.999)
            
            # Optimize radii given centers
            new_r, new_s = solve_lp_radii(new_c, n)
            
            if new_s > curr_s + 1e-7:
                curr_c = new_c
                curr_r = new_r
                curr_s = new_s
                
        if curr_s > best_sum:
            best_sum = curr_s
            best_c = curr_c.copy()
            best_r = curr_r.copy()
            
    # Phase 2: Stochastic Jiggle Search to escape local minima
    if best_c is not None:
        curr_c = best_c.copy()
        curr_r = best_r.copy()
        curr_s = best_sum
        step = 0.025
        
        for _ in range(4000):
            idx = rng.integers(n)
            old_pos = curr_c[idx].copy()
            
            # Random move
            move = rng.uniform(-step, step, 2)
            new_pos = np.clip(curr_c[idx] + move, 0.001, 0.999)
            curr_c[idx] = new_pos
            
            r_try, s_try = solve_lp_radii(curr_c, n)
            if s_try > curr_s + 1e-8:
                curr_s = s_try
                curr_r = r_try.copy()
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_c = curr_c.copy()
                    best_r = curr_r.copy()
                step *= 0.9985
            else:
                curr_c[idx] = old_pos
                if rng.random() < 0.015:
                    step *= 0.95
                    
    # Fallback safety net
    if best_c is None:
        best_c = configs[0]
        best_r, best_sum = solve_lp_radii(best_c, n)
        
    # Phase 3: Strict Safety Scaling to guarantee 1e-12 validator tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_r *= scale * 0.9999994
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
