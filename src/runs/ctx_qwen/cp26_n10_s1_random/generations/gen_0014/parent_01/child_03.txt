import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

def decode(params):
    """Converts reparameterized variables (r, u, v) to centers and radii."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    # Reparameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    return np.column_stack((x, y)), r

def con_func(params):
    """Inequality constraints >= 0 for pairwise non-overlap."""
    c, r = decode(params)
    dx = c[PAIR_I, 0] - c[PAIR_J, 0]
    dy = c[PAIR_I, 1] - c[PAIR_J, 1]
    d2 = dx**2 + dy**2
    rs = r[PAIR_I] + r[PAIR_J]
    return d2 - rs**2

def obj_func(params):
    """Objective: minimize negative sum of radii => maximize sum of radii."""
    return -np.sum(params[:N])

def solve_lp(c):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                      np.minimum(c[:, 1], 1.0 - c[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = c[PAIR_I] - c[PAIR_J]
    dists = np.hypot(diffs[:, 0], diffs[:, 1])
    
    A_ub = np.zeros((M_PAIRS, N))
    A_ub[np.arange(M_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(M_PAIRS), PAIR_J] = 1.0
    b_ub = dists
    
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

    ]

    best_r *= scale * 0.9999995
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
