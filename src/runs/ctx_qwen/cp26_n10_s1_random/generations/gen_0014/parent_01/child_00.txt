import numpy as np
from scipy.optimize import minimize, linprog

N = 26
M_PAIRS = N * (N - 1) // 2
PAIR_I, PAIR_J = np.triu_indices(N, k=1)

# Precompute constant LP constraint matrix for pairwise non-overlap: r_i + r_j <= d_ij
A_LP = np.zeros((M_PAIRS, N))
A_LP[np.arange(M_PAIRS), PAIR_I] = 1.0
A_LP[np.arange(M_PAIRS), PAIR_J] = 1.0

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[PAIR_I] - centers[PAIR_J]
    dists = np.hypot(diffs[:, 0], diffs[:, 1])
    
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=dists, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def slsqp_obj(vars_arr):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[:N])

def slsqp_cons(vars_arr):
    """Inequality constraints >= 0: pairwise non-overlap using parameterized coordinates."""
    r = vars_arr[:N]
    u = vars_arr[N:2*N]
    v = vars_arr[2*N:3*N]
    
    # Parameterization automatically enforces r <= x <= 1-r when u,v in [0,1]
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Return d^2 - (r_i + r_j)^2 >= 0 for all pairs
    return d2[PAIR_I, PAIR_J] - rs[PAIR_I, PAIR_J]**2

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_r = None

                x += 0.16

        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
