import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun
    return np.full(n, 1e-9), 0.0

def objective_centers(x):
    """Objective for L-BFGS-B: maximize LP sum of radii."""
    return -solve_lp(x.reshape(N, 2))[1]

def objective_joint(x):
    """Objective for joint SLSQP: maximize sum of radii."""
    return -np.sum(x[2*N:])

def constraints_joint(x):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    cx, cy, r = x[:N], x[N:2*N], x[2*N:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    idx = np.triu_indices(N, k=1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        if len(pts) >= N:

            inits.append(base)

                p = base + rng.uniform(-0.03, 0.03, base.shape)
                inits.append(np.clip(p, 0.04, 0.96))

        inits.append(rng.uniform(0.12, 0.88, (N, 2)))

        x, y = best_c[i]

    best_sum = float(np.sum(best_r))
