import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    
    A_ub = np.zeros((m + 4 * n, n))
    b_ub = np.zeros(m + 4 * n)
    
    diff = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diff ** 2, axis=1))
    A_ub[:m, idx_i] = 1.0
    A_ub[:m, idx_j] = 1.0
    b_ub[:m] = dists
    
    for i in range(n):
        x, y = centers[i]
        A_ub[m + i, i] = 1.0
        b_ub[m + i] = x
        A_ub[m + n + i, i] = 1.0
        b_ub[m + n + i] = 1.0 - x
        A_ub[m + 2 * n + i, i] = 1.0
        b_ub[m + 2 * n + i] = y
        A_ub[m + 3 * n + i, i] = 1.0
        b_ub[m + 3 * n + i] = 1.0 - y
        
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return None, 0.0, None

def get_lp_gradient(centers, res):
    """Computes gradient of LP sum of radii w.r.t centers using dual variables."""
    n = centers.shape[0]
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    grad = np.zeros((n, 2))
    
    try:
        marg = getattr(res, 'marginals', None)
        if marg is None:
            return grad
        lams = getattr(marg, 'ineqlin', None)
        if lams is None:
            return grad
        lams = np.asarray(lams)
        
        valid_pair = lams[:m] > 1e-7
        if np.any(valid_pair):
            k = np.where(valid_pair)[0]
            i, j = idx_i[k], idx_j[k]
            lam = lams[k]
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff, axis=1, keepdims=True)
            dist = np.maximum(dist, 1e-12)
            factors = (lam / dist)
            np.add.at(grad, i, diff * factors)
            np.add.at(grad, j, -diff * factors)
            
        for i in range(n):
            gx = lams[m + i] - lams[m + n + i]
            gy = lams[m + 2 * n + i] - lams[m + 3 * n + i]
            grad[i, 0] += gx
            grad[i, 1] += gy
    except Exception:
        pass
    return grad

def obj_lp(x):
    """Objective and gradient for L-BFGS-B: maximize LP sum of radii."""
    c = x.reshape(N, 2)
    c = np.clip(c, 1e-4, 1.0 - 1e-4)
    r, s, res = solve_lp(c)
    if r is None:
        return 1e6, np.zeros_like(x)
    g = get_lp_gradient(c, res)
    return -s, -g.flatten()

def cons_joint(v):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x, y, r = v[:N], v[N:2*N], v[2*N:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    idx = np.triu_indices(N, k=1)
    c = np.concatenate([c, dx[idx] ** 2 + dy[idx] ** 2 - dr[idx] ** 2])
    return c

def obj_joint(v):
    """Objective for joint SLSQP: maximize sum of radii."""
    return -np.sum(v[2*N:])
