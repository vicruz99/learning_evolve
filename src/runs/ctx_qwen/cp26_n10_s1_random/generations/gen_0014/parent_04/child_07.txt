import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    A_ub = np.zeros((M_PAIRS, n))
    A_ub[np.arange(M_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(M_PAIRS), PAIR_J] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[PAIR_I, PAIR_J]
    
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return None, 0.0, None

def lp_gradient(centers, res):
    """Computes gradient of LP sum of radii w.r.t centers using dual variables."""
    grad = np.zeros((N, 2))
    try:
        lams = getattr(getattr(res, 'ineqlin', None), 'marginals', None)
        if lams is None: return grad
        valid = lams > 1e-8
        if not np.any(valid): return grad
        k = np.where(valid)[0]
        i, j = PAIR_I[k], PAIR_J[k]
        lam = lams[k]
        diff = centers[i] - centers[j]
        dist = np.linalg.norm(diff, axis=1)
        dist_safe = np.maximum(dist, 1e-12)
        factors = (lam / dist_safe)[:, np.newaxis]
        np.add.at(grad, i, diff * factors)
        np.add.at(grad, j, -diff * factors)
    except Exception:
        pass
    return grad

def obj_lp(x):
    """Objective and gradient for L-BFGS-B: maximize LP sum of radii."""
    c = x.reshape(N, 2)
    r, s, res = solve_lp(c)
    if r is None:
        return 1e6, np.zeros_like(x)
    g = lp_gradient(c, res)
    return -s, -g.flatten()

def cons_slsqp(v):
    """Inequality constraints >= 0 for parameterized SLSQP optimization."""
    r = v[:N]
    u = v[N:2*N]
    vc = v[2*N:3*N]
    
    # Automatic boundary satisfaction: x = r + (1-2r)*u, y = r + (1-2r)*vc
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * vc
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    rs2 = rs**2
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    return d2[mask] - rs2[mask]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Phase 1: Generate diverse initial configurations
    configs = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,5,5], [6,6,5,5,4], [5,7,5,5,4]]
    
    for pat in patterns:
        if sum(pat) != N: continue
        pts = []
        r0 = 0.101
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        base = np.array(pts[:N])
        configs.append(base)
        
        for _ in range(4):
            p = base.copy()
            p += rng.uniform(-0.025, 0.025, p.shape)
            p = np.clip(p, 0.03, 0.97)
            configs.append(p)
            
    for _ in range(8):
        configs.append(rng.uniform(0.15, 0.85, (N, 2)))

    best_r = None

        candidates += [current_c[i] + np.array([-s, s]) / 1.414 for s in [step, -step]]
        candidates += [current_c[i] + rng.uniform(-step, step, 2)]

        best_val = best_s

            test_c[i] = cand
            _, s, _ = solve_lp(test_c)

                best_move = cand

            r_lp, s_lp, _ = solve_lp(current_c)
