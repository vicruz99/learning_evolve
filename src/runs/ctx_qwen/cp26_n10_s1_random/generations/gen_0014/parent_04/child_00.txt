

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
M_PAIRS = len(IDX_I)

# Precompute constant LP constraint matrix structure
A_LP = np.zeros((M_PAIRS + 4 * N, N))
for k, (i, j) in enumerate(zip(IDX_I, IDX_J)):
    A_LP[k, i] = 1.0
    A_LP[k, j] = 1.0
for i in range(N):
    A_LP[M_PAIRS + i, i] = 1.0
    A_LP[M_PAIRS + N + i, i] = 1.0
    A_LP[M_PAIRS + 2*N + i, i] = 1.0
    A_LP[M_PAIRS + 3*N + i, i] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    b_ub = np.zeros(M_PAIRS + 4 * N)
    b_ub[:M_PAIRS] = dists[IDX_I, IDX_J]
    
    for i in range(n):
        x, y = centers[i]
        b_ub[M_PAIRS + i] = x
        b_ub[M_PAIRS + N + i] = 1.0 - x
        b_ub[M_PAIRS + 2*N + i] = y
        b_ub[M_PAIRS + 3*N + i] = 1.0 - y
        
    try:
        res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return None, 0.0, None

def lp_obj_grad(x_flat):
    """Objective and gradient for L-BFGS-B: minimize negative LP sum of radii."""
    c = x_flat.reshape(N, 2)
    r, s, res = solve_lp(c)
    if r is None:
        return 1e6, np.zeros_like(x_flat)
        
    grad = np.zeros((N, 2))
    try:
        marg = getattr(getattr(res, 'ineqlin', None), 'marginals', None)
        if marg is not None:
            marg = np.asarray(marg)
            lams = marg[:M_PAIRS]
            mask = lams > 1e-8
            if np.any(mask):
                idx = np.where(mask)[0]
                lam_vals = lams[idx]
                ii, jj = IDX_I[idx], IDX_J[idx]
                dx = c[ii, 0] - c[jj, 0]
                dy = c[ii, 1] - c[jj, 1]
                d = np.sqrt(dx**2 + dy**2)
                d = np.where(d < 1e-12, 1e-12, d)
                fx = lam_vals * dx / d
                fy = lam_vals * dy / d
                np.add.at(grad[:, 0], ii, fx)
                np.add.at(grad[:, 1], ii, fy)
                np.add.at(grad[:, 0], jj, -fx)
                np.add.at(grad[:, 1], jj, -fy)
                
            for i in range(N):
                grad[i, 0] += marg[M_PAIRS + i] - marg[M_PAIRS + N + i]
                grad[i, 1] += marg[M_PAIRS + 2*N + i] - marg[M_PAIRS + 3*N + i]
    except Exception:
        pass
        
    return -s, -grad.flatten()

def obj_equal(v):
    """Objective for equal-radius optimization: minimize negative shared radius t."""
    return -v[-1]

def cons_equal(v):
    """Inequality constraints >= 0 for equal-radius packing."""
    xs = v[:N]
    ys = v[N:2*N]
    t = v[-1]
    c = np.concatenate([xs - t, 1.0 - xs - t, ys - t, 1.0 - ys - t])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    d2 = dx**2 + dy**2
    c = np.concatenate([c, d2[IDX_I, IDX_J] - 4.0*t**2])
    return c

def obj_joint(v):
    """Objective for joint SLSQP: maximize sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x, y, r = v[:N], v[N:2*N], v[2*N:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    c = np.concatenate([c, dx[IDX_I, IDX_J]**2 + dy[IDX_I, IDX_J]**2 - dr[IDX_I, IDX_J]**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(123)
    best_s = -1.0
    best_c = None
    best_r = None
    
    # Phase 1: Generate diverse hexagonal initial configurations
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [5,7,5,5,4], [6,5,5,6,4], [5,6,6,4,5], [5,5,5,5,6],
        [6,6,5,5,4], [4,5,6,5,6], [5,4,6,6,5], [6,4,5,6,5],
        [5,6,4,6,5], [6,5,6,4,5]
    ]
    
    configs = []
    for pat in patterns:
        if sum(pat) < N: continue
        pts = []
        y = 0.101
        row = 0
        for cnt in pat:
            shift = 0.101 if row % 2 == 1 else 0.0
            x = 0.101 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 0.202
            y += np.sqrt(3) * 0.101
            row += 1
        base = np.array(pts[:N])
        configs.append(base)
        for _ in range(6):
            p = base.copy()
            p += rng.uniform(-0.02, 0.02, p.shape)
            p = np.clip(p, 0.05, 0.95)
            configs.append(p)
            
    for _ in range(12):
        configs.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    bounds_eq = [(0.0, 1.0)]*(2*N) + [(0.08, 0.12)]
    bounds_c = [(0.001, 0.999)]*(2*N)
    
    # Phase 2: Equal radius optimization to find tight contact graphs
    for cfg in configs:
        x0 = np.concatenate([cfg.flatten(), [0.10]])
        try:
            res = minimize(obj_equal, x0, method='SLSQP', bounds=bounds_eq,
                           constraints={'type': 'ineq', 'fun': cons_equal},
                           options={'maxiter': 4000, 'ftol': 1e-14})
            if np.isfinite(res.fun) and res.x[-1] > 0.095:
                c_eq = res.x[:2*N].reshape(N, 2)
                r_lp, s_lp, _ = solve_lp(c_eq)
                if r_lp is not None and s_lp > best_s:
                    best_s = s_lp
                    best_c = c_eq.copy()
                    best_r = r_lp.copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = configs[0]
        best_r, best_s, _ = solve_lp(best_c)
        
    # Phase 3: LP gradient ascent on centers (L-BFGS-B)
    for _ in range(10):
        try:
            res = minimize(lp_obj_grad, best_c.flatten(), method='L-BFGS-B', jac=True,
                           bounds=bounds_c, options={'maxiter': 1500, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(N, 2)
                r_opt, s_opt, _ = solve_lp(c_opt)
                if r_opt is not None and s_opt > best_s:
                    best_s = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            pass
            
    # Phase 4: Coordinate descent / Jiggle search with LP evaluation
    curr_c = best_c.copy()
    curr_r = best_r.copy()
    curr_s = best_s
    step = 0.012
    
    for it in range(5000):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        new_c = curr_c[idx] + rng.uniform(-step, step, 2)
        new_c = np.clip(new_c, 0.005, 0.995)
        curr_c[idx] = new_c
        
        r_try, s_try, _ = solve_lp(curr_c)
        if r_try is not None and s_try > curr_s + 1e-9:
            curr_s = s_try
            curr_r = r_try.copy()
            if curr_s > best_s:
                best_s = curr_s
                best_c = curr_c.copy()
                best_r = curr_r.copy()
            step *= 0.998
        else:
            curr_c[idx] = old
            if rng.random() < 0.03:
                step *= 0.96
                
    # Phase 5: Joint SLSQP polish for precise constraint handling
    for _ in range(8):
        pert_c = best_c + rng.uniform(-0.005, 0.005, best_c.shape)
        pert_c = np.clip(pert_c, 0.01, 0.99)
        pert_r = best_r * 0.995
        v0 = np.concatenate([pert_c.flatten(), pert_r])
        bounds_j = [(0.0, 1.0)]*(2*N) + [(1e-6, 0.5)]*N
        try:
            res_j = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                             constraints={'type': 'ineq', 'fun': cons_joint},
                             options={'maxiter': 5000, 'ftol': 1e-14})
            if np.isfinite(res_j.fun):
                c_j = res_j.x[:2*N].reshape(N, 2)
                r_j, s_j, _ = solve_lp(c_j)
                if r_j is not None and s_j > best_s:
                    best_s = s_j
                    best_c = c_j.copy()
                    best_r = r_j.copy()
        except Exception:
            pass
            
    # Fallback safety net
    if best_c is None:
        grid = np.array([(i * 0.18 + 0.1, j * 0.18 + 0.1) for j in range(5) for i in range(5)] + [[0.5, 0.5]])
        best_c = grid[:N]
        best_r, best_s, _ = solve_lp(best_c)
        
    # Final strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(N):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999995
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
```
