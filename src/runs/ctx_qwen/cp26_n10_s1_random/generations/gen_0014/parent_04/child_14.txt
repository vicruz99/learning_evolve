1. **Diverse Initialization**: Generate multiple high-quality starting configurations based on hexagonal lattices with varying row distributions (e.g., [6,5,6,5,4], [5,6,5,6,4]) and random perturbations. This covers the most promising topological basins for N=26.
2. **LP Gradient Ascent**: Use `scipy.optimize.minimize` with `L-BFGS-B` and exact dual gradients from `linprog`. The objective maximizes the sum of radii by moving centers along the gradient derived from active pairwise contact constraints. This efficiently navigates the center space while `linprog` handles optimal radius assignment.
3. **Iterative Coordinate Descent (Jiggle)**: After gradient ascent, perform a randomized local search where individual circles are perturbed and accepted if they increase the LP-optimal sum of radii. This helps escape flat regions and tightens contacts.
4. **Joint SLSQP Polish**: Convert the best configuration into a joint (centers, radii) optimization problem using `SLSQP`. This smooth quadratic constraint formulation ensures precise handling of boundary and non-overlap constraints, often extracting small gains missed by LP-only methods.
5. **Robust Safety Scaling**: Finally, compute a strict scaling factor based on distances to boundaries and inter-circle distances, applying a minimal margin to guarantee strict compliance with the 1e-12 validation tolerance.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun, res
    return np.zeros(n), 0.0, None

def compute_gradient(centers, res):
    """Computes gradient of LP sum of radii w.r.t centers using dual variables."""
    n = centers.shape[0]
    grad = np.zeros((n, 2))
    if res is None: 
        return grad
    
    try:
        if hasattr(res, 'marginals'):
            duals = np.asarray(res.marginals.ineqlin)
        else:
            duals = np.asarray(res.ineqlin.marginals)
    except Exception:
        return grad
        
    idx_i, idx_j = np.triu_indices(n, k=1)
    valid = duals > 1e-8
    if not np.any(valid):
        return grad
        
    i_idx = idx_i[valid]
    j_idx = idx_j[valid]
    lam = duals[valid]
    
    c_i = centers[i_idx]
    c_j = centers[j_idx]
    diff = c_i - c_j
    dist = np.linalg.norm(diff, axis=1)
    dist = np.maximum(dist, 1e-12)
    factors = lam / dist
    
    np.add.at(grad[:, 0], i_idx, diff[:, 0] * factors)
    np.add.at(grad[:, 1], i_idx, diff[:, 1] * factors)
    np.add.at(grad[:, 0], j_idx, -diff[:, 0] * factors)
    np.add.at(grad[:, 1], j_idx, -diff[:, 1] * factors)
    
    return grad

def obj_and_grad(x):
    """Objective and gradient for L-BFGS-B: maximize LP sum of radii."""
    c = x.reshape(N, 2)
    r, s, res = solve_lp(c)
    if r is None: 
        return 1e6, np.zeros_like(x)
    g = compute_gradient(c, res)
    return -s, -g.flatten()

def obj_slsqp(v):
    """Objective for joint SLSQP: maximize sum of radii."""
    return -np.sum(v[2*N:])

def cons_slsqp(v):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    idx = np.triu_indices(N, k=1)
    c = np.concatenate([c, d2[idx] - rs[idx]**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Generate diverse hexagonal initial configurations
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], [4,6,6,6,4],
        [5,6,6,4,5], [6,5,5,6,4], [5,7,4,5,5], [7,5,5,5,4], [5,5,5,5,6]
    ]
    
    starts = []
    for pat in patterns:
        if sum(pat) < N: continue
        pts = []
        r0 = 0.095
        y = r0
        row = 0
        for cnt in pat:
            shift = r0 if row % 2 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        cfg = np.array(pts[:N])
        mn, mx = cfg.min(axis=0), cfg.max(axis=0)
        cfg = (cfg - mn) / (mx - mn + 1e-9) * 0.85 + 0.075
        starts.append(cfg)
        for _ in range(4):
            p = cfg + rng.uniform(-0.02, 0.02, cfg.shape)
            starts.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(10):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    bounds_c = [(0.005, 0.995)] * (2 * N)
    
    # Phase 2: L-BFGS-B optimization with LP dual gradients
    for cfg in starts:
        c = np.clip(cfg, 0.01, 0.99)
        try:
            res = minimize(obj_and_grad, c.flatten(), method='L-BFGS-B', jac=True,
                           bounds=bounds_c, options={'maxiter': 1500, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(N, 2)
                r_opt, s_opt, _ = solve_lp(c_opt)
                if r_opt is not None and s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Phase 3: Iterative coordinate descent / jiggle search to escape local minima
    if best_centers is not None:
        curr_c = best_centers.copy()
        curr_r = best_radii.copy()
        curr_s = best_sum
        step = 0.015
        
        for _ in range(3000):
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            move = rng.uniform(-step, step, 2)
            new_pos = np.clip(curr_c[idx] + move, 0.01, 0.99)
            curr_c[idx] = new_pos
            
            r_try, s_try, _ = solve_lp(curr_c)
            if r_try is not None and s_try > curr_s + 1e-8:
                curr_s = s_try
                curr_r = r_try.copy()
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_centers = curr_c.copy()
                    best_radii = curr_r.copy()
                step *= 0.999
            else:
                curr_c[idx] = old
                if rng.random() < 0.03: step *= 0.95

    # Phase 4: Joint SLSQP polish for precise constraint handling
    if best_centers is not None:
        c_p = best_centers + rng.uniform(-0.004, 0.004, best_centers.shape)
        c_p = np.clip(c_p, 0.01, 0.99)
        r_p = best_radii * 0.995
        v0 = np.concatenate([c_p.flatten(), r_p])
        
        bounds_slqp = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
        
        try:
            res_s = minimize(obj_slsqp, v0, method='SLSQP', bounds=bounds_slqp,
                             constraints={'type': 'ineq', 'fun': cons_slsqp},
                             options={'maxiter': 6000, 'ftol': 1e-14})
            if np.isfinite(res_s.fun):
                c_s = np.column_stack((res_s.x[:N], res_s.x[N:2*N]))
                r_s, s_s, _ = solve_lp(c_s)
                if r_s is not None and s_s > best_sum:
                    best_sum = s_s
                    best_centers = c_s.copy()
                    best_radii = r_s.copy()
        except Exception:
            pass

    # Final strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12: 
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
