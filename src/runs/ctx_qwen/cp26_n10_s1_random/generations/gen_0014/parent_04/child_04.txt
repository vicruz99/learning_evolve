import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-12)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[PAIR_I] - centers[PAIR_J]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    
    A_ub = np.zeros((M_PAIRS, n))
    A_ub[np.arange(M_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(M_PAIRS), PAIR_J] = 1.0
    b_ub = dists
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            r = np.maximum(res.x, 0.0)
            return r, -res.fun, res
    except Exception:
        pass
    return np.zeros(n), 0.0, None

def get_lp_grad(centers, res):
    """Computes gradient of LP sum of radii w.r.t centers using dual variables."""
    n = centers.shape[0]
    grad = np.zeros((n, 2))
    if res is None:
        return grad
    try:
        marg = getattr(getattr(res, 'marginals', None), 'ineqlin', None)
        if marg is None:
            return grad
        lams = np.asarray(marg)
        valid = lams > 1e-8
        if not np.any(valid):
            return grad
        idx = np.where(valid)[0]
        lam = lams[idx]
        i = PAIR_I[idx]
        j = PAIR_J[idx]
        
        diff = centers[i] - centers[j]
        dist = np.linalg.norm(diff, axis=1)
        dist = np.maximum(dist, 1e-12)
        factors = (lam / dist)[:, np.newaxis]
        
        np.add.at(grad, i, diff * factors)
        np.add.at(grad, j, -diff * factors)
    except Exception:
        pass
    return grad

def obj_lp(v):
    """Objective and gradient for L-BFGS-B: maximize LP sum of radii."""
    c = v.reshape(N, 2)
    r, s, res = solve_lp(c)
    if s < 1e-6:
        return 1e6, np.zeros_like(v)
    g = get_lp_grad(c, res)
    return -s, -g.flatten()

def obj_slsqp(v):
    """Objective for joint SLSQP: maximize sum of radii."""
    return -np.sum(v[2*N:])

def cons_slsqp(v):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    idx = np.triu_indices(N, k=1)
    c = np.concatenate([c, d2[idx] - dr[idx]**2])
    return c

def generate_configs(rng):
    """Generates diverse initial configurations."""
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,6,4], [5,5,6,5,5],
        [6,6,5,5,4], [5,7,5,5,4], [6,4,6,5,5], [5,6,4,6,5],
        [7,5,5,5,4], [5,5,5,6,5], [6,5,5,6,4], [5,8,5,5,3]
    ]
    
    for pat in patterns:
        if sum(pat) < N: continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        base = np.array(pts[:N])
        mn = base.min(axis=0)
        mx = base.max(axis=0)
        span = mx - mn + 1e-9
        norm_base = (base - mn) / span * 0.86 + 0.07
        configs.append(norm_base)
        
        for _ in range(5):
            p = norm_base + rng.uniform(-0.025, 0.025, norm_base.shape)
            configs.append(np.clip(p, 0.02, 0.98))
            
    for _ in range(12):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    configs = generate_configs(rng)
    
    best_s = -1.0
    best_c = None
    best_r = None
    
    bounds_c = [(0.001, 0.999)] * (2 * N)
    bounds_slqp = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    
    # Phase 1: SLSQP Joint Optimization from diverse starts
    for cfg in configs:
        v0 = np.zeros(3 * N)
        v0[:N] = cfg[:, 0]
        v0[N:2*N] = cfg[:, 1]
        v0[2*N:] = 0.08
        
        try:
            res = minimize(obj_slsqp, v0, method='SLSQP', bounds=bounds_slqp,
                           constraints={'type': 'ineq', 'fun': cons_slsqp},
                           options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            if np.isfinite(res.fun):
                c_opt = np.column_stack((res.x[:N], res.x[N:2*N]))
                r_opt, s_opt, _ = solve_lp(c_opt)
                if r_opt is not None and s_opt > best_s:
                    best_s = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = configs[0]
        best_r, best_s, _ = solve_lp(best_c)
        
    # Phase 2: LP Gradient Ascent (L-BFGS-B) with random restarts
    for _ in range(8):
        c_pert = best_c + rng.uniform(-0.012, 0.012, best_c.shape)
        c_pert = np.clip(c_pert, 0.001, 0.999)
        try:
            res = minimize(obj_lp, c_pert.flatten(), method='L-BFGS-B', jac=True,
                           bounds=bounds_c, options={'maxiter': 2500, 'ftol': 1e-15})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(N, 2)
                r_opt, s_opt, _ = solve_lp(c_opt)
                if r_opt is not None and s_opt > best_s:
                    best_s = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            pass
            
    # Phase 3: Local Jiggle Search (Coordinate Descent)
    step = 0.025
    for it in range(6000):
        idx = rng.integers(N)
        old_pos = best_c[idx].copy()
        move = rng.uniform(-step, step, 2)
        new_pos = np.clip(best_c[idx] + move, 0.001, 0.999)
        best_c[idx] = new_pos
        
        r_try, s_try, _ = solve_lp(best_c)
        if r_try is not None and s_try > best_s + 1e-9:
            best_s = s_try
            best_r = r_try.copy()
            step *= 1.002
        else:
            best_c[idx] = old_pos
            if rng.random() < 0.15:
                step *= 0.96
                
    # Phase 4: Final SLSQP Polish
    v0 = np.concatenate([best_c.flatten(), best_r])
    try:
        res = minimize(obj_slsqp, v0, method='SLSQP', bounds=bounds_slqp,
                       constraints={'type': 'ineq', 'fun': cons_slsqp},
                       options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
        if np.isfinite(res.fun):
            cx = res.x[:N]
            cy = res.x[N:2*N]
            best_c = np.column_stack((cx, cy))
            best_r, best_s, _ = solve_lp(best_c)
    except Exception:
        pass
        
    # Strict safety scaling to guarantee numerical validity against checker tolerance
    scale = 1.0
    for i in range(N):
        x, y = best_c[i]
        r = best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999995
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
