import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIRS_I)

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[PAIRS_I] - centers[PAIRS_J]
    dists = np.hypot(diffs[:, 0], diffs[:, 1])
    
    A_ub = np.zeros((M_PAIRS, n))
    A_ub[np.arange(M_PAIRS), PAIRS_I] = 1.0
    A_ub[np.arange(M_PAIRS), PAIRS_J] = 1.0
    b_ub = dists
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun, res
    return np.full(n, 1e-6), 0.0, None

def selsqp_objective(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def selsqp_constraints(v):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + M_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[PAIRS_I] - cx[PAIRS_J]
    dy = cy[PAIRS_I] - cy[PAIRS_J]
    dr = r[PAIRS_I] + r[PAIRS_J]
    c[4*N:] = dx**2 + dy**2 - dr**2
    return c

def generate_configs(rng):
    """Generates diverse initial configurations."""
    configs = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 6, 4, 5],
        [5, 7, 5, 5, 4], [7, 5, 5, 5, 4], [4, 5, 6, 5, 6],
        [6, 4, 6, 5, 5], [5, 5, 5, 6, 5], [6, 5, 5, 5, 5]
    ]
    
    for pat in patterns:
        if sum(pat) < N: continue
        pts = []
        r0 = 0.10
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        base = np.array(pts[:N])
        
        mn = base.min(axis=0)
        mx = base.max(axis=0)
        span = mx - mn + 1e-9
        norm_base = (base - mn) / span * 0.85 + 0.075
        configs.append(norm_base)
        
        for _ in range(3):
            p = norm_base + rng.uniform(-0.02, 0.02, norm_base.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(8):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    configs = generate_configs(rng)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    bounds_opt = [(0.001, 0.999)] * (2*N) + [(1e-6, 0.5)] * N
    
    # Phase 1: SLSQP multi-start from feasible LP radii
    for cfg in configs:
        r_init, _, _ = solve_lp_radii(cfg)
        x0 = np.concatenate([cfg.flatten(), r_init * 0.99])
        
        try:
            res = minimize(selsqp_objective, x0, method='SLSQP',
                           bounds=bounds_opt,
                           constraints={'type': 'ineq', 'fun': selsqp_constraints},
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if np.isfinite(res.fun):
                cx_opt = res.x[:N]
                cy_opt = res.x[N:2*N]
                c_opt = np.column_stack((cx_opt, cy_opt))
                
                r_lp, s_lp, _ = solve_lp_radii(c_opt)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_c = c_opt.copy()
                    best_r = r_lp.copy()
        except Exception:
            pass
            
    # Phase 2: Stochastic jiggle search on centers with LP evaluation
    if best_c is not None:
        curr_c = best_c.copy()
        curr_r = best_r.copy()
        curr_s = best_sum
        step = 0.015
        
        for it in range(4000):
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            
            move = rng.uniform(-step, step, 2)
            new_pos = np.clip(curr_c[idx] + move, 0.005, 0.995)
            curr_c[idx] = new_pos
            
            r_try, s_try, _ = solve_lp_radii(curr_c)
            if s_try > curr_s + 1e-9:
                curr_s = s_try
                curr_r = r_try.copy()
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_c = curr_c.copy()
                    best_r = curr_r.copy()
                step *= 0.999
            else:
                curr_c[idx] = old
                if rng.random() < 0.02:
                    step *= 0.95
                    
        # Phase 3: Final SLSQP polish from best jiggle result
        x0_final = np.concatenate([best_c.flatten(), best_r * 0.99])
        try:
            res_final = minimize(selsqp_objective, x0_final, method='SLSQP',
                                 bounds=bounds_opt,
                                 constraints={'type': 'ineq', 'fun': selsqp_constraints},
                                 options={'maxiter': 6000, 'ftol': 1e-13})
            if np.isfinite(res_final.fun):
                cx_f = res_final.x[:N]
                cy_f = res_final.x[N:2*N]
                c_f = np.column_stack((cx_f, cy_f))
                r_f, s_f, _ = solve_lp_radii(c_f)
                if s_f > best_sum:
                    best_sum = s_f
                    best_c = c_f
                    best_r = r_f
        except Exception:
            pass

    # Fallback safety net
    if best_c is None:
        best_c = configs[0]
        best_r, best_sum, _ = solve_lp_radii(best_c)

    # Strict numerical safety scaling
    scale = 1.0
    for i in range(N):
        x, y = best_c[i]
        r = best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
