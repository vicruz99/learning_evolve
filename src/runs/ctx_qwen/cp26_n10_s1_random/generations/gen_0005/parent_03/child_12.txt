import numpy as np
from scipy.optimize import minimize, linprog

def get_hex_config(n, row_counts, r_init):
    """Generates a hexagonal lattice initialization with specified row counts."""
    pts = []
    y = r_init
    row_idx = 0
    for cnt in row_counts:
        shift = r_init if row_idx % 2 == 1 else 0.0
        x = r_init + shift
        for _ in range(cnt):
            if len(pts) < n:
                pts.append([x, y])
            x += 2.0 * r_init
        y += np.sqrt(3) * r_init
        row_idx += 1
    return np.array(pts[:n])

def penalty_func(params, n, mu):
    """Penalty-based objective for unconstrained optimization."""
    c = params[:2*n].reshape(n, 2)
    r = params[2*n:]
    
    # Boundary penalty
    dist_wall = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), np.minimum(c[:, 1], 1.0 - c[:, 1]))
    b_viol = np.maximum(0.0, r - dist_wall)
    b_pen = np.sum(b_viol**2)
    
    # Pairwise overlap penalty
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    dists = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    o_viol = np.maximum(0.0, r_sum - dists)
    
    # Only count upper triangle to avoid double counting and self-interaction
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    o_pen = np.sum(o_viol[mask]**2)
    
    return -np.sum(r) + mu * (b_pen + o_pen)

def neg_sum_radii(vars_array, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars_array[2*n:])

def exact_constraints(vars_array, n):
    """Returns inequality constraints >= 0 for valid packing."""
    c = vars_array[:2*n].reshape(n, 2)
    r = vars_array[2*n:]
    cons = []
    
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap constraints (squared)
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    d2 = dx**2 + dy**2
    np.fill_diagonal(d2, 1.0)
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cons.append(d2[mask] - r_sum_sq[mask])
    
    return np.concatenate(cons)

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    limits = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    
    c_obj = np.ones(n) * -1.0
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun
    return np.full(n, 1e-6), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds_vars = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    configs = []
    row_dists = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],

    # Generate hexagonal lattice starts
    for rd in row_dists:
        if sum(rd) != n: continue
        for r_init in [0.09, 0.10, 0.10135, 0.11]:
            cfg = get_hex_config(n, rd, r_init)
            configs.append(np.clip(cfg, 0.05, 0.95))
            
    # Rotated lattices to break symmetry
    for rd in row_dists[:3]:
        if sum(rd) != n: continue

            cos_t, sin_t = np.cos(angle), np.sin(angle)

            cfg_rot = cfg @ rot
            cfg_rot = (cfg_rot - cfg_rot.min(axis=0)) / (cfg_rot.max(axis=0) - cfg_rot.min(axis=0))
            cfg_rot = cfg_rot * 0.88 + 0.06
            configs.append(np.clip(cfg_rot, 0.05, 0.95))

            d = np.linalg.norm(c[i]-c[j])
            r_sum = r[i] + r[j]
            if r_sum > 1e-12:
                scale = min(scale, d / r_sum)
                
    r *= scale * 0.9999995
    best_sum = float(np.sum(r))
    best_radii = r
    
    return best_centers, best_radii, best_sum
