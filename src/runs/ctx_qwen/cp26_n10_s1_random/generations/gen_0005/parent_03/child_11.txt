2. **Linear Programming Radius Assignment**: For each optimized center configuration, solve an LP to exactly maximize the sum of individual radii subject to pairwise distance and boundary constraints. This gives a tight upper bound for those centers and provides an excellent starting point for joint optimization.
3. **Joint Penalty-Based Refinement**: Use a penalty method with `L-BFGS-B` to simultaneously optimize centers and radii. The objective minimizes `-sum(r) + λ * penalty(violations)`. 4. **Final LP & Safety Scaling**: Run a final LP on the best centers found, then apply a conservative uniform scaling factor to guarantee strict satisfaction of the `1e-12` validation tolerance.

import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun
    return np.full(n, 1e-6), 0.0

def obj_eq_radii(vars_flat, n):
    """Objective to maximize minimum separation for equal-radius packing."""
    c = vars_flat.reshape(n, 2)
    dists = np.sqrt(np.sum((c[:, None, :] - c[None, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pair = np.min(dists)
    to_wall = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                         np.minimum(c[:, 1], 1.0 - c[:, 1]))
    min_wall = np.min(to_wall)
    return -min(min_pair, 2.0 * min_wall)

def joint_penalty(vars_flat, n, lam):
    """Penalty-based objective for joint center and radius optimization."""
    c = vars_flat[:2*n].reshape(n, 2)
    r = vars_flat[2*n:]
    obj = -np.sum(r)
    
    # Boundary penalty
    pen = np.sum(np.maximum(0.0, r - c[:, 0])**2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - c[:, 0]))**2)
    pen += np.sum(np.maximum(0.0, r - c[:, 1])**2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - c[:, 1]))**2)
    
    # Overlap penalty
    dists = np.sqrt(np.sum((c[:, None, :] - c[None, :, :])**2, axis=2))
    np.fill_diagonal(dists, 1e6)
    r_sum = r[:, None] + r[None, :]
    pen += np.sum(np.maximum(0.0, r_sum - dists)**2)
    
    return obj + lam * pen

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    # Phase 0: Generate diverse initial configurations
    configs = []
    row_dist = [5, 6, 5, 6, 4]  # Optimal hex row distribution for N=26
    for r0 in [0.09, 0.10, 0.101, 0.102]:
        pts = []
        y = r0
        row_idx = 0
        for cnt in row_dist:
            shift = r0 if row_idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) < n: pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row_idx += 1
        configs.append(np.array(pts[:n]))
        
    for _ in range(10):
        base = configs[0].copy()
        base += rng.uniform(-0.03, 0.03, base.shape)
        base = np.clip(base, 0.05, 0.95)
        configs.append(base)
        
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Phase 1: Optimize centers for equal radii, then assign optimal radii via LP
    for cfg in configs:
        res = minimize(obj_eq_radii, cfg.flatten(), args=(n,), method='L-BFGS-B',
                       bounds=[(0.01, 0.99)]*(2*n), options={'maxiter': 5000})
        c_opt = res.x.reshape(n, 2)
        r_lp, s_lp = solve_radii_lp(c_opt)
        if s_lp > best_sum:
            best_sum = s_lp
            best_c = c_opt.copy()
            best_r = r_lp.copy()
            
    # Phase 2: Joint optimization with increasing penalty to refine variable radii
    current_c = best_c.copy()
    current_r = best_r.copy()
    lam = 200.0
    
    for step in range(30):
        x0 = np.concatenate([current_c.flatten(), current_r])
        bounds = [(0.02, 0.98)]*(2*n) + [(1e-5, 0.5)]*n
        res = minimize(joint_penalty, x0, args=(n, lam), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 5000, 'ftol': 1e-12})
        current_c = res.x[:2*n].reshape(n, 2)
        current_r = np.maximum(res.x[2*n:], 1e-5)
        
        # Strict validity check before accepting
        valid = True
        if np.any(current_r > current_c[:, 0] - 1e-9) or np.any(current_r > 1.0 - current_c[:, 0] - 1e-9): valid = False
        if np.any(current_r > current_c[:, 1] - 1e-9) or np.any(current_r > 1.0 - current_c[:, 1] - 1e-9): valid = False
        dists = np.sqrt(np.sum((current_c[:, None, :] - current_c[None, :, :])**2, axis=2))
        np.fill_diagonal(dists, 1e6)
        if np.any(dists < current_r[:, None] + current_r[None, :] - 1e-9): valid = False
            
        if valid:
            s = np.sum(current_r)
            if s > best_sum:
                best_sum = s
                best_c = current_c.copy()
                best_r = current_r.copy()
                
        lam *= 1.3
        # Periodic perturbation to escape shallow local minima
        if step % 4 == 0 and step > 0:
            current_c += rng.uniform(-0.003, 0.003, current_c.shape)
            current_c = np.clip(current_c, 0.05, 0.95)
            current_r *= 0.995

    # Phase 3: Final LP refinement on best centers
    r_lp, s_lp = solve_radii_lp(best_c)
    if s_lp > best_sum:
        best_sum = s_lp
        best_r = r_lp
        
    # Phase 4: Safety scaling to strictly satisfy 1e-12 validator tolerance
    scale = 1.0
    for i in range(n):
        scale = min(scale, best_c[i,0]/best_r[i], (1-best_c[i,0])/best_r[i],
                    best_c[i,1]/best_r[i], (1-best_c[i,1])/best_r[i])
    dists = np.sqrt(np.sum((best_c[:, None, :] - best_c[None, :, :])**2, axis=2))
    np.fill_diagonal(dists, 1e6)
    for i in range(n):
        for j in range(i+1, n):
            scale = min(scale, dists[i,j] / (best_r[i] + best_r[j]))
            
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
