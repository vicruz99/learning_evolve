import numpy as np
from scipy.optimize import minimize, linprog

def obj_func(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def cons_func(vars_arr, n):
    """Returns inequality constraints >= 0 for valid packing."""
    c = vars_arr[:2*n].reshape(n, 2)
    r = vars_arr[2*n:]
    
    # Boundary constraints
    b1 = c[:, 0] - r
    b2 = 1.0 - c[:, 0] - r
    b3 = c[:, 1] - r
    b4 = 1.0 - c[:, 1] - r
    
    # Pairwise non-overlap: ||c_i - c_j||^2 >= (r_i + r_j)^2
    dx = c[:, 0, None] - c[:, 0, None].T
    dy = c[:, 1, None] - c[:, 1, None].T
    d2 = dx**2 + dy**2
    np.fill_diagonal(d2, 1.0)  # Placeholder, ignored by triu mask
    
    rs = r[:, None] + r[None, :]
    idx = np.triu_indices(n, k=1)
    p = d2[idx] - rs[idx]**2
    
    return np.concatenate([b1, b2, b3, b4, p])

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    lims = np.maximum(lims, 1e-9)
    
    A = np.zeros((n*(n-1)//2, n))
    b = np.zeros(n*(n-1)//2)
    dists = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A[k, i] = 1.0
        A[k, j] = 1.0
        b[k] = dists[i, j]
        
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, 
                      bounds=[(0.0, l) for l in lims], method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-6)

def relax_centers(centers, radii, steps=150):
    """Force-directed relaxation to reduce overlaps and improve layout."""
    c = centers.copy()
    r = radii.copy()
    
    for t in range(steps):
        diff = c[:, None, :] - c[None, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(d, np.inf)
        
        ov = np.maximum(0.0, r[:, None] + r[None, :] - d)
        dir_vec = diff / np.where(d > 1e-12, d, 1.0)
        force = np.sum(ov[:, :, None] * dir_vec, axis=1)
        
        step_size = 0.04 * (1.0 - t / steps)
        c += force * step_size
        c = np.clip(c, 0.001, 0.999)
        
    return c
