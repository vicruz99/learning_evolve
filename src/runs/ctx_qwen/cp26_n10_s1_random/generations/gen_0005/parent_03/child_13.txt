2. **Phase 1: Equal Radius Optimization**: Optimize centers to maximize a shared radius $R$. This is often easier to converge than joint optimization and quickly finds dense configurations. We use `SLSQP` with explicit distance and boundary constraints.
3. **Phase 2: LP Radius Refinement**: For fixed centers, solve a Linear Program to maximize $\sum r_i$. This exactly computes the optimal variable radii for any given center layout, often gaining ~0.1-0.3% over equal radii by utilizing boundary gaps.
4. **Phase 3: Joint Sum-Radii Optimization**: Use the LP radii as initialization for a direct joint optimization of centers and radii using `SLSQP`. This fine-tunes the layout to squeeze out the last bits of radius sum.
5. **Multi-Restart & Perturbation**: Run the pipeline from multiple perturbed starts to escape local minima. Each iteration builds upon the best found configuration.

import numpy as np
from scipy.optimize import minimize, linprog

def generate_hex_config(n, row_counts, r_init):
    """Generates a hexagonal lattice initialization with specified row counts."""
    pts = []
    y = r_init
    row_idx = 0
    for cnt in row_counts:
        shift = r_init if row_idx % 2 == 1 else 0.0
        x = r_init + shift
        for _ in range(cnt):
            if len(pts) < n:
                pts.append([x, y])
            x += 2.0 * r_init
        y += np.sqrt(3) * r_init
        row_idx += 1
    return np.array(pts[:n])

def equal_r_objective(vars_arr, n):
    """Objective: minimize -R => maximize R"""
    return -vars_arr[-1]

def equal_r_constraints(vars_arr, n):
    """Inequality constraints >= 0 for equal radius packing."""
    c = vars_arr[:2*n].reshape(n, 2)
    R = vars_arr[-1]
    
    # Boundary constraints
    b = np.concatenate([
        c[:, 0] - R,
        1.0 - c[:, 0] - R,
        c[:, 1] - R,
        1.0 - c[:, 1] - R
    ])
    
    # Pairwise squared distance constraints: ||c_i - c_j||^2 >= 4*R^2
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    d2 = dx**2 + dy**2
    np.fill_diagonal(d2, 1.0)
    
    idx = np.triu_indices(n, k=1)
    p = d2[idx] - 4.0 * R**2
    return np.concatenate([b, p])

def sum_r_objective(vars_arr, n):
    """Objective: minimize -sum(r) => maximize sum(r)"""
    return -np.sum(vars_arr[2*n:])

def sum_r_constraints(vars_arr, n):
    """Inequality constraints >= 0 for variable radius packing."""
    c = vars_arr[:2*n].reshape(n, 2)
    r = vars_arr[2*n:]
    
    # Boundary constraints
    b = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    # Pairwise squared distance constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    d2 = dx**2 + dy**2
    np.fill_diagonal(d2, 1.0)
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    idx = np.triu_indices(n, k=1)
    p = d2[idx] - r_sum_sq[idx]
    return np.concatenate([b, p])

def solve_radii_lp(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    
    c_obj = np.ones(n) * -1.0
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    idx = np.triu_indices(n, k=1)
    m = len(idx[0])
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    
    A_ub[:, idx[0]] += 1.0
    A_ub[:, idx[1]] += 1.0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)
    
    # Row distributions that sum to 26
    row_dists = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5]
    ]
    
    initial_configs = []
    for rd in row_dists:
        cfg = generate_hex_config(n, rd, 0.095)
        initial_configs.append(np.clip(cfg, 0.05, 0.95))
        
        # Perturbed versions
        for _ in range(3):
            pert = cfg + rng.uniform(-0.015, 0.015, cfg.shape)
            initial_configs.append(np.clip(pert, 0.05, 0.95))
            
        # Rotated versions
        for angle in [0.03, -0.03, 0.06]:
            cos_t, sin_t = np.cos(angle), np.sin(angle)
            rot = np.array([[cos_t, -sin_t], [sin_t, cos_t]])
            cfg_rot = cfg @ rot
            cfg_rot = (cfg_rot - cfg_rot.min(axis=0)) / (cfg_rot.max(axis=0) - cfg_rot.min(axis=0))
            cfg_rot = cfg_rot * 0.88 + 0.06
            initial_configs.append(np.clip(cfg_rot, 0.05, 0.95))

    bounds_eq = [(0.0, 1.0)] * (2 * n) + [(0.01, 0.12)]
    cons_eq = {'type': 'ineq', 'fun': equal_r_constraints, 'args': (n,)}
    
    # Phase 1: Optimize equal R to find dense center layouts
    for cfg in initial_configs:
        x0 = np.concatenate([cfg.flatten(), [0.09]])
        try:
            res = minimize(equal_r_objective, x0, args=(n,), method='SLSQP',
                           bounds=bounds_eq, constraints=cons_eq,
                           options={'maxiter': 1500, 'ftol': 1e-13})
            if -res.fun > 0.095:
                c_opt = res.x[:2*n].reshape(n, 2)
                # Phase 2: LP refinement
                r_lp, s_lp = solve_radii_lp(c_opt, n)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c_opt.copy()
                    best_radii = r_lp.copy()
        except Exception:
            continue

    # Phase 3: Joint optimization of centers and radii starting from best LP result
    if best_centers is not None:
        bounds_joint = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
        cons_joint = {'type': 'ineq', 'fun': sum_r_constraints, 'args': (n,)}
        
        for iteration in range(6):
            # Perturb centers slightly
            pert_centers = best_centers + rng.uniform(-0.003, 0.003, best_centers.shape)
            pert_centers = np.clip(pert_centers, 0.02, 0.98)
            r_init = best_radii * 0.97
            x0 = np.concatenate([pert_centers.flatten(), r_init])
            
            try:
                res = minimize(sum_r_objective, x0, args=(n,), method='SLSQP',
                               bounds=bounds_joint, constraints=cons_joint,
                               options={'maxiter': 2000, 'ftol': 1e-13})
                if np.isfinite(res.fun):
                    c_opt = res.x[:2*n].reshape(n, 2)
                    r_opt = res.x[2*n:]
                    
                    # Re-verify and LP refine
                    r_lp, s_lp = solve_radii_lp(c_opt, n)
                    if s_lp > best_sum:
                        best_sum = s_lp
                        best_centers = c_opt.copy()
                        best_radii = r_lp.copy()
            except Exception:
                continue

    # Fallback safety net
    if best_centers is None:
        best_centers = np.clip(generate_hex_config(n, [5, 6, 5, 6, 4], 0.09), 0.1, 0.9)
        best_radii, best_sum = solve_radii_lp(best_centers, n)
        
    # Final strict safety scaling to guarantee 1e-12 validator tolerance
    scale = 1.0
    c = best_centers
    r = best_radii
    for i in range(n):
        if r[i] > 1e-12:
            scale = min(scale, c[i,0]/r[i], (1-c[i,0])/r[i], c[i,1]/r[i], (1-c[i,1])/r[i])
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(c[i]-c[j])
            r_sum = r[i] + r[j]
            if r_sum > 1e-12:
                scale = min(scale, d / r_sum)
                
    r *= scale * 0.9999995
    best_sum = float(np.sum(r))
    best_radii = r
    
    return best_centers, best_radii, best_sum
