import numpy as np
from scipy.optimize import minimize, linprog

def compute_min_dist_neg(centers_flat):
    """Objective for center optimization: maximize minimum distance."""
    c = centers_flat.reshape(-1, 2)
    c = np.clip(c, 1e-4, 1.0 - 1e-4)
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pair = np.min(dists)
    
    d_bound = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                         np.minimum(c[:, 1], 1.0 - c[:, 1]))
    min_bound = np.min(d_bound)
    
    # Return negative to minimize
    return -min(min_pair / 2.0, min_bound)

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    limits = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    
    c_obj = np.ones(n) * -1.0
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun
    return np.full(n, 1e-6), 0.0

def joint_objective(vars_array, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(vars_array[2*n:])

def joint_constraints(vars_array, n):
    """Returns inequality constraints >= 0 for valid packing."""
    c = vars_array[:2*n].reshape(n, 2)
    r = vars_array[2*n:]
    cons = []
    
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap constraints
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    d2 = dx**2 + dy**2
    np.fill_diagonal(d2, 1.0)
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cons.append(d2[mask] - r_sum[mask]**2)
    return np.concatenate(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # 1. Generate diverse initial configurations
    configs = []
    row_dist = [5, 6, 5, 6, 4]
    
    for r_init in [0.09, 0.10, 0.10135, 0.105, 0.11]:
        pts = []
        y = r_init
        row_idx = 0
        for cnt in row_dist:
            shift = r_init if row_idx % 2 == 1 else 0.0
            x = r_init + shift
            for _ in range(cnt):
                if len(pts) < n:
                    pts.append([x, y])
                x += 2.0 * r_init
            y += np.sqrt(3) * r_init
            row_idx += 1
        cfg = np.array(pts[:n])
        # Normalize and center
        cfg = (cfg - cfg.min(axis=0)) / (cfg.max(axis=0) - cfg.min(axis=0))
        cfg = cfg * 0.8 + 0.1
        configs.append(cfg)
        
    np.random.seed(42)
    # Add perturbed versions
    for _ in range(5):
        cfg_p = configs[0] + np.random.uniform(-0.025, 0.025, configs[0].shape)
        configs.append(np.clip(cfg_p, 0.05, 0.95))
        
    # 2. Center optimization + LP refinement loop
    for cfg in configs:
        try:
            res = minimize(compute_min_dist_neg, cfg.flatten(), method='Nelder-Mead',
                           options={'maxiter': 1500, 'xatol': 1e-9, 'fatol': 1e-10})
            c_opt = res.x.reshape(-1, 2)
        except:
            c_opt = cfg
            
        r_lp, s_lp = solve_radii_lp(c_opt)
        if s_lp > best_sum:
            best_sum = s_lp
            best_centers = c_opt.copy()
            best_radii = r_lp.copy()
            
    # 3. Local perturbation refinement
    if best_centers is not None:
        for _ in range(6):
            pert = best_centers + np.random.uniform(-0.004, 0.004, best_centers.shape)
            pert = np.clip(pert, 0.05, 0.95)
            try:
                res = minimize(compute_min_dist_neg, pert.flatten(), method='Nelder-Mead',
                               options={'maxiter': 1000, 'xatol': 1e-9})
                c_opt = res.x.reshape(-1, 2)
            except:
                c_opt = pert
                
            r_lp, s_lp = solve_radii_lp(c_opt)
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = c_opt.copy()
                best_radii = r_lp.copy()
                
    # 4. Joint SLSQP polish
    if best_centers is not None:
        bounds_vars = [(0.0, 1.0)] * (2 * n) + [(1e-4, 0.5)] * n
        r0 = best_radii * 0.98
        x0 = np.concatenate([best_centers.flatten(), r0])
        try:
            res = minimize(joint_objective, x0, method='SLSQP', args=(n,),
                          bounds=bounds_vars,
                          constraints={'type': 'ineq', 'fun': joint_constraints, 'args': (n,)},
                          options={'maxiter': 3000, 'ftol': 1e-12})
            if np.isfinite(res.fun):
                c_opt = res.x[:2*n].reshape(n, 2)
                r_lp, s_lp = solve_radii_lp(c_opt)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c_opt.copy()
                    best_radii = r_lp.copy()
        except:
            pass

    # Fallback safety net
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum = solve_radii_lp(best_centers)
        
    # 5. Final safety scaling to strictly satisfy 1e-12 validator tolerance
    scale = 1.0
    c = best_centers
    r = best_radii
    for i in range(n):
        if r[i] > 1e-12:
            scale = min(scale, c[i,0]/r[i], (1-c[i,0])/r[i], c[i,1]/r[i], (1-c[i,1])/r[i])
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(c[i]-c[j])
            r_sum = r[i] + r[j]
            if r_sum > 1e-12:
                scale = min(scale, d / r_sum)
                
    r *= scale * 0.9999995
    best_sum = float(np.sum(r))
    best_radii = r
    
    return best_centers, best_radii, best_sum
