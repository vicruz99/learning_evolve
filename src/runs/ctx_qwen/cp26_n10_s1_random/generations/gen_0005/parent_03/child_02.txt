import numpy as np
from scipy.optimize import minimize, linprog

def get_hex_config(n, row_counts, r_init):
    """Generates a hexagonal lattice initialization with specified row counts."""
    pts = []
    y = r_init
    row_idx = 0
    for cnt in row_counts:
        shift = r_init if row_idx % 2 == 1 else 0.0
        x = r_init + shift
        for _ in range(cnt):
            if len(pts) < n:
                pts.append([x, y])
            x += 2.0 * r_init
        y += np.sqrt(3) * r_init
        row_idx += 1
    return np.array(pts[:n])

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    
    c_obj = np.ones(n) * -1.0
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def penalty_obj(centers_flat, radii, n):
    """Objective for center relaxation: minimizes squared constraint violations."""
    centers = centers_flat.reshape(n, 2)
    
    # Boundary violations
    b_viol = np.concatenate([
        np.maximum(0.0, radii - centers[:, 0]),
        np.maximum(0.0, radii - (1.0 - centers[:, 0])),
        np.maximum(0.0, radii - centers[:, 1]),
        np.maximum(0.0, radii - (1.0 - centers[:, 1]))
    ])
    
    # Pairwise overlap violations
    dx = centers[:, 0:1] - centers[:, 0:1].T
    dy = centers[:, 1:2] - centers[:, 1:2].T
    dists = np.sqrt(dx**2 + dy**2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    idx = np.triu_indices(n, k=1)
    p_viol = np.maximum(0.0, r_sum[idx] - dists[idx])
    
    return np.sum(b_viol**2) + np.sum(p_viol**2)

def slesqp_obj(vars_arr, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def slesqp_cons(vars_arr, n):
    """Inequality constraints >= 0 for joint optimization."""
    c = vars_arr[:2*n].reshape(n, 2)
    r = vars_arr[2*n:]
    
    b = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    d2 = dx**2 + dy**2
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    idx = np.triu_indices(n, k=1)
    return np.concatenate([b, d2[idx] - r_sum[idx]**2])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    np.random.seed(42)
    bounds_c = [(0.0, 1.0)] * (2 * n)
    
    # 1. Generate diverse initial configurations
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [5, 6, 4, 6, 5], [6, 4, 5, 6, 5],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6]
    ]
    
    configs = []
    for pat in patterns:
        if sum(pat) < n: continue
        cfg = get_hex_config(n, pat, 0.095)
        configs.append(cfg)
        for _ in range(3):
            cfg_p = cfg + np.random.uniform(-0.035, 0.035, cfg.shape)
            configs.append(np.clip(cfg_p, 0.05, 0.95))
            
    for _ in range(6):
        configs.append(np.random.uniform(0.15, 0.85, (n, 2)))
        
    # 2. Alternating LP + Center Relaxation Optimization
    for cfg in configs:
        cur_c = cfg.copy()
        cur_r, cur_s = solve_radii_lp(cur_c)
        
        # Iteratively expand radii and relax centers
        for _ in range(8):
            cur_r *= 1.015  # Gentle expansion to drive optimization
            res_c = minimize(
                penalty_obj, cur_c.flatten(), args=(cur_r, n),
                method='L-BFGS-B', bounds=bounds_c,
                options={'maxiter': 3000, 'ftol': 1e-15}
            )
            cur_c = res_c.x.reshape(n, 2)
            
            # Re-solve LP for true max radii at new centers
            new_r, new_s = solve_radii_lp(cur_c)
            if new_s > cur_s:
                cur_r, cur_s = new_r, new_s
            else:
                break  # Converged or hit local limit
                
        if cur_s > best_sum:
            best_sum = cur_s
            best_centers = cur_c.copy()
            best_radii = cur_r.copy()
            
    # 3. Joint SLSQP Polish
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii])
        bounds_vars = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
        
        try:
            res = minimize(
                slesqp_obj, x0, args=(n,),
                method='SLSQP', bounds=bounds_vars,
                constraints={'type': 'ineq', 'fun': slesqp_cons, 'args': (n,)},
                options={'maxiter': 5000, 'ftol': 1e-13, 'iprint': -1}
            )
            if np.isfinite(res.fun):
                c_opt = res.x[:2*n].reshape(n, 2)
                r_opt, s_opt = solve_radii_lp(c_opt)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Fallback safety net
    if best_centers is None:
        best_centers = np.clip(get_hex_config(n, [5,6,5,6,4], 0.09), 0.1, 0.9)
        best_radii, best_sum = solve_radii_lp(best_centers)
        
    # 4. Final strict validity scaling
    scale = 1.0
    c = best_centers
    r = best_radii
    for i in range(n):
        if r[i] > 1e-12:
            scale = min(scale, c[i,0]/r[i], (1.0-c[i,0])/r[i], c[i,1]/r[i], (1.0-c[i,1])/r[i])
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(c[i]-c[j])
            r_sum = r[i] + r[j]
            if r_sum > 1e-12:
                scale = min(scale, d / r_sum)
                
    r *= scale * 0.9999995
    best_sum = float(np.sum(r))
    best_radii = r
    
    return best_centers, best_radii, best_sum
