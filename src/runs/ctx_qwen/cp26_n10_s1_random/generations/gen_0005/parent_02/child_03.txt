2. **Multi-Phase Optimization**: Jointly optimizing centers and radii with SLSQP often gets stuck in local minima due to squared-distance constraints. We will use a phased approach:
   - Phase 1: Generate highly diverse initial configurations (hexagonal lattices with varying densities, randomized shifts, corner/edge biased placements).
   - Phase 2: Run SLSQP from each start, but crucially, *immediately* refine radii with LP after convergence. This decouples the non-convex geometry from the radius maximization.

3. **Vectorized Constraints**: The constraint evaluation is fully vectorized for speed, allowing more iterations and tighter tolerances.

import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    bounds = []
    
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mx, 1e-9)))
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def obj_sum_r(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def cons_sum_r(x, n):
    """Inequality constraints >= 0 for valid packing (vectorized)."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap constraints: squared distance >= squared sum of radii
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    d2 = cx_m**2 + cy_m**2
    rs2 = r_m**2
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    return np.concatenate([c, d2[mask] - rs2[mask]])

def generate_initial_configs(n, rng):
    """Generate diverse starting configurations."""
    configs = []
    
    # 1. Hexagonal lattices with varying base radii and shifts
    for r0 in np.linspace(0.075, 0.105, 6):
        pts = []
        y = r0
        row = 0
        while y + r0 < 1.0 and len(pts) < n:
            shift = r0 if row % 2 == 1 else 0.0
            x = r0 + shift
            while x + r0 < 1.0 and len(pts) < n:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        while len(pts) < n:
            pts.append([0.5, 0.5])
            
        configs.append(np.array(pts[:n]))
        
        # Perturbed hex grids
        pert = np.array(pts[:n]) + rng.uniform(-0.015, 0.015, (n, 2))
        configs.append(np.clip(pert, 0.05, 0.95))
        
    # 2. Corner & Edge biased configurations
    corners = np.array([[0.2, 0.2], [0.8, 0.2], [0.2, 0.8], [0.8, 0.8]])
    mid_edges = np.array([[0.5, 0.15], [0.5, 0.85], [0.15, 0.5], [0.85, 0.5]])
    center_pts = corners + mid_edges
    rest = rng.uniform(0.3, 0.7, (n - len(center_pts), 2))
    configs.append(np.vstack([center_pts, rest]))
    
    # 3. Random dense configurations
    for _ in range(12):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    configs = generate_initial_configs(n, rng)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_opt = [(0.0, 1.0)] * (2 * n) + [(1e-7, 0.5)] * n
    cons_dict = {'type': 'ineq', 'fun': cons_sum_r, 'args': (n,)}
    
    # Phase 1: Broad SLSQP search from diverse starts
    for cfg in configs:
        x0 = np.zeros(3 * n)
        x0[:n] = cfg[:, 0]
        x0[n:2 * n] = cfg[:, 1]
        x0[2 * n:] = 0.07  # Feasible start
        
        try:
            res = minimize(
                obj_sum_r, x0, args=(n,), method='SLSQP', bounds=bounds_opt,
                constraints=cons_dict, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
            )
            if not np.isfinite(res.fun):
                continue
                
            centers_opt = np.column_stack((res.x[:n], res.x[n:2 * n]))
            
            # Refine radii exactly with LP
            r_lp = solve_radii_lp(centers_opt)
            if r_lp is not None:
                s = np.sum(r_lp)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers_opt.copy()
                    best_radii = r_lp.copy()
        except Exception:
            continue
            
    # Phase 2: Local perturbation refinement to escape local minima
    if best_centers is not None:
        for trial in range(10):
            perturbation_scale = 0.008 * (1.0 - trial * 0.1)
            pert = best_centers + rng.normal(0, perturbation_scale, best_centers.shape)
            pert = np.clip(pert, 0.02, 0.98)
            
            x0 = np.zeros(3 * n)
            x0[:n] = pert[:, 0]
            x0[n:2 * n] = pert[:, 1]
            x0[2 * n:] = np.mean(best_radii) * 0.95
            
            try:
                res = minimize(
                    obj_sum_r, x0, args=(n,), method='SLSQP', bounds=bounds_opt,
                    constraints=cons_dict, options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False}
                )
                if not np.isfinite(res.fun):
                    continue
                    
                centers_opt = np.column_stack((res.x[:n], res.x[n:2 * n]))
                r_lp = solve_radii_lp(centers_opt)
                if r_lp is not None:
                    s = np.sum(r_lp)
                    if s > best_sum:
                        best_sum = s
                        best_centers = centers_opt.copy()
                        best_radii = r_lp.copy()
            except Exception:
                continue

    # Fallback (should rarely trigger given robust phases)
    if best_centers is None:
        best_centers = configs[0]
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    # Phase 3: Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
