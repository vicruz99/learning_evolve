import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        for lim in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(lim)
            
    # Pairwise non-overlap: r_i + r_j <= dist(i, j)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                      bounds=[(0.0, None)] * n, method='highs')
        if res.success and np.isfinite(res.fun):
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def hill_climb_lp(centers, n, rng, iterations=2000):
    """Stochastic hill-climbing on centers maximizing LP sum of radii."""
    current_centers = centers.copy()
    best_sum, _ = solve_lp_radii(current_centers)
    best_centers = current_centers.copy()
    
    step = 0.025
    for _ in range(iterations):
        i = rng.integers(n)
        old_c = current_centers[i].copy()
        
        # Propose random move
        new_c = old_c + rng.uniform(-step, step, 2)
        new_c = np.clip(new_c, 1e-4, 1.0 - 1e-4)
        current_centers[i] = new_c
        
        curr_sum, _ = solve_lp_radii(current_centers)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = current_centers.copy()
        else:
            current_centers[i] = old_c  # Revert if worse
            
        # Adaptive step decay
        step *= 0.9997
        if step < 1e-6:
            step = 1e-6  # Maintain minimal exploration
            
    return best_centers, best_sum

def obj_sum_r(x, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2 * n:])

def cons_sum_r(x, n):
    """Inequality constraints >= 0 for SLSQP."""
    cx = x[:n]
    cy = x[n:2 * n]
    r = x[2 * n:]
    
    c = []
    # Boundary constraints
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    # Pairwise non-overlap (smooth squared formulation)
    cx_m = cx[:, np.newaxis] - cx[np.newaxis, :]
    cy_m = cy[:, np.newaxis] - cy[np.newaxis, :]
    r_m = r[:, np.newaxis] + r[np.newaxis, :]
    d2 = cx_m**2 + cy_m**2
    rs2 = r_m**2
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.extend(d2[mask] - rs2[mask])
    return np.array(c)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_opt = [(0.0, 1.0)] * (2 * n) + [(1e-7, 0.5)] * n

                pts.append([x, y])

            row += 1
        while len(pts) < n:
            pts.append([0.5, 0.5])
        inits.append(np.array(pts[:n]))

            x += 0.16

            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
