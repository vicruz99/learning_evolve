import numpy as np
from scipy.optimize import minimize, linprog
import operator

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        for lim in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = lim
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = d
            k += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective_slsqp(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def constraints_slsqp(vars_arr, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c = []
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    d2 = cx_m**2 + cy_m**2
    rs2 = r_m**2
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.extend(d2[mask] - rs2[mask])
    return np.array(c)

def hill_climb_lp(centers, n, rng, iterations=8000):
    """Stochastic hill-climbing on center coordinates using LP to evaluate objective."""
    best_centers = centers.copy()
    r, best_val = solve_lp_radii(best_centers)
    step = 0.012
    for _ in range(iterations):
        i = rng.integers(n)
        old_pos = best_centers[i].copy()
        best_centers[i] += rng.uniform(-step, step, 2)
        best_centers[i] = np.clip(best_centers[i], 1e-4, 1.0 - 1e-4)
        
        r, val = solve_lp_radii(best_centers)
        if val > best_val + 1e-9:
            best_val = val
        else:
            best_centers[i] = old_pos
        step *= 0.9997
    return best_centers, best_val
