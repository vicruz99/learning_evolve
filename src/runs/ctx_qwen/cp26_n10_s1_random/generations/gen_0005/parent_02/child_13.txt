import numpy as np
from scipy.optimize import minimize, linprog

def compute_lp_sum(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    n_bound = 4 * n
    n_pair = n * (n - 1) // 2
    A_ub = np.zeros((n_bound + n_pair, n))
    b_ub = np.zeros(n_bound + n_pair)
    
    idx = 0
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        limits = (x, 1.0 - x, y, 1.0 - y)
        for lim in limits:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, 0.5)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, None

def generate_initial_configs(n, rng):
    """Generate diverse starting configurations for optimization."""
    configs = []
    # Hexagonal lattices with varying densities
    for r0 in np.linspace(0.08, 0.11, 8):
        pts = []
        y = r0
        row = 0
        while y + r0 < 1.0 and len(pts) < n + 5:
            shift = r0 if row % 2 == 1 else 0.0
            x = r0 + shift
            while x + r0 < 1.0 and len(pts) < n + 5:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        # Select points closest to center to minimize edge distortion
        dists_to_center = np.sum((np.array(pts) - 0.5)**2, axis=1)
        idxs = np.argsort(dists_to_center)[:n]
        configs.append(np.array(pts)[idxs])
        
    # Grid-based initialization
    gs = np.linspace(0.1, 0.9, 5)
    grid = np.array([[x, y] for y in gs for x in gs])
    configs.append(grid)
    configs.append(np.vstack([grid[:25], [0.5, 0.5]]))
    
    # Random placements
    for _ in range(10):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
    return configs

def obj_slsqp(x, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def cons_slsqp(x, n):
    """Inequality constraints >= 0 for SLSQP."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    c = []
    # Boundary constraints
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    # Pairwise non-overlap constraints (squared for smoothness)
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    d2 = cx_m**2 + cy_m**2
    rs2 = r_m**2
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.extend(d2[mask] - rs2[mask])
    return np.array(c)
