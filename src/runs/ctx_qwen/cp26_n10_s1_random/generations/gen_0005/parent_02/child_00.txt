import numpy as np
from scipy.optimize import minimize, linprog

def objective_slsqp(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def constraints_slsqp(x, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    num_pairs = n * (n - 1) // 2
    c = np.empty(4 * n + num_pairs)
    
    # Boundary constraints
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    
    # Pairwise non-overlap constraints (squared for smooth gradients)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            c[4*n + idx] = (cx[i] - cx[j])**2 + (cy[i] - cy[j])**2 - (r[i] + r[j])**2
            idx += 1
    return c

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_pairs = n * (n - 1) // 2
    num_cons = 4 * n + num_pairs
    
    A_ub = np.zeros((num_cons, n))
    b_ub = np.zeros(num_cons)
    bounds = []
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mx, 1e-9)))
        for lim in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def hill_climb_centers(init_centers, n, rng, steps=2500, init_step=0.025):
    """Local search on centers using LP to evaluate radius sum."""
    centers = init_centers.copy()
    best_radii = solve_lp_radii(centers)
    if best_radii is None:
        # Fallback if LP fails initially
        r0 = min(np.min(centers), np.min(1.0 - centers)) / 2.0
        best_radii = np.full(n, r0)
        
    best_sum = np.sum(best_radii)
    best_centers = centers.copy()
    
    step = init_step
    for _ in range(steps):
        i = rng.integers(0, n)
        old_c = centers[i].copy()
        
        # Perturb one center
        centers[i] += rng.uniform(-step, step, 2)
        centers[i] = np.clip(centers[i], 1e-5, 1.0 - 1e-5)
        
        radii = solve_lp_radii(centers)
        if radii is not None:
            curr_sum = np.sum(radii)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_radii = radii.copy()
                best_centers = centers.copy()
            else:
                centers[i] = old_c
        else:
            centers[i] = old_c
            
        # Decaying step size
        step *= 0.9993
        
    return best_centers, best_radii, best_sum
