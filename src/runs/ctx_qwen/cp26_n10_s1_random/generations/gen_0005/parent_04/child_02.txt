To maximize the sum of radii for 26 circles in a unit square, we exploit the fact that circles can have different sizes. A highly effective strategy combines:
1. **Diverse Initializations**: Hexagonal lattice patterns, grid layouts, and random placements provide good starting points. Hexagonal patterns naturally minimize overlap and maximize packing density.
2. **Joint Center-Radius Optimization**: We use SLSQP to optimize both centers and radii simultaneously, enforcing non-overlap and boundary constraints. This finds a locally optimal configuration where circles are pushed against each other and walls.
3. **Linear Programming Radius Refinement**: After SLSQP finds a set of centers, fixing those centers and solving a Linear Program (LP) to maximize the sum of radii is exact and fast. The LP constraints are simply `r_i + r_j <= dist(i,j)` and `r_i <= dist(i, boundary)`. This step significantly boosts the sum compared to uniform scaling, as it intelligently assigns larger radii to circles with more space and smaller radii to tight gaps.
4. **Iterative Perturbation & Refinement**: We perturb the best-found centers slightly and repeat the optimization + LP cycle to escape local minima and squeeze out additional space.
5. **Strict Safety Margins**: A tiny multiplicative factor (0.99999999) is applied to LP radii to guarantee strict satisfaction of the `1e-12` tolerance in the validation function without meaningfully reducing the objective value.

This pipeline leverages the strengths of nonlinear constrained optimization for geometry placement and linear programming for optimal radius allocation, consistently pushing past simple uniform-scaling baselines.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, 0.5)] * n
    
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
            
    m_pairs = len(pairs)
    m_total = m_pairs + n
    
    A_ub = np.zeros((m_total, n))
    b_ub = np.zeros(m_total)
    
    idx = 0
    for i, j in pairs:
        d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        b_ub[idx] = d
        idx += 1
        
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(lim, 0.0)
        idx += 1
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def objective(vars_array, n):
    """Objective: minimize negative sum of radii => maximize sum of radii"""
    return -np.sum(vars_array[2 * n:])

def constraints(vars_array, n):
    """Computes inequality constraints >= 0 for valid packing."""
    cx = vars_array[:n]
    cy = vars_array[n:2 * n]
    r = vars_array[2 * n:]
    
    c = np.empty(4 * n + n * (n - 1) // 2)
    c[:n] = cx - r
    c[n:2 * n] = 1.0 - cx - r
    c[2 * n:3 * n] = cy - r
    c[3 * n:4 * n] = 1.0 - cy - r
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    d = np.sqrt(cx_m**2 + cy_m**2)
    r_m = r[:, None] + r[None, :]
    
    c[4 * n:] = d[idx_i, idx_j] - r_m[idx_i, idx_j]
    return c

def make_hex(rows_pattern, r0):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N_CIRCLES:
                break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < N_CIRCLES:
        pts.append([0.5, 0.5])
    return np.array(pts[:N_CIRCLES])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    bounds_vars = [(0.0, 1.0)] * (2 * n) + [(1e-5, 0.45)] * n
    cons = {'type': 'ineq', 'fun': constraints, 'args': (n,)}
    
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    inits = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [5, 7, 5, 5, 4], [6, 4, 6, 5, 5],
        [5, 5, 5, 5, 6], [6, 6, 4, 5, 5]
    ]
    for p in patterns:
        if sum(p) < n:
            continue
        pts = make_hex(p, 0.09)
        inits.append(pts)
        for _ in range(4):
            pert = pts + rng.uniform(-0.025, 0.025, pts.shape)
            inits.append(np.clip(pert, 0.05, 0.95))
            
    for _ in range(15):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # Grid init
    g = np.linspace(0.15, 0.85, 5)
    grid = np.array([[x, y] for y in g for x in g])
    grid = np.vstack([grid, [0.5, 0.5]])
    inits.append(grid)
    
    # Phase 1: Broad search with SLSQP + LP refinement
    for cfg in inits:
        x0 = np.zeros(3 * n)
        x0[:n] = cfg[:, 0]
        x0[n:2 * n] = cfg[:, 1]
        x0[2 * n:] = 0.05  # Safe initial radius to guarantee feasibility
        
        try:
            res = minimize(objective, x0, args=(n,), method='SLSQP', bounds=bounds_vars,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if np.isfinite(res.fun):
                c_opt = res.x[:2 * n].reshape(n, 2)
                r_lp, _ = solve_radii_lp(c_opt)
                if r_lp is not None:
                    r_safe = r_lp * 0.99999999
                    s_safe = np.sum(r_safe)
                    if s_safe > best_sum:
                        best_sum = s_safe
                        best_centers = c_opt.copy()
                        best_radii = r_safe.copy()
        except Exception:
            continue
            
    # Phase 2: Iterative local refinement around the best found configuration
    if best_centers is not None:
        for _ in range(8):
            c_pert = np.clip(best_centers + rng.uniform(-0.008, 0.008, best_centers.shape), 0.05, 0.95)
            x0 = np.zeros(3 * n)
            x0[:n] = c_pert[:, 0]
            x0[n:2 * n] = c_pert[:, 1]
            x0[2 * n:] = best_radii * 0.96
            
            try:
                res = minimize(objective, x0, args=(n,), method='SLSQP', bounds=bounds_vars,
                               constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
                if np.isfinite(res.fun):
                    c_ref = res.x[:2 * n].reshape(n, 2)
                    r_lp, _ = solve_radii_lp(c_ref)
                    if r_lp is not None:
                        r_safe = r_lp * 0.99999999
                        if np.sum(r_safe) > best_sum:
                            best_sum = np.sum(r_safe)
                            best_centers = c_ref.copy()
                            best_radii = r_safe.copy()
            except Exception:
                pass
                
    # Fallback if optimization unexpectedly fails
    if best_centers is None:
        c_fb = make_hex([5, 6, 5, 6, 4], 0.09)
        r_fb, _ = solve_radii_lp(c_fb)
        if r_fb is not None:
            best_centers = c_fb
            best_radii = r_fb * 0.99999999
            best_sum = np.sum(best_radii)
        else:
            best_centers = c_fb
            best_radii = np.full(n, 0.09) * 0.99999999
            best_sum = np.sum(best_radii)
            
    return best_centers, best_radii, float(best_sum)
