import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    pairs_i, pairs_j = np.triu_indices(n, k=1)
    m_pairs = len(pairs_i)
    m_boundary = 4 * n
    
    A_ub = np.zeros((m_pairs + m_boundary, n))
    b_ub = np.zeros(m_pairs + m_boundary)
    
    idx = 0
    for i, j in zip(pairs_i, pairs_j):
        d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        b_ub[idx] = d
        idx += 1
        
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def joint_objective(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def joint_constraints(v, n):
    """Inequality constraints >= 0: boundary and pairwise non-overlap."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    con = np.empty(4 * n)
    con[:n] = cx - r
    con[n:2 * n] = 1.0 - cx - r
    con[2 * n:3 * n] = cy - r
    con[3 * n:4 * n] = 1.0 - cy - r
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    rs = r[idx_i] + r[idx_j]
    con = np.concatenate([con, dx**2 + dy**2 - rs**2])
    
    return con

def make_hex(pattern, r0):
    """Generates initial positions on a hexagonal lattice."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
