import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(lim)
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def penalty_obj(vars_arr, n, mu, triu_i, triu_j):
    """Penalized objective for interleaved [x0, y0, r0, x1, y1, r1, ...] layout."""
    cx = vars_arr[0::3]
    cy = vars_arr[1::3]
    r = vars_arr[2::3]
    
    obj = -np.sum(r)
    
    # Boundary violations
    b = np.maximum(0.0, r - cx)
    b = np.concatenate([b, np.maximum(0.0, r + cx - 1.0)])
    b = np.concatenate([b, np.maximum(0.0, r - cy)])
    b = np.concatenate([b, np.maximum(0.0, r + cy - 1.0)])
    obj += mu * np.sum(b**2)
    
    # Overlap violations
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    o = np.maximum(0.0, r_sum - dist)
    obj += mu * np.sum(o[triu_i, triu_j]**2)
    
    return obj

def slsqp_obj(vars_arr, n):
    """Objective for SLSQP: maximize sum of radii (contiguous layout)."""
    return -np.sum(vars_arr[2*n:])

def slsqp_cons(vars_arr, n):
    """Inequality constraints >= 0 for SLSQP (contiguous layout)."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    c = np.empty(4*n + n*(n-1)//2)
    idx = 0
    
    c[idx:idx+n] = cx - r; idx += n
    c[idx:idx+n] = 1.0 - cx - r; idx += n
    c[idx:idx+n] = cy - r; idx += n
    c[idx:idx+n] = 1.0 - cy - r; idx += n
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    c[idx:] = dist[idx_i, idx_j] - r_sum[idx_i, idx_j]
    
    return c
