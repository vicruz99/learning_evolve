import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(lim)
        
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def constraint_func(vars_arr, n, idx_i, idx_j):
    """
    Computes inequality constraints >= 0 for valid packing.
    Variables layout: [x0..x25, y0..y25, r0..r25]
    """
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    # Boundary constraints
    c = np.empty(4*n + (n*(n-1))//2)
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dist_sq = dx**2 + dy**2
    r_sum_sq = (r[idx_i] + r[idx_j])**2
    
    c[4*n:] = dist_sq - r_sum_sq
    return c

def obj_func(vars_arr, n):
    """Objective: minimize negative sum of radii => maximize sum of radii"""
    return -np.sum(vars_arr[2*n:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    idx_i, idx_j = np.triu_indices(n, k=1)
    
    # Bounds: centers in [0, 1], radii in [small, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    cons_dict = {'type': 'ineq', 'fun': constraint_func, 'args': (n, idx_i, idx_j)}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)
    inits = []
    
    # Hexagonal lattice patterns summing to 26
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [4, 6, 6, 6, 4], 
        [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 7, 5, 5, 4], [6, 4, 6, 5, 5]
    ]
    
    def make_hex(rows, r0=0.09):
        pts = []
        y = r0
        for ri, cnt in enumerate(rows):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2 * r0
            y += np.sqrt(3) * r0
        while len(pts) < n:
            pts.append([0.5, 0.5])
        return np.array(pts[:n])

    # Generate structured and perturbed initial configurations
    for p in patterns:
        if sum(p) != n: continue
        pts = make_hex(p)
        inits.append(pts)
        for _ in range(3):
            pts_pert = np.clip(pts + rng.uniform(-0.025, 0.025, pts.shape), 0.05, 0.95)
            inits.append(pts_pert)
            
    # Add random configurations for diversity
    for _ in range(10):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # Optimize each configuration using alternating SLSQP + LP

        x0[:n] = cfg[:, 0]
        x0[n:2*n] = cfg[:, 1]
        x0[2*n:] = 0.085  # Feasible starting radius

        for step in range(4):

                # Optimize centers and radii jointly
                res = minimize(
                    obj_func, x0, args=(n,), method='SLSQP', bounds=bounds,
                    constraints=cons_dict, options={'maxiter': 2500, 'ftol': 1e-11, 'disp': False}
                )

                new_radii = solve_radii_lp(centers)
                
                # Update variables for next iteration

                x0[n:2*n] = centers[:, 1]
                x0[2*n:] = new_radii

                break

                valid = False
                
        if valid:
            s = np.sum(radii_cand)
            if s > best_sum:
                best_sum = s
                best_centers = centers_cand.copy()
                best_radii = radii_cand.copy()
                
    # Fallback if optimization fails unexpectedly
    if best_centers is None:
        best_centers = make_hex([5, 6, 5, 6, 4])
        best_radii = np.full(n, 0.09)
        best_sum = np.sum(best_radii)

    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
