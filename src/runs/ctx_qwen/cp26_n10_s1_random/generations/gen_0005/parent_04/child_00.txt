import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIU_INDICES = np.triu_indices(N_CIRCLES, k=1)

def penalty_objective(v, mu):
    """Computes penalized objective: minimize -sum(r) + mu * (boundary_violations + overlap_violations)"""
    cx, cy, r = v[:N_CIRCLES], v[N_CIRCLES:2*N_CIRCLES], v[2*N_CIRCLES:]
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary penalties: circle must be inside [0,1]x[0,1]
    b = np.maximum(0.0, r - cx)
    pen += np.sum(b**2)
    b = np.maximum(0.0, r + cx - 1.0)
    pen += np.sum(b**2)
    b = np.maximum(0.0, r - cy)
    pen += np.sum(b**2)
    b = np.maximum(0.0, r + cy - 1.0)
    pen += np.sum(b**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    o = np.maximum(0.0, r_sum[TRIU_INDICES] - dist[TRIU_INDICES])
    pen += np.sum(o**2)
    return obj + mu * pen

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to maximize sum of radii subject to non-overlap and boundary constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(lim)
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                      bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def generate_hex_config(rows_pattern, r0):
    """Generates an initial hexagonal lattice configuration."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N_CIRCLES:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < N_CIRCLES:
        pts.append([0.5, 0.5])
    return np.array(pts[:N_CIRCLES])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    inits = []
    
    # Structured hexagonal patterns known to be near-optimal for square packing
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], [5,6,4,6,5]]
    for p in patterns:
        if sum(p) < N_CIRCLES:
            continue
        c = generate_hex_config(p, 0.095)
        inits.append(c)
        for _ in range(3):
            pc = np.clip(c + rng.uniform(-0.04, 0.04, c.shape), 0.05, 0.95)
            inits.append(pc)
            
    # Random starts to escape symmetric local minima
    for _ in range(5):
        inits.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)))
        
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds_vars = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(1e-5, 0.5)] * N_CIRCLES
    
    for cfg in inits:
        v0 = np.zeros(3 * N_CIRCLES)
        v0[:N_CIRCLES] = cfg[:, 0]
        v0[N_CIRCLES:2*N_CIRCLES] = cfg[:, 1]
        v0[2*N_CIRCLES:] = 0.09
        
        # Continuation method: gradually increase penalty weight
        v_cur = v0.copy()
        for mu in [50.0, 200.0, 1000.0, 5000.0]:
            res = minimize(penalty_objective, v_cur, args=(mu,), method='L-BFGS-B', 
                           bounds=bounds_vars, options={'maxiter': 2000, 'ftol': 1e-12})
            v_cur = res.x
            
        cx_opt, cy_opt, r_opt = v_cur[:N_CIRCLES], v_cur[N_CIRCLES:2*N_CIRCLES], v_cur[2*N_CIRCLES:]
        centers_opt = np.column_stack((cx_opt, cy_opt))
        
        # Exact radius optimization for fixed centers
        r_lp, s_lp = solve_lp_radii(centers_opt)
        if r_lp is not None:
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = centers_opt.copy()
                best_radii = r_lp.copy()
                
    # Fallback configuration
    if best_centers is None:
        best_centers = generate_hex_config([5,6,5,6,4], 0.09)
        r_lp, _ = solve_lp_radii(best_centers)
        best_radii = r_lp
        best_sum = np.sum(best_radii)
        
    # Strict safety scaling to guarantee validity within 1e-12 tolerance
    scale = 1.0
    for i in range(N_CIRCLES):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-9:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
