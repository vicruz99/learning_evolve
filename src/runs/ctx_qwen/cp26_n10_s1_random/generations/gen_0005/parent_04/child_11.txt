import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    A_ub_rows = []
    b_ub_vals = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_vals.append(lim)
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(d)
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub_rows), b_ub=np.array(b_ub_vals), 
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def penalty_objective(v, n, mu):
    """Computes penalized objective: minimize -sum(r) + mu * violations^2"""
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    
    obj = -np.sum(r)
    
    # Boundary penalties
    b1 = np.maximum(0.0, r - cx)
    b2 = np.maximum(0.0, r - (1.0 - cx))
    b3 = np.maximum(0.0, r - cy)
    b4 = np.maximum(0.0, r - (1.0 - cy))
    obj += mu * np.sum(b1**2 + b2**2 + b3**2 + b4**2)
    
    # Overlap penalties
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2 + 1e-12)
    r_sum = r[:, None] + r[None, :]
    viol = np.maximum(0.0, r_sum - dist)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    obj += mu * np.sum(viol[idx_i, idx_j]**2)
    
    return obj

def generate_hex_config(n, r0, shift_x=0.0, shift_y=0.0):
    """Generates initial center positions on a hexagonal lattice."""
    pts = []
    y = r0 + shift_y
    row = 0
    while len(pts) < n:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift + shift_x
        while x + r0 <= 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds_vars = [(0.0, 1.0), (0.0, 1.0), (1e-5, 0.5)] * n
    
    # Generate diverse initial configurations
    inits = []
    hex_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [4, 6, 6, 6, 4], 
        [5, 5, 6, 5, 5], [6, 6, 5, 5, 4], [5, 4, 6, 5, 6]
    ]
    
    for pat in hex_patterns:
        if sum(pat) < n: 
            continue
        # Generate base hex
        pts = []
        y = 0.09
        row = 0
        for cnt in pat:
            shift = 0.09 if row % 2 == 1 else 0.0
            x = 0.09 + shift
            for _ in range(cnt):
                if len(pts) >= n: 
                    break
                pts.append([x, y])
                x += 0.18
            y += 0.09 * np.sqrt(3)
            row += 1
        base = np.array(pts[:n])
        
        # Add perturbations
        for _ in range(4):
            pert = base.copy()
            pert += rng.uniform(-0.03, 0.03, pert.shape)
            pert = np.clip(pert, 0.05, 0.95)
            inits.append(pert)
            
    # Add random starts
    for _ in range(10):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    def optimize_from_init(cfg):
        v0 = np.zeros(3 * n)
        v0[0::3] = cfg[:, 0]
        v0[1::3] = cfg[:, 1]
        v0[2::3] = 0.06  # Start small to avoid huge penalties
        
        # Stage 1: Gentle penalty to find basin
        res1 = minimize(penalty_objective, v0, args=(n, 500.0), method='L-BFGS-B', 
                        bounds=bounds_vars, options={'maxiter': 2000, 'ftol': 1e-12})
        
        # Stage 2: Strong penalty
        res2 = minimize(penalty_objective, res1.x, args=(n, 5000.0), method='L-BFGS-B',
                        bounds=bounds_vars, options={'maxiter': 3000, 'ftol': 1e-12})
                        
        # Stage 3: Very strong penalty for strict feasibility
        res3 = minimize(penalty_objective, res2.x, args=(n, 20000.0), method='L-BFGS-B',
                        bounds=bounds_vars, options={'maxiter': 3000, 'ftol': 1e-12})
                        
        opt_c = res3.x[:2*n].reshape(n, 2)
        return opt_c

    # Phase 1: Initial broad search
    for cfg in inits:
        opt_c = optimize_from_init(cfg)
        radii, s_lp = solve_radii_lp(opt_c)
        if radii is not None and s_lp > best_sum:
            best_sum = s_lp
            best_centers = opt_c.copy()
            best_radii = radii.copy()
            
    # Phase 2: Refinement loop using LP radii to guide center optimization
    if best_centers is not None:
        for cycle in range(4):
            # Perturb best centers slightly to escape local minima
            pert_c = best_centers + rng.uniform(-0.005, 0.005, best_centers.shape)
            pert_c = np.clip(pert_c, 0.05, 0.95)
            
            # Optimize centers starting from perturbation
            opt_c = optimize_from_init(pert_c)
            
            # Solve LP for exact radii
            radii, s_lp = solve_radii_lp(opt_c)
            if radii is not None and s_lp > best_sum:
                best_sum = s_lp
                best_centers = opt_c.copy()
                best_radii = radii.copy()
                
    # Fallback if everything fails
    if best_centers is None:
        best_centers = generate_hex_config(n, 0.095)
        best_radii = np.full(n, 0.095)
        best_sum = np.sum(best_radii)
        
    # Final safety scaling to guarantee strict validity within numerical tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
