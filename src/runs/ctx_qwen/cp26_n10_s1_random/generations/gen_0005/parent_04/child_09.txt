import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """
    Given fixed centers, solves the LP to maximize sum of radii.
    Constraints: r_i <= dist(i, boundary) and r_i + r_j <= dist(i, j)
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, 0.5)] * n
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(lim)
        
    # Pairwise distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def penalty_objective(vars_arr, n, mu):
    """
    Computes the penalized objective function for circle packing.
    Minimizes negative sum of radii while penalizing boundary and overlap violations.
    """
    cx = vars_arr[0::3]
    cy = vars_arr[1::3]
    r = vars_arr[2::3]

    obj = -np.sum(r)

    # Boundary violation penalties
    bv = np.concatenate([
        np.maximum(0.0, r - cx),
        np.maximum(0.0, r + cx - 1.0),
        np.maximum(0.0, r - cy),
        np.maximum(0.0, r + cy - 1.0)
    ])

    # Pairwise overlap penalties
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, 1.0)
    r_sum = r[:, None] + r[None, :]
    o_viol = np.maximum(0.0, r_sum - dist)
    
    triu_idx = np.triu_indices(n, k=1)
    obj += mu * (np.sum(bv**2) + np.sum(o_viol[triu_idx]**2))
    return obj

def make_hex_config(rows_pattern, r0):
    """Generates an initial hexagonal lattice configuration."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
