import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, A_ub, idx_i, idx_j, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 0.0)) for l in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def hill_climb(centers, A_ub, idx_i, idx_j, n, iterations=40000):
    """Hill-climbs on center positions, evaluating sum of radii exactly via LP."""
    best_centers = centers.copy()
    radii, best_sum = solve_lp_radii(best_centers, A_ub, idx_i, idx_j, n)
    if radii is None:
        return best_centers, np.zeros(n), 0.0
    best_radii = radii
    
    step = 0.04

        i = np.random.randint(n)
        old_pos = centers[i].copy()
        best_local_sum = best_sum
        best_pos = old_pos
        best_r = None

        for _ in range(2):
            new_pos = centers[i] + np.random.uniform(-step, step, 2)
            new_pos = np.clip(new_pos, 1e-4, 0.999)
            centers[i] = new_pos
            
            lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                              np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
            bounds = [(0.0, max(l, 0.0)) for l in lims]

            b_ub = dists[idx_i, idx_j]

                        best_local_sum = s_new
                        best_pos = new_pos.copy()
                        best_r = res.x
            except Exception:
                pass

            best_centers = centers.copy()
            best_radii = best_r
            step *= 1.0001
        else:
            step *= 0.9998
            
        if it % 5000 == 0:
            step = max(step * 0.5, 0.001)
            
    return best_centers, best_radii, best_sum

def obj_var(v, n):
    return -np.sum(v[2*n:])

def cons_var(v, n, idx_i, idx_j):
    xs = v[:n]
    ys = v[n:2*n]
    rs = v[2*n:]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[idx_i] - xs[idx_j]
    dy = ys[idx_i] - ys[idx_j]
    dr = rs[idx_i] + rs[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def run_packing():
