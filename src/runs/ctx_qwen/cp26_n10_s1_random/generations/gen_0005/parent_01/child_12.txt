import numpy as np
from scipy.optimize import minimize, linprog

def penalty_objective(vars, n, mu):
    """Objective with smooth penalty for boundary and overlap constraints."""
    cx = vars[:n]
    cy = vars[n:2*n]
    r = vars[2*n:]
    
    obj = -np.sum(r)
    penalty = 0.0
    
    # Boundary penalties: r <= x, r <= 1-x, r <= y, r <= 1-y
    diff_x1 = r - cx
    diff_x2 = r - (1.0 - cx)
    diff_y1 = r - cy
    diff_y2 = r - (1.0 - cy)
    
    penalty += np.sum(np.maximum(0.0, diff_x1)**2)
    penalty += np.sum(np.maximum(0.0, diff_x2)**2)
    penalty += np.sum(np.maximum(0.0, diff_y1)**2)
    penalty += np.sum(np.maximum(0.0, diff_y2)**2)
    
    # Pairwise penalties: r_i + r_j <= dist(i,j)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    
    r_sum = r[:, None] + r[None, :]
    diff = r_sum - dist
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    violations = np.maximum(0.0, diff[mask])
    penalty += np.sum(violations**2)
    
    return obj + mu * penalty

def solve_lp_radii(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 0.0)
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def constraints_var(vars, n):
    """Inequality constraints >= 0 for variable radius packing."""
    xs = vars[:n]
    ys = vars[n:2*n]
    rs = vars[2*n:]
    
    # Boundary constraints
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Pairwise non-overlap constraints (squared distances for smoothness)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    idx = np.triu_indices(n, k=1)
    c = np.concatenate([c, (dx[idx]**2 + dy[idx]**2) - dr[idx]**2])
    
    return c

def objective_sum_radii(vars, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2*n:])

def generate_initial_configs(n):
    """Generates diverse high-quality initial center configurations."""
    configs = []
    rng = np.random.default_rng(42)
    
    # Hexagonal patterns with various row distributions summing to >= 26
    row_dists = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], 
        [4, 6, 6, 6, 4], [5, 6, 6, 5, 4], [6, 6, 5, 5, 4]
    ]
    
    for rc in row_dists:
        pts = []
        y = 0.08
        for idx, cnt in enumerate(rc):
            shift = 0.09 if idx % 2 == 1 else 0.0
            width = (cnt - 1) * 0.18
            x_start = 0.5 - width / 2.0 + shift
            for c in range(cnt):
                if len(pts) < n:
                    pts.append([x_start + c * 0.18, y])
            y += 0.155
            if len(pts) >= n: break
        configs.append(np.array(pts[:n]))
        
        # Add perturbed versions
        for _ in range(2):
            cfg_pert = np.clip(pts[:n] + rng.uniform(-0.02, 0.02, (n, 2)), 0.05, 0.95)
            configs.append(cfg_pert)

    # Centered hexagonal lattice
    pts = []
    r0 = 0.09
    y = r0
    row = 0
    while len(pts) < n + 10:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x + r0 <= 1.0:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
        row += 1
    pts = np.array(pts)
    mx, my = pts.mean(axis=0)
    pts = pts - np.array([mx - 0.5, my - 0.5])
    configs.append(pts[:n])
    
    # Random dense starts
    for _ in range(3):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    configs = generate_initial_configs(n)
    
    for cfg in configs:
        # Phase 1: Penalty-based L-BFGS-B optimization with continuation
        v = np.concatenate([cfg[:, 0], cfg[:, 1], np.full(n, 0.08)])
        bounds_pen = [(0.0, 1.0)] * (2*n) + [(1e-4, 0.5)] * n
        
        try:
            for mu in [50.0, 200.0, 1000.0, 5000.0, 20000.0]:
                res = minimize(penalty_objective, v, args=(n, mu), method='L-BFGS-B',
                               bounds=bounds_pen, options={'maxiter': 2000, 'ftol': 1e-12})
                v = res.x
        except Exception:
            continue
            
        cx_pen = v[:n]
        cy_pen = v[n:2*n]
        centers_pen = np.column_stack((cx_pen, cy_pen))
        
        # Phase 2: LP refinement to get optimal radii for optimized centers
        radii_lp, sum_lp = solve_lp_radii(centers_pen, n)
        if radii_lp is None:
            continue
            
        # Phase 3: SLSQP joint optimization with hard constraints to polish
        v0 = np.zeros(3*n)
        v0[:n] = centers_pen[:, 0]
        v0[n:2*n] = centers_pen[:, 1]
        # Shrink slightly to guarantee initial feasibility for SLSQP
        v0[2*n:] = np.maximum(radii_lp * 0.97, 1e-4)
        
        bounds_sl = [(0.0, 1.0)] * (2*n) + [(1e-5, 0.5)] * n
        
        try:
            res_sl = minimize(objective_sum_radii, v0, args=(n,),
                              method='SLSQP', bounds=bounds_sl,
                              constraints={'type': 'ineq', 'fun': constraints_var, 'args': (n,)},
                              options={'maxiter': 4000, 'ftol': 1e-13})
            
            if np.isfinite(res_sl.fun):
                cx_sl = res_sl.x[:n]
                cy_sl = res_sl.x[n:2*n]
                centers_sl = np.column_stack((cx_sl, cy_sl))
                
                # Final LP refinement on polished centers
                radii_final, sum_final = solve_lp_radii(centers_sl, n)
                if radii_final is not None and sum_final > best_sum:
                    best_sum = sum_final
                    best_centers = centers_sl
                    best_radii = radii_final
        except Exception:
            # Fallback to LP result if SLSQP fails
            if sum_lp > best_sum:
                best_sum = sum_lp
                best_centers = centers_pen
                best_radii = radii_lp

    # Guaranteed fallback configuration
    if best_centers is None:
        cfg = configs[0]
        best_centers = cfg
        radii_fb, _ = solve_lp_radii(cfg, n)
        best_radii = radii_fb if radii_fb is not None else np.full(n, 0.08)
        best_sum = np.sum(best_radii)

    # Final safety scaling to guarantee strict numerical validity against checker tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
