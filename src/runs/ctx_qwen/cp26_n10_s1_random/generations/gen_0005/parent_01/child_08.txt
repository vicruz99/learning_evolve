

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we leverage the fact that for fixed centers, finding optimal radii is a Linear Programming (LP) problem. The objective $\sum r_i$ with constraints $r_i + r_j \le d_{ij}$ and $r_i \le \text{dist}(c_i, \text{boundary})$ is convex and can be solved exactly and very quickly using `scipy.optimize.linprog` with the HiGHS solver.

Instead of relying solely on gradient-based optimizers (like SLSQP) which can struggle with the high dimensionality and non-smooth active constraints of 325 pairwise non-overlap conditions, we use a **Hill-Climbing optimization on the centers** where the objective function is the LP-optimal sum of radii. This approach:
1. Directly optimizes the true objective we care about.
2. Avoids constraint violation issues during optimization since LP guarantees feasibility.
3. Is highly efficient because the LP constraint matrix structure remains constant; only the RHS (`b_ub`) changes with centers.

We initialize with multiple diverse hexagonal lattice patterns and random configurations. For each, we run a stochastic hill-climb with decaying step sizes to find local optima. Finally, we perform multiple refinement passes from the best found configuration to escape shallow local minima. A strict safety scaling is applied at the end to guarantee numerical validity against the checker's $10^{-12}$ tolerance.
</strategy>

import numpy as np
from scipy.optimize import linprog

def build_lp_matrix(n):
    """Precomputes the constant constraint matrix for the LP radius optimization."""
    n_pairs = n * (n - 1) // 2
    n_bound = 4 * n
    A_ub = np.zeros((n_pairs + n_bound, n))
    
    idx = 0
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        for k in range(4):
            A_ub[idx, i] = 1.0
            idx += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    pair_indices = []
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            pair_indices.append((i, j))
            idx += 1
            
    return A_ub, pair_indices

def solve_lp_fast(centers, A_ub, pair_indices, n):
    """Solves the LP to maximize sum of radii for fixed centers efficiently."""
    xs, ys = centers[:, 0], centers[:, 1]
    n_pairs = len(pair_indices)
    b_ub = np.empty(4 * n + n_pairs)
    
    idx = 0
    # Fill boundary limits
    for i in range(n):
        b_ub[idx] = xs[i]
        b_ub[idx+1] = 1.0 - xs[i]
        b_ub[idx+2] = ys[i]
        b_ub[idx+3] = 1.0 - ys[i]
        idx += 4
        
    # Fill pairwise distances
    for i, j in pair_indices:
        dx = xs[i] - xs[j]
        dy = ys[i] - ys[j]
        b_ub[idx] = np.sqrt(dx*dx + dy*dy)
        idx += 1
        
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def hill_climb(centers, n, A_ub, pair_indices, iterations=4000):
    """Stochastic hill-climbing on center coordinates, evaluated via LP."""
    best_centers = centers.copy()
    radii, best_sum = solve_lp_fast(centers, A_ub, pair_indices, n)
    if radii is None:
        return centers, np.full(n, 0.05), 1.3
        
    best_radii = radii.copy()
    step_base = 0.025
    
    for it in range(iterations):
        # Decaying step size for coarse-to-fine optimization
        curr_step = step_base * (1.0 - 0.4 * (it / iterations))
        if curr_step < 1e-6: curr_step = 1e-6
        
        i = np.random.randint(n)
        old_c = centers[i].copy()
        
        # Perturb one circle randomly
        centers[i, 0] += np.random.uniform(-curr_step, curr_step)
        centers[i, 1] += np.random.uniform(-curr_step, curr_step)
        # Keep strictly inside to maintain LP feasibility bounds > 0
        centers[i] = np.clip(centers[i], 1e-4, 1.0 - 1e-4)
        
        radii, curr_sum = solve_lp_fast(centers, A_ub, pair_indices, n)
        
        # Accept if improvement
        if radii is not None and curr_sum > best_sum + 1e-9:
            best_sum = curr_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
        else:
            centers[i] = old_c
            
    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    A_ub, pair_indices = build_lp_matrix(n)
    
    # Diverse high-quality initial configurations
    configs = []
    row_dists = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], 
        [4, 6, 6, 6, 4], [5, 6, 6, 5, 4], [5, 7, 5, 5, 4]
    ]
    
    for rc in row_dists:
        pts = []
        y = 0.085
        for idx, cnt in enumerate(rc):
            shift = 0.09 if idx % 2 == 1 else 0.0
            width = (cnt - 1) * 0.18
            x_start = 0.5 - width / 2.0 + shift
            for c in range(cnt):
                if len(pts) < n:
                    pts.append([x_start + c * 0.18, y])
            y += 0.155
            if len(pts) >= n: break
        configs.append(np.array(pts[:n]))
        
    # Add random dense starts to break lattice symmetry bias
    for _ in range(4):
        c = np.random.uniform(0.15, 0.85, (n, 2))
        configs.append(c)
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Explore multiple basins
    for cfg in configs:
        c_opt, r_opt, s_opt = hill_climb(cfg.copy(), n, A_ub, pair_indices, iterations=3000)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            best_radii = r_opt
            
    # Phase 2: Intensive local refinement around the best found configuration
    if best_centers is not None:
        for _ in range(6):
            pert = best_centers + np.random.uniform(-0.008, 0.008, (n, 2))
            pert = np.clip(pert, 0.05, 0.95)
            c_ref, r_ref, s_ref = hill_climb(pert, n, A_ub, pair_indices, iterations=4000)
            if s_ref > best_sum:
                best_sum = s_ref
                best_centers = c_ref
                best_radii = r_ref
                
    # Fallback safety
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum = solve_lp_fast(best_centers, A_ub, pair_indices, n)
        
    # Strict safety scaling to guarantee validity against 1e-12 checker tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt((best_centers[i, 0] - best_centers[j, 0])**2 + (best_centers[i, 1] - best_centers[j, 1])**2)
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
