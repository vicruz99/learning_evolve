import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    # Prevent negative bounds due to numerical precision
    limits = np.maximum(limits, 1e-9)
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def hill_climb(centers_init, n, iterations=4000, rng=None):
    """Optimizes centers by maximizing the LP-derived sum of radii."""
    if rng is None: rng = np.random.default_rng(42)
    centers = centers_init.copy()
    radii, best_sum = solve_lp_radii(centers, n)
    if radii is None:
        best_radii = np.full(n, 0.01)
        best_sum = 0.26
    else:
        best_radii = radii.copy()
        
    best_centers = centers.copy()
    step = 0.05
    stagnation = 0
    
    for it in range(iterations):
        step *= 0.9985
        # Shake if stuck in a shallow local minimum
        if stagnation > 1000 and step < 0.005:
            step = 0.02 
            stagnation = 0
            
        i = rng.integers(n)
        old_c = centers[i].copy()
        centers[i] += rng.uniform(-step, step, 2)
        centers[i] = np.clip(centers[i], 1e-4, 1.0 - 1e-4)
        
        radii, curr_sum = solve_lp_radii(centers, n)
        if radii is not None and curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            stagnation = 0
        else:
            centers[i] = old_c
            stagnation += 1
            
    return best_centers, best_radii, best_sum

def generate_inits(n):
    """Generates diverse high-quality initial center configurations."""
    configs = []
    rng = np.random.default_rng(123)
    
    # Hexagonal lattice patterns with various row distributions
    rows_pats = [[5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [4,6,6,6,4]]
    for pat in rows_pats:
        pts = []
        y = 0.08
        for r_idx, cnt in enumerate(pat):
            shift = 0.09 if r_idx % 2 == 1 else 0.0
            width = (cnt - 1) * 0.18
            x_start = 0.5 - width/2.0 + shift
            for c in range(cnt):
                if len(pts) < n:
                    pts.append([x_start + c * 0.18, y])
            y += 0.155
        configs.append(np.array(pts[:n]))
        
    # Grid 5x5 + 1 circle
    g = np.linspace(0.12, 0.88, 5)
    grid = np.array([[x,y] for y in g for x in g])
    grid = np.vstack([grid, [0.5, 0.05]])
    configs.append(grid)
    
    # Random placements refined by force simulation
    for _ in range(2):
        c = rng.uniform(0.15, 0.85, (n, 2))
        r_sim = 0.05
        for _ in range(800):
            forces = np.zeros_like(c)
            for i in range(n):
                for d in range(2):
                    if c[i,d] - r_sim < 0: forces[i,d] += 10.0 * (r_sim - c[i,d])
                    if c[i,d] + r_sim > 1.0: forces[i,d] -= 10.0 * (c[i,d] + r_sim - 1.0)
                for j in range(i+1, n):
                    dx = c[i,0] - c[j,0]
                    dy = c[i,1] - c[j,1]
                    d = np.sqrt(dx*dx + dy*dy) + 1e-9
                    if d < 2.0*r_sim:
                        f = 5.0 * (2.0*r_sim - d) / d
                        forces[i,0] += dx*f; forces[i,1] += dy*f
                        forces[j,0] -= dx*f; forces[j,1] -= dy*f
            c += forces * 0.005
            c = np.clip(c, 1e-3, 1.0-1e-3)
        configs.append(c)
        
    return configs

def objective_joint(vars_arr, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def constraints_joint(vars_arr, n):
    """Smooth squared-distance constraints for joint optimization."""
    xs = vars_arr[:n]
    ys = vars_arr[n:2*n]
    rs = vars_arr[2*n:]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    idx = np.triu_indices(n, k=1)
    c = np.concatenate([c, (dx[idx]**2 + dy[idx]**2) - dr[idx]**2])
    return c
