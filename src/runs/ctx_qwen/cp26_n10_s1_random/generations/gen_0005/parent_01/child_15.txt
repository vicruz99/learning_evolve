from scipy.optimize import minimize, linprog

N = 26

# Precompute LP constraint matrix structure (constant for fixed N)
TRIU_IDX = np.triu_indices(N, k=1)
A_LP = np.zeros((len(TRIU_IDX[0]), N))
for k, (i, j) in enumerate(zip(TRIU_IDX[0], TRIU_IDX[1])):
    A_LP[k, i] = 1.0
    A_LP[k, j] = 1.0

def _lp_sum(centers):
    """Returns max sum of radii for fixed centers via LP."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 0.0)) for l in lims]

    d2 = diff[:, :, 0]**2 + diff[:, :, 1]**2
    d2[np.arange(N), np.arange(N)] = np.inf
    b_ub = np.sqrt(d2)[TRIU_IDX]
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        return -res.fun if res.success else 0.0
    except Exception:
        return 0.0

def _lp_radii(centers):
