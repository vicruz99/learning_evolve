import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Constraints: r_i + r_j <= dist(i,j)
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    bounds = [(0.0, lims[i]) for i in range(n)]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def constraint_func(vars_arr, n):
    """Computes inequality constraints >= 0 for valid packing."""
    xs = vars_arr[:n]
    ys = vars_arr[n:2*n]
    rs = vars_arr[2*n:]
    
    # Boundary constraints: circles inside [0,1]x[0,1]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Pairwise non-overlap constraints (squared for smooth gradients)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    idx = np.triu_indices(n, k=1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def objective_func(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def hill_climb_lp(init_c, n, steps=1500, rng=None):
    """Stochastic hill-climbing on centers, evaluating objective via LP."""
    if rng is None:
        rng = np.random.default_rng(123)
        
    centers = init_c.copy()
    _, best_sum = solve_lp_radii(centers, n)
    if best_sum < 1e-6:
        return centers, 0.0
        
    best_c = centers.copy()
    step_size = 0.025
    
    for _ in range(steps):
        i = rng.integers(n)
        old_c = centers[i].copy()
        centers[i] += rng.uniform(-step_size, step_size, 2)
        centers[i] = np.clip(centers[i], 1e-4, 1.0 - 1e-4)
        
        _, curr_sum = solve_lp_radii(centers, n)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_c = centers.copy()
        else:
            centers[i] = old_c
            
        step_size *= 0.9985
    return best_c, best_sum

def generate_initial_configs(n):
    """Generates diverse initial center configurations."""
    configs = []
    rng = np.random.default_rng(42)
    
    # 1. Hexagonal lattices with different row distributions
    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [5, 6, 6, 5, 4], [6, 6, 5, 5, 4]
    ]
    
    for rc in row_patterns:
        pts = []
        y = 0.06
        for idx, cnt in enumerate(rc):
            shift = 0.06 if idx % 2 == 1 else 0.0
            width = (cnt - 1) * 0.12
            x_start = 0.5 - width / 2.0 + shift
            for c in range(cnt):
                if len(pts) < n:
                    pts.append([x_start + c * 0.12, y])
            y += 0.104 # ~ sqrt(3)/2 * 0.12
            if len(pts) >= n: break
        configs.append(np.array(pts[:n]))
        
    # 2. Perturbed hex lattices to break symmetry
    for cfg in configs[:3]:
        for _ in range(2):
            pert = cfg.copy()
            pert += rng.uniform(-0.02, 0.02, pert.shape)
            pert = np.clip(pert, 0.05, 0.95)
            configs.append(pert)
            
    # 3. Random dense configurations
    for _ in range(5):
        c = rng.uniform(0.1, 0.9, (n, 2))
        configs.append(c)
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_vars = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.25)] * n
    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}
    
    configs = generate_initial_configs(n)
    rng = np.random.default_rng(42)
    
    for cfg in configs:
        # Phase 1: LP-based hill climbing to find good center arrangement
        hc_c, hc_sum = hill_climb_lp(cfg, n, steps=1500, rng=rng)
        radii_lp, _ = solve_lp_radii(hc_c, n)
        if radii_lp is None:
            continue
            
        # Phase 2: Joint SLSQP optimization of centers and radii
        v0 = np.zeros(3 * n)
        v0[:n] = hc_c[:, 0]
        v0[n:2*n] = hc_c[:, 1]
        v0[2*n:] = np.maximum(radii_lp * 0.98, 1e-6)
        
        try:
            res = minimize(
                objective_func, v0, args=(n,),
                method='SLSQP', bounds=bounds_vars,
                constraints=cons,
                options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False}
            )
            
            if np.isfinite(res.fun):
                cx = res.x[:n]
                cy = res.x[n:2*n]
                r = res.x[2*n:]
                centers_opt = np.column_stack((cx, cy))
                
                # Strict validation
                valid = True
                if np.any(cx < r - 1e-10) or np.any(cx > 1.0 - r + 1e-10): valid = False
                if np.any(cy < r - 1e-10) or np.any(cy > 1.0 - r + 1e-10): valid = False
                
                if valid:
                    idx = np.triu_indices(n, k=1)
                    dx = cx[idx[0]] - cx[idx[1]]
                    dy = cy[idx[0]] - cy[idx[1]]
                    dr = r[idx[0]] + r[idx[1]]
                    if np.any(dx**2 + dy**2 < dr**2 - 1e-10):
                        valid = False
                        
                if valid:
                    s = np.sum(r)
                    if s > best_sum:
                        best_sum = s
                        best_centers = centers_opt.copy()
                        best_radii = r.copy()
        except Exception:
            continue
            
    # Fallback configuration if optimization fails unexpectedly
    if best_centers is None:
        best_centers = configs[0]
        radii_fb, _ = solve_lp_radii(best_centers, n)
        best_radii = radii_fb if radii_fb is not None else np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    # Final safety scaling to guarantee strict numerical validity for the checker
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
