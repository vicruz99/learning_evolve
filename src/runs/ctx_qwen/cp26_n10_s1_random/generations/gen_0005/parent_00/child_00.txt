import numpy as np
from scipy.optimize import linprog

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    # Boundary constraints: r_i <= dist to each wall
    for i in range(n):
        x, y = centers[i]
        limits = (x, 1.0 - x, y, 1.0 - y)
        for lim in limits:
            A_ub[idx, i] = 1.0
            b_ub[idx] = max(lim, 0.0)
            idx += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def generate_inits(n):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattices with varying base scales
    for scale in [0.085, 0.09, 0.095, 0.10, 0.105]:
        pts = []
        y = scale
        row = 0
        while len(pts) < n + 5:
            shift = scale if row % 2 == 1 else 0.0
            x = scale + shift
            while x < 1.0 - scale and len(pts) < n + 5:
                pts.append([x, y])
                x += 2.0 * scale
            y += np.sqrt(3) * scale
            row += 1
        pts = np.array(pts[:n])
        # Normalize and center in the square
        m = pts.min(axis=0)
        M = pts.max(axis=0)
        pts = (pts - m) / (M - m) * 0.86 + 0.07
        inits.append(pts)
        
    # 2. Regular grid + center point
    gx = np.linspace(0.12, 0.88, 5)
    gy = np.linspace(0.12, 0.88, 5)
    grid = np.array([(x, y) for y in gy for x in gx])
    grid = np.vstack([grid, [0.5, 0.5]])
    inits.append(grid)
    
    return inits

                best_centers = curr_c.copy()
                best_radii = r.copy()

                step = step * 0.9991  # Slow decay on success

                curr_c[i] = old_pos  # Revert
                step = step * 0.9997
                
            # Periodic random restarts from best to escape local optima
