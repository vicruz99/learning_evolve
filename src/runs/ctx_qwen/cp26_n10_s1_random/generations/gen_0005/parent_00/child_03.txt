import numpy as np
from scipy.optimize import minimize, linprog

def compute_constraints(vars_array):
    """Computes pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2"""
    N = 26
    r = vars_array[:N]
    u = vars_array[N:2*N]
    v = vars_array[2*N:3*N]
    
    # Parameterization automatically satisfies boundary constraints:
    # x = r + (1-2r)*u  =>  r <= x <= 1-r when u in [0,1]
    denom = 1.0 - 2.0 * r
    x = r + denom * u
    y = r + denom * v
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    return d2[i_idx, j_idx] - rs[i_idx, j_idx]**2

def objective(vars_array):
    """Objective: minimize negative sum of radii => Maximize sum of radii"""
    return -np.sum(vars_array[:26])

def get_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_b = 4 * n
    num_p = n * (n - 1) // 2
    A_ub = np.zeros((num_b + num_p, n))
    b_ub = np.zeros(num_b + num_p)
    bounds = []
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lims = (x, 1.0 - x, y, 1.0 - y)
        max_r = min(lims)
        bounds.append((0.0, max(max_r - 1e-7, 1e-7)))
        for lim in lims:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except:
        pass
    return np.full(n, 0.01), 0.0

def force_init(n, rng):
    """Generates a well-spread initial configuration using repulsive forces."""
    pts = rng.uniform(0.2, 0.8, (n, 2))
    temp = 0.1
    for _ in range(800):
        forces = np.zeros_like(pts)
        diffs = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.linalg.norm(diffs, axis=2) + 1e-5
        inv_d2 = 1.0 / dists**2
        np.fill_diagonal(inv_d2, 0)
        forces = np.sum(diffs * inv_d2[:, :, np.newaxis], axis=1)
        
        # Wall repulsion to keep points away from edges initially
        wall_f = np.zeros_like(pts)
        wall_f[:, 0] += np.where(pts[:, 0] < 0.25, 0.2, 0.0) - np.where(pts[:, 0] > 0.75, 0.2, 0.0)
        wall_f[:, 1] += np.where(pts[:, 1] < 0.25, 0.2, 0.0) - np.where(pts[:, 1] > 0.75, 0.2, 0.0)
        
        pts += temp * (forces + wall_f)
        pts = np.clip(pts, 0.05, 0.95)
        temp *= 0.995
    return pts

def hill_climb_centers(init_centers, rng):
    """Local search on centers optimizing LP-sum of radii."""
    centers = init_centers.copy()
    best_r, best_sum = get_lp_radii(centers)
    step = 0.02
    
    for _ in range(1200):
        i = rng.integers(26)
        old_c = centers[i].copy()
        
        best_loc_sum = best_sum
        best_loc_c = old_c.copy()
        
        # Try 4 cardinal directions + 2 random perturbations
        dirs = [[0, 1], [0, -1], [1, 0], [-1, 0]]
        for d in dirs:
            new_c = old_c + step * np.array(d, dtype=float)
            new_c = np.clip(new_c, 0.01, 0.99)
            centers[i] = new_c
            r_c, s_c = get_lp_radii(centers)
            if r_c is not None and s_c > best_loc_sum:
                best_loc_sum = s_c
                best_loc_c = new_c.copy()
                
        for _ in range(2):
            new_c = old_c + rng.uniform(-step, step, 2)
            new_c = np.clip(new_c, 0.01, 0.99)
            centers[i] = new_c
            r_c, s_c = get_lp_radii(centers)
            if r_c is not None and s_c > best_loc_sum:
                best_loc_sum = s_c
                best_loc_c = new_c.copy()
                
        centers[i] = best_loc_c
        if best_loc_sum > best_sum:
            best_sum = best_loc_sum
            best_r = get_lp_radii(centers)[0]
        step *= 0.998
            
    return centers, best_r, best_sum
