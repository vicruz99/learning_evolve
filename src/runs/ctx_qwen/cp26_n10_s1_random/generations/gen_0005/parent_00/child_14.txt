import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_constraints(z):
    """
    Computes pairwise non-overlap constraints.
    Parameterization automatically satisfies boundary constraints:
    x = r + (1-2r)*u, y = r + (1-2r)*v with u,v in [0,1]
    """
    r = z[:N]
    u = z[N:2*N]
    v = z[2*N:3*N]
    
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    return d2[i_idx, j_idx] - rs[i_idx, j_idx]**2

def objective(z):
    """Objective: minimize negative sum of radii => Maximize sum of radii"""
    return -np.sum(z[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds = [(1e-5, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_sum = -np.inf
    best_z = None
    
    # 1. Generate diverse initial configurations
    inits = []
    
    # Hexagonal patterns with various row distributions
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 6, 6, 5, 4], [6, 5, 5, 6, 4], [5, 6, 5, 5, 5]
    ]
    
    for pat in patterns:
        pts = []
        y = 0.09
        row = 0
        while len(pts) < N:
            shift = 0.09 if row % 2 == 1 else 0.0
            x = 0.09 + shift
            for _ in range(pat[row % len(pat)]):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 0.18
            y += 0.155
            row += 1
        inits.append(np.array(pts[:N]))
        
    # Random dense configurations
    for _ in range(12):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # Convert to optimization variable space
    def cfg_to_z(cfg):
        r0 = np.full(N, 0.09)
        denom = 1.0 - 2.0 * r0
        u = np.clip((cfg[:, 0] - r0) / denom, 0.0, 1.0)
        v = np.clip((cfg[:, 1] - r0) / denom, 0.0, 1.0)
        return np.concatenate([r0, u, v])
        
    z_inits = [cfg_to_z(c) for c in inits]
    
    # Phase 1: Multi-start SLSQP optimization
    for z0 in z_inits:
        for _ in range(5):
            zp = z0.copy()
            zp[:N] *= rng.uniform(0.95, 1.05, N)
            zp[N:] += rng.uniform(-0.03, 0.03, 2*N)
            zp = np.clip(zp, [1e-5]*N + [0.0]*(2*N), [0.5]*N + [1.0]*(2*N))
            
            try:
                res = minimize(objective, zp, method='SLSQP', bounds=bounds, 
                               constraints=cons, options={'maxiter': 7000, 'ftol': 1e-14})
                if np.isfinite(res.fun):
                    c_vals = compute_constraints(res.x)
                    if np.min(c_vals) > -1e-4:
                        s = -res.fun
                        if s > best_sum:
                            best_sum = s
                            best_z = res.x.copy()
            except Exception:
                pass
                
    if best_z is None:
        best_z = z_inits[0]
        best_sum = np.sum(best_z[:N])
        
    # Phase 2: Iterative perturbation and refinement around best solution
    for iteration in range(50):
        zp = best_z.copy()
        noise_scale = 0.025 * (0.92 ** iteration)
        zp[:N] *= rng.uniform(1 - noise_scale, 1 + noise_scale, N)
        zp[N:] += rng.uniform(-noise_scale, noise_scale, 2*N)
        zp = np.clip(zp, [1e-5]*N + [0.0]*(2*N), [0.5]*N + [1.0]*(2*N))
        
        try:
            res = minimize(objective, zp, method='SLSQP', bounds=bounds,

                c_vals = compute_constraints(res.x)
                if np.min(c_vals) > -1e-4:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_z = res.x.copy()
        except Exception:
            pass
            
    # Decode best solution to centers and radii
    r_opt = best_z[:N]
    u_opt = best_z[N:2*N]
    v_opt = best_z[2*N:3*N]
    x_opt = r_opt + (1.0 - 2.0 * r_opt) * u_opt
    y_opt = r_opt + (1.0 - 2.0 * r_opt) * v_opt
    centers_opt = np.column_stack((x_opt, y_opt))

    # Extracts exact geometric slack that nonlinear solvers often miss
    c_obj = -np.ones(N)
    A_ub_rows = []
    b_ub_vals = []
    lp_bounds = []
    
    for i in range(N):
        max_r_wall = min(x_opt[i], 1.0 - x_opt[i], y_opt[i], 1.0 - y_opt[i])
        lp_bounds.append((0.0, max(max_r_wall - 1e-8, 1e-8)))

        limits = (x_opt[i], 1.0 - x_opt[i], y_opt[i], 1.0 - y_opt[i])
        for lim in limits:
            row = np.zeros(N)
            row[i] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(lim)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.hypot(x_opt[i] - x_opt[j], y_opt[i] - y_opt[j])
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(dist)
            
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    try:
        lp_res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=lp_bounds, method='highs')
        if lp_res.success:
            r_final = lp_res.x
        else:
            r_final = r_opt.copy()
    except Exception:
        r_final = r_opt.copy()
        
    # Final safety scaling to guarantee strict numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(N):
        x, y = centers_opt[i]
        r = r_final[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(centers_opt[i,0]-centers_opt[j,0], centers_opt[i,1]-centers_opt[j,1])
            rs = r_final[i] + r_final[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    r_final *= scale * 0.999999
    sum_radii = np.sum(r_final)
    
    return centers_opt, r_final, float(sum_radii)
