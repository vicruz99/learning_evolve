This strictly enforces boundary constraints while allowing SLSQP to freely optimize radii and positions. import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = n * 4 + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        lims = (x, 1.0 - x, y, 1.0 - y)
        for lim in lims:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    dists_upper = dists[i_idx, j_idx]
    
    for k in range(len(dists_upper)):
        i, j = i_idx[k], j_idx[k]
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        b_ub[idx] = dists_upper[k]
        idx += 1
        
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def compute_constraints(vars_array):
    """Computes pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2"""
    r = vars_array[:N]
    u = vars_array[N:2*N]
    v = vars_array[2*N:3*N]
    
    # Parameterization automatically satisfies boundary constraints:
    # x = r + (1-2r)*u  =>  r <= x <= 1-r when u in [0,1]
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    return d2[i_idx, j_idx] - rs[i_idx, j_idx]**2

def objective(vars_array):
    """Objective: minimize negative sum of radii => Maximize sum of radii"""
    return -np.sum(vars_array[:N])

def run_packing():
    rng = np.random.default_rng(42)
    bounds_vars = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # --- Initialization: Hexagonal Lattice ---
    pts = []
    r0_init = 0.095
    y = r0_init
    row = 0
    while len(pts) < N + 10:
        shift = r0_init if row % 2 == 1 else 0.0
        x = r0_init + shift
        while x + r0_init <= 1.0:
            pts.append([x, y])
            x += 2.0 * r0_init
        y += np.sqrt(3) * r0_init
        row += 1
    base_centers = np.array(pts[:N])
    
    # Initial LP evaluation
    best_centers = base_centers.copy()
    _, best_sum = solve_radii_lp(best_centers)
    
    # --- Phase 1: SLSQP from diverse starts ---
    starts = []
    # Perturbed hex
    for _ in range(6):
        c = base_centers + rng.uniform(-0.025, 0.025, base_centers.shape)
        starts.append(np.clip(c, 0.05, 0.95))
    # Grid
    gx = np.linspace(0.12, 0.88, 5)
    gy = np.linspace(0.12, 0.88, 5)
    starts.append(np.array([(x, y) for y in gy for x in gx]))
    # Random dense
    for _ in range(4):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    for cfg in starts:
        r0 = np.full(N, 0.085)
        denom = 1.0 - 2.0 * r0
        u = np.clip((cfg[:, 0] - r0) / denom, 0.0, 1.0)
        v = np.clip((cfg[:, 1] - r0) / denom, 0.0, 1.0)
        x0 = np.concatenate([r0, u, v])
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_vars,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_vals = compute_constraints(res.x)
                if np.min(c_vals) > -1e-4:
                    r_s = res.x[:N]
                    u_s = res.x[N:2*N]
                    v_s = res.x[2*N:3*N]
                    cx = r_s + (1.0 - 2.0 * r_s) * u_s
                    cy = r_s + (1.0 - 2.0 * r_s) * v_s
                    centers_decoded = np.column_stack((cx, cy))
                    
                    r_lp, s_lp = solve_radii_lp(centers_decoded)
                    if r_lp is not None and s_lp > best_sum:
                        best_sum = s_lp
                        best_centers = centers_decoded.copy()
        except Exception:
            pass
            
    # --- Phase 2: Hill Climbing on Centers with LP Evaluation ---
    step = 0.04
    for it in range(9000):
        i = rng.integers(N)
        old_pos = best_centers[i].copy()
        best_centers[i] += rng.uniform(-step, step, 2)
        best_centers[i] = np.clip(best_centers[i], 1e-6, 1.0 - 1e-6)
        
        r_lp, s_lp = solve_radii_lp(best_centers)
        if r_lp is not None and s_lp > best_sum:
            best_sum = s_lp
        else:
            best_centers[i] = old_pos
            
        step *= 0.9995
        if step < 0.002:
            step = 0.015 # Restart step size to escape micro-plateaus
            
    # --- Phase 3: Final SLSQP Refinement on Hill-Climb Result ---
    r_lp_cur, _ = solve_radii_lp(best_centers)
    if r_lp_cur is not None:
        r_init = r_lp_cur * 0.96
        cx, cy = best_centers[:, 0], best_centers[:, 1]
        denom = 1.0 - 2.0 * r_init
        u_init = np.clip((cx - r_init) / denom, 0.0, 1.0)
        v_init = np.clip((cy - r_init) / denom, 0.0, 1.0)
        x0 = np.concatenate([r_init, u_init, v_init])
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_vars,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_vals = compute_constraints(res.x)
                if np.min(c_vals) > -1e-4:
                    r_s = res.x[:N]
                    u_s = res.x[N:2*N]
                    v_s = res.x[2*N:3*N]
                    cx = r_s + (1.0 - 2.0 * r_s) * u_s
                    cy = r_s + (1.0 - 2.0 * r_s) * v_s
                    centers_decoded = np.column_stack((cx, cy))
                    r_lp, s_lp = solve_radii_lp(centers_decoded)
                    if r_lp is not None and s_lp > best_sum:
                        best_sum = s_lp
                        best_centers = centers_decoded.copy()
        except Exception:
            pass

    # --- Final LP Extraction & Safety Scaling ---
    r_final, best_sum = solve_radii_lp(best_centers)
    if r_final is None:
        r_final = np.full(N, 0.08)
        best_sum = np.sum(r_final)
        
    # Strict scaling to guarantee numerical validity against checker tolerance
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = r_final[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = r_final[i] + r_final[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    r_final *= scale * 0.999999
    best_sum = float(np.sum(r_final))
    
    return best_centers, r_final, best_sum
