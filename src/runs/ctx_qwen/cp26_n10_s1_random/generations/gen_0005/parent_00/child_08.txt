import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute the constant LP constraint matrix structure
# Constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
#              r_i + r_j <= dist(i, j)
def _setup_lp_matrix():
    n_pairs = N * (N - 1) // 2
    A = np.zeros((4 * N + n_pairs, N))
    
    # Boundary constraints: 1 * r_i <= bound
    for i in range(N):
        A[i, i] = 1.0
        A[N + i, i] = 1.0
        A[2*N + i, i] = 1.0
        A[3*N + i, i] = 1.0
        
    # Pairwise constraints: r_i + r_j <= dist
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            row = 4 * N + idx
            A[row, i] = 1.0
            A[row, j] = 1.0
            idx += 1
    return A

A_LP = _setup_lp_matrix()
C_OBJ = -np.ones(N)  # Minimize negative sum => Maximize sum
LP_BOUNDS = [(0.0, None)] * N

def _solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers. Returns (sum_radii, radii)."""
    b = np.empty(4 * N + N * (N - 1) // 2)
    
    # Boundary distances
    b[:N] = centers[:, 0]
    b[N:2*N] = 1.0 - centers[:, 0]
    b[2*N:3*N] = centers[:, 1]
    b[3*N:4*N] = 1.0 - centers[:, 1]
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b[4*N:] = dists[np.triu_indices(N, k=1)]
    
    res = linprog(C_OBJ, A_ub=A_LP, b_ub=b, bounds=LP_BOUNDS, method='highs')
    if res.success and res.fun < 0:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def _hill_climb(centers, rng, steps=2500):
    """Randomized hill climbing on center coordinates."""
    c = centers.copy()
    best_sum, best_r = _solve_lp(c)
    best_c = c.copy()
    
    step = 0.04
    for _ in range(steps):
        i = rng.integers(N)
        old_pos = c[i].copy()
        
        # Random perturbation
        c[i] += rng.uniform(-step, step, 2)
        c[i] = np.clip(c[i], 1e-4, 1.0 - 1e-4)
        
        curr_sum, _ = _solve_lp(c)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_c = c.copy()
        else:
            c[i] = old_pos
            
        # Decay step size for fine-tuning
        step *= 0.9992
        
    return best_c, best_sum

def _nelder_mead_refine(centers):
    """Derivative-free local optimization on centers."""
    def obj(x_flat):
        return -_solve_lp(x_flat.reshape(N, 2))[0]
        
    res = minimize(
        obj, centers.flatten(), method='Nelder-Mead',
        options={'maxiter': 6000, 'xatol': 1e-8, 'fatol': 1e-10}
    )
    return res.x.reshape(N, 2), -res.fun

def run_packing():
    rng = np.random.default_rng(42)
    inits = []
    
    # 1. Hexagonal lattice patterns (optimal for dense packing)
    for r0 in [0.088, 0.092, 0.096]:
        pts = []
        y = r0
        row = 0
        while len(pts) < N + 10:
            shift = r0 if row % 2 == 1 else 0.0
            x = r0 + shift
            while x + r0 <= 1.0:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        inits.append(np.array(pts[:N]))
        
    # 2. Regular grid + center point
    g = np.linspace(0.14, 0.86, 5)
    pts_grid = np.array([(x, y) for y in g for x in g])
    inits.append(np.vstack([pts_grid, [0.5, 0.5]]))
    
    # 3. Diverse random starts to escape structural biases
    for _ in range(5):
        inits.append(rng.uniform(0.2, 0.8, (N, 2)))
        
    best_centers = None
    best_sum = -1.0
    best_radii = None
    
    # Optimization pipeline: Hill Climbing -> Nelder-Mead
    for cfg in inits:
        hc_c, _ = _hill_climb(cfg, rng, steps=3000)
        nm_c, nm_s = _nelder_mead_refine(hc_c)
        
        if nm_s > best_sum:
            best_sum = nm_s
            best_centers = nm_c.copy()
            _, best_radii = _solve_lp(nm_c)
            
    # Strict safety scaling to guarantee numerical validity against checker tolerance
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
