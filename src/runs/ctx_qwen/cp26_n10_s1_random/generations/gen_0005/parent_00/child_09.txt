import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_pairs = n * (n - 1) // 2
    num_constraints = n + num_pairs
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    bounds = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(lim, 0.0)))
        A_ub[i, i] = 1.0
        b_ub[i] = lim
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    idx = n
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 0.0), 0.0

def slsqp_objective(vars_array, n):
    return -np.sum(vars_array[:n])

def slsqp_constraints(vars_array, n):
    r = vars_array[:n]
    u = vars_array[n:2*n]
    v = vars_array[2*n:3*n]
    
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = (r[:, np.newaxis] + r[np.newaxis, :])**2
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    return d2[i_idx, j_idx] - rs[i_idx, j_idx]

def run_packing():
    n = N
    rng = np.random.default_rng(42)
    
    # 1. Generate diverse initial configurations
    inits = []
    
    # Hexagonal lattices with varying densities
    for density in [0.075, 0.085, 0.095, 0.105]:
        pts = []
        y = density
        row = 0
        while len(pts) < n + 5:
            shift = density if row % 2 == 1 else 0.0
            x = density + shift
            while x + density <= 1.0:
                pts.append([x, y])
                x += 2.0 * density
            y += np.sqrt(3) * density
            row += 1
        inits.append(np.array(pts[:n]))
        
    # Grid patterns
    for s in [5, 6, 7]:
        grid = []
        for i in range(s):
            for j in range(s):
                grid.append([(j + 0.5) / s, (i + 0.5) / s])
        grid = np.array(grid) + rng.uniform(-0.03, 0.03, (len(grid), 2))
        grid = np.clip(grid, 0.05, 0.95)
        inits.append(grid[:n])
        
    # Random starts
    for _ in range(4):
        inits.append(rng.uniform(0.05, 0.95, (n, 2)))
        
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds_slsqp = [(1e-6, 0.5)] * n + [(0.0, 1.0)] * n + [(0.0, 1.0)] * n
    cons = {'type': 'ineq', 'fun': slsqp_constraints, 'args': (n,)}
    
    # 2. SLSQP Phase
    for cfg in inits:
        r0 = np.full(n, 0.09)
        denom = 1.0 - 2.0 * r0
        u0 = np.clip((cfg[:, 0] - r0) / denom, 0.0, 1.0)
        v0 = np.clip((cfg[:, 1] - r0) / denom, 0.0, 1.0)
        x0 = np.concatenate([r0, u0, v0])
        
        try:
            res = minimize(slsqp_objective, x0, args=(n,), method='SLSQP', 
                           bounds=bounds_slsqp, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                r_opt = res.x[:n]
                u_opt = res.x[n:2*n]
                v_opt = res.x[2*n:3*n]
                cx = r_opt + (1.0 - 2.0 * r_opt) * u_opt
                cy = r_opt + (1.0 - 2.0 * r_opt) * v_opt
                centers_slsqp = np.column_stack((cx, cy))
                
                r_lp, s_lp = solve_radii_lp(centers_slsqp)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = centers_slsqp.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass

    # 3. Local Search (Simulated Annealing) on Centers using LP objective
    if best_centers is not None:
        current_centers = best_centers.copy()
        current_sum = best_sum
        stale_count = 0
        
        n_steps = 12000
        step_max = 0.045
        temp_init = 0.025
        
        for k in range(n_steps):
            step = step_max * (0.9985 ** k)
            temp_k = temp_init * (0.9992 ** k)
            
            old_centers = current_centers.copy()
            
            # Move types
            move = rng.choice(['single', 'multi', 'global'])
            if move == 'single':
                i = rng.integers(n)
                current_centers[i] += rng.uniform(-step, step, 2)
            elif move == 'multi':
                idx = rng.choice(n, size=4, replace=False)
                current_centers[idx] += rng.uniform(-step, step, (4, 2))
            else:
                current_centers += rng.uniform(-step * 0.5, step * 0.5, current_centers.shape)

            _, new_sum = solve_radii_lp(current_centers)
            
            delta = new_sum - current_sum
            if delta > 0 or rng.random() < np.exp(delta / max(temp_k, 1e-9)):
                current_sum = new_sum
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = current_centers.copy()
                    best_radii, _ = solve_radii_lp(best_centers)
                    stale_count = 0
                else:
                    stale_count += 1
            else:
                current_centers = old_centers.copy()
                stale_count += 1
                
            # Micro-restart if stuck to escape shallow local minima
            if stale_count > 600:
                current_centers += rng.uniform(-0.015, 0.015, current_centers.shape)
                current_centers = np.clip(current_centers, 0.05, 0.95)
                current_sum = best_sum
                stale_count = 0

    # 4. Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
