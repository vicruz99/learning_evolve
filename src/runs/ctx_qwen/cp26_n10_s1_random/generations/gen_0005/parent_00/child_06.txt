import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    bounds = []
    
    # Boundary constraints and upper bounds
    for i in range(n):
        x, y = centers[i]
        lims = [x, 1.0 - x, y, 1.0 - y]
        max_r = min(lims)
        bounds.append((0.0, max(max_r, 1e-9)))
        
        for lim in lims:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(lim)
            
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def hill_climb_opt(centers, rng, n_iter=2500):
    """Stochastic hill-climbing on center coordinates using LP objective."""
    n = centers.shape[0]
    _, best_sum = solve_lp_radii(centers)
    best_centers = centers.copy()
    
    base_step = 0.045
    for it in range(n_iter):
        # Exponential decay for fine-tuning
        current_step = base_step * (0.997 ** it)
        
        i = rng.integers(n)
        old_pos = centers[i].copy()
        
        best_local_sum = best_sum
        best_local_pos = old_pos
        
        # Try multiple random moves to find a better one
        for _ in range(3):
            new_pos = old_pos + rng.uniform(-current_step, current_step, 2)
            # Keep strictly inside to ensure LP feasibility
            new_pos = np.clip(new_pos, 1e-3, 1.0 - 1e-3)
            centers[i] = new_pos
            
            _, s_test = solve_lp_radii(centers)
            if s_test > best_local_sum:
                best_local_sum = s_test
                best_local_pos = new_pos.copy()
                
        centers[i] = best_local_pos
        if best_local_sum > best_sum:
            best_sum = best_local_sum
            best_centers = centers.copy()
            
    return best_centers, best_sum

def objective_slqp(vars_arr, n):
    r = vars_arr[:n]
    return -np.sum(r)

def constraints_slqp(vars_arr, n):
    r = vars_arr[:n]
    x = vars_arr[n:2*n]
    y = vars_arr[2*n:3*n]
    
    c = []
    # Boundary constraints
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise non-overlap constraints (squared)
    xi = x[:, None]
    yi = y[:, None]
    ri = r[:, None]
    dx = xi - xi.T
    dy = yi - yi.T
    d2 = dx**2 + dy**2
    rs = ri + ri.T
    idx = np.triu_indices(n, k=1)
    c.append(d2[idx] - rs[idx]**2)
    
    return np.concatenate(c)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattices with varying densities
    for spacing in [0.17, 0.18, 0.19, 0.20, 0.22]:
        lattice = []
        row = 0
        cy = 0.0
        while cy < 1.0:
            shift = spacing / 2.0 if row % 2 == 1 else 0.0
            cx = 0.0
            while cx < 1.0:
                lattice.append([cx + shift, cy])
                cx += spacing
            cy += spacing * np.sqrt(3) / 2.0
            row += 1
            
        lattice = np.array(lattice)
        # Pick points closest to center to minimize edge effects
        dists = np.hypot(lattice[:, 0] - 0.5, lattice[:, 1] - 0.5)
        order = np.argsort(dists)
        inits.append(lattice[order[:n]])
        
    # 2. Regular grids
    for k in range(4, 7):
        pts = np.linspace(0.12, 0.88, k)
        grid = np.array([[x, y] for x in pts for y in pts])
        grid = grid[np.argsort(np.hypot(grid[:, 0] - 0.5, grid[:, 1] - 0.5))][:n]
        inits.append(grid)
        
    # 3. Random dense starts
    for _ in range(5):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = -np.inf
    best_centers = None
    
    bounds_vars = [(1e-6, 0.5)] * n + [(0.0, 1.0)] * n + [(0.0, 1.0)] * n
    cons_dict = {'type': 'ineq', 'fun': constraints_slqp, 'args': (n,)}
    
    for cfg in inits:
        # Phase 1: SLSQP to smooth out overlaps and find local structure
        r0 = np.full(n, 0.09)
        x0 = np.concatenate([r0, cfg[:, 0], cfg[:, 1]])
        
        try:
            res = minimize(objective_slqp, x0, args=(n,), method='SLSQP',
                           bounds=bounds_vars, constraints=cons_dict,
                           options={'maxiter': 3000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                centers_slsqp = np.column_stack((res.x[n:2*n], res.x[2*n:3*n]))
                r_lp, s_lp = solve_lp_radii(centers_slsqp)
                if s_lp > best_sum:

                    best_centers = centers_slsqp.copy()
        except Exception:
            pass

            best_centers = centers_hc.copy()

        # Fine hill-climb
        centers_fine, s_fine = hill_climb_opt(best_centers, rng, n_iter=3000)
        if s_fine > best_sum:
            best_sum = s_fine
            best_centers = centers_fine.copy()
            
        # Final SLSQP polish
        r_init = np.full(n, 0.09)
        x0 = np.concatenate([r_init, best_centers[:, 0], best_centers[:, 1]])

                r_lp_final, s_lp_final = solve_lp_radii(centers_final)
                if s_lp_final > best_sum:
                    best_sum = s_lp_final
                    best_centers = centers_final.copy()
        except Exception:
            pass

    # Fallback if optimization unexpectedly fails
    if best_centers is None:
        best_centers = inits[0]
        
    # Extract final optimal radii via LP
    final_radii, final_sum = solve_lp_radii(best_centers)
    
    # Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y = best_centers[i]
        r = final_radii[i]
        if r > 1e-9:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)

        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = final_radii[i] + final_radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    final_radii *= scale * 0.999999
    final_sum = float(np.sum(final_radii))
    
    return best_centers, final_radii, final_sum
