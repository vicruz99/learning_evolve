import numpy as np
from scipy.optimize import linprog

def solve_radii_lp(centers):
    """
    Given fixed centers, solves the LP to maximize sum of radii.
    Constraints: r_i + r_j <= dist(i,j) and r_i <= distance to walls.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    bounds = []
    
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(lim - 1e-9, 1e-9)))
        
        # Boundary constraints: r_i <= x, 1-x, y, 1-y
        for val in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(val)
            
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def hill_climb(init_centers, n, rng, max_iter=2500):
    """
    Local search on center coordinates to maximize sum of radii.
    Uses LP to evaluate the objective at each step.
    """
    centers = init_centers.copy()
    best_centers = centers.copy()
    best_radii, best_sum = solve_radii_lp(centers)
    step = 0.015
    
    for _ in range(max_iter):
        i = rng.integers(n)
        old_pos = centers[i].copy()
        centers[i] += rng.uniform(-step, step, 2)
        centers[i] = np.clip(centers[i], 1e-5, 1.0 - 1e-5)
        
        radii, curr_sum = solve_radii_lp(centers)
        if curr_sum > best_sum + 1e-9:
            best_sum = curr_sum
            best_centers = centers.copy()
            best_radii = radii
            # Slightly increase step on success to explore promising regions
            step = min(step * 1.02, 0.02)
        else:
            centers[i] = old_pos
            # Decay step size for fine-tuning
            step *= 0.998
            
    return best_centers, best_radii, best_sum
