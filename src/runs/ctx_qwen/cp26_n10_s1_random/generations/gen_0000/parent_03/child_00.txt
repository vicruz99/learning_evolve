import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses SLSQP optimization with multiple random restarts to find a good local optimum.
    """
    n = 26
    
    # Helper to generate an initial hexagonal packing configuration
    def get_initial_config(seed=0):
        rng = np.random.RandomState(seed)
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        
        # Start with a small radius that definitely fits
        # Radius 0.08 fits 26 circles in hexagonal packing easily
        # Diameter 0.16. Row width 1 fits 6 circles? 6*0.16 = 0.96. Yes.
        # But let's stick to a safe 0.05 to avoid any boundary issues initially.
        r_init = 0.05
        
        # Generate hexagonal grid points
        # Spacing
        dx = 2 * r_init
        dy = math.sqrt(3) * r_init
        
        row = 0
        col = 0
        idx = 0
        
        # Fill grid
        while idx < n:
            # Determine x offset for the row
            # Even rows (0, 2, ...) start at x=r_init
            # Odd rows (1, 3, ...) start at x=3r_init (shifted by dx)
            # Actually, standard hex packing:
            # Row 0: r, 3r, 5r...
            # Row 1: 2r, 4r, 6r... (shifted by r)
            
            x_start = r_init
            if row % 2 == 1:
                x_start = 3 * r_init # Shift by one diameter? 
                # Wait, distance between (r, 0) and (2r, sqrt(3)r) is 2r.
                # So x shift is r.
                # If row 0 is at x = r, 3r, 5r...
                # Row 1 should be at x = 2r, 4r, 6r...
                # But 2r = r + r. So shift is r.
                # However, if we use dx = 2r, then row 0 indices are 0, 2, 4 (indices of 2r steps)
                # Let's just use coordinates directly.
                pass 
            
            # Let's simplify: just place points on a hex lattice
            y = r_init + row * dy
            
            if y + r_init > 1.0:
                # If row doesn't fit vertically, wrap or break?
                # Better to just reset or stop. But we need 26.
                # Let's assume r_init is small enough.
                pass
            
            x = r_init
            if row % 2 == 1:
                x = 3 * r_init # Shifted row
            
            # Add circles in this row
            while x + r_init <= 1.0 and idx < n:
                centers[idx] = [x, y]
                radii[idx] = r_init
                idx += 1
                x += 2 * r_init
            
            row += 1
            
        # Add some random perturbation to break symmetry and help optimization
        centers += rng.uniform(-0.01, 0.01, size=centers.shape)
        radii += rng.uniform(-0.001, 0.001, size=radii.shape)
        
        # Ensure valid bounds for initial guess
        centers = np.clip(centers, 1e-5, 1 - 1e-5)
        radii = np.clip(radii, 1e-5, 0.4)
        
        return centers, radii

    def objective(z):
        # z = [x1, y1, r1, ..., x26, y26, r26]
        # But we grouped them as centers then radii in logic above?
        # Let's flatten centers and radii.
        # z[:n*2] are centers, z[n*2:] are radii.
        r = z[n*2:]
        return -np.sum(r)

    def inequality_constraints(z):
        # Returns a list of constraint functions/values
        # We need:
        # 1. x_i - r_i >= 0
        # 2. 1 - x_i - r_i >= 0
        # 3. y_i - r_i >= 0
        # 4. 1 - y_i - r_i >= 0
        # 5. dist(i, j) - (r_i + r_j) >= 0
        
        centers = z[:n*2].reshape(n, 2)
        r = z[n*2:]
        
        cons = []
        
        # Boundary constraints
        # x >= r
        cons.append(centers[:, 0] - r)
        # 1 - x >= r => 1 - x - r >= 0
        cons.append(1.0 - centers[:, 0] - r)
        # y >= r
        cons.append(centers[:, 1] - r)
        # 1 - y >= r
        cons.append(1.0 - centers[:, 1] - r)
        
        # Overlap constraints
        # We need to compute distances for all pairs
        # This can be large, but n=26 is small (325 pairs)
        # Vectorized calculation
        
        # diff shape (n, n, 2)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # dists[i, j] - (r[i] + r[j]) >= 0
        # We only need upper triangle i < j
        # Create a mask
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        active_dists = dists[mask]
        active_sum_r = r[np.where(mask)[0]] + r[np.where(mask)[1]]
        
        cons.append(active_dists - active_sum_r)
        
        # Flatten all constraints into a single array for scipy?
        # Or return list of dicts?
        # scipy.optimize.minimize accepts a list of constraint dicts.
        # But passing a single function returning a 1D array is often cleaner for 'ineq'
        # However, constraints must be defined individually or as a block.
        # If we return a 1D array, it treats each element as a constraint >= 0.
        
        all_cons = np.concatenate(cons)
        return all_cons

    def constraint_func(z):
        return inequality_constraints(z)

    # Setup bounds
    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
    for i in range(n):
        bounds.append((0, 0.5)) # r

    best_z = None
    best_val = -np.inf
    
    # Try multiple random seeds
    for seed in range(5):
        centers_init, radii_init = get_initial_config(seed)
        z0 = np.concatenate([centers_init.flatten(), radii_init])
        
        # Constraints definition for SLSQP
        cons = {'type': 'ineq', 'fun': constraint_func}
        
        try:
            res = minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success or res.fun < best_val: # res.fun is negative sum, so smaller is better
                 # Actually we want to maximize sum, so minimize -sum.
                 # best_val should track the minimum of objective (most negative).
                 # But let's check valid result.
                 if res.fun < best_val:
                     best_val = res.fun
                     best_z = res.x
        except Exception as e:
            continue

    if best_z is None:
        # Fallback to initial guess of last run
        centers_init, radii_init = get_initial_config(0)
        best_z = np.concatenate([centers_init.flatten(), radii_init])

    # Extract result
    centers = best_z[:n*2].reshape(n, 2)
    radii = best_z[n*2:]
    
    # Clip any tiny negative values due to numerical errors
    radii = np.maximum(radii, 0.0)
    centers = np.clip(centers, 0.0, 1.0)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
