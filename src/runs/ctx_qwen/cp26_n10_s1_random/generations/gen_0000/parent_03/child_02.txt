import numpy as np
from scipy.optimize import minimize

def compute_cost(vars, n, penalty_weight):
    """
    Computes the objective function: negative sum of radii + penalty for constraint violations.
    """
    centers = vars[:n*2].reshape(n, 2)
    radii = vars[n*2:]
    
    # We minimize the negative sum of radii (equivalent to maximizing sum)
    cost = -np.sum(radii)
    penalty = 0.0
    
    # Boundary constraints penalties
    # x - r >= 0  => r - x <= 0
    # x + r <= 1  => x + r - 1 <= 0
    # y - r >= 0  => r - y <= 0
    # y + r <= 1  => y + r - 1 <= 0
    violations = np.maximum(0.0, np.concatenate([
        radii - centers[:, 0],
        centers[:, 0] + radii - 1.0,
        radii - centers[:, 1],
        centers[:, 1] + radii - 1.0
    ]))
    penalty += np.sum(violations**2)
    
    # Pairwise non-overlap constraints
    # dist(i, j) >= r_i + r_j  =>  r_i + r_j - dist(i,j) <= 0
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    
    overlap = np.maximum(0.0, radii[:, np.newaxis] + radii[np.newaxis, :] - dist)
    penalty += np.sum(overlap**2)
    
    return cost + penalty_weight * penalty

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initial grid layout: 5x5 + 1 center circle
    init_xs = []
    init_ys = []
    for i in range(5):
        for j in range(5):
            init_xs.append(0.1 + i * 0.2)
            init_ys.append(0.1 + j * 0.2)
    init_xs.append(0.5)
    init_ys.append(0.5)
    
    init_xs = np.array(init_xs[:n])
    init_ys = np.array(init_ys[:n])
    
    centers = np.column_stack([init_xs, init_ys])
    # Add small perturbation to break symmetry and avoid flat regions
    centers += np.random.uniform(-0.02, 0.02, size=centers.shape)
    radii = np.full(n, 0.08)
    
    # Flatten to 1D optimization vector
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: centers in [0,1], radii in [1e-6, 0.5]
    bounds = [(0, 1)] * (2*n) + [(1e-6, 0.5)] * n
    
    current_x = x0
    # Continuation schedule: gradually increase penalty weight to tighten constraints
    weights = [50, 500, 5000, 50000, 200000]
    
    for w in weights:
        res = minimize(compute_cost, current_x, args=(n, w), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-12})
        current_x = res.x
        
    centers_final = current_x[:2*n].reshape(n, 2)
    radii_final = current_x[2*n:]
    
    # Final validity enforcement: scale down radii if any numerical overlap persists
    min_valid_scale = 1.0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers_final[i] - centers_final[j])
            req = radii_final[i] + radii_final[j]
            if req > d:
                min_valid_scale = min(min_valid_scale, d / req)
                
    if min_valid_scale < 1.0:
        radii_final *= max(0.9999999, min_valid_scale)
        
    return centers_final, radii_final, np.sum(radii_final)
