import numpy as np
from scipy.optimize import minimize


def shrink_radii(centers, radii, n, iterations=50):
    """Shrink radii until no constraint violations"""
    for _ in range(iterations):
        max_viol = 0.0
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            max_viol = max(max_viol, max(0, r - x), max(0, x + r - 1), max(0, r - y), max(0, y + r - 1))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                max_viol = max(max_viol, max(0, radii[i] + radii[j] - d))
        if max_viol < 1e-12:
            break
        scale = 1.0 - max_viol / (0.05 + max_viol) * 0.02
        radii *= scale
    return radii


def objective_func(params, n, pw):
    """Objective: minimize negative sum of radii plus penalty for violations"""
    centers = params[:2 * n].reshape((n, 2))
    radii = np.maximum(params[2 * n:], 1e-8)

    obj = -np.sum(radii)

    # Boundary penalty

        obj += pw * (max(0, r - x) ** 2 + max(0, x + r - 1) ** 2 +

            gap = radii[i] + radii[j] - d

def hex_init(n, r, row_counts):
    """Initialize circles in hexagonal pattern"""
    centers = np.zeros((n, 2))
    radii = np.full(n, r)
    idx = 0

        y = r + row * np.sqrt(3) * r
        for col in range(count):
            x = r + col * 2 * r + (row % 2) * r
            centers[idx] = [x, y]
            idx += 1
    return centers, radii


def grid_init(n, r):
    """Initialize circles in a grid pattern"""
    centers = np.zeros((n, 2))
    radii = np.full(n, r)
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    # Adjust to fit in unit square
    spacing_x = (1 - 2 * r) / max(cols - 1, 1)
    spacing_y = (1 - 2 * r) / max(rows - 1, 1)
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = r + col * spacing_x
            y = r + row * spacing_y
            centers[idx] = [x, y]
            idx += 1
    return centers, radii
