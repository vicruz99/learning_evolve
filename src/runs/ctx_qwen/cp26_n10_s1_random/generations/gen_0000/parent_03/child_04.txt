def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def compute_penalty(centers, r):
    """
    Computes the penalty function for a given set of centers and fixed radius r.
    Penalty is sum of squared overlaps and boundary violations.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary violations
    # x >= r => r - x <= 0
    # x <= 1 - r => x + r - 1 <= 0
    # y >= r => r - y <= 0
    # y <= 1 - r => y + r - 1 <= 0
    
    # Vectorized boundary check
    # left wall
    violation_x_min = np.maximum(0, r - centers[:, 0])
    # right wall
    violation_x_max = np.maximum(0, centers[:, 0] + r - 1)
    # bottom wall
    violation_y_min = np.maximum(0, r - centers[:, 1])
    # top wall
    violation_y_max = np.maximum(0, centers[:, 1] + r - 1)
    
    boundary_pen = (violation_x_min**2).sum() + (violation_x_max**2).sum() + \
                   (violation_y_min**2).sum() + (violation_y_max**2).sum()
    
    # Pairwise overlaps
    # dist^2 < (2r)^2 => overlap
    # overlap amount = 2r - dist
    # squared overlap = (2r - dist)^2
    
    # Compute pairwise distances
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # We only need upper triangle
    # mask for i < j
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pair_dists = dists[mask]
    
    # Overlap: if dist < 2r
    # safe_dist = 2r
    # penalty = sum(max(0, 2r - dist)^2)
    
    overlaps = np.maximum(0, 2 * r - pair_dists)
    overlap_pen = np.sum(overlaps**2)
    
    return boundary_pen + overlap_pen

    def objective(centers_flat):
        centers = centers_flat.reshape(-1, 2)
        return compute_penalty(centers, r)
    return objective
