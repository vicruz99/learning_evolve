from scipy.optimize import minimize

def compute_objective(v, n, penalty_weight):
    """
    Computes the objective function: -sum(radii) + penalty for constraint violations.
    """

    radii = v[2*n:]

    obj = -np.sum(radii)

    # Compute pairwise distances efficiently using broadcasting
    c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(c_diff**2, axis=2))

    overlaps = np.maximum(0, overlaps)
    obj += penalty_weight * np.sum(overlaps**2)
    
    # Boundary penalties
    # Ensure circles stay within [0, 1] x [0, 1]
    obj += penalty_weight * np.sum(np.maximum(0, radii - centers[:, 0])**2)       # Left boundary
    obj += penalty_weight * np.sum(np.maximum(0, centers[:, 0] + radii - 1)**2)   # Right boundary
    obj += penalty_weight * np.sum(np.maximum(0, radii - centers[:, 1])**2)       # Bottom boundary
    obj += penalty_weight * np.sum(np.maximum(0, centers[:, 1] + radii - 1)**2)   # Top boundary
    
    return obj

def run_packing():
    n = 26

    y_start = 0.1

    for row in range(6):

        x_offset = 0.0 if row % 2 == 0 else 0.09

            if idx >= n:
                break
            centers[idx, 0] = 0.1 + col * col_width + x_offset
            centers[idx, 1] = y_start + row * row_height
            idx += 1
            
    x0 = np.concatenate([centers.flatten(), radii])

    best_obj = float('inf')
    
    # Optimization with penalty annealing and restarts
    np.random.seed(42)
    for restart in range(5):
        # Increase penalty weight to enforce constraints more strictly
        penalty = 100.0 * (10 ** restart)

        current_v = best_v + np.random.normal(0, 0.002 * (0.85 ** restart), size=best_v.shape)

    radii_final = np.maximum(radii_final, 1e-9)
    centers_final[:, 0] = np.clip(centers_final[:, 0], radii_final, 1 - radii_final)
    centers_final[:, 1] = np.clip(centers_final[:, 1], radii_final, 1 - radii_final)
    
    return centers_final, radii_final, np.sum(radii_final)
