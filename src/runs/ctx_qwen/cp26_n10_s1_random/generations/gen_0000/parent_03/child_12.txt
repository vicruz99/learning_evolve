def compute_penalty_and_grad(params, n):
    """
    Computes the penalty function value and its gradient.
    Penalty = Sum of squared overlaps + Sum of squared boundary violations.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    penalty = 0.0
    grad_centers = np.zeros_like(centers)
    grad_radii = np.zeros_like(radii)
    
    # 1. Inter-circle constraints: dist(i, j) >= r_i + r_j
    # Violation: max(0, r_i + r_j - dist)
    # Penalty term: violation^2
    
    # We iterate over all pairs
    for i in range(n):

            diff = centers[i] - centers[j]
            dist_sq = np.sum(diff**2)
            dist = np.sqrt(dist_sq + 1e-12) # Avoid division by zero
            
            r_sum = radii[i] + radii[j]
            
            # Check if overlapping
            if r_sum > dist:
                violation = r_sum - dist
                penalty += violation**2
                
                # Gradient w.r.t centers
                # d(violation)/d(x_i) = -1 * d(dist)/d(x_i) = - (x_i - x_j)/dist
                # d(violation^2)/d(x_i) = 2 * violation * d(violation)/d(x_i)

                grad_centers[j] -= factor * diff
                
                # Gradient w.r.t radii

                grad_radii[i] += 2.0 * violation
                grad_radii[j] += 2.0 * violation

            grad_radii[i] += 2.0 * viol_bottom

        if viol_top > 0:
            penalty += viol_top**2
            grad_centers[i, 1] += 2.0 * viol_top
            grad_radii[i] += 2.0 * viol_top

    return penalty, grad_centers, grad_radii

def objective_and_grad(params, n, penalty_weight):

    Objective function: -sum(radii) + penalty_weight * Penalty
    Gradient: derivative of objective.

    radii = params[2*n:]
    obj = -np.sum(radii) + penalty_weight * penalty

    # Add penalty gradient contribution
    grad_params += penalty_weight * np.concatenate([grad_centers.flatten(), grad_radii])
    
    return obj, grad_params
