def compute_overlap_violations(centers, radii):
    """Compute total violation of non-overlap and boundary constraints."""
    n = len(radii)
    violation = 0.0
    
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            violation += (x - r) ** 2
        if x + r > 1:
            violation += (x + r - 1) ** 2
        if y - r < 0:
            violation += (y - r) ** 2
        if y + r > 1:
            violation += (y + r - 1) ** 2
    
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):

            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                violation += (min_dist - dist) ** 2
    
    return violation
