import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def hexagonal_grid_init(n_circles):
    # Determine grid dimensions
    # A 5x5 grid is 25 circles. 26 needs 1 more.
    # We'll use a slightly staggered approach to fit 26.
    # 5 rows. 5, 5, 5, 5, 6 circles? Or 6, 5, 5, 5, 5?
    # Let's try to fit 26 in a dense pattern.
    # Standard hexagonal packing: spacing dx = sqrt(3)*r, dy = 1.5r? No.
    # Horizontal spacing 2r, vertical sqrt(3)r.
    # Let's just generate a random dense packing that respects constraints roughly.
    
    centers = np.zeros((n_circles, 2))
    radii = np.ones(n_circles) * 0.1 # Initial guess
    
    # 5x5 grid for 25, place 26th in center if possible, or perturb
    # 5x5 grid centers
    x_coords = np.linspace(0.1, 0.9, 5)
    y_coords = np.linspace(0.1, 0.9, 5)
    
    idx = 0
    for x in x_coords:
        for y in y_coords:
            if idx < n_circles:
                centers[idx] = [x, y]
                idx += 1
    
    if idx < n_circles:
        # Place remaining in gaps
        # For 26th, place at (0.5, 0.5) and shrink others slightly?
        # Or just random perturbation
        for i in range(idx, n_circles):
            centers[i] = [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]
            
    return centers, radii

def get_constraint_violations(centers, radii):
    n = len(radii)
    violations = 0.0
    penalty_weight = 1000.0
    
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0: violations += penalty_weight * (r - x)**2
        if x + r > 1: violations += penalty_weight * (x + r - 1)**2
        if y - r < 0: violations += penalty_weight * (r - y)**2
        if y + r > 1: violations += penalty_weight * (y + r - 1)**2
        
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                violations += penalty_weight * (min_dist - dist)**2
                
    return violations

def objective_function(vars, n):
    # vars = [x1, y1, r1, x2, y2, r2, ...]
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Maximize sum of radii -> Minimize -sum
    obj = -np.sum(radii)
    
    # Add soft constraints
    obj += get_constraint_violations(centers, radii)
    
    return obj

def run_packing():
    n = 26
    np.random.seed(42)
    
    # 1. Initial Configuration
    centers, radii = hexagonal_grid_init(n)
    
    # Combine into vector
    x0 = np.concatenate([centers.flatten(), radii])
    
    # 2. Simulated Annealing (Simplified Local Search)
    current_x = x0
    current_score = objective_function(current_x, n)
    
    temperature = 1.0
    temp_decay = 0.99
    max_iter = 2000
    
    for i in range(max_iter):
        # Propose new solution
        # Perturb centers slightly
        perturbation = np.random.normal(0, 0.02, size=x0.shape)
        proposed_x = current_x + perturbation
        
        # Ensure radii are non-negative
        proposed_x[2*n:] = np.maximum(proposed_x[2*n:], 0)
        
        proposed_score = objective_function(proposed_x, n)
        
        # Accept if better or with probability
        if proposed_score < current_score:
            current_x = proposed_x
            current_score = proposed_score
        else:
            if temperature > 1e-5:
                prob = np.exp(-(proposed_score - current_score) / temperature)
                if np.random.rand() < prob:
                    current_x = proposed_x
                    current_score = proposed_score
        
        temperature *= temp_decay

    # 3. Refined Constrained Optimization
    # Extract best guess
    best_centers = current_x[:2*n].reshape(n, 2)
    best_radii = current_x[2*n:]
    
    # Define constraints for SLSQP
    cons = []
    bounds = []
    
    # Bounds for each circle
    for i in range(n):
        # x in [0, 1], y in [0, 1], r >= 0
        # Actually, x >= r, x <= 1-r, etc.
        # We can let solver handle r >= 0, and use inequality constraints for others.
        # But simpler: bounds on x, y are [0,1], r is [0, 0.5]
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r
        
    # Inequality constraints: g(x) >= 0
    # Boundary: x - r >= 0 -> x - r >= 0
    #           x + r <= 1 -> 1 - x - r >= 0
    #           y - r >= 0
    #           y + r <= 1
    
    for i in range(n):
        idx_x = i * 3
        idx_y = i * 3 + 1
        idx_r = i * 3 + 2 + 2*n # r is at end of vector
        
        # x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i, idx_x=idx_x, idx_r=idx_r: v[idx_x] - v[idx_r]})
        # 1 - x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i, idx_x=idx_x, idx_r=idx_r: 1.0 - v[idx_x] - v[idx_r]})
        # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i, idx_y=idx_y, idx_r=idx_r: v[idx_y] - v[idx_r]})
        # 1 - y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i, idx_y=idx_y, idx_r=idx_r: 1.0 - v[idx_y] - v[idx_r]})

    # Pairwise distance constraints: dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi = i * 3
            idx_yi = i * 3 + 1
            idx_ri = i * 3 + 2 + 2*n
            
            idx_xj = j * 3
            idx_yj = j * 3 + 1
            idx_rj = j * 3 + 2 + 2*n
            
            def distance_constraint(v, i=i, j=j, 
                                     idx_xi=idx_xi, idx_yi=idx_yi, idx_ri=idx_ri,
                                     idx_xj=idx_xj, idx_yj=idx_yj, idx_rj=idx_rj):
                xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r_sq = (ri + rj)**2
                return dist_sq - sum_r_sq

            cons.append({'type': 'ineq', 'fun': distance_constraint})

    # Objective function for SLSQP (Minimize -sum(radii))
    def objective_slsqp(v):
        radii_vec = v[2*n:]
        return -np.sum(radii_vec)

    # Run optimization
    try:
        result = minimize(objective_slsqp, current_x, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 1000, 'ftol': 1e-9})
        
        if result.success:
            final_centers = result.x[:2*n].reshape(n, 2)
            final_radii = result.x[2*n:]
        else:
            # Fallback to SA result if SLSQP fails
            final_centers = best_centers
            final_radii = best_radii
            
    except Exception:
        final_centers = best_centers
        final_radii = best_radii

    # Final validation check (optional, for safety)
    # If invalid, try to shrink radii slightly to fix overlaps (simple fix)
    # But we want to maximize, so we assume the optimizer did its job.
    # Let's enforce non-negativity strictly
    final_radii = np.maximum(final_radii, 0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
