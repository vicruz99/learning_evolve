from scipy.optimize import minimize


def hexagonal_init(n, config, r_init, seed=0):
    """Initialize circles in hexagonal pattern."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r_init
    idx = 0
    for row, size in enumerate(config):
        if idx >= n:
            break
        y = r_init + row * np.sqrt(3) * r_init
        for col in range(size):
            if row % 2 == 0:
                x = r_init + col * 2 * r_init
            else:
                x = r_init + r_init + col * 2 * r_init
            centers[idx] = [x + rng.uniform(-0.005, 0.005), y + rng.uniform(-0.005, 0.005)]
            idx += 1
    # Scale to fit in unit square
    max_extent = np.max(centers + radii[:, None], axis=0)
    scale = 1.0 / np.max(max_extent)
    centers *= scale
    radii *= scale
    return centers, radii
