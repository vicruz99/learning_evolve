import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    num_vars = n * 3  # x, y, r for each circle
    
    # 1. Initialization: Hexagonal Grid
    # Generate points in a hexagonal pattern to ensure high initial density
    centers_init = []
    r_est = 0.1  # Estimated radius for spacing
    dy = r_est * np.sqrt(3)
    dx = 2 * r_est
    
    # Generate enough points to cover the square, then pick first n
    # or arrange them nicely. 
    # A simple way is to iterate rows and columns.
    row = 0
    while len(centers_init) < n:
        y = r_est + row * dy
        if y + r_est > 1.0:
            # If row is too high, shift it down slightly or stop?
            # Actually, we just want points inside [0,1]x[0,1] roughly.
            # If we can't fit more rows, we might run out of space, 
            # but with 26 circles, 5-6 rows fit easily.
            pass 
        
        col = 0
        x_offset = 0.0 if (row % 2) == 0 else r_est
        
        while len(centers_init) < n:
            x = x_offset + r_est + col * dx
            if x + r_est <= 1.0:
                centers_init.append([x, y])
            col += 1
            if x + r_est > 1.0 and col > 10: # Safety break
                break
        row += 1
        if row > 20: # Safety break
            break
            
    # Convert to numpy array
    centers_init = np.array(centers_init[:n])
    radii_init = np.full(n, 0.01) # Start with small radii
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(num_vars)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]
        
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([
            (0.0, 1.0), # x
            (0.0, 1.0), # y
            (0.0, 0.5)  # r
        ])

    # 2. Objective Function
    def objective(vars):
        # Unpack
        centers = vars[:2*n].reshape(n, 2)
        radii = vars[2*n:]
        
        # Negative sum of radii (to maximize sum)
        score = -np.sum(radii)
        
        penalty = 0.0
        lambda_pen = 1000.0 # Penalty weight
        
        # Overlap constraints: dist(i, j) >= r_i + r_j
        # Iterate over unique pairs
        for i in range(n):
            xi, yi = centers[i]
            ri = radii[i]
            
            # Boundary constraints
            # r <= x, r <= 1-x, r <= y, r <= 1-y
            if ri > xi:
                penalty += lambda_pen * (ri - xi)**2
            if ri > 1 - xi:
                penalty += lambda_pen * (ri - (1 - xi))**2
            if ri > yi:
                penalty += lambda_pen * (ri - yi)**2
            if ri > 1 - yi:
                penalty += lambda_pen * (ri - (1 - yi))**2
                
            # Overlap with subsequent circles
            for j in range(i + 1, n):
                xj, yj = centers[j]
                rj = radii[j]
                
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                r_sum = ri + rj
                
                if r_sum > dist:
                    overlap = r_sum - dist
                    penalty += lambda_pen * overlap**2
                    
        return score + penalty

    # 3. Optimization
    # Using L-BFGS-B for bound constraints
    # Options to limit iterations/time if needed, but default is usually fine for this size
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 1000})
    
    # Extract solution
    best_centers = res.x[:2*n].reshape(n, 2)
    best_radii = res.x[2*n:]
    
    # Post-processing: Ensure non-negative radii and clip tiny violations if any
    # The optimizer should handle bounds, but just in case
    best_radii = np.maximum(best_radii, 0.0)
    
    sum_radii = np.sum(best_radii)
    
    return best_centers, best_radii, sum_radii
