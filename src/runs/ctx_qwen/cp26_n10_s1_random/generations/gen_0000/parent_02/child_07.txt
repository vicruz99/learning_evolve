import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n) # Maximize sum(r) => Minimize -sum(r)
    
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1
        # We add 4 rows for the 4 boundaries
        A_ub.extend([row, row, row, row])
        b_ub.extend([x, 1 - x, y, 1 - y])
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.fun * -1, res.x
        else:
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

            x_start = 0.15
        else:
            x_start = 0.05
            
        for col in range(7):
            if idx >= n: break

            if 0 <= x <= 1 and 0 <= y <= 1:
                centers[idx] = [x, y]
                idx += 1
        if idx >= n: break

    # Fallback if grid didn't fill

        idx += 1

    best_centers = centers.copy()
    best_sum, best_radii = solve_radii(centers)
    current_sum = best_sum
    
    # Coordinate-wise optimization with Simulated Annealing
    temperature = 0.05
    cooling_rate = 0.995
    
    for iteration in range(300):
        # Shuffle circle order for diversity
        indices = np.random.permutation(n)
        
        for i in indices:
            x, y = centers[i]
            # Test 4 directions

            best_move_sum = current_sum
            best_move_center = centers[i].copy()
            move_made = False
            
            for dx, dy in directions:
                new_x, new_y = x + dx, y + dy
                # Boundary check
                if not (0 <= new_x <= 1 and 0 <= new_y <= 1):
                    continue
                
                centers[i] = [new_x, new_y]
                move_sum, _ = solve_radii(centers)
                
                # Simulated Annealing acceptance
                if move_sum > best_move_sum:

                        best_move_sum = move_sum
                        best_move_center = centers[i].copy()
                        move_made = True
                
                centers[i] = [x, y] # Reset

                centers[i] = best_move_center
                current_sum = best_move_sum

                    best_radii = solve_radii(centers)[1] # Re-calc radii for best_centers

        temperature *= cooling_rate

    # Final cleanup to remove any potential NaNs or negatives
    radii = np.maximum(best_radii, 0.0)
    # Clamp centers to [0, 1] just in case
    best_centers = np.clip(best_centers, 1e-9, 1 - 1e-9)
    
    return best_centers, radii, float(best_sum)
