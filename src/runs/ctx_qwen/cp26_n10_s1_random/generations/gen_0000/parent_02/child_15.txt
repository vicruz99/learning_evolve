import numpy as np
from scipy.optimize import minimize

def get_overlap_penalty(centers, radii):
    """Calculate penalty for circle overlaps."""
    n = len(radii)
    penalty = 0.0
    # Vectorized distance calculation
    # diff has shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    
    # r_sum has shape (n, n)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap occurs where dist < r_sum
    overlap = r_sum - dist
    
    # Only consider positive overlaps (and ignore diagonal where i==j)
    # Create a mask for lower triangle to avoid double counting and self-comparison
    mask = np.tril(np.ones((n, n), dtype=bool), k=-1)
    
    valid_overlaps = np.maximum(0, overlap[mask])
    penalty += np.sum(valid_overlaps**2)
    
    return penalty

def get_boundary_penalty(centers, radii):
    """Calculate penalty for circles exceeding unit square boundaries."""
    # Violations: x-r < 0, x+r > 1, y-r < 0, y+r > 1
    # Rewritten as: r > x, r > 1-x, r > y, r > 1-y
    # We penalize (r - x)^2 if r > x, etc.
    
    # Left boundary
    v1 = np.maximum(0, radii - centers[:, 0])
    # Right boundary
    v2 = np.maximum(0, radii - (1 - centers[:, 0]))
    # Bottom boundary
    v3 = np.maximum(0, radii - centers[:, 1])
    # Top boundary
    v4 = np.maximum(0, radii - (1 - centers[:, 1]))
    
    penalty = np.sum(v1**2) + np.sum(v2**2) + np.sum(v3**2) + np.sum(v4**2)
    return penalty

def objective_function(x_vars, centers_shape, penalty_weight=100.0):

    Objective function for scipy optimizer.
    x_vars: flattened array of [x1, y1, r1, x2, y2, r2, ...]
    """
    n = centers_shape[0]

    radii = x_vars[2*n:]

    obj = -np.sum(radii)

    boundary_pen = get_boundary_penalty(centers, radii)
    
    total_penalty = overlap_pen + boundary_pen
    
    return obj + penalty_weight * total_penalty
