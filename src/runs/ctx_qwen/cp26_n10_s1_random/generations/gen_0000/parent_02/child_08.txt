def generate_constraints(n):

    2. Non-overlap constraints between every pair of circles.

                dy = v[idx_yi] - v[idx_yj]

                'type': 'ineq',
                'fun': constraint_fun,

    row = 0
    while idx < n:

            idx += 1
        row += 1
