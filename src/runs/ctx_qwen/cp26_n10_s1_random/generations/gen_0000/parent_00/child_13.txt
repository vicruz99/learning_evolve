        return cons

        # dist_ij >= r_i + r_j  => dist_ij - r_i - r_j >= 0
        
        # Extract centers and radii
        # vars_flat layout: x0, y0, r0, x1, y1, r1, ...

        radii = np.zeros(n_circles)

            idx = 3 * i
            centers[i, 0] = vars_flat[idx]
            centers[i, 1] = vars_flat[idx+1]
            radii[i] = vars_flat[idx+2]
            
        cons = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                cons.append(dist - radii[i] - radii[j])
        return cons

    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 1] (practically much smaller)
    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((1e-6, 0.5)) # r (positive, max 0.5)

    def get_hex_grid_guess(r_offset=0.0, angle=0.0):

                cons.append(1 - y - r)

    if best_centers is None:
