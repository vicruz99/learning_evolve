import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective: maximize sum of radii -> minimize negative sum."""
    radii = vars[2::3]
    return -np.sum(radii)

def constraint_fun(vars):
    """Constraints: boundaries and non-overlap."""
    centers = np.column_stack((vars[0::3], vars[1::3]))
    radii = vars[2::3]
    
    cons = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    for i in range(N_CIRCLES):
        cons.append(centers[i, 0] - radii[i])
        cons.append(1 - centers[i, 0] - radii[i])
        cons.append(centers[i, 1] - radii[i])
        cons.append(1 - centers[i, 1] - radii[i])
        
    # Distance constraints: ||ci - cj|| >= ri + rj
    # Vectorized computation for performance
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    triu_indices = np.triu_indices(N_CIRCLES, k=1)
    pair_dists = dist_matrix[triu_indices]
    pair_radii = radii[triu_indices[0]] + radii[triu_indices[1]]
    
    cons.extend(pair_dists - pair_radii)
    
    return np.array(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Run optimization to pack 26 circles and return best configuration."""
    n_vars = 3 * N_CIRCLES
    
    # Bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Initialization strategies
    initial_guesses = []
    
    # 1. Perturbed 5x5 grid + 1 center
    grid = np.linspace(0.15, 0.85, 5)
    c_grid = np.array([[x, y] for x in grid for y in grid])
    c_grid = np.vstack([c_grid, [0.5, 0.5]])
    c_grid = c_grid + np.random.default_rng(42).normal(0, 0.01, c_grid.shape)
    initial_guesses.append(np.hstack([c_grid.flatten(), np.full(N_CIRCLES, 0.06)]))
    
    # 2. Random initializations
    rng = np.random.default_rng(123)
    for _ in range(8):
        c_rand = rng.uniform(0.2, 0.8, (N_CIRCLES, 2))
        initial_guesses.append(np.hstack([c_rand.flatten(), np.full(N_CIRCLES, 0.05)]))
        
    for guess in initial_guesses:
        res = minimize(
            objective,
            guess,
            method='SLSQP',
            jac='2-point',
            constraints={'type': 'ineq', 'fun': constraint_fun},
            bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False}
        )
        
        current_sum = -res.fun
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_centers = res.x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
            best_radii = res.x[2*N_CIRCLES:]
            
    # Clip tiny negative values from numerical noise
    best_radii = np.clip(best_radii, 0.0, None)
    
    return best_centers, best_radii, float(np.sum(best_radii))
