import numpy as np
from scipy.optimize import minimize


def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    for trial in range(120):
        centers, radii = get_initial(n, trial)
        x0 = np.concatenate([centers.ravel(), radii])

        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        cons = [{'type': 'ineq', 'fun': constraint_fn, 'args': (n,)}]

        res = minimize(
            objective_fn, x0, method='SLSQP', bounds=bounds,
            constraints=cons, args=(n,),
            options={'maxiter': 2000, 'ftol': 1e-15, 'disp': False}
        )

        c = res.x[:2 * n].reshape(n, 2)
        r = res.x[2 * n:]
        s = np.sum(r)

        if is_feasible(c, r, n) and s > best_sum:
            best_sum = s
            best_centers = c.copy()
            best_radii = r.copy()

    return best_centers, best_radii, best_sum


def objective_fn(x, n):
    return -np.sum(x[2 * n:])


def constraint_fn(x, n):
    c = []
    for i in range(n):
        ci = 3 * i
        c.append(x[ci] - x[ci + 2])
        c.append(1.0 - x[ci] - x[ci + 2])
        c.append(x[ci + 1] - x[ci + 2])
        c.append(1.0 - x[ci + 1] - x[ci + 2])
    for i in range(n):
        for j in range(i + 1, n):
            ci = 3 * i
            cj = 3 * j
            dx = x[ci] - x[cj]
            dy = x[ci + 1] - x[cj + 1]
            c.append(np.sqrt(dx * dx + dy * dy) - x[ci + 2] - x[cj + 2])
    return np.array(c)


def is_feasible(c, r, n):
    for i in range(n):
        if c[i, 0] - r[i] < -1e-10 or c[i, 0] + r[i] > 1 + 1e-10:
            return False
        if c[i, 1] - r[i] < -1e-10 or c[i, 1] + r[i] > 1 + 1e-10:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((c[i] - c[j]) ** 2))
            if d < r[i] + r[j] - 1e-10:
                return False
    return True


def get_initial(n, trial):
    np.random.seed(trial)
    if trial < 40:
        return hex_init(n, trial)
    elif trial < 80:
        return greedy_init(n, trial)
    else:
        return rand_init(n, trial)


def hex_init(n, seed):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    cols = 4 + (seed % 4)
    spacing = 1.0 / (cols + 2)
    dy = spacing * np.sqrt(3) / 2

    k = 0
    for row in range(12):
        for col in range(cols + 2):
            if k >= n:
                return centers, radii
            x = (col + 1.5) * spacing
            if row % 2 == 1:
                x += spacing / 2
            y = (row + 1.5) * dy
            centers[k] = [x, y]
            radii[k] = spacing / 2.5
            k += 1

    return centers, radii


def greedy_init(n, seed):
    np.random.seed(seed + 10000)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    for i in range(n):
        pt, r = best_placement(centers[:i], radii[:i])
        centers[i] = pt
        radii[i] = r

    return centers, radii


def best_placement(centers, radii):
    best_pt = None
    best_r = -1.0

    grid = 60
    xs = np.linspace(0.005, 0.995, grid)
    ys = np.linspace(0.005, 0.995, grid)

    for x in xs:
        for y in ys:
            r = min(x, 1.0 - x, y, 1.0 - y)
            for j in range(len(radii)):
                dx = x - centers[j, 0]
                dy = y - centers[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                r = min(r, d - radii[j])
            if r > best_r:
                best_r = r
                best_pt = np.array([x, y])

    if best_pt is None:
        best_pt = np.array([0.5, 0.5])
        best_r = 0.01

    return best_pt, best_r


def rand_init(n, seed):
    np.random.seed(seed + 20000)
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.015)
    return centers, radii
