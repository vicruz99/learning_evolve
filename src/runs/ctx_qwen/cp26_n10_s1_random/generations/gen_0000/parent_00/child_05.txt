import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n, mu):
    cs = vars[:2*n].reshape(n, 2)
    rs = vars[2*n:]
    
    pen = 0.0
    # Boundary penalties
    for i in range(n):
        x, y = cs[i]
        r = rs[i]
        if x - r < 0: pen += mu * (x - r)**2
        if x + r > 1: pen += mu * (x + r - 1)**2
        if y - r < 0: pen += mu * (y - r)**2
        if y + r > 1: pen += mu * (y + r - 1)**2
        
    # Overlap penalties
    for i in range(n):
        xi, yi = cs[i]
        ri = rs[i]
        for j in range(i + 1, n):
            dx = xi - cs[j, 0]
            dy = yi - cs[j, 1]
            d = np.sqrt(dx*dx + dy*dy + 1e-12)
            if ri + rs[j] - d > 0:
                pen += mu * (ri + rs[j] - d)**2
                
    return -np.sum(rs) + pen

def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0

        rows = [5, 6, 5, 6, 4]
        pos = []

            x_start = 0.15 + (idx % 2) * (x_step / 2)
            for _ in range(cnt):
                pos.append([x_start, y_curr])
                x_start += x_step
            y_curr += y_step
        centers = np.array(pos)

        centers = np.clip(centers, 0.05, 0.95)
        
        radii = np.full(n, 0.08)
        x0 = np.concatenate([centers.flatten(), radii])
        
        bounds = [(0.0, 1.0)] * (2*n) + [(1e-4, 0.5)] * n
        
        current_mu = 10.0
        for step in range(8):
            res = minimize(compute_objective, x0, args=(n, current_mu), method='L-BFGS-B', bounds=bounds,
                           options={'ftol': 1e-12, 'gtol': 1e-8, 'maxiter': 500})
            x0 = res.x
            current_mu *= 3.0
            
        centers_opt = x0[:2*n].reshape(n, 2)
        radii_opt = x0[2*n:]

        valid = True
        for i in range(n):
            x, y = centers_opt[i]
            r = radii_opt[i]

                valid = False
                break

                        valid = False
                        break
                if not valid:
                    break
                    
        if valid:
            s = np.sum(radii_opt)
            if s > best_sum:
                best_sum = s
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
                
    return best_centers, best_radii, best_sum
