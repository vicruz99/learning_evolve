import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- 1. Initialization: Hexagonal Lattice ---
    # We create a standard hexagonal lattice.
    # Spacing x = 1, y = sqrt(3)/2.
    # We pack as many as possible in a 5x5-ish grid and add the 26th in a gap.
    centers = []
    r_init = 0.11
    for row in range(5):
        for col in range(5):
            x = (col + 0.5) * 2 * r_init
            # Shift every other row to create hexagonal pattern
            if row % 2 == 1:
                x += r_init
            y = (row + 0.5) * np.sqrt(3) * r_init
            centers.append([x, y])
    
    # Add the 26th circle in a calculated gap (center of a cluster)
    centers.append([0.5, 0.5])
    
    # Normalize coordinates to be within [0, 1]
    centers = np.array(centers)
    centers -= np.min(centers, axis=0)
    centers /= np.max(centers, axis=0)
    
    # Adjust 26th circle to be slightly smaller and repositioned if needed
    centers[-1] = [0.55, 0.55]
    radii_init = np.full(n, r_init)
    radii_init[-1] = 0.05

    # --- 2. Optimization ---
    def objective(vars):
        # We maximize sum of radii, so we minimize negative sum
        radii = vars[2 * n : 3 * n]
        return -np.sum(radii)

    def constraints(vars):
        centers = vars[:2 * n].reshape(n, 2)
        radii = vars[2 * n : 3 * n]
        cons = []

        # Boundary constraints: x - r >= 0, x + r <= 1, same for y
        for i in range(n):
            cons.append(centers[i, 0] - radii[i])   # Left
            cons.append(1 - (centers[i, 0] + radii[i])) # Right
            cons.append(centers[i, 1] - radii[i])   # Bottom
            cons.append(1 - (centers[i, 1] + radii[i])) # Top

        # Overlap constraints: dist >= r1 + r2 => dist^2 - (r1+r2)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((centers[i] - centers[j]) ** 2)
                r_sum = radii[i] + radii[j]
                cons.append(dist_sq - r_sum ** 2)
        return np.array(cons)

    # Initial vector: [x1, y1, ..., xn, yn, r1, ..., rn]
    x0 = np.concatenate([centers.flatten(), radii_init])

    # Bounds for variables
    # x, y in [0, 1] (handled by constraints, but good for bounds)
    # r in [1e-6, 0.5]
    bounds = [(0, 1)] * (2 * n) + [(1e-6, 0.5)] * n

    # Use SLSQP solver
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                   constraints={'type': 'ineq', 'fun': constraints}, 
                   options={'maxiter': 2000, 'ftol': 1e-12})

    # --- 3. Post-processing and Validation ---
    final_vars = res.x
    final_centers = final_vars[:2 * n].reshape(n, 2)
    final_radii = final_vars[2 * n : 3 * n]
    
    # Ensure radii are positive
    final_radii = np.maximum(final_radii, 1e-9)
    
    # Validate
    is_valid = validate_packing(final_centers, final_radii)
    if not is_valid:
        # Fallback to a simple valid grid if optimization fails
        grid_r = 0.09
        final_centers = np.array([(i * 2 * grid_r + grid_r, j * 2 * grid_r + grid_r) 
                                  for j in range(5) for i in range(5)] + [[0.55, 0.55]])
        final_radii = np.full(26, grid_r)
        final_radii[-1] = 0.04
        
    total_sum = np.sum(final_radii)
    return final_centers, final_radii, total_sum

if __name__ == "__main__":
    centers, radii, total = run_packing()
    print(f"Valid: {validate_packing(centers, radii)}")
    print(f"Total sum of radii: {total}")
