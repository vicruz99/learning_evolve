import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # 1. Generate Initial Configurations
    # Strategy 1: Hexagonal-like packing
    def generate_hex_pattern(n):
        centers = np.zeros((n, 2))
        r_est = 0.1
        row_h = r_est * np.sqrt(3)
        row_w = r_est * 2.0
        idx = 0
        y = r_est
        shift = 0
        while idx < n:
            x = r_est + shift
            count = 0
            while x + r_est <= 1.0 and idx < n:
                centers[idx] = [x, y]
                x += row_w
                count += 1
                idx += 1
            y += row_h
            shift = row_w / 2.0
        return centers[:n]

    # Strategy 2: Perturbed Grid
    def generate_grid_pattern(n):
        cols = 5
        rows = 6
        step_x = 1.0 / cols
        step_y = 1.0 / rows
        centers = []
        count = 0
        for r in range(rows):
            for c in range(cols):
                if count >= n:
                    break
                centers.append([step_x * (c + 0.5), step_y * (r + 0.5)])
                count += 1
            if count >= n:
                break
        return np.array(centers[:n])

    # Strategy 3: Random
    def generate_random(n):
        return np.random.rand(n, 2)

    init_configs = [
        generate_hex_pattern(n),
        generate_grid_pattern(n),
        generate_random(n),
        generate_random(n) # Additional random restart
    ]

    best_sum = 0.0
    best_centers = None
    best_radii = None

    # 2. Optimization Function
    def solve_radii(centers):
        """Solve LP to find max sum of radii for fixed centers."""
        # Bounds for radii: 0 <= r <= dist to nearest boundary
        dist_to_bounds = np.array([
            min(c[0], 1-c[0], c[1], 1-c[1]) for c in centers
        ])
        
        # Constraints: r_i + r_j <= dist_ij
        # We only need to consider pairs that are close to reduce problem size,
        # but for n=26, n*(n-1)/2 = 325 constraints is small for LP.
        
        # Objective: max sum(r) -> min -sum(r)
        c_obj = -np.ones(n)
        
        bounds = [(0, b) for b in dist_to_bounds]
        
        A_ub = []
        b_ub = []
        
        dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
        
        for i in range(n):
            for j in range(i + 1, n):
                # r_i + r_j <= dist(i, j)
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dists[i, j])
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        try:
            res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                radii = -res.fun # Sum of radii
                # Extract radii
                r_vals = -res.x
                return r_vals, radii
        except Exception:
            pass
            
        return None, 0.0

    # 3. Simulated Annealing / Local Search
    def optimize_config(init_centers):
        centers = init_centers.copy()
        current_radii, current_sum = solve_radii(centers)
        
        if current_radii is None:
            return centers, np.zeros(n), 0.0

        best_sum = current_sum
        best_radii = current_radii
        best_centers = centers.copy()
        
        temperature = 0.05
        decay = 0.995
        steps = 2000
        
        # Precompute distances for efficiency
        # But since centers change, we recompute in solve_radii
        
        for step in range(steps):
            # Perturbation
            noise = np.random.randn(n, 2) * 0.02 * (1.0 / (step + 10))
            new_centers = centers + noise
            
            # Ensure centers are within [0, 1] (strictly, radius > 0 requires this)
            # But LP handles bounds, so we just clip to keep valid for geometry
            new_centers = np.clip(new_centers, 1e-5, 1-1e-5)
            
            new_radii, new_sum = solve_radii(new_centers)
            
            if new_radii is not None:
                delta = new_sum - current_sum
                
                # Accept if better, or with probability
                if delta > 0 or np.random.rand() < np.exp(delta / temperature):
                    centers = new_centers
                    current_radii = new_radii
                    current_sum = new_sum
                    
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_radii = current_radii
                        best_centers = centers.copy()
            
            temperature *= decay
            
        return best_centers, best_radii, best_sum

    # 4. Run Optimization for all initial configs
    for init_c in init_configs:
        # Rescale/adjust init if needed? 
        # Hex pattern might be slightly out, but LP handles it.
        # Grid is good.
        
        # Clip init centers to valid range
        init_c = np.clip(init_c, 1e-5, 1-1e-5)
        
        c, r, s = optimize_config(init_c)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r

    # 5. Final Validation & Adjustment
    # The LP ensures r_i + r_j <= dist, but floating point might be tight.
    # We might need to slightly shrink radii to pass validation strictly.
    # Validation allows 1e-12 error, so LP solution should be fine.
    # However, to be safe, we can enforce strict inequality in validation check logic if needed.
    # But standard LP solution is precise.
    
    # Check if radii are valid
    if best_radii is None or best_sum == 0:
        # Fallback to simple grid if optimization failed
        c = generate_grid_pattern(n)
        r = np.array([0.1] * n) # Approx
        # Re-solve radii for this fallback
        r, s = solve_radii(c)
        best_centers = c
        best_radii = r
        best_sum = s

    return best_centers, best_radii, best_sum
