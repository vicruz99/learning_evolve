import numpy as np
from scipy.optimize import minimize


def run_packing():

    best_centers = None
    best_radii = None
    best_sum = -1

    for trial in range(60):
        centers, radii = grow_and_optimize(n, trial)
        if centers is not None and check_validity(centers, radii, n):
            s = np.sum(radii)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()

    if best_centers is None:
        centers = np.array([[0.5, 0.5]] * n)
        radii = np.ones(n) * 0.05
        return centers, radii, float(np.sum(radii))

    return best_centers, best_radii, float(best_sum)


def init_hexagonal(n, seed):
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.015
    idx = 0
    row = 0
    while idx < n:
        cols = 5 if row % 2 == 0 else 4
        for col in range(cols):
            if idx >= n:
                break
            x = (col + 0.5 + (0.5 if row % 2 == 1 else 0)) / 5.0
            y = (row + 0.5) / 5.5
            centers[idx] = [x, y]
            idx += 1
        row += 1
    centers += np.random.uniform(-0.018, 0.018, (n, 2))
    centers = np.clip(centers, 0.04, 0.96)
    return centers, radii


def grow_and_optimize(n, seed):
    centers, radii = init_hexagonal(n, seed)

    for outer in range(800):
        # Force-based relaxation
        for inner in range(300):
            forces = np.zeros((n, 2))
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist_sq = dx * dx + dy * dy
                    dist = np.sqrt(dist_sq) + 1e-12
                    target = radii[i] + radii[j]
                    if dist < target:
                        f = (target - dist) / dist * 8.0
                        forces[i, 0] += dx * f
                        forces[i, 1] += dy * f
                        forces[j, 0] -= dx * f
                        forces[j, 1] -= dy * f
                for d in range(2):
                    if centers[i, d] < radii[i]:
                        forces[i, d] += 200.0 * (radii[i] - centers[i, d])
                    if centers[i, d] > 1.0 - radii[i]:
                        forces[i, d] -= 200.0 * (centers[i, d] - (1.0 - radii[i]))
            centers += forces * 0.008
        # Clamp centers
        centers = np.clip(centers, 0.001, 0.999)

        # Grow radii slowly
        radii *= 1.0008

    # Ensure boundary validity
    for i in range(n):
        x, y = centers[i]
        radii[i] = min(radii[i], x, 1.0 - x, y, 1.0 - y, 0.5)

    # Resolve overlaps by reducing radii
    for _ in range(2000):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-10:
                    total = radii[i] + radii[j]
                    if total > 1e-15:
                        scale = dist / total
                        radii[i] *= scale
                        radii[j] *= scale
                    changed = True
        if not changed:
            break

    radii = np.maximum(radii, 0.0)

    if not check_validity(centers, radii, n):
        return None, None

    # Optional scipy refinement
    centers, radii = scipy_refine(centers, radii, n)

    return centers, radii


def scipy_refine(centers, radii, n):
    # Combine into single vector
    x0 = np.concatenate([centers.flatten(), radii])

    def objective(x):
        c = x[:2 * n].reshape(n, 2)
        r = x[2 * n:]
        obj = -np.sum(r)
        # Penalty for boundary
        for i in range(n):
            x_coord, y_coord = c[i]
            rad = r[i]
            if x_coord < rad:
                obj += 500.0 * (rad - x_coord) ** 2
            if x_coord > 1.0 - rad:
                obj += 500.0 * (x_coord - (1.0 - rad)) ** 2
            if y_coord < rad:
                obj += 500.0 * (rad - y_coord) ** 2
            if y_coord > 1.0 - rad:
                obj += 500.0 * (y_coord - (1.0 - rad)) ** 2
        # Penalty for overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                if d < r[i] + r[j]:
                    obj += 500.0 * (r[i] + r[j] - d) ** 2
        return obj

    result = minimize(objective, x0, method='BFGS', options={'maxiter': 2000, 'disp': False})

    new_centers = result.x[:2 * n].reshape(n, 2)
    new_radii = result.x[2 * n:]

    # Clean up
    for i in range(n):
        x_coord, y_coord = new_centers[i]
        new_radii[i] = min(new_radii[i], x_coord, 1.0 - x_coord, y_coord, 1.0 - y_coord, 0.5)
        new_radii[i] = max(new_radii[i], 0.0)

    for _ in range(1000):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = new_centers[i, 0] - new_centers[j, 0]
                dy = new_centers[i, 1] - new_centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < new_radii[i] + new_radii[j] - 1e-10:
                    total = new_radii[i] + new_radii[j]
                    if total > 1e-15:
                        scale = dist / total
                        new_radii[i] *= scale
                        new_radii[j] *= scale
                    changed = True
        if not changed:
            break

    new_radii = np.maximum(new_radii, 0.0)
    return new_centers, new_radii


def check_validity(centers, radii, n):
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < -1e-12:
            return False
        if x - r < -1e-10 or x + r > 1.0 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1.0 + 1e-10:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True


def ensure_validity(centers, radii, n):
    for i in range(n):
        x, y = centers[i]
        radii[i] = min(radii[i], x, 1.0 - x, y, 1.0 - y, 0.5)
