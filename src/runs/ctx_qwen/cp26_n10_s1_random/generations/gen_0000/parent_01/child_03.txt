1. **LP-based radius optimization**: Given fixed centers, the optimal radii can be found by solving a linear program (maximize sum of radii subject to boundary and non-overlap constraints). This is efficient and exact.

2. **Differential evolution for center optimization**: Use scipy's differential_evolution to search over center positions, evaluating each candidate by solving the LP for radii. DE is a robust global optimizer that handles non-differentiable objectives well.

3. **Multiple restarts with varied initial patterns**: Try hexagonal, grid, and perturbed patterns to escape local optima.

4. **Post-processing refinement**: After finding a good solution, perform additional local optimization using SLSQP on both centers and radii jointly to squeeze out more improvement.

The key insight is that the radius subproblem (given centers) is a linear program, while the center problem is non-convex. By separating these, we get an efficient inner loop.

import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize


def solve_lp(centers, n):
    """Solve LP to find optimal radii for fixed centers."""
    c_obj = -np.ones(n)
    
    A_ub_rows = []
    b_ub_list = []
    
    for i in range(n):
        ci = centers[i]
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_list.append(ci[0])
        
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_list.append(1.0 - ci[0])
        
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_list.append(ci[1])
        
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_list.append(1.0 - ci[1])
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_list.append(dist)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_list)
    
    bounds = [(0, 0.5) for _ in range(n)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    
    return np.full(n, 0.01)


def neg_sum_radii_for_centers(params, n):
    centers = np.zeros((n, 2))
    for i in range(n):
        centers[i, 0] = params[2 * i]
        centers[i, 1] = params[2 * i + 1]
    radii = solve_lp(centers, n)
    return -np.sum(radii)


def make_hex_initial(n, perturbation=0.0):
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    spacing = 0.18
    
    while idx < n:
        if row % 2 == 0:
            ncols = 5
            x0 = 0.12
        else:
            ncols = 6
            x0 = 0.12 + spacing / 2.0
        
        for col in range(ncols):
            if idx >= n:
                break
            x = x0 + col * spacing
            y = 0.1 + row * spacing * np.sqrt(3.0) / 2.0
            
            if perturbation > 0:
                phase = row * 10 + col * 3.7
                x += perturbation * np.sin(phase)
                y += perturbation * np.cos(phase * 1.3)
            
            x = np.clip(x, 0.08, 0.92)
            y = np.clip(y, 0.08, 0.92)
            centers[idx] = [x, y]
            idx += 1
        row += 1
    
    return centers


def make_hex_initial_variant2(n, perturbation=0.0):
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    spacing = 0.175
    
    while idx < n:
        if row % 2 == 0:
            ncols = 6
            x0 = 0.08
        else:
            ncols = 5
            x0 = 0.08 + spacing / 2.0
        
        for col in range(ncols):
            if idx >= n:
                break
            x = x0 + col * spacing
            y = 0.1 + row * spacing * np.sqrt(3.0) / 2.0
            
            if perturbation > 0:
                phase = row * 7 + col * 2.1
                x += perturbation * np.cos(phase)
                y += perturbation * np.sin(phase * 1.7)
            
            x = np.clip(x, 0.08, 0.92)
            y = np.clip(y, 0.08, 0.92)
            centers[idx] = [x, y]
            idx += 1

    return centers


def make_grid_initial(n, spacing=0.16):
    centers = np.zeros((n, 2))
    idx = 0
    y = 0.1
    while idx < n:
        x = 0.1
        while x <= 0.9 and idx < n:
            centers[idx] = [x, y]
            idx += 1
            x += spacing
        y += spacing
    return centers


def flat_centers(centers, n):
    params = np.zeros(2 * n)
    for i in range(n):
        params[2 * i] = centers[i, 0]
        params[2 * i + 1] = centers[i, 1]
    return params


def unflat_centers(params, n):
    centers = np.zeros((n, 2))
    for i in range(n):
        centers[i, 0] = params[2 * i]
        centers[i, 1] = params[2 * i + 1]
    return centers


def full_flat(centers, radii, n):
    params = np.zeros(3 * n)
    for i in range(n):
        params[3 * i] = centers[i, 0]
        params[3 * i + 1] = centers[i, 1]
        params[3 * i + 2] = radii[i]
    return params


def full_unflat(params, n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = params[3 * i]
        centers[i, 1] = params[3 * i + 1]
        radii[i] = params[3 * i + 2]
    return centers, radii


def objective(params, n):
    return -np.sum(params[2::3])


def boundary_constraints(params, n):
    vals = np.zeros(4 * n)
    for i in range(n):
        vals[4 * i] = params[3 * i] - params[3 * i + 2]
        vals[4 * i + 1] = 1.0 - params[3 * i] - params[3 * i + 2]
        vals[4 * i + 2] = params[3 * i + 1] - params[3 * i + 2]
        vals[4 * i + 3] = 1.0 - params[3 * i + 1] - params[3 * i + 2]
    return vals


def distance_constraints(params, n):
    nc = n * (n - 1) // 2
    vals = np.zeros(nc)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = params[3 * i] - params[3 * j]
            dy = params[3 * i + 1] - params[3 * j + 1]
            rsum = params[3 * i + 2] + params[3 * j + 2]
            vals[k] = dx * dx + dy * dy - rsum * rsum
            k += 1
    return vals


def make_bounds(n):
    b = []
    for _ in range(n):
        b.append((0.0, 1.0))
        b.append((0.0, 1.0))
        b.append((0.0, 0.5))
    return b


def refine_with_slscp(centers, radii, n):
    params0 = full_flat(centers, radii, n)
    
    bounds = make_bounds(n)
    
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints, 'args': (n,)},
        {'type': 'ineq', 'fun': distance_constraints, 'args': (n,)},
    ]
    
    result = minimize(
        objective,
        params0,
        args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 2000, 'ftol': 1e-15, 'disp': False}
    )
    
    return full_unflat(result.x, n)


def run_packing():
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    initial_configs = []
    
    for pert in [0.0, 0.01, 0.02, 0.03, 0.05]:
        initial_configs.append(make_hex_initial(n, pert))
        initial_configs.append(make_hex_initial_variant2(n, pert))
    
    initial_configs.append(make_grid_initial(n, 0.15))
    initial_configs.append(make_grid_initial(n, 0.17))
    initial_configs.append(make_grid_initial(n, 0.19))
    
    for init_centers in initial_configs:
        params0 = flat_centers(init_centers, n)
        
        bounds_de = [(0.08, 0.92) for _ in range(2 * n)]
        
        try:
            de_result = differential_evolution(
                neg_sum_radii_for_centers,
                bounds_de,
                args=(n,),
                seed=42,
                maxiter=150,
                popsize=15,
                mutation=(0.3, 1.2),
                recombination=0.8,
                tol=1e-10,
                polish=False,
                workers=1,
                init='latinhypercube'
            )
            
            c = unflat_centers(de_result.x, n)
            r = solve_lp(c, n)
            s = np.sum(r)
            
            if s > best_sum:
                best_sum = s
                best_centers = c.copy()
                best_radii = r.copy()
        except Exception:
            pass
    
    if best_centers is not None:
        c_refined, r_refined = refine_with_slscp(best_centers, best_radii, n)
        s_refined = np.sum(r_refined)
        if s_refined > best_sum:
            best_centers = c_refined
            best_radii = r_refined
            best_sum = s_refined
    
    for trial in range(8):
        noise = 0.02 * np.sqrt(trial + 1)
        c_noisy = best_centers + noise * np.random.randn(n, 2)
        c_noisy[:, 0] = np.clip(c_noisy[:, 0], 0.08, 0.92)
        c_noisy[:, 1] = np.clip(c_noisy[:, 1], 0.08, 0.92)
        
        r_noisy = solve_lp(c_noisy, n)
        
        c_refined, r_refined = refine_with_slscp(c_noisy, r_noisy, n)
        s_refined = np.sum(r_refined)
        
        if s_refined > best_sum:
            best_sum = s_refined
            best_centers = c_refined.copy()
            best_radii = r_refined.copy()
    
    return best_centers, best_radii, best_sum
