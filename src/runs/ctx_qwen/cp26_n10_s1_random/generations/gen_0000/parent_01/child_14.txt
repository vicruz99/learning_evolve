def run_packing():

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.

    # Helper function to calculate the objective value (negative sum of radii)
    def objective(v):
        # v is flattened array: [x0, y0, r0, x1, y1, r1, ...]
        # Radii are at indices 2, 5, 8, ...
        radii = v[2::3]
        return -np.sum(radii)

    # Helper function to evaluate inequality constraints (must be >= 0)
    def constraints(v):
        # Reshape v to (n, 3) -> [x, y, r]
        # But v is 1D [x0, y0, r0, x1, y1, r1...]
        # Let's extract arrays directly
        xs = v[0::3]
        ys = v[1::3]
        rs = v[2::3]
        
        cons = []
        
        # 1. Boundary Constraints
        # x - r >= 0
        cons.extend(xs - rs)
        # 1 - x - r >= 0  => 1 >= x + r
        cons.extend(1.0 - xs - rs)
        # y - r >= 0
        cons.extend(ys - rs)
        # 1 - y - r >= 0
        cons.extend(1.0 - ys - rs)
        
        # 2. Non-overlap Constraints
        # dist^2 >= (r_i + r_j)^2
        # Only check i < j
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = (xs[i] - xs[j])**2 + (ys[i] - ys[j])**2
                rad_sum = rs[i] + rs[j]
                cons.append(dist_sq - rad_sum**2)
        
        return np.array(cons)
