    # Order: x, y, r
    vars_arr = vars_flat.reshape(n, 3)
    x = vars_arr[:, 0]
    y = vars_arr[:, 1]
    r = vars_arr[:, 2]
    
    # Objective: maximize sum(r) -> minimize -sum(r)
    cost = -np.sum(r)
    
    # Penalty weight
    PENALTY = 1000.0
    
    # 1. Boundary Constraints
    # x - r >= 0  => penalty if r > x
    # x + r <= 1  => penalty if x + r > 1
    # y - r >= 0
    # y + r <= 1
    
    # x constraints
    cost += PENALTY * np.sum(np.maximum(0.0, r - x)**2)
    cost += PENALTY * np.sum(np.maximum(0.0, x + r - 1.0)**2)
    
    # y constraints
    cost += PENALTY * np.sum(np.maximum(0.0, r - y)**2)
    cost += PENALTY * np.sum(np.maximum(0.0, y + r - 1.0)**2)
    
    # 2. Overlap Constraints
    # dist(i, j) >= r_i + r_j
    # penalty if r_i + r_j > dist(i, j)
    
    # Compute pairwise distances
    # x_diff[i, j] = x[i] - x[j]
    x_diff = x[:, None] - x[None, :]
    y_diff = y[:, None] - y[None, :]
    dist_sq = x_diff**2 + y_diff**2
    dist = np.sqrt(np.maximum(dist_sq, 1e-15))
    
    sum_r = r[:, None] + r[None, :]
    
    # Overlap amount: positive means violation
    overlap = sum_r - dist
    
    # Zero out diagonal (distance 0, sum_r 2r, overlap 2r - but self-overlap is irrelevant)
    np.fill_diagonal(overlap, 0.0)
    
    # Sum squared positive overlaps
    # Only count each pair once. Upper triangle.
    # Using a mask is efficient enough for n=26.
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cost += PENALTY * np.sum(np.maximum(0.0, overlap[mask])**2)
    
    return cost
