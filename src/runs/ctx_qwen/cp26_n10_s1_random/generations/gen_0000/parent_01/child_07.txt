import numpy as np
from scipy.optimize import minimize


def hexagonal_initial(n, r_base, seed=0):
    """Create hexagonal initial configuration"""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r_base
    
    r = r_base * 0.8
    idx = 0
    for row in range(10):
        for col in range(10):
            if idx >= n:
                break
            x = r + col * 1.85 * r
            y = r + row * 1.6 * r
            if row % 2 == 1:
                x += r * 0.925
            if x + r <= 0.99 and y + r <= 0.99 and x - r >= 0.01 and y - r >= 0.01:
                centers[idx] = [x, y]
                idx += 1
        if idx >= n:
            break
    
    while idx < n:
        centers[idx] = np.random.uniform(0.15, 0.85, 2)
        idx += 1
    
    return centers, radii


def grid_initial(n, seed=0):

    np.random.seed(seed)

    radii = np.ones(n) * 0.08
    
    idx = 0
    # 5x5 grid
    for row in range(5):
        for col in range(5):
            if idx >= n:
                break
            x = 0.1 + col * 0.18
            y = 0.1 + row * 0.18
            centers[idx] = [x, y]
            idx += 1
    
    # Fill remaining with random positions
    while idx < n:
        centers[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
    
    return centers, radii

def random_initial(n, seed=0):
    """Create random initial configuration"""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.ones(n) * 0.06
    return centers, radii


def penalized_objective(x, n=26, penalty_weight=500):
    """Objective function with penalty terms for constraints"""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    # Main objective: maximize sum of radii
    obj = -np.sum(radii)
    
    pw = penalty_weight
    
    # Boundary constraint penalties
    for i in range(n):
        # Left boundary: x - r >= 0
        violation = max(0.0, -centers[i, 0] + radii[i])
        obj += pw * violation * violation
        
        # Right boundary: x + r <= 1
        violation = max(0.0, centers[i, 0] + radii[i] - 1.0)
        obj += pw * violation * violation
        
        # Bottom boundary: y - r >= 0
        violation = max(0.0, -centers[i, 1] + radii[i])
        obj += pw * violation * violation
        
        # Top boundary: y + r <= 1
        violation = max(0.0, centers[i, 1] + radii[i] - 1.0)
        obj += pw * violation * violation
    
    # Overlap constraint penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-12)
            violation = max(0.0, radii[i] + radii[j] - dist)
            obj += pw * violation * violation
    
    # Non-negative radius penalties
    for i in range(n):
        violation = max(0.0, -radii[i])
        obj += pw * violation * violation
    
    return obj


def optimize_packing(centers0, radii0, n=26, penalty_weight=500, max_iter=20000):
    """Optimize circle packing using L-BFGS-B with penalty method"""
    x0 = np.concatenate([centers0.flatten(), radii0])
    
    # Bounds for centers and radii
    bounds = []
    for i in range(2 * n):
        bounds.append((0.0001, 0.9999))
    for i in range(n):
        bounds.append((0.00001, 0.45))
    
    result = minimize(
        penalized_objective,
        x0,
        args=(n, penalty_weight),
        method='L-BFGS-B',
        bounds=bounds,
        options={
            'maxiter': max_iter,
            'ftol': 1e-15,
            'gtol': 1e-12,
            'disp': False
        }
    )
    
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    
    return centers_opt, radii_opt, result.fun


def correct_feasibility(centers, radii, n=26):
    """Correct any slight constraint violations"""
    centers = centers.copy()
    radii = radii.copy()
    
    # Enforce boundary constraints
    for i in range(n):
        radii[i] = min(radii[i], centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = max(radii[i], 0.00001)
    
    # Resolve overlaps by reducing radii
    for iteration in range(100):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy + 1e-12)
                if dist < radii[i] + radii[j] - 1e-10:
                    violation = (radii[i] + radii[j] - dist) / 2
                    max_violation = max(max_violation, violation)
                    radii[i] = max(radii[i] - violation * 0.9, 0.00001)
                    radii[j] = max(radii[j] - violation * 0.9, 0.00001)
        
        if max_violation < 1e-10:
            break
    
    # Re-enforce boundary constraints after overlap correction
    for i in range(n):
        radii[i] = min(radii[i], centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = max(radii[i], 0.00001)
    
    return centers, radii
