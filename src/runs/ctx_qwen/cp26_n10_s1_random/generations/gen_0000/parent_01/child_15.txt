import numpy as np
from scipy.optimize import minimize

def get_initial_config(n_circles=26):
    """
    Generates an initial hexagonal packing configuration for n_circles.
    """
    # We aim for a layout that fits well in a square.
    # 6 rows with lengths 5, 4, 5, 4, 5, 3 sums to 26.
    row_lengths = [5, 4, 5, 4, 5, 3]
    
    # Check if total matches
    assert sum(row_lengths) == n_circles, "Row lengths sum mismatch"
    
    centers = []
    
    # Initial radius guess. 
    # For 6 rows of hexagonal packing, height ~ r(2 + 5*sqrt(3)) approx 10.66r
    # Width for 5 circles ~ 2*5*r = 10r
    # To fit in 1x1, r should be approx 0.09.
    r = 0.09
    
    current_y = r
    for i, length in enumerate(row_lengths):
        # Determine x-offset for this row
        # Even rows (0, 2, 4) start at x = r
        # Odd rows (1, 3, 5) start at x = 2r (shifted by r)
        if i % 2 == 0:
            x_start = r
        else:
            x_start = 2 * r
            
        for j in range(length):
            x = x_start + j * (2 * r)
            centers.append([x, current_y])
            
        # Move to next row y
        current_y += np.sqrt(3) * r
        
    centers = np.array(centers)
    radii = np.full(n_circles, r)
    
    return centers, radii

def compute_constraints(v, n):
    """
    Computes constraint violations for SLSQP.
    Returns an array where values <= 0 satisfy constraints.
    Constraints:
    1. x_i >= r_i  => r_i - x_i <= 0
    2. x_i <= 1 - r_i => x_i + r_i - 1 <= 0
    3. y_i >= r_i  => r_i - y_i <= 0
    4. y_i <= 1 - r_i => y_i + r_i - 1 <= 0
    5. dist^2 >= (r_i + r_j)^2 => (r_i + r_j)^2 - dist^2 <= 0
    """
    constraints = []
    
    # Reshape vector
    # v is [x0, y0, r0, x1, y1, r1, ...]
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # r - x <= 0
        constraints.append(r - x)
        # x + r - 1 <= 0
        constraints.append(x + r - 1.0)
        # r - y <= 0
        constraints.append(r - y)
        # y + r - 1 <= 0
        constraints.append(y + r - 1.0)
        
    # Overlap constraints
    # To speed up, we can vectorize or loop. Loop is fine for N=26 (325 pairs)
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        for j in range(i + 1, n):
            xj, yj = centers[j]
            rj = radii[j]
            
            dx = xi - xj
            dy = yi - yj
            dist_sq = dx*dx + dy*dy
            sum_r = ri + rj
            sum_r_sq = sum_r * sum_r
            
            # Constraint: sum_r_sq - dist_sq <= 0
            # If dist_sq < sum_r_sq, this is positive (violation)
            constraints.append(sum_r_sq - dist_sq)
            
    return np.array(constraints)

def objective(v, n):
    """
    Objective function: minimize -sum(radii)
    """
    radii = v[2*n:]
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Get initial guess
    centers_init, radii_init = get_initial_config(n)
    
    # Flatten to vector
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # Bounds
    # x, y in [0, 1]
    # r in [0, 0.5] (max possible radius in unit square is 0.5)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # Constraints dictionary for SLSQP
    cons = {'type': 'ineq', 'fun': lambda v: -compute_constraints(v, n)}
    # Note: SLSQP expects fun(x) >= 0 for 'ineq'. 
    # Our compute_constraints returns violations where <= 0 is valid.
    # So we negate it: -violation >= 0 <=> violation <= 0.
    
    # We might run multiple restarts with small perturbations to find global optimum
    best_v = x0
    best_obj = objective(x0, n)
    
    # Try a few random perturbations
    np.random.seed(42)
    
    # Initial run
    res = minimize(objective, x0, args=(n,), method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
    if res.success or res.fun < best_obj: # Note: fun is negative sum, so smaller is better
        best_v = res.x
        best_obj = res.fun

        # Perturb centers slightly, keep radii

        # Ensure centers stay within bounds loosely
        centers_pert = np.clip(centers_pert, 0, 1)
        
        v_pert = np.concatenate([centers_pert.flatten(), radii_init])
        
        res = minimize(objective, v_pert, args=(n,), method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
        
        if res.fun < best_obj:
            best_v = res.x
            best_obj = res.fun
            # If we found a better one, maybe restart from there?
            # But let's just collect best from independent runs

    best_radii = best_v[2*n:]
    sum_radii = -best_obj # Since obj was -sum

    # Although SLSQP should respect bounds, numerical issues can occur
    best_radii = np.maximum(best_radii, 0.0)
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    # Recalculate sum
    sum_radii = np.sum(best_radii)
    
    return best_centers, best_radii, sum_radii
