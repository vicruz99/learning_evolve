def generate_hex_grid(n):
    """
    Generates a hexagonal grid of n points inside [0,1]x[0,1].

    # Let's assume a spacing of roughly 0.2
    dx = 0.18
    dy = dx * np.sqrt(3) / 2
    
    # Generate points in a range larger than [0,1] and filter
    # Start from a corner or center? Center is usually better for square container
    # Let's try generating a grid centered at (0.5, 0.5)

    # x = i * dx + (j % 2) * (dx / 2)
    # y = j * dy
    
    # Determine range for i, j
    # Approx width 1, height 1

    # j range: -6 to 6
    
    raw_points = []
    for j in range(-8, 9):
        for i in range(-8, 9):

def objective_and_grad(vars_flat, centers_init, penalty_weight):

    Objective function and its gradient.
    vars_flat: array of shape (3 * n) containing x, y, r for each circle.
    """
    n = len(centers_init)
    centers = vars_flat[:2*n].reshape(n, 2)
    radii = vars_flat[2*n:]
    
    obj = -np.sum(radii)
    grad = np.zeros_like(vars_flat)

    grad[2*n:] = -1.0
    
    # Penalty for boundary constraints
    # x - r >= 0  => violation if x < r
    # r - x <= 0  => violation
    # 1 - x - r >= 0 => violation if x + r > 1
    # ...

    # Left: x >= r => r - x <= 0
    viol_left = np.maximum(0, radii - centers[:, 0])
    obj += penalty_weight * np.sum(viol_left**2)
    # Gradient of penalty term: 2 * (r-x) * dr/dv = 2*(r-x)*1, dx/dv = -1 (since deriv wrt r is 1, wrt x is -1)
    # Wait, term is (r-x)^2. d/dr = 2(r-x), d/dx = -2(r-x).
    grad[2*n:] += penalty_weight * 2 * viol_left
    grad[::2] += penalty_weight * 2 * viol_left * (-1) # x component (even indices in centers)
    
    # Right: 1 - x - r >= 0 => x + r - 1 <= 0
    viol_right = np.maximum(0, centers[:, 0] + radii - 1)
    obj += penalty_weight * np.sum(viol_right**2)
    grad[2*n:] += penalty_weight * 2 * viol_right
    grad[::2] += penalty_weight * 2 * viol_right
    
    # Bottom: y >= r => r - y <= 0
    viol_bottom = np.maximum(0, radii - centers[:, 1])
    obj += penalty_weight * np.sum(viol_bottom**2)
    grad[2*n:] += penalty_weight * 2 * viol_bottom
    grad[1::2] += penalty_weight * 2 * viol_bottom * (-1) # y component (odd indices)
    
    # Top: 1 - y - r >= 0 => y + r - 1 <= 0
    viol_top = np.maximum(0, centers[:, 1] + radii - 1)
    obj += penalty_weight * np.sum(viol_top**2)
    grad[2*n:] += penalty_weight * 2 * viol_top
    grad[1::2] += penalty_weight * 2 * viol_top
    
    # Overlap constraints
    # dist(i, j) >= r_i + r_j
    # r_i + r_j - dist(i, j) <= 0
    # We iterate over pairs
    # To optimize, we can vectorize or just loop (n=26 is small)
    
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff)
            if dist < 1e-9: dist = 1e-9 # Avoid division by zero
            
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                obj += penalty_weight * overlap**2
                
                # Gradients
                # d/dr_i = 2*overlap
                # d/dr_j = 2*overlap
                # d/dx_i = 2*overlap * (- d(dist)/dx_i) = 2*overlap * (-(x_i-x_j)/dist)
                # d/dy_i = ...
                # Note: gradient of -dist wrt x_i is -(x_i-x_j)/dist
                
                term = penalty_weight * 2 * overlap
                
                grad[2*n + i] += term
                grad[2*n + j] += term
                
                # Position gradients
                # Deriv of (r_i + r_j - sqrt((dx)^2 + ...))^2
                # 2*(...) * (-(x_i - x_j)/dist)

                grad[i*2] += common_pos_grad * (centers[i, 0] - centers[j, 0])

                grad[j*2+1] += common_pos_grad * (centers[j, 1] - centers[i, 1])
                
    return obj, grad
