import numpy as np
from scipy.optimize import minimize, Bounds, NonlinearConstraint

def objective_func(x):
    """Objective function to minimize: -sum(radii)"""
    n = 26
    return -np.sum(x[2*n:])

def constraint_func(x):
    """Vectorized constraint function: returns array of constraint values >= 0"""
    n = 26
    c = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    
    num_pair = n * (n - 1) // 2
    cons = np.empty(4*n + num_pair)
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons[:n] = c[:, 0] - r                # x - r >= 0
    cons[n:2*n] = 1 - c[:, 0] - r         # 1 - x - r >= 0
    cons[2*n:3*n] = c[:, 1] - r           # y - r >= 0
    cons[3*n:4*n] = 1 - c[:, 1] - r       # 1 - y - r >= 0
    
    # Pairwise non-overlap: dist >= r_i + r_j
    dx = c[:, 0, None] - c[None, :, 0]
    dy = c[:, 1, None] - c[None, :, 1]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cons[4*n:] = dist[mask] - r_sum[mask]
    
    return cons

def run_packing():
    n = 26
    x0_list = []
    
    # Initial guess 1: Hexagonal lattice packing
    r0 = 0.085
    centers = []
    y = r0
    row = 0
    while len(centers) < n:
        if row % 2 == 0:
            xs = [r0 + i * 2 * r0 for i in range(5)]
        else:
            xs = [2 * r0 + i * 2 * r0 for i in range(5)]
        for x in xs:
            if x + r0 <= 1 and len(centers) < n:
                centers.append([x, y])
        y += np.sqrt(3) * r0
        row += 1
    c_hex = np.array(centers[:n])
    r_hex = np.full(n, r0)
    x0_list.append(np.concatenate([c_hex.ravel(), r_hex]))

    centers_grid = np.vstack([cx.ravel(), cy.ravel()]).T[:n]
    r_grid = np.full(n, 0.085)
    x0_list.append(np.concatenate([centers_grid.ravel(), r_grid]))
    
    # Setup bounds and constraints
    lb = np.zeros(3*n)
    ub = np.ones(3*n)
    ub[2*n:] = 0.5
    bounds = Bounds(lb, ub, keep_feasible=True)
    
    nl_con = NonlinearConstraint(constraint_func, 0, np.inf)
    
    best_res = None
    best_val = -np.inf
    
    for x0 in x0_list:
        res = minimize(objective_func, x0, method='trust-constr', bounds=bounds, constraints=nl_con, options={'maxiter': 2000, 'verbose': 0})
        current_val = -res.fun
        if current_val > best_val:
            best_val = current_val
            best_res = res
            
    if best_res is None:
        return np.zeros((n, 2)), np.zeros(n), 0.0
        
    centers = best_res.x[:2*n].reshape(n, 2)
    radii = best_res.x[2*n:]
    
    # Ensure strict feasibility and positivity
    radii = np.clip(radii, 1e-7, 0.5)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
    
    return centers, radii, np.sum(radii)
