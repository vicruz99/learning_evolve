def objective_and_constraints(x):
    """
    Computes the negative sum of radii and all constraint violations.
    x is a flat array of size 3*N: [x1, y1, r1, x2, y2, r2, ...]
    """
    n = 26
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    obj = -np.sum(radii)
    
    # Constraints vector
    constraints_vals = []
    
    # 1. Boundary constraints
    # x - r >= 0
    constraints_vals.append(centers[:, 0] - radii)
    # 1 - x - r >= 0
    constraints_vals.append(1.0 - centers[:, 0] - radii)
    # y - r >= 0
    constraints_vals.append(centers[:, 1] - radii)
    # 1 - y - r >= 0
    constraints_vals.append(1.0 - centers[:, 1] - radii)
    
    # 2. Non-overlap constraints
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    # We only compute for i < j
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            # Using squared distance to avoid sqrt, constraint: dist_sq >= r_sum^2
            # dist_sq - r_sum^2 >= 0
            constraints_vals.append(dist_sq - r_sum*r_sum)
            
    return obj, np.concatenate(constraints_vals)

    # Actually, r <= 0.5 is always true for unit square.
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    # Initial constraint check
    _, cons = objective_and_constraints(x0)
    
    # Define constraints for SLSQP
    # SLSQP expects a list of dictionaries or a single constraint dict for vector
    # Here we use a single dict with a vector function
    cons_dict = {
        'type': 'ineq',
        'fun': lambda x: objective_and_constraints(x)[1]
    }
    
    # Optimization options
    options = {
        'maxiter': 500,
        'ftol': 1e-12,
        'disp': False
    }
    
    try:
        res = minimize(
            fun=lambda x: objective_and_constraints(x)[0],
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_dict,
            options=options
        )
        return res.x
    except Exception:
        return None
