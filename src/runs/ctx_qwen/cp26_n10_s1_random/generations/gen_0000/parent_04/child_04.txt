import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def objective_function(params, n):
    """
    Objective function to minimize: negative sum of radii + penalty for constraints.
    """
    # Unpack parameters
    centers = params[:2 * n].reshape((n, 2))
    radii = params[2 * n:]

    # Penalty terms
    penalty = 0.0
    weight = 1000.0

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # x - r >= 0 => r - x <= 0
        if r > x:
            penalty += weight * (r - x)**2
        
        # x + r <= 1 => x + r - 1 <= 0
        if x + r > 1:
            penalty += weight * (x + r - 1)**2
            
        # y - r >= 0 => r - y <= 0
        if r > y:
            penalty += weight * (r - y)**2
            
        # y + r <= 1 => y + r - 1 <= 0
        if y + r > 1:
            penalty += weight * (y + r - 1)**2

    # Overlap constraints
    # We check a subset or all. For 26 circles, 325 pairs is manageable.
    # To speed up, we can vectorize or just loop.
    for i in range(n):
        r_i = radii[i]
        c_i = centers[i]
        for j in range(i + 1, n):
            r_j = radii[j]
            c_j = centers[j]
            
            dist_sq = np.sum((c_i - c_j)**2)
            dist = np.sqrt(dist_sq)
            
            # Check if dist < r_i + r_j
            overlap = r_i + r_j - dist
            if overlap > 0:
                penalty += weight * overlap**2

    # Objective: minimize negative sum of radii + penalty
    return -np.sum(radii) + penalty

def get_hexagonal_initialization(n, seed=0):

            temp_n -= take

    else:
