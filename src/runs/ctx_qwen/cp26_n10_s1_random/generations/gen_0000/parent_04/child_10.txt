import numpy as np
import scipy.optimize as opt

def get_initial_hexagonal_positions(n_circles, radius_init=0.08):
    """
    Generates an initial configuration of circles in a hexagonal lattice pattern.
    """
    centers = []
    radii = []
    
    # Approximate number of circles per row based on hexagonal packing density
    # Or simply fill rows
    row_y = radius_init
    row_idx = 0
    
    while len(centers) < n_circles:
        # Horizontal shift for alternating rows
        shift = radius_init if row_idx % 2 == 1 else 0
        start_x = radius_init + shift
        step_x = 2 * radius_init
        
        # Determine how many circles fit in this row
        # Max x is 1 - radius_init
        max_x = 1 - radius_init
        
        count_in_row = int((max_x - start_x) / step_x) + 1
        
        for k in range(count_in_row):
            if len(centers) < n_circles:
                x = start_x + k * step_x
                centers.append([x, row_y])
                radii.append(radius_init)
        
        row_y += radius_init * np.sqrt(3)
        row_idx += 1
        
    return np.array(centers[:n_circles]), np.array(radii[:n_circles])

            constraints = []

                idx_x = i * 3
                idx_y = i * 3 + 1
                idx_r = n * 3 + i

                y = vars[idx_y]
                r = vars[idx_r]

                constraints.append(1.0 - x - r)

                # y <= 1-r
                constraints.append(1.0 - y - r)

                constraints.append(r)
                
            return np.array(constraints)
