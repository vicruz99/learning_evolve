def objective(params, n):
    """
    Objective function: minimize -sum(radii)
    params = [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = params.reshape(n, 3)[:, :2]
    radii = params.reshape(n, 3)[:, 2]
    return -np.sum(radii)

def gradient_objective(params, n):
    """
    Gradient of the objective function.
    d/dr_i = -1. d/dx_i = 0, d/dy_i = 0.
    """
    grad = np.zeros_like(params)
    # Indices for radii are 2, 5, 8, ...
    for i in range(n):
        grad[3*i + 2] = -1.0
    return grad

def get_constraints(n):
    """
    Returns a list of constraint dictionaries for scipy.optimize.
    """
    constraints = []
    
    # 1. Boundary constraints
    # x_i >= r_i  => x_i - r_i >= 0
    # x_i <= 1 - r_i => 1 - x_i - r_i >= 0
    # y_i >= r_i => y_i - r_i >= 0
    # y_i <= 1 - r_i => 1 - y_i - r_i >= 0
    
    for i in range(n):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # Left
        constraints.append({
            'type': 'ineq',
            'fun': lambda p, i=i: p[idx_x] - p[idx_r],
            'jac': lambda p, i=i: np.array([0]*(3*i) + [1, 0, -1] + [0]*(3*(n-1-i)))
        })
        # Right
        constraints.append({
            'type': 'ineq',
            'fun': lambda p, i=i: 1.0 - p[idx_x] - p[idx_r],
            'jac': lambda p, i=i: np.array([0]*(3*i) + [-1, 0, -1] + [0]*(3*(n-1-i)))
        })
        # Bottom
        constraints.append({
            'type': 'ineq',
            'fun': lambda p, i=i: p[idx_y] - p[idx_r],
            'jac': lambda p, i=i: np.array([0]*(3*i) + [0, 1, -1] + [0]*(3*(n-1-i)))
        })
        # Top
        constraints.append({
            'type': 'ineq',
            'fun': lambda p, i=i: 1.0 - p[idx_y] - p[idx_r],
            'jac': lambda p, i=i: np.array([0]*(3*i) + [0, -1, -1] + [0]*(3*(n-1-i)))
        })
        
        # Non-negative radius
        constraints.append({
            'type': 'ineq',
            'fun': lambda p, i=i: p[idx_r],
            'jac': lambda p, i=i: np.array([0]*(3*i) + [0, 0, 1] + [0]*(3*(n-1-i)))
        })

    # 2. Overlap constraints
    # dist(i, j) >= r_i + r_j
    # sqrt(...) - (r_i + r_j) >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
            idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
            
            def fun(p, i=i, j=j):
                xi, yi, ri = p[idx_xi], p[idx_yi], p[idx_ri]
                xj, yj, rj = p[idx_xj], p[idx_yj], p[idx_rj]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                return dist - (ri + rj)
            
            def jac(p, i=i, j=j):
                grad = np.zeros(3 * n)
                xi, yi, ri = p[idx_xi], p[idx_yi], p[idx_ri]
                xj, yj, rj = p[idx_xj], p[idx_yj], p[idx_rj]
                
                dx = xi - xj
                dy = yi - yj
                dist = np.sqrt(dx*dx + dy*dy)
                
                if dist < 1e-9:
                    # Avoid division by zero, though physically impossible if ri, rj > 0
                    grad[idx_xi] = 1.0
                    grad[idx_xj] = -1.0
                else:
                    # Derivatives of sqrt(dx^2 + dy^2) w.r.t xi, yi, xj, yj
                    term = 1.0 / dist
                    grad[idx_xi] = dx * term
                    grad[idx_yi] = dy * term
                    grad[idx_xj] = -dx * term
                    grad[idx_yj] = -dy * term
                
                # Derivatives w.r.t ri, rj are -1
                grad[idx_ri] = -1.0
                grad[idx_rj] = -1.0
                
                return grad
            
            constraints.append({
                'type': 'ineq',
                'fun': fun,
                'jac': jac
            })
            
    return constraints
