import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    best_sum_radii = 0.0
    best_state = None
    
    # Run optimization multiple times with different initial seeds to find a better local optimum
    for seed in range(10):
        rng = np.random.default_rng(seed)
        
        # 1. Initial Placement: Perturbed Hexagonal Grid
        # We arrange points in a grid-like structure with offsets to mimic hexagonal packing
        centers = np.zeros((n_circles, 2))
        radii = np.full(n_circles, 0.08)  # Start with small radii
        
        idx = 0
        # Attempt to fill rows with offsets
        y = 0.15
        while idx < n_circles:
            x = 0.15
            row_offset = 0.15 if int((y - 0.15) / 0.15) % 2 == 1 else 0.0
            while x <= 0.85:
                if idx < n_circles:
                    centers[idx] = [x + row_offset + rng.uniform(-0.01, 0.01), 
                                    y + rng.uniform(-0.01, 0.01)]
                    idx += 1
                x += 0.20
            y += 0.18
            
        # Clip centers to ensure they are within bounds initially
        centers = np.clip(centers, 0.05, 0.95)

        # Define variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
        initial_vars = np.hstack([centers.flatten(), radii])

        def objective(vars):
            """We want to maximize sum of radii, so we minimize the negative sum."""
            return -np.sum(vars[2*n_circles:])

        def constraints(vars):
            """Returns an array of constraint values. SLSQP requires them to be >= 0."""
            centers = vars[:2*n_circles].reshape(n_circles, 2)
            radii = vars[2*n_circles:]
            cons = []

            # Boundary constraints: 0 <= x - r and x + r <= 1 (implies r <= x <= 1-r)
            for i in range(n_circles):
                x, y = centers[i]
                r = radii[i]
                cons.append(x - r)          # x >= r
                cons.append(1.0 - (x + r))  # x + r <= 1.0
                cons.append(y - r)          # y >= r
                cons.append(1.0 - (y + r))  # y + r <= 1.0
            
            # Non-overlap constraints: dist >= r_i + r_j  =>  dist^2 - (r_i+r_j)^2 >= 0
            # Note: Using squared distance helps, but scipy handles the square root in the constraint function
            for i in range(n_circles):
                for j in range(i + 1, n_circles):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    cons.append(dist - (radii[i] + radii[j]))
            
            return np.array(cons)

        # Set up constraints for scipy
        scipy_constraints = {'type': 'ineq', 'fun': constraints}

        # Define bounds for variables: x, y in [0, 1], r >= 0
        bounds = []
        for _ in range(n_circles):
            bounds.extend([(0, 1), (0, 1), (0, 0.5)]) # r capped at 0.5 for safety

        try:
            res = minimize(objective, initial_vars, method='SLSQP', bounds=bounds, 
                           constraints=scipy_constraints, options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success or (not np.isnan(res.fun) and np.all(constraints(res.x) >= -1e-7)):
                current_sum = -res.fun
                if current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    best_state = res.x.copy()
        except Exception:
            continue

    if best_state is None:
        # Fallback to a simple valid grid if optimization fails
        centers = np.zeros((n_circles, 2))
        radii = np.full(n_circles, 0.09)
        idx = 0
        for r in range(5):
            for c in range(5):
                if idx < n_circles:
                    centers[idx] = [0.1 + c*0.2, 0.1 + r*0.2]
                    idx += 1
        # Add last circle
        centers[25] = [0.5, 0.5]
        radii[25] = 0.05
        
        return centers, radii, np.sum(radii)

    # Extract final results
    best_centers = best_state[:2*n_circles].reshape(n_circles, 2)
    best_radii = best_state[2*n_circles:]
    
    # Ensure strict non-negativity for radii
    best_radii = np.maximum(best_radii, 0)
    
    return best_centers, best_radii, best_sum_radii
