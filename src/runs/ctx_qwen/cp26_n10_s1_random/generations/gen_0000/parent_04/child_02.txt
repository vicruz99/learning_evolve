To maximize the sum of radii for 26 circles in a unit square, we formulate the problem as a constrained non-linear optimization task:
1. **Objective**: Maximize $\sum_{i=1}^{26} r_i$, equivalently minimize $-\sum r_i$.
2. **Constraints**: 
   - Boundary: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$ for all $i$.
   - Non-overlap: $\|c_i - c_j\| \ge r_i + r_j$ for all $i < j$. We square both sides to keep constraints polynomial: $\|c_i - c_j\|^2 - (r_i + r_j)^2 \ge 0$.
3. **Initialization**: A hexagonal lattice pattern provides a dense, non-overlapping starting configuration that is much closer to the optimum than random placement. Small random perturbations help escape symmetric local minima.
4. **Solver**: `scipy.optimize.minimize` with the `SLSQP` method is highly effective for medium-scale problems with many inequality constraints. We vectorize the constraint function for fast evaluation during gradient computations.
5. **Robustness**: We run the optimizer from 5 different initial seeds and keep the best result. This multi-restart strategy mitigates the risk of getting stuck in poor local optima due to the non-convex nature of the pairwise distance constraints.
6. **Post-processing**: Numerical clipping ensures strict compliance with the unit square bounds and non-negative radii, satisfying the validation function's tolerance.

import numpy as np
from scipy.optimize import minimize

def _objective(X, n):
    """Minimize negative sum of radii to maximize total sum."""
    return -np.sum(X[2::3])

def _constraint_fun(X, n):
    """
    Vectorized constraint function.
    Returns array of constraint values, all must be >= 0.
    Includes boundary constraints and pairwise non-overlap constraints.
    """
    X = X.reshape((n, 3))
    cx, cy = X[:, 0], X[:, 1]
    r = X[:, 2]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b_cons = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Pairwise constraints: ||ci - cj||^2 >= (ri + rj)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist_sq = dx**2 + dy**2
    r_sum_sq = (r[:, None] + r[None, :])**2
    
    # Extract upper triangle to avoid duplicate constraints
    triu_idx = np.triu_indices(n, k=1)
    p_cons = dist_sq[triu_idx] - r_sum_sq[triu_idx]
    
    return np.concatenate([b_cons, p_cons])

def _generate_initial(n, seed):
    """Generate initial positions using a perturbed hexagonal grid."""
    rng = np.random.default_rng(seed)
    pts = []
    y = 0.15
    row_idx = 0
    
    # Hexagonal packing layout
    while len(pts) < n:
        x = 0.15
        offset = 0.13 if row_idx % 2 == 1 else 0.0
        while x < 0.85 and len(pts) < n:
            pts.append([x + offset, y])
            x += 0.25
        y += 0.22
        row_idx += 1
        
    pts = np.array(pts[:n])
    # Add controlled random noise to break symmetry
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.1, 0.9)
    
    # Format as flat vector: [x0, y0, r0, x1, y1, r1, ...]
    X0 = np.zeros(3 * n)
    X0[0::3] = pts[:, 0]
    X0[1::3] = pts[:, 1]
    X0[2::3] = 0.05  # Safe initial radius
    return X0

def run_packing():
    """Optimize circle packing to maximize sum of radii."""
    n = 26
    best_X = None
    best_obj = np.inf
    
    # Multiple restarts to escape local minima
    for seed in range(5):
        X0 = _generate_initial(n, seed)
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        cons = {'type': 'ineq', 'fun': _constraint_fun, 'args': (n,)}
        
        try:
            res = minimize(_objective, X0, args=(n,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
            if res.fun < best_obj:
                best_obj = res.fun
                best_X = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization fails completely
    if best_X is None:
        best_X = _generate_initial(n, 0)
        
    # Reshape and format output
    centers = best_X.reshape((n, 3))[:, :2]
    radii = best_X.reshape((n, 3))[:, 2]
    
    # Clip to strictly satisfy bounds (handles minor numerical drift)
    centers = np.clip(centers, 0.0, 1.0)
    radii = np.clip(radii, 0.0, 0.5)
    
    return centers, radii, -best_obj
