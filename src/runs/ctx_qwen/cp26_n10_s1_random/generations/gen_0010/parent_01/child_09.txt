import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = N
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def objective(vars_arr):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars_arr[2*N:])

def constraints(vars_arr):
    """Inequality constraints >= 0 for valid packing (squared formulation for smoothness)."""
    x = vars_arr[:N]
    y = vars_arr[N:2*N]
    r = vars_arr[2*N:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    idx_i, idx_j = np.triu_indices(N, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,6,4], [5,5,6,5,5],
        [6,6,5,5,4], [5,6,4,6,5], [6,5,5,6,4], [5,4,6,6,5],
        [7,5,5,5,4], [5,5,5,5,6], [6,6,6,4,4], [4,5,6,5,6],
        [5,7,5,5,4], [5,5,7,5,4], [6,5,4,5,6], [5,6,5,4,6]
    ]
    for pat in patterns:
        if sum(pat) != N: continue
        pts = []
        r0 = 0.098
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx%2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:N])
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        if span[0] > 0.01 and span[1] > 0.01:
            scale = 0.85 / max(span[0]/(1-2*r0), span[1]/(1-2*r0))
            pts = (pts - mn) * scale + 0.5 * (1.0 - scale)
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        for _ in range(3):
            p = pts + rng.uniform(-0.015, 0.015, pts.shape)
            inits.append(np.clip(p, 0.05, 0.95))
            
    # Corner-exploiting configuration
    corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    hex_pts = []
    r0 = 0.1
    y = 0.25
    for idx in range(5):
        shift = r0 if idx%2 == 1 else 0.0
        x = 0.25 + shift
        for _ in range(5):
            if len(hex_pts) >= N-4: break
            hex_pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    pts = np.array(corners + hex_pts[:N-4])
    pts = np.clip(pts, 0.05, 0.95)
    inits.append(pts)
    for _ in range(3):
        p = pts + rng.uniform(-0.02, 0.02, pts.shape)
        inits.append(np.clip(p, 0.05, 0.95))

    for _ in range(5):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return inits

def hill_climb(centers, rng, steps=800):
    """Local search on centers using LP sum of radii as objective."""
    c = centers.copy()
    r, s = solve_lp(c)
    step = 0.012
    for _ in range(steps):
        # Move 1 or 2 circles simultaneously to escape local ridges
        n_move = 1 if rng.random() < 0.65 else 2
        idxs = rng.choice(N, size=n_move, replace=False)
        old = c[idxs].copy()
        c[idxs] += rng.uniform(-step, step, (n_move, 2))
        c[idxs] = np.clip(c[idxs], 1e-4, 1.0-1e-4)
        
        nr, ns = solve_lp(c)
        if ns > s + 1e-8:
            s = ns
            r = nr
        else:
            c[idxs] = old
        step *= 0.997
    return c, r, s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    inits = generate_inits(rng)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    # Phase 1: Broad search over diverse structures
    for init_c in inits:
        c, r, s = hill_climb(init_c, rng, steps=800)
        if s > best_sum:
            best_sum = s
            best_c = c.copy()
            best_r = r.copy()
            
    # Phase 2: Focused iterative refinement on the best configuration found
    for _ in range(40):
        c_p = best_c.copy()
        n_pert = rng.integers(3, 7)
        idxs = rng.choice(N, size=n_pert, replace=False)
        c_p[idxs] += rng.uniform(-0.008, 0.008, (n_pert, 2))
        c_p = np.clip(c_p, 0.05, 0.95)
        
        c_try, r_try, s_try = hill_climb(c_p, rng, steps=500)
        if s_try > best_sum + 1e-7:
            best_sum = s_try
            best_c = c_try.copy()
            best_r = r_try.copy()

    # Phase 3: Smooth SLSQP joint polish
    if best_c is not None:
        x0 = np.zeros(3*N)
        x0[:N] = best_c[:,0]
        x0[N:2*N] = best_c[:,1]
        x0[2*N:] = best_r * 0.96  # Start strictly feasible
        
        bounds_vars = [(0.0, 1.0)]*(2*N) + [(1e-6, 0.5)]*N
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_vars,
                           constraints={'type': 'ineq', 'fun': constraints},
                           options={'maxiter': 12000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_opt = np.column_stack((res.x[:N], res.x[N:2*N]))
                r_opt, s_opt = solve_lp(c_opt)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            pass

    # Phase 4: Strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N):
        x, y, r_val = best_c[i,0], best_c[i,1], best_r[i]
        if r_val > 1e-12:
            scale = min(scale, x/r_val, (1.0-x)/r_val, y/r_val, (1.0-y)/r_val)
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
