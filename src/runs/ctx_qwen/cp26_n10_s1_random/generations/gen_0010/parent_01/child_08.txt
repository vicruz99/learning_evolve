import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_sum(centers, idx_i, idx_j, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Radius upper bounds determined by distance to nearest wall
    lims = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]), 
                      np.minimum(centers[:,1], 1.0-centers[:,1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    # Pairwise distances: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.full(n, 1e-9)

def obj_centers(x, idx_i, idx_j, A_ub):
    """Objective for center optimization: negative sum of optimal radii."""
    c = np.clip(x.reshape(-1, 2), 0.001, 0.999)
    s, _ = solve_lp_sum(c, idx_i, idx_j, A_ub)
    return -s

def get_hex_configs(n):
    """Generates diverse hexagonal initial layouts."""
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5],
        [4,6,6,6,4], [6,4,6,6,4], [5,6,6,4,5], [5,4,6,6,5],
        [6,5,5,5,5], [5,5,5,5,6], [7,5,6,8], [6,7,6,7]
    ]
    configs = []
    for pat in patterns:
        if sum(pat) != n: 
            continue
        pts = []
        r0 = 0.098
        y = r0
        for i, cnt in enumerate(pat):
            shift = r0 if i%2==1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: 
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:n])
        # Normalize and center configuration within the square
        mn, mx = pts.min(axis=0), pts.max(axis=0)
        span = mx - mn
        if span[0] > 1e-4 and span[1] > 1e-4:
            pts = (pts - mn) / span * 0.82 + 0.09
        configs.append(np.clip(pts, 0.02, 0.98))
    return configs

def run_packing():
    n = 26
    rng = np.random.default_rng(2024)
    
    # Precompute LP constraint matrix structure
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    # Generate starts
    configs = get_hex_configs(n)
    for _ in range(12):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = 0.0
    best_c = None
    best_r = np.zeros(n)
    
    # Phase 1: Multi-start Nelder-Mead on centers
    for cfg in configs:
        try:
            res = minimize(obj_centers, cfg.flatten(), args=(idx_i, idx_j, A_ub), 
                           method='Nelder-Mead', options={'maxiter': 2000, 'xatol': 1e-9, 'fatol': 1e-12})
            c_opt = np.clip(res.x.reshape(n, 2), 0.001, 0.999)
            s, r = solve_lp_sum(c_opt, idx_i, idx_j, A_ub)
            if s > best_sum:
                best_sum = s
                best_c = c_opt.copy()
                best_r = r.copy()
        except Exception:
            pass
            
    # Fallback if optimization failed
    if best_c is None:
        best_c = configs[0]
        best_sum, best_r = solve_lp_sum(best_c, idx_i, idx_j, A_ub)
        
    # Phase 2: Local perturbation search to fine-tune centers
    step = 0.006
    for _ in range(3000):
        i = rng.integers(n)
        old = best_c[i].copy()
        best_c[i] += rng.uniform(-step, step, 2)
        best_c[i] = np.clip(best_c[i], 0.001, 0.999)
        
        s_new, _ = solve_lp_sum(best_c, idx_i, idx_j, A_ub)
        if s_new > best_sum + 1e-8:
            best_sum = s_new
        else:
            best_c[i] = old
        step *= 0.9985
        
    # Recompute optimal radii for final centers
    best_sum, best_r = solve_lp_sum(best_c, idx_i, idx_j, A_ub)
    
    # Phase 3: Strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.999999
    final_sum = float(np.sum(best_r))
    
    return best_c, best_r, final_sum
