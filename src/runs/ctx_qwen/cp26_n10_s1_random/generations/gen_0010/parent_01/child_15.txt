import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Maximum radius for each circle limited by distance to nearest boundary
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    dx = centers[idx_i, 0] - centers[idx_j, 0]
    dy = centers[idx_i, 1] - centers[idx_j, 1]
    b_ub = np.hypot(dx, dy)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            r = np.maximum(res.x, 0.0)
            return r, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def lp_objective(x, n):
    """Objective for derivative-free optimizers: negative sum of radii."""
    c = x.reshape(n, 2)
    _, val = solve_lp(c)
    return -val

def simulated_annealing(centers, rng, steps=8000, temp_init=0.05):
    """Global optimization of centers using Simulated Annealing."""
    n = centers.shape[0]
    best_c = centers.copy()
    curr_c = centers.copy()
    _, best_val = solve_lp(best_c)
    curr_val = best_val
    
    temp = temp_init
    step = 0.04
    
    for _ in range(steps):
        idx = rng.integers(n)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.uniform(-step, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 1e-4, 1.0 - 1e-4)
        
        _, new_val = solve_lp(curr_c)
        delta = new_val - curr_val
        
        # Accept if better, or with probability based on temperature
        if delta > 0 or rng.random() < np.exp(delta / temp):
            curr_val = new_val
            if curr_val > best_val:
                best_val = curr_val
                best_c = curr_c.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.998
        step *= 0.999
        
    return best_c, best_val
