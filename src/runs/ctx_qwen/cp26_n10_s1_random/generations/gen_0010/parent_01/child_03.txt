import numpy as np
from scipy.optimize import minimize, linprog

def compute_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, l) for l in lims]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    np.fill_diagonal(dists, np.inf)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def slsqp_constraints(v, n, idx_i, idx_j):
    """Inequality constraints >= 0 for SLSQP. Uses squared distances for better conditioning."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    
    # Boundary constraints
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap: d^2 >= (r_i + r_j)^2
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c_pair = dx**2 + dy**2 - dr**2
    
    return np.concatenate([c, c_pair])

def slsqp_objective(v, n):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(v[2*n:])
