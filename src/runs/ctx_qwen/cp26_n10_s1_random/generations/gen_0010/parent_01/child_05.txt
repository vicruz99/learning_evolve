import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def constraints_slsqp(x, n):
    """Inequality constraints for SLSQP: must be >= 0."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    con = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def obj_slsqp(x, n):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(x[2*n:])

def neg_lp_sum(x, n):
    """Objective for Nelder-Mead: maximize LP sum of radii."""
    c = x.reshape(n, 2)
    c = np.clip(c, 0.001, 0.999)
    _, s = solve_lp_radii(c)
    return -s

def get_hex_start(pat, rng):
    """Generates a hexagonal lattice configuration normalized to [0.1, 0.9]."""
    n = 26
    pts = []
    r0 = 0.095
    y = r0
    for idx, cnt in enumerate(pat):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    pts = np.array(pts[:n])
    
    mn = pts.min(axis=0)
    mx = pts.max(axis=0)
    span = mx - mn
    if span[0] > 0: pts[:, 0] = (pts[:, 0] - mn[0]) / span[0] * 0.8 + 0.1
    if span[1] > 0: pts[:, 1] = (pts[:, 1] - mn[1]) / span[1] * 0.8 + 0.1
    
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.05, 0.95)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5],
        [4,6,6,6,4], [5,6,4,6,5], [6,4,6,5,5], [5,5,5,5,6],
        [7,6,6,7], [6,7,6,7], [7,7,6,6], [6,6,7,7],
        [5,5,5,5,5,1], [4,5,6,5,6], [5,4,6,5,6], [6,5,5,5,5]
    ]
    
    starts = []
    for pat in patterns:
        if sum(pat) < n: continue
        for _ in range(2):
            starts.append(get_hex_start(pat, rng))
            
    for _ in range(6):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    bounds_vars = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints_slsqp, 'args': (n,)}
    
    # Phase 1: Nelder-Mead on centers (LP objective)
    for s in starts:
        try:
            res_nm = minimize(neg_lp_sum, s.flatten(), args=(n,), method='Nelder-Mead',
                              options={'maxiter': 2500, 'xatol': 1e-8, 'fatol': 1e-10})
            c_opt = np.clip(res_nm.x.reshape(n, 2), 0.01, 0.99)
            r_lp, s_lp = solve_lp_radii(c_opt)
            if s_lp > best_sum:
                best_sum = s_lp
                best_c = c_opt.copy()
                best_r = r_lp.copy()
        except Exception:
            continue
            
    if best_c is None:
        best_c = starts[0]
        best_r, best_sum = solve_lp_radii(best_c)
        
    # Phase 2: SLSQP Joint Refinement
    x0 = np.zeros(3*n)
    x0[:n] = best_c[:, 0]
    x0[n:2*n] = best_c[:, 1]
    x0[2*n:] = best_r * 0.95
    
    for _ in range(3):
        try:
            res_slqp = minimize(obj_slsqp, x0, args=(n,), method='SLSQP', bounds=bounds_vars,
                                constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
            if np.isfinite(res_slqp.fun):
                c_opt = np.column_stack((res_slqp.x[:n], res_slqp.x[n:2*n]))
                if np.min(constraints_slsqp(res_slqp.x, n)) > -1e-5:
                    r_lp, s_lp = solve_lp_radii(c_opt)
                    if s_lp > best_sum:
                        best_sum = s_lp
                        best_c = c_opt.copy()
                        best_r = r_lp.copy()
                    x0[:n] = c_opt[:, 0]
                    x0[n:2*n] = c_opt[:, 1]
                    x0[2*n:] = best_r * 0.95
        except Exception:
            break
            
    # Phase 3: Coordinate Ascent / Hill Climbing on Centers
    step = 0.018
    for it in range(3000):
        idx = rng.integers(n)
        old = best_c[idx].copy()
        best_c[idx] += rng.uniform(-step, step, 2)
        best_c[idx] = np.clip(best_c[idx], 0.005, 0.995)
        
        r_try, s_try = solve_lp_radii(best_c)
        if s_try > best_sum + 1e-7:
            best_sum = s_try
            best_r = r_try.copy()
            step = min(step * 1.015, 0.04)
        else:
            best_c[idx] = old
            step *= 0.994
            
    # Final LP squeeze
    r_final, s_final = solve_lp_radii(best_c)
    if s_final > best_sum:
        best_r = r_final
        best_sum = s_final
        
    # Phase 4: Strict Safety Scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
