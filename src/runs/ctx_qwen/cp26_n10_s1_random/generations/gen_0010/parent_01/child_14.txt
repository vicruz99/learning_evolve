import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
N_PAIRS = len(TRIU_I)

def compute_dists(c):
    """Computes pairwise Euclidean distance matrix."""
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    return np.hypot(diff[:, :, 0], diff[:, :, 1])

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers.
    Returns radii, sum_radii, and LP result object (for marginals)."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in limits]
    
    dists = compute_dists(centers)
    b_ub = dists[TRIU_I, TRIU_J]
    
    A_ub = np.zeros((N_PAIRS, N))
    A_ub[np.arange(N_PAIRS), TRIU_I] = 1.0
    A_ub[np.arange(N_PAIRS), TRIU_J] = 1.0
    
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0, None

def get_gradient(centers, res_obj):
    """Computes gradient of sum of radii w.r.t centers using LP dual variables."""
    if res_obj is None:
        return None
    try:
        marg = res_obj.ineqlin.marginals
    except AttributeError:
        try:
            marg = res_obj.marginals.ineqlin
        except AttributeError:
            return None
            
    grad = np.zeros((N, 2))
    valid = marg > 1e-7
    if not np.any(valid):
        return None
        
    idx = np.where(valid)[0]
    lam = marg[idx]
    i_ids = TRIU_I[idx]
    j_ids = TRIU_J[idx]
    
    diff = centers[i_ids] - centers[j_ids]
    dists = np.hypot(diff[:, 0], diff[:, 1])
    dists = np.where(dists > 1e-9, dists, 1.0)
    
    factors = lam[:, None] / dists[:, None]
    vecs = diff * factors
    
    np.add.at(grad, i_ids, vecs)
    np.add.at(grad, j_ids, -vecs)
    
    return grad

def objective_joint(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def constraints_joint(x, n):
    """Inequality constraints >= 0 for valid packing (squared distances for stability)."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    cons = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
    
    dx = cx[TRIU_I] - cx[TRIU_J]
    dy = cy[TRIU_I] - cy[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    
    # Squared distance constraint: d^2 >= (r_i + r_j)^2
    d2 = dx**2 + dy**2
    r2 = dr**2
    cons.append(d2 - r2)
    return np.concatenate(cons)

def run_gradient_ascent(c, steps=150):
    """Iteratively moves centers in the direction of the LP-derived gradient."""
    step_size = 0.015
    for _ in range(steps):
        r, s, res_obj = solve_lp(c)
        g = get_gradient(c, res_obj)
        if g is None:
            break
        gn = np.linalg.norm(g)
        if gn < 1e-8:
            break
        c_new = c + step_size * g / gn
        c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
        r_new, s_new, _ = solve_lp(c_new)
        if s_new > s + 1e-8:
            c = c_new
        else:
            step_size *= 0.85
    return c

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [5,6,4,6,5], [5,4,6,6,5], [6,5,5,6,4],
        [5,5,5,5,6], [7,5,5,5,4], [4,5,7,5,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,5,5,5], [5,5,5,6,5], [6,5,6,4,5]
    ]
    for pat in pats:
        if sum(pat) != N: continue
        pts = []
        r0 = 0.098
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3.0)
        pts = np.array(pts[:N])
        
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        if span[0] > 1e-4 and span[1] > 1e-4:
            scale = 0.85 / max(span[0], span[1])
            pts = (pts - mn) * scale + 0.5 * (1.0 - scale)
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
        for _ in range(3):
            p = pts + rng.uniform(-0.015, 0.015, pts.shape)
            inits.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(10):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    inits = generate_inits(rng)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    bounds_jt = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    
    for cfg in inits:
        # Phase 1: Gradient ascent on centers using LP duals
        c_opt = run_gradient_ascent(cfg.copy(), steps=200)
        r_lp, s_lp, _ = solve_lp(c_opt)
        
        if s_lp > best_sum:
            best_sum = s_lp
            best_c = c_opt.copy()
            best_r = r_lp.copy()
            
        # Phase 2: Joint SLSQP polish to deform layout and radii
        x0 = np.concatenate([c_opt.flatten(), r_lp * 0.96])
        try:
            res = minimize(objective_joint, x0, args=(N,), method='SLSQP',
                           bounds=bounds_jt,
                           constraints={'type': 'ineq', 'fun': constraints_joint, 'args': (N,)},
                           options={'maxiter': 6000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_pol = res.x[:2*N].reshape(N, 2)
                r_pol, s_pol, _ = solve_lp(c_pol)
                if s_pol > best_sum:
                    best_sum = s_pol
                    best_c = c_pol.copy()
                    best_r = r_pol.copy()
        except Exception:
            pass
            
        # Phase 3: Shake and re-optimize to escape local minima
        c_shake = best_c.copy()
        c_shake += rng.uniform(-0.01, 0.01, c_shake.shape)
        c_shake = np.clip(c_shake, 0.05, 0.95)
        c_shake = run_gradient_ascent(c_shake, steps=100)
        r_sh, s_sh, _ = solve_lp(c_shake)
        if s_sh > best_sum:
            best_sum = s_sh
            best_c = c_shake.copy()
            best_r = r_sh.copy()
            
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum, _ = solve_lp(best_c)
        
    # Phase 4: Strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.999999
    return best_c, best_r, float(np.sum(best_r))
