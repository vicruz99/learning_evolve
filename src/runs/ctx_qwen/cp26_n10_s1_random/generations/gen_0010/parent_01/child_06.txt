import numpy as np
from scipy.optimize import linprog, minimize

def get_triu_indices(n):
    return np.triu_indices(n, k=1)

def compute_lp_and_grad(centers, n, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient w.r.t centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-10)) for lim in limits]
    
    diff = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diff**2, axis=1) + 1e-15)
    
    m = len(idx_i)
    A_ub = np.zeros((m + n, n))
    b_ub = np.zeros(m + n)
    
    # Pairwise constraints: r_i + r_j <= dist
    A_ub[:m, idx_i] = 1.0
    A_ub[:m, idx_j] = 1.0
    b_ub[:m] = dists
    
    # Boundary constraints: r_i <= limit
    for i in range(n):
        A_ub[m + i, i] = 1.0
        b_ub[m + i] = limits[i]
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, 0.0, None
            
        r_opt = res.x
        sum_r = -res.fun
        
        # Extract duals for pairwise constraints
        try:
            mu = res.marginals.ineqlin
        except AttributeError:
            try:
                mu = res.ineqlin.marginals
            except AttributeError:
                mu = None
                
        grad = np.zeros((n, 2))
        if mu is not None and len(mu) >= m:
            lam = np.maximum(mu[:m], 0.0)
            active = np.where(lam > 1e-8)[0]
            for k_idx in active:
                lam_v = lam[k_idx]
                i_c = idx_i[k_idx]
                j_c = idx_j[k_idx]
                d_v = dists[k_idx]
                if d_v < 1e-9:
                    d_v = 1.0
                vec = diff[k_idx] / d_v
                grad[i_c] += lam_v * vec
                grad[j_c] -= lam_v * vec
                
        return r_opt, sum_r, grad
    except Exception:
        return None, 0.0, None

def optimize_centers_grad(c_init, n, idx_i, idx_j, rng, max_iter=1500):
    """Gradient ascent on centers using LP duals."""
    c = c_init.copy()
    curr_r, curr_s, _ = compute_lp_and_grad(c, n, idx_i, idx_j)
    if curr_r is None:
        return c, np.full(n, 1e-5), 0.0
        
    step = 0.02
    for it in range(max_iter):
        _, _, g = compute_lp_and_grad(c, n, idx_i, idx_j)
        if g is None:
            break
            
        gn = np.linalg.norm(g)
        if gn < 1e-9:
            c += rng.uniform(-0.015, 0.015, c.shape)
            c = np.clip(c, 0.01, 0.99)
            continue
            
        # Line search
        best_c = c
        best_s = curr_s
        for _ in range(20):
            c_try = c + step * g / gn
            c_try = np.clip(c_try, 0.01, 0.99)
            r_try, s_try, _ = compute_lp_and_grad(c_try, n, idx_i, idx_j)
            if r_try is not None and s_try > best_s:
                best_s = s_try
                best_c = c_try.copy()
                curr_r = r_try
                curr_s = s_try
            else:
                break
                
        c = best_c
        step = min(step * 1.02, 0.05)
        
        # Occasional large jump to escape local minima
        if it % 200 == 0 and rng.random() < 0.4:
            c_jump = c + rng.uniform(-0.04, 0.04, c.shape)
            c_jump = np.clip(c_jump, 0.05, 0.95)
            r_j, s_j, _ = compute_lp_and_grad(c_jump, n, idx_i, idx_j)
            if r_j is not None and s_j > curr_s + 1e-6:
                c = c_jump
                curr_r = r_j
                curr_s = s_j
                step = 0.02
                
    return c, curr_r, curr_s

def constraints_slsqp(v, n, idx_i, idx_j):
    """Inequality constraints >= 0 for SLSQP."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    
    c_list = []
    c_list.append(x - r)
    c_list.append(1.0 - x - r)
    c_list.append(y - r)
    c_list.append(1.0 - y - r)
    
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    d = np.sqrt(dx**2 + dy**2 + 1e-12)
    c_list.append(d - dr)
    
    return np.concatenate(c_list)

def objective_slsqp(v, n):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
