import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_and_grad(centers):
    """
    Solves LP to maximize sum of radii for fixed centers and returns 
    optimal radii, sum of radii, and gradient w.r.t centers via dual variables.
    """
    n = centers.shape[0]
    idx_i, idx_j = np.triu_indices(n, k=1)
    diff = centers[idx_i] - centers[idx_j]
    dists = np.linalg.norm(diff, axis=1)
    dists = np.maximum(dists, 1e-9)
    
    m_pairs = len(idx_i)
    m_bound = 4 * n
    m = m_pairs + m_bound
    
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m_pairs), idx_i] = 1.0
    A_ub[np.arange(m_pairs), idx_j] = 1.0
    
    b_ub = np.zeros(m)
    b_ub[:m_pairs] = dists
    
    # Explicit boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        b_ub[m_pairs + 4*i] = x
        b_ub[m_pairs + 4*i + 1] = 1.0 - x
        b_ub[m_pairs + 4*i + 2] = y
        b_ub[m_pairs + 4*i + 3] = 1.0 - y
        for offset in range(4):
            A_ub[m_pairs + 4*i + offset, i] = 1.0
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None
        radii = res.x
        sum_r = -res.fun
    except Exception:
        return None, None, None
        
    grad_c = np.zeros_like(centers)
    try:
        # Handle scipy version differences for marginal access
        if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            marginals = res.ineqlin.marginals
        elif hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
            marginals = res.marginals.ineqlin
        else:
            marginals = None
            
        if marginals is not None:
            # Pairwise contact gradients
            pair_lams = marginals[:m_pairs]
            active = pair_lams > 1e-8
            if np.any(active):
                idx_act = np.where(active)[0]
                lams = pair_lams[idx_act]
                i_a = idx_i[idx_act]
                j_a = idx_j[idx_act]
                d_a = dists[idx_act]
                vecs = (centers[i_a] - centers[j_a]) / d_a[:, None]
                np.add.at(grad_c, i_a, vecs * lams[:, None])
                np.add.at(grad_c, j_a, -vecs * lams[:, None])
                
            # Boundary contact gradients
            bound_lams = marginals[m_pairs:]
            active_b = bound_lams > 1e-8
            if np.any(active_b):
                idx_act_b = np.where(active_b)[0]
                lams_b = bound_lams[idx_act_b]
                wall_idx = idx_act_b % 4
                circ_idx = idx_act_b // 4
                for idx in range(len(idx_act_b)):
                    ci = circ_idx[idx]
                    w = wall_idx[idx]
                    lam = lams_b[idx]
                    if w == 0: grad_c[ci, 0] += lam
                    elif w == 1: grad_c[ci, 0] -= lam
                    elif w == 2: grad_c[ci, 1] += lam
                    elif w == 3: grad_c[ci, 1] -= lam
    except Exception:
        pass
        
    return radii, sum_r, grad_c

def constraints_slsqp(vars_arr, n):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    dist = np.sqrt(dx**2 + dy**2 + 1e-12)
    c.append(dist - dr)
    return np.concatenate(c)

def objective_slsqp(vars_arr, n):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars_arr[2*n:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
