import numpy as np
from scipy.optimize import linprog, minimize

def compute_lp(centers, n, idx_i, idx_j, A_ub):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in limits]
    
    diffs = centers[idx_i] - centers[idx_j]
    b_ub = np.sqrt(np.sum(diffs**2, axis=1))
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None, None, None
        
    if not res.success or not np.isfinite(res.fun):
        return None, None, None
        
    r = res.x
    s = -res.fun
    
    grad = np.zeros((n, 2))
    try:
        lams = res.ineqlin.marginals
    except Exception:
        try:
            lams = res.marginals.ineqlin
        except Exception:
            lams = None
            
    if lams is not None:
        try:
            lams_arr = np.asarray(lams).ravel()
        except Exception:
            lams_arr = np.zeros(1)
            
        valid = lams_arr > 1e-7
        if np.any(valid):
            ii = idx_i[valid]
            jj = idx_j[valid]
            lam_v = lams_arr[valid]
            ds = np.sqrt(np.sum((centers[ii] - centers[jj])**2, axis=1))
            ds = np.where(ds > 1e-8, ds, 1.0)
            diff = centers[ii] - centers[jj]
            factor = lam_v[:, None] / ds[:, None]
            np.add.at(grad, ii, diff * factor)
            np.add.at(grad, jj, -diff * factor)
            
    return r, s, grad

def constraints_slsqp(vars_arr, n, idx_i, idx_j):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(c)

def objective_slsqp(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Precompute LP constraint matrix structure
    A_ub = np.zeros((m_pairs, n))
    A_ub[np.arange(m_pairs), idx_i] = 1.0
    A_ub[np.arange(m_pairs), idx_j] = 1.0
    
    rng = np.random.default_rng(42)
    best_c, best_r, best_s = None, None, -1.0

    # 1. Hexagonal row patterns

        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,6,4], [6,6,5,5,4],
        [5,5,6,5,5], [6,4,6,4,6], [5,6,6,5,4], [7,5,5,5,4],
        [5,5,5,5,6], [6,6,6,4,4], [4,5,6,5,6], [5,4,6,5,6],
        [5,5,5,6,5], [6,5,5,5,5], [5,6,4,6,5], [4,6,5,6,5]
    ]
    
    for pat in row_pats:

            continue

        best_c = starts[0]
        best_r, best_s, _ = compute_lp(best_c, n, idx_i, idx_j, A_ub)
        if best_r is None:
            best_r = np.full(n, 0.08)
            best_s = float(np.sum(best_r))

    # Phase 4: Strict Safety Scaling
    scale = 1.0
    for i in range(n):
        x, y, r_val = best_c[i, 0], best_c[i, 1], best_r[i]
        if r_val > 1e-12:
            scale = min(scale, x/r_val, (1.0-x)/r_val, y/r_val, (1.0-y)/r_val)

            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_r *= scale * 0.9999995
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
