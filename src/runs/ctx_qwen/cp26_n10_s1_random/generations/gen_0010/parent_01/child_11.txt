import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_and_grad(centers):
    """Solves LP to maximize sum of radii for fixed centers and extracts gradient."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, None, None
            
        r = res.x
        s = -res.fun
        
        grad = np.zeros_like(centers)
        lam = None
        try:
            lam = res.ineqlin.marginals
        except AttributeError:
            try:
                lam = res.marginals.ineqlin
            except AttributeError:
                lam = None
                
        if lam is not None:
            active = lam > 1e-7
            if np.any(active):
                k_act = np.where(active)[0]
                i_a = idx_i[k_act]
                j_a = idx_j[k_act]
                l_a = lam[k_act]
                
                diff_a = centers[i_a] - centers[j_a]
                dist_a = np.linalg.norm(diff_a, axis=1)
                dist_a = np.maximum(dist_a, 1e-9)
                
                factor = (l_a / dist_a)[:, np.newaxis]
                np.add.at(grad, i_a, diff_a * factor)
                np.add.at(grad, j_a, -diff_a * factor)
                
        return r, s, grad
    except Exception:
        return None, None, None

def constraints_slsqp(vars_arr, n):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def objective_slsqp(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def generate_starts(n, rng):
    """Generates diverse initial center configurations."""
    starts = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [6,6,4,6,4], 
        [5,5,6,5,5], [6,4,6,4,6], [5,6,6,5,4], [6,5,4,6,5],
        [5,5,5,5,6], [6,6,6,4,4], [4,5,6,5,6], [5,4,6,5,6]
    ]
    
    for pat in patterns:
        if sum(pat) != n: continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx%2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:n])
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        if span[0] > 1e-4 and span[1] > 1e-4:
            pts = (pts - mn) / span * 0.8 + 0.1
        starts.append(pts)
        for _ in range(2):
            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            starts.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(6):
        starts.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    corners = np.array([[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]])
    rem = rng.uniform(0.2, 0.8, (n-4, 2))
    cf = np.vstack([corners, rem])
    starts.append(cf)
    
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    starts = generate_starts(n, rng)
    
    for cfg in starts:
        c = cfg.copy()
        curr_r, curr_s, _ = solve_lp_and_grad(c)
        if curr_r is None: continue
        
        step = 0.04
        for it in range(400):
            r, s, g = solve_lp_and_grad(c)
            if r is None: break
            
            gn = np.linalg.norm(g)
            if gn > 1e-7:
                c_new = c + step * g / gn
                c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
                r_new, s_new, _ = solve_lp_and_grad(c_new)
                
                if r_new is not None and s_new > s + 1e-8:
                    c = c_new
                    curr_r = r_new
                    curr_s = s_new
                    step = min(step * 1.05, 0.08)
                else:
                    step *= 0.90
            else:
                if rng.random() < 0.15:
                    c_shake = c + rng.uniform(-0.025, 0.025, c.shape)
                    c_shake = np.clip(c_shake, 0.05, 0.95)
                    r_sh, s_sh, _ = solve_lp_and_grad(c_shake)
                    if r_sh is not None and s_sh > curr_s:
                        c = c_shake
                        curr_r = r_sh
                        curr_s = s_sh
                        step = 0.04
                        
        if curr_s > best_sum:
            best_sum = curr_s
            best_centers = c.copy()
            best_radii = curr_r.copy()
            
        # SLSQP Polish to fine-tune centers & radii simultaneously
        x0 = np.zeros(3*n)
        x0[:n] = c[:,0]
        x0[n:2*n] = c[:,1]
        x0[2*n:] = curr_r * 0.96
        
        bounds_slqp = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
        try:
            res = minimize(objective_slsqp, x0, args=(n,), method='SLSQP', 
                           bounds=bounds_slqp,
                           constraints={'type': 'ineq', 'fun': constraints_slsqp, 'args': (n,)},
                           options={'maxiter': 6000, 'ftol': 1e-12})
            if np.isfinite(res.fun):
                c_pol = np.column_stack((res.x[:n], res.x[n:2*n]))
                r_lp, s_lp = solve_lp_and_grad(c_pol)
                if r_lp is not None and s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c_pol.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass

    # Final Safety Scaling to guarantee numerical validity
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i,0], best_centers[i,1], best_radii[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
                
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-12:
                    scale = min(scale, d/rs)
                    
        best_radii *= scale * 0.9999995
        best_sum = float(np.sum(best_radii))
    else:
        best_centers = rng.uniform(0.1, 0.9, (n, 2))
        best_radii = np.full(n, 0.05)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
