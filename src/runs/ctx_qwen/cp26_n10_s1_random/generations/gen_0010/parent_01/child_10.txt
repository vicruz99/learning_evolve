import numpy as np
from scipy.optimize import minimize, linprog

# Precompute constant structures for N=26
N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
M_PAIRS = len(IDX_I)
M_BOUND = 4 * N
M_TOTAL = M_BOUND + M_PAIRS

# Constant LP constraint matrix structure
A_LP = np.zeros((M_TOTAL, N))
for i in range(N):
    A_LP[i, i] = 1.0
    A_LP[N + i, i] = 1.0
    A_LP[2 * N + i, i] = 1.0
    A_LP[3 * N + i, i] = 1.0

k = 0
for ii, jj in zip(IDX_I, IDX_J):
    A_LP[M_BOUND + k, ii] = 1.0
    A_LP[M_BOUND + k, jj] = 1.0
    k += 1

def solve_lp_sum(centers):
    """Fast LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    b_ub = np.empty(M_TOTAL)
    b_ub[:4*N] = np.concatenate([centers[:, 0], 1.0 - centers[:, 0], 
                                 centers[:, 1], 1.0 - centers[:, 1]])
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    b_ub[M_BOUND:] = dists[IDX_I, IDX_J]
    
    bounds = [(0.0, lim) for lim in lims]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-9), 0.0

def objective_joint(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def constraints_joint(v, n):
    """Inequality constraints >= 0 for valid packing."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    dx = x[IDX_I] - x[IDX_J]
    dy = y[IDX_I] - y[IDX_J]
    dr = r[IDX_I] + r[IDX_J]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def generate_hex_configs(rng):
    """Generates diverse initial center configurations."""
    configs = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5],
        [4,6,6,6,4], [6,5,4,6,5], [5,6,4,6,5], [5,5,5,5,6],
        [7,5,5,5,4], [5,7,5,5,4], [6,4,6,5,5], [5,6,6,4,5]
    ]
    for pat in pats:
        if sum(pat) != N: continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:N])
        
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        if span[0] > 1e-4 and span[1] > 1e-4:
            scale = 0.85 / max(span[0]/(1-2*r0), span[1]/(1-2*r0))
            pts = (pts - mn) * scale + 0.5 * (1.0 - scale)
        pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts)
        
        for _ in range(2):
            p = pts + rng.uniform(-0.015, 0.015, pts.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(6):
        configs.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return configs

def sa_optimize(cfg, rng, steps=4000):
    """Simulated annealing on centers maximizing LP sum of radii."""
    c = cfg.copy()
    _, s_curr = solve_lp_sum(c)
    best_c = c.copy()
    best_s = s_curr
    temp = 0.04
    step = 0.02
    
    for _ in range(steps):
        idx = rng.integers(N)
        old = c[idx].copy()
        c[idx] += rng.uniform(-step, step, 2)
        c[idx] = np.clip(c[idx], 1e-4, 1.0 - 1e-4)
        
        _, s_new = solve_lp_sum(c)
        
        delta = s_new - best_s
        if delta > 0 or rng.random() < np.exp(delta / max(temp, 1e-6)):
            best_c = c.copy()
            best_s = s_new
        else:
            c[idx] = old
            
        temp *= 0.998
        step *= 0.9995
        
    return best_c, best_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    bounds_vars = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_joint, 'args': (N,)}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    configs = generate_hex_configs(rng)
    
    for cfg in configs:

        c_sa, s_sa = sa_optimize(cfg, rng, steps=3000)

        x0 = np.zeros(3*N)

                           bounds=bounds_vars, constraints=cons,

                cy = res.x[N:2*N]

                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass
            
        # Phase 3: Fine-tune LP hill climb on SA result
        curr_c = c_sa.copy()
        curr_s = s_sa
        step = 0.008

            improved = False

                idx = rng.integers(N)

                    improved = True
                else:
                    curr_c[idx] = old
            step *= 0.85
            
        if curr_s > best_sum:
            best_sum = curr_s
            best_centers = curr_c.copy()
            best_radii, _ = solve_lp_sum(best_centers)

    # Fallback safety net
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum = solve_lp_sum(best_centers)

    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

        for j in range(i+1, N):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], 
                         best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999995
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
