def solve_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    m_b = 4 * n
    m_p = n * (n - 1) // 2
    A = np.zeros((m_b + m_p, n))
    b = np.zeros(m_b + m_p)
    idx = 0
    
    limits = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        limits[i] = lim
        for _ in range(4):
            A[idx, i] = 1.0
            b[idx] = lim
            idx += 1
            
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    ii, jj = np.triu_indices(n, k=1)
    b[idx:] = dists[ii]
    A[idx:, ii] = 1.0
    A[idx:, jj] = 1.0
    
    bounds = [(0.0, max(lim, 1e-9)) for lim in limits]
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def slsqp_obj(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def slsqp_cons(v, n, ii, jj):

    x = v[:n]
    y = v[n:2*n]

    dy = y[ii] - y[jj]
    dr = r[ii] + r[jj]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
