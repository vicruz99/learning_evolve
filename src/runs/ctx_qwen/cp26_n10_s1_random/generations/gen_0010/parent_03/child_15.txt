import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    dx = centers[idx_i, 0] - centers[idx_j, 0]
    dy = centers[idx_i, 1] - centers[idx_j, 1]
    b_ub = np.sqrt(dx**2 + dy**2)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def lp_obj_wrapper(x_flat, n):
    """Objective for center optimization: minimize negative LP sum of radii."""
    centers = np.clip(x_flat.reshape(n, 2), 1e-6, 1.0 - 1e-6)
    _, s = solve_lp_radii(centers, n)
    return -s if np.isfinite(s) else 1e6

def get_hex_config(row_counts, r0, n):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def force_simulate(centers, radii, n, steps):
    """Physics simulation to expand circles until they touch."""
    vel = np.zeros_like(centers)
    dt = 0.003
    damping = 0.8
    k_rep = 150.0
    k_wall = 400.0

        req = radii[:, None] + radii[None, :]
        overlap = np.maximum(0.0, req - dists)

        fy = np.sum(diff[:, :, 1] * rep, axis=1)
        forces = np.column_stack((fx, fy))
        
        w_rep = np.zeros_like(centers)
        w_rep[:, 0] += np.maximum(0.0, radii - centers[:, 0]) * k_wall
        w_rep[:, 0] -= np.maximum(0.0, centers[:, 0] - (1.0 - radii)) * k_wall
        w_rep[:, 1] += np.maximum(0.0, radii - centers[:, 1]) * k_wall
        w_rep[:, 1] -= np.maximum(0.0, centers[:, 1] - (1.0 - radii)) * k_wall
        forces += w_rep
        
        vel = damping * vel + forces * dt
        centers += vel
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    return centers, radii

def cons_slsqp(v, n, ii, jj):
    """Inequality constraints >= 0 for SLSQP."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[ii] - x[jj]
    dy = y[ii] - y[jj]
    dr = r[ii] + r[jj]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def obj_slsqp(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])
