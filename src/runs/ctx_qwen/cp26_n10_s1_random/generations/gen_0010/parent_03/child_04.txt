import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_and_grad(centers, n):
    """Solves LP to maximize sum of radii for fixed centers and returns gradient."""
    ii, jj = np.triu_indices(n, k=1)
    m_p = len(ii)
    
    # Pairwise distances
    diff = centers[ii] - centers[jj]
    dists = np.hypot(diff[:, 0], diff[:, 1])
    
    # Boundary limits
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 0.0)
    
    # Construct constraint matrix: pairwise first, then boundaries
    m_b = n
    A_ub = np.zeros((m_p + m_b, n))
    A_ub[:m_p, ii] = 1.0
    A_ub[:m_p, jj] = 1.0
    A_ub[m_p:, np.arange(n)] = 1.0
    
    b_ub = np.concatenate([dists, lims])
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, 0.0, None
            
        r = res.x
        s = -res.fun
        
        # Compute gradient using LP marginals (dual variables)
        grad = np.zeros((n, 2))
        try:
            marg = res.ineqlin.marginals
        except AttributeError:
            try:
                marg = res.marginals.ineqlin
            except AttributeError:
                marg = None
                
        if marg is not None:
            # Only pairwise constraints contribute to gradient w.r.t centers
            active = marg[:m_p] > 1e-6
            if np.any(active):
                k = np.where(active)[0]
                lam = marg[:m_p][k]
                d = dists[k]
                d_safe = np.where(d > 1e-8, d, 1.0)
                vec = (centers[ii[k]] - centers[jj[k]]) / d_safe[:, None]
                
                np.add.at(grad, ii[k], vec * lam[:, None])
                np.add.at(grad, jj[k], -vec * lam[:, None])
                
        return r, s, grad
    except Exception:
        return None, 0.0, None

def objective_and_grad(x, n):
    """Objective and gradient for optimizer: minimize -sum(radii)."""
    centers = x.reshape(n, 2)
    r, s, grad = solve_lp_and_grad(centers, n)
    if grad is None:
        return 1e6, np.zeros_like(x)
    return -s, grad.flatten()

def generate_hex_config(row_counts, r0, shift_x, shift_y):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0 + shift_y
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift + shift_x
        for _ in range(cnt):
            if len(pts) >= 26:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(pts[:26])
