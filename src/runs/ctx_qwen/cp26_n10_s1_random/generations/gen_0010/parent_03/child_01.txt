import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    lims = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    dists = np.hypot(dx, dy)
    
    m = len(i_idx)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), i_idx] = 1.0
    A_ub[np.arange(m), j_idx] = 1.0
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def neg_lp_sum(x_flat, n):
    """Objective for center optimization: minimize negative LP sum of radii."""
    c = x_flat.reshape(n, 2)
    c = np.clip(c, 1e-5, 1.0 - 1e-5)
    _, s = solve_lp(c, n)
    return -s

def cons_slsqp(v, n, ii, jj):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[ii] - x[jj]
    dy = y[ii] - y[jj]
    dr = r[ii] + r[jj]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def obj_slsqp(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])
