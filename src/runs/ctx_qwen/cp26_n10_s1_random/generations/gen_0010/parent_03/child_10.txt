from scipy.optimize import minimize, linprog

# Precompute pair indices for efficiency
N_CIRCLES = 26
IDX_I, IDX_J = np.triu_indices(N_CIRCLES, k=1)
N_PAIRS = len(IDX_I)

def solve_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[IDX_I, IDX_J]
    A_ub[np.arange(m), IDX_I] = 1.0
    A_ub[np.arange(m), IDX_J] = 1.0
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def slsqp_obj(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def slsqp_cons(v, n):
    """Inequality constraints >= 0 for SLSQP."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    
    c_boundary = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    dx = x[IDX_I] - x[IDX_J]
    dy = y[IDX_I] - y[IDX_J]
    dr = r[IDX_I] + r[IDX_J]
    c_pairwise = dx**2 + dy**2 - dr**2
    
    return np.concatenate([c_boundary, c_pairwise])

def penalty_obj(v, n, mu):
    """Smooth penalty objective for L-BFGS-B."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    
    pen = mu * (np.sum(np.maximum(0.0, r - x)**2) +
                np.sum(np.maximum(0.0, r - (1.0 - x))**2) +
                np.sum(np.maximum(0.0, r - y)**2) +
                np.sum(np.maximum(0.0, r - (1.0 - y))**2))
                
    dx = x[IDX_I] - x[IDX_J]
    dy = y[IDX_I] - y[IDX_J]
    d = np.sqrt(dx**2 + dy**2)
    dr = r[IDX_I] + r[IDX_J]
    pen += mu * np.sum(np.maximum(0.0, dr - d)**2)
    
    return -np.sum(r) + pen

def generate_inits(n, rng):

    inits = []
    row_patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [5,6,4,6,5], [7,6,6,7], [8,7,7,4],

    ]

            continue

                        break

            inits.append(cfg)
            
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return inits

def hill_climb_centers(centers, n, rng, steps=1500):
    """Local search on centers using LP sum of radii as objective."""
    current_r, current_s = solve_lp(centers, n)
    if current_r is None:
        return centers, current_s
        
    step = 0.025
    for _ in range(steps):
        idx = rng.integers(n)
        old = centers[idx].copy()
        centers[idx] += rng.uniform(-step, step, 2)
        centers[idx] = np.clip(centers[idx], 0.005, 0.995)
        
        r_try, s_try = solve_lp(centers, n)
        if r_try is not None and s_try > current_s + 1e-9:
            current_r, current_s = r_try, s_try
        else:
            centers[idx] = old
        step *= 0.996
    return centers, current_s
