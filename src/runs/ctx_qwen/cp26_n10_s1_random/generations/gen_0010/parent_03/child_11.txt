import numpy as np
from scipy.optimize import linprog, minimize

N = 26
II, JJ = np.triu_indices(N, k=1)
NPAIRS = len(II)
M_CON = N + NPAIRS

# Precompute the sparse structure of the LP constraint matrix
A_UB = np.zeros((M_CON, N))
A_UB[:N, :] = np.eye(N)
A_UB[N:, II] = 1.0
A_UB[N:, JJ] = 1.0

def solve_lp(c):
    """Solves LP to maximize sum of radii for fixed centers and returns gradient."""
    x, y = c[:, 0], c[:, 1]
    ub = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    ub = np.maximum(ub, 1e-9)
    
    b_ub = np.empty(M_CON)
    b_ub[:N] = ub
    dx = c[II, 0] - c[JJ, 0]
    dy = c[II, 1] - c[JJ, 1]
    b_ub[N:] = np.sqrt(dx**2 + dy**2)
    
    try:
        res = linprog(-np.ones(N), A_ub=A_UB, b_ub=b_ub, bounds=[(0, None)]*N, method='highs')
        if res.success and np.isfinite(res.fun):
            r = res.x
            s = -res.fun
            grad = np.zeros_like(c)
            
            # Extract dual variables for gradient computation
            try:
                duals = np.abs(res.ineqlin.marginals)
            except AttributeError:
                try:
                    duals = np.abs(res.marginals.ineqlin)
                except AttributeError:
                    duals = None
                    
            if duals is not None:
                # Only pairwise constraints contribute to center gradients
                active = duals[N:] > 1e-7
                if np.any(active):
                    idx_v = np.where(active)[0]
                    dists_v = np.sqrt((c[II[idx_v], 0] - c[JJ[idx_v], 0])**2 + (c[II[idx_v], 1] - c[JJ[idx_v], 1])**2)
                    dists_v = np.maximum(dists_v, 1e-9)
                    factors = duals[N:][idx_v] / dists_v
                    diff = c[II[idx_v]] - c[JJ[idx_v]]
                    np.add.at(grad, II[idx_v], diff * factors[:, None])
                    np.add.at(grad, JJ[idx_v], -diff * factors[:, None])
            return r, s, grad
    except Exception:
        pass
    return None, 0.0, None

def get_constraints(v, n, ii, jj):
    """Inequality constraints >= 0 for valid joint packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    con = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[ii] - cx[jj]
    dy = cy[ii] - cy[jj]
    dr = r[ii] + r[jj]
    con = np.concatenate([con, dx**2 + dy**2 - dr**2])
    return con

def obj_joint(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

        [3,6,6,6,5], [5,7,4,5,5], [4,5,7,5,5], [6,4,6,5,5]
    ]

            cfg = np.clip(cfg, 0.05, 0.95)
            configs.append(cfg)

        configs.append(rng.uniform(0.15, 0.85, (n, 2)))

    best_r = None

        if r > 1e-12:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)

            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
    best_r *= scale * 0.9999999
    return best_c, best_r, float(np.sum(best_r))
