def solve_lp_fast(centers, n, A_ub_struct, pair_idxs):
    """Solves LP to maximize sum of radii for fixed centers using precomputed structure."""
    # Compute max radius per circle limited by boundaries
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lims[i]) for i in range(n)]
    
    # Compute pairwise distances for RHS
    diffs = centers[pair_idxs[:, 0]] - centers[pair_idxs[:, 1]]
    b_ub = np.sqrt(np.sum(diffs**2, axis=1))
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_struct, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun) and np.all(res.x >= -1e-9):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0
