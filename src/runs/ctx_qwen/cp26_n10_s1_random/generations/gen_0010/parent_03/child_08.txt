import numpy as np
from scipy.optimize import linprog, minimize

N = 26
idx_i, idx_j = np.triu_indices(N, 1)
M = len(idx_i)
A_lp = np.zeros((M, N))
A_lp[np.arange(M), idx_i] = 1.0
A_lp[np.arange(M), idx_j] = 1.0

def solve_lp_fast(centers):
    lims = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]),
                      np.minimum(centers[:,1], 1.0-centers[:,1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0, l) for l in lims]
    diffs = centers[idx_i] - centers[idx_j]
    b_ub = np.sqrt(np.sum(diffs**2, axis=1))
    try:
        res = linprog(-np.ones(N), A_ub=A_lp, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except:
        pass
    return None, 0.0

def lp_obj(x):
    _, s = solve_lp_fast(x.reshape(N, 2))
    return -s

def generate_inits():
    inits = []
    rng = np.random.default_rng(42)
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [5,4,6,6,5], [5,6,4,6,5], [6,5,5,6,4],
        [7,5,5,5,4], [5,5,5,5,6], [6,6,6,4,4], [5,5,5,6,5],
        [4,5,6,6,5], [5,6,5,4,6], [6,4,6,5,5], [5,5,4,6,6]
    ]
    for pat in patterns:
        if sum(pat) < N: continue
        for _ in range(2):
            r0 = 0.09 + rng.uniform(-0.008, 0.008)
            pts = []
            y = r0
            for idx, cnt in enumerate(pat):
                shift = r0 if idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= N: break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            cfg = np.array(pts[:N])
            mn, mx = cfg.min(axis=0), cfg.max(axis=0)
            cfg = (cfg - mn) / (mx - mn + 1e-9) * 0.82 + 0.09
            cfg += rng.uniform(-0.012, 0.012, cfg.shape)
            cfg = np.clip(cfg, 0.03, 0.97)
            inits.append(cfg)
            
    # Rotated variants to better align with square boundaries
    for angle_deg in [5, 10, 15, 25, 35]:
        angle = np.radians(angle_deg)
        rot = np.array([[np.cos(angle), -np.sin(angle)],
                        [np.sin(angle), np.cos(angle)]])
        for cfg in inits[:4]:
            c_cfg = cfg - 0.5
            c_cfg = c_cfg @ rot.T
            c_cfg += 0.5
            c_cfg = np.clip(c_cfg, 0.02, 0.98)
            inits.append(c_cfg)
            
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
    return inits

def run_packing():

    rng = np.random.default_rng(123)
    inits = generate_inits()
    
    # Phase 1: Nelder-Mead on centers (robust for non-smooth LP objective)
    for cfg in inits:
        try:
            res = minimize(lp_obj, cfg.flatten(), method='Nelder-Mead',
                           options={'maxiter': 1500, 'xatol': 1e-7, 'fatol': 1e-8})
            c_opt = res.x.reshape(N, 2)
            r, s = solve_lp_fast(c_opt)
            if r is not None and s > best_s:
                best_s = s
                best_c = c_opt.copy()
                best_r = r.copy()
        except:
            continue
            
    # Phase 2: Hill Climbing with Shake Mechanism
    if best_c is not None:
        curr_c = best_c.copy()
        curr_s = best_s
        step = 0.025
        for it in range(2500):
            n_pert = rng.integers(1, 6)
            idx = rng.choice(N, size=n_pert, replace=False)

                curr_s = s_try
                if s_try > best_s:
                    best_s = s_try
                    best_c = curr_c.copy()
                    best_r = r_try.copy()
            else:
                curr_c[idx] = old
            step *= 0.996
            
        # Shake to escape local minima
        for _ in range(10):
            shake = best_c + rng.uniform(-0.015, 0.015, best_c.shape)
            shake = np.clip(shake, 0.02, 0.98)
            r_sh, s_sh = solve_lp_fast(shake)
            if r_sh is not None and s_sh > best_s:
                best_s = s_sh
                best_c = shake.copy()
                best_r = r_sh.copy()
                curr_c = best_c.copy()
                curr_s = best_s
                step = 0.015

                    idx = rng.choice(N, size=n_pert, replace=False)

        def obj_j(v): return -np.sum(v[2*N:])

            x, y, r = v[:N], v[N:2*N], v[2*N:]
            c = np.concatenate([x-r, 1.0-x-r, y-r, 1.0-y-r])
            dx = x[idx_i] - x[idx_j]
            dy = y[idx_i] - y[idx_j]
            dr = r[idx_i] + r[idx_j]
            c = np.concatenate([c, dx**2 + dy**2 - dr**2])
            return c
        bounds = [(0.0, 1.0)]*(2*N) + [(1e-6, 0.5)]*N
        try:
            res = minimize(obj_j, v0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': con_j},
                           options={'maxiter': 8000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_p = np.column_stack((res.x[:N], res.x[N:2*N]))
                r_p, s_p = solve_lp_fast(c_p)
                if r_p is not None and s_p > best_s:
                    best_c = c_p
                    best_r = r_p
                    best_s = s_p
        except:
            pass

    # Phase 4: Strict safety scaling
    if best_c is not None:
        scale = 1.0
        for i in range(N):
            x, y, r = best_c[i,0], best_c[i,1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
        dx = best_c[idx_i,0] - best_c[idx_j,0]
        dy = best_c[idx_i,1] - best_c[idx_j,1]
        d = np.sqrt(dx**2 + dy**2)
        rs = best_r[idx_i] + best_r[idx_j]
        mask = rs > 1e-12
        if np.any(mask):
            scale = min(scale, np.min(d[mask]/rs[mask]))
        best_r *= scale * 0.9999999
        best_s = float(np.sum(best_r))
    else:
        best_c = np.random.uniform(0.1, 0.9, (N, 2))
        best_r = np.full(N, 0.08)
        best_s = float(np.sum(best_r))
        
    return best_c, best_r, best_s
