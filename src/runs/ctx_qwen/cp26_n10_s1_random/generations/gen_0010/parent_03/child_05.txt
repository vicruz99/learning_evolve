import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers, n, ii, jj):
    """Solves LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    bounds = [(0.0, lim) for lim in limits]
    
    dx = centers[ii, 0] - centers[jj, 0]
    dy = centers[ii, 1] - centers[jj, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    m = len(ii)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), ii] = 1.0
    A_ub[np.arange(m), jj] = 1.0
    b_ub = dists
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def cons_joint(v, n, ii, jj):
    """Inequality constraints >= 0 for joint SLSQP optimization."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[ii] - x[jj]
    dy = y[ii] - y[jj]
    dr = r[ii] + r[jj]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def obj_joint(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def clearance_func(v, n):
    """Negative minimum clearance for equal-radius center optimization."""
    c = v.reshape(n, 2)
    dx = c[:, np.newaxis, 0] - c[np.newaxis, :, 0]
    dy = c[:, np.newaxis, 1] - c[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    min_pair = np.min(dists) / 2.0
    
    d_wall = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                        np.minimum(c[:, 1], 1.0 - c[:, 1]))
    min_wall = np.min(d_wall)
    return -min(min_pair, min_wall)

def make_hex(rows, r0):
    """Generates initial positions on a hexagonal lattice."""
    n = 26
    pts = []
    y = r0
    for idx, cnt in enumerate(rows):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

    best_sum = -1.0

    if best_c is None:

        x, y, r = best_c[i,0], best_c[i,1], best_r[i]

            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)

            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999995
    final_sum = float(np.sum(best_r))
    
    return best_c, best_r, final_sum
