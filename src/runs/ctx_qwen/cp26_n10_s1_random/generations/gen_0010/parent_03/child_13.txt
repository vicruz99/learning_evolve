import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    limits = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    limits = np.maximum(limits, 1e-9)
    bounds = [(0.0, lim) for lim in limits]
    
    dx = centers[:, 0:1] - centers[:, 0:1].T
    dy = centers[:, 1:2] - centers[:, 1:2].T
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def force_sim(c0, n, steps=3000):
    """Physics simulation to expand circles and find tight packing."""
    c = c0.copy()
    r = np.full(n, 0.03)
    vel = np.zeros_like(c)
    dt = 0.002
    damp = 0.94
    k_rep = 800.0
    k_wall = 1200.0
    
    for _ in range(steps):
        r *= 1.0004
        diff = c[:, None, :] - c[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, 1e9)
        dists_safe = np.maximum(dists, 1e-8)
        
        req = r[:, None] + r[None, :]
        overlap = np.maximum(0.0, req - dists)
        rep = (overlap * k_rep) / dists_safe
        
        fx = np.sum(diff[:, :, 0] * rep, axis=1)
        fy = np.sum(diff[:, :, 1] * rep, axis=1)
        forces = np.column_stack((fx, fy))
        
        w_rep = np.zeros_like(c)
        w_rep[:, 0] += np.maximum(0.0, r - c[:, 0]) * k_wall
        w_rep[:, 0] -= np.maximum(0.0, c[:, 0] - (1.0 - r)) * k_wall
        w_rep[:, 1] += np.maximum(0.0, r - c[:, 1]) * k_wall
        w_rep[:, 1] -= np.maximum(0.0, c[:, 1] - (1.0 - r)) * k_wall
        forces += w_rep
        
        vel = damp * vel + forces * dt
        c += vel
        c = np.clip(c, 1e-5, 1.0 - 1e-5)
    return c

def obj_slqp(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_slqp(v, n):
    """Inequality constraints >= 0 for SLSQP."""
    cx = v[:n]
    cy = v[n:2*n]
    rad = v[2*n:]
    c = np.concatenate([cx - rad, 1.0 - cx - rad, cy - rad, 1.0 - cy - rad])
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    idx = np.triu_indices(n, k=1)
    rs = rad[:, None] + rad[None, :]
    c = np.concatenate([c, d2[idx] - rs[idx]**2])
    return c
