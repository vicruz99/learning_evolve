import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
ii, jj = np.triu_indices(N_CIRCLES, k=1)
N_PAIRS = len(ii)
A_PAIRWISE = np.zeros((N_PAIRS, N_CIRCLES))
A_PAIRWISE[np.arange(N_PAIRS), ii] = 1.0
A_PAIRWISE[np.arange(N_PAIRS), jj] = 1.0

def solve_lp_for_centers(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = N_CIRCLES
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-9)) for lim in limits]
    
    dx = centers[ii, 0] - centers[jj, 0]
    dy = centers[ii, 1] - centers[jj, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_PAIRWISE, b_ub=dists, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def neg_lp_sum(x_flat):
    """Objective for center optimization: minimize negative LP sum of radii."""
    c = x_flat.reshape(N_CIRCLES, 2)
    c = np.clip(c, 1e-4, 1.0 - 1e-4)
    _, s = solve_lp_for_centers(c)
    return -s if s > 0 else 1e5

def joint_obj(v, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def joint_cons(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[triu_i] - x[triu_j]
    dy = y[triu_i] - y[triu_j]
    dr = r[triu_i] + r[triu_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

            x = r0 + shift

                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        cfg = np.array(pts[:n])

        cfg[2] = [0.1, 0.9]
        cfg[3] = [0.9, 0.9]

            best_r = np.full(n, 0.08)

        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)

            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])

            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999999
    return best_c, best_r, float(np.sum(best_r))
