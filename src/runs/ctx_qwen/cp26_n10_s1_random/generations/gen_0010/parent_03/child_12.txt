from scipy.optimize import minimize, linprog

def solve_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    m = m_bound + m_pair
    A = np.zeros((m, n))
    b = np.zeros(m)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        limits = (x, 1.0 - x, y, 1.0 - y)
        for lim in limits:
            A[idx, i] = 1.0
            b[idx] = lim
            idx += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1
            
    bounds = [(0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def lp_obj_flat(x_flat, n):
    """Objective for Nelder-Mead: minimize negative LP sum of radii."""
    c = x_flat.reshape(n, 2)
    _, s = solve_lp(c, n)
    return -s

def cons_slsqp(v, n, ii, jj):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[ii] - x[jj]
    dy = y[ii] - y[jj]
    dr = r[ii] + r[jj]
    return np.concatenate([c, dx**2 + dy**2 - dr**2])

def obj_slsqp(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])
