import numpy as np
from scipy.optimize import linprog, minimize

def get_lp_and_grad(centers, n, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers and returns gradient."""
    dx = centers[idx_i, 0] - centers[idx_j, 0]
    dy = centers[idx_i, 1] - centers[idx_j, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    bounds = [(0.0, lim) for lim in limits]
    
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    b_ub = dists
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None
            
        r = res.x
        s = -res.fun
        
        grad = np.zeros((n, 2))
        lams = None
        if hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
            lams = res.marginals.ineqlin
        elif hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            lams = res.ineqlin.marginals
            
        if lams is not None:
            valid = lams > 1e-7
            if np.any(valid):
                k_idx = np.where(valid)[0]
                ii = idx_i[k_idx]
                jj = idx_j[k_idx]
                d = dists[k_idx]
                d = np.where(d > 1e-8, d, 1.0)
                diffs = centers[ii] - centers[jj]
                factors = (lams[k_idx][:, np.newaxis] / d[:, np.newaxis])
                
                np.add.at(grad, ii, diffs * factors)
                np.add.at(grad, jj, -diffs * factors)
                
        return r, s, grad
    except Exception:
        return None, None, None

def generate_inits(n, rng, num_inits=30):
    """Generates diverse initial center configurations."""
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [7,6,6,7], [8,6,6,6], 
        [9,5,6,6], [6,6,5,5,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,5,5,5,5], [5,5,5,5,6], [10,8,8], [9,9,8],
        [6,7,6,7], [5,6,6,6,3], [4,5,6,6,5]
    ]
    for pat in patterns:
        if sum(pat) < n: continue
        for _ in range(2):
            r0 = rng.uniform(0.09, 0.11)
            pts = []
            y = r0
            for idx, cnt in enumerate(pat):
                shift = r0 if idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            cfg = np.array(pts[:n])
            mn, mx = cfg.min(axis=0), cfg.max(axis=0)
            span = mx - mn
            if span[0] > 0: cfg[:, 0] = (cfg[:, 0] - mn[0]) / span[0] * 0.8 + 0.1
            if span[1] > 0: cfg[:, 1] = (cfg[:, 1] - mn[1]) / span[1] * 0.8 + 0.1
            cfg += rng.uniform(-0.02, 0.02, cfg.shape)
            cfg = np.clip(cfg, 0.03, 0.97)
            configs.append(cfg)
            
    for _ in range(10):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    for _ in range(5):
        cfg = rng.uniform(0.05, 0.95, (n, 2))
        corners = np.array([[0.05,0.05], [0.95,0.05], [0.05,0.95], [0.95,0.95]])
        cfg[:4] = corners + rng.uniform(-0.02, 0.02, (4,2))
        cfg = np.clip(cfg, 0.02, 0.98)
        configs.append(cfg)
        
    return configs[:num_inits]

def obj_slsqp(v, n):
    return -np.sum(v[2*n:])

def cons_slsqp(v, n, idx_i, idx_j):
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
