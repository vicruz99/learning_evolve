import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_and_grad(c, n, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient."""
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    wall = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), np.minimum(c[:, 1], 1.0 - c[:, 1]))
    bounds = [(0.0, max(w, 1e-9)) for w in wall]
    
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None
            
        r = res.x
        s = -res.fun
        
        grad = np.zeros((n, 2))
        try:
            lams = res.ineqlin.marginals
        except AttributeError:
            try:
                lams = res.marginals.ineqlin
            except AttributeError:
                lams = np.zeros(m)
                
        active = lams > 1e-7
        if np.any(active):
            k_idx = np.where(active)[0]
            lams_v = lams[k_idx]
            ii = idx_i[k_idx]
            jj = idx_j[k_idx]
            ds = b_ub[k_idx]
            ds = np.where(ds > 1e-8, ds, 1e-8)
            diffs = c[ii] - c[jj]
            factors = lams_v[:, None] / ds[:, None]
            np.add.at(grad, ii, diffs * factors)
            np.add.at(grad, jj, -diffs * factors)
            
        return r, s, grad
    except Exception:
        return None, None, None

def force_simulate(c, r, steps=600):
    """Vectorized force-directed simulation to spread circles and grow radii."""
    n = len(r)
    vel = np.zeros_like(c)
    dt = 0.005
    damp = 0.85
    for _ in range(steps):
        r *= 1.00005
        diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist, 1e9)
        overlap = np.maximum(0.0, r[:, np.newaxis] + r[np.newaxis, :] - dist)
        force_mag = overlap * 100.0 / (dist + 1e-9)
        f = np.sum(diff * force_mag[:, :, np.newaxis], axis=1)
        
        wf = np.zeros_like(c)
        mask_l = c[:, 0] < r
        mask_r = c[:, 0] > 1.0 - r
        mask_b = c[:, 1] < r
        mask_t = c[:, 1] > 1.0 - r
        wf[mask_l, 0] += (r[mask_l] - c[mask_l, 0]) * 200.0
        wf[mask_r, 0] -= (c[mask_r, 0] + r[mask_r] - 1.0) * 200.0
        wf[mask_b, 1] += (r[mask_b] - c[mask_b, 1]) * 200.0
        wf[mask_t, 1] -= (c[mask_t, 1] + r[mask_t] - 1.0) * 200.0
        f += wf
        
        vel = damp * vel + f * dt
        c += vel
        c = np.clip(c, 0.001, 0.999)
    return c, r

def obj_joint(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def cons_joint(x, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def optimize_track(cfg, n, idx_i, idx_j, rng):
    """Optimization pipeline for a single configuration track."""
    c = cfg.copy()
    r_init = np.full(n, 0.06)
    c, _ = force_simulate(c, r_init, steps=600)
    
    r_lp, s_lp, _ = solve_lp_and_grad(c, n, idx_i, idx_j)
    if r_lp is None:
        return None, None, -1.0
        
    best_s = s_lp
    best_c = c.copy()
    best_r = r_lp.copy()
    
    step = 0.03
    for _ in range(2500):
        r, s, g = solve_lp_and_grad(c, n, idx_i, idx_j)
        if r is None: break
        
        gn = np.linalg.norm(g)
        if gn > 1e-6:
            c_new = c + step * g / gn
            c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
            r_new, s_new, _ = solve_lp_and_grad(c_new, n, idx_i, idx_j)
            
            if r_new is not None and s_new > s + 1e-8:
                c = c_new
                s = s_new
                step = min(step * 1.015, 0.08)
            else:
                step *= 0.94
        else:
            if rng.random() < 0.15:
                c_shake = c + rng.uniform(-0.025, 0.025, c.shape)
                c_shake = np.clip(c_shake, 0.05, 0.95)
                r_sh, s_sh, _ = solve_lp_and_grad(c_shake, n, idx_i, idx_j)
                if r_sh is not None and s_sh > s + 1e-8:
                    c = c_shake
                    s = s_sh
                    step = 0.03
                    
        if s > best_s:
            best_s = s
            best_c = c.copy()
            best_r = r.copy() if r is not None else best_r
            
    # SLSQP joint polish
    x0 = np.concatenate([best_c[:, 0], best_c[:, 1], best_r * 0.97])
    bounds_slqp = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    try:
        res = minimize(obj_joint, x0, args=(n,), method='SLSQP', bounds=bounds_slqp,
                       constraints={'type': 'ineq', 'fun': cons_joint, 'args': (n, idx_i, idx_j)},
                       options={'maxiter': 4000, 'ftol': 1e-12})
        if np.isfinite(res.fun):
            c_pol = np.column_stack((res.x[:n], res.x[n:2*n]))
            r_pol, s_pol, _ = solve_lp_and_grad(c_pol, n, idx_i, idx_j)
            if r_pol is not None and s_pol > best_s:
                best_c = c_pol
                best_r = r_pol
                best_s = s_pol
    except Exception:
        pass
        
    return best_c, best_r, best_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    idx_i, idx_j = np.triu_indices(n, k=1)
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    patterns = [

    ]
    
    configs = []
    for pat in patterns:
        if sum(pat) != n: continue
        for r0 in [0.090, 0.095, 0.100, 0.105]:
            pts = []
            y = r0
            for ri, cnt in enumerate(pat):
                shift = r0 if ri % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) < n: pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            configs.append(np.array(pts[:n]))
            
    for _ in range(12):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    for cfg in configs:
        cfg_jit = cfg + rng.uniform(-0.02, 0.02, cfg.shape)
        cfg_jit = np.clip(cfg_jit, 0.05, 0.95)
        
        c_res, r_res, s_res = optimize_track(cfg_jit, n, idx_i, idx_j, rng)
        if s_res > best_sum:
            best_sum = s_res
            best_c = c_res
            best_r = r_res
            
    if best_c is None:
        best_c = configs[0]
        best_r, best_sum, _ = solve_lp_and_grad(best_c, n, idx_i, idx_j)
        if best_r is None:
            best_r = np.full(n, 0.08)
            best_sum = np.sum(best_r)
            
    # Strict numerical safety scaling
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    diff = best_c[:, np.newaxis, :] - best_c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    r_pair = best_r[:, np.newaxis] + best_r[np.newaxis, :]
    scale = min(scale, np.min(dists[idx_i, idx_j] / np.maximum(r_pair[idx_i, idx_j], 1e-12)))
    
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
