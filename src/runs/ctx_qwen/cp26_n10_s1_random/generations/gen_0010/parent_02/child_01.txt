import numpy as np
from scipy.optimize import linprog, minimize
import warnings
warnings.filterwarnings('ignore')

N_CIRCLES = 26

def get_lp_structure(n):
    """Precomputes indices and constraint matrix structure for pairwise constraints."""
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    return A_ub, idx_i, idx_j

def solve_lp_and_grad(centers, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, gradient, and distances."""
    n = centers.shape[0]
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(w, 1e-12)) for w in wall]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None, None
    except Exception:
        return None, None, None, None
        
    r = res.x
    s = -res.fun
    grad = np.zeros((n, 2))
    
    try:
        lams = None
        if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            lams = res.ineqlin.marginals
        elif hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
            lams = res.marginals.ineqlin
        elif hasattr(res, 'ineqfunc') and hasattr(res.ineqfunc, 'marginals'):
            lams = res.ineqfunc.marginals
            
        if lams is not None:
            lams = np.abs(lams)
            valid = lams > 1e-7
            if np.any(valid):
                k_idx = np.where(valid)[0]
                i_a = idx_i[k_idx]
                j_a = idx_j[k_idx]
                d_a = b_ub[k_idx]
                lam = lams[k_idx]
                diff_a = centers[i_a] - centers[j_a]
                inv_d = 1.0 / np.maximum(d_a, 1e-12)
                vec = diff_a * inv_d[:, np.newaxis] * lam[:, np.newaxis]
                np.add.at(grad, i_a, vec)
                np.add.at(grad, j_a, -vec)
    except Exception:
        pass
        
    return r, s, grad, b_ub

def objective_joint(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def constraints_joint(v, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def hex_init(n, rows, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for i, cnt in enumerate(rows):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:n])

def optimize_config(cfg, A_ub, idx_i, idx_j, rng):
    """Runs multi-phase optimization on a single configuration."""
    n = cfg.shape[0]
    c = np.clip(cfg, 1e-4, 1.0 - 1e-4)
    
    # Phase 1: Gradient ascent with adaptive steps and shakes
    step = 0.035
    for _ in range(300):
        r, s, g, _ = solve_lp_and_grad(c, A_ub, idx_i, idx_j)
        if r is None: break
        gn = np.linalg.norm(g)
        if gn > 1e-6:
            c_new = c + step * g / gn
            c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
            r_new, s_new, _, _ = solve_lp_and_grad(c_new, A_ub, idx_i, idx_j)
            if r_new is not None and s_new > s + 1e-8:
                c = c_new
                step = min(step * 1.015, 0.07)
            else:
                step *= 0.94
        else:
            if rng.random() < 0.25:
                c_shake = c + rng.uniform(-0.025, 0.025, c.shape)
                c_shake = np.clip(c_shake, 1e-4, 1.0 - 1e-4)
                r_sh, s_sh, _, _ = solve_lp_and_grad(c_shake, A_ub, idx_i, idx_j)
                if r_sh is not None and s_sh > s:
                    c = c_shake
                    step = 0.035
                    
    # Phase 2: Coordinate-wise hill climbing
    curr_r, curr_s, _, _ = solve_lp_and_grad(c, A_ub, idx_i, idx_j)
    step_hc = 0.022
    for _ in range(1500):
        i = rng.integers(n)
        old = c[i].copy()
        c[i] += rng.uniform(-step_hc, step_hc, 2)
        c[i] = np.clip(c[i], 1e-4, 1.0 - 1e-4)
        r_try, s_try, _, _ = solve_lp_and_grad(c, A_ub, idx_i, idx_j)
        if r_try is not None and s_try > curr_s + 1e-8:
            curr_s = s_try
            curr_r = r_try
        else:
            c[i] = old
        step_hc *= 0.9975
        
    # Phase 3: SLSQP joint polish
    if curr_r is not None:
        x0 = np.concatenate([c[:, 0], c[:, 1], curr_r * 0.992])
        bounds_slqp = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
        try:
            res = minimize(objective_joint, x0, args=(n, idx_i, idx_j), method='SLSQP',
                           bounds=bounds_slqp,
                           constraints={'type': 'ineq', 'fun': constraints_joint, 'args': (n, idx_i, idx_j)},
                           options={'maxiter': 2500, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_pol = res.x[:2*n].reshape(n, 2)
                r_pol, s_pol, _, _ = solve_lp_and_grad(c_pol, A_ub, idx_i, idx_j)
                if r_pol is not None and s_pol > curr_s:
                    c = c_pol
                    curr_r = r_pol
                    curr_s = s_pol
        except Exception:
            pass
            
    return c, curr_r, curr_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    A_ub, idx_i, idx_j = get_lp_structure(n)
    rng = np.random.default_rng(42)
    
    # Diverse hexagonal row distributions summing to 26

    for pat in patterns:

        for r0 in [0.092, 0.098, 0.104]:
            c = hex_init(n, pat, r0, rng)
            configs.append(c)
            c2 = c + rng.uniform(-0.028, 0.028, c.shape)
            configs.append(np.clip(c2, 0.05, 0.95))
            
    for _ in range(5):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = 0.0
    best_c = None
    best_r = None
    
    for cfg in configs:
        c_opt, r_opt, s_opt = optimize_config(cfg, A_ub, idx_i, idx_j, rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            best_r = r_opt.copy()
            
    if best_c is None:
        best_c = hex_init(n, [6, 5, 6, 5, 4], 0.1, rng)
        best_r = np.full(n, 0.095)
        best_sum = np.sum(best_r)
        
    # Final strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0 - x)/r, y/r, (1.0 - y)/r)
            
    diff = best_c[:, np.newaxis, :] - best_c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

    r_pair = best_r[:, np.newaxis] + best_r[np.newaxis, :]

    best_r *= scale * 0.999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
