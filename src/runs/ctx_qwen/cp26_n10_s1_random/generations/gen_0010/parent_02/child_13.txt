import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

def get_lp_data(centers, n, idx_i, idx_j):
    """Precomputes wall limits and pairwise distances for LP constraints."""
    w = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(val, 1e-12)) for val in w]
    diff = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    return bounds, dists

def solve_lp_and_grad(centers, n, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii and computes gradient w.r.t centers."""
    bounds, dists = get_lp_data(centers, n, idx_i, idx_j)
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None, None
            
        r = res.x
        s = -res.fun
        
        # Extract dual variables (marginals) for pairwise constraints
        try:
            lams = res.marginals.ineqlin
        except AttributeError:
            try:
                lams = res.ineqlin.marginals
            except AttributeError:
                lams = None
                
        grad_c = np.zeros((n, 2))
        if lams is not None and np.any(lams > 1e-9):
            active = lams > 1e-9
            k_idx = np.where(active)[0]
            if len(k_idx) == 0:
                return r, s, grad_c, lams
                
            lam_vals = lams[k_idx]
            i_idx = idx_i[k_idx]
            j_idx = idx_j[k_idx]

            vec = diff * factors
            
            np.add.at(grad_c, i_idx, vec)
            np.add.at(grad_c, j_idx, -vec)
            
        return r, s, grad_c, lams
    except Exception:
        return None, None, None, None

def optimize_centers_lp(c, n, A_ub, idx_i, idx_j, rng, max_iter=700):

        return c, r, s

                    r, s, grad = r_new, s_new, grad_new
            continue

            r, s, grad = r_new, s_new, grad_new

        else:
            step *= 0.88

def objective_joint(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def constraints_joint(v, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
