def hex_init(n, rows, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for i, cnt in enumerate(rows):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: 
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:n])

def obj_eq(v, n):
    """Objective for equal-radius optimization."""
    return -v[2*n]

def cons_eq(v, n):
    """Constraints for equal-radius packing."""
    cx = v[:n]
    cy = v[n:2*n]
    t = v[2*n]
    c = np.concatenate([cx - t, 1.0 - cx - t, cy - t, 1.0 - cy - t])
    idx = np.triu_indices(n, k=1)
    dx = cx[idx[0]] - cx[idx[1]]
    dy = cy[idx[0]] - cy[idx[1]]
    c = np.concatenate([c, dx**2 + dy**2 - 4.0*t**2])
    return c

def obj_joint(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_joint(v, n):

    cx = v[:n]

    dy = cy[idx[0]] - cy[idx[1]]
    dr = r[idx[0]] + r[idx[1]]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])

def lp_solve(centers, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]

                      np.minimum(centers[:,1], 1.0-centers[:,1]))

        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

            return res.x, -res.fun, res.ineqlin.marginals if hasattr(res, 'ineqlin') else None
    except: 
        pass
    return np.full(n, 1e-6), 0.0, None

def lp_grad(centers, radii, marginals, idx_i, idx_j, A_ub):
    """Compute gradient of LP objective w.r.t. centers using dual variables."""
    n = centers.shape[0]
    grad = np.zeros((n, 2))
    m = len(idx_i)
    if marginals is None:
        return grad
    
    active = marginals > 1e-8
    if not np.any(active):
        return grad
    
    k_active = np.where(active)[0]
    lam = marginals[k_active]
    ii = idx_i[k_active]
    jj = idx_j[k_active]
    
    diff = centers[ii] - centers[jj]
    dist = np.sqrt(np.sum(diff**2, axis=1))
    dist = np.maximum(dist, 1e-12)
    
    factors = lam[:, np.newaxis] / dist[:, np.newaxis]
    contrib = diff * factors
    
    np.add.at(grad, ii, contrib)
    np.add.at(grad, jj, -contrib)
    
    return grad
