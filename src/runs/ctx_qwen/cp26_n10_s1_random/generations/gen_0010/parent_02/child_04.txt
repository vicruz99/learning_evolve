import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

def solve_lp_and_get_marginals(centers, n, A_ub, idx_i, idx_j):
    """Solves LP for radii and extracts marginals for gradient computation."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-12)
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[idx_i] - centers[idx_j]
    b_ub = np.sqrt(np.sum(diffs**2, axis=1))
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None, None
            
        r = res.x
        s = -res.fun
        
        # Extract marginals safely
        marg_ineq = None
        marg_upper = None
        try:
            marg_ineq = res.marginals.ineqlin
            marg_upper = res.marginals.upper
        except AttributeError:
            try:
                marg_ineq = res.ineqlin.marginals
                marg_upper = res.upper.marginals
            except AttributeError:
                pass
                
        return r, s, marg_ineq, marg_upper
    except Exception:
        return None, None, None, None

def compute_gradient(centers, n, marg_ineq, marg_upper, idx_i, idx_j):
    """Computes gradient of sum of radii w.r.t centers using LP marginals."""
    grad = np.zeros((n, 2))
    if marg_ineq is None or marg_upper is None:
        return grad
        
    # Pairwise contacts
    if marg_ineq is not None:
        active = marg_ineq > 1e-7
        if np.any(active):
            k_idx = np.where(active)[0]
            lam = marg_ineq[k_idx]
            i_a = idx_i[k_idx]
            j_a = idx_j[k_idx]
            d_vec = centers[i_a] - centers[j_a]
            d_norm = np.linalg.norm(d_vec, axis=1, keepdims=True)
            d_norm = np.where(d_norm < 1e-12, 1.0, d_norm)
            factors = (lam[:, None] / d_norm)
            np.add.at(grad, i_a, d_vec * factors)
            np.add.at(grad, j_a, -d_vec * factors)
            
    # Wall contacts (upper bounds correspond to r <= dist_to_wall)
    # Upper bounds are ordered as [x, 1-x, y, 1-y] for each circle
    if marg_upper is not None:
        for i in range(n):
            # Left wall: x - r >= 0  => r <= x. Dual pushes +x direction
            if marg_upper[4*i] > 1e-7:
                grad[i, 0] += marg_upper[4*i]
            # Right wall: 1-x - r >= 0 => r <= 1-x. Dual pushes -x direction
            if marg_upper[4*i+1] > 1e-7:
                grad[i, 0] -= marg_upper[4*i+1]
            # Bottom wall: y - r >= 0 => r <= y. Dual pushes +y direction
            if marg_upper[4*i+2] > 1e-7:
                grad[i, 1] += marg_upper[4*i+2]
            # Top wall: 1-y - r >= 0 => r <= 1-y. Dual pushes -y direction
            if marg_upper[4*i+3] > 1e-7:
                grad[i, 1] -= marg_upper[4*i+3]
                
    return grad

def gradient_ascent_opt(centers, n, A_ub, idx_i, idx_j, rng, max_iter=4000):
    """Performs gradient ascent on centers to maximize sum of radii."""
    curr_c = centers.copy()
    curr_r, curr_s, _, _ = solve_lp_and_get_marginals(curr_c, n, A_ub, idx_i, idx_j)
    if curr_r is None:
        return curr_c, np.full(n, 1e-6), 0.0
        
    base_step = 0.008
    best_c, best_r, best_s = curr_c.copy(), curr_r.copy(), curr_s
    
    for it in range(max_iter):
        r, s, m_ineq, m_upper = solve_lp_and_get_marginals(curr_c, n, A_ub, idx_i, idx_j)
        if r is None: break
            
        grad = compute_gradient(curr_c, n, m_ineq, m_upper, idx_i, idx_j)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-9:
            # Inject jitter to escape plateau
            if rng.random() < 0.05:
                curr_c += rng.uniform(-0.005, 0.005, curr_c.shape)
                curr_c = np.clip(curr_c, 1e-4, 1.0-1e-4)
            continue
            
        step = base_step * (0.998 ** (it // 50))
        new_c = curr_c + step * grad / g_norm
        new_c = np.clip(new_c, 1e-4, 1.0-1e-4)
        
        r_new, s_new, _, _ = solve_lp_and_get_marginals(new_c, n, A_ub, idx_i, idx_j)
        if r_new is not None and s_new > s + 1e-9:
            curr_c = new_c
            curr_r = r_new
            curr_s = s_new
            if s_new > best_s + 1e-7:
                best_c, best_r, best_s = curr_c.copy(), curr_r.copy(), s_new
        else:
            # Try smaller step or random perturbation if gradient ascent stalls
            if rng.random() < 0.15:
                pert = curr_c + rng.uniform(-0.003, 0.003, curr_c.shape)
                pert = np.clip(pert, 1e-4, 1.0-1e-4)
                r_p, s_p, _, _ = solve_lp_and_get_marginals(pert, n, A_ub, idx_i, idx_j)
                if r_p is not None and s_p > curr_s:
                    curr_c, curr_r, curr_s = pert, r_p, s_p
                    if s_p > best_s + 1e-7:
                        best_c, best_r, best_s = curr_c.copy(), curr_r.copy(), s_p
                        
    return best_c, best_r, best_s

def penalty_objective(v, n, mu):
    """Smooth penalty objective for L-BFGS-B."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    obj = -np.sum(r)
    pen = 0.0
    pen += np.sum(np.maximum(0.0, r - cx)**2)
    pen += np.sum(np.maximum(0.0, r - (1.0-cx))**2)
    pen += np.sum(np.maximum(0.0, r - cy)**2)
    pen += np.sum(np.maximum(0.0, r - (1.0-cy))**2)
    idx = np.triu_indices(n, k=1)
    dx = cx[idx[0]] - cx[idx[1]]
    dy = cy[idx[0]] - cy[idx[1]]
    dist = np.sqrt(dx**2 + dy**2)
    rs = r[idx[0]] + r[idx[1]]
    pen += np.sum(np.maximum(0.0, rs - dist)**2)
    return obj + mu * pen

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
