

<strategy>
1. **Subgradient Ascent on LP Objective**: The sum of radii for fixed centers is a concave, piecewise-linear function. Its subgradient can be derived from the dual variables (marginals) of the linear program that computes optimal radii. I will use these subgradients to perform projected gradient ascent on the centers, which efficiently pushes circles apart to increase the sum of radii.
2. **Diverse Initializations**: Generate multiple hexagonal lattice patterns with varying row counts and initial radii, plus random starts. Normalize them to fit comfortably within the unit square.
3. **Adaptive Step Size & Shake Escape**: Use an adaptive step size for gradient ascent. When the gradient norm drops (plateau), apply random perturbations to a subset of circles to escape local minima, similar to simulated annealing.
4. **SLSQP Joint Polish**: After gradient ascent finds a good configuration, refine both centers and radii simultaneously using SLSQP to handle the non-linear constraints smoothly.
5. **Strict Safety Scaling**: Finally, compute the exact safety margin for boundaries and pairwise distances, and scale radii down slightly to guarantee strict validity against the 1e-12 tolerance in the validator.
6. **Robustness**: Handle scipy version differences for LP marginals, ensure numerical stability with small epsilons, and include multiple restarts from the best found state to maximize the chance of reaching the target sum of radii.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

def get_lp_and_grad(centers, n, idx_i, idx_j, A_ub):
    """Solves LP for optimal radii and computes subgradient w.r.t centers."""
    wall = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]), 
                      np.minimum(centers[:,1], 1.0 - centers[:,1]))
    bounds = [(0.0, max(w, 1e-12)) for w in wall]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, None, None
            
        r = np.maximum(res.x, 0.0)
        s = -res.fun
        
        # Compute subgradient from LP dual variables
        grad = np.zeros((n, 2))
        try:
            marginals = res.marginals.ineqlin
        except AttributeError:
            try:
                marginals = res.ineqlin.marginals
            except AttributeError:
                marginals = None
                
        if marginals is not None:
            active = marginals > 1e-8
            if np.any(active):
                k_idx = np.where(active)[0]
                i_a = idx_i[k_idx]
                j_a = idx_j[k_idx]
                d_a = b_ub[k_idx]
                lam = marginals[k_idx]
                
                d_safe = np.maximum(d_a, 1e-9)
                vec = diff[i_a, j_a] / d_safe[:, np.newaxis] * lam[:, np.newaxis]
                
                np.add.at(grad, i_a, vec)
                np.add.at(grad, j_a, -vec)
                
        return r, s, grad
    except Exception:
        return None, None, None

def objective_joint(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def constraints_joint(v, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    
    return c

def generate_inits(n, rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [4,6,6,6,4],
        [5,5,6,5,5], [5,7,5,5,4], [5,5,5,5,6], [6,5,5,5,5],
        [6,6,6,4,4], [5,6,4,6,5]
    ]
    
    for pat in patterns:
        if sum(pat) != n: continue
        for r0 in [0.085, 0.090, 0.095, 0.100]:
            pts = []
            y = r0
            for i, cnt in enumerate(pat):
                shift = r0 if i % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            pts = np.array(pts[:n])
            
            # Normalize to fit within [0.12, 0.88]
            mn = pts.min(axis=0)
            mx = pts.max(axis=0)
            span = mx - mn
            if span[0] > 0: pts[:, 0] = (pts[:, 0] - mn[0]) / span[0] * 0.76 + 0.12
            if span[1] > 0: pts[:, 1] = (pts[:, 1] - mn[1]) / span[1] * 0.76 + 0.12
            inits.append(pts)
            
            # Perturbed variant
            p = pts + rng.uniform(-0.025, 0.025, pts.shape)
            inits.append(np.clip(p, 0.08, 0.92))
            
    # Pure random starts
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    inits = generate_inits(n, rng)
    bounds_opt = [(0.0, 1.0)]*(2*n) + [(1e-5, 0.5)]*n
    
    # Phase 1: Gradient ascent from diverse starts
    for cfg in inits:
        c = cfg.copy()
        r, s, g = get_lp_and_grad(c, n, idx_i, idx_j, A_ub)
        if r is None: continue
        
        step = 0.015
        for it in range(2000):
            r, s, g = get_lp_and_grad(c, n, idx_i, idx_j, A_ub)
            if r is None: break
            
            gn = np.linalg.norm(g)
            if gn > 1e-5:
                c_new = c + step * g / gn
                c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
                r_new, s_new, _ = get_lp_and_grad(c_new, n, idx_i, idx_j, A_ub)
                
                if r_new is not None and s_new > s + 1e-8:
                    c = c_new
                    s = s_new
                    step = min(step * 1.008, 0.04)
                else:
                    step *= 0.96
            else:
                # Escape plateau with random shake
                if rng.random() < 0.08:
                    shake_idx = rng.choice(n, size=max(2, n//3), replace=False)
                    c_shake = c.copy()
                    c_shake[shake_idx] += rng.uniform(-0.015, 0.015, (len(shake_idx), 2))
                    c_shake = np.clip(c_shake, 0.08, 0.92)
                    r_sh, s_sh, _ = get_lp_and_grad(c_shake, n, idx_i, idx_j, A_ub)
                    if r_sh is not None and s_sh > s:
                        c = c_shake
                        s = s_sh
                        step = 0.015
                        
        if s > best_sum:
            best_sum = s
            best_c = c.copy()
            best_r = r.copy()
            
        # SLSQP polish for this configuration
        x0 = np.concatenate([c.flatten(), best_r * 0.97])
        try:
            res = minimize(objective_joint, x0, args=(n, idx_i, idx_j), method='SLSQP',
                           bounds=bounds_opt,
                           constraints={'type': 'ineq', 'fun': constraints_joint, 'args': (n, idx_i, idx_j)},
                           options={'maxiter': 3000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_pol = res.x[:2*n].reshape(n, 2)
                r_pol, s_pol, _ = get_lp_and_grad(c_pol, n, idx_i, idx_j, A_ub)
                if r_pol is not None and s_pol > best_sum:
                    best_sum = s_pol
                    best_c = c_pol.copy()
                    best_r = r_pol.copy()
        except Exception:
            pass

    # Phase 2: Refinement from best state via multiple perturbations
    if best_c is not None:
        for _ in range(15):
            c_try = best_c.copy()
            idx_pert = rng.choice(n, size=n//4, replace=False)
            c_try[idx_pert] += rng.uniform(-0.01, 0.01, (len(idx_pert), 2))
            c_try = np.clip(c_try, 0.05, 0.95)
            
            r, s, g = get_lp_and_grad(c_try, n, idx_i, idx_j, A_ub)
            if r is not None and s > best_sum:
                step = 0.01
                for it in range(800):
                    r, s, g = get_lp_and_grad(c_try, n, idx_i, idx_j, A_ub)
                    if r is None: break
                    gn = np.linalg.norm(g)
                    if gn > 1e-5:
                        c_new = c_try + step * g / gn
                        c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
                        r_new, s_new, _ = get_lp_and_grad(c_new, n, idx_i, idx_j, A_ub)
                        if r_new is not None and s_new > s + 1e-8:
                            c_try = c_new
                            s = s_new
                            step = min(step * 1.01, 0.03)
                        else:
                            step *= 0.95
                    else:
                        if rng.random() < 0.1:
                            idx_sh = rng.choice(n, size=3, replace=False)
                            c_try[idx_sh] += rng.uniform(-0.01, 0.01, (3, 2))
                            c_try = np.clip(c_try, 0.05, 0.95)
                            
                if s > best_sum:
                    best_sum = s
                    best_c = c_try.copy()
                    best_r = r.copy()

    # Phase 3: Strict numerical safety scaling
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    diff = best_c[:, np.newaxis, :] - best_c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    r_pair = best_r[:, np.newaxis] + best_r[np.newaxis, :]
    scale = min(scale, np.min(dists[idx_i, idx_j] / np.maximum(r_pair[idx_i, idx_j], 1e-12)))
    
    best_r *= scale * 0.999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
