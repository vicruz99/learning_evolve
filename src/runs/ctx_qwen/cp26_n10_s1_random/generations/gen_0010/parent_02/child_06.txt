2. **Dual-Variable Gradient Ascent**: The LP solver returns dual variables (marginals) for each constraint. This allows us to perform precise gradient ascent on the centers to increase the sum of radii.
3. **Force-Directed Untangling**: Before gradient ascent, a short vectorized force simulation resolves initial overlaps and spreads circles out, ensuring the LP starts from a feasible configuration with non-trivial radii.

5. **SLSQP Polishing**: After the LP-gradient phase converges, a constrained non-linear optimizer (`SLSQP`) performs a final joint polish of centers and radii.

import numpy as np
from scipy.optimize import linprog, minimize
import warnings
warnings.filterwarnings('ignore')

def compute_lp_data(centers):
    """Precompute distance matrix and wall limits for LP."""
    n = centers.shape[0]
    wall = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]), 
                      np.minimum(centers[:,1], 1.0-centers[:,1]))
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    return wall, dists

def solve_lp_radii(centers, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, and marginals."""
    n = centers.shape[0]
    wall, _ = compute_lp_data(centers)
    bounds = [(0.0, max(w, 1e-9)) for w in wall]
    dists = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            # Extract marginals (dual variables) safely across scipy versions
            marg = np.zeros(len(b_ub))
            if hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
                marg = res.marginals.ineqlin
            elif hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
                marg = res.ineqlin.marginals
            return res.x, -res.fun, np.abs(marg)
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0, np.zeros(len(b_ub))

def compute_center_gradient(centers, marginals, idx_i, idx_j):
    """Computes gradient of sum of radii w.r.t centers using LP marginals."""
    n = centers.shape[0]
    grad = np.zeros_like(centers)
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    active = marginals > 1e-8
    if np.any(active):
        idx_k = np.where(active)[0]
        i_k = idx_i[idx_k]
        j_k = idx_j[idx_k]
        lam = marginals[idx_k]
        d_k = dists[i_k, j_k]
        d_k = np.where(d_k > 1e-8, d_k, 1.0)
        
        # Gradient contribution: lam_ij * (c_i - c_j) / d_ij
        vec = diff[i_k, j_k] / d_k[:, None] * lam[:, None]
        for k in range(len(idx_k)):
            grad[i_k[k]] += vec[k]
            grad[j_k[k]] -= vec[k]
    return grad

def force_simulation(centers, steps=250):
    """Vectorized force-directed layout to resolve overlaps and spread circles."""
    n = centers.shape[0]
    radii = np.full(n, 0.085)
    vel = np.zeros_like(centers)
    dt = 0.005
    damp = 0.85
    
    for _ in range(steps):
        radii *= 1.0003
        diff = centers[:, None, :] - centers[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist, 1e9)
        
        overlap = np.maximum(0.0, radii[:, None] + radii[None, :] - dist)
        f_rep = overlap * 120.0 / (dist + 1e-9)
        f = np.sum(diff * f_rep[:, :, None], axis=1)
        
        # Wall repulsion
        mask_l = centers[:, 0] < radii
        mask_r = centers[:, 0] > 1.0 - radii
        mask_b = centers[:, 1] < radii
        mask_t = centers[:, 1] > 1.0 - radii
        f[mask_l, 0] += (radii[mask_l] - centers[mask_l, 0]) * 300.0
        f[mask_r, 0] -= (centers[mask_r, 0] + radii[mask_r] - 1.0) * 300.0
        f[mask_b, 1] += (radii[mask_b] - centers[mask_b, 1]) * 300.0
        f[mask_t, 1] -= (centers[mask_t, 1] + radii[mask_t] - 1.0) * 300.0
        
        vel = damp * vel + f * dt
        centers += vel
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    return centers

def hex_init(n, rows, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for i, cnt in enumerate(rows):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([rng.uniform(0.3, 0.7), rng.uniform(0.3, 0.7)])
    return np.array(pts[:n])

def objective_joint(v, n):
    return -np.sum(v[2*n:])

def constraints_joint(v, n, idx_i, idx_j):
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    dr = r[:, None] + r[None, :]
    c = np.concatenate([c, d2[idx_i, idx_j] - dr[idx_i, idx_j]**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    idx_i, idx_j = np.triu_indices(n, k=1)
    m_pairs = len(idx_i)
    A_ub = np.zeros((m_pairs, n))
    A_ub[np.arange(m_pairs), idx_i] = 1.0
    A_ub[np.arange(m_pairs), idx_j] = 1.0
    
    rng = np.random.default_rng(42)
    
    # Diverse structural patterns
    row_pats = [[6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], [7,5,5,5,4], [6,6,6,4,4], [5,6,4,6,5]]
    configs = []
    for pat in row_pats:
        for r0 in [0.085, 0.090, 0.095, 0.100]:
            c = hex_init(n, pat, r0, rng)
            mn, mx = c.min(axis=0), c.max(axis=0)
            c = (c - mn) / (mx - mn + 1e-9) * 0.82 + 0.09
            configs.append(c)
            configs.append(np.clip(c + rng.uniform(-0.025, 0.025, c.shape), 0.05, 0.95))
            
    for _ in range(6):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = 0.0
    best_c, best_r = None, None
    
    bounds_opt = [(0.0, 1.0)]*(2*n) + [(1e-5, 0.5)]*n
    
    for cfg in configs:
        c = force_simulation(cfg.copy(), steps=200)
        r, s, marg = solve_lp_radii(c, A_ub, idx_i, idx_j)
        
        # LP-Guided Gradient Ascent on Centers
        lr = 0.025
        for step in range(500):
            g = compute_center_gradient(c, marg, idx_i, idx_j)
            gn = np.linalg.norm(g)
            if gn < 1e-8: break
            
            c_new = c + lr * g / gn
            c_new = np.clip(c_new, 0.015, 0.985)
            
            r_new, s_new, marg_new = solve_lp_radii(c_new, A_ub, idx_i, idx_j)
            if s_new > s + 1e-9:
                c, r, s, marg = c_new, r_new, s_new, marg_new
                lr = min(lr * 1.08, 0.05)
            else:
                lr *= 0.88
                if lr < 1e-5: break
                
        if s > best_sum:
            best_sum = s
            best_c, best_r = c.copy(), r.copy()
            
        # SLSQP Joint Polish
        x0 = np.concatenate([c.flatten(), r * 0.995])
        try:
            res = minimize(objective_joint, x0, args=(n,), method='SLSQP', bounds=bounds_opt,
                           constraints={'type': 'ineq', 'fun': constraints_joint, 'args': (n, idx_i, idx_j)},
                           options={'maxiter': 2000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_p = res.x[:2*n].reshape(n, 2)
                r_p, s_p, _ = solve_lp_radii(c_p, A_ub, idx_i, idx_j)
                if s_p > best_sum:
                    best_sum = s_p
                    best_c, best_r = c_p.copy(), r_p.copy()
        except Exception:
            pass

    # Strict numerical safety scaling
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    diff = best_c[:, None, :] - best_c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    r_pair = best_r[:, None] + best_r[None, :]
    scale = min(scale, np.min(dists[idx_i, idx_j] / np.maximum(r_pair[idx_i, idx_j], 1e-12)))
    
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
