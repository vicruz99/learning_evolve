import numpy as np
from scipy.optimize import linprog, minimize

def get_lp_and_grad(centers):
    """Solves LP for radii given centers and computes gradient of optimal sum w.r.t centers."""
    n = centers.shape[0]
    idx_i, idx_j = np.triu_indices(n, k=1)
    m_pairs = len(idx_i)
    m_walls = 4 * n
    m_total = m_pairs + m_walls
    
    A = np.zeros((m_total, n))
    b = np.zeros(m_total)
    
    # Pairwise distance constraints: r_i + r_j <= d_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    A[np.arange(m_pairs), idx_i] = 1.0
    A[np.arange(m_pairs), idx_j] = 1.0
    b[:m_pairs] = dists[idx_i, idx_j]
    
    # Wall constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    A[m_pairs : m_pairs+n, np.arange(n)] = 1.0
    b[m_pairs : m_pairs+n] = centers[:, 0]
    A[m_pairs+n : m_pairs+2*n, np.arange(n)] = 1.0
    b[m_pairs+n : m_pairs+2*n] = 1.0 - centers[:, 0]
    A[m_pairs+2*n : m_pairs+3*n, np.arange(n)] = 1.0
    b[m_pairs+2*n : m_pairs+3*n] = centers[:, 1]
    A[m_pairs+3*n : m_pairs+4*n, np.arange(n)] = 1.0
    b[m_pairs+3*n : m_pairs+4*n] = 1.0 - centers[:, 1]
    
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    except Exception:
        return None, None, None
        
    if not res.success or not np.isfinite(res.fun):
        return None, None, None
        
    radii = res.x
    sum_r = -res.fun
    
    # Compute gradient using marginals (shadow prices)
    grad = np.zeros((n, 2))
    try:
        marg = res.ineqlin.marginals
    except AttributeError:
        marg = np.zeros(m_total)
        
    # Pairwise gradient contributions
    active_pairs = marg[:m_pairs] > 1e-6
    if np.any(active_pairs):
        k_idx = np.where(active_pairs)[0]
        lam = marg[:m_pairs][k_idx]
        ii = idx_i[k_idx]
        jj = idx_j[k_idx]
        d = dists[ii, jj]
        d = np.where(d < 1e-9, 1e-9, d)
        diff_k = centers[ii] - centers[jj]
        factor = lam / d
        for k in range(len(k_idx)):
            grad[ii[k]] += diff_k[k] * factor[k]
            grad[jj[k]] -= diff_k[k] * factor[k]
            
    # Wall gradient contributions
    active_x = marg[m_pairs : m_pairs+n] > 1e-6
    if np.any(active_x):
        idx = np.where(active_x)[0]
        grad[idx, 0] += marg[m_pairs : m_pairs+n][idx]
        
    active_1x = marg[m_pairs+n : m_pairs+2*n] > 1e-6
    if np.any(active_1x):
        idx = np.where(active_1x)[0]
        grad[idx, 0] -= marg[m_pairs+n : m_pairs+2*n][idx]
        
    active_y = marg[m_pairs+2*n : m_pairs+3*n] > 1e-6
    if np.any(active_y):
        idx = np.where(active_y)[0]
        grad[idx, 1] += marg[m_pairs+2*n : m_pairs+3*n][idx]
        
    active_1y = marg[m_pairs+3*n : m_pairs+4*n] > 1e-6
    if np.any(active_1y):
        idx = np.where(active_1y)[0]
        grad[idx, 1] -= marg[m_pairs+3*n : m_pairs+4*n][idx]
        
    return radii, sum_r, grad

def make_hex(n, pat, rng):
    """Generates a hexagonal lattice configuration scaled to fit in [0.1, 0.9]."""
    pts = []
    r0 = 0.095
    y = r0
    for i, cnt in enumerate(pat):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    pts = np.array(pts[:n])
    # Normalize to [0.1, 0.9] with slight random jitter
    mn = pts.min(axis=0)
    mx = pts.max(axis=0)
    span = mx - mn + 1e-9
    pts = (pts - mn) / span * 0.8 + 0.1
    pts += rng.uniform(-0.005, 0.005, pts.shape)
    return np.clip(pts, 0.05, 0.95)

def objective_slsqp(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_slsqp(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
