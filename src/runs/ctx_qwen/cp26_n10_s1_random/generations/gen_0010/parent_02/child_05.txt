import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]),
                      np.minimum(centers[:,1], 1.0-centers[:,1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    idx = np.triu_indices(n, k=1)
    m = len(idx[0])
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx[0]] = 1.0
    A_ub[np.arange(m), idx[1]] = 1.0
    b_ub = dists[idx]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def obj_penalty(v, n, mu):
    """Smooth penalty objective for L-BFGS-B: minimize -sum(r) + mu * violations^2."""
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary violations
    pen += np.sum(np.maximum(0.0, r - c[:,0])**2)
    pen += np.sum(np.maximum(0.0, r - (1.0-c[:,0]))**2)
    pen += np.sum(np.maximum(0.0, r - c[:,1])**2)
    pen += np.sum(np.maximum(0.0, r - (1.0-c[:,1]))**2)
    
    # Pairwise overlap violations
    diff = c[:, None, :] - c[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, 1e9)
    viol = np.maximum(0.0, r[:, None] + r[None, :] - dist)
    pen += np.sum(np.triu(viol**2, k=1))
    
    return obj + mu * pen

def generate_inits(n, rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # Hexagonal lattice patterns
    pats = [[6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], [4,6,6,6,4]]
    for pat in pats:
        pts = []
        r0 = 0.09
        y = r0
        for i, cnt in enumerate(pat):
            shift = r0 if i%2==1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        inits.append(np.array(pts[:n]))
        # Perturbed variants
        inits.append(inits[-1] + rng.uniform(-0.02, 0.02, (n, 2)))
        
    # Structured grid
    gx = np.linspace(0.15, 0.85, 6)
    gy = np.linspace(0.15, 0.85, 6)
    grid_pts = []
    for y in gy:
        for x in gx:
            if len(grid_pts) >= n: break
            grid_pts.append([x, y])
    inits.append(np.array(grid_pts[:n]))
    
    # Random placements
    for _ in range(8):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_c = None
    best_r = None
    
    inits = generate_inits(n, rng)
    bounds_opt = [(0.0, 1.0)] * (2 * n) + [(1e-4, 0.5)] * n
    
    # Phase 1: Penalty-based L-BFGS-B from diverse starts
    for cfg in inits:
        r_lp, _ = solve_lp(cfg)
        if r_lp is None:
            r_lp = np.full(n, 0.08)
        v0 = np.concatenate([cfg.flatten(), r_lp])
        
        try:
            res = minimize(obj_penalty, v0, args=(n, 20000.0), method='L-BFGS-B',

            v_opt = res.x
            c_opt = v_opt[:2 * n].reshape(n, 2)
            
            # Exact LP refinement for radii
            r_final, s_final = solve_lp(c_opt)
            if r_final is not None and s_final > best_sum:

                best_c = c_opt.copy()
                best_r = r_final.copy()
        except Exception:
            pass

    # Phase 2: LP-Guided Hill Climbing on Centers
    if best_c is not None:
        curr_c = best_c.copy()
        curr_s = best_sum
        step = 0.02
        for it in range(2500):
            i = rng.integers(n)
            old = curr_c[i].copy()
            curr_c[i] += rng.uniform(-step, step, 2)
            curr_c[i] = np.clip(curr_c[i], 1e-4, 1.0 - 1e-4)
            
            r_try, s_try = solve_lp(curr_c)
            if s_try > curr_s + 1e-9:
                curr_s = s_try
                best_sum = curr_s
                best_c = curr_c.copy()
                best_r = r_try.copy()
            else:
                curr_c[i] = old
            step *= 0.997  # Slow decay for fine-tuning

    # Phase 3: Strict numerical safety scaling
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)

    r_pair = best_r[idx[0]] + best_r[idx[1]]
    scale = min(scale, np.min(dists[idx] / np.maximum(r_pair, 1e-12)))
    
    best_r *= scale * 0.999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
