import numpy as np
from scipy.optimize import linprog, minimize

def compute_lp_and_grad(centers, n, idx_i, idx_j, A_ub):
    """Solves LP for max sum of radii and computes analytical gradient w.r.t centers."""
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(w, 1e-12)) for w in wall]
    
    diff = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    b_ub = dists
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success or not np.isfinite(res.fun):
        return None, None, None
        
    r = res.x
    s = -res.fun
    
    grad_c = np.zeros((n, 2))
    try:
        marginals = res.ineqlin.marginals
        # d(opt_val)/d(b_i) = marginal. Since we minimize -sum(r), d(sum(r))/d(b_i) = -marginal
        lam = np.maximum(-marginals, 0.0)
        active = lam > 1e-7
        if np.any(active):
            k_act = np.where(active)[0]
            i_a = idx_i[k_act]
            j_a = idx_j[k_act]
            d_a = dists[k_act]
            lam_a = lam[k_act]
            
            dir_vec = diff[k_act] / np.maximum(d_a[:, None], 1e-12)
            weights = lam_a[:, None]
            
            contrib = dir_vec * weights
            np.add.at(grad_c, i_a, contrib)
            np.add.at(grad_c, j_a, -contrib)
    except Exception:
        pass
        
    return r, s, grad_c

def hex_init(n, rows, r0, rng, angle=0.0):
    """Generates initial positions on a hexagonal lattice, optionally rotated."""
    pts = []
    y = r0
    for i, cnt in enumerate(rows):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    pts = np.array(pts[:n])
    
    if angle != 0.0:
        rad = np.deg2rad(angle)
        c = np.array([0.5, 0.5])
        pts = (pts - c) @ np.array([[np.cos(rad), -np.sin(rad)], [np.sin(rad), np.cos(rad)]]) + c
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def obj_slsqp(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_slsqp(v, n, idx_i, idx_j):
    """Inequality constraints >= 0 for SLSQP."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    rows_pats = [[6,5,6,5,4], [5,6,5,6,4], [6,6,4,6,4], [4,6,5,6,5], [5,5,6,5,5]]
    angles = [0, 10, 20, 30, 45]
    
    configs = []
    for pat in rows_pats:
        for r0 in [0.090, 0.095, 0.100]:
            for ang in angles:
                configs.append(hex_init(n, pat, r0, rng, ang))
            c_base = hex_init(n, pat, 0.095, rng, 0)

                c_pert = c_base + rng.uniform(-0.03, 0.03, c_base.shape)
                configs.append(np.clip(c_pert, 0.05, 0.95))
                
    for _ in range(8):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))

    for cfg in configs:
        c = cfg.copy()
        
        # Phase 1: LP Gradient Ascent on Centers
        step = 0.025

            if r is None: break

            if gn > 1e-8:
                c_new = c + step * g / gn
                c_new = np.clip(c_new, 1e-4, 0.999)
                r_new, s_new, _ = compute_lp_and_grad(c_new, n, idx_i, idx_j, A_ub)
                if r_new is not None and s_new > s + 1e-9:
                    c = c_new
                    step = min(step * 1.01, 0.05)
                else:
                    step *= 0.9
            else:
                if rng.random() < 0.2:
                    c += rng.uniform(-0.01, 0.01, c.shape)
                    c = np.clip(c, 0.05, 0.95)

        r_cur, s_cur, _ = compute_lp_and_grad(c, n, idx_i, idx_j, A_ub)

        curr_s = s_cur

                curr_s = s_try
                if s_try > best_sum:
                    best_sum = s_try
                    best_c = curr_c.copy()
                    best_r = r_try.copy()
            else:
                curr_c[i] = old
                
        # Phase 3: SLSQP Joint Polish
        if s_cur > best_sum * 0.99:
            x0 = np.concatenate([c.flatten(), r_cur * 0.98])
            try:
                res = minimize(obj_slsqp, x0, args=(n,), method='SLSQP', bounds=bounds_opt,

                    cy = res.x[n:2*n]
                    centers_opt = np.column_stack((cx, cy))
                    lp_r, lp_s, _ = compute_lp_and_grad(centers_opt, n, idx_i, idx_j, A_ub)
                    if lp_r is not None and lp_s > best_sum:
                        best_sum = lp_s
                        best_c = centers_opt.copy()
                        best_r = lp_r.copy()
            except Exception:
                pass
                
    if best_c is None:
        best_c = configs[0]
        best_r = np.full(n, 0.085)
        best_sum = np.sum(best_r)
        
    # Final safety scaling to strictly satisfy 1e-12 validator tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
            
    diff = best_c[:, None, :] - best_c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    r_pair = best_r[:, None] + best_r[None, :]
    scale = min(scale, np.min(dists[idx_i, idx_j] / np.maximum(r_pair[idx_i, idx_j], 1e-12)))
    
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
