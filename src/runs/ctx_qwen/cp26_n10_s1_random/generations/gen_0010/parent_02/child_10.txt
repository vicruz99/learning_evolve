import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

def solve_lp(centers, A_ub, idx_i, idx_j, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(w, 1e-9)) for w in wall]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def obj_centers(x_flat, n, A_ub, idx_i, idx_j):
    """Objective for center optimization: minimize negative LP sum of radii."""
    c = x_flat.reshape(n, 2)
    _, s = solve_lp(c, A_ub, idx_i, idx_j, n)
    return -s

def constraints_joint(v, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid packing (centers and radii)."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def objective_joint(v, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Diverse hexagonal row distributions summing to 26
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,4,6,4], [4,6,5,6,5], 
        [5,5,6,5,5], [6,5,5,6,4], [5,4,6,6,5], [6,4,5,6,5],
        [5,6,4,6,5], [4,5,7,5,5], [5,7,4,5,5], [6,6,5,5,4],
        [5,5,5,5,6], [6,5,6,4,5], [5,6,6,5,4]
    ]
    
    starts = []
    for pat in patterns:
        for r0 in [0.092, 0.096, 0.100]:
            pts = []
            y = r0
            for i, cnt in enumerate(pat):
                shift = r0 if i % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            pts = np.array(pts[:n])
            mn = pts.min(axis=0)
            mx = pts.max(axis=0)
            span = mx - mn
            if span[0] > 0: pts[:, 0] = (pts[:, 0] - mn[0]) / span[0] * 0.8 + 0.1
            if span[1] > 0: pts[:, 1] = (pts[:, 1] - mn[1]) / span[1] * 0.8 + 0.1
            starts.append(pts.flatten())
            
    # Add controlled random starts
    for _ in range(12):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)).flatten())
        
    # Phase 1: Nelder-Mead on centers using LP objective
    for x0 in starts:
        try:
            res = minimize(obj_centers, x0, args=(n, A_ub, idx_i, idx_j), 
                           method='Nelder-Mead', 
                           options={'maxiter': 2500, 'fatol': 1e-9, 'xatol': 1e-9})
            c_opt = res.x.reshape(n, 2)
            r_lp, s_lp = solve_lp(c_opt, A_ub, idx_i, idx_j, n)
            if r_lp is not None and s_lp > best_sum:
                best_sum = s_lp
                best_c = c_opt.copy()
                best_r = r_lp.copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = starts[0].reshape(n, 2)
        best_r, best_sum = solve_lp(best_c, A_ub, idx_i, idx_j, n)
        
    # Phase 2: Joint SLSQP polish from best centers
    bounds_opt = [(0.0, 1.0)]*(2*n) + [(1e-6, 0.5)]*n
    x0_jt = np.concatenate([best_c[:,0], best_c[:,1], best_r * 0.99])
    
    try:
        res_jt = minimize(objective_joint, x0_jt, args=(n,), method='SLSQP', bounds=bounds_opt,
                           constraints={'type': 'ineq', 'fun': constraints_joint, 'args': (n, idx_i, idx_j)},
                           options={'maxiter': 5000, 'ftol': 1e-12})
        if np.isfinite(res_jt.fun):
            if np.min(constraints_joint(res_jt.x, n, idx_i, idx_j)) > -1e-5:
                cx = res_jt.x[:n]
                cy = res_jt.x[n:2*n]
                c_pol = np.column_stack((cx, cy))
                r_final, s_final = solve_lp(c_pol, A_ub, idx_i, idx_j, n)
                if s_final > best_sum:
                    best_sum = s_final
                    best_c = c_pol.copy()
                    best_r = r_final.copy()
    except Exception:
        pass
        
    # Phase 3: Adaptive Hill Climbing on centers
    curr_c = best_c.copy()
    curr_s = best_sum
    for step in range(2000):
        scale = 0.006 * (1.0 - step/2000.0)**0.4
        i = rng.integers(n)
        old = curr_c[i].copy()
        curr_c[i] += rng.uniform(-scale, scale, 2)
        curr_c[i] = np.clip(curr_c[i], 1e-4, 1.0 - 1e-4)
        
        r_new, s_new = solve_lp(curr_c, A_ub, idx_i, idx_j, n)
        if s_new > curr_s + 1e-9:
            curr_s = s_new
            best_c = curr_c.copy()
            best_r = r_new.copy()
            best_sum = curr_s
        else:
            curr_c[i] = old
            
    # Phase 4: Strict numerical safety scaling
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0 - x)/r, y/r, (1.0 - y)/r)
            
    diff = best_c[:, np.newaxis, :] - best_c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_pair = best_r[:, np.newaxis] + best_r[np.newaxis, :]
    scale = min(scale, np.min(dists[idx_i, idx_j] / np.maximum(r_pair[idx_i, idx_j], 1e-12)))
    
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
