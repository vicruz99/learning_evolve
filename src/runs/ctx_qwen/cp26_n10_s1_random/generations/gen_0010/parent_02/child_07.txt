import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers, A_lp, idx_i, idx_j, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    wall = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]), 
                      np.minimum(centers[:,1], 1.0-centers[:,1]))
    bounds = [(0.0, max(w, 1e-12)) for w in wall]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_lp, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def obj_clearance(v, n):
    """Objective for Nelder-Mead: maximize minimum clearance."""
    c = v.reshape(n, 2)
    w = np.minimum(np.minimum(c[:,0], 1.0-c[:,0]), np.minimum(c[:,1], 1.0-c[:,1]))
    d = np.sqrt(np.sum((c[:,None,:] - c[None,:,:])**2, axis=2))
    np.fill_diagonal(d, 1e9)
    return -np.min(np.min(w), np.min(d)/2.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m_pairs = len(idx_i)
    A_lp = np.zeros((m_pairs, n))
    A_lp[np.arange(m_pairs), idx_i] = 1.0
    A_lp[np.arange(m_pairs), idx_j] = 1.0
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # 1. Generate diverse initial configurations
    configs = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [6,6,4,6,4], [4,6,5,6,5], [5,5,6,5,5]]
    for pat in pats:
        for r0 in [0.095, 0.100, 0.105]:
            pts = []
            y = r0
            for i, cnt in enumerate(pat):
                shift = r0 if i % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            while len(pts) < n:
                pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
            cfg = np.array(pts[:n])
            mn, mx = cfg.min(axis=0), cfg.max(axis=0)
            cfg = (cfg - mn) / (mx - mn + 1e-9) * 0.7 + 0.15
            configs.append(cfg)
            for _ in range(2):
                configs.append(np.clip(cfg + rng.uniform(-0.02, 0.02, cfg.shape), 0.05, 0.95))
                
    for _ in range(10):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # 2. Optimize each configuration
    for cfg in configs:
        # Clearance optimization to spread circles evenly
        res = minimize(obj_clearance, cfg.flatten(), args=(n,), method='Nelder-Mead',
                       options={'maxiter': 2000, 'xatol': 1e-9, 'fatol': 1e-10})
        c_opt = np.clip(res.x.reshape(n, 2), 0.01, 0.99)
        r_lp, s_lp = solve_lp(c_opt, A_lp, idx_i, idx_j, n)
        
        curr_c = c_opt.copy()
        curr_s = s_lp
        curr_r = r_lp.copy() if s_lp > 0 else np.full(n, 1e-6)
        
        # Hill climbing with cooling
        for it in range(1500):
            scale = 0.018 * (1.0 - it / 1500.0)**0.5
            i = rng.integers(n)
            old = curr_c[i].copy()
            curr_c[i] += rng.uniform(-scale, scale, 2)
            curr_c[i] = np.clip(curr_c[i], 0.005, 0.995)
            
            r_new, s_new = solve_lp(curr_c, A_lp, idx_i, idx_j, n)
            if s_new > curr_s + 1e-8:
                curr_s = s_new
                curr_r = r_new.copy()
            else:
                curr_c[i] = old
                
        if curr_s > best_sum:
            best_sum = curr_s
            best_c = curr_c.copy()
            best_r = curr_r.copy()
            
        # 3. Multi-point basin hopping from best found
        for _ in range(15):
            p_c = best_c.copy()
            idx = rng.choice(n, size=5, replace=False)
            p_c[idx] += rng.uniform(-0.012, 0.012, (5, 2))
            p_c = np.clip(p_c, 0.05, 0.95)
            
            res2 = minimize(obj_clearance, p_c.flatten(), args=(n,), method='Nelder-Mead',
                            options={'maxiter': 600})
            c_ref = np.clip(res2.x.reshape(n, 2), 0.01, 0.99)
            r_ref, s_ref = solve_lp(c_ref, A_lp, idx_i, idx_j, n)
            
            if s_ref > best_sum + 1e-6:
                temp_c = c_ref.copy()
                temp_s = s_ref
                temp_r = r_ref.copy()
                for _ in range(600):
                    sc = 0.008 * (1.0 - _ / 600.0)
                    idx2 = rng.integers(n)
                    old2 = temp_c[idx2].copy()
                    temp_c[idx2] += rng.uniform(-sc, sc, 2)
                    temp_c[idx2] = np.clip(temp_c[idx2], 0.005, 0.995)
                    r_t, s_t = solve_lp(temp_c, A_lp, idx_i, idx_j, n)
                    if s_t > temp_s + 1e-8:
                        temp_s = s_t
                        temp_r = r_t.copy()
                    else:
                        temp_c[idx2] = old2
                        
                if temp_s > best_sum:
                    best_sum = temp_s
                    best_c = temp_c.copy()
                    best_r = temp_r.copy()

    # 4. Strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    diff = best_c[:, None, :] - best_c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    r_pair = best_r[:, None] + best_r[None, :]
    scale = min(scale, np.min(dists[idx_i, idx_j] / np.maximum(r_pair[idx_i, idx_j], 1e-12)))
    
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
