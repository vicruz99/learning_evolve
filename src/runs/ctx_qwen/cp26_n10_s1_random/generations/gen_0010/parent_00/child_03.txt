import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    bounds = []
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(lim, 1e-9)))
        
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def lp_objective(x_flat, n):
    """Objective for center optimization: maximize LP sum of radii."""
    centers = np.clip(x_flat.reshape(n, 2), 1e-4, 1.0 - 1e-4)
    _, s = solve_lp_radii(centers)
    return -s

def slqp_objective(x, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(x[2 * n:])

def slqp_constraints(x, n, triu_i, triu_j):

    cx = x[:n]
    cy = x[n:2 * n]
    r = x[2 * n:]

    dx = cx[triu_i] - cx[triu_j]
    dy = cy[triu_i] - cy[triu_j]
    dr = r[triu_i] + r[triu_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def force_spread(cfg, steps=500):

    n = len(cfg)
    c = cfg.copy()
    r = np.full(n, 0.08)
    v = np.zeros_like(c)
    dt = 0.003
    damp = 0.90

        f = np.zeros_like(c)

        np.fill_diagonal(dists, np.inf)

        f[:, 1] += np.sum(diff[:, :, 1] * force_mat, axis=1)

        f[:, 0] -= np.clip(c[:, 0] + r - 1.0, 0, None) * 40.0
        f[:, 1] += np.clip(r - c[:, 1], 0, None) * 40.0
        f[:, 1] -= np.clip(c[:, 1] + r - 1.0, 0, None) * 40.0
        
        v = damp * v + f * dt
        c += v
        c = np.clip(c, 0.02, 0.98)
    return c
