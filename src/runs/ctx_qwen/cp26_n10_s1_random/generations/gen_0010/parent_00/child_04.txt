import numpy as np
from scipy.optimize import minimize, linprog

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
N_PAIRS = len(IDX_I)

# Precompute constant structure of the LP constraint matrix
A_UB = np.zeros((N_PAIRS, N))
A_UB[np.arange(N_PAIRS), IDX_I] = 1.0
A_UB[np.arange(N_PAIRS), IDX_J] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, and gradient."""
    c_obj = -np.ones(N)
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    bounds = [(0.0, lim) for lim in limits]
    
    diff = centers[IDX_I] - centers[IDX_J]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    
    res = linprog(c_obj, A_ub=A_UB, b_ub=dists, bounds=bounds, method='highs')
    if not res.success:
        return np.full(N, 1e-9), 0.0, None
    
    r = res.x
    s = -res.fun
    grad = np.zeros((N, 2))
    
    # Extract marginals for gradient computation
    try:
        marg = res.ineqlin.marginals
    except AttributeError:
        try:
            marg = res.marginals.ineqlin
        except AttributeError:
            marg = None
            
    if marg is not None:
        active = marg > 1e-8
        if np.any(active):
            d_a = dists[active]
            mu = marg[active]
            i_a = IDX_I[active]
            j_a = IDX_J[active]
            
            d_safe = np.where(d_a < 1e-12, 1.0, d_a)
            diff_a = centers[i_a] - centers[j_a]
            vec = diff_a * (mu / d_safe)[:, np.newaxis]
            
            np.add.at(grad, i_a, vec)
            np.add.at(grad, j_a, -vec)
            
    return r, s, grad

def get_objective(vars_arr):
    return -np.sum(vars_arr[2 * N:])

def get_constraints(vars_arr):
    x = vars_arr[:N]
    y = vars_arr[N:2 * N]
    r = vars_arr[2 * N:]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[IDX_I] - x[IDX_J]
    dy = y[IDX_I] - y[IDX_J]
    dr = r[IDX_I] + r[IDX_J]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def gradient_ascent(c, rng, max_iter=150):
    """Gradient ascent on centers using LP marginals."""
    r, s, g = solve_lp(c)
    step = 0.025
    for _ in range(max_iter):
        if g is None or np.linalg.norm(g) < 1e-9:
            break
        dir_n = g / np.linalg.norm(g)
        c_new = np.clip(c + step * dir_n, 1e-4, 1.0 - 1e-4)
        r_new, s_new, g_new = solve_lp(c_new)
        
        if s_new > s + 1e-9:
            c, s, g = c_new, s_new, g_new
            step = min(step * 1.03, 0.06)
        else:
            step *= 0.92
    return c

def coordinate_hill_climb(c, rng, max_iter=400):
    """Coordinate-wise perturbation to escape local minima."""
    r, s, _ = solve_lp(c)
    step = 0.015
    for _ in range(max_iter):
        i = rng.integers(N)
        old_c = c[i].copy()
        c[i] = np.clip(old_c + rng.uniform(-step, step, 2), 1e-4, 1.0 - 1e-4)
        
        r_new, s_new, _ = solve_lp(c)
        if s_new > s + 1e-9:
            s = s_new
        else:
            c[i] = old_c
        step *= 0.995
    return c

def generate_starts(rng):
    """Generates diverse initial center configurations."""
    starts = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4]
    ]
    
    for pat in patterns:
        if sum(pat) != N:
            continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N:
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        starts.append(np.array(pts[:N]))
        
    # Random and corner-biased starts
    for _ in range(12):
        starts.append(rng.uniform(0.12, 0.88, (N, 2)))
        
    # Corner bias: place some circles near corners, rest random
    for _ in range(6):
        c = rng.uniform(0.15, 0.85, (N, 2))
        c[:4] = rng.uniform(0.05, 0.25, (4, 2))
        c[4:8] = rng.uniform(0.75, 0.95, (4, 2))
        starts.append(c)
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_vars = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    starts = generate_starts(rng)
    
    for cfg in starts:
        # Phase 1: Gradient ascent to quickly reach a local peak
        c_curr = gradient_ascent(cfg.copy(), rng, max_iter=120)
        
        # Phase 2: Coordinate hill-climbing to break symmetries
        c_curr = coordinate_hill_climb(c_curr, rng, max_iter=300)
        
        r_curr, s_curr, _ = solve_lp(c_curr)
        
        # Phase 3: Joint SLSQP polish
        r_init = np.full(N, 0.08)
        for i in range(N):
            dw = min(c_curr[i, 0], 1.0 - c_curr[i, 0], c_curr[i, 1], 1.0 - c_curr[i, 1])
            d_nn = np.min(np.linalg.norm(c_curr - c_curr[i], axis=1))
            r_init[i] = min(dw, d_nn / 2.0) * 0.85
            
        x0 = np.concatenate([c_curr[:, 0], c_curr[:, 1], r_init])
        
        try:
            res = minimize(get_objective, x0, method='SLSQP', bounds=bounds_vars,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_pol = res.x[:2 * N].reshape(N, 2)
                r_pol, s_pol, _ = solve_lp(c_pol)
                if s_pol > s_curr:
                    s_curr = s_pol
                    c_curr = c_pol
        except Exception:
            pass
            
        if s_curr > best_sum:
            best_sum = s_curr
            best_centers = c_curr.copy()
            best_radii = r_curr.copy()
            
    # Fallback safety net
    if best_centers is None:
        best_centers = np.random.uniform(0.1, 0.9, (N, 2))
        best_radii, best_sum, _ = solve_lp(best_centers)
        
    # Phase 4: Final exact LP squeeze and strict safety scaling
    final_r, final_s, _ = solve_lp(best_centers)
    if final_r is not None:
        best_radii = final_r
        best_sum = final_s

    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0],
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
