import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers, idx_i, idx_j, A_ub_pair):
    """Solves LP to maximize sum of radii for fixed centers.
    Returns radii, sum_radii, marginal values, pairwise distances."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub_pair, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            r = res.x
            s = -res.fun
            try:
                lams = np.abs(res.ineqlin.marginals)
            except:
                lams = np.zeros(len(idx_i))
            return r, s, lams, b_ub
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0, np.zeros(len(idx_i)), b_ub

def get_lp_gradient(centers, n, lams, b_ub, idx_i, idx_j):
    """Computes gradient of LP sum of radii w.r.t centers using dual variables."""
    grad = np.zeros((n, 2))
    valid = lams > 1e-8
    if np.any(valid):
        ii = idx_i[valid]
        jj = idx_j[valid]
        d = b_ub[valid]
        lam = lams[valid]
        diff = centers[ii] - centers[jj]
        d_safe = np.maximum(d, 1e-12)
        vec = (diff / d_safe[:, np.newaxis]) * lam[:, np.newaxis]
        for k in range(len(valid)):
            grad[ii[k]] += vec[k]
            grad[jj[k]] -= vec[k]
    return grad

def obj_slsqp(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def cons_slsqp(vars_arr, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid packing."""
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def make_hex_config(pat, r0, rng):
    """Generates a hexagonal lattice configuration from a row pattern."""
    n = 26
    pts = []
    y = r0
    for idx, cnt in enumerate(pat):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
    pts = np.array(pts[:n])
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    idx_i, idx_j = np.triu_indices(n, k=1)
    m_pairs = len(idx_i)
    A_ub_pair = np.zeros((m_pairs, n))
    A_ub_pair[np.arange(m_pairs), idx_i] = 1.0
    A_ub_pair[np.arange(m_pairs), idx_j] = 1.0
    
    bounds_vars = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    cons_dict = {'type': 'ineq', 'fun': cons_slsqp, 'args': (n, idx_i, idx_j)}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Generate diverse initial configurations
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,5,5], [5,5,6,5,5], 
        [6,6,5,5,4], [6,5,4,6,5], [5,4,6,6,5], [5,6,6,5,4]
    ]
    starts = []
    for pat in patterns:
        if sum(pat) != n: continue
        cfg = make_hex_config(pat, 0.095, rng)
        starts.append(cfg)
        for _ in range(6):
            p = cfg + rng.uniform(-0.025, 0.025, cfg.shape)
            p = np.clip(p, 0.04, 0.96)
            starts.append(p)
            
    for _ in range(12):
        starts.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # Phase 2: LP Gradient Ascent on Centers
    for cfg in starts:
        c_curr = cfg.copy()
        r_curr, s_curr, lams, b_ub = solve_lp(c_curr, idx_i, idx_j, A_ub_pair)
        step = 0.03

            gn = np.linalg.norm(grad)

                    r_curr = r_new
                    s_curr = s_new

                else:
                    step *= 0.90
            else:
                c_curr += rng.uniform(-0.01, 0.01, c_curr.shape)
                c_curr = np.clip(c_curr, 0.05, 0.95)
                
        if s_curr > best_sum:
            best_sum = s_curr
            best_centers = c_curr.copy()
            best_radii = r_curr.copy()

    start_sums = []

            if np.isfinite(res.fun):

                cy = res.x[n:2*n]
                c_mat = np.column_stack((cx, cy))
                r_lp, s_lp, _, _ = solve_lp(c_mat, idx_i, idx_j, A_ub_pair)

                    best_sum = s_lp
                    best_centers = c_mat.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass
            
    # Phase 4: Stochastic Hill Climbing on Centers (LP evaluation)
    if best_centers is not None:
        c_opt = best_centers.copy()
        s_opt = best_sum
        step = 0.015
        for it in range(6000):
            i = rng.integers(n)
            old = c_opt[i].copy()

            c_opt[i] = np.clip(c_opt[i], 1e-4, 1.0-1e-4)
            
            r_try, s_try, _, _ = solve_lp(c_opt, idx_i, idx_j, A_ub_pair)
            if s_try > s_opt + 1e-9:
                s_opt = s_try
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_try.copy()
                step = min(step * 1.005, 0.04)
            else:
                c_opt[i] = old
                step *= 0.997
                
    # Fallback safety net
    if best_centers is None:
        best_centers = starts[0].copy()
        best_radii, best_sum, _, _ = solve_lp(best_centers, idx_i, idx_j, A_ub_pair)
        
    # Phase 5: Strict Safety Scaling
    scale = 1.0
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], 
                         best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
