

<strategy>
1. **Robust LP Solver**: Maintain the efficient `linprog` (HiGHS) to compute optimal radii for fixed centers. This guarantees we always extract the maximum possible sum for any given layout.
2. **Derivative-Free Global/Local Search**: Apply `scipy.optimize.minimize` with `Nelder-Mead` directly on the LP objective. Since the LP objective is non-smooth but piecewise linear, derivative-free methods work exceptionally well. Centers are clipped inside the objective to naturally respect boundaries.
3. **Aggressive Coordinate Descent (Hill Climbing)**: After each optimization pass, run an extended coordinate descent (2500+ steps). We perturb one circle at a time and accept moves that strictly increase the LP sum. The step size decays slowly to allow fine-grained adjustments near the optimum.
4. **Joint SLSQP Polish**: Run a constrained joint optimization (`SLSQP`) on centers and radii simultaneously. This smooths out any remaining kinks from the non-smooth LP objective and pushes circles tightly against all active constraints.
5. **Multiple Restarts & Perturbations**: Generate diverse initial hexagonal patterns and random starts. Run the full pipeline on each, keeping the best. After SLSQP, run another hill-climb pass to capture any residual gains.
6. **Strict Safety Scaling**: Finally, scale down radii minimally to guarantee numerical validity against the `1e-12` tolerance in the validator, ensuring a valid submission while preserving the maximized sum.
</strategy>
```python
import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        lim = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def objective_centers(c_flat, n):
    """Objective for center optimization: minimize negative LP sum of radii."""
    c = c_flat.reshape(n, 2)
    c = np.clip(c, 1e-4, 1.0 - 1e-4)
    r, s = solve_lp_radii(c)
    return -s

def hill_climb_optimization(centers, n, rng, steps=3000):
    """Coordinate descent on centers to maximize LP sum of radii."""
    current_c = centers.copy()
    r, best_s = solve_lp_radii(current_c)
    
    for step in range(steps):
        i = rng.integers(n)
        old_c = current_c[i].copy()
        
        # Adaptive step size with slow decay for fine-tuning
        step_size = 0.012 * (0.998 ** step)
        new_c = old_c + rng.uniform(-step_size, step_size, 2)
        new_c = np.clip(new_c, 1e-4, 1.0 - 1e-4)
        
        current_c[i] = new_c
        r_try, s_try = solve_lp_radii(current_c)
        
        if s_try > best_s:
            best_s = s_try
            r = r_try.copy()
        else:
            current_c[i] = old_c
            
    return current_c, r, best_s

def obj_joint(x, n):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def cons_joint(x, n):
    """Inequality constraints >= 0 for joint SLSQP."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    
    c_list = []
    c_list.append(cx - r)
    c_list.append(1.0 - cx - r)
    c_list.append(cy - r)
    c_list.append(1.0 - cy - r)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c_list.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(c_list)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Diverse hexagonal row patterns summing to 26
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,5,5], 
        [5,5,6,5,5], [6,6,5,5,4], [6,5,4,6,5],
        [5,5,5,5,6], [6,4,6,5,5], [4,5,6,6,5], [5,6,4,6,5]
    ]
    
    starts = []
    for pat in patterns:
        if sum(pat) != n: 
            continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: 
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        starts.append(np.array(pts[:n]))
        
    # Add random starts
    for _ in range(12):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    bounds_j = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    for cfg in starts:
        # Phase 1: Nelder-Mead optimization on centers
        try:
            res_nm = minimize(objective_centers, cfg.flatten(), args=(n,), 
                              method='Nelder-Mead', 
                              options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10})
            c_opt = np.clip(res_nm.x.reshape(n, 2), 0.01, 0.99)
            r_opt, s_opt = solve_lp_radii(c_opt)
            
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            pass
            
        # Phase 2: Hill climbing refinement
        try:
            c_hc, r_hc, s_hc = hill_climb_optimization(cfg, n, rng, steps=2500)
            if s_hc > best_sum:
                best_sum = s_hc
                best_centers = c_hc.copy()
                best_radii = r_hc.copy()
        except Exception:
            pass

    # Phase 3: Joint SLSQP polish from the best configuration found
    if best_centers is not None:
        # Perturb slightly to avoid staying exactly on a flat region
        pert_centers = best_centers + rng.uniform(-0.002, 0.002, best_centers.shape)
        pert_centers = np.clip(pert_centers, 0.02, 0.98)
        
        x0 = np.concatenate([pert_centers[:,0], pert_centers[:,1], best_radii * 0.97])
        
        try:
            res_j = minimize(obj_joint, x0, args=(n,), method='SLSQP', bounds=bounds_j,
                             constraints={'type': 'ineq', 'fun': cons_joint, 'args': (n,)},
                             options={'maxiter': 8000, 'ftol': 1e-13})
            if np.isfinite(res_j.fun):
                cx = res_j.x[:n]
                cy = res_j.x[n:2*n]
                c_mat = np.column_stack((cx, cy))
                r_lp, s_lp = solve_lp_radii(c_mat)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c_mat.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass
            
        # One more hill climb from the SLSQP result to capture residual gains
        try:
            c_hc2, r_hc2, s_hc2 = hill_climb_optimization(best_centers, n, rng, steps=1500)
            if s_hc2 > best_sum:
                best_sum = s_hc2
                best_centers = c_hc2.copy()
                best_radii = r_hc2.copy()
        except Exception:
            pass

    # Fallback if all else fails
    if best_centers is None:
        grid = np.array([(i * 0.18 + 0.1, j * 0.18 + 0.1) for j in range(5) for i in range(5)] + [[0.5, 0.5]])
        best_centers = grid[:n]
        best_radii, best_sum = solve_lp_radii(best_centers)

    # Phase 4: Final strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0 - x)/r, y/r, (1.0 - y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i,0] - best_centers[j,0], 
                         best_centers[i,1] - best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
```
