import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_radii(centers, return_marginals=False):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in limits]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        r = res.x
        s = -res.fun
        if return_marginals:
            try:
                lams = np.abs(res.marginals.ineqlin)
            except AttributeError:
                try:
                    lams = np.abs(res.ineqlin.marginals)
                except Exception:
                    lams = np.zeros(m)
            return r, s, lams
        return r, s
    return np.full(n, 1e-9), 0.0, np.zeros(m)

def lp_gradient_ascent(centers, steps=1500):
    """Moves centers using LP dual gradients to increase sum of radii."""
    n = centers.shape[0]
    c = centers.copy()
    idx_i, idx_j = np.triu_indices(n, k=1)
    step_size = 0.008
    
    for it in range(steps):
        r, s, lams = solve_lp_radii(c, return_marginals=True)
        
        grad = np.zeros((n, 2))
        valid = lams > 1e-7
        if np.any(valid):
            v_idx = np.where(valid)[0]
            lams_v = lams[v_idx]
            i_a = idx_i[v_idx]
            j_a = idx_j[v_idx]
            diffs = c[i_a] - c[j_a]
            d_a = np.sqrt(np.sum(diffs**2, axis=1))
            d_a = np.where(d_a < 1e-8, 1.0, d_a)
            factors = lams_v[:, np.newaxis] / d_a[:, np.newaxis]
            
            for k in range(len(v_idx)):
                grad[i_a[k]] += diffs[k] * factors[k]
                grad[j_a[k]] -= diffs[k] * factors[k]
                
        # Boundary repulsion forces
        for i in range(n):
            x, y = c[i]
            if x < r[i] + 0.03: grad[i, 0] += 10.0
            if x > 1.0 - r[i] - 0.03: grad[i, 0] -= 10.0
            if y < r[i] + 0.03: grad[i, 1] += 10.0
            if y > 1.0 - r[i] - 0.03: grad[i, 1] -= 10.0
            
        gn = np.linalg.norm(grad)
        if gn > 1e-6:
            c += step_size * grad / gn
            c = np.clip(c, 1e-4, 1.0 - 1e-4)
        step_size *= 0.997
        
    return c

def simulate_forces(cfg, steps=2000):
    """Physics simulation with growing radii to find tight packings."""
    n = len(cfg)
    c = cfg.copy()
    r = np.full(n, 0.015)
    v = np.zeros_like(c)
    dt = 0.0015
    damp = 0.93
    
    for _ in range(steps):
        r += 0.000035
        f = np.zeros_like(c)
        
        diff = c[:, None, :] - c[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        req = r[:, None] + r[None, :]
        overlap = np.maximum(0.0, req - dists)
        safe_d = np.where(dists < 1e-9, 1e-9, dists)
        force_mat = (overlap / safe_d) * 60.0
        f[:, 0] += np.sum(diff[:,:,0] * force_mat, axis=1)
        f[:, 1] += np.sum(diff[:,:,1] * force_mat, axis=1)
        
        f[:, 0] += np.clip(r - c[:, 0], 0, None) * 30.0
        f[:, 0] -= np.clip(c[:, 0] + r - 1.0, 0, None) * 30.0
        f[:, 1] += np.clip(r - c[:, 1], 0, None) * 30.0
        f[:, 1] -= np.clip(c[:, 1] + r - 1.0, 0, None) * 30.0
        
        v = damp * v + f * dt
        c += v
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
        
    return c, r

def get_constraints(vars_arr, n):
    """Inequality constraints >= 0 for valid packing."""
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def get_objective(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

    return best_centers, best_radii, best_sum
