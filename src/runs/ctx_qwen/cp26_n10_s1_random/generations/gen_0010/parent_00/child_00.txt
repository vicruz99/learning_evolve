import numpy as np
from scipy.optimize import minimize, linprog

def compute_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lims[i]) for i in range(n)]
    
    ii, jj = np.triu_indices(n, k=1)
    m = len(ii)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), ii] = 1.0
    A_ub[np.arange(m), jj] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    b_ub = dists[ii, jj]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None
            
        r = res.x
        s = -res.fun
        grad = np.zeros((n, 2))
        
        # Extract pair-wise marginals
        lams = None
        try:
            lams = res.marginals.ineqlin
        except AttributeError:
            try:
                lams = res.ineqlin.marginals
            except AttributeError:
                pass
                
        if lams is not None:
            active = np.abs(lams) > 1e-7
            if np.any(active):
                idx = np.where(active)[0]
                lams_a = np.abs(lams[idx])
                ii_a = ii[idx]
                jj_a = jj[idx]
                d_a = b_ub[idx]
                d_a = np.where(d_a > 1e-9, d_a, 1.0)
                vec = (centers[ii_a] - centers[jj_a]) / d_a[:, np.newaxis]
                for k in range(len(idx)):
                    grad[ii_a[k]] += lams_a[k] * vec[k]
                    grad[jj_a[k]] -= lams_a[k] * vec[k]
                    
        # Extract boundary marginals
        try:
            b_margs = res.marginals.bounds
            for i in range(n):
                mu = np.abs(b_margs.upper[i])
                if mu > 1e-7:
                    x, y = centers[i]
                    if abs(x - lims[i]) < 1e-7: grad[i, 0] += mu
                    if abs(1.0 - x - lims[i]) < 1e-7: grad[i, 0] -= mu
                    if abs(y - lims[i]) < 1e-7: grad[i, 1] += mu
                    if abs(1.0 - y - lims[i]) < 1e-7: grad[i, 1] -= mu
        except AttributeError:
            pass
            
        return r, s, grad
    except Exception:
        return None, None, None

def generate_configs(n, rng):
    """Generates diverse initial center configurations."""
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,6,4], [5,5,6,5,5], 
        [6,6,5,5,4], [5,4,6,6,5], [6,5,5,6,4], [5,5,5,5,6]
    ]
    for pat in patterns:
        if sum(pat) != n: continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        cfg = np.array(pts[:n])
        configs.append(cfg)
        for _ in range(4):
            p = cfg + rng.uniform(-0.018, 0.018, cfg.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(10):
        configs.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    return configs

def joint_cons(v, n):
    """Inequality constraints for SLSQP joint optimization."""
    cx = v[:n]
    cy = v[n:2*n]
    rr = v[2*n:]
    c_vals = [cx - rr, 1.0 - cx - rr, cy - rr, 1.0 - cy - rr]
    ii, jj = np.triu_indices(n, k=1)
    dx = cx[ii] - cx[jj]
    dy = cy[ii] - cy[jj]
    dr = rr[ii] + rr[jj]
    c_vals.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c_vals)

def joint_obj(v):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(v[2*26:])

def run_packing():

    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_c = None
    best_r = None

    # Phase 1: LP Gradient Ascent from diverse starts
    for cfg in inits:
        c = cfg.copy()
        r, s, g = compute_lp(c, n)
        if r is None: continue

        stall_count = 0
        for it in range(3000):
            gn = np.linalg.norm(g)
            
            if gn < 1e-5:
                stall_count += 1

                    c_shake = c + rng.uniform(-0.015, 0.015, c.shape)

                    if r_sh is not None and s_sh > s:
                        c, r, s, g = c_shake, r_sh, s_sh, g_sh
                        step = 0.025
                        stall_count = 0
                step *= 0.99
                continue

            r_new, s_new, g_new = compute_lp(c_new, n)
            
            if r_new is not None and s_new > s + 1e-8:

                step *= 0.94
                stall_count += 1

            best_r = r.copy()

            if np.isfinite(res.fun):

                curr_s = s_try
                best_c = curr_c.copy()
                best_r = r_try.copy()

                curr_c[i] = old

    for i in range(n):

            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999995
    return best_c, best_r, float(np.sum(best_r))
