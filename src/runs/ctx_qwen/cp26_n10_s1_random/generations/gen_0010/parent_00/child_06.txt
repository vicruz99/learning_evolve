import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_and_grad(centers):
    """Solves LP to maximize sum of radii for fixed centers and computes analytical gradient."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m_pairs = len(idx_i)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(m_pairs + n)
    A_ub = np.zeros((m_pairs + n, n))
    
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i,j)
    A_ub[:m_pairs, idx_i] = 1.0
    A_ub[:m_pairs, idx_j] = 1.0
    b_ub[:m_pairs] = dists[idx_i, idx_j]
    
    # Boundary constraints: r_i <= dist_to_wall(i)
    b_ub[m_pairs:] = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]),
                                 np.minimum(centers[:,1], 1.0-centers[:,1]))
    A_ub[m_pairs:, :] = np.eye(n)
    
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None
            
        radii = res.x
        sum_r = -res.fun
        
        grad = np.zeros((n, 2))
        try:
            # Marginals correspond to the rows of A_ub
            marginals = np.abs(res.ineqlin.marginals)
        except:
            return radii, sum_r, None
            
        pair_marg = marginals[:m_pairs]
        
        # Gradient contribution from pair constraints
        # d(sum_r)/d(c_i) = sum_{j!=i} lambda_{ij} * (c_i - c_j) / dist_{ij}
        diff_ij = centers[idx_i] - centers[idx_j]
        d_ij = dists[idx_i, idx_j]
        d_ij = np.where(d_ij < 1e-9, 1e-9, d_ij)
        
        factors = pair_marg / d_ij
        vec = diff_ij * factors[:, np.newaxis]
        
        np.add.at(grad, idx_i, vec)
        np.add.at(grad, idx_j, -vec)
        
        return radii, sum_r, grad
    except Exception:
        return None, None, None

def obj_and_jac(x, n):
    """Objective and Jacobian for L-BFGS-B: minimizes negative sum of radii."""
    c = x.reshape(n, 2)
    r, s, g = solve_lp_and_grad(c)
    if g is None:
        return 1e6, np.zeros_like(x)
    return -s, g.flatten()

def get_constraints(vars_arr, n):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def get_objective(vars_arr, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def slsqp_polish(x0, n):
    """Joint SLSQP refinement of centers and radii."""
    bounds = [(0.0, 1.0)]*(2*n) + [(1e-6, 0.5)]*n
    cons = {'type': 'ineq', 'fun': get_constraints, 'args': (n,)}
    try:
        res = minimize(get_objective, x0, args=(n,), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
        if np.isfinite(res.fun):
            return res.x
    except Exception:
        pass
    return x0

def hill_climb_lp(centers, n, rng, iters=3000):
    """Stochastic coordinate descent on centers evaluated via exact LP."""
    c = centers.copy()
    r, s, _ = solve_lp_and_grad(c)
    if r is None: return c
    best_s = s
    
    for step in range(iters):
        i = rng.integers(n)
        old = c[i].copy()
        # Adaptive decaying step size
        step_size = 0.012 * (0.996 ** step)
        c[i] += rng.uniform(-step_size, step_size, 2)
        c[i] = np.clip(c[i], 1e-4, 1.0 - 1e-4)
        
        r_new, s_new, _ = solve_lp_and_grad(c)
        if r_new is not None and s_new > best_s + 1e-9:
            best_s = s_new
        else:
            c[i] = old
    return c

def generate_inits(n, rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,6,4], 
        [5,5,6,5,5], [6,6,5,5,4], [6,5,4,6,5],
        [5,4,6,6,5], [7,5,5,5,4], [5,7,4,5,5]
    ]
    
    for pat in patterns:
        if sum(pat) != n: continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx%2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:n])
        mn, mx = pts.min(axis=0), pts.max(axis=0)
        span = mx - mn
        pts = (pts - mn) / span * 0.8 + 0.1
        inits.append(pts)
        # Perturbed variants to break symmetry
        for _ in range(4):
            p = pts + rng.uniform(-0.015, 0.015, pts.shape)
            inits.append(np.clip(p, 0.05, 0.95))
            
    # Random starts
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

        best_c = inits[0]
        best_r, best_s, _ = solve_lp_and_grad(best_c)

    scale = 1.0
    for i in range(n):
        x, y = best_c[i]
        r = best_r[i]
        if r > 1e-12:

    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
