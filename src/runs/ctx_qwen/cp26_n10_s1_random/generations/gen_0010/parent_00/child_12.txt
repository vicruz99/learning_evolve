import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_and_grad(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns optimal radii, sum of radii, and gradient of sum w.r.t centers.
    """
    n = centers.shape[0]
    m = n * (n - 1) // 2
    
    A_ub = np.zeros((m + 4*n, n))
    b_ub = np.zeros(m + 4*n)
    
    idx = 0
    # Pairwise non-overlap constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            idx += 1
            
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, None, None
        
    radii = res.x
    sum_r = -res.fun
    
    # Extract dual variables (marginals) safely across scipy versions
    if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
        lams = np.abs(res.ineqlin.marginals)
    elif hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
        lams = np.abs(res.marginals.ineqlin)
    else:
        lams = np.zeros(len(b_ub))
        
    grad = np.zeros((n, 2))
    k = 0
    # Gradient from pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            lam = lams[k]
            if lam > 1e-7:
                d = b_ub[k]
                if d > 1e-9:
                    fx = lam * (centers[i,0] - centers[j,0]) / d
                    fy = lam * (centers[i,1] - centers[j,1]) / d
                    grad[i, 0] += fx
                    grad[i, 1] += fy
                    grad[j, 0] -= fx
                    grad[j, 1] -= fy
            k += 1
            
    # Gradient from boundary constraints
    grad[:, 0] += lams[m:m+n] - lams[m+n:m+2*n]
    grad[:, 1] += lams[m+2*n:m+3*n] - lams[m+3*n:m+4*n]
        
    return radii, sum_r, grad

def joint_obj(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def joint_cons(x, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    c_list = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c_list.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c_list)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    configs = []
    # High-quality hexagonal row distributions summing to 26
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,6,6,4], [5,5,6,5,5], [6,6,5,5,4]]
    
    for pat in patterns:
        if sum(pat) != n: continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) < n: pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        cfg = np.array(pts[:n])
        # Center and scale to fit comfortably inside [0.1, 0.9]
        mn, mx = cfg.min(axis=0), cfg.max(axis=0)
        cfg = (cfg - mn) / (mx - mn + 1e-9) * 0.8 + 0.1
        configs.append(cfg)
        # Add perturbed variants to break symmetry traps
        for _ in range(3):
            p = cfg + rng.uniform(-0.02, 0.02, cfg.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    # Add fully random starts
    for _ in range(5):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    bounds_jt = [(0.0, 1.0)]*(2*n) + [(1e-6, 0.5)]*n
    
    for cfg in configs:
        c = cfg.copy()
        step = 0.025
        # Phase 1: LP Gradient Ascent on centers
        for it in range(700):
            r, s, g = solve_lp_and_grad(c)
            if r is None: break
            if s > best_sum:
                best_sum = s
                best_c = c.copy()
                best_r = r.copy()
                
            gn = np.linalg.norm(g)
            if gn > 1e-8:
                # Adaptive step with line search
                try_step = step
                improved = False
                for _ in range(4):
                    c_new = c + try_step * g / gn
                    c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
                    r_new, s_new, _ = solve_lp_and_grad(c_new)
                    if r_new is not None and s_new > s + 1e-9:
                        c = c_new
                        step = min(try_step * 1.15, 0.05)
                        improved = True
                        break
                    try_step *= 0.5
                if not improved:
                    step *= 0.92
            else:
                # Random shake to escape local plateaus
                if rng.random() < 0.15:
                    c += rng.uniform(-0.015, 0.015, c.shape)
                    c = np.clip(c, 0.05, 0.95)
                    
        # Phase 2: Joint SLSQP polish to fine-tune centers & radii simultaneously
        if best_c is not None:
            x0 = np.concatenate([best_c.flatten(), best_r * 0.99])
            try:
                res = minimize(joint_obj, x0, args=(n,), method='SLSQP', bounds=bounds_jt,
                               constraints={'type': 'ineq', 'fun': joint_cons, 'args': (n,)},
                               options={'maxiter': 2000, 'ftol': 1e-12})
                if np.isfinite(res.fun):
                    c_p = res.x[:2*n].reshape(n, 2)
                    r_p, s_p, _ = solve_lp_and_grad(c_p)
                    if r_p is not None and s_p > best_sum:
                        best_sum = s_p
                        best_c = c_p
                        best_r = r_p
            except Exception: pass

    # Phase 3: Strict safety scaling to guarantee numerical validity
    if best_c is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                rs = best_r[i] + best_r[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)
        best_r *= scale * 0.999999
        best_sum = float(np.sum(best_r))
    else:
        best_c = np.random.uniform(0.1, 0.9, (n, 2))
        best_r = np.full(n, 0.08)
        best_sum = float(np.sum(best_r))
        
    return best_c, best_r, best_sum
