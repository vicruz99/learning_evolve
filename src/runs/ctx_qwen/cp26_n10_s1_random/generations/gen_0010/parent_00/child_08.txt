import numpy as np
from scipy.optimize import minimize, linprog

def compute_pairwise_dists(centers):
    """Computes the n x n pairwise Euclidean distance matrix."""
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.sqrt(np.sum(diff**2, axis=2))

def solve_lp_and_get_grad(centers):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    dists = compute_pairwise_dists(centers)
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None
            
        r = res.x
        s = -res.fun
        
        # Extract marginals for gradient computation
        lams = None
        try:
            lams = np.abs(res.ineqlin.marginals)
        except AttributeError:
            try:
                lams = np.abs(res.marginals.ineqlin)
            except AttributeError:
                pass
                
        grad = np.zeros((n, 2))
        if lams is not None:
            active = lams > 1e-8
            if np.any(active):
                k_idx = np.where(active)[0]
                ii = idx_i[k_idx]
                jj = idx_j[k_idx]
                d_vals = b_ub[k_idx]
                lams_v = lams[k_idx]
                d_vals = np.maximum(d_vals, 1e-12)
                diff = centers[ii] - centers[jj]
                factors = lams_v / d_vals
                np.add.at(grad, ii, diff * factors[:, np.newaxis])
                np.add.at(grad, jj, -diff * factors[:, np.newaxis])
                
        return r, s, grad
    except Exception:
        return None, None, None

def obj_joint(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def cons_joint(x, n):
    """Inequality constraints >= 0 for SLSQP."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
