import numpy as np
from scipy.optimize import linprog, minimize

def compute_lp_and_grad(centers):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient."""
    n = centers.shape[0]
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    
    # Constraint matrix for r_i + r_j <= dist_ij
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    # Bounds for r_i: 0 <= r_i <= min(dist to walls)
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-9)) for lim in limits]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, None, None
        
    r = res.x
    s = -res.fun
    
    # Compute gradient using LP dual variables (marginals)
    grad = np.zeros((n, 2))
    try:
        lams = None
        if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            lams = res.ineqlin.marginals
        elif hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
            lams = res.marginals.ineqlin
            
        if lams is not None:
            valid = lams > 1e-7
            if np.any(valid):
                k_idx = np.where(valid)[0]
                i_a = idx_i[k_idx]
                j_a = idx_j[k_idx]
                d_a = b_ub[k_idx]
                lam = lams[k_idx]
                d_safe = np.maximum(d_a, 1e-12)
                diff_vec = centers[i_a] - centers[j_a]
                factors = lam[:, np.newaxis] / d_safe[:, np.newaxis]
                grad_contrib = diff_vec * factors
                np.add.at(grad, i_a, grad_contrib)
                np.add.at(grad, j_a, -grad_contrib)
    except Exception:
        pass
        
    return r, s, grad

def lp_obj_and_grad(x_flat):
    """Objective and gradient for L-BFGS-B: maximize LP sum of radii."""
    n = len(x_flat) // 2
    c = x_flat.reshape(n, 2)
    r, s, g = compute_lp_and_grad(c)
    if r is None:
        return 1e6, np.full_like(x_flat, 0.0)
    return -s, -g.flatten()

def get_constraints(vars_arr, n):
    """Computes inequality constraints for SLSQP: must be >= 0."""
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def constraint_wrapper(vars_arr, n):
    """Wrapper to avoid lambda/closure restrictions."""
    return get_constraints(vars_arr, n)

def get_objective(vars_arr, n):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars_arr[2*n:])

def generate_hex_config(pat, r0):
    """Generates a hexagonal lattice configuration."""
    n = 26
    pts = []
    y = r0
    for idx, cnt in enumerate(pat):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    return np.array(pts[:n])
