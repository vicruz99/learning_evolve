import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    if n == 0:
        return np.array([]), 0.0
        
    # Upper bounds for radii based on distance to nearest boundary
    wall_dists = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    bounds = [(0.0, max(d, 1e-9)) for d in wall_dists]
    
    # Pairwise distance constraints: r_i + r_j <= dist(i, j)
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    # Compute pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def lp_objective(x_flat, n):
    """Objective for center optimization: minimize negative LP sum of radii."""
    centers = x_flat.reshape(n, 2)
    # Clip to stay strictly inside [0,1]x[0,1]
    centers = np.clip(centers, 0.005, 0.995)
    _, s = solve_lp_radii(centers)
    return -s

def generate_initial_configs(n, rng):

    starts = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,5,5], 
        [5,5,6,5,5], [6,6,5,5,4], [6,5,4,6,5],

    ]

            continue

                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:n])
        starts.append(pts)

            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            p = np.clip(p, 0.05, 0.95)
            starts.append(p)
            
    # Random starts
    for _ in range(10):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return starts

def force_simulate(cfg, steps=1500):
    """Force-directed repulsion to tighten and validly pack circles."""
    n = len(cfg)
    c = cfg.copy()
    r = np.full(n, 0.09)
    v = np.zeros_like(c)
    dt = 0.0015
    damp = 0.93
    
    for _ in range(steps):
        f = np.zeros_like(c)
        diff = c[:, None, :] - c[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        
        req = r[:, None] + r[None, :]
        overlap = np.maximum(0.0, req - dists)
        safe_dists = np.where(dists > 1e-9, dists, 1.0)
        force_mat = (overlap / safe_dists) * 80.0
        f[:, 0] += np.sum(diff[:, :, 0] * force_mat, axis=1)
        f[:, 1] += np.sum(diff[:, :, 1] * force_mat, axis=1)
        
        # Wall repulsion
        f[:, 0] += np.clip(r - c[:, 0], 0, None) * 40.0
        f[:, 0] -= np.clip(c[:, 0] + r - 1.0, 0, None) * 40.0
        f[:, 1] += np.clip(r - c[:, 1], 0, None) * 40.0
        f[:, 1] -= np.clip(c[:, 1] + r - 1.0, 0, None) * 40.0
        
        v = damp * v + f * dt
        c += v
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
        r = np.minimum(r * 1.000015, 0.15)
    return c

def joint_slsqp_obj(v, n):
    return -np.sum(v[2*n:])

def joint_slsqp_cons(v, n):
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c_cons = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c_cons = np.concatenate([c_cons, dx**2 + dy**2 - dr**2])
    return c_cons
