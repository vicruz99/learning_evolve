

```python
import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26
A_PAIR = np.zeros((N_CIRCLES*(N_CIRCLES-1)//2, N_CIRCLES))
_p_idx = 0
for _i in range(N_CIRCLES):
    for _j in range(_i+1, N_CIRCLES):
        A_PAIR[_p_idx, _i] = 1.0
        A_PAIR[_p_idx, _j] = 1.0
        _p_idx += 1

def solve_lp(centers):
    n = centers.shape[0]
    bounds = []
    b = np.zeros(n*(n-1)//2)
    k = 0
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0-x, y, 1.0-y)
        bounds.append((0.0, max(lim, 1e-9)))
    for i in range(n):
        for j in range(i+1, n):
            b[k] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            k += 1
    try:
        res = linprog(-np.ones(n), A_ub=A_PAIR, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def obj_cent(v):
    c = v.reshape(N_CIRCLES, 2)
    c = np.clip(c, 1e-4, 1.0 - 1e-4)
    _, s = solve_lp(c)
    return -s

def force_simulate(centers, steps=300):
    n = centers.shape[0]
    radii = np.full(n, 0.08)
    vel = np.zeros_like(centers)
    dt = 0.005
    damping = 0.92
    k_rep = 100.0
    k_wall = 50.0
    
    for _ in range(steps):
        radii *= 1.00005
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                dist = np.hypot(dx, dy)
                req = radii[i] + radii[j]
                if dist < req and dist > 1e-9:
                    overlap = req - dist
                    f = k_rep * overlap / dist
                    fx = f * dx
                    fy = f * dy
                    forces[i,0] += fx
                    forces[i,1] += fy
                    forces[j,0] -= fx
                    forces[j,1] -= fy
                    
            x, y = centers[i]
            r = radii[i]
            if x - r < 0: forces[i,0] += k_wall * (r - x)
            if x + r > 1.0: forces[i,0] -= k_wall * (x + r - 1.0)
            if y - r < 0: forces[i,1] += k_wall * (r - y)
            if y + r > 1.0: forces[i,1] -= k_wall * (y + r - 1.0)
            
        vel = damping * vel + forces * dt
        centers += vel
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
        
    return centers

def generate_inits(rng):
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], 
        [4,6,6,6,4], [7,5,5,5,4], [5,7,4,5,5], [5,5,5,5,6], 
        [6,5,5,5,5], [5,4,6,6,5], [5,6,4,6,5], [6,4,5,6,5]
    ]
    for pat in patterns:
        if sum(pat) < N_CIRCLES: continue
        pts = []
        r0 = 0.095
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri%2==1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N_CIRCLES: break
                pts.append([x, y])
                x += 2.0*r0
            y += np.sqrt(3)*r0
        cfg = np.array(pts[:N_CIRCLES])
        mn = cfg.min(axis=0)
        mx = cfg.max(axis=0)
        span = mx - mn + 1e-9
        cfg = (cfg - mn) / span * 0.75 + 0.125
        inits.append(cfg)
        for _ in range(4):
            p = cfg + rng.uniform(-0.025, 0.025, (N_CIRCLES, 2))
            inits.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(12):
        inits.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)))
    return inits

def joint_obj(v):
    return -np.sum(v[2*N_CIRCLES:])

def joint_cons(v):
    n = N_CIRCLES
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    con = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    d2 = dx**2 + dy**2
    np.fill_diagonal(d2, 1.0)
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    idx = np.triu_indices(n, k=1)
    con = np.concatenate([con, d2[idx] - r_sum[idx]**2])
    return con

def run_packing():
    rng = np.random.default_rng(42)
    inits = generate_inits(rng)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    # Phase 1: Force simulation + LP
    for cfg in inits:
        c_sim = force_simulate(cfg.copy(), steps=300)
        r, s = solve_lp(c_sim)
        if s > best_sum:
            best_sum = s
            best_c = c_sim.copy()
            best_r = r.copy()
            
    # Phase 2: Nelder-Mead on centers for diverse starts
    candidates = [best_c] + inits[:8]
    for cfg in candidates:
        try:
            res = minimize(obj_cent, cfg.flatten(), method='Nelder-Mead',
                           options={'maxiter': 4000, 'xatol': 1e-8, 'fatol': 1e-9})
            c_opt = res.x.reshape(N_CIRCLES, 2)
            c_opt = np.clip(c_opt, 0.01, 0.99)
            r, s = solve_lp(c_opt)
            if s > best_sum:
                best_sum = s
                best_c = c_opt.copy()
                best_r = r.copy()
        except Exception:
            pass
            
    # Phase 3: Iterative Hill Climbing on centers
    if best_c is not None:
        curr_c = best_c.copy()
        curr_s = best_sum
        step = 0.025
        for it in range(5000):
            idx = rng.integers(N_CIRCLES)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.uniform(-step, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
            r_try, s_try = solve_lp(curr_c)
            if s_try > curr_s + 1e-8:
                curr_s = s_try
                best_sum = curr_s
                best_c = curr_c.copy()
                best_r = r_try.copy()
                step = min(step * 1.015, 0.06)
            else:
                curr_c[idx] = old
                step *= 0.997
                
    # Phase 4: Joint SLSQP polish from best and perturbed states
    if best_c is not None:
        bases = [best_c]
        for _ in range(5):
            p = best_c + rng.uniform(-0.005, 0.005, (N_CIRCLES, 2))
            bases.append(np.clip(p, 0.05, 0.95))
            
        for base in bases:
            v0 = np.concatenate([base[:,0], base[:,1], best_r * 0.98])
            bounds = [(0.0, 1.0)]*(2*N_CIRCLES) + [(1e-6, 0.5)]*N_CIRCLES
            try:
                res = minimize(joint_obj, v0, method='SLSQP', bounds=bounds,
                               constraints={'type': 'ineq', 'fun': joint_cons},
                               options={'maxiter': 6000, 'ftol': 1e-13})
                if np.isfinite(res.fun):
                    c_p = np.column_stack((res.x[:N_CIRCLES], res.x[N_CIRCLES:2*N_CIRCLES]))
                    r_p, s_p = solve_lp(c_p)
                    if s_p > best_sum:
                        best_sum = s_p
                        best_c = c_p
                        best_r = r_p
            except Exception:
                pass
                
    # Final strict safety scaling to guarantee validation tolerance
    scale = 1.0
    for i in range(N_CIRCLES):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
```
