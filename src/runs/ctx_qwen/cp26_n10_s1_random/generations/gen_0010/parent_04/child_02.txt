import numpy as np
from scipy.optimize import minimize, linprog

def build_constraints(v, n, idx_i, idx_j):
    """Computes inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    
    con = []
    # Boundary constraints: circles inside [0,1]x[0,1]
    con.append(cx - r)
    con.append(1.0 - cx - r)
    con.append(cy - r)
    con.append(1.0 - cy - r)
    
    # Pairwise non-overlap constraints: dist(i,j) >= r_i + r_j
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dist = np.hypot(dx, dy)
    r_sum = r[idx_i] + r[idx_j]
    con.append(dist - r_sum)
    
    return np.concatenate(con)

def obj_func(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def solve_lp_radii(centers, n, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]), 
                      np.minimum(centers[:,1], 1.0 - centers[:,1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    dx = centers[idx_i, 0] - centers[idx_j, 0]
    dy = centers[idx_i, 1] - centers[idx_j, 1]
    dists = np.hypot(dx, dy)
    
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    for k in range(m):
        A_ub[k, idx_i[k]] = 1.0
        A_ub[k, idx_j[k]] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-6)

def generate_hex(n, pattern, r0):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for idx, cnt in enumerate(pattern):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    idx_i, idx_j = np.triu_indices(n, k=1)
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    # Diverse row distributions summing to >= 26
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], 
        [4,6,6,6,4], [5,7,5,5,4], [6,5,5,6,4], [5,6,6,4,5],
        [7,5,5,5,4], [5,5,5,5,6], [6,4,6,5,5], [5,6,4,6,5]
    ]
    
    configs = []
    for pat in patterns:
        if sum(pat) < n:
            continue
        for r0 in [0.09, 0.095, 0.10, 0.105, 0.11]:
            pts = generate_hex(n, pat, r0)
            # Normalize to fit comfortably inside [0.05, 0.95]
            mn = pts.min(axis=0)
            mx = pts.max(axis=0)
            span = mx - mn + 1e-9
            pts = (pts - mn) / span * 0.85 + 0.075
            pts = np.clip(pts, 0.02, 0.98)
            configs.append(pts)
            
    # Add dense random starts
    for _ in range(10):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    
    # Phase 1: Multi-start joint optimization
    for cfg in configs:
        r_init = solve_lp_radii(cfg, n, idx_i, idx_j)
        v0 = np.concatenate([cfg[:,0], cfg[:,1], r_init])
        
        try:
            res = minimize(obj_func, v0, args=(n,), method='SLSQP',
                           bounds=bounds,
                           constraints={'type': 'ineq', 'fun': build_constraints, 'args': (n, idx_i, idx_j)},
                           options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            
            if np.isfinite(res.fun):
                c_opt = np.column_stack((res.x[:n], res.x[n:2*n]))
                # Refine radii with LP for fixed optimal centers to extract maximal sum
                r_lp = solve_lp_radii(c_opt, n, idx_i, idx_j)
                s_lp = np.sum(r_lp)
                
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_c = c_opt.copy()
                    best_r = r_lp.copy()
        except Exception:
            continue
            
    # Phase 2: Perturbation search around best found to escape local minima
    if best_c is not None:
        for _ in range(20):
            pert = best_c + rng.uniform(-0.006, 0.006, (n, 2))
            pert = np.clip(pert, 0.01, 0.99)
            
            r_p = solve_lp_radii(pert, n, idx_i, idx_j)
            v0_p = np.concatenate([pert[:,0], pert[:,1], r_p])
            
            try:
                res_p = minimize(obj_func, v0_p, args=(n,), method='SLSQP',
                                 bounds=bounds,
                                 constraints={'type': 'ineq', 'fun': build_constraints, 'args': (n, idx_i, idx_j)},
                                 options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
                if np.isfinite(res_p.fun):
                    c_p = np.column_stack((res_p.x[:n], res_p.x[n:2*n]))
                    r_lp_p = solve_lp_radii(c_p, n, idx_i, idx_j)
                    s_lp_p = np.sum(r_lp_p)
                    if s_lp_p > best_sum:
                        best_sum = s_lp_p
                        best_c = c_p.copy()
                        best_r = r_lp_p.copy()
            except Exception:
                continue

    # Final strict safety scaling to guarantee validation tolerance
    if best_c is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            if r > 1e-9:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                rs = best_r[i] + best_r[j]
                if rs > 1e-9:
                    scale = min(scale, d / rs)
                
        best_r *= scale * 0.9999999
        best_sum = float(np.sum(best_r))
    else:
        # Guaranteed fallback
        best_c = generate_hex(n, [6, 5, 6, 5, 4], 0.09)
        best_r = solve_lp_radii(best_c, n, idx_i, idx_j)
        best_sum = float(np.sum(best_r))
        
    return best_c, best_r, best_sum
