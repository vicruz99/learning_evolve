import numpy as np
from scipy.optimize import linprog, minimize

def build_lp_structure(n):
    """Precomputes the constant inequality matrix structure for the LP solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def update_lp_rhs(centers, b_ub, idx_i, idx_j):
    """Updates the right-hand side of LP constraints for given centers."""
    n = centers.shape[0]
    k = 0
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        b_ub[k] = x; k += 1
        b_ub[k] = 1.0 - x; k += 1
        b_ub[k] = y; k += 1
        b_ub[k] = 1.0 - y; k += 1
    # Pairwise constraints
    for idx in range(len(idx_i)):
        i, j = idx_i[idx], idx_j[idx]
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b_ub[k] = np.hypot(dx, dy)
        k += 1
    return b_ub

def solve_lp_with_grad(centers, A_ub, b_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii and computes gradient w.r.t centers."""
    b_ub = update_lp_rhs(centers, A_ub, b_ub, idx_i, idx_j)
    n = centers.shape[0]
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return np.full(n, 1e-7), 0.0, np.zeros_like(centers)
            
        radii = np.maximum(res.x, 1e-9)
        sum_r = -res.fun
        
        grad = np.zeros_like(centers)
        try:
            # Extract marginals for pairwise constraints
            marg = res.ineqlin.marginals
        except AttributeError:
            try:
                marg = res.marginals.ineqlin
            except:
                marg = None
                
        if marg is not None:
            pair_marg = marg[4*n:]
            pair_rhs = b_ub[4*n:]
            for idx in range(len(idx_i)):
                if pair_marg[idx] > 1e-7:
                    i, j = idx_i[idx], idx_j[idx]
                    d = pair_rhs[idx]
                    if d > 1e-9:
                        diff = centers[i] - centers[j]
                        factor = pair_marg[idx] / d
                        grad[i] += diff * factor
                        grad[j] -= diff * factor
                        
        return radii, sum_r, grad
    except Exception:
        return np.full(n, 1e-7), 0.0, np.zeros_like(centers)

def generate_hex(n, rows, r0, rng):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:n])

def joint_constraints(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    
    con = [
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ]
    
    # Pairwise: dist >= r_i + r_j
    dx = cx[triu_i] - cx[triu_j]
    dy = cy[triu_i] - cy[triu_j]
    dist = np.hypot(dx, dy)
    r_sum = r[triu_i] + r[triu_j]
    con.append(dist - r_sum)
    
    return np.concatenate(con)
