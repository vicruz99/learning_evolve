import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Precomputes the constant inequality matrix structure for the LP solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        for _ in range(4):
            A[k, i] = 1.0
            k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp_fast(centers, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    b = np.empty(A_ub.shape[0])
    k = 0
    for i in range(n):
        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            b[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def obj_lp(x, n, A):
    """Objective for center optimization: maximize sum of radii via LP."""
    c = x.reshape(n, 2)
    _, s = solve_lp_fast(c, A)
    return -s

def joint_obj(v, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def joint_cons(v, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid packing (squared distances for smoothness)."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    con = []
    con.append(cx - r)
    con.append(1.0 - cx - r)
    con.append(cy - r)
    con.append(1.0 - cy - r)
    
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    r_sum = r[idx_i] + r[idx_j]
    con.append(dx**2 + dy**2 - r_sum**2)
    
    return np.concatenate(con)

def generate_hex(n, rows, r0):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
