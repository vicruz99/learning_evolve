import numpy as np
from scipy.optimize import linprog, minimize

def get_lp_marginals(res):
    """Safely extract marginals from linprog result across scipy versions."""
    try:
        return res.ineqlin.marginals
    except AttributeError:
        try:
            return res.marginals.ineqlin
        except AttributeError:
            return None

def compute_lp_and_grad(centers, n):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns radii, sum of radii, and gradient of the sum w.r.t center coordinates.
    """
    m_pairs = n * (n - 1) // 2
    A_ub = np.zeros((4 * n + m_pairs, n))
    b_ub = np.zeros(4 * n + m_pairs)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        # r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    idx_i, idx_j = np.triu_indices(n, k=1)
    diffs = centers[idx_i] - centers[idx_j]
    dists = np.hypot(diffs[:, 0], diffs[:, 1])
    np.putmask(dists, dists < 1e-9, 1e-9)
    
    for k in range(m_pairs):
        # r_i + r_j <= dist_ij
        A_ub[4 * n + k, idx_i[k]] = 1.0
        A_ub[4 * n + k, idx_j[k]] = 1.0
        b_ub[4 * n + k] = dists[k]
        
    bounds = [(0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, None, None
            
        r = res.x
        s = -res.fun
        
        # Compute subgradient using LP marginals
        grad = np.zeros((n, 2))
        marginals = get_lp_marginals(res)
        if marginals is not None:
            # Pairwise constraints contributions
            pair_marginals = marginals[4 * n:]
            valid = pair_marginals > 1e-8
            if np.any(valid):
                k_idx = np.where(valid)[0]
                ii = idx_i[k_idx]
                jj = idx_j[k_idx]
                lam = pair_marginals[k_idx]
                d = dists[k_idx]
                vec = (centers[ii] - centers[jj]) / d[:, None] * lam[:, None]
                np.add.at(grad, ii, vec)
                np.add.at(grad, jj, -vec)
                
            # Boundary constraints contributions
            b_marginals = marginals[:4 * n].reshape(n, 4)
            for i in range(n):
                if b_marginals[i, 0] > 1e-8: grad[i, 0] += b_marginals[i, 0]
                if b_marginals[i, 1] > 1e-8: grad[i, 0] -= b_marginals[i, 1]
                if b_marginals[i, 2] > 1e-8: grad[i, 1] += b_marginals[i, 2]
                if b_marginals[i, 3] > 1e-8: grad[i, 1] -= b_marginals[i, 3]
        return r, s, grad
    except Exception:
        return None, None, None

def local_search_lp(centers, rng, max_iter=1500):
    """Iterative LP-guided local search to optimize center positions."""
    n = centers.shape[0]
    curr_c = centers.copy()
    _, best_s, _ = compute_lp_and_grad(curr_c, n)
    if best_s is None: return curr_c
    
    step = 0.025
    for it in range(max_iter):
        r, s, g = compute_lp_and_grad(curr_c, n)
        if r is None: break
        
        gn = np.linalg.norm(g)
        if gn > 1e-6:
            direction = g / gn
            c_try = curr_c + step * direction
            c_try = np.clip(c_try, 1e-4, 1.0 - 1e-4)
            r_try, s_try, _ = compute_lp_and_grad(c_try, n)
            
            if r_try is not None and s_try > s + 1e-7:
                curr_c = c_try
                best_s = s_try
                step = min(step * 1.02, 0.05)
            else:
                step *= 0.95
        else:
            # Plateau: random shake to escape
            if rng.random() < 0.25:
                c_shake = curr_c + rng.uniform(-step, step, (n, 2))
                c_shake = np.clip(c_shake, 1e-4, 1.0 - 1e-4)
                r_sh, s_sh, _ = compute_lp_and_grad(c_shake, n)
                if r_sh is not None and s_sh > s + 1e-7:
                    curr_c = c_shake
                    best_s = s_sh
            step *= 0.998
            
    return curr_c

def obj_slsqp(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def cons_slsqp(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r,
        np.hypot(cx[triu_i] - cx[triu_j], cy[triu_i] - cy[triu_j]) - (r[triu_i] + r[triu_j])
    ])
    return c

def run_slsqp_polish(centers, radii, n, rng):
    """Fine-tune centers and radii jointly using SLSQP."""
    x0 = np.concatenate([centers[:, 0], centers[:, 1], radii * 0.99])
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    triu_i, triu_j = np.triu_indices(n, k=1)
    
    try:
        res = minimize(obj_slsqp, x0, args=(n,), method='SLSQP',
                       bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons_slsqp, 'args': (n, triu_i, triu_j)},
                       options={'maxiter': 5000, 'ftol': 1e-13})
        if np.isfinite(res.fun):
            return np.column_stack((res.x[:n], res.x[n:2 * n])), np.maximum(res.x[2 * n:], 1e-9)
    except Exception:
        pass
    return centers, radii

def generate_hex_config(n, rows, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:n])

def generate_initial_configs(n, rng):
    """Creates diverse starting configurations."""
    configs = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [5, 7, 5, 5, 4],
        [6, 5, 5, 6, 4], [5, 6, 6, 4, 5], [7, 5, 5, 5, 4]
    ]
    for pat in patterns:
        if sum(pat) >= n:
            cfg = generate_hex_config(n, pat, 0.095, rng)
            configs.append(cfg)
            cfg_p = cfg + rng.uniform(-0.02, 0.02, (n, 2))
            configs.append(np.clip(cfg_p, 0.05, 0.95))
            
    for _ in range(15):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # Corner and edge biased starts
    for _ in range(5):
        cfg = np.zeros((n, 2))
        corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        for i in range(n):
            cfg[i] = rng.uniform(0.05, 0.95, 2)
            if i < 4:
                cfg[i] = corners[i] + rng.uniform(-0.05, 0.05, 2)
        configs.append(np.clip(cfg, 0.02, 0.98))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    configs = generate_initial_configs(n, rng)
    
    for cfg in configs:
        # Phase 1: LP-based gradient ascent on centers
        c_opt = local_search_lp(cfg.copy(), rng, max_iter=1500)
        r_lp, s_lp, _ = compute_lp_and_grad(c_opt, n)
        
        if r_lp is not None and s_lp > best_sum:
            best_sum = s_lp
            best_centers = c_opt.copy()
            best_radii = r_lp.copy()
            
        # Phase 2: SLSQP Joint Polish
        if best_centers is not None:
            c_pol, r_pol = run_slsqp_polish(best_centers, best_radii, n, rng)
            r_ref, s_ref, _ = compute_lp_and_grad(c_pol, n)
            if r_ref is not None and s_ref > best_sum:
                best_sum = s_ref
                best_centers = c_pol.copy()
                best_radii = r_ref.copy()
                
        # Phase 3: Perturbation & Restart to escape local minima
        for _ in range(8):
            pert = best_centers + rng.uniform(-0.012, 0.012, (n, 2))
            pert = np.clip(pert, 0.02, 0.98)
            c_p = local_search_lp(pert.copy(), rng, max_iter=800)
            r_p, s_p, _ = compute_lp_and_grad(c_p, n)
            if r_p is not None and s_p > best_sum:
                best_sum = s_p
                best_centers = c_p.copy()
                best_radii = r_p.copy()
                
    # Final strict safety scaling to guarantee validation tolerance
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-9:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-9:
                    scale = min(scale, d / rs)
                
        best_radii *= scale * 0.9999999
        best_sum = float(np.sum(best_radii))
    else:
        best_centers = generate_hex_config(n, [6, 5, 6, 5, 4], 0.09, rng)
        best_radii = np.full(n, 0.09)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
