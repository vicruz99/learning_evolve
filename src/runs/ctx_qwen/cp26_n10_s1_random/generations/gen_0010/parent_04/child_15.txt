import numpy as np
from scipy.optimize import minimize, linprog

# Global constants and precomputed structures for efficiency
N_CIRCLES = 26
M_PAIRS = N_CIRCLES * (N_CIRCLES - 1) // 2

A_PAIRS = np.zeros((M_PAIRS, N_CIRCLES))
I_PAIRS, J_PAIRS = np.triu_indices(N_CIRCLES, k=1)
_k = 0
for i in range(N_CIRCLES):
    for j in range(i + 1, N_CIRCLES):
        A_PAIRS[_k, i] = 1.0
        A_PAIRS[_k, j] = 1.0
        _k += 1

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diffs[:, :, 0], diffs[:, :, 1])
    np.fill_diagonal(dists, np.inf)
    b_pairs = dists[I_PAIRS, J_PAIRS]
    
    try:
        res = linprog(-np.ones(N_CIRCLES), A_ub=A_PAIRS, b_ub=b_pairs, 
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N_CIRCLES, 1e-6), 0.0

def joint_objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * N_CIRCLES:])

def joint_constraints(v):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:N_CIRCLES]
    cy = v[N_CIRCLES:2 * N_CIRCLES]
    r = v[2 * N_CIRCLES:]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.hypot(dx, dy)
    c = np.concatenate([c, dist[I_PAIRS, J_PAIRS] - (r[I_PAIRS] + r[J_PAIRS])])
    
    return c

def hill_climb_centers(centers, rng, iters=2000):
    """Local search optimizing centers by maximizing LP sum of radii."""
    n = centers.shape[0]
    cur_c = centers.copy()
    cur_r, cur_s = solve_lp_radii(cur_c)
    step = 0.025
    
    for _ in range(iters):
        idx = rng.integers(n)
        old_pos = cur_c[idx].copy()
        
        # Random perturbation
        cur_c[idx] += rng.uniform(-step, step, 2)
        cur_c[idx] = np.clip(cur_c[idx], 0.01, 0.99)
        
        new_r, new_s = solve_lp_radii(cur_c)
        
        if new_s > cur_s + 1e-9:
            cur_r = new_r
            cur_s = new_s
            step = min(step * 1.04, 0.06)
        else:
            cur_c[idx] = old_pos
            step *= 0.997
            
        # Periodic shake to escape plateaus
        if rng.random() < 0.05:
            cur_c[rng.integers(n)] += rng.uniform(-0.01, 0.01, 2)
            cur_c = np.clip(cur_c, 0.01, 0.99)
            cur_r, cur_s = solve_lp_radii(cur_c)
            
    return cur_c, cur_r, cur_s

def run_packing():
    rng = np.random.default_rng(42)
    bounds_vars = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(1e-6, 0.5)] * N_CIRCLES
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Generate diverse hexagonal and random initial configurations
    configs = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [5, 7, 5, 5, 4],
        [7, 5, 5, 5, 4], [5, 5, 5, 6, 5], [5, 5, 5, 5, 6],
        [5, 6, 4, 6, 5], [6, 4, 6, 5, 5]
    ]
    
    for pat in patterns:
        if sum(pat) < N_CIRCLES: continue
        for sx in [0.0, 0.04, -0.04]:
            for sy in [0.0, 0.02]:
                pts = []
                y = 0.09
                for ri, cnt in enumerate(pat):
                    shift = 0.09 if ri % 2 == 1 else 0.0
                    x = 0.09 + shift + sx
                    for _ in range(cnt):
                        if len(pts) >= N_CIRCLES: break
                        pts.append([x, y + sy])
                        x += 0.18
                    y += 0.09 * np.sqrt(3)
                if len(pts) >= N_CIRCLES:
                    cfg = np.array(pts[:N_CIRCLES])
                    cfg += rng.uniform(-0.01, 0.01, (N_CIRCLES, 2))
                    cfg = np.clip(cfg, 0.05, 0.95)
                    configs.append(cfg)
                    
    for _ in range(12):
        configs.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)))
        
    # Phase 1: Multi-start SLSQP optimization
    for cfg in configs:
        r_lp, s_lp = solve_lp_radii(cfg)
        v0 = np.concatenate([cfg[:, 0], cfg[:, 1], r_lp * 0.96])
        
        try:
            res = minimize(joint_objective, v0, method='SLSQP', bounds=bounds_vars,
                           constraints={'type': 'ineq', 'fun': joint_constraints},
                           options={'maxiter': 5000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_opt = np.column_stack((res.x[:N_CIRCLES], res.x[N_CIRCLES:2*N_CIRCLES]))
                r_opt, s_opt = solve_lp_radii(c_opt)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            continue
            
    # Phase 2: Exact LP-based Hill Climbing & Perturbation Search
    if best_centers is not None:
        # Direct polishing
        c_hc, r_hc, s_hc = hill_climb_centers(best_centers, rng, iters=2500)
        if s_hc > best_sum:
            best_sum = s_hc
            best_centers = c_hc.copy()
            best_radii = r_hc.copy()
            
        # Aggressive perturbation search to escape local optima
        for _ in range(40):
            pert = best_centers + rng.uniform(-0.009, 0.009, (N_CIRCLES, 2))
            pert = np.clip(pert, 0.02, 0.98)
            c_p, r_p, s_p = hill_climb_centers(pert, rng, iters=1200)
            if s_p > best_sum:
                best_sum = s_p
                best_centers = c_p.copy()
                best_radii = r_p.copy()
                
    # Phase 3: Strict Safety Scaling to guarantee validator tolerance
    if best_centers is not None:
        scale = 1.0
        for i in range(N_CIRCLES):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-9:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
                
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-9:
                    scale = min(scale, d / rs)
                    
        best_radii *= scale * 0.9999999
        best_sum = float(np.sum(best_radii))
    else:
        # Guaranteed fallback
        best_centers = np.random.uniform(0.2, 0.8, (N_CIRCLES, 2))
        best_radii = np.full(N_CIRCLES, 0.05)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
