N_CIRCLES = 26
N_PAIRS = N_CIRCLES * (N_CIRCLES - 1) // 2
TRIU_I, TRIU_J = np.triu_indices(N_CIRCLES, k=1)

A_LP_PAIRS = np.zeros((N_PAIRS, N_CIRCLES))

def solve_lp_radii(centers, n, triu_i, triu_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[triu_i, triu_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_LP_PAIRS, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def obj_single_circle(xy, curr_c, idx, n, triu_i, triu_j):
    """Objective for single-circle optimization: minimize negative LP sum of radii."""
    temp_c = curr_c.copy()
    temp_c[idx] = np.clip(xy, 0.005, 0.995)
    _, s = solve_lp_radii(temp_c, n, triu_i, triu_j)
    return -s

def joint_obj(v, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def joint_cons(v, n, triu_i, triu_j):

    cx, cy, r = v[:n], v[n:2*n], v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])

    rs = r[triu_i] + r[triu_j]
    c = np.concatenate([c, d2[triu_i, triu_j] - rs**2])
    return c
