import numpy as np
from scipy.optimize import linprog, minimize

def build_A_ub(n):
    """Precomputes the constant inequality matrix structure for the LP solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp_and_grad(centers, A_ub):
    """Solves LP to maximize sum of radii for fixed centers and returns gradient."""
    n = centers.shape[0]
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            b[k] = np.linalg.norm(centers[i] - centers[j])
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            return np.full(n, 1e-6), 0.0, np.zeros((n, 2))
            
        radii = res.x
        s = np.sum(radii)
        
        grad = np.zeros((n, 2))
        try:
            marginals = -res.ineqlin.marginals  # Negate for ascent direction
        except AttributeError:
            try:
                marginals = -res.marginals.ineqlin
            except AttributeError:
                return radii, s, grad
                
        k = 4 * n
        for i in range(n):
            for j in range(i + 1, n):
                lam = marginals[k]
                if lam > 1e-8:
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    if dist > 1e-9:
                        grad[i] += lam * diff / dist
                        grad[j] -= lam * diff / dist
                k += 1
        return radii, s, grad
    except Exception:
        return np.full(n, 1e-6), 0.0, np.zeros((n, 2))

def obj_centers(v, n, A_ub):
    """Objective for center optimization: maximize LP sum of radii."""
    c = np.clip(v.reshape(n, 2), 1e-4, 1.0 - 1e-4)
    _, s, _ = solve_lp_and_grad(c, A_ub)
    return -s

            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)

    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
