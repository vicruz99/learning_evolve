import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_radii(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    m_b = 4 * n
    m_p = n * (n - 1) // 2
    A = np.zeros((m_b + m_p, n))
    b = np.zeros(m_b + m_p)
    idx = 0
    
    for i in range(n):
        x, y = centers[i]
        lims = [x, 1.0 - x, y, 1.0 - y]
        for lim in lims:
            A[idx, i] = 1.0
            b[idx] = max(lim, 1e-9)
            idx += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-5)

def penalty_objective(v, n, mu):
    """Objective: minimize -sum(r) + penalty for constraint violations."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary penalties
    pen += np.sum(np.maximum(0.0, r - x) ** 2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - x)) ** 2)
    pen += np.sum(np.maximum(0.0, r - y) ** 2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - y)) ** 2)
    
    # Pairwise overlap penalties
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist = np.sqrt(dx ** 2 + dy ** 2)
    np.fill_diagonal(dist, 1e6)
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    viol = np.maximum(0.0, r_sum - dist)
    pen += np.sum(np.triu(viol ** 2, k=1))
    
    return obj + mu * pen

def force_simulate(centers, n, steps=600):
    """Physics simulation to find dense, valid initial layouts."""
    r = np.full(n, 0.04)
    vel = np.zeros_like(centers)
    dt = 0.008
    damping = 0.80
    k_rep = 12.0
    k_wall = 25.0
    growth = 1.00007
    
    for _ in range(steps):
        r *= growth
        forces = np.zeros_like(centers)
        
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff ** 2, axis=2))
        np.fill_diagonal(d, 1e9)
        
        overlap = r[:, np.newaxis] + r[np.newaxis, :] - d
        mask = overlap > 0
        safe_d = np.where(d > 1e-8, d, 1e-8)
        f_mat = np.where(mask, overlap / safe_d * diff, 0.0)
        forces += np.sum(f_mat, axis=1) * k_rep
        
        for i in range(n):
            if centers[i, 0] < r[i]: forces[i, 0] += k_wall * (r[i] - centers[i, 0])
            if centers[i, 0] > 1.0 - r[i]: forces[i, 0] -= k_wall * (centers[i, 0] - (1.0 - r[i]))
            if centers[i, 1] < r[i]: forces[i, 1] += k_wall * (r[i] - centers[i, 1])
            if centers[i, 1] > 1.0 - r[i]: forces[i, 1] -= k_wall * (centers[i, 1] - (1.0 - r[i]))
            
        vel = damping * vel + forces * dt
        centers += vel
        centers = np.clip(centers, 0.0, 1.0)
        
    return centers, r
