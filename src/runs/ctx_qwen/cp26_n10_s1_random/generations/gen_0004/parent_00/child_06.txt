from scipy.optimize import minimize, linprog

N = 26
TRIU_MASK = np.triu(np.ones((N, N), dtype=bool), k=1)

def obj_func(vars_arr):
    """Objective: minimize negative sum of radii => maximize sum of radii"""
    return -np.sum(vars_arr[2::3])

def cons_func(vars_arr):
    """Inequality constraints >= 0 for valid packing"""
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.extend(x - r)
    c.extend(1.0 - x - r)
    c.extend(y - r)
    c.extend(1.0 - y - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.extend(d2[TRIU_MASK] - rs[TRIU_MASK]**2)
    
    return np.array(c)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers"""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Boundary limits for each radius
    bounds = [(0.0, max(min(c_[0], 1.0-c_[0], c_[1], 1.0-c_[1]), 0.0)) for c_ in centers]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def get_hex(r0, rows_cfg):
    """Generates a hexagonal lattice configuration with specified row counts"""
    pts = []
    dy = np.sqrt(3)*r0
    y = r0
    for idx, cnt in enumerate(rows_cfg):
        shift = r0 if idx%2==1 else 0.0
        w = (cnt-1)*2*r0
        x0 = (1.0-w)/2.0 + shift
        for _ in range(cnt):
            pts.append([x0, y])
            x0 += 2*r0
        y += dy
    return np.array(pts[:N])
