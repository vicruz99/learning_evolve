import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_INDICES = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_constraints = n * (n - 1) // 2 + 4 * n
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        limits = [x, 1.0 - x, y, 1.0 - y]
        for lim in limits:
            A_ub[idx, i] = 1.0
            b_ub[idx] = max(lim, 0.0)
            idx += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective(vars_arr):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_arr[2::3])

def get_constraints(vars_arr):
    """Computes inequality constraints >= 0 for valid packing."""
    n = N
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    
    cons = np.empty(4 * n + n * (n - 1) // 2)
    idx = 0
    
    # Boundary constraints
    cons[idx:idx+n] = x - r; idx += n
    cons[idx:idx+n] = 1.0 - x - r; idx += n
    cons[idx:idx+n] = y - r; idx += n
    cons[idx:idx+n] = 1.0 - y - r; idx += n
    
    # Pairwise non-overlap constraints (squared distance >= squared sum of radii)
    xx = x[:, np.newaxis] - x[np.newaxis, :]
    yy = y[:, np.newaxis] - y[np.newaxis, :]
    rr = r[:, np.newaxis] + r[np.newaxis, :]
    d2 = xx**2 + yy**2
    r2 = rr**2
    cons[idx:] = (d2 - r2)[TRIU_INDICES]
    
    return cons

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_opt = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': get_constraints}
    
    configs = []
    
    # 1. Hexagonal lattices with various base radii and row shifts
    for r0 in np.linspace(0.07, 0.11, 6):
        for shift in [0.0, r0 / 2, r0]:
            pts = []
            y = r0
            row = 0
            while len(pts) < N + 10:
                x = r0 + shift + (row % 2) * r0
                while x + r0 <= 1.0 and len(pts) < N + 10:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
                row += 1
            configs.append(np.array(pts[:N]))
            
    # 2. Grid patterns
    for step in [0.18, 0.19, 0.20, 0.21]:
        xs = np.arange(0.5 * step, 1.0, step)
        ys = np.arange(0.5 * step, 1.0, step)
        pts = [(x, y) for y in ys for x in xs]
        while len(pts) < N:
            pts.append([0.5, 0.5])
        configs.append(np.array(pts[:N]))
        
    # 3. Random dense configurations
    for _ in range(8):
        configs.append(np.random.uniform(0.1, 0.9, (N, 2)))

    for cfg in configs:
