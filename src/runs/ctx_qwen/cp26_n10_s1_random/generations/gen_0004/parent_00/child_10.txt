import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_IDX = np.triu_indices(N, k=1)

def objective(vars_arr):
    """Minimize negative sum of radii => Maximize sum of radii"""
    return -np.sum(vars_arr[2*N:])

def constraints(vars_arr):
    """Computes inequality constraints >= 0 for valid packing"""
    cx = vars_arr[:N]
    cy = vars_arr[N:2*N]
    r = vars_arr[2*N:]
    
    # Boundary constraints: x-r>=0, 1-x-r>=0, y-r>=0, 1-y-r>=0
    c = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    c.append(d2[TRIU_IDX] - rs[TRIU_IDX]**2)
    
    return np.concatenate(c)

def get_hex_init(n, r0=0.1, row_counts=None):
    """Generates a hexagonal lattice initialization"""
    if row_counts is None:
        pts = []
        y = r0
        row = 0
        while len(pts) < n + 5:
            shift = r0 if row % 2 == 1 else 0.0
            x = r0 + shift
            while x + r0 <= 1.0:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        return np.array(pts[:n])
    else:
        pts = []
        y = r0
        for i, cnt in enumerate(row_counts):
            shift = r0 if i % 2 == 1 else 0.0
            x_start = r0 + shift
            for _ in range(cnt):
                pts.append([x_start, y])
                x_start += 2.0 * r0
            y += np.sqrt(3) * r0
        return np.array(pts[:n])

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    # Boundary constraints: r_i <= dist(i, boundary)
    for i in range(n):
        x, y = centers[i]
        limits = [x, 1.0 - x, y, 1.0 - y]
        for lim in limits:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(lim)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def run_packing():
    rng = np.random.default_rng(42)
    bounds_vars = [(0.0, 1.0)] * (2 * N) + [(1e-5, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = []
    
    # 1. Hexagonal lattices with various densities
    for r0 in [0.09, 0.095, 0.10]:
        cfg = get_hex_init(N, r0)
        min_c = cfg.min(axis=0)
        max_c = cfg.max(axis=0)
        cfg = (cfg - min_c) / (max_c - min_c) * 0.94 + 0.03
        configs.append(cfg)
        for _ in range(3):
            cfg_p = cfg + rng.uniform(-0.02, 0.02, cfg.shape)
            cfg_p = np.clip(cfg_p, 0.05, 0.95)
            configs.append(cfg_p)
            
    # 2. Specific row distributions known to be good for N=26
    row_dists = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], [4, 6, 6, 6, 4]]
    for rd in row_dists:
        if sum(rd) >= N:
            cfg = get_hex_init(N, 0.095, rd)
            min_c = cfg.min(axis=0)
            max_c = cfg.max(axis=0)
            cfg = (cfg - min_c) / (max_c - min_c) * 0.94 + 0.03
            configs.append(cfg)
            
    # 3. Regular grid + perturbations
    grid = np.array([[0.15 + 0.175 * (i % 5), 0.15 + 0.175 * (i // 5)] for i in range(25)])
    grid = np.vstack([grid, [0.5, 0.5]])
    configs.append(grid)
    for _ in range(2):
        configs.append(np.clip(grid + rng.uniform(-0.03, 0.03, grid.shape), 0.05, 0.95))
        
    # Optimization loop
    for cfg in configs:
        x0 = np.zeros(3 * N)
        x0[:N] = cfg[:, 0]
        x0[N:2*N] = cfg[:, 1]
        x0[2*N:] = 0.09  # Strictly feasible initial radius
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_vars,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-8:
                    cx = res.x[:N]
                    cy = res.x[N:2*N]
                    centers = np.column_stack((cx, cy))
                    radii = res.x[2*N:]
                    
                    # LP refinement on centers to extract maximum possible radii
                    lp_r, lp_sum = solve_lp_radii(centers)
                    if lp_r is not None and lp_sum > 0:
                        s = lp_sum
                        c_opt = centers
                        r_opt = lp_r
                    else:
                        s = np.sum(radii)
                        c_opt = centers
                        r_opt = radii
                        
                    if s > best_sum:
                        best_sum = s
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
        except Exception:
            continue
            
    # Final strict safety scaling to guarantee numerical validity
    if best_centers is not None:
        scale = 1.0
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-9:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
                
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-9:
                    scale = min(scale, d / rs)
                    
        best_radii *= max(scale * 0.9999999, 0.0)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
