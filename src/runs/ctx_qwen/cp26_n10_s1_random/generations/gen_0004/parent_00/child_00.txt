The LP maximizes $\sum r_i$ subject to $r_i + r_j \leq d_{ij}$ and boundary constraints $r_i \leq \min(x_i, 1-x_i, y_i, 1-y_i)$.

My strategy leverages this property:

3. **Hybrid Local Optimization**: For each initialization, use `L-BFGS-B` (which respects bounds) to quickly approach a local optimum, followed by `Nelder-Mead` (derivative-free) to escape flat regions and refine the non-smooth LP objective.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    A_ub = []
    b_ub = []
    
    xs, ys = centers[:, 0], centers[:, 1]
    # Boundary constraints: r_i <= dist to each wall
    limits = np.minimum(np.minimum(xs, 1.0 - xs), np.minimum(ys, 1.0 - ys))
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(limits[i])
        
    # Pairwise constraints: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(xs[i] - xs[j], ys[i] - ys[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun
    return None, 0.0

def objective(x):
    """Objective function for center optimization: minimize negative sum of radii."""
    try:
        centers = x.reshape(-1, 2)
        # Keep centers strictly inside to ensure LP feasibility and safety
        if np.any(centers < 0.002) or np.any(centers > 0.998):
            return 1.0e6
        _, s = solve_lp_radii(centers)
        return -s
    except Exception:
        return 1.0e6

def run_packing():
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_centers = None
    
    # Bounds for center optimization
    bounds_opt = [(0.005, 0.995)] * (2 * N_CIRCLES)
    inits = []
    
    # 1. Hexagonal lattices with varying densities
    for scale in [0.08, 0.085, 0.09, 0.095, 0.10]:
        pts = []
        y = scale
        row = 0
        while len(pts) < N_CIRCLES + 5:
            shift = scale if row % 2 == 1 else 0.0
            x = scale + shift
            while x + scale <= 1.0:
                pts.append([x, y])
                x += 2.0 * scale
            y += np.sqrt(3) * scale
            row += 1
        pts = np.array(pts[:N_CIRCLES])
        pts += rng.uniform(-0.012, 0.012, pts.shape)
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts.flatten())
        
    # 2. Perturbed grid starts
    grid = np.array([(0.15 + i*0.17, 0.15 + j*0.17) for i in range(5) for j in range(5)])
    grid = np.vstack([grid, [0.5, 0.5]])
    for _ in range(5):
        pert = grid + rng.uniform(-0.05, 0.05, grid.shape)
        pert = np.clip(pert, 0.05, 0.95)
        inits.append(pert.flatten())
        
    # 3. Random dense starts
    for _ in range(3):
        g = rng.uniform(0.2, 0.8, (N_CIRCLES, 2))
        inits.append(g.flatten())
        
    # Optimization loop
    for x0 in inits:
        try:
            # Phase 1: L-BFGS-B respects bounds and converges quickly
            res1 = minimize(objective, x0, method='L-BFGS-B', bounds=bounds_opt,
                            options={'maxiter': 1200, 'ftol': 1e-13})
            x_curr = res1.x
            
            # Phase 2: Nelder-Mead refines non-smooth LP objective
            res2 = minimize(objective, x_curr, method='Nelder-Mead',
                            options={'maxiter': 1500, 'fatol': 1e-13, 'xatol': 1e-9})
            
            # Keep the best of the two phases
            if -res2.fun > -res1.fun:
                x_curr = res2.x
                
            s_curr = max(-res1.fun, -res2.fun)
            if np.isfinite(s_curr) and s_curr > best_sum:
                best_sum = s_curr
                best_centers = x_curr.reshape(-1, 2)
        except Exception:
            continue
            
    # Fallback if all optimizations fail
    if best_centers is None:
        best_centers = inits[0].reshape(-1, 2)
        
    # Final exact LP to get precise radii for the best centers
    radii, s = solve_lp_radii(best_centers)
    if radii is not None:
        best_radii = radii
        best_sum = s
    else:
        best_radii = np.full(N_CIRCLES, 0.08)
        best_sum = N_CIRCLES * 0.08
        
    # Strict safety scaling to guarantee validation passes with 1e-12 tolerance
    scale = 1.0
    for i in range(N_CIRCLES):
        r_i = best_radii[i]
        if r_i > 1e-9:
            scale = min(scale, best_centers[i,0]/r_i, (1.0-best_centers[i,0])/r_i,
                        best_centers[i,1]/r_i, (1.0-best_centers[i,1])/r_i)
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
