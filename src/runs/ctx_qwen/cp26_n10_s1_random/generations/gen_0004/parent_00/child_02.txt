import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(vars_arr):
    """Minimize negative sum of radii => Maximize sum of radii"""
    return -np.sum(vars_arr[2::3])

def get_constraints(vars_arr):
    """Computes inequality constraints >= 0 for valid packing"""
    n = N
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    
    cons = []
    # Boundary constraints: r <= x, 1-x, y, 1-y
    cons.extend(x - r)
    cons.extend(1.0 - x - r)
    cons.extend(y - r)
    cons.extend(1.0 - y - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    i, j = np.triu_indices(n, 1)
    dx = x[i] - x[j]
    dy = y[i] - y[j]
    d2 = dx**2 + dy**2
    rs = r[i] + r[j]
    cons.extend(d2 - rs**2)
    
    return np.array(cons)

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers"""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        limits = (x, 1.0 - x, y, 1.0 - y)
        for lim in limits:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(lim)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x
    except Exception:
        pass
    return None

def get_hex_init(r0=0.101):
    """Generates a hexagonal lattice initialization clustered at the center"""
    pts = []
    y = r0
    row = 0
    while y + r0 <= 1.0:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x + r0 <= 1.0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
        if len(pts) > N + 15:
            break
            
    pts = np.array(pts)
    # Pick points closest to the center to minimize edge effects
    dists = np.sum((pts - 0.5) ** 2, axis=1)
    idx = np.argsort(dists)[:N]
    return pts[idx]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Generate diverse starting configurations
    starts = []
    for r_init in [0.100, 0.101, 0.102]:
        starts.append(get_hex_init(r_init))
    for _ in range(6):
        cfg = get_hex_init(0.101).copy()
        cfg += rng.uniform(-0.012, 0.012, cfg.shape)
        cfg = np.clip(cfg, 1e-3, 1.0 - 1e-3)
        starts.append(cfg)
        
    # Optimization loop
    for cfg in starts:
        x0 = np.zeros(3 * N)
        x0[0::3] = cfg[:, 0]
        x0[1::3] = cfg[:, 1]
        x0[2::3] = 0.101  # Strictly feasible initial radius
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            
            if res.success and np.isfinite(res.fun):
                cx = res.x[0::3]
                cy = res.x[1::3]
                centers_tmp = np.column_stack((cx, cy))
                
                # Exact LP refinement for radii
                lp_r = solve_lp_radii(centers_tmp)
                if lp_r is not None:
                    s = np.sum(lp_r)
                    if s > best_sum:
                        best_sum = s
                        best_centers = centers_tmp.copy()
                        best_radii = lp_r.copy()
        except Exception:
            continue
            
    # Fallback if optimization unexpectedly fails
    if best_centers is None:
        best_centers = get_hex_init(0.10)
        best_radii = np.full(N, 0.10)
        best_sum = np.sum(best_radii)
        
    # Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-9:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    # Apply conservative scale
    best_radii *= max(scale * 0.999999, 0.0)
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
