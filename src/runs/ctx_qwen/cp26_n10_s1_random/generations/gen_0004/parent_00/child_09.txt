To maximize the sum of radii for 26 circles in a unit square, I will employ a robust two-stage optimization strategy that overcomes local minima traps common in circle packing:

1. **Center Optimization via Clearance Maximization**: Instead of directly optimizing radii and positions together (which creates a highly non-convex landscape), I first optimize only the centers to maximize the minimum clearance (distance to boundaries and half the pairwise distance). This proxy objective strongly correlates with the optimal equal-radius packing configuration and is much smoother for derivative-free optimizers like Nelder-Mead.
2. **Diverse Hexagonal Initializations**: I generate multiple starting configurations using hexagonal lattices with various row distributions that sum to 26 (e.g., `[6, 5, 6, 5, 4]`), plus grids and random perturbations. This covers the known high-density layouts for this circle count.
3. **Linear Programming for Radii**: Once tight center configurations are found, I solve a Linear Program (LP) to exactly maximize $\sum r_i$ subject to $r_i + r_j \le d_{ij}$ and boundary constraints. LP is convex, extremely fast for $N=26$, and extracts the absolute best radii for any fixed center layout, including allowing unequal radii if beneficial.
4. **SLSQP Local Refinement**: The best configuration from the LP stage is refined using SLSQP on all variables $(x, y, r)$ simultaneously to squeeze out remaining slack and smooth out numerical boundaries.
5. **Strict Safety Scaling**: A final deterministic scaling step guarantees all constraints are strictly satisfied within the checker's $10^{-12}$ tolerance.

All helper functions are defined at the top level with no closures or lambda functions to comply with the rules.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mx, 0.0)))
        
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def compute_clearance(x):
    """Computes maximum equal radius feasible for given centers."""
    c = x.reshape(N, 2)
    d_wall = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                        np.minimum(c[:, 1], 1.0 - c[:, 1]))
    min_wall = np.min(d_wall)
    
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    d_pair = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(d_pair, np.inf)
    min_pair = np.min(d_pair) / 2.0
    
    return min(min_wall, min_pair)

def neg_clearance(x):
    return -compute_clearance(x)

def generate_hex_config(row_counts, r_init=0.1):
    """Generates a hexagonal lattice configuration for N circles."""
    pts = []
    y = r_init
    for idx, cnt in enumerate(row_counts):
        shift = r_init if idx % 2 == 1 else 0.0
        row_width = (cnt - 1) * 2 * r_init
        x_start = (1.0 - row_width) / 2.0 + shift
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2 * r_init
        y += np.sqrt(3) * r_init
    pts = np.array(pts[:N])
    return pts

def objective_slqp(x):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def constraints_slqp(x):
    """Inequality constraints for SLSQP: all must be >= 0."""
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    c = []
    
    # Boundary constraints
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    # Pairwise separation constraints
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    d2 = cx_m**2 + cy_m**2
    rs2 = r_m**2
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.extend((d2 - rs2)[mask])
    
    return np.array(c)

def run_packing():
    np.random.seed(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Diverse row distributions for 26 circles that approximate hexagonal packing
    row_dists = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 6, 4], [4, 6, 6, 6, 4], [6, 5, 5, 5, 5],
        [5, 6, 5, 5, 5], [7, 5, 6, 4, 4], [6, 6, 6, 4, 4]
    ]
    
    starts = []
    for rc in row_dists:
        if sum(rc) >= N:
            c = generate_hex_config(rc, 0.1)
            starts.append(c)
            for _ in range(3):
                p = c + np.random.uniform(-0.02, 0.02, c.shape)
                p = np.clip(p, 0.05, 0.95)
                starts.append(p)
                
    # Grid starts
    for r0 in [0.09, 0.10]:
        gx = np.linspace(r0, 1-r0, 5)
        gy = np.linspace(r0, 1-r0, 5)
        c = np.array([(x, y) for y in gy for x in gx])
        c = np.vstack([c, [0.5, 0.5]])
        starts.append(c)
        
    bounds_slqp = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    
    # Phase 1: Nelder-Mead on centers to maximize clearance
    for cfg in starts:
        try:
            res_nm = minimize(neg_clearance, cfg.flatten(), method='Nelder-Mead',
                              options={'maxiter': 5000, 'xatol': 1e-9, 'fatol': 1e-12})
            c_opt = np.clip(res_nm.x.reshape(N, 2), 1e-6, 1.0 - 1e-6)
            
            # Phase 2: LP for exact optimal radii given centers
            r_lp, s_lp = solve_lp_radii(c_opt)
            if r_lp is not None and s_lp > best_sum:
                best_sum = s_lp
                best_centers = c_opt
                best_radii = r_lp
        except Exception:
            pass
            
    # Phase 3: SLSQP refinement on all variables
    if best_centers is not None:
        # Start slightly inside feasible region for SLSQP
        x0 = np.concatenate([best_centers.flatten(), best_radii * 0.99])
        try:
            res_slqp = minimize(objective_slqp, x0, method='SLSQP', bounds=bounds_slqp,
                                constraints={'type': 'ineq', 'fun': constraints_slqp},
                                options={'maxiter': 3000, 'ftol': 1e-14})
            if np.isfinite(res_slqp.fun):
                c_f = res_slqp.x[:2*N].reshape(N, 2)
                r_f = res_slqp.x[2*N:]
                s_f = np.sum(r_f)
                if s_f > best_sum:
                    best_sum = s_f
                    best_centers = c_f
                    best_radii = r_f
        except Exception:
            pass
            
    # Fallback configuration if optimization fails unexpectedly
    if best_centers is None:
        c_fb = generate_hex_config([6, 5, 6, 5, 4], 0.1)
        r_fb, s_fb = solve_lp_radii(c_fb)
        best_centers = c_fb
        best_radii = r_fb
        best_sum = s_fb
        
    # Strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_radii *= max(scale * 0.9999995, 0.0)
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
