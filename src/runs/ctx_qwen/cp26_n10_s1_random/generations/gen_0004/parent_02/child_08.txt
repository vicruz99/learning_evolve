import numpy as np
from scipy.optimize import minimize, linprog

def obj_equal_radius(v, n):
    """Objective: maximize t (minimize -t) for equal radius packing."""
    return -v[-1]

def cons_equal_radius(v, n):
    """Constraints for equal radius packing: walls and pairwise distances >= 2t."""
    xs = v[:n]
    ys = v[n:2*n]
    t = v[-1]
    
    # Boundary constraints: x >= t, 1-x >= t, y >= t, 1-y >= t
    c_wall = np.concatenate([xs - t, 1.0 - xs - t, ys - t, 1.0 - ys - t])
    
    # Pairwise constraints: ||c_i - c_j|| >= 2t
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    d = np.sqrt(dx**2 + dy**2)
    iu, ju = np.triu_indices(n, k=1)
    c_pairs = d[iu, ju] - 2.0*t
    
    return np.concatenate([c_wall, c_pairs])

def solve_lp_for_radii(centers, n):
    """Solves LP to maximize sum(r_i) given fixed centers."""
    c_obj = -np.ones(n)
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = centers[i]
        limits = (x, 1.0-x, y, 1.0-y)
        for lim in limits:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(lim)
            
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(d)
            
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    bounds = [(0.0, None)]*n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def hill_climb_centers_with_lp(centers, n, iters):
    """Local search: perturb centers one-by-one, accept if LP sum of radii improves."""
    best_c = centers.copy()
    radii, best_sum = solve_lp_for_radii(centers, n)
    best_r = radii
    
    base_step = 0.015
    for it in range(iters):
        step = base_step * (0.999 ** it)
        idx = np.random.randint(n)
        old_pos = best_c[idx].copy()
        
        # Perturb one center
        best_c[idx] += np.random.uniform(-step, step, 2)
        best_c[idx] = np.clip(best_c[idx], 1e-4, 1.0 - 1e-4)
        
        radii, curr_sum = solve_lp_for_radii(best_c, n)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_r = radii
        else:
            best_c[idx] = old_pos
            
    return best_c, best_r, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    
    # Generate diverse hexagonal lattice initializations
    configs = []
    r0 = 0.092
    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 6, 5, 4]
    ]
    
    for rp in row_patterns:
        pts = []
        y = r0
        for i, cnt in enumerate(rp):
            shift = r0 if i % 2 == 1 else 0.0
            row_w = (cnt - 1) * 2 * r0
            x_start = 0.5 - row_w / 2.0 + shift
            for k in range(cnt):
                if len(pts) >= n: break
                pts.append([x_start + k * 2 * r0, y])
            y += np.sqrt(3) * r0
        while len(pts) < n:
            pts.append([0.5, 0.5])
        cfg = np.array(pts[:n])
        
        # Add controlled perturbations to escape symmetry
        for _ in range(3):
            pert = cfg + np.random.uniform(-0.012, 0.012, cfg.shape)
            pert = np.clip(pert, 0.05, 0.95)
            configs.append(pert)
        configs.append(cfg)
        
    best_overall_sum = 0.0
    best_overall_c = None
    best_overall_r = None
    
    bounds_eq = [(0.0, 1.0)]*(2*n) + [(0.06, 0.14)]
    
    for cfg in configs:
        # Phase 1: Optimize centers for max equal radius (dense skeleton)
        v0 = np.concatenate([cfg.flatten(), [0.08]])
        try:
            res = minimize(obj_equal_radius, v0, args=(n,), method='SLSQP', 
                           bounds=bounds_eq, 
                           constraints={'type': 'ineq', 'fun': cons_equal_radius, 'args': (n,)},
                           options={'maxiter': 4000, 'ftol': 1e-12})
            c_opt = res.x[:2*n].reshape(n, 2)
        except Exception:
            c_opt = cfg
            
        # Phase 2: Hill-climb centers while re-optimizing radii via LP
        c_hill, r_hill, s_hill = hill_climb_centers_with_lp(c_opt, n, iters=1500)
        if s_hill > best_overall_sum:
            best_overall_sum = s_hill
            best_overall_c = c_hill
            best_overall_r = r_hill
            
    # Fallback if optimization fails unexpectedly
    if best_overall_c is None:
        best_overall_c = np.random.uniform(0.1, 0.9, (n, 2))
        best_overall_r = np.full(n, 0.08)
        best_overall_sum = np.sum(best_overall_r)
        
    # Phase 3: Strict safety scaling to guarantee checker validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_overall_c[i, 0], best_overall_c[i, 1], best_overall_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_overall_c[i,0]-best_overall_c[j,0], 
                         best_overall_c[i,1]-best_overall_c[j,1])
            rs = best_overall_r[i] + best_overall_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_overall_r *= scale * 0.9999999
    best_overall_sum = np.sum(best_overall_r)
    
    return best_overall_c, best_overall_r, float(best_overall_sum)
