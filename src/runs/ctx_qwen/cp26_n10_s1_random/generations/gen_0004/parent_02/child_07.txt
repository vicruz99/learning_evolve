import numpy as np
from scipy.optimize import minimize, linprog

def get_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed circle centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        mr = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mr, 1e-9)))
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_slsqp(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def con_slsqp(v, n):
    """Inequality constraints >= 0 for valid packing."""
    xs, ys, rs = v[:n], v[n:2*n], v[2*n:]
    c = np.empty(4 * n + n * (n - 1) // 2)
    idx = 0
    
    # Boundary constraints
    c[idx:idx + n] = xs - rs; idx += n
    c[idx:idx + n] = 1.0 - xs - rs; idx += n
    c[idx:idx + n] = ys - rs; idx += n
    c[idx:idx + n] = 1.0 - ys - rs; idx += n
    
    # Pairwise non-overlap constraints (squared distance >= squared radius sum)
    iu, ju = np.triu_indices(n, 1)
    dx = xs[iu] - xs[ju]
    dy = ys[iu] - ys[ju]
    dr = rs[iu] + rs[ju]
    c[idx:] = dx * dx + dy * dy - dr * dr
    
    return c

    rng = np.random.default_rng(42)

    patterns = [
