import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, 0.5)] * n
    
    num_b = 4 * n
    num_p = n * (n - 1) // 2
    A = np.zeros((num_b + num_p, n))
    b = np.zeros(num_b + num_p)
    
    idx = 0
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        A[idx, i] = 1.0; b[idx] = x; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - x; idx += 1
        A[idx, i] = 1.0; b[idx] = y; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - y; idx += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, None

def objective_func(x, n):
    """Objective: maximize sum of radii by optimizing centers."""
    c = x.reshape(n, 2)
    # Strictly keep centers inside bounds to ensure LP feasibility
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    s, _ = solve_lp_radii(c)
    return -s
