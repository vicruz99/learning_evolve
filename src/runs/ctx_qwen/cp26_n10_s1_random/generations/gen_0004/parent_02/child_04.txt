import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
IU, JU = np.triu_indices(N_CIRCLES, k=1)
NUM_PAIRS = len(IU)

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = N_CIRCLES
    c_clipped = np.clip(centers, 1e-4, 1.0 - 1e-4)
    
    # Bounds for radii based on distance to boundaries
    bounds = []
    for i in range(n):
        x, y = c_clipped[i]
        mr = min(x, 1.0-x, y, 1.0-y)
        bounds.append((0.0, max(mr, 1e-7)))
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    A_ub = np.zeros((NUM_PAIRS, n))
    b_ub = np.zeros(NUM_PAIRS)
    for k in range(NUM_PAIRS):
        i, j = IU[k], JU[k]
        d = np.hypot(c_clipped[i,0]-c_clipped[j,0], c_clipped[i,1]-c_clipped[j,1])
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = d
        
    c_obj = -np.ones(n)  # Maximize sum(r) <=> minimize -sum(r)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return None, 0.0

def neg_lp_sum(x_flat):
    """Objective for Nelder-Mead: minimize negative LP sum."""
    centers = x_flat.reshape(-1, 2)
    _, s = solve_lp(centers)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    rng = np.random.default_rng(42)
    
    inits = []
    
    # 1. Hexagonal lattices with varying base radii
    for r0 in [0.085, 0.09, 0.095, 0.10, 0.105]:
        pts = []
        y = r0
        row = 0
        while y + r0 < 1.0 and len(pts) < n:
            shift = r0 if row % 2 == 1 else 0.0
            x = r0 + shift
            while x + r0 < 1.0 and len(pts) < n:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        while len(pts) < n:
            pts.append([0.5 + rng.uniform(-0.15, 0.15), 0.5 + rng.uniform(-0.15, 0.15)])
        inits.append(np.array(pts[:n]))
        
    # 2. Regular grids
    for sp in [0.18, 0.19, 0.20, 0.21]:
        pts = []
        for i in range(6):
            for j in range(5):
                if len(pts) < n:
                    pts.append([sp * (i + 0.5), sp * (j + 0.5)])
        inits.append(np.array(pts[:n]))
        
    # 3. Random dense configurations
    for _ in range(5):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Phase 1: Nelder-Mead optimization from each initialization
    for cfg in inits:
        try:
            res = minimize(neg_lp_sum, cfg.flatten(), method='Nelder-Mead',
                           options={'maxiter': 800, 'xatol': 1e-6, 'fatol': 1e-8})
            c_opt = res.x.reshape(-1, 2)
            radii, s = solve_lp(c_opt)
            if radii is not None and s > best_sum:
                best_sum = s
                best_c = c_opt.copy()
                best_r = radii.copy()
        except Exception:
            continue
            
    # Phase 2: Local restarts around the best configuration to escape local minima
    if best_c is not None:
        for _ in range(15):
            pert = best_c + rng.normal(0, 0.008, best_c.shape)
            try:
                res = minimize(neg_lp_sum, pert.flatten(), method='Nelder-Mead',
                               options={'maxiter': 400, 'xatol': 1e-6, 'fatol': 1e-8})
                c_opt = res.x.reshape(-1, 2)
                radii, s = solve_lp(c_opt)
                if radii is not None and s > best_sum:
                    best_sum = s
                    best_c = c_opt.copy()
                    best_r = radii.copy()
            except Exception:
                continue
                
    # Fallback in case optimization unexpectedly fails
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum = solve_lp(best_c)
        
    # Phase 3: Strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.999999
    best_sum = np.sum(best_r)
    
    return best_c, best_r, float(best_sum)
