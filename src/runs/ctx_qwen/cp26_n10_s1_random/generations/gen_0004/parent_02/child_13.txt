import numpy as np
from scipy.optimize import minimize, linprog

def cons_equal_t(v, n):
    xs = v[:n]
    ys = v[n:2*n]
    t = v[-1]
    c = []
    c.extend(xs - t)
    c.extend(1.0 - xs - t)
    c.extend(ys - t)
    c.extend(1.0 - ys - t)
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    d2 = dx**2 + dy**2
    iu, ju = np.triu_indices(n, 1)
    c.extend(d2[iu, ju] - 4.0 * t * t)
    return np.array(c)

def obj_equal_t(v):
    return -v[-1]

def cons_var(v, n):
    xs = v[:n]
    ys = v[n:2*n]
    rs = v[2*n:]
    c = []
    c.extend(xs - rs)
    c.extend(1.0 - xs - rs)
    c.extend(ys - rs)
    c.extend(1.0 - ys - rs)
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    d2 = dx**2 + dy**2
    rs2 = dr**2
    iu, ju = np.triu_indices(n, 1)
    c.extend(d2[iu, ju] - rs2[iu, ju])
    return np.array(c)

def obj_var(v, n):
    return -np.sum(v[2*n:])

def solve_lp_radii(centers, n):
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(max_r, 0.0)))
        
    A_ub = []
    b_ub = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def get_hex_init(n, row_counts, r0=0.095):
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        row_width = (cnt - 1) * 2 * r0
        x_start = 0.5 - row_width / 2.0 + shift
        for k in range(cnt):
            if len(pts) >= n: break
            x = x_start + k * 2 * r0
            pts.append([x, y])
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    row_patterns = [

    configs = []

            pert = base + rng.uniform(-0.01, 0.01, base.shape)
            configs.append(np.clip(pert, 0.05, 0.95))
            
    best_t = 0.0
    best_v_t = None
    
    # Phase 1: Maximize equal radius t

                           options={'maxiter': 15000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                t_val = -res.fun
                if t_val > best_t:
                    best_t = t_val
                    best_v_t = res.x.copy()

            continue

        best_sum_var = n * best_t

    lp_radii, lp_sum = solve_lp_radii(final_centers, n)

    scale = 1.0
    for i in range(n):
        x, y, r = final_centers[i, 0], final_centers[i, 1], final_radii[i]

            rs_sum = final_radii[i] + final_radii[j]
            if rs_sum > 1e-12:
                scale = min(scale, d / rs_sum)
                
    final_radii *= scale * 0.9999999
    final_sum = np.sum(final_radii)
    
    return final_centers, final_radii, float(final_sum)
