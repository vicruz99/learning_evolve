import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    b_x = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    b_y = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    b_ub = np.minimum(b_x, b_y)
    
    # Pairwise distances
    dx = centers[:, 0, np.newaxis] - centers[:, 0]
    dy = centers[:, 1, np.newaxis] - centers[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    pair_dists = dists[np.triu_indices(n, k=1)]
    
    n_pair = n * (n - 1) // 2
    A_ub = np.zeros((n + n_pair, n))
    b = np.zeros(n + n_pair)
    
    A_ub[:n] = np.eye(n)
    b[:n] = b_ub
    
    iu, ju = np.triu_indices(n, k=1)
    A_ub[n:, iu] = 1.0
    A_ub[n:, ju] = 1.0
    b[n:] = pair_dists
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun
    return None, 0.0

def hill_climb_lp(centers_init):

    rng = np.random.default_rng(42)
    centers = centers_init.copy()
    radii, best_sum = solve_lp_radii(centers)
    
    if radii is None:
        return centers, np.zeros(N), 0.0

    best_radii = radii.copy()

        i = rng.integers(N)
        old_c = centers[i].copy()

            centers[i] = old_c
            continue

            centers[i] = old_c
            continue

            current_sum = s
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = r.copy()
        else:
            centers[i] = old_c
            
        T *= 0.9997 # Cooling schedule
        
    return best_centers, best_radii, best_sum

def objective_slqp(x):
    return -np.sum(x[2*N:])

def constraints_slqp(x):
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    c = []
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    cx_m = cx[:, np.newaxis] - cx[np.newaxis, :]
    cy_m = cy[:, np.newaxis] - cy[np.newaxis, :]
    dr_m = r[:, np.newaxis] + r[np.newaxis, :]
    
    d2 = cx_m**2 + cy_m**2
    dr2 = dr_m**2
    
    iu, ju = np.triu_indices(N, k=1)
    c.extend(d2[iu, ju] - dr2[iu, ju])
    return np.array(c)
