def get_constraints(x, n):
    """Compute all inequality constraints >= 0 for valid packing."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: circles must stay inside [0,1]x[0,1]
    c_bound = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Pairwise non-overlap constraints: squared distance >= squared sum of radii
    cx_diff = cx[:, np.newaxis] - cx[np.newaxis, :]
    cy_diff = cy[:, np.newaxis] - cy[np.newaxis, :]
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    d_sq = cx_diff**2 + cy_diff**2
    r_sq = r_sum**2
    
    idx = np.triu_indices(n, k=1)
    c_pair = d_sq[idx] - r_sq[idx]
    
    return np.concatenate([c_bound, c_pair])

def obj_func(x, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)  # Maximize sum(r) <=> minimize -sum(r)
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(max_r, 0.0)))
        
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0
