import numpy as np
from scipy.optimize import linprog

def solve_lp_radii(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j) and r_i <= dist(i, boundary)
    """
    n = centers.shape[0]
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    num_boundary = 4 * n
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_boundary + num_pairs, n))
    b_ub = np.zeros(num_boundary + num_pairs)
    
    idx = 0
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def generate_hex_configs(n, base_r, num_vars):
    """Generates diverse hexagonal lattice configurations."""
    configs = []
    for v in range(num_vars):
        pts = []
        y = base_r + v * 0.015
        row = 0
        shift_alt = v % 2
        while len(pts) < n:
            shift = base_r if (row + shift_alt) % 2 == 1 else 0.0
            x = base_r + shift
            while x + base_r <= 1.0 and len(pts) < n:
                pts.append([x, y])
                x += 2.0 * base_r
            y += np.sqrt(3) * base_r
            row += 1
        # Pad with center points if lattice yields fewer than n
        while len(pts) < n:
            pts.append([0.5, 0.5])
        configs.append(np.array(pts[:n]))
    return configs

def hill_climb_lp(init_centers, n, iterations=3000):
    """Hill-climbing on center coordinates, using LP to evaluate objective."""
    centers = init_centers.copy()
    radii, best_sum = solve_lp_radii(centers)
    
    if radii is None:
        return init_centers, np.zeros(n), 0.0
        
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = float(best_sum)
    
    step = 0.025
    no_improve = 0
    stagnation_threshold = 400
    
    for it in range(iterations):
        i = np.random.randint(n)
        old_c = centers[i].copy()
        
        # Perturb one random circle
        centers[i, 0] += np.random.uniform(-step, step)
        centers[i, 1] += np.random.uniform(-step, step)
        centers[i] = np.clip(centers[i], 1e-4, 1.0 - 1e-4)
        
        radii, curr_sum = solve_lp_radii(centers)
        if radii is not None and curr_sum > best_sum:
            best_centers = centers.copy()
            best_radii = radii
            best_sum = float(curr_sum)
            no_improve = 0
        else:
            centers[i] = old_c
            no_improve += 1
            
        # Reset step size if stuck to escape local minima
        if no_improve > stagnation_threshold:
            step = 0.025
            no_improve = 0
            
        step *= 0.9990
        
    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None

    inits = []

        inits.append(pts)

        c, r, s = hill_climb_lp(cfg, n, iterations=2500)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r

    if best_radii is not None and best_sum > 0:
        scale = 0.9999995
        best_radii *= scale
        best_sum = float(np.sum(best_radii))
        
    # Fallback if optimization unexpectedly fails
    if best_centers is None:
        best_centers = inits[0]
        best_radii = np.full(n, 0.08)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
