import numpy as np
from scipy.optimize import linprog

def solve_radii_lp(centers, n, idx_i, idx_j):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j) and r_i <= min dist to boundary.
    """
    c_obj = -np.ones(n)
    
    # Upper bounds for each radius based on distance to boundaries
    limits = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    bounds = [(0.0, max(0.0, lim)) for lim in limits]
    
    # Pairwise distance constraints: r_i + r_j <= ||c_i - c_j||
    diffs = centers[idx_i] - centers[idx_j]
    b_ub = np.sqrt(np.sum(diffs**2, axis=1))
    
    A_ub = np.zeros((len(idx_i), n))
    A_ub[np.arange(len(idx_i)), idx_i] = 1.0
    A_ub[np.arange(len(idx_i)), idx_j] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def run_simulated_annealing(centers_init, n, idx_i, idx_j, steps=2000):
    """Optimizes centers to maximize LP-optimal sum of radii using Simulated Annealing."""

    best_centers = centers.copy()
    best_radii = radii.copy()

        # Perturb a random circle

        r_new, s_new = solve_radii_lp(c_new, n, idx_i, idx_j)

                best_sum = curr_sum
                best_centers = centers.copy()

    return best_centers, best_radii, best_sum

def generate_hex_lattice(row_counts, padding=0.02):
    """Generates a hexagonal lattice configuration for a given row distribution."""
    pts = []
    r_est = 0.10
    dy = np.sqrt(3) * r_est
    y = r_est
    
    for row_idx, cnt in enumerate(row_counts):
        shift = r_est if row_idx % 2 == 1 else 0.0
        x_start = r_est + shift
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += dy
        
    pts = np.array(pts[:sum(row_counts)])
    if len(pts) == 0:
        return None
        
    # Normalize and scale to fit in [padding, 1-padding]
    min_c = pts.min(axis=0)
    max_c = pts.max(axis=0)
    span = max_c - min_c
    if span.max() == 0:
        return None
        
    scale = (1.0 - 2 * padding) / span.max()
    pts = (pts - min_c) * scale + padding
    return pts
