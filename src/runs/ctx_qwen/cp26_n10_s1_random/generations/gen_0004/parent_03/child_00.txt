import numpy as np
from scipy.optimize import minimize, linprog

def obj_func(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:3*n])

def con_func(vars_arr, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid packing."""
    c = vars_arr[:2*n].reshape(n, 2)
    r = vars_arr[2*n:3*n]
    
    # Boundary constraints
    bc = np.concatenate([c[:, 0] - r, 1 - c[:, 0] - r, c[:, 1] - r, 1 - c[:, 1] - r])
    
    # Pairwise non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    diffs = c[idx_i] - c[idx_j]
    dist_sq = np.sum(diffs**2, axis=1)
    r_sum = r[idx_i] + r[idx_j]
    pc = dist_sq - r_sum**2
    
    return np.concatenate([bc, pc])

def compute_lp_sum(centers, n, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 0.0)
    
    diffs = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    
    A_ub = np.zeros((len(idx_i), n))
    A_ub[np.arange(len(idx_i)), idx_i] = 1.0
    A_ub[np.arange(len(idx_i)), idx_j] = 1.0
    
    bounds = [(0.0, lim) for lim in limits]
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    idx_i, idx_j = np.triu_indices(n, k=1)
    rng = np.random.default_rng(42)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds_vars = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    configs = []
    
    # 1. Structured hexagonal patterns with varied row counts
    row_counts_list = [
        [5, 5, 5, 5, 4, 2], [5, 5, 5, 5, 5, 1], [6, 5, 5, 5, 5], 
        [5, 6, 5, 5, 5], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [7, 6, 6, 5, 2], [6, 6, 6, 5, 3], [5, 5, 5, 6, 5]
    ]
    
    for rc in row_counts_list:
        pts = []
        y = 0.09
        row_idx = 0
        while len(pts) < n:
            cnt = rc[row_idx % len(rc)]
            shift = 0.09 if row_idx % 2 == 1 else 0.0
            row_width = (cnt - 1) * 0.18
            x_start = (1.0 - row_width) / 2.0 + shift
            for _ in range(cnt):
                if len(pts) < n:
                    pts.append([x_start, y])
                    x_start += 0.18
            y += 0.09 * np.sqrt(3)
            row_idx += 1
        configs.append(np.array(pts[:n]))
        
    # 2. Regular grid starts with different spacings
    for s in [0.17, 0.18, 0.19, 0.20, 0.21]:
        pts = []

            while x < 1.0 - s/2 and len(pts) < n:

                x += s
            y += s
        configs.append(np.array(pts[:n]))
        
    # 3. Perturbed versions of best structured starts

                    scale = min(scale, d / rs)
        best_radii *= scale * 0.99999999
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
