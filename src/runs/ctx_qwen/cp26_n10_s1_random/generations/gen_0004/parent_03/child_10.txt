def compute_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Upper bounds for each radius based on distance to boundaries
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(0.0, lim)) for lim in lims]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A = np.zeros((m, n))
    A[np.arange(m), idx_i] = 1.0
    A[np.arange(m), idx_j] = 1.0
    b = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def obj_slsqp(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def cons_slsqp(x, n, idx_i, idx_j):

    c = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    
    # Boundary constraints
    bc = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, 
                         c[:, 1] - r, 1.0 - c[:, 1] - r])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    diff = c[idx_i] - c[idx_j]
    dist_sq = np.sum(diff**2, axis=1)
    r_sum = r[idx_i] + r[idx_j]
    pc = dist_sq - r_sum**2
    
    return np.concatenate([bc, pc])
