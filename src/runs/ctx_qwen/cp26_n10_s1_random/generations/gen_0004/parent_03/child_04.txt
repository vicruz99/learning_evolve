from scipy.optimize import minimize, linprog

def obj_equal(x, n):
    """Objective function for equal-radius optimization: maximize r => minimize -r"""
    return -x[-1]

def cons_equal(x, n):
    """Inequality constraints for equal-radius packing: all must be >= 0"""
    r = x[-1]
    c = x[:-1].reshape(n, 2)
    cx, cy = c[:, 0], c[:, 1]
    
    # Boundary constraints
    bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise separation constraints: ||c_i - c_j||^2 >= (2r)^2
    diffs = c[:, None, :] - c[None, :, :]
    d2 = np.sum(diffs**2, axis=2)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pc = d2[mask] - 4.0 * r**2
    
    return np.concatenate([bc, pc])

def get_lp_data(centers, idx_i, idx_j):

    n = centers.shape[0]
    limits = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    bounds = [(0.0, lim) for lim in np.maximum(limits, 0.0)]
    
    A_ub = np.zeros((len(idx_i), n))

    A_ub[np.arange(len(idx_i)), idx_j] = 1.0
    
    diffs = centers[idx_i] - centers[idx_j]
    b_ub = np.sqrt(np.sum(diffs**2, axis=1))
    
    return A_ub, b_ub, bounds
