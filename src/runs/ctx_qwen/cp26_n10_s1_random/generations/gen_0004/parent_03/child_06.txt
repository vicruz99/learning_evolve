import numpy as np
from scipy.optimize import linprog

def solve_radii_lp(centers, n, idx_i, idx_j, A_ub):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j) and r_i <= dist(i, boundary)
    """
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(0.0, lim)) for lim in limits]
    b_ub = np.linalg.norm(centers[idx_i] - centers[idx_j], axis=1)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

    rng = np.random.default_rng(42)

                y += np.sqrt(3) * r0
