import numpy as np
from scipy.optimize import linprog

def solve_lp_radii(centers, idx_i, idx_j, A_ub, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    x = centers[:, 0]
    y = centers[:, 1]
    ub = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    bounds = [(0.0, max(0.0, ub[k])) for k in range(n)]
    
    diffs = centers[idx_i] - centers[idx_j]
    b_ub = np.sqrt(np.sum(diffs**2, axis=1))
    
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def force_layout(centers, steps=1200):

    n = len(centers)
    c = centers.copy()
    dt = 0.008
    for _ in range(steps):

        diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        
        # Soft repulsion to prevent overlaps and encourage spacing
        mask = dists < 0.45
        safe_d = np.where(mask, dists, 1.0)

        dirs = diffs / safe_d[:, :, np.newaxis]
        forces += np.sum(dirs * rep[:, :, np.newaxis], axis=1)

            if c[i, 1] > 0.97: forces[i, 1] -= 30.0
            
        c += forces * dt
        c = np.clip(c, 0.02, 0.98)
    return c

def local_search(centers, idx_i, idx_j, A_ub, n, iters=8000, rng=None):

    best_c = centers.copy()

    step = 0.035

                best_c = cand
                best_s = s
        step *= 0.998
    return best_c, best_s
