import numpy as np
from scipy.optimize import minimize, linprog

def get_constraints_equal(vars, idx_i, idx_j):
    """Computes inequality constraints >= 0 for equal-radius packing."""
    n = 26
    cx = vars[:n]
    cy = vars[n:2*n]
    r = vars[2*n]
    
    # Boundary constraints
    bc = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    # Pairwise non-overlap constraints
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    pc = (dx**2 + dy**2) - 4.0 * r**2
    
    return np.concatenate([bc, pc])

def obj_equal(vars):
    """Objective: maximize shared radius r."""
    return -vars[2*26]

def solve_lp_radii(centers, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = 26
    x = centers[:, 0]
    y = centers[:, 1]
    
    limits = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    bounds = [(0.0, max(0.0, float(l))) for l in limits]
    
    A_ub = np.zeros((len(idx_i), n))
    A_ub[np.arange(len(idx_i)), idx_i] = 1.0
    A_ub[np.arange(len(idx_i)), idx_j] = 1.0
    
    diffs = centers[idx_i] - centers[idx_j]
    b_ub = np.sqrt(np.sum(diffs**2, axis=1))
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def run_packing():

    rng = np.random.default_rng(123)
    
    best_sum = 0.0

    best_radii = None

    ]
    
    for rows in row_counts_list:
        pts = []
        r0 = 0.105

        y = r0

            x_start = (1.0 - row_width) / 2.0 + shift
            for _ in range(cnt):
                if len(pts) < n:

                    x_start += 2.0 * r0
            y += dy
        if len(pts) >= n:
            configs.append(np.array(pts[:n]))

    base_grid = np.array([[0.1 + i*0.16, 0.1 + j*0.16] for i in range(5) for j in range(5)])

    configs.append(base_grid)

    # Rotated lattices to better fit the square boundaries

        rotated = base_grid - 0.5

        best_radii = np.full(n, 0.09)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
