import numpy as np
from scipy.optimize import minimize, linprog

def compute_clearance(centers):
    """Computes the maximum equal radius feasible for the given centers."""
    n = centers.shape[0]
    # Distance to boundaries
    d_wall = np.min(np.stack([centers[:, 0], 1.0 - centers[:, 0], 
                              centers[:, 1], 1.0 - centers[:, 1]], axis=1), axis=1)
    min_wall = np.min(d_wall)
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pair = np.min(dists) / 2.0
    
    return min(min_wall, min_pair)

def neg_clearance(x):
    """Objective for Nelder-Mead: minimize negative clearance."""
    return -compute_clearance(x.reshape(-1, 2))

def pack_obj(vars_arr, *args):
    """Objective for SLSQP: minimize negative radius."""
    return -vars_arr[-1]

def pack_constraints(vars_arr, n):
    """Inequality constraints for SLSQP: all must be >= 0."""
    c = vars_arr[:2 * n].reshape(n, 2)
    r = vars_arr[-1]
    
    # Boundary constraints
    bc = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, 
                         c[:, 1] - r, 1.0 - c[:, 1] - r])
    
    # Pairwise separation constraints (vectorized)
    diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist_sq = np.sum(diffs**2, axis=2)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pc = dist_sq[mask] - 4.0 * r * r
    
    return np.concatenate([bc, pc])

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    limits = np.min(np.stack([centers[:, 0], 1.0 - centers[:, 0], 
                              centers[:, 1], 1.0 - centers[:, 1]], axis=1), axis=1)
    bounds = [(0.0, max(0.0, lim)) for lim in limits]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diffs = centers[idx_i] - centers[idx_j]
    b_ub = np.sqrt(np.sum(diffs**2, axis=1))
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def run_packing():
    n = 26
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Generate diverse hexagonal lattice initializations
    starts = []
    for base_r in [0.085, 0.095, 0.105]:
        pts = []
        y = base_r
        row = 0
        while len(pts) < n + 5:
            x = base_r + (row % 2) * base_r
            while x + base_r <= 1.0 and len(pts) < n + 5:
                pts.append([x, y])
                x += 2.0 * base_r
            y += np.sqrt(3) * base_r
            row += 1
        arr = np.array(pts[:n])
        # Normalize to fit well inside [0, 1]
        arr = (arr - arr.min(axis=0)) / (arr.max(axis=0) - arr.min(axis=0)) * 0.92 + 0.04
        starts.append(arr)
        
    np.random.seed(42)
    for base in starts:
        for _ in range(6):
            # Perturb configuration
            cfg = base + np.random.uniform(-0.012, 0.012, (n, 2))
            cfg = np.clip(cfg, 0.02, 0.98)
            
            # Phase 1: Nelder-Mead global search on centers
            res_nm = minimize(neg_clearance, cfg.flatten(), method='Nelder-Mead',
                              options={'maxiter': 25000, 'xatol': 1e-9, 'fatol': 1e-11})
            c_opt = res_nm.x.reshape(n, 2)
            c_opt = np.clip(c_opt, 1e-6, 1.0 - 1e-6)
            
            # Phase 2: SLSQP local refinement for equal radius
            r_eq = compute_clearance(c_opt)
            vars_init = np.concatenate([c_opt.flatten(), [max(0.05, r_eq * 0.999)]])
            bounds_sqp = [(0.0, 1.0)] * (2 * n) + [(0.05, 0.15)]
            cons_sqp = {'type': 'ineq', 'fun': pack_constraints, 'args': (n,)}
            
            try:
                res_sqp = minimize(pack_obj, vars_init, args=(n,), method='SLSQP',
                                   bounds=bounds_sqp, constraints=cons_sqp,
                                   options={'maxiter': 2000, 'ftol': 1e-13})
                if res_sqp.success:
                    c_opt = res_sqp.x[:2 * n].reshape(n, 2)
            except Exception:
                pass
                
            # Phase 3: LP to maximize sum of (potentially unequal) radii
            r_lp, s_lp = solve_lp(c_opt)
            if s_lp > best_sum:
                best_sum = s_lp
                best_c = c_opt.copy()
                best_r = r_lp.copy()
                
    # Fallback safety
    if best_r is not None:
        best_r *= 0.999999
        best_sum = float(np.sum(best_r))
    else:
        best_c = np.random.uniform(0.2, 0.8, (n, 2))
        best_r, best_sum = solve_lp(best_c)
        
    return best_c, best_r, best_sum
