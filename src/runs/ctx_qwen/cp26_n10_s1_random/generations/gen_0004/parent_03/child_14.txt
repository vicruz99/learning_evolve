import numpy as np
from scipy.optimize import minimize, linprog

def compute_constraints_var(vars_flat, n):
    """Constraints for variable radii optimization: dist^2 >= (ri+rj)^2"""
    cx = vars_flat[:n]
    cy = vars_flat[n:2*n]
    r = vars_flat[2*n:]
    
    bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    i, j = np.triu_indices(n, k=1)
    dx = cx[i] - cx[j]
    dy = cy[i] - cy[j]
    dist_sq = dx**2 + dy**2
    r_sum = r[i] + r[j]
    pc = dist_sq - r_sum**2
    
    return np.concatenate([bc, pc])

def compute_constraints_eq(vars_flat, n):
    """Constraints for equal radius optimization: dist^2 >= (2t)^2"""
    c = vars_flat[:2*n].reshape(n, 2)
    t = vars_flat[-1]
    
    bc = np.concatenate([c[:, 0] - t, 1.0 - c[:, 0] - t, c[:, 1] - t, 1.0 - c[:, 1] - t])
    
    i, j = np.triu_indices(n, k=1)
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dist_sq = dx**2 + dy**2
    pc = dist_sq - 4.0 * t**2
    
    return np.concatenate([bc, pc])

def obj_var(vars_flat, n):
    return -np.sum(vars_flat[2*n:])

def obj_eq(vars_flat, n):
    return -vars_flat[-1]

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # 1. Generate diverse initial configurations
    configs = []
    
    # Hexagonal pattern: 6-5-6-5-4 rows (optimal topology for N=26)
    rows = [6, 5, 6, 5, 4]
    r_init = 0.08
    pts = []
    y = r_init
    for idx, cnt in enumerate(rows):
        shift = r_init if idx % 2 == 1 else 0.0
        x = r_init + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_init
        y += np.sqrt(3) * r_init
    configs.append(np.array(pts[:n]))
    
    # Perturbed hexagons
    for _ in range(4):
        configs.append(configs[0] + rng.uniform(-0.04, 0.04, (n, 2)))
        
    # Regular 5x5 grid + center
    grid = np.array([[0.1 + i*0.18, 0.1 + j*0.18] for j in range(5) for i in range(5)])
    grid = np.vstack([grid, [0.5, 0.5]])
    configs.append(grid)
    configs.append(grid + rng.uniform(-0.03, 0.03, (n, 2)))
    
    # Random dense
    for _ in range(3):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # 2. Optimization Loop
    for cfg in configs:
        cfg = np.clip(cfg, 0.05, 0.95)
        
        # Compute a strictly feasible initial radius for this layout
        dists = np.sqrt(np.sum((cfg[:, None, :] - cfg[None, :, :])**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        min_pair = np.min(dists, axis=1) / 2.0
        min_wall = np.minimum(np.minimum(cfg[:, 0], 1.0 - cfg[:, 0]), 
                              np.minimum(cfg[:, 1], 1.0 - cfg[:, 1]))
        r_start = np.minimum(min_pair, min_wall)
        r_start = np.maximum(r_start, 0.01)
        
        # --- Track A: Variable Radii ---
        x0_var = np.concatenate([cfg.flatten(), r_start])
        bounds_var = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
        try:
            res_var = minimize(obj_var, x0_var, args=(n,), method='SLSQP', bounds=bounds_var,
                          constraints={'type': 'ineq', 'fun': compute_constraints_var, 'args': (n,)},
                          options={'maxiter': 8000, 'ftol': 1e-12})
            
            if np.isfinite(res_var.fun):
                if np.min(compute_constraints_var(res_var.x, n)) >= -1e-7:
                    s = np.sum(res_var.x[2*n:])
                    if s > best_sum:
                        best_sum = s
                        best_centers = res_var.x[:2*n].reshape(n, 2).copy()
                        best_radii = res_var.x[2*n:].copy()
        except Exception:
            pass
            
        # --- Track B: Equal Radii ---
        x0_eq = np.concatenate([cfg.flatten(), [np.mean(r_start)]])

        except Exception:
            pass

        best_radii = np.full(n, 0.08)

    limits = np.minimum(np.minimum(best_centers[:, 0], 1.0 - best_centers[:, 0]),

    limits = np.maximum(limits, 0.0)
    bounds_lp = [(0.0, lim) for lim in limits]
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    A_ub = np.zeros((len(i_idx), n))
    A_ub[np.arange(len(i_idx)), i_idx] = 1.0
    A_ub[np.arange(len(i_idx)), j_idx] = 1.0
    
    diffs = best_centers[i_idx] - best_centers[j_idx]
    b_ub = np.sqrt(np.sum(diffs**2, axis=1))
    
    try:
        lp_res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if lp_res.success and np.isfinite(lp_res.fun):
            best_radii = lp_res.x
            best_sum = np.sum(best_radii)
    except Exception:
        pass
        
    # Safety shrink to guarantee strict validity against 1e-12 tolerance
    best_radii *= 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
