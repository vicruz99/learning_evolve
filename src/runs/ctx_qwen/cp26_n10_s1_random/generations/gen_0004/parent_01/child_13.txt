import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_clearance(centers):
    centers = np.clip(centers, 0.0, 1.0)
    d_wall = np.min(np.array([
        centers[:, 0], 1.0 - centers[:, 0],
        centers[:, 1], 1.0 - centers[:, 1]
    ]), axis=0)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    d_pair = np.min(dists) / 2.0
    
    return np.min(d_wall), np.min(d_pair)

def neg_clearance(centers_flat):
    centers = centers_flat.reshape(N, 2)
    cw, cp = compute_clearance(centers)
    return -np.min(cw, cp)

def generate_hex_configs():
    configs = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4],
        [5, 6, 4, 6, 5], [6, 5, 5, 6, 4], [5, 7, 5, 5, 4]
    ]
    
    for pat in patterns:
        if sum(pat) != N: 
            continue
        for r0 in [0.09, 0.095, 0.105, 0.11]:
            pts = []
            y = r0
            for ri, cnt in enumerate(pat):
                shift_x = r0 if ri % 2 == 1 else 0.0
                x = r0 + shift_x
                for _ in range(cnt):
                    if len(pts) >= N: 
                        break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            if len(pts) == N:
                configs.append(np.array(pts))
                
    np.random.seed(42)
    for cfg in configs[:10]:
        for _ in range(8):
            cfg_pert = cfg + np.random.uniform(-0.02, 0.02, cfg.shape)
            cfg_pert = np.clip(cfg_pert, 0.05, 0.95)
            configs.append(cfg_pert)
            
    return configs

def run_packing():
    configs = generate_hex_configs()
    
    best_clearance = -1.0
    best_centers = None
    
    for cfg in configs:
        res = minimize(neg_clearance, cfg.flatten(), method='Nelder-Mead',
                       options={'maxiter': 10000, 'xatol': 1e-9, 'fatol': 1e-11})
        c_opt = res.x.reshape(N, 2)
        c_opt = np.clip(c_opt, 0.0, 1.0)
        cw, cp = compute_clearance(c_opt)
        clr = np.min(cw, cp)
        
        if clr > best_clearance:
            best_clearance = clr
            best_centers = c_opt
            
    if best_centers is None:
        best_centers = configs[0]
        
    n = N
    c_obj = -np.ones(n)
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs + 4 * n, n))
    b_ub = np.zeros(num_pairs + 4 * n)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                            best_centers[i, 1] - best_centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    for i in range(n):
        x, y = best_centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    lp_res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if lp_res.success and np.all(lp_res.x >= -1e-9):
        radii = np.maximum(lp_res.x, 0.0)
    else:
        radii = np.full(n, best_clearance)
        
    radii *= 0.9999999
    sum_radii = np.sum(radii)
    
    return best_centers, radii, float(sum_radii)
