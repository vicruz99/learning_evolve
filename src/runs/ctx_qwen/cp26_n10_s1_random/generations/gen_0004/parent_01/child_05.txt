import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs + 4 * n, n))
    b_ub = np.zeros(num_pairs + 4 * n)
    
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    i_idx, j_idx = np.triu_indices(n, k=1)
    for k in range(len(i_idx)):
        i, j = i_idx[k], j_idx[k]
        dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dist
        
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    idx = num_pairs
    for i in range(n):
        x, y = centers[i]
        limits = (x, 1.0 - x, y, 1.0 - y)
        for lim in limits:
            A_ub[idx, i] = 1.0
            b_ub[idx] = max(lim, 0.0)
            idx += 1
            
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def hill_climb_centers(init_centers, iterations=1500):
    """Stochastic hill-climbing on center coordinates, using LP to evaluate objective."""
    centers = init_centers.copy()
    radii, best_sum = solve_lp(centers)

        best_sum = 0.0
        radii = np.zeros(N)

    best_radii = radii.copy()

        centers[i] += np.random.uniform(-step_size, step_size, 2)
        centers[i] = np.clip(centers[i], 1e-6, 1.0 - 1e-6)

            step_size = max(1e-5, step_size * 0.995)
            
    return best_centers, best_radii, best_sum

def objective_joint(vars_flat):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def constraints_joint(vars_flat):

    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    b = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    i, j = np.triu_indices(N, k=1)
    dx = x[i] - x[j]
    dy = y[i] - y[j]
    dist = np.sqrt(dx**2 + dy**2)
    p = dist - (r[i] + r[j])
    
    return np.concatenate([b, p])

def generate_hex_patterns():
    """Generates hexagonal lattice configurations with varying row counts."""
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4],
        [5, 7, 5, 5, 4], [7, 5, 6, 5, 3], [5, 6, 4, 6, 5]
    ]
    configs = []
    for pat in patterns:
        if sum(pat) != N: 
            continue

        r0 = 0.1
        y = r0

                pts.append([x, y])
                x += 2.0 * r0
                cnt -= 1
            y += r0 * np.sqrt(3)

        scale = 0.90 / (max_c - min_c).max()
        arr = (arr - min_c) * scale + (1.0 - scale) / 2.0
        configs.append(arr)
    return configs
