import numpy as np
from scipy.optimize import minimize, linprog

N = 26
MASK = np.triu(np.ones((N, N), dtype=bool), k=1)

def objective(x):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(x[2*N:])

def constraints(x):
    """Inequality constraints: all must be >= 0."""
    cx, cy, r = x[:N], x[N:2*N], x[2*N:]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    # Pairwise non-overlap: squared distance >= squared sum of radii
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    
    c = np.concatenate([c, (cx_m**2 + cy_m**2)[MASK] - r_m[MASK]**2])
    return c

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    bounds = []
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        mr = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mr, 0.0)))
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def generate_hex(row_counts, r0, perturb=0.0):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r0
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
        
    pts = np.array(pts[:N])
    if perturb > 0:
        pts += np.random.uniform(-perturb, perturb, pts.shape)
    return np.clip(pts, 0.05, 0.95)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # Bounds: centers in [0, 1], radii in [1e-5, 0.5]
    bounds = [(0.0, 1.0)] * 2*N + [(1e-5, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [5,5,5,5,6], [6,6,5,5,4], [5,6,4,6,5], [7,5,5,5,4],
        [5,5,7,4,5], [6,4,6,5,5], [5,7,5,4,5], [4,5,6,6,5]
    ]
    
    starts = []
    for pat in patterns:
        if sum(pat) != 26: 
            continue
        for r0 in [0.075, 0.085, 0.095, 0.105]:
            starts.append(generate_hex(pat, r0))
            for _ in range(4):
                starts.append(generate_hex(pat, r0, perturb=0.025))
                
    # Add random valid starts for robustness
    for _ in range(8):
        starts.append(np.random.uniform(0.1, 0.9, (N, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: SLSQP + LP from multiple starts
    for cfg in starts:
        x0 = np.zeros(3*N)
        x0[:N] = cfg[:, 0]
        x0[N:2*N] = cfg[:, 1]
        x0[2*N:] = 0.05  # Strictly feasible initial radius
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            
            # Verify constraints are satisfied within numerical tolerance
            if np.min(constraints(res.x)) >= -1e-6:
                c_opt = np.column_stack((res.x[:N], res.x[N:2*N]))
                
                # LP refinement to extract maximal radii for these centers
                lp_r, lp_s = solve_lp_radii(c_opt)
                if lp_r is not None and lp_s > best_sum:
                    best_centers = c_opt
                    best_radii = lp_r
                    best_sum = lp_s
        except Exception:
            continue
            
    # Phase 2: Iterative local search around the best found configuration
    if best_centers is not None:
        cur_centers = best_centers.copy()
        cur_radii = best_radii.copy()
        
        for _ in range(8):
            # Perturb centers to escape local minima
            perturbed = cur_centers + np.random.uniform(-0.015, 0.015, cur_centers.shape)
            perturbed = np.clip(perturbed, 0.05, 0.95)
            
            x0 = np.zeros(3*N)
            x0[:N] = perturbed[:, 0]
            x0[N:2*N] = perturbed[:, 1]
            x0[2*N:] = cur_radii
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False})
                
                if np.min(constraints(res.x)) >= -1e-6:
                    c_new = np.column_stack((res.x[:N], res.x[N:2*N]))
                    lp_r, lp_s = solve_lp_radii(c_new)
                    
                    if lp_r is not None and lp_s > best_sum:
                        best_centers = c_new
                        best_radii = lp_r
                        best_sum = lp_s
                        cur_centers = c_new
                        cur_radii = lp_r
            except Exception:
                pass
                
    # Final safety scaling to guarantee strict validity against checker tolerance
    if best_radii is not None:
        scale = 1.0
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-9:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
                
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-9:
                    scale = min(scale, d / rs)
                    
        best_radii *= max(scale * 0.9999999, 0.0)
        best_sum = float(np.sum(best_radii))
        
    # Fallback configuration (should not be reached with proper optimization)
    if best_centers is None:
        best_centers = np.random.uniform(0.1, 0.9, (N, 2))
        best_radii = np.full(N, 0.05)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
