import numpy as np
from scipy.optimize import minimize, linprog
from operator import itemgetter

N = 26
PAIR_INDICES = np.triu_indices(N, k=1)

def obj_slsqp(vars):
    """Objective: minimize negative sum of radii => maximize sum(r)"""
    return -np.sum(vars[2::3])

def cons_slsqp(vars):
    """
    Inequality constraints: all must be >= 0.
    Uses squared distances to avoid gradient singularities at contact points.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints
    b = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap: ||c_i - c_j||^2 - (r_i + r_j)^2 >= 0
    i_idx, j_idx = PAIR_INDICES
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    d2 = dx**2 + dy**2
    rs = (r[i_idx] + r[j_idx])**2
    
    o = d2 - rs
    return np.concatenate([b, o])

def get_lp_sum(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    
    # Boundary constraints matrix part
    A_ub_part = np.empty((4 * n, n))
    b_ub_part = np.empty(4 * n)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mx, 0.0)))
        
        A_ub_part[idx, i] = 1.0; b_ub_part[idx] = x; idx += 1
        A_ub_part[idx, i] = 1.0; b_ub_part[idx] = 1.0 - x; idx += 1
        A_ub_part[idx, i] = 1.0; b_ub_part[idx] = y; idx += 1
        A_ub_part[idx, i] = 1.0; b_ub_part[idx] = 1.0 - y; idx += 1
        
    # Pairwise constraints matrix part (vectorized)
    i_idx, j_idx = PAIR_INDICES
    d_vals = np.hypot(centers[i_idx, 0] - centers[j_idx, 0], 
                      centers[i_idx, 1] - centers[j_idx, 1])
                      
    pair_rows = np.zeros((len(i_idx), n))
    pair_rows[np.arange(len(i_idx)), i_idx] = 1.0
    pair_rows[np.arange(len(i_idx)), j_idx] = 1.0
    
    A_ub = np.vstack([A_ub_part, pair_rows])
    b_ub = np.hstack([b_ub_part, d_vals])
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def hill_climb(centers, iters=4000):
    """Perturb centers to maximize LP sum of radii."""
    best_c = centers.copy()
    _, best_s = get_lp_sum(best_c)
    step = 0.025
    
    for _ in range(iters):
        i = np.random.randint(N)
        old = best_c[i].copy()
        best_c[i] += np.random.uniform(-step, step, 2)
        best_c[i] = np.clip(best_c[i], 1e-5, 1.0 - 1e-5)
        
        _, s = get_lp_sum(best_c)
        if s > best_s:
            best_s = s
        else:
            best_c[i] = old
        step *= 0.9995
    return best_c

def generate_hex(r0, rows, perturb=0.0):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0
    for r in range(rows):
        shift = r0 if r % 2 == 1 else 0.0
        x = r0 + shift
        while x + r0 <= 1.0 and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    pts = np.array(pts[:N])
    if perturb > 0:
        pts += np.random.uniform(-perturb, perturb, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts
