import numpy as np
from scipy.optimize import linprog, differential_evolution

N = 26

def solve_lp_sum(centers_flat):
    """
    Solves the LP to maximize sum of radii for a fixed set of centers.
    Returns the maximum sum of radii.
    """
    centers = centers_flat.reshape(-1, 2)
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Maximize sum(r) <=> minimize -sum(r)
    
    idx = 0
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs + 4 * n, n))
    b_ub = np.zeros(num_pairs + 4 * n)
    
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun
    except Exception:
        return 0.0
    return 0.0

def de_objective(centers_flat):
    """Objective for DE: minimize negative sum of radii."""
    return -solve_lp_sum(centers_flat)

def get_final_radii(centers):
    """
    Extracts the optimal radii for the given centers using LP.
    Applies a tiny safety margin to ensure strict validation pass.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    idx = 0
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs + 4 * n, n))
    b_ub = np.zeros(num_pairs + 4 * n)
    
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            # Apply strict safety scaling to prevent 1e-12 tolerance failures
            return res.x * 0.999999
    except Exception:
        pass
    return np.full(n, 0.0)

def run_packing():
    np.random.seed(42)
    
    # Search space for centers: strictly inside the unit square
    bounds = [(0.01, 0.99)] * (2 * N)
    
    # Differential Evolution for global optimization of center coordinates
    res = differential_evolution(
        de_objective,
        bounds,
        strategy='best1bin',
        maxiter=300,
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=42,
        polish=True,  # Runs Nelder-Mead local search on best result
        workers=1
    )
    
    best_centers = np.clip(res.x.reshape(-1, 2), 0.0, 1.0)
    best_radii = get_final_radii(best_centers)
    sum_r = float(np.sum(best_radii))
    
    return best_centers, best_radii, sum_r
