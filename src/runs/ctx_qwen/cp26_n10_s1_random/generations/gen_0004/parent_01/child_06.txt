import numpy as np
from scipy.optimize import linprog

N = 26

def solve_lp_radii(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j) and r_i <= dist(i, boundary)
    """
    n = centers.shape[0]
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(max_r, 0.0)))
        
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, -1e9

def hill_climb_lp(init_centers, n, iterations=5000):
    """
    Stochastic hill-climbing on center coordinates, using LP to evaluate objective.
    Uses exponentially decaying step size for fine-tuning.
    """
    centers = init_centers.copy()
    best_centers = centers.copy()
    _, best_sum = solve_lp_radii(centers)
    if best_sum < 0:
        return best_centers, best_sum
        
    step_base = 0.03
    for it in range(iterations):
        # Exponential decay of step size
        step = step_base * (0.9990 ** it)
        
        i = np.random.randint(n)
        old_c = centers[i].copy()
        
        # Perturb one random circle
        centers[i] += np.random.uniform(-step, step, 2)
        # Keep strictly inside [0,1] to ensure LP feasibility
        centers[i] = np.clip(centers[i], 1e-4, 1.0 - 1e-4)
        
        radii, curr_sum = solve_lp_radii(centers)
        if radii is not None and curr_sum > best_sum + 1e-10:
            best_sum = curr_sum
            best_centers = centers.copy()
        else:
            centers[i] = old_c  # Revert if no improvement
            
    return best_centers, best_sum

def generate_hex_config(row_counts, r_init, shift_y=0.0):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r_init + shift_y
    for ri, cnt in enumerate(row_counts):
        shift_x = r_init if ri % 2 == 1 else 0.0
        x_start = r_init + shift_x
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x_start, y])
            x_start += 2.0 * r_init
        y += r_init * np.sqrt(3)
        
    pts = np.array(pts[:N])
    # Normalize to fit comfortably in [0,1] with padding
    mn = pts.min(axis=0)
    mx = pts.max(axis=0)
    span = max(mx[0]-mn[0], mx[1]-mn[1])
    if span > 1e-5:
        pts = (pts - mn) / span * 0.85 + 0.075
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # Hexagonal row patterns that sum to 26
    patterns = [
        [6, 6, 5, 5, 4], [5, 6, 6, 5, 4], [5, 5, 6, 6, 4],
        [4, 6, 6, 6, 4], [6, 5, 6, 5, 4], [5, 7, 5, 5, 4],
        [5, 6, 4, 6, 5], [7, 5, 6, 4, 4], [6, 6, 6, 4, 4],
        [5, 5, 5, 6, 5], [6, 5, 5, 5, 5], [5, 5, 5, 5, 6]
    ]
    
    starts = []
    # Generate structured hexagonal starts with varying base radii
    for pat in patterns:
        if sum(pat) != 26: 
            continue
        for r0 in [0.09, 0.10, 0.11]:
            starts.append(generate_hex_config(pat, r0))
            
    # Add a regular grid + center start
    grid_pts = []
    for r in range(5):
        for c in range(5):
            grid_pts.append([0.10 + c*0.20, 0.10 + r*0.20])
    grid_pts.append([0.5, 0.5])
    starts.append(np.array(grid_pts[:26]))
    
    # Add random starts for robustness
    for _ in range(5):
        starts.append(np.random.uniform(0.1, 0.9, size=(N, 2)))
        
    # Phase 1: Evaluate all starts with LP to find the most promising candidates
    scored_starts = []
    for s in starts:
        _, sc = solve_lp_radii(s)
        scored_starts.append((sc, s))
        
    # Sort by objective value (higher is better)
    scored_starts.sort(key=lambda x: x[0], reverse=True)
    
    best_sum = scored_starts[0][0]
    best_centers = scored_starts[0][1].copy()
    
    # Phase 2: Run hill-climbing refinement on the top 3 starts
    for sc, s_cfg in scored_starts[:3]:
        c_opt, s_opt = hill_climb_lp(s_cfg, N, iterations=6000)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            
    # Phase 3: Fine-tune the absolute best found so far with a shorter, precise search
    c_final, s_final = hill_climb_lp(best_centers, N, iterations=4000)
    if s_final > best_sum:
        best_sum = s_final
        best_centers = c_final
        
    # Phase 4: Exact LP refinement for radii given the optimized centers
    radii, final_sum = solve_lp_radii(best_centers)
    if radii is None:
        # Fallback to safe equal packing if LP fails unexpectedly
        radii = np.full(N, 0.08)
        final_sum = N * 0.08
        
    # Apply tiny safety margin to guarantee strict numerical validity against 1e-12 tolerance
    radii *= 0.9999999
    final_sum = float(np.sum(radii))
    
    return best_centers, radii, final_sum
