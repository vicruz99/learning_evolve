import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def obj_joint(vars_flat):
    return -np.sum(vars_flat[2::3])

def cons_joint(vars_flat):
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    c = np.empty(4*N + N*(N-1)//2)
    idx = 0
    c[idx:idx+N] = x - r; idx += N
    c[idx:idx+N] = 1.0 - x - r; idx += N
    c[idx:idx+N] = y - r; idx += N
    c[idx:idx+N] = 1.0 - y - r; idx += N
    i, j = np.triu_indices(N, k=1)
    dx = x[i] - x[j]
    dy = y[i] - y[j]
    c[idx:] = dx**2 + dy**2 - (r[i] + r[j])**2
    return c

def obj_equal(vars_flat):
    return -vars_flat[-1]

def cons_equal(vars_flat):
    x = vars_flat[0::2]
    y = vars_flat[1::2]
    r = vars_flat[-1]
    c = np.empty(4*N + N*(N-1)//2)
    idx = 0
    c[idx:idx+N] = x - r; idx += N
    c[idx:idx+N] = 1.0 - x - r; idx += N
    c[idx:idx+N] = y - r; idx += N
    c[idx:idx+N] = 1.0 - y - r; idx += N
    i, j = np.triu_indices(N, k=1)
    dx = x[i] - x[j]
    dy = y[i] - y[j]
    c[idx:] = dx**2 + dy**2 - 4.0*r**2
    return c

def run_packing():
    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Equal radius optimization
    bounds_eq = [(0.0, 1.0)] * (2*N) + [(0.05, 0.15)]
    starts_eq = []
    
    for r0 in [0.098, 0.100, 0.10139, 0.102]:
        pts = []
        y = r0
        row = 0
        while len(pts) < N and y + r0 <= 1.0:
            shift = r0 if row % 2 == 1 else 0.0
            x = r0 + shift
            while x + r0 <= 1.0 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
            row += 1
        while len(pts) < N:
            pts.append([0.5, 0.5])
        pts = np.array(pts[:N])
        starts_eq.append(np.concatenate([pts.flatten(), [r0]]))
        
        for _ in range(10):
            p = pts + np.random.uniform(-0.008, 0.008, pts.shape)
            p = np.clip(p, 0.02, 0.98)
            starts_eq.append(np.concatenate([p.flatten(), [r0]]))
            
    for x0 in starts_eq:
        try:
            res = minimize(obj_equal, x0, method='SLSQP', bounds=bounds_eq,
                           constraints={'type': 'ineq', 'fun': cons_equal},
                           options={'maxiter': 3000, 'ftol': 1e-12})
            if res.x[-1] > 0.1013:
                c = res.x[:2*N].reshape(N, 2)
                r = res.x[-1]
                s = r * N
                if s > best_sum:
                    best_sum = s
                    best_centers = c.copy()
                    best_radii = np.full(N, r)
        except Exception:
            continue

    # Phase 2: Joint optimization from best equal-radius config
    if best_centers is not None:
        x0_j = np.zeros(3*N)
        x0_j[0::3] = best_centers[:, 0]
        x0_j[1::3] = best_centers[:, 1]
        x0_j[2::3] = best_radii
        
        bounds_j = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
        try:
            res_j = minimize(obj_joint, x0_j, method='SLSQP', bounds=bounds_j,
                             constraints={'type': 'ineq', 'fun': cons_joint},
                             options={'maxiter': 5000, 'ftol': 1e-12})
            c_j = res_j.x[:2*N].reshape(N, 2)
            r_j = res_j.x[2::3]
            s_j = np.sum(r_j)
            if s_j > best_sum:
                best_sum = s_j
                best_centers = c_j
                best_radii = r_j
        except Exception:
            pass

    # Phase 3: LP refinement for radii given fixed centers
    n_pairs = N*(N-1)//2
    A_ub = np.zeros((n_pairs + 4*N, N))
    b_ub = np.zeros(n_pairs + 4*N)

            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1

        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1

        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1

        lp_res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if lp_res.success and np.isfinite(lp_res.fun):
            r_lp = np.maximum(lp_res.x, 0.0) * 0.9999999
            s_lp = np.sum(r_lp)
            if s_lp > best_sum:
                best_sum = s_lp
                best_radii = r_lp
    except Exception:
        pass
        
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
