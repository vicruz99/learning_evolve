import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_lp_sum(centers):
    """
    Computes the maximum sum of radii for fixed centers using LP.
    Returns (sum_radii, radii_array).
    """
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    n = centers.shape[0]
    
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs + 4 * n, n))
    b_ub = np.zeros(num_pairs + 4 * n)
    
    idx = 0
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    # Pairwise constraints: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if res.success and np.isfinite(res.fun):
            radii = np.maximum(res.x, 0.0)
            return -res.fun, radii
    except Exception:
        pass
    return 0.0, np.zeros(n)

def neg_lp_sum(x_flat):
    """Objective for center optimization: minimize negative LP sum of radii."""
    centers = x_flat.reshape(-1, 2)
    val, _ = compute_lp_sum(centers)
    return -val

def generate_configs(rng):

    configs = []
    hex_pats = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4],
        [5, 6, 4, 6, 5], [5, 7, 5, 5, 4], [6, 4, 6, 6, 4],
        [5, 5, 4, 6, 6], [4, 5, 6, 5, 6], [6, 5, 5, 5, 5]
    ]
    
    for pat in hex_pats:
        if sum(pat) != 26: 
            continue
        for r0 in [0.08, 0.09, 0.10, 0.11]:
            pts = []
            y = r0
            for ri, cnt in enumerate(pat):
                shift = r0 if ri % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= 26: 
                        break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            pts = np.array(pts[:26])
            mn = pts.min(axis=0)
            mx = pts.max(axis=0)
            scale = 0.88 / (mx - mn).max()
            pts = (pts - mn) * scale + (1.0 - scale) / 2.0
            configs.append(pts)
            
            for _ in range(2):
                pert = pts + rng.uniform(-0.02, 0.02, pts.shape)
                pert = np.clip(pert, 0.05, 0.95)
                configs.append(pert)
                
    # Grid-based initializations
    for s in [0.18, 0.19, 0.20]:
        pts = []
        y = s / 2
        r = 0
        while y <= 1.0 and len(pts) < 26:
            x = s / 2
            shift = s / 2 if r % 2 == 1 else 0.0
            x += shift

                pts.append([x, y])
                x += s

            r += 1

    # Random scatter

        configs.append(rng.uniform(0.15, 0.85, (26, 2)))

def run_packing():

    best_centers = None

        try:

            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(-1, 2)

        best_centers = configs[0]
        
    # Phase 2: Hill-Climbing Refinement on Centers
    curr_centers = best_centers.copy()
    curr_sum, _ = compute_lp_sum(curr_centers)

        i = rng.integers(N)

    _, final_radii = compute_lp_sum(best_centers)

    sum_r = np.sum(final_radii)
    return best_centers, final_radii, float(sum_r)
