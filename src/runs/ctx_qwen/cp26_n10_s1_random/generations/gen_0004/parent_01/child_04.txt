import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(vars_flat):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars_flat[2::3])

def constraints(vars_flat):
    """
    Inequality constraints: all must be >= 0.
    Includes boundary containment and pairwise non-overlap.
    Uses squared distances for smooth gradients.
    """
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap: ||c_i - c_j||^2 >= (r_i + r_j)^2
    # Using squared distance avoids sqrt singularities and provides smooth gradients
    i, j = np.triu_indices(N, k=1)
    dx = x[i] - x[j]
    dy = y[i] - y[j]
    d2 = dx**2 + dy**2
    r_sum = r[i] + r[j]
    p = d2 - r_sum**2
    
    return np.concatenate([b, p])

def generate_hex_config(row_counts, r_init, scale=1.0, perturb=0.0):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r_init
    dy = np.sqrt(3) * r_init
    for ri, cnt in enumerate(row_counts):
        shift_x = r_init if ri % 2 == 1 else 0.0
        x_start = r_init + shift_x
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x_start, y])
            x_start += 2.0 * r_init
        y += dy
        
    pts = np.array(pts[:N])
    if scale != 1.0:
        pts = (pts - pts.mean(axis=0)) * scale + 0.5
    if perturb > 0:
        pts += np.random.uniform(-perturb, perturb, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    # Boundary constraints: r_i <= dist(i, boundary)
    for i in range(n):
        x, y = centers[i]
        limits = (x, 1.0 - x, y, 1.0 - y)
        for lim in limits:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(lim)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def run_packing():

    best_radii = None

        [4, 5, 6, 6, 5], [5, 5, 4, 6, 6]
    ]
    
    # Generate diverse initial configurations
    for pat in patterns:
        if sum(pat) != 26: 
            continue

    best_radii *= max(scale * 0.9999995, 0.0)
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
