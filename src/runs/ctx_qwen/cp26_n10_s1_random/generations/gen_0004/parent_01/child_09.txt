import numpy as np
from scipy.optimize import linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(max_r, 0.0)))
        
    # Compute pairwise distances
    dists = np.sqrt(((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2).sum(axis=2))
    ii, jj = np.triu_indices(n, k=1)
    
    # Build LP constraints: r_i + r_j <= dist(i, j)
    n_pairs = len(ii)
    A_ub = np.zeros((n_pairs, n))
    A_ub[np.arange(n_pairs), ii] = 1.0
    A_ub[np.arange(n_pairs), jj] = 1.0
    b_ub = dists[ii, jj]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= -1e-9):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def hill_climb(init_centers, n, iterations=2500, step_base=0.025):
    """Stochastic hill-climbing on center coordinates, evaluating objective via LP."""
    centers = init_centers.copy()
    best_centers = centers.copy()
    best_radii, best_sum = solve_lp_radii(centers)
    
    if best_radii is None:
        return init_centers, np.full(n, 0.05), 0.05 * n
        
    for it in range(iterations):
        step = step_base * (0.998 ** it)
        i = np.random.randint(n)
        old_pos = centers[i].copy()
        centers[i] += np.random.uniform(-step, step, 2)
        centers[i] = np.clip(centers[i], 1e-4, 1.0 - 1e-4)
        
        radii, curr_sum = solve_lp_radii(centers)
        if radii is not None and curr_sum > best_sum + 1e-9:
            best_centers = centers.copy()
            best_radii = radii
            best_sum = curr_sum
        else:
            centers[i] = old_pos
            
    return best_centers, best_radii, best_sum

def generate_hex_config(row_counts, r_init):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r_init
    for ri, cnt in enumerate(row_counts):
        shift_x = r_init if ri % 2 == 1 else 0.0
        x_start = r_init + shift_x
        for _ in range(cnt):
            if len(pts) >= 26:
                break
            pts.append([x_start, y])
            x_start += 2.0 * r_init
        y += r_init * np.sqrt(3)
        
    pts = np.array(pts[:26])
    if pts.size > 0:
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        # Scale to comfortably fit in [0.1, 0.9]
        scale = 0.8 / (mx - mn).max()
        pts = (pts - mn) * scale + 0.1
    return pts

    starts = []

            cfg = generate_hex_config(pat, r0)
            # Perturb to break symmetry and help escape local minima
            cfg += np.random.uniform(-0.015, 0.015, cfg.shape)
            cfg = np.clip(cfg, 0.1, 0.9)
            starts.append(cfg)

            best_sum = b_s
            best_centers = b_c.copy()
            best_radii = b_r.copy()

        best_centers = starts[0]

        best_sum = 0.08 * n

    best_radii = np.maximum(best_radii, 0.0) * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
