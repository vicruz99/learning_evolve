import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_lp_sum(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(max_r, 1e-9)))
        
    A_ub = []
    b_ub = []
    
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def objective(vars_arr):
    """Objective: minimize negative sum of radii"""
    return -np.sum(vars_arr[2::3])

def constraints(vars_arr):
    """Computes inequality constraints >= 0 for valid packing."""
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    
    cons = []
    # Linear boundary constraints are exact and stable
    cons.extend(x - r)
    cons.extend(1.0 - x - r)
    cons.extend(y - r)
    cons.extend(1.0 - y - r)
    
    # Vectorized squared pairwise constraints: ||c_i - c_j||^2 - (r_i + r_j)^2 >= 0
    xi = x[:, None]
    yi = y[:, None]
    ri = r[:, None]
    
    dx = xi - xi.T
    dy = yi - yi.T
    dr = ri + ri.T
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    dist_sq = dx[mask]**2 + dy[mask]**2
    r_sum_sq = dr[mask]**2
    
    cons.extend(dist_sq - r_sum_sq)
    return np.array(cons)

def generate_hex_start(r0, row_counts, perturb=0.0):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r0
    for ri, cnt in enumerate(row_counts):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        
    # Fill remaining spots if lattice didn't yield enough
    while len(pts) < N:
        pts.append([0.5 + np.random.uniform(-0.2, 0.2), 
                    0.5 + np.random.uniform(-0.2, 0.2)])
    pts = np.array(pts[:N])
    if perturb > 0:
        pts += np.random.uniform(-perturb, perturb, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts
