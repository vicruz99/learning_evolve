We then treat the LP objective value as a black-box function of the centers and optimize the centers using the Powell method (derivative-free), which efficiently navigates the non-smooth landscape created by the LP constraints. import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def solve_lp_for_centers(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    bounds = [(0, None)] * n
    num_constraints = n * (n - 1) // 2 + 4 * n
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            dist = max(dist, 1e-9)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    for i in range(n):
        x, y = centers[i]
        limits = [x, 1.0 - x, y, 1.0 - y]
        for lim in limits:
            A_ub[idx, i] = 1.0
            b_ub[idx] = max(lim, 0.0)
            idx += 1
            
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return None, 0.0

def objective_lp(centers_flat):
    """Objective for Powell: maximize LP sum of radii => minimize negative sum."""
    centers = np.clip(centers_flat.reshape(N, 2), 1e-3, 1.0 - 1e-3)
    _, s = solve_lp_for_centers(centers)
    return -s

def generate_hex_config(row_counts, scale=0.85):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    r0 = 0.10
    y = r0
    for ri, cnt in enumerate(row_counts):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        
    pts = np.array(pts[:N])
    mn = pts.min(axis=0)
    mx = pts.max(axis=0)
    span = mx - mn
    span = max(span[0], span[1])
    if span > 0:
        pts = (pts - mn) / span * scale + (1.0 - scale) / 2.0
    return pts

def obj_slqp(vars_arr):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(vars_arr[2::3])

def cons_slqp(vars_arr):
    """Inequality constraints for SLSQP: all must be >= 0."""
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    n = len(r)
    c = np.empty(4 * n + n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        c[idx] = x[i] - r[i]; idx += 1
        c[idx] = 1.0 - x[i] - r[i]; idx += 1
        c[idx] = y[i] - r[i]; idx += 1
        c[idx] = 1.0 - y[i] - r[i]; idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            c[idx] = (x[i] - x[j])**2 + (y[i] - y[j])**2 - (r[i] + r[j])**2
            idx += 1
    return c

def run_packing():
    np.random.seed(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    patterns = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [5, 6, 4, 6, 5]]
    configs = []
    for pat in patterns:
        if sum(pat) != N: 
            continue
        configs.append(generate_hex_config(pat))
        
    for _ in range(8):
        base = configs[np.random.randint(len(configs))]
        pert = base + np.random.uniform(-0.02, 0.02, base.shape)
        configs.append(np.clip(pert, 0.05, 0.95))
        
    # Phase 1: Optimize centers using Powell method on the LP objective
    for cfg in configs:
        try:
            res = minimize(objective_lp, cfg.flatten(), method='Powell',
                           options={'maxiter': 3000, 'xtol': 1e-7, 'ftol': 1e-9})
            c_opt = res.x.reshape(N, 2)
            radii, s = solve_lp_for_centers(c_opt)
            if radii is not None and s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = radii.copy()
        except Exception:
            continue

    # Phase 2: Joint SLSQP refinement starting from the best LP centers
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii
        
        bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        try:
            res_slqp = minimize(obj_slqp, x0, method='SLSQP',
                                bounds=bounds, constraints={'type': 'ineq', 'fun': cons_slqp},
                                options={'maxiter': 2000, 'ftol': 1e-12})
            c_slqp = np.column_stack((res_slqp.x[0::3], res_slqp.x[1::3]))
            r_slqp = res_slqp.x[2::3]
            if np.sum(r_slqp) > best_sum:
                best_centers = c_slqp
                best_radii = r_slqp
                best_sum = np.sum(r_slqp)
        except Exception:
            pass

    # Final safety clamping and scaling
    if best_radii is not None:
        best_radii = np.maximum(best_radii, 0.0) * 0.9999999
        best_sum = np.sum(best_radii)
    else:
        best_centers = configs[0]
        best_radii = np.full(N, 0.09)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
