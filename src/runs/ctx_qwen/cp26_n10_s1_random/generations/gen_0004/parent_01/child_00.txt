from scipy.optimize import minimize, linprog

N = 26

def objective(vars_flat):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars_flat[2::3])

def get_constraints(vars_flat):
    """
    Inequality constraints: all must be >= 0.
    Includes boundary containment and pairwise non-overlap.
    """
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap: ||c_i - c_j|| >= r_i + r_j
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dist = np.sqrt(dx*dx + dy*dy)
    p = dist - (r[i_idx] + r[j_idx])
    
    return np.concatenate([b, p])

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    dists = np.sqrt(dx*dx + dy*dy)
    
    for k in range(len(i_idx)):
        row = np.zeros(n)
        row[i_idx[k]] = 1.0
        row[j_idx[k]] = 1.0
        A_ub.append(row)
        b_ub.append(dists[k])
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Boundary constraints handled efficiently as variable bounds
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, max_r)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def generate_hex_config(row_counts, r0, perturb=0.0):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r0
    for ri, cnt in enumerate(row_counts):
        shift_x = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift_x

            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)

        pts.append([0.5 + np.random.uniform(-0.2, 0.2), 
                    0.5 + np.random.uniform(-0.2, 0.2)])
                    
    pts = np.array(pts[:N])

    mx = pts.max(axis=0)

    if span.max() > 1e-6:
        scale = 0.82 / span.max()
        pts = (pts - mn) * scale + (1.0 - 0.82*scale) / 2.0
        
    if perturb > 0:
        pts += np.random.uniform(-perturb, perturb, pts.shape)
        
    return np.clip(pts, 0.02, 0.98)
