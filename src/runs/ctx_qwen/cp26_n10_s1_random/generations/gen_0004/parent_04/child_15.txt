import numpy as np
from scipy.optimize import linprog

def solve_lp(centers, A_ub, idx):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j) and r_i <= dist(i, boundary)
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    x, y = centers[:, 0], centers[:, 1]
    limits = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    limits = np.maximum(limits, 0.0)
    bounds = [(0.0, lim) for lim in limits]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[idx]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    # Precompute constant pairwise constraint matrix for LP
    idx = np.triu_indices(n, k=1)
    m = len(idx[0])
    A_ub = np.zeros((m, n))
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    # Initialize with a dense hexagonal lattice pattern
    pts = []
    y = 0.085
    row = 0
    rc = [6, 5, 6, 5, 4]
    for cnt in rc:
        shift = 0.085 if row % 2 == 1 else 0.0
        width = (cnt - 1) * 0.17
        x_start = 0.5 - width/2.0 + shift
        for c in range(cnt):
            if len(pts) < n:
                pts.append([x_start + c*0.17, y])
        y += 0.14722
        row += 1
    centers = np.array(pts[:n])
    centers = np.clip(centers, 0.05, 0.95)
    
    # Initial LP evaluation
    best_radii, best_sum = solve_lp(centers, A_ub, idx)
    if best_radii is None:
        best_radii = np.full(n, 0.05)
        best_sum = 0.05 * n
    best_centers = centers.copy()
    
    # Hill-climbing on centers with LP objective evaluation
    step = 0.035
    for it in range(25000):
        i = rng.integers(n)
        old_c = centers[i].copy()
        centers[i] += rng.uniform(-step, step, 2)
        centers[i] = np.clip(centers[i], 1e-4, 1.0 - 1e-4)
        
        radii, s = solve_lp(centers, A_ub, idx)
        if s > best_sum:
            best_centers = centers.copy()
            best_radii = radii
            best_sum = s
        else:
            centers[i] = old_c
            
        if it % 2500 == 0:
            step *= 0.93
            
    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
