import numpy as np
from scipy.optimize import minimize, linprog

def objective(vars_arr):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_arr[2::3])

def get_constraints(vars_arr, n):
    """Compute all inequality constraints >= 0 for valid packing."""
    xs = vars_arr[0::3]
    ys = vars_arr[1::3]
    rs = vars_arr[2::3]
    
    # Boundary constraints: circles must stay inside [0,1]x[0,1]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Pairwise non-overlap constraints: squared distance >= squared sum of radii
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    idx = np.triu_indices(n, k=1)
    c = np.concatenate([c, (dx[idx]**2 + dy[idx]**2) - dr[idx]**2])
    
    return c

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    
    # Max radius for each circle limited by boundaries
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 0.0)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in limits]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Constraint matrix: r_i + r_j <= dist(i, j)
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def generate_inits(n, rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # Hexagonal lattice patterns with various row distributions
    row_counts_list = [
        [6,5,5,5,5], [5,6,5,5,5], [5,5,6,5,5], [5,5,5,6,5], [5,5,5,5,6],
        [5,5,5,5,5,1], [4,6,6,5,5], [5,5,4,6,6], [6,6,6,4,4], [5,5,5,5,4,2],
        [5,5,5,4,5,2], [4,5,6,5,6], [5,6,4,5,6], [5,4,6,5,6], [6,5,6,4,5]
    ]
    
    for rc in row_counts_list:
        pts = []
        y = 0.075
        for r_idx, cnt in enumerate(rc):
            shift = 0.075 if r_idx % 2 == 1 else 0.0
            width = (cnt - 1) * 0.15
            x_start = 0.5 - width / 2.0 + shift
            for c in range(cnt):
                if len(pts) < n:
                    pts.append([x_start + c * 0.15, y])
            y += 0.1299
        if len(pts) >= n:
            inits.append(np.array(pts[:n]))
            
    # Random perturbations of the base hex config
    for _ in range(15):
        base = inits[0].copy()
        base += rng.uniform(-0.02, 0.02, base.shape)
        inits.append(np.clip(base, 0.05, 0.95))
        
    # Random placements
    inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return inits

def check_validity(centers, radii, n):
    """Fast validity check for internal optimization steps."""
    # Boundary check
    if not (np.all(centers[:, 0] >= radii - 1e-9) and 
            np.all(centers[:, 0] <= 1.0 - radii + 1e-9) and
            np.all(centers[:, 1] >= radii - 1e-9) and 
            np.all(centers[:, 1] <= 1.0 - radii + 1e-9)):
        return False
        
    # Overlap check
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dr = radii[:, None] + radii[None, :]
    
    idx = np.triu_indices(n, k=1)
    dist_sq = dx[idx]**2 + dy[idx]**2
    rs_sq = dr[idx]**2
    
    return np.all(dist_sq >= rs_sq - 1e-9)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * n
    cons = {'type': 'ineq', 'fun': get_constraints, 'args': (n,)}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Phase 1: Multi-start SLSQP with perturbation escape
    for cfg in inits:

        v0[1::3] = cfg[:, 1]
        v0[2::3] = 0.06  # Strictly feasible initial radius
        
        current_v = v0.copy()
        for trial in range(4):
            try:
                res = minimize(
                    objective, current_v, method='SLSQP', bounds=bounds,
                    constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14}
                )

                    centers_temp = np.column_stack((cx, cy))

                            best_radii = r.copy()

                    current_v[1::3] = np.clip(current_v[1::3] + rng.uniform(-0.005, 0.005, n), 0.0, 1.0)

            except Exception:
                break

                                best_sum = s

            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-12:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)

            for j in range(i + 1, n):
                d = np.linalg.norm(best_centers[i] - best_centers[j])
                rs = best_radii[i] + best_radii[j]

        best_sum = np.sum(best_radii)

    if best_centers is None:

        r_fb = 0.08
        y = r_fb
        row = 0
        while len(fallback_pts) < n:

            row += 1
        best_centers = np.array(fallback_pts[:n])
        best_radii = np.full(n, r_fb)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
