from scipy.optimize import linprog, minimize

def get_lp_structure(n):
    """Precomputes the constraint matrix structure for the LP."""
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx[0]] = 1.0
    A_ub[np.arange(m), idx[1]] = 1.0
    return A_ub, idx

def solve_lp_radii(centers, A_ub, idx):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i, j) and r_i <= dist(i, boundary)
    """
    n = centers.shape[0]
    
    # Boundary limits
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)  # Prevent degenerate bounds
    bounds = [(0.0, lim) for lim in lims]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Extract upper triangular distances for b_ub
    b_ub = dists[idx]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def hill_climb_centers(centers, A_ub, idx, rng, iterations=2000):

    Hill-climbing on center coordinates to maximize LP objective.
    Moves one circle at a time with exponentially decaying step size.
    """

        return centers, 0.0

    best_sum = curr_sum
    step = 0.045

        else:
            centers[i] = old_c  # Revert move
            
        # Decay step size
        step *= 0.9991
        
    return best_centers, best_sum
