import numpy as np
from scipy.optimize import minimize, linprog


def solve_radii_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 0.0)
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in limits]
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs ** 2, axis=2))
    np.fill_diagonal(dists, 2.0)
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0


def generate_hex_config(row_counts, r_base, n):
    """Generate hexagonal lattice configuration."""
    pts = []
    y = r_base
    dy = np.sqrt(3) * r_base
    for idx, cnt in enumerate(row_counts):
        shift = r_base if idx % 2 == 1 else 0.0
        x = r_base + shift
        for c in range(cnt):
            if len(pts) < n:
                pts.append([x + c * 2.0 * r_base, y])
            x += 2.0 * r_base
        y += dy
    return np.array(pts[:n])


def normalize_config(centers, margin=0.05):
    """Normalize configuration to fit in [margin, 1-margin]."""
    mn = centers.min(axis=0)
    mx = centers.max(axis=0)
    span = mx - mn
    if span.max() < 1e-12:
        return np.full_like(centers, 0.5)
    cfg = (centers - mn) / span
    size = 1.0 - 2.0 * margin
    return cfg * size + margin


def force_relax(centers, n, r_target, steps=800):
    """Force-directed relaxation toward target radius."""
    centers = centers.copy()
    r = r_target * 0.3
    r_final = r_target
    dt = 0.003
    k_rep = 200.0
    k_wall = 800.0

    for step in range(steps):
        progress = step / steps
        r = r + (r_final - r) * 0.01
        r = min(r, r_final)

        forces = np.zeros_like(centers)

        for i in range(n):
            for dim in range(2):
                if centers[i, dim] - r < 0:
                    forces[i, dim] += k_wall * (r - centers[i, dim])
                if centers[i, dim] + r > 1.0:
                    forces[i, dim] -= k_wall * (centers[i, dim] + r - 1.0)

            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < 2.0 * r + 0.01 and dist > 1e-10:
                    overlap = max(2.0 * r - dist, 0)
                    f = k_rep * overlap / (dist + 1e-10)
                    forces[i, 0] += dx * f
                    forces[i, 1] += dy * f
                    forces[j, 0] -= dx * f
                    forces[j, 1] -= dy * f

        centers += forces * dt
        centers = np.clip(centers, 1e-6, 1.0 - 1e-6)

    return centers, r


def equal_radius_obj(v, n):
    return -v[-1]


def equal_radius_cons(v, n):
    t = v[-1]
    xs = v[:n]
    ys = v[n:]
    c = list(xs - t)
    c.extend(1.0 - xs - t)
    c.extend(ys - t)
    c.extend(1.0 - ys - t)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    idx = np.triu_indices(n, k=1)
    d2 = dx[idx] ** 2 + dy[idx] ** 2
    c.extend(d2 - 4.0 * t ** 2)
    return np.array(c)


def vr_obj(v):
    return -np.sum(v[2::3])


def vr_cons(v, n):
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    c = list(xs - rs)
    c.extend(1.0 - xs - rs)
    c.extend(ys - rs)
    c.extend(1.0 - ys - rs)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    idx = np.triu_indices(n, k=1)
    d2 = dx[idx] ** 2 + dy[idx] ** 2
    rs2 = dr[idx] ** 2
    c.extend(d2 - rs2)
    return np.array(c)


def validate_packing_internal(centers, radii, tol=1e-9):
    """Internal validation check."""
    n = centers.shape[0]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r - tol or x > 1.0 - r + tol or y < r - tol or y > 1.0 - r + tol:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            d2 = (centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2
            rs = radii[i] + radii[j]
            if d2 < rs ** 2 - tol:
                return False
    return True
