from scipy.optimize import minimize, linprog

N = 26

def obj_joint(v):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(v[2::3])

def cons_joint(v):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap constraints
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    idx = np.triu_indices(N, k=1)
    c = np.concatenate([c, (dx[idx]**2 + dy[idx]**2) - dr[idx]**2])
    
    return c

def solve_radii_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Maximum radius per circle limited by boundaries
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    bounds = [(0.0, lim) for lim in limits]
    
    A_ub_rows = []
    b_ub_rows = []
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_rows.append(d)
            
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def get_min_clearance(centers):
    """Computes maximum equal radius feasible for given centers."""
    n = centers.shape[0]
    dists_wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_pair = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists_pair, np.inf)
    return min(np.min(dists_wall), np.min(dists_pair) / 2.0)
