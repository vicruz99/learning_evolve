import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    
    # Upper bounds from boundaries
    lims = np.minimum(centers, 1.0 - centers).min(axis=1)
    lims = np.maximum(lims, 0.0)
    bounds = [(0.0, lim) for lim in lims]
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Constraint matrix: r_i + r_j <= dist(i, j)
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def force_expand(centers, n, steps=600):
    """Force-directed expansion to find dense initial configurations."""
    r = np.full(n, 0.025)
    dt = 0.015
    k_rep = 6.0
    k_wall = 30.0
    
    for _ in range(steps):
        r *= 1.0004
        forces = np.zeros_like(centers)
        
        # Circle-circle repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                min_d = r[i] + r[j]
                if dist < min_d and dist > 1e-8:
                    overlap = min_d - dist
                    f = k_rep * overlap / dist
                    fx, fy = dx * f, dy * f
                    forces[i] += np.array([fx, fy])
                    forces[j] -= np.array([fx, fy])
                    
        # Wall repulsion
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < 0: forces[i, 0] += k_wall * (r[i] - x)
            if x + r[i] > 1: forces[i, 0] -= k_wall * (x + r[i] - 1)
            if y - r[i] < 0: forces[i, 1] += k_wall * (r[i] - y)
            if y + r[i] > 1: forces[i, 1] -= k_wall * (y + r[i] - 1)
            
        centers += forces * dt
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
        
    return centers

def obj_slqp(x, n):
    """Objective: maximize sum of radii."""
    return -np.sum(x[2::3])

def cons_slqp(x, n):
    """Inequality constraints >= 0 for valid packing."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap (squared formulation for smoothness)
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    
    idx = np.triu_indices(n, k=1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    inits = []

    for density in [0.09, 0.10, 0.11]:

            shift = density if row % 2 == 1 else 0.0

            y += density * np.sqrt(3)

        inits.append(np.array(pts[:n]))

                    scale = min(scale, d / rs)

        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
