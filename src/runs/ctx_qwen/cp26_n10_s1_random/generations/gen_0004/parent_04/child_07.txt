import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrices(centers):
    """Builds constraint matrices for the radii LP given fixed centers."""
    n = centers.shape[0]
    # Boundary limits
    lims = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    lims = np.minimum(lims, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 0.0)) for l in lims]
    
    # Pairwise distance constraints
    idx = np.triu_indices(n, k=1)
    m = len(idx[0])
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    dx = centers[idx[0], 0] - centers[idx[1], 0]
    dy = centers[idx[0], 1] - centers[idx[1], 1]
    b_ub = np.hypot(dx, dy)
    
    for k in range(m):
        A_ub[k, idx[0][k]] = 1.0
        A_ub[k, idx[1][k]] = 1.0
        
    return A_ub, b_ub, bounds

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    A_ub, b_ub, bounds = build_lp_matrices(centers)
    c_obj = -np.ones(centers.shape[0])
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def expand_circles(centers, n, steps=2000, dt=0.01):
    """Force-directed expansion to pre-tighten packing."""
    r = 0.05
    for _ in range(steps):
        forces = np.zeros_like(centers)
        r += 0.000035
        
        # Wall repulsion
        for i in range(n):
            for axis in range(2):
                if centers[i, axis] < r:
                    forces[i, axis] += (r - centers[i, axis]) * 200.0
                elif centers[i, axis] > 1.0 - r:
                    forces[i, axis] -= (centers[i, axis] - (1.0 - r)) * 200.0
                    
        # Circle repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                if dist < 2.0 * r and dist > 1e-8:
                    overlap = 2.0 * r - dist
                    f = overlap * 80.0 / dist
                    forces[i, 0] += dx * f
                    forces[i, 1] += dy * f
                    forces[j, 0] -= dx * f
                    forces[j, 1] -= dy * f
                    
        centers += forces * dt
        centers = np.clip(centers, 0.002, 0.998)
    return centers

def equal_radius_constraints(vars_arr, n):
    """Constraints for equal-radius optimization."""
    xs = vars_arr[:n]
    ys = vars_arr[n:2*n]
    r = vars_arr[-1]
    
    c = np.zeros(4 * n + n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        c[idx] = xs[i] - r; idx += 1
        c[idx] = 1.0 - xs[i] - r; idx += 1
        c[idx] = ys[i] - r; idx += 1
        c[idx] = 1.0 - ys[i] - r; idx += 1
        
    ii, jj = np.triu_indices(n, k=1)
    dx = xs[ii] - xs[jj]
    dy = ys[ii] - ys[jj]
    c[idx:] = dx * dx + dy * dy - 4.0 * r * r
    return c

def equal_radius_obj(vars_arr):
    """Objective for equal-radius optimization."""
    return -vars_arr[-1]

def optimize_equal_radius(centers_init, n):
    """Optimizes positions to maximize a shared radius."""
    x0 = np.concatenate([centers_init.flatten(), [0.08]])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.05, 0.15)]
    cons = {'type': 'ineq', 'fun': equal_radius_constraints, 'args': (n,)}
    
    res = minimize(equal_radius_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
                   
    if np.isfinite(res.fun) and res.x[-1] > 0.01:
        return res.x[:2 * n].reshape(n, 2), res.x[-1]
    return centers_init, 0.08

def hill_climb_lp(centers, n, iters=1200):
    """Local search on centers guided by LP objective."""
    best_centers = centers.copy()
    radii, best_val = solve_lp_radii(centers)
    if radii is None:
        return centers, 0.0
        
    step = 0.025
    for _ in range(iters):
        i = np.random.randint(n)
        old = centers[i].copy()
        centers[i] += np.random.uniform(-step, step, 2)
        centers[i] = np.clip(centers[i], 0.005, 0.995)
        
        radii, val = solve_lp_radii(centers)
        if radii is not None and val > best_val:
            best_val = val
            best_centers = centers.copy()
        else:
            centers[i] = old
            
        step *= 0.999
    return best_centers, best_val
