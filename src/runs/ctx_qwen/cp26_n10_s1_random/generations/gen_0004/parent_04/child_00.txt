import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Max radius limited by boundaries
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 0.0)
    
    bounds = [(0.0, lim) for lim in limits]
    c_obj = -np.ones(n)
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    idx = np.triu_indices(n, k=1)
    m = len(idx[0])
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_equal(vars_arr):
    """Objective: minimize negative equal radius."""
    return -vars_arr[-1]

def cons_equal(vars_arr, n):
    """Constraints for equal radius packing."""
    xs = vars_arr[:n]
    ys = vars_arr[n:2*n]
    r = vars_arr[2*n]
    
    # Boundary constraints
    c = np.concatenate([xs - r, 1.0 - xs - r, ys - r, 1.0 - ys - r])
    
    # Pairwise non-overlap: dist^2 >= 4r^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    idx = np.triu_indices(n, k=1)
    c = np.concatenate([c, (dx[idx]**2 + dy[idx]**2) - 4.0 * r**2])
    
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    rng = np.random.default_rng(42)
    
    # Phase 1: Multi-start optimization of equal radius packing
    for trial in range(12):
        # Generate base hexagonal lattice
        pts = []
        y = 0.08
        row = 0
        while len(pts) < n:
            shift = 0.08 if row % 2 == 1 else 0.0
            x = 0.08 + shift
            while len(pts) < n and x + 0.08 < 1.0:
                pts.append([x, y])
                x += 0.16
            y += 0.08 * np.sqrt(3)
            row += 1
        cfg = np.array(pts[:n])
        
        # Random perturbation to explore configuration space
        cfg += rng.uniform(-0.025, 0.025, cfg.shape)
        cfg = np.clip(cfg, 0.12, 0.88)  # Ensure strict initial feasibility
        
        x0 = np.concatenate([cfg.flatten(), [0.07]])
        bounds_eq = [(0.0, 1.0)] * (2*n) + [(0.05, 0.15)]
        
        try:
            res = minimize(
                obj_equal, x0, method='SLSQP', bounds=bounds_eq,
                constraints={'type': 'ineq', 'fun': cons_equal, 'args': (n,)},
                options={'maxiter': 6000, 'ftol': 1e-14, 'iprint': -1}
            )
            
            if res.success and res.x[-1] > 0.1005:
                c_eq = res.x[:2*n].reshape(n, 2)
                radii_lp, sum_lp = solve_radii_lp(c_eq)
                
                if radii_lp is not None and sum_lp > best_sum:
                    best_centers = c_eq.copy()
                    best_radii = radii_lp.copy()
                    best_sum = sum_lp
        except Exception:
            continue
            
    # Phase 2: Guided local search with LP refinement
    if best_centers is not None:
        curr_centers = best_centers.copy()
        step_size = 0.008
        
        for _ in range(2500):
            i = rng.integers(n)
            old_pos = curr_centers[i].copy()
            
            # Perturb one circle
            curr_centers[i] += rng.uniform(-step_size, step_size, 2)
            curr_centers[i] = np.clip(curr_centers[i], 1e-4, 1.0 - 1e-4)
            
            r_new, s_new = solve_radii_lp(curr_centers)
            
            if r_new is not None and s_new > best_sum:
                best_centers = curr_centers.copy()
                best_radii = r_new.copy()
                best_sum = s_new
                step_size = min(step_size * 1.02, 0.025)  # Expand search if improving
            else:
                curr_centers[i] = old_pos  # Revert
                step_size *= 0.996  # Contract search
                
    # Phase 3: Strict numerical safety scaling
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
                
        for i in range(n):
            for j in range(i+1, n):
                d = np.linalg.norm(best_centers[i] - best_centers[j])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-12:
                    scale = min(scale, d/rs)
                    
        best_radii *= scale * 0.999999
        best_sum = np.sum(best_radii)
    else:
        # Fallback configuration
        best_centers = np.tile([0.5, 0.5], (n, 1))
        best_centers[:, 0] += np.linspace(-0.4, 0.4, n)
        best_radii = np.full(n, 0.05)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
