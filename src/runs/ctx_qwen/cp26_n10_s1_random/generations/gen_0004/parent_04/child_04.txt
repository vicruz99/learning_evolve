import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_clearance(centers):
    n = centers.shape[0]
    d_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pair = np.min(dists) / 2.0
    return min(np.min(d_bound), min_pair)

def solve_lp_radii(centers):
    n = centers.shape[0]
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(0.0, lim - 1e-9)) for lim in limits]
    c_obj = -np.ones(n)

    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)

    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_clearance(x, n):
    return -compute_clearance(x.reshape(n, 2))

def obj_slqp(v, n):
    return -v[2 * n]

def con_slqp(v, n):
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n]
    c = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
    ii, jj = np.triu_indices(n, 1)
    dx = cx[ii] - cx[jj]
    dy = cy[ii] - cy[jj]
    c.append((dx**2 + dy**2) - 4.0 * r**2)
    return np.concatenate(c)

def generate_hex_configs(n):
    configs = []
    row_patterns = [
        [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 5, 6, 5, 5],
        [5, 5, 5, 6, 5], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4],
        [5, 5, 5, 5, 4, 2], [4, 6, 6, 6, 4], [5, 6, 6, 5, 4],
        [6, 5, 5, 5, 4, 1]
    ]
    for pat in row_patterns:
        if sum(pat) != n:
            continue
        pts = []
        y = 0.08
        for idx, cnt in enumerate(pat):
            shift = 0.08 if idx % 2 == 1 else 0.0
            width = (cnt - 1) * 0.16
            x_start = 0.5 - width / 2.0 + shift
            for c in range(cnt):
                pts.append([x_start + c * 0.16, y])
            y += 0.13856
        configs.append(np.array(pts))
    return configs

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    rng = np.random.default_rng(123)
    hex_configs = generate_hex_configs(n)

    # Phase 1: Nelder-Mead on centers to maximize clearance
    for cfg in hex_configs:
        for _ in range(6):
            init = cfg + rng.uniform(-0.015, 0.015, cfg.shape)
            init = np.clip(init, 0.05, 0.95)
            try:
                res = minimize(obj_clearance, init.flatten(), args=(n,),
                               method='Nelder-Mead', options={'maxiter': 5000, 'xatol': 1e-7, 'fatol': 1e-9})
                centers_nm = res.x.reshape(n, 2)
                lp_r, lp_s = solve_lp_radii(centers_nm)
                if lp_r is not None and lp_s > best_sum:
                    best_sum = lp_s
                    best_centers = centers_nm.copy()
                    best_radii = lp_r.copy()
            except Exception:
                continue

    # Phase 2: SLSQP equal radius refinement
    if best_centers is not None:
        bounds_slqp = [(0.0, 1.0)] * (2 * n) + [(0.05, 0.15)]
        for _ in range(12):
            init_c = best_centers + rng.uniform(-0.005, 0.005, (n, 2))
            init_c = np.clip(init_c, 0.05, 0.95)
            r_init = compute_clearance(init_c) * 0.98
            vars0 = np.zeros(2 * n + 1)
            vars0[:2 * n] = init_c.flatten()
            vars0[2 * n] = r_init
            try:
                res = minimize(obj_slqp, vars0, args=(n,), method='SLSQP', bounds=bounds_slqp,
                               constraints={'type': 'ineq', 'fun': con_slqp, 'args': (n,)},
                               options={'maxiter': 3000, 'ftol': 1e-12})
                if res.success:
                    c_opt = res.x[:2 * n].reshape(n, 2)
                    lp_r2, lp_s2 = solve_lp_radii(c_opt)
                    if lp_r2 is not None and lp_s2 > best_sum:
                        best_sum = lp_s2
                        best_centers = c_opt
                        best_radii = lp_r2
            except Exception:
                continue

    # Safety scaling to guarantee strict numerical validity
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-9:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(best_centers[i] - best_centers[j])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-9:
                    scale = min(scale, d / rs)
        best_radii *= scale * 0.999999
        best_sum = np.sum(best_radii)
    else:
        best_centers = hex_configs[0]
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
