The strategy employs a two-phase optimization:

This smooths out the configuration into an optimal packing lattice. We seed this with multiple hexagonal row patterns and random perturbations to escape local minima.
2. **Radius Optimization (Linear Programming)**: Once optimal centers are found, we solve an LP to maximize $\sum r_i$ subject to $r_i + r_j \le d_{ij}$ and boundary constraints. This extracts any remaining slack and handles unequal radii optimally.
3. **Local Perturbation Search**: We randomly perturb the best centers slightly and re-run the LP to ensure we haven't missed a nearby superior configuration.
4. **Safety Scaling**: Finally, we apply a conservative scaling factor to guarantee strict numerical validity against the checker's $10^{-12}$ tolerance.

This approach decouples the non-smooth geometric arrangement from the radius assignment, which is more robust than joint SLSQP optimization for this problem size.

import numpy as np
from scipy.optimize import minimize, linprog

def min_clearance(centers):
    """Computes the maximum feasible equal radius for given centers."""
    n = centers.shape[0]
    # Distance to walls
    d_wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    d_pair = np.min(dists) / 2.0
    return min(np.min(d_wall), d_pair)

def neg_clearance(x):
    """Objective for center optimization: maximize min clearance."""
    return -min_clearance(x.reshape(-1, 2))

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, max_r)))
        
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    m = len(idx[0])
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def generate_inits(n, rng):
    """Generates diverse initial center configurations."""
    inits = []
    # Known effective hexagonal row distributions for N=26
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [5, 6, 6, 5, 4], [4, 6, 6, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 5, 6, 5], [7, 5, 6, 4, 4], [5, 5, 5, 5, 6]
    ]
    
    for pat in patterns:
        if sum(pat) < n: continue
        pts = []
        y = 0.09
        for idx, cnt in enumerate(pat):
            shift = 0.09 if idx % 2 == 1 else 0.0
            x_start = 0.5 - (cnt - 1) * 0.09 + shift
            for _ in range(cnt):
                if len(pts) < n:
                    pts.append([x_start, y])
                x_start += 0.18
            y += 0.09 * np.sqrt(3)
        if len(pts) >= n:
            inits.append(np.array(pts[:n]))
            
    # Random configurations
    for _ in range(10):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # Perturbations of the first hex pattern
    if len(inits) > 0:
        for _ in range(15):
            base = inits[0].copy()
            base += rng.uniform(-0.025, 0.025, base.shape)
            base = np.clip(base, 0.05, 0.95)
            inits.append(base)
            
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    inits = generate_inits(n, rng)
    
    best_centers = None
    best_lp_sum = -1.0
    best_lp_radii = None
    
    # Phase 1: Optimize geometry to maximize equal radius clearance
    for cfg in inits:
        res = minimize(neg_clearance, cfg.flatten(), method='Nelder-Mead',
                       options={'maxiter': 20000, 'xatol': 1e-9, 'fatol': 1e-10})
        opt_centers = res.x.reshape(n, 2)
        
        # Phase 2: LP to extract optimal (possibly unequal) radii
        r_lp, s_lp = solve_radii_lp(opt_centers)
        if r_lp is not None and s_lp > best_lp_sum:
            best_lp_sum = s_lp
            best_centers = opt_centers.copy()
            best_lp_radii = r_lp.copy()
            
    # Phase 3: Local perturbation search to escape flat optima
    for _ in range(150):
        pert = best_centers + rng.uniform(-0.003, 0.003, (n, 2))
        pert = np.clip(pert, 0.005, 0.995)
        r_lp, s_lp = solve_radii_lp(pert)
        if r_lp is not None and s_lp > best_lp_sum:
            best_lp_sum = s_lp
            best_centers = pert.copy()
            best_lp_radii = r_lp.copy()
            
    # Phase 4: Safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_lp_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_lp_radii[i] + best_lp_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_lp_radii *= scale * 0.999999
    best_lp_sum = np.sum(best_lp_radii)
    
    return best_centers, best_lp_radii, float(best_lp_sum)
