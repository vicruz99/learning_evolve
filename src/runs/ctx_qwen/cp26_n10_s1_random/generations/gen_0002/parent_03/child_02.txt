import numpy as np
from scipy.optimize import minimize

def _objective(vars, n):
    """Minimize negative sum of radii => Maximize sum of radii"""
    return -np.sum(vars[2*n:3*n])

def _constraint_fun(vars, n):
    """Compute inequality constraints >= 0 for valid packing"""
    cx = vars[:n]
    cy = vars[n:2*n]
    r = vars[2*n:3*n]
    c = []
    # Boundary constraints: circles must stay inside [0,1]x[0,1]
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    # Overlap constraints: squared distance >= squared sum of radii
    for i in range(n):
        for j in range(i+1, n):
            c.append((cx[i]-cx[j])**2 + (cy[i]-cy[j])**2 - (r[i]+r[j])**2)
    return np.array(c)

def _validate_and_update(cx, cy, r, n, best_state):
    """Validate a candidate solution and update best_state if improved"""
    valid = True
    # Boundary check with tolerance
    if np.any(cx < r - 1e-8) or np.any(cx > 1-r + 1e-8) or \
       np.any(cy < r - 1e-8) or np.any(cy > 1-r + 1e-8):
        return
    # Overlap check with tolerance
    dists_sq = (cx[:,None]-cx[None,:])**2 + (cy[:,None]-cy[None,:])**2
    r_sum = r[:,None] + r[None,:]
    mask = np.triu(np.ones((n,n), dtype=bool), k=1)
    if np.any(dists_sq[mask] < (r_sum[mask])**2 - 1e-8):
        return
        
    s = np.sum(r)
    if s > best_state[0]:
        best_state[0] = s
        best_state[1] = np.column_stack((cx, cy))
