from scipy.optimize import minimize

N_CIRCLES = 26

def penalty_obj(x, mu):
    """Objective: negative sum of radii + penalty for constraint violations."""
    n = N_CIRCLES
    cx, cy, r = x[:n], x[n:2*n], x[2*n:3*n]
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary penalties: circles must stay inside [0,1]x[0,1]
    pen += np.sum(np.maximum(0, r - cx)**2)
    pen += np.sum(np.maximum(0, r - (1.0 - cx))**2)
    pen += np.sum(np.maximum(0, r - cy)**2)
    pen += np.sum(np.maximum(0, r - (1.0 - cy))**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    diff_cx = cx[:, None] - cx[None, :]
    diff_cy = cy[:, None] - cy[None, :]
    dist = np.sqrt(diff_cx**2 + diff_cy**2)
    np.fill_diagonal(dist, np.inf)
    
    r_sum = r[:, None] + r[None, :]
    viol = np.maximum(0, r_sum - dist)
    triu = np.triu_indices(n, k=1)
    pen += np.sum(viol[triu]**2)
    
    return obj + mu * pen

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
