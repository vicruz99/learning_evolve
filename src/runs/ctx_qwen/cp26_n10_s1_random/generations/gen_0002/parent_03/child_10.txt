import numpy as np
from scipy.optimize import minimize

def objective_and_penalty(x, n, mu, triu_idx):
    """
    Computes the negative sum of radii plus a smooth penalty for constraint violations.
    x layout: [x0, y0, x1, y1, ..., x25, y25, r0, r1, ..., r25]
    """
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    # Primary objective: maximize sum(r) => minimize -sum(r)
    obj = -np.sum(radii)
    
    # Boundary violations: circle must be inside [0,1]x[0,1]
    # Left: r > x  => violation = r - x
    v = radii - centers[:, 0]
    pen_b = np.sum(np.maximum(0, v)**2)
    # Right: r > 1-x => violation = r + x - 1
    v = radii + centers[:, 0] - 1.0
    pen_b += np.sum(np.maximum(0, v)**2)
    # Bottom: r > y => violation = r - y
    v = radii - centers[:, 1]
    pen_b += np.sum(np.maximum(0, v)**2)
    # Top: r > 1-y => violation = r + y - 1
    v = radii + centers[:, 1] - 1.0
    pen_b += np.sum(np.maximum(0, v)**2)
    
    # Overlap violations: dist(i,j) < r_i + r_j => violation = r_i + r_j - dist
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    # Avoid self-distance issues
    dists[np.arange(n), np.arange(n)] = 10.0
    r_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
    o_viol = r_sums - dists
    
    # Only count upper triangle to avoid double counting
    pen_o = np.sum(np.maximum(0, o_viol[triu_idx])**2)
    
    return obj + mu * (pen_b + pen_o)
