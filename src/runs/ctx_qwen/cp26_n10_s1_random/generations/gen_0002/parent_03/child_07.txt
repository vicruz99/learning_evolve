import numpy as np
from scipy.optimize import minimize

def generate_hex_init(n, seed=42):
    """Generates a hexagonal lattice initialization for n circles."""
    np.random.seed(seed)
    r_est = 0.101
    dx = 2.0 * r_est
    dy = np.sqrt(3.0) * r_est
    
    pts = []
    y = r_est
    row = 0
    # Hexagonal pattern tuned for 26 circles
    while len(pts) < n:
        shift = dx / 2.0 if row % 2 == 1 else 0.0
        x = r_est + shift
        while x + r_est <= 1.0 and len(pts) < n:
            pts.append([x, y])
            x += dx
        y += dy
        row += 1
        
    pts = np.array(pts[:n])
    # Center and scale to fit comfortably in [0,1]^2
    pts -= pts.min(axis=0)
    pts /= pts.max(axis=0)
    pts = pts * 0.94 + 0.03
    
    # Add small random perturbation to break symmetry
    pts += np.random.uniform(-0.005, 0.005, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def compute_objective(x, n):
    """Objective: minimize negative radius t => maximize t"""
    return -x[2 * n]

def compute_constraints(x, n):
    """
    Computes inequality constraints >= 0 for valid equal-radius packing.
    Variables: x[0:2n] are centers, x[2n] is radius t.
    """
    c = x[:2 * n].reshape(n, 2)
    t = x[2 * n]
    
    # Boundary constraints: x >= t, 1-x >= t, y >= t, 1-y >= t
    b1 = c[:, 0] - t
    b2 = 1.0 - c[:, 0] - t
    b3 = c[:, 1] - t
    b4 = 1.0 - c[:, 1] - t
    
    # Pairwise non-overlap constraints: ||c_i - c_j|| >= 2t
    diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps = dists[mask] - 2.0 * t
    
    return np.concatenate([b1, b2, b3, b4, overlaps])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.05, 0.5)]
    cons = {'type': 'ineq', 'fun': compute_constraints, 'args': (n,)}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Try multiple initial configurations
    seeds = [42, 101, 200, 305, 410, 515, 620, 725, 830, 935]
    
    for seed in seeds:
        centers_init = generate_hex_init(n, seed)
        x0 = np.concatenate([centers_init.flatten(), [0.095]])
        
        try:
            res = minimize(
                compute_objective,
                x0,
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 8000, 'ftol': 1e-13}
            )
            
            if np.isfinite(res.fun):
                cx = res.x[:n]
                cy = res.x[n:2*n]
                t_opt = res.x[2*n]
                centers = np.column_stack((cx, cy))
                
                # Validate constraints strictly
                valid = True
                for i in range(n):
                    if (cx[i] < t_opt - 1e-12 or cx[i] > 1.0 - t_opt + 1e-12 or
                        cy[i] < t_opt - 1e-12 or cy[i] > 1.0 - t_opt + 1e-12):
                        valid = False
                        break
                if valid:
                    for i in range(n):
                        for j in range(i + 1, n):
                            d = np.sqrt((cx[i]-cx[j])**2 + (cy[i]-cy[j])**2)
                            if d < 2.0 * t_opt - 1e-12:
                                valid = False
                                break
                        if not valid:
                            break
                            
                if valid:
                    # Post-processing: compute exact maximal feasible radii for each circle
                    # This often squeezes out extra sum and guarantees robustness
                    radii = np.zeros(n)
                    for i in range(n):
                        # Distance to boundaries
                        min_d = min(cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
                        # Distance to other circles
                        for j in range(n):
                            if i != j:
                                d = np.sqrt((cx[i]-cx[j])**2 + (cy[i]-cy[j])**2)
                                if d < min_d:
                                    min_d = d
                        radii[i] = min_d / 2.0
                        
                    # Apply tiny safety margin for numerical stability in validator
                    radii *= 0.9999999
                    current_sum = np.sum(radii)
                    
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_centers = centers.copy()
                        best_radii = radii.copy()
        except Exception:
            continue
            
    # Fallback to a valid hex packing if optimization unexpectedly fails
    if best_centers is None:
        fb_centers = generate_hex_init(n, 42)
        fb_r = np.full(n, 0.095)
        best_centers = fb_centers
        best_radii = fb_r
        best_sum = np.sum(fb_r)
        
    return best_centers, best_radii, float(best_sum)
