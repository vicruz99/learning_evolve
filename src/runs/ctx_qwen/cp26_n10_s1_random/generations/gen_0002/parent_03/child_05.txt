def compute_penalty(v, n, mu):
    """
    Computes the objective (-r) plus penalty for constraint violations.
    v: flat array [x0, y0, ..., xn, yn, r]
    """
    cx = v[0::2]
    cy = v[1::2]
    r = v[-1]
    
    # Objective: maximize r => minimize -r
    obj = -r
    
    # Boundary penalties: circle must be inside [0,1]x[0,1]
    # Violation if r > x, r > 1-x, r > y, r > 1-y
    p = np.sum(np.maximum(0, r - cx)**2)
    p += np.sum(np.maximum(0, cx + r - 1.0)**2)
    p += np.sum(np.maximum(0, r - cy)**2)
    p += np.sum(np.maximum(0, cy + r - 1.0)**2)
    
    # Overlap penalties: dist(i,j) >= 2r
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Upper triangular indices to count each pair once
    tri_i, tri_j = np.triu_indices(n, k=1)
    pair_dists = dists[tri_i, tri_j]
    overlap = np.maximum(0, 2.0 * r - pair_dists)
    p += np.sum(overlap**2)
    
    return obj + mu * p

def get_hex_init(r0, n):
    """Generates a hexagonal lattice of n points with spacing based on r0."""
    pts = []
    y = r0
    row = 0
    while len(pts) < n:
        shift = r0 if row % 2 == 1 else 0.0

        while x + r0 <= 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    return np.array(pts[:n])
