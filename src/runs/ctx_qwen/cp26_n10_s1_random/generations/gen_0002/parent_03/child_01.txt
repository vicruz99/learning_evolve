import numpy as np
from scipy.optimize import minimize

def compute_penalized_objective(vars, n, mu, triu_idx):
    """
    Objective function for L-BFGS-B: minimizes negative sum of radii plus penalty for constraint violations.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary penalties: circles must stay within [0, 1]
    # Left: x >= r  => r - x <= 0
    v = r - x
    pen += np.sum(np.maximum(0, v)**2)
    # Right: 1 - x >= r => r + x - 1 <= 0
    v = r + x - 1.0
    pen += np.sum(np.maximum(0, v)**2)
    # Bottom: y >= r
    v = r - y
    pen += np.sum(np.maximum(0, v)**2)
    # Top: 1 - y >= r
    v = r + y - 1.0
    pen += np.sum(np.maximum(0, v)**2)
    
    # Overlap penalties: dist(i, j) >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2 + 1e-12)
    np.fill_diagonal(dist, 1.0)
    
    r_sum = r[:, None] + r[None, :]
    viol = r_sum - dist
    pen += np.sum(np.maximum(0, viol[triu_idx])**2)
    
    return obj + mu * pen

def obj_slqp(vars, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def cons_slqp(vars, n):
    """Inequality constraints for SLSQP: boundary and non-overlap."""
    cx = vars[0::3]
    cy = vars[1::3]
    r = vars[2::3]
    c = []
    # Boundary constraints
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    # Pairwise non-overlap constraints: dist(i,j) >= r_i + r_j
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    r_sum = r[:, None] + r[None, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.extend(dist[mask] - r_sum[mask])
    return np.array(c)

def make_hex(rows_pattern, r):
    """Generates a hexagonal lattice pattern of circles."""
    pts = []
    y = r
    dy = np.sqrt(3) * r
    for idx, cnt in enumerate(rows_pattern):
        shift = r if idx % 2 == 1 else 0.0
        x = r + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2 * r
        y += dy
    return np.array(pts)
