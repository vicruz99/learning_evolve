import numpy as np
from scipy.optimize import minimize

def compute_objective_and_penalty(vars_arr, n, mu, tri_i, tri_j):
    """
    Computes the objective: negative sum of radii + penalty for constraint violations.
    Vectorized for performance.
    """
    c = vars_arr[:2*n].reshape(n, 2)
    r = vars_arr[2*n:]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary penalties: circles must stay within [0, 1]
    pen += np.sum(np.maximum(0, r - c[:, 0])**2)
    pen += np.sum(np.maximum(0, r - (1.0 - c[:, 0]))**2)
    pen += np.sum(np.maximum(0, r - c[:, 1])**2)
    pen += np.sum(np.maximum(0, r - (1.0 - c[:, 1]))**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    violations = r_sum[tri_i, tri_j] - dists[tri_i, tri_j]
    pen += np.sum(np.maximum(0, violations)**2)
    
    return obj + mu * pen

def get_hex_init(r0, counts, shift_rows=True):
    """Generates an initial hexagonal grid of n circles with estimated radius r0."""
    pts = []
    y = r0
    for idx, cnt in enumerate(counts):
        shift = r0 if (shift_rows and idx % 2 == 1) else 0.0
        # Center the row horizontally
        x_start = (6 - cnt) * r0 
        for c in range(cnt):
            x = x_start + c * 2 * r0 + shift + r0
            pts.append([x, y])
        y += np.sqrt(3) * r0
    return np.array(pts)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    tri_i, tri_j = np.triu_indices(n, k=1)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    inits = []
    r0 = 0.085
    
    # 1. Standard hex lattice variants (targeting known optimal row structures)
    row_counts = [
        [5, 6, 5, 6, 4],
        [6, 5, 6, 5, 4],
        [5, 5, 6, 5, 5],
        [4, 6, 6, 5, 5],
        [6, 6, 5, 5, 4]
    ]
    for counts in row_counts:
        cfg = get_hex_init(r0, counts)
        # Normalize to comfortably fit inside [0, 1]
        cfg = (cfg - cfg.min(axis=0)) / (cfg.max(axis=0) - cfg.min(axis=0))
        cfg = cfg * 0.85 + 0.075
        inits.append(cfg)
        
    # 2. Perturbed versions to escape symmetry traps
    for cfg in inits[:3]:
        for _ in range(3):
            cfg_pert = cfg + np.random.uniform(-0.02, 0.02, cfg.shape)
            inits.append(np.clip(cfg_pert, 0.05, 0.95))
            
    # 3. Random feasible placements
    for _ in range(5):
        cfg = np.random.uniform(0.15, 0.85, size=(n, 2))
        inits.append(cfg)
        
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    mu = 80000.0
    
    for cfg in inits:
        x0 = np.concatenate([cfg.flatten(), np.full(n, 0.08)])
        
        res = minimize(
            compute_objective_and_penalty,
            x0,
            args=(n, mu, tri_i, tri_j),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 20000, 'ftol': 1e-14}
        )
        
        c_opt = res.x[:2*n].reshape(n, 2)
        
        # Geometric refinement: compute exact maximum feasible radii for these centers.
        # This guarantees strict validity and often improves the sum slightly by 
        # pushing radii to the exact constraint boundaries.
        r_refined = np.zeros(n)
        for i in range(n):
            min_d = min(c_opt[i, 0], 1.0 - c_opt[i, 0], c_opt[i, 1], 1.0 - c_opt[i, 1])
            for j in range(n):
                if i != j:
                    d = np.linalg.norm(c_opt[i] - c_opt[j])
                    if d < min_d:
                        min_d = d
            r_refined[i] = min_d / 2.0
            
        # Verify validity against grader tolerance
        valid = True
        for i in range(n):
            x, y = c_opt[i]
            r = r_refined[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False
                break
        if valid:
            for i in range(n):
                for j in range(i+1, n):
                    if np.linalg.norm(c_opt[i] - c_opt[j]) < r_refined[i] + r_refined[j] - 1e-12:
                        valid = False
                        break
                if not valid: break
                
        if valid:
            s = np.sum(r_refined)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = r_refined.copy()
                
    # Safety scaling to guarantee strict validity within numerical tolerance
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            x, y = best_centers[i]
            r = best_radii[i]
            if r < 1e-9: continue
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
        for i in range(n):
            for j in range(i+1, n):
                dist = np.linalg.norm(best_centers[i] - best_centers[j])
                r_sum = best_radii[i] + best_radii[j]
                if r_sum > 1e-9:
                    scale = min(scale, dist / r_sum)
        best_radii *= scale * 0.99999
        best_sum = np.sum(best_radii)
        
    # Fallback (should not be reached with robust initialization)
    if best_centers is None:
        best_centers = inits[0]
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
