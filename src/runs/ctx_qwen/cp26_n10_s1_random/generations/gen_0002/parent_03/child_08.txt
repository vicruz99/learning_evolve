import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_max_radii(centers):
    """Computes the maximum valid radius for each circle given fixed centers."""
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        # Distance to square boundaries
        d_wall = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
        # Distance to all other centers
        d_centers = np.linalg.norm(centers - centers[i], axis=1)
        d_centers[i] = np.inf
        # Radius is half the minimum distance to boundary or neighbor
        radii[i] = min(d_wall, np.min(d_centers) / 2.0)
    return radii

def obj_centers_wrapper(c_flat, n):
    """Objective for Nelder-Mead: maximize sum of exact maximal radii."""
    return -np.sum(compute_max_radii(c_flat.reshape(n, 2)))

def slsqp_objective(vars, n):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(vars[2*n:])

def slsqp_constraints(vars, n):
    """Explicit inequality constraints >= 0 for SLSQP."""
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    cons = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons.extend(c[:, 0] - r)
    cons.extend(1.0 - c[:, 0] - r)
    cons.extend(c[:, 1] - r)
    cons.extend(1.0 - c[:, 1] - r)
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            cons.append(dx*dx + dy*dy - (r[i] + r[j])**2)
    return np.array(cons)

    penalty += np.sum(np.maximum(0, -v)**2)
