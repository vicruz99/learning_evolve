from scipy.optimize import minimize

def compute_loss(vars, n):
    """
    Computes the objective: negative sum of radii + penalty for constraint violations.
    """
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    obj = -np.sum(r)
    
    mu = 200000.0  # High penalty weight to enforce constraints strictly
    
    # Boundary penalties: circles must stay within [0, 1]
    v = np.maximum(0, r - c[:, 0])
    v += np.maximum(0, r - (1.0 - c[:, 0]))
    v += np.maximum(0, r - c[:, 1])
    v += np.maximum(0, r - (1.0 - c[:, 1]))
    pen_bound = np.sum(v**2)

    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

    overlap = np.maximum(0, r_sum[tri_i, tri_j] - dists[tri_i, tri_j])
    pen_overlap = np.sum(overlap**2)
    
    return obj + mu * (pen_bound + pen_overlap)

def make_hex(rows_counts, n, r0=0.09):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    dy = np.sqrt(3) * r0
    y = r0
    for ri, cnt in enumerate(rows_counts):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) < n:
                pts.append([x, y])
            x += 2 * r0
        y += dy
    pts = np.array(pts[:n])
    # Normalize to comfortably fit inside [0, 1]
    pts -= pts.min(axis=0)
    pts /= pts.max(axis=0)
    pts = pts * 0.92 + 0.04
    return pts
