Strategy:
1. **Phase 1 (Equal Radii Optimization)**: Maximize a single radius variable `t` subject to non-overlap and boundary constraints. This focuses the optimizer on finding the best geometric arrangement without being distracted by variable radius trade-offs. We use `SLSQP` with multiple high-quality initializations (hexagonal lattice, perturbed lattices, and a grid) to escape local minima.
2. **Phase 2 (Variable Radii Relaxation)**: Use the best centers from Phase 1 as a starting point and optimize individual radii to see if slight asymmetries can squeeze out more volume. This is often beneficial near boundaries.
3. **Post-Processing (Exact Maximal Radii)**: For the final optimized centers, compute the exact maximal feasible radius for each circle as half the minimum distance to any boundary or other circle. This guarantees strict validity and often captures the theoretical upper bound for the given center positions.
4. **Robustness**: Vectorize constraint evaluations, use `triu_indices` to avoid redundant constraints, and apply a tiny safety margin to pass the validator's numerical tolerance strictly.

import numpy as np
from scipy.optimize import minimize

def obj_equal(vars, n):
    """Objective for Phase 1: maximize equal radius t => minimize -t"""
    return -vars[-1]

def cons_equal(vars, n):
    """Constraints for Phase 1: boundary and pairwise non-overlap for equal radius t"""
    t = vars[-1]
    cx = vars[0::2]
    cy = vars[1::2]
    c = list(cx - t)
    c.extend(1.0 - cx - t)
    c.extend(cy - t)
    c.extend(1.0 - cy - t)
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist_sq = dx**2 + dy**2
    triu_idx = np.triu_indices(n, k=1)
    c.extend(dist_sq[triu_idx] - 4.0 * t**2)
    return np.array(c)

def obj_var(vars, n):
    """Objective for Phase 2: maximize sum of radii => minimize -sum(r)"""
    return -np.sum(vars[2*n:])

def cons_var(vars, n):
    """Constraints for Phase 2: boundary and pairwise non-overlap for variable radii"""
    cx = vars[0:n]
    cy = vars[n:2*n]
    r = vars[2*n:3*n]
    c = list(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    triu_idx = np.triu_indices(n, k=1)
    r_sum = r[:, None] + r[None, :]
    c.extend(dist[triu_idx] - r_sum[triu_idx])
    return np.array(c)

def run_packing():
    n = 26
    np.random.seed(42)
    
    best_t = 0.0
    best_centers_eq = None
    
    # Generate diverse initial configurations
    inits = []
    # 1. Hexagonal lattice
    pts = []
    r0 = 0.09
    y = r0
    row = 0
    while len(pts) < n:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x + r0 <= 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
    inits.append(np.array(pts[:n]))
    
    # 2. Perturbed hex lattices to break symmetry
    for _ in range(5):
        p = inits[0] + np.random.uniform(-0.02, 0.02, inits[0].shape)
        inits.append(np.clip(p, 0.05, 0.95))
        
    # 3. Grid based initialization
    grid = np.array([(0.1+i*0.2, 0.1+j*0.2) for i in range(5) for j in range(5)])
    grid = np.vstack([grid, [0.5, 0.5]])
    inits.append(grid)
    
    bounds_equal = [(0.0, 1.0)] * (2*n) + [(0.05, 0.11)]
    
    # Phase 1: Optimize equal radii
    for cfg in inits:
        x0 = np.zeros(2*n + 1)
        x0[0::2] = cfg[:, 0]
        x0[1::2] = cfg[:, 1]
        x0[-1] = 0.085
        
        try:
            res = minimize(obj_equal, x0, args=(n,), method='SLSQP', bounds=bounds_equal,
                           constraints={'type': 'ineq', 'fun': cons_equal},
                           options={'maxiter': 4000, 'ftol': 1e-12})
            if res.success or res.fun < -0.101:
                t_opt = res.x[-1]
                if t_opt > best_t:
                    best_t = t_opt
                    best_centers_eq = np.column_stack((res.x[0::2], res.x[1::2]))
        except Exception:
            pass
            
    if best_centers_eq is None:
        best_centers_eq = inits[0]
        best_t = 0.08
        
    # Phase 2: Optimize variable radii
    bounds_var = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    x0_var = np.concatenate([best_centers_eq.flatten(), np.full(n, best_t)])
    
    try:
        res = minimize(obj_var, x0_var, args=(n,), method='SLSQP', bounds=bounds_var,
                       constraints={'type': 'ineq', 'fun': cons_var},
                       options={'maxiter': 4000, 'ftol': 1e-12})
        if np.isfinite(res.fun):
            cx = res.x[:n]
            cy = res.x[n:2*n]
            r = res.x[2*n:3*n]
            centers = np.column_stack((cx, cy))
            s = np.sum(r)
            if s > best_sum:
                best_sum = s
                best_centers = centers
                best_radii = r
    except Exception:
        pass
        
    # Fallback to equal radii if variable optimization didn't improve
    if best_centers is None or best_sum < 26 * best_t * 0.999:
        best_centers = best_centers_eq
        best_radii = np.full(n, best_t)
        best_sum = np.sum(best_radii)
        
    # Post-processing: compute exact maximal feasible radii for the optimized centers
    # This guarantees strict validity and often captures the theoretical limit for fixed centers
    radii_final = np.zeros(n)
    for i in range(n):
        min_d = min(best_centers[i, 0], 1.0 - best_centers[i, 0],
                    best_centers[i, 1], 1.0 - best_centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.linalg.norm(best_centers[i] - best_centers[j])
                if d < min_d:
                    min_d = d
        radii_final[i] = min_d / 2.0
        
    # Apply tiny safety margin to comfortably pass validator tolerance
    radii_final *= 0.999999
    sum_radii = np.sum(radii_final)
    
    return best_centers, radii_final, float(sum_radii)
