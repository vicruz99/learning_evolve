import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def penalty_objective(x_flat, n, mu):
    """Objective: -sum(radii) + mu * penalty for constraint violations."""
    c = x_flat[:2*n].reshape(n, 2)
    r = x_flat[2*n:]
    
    obj = -np.sum(r)
    
    # Boundary penalties: ensure circles stay within [0,1]
    bx = c[:, 0] - r
    rx = 1.0 - c[:, 0] - r
    by = c[:, 1] - r
    ry = 1.0 - c[:, 1] - r
    pen = np.sum(np.maximum(-bx, 0.0)**2) + np.sum(np.maximum(-rx, 0.0)**2) + \
          np.sum(np.maximum(-by, 0.0)**2) + np.sum(np.maximum(-ry, 0.0)**2)
          
    # Overlap penalties: ensure dist(i,j) >= r_i + r_j
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    viol = r_sum - dists
    tri_i, tri_j = np.triu_indices(n, 1)
    pen += np.sum(np.maximum(viol[tri_i, tri_j], 0.0)**2)
    
    return obj + mu * pen

def exact_constraints(x_flat, n):
    """Inequality constraints >= 0 for exact SLSQP refinement."""
    c = x_flat[:2*n].reshape(n, 2)
    r = x_flat[2*n:]
    
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    tri_i, tri_j = np.triu_indices(n, 1)
    con.append(dists[tri_i, tri_j] - r_sum[tri_i, tri_j])
    
    return np.concatenate(con)

def slsqp_objective(x_flat, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x_flat[2*n:])
