import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def compute_constraints_joint(vars_flat):
    """Vectorized inequality constraints: >= 0 for valid packing."""
    n = N_CIRCLES
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    c = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c.extend(x - r)
    c.extend(1.0 - x - r)
    c.extend(y - r)
    c.extend(1.0 - y - r)
    
    # Non-overlap constraints: dist(i,j) >= r_i + r_j
    r_sums = r[:, None] + r[None, :]
    xs = x[:, None]
    ys = y[:, None]
    dx = xs - xs.T
    dy = ys - ys.T
    d = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(d, np.inf)
    
    iu, ju = np.triu_indices(n, k=1)
    c.extend(d[iu, ju] - r_sums[iu, ju])
    return np.concatenate(c)

def objective_joint(vars_flat):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def solve_radii_lp(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    Constraints: r_i <= dist_to_boundary, r_i + r_j <= dist(i,j), r_i >= 0
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        xi, yi = centers[i]
        lim = min(xi, 1.0 - xi, yi, 1.0 - yi)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(max(lim, 1e-9))
        
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(max(d, 1e-9))
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-5)

def run_packing():
    n = N_CIRCLES
    bounds_vars = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * n
    cons = {'type': 'ineq', 'fun': compute_constraints_joint}

    v0[1::3] = pts[:, 1]

    inits.append(v0)

    # Safety shrink to strictly satisfy validation tolerances
    best_radii *= 0.999999
    best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
