from scipy.optimize import minimize

def objective(vars):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum of radii)."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Vectorized inequality constraints for valid packing.
    Returns an array of constraint values, all of which must be >= 0.
    """
    n = len(vars) // 3
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    c = []
    # Boundary constraints: circles must stay within [0, 1]
    c.extend(x - r)
    c.extend(1.0 - x - r)
    c.extend(y - r)
    c.extend(1.0 - y - r)
    
    # Pairwise non-overlap constraints using squared distances for smoothness
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Extract upper triangle to avoid duplicates and self-interaction
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.extend((d_sq - r_sum_sq)[mask])
    
    return np.array(c)

def run_packing():
    n = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * n
