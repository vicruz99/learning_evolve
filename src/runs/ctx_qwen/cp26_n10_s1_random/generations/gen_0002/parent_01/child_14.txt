import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n):
    """Objective: maximize radius r (minimize -r)."""
    return -vars[-1]

def compute_constraints(vars, n):
    """
    Inequality constraints >= 0 for valid packing.
    Ensures circles are inside [0,1]^2 and non-overlapping.
    """
    c = vars[:-1].reshape(n, 2)
    r = vars[-1]
    con = []
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap: dist(i,j)^2 >= 4r^2
    iu, ju = np.triu_indices(n, k=1)
    dx = c[iu, 0] - c[ju, 0]
    dy = c[iu, 1] - c[ju, 1]
    con.append(dx**2 + dy**2 - 4.0 * r**2)
    
    return np.concatenate(con)

def run_packing() -> tuple:
    n = 26
    best_r = 0.0
    best_centers = None
    
    # Prepare diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice initialization
    pts_hex = []
    r0 = 0.101
    y = r0
    row = 0
    while len(pts_hex) < n:
        shift = r0 if row % 2 else 0.0
        x = r0 + shift
        while x + r0 <= 1.0 and len(pts_hex) < n:
            pts_hex.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    inits.append(np.array(pts_hex[:n]))
    
    # 2. Regular grid + center
    pts_grid = []
    for i in range(5):
        for j in range(5):
            pts_grid.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    pts_grid.append([0.5, 0.5])
    inits.append(np.array(pts_grid[:n]))
    
    # 3. Randomized perturbations of the hex lattice to escape symmetry traps
    np.random.seed(42)
    for _ in range(12):
        cfg = inits[0].copy()
        cfg += np.random.uniform(-0.025, 0.025, cfg.shape)
        cfg = np.clip(cfg, 0.05, 0.95)
        inits.append(cfg)
        
    # Optimization bounds: centers in [0,1], radius in [0.05, 0.15]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.05, 0.15)]
    cons = {'type': 'ineq', 'fun': compute_constraints, 'args': (n,)}
    
    for idx, cfg in enumerate(inits):
        # Initial guess: coordinates + starting radius
        x0 = np.concatenate([cfg.flatten(), [0.095]])
        
        # Primary optimization
        try:
            res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
            
            if res.x[-1] > best_r:
                best_r = res.x[-1]
                best_centers = res.x[:2 * n].reshape(n, 2)
                
            # Secondary refinement pass from the converged point
            if res.x[-1] > 0.09:
                try:
                    res2 = minimize(compute_objective, res.x, method='SLSQP', bounds=bounds,
                                    constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
                    if res2.x[-1] > best_r:
                        best_r = res2.x[-1]
                        best_centers = res2.x[:2 * n].reshape(n, 2)
                except Exception:
                    pass
        except Exception:
            continue
            
    # Fallback safety configuration
    if best_r < 0.09 or best_centers is None:
        best_r = 0.09
        best_centers = inits[0]
        
    # Final validation and strict tolerance adjustment
    centers = best_centers
    r_val = best_r
    
    valid = True
    # Check boundaries
    for i in range(n):
        x, y = centers[i]
        if x - r_val < -1e-9 or x + r_val > 1.0 + 1e-9:
            valid = False; break
        if y - r_val < -1e-9 or y + r_val > 1.0 + 1e-9:
            valid = False; break
            
    # Check overlaps
    if valid:
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                            (centers[i, 1] - centers[j, 1])**2)
                if d < 2.0 * r_val - 1e-9:
                    valid = False; break
            if not valid: break

            valid = True

                if x - r_val < -1e-9 or x + r_val > 1.0 + 1e-9:
                    valid = False; break

                            valid = False; break

            if valid: break

    return centers, radii, float(np.sum(radii))
