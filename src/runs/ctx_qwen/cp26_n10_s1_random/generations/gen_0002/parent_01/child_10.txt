import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def penalty_objective(vars, n, mu):
    """
    Objective: -sum(r) + mu * penalty(boundary & overlap violations)
    """
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary penalties: circles must stay within [0, 1]
    # x >= r  => r - x <= 0
    pen += np.sum(np.maximum(0.0, r - c[:, 0])**2)
    # x + r <= 1 => x + r - 1 <= 0
    pen += np.sum(np.maximum(0.0, c[:, 0] + r - 1.0)**2)
    # y >= r
    pen += np.sum(np.maximum(0.0, r - c[:, 1])**2)
    # y + r <= 1
    pen += np.sum(np.maximum(0.0, c[:, 1] + r - 1.0)**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    viol = r_sum - dists
    pen += np.sum(np.maximum(0.0, viol)**2)
    
    return obj + mu * pen

def run_packing() -> tuple:
    n = N_CIRCLES
    bounds = [(0.01, 0.99)] * (2*n) + [(1e-5, 0.45)] * n
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # --- Initialization: Hexagonal Lattice + Perturbations ---
    np.random.seed(42)
    inits = []
    r0 = 0.10
    pts = []
    y = r0
    row = 0
    while y < 1.0 and len(pts) < n + 5:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x < 1.0 and len(pts) < n + 5:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    hex_pts = np.array(pts[:n])
    
    # Base config
    v_base = np.zeros(3*n)
    v_base[0::3] = hex_pts[:, 0]
    v_base[1::3] = hex_pts[:, 1]
    v_base[2::3] = 0.085
    inits.append(v_base)
    
    # Perturbed configs
    for i in range(9):
        cfg = hex_pts.copy()
        cfg += np.random.uniform(-0.035, 0.035, cfg.shape)
        cfg = np.clip(cfg, 0.05, 0.95)
        v = np.zeros(3*n)
        v[0::3] = cfg[:, 0]
        v[1::3] = cfg[:, 1]
        v[2::3] = 0.08 + i * 0.005
        inits.append(v)
        
    # --- Optimization Loop with Continuation ---
    for v_init in inits:
        curr_vars = v_init.copy()
        mu = 200.0
        
        # Continuation: increase penalty weight to enforce constraints strictly
        for _ in range(6):
            res = minimize(penalty_objective, curr_vars, args=(n, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 8000, 'ftol': 1e-14, 'gtol': 1e-10})
            curr_vars = res.x
            mu *= 4.0
            
        c_opt = curr_vars[:2*n].reshape(n, 2)
        r_opt = curr_vars[2*n:]
        
        # --- Post-processing: Exact feasible radii assignment ---
        # Recompute radii based strictly on clearances to guarantee validity
        # and capture any slack left by the optimizer.
        r_feas = np.zeros(n)
        for i in range(n):
            # Distance to boundaries
            d_boundary = min(c_opt[i, 0], 1.0 - c_opt[i, 0],
                             c_opt[i, 1], 1.0 - c_opt[i, 1])
            # Distance to other centers
            d_pairs = np.linalg.norm(c_opt - c_opt[i], axis=1)
            d_pairs[i] = np.inf
            d_pairwise = np.min(d_pairs)
            
            # Radius is limited by both, with a tiny safety buffer
            r_feas[i] = min(d_boundary, d_pairwise / 2.0) - 1e-9
            
        # Validate and update best
        if np.all(r_feas > 1e-10):
            s = np.sum(r_feas)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = r_feas.copy()
                
    # Fallback (should not be reached with this robust setup)
    if best_centers is None:
        best_centers = hex_pts
        best_radii = np.full(n, 0.08)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, float(best_sum)
