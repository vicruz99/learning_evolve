import numpy as np
from scipy.optimize import minimize

def smooth_min_clearance(centers, k=50.0):
    """
    Computes a smooth approximation of the minimum clearance 
    (distance to boundaries and half pairwise distances).
    Maximizing this value pushes circles apart and into corners optimally.
    """
    n = centers.shape[0]
    # Distances to four boundaries
    b_left = centers[:, 0]
    b_right = 1.0 - centers[:, 0]
    b_bottom = centers[:, 1]
    b_top = 1.0 - centers[:, 1]
    b_clr = np.minimum(np.minimum(b_left, b_right), np.minimum(b_bottom, b_top))
    
    # Pairwise half-distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    # Extract upper triangle to avoid duplicates
    upper_tri = dists[np.triu_indices(n, k=1)] / 2.0
    
    all_vals = np.concatenate([b_clr, upper_tri])
    
    # Stable log-sum-exp approximation of min(all_vals)
    max_v = np.max(all_vals)
    shifted = all_vals - max_v
    return -np.log(np.sum(np.exp(-k * shifted))) / k - max_v

def objective(x, n, k):
    """Objective: minimize negative smooth min clearance => maximize clearance."""
    return -smooth_min_clearance(x.reshape(n, 2), k)

def run_packing():
    n = 26
    k = 50.0  # Sharpness of smooth min approximation
    
    bounds = [(0.0, 1.0)] * (2 * n)
    
    best_val = np.inf
    best_centers = None
    
    np.random.seed(42)
    inits = []
    
    # 1. Hexagonal lattice initialization (optimal density structure)
    r0 = 0.10
    pts = []
    y = r0
    row = 0
    while len(pts) < n:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x + r0 <= 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    hex_init = np.array(pts[:n])
    inits.append(hex_init)
    
    # 2. Systematic perturbations to escape local minima and break symmetry
    for _ in range(20):
        p = hex_init + np.random.uniform(-0.025, 0.025, hex_init.shape)
        p = np.clip(p, 0.05, 0.95)
        inits.append(p)
        
    # 3. Random starts for robustness
    for _ in range(10):
        inits.append(np.random.uniform(0.08, 0.92, (n, 2)))
        
    # Optimize from each configuration
    for cfg in inits:
        try:
            res = minimize(objective, cfg.flatten(), args=(n, k), method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 15000, 'ftol': 1e-15, 'gtol': 1e-13})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(n, 2)
        except Exception:
            continue
            
    # Fallback if optimization fails
    if best_centers is None:
        best_centers = hex_init
        
    # Compute exact clearances for the best layout found
    c = best_centers
    b_clr = np.minimum(c[:, 0], 1.0 - c[:, 0])
    b_clr = np.minimum(b_clr, c[:, 1])
    b_clr = np.minimum(b_clr, 1.0 - c[:, 1])
    
    diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    upper_tri = dists[np.triu_indices(n, k=1)] / 2.0
    
    # The limiting radius is the minimum clearance
    r_max = np.min(np.concatenate([b_clr, upper_tri]))
    
    # Apply a tiny safety margin to strictly satisfy the 1e-12 tolerance in validation
    r_max *= 0.999999
    
    radii = np.full(n, r_max)
    sum_r = float(np.sum(radii))
    
    return best_centers, radii, sum_r
