import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    # --- Phase 1: High-Quality Hexagonal Initialization ---
    # Generate a dense hex lattice and select the 26 points closest to the center
    pts = []
    s = 0.14
    y = s
    row = 0
    while y < 1.2:
        shift = s if row % 2 == 1 else 0.0
        x = s + shift
        while x < 1.2:
            pts.append([x, y])
            x += 2.0 * s
        y += s * np.sqrt(3)
        row += 1
    pts = np.array(pts)
    dists_to_center = np.sum((pts - 0.5)**2, axis=1)
    best_idx = np.argsort(dists_to_center)[:n]
    centers = pts[best_idx].copy()
    
    # Normalize to fit comfortably inside [0,1]
    mn = centers.min(axis=0)
    mx = centers.max(axis=0)
    centers = (centers - mn) / (mx - mn) * 0.88 + 0.06
    
    # --- Phase 2: Radius Expansion with Force Simulation ---
    best_r = 0.05
    best_cfg = centers.copy()
    current_cfg = centers.copy()
    R = 0.05
    
    # Gradually expand radius until no valid configuration can be maintained
    while R < 0.103:
        valid_R = False
        # Try a few perturbations per radius step to escape local minima
        for p in range(3):
            cfg = current_cfg.copy()
            if p > 0:
                cfg += np.random.uniform(-0.005, 0.005, cfg.shape)
                cfg = np.clip(cfg, R, 1.0 - R)
            
            # Force relaxation steps
            for _ in range(300):
                forces = np.zeros_like(cfg)
                
                # Boundary repulsion forces
                for i in range(n):
                    if cfg[i, 0] < R: forces[i, 0] += (R - cfg[i, 0]) * 200.0
                    if cfg[i, 0] > 1.0 - R: forces[i, 0] -= (cfg[i, 0] + R - 1.0) * 200.0
                    if cfg[i, 1] < R: forces[i, 1] += (R - cfg[i, 1]) * 200.0
                    if cfg[i, 1] > 1.0 - R: forces[i, 1] -= (cfg[i, 1] + R - 1.0) * 200.0
                    
                # Pairwise repulsion forces (vectorized)
                diffs = cfg[:, np.newaxis, :] - cfg[np.newaxis, :, :]
                dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-12)
                np.fill_diagonal(dists, np.inf)
                overlap = np.maximum(0.0, 2.0 * R - dists)
                
                # Direction vectors, zeroed where no overlap
                mask = overlap > 0.0
                dir_vec = np.where(mask[:, :, np.newaxis], diffs / dists[:, :, np.newaxis], 0.0)
                
                # Accumulate repulsive forces
                forces -= np.sum(dir_vec * overlap[:, :, np.newaxis] * 100.0, axis=1)
                
                # Update positions
                cfg += forces * 0.005
                cfg = np.clip(cfg, R, 1.0 - R)
                
            # Strict validity check for current R
            valid = True
            for i in range(n):
                if cfg[i, 0] < R or cfg[i, 0] > 1.0 - R or \
                   cfg[i, 1] < R or cfg[i, 1] > 1.0 - R:
                    valid = False
                    break
                    
            if valid:
                d_mat = np.sqrt(np.sum((cfg[:, np.newaxis, :] - cfg[np.newaxis, :, :])**2, axis=2))
                if np.any(d_mat[np.triu_indices(n, k=1)] < 2.0 * R - 1e-6):
                    valid = False
                    
            if valid:
                best_r = R
                best_cfg = cfg.copy()
                current_cfg = cfg.copy()
                valid_R = True
                break # Found valid configuration for this R, move to next
        
        if valid_R:
            R += 0.0005
        else:
            break # Cannot expand further
            
    # --- Phase 3: SLSQP Fine-Tuning ---
    # Optimize sum of radii jointly with centers using the expanded config as seed
    x0 = np.zeros(3 * n)
    x0[0::3] = best_cfg[:, 0]
    x0[1::3] = best_cfg[:, 1]
    x0[2::3] = best_r
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-5, 0.5)] * n
    
    def objective(v):
        return -np.sum(v[2::3])
        
    def constraints(v):
        cx = v[0::3]
        cy = v[1::3]
        r = v[2::3]
        
        c = []
        # Boundary constraints
        c.append(cx - r)
        c.append(1.0 - cx - r)
        c.append(cy - r)
        c.append(1.0 - cy - r)
        
        # Non-overlap constraints
        dx = cx[:, np.newaxis] - cx[np.newaxis, :]
        dy = cy[:, np.newaxis] - cy[np.newaxis, :]
        d = np.sqrt(dx**2 + dy**2)
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        
        idx = np.triu_indices(n, k=1)
        c.append(d[idx] - r_sum[idx])
        
        return np.concatenate(c)
    
    try:
        res = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'ineq', 'fun': constraints},
            options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False}
        )
        
        if np.isfinite(res.fun):
            final_centers = np.column_stack((res.x[0::3], res.x[1::3]))
            final_radii = res.x[2::3].copy()
        else:
            final_centers = best_cfg.copy()
            final_radii = np.full(n, best_r)
    except Exception:
        final_centers = best_cfg.copy()
        final_radii = np.full(n, best_r)
        
    # Safety margin to strictly satisfy the 1e-12 tolerance in validate_packing
    final_radii *= 0.99999
    
    return final_centers, final_radii, float(np.sum(final_radii))
