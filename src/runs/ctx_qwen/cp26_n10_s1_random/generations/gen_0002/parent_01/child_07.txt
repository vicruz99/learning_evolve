import numpy as np
from scipy.optimize import minimize

N = 26

def check_valid(c, r):
    """Strict validation matching grader tolerances."""
    if np.any(r < -1e-12): return False
    if np.any(c[:, 0] - r < -1e-9) or np.any(c[:, 0] + r > 1.0 + 1e-9): return False
    if np.any(c[:, 1] - r < -1e-9) or np.any(c[:, 1] + r > 1.0 + 1e-9): return False
    
    dists = np.linalg.norm(c[:, None, :] - c[None, :, :], axis=2)
    rs = r[:, None] + r[None, :]
    iu, ju = np.triu_indices(N, 1)
    return np.all(dists[iu, ju] >= rs[iu, ju] - 1e-9)

def hex_init(r0, shift_x=0.0, shift_y=0.0):
    """Generate hexagonal lattice points."""
    pts = []
    y = r0 + shift_y
    row = 0
    while len(pts) < N:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift + shift_x
        while x + r0 <= 1.0 and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    return np.array(pts[:N])

def obj(vars):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def con(vars):
    """Inequality constraints >= 0 for valid packing."""
    c = vars[:2*N].reshape(-1, 2)
    r = vars[2*N:]
    con_list = []
    con_list.append(c[:, 0] - r)
    con_list.append(1.0 - c[:, 0] - r)
    con_list.append(c[:, 1] - r)
    con_list.append(1.0 - c[:, 1] - r)
    
    dx = c[:, 0, None] - c[None, :, 0]
    dy = c[:, 1, None] - c[None, :, 1]
    d = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(d, np.inf)
    
    iu, ju = np.triu_indices(N, 1)
    con_list.append(d[iu, ju] - (r[iu] + r[ju]))
    return np.concatenate(con_list)

def obj_eq(vars):
    """Objective for equal radius: minimize negative radius."""
    return -vars[-1]

def con_eq(vars):
    """Constraints for equal radius packing."""
    t = vars[-1]
    c = vars[:-1].reshape(-1, 2)
    cons = []
    cons.extend(c[:, 0] - t)
    cons.extend(1.0 - c[:, 0] - t)
    cons.extend(c[:, 1] - t)
    cons.extend(1.0 - c[:, 1] - t)
    dx = c[:, 0, None] - c[None, :, 0]
    dy = c[:, 1, None] - c[None, :, 1]
    d = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(d, np.inf)
    iu, ju = np.triu_indices(N, 1)
    cons.extend(d[iu, ju] - 2.0*t)
    return np.array(cons)

def run_packing():
    np.random.seed(42)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    eq_bounds = [(0.0, 1.0)]*(2*N) + [(0.05, 0.15)]
    
    # Phase 1: Equal radius optimization to find optimal lattice structure
    for trial in range(10):
        sx = np.random.uniform(-0.02, 0.02)
        sy = np.random.uniform(-0.02, 0.02)
        c0 = hex_init(0.09, sx, sy)
        x0 = np.concatenate([c0.flatten(), [0.09]])
        
        res = minimize(obj_eq, x0, method='SLSQP', bounds=eq_bounds,
                       constraints={'type': 'ineq', 'fun': con_eq},
                       options={'maxiter': 4000, 'ftol': 1e-13})
        if res.success:
            r_eq = res.x[-1]
            c_eq = res.x[:-1].reshape(-1, 2)
            if check_valid(c_eq, np.full(N, r_eq)):
                s = r_eq * N
                if s > best_sum:
                    best_sum = s
                    best_c = c_eq.copy()
                    best_r = np.full(N, r_eq)

    # Fallback if Phase 1 fails unexpectedly
    if best_c is None:
        best_c = hex_init(0.09)
        best_r = np.full(N, 0.09)
        best_sum = np.sum(best_r)

    # Phase 2: Unequal radius optimization to exploit asymmetries/boundaries
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.5)] * N
    
    for _ in range(15):
        pert_c = best_c + np.random.uniform(-0.008, 0.008, (N, 2))
        pert_c = np.clip(pert_c, 0.01, 0.99)
        pert_r = best_r + np.random.uniform(-0.005, 0.005, N)
        pert_r = np.clip(pert_r, 1e-5, 0.5)
        
        x0_p = np.concatenate([pert_c.flatten(), pert_r])
        
        res = minimize(obj, x0_p, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': con},
                       options={'maxiter': 5000, 'ftol': 1e-13})
        
        if res.success:
            c_opt = res.x[:2*N].reshape(-1, 2)
            r_opt = res.x[2*N:]
            
            valid = check_valid(c_opt, r_opt)
            if not valid:
                scale = 0.995
                while not valid and scale > 0.9:
                    r_opt *= scale
                    valid = check_valid(c_opt, r_opt)
            
            if valid:
                s = np.sum(r_opt)
                if s > best_sum:
                    best_sum = s
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()

    # Final safety scaling to guarantee strict 1e-12 compliance
    min_scale = 1.0
    c = best_c
    r = best_r
    
    for i in range(N):
        if r[i] > 1e-9:
            min_scale = min(min_scale, c[i,0]/r[i], (1-c[i,0])/r[i], c[i,1]/r[i], (1-c[i,1])/r[i])
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.linalg.norm(c[i] - c[j])
            if d > 1e-9 and (r[i] + r[j]) > 1e-9:
                min_scale = min(min_scale, d / (r[i] + r[j]))
                
    final_scale = min_scale * 0.999999
    best_r *= final_scale
    best_sum = np.sum(best_r)
    
    return best_c, best_r, best_sum
