def objective(vars_arr, n, weight):

    cy = vars_arr[1::3]
    r = vars_arr[2::3]
    
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary penalties: circle must stay within [0, 1]
    pen += np.sum(np.maximum(0, r - cx)**2)
    pen += np.sum(np.maximum(0, r - (1.0 - cx))**2)
    pen += np.sum(np.maximum(0, r - cy)**2)
    pen += np.sum(np.maximum(0, r - (1.0 - cy))**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, 1e9)  # Ignore self-distances
    
    r_sum = r[:, None] + r[None, :]
    tri_idx = np.triu_indices(n, k=1)
    overlap = np.maximum(0, r_sum[tri_idx] - dist[tri_idx])
    pen += np.sum(overlap**2)
    
    return obj + weight * pen
