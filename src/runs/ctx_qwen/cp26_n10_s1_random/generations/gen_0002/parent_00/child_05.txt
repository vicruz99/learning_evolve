import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def equal_radius_constraints(x):
    """Inequality constraints for equal-radius packing: all must be >= 0."""
    c = x[:2*N].reshape(N, 2)
    t = x[-1]
    cons = []
    
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    cons.append(c[:, 0] - t)
    cons.append(1.0 - c[:, 0] - t)
    cons.append(c[:, 1] - t)
    cons.append(1.0 - c[:, 1] - t)
    
    # Pairwise non-overlap: dist^2 >= 4*t^2
    diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist_sq = np.sum(diffs**2, axis=2)
    np.fill_diagonal(dist_sq, 1e12)  # Large value to ignore self-distance
    triu_idx = np.triu_indices(N, k=1)
    cons.append(dist_sq[triu_idx] - 4.0 * t**2)
    
    return np.concatenate([arr.flatten() for arr in cons])

def objective_equal(x):
    """Objective: maximize radius t => minimize -t."""
    return -x[-1]

def solve_radii_lp(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    
    # Boundary limits for each radius
    b_dist = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    bounds = [(0.0, bd) for bd in b_dist]
    
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    triu_i, triu_j = np.triu_indices(n, k=1)
    dists = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=2))
    rhs = dists[triu_i, triu_j]
    
    m = len(rhs)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), triu_i] = 1.0
    A_ub[np.arange(m), triu_j] = 1.0
    
    # Maximize sum(r) => minimize -sum(r)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A_ub, b_ub=rhs, bounds=bounds, method='highs')
    
    return res.x if res.success else None

def generate_hex_config(rows_counts, r_init, shift_x=0.0, shift_y=0.0, seed=0):
    """Generate initial hexagonal lattice configuration."""
    rng = np.random.default_rng(seed)
    pts = []
    y = r_init + shift_y + 0.05
    dy = np.sqrt(3) * r_init
    
    for idx, count in enumerate(rows_counts):
        x_start = r_init + shift_x + (idx % 2) * r_init + 0.05
        step = max((1.0 - 2 * r_init - 0.1) / max(count - 1, 1), 0.01)
        for k in range(count):
            if len(pts) >= N: break
            x = x_start + k * step
            if x + r_init <= 1.0 and y + r_init <= 1.0:
                pts.append([x, y])
        y += dy
        
    # Pad with random points if lattice yields too few
    while len(pts) < N:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
        
    return np.array(pts[:N])

def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    np.random.seed(42)
    
    # Diverse initial configurations
    configs = []
    configs.append(generate_hex_config([5, 6, 5, 6, 4], 0.09, seed=0))
    configs.append(generate_hex_config([6, 5, 6, 5, 4], 0.09, seed=1))
    configs.append(generate_hex_config([5, 5, 6, 6, 4], 0.09, seed=2))
    
    # Perturbed hexagonal starts
    for i in range(3):
        cfg = configs[0].copy()
        cfg += np.random.normal(0, 0.02, cfg.shape)
        cfg = np.clip(cfg, 0.05, 0.95)
        configs.append(cfg)
        
    # Regular grid start
    grid = np.array([(i*0.19+0.1, j*0.19+0.1) for j in range(5) for i in range(5)] + [[0.5, 0.5]])
    configs.append(grid[:N])
    
    bounds_eq = [(0.0, 1.0)] * (2*N) + [(0.05, 0.12)]
    
    # Stage 1: Optimize centers for maximum equal radius
    for cfg in configs:
        x0 = np.concatenate([cfg.flatten(), [0.085]])
        cons = {'type': 'ineq', 'fun': equal_radius_constraints}
        
        try:
            res = minimize(objective_equal, x0, method='SLSQP', bounds=bounds_eq,
                         constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
            
            # Only proceed if we found a reasonable radius
            if res.x[-1] > 0.101:
                c_opt = res.x[:2*N].reshape(N, 2)
                
                # Stage 2: Extract optimal variable radii via LP
                r_opt = solve_radii_lp(c_opt)
                if r_opt is not None and np.all(r_opt > 0):
                    s = np.sum(r_opt)
                    if s > best_sum:
                        best_sum = s
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
        except Exception:
            continue
            
    # Stage 3: Local refinement by perturbing centers and re-solving LP
    if best_centers is not None:
        current_sum = best_sum
        for _ in range(3):  # 3 rounds of refinement
            improved = False
            for _ in range(30):
                # Small Gaussian perturbation
                pert = best_centers + np.random.normal(0, 0.004, best_centers.shape)
                pert = np.clip(pert, 0.01, 0.99)
                
                r_pert = solve_radii_lp(pert)
                if r_pert is not None:
                    s_pert = np.sum(r_pert)
                    if s_pert > current_sum + 1e-9:
                        current_sum = s_pert
                        best_centers = pert.copy()
                        best_radii = r_pert.copy()
                        improved = True
            if not improved:
                break
                
    # Final safety projection to guarantee strict validity within tolerance
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if r < 1e-9: continue
        scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
        
    for i in range(N):
        for j in range(i+1, N):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    # Apply shrink with minimal margin for numerical safety
    best_radii *= max(scale * 0.999999, 0.0)
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
