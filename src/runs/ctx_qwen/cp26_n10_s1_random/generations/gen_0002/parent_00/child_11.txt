import numpy as np
from scipy.optimize import minimize, linprog

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def get_constraints(vars_arr, n):
    """
    Computes inequality constraints >= 0 for SLSQP.
    Variables layout: [x0, y0, x1, y1, ..., r0, r1, ...] -> actually interleaved in vars_arr as [x0,y0,r0,x1,y1,r1...]
    But here we assume vars_arr is flat: [x0, x1, ..., x25, y0, y1, ..., y25, r0, r1, ..., r25] for easier slicing
    Wait, standard is [x0, y0, r0, x1, y1, r1, ...] -> vars_arr[0::3] etc.
    I'll stick to the interleaved format used in optimization.
    """
    cx = vars_arr[0::3]
    cy = vars_arr[1::3]
    r = vars_arr[2::3]
    cons = []
    
    # Boundary constraints: r <= x <= 1-r  =>  x-r >= 0, 1-x-r >= 0
    cons.append(cx - r)
    cons.append(1.0 - cx - r)
    cons.append(cy - r)
    cons.append(1.0 - cy - r)
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    triu_idx = np.triu_indices(n, k=1)
    cons.append(dist_sq[triu_idx] - r_sum[triu_idx]**2)
    
    return np.concatenate(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_result = None
    
    np.random.seed(42)

            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
    base_cfg = np.array(pts[:n])

    min_c = base_cfg.min(axis=0)
    max_c = base_cfg.max(axis=0)
    base_cfg = (base_cfg - min_c) / (max_c - min_c)
    base_cfg = base_cfg * 0.85 + 0.075
    
    # 2. Create diverse initial configurations via perturbation
    configs = [base_cfg]
    for _ in range(12):
        cfg = base_cfg.copy()
        cfg += np.random.normal(0, 0.015, cfg.shape)
        cfg = np.clip(cfg, 0.05, 0.95)
        configs.append(cfg)
        
    # Bounds: centers in [0,1], radii in [1e-6, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    
    for cfg in configs:
        # 3. Feasible radius initialization to guarantee SLSQP starts in valid region
        r_init = np.zeros(n)
        for i in range(n):
            min_d = min(cfg[i,0], 1-cfg[i,0], cfg[i,1], 1-cfg[i,1])
            for j in range(n):
                if i != j:
                    d = np.linalg.norm(cfg[i] - cfg[j])
                    if d < min_d:
                        min_d = d
            r_init[i] = min_d * 0.45  # Strictly feasible
            
        x0 = np.zeros(3*n)
        x0[0::3] = cfg[:, 0]
        x0[1::3] = cfg[:, 1]
        x0[2::3] = r_init

            res = minimize(
                lambda v: -np.sum(v[2::3]),
                x0,
                method='SLSQP',
                bounds=bounds,

        best_result = (fb_centers, fb_radii, np.sum(fb_radii))
