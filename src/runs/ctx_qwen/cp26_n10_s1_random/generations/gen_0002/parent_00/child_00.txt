import numpy as np
from scipy.optimize import minimize

def compute_obj_and_penalty(vars_arr, n, mu, tri_i, tri_j):
    """
    Computes the objective (negative sum of radii) plus a weighted penalty 
    for boundary and overlap violations.
    """
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    obj = -np.sum(r)
    penalty = 0.0
    
    # Boundary penalties: circle must stay within [0, 1]
    # r <= x  => x - r >= 0
    penalty += np.sum(np.maximum(0.0, r - x)**2)
    penalty += np.sum(np.maximum(0.0, r - (1.0 - x))**2)
    penalty += np.sum(np.maximum(0.0, r - y)**2)
    penalty += np.sum(np.maximum(0.0, r - (1.0 - y))**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    
    r_sum = r[:, None] + r[None, :]
    # Only evaluate upper triangular pairs to avoid duplication
    violations = r_sum[tri_i, tri_j] - dist[tri_i, tri_j]
    penalty += np.sum(np.maximum(0.0, violations)**2)
    
    return obj + mu * penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    np.random.seed(42)
    tri_i, tri_j = np.triu_indices(n, k=1)
    
    # Variable bounds: [x0, y0, r0, x1, y1, r1, ...]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    configs = []
    
    # 1. Hexagonal lattice pattern (6-5-6-5-4 rows) - highly dense topology for N=26
    def create_hex_pattern():
        pts = []
        R = 0.10
        dy = np.sqrt(3) * R
        y = R
        row = 0
        counts = [6, 5, 6, 5, 4]
        for cnt in counts:
            shift = R if row % 2 == 1 else 0.0
            x = R + shift
            for _ in range(cnt):
                pts.append([x, y])
                x += 2.0 * R
            y += dy
            row += 1
        return np.array(pts[:n])
        
    configs.append(create_hex_pattern())
    
    # 2. Perturbed hexagonal patterns to escape symmetry/local minima
    for _ in range(6):
        cfg = create_hex_pattern() + np.random.normal(0, 0.008, size=(n, 2))
        cfg = np.clip(cfg, 0.05, 0.95)
        configs.append(cfg)
        
    # 3. Regular 5x5 grid + 1 center circle
    grid = np.array([(0.1 + i*0.2, 0.1 + j*0.2) for j in range(5) for i in range(5)] + [(0.5, 0.5)])
    configs.append(grid[:n])
    
    # 4. Random valid starts
    for _ in range(4):
        cfg = np.random.uniform(0.15, 0.85, size=(n, 2))
        configs.append(cfg)
        
    # Optimize from each configuration
    for cfg in configs:
        x0 = np.zeros(3 * n)
        x0[0::3] = cfg[:, 0]
        x0[1::3] = cfg[:, 1]
        x0[2::3] = 0.06  # Start with a small feasible radius
        
        # Sequential penalty method: gradually increase mu to enforce constraints strictly
        mu = 2000.0
        curr_x = x0.copy()
        
        for _ in range(6):
            res = minimize(
                compute_obj_and_penalty,
                curr_x,
                args=(n, mu, tri_i, tri_j),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 4000, 'ftol': 1e-13}
            )
            curr_x = res.x
            mu *= 3.0  # Increase penalty weight iteratively
            
        c_opt = curr_x[:2*n].reshape(n, 2)
        r_opt = curr_x[2*n:]
        
        # Safety projection: compute exact maximal scaling factor to guarantee validity
        scale = 1.0
        for i in range(n):
            x, y, r = c_opt[i, 0], c_opt[i, 1], r_opt[i]
            if r < 1e-9: continue
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
        for i in range(n):
            for j in range(i+1, n):
                d = np.linalg.norm(c_opt[i] - c_opt[j])
                rs = r_opt[i] + r_opt[j]
                if rs < 1e-9: continue
                scale = min(scale, d / rs)
                
        # Apply scaling with a tiny margin for numerical stability against checker tolerance
        r_final = r_opt * scale * 0.999995
        current_sum = np.sum(r_final)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = c_opt.copy()
            best_radii = r_final.copy()
            
    # Fallback configuration (should rarely be reached)
    if best_centers is None:
        r_fb = 0.09
        best_centers = np.array([(r_fb + i*2*r_fb, r_fb + j*2*r_fb) for j in range(5) for i in range(5)] + [(0.5, 0.5)])
        best_radii = np.full(n, r_fb)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
