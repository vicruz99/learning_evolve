from scipy.optimize import minimize

def compute_constraints(vars_arr, n):
    """Computes inequality constraints >= 0 for equal-radius packing."""
    t = vars_arr[-1]
    xs = vars_arr[0::2]
    ys = vars_arr[1::2]
    
    # Boundary constraints: x >= t, 1-x >= t, y >= t, 1-y >= t
    b1 = xs - t
    b2 = 1.0 - xs - t
    b3 = ys - t
    b4 = 1.0 - ys - t
    
    # Pairwise non-overlap: dist^2 >= (2t)^2
    dx = xs[:, np.newaxis] - xs[np.newaxis, :]
    dy = ys[:, np.newaxis] - ys[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    np.fill_diagonal(dist_sq, np.inf)
    
    tri_idx = np.triu_indices(n, k=1)
    p = dist_sq[tri_idx] - 4.0 * t**2
    
    return np.concatenate([b1, b2, b3, b4, p])

def objective_func(vars_arr):
    """Objective: maximize t => minimize -t"""
    return -vars_arr[-1]
