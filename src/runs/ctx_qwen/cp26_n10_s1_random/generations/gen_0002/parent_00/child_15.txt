import numpy as np
from scipy.optimize import minimize

def get_constraints_equal(x, n):
    """Inequality constraints for equal-radius packing: boundary and pairwise non-overlap."""
    centers = x[:2 * n].reshape(n, 2)
    r = x[2 * n]
    cons = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons.extend(centers[:, 0] - r)
    cons.extend(1.0 - (centers[:, 0] + r))
    cons.extend(centers[:, 1] - r)
    cons.extend(1.0 - (centers[:, 1] + r))
    
    # Pairwise non-overlap: ||c_i - c_j||^2 >= 4r^2
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_sq = np.sum(diffs**2, axis=2)
    np.fill_diagonal(dists_sq, np.inf)
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cons.extend(dists_sq[mask] - 4.0 * r * r)
    return np.array(cons)

def objective_equal(x):
    """Objective for equal packing: minimize negative radius => maximize r."""
    return -x[-1]

def get_constraints_unequal(x, n):
    """Inequality constraints for unequal-radius packing."""
    centers = x[:2 * n].reshape(n, 2)
    r = x[2 * n:]
    cons = []
    
    # Boundary constraints
    cons.extend(centers[:, 0] - r)
    cons.extend(1.0 - (centers[:, 0] + r))
    cons.extend(centers[:, 1] - r)
    cons.extend(1.0 - (centers[:, 1] + r))
    
    # Pairwise non-overlap: ||c_i - c_j||^2 >= (r_i + r_j)^2
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_sq = np.sum(diffs**2, axis=2)
    np.fill_diagonal(dists_sq, np.inf)
    
    r_sum_sq = (r[:, np.newaxis] + r[np.newaxis, :])**2
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cons.extend(dists_sq[mask] - r_sum_sq[mask])
    return np.array(cons)

def objective_unequal(x):
    """Objective for unequal packing: minimize negative sum of radii."""
    return -np.sum(x[2 * 26:])

def force_relaxation(centers, n, steps=4000):
    """Physics-based relaxation to push circles apart while gradually expanding radii."""
    radii = np.full(n, 0.095)
    vel = np.zeros_like(centers)
    dt = 0.004
    damping = 0.90
    
    for _ in range(steps):
        radii *= 1.00004
        forces = np.zeros_like(centers)
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                d = np.linalg.norm(diff)
                req = radii[i] + radii[j]
                if d < req and d > 1e-7:
                    f = 15.0 * (req - d) / d
                    forces[i] += f * diff
                    forces[j] -= f * diff
                    
        # Wall repulsion
        for i in range(n):
            for k in range(2):
                if centers[i, k] < radii[i]:
                    forces[i, k] += 10.0 * (radii[i] - centers[i, k])
                elif centers[i, k] > 1.0 - radii[i]:
                    forces[i, k] -= 10.0 * (centers[i, k] - (1.0 - radii[i]))
                    
        vel = damping * vel + forces * dt
        centers += vel
        centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
        
    # Compute geometrically valid radii from final positions
    final_r = np.zeros(n)
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.linalg.norm(centers[i] - centers[j])
                if d < min_d:
                    min_d = d
        final_r[i] = min_d / 2.0 * 0.99999
        
    return centers, final_r

        fc, fr = force_relaxation(start_cfg, n)

            best_centers = fc
