import numpy as np
from scipy.optimize import minimize, linprog

def get_triu_indices(n):
    return np.triu_indices(n, k=1)

def penalty_objective(vars_arr, n, mu, triu_idx):
    cx = vars_arr[0::3]
    cy = vars_arr[1::3]
    rad = vars_arr[2::3]
    
    # Objective: maximize sum of radii => minimize negative sum
    obj = -np.sum(rad)
    pen = 0.0
    
    # Boundary penalties
    v = rad - cx; pen += np.sum(np.maximum(0, v)**2)
    v = rad - (1.0 - cx); pen += np.sum(np.maximum(0, v)**2)
    v = rad - cy; pen += np.sum(np.maximum(0, v)**2)
    v = rad - (1.0 - cy); pen += np.sum(np.maximum(0, v)**2)
    
    # Overlap penalties
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    dist = np.sqrt(cx_m**2 + cy_m**2)
    np.fill_diagonal(dist, 1e9)
    rad_sum = rad[:, None] + rad[None, :]
    v = rad_sum - dist
    pen += np.sum(np.maximum(0, v[triu_idx])**2)
    
    return obj + mu * pen

def compute_lp_radii(centers, n):
    # LP: maximize sum(r) s.t. r_i <= boundaries, r_i + r_j <= dist(i,j)
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1.0
        # r_i <= x
        A_ub.append(row)
        b_ub.append(x)
        # r_i <= 1-x
        A_ub.append(row.copy())
        b_ub.append(1.0 - x)
        # r_i <= y
        A_ub.append(row.copy())
        b_ub.append(y)
        # r_i <= 1-y
        A_ub.append(row.copy())
        b_ub.append(1.0 - y)
        
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

def optimize_packing(init_centers, init_radii, n, triu_idx):
    c = init_centers.copy()
    r = init_radii.copy()
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    # Continuation method: gradually increase penalty weight
    mu = 500.0
    for step in range(20):
        x0 = np.zeros(3 * n)
        x0[0::3] = c[:, 0]
        x0[1::3] = c[:, 1]
        x0[2::3] = r
        
        res = minimize(penalty_objective, x0, args=(n, mu, triu_idx), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-14})
        
        c = np.column_stack((res.x[0::3], res.x[1::3]))
        r = np.maximum(res.x[2::3], 1e-6)
        mu *= 1.5
        
    return c, r

def generate_configs(n):
    configs = []
    pts = []
    r0 = 0.085
    y = r0
    row = 0
    # Generate hexagonal lattice
    while len(pts) < n:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x < 1.0 - r0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
        row += 1
    configs.append(np.array(pts[:n]))
    
    np.random.seed(42)
    # Perturbed versions to escape local minima
    for i in range(10):
        cfg = configs[0].copy()
        cfg += np.random.uniform(-0.03, 0.03, cfg.shape)
        cfg = np.clip(cfg, 0.05, 0.95)
        configs.append(cfg)
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    triu_idx = get_triu_indices(n)
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    configs = generate_configs(n)
    
    for cfg in configs:
        c_opt, r_opt = optimize_packing(cfg, np.full(n, 0.08), n, triu_idx)
        r_lp = compute_lp_radii(c_opt, n)
        
        if r_lp is not None:
            current_sum = np.sum(r_lp)
            if current_sum > best_sum:
                # Compute exact safe scaling factor to guarantee strict validity
                scale = 1.0
                for i in range(n):
                    x, y = c_opt[i]
                    r = r_lp[i]
                    if r < 1e-12: continue
                    scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
                    
                for i in range(n):
                    for j in range(i+1, n):
                        d = np.hypot(c_opt[i,0]-c_opt[j,0], c_opt[i,1]-c_opt[j,1])
                        rs = r_lp[i] + r_lp[j]
                        if rs < 1e-12: continue
                        scale = min(scale, d / rs)
                        
                # Apply minimal safety margin
                r_safe = r_lp * scale * 0.999995
                s_safe = np.sum(r_safe)
                if s_safe > best_sum:
                    best_sum = s_safe
                    best_centers = c_opt.copy()
                    best_radii = r_safe.copy()
                    
    # Fallback
    if best_centers is None:
        best_centers = configs[0]
        best_radii = np.full(n, 0.09)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
