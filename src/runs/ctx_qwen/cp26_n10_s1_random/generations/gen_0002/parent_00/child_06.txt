def obj_eq(v):

def con_eq(v):
    """Inequality constraints: boundaries and non-overlap >= 0"""

    r = v[-1]
    c = v[:-1].reshape(n, 2)
    cons = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    cons.extend(c[:, 0] - r)
    cons.extend(1.0 - c[:, 0] - r)
    cons.extend(c[:, 1] - r)
    cons.extend(1.0 - c[:, 1] - r)

            cons.append(np.dot(diff, diff) - 4.0 * r * r)
    return np.array(cons)
