import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    def hex_init(perturb=0.0):
        """Generates a hexagonal lattice initialization matching the 5-6-5-6-4 structure."""
        rows_counts = [5, 6, 5, 6, 4]
        pts = []
        r_est = 0.101
        dy = np.sqrt(3) * r_est
        y = r_est + 0.05
        for idx, cnt in enumerate(rows_counts):
            shift = r_est / 2 if idx % 2 == 1 else 0.0
            x_start = r_est + shift + 0.05
            step = (1.0 - 2 * r_est - 0.1 - shift) / max(cnt - 1, 1)
            for k in range(cnt):
                if len(pts) >= n: break
                pts.append([x_start + k * step, y])
            y += dy
        cfg = np.array(pts[:n])
        if perturb > 0:
            cfg += np.random.normal(0, perturb, cfg.shape)
            cfg = np.clip(cfg, 0.05, 0.95)
        return cfg

    # Create diverse starting configurations
    configs = [hex_init(0.0)]
    for _ in range(8):
        configs.append(hex_init(0.03))

    best_sum = -1.0
    best_centers = None
    best_radii = None

    def objective(v):
        """Penalty-based objective: maximize t while enforcing boundaries and non-overlap."""
        c = v[:2*n].reshape(n, 2)
        t = v[2*n]
        obj = -t
        mu = 80000.0
        pen = 0.0
        
        # Boundary penalties: t <= x <= 1-t, t <= y <= 1-t
        pen += np.sum(np.maximum(0, t - c[:, 0])**2)
        pen += np.sum(np.maximum(0, t - (1.0 - c[:, 0]))**2)
        pen += np.sum(np.maximum(0, t - c[:, 1])**2)
        pen += np.sum(np.maximum(0, t - (1.0 - c[:, 1]))**2)
        
        # Overlap penalties: dist(i,j) >= 2*t
        dx = c[:, 0, np.newaxis] - c[:, 0]
        dy = c[:, 1, np.newaxis] - c[:, 1]
        dists = np.sqrt(dx**2 + dy**2)
        np.fill_diagonal(dists, np.inf)
        triu = np.triu_indices(n, k=1)
        viol = np.maximum(0, 2.0*t - dists[triu])**2
        pen += np.sum(viol)
        
        return obj + mu * pen

    bounds = [(0.0, 1.0)] * (2*n) + [(0.08, 0.12)]

    for cfg in configs:
        v0 = np.concatenate([cfg.flatten(), [0.10]])
        try:
            res = minimize(objective, v0, method='L-BFGS-B', bounds=bounds,

            c_opt = res.x[:2*n].reshape(n, 2)
            t_opt = res.x[2*n]

                    best_sum = curr_sum
                    best_centers = c_opt.copy()
                    best_radii = np.full(n, t_opt)
        except Exception:
            pass

    # Fallback configuration
    if best_centers is None:
        best_centers = hex_init()
        best_radii = np.full(n, 0.095)
        best_sum = n * 0.095

    # Final safety projection to guarantee strict validity against validator tolerance
    radii = best_radii.copy()
    centers = best_centers.copy()

    # Check boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            rs = radii[i] + radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)

    radii *= scale * 0.999999
    final_sum = np.sum(radii)
    
    return centers, radii, final_sum
