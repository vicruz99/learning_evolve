import numpy as np
from scipy.optimize import minimize

def compute_penalty(x, n, mu, tri_mask):
    """Computes objective (negative sum of radii) plus smooth penalty for constraint violations."""
    c = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    
    # Boundary penalties: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    pen = 0.0
    pen += np.sum(np.maximum(0, r - c[:,0])**2)
    pen += np.sum(np.maximum(0, r - (1.0 - c[:,0]))**2)
    pen += np.sum(np.maximum(0, r - c[:,1])**2)
    pen += np.sum(np.maximum(0, r - (1.0 - c[:,1]))**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = r[:, None] + r[None, :]
    
    # Use precomputed upper triangular mask for efficiency
    violations = r_sum[tri_mask] - dists[tri_mask]
    pen += np.sum(np.maximum(0, violations)**2)
    
    return -np.sum(r) + mu * pen

def compute_exact_radii(centers, n):
    """Computes the maximum valid radius for each center given the layout."""
    c = centers
    # Distance to boundaries
    b = np.minimum(
        np.minimum(c[:,0], 1.0 - c[:,0]),
        np.minimum(c[:,1], 1.0 - c[:,1])
    )
    # Pairwise half-distances
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    half_dists = dists / 2.0
    
    # Radius is limited by closest boundary and closest neighbor
    radii = np.minimum(b, np.min(half_dists, axis=1))
    return radii
