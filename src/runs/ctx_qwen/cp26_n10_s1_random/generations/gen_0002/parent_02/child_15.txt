import numpy as np
from scipy.optimize import minimize

def objective_penalty(vars, n, mu):
    """
    Objective function combining negative sum of radii with a smooth penalty 
    for boundary violations and circle overlaps.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary penalties: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    pen += np.sum(np.maximum(0, -(x - r))**2)
    pen += np.sum(np.maximum(0, -(1.0 - x - r))**2)
    pen += np.sum(np.maximum(0, -(y - r))**2)
    pen += np.sum(np.maximum(0, -(1.0 - y - r))**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    
    r_sum = r[:, None] + r[None, :]
    viol = r_sum - dist
    triu = np.triu(np.ones((n, n), dtype=bool), k=1)
    pen += np.sum(np.maximum(0, viol[triu])**2)
    
    return obj + mu * pen

def generate_start(idx, n, rng):
    """Generates diverse initial configurations for the optimizer."""
    pts = np.zeros((n, 2))
    if idx == 0:
        rows = [5, 6, 5, 6, 4]
        r0 = 0.1
        y = r0
        row_idx = 0
        cnt = 0

            shift = r0 if row_idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(c):
                if cnt < n:
                    pts[cnt] = [x, y]
                    cnt += 1
                x += 2 * r0
            y += np.sqrt(3) * r0
            row_idx += 1
    elif idx == 1:
        rows = [6, 5, 6, 5, 4]

        row_idx = 0
        cnt = 0

                    cnt += 1
                x += 2 * r0
            y += np.sqrt(3) * r0
            row_idx += 1
    elif idx == 2:
        c = 0

    v = np.zeros(3*n)

def run_packing():
    n = 26
    mu = 100000.0  # High penalty weight to strictly enforce constraints
    bounds = [(0, 1)] * (2*n) + [(1e-5, 0.5)] * n
    
    best_sum = -1.0
    best_vars = None
    rng = np.random.default_rng(42)
    
    # Generate a diverse set of starting points
    starts = []
    for i in range(3):
        starts.append(generate_start(0, n, rng))
        starts.append(generate_start(1, n, rng))
        starts.append(generate_start(2, n, rng))
        starts.append(generate_start(3, n, rng))
        
    # Add perturbed versions to escape symmetric local minima
    for s in starts.copy():
        p = s + rng.uniform(-0.02, 0.02, 3*n)
        p[:2*n] = np.clip(p[:2*n], 0.05, 0.95)
        starts.append(p)
        
    # Phase 1: Penalty-based optimization from multiple starts
    for x0 in starts:
        res = minimize(objective_penalty, x0, args=(n, mu), method='L-BFGS-B',

        r_opt = res.x[2::3]

    for i in range(n):

            rs = final_radii[i] + final_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    # Apply scaling with a tiny buffer for numerical stability
    final_radii *= scale * 0.99999
    best_sum = np.sum(final_radii)
    
    return final_centers, final_radii, best_sum
