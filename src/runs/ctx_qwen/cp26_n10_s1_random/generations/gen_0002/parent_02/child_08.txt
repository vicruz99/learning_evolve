import numpy as np
from scipy.optimize import minimize

def compute_clearance(c):
    """
    Computes clearance values: distances to boundaries and half-pairwise distances.
    """
    n = c.shape[0]
    # Distance to each of the 4 boundaries
    b = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                   np.minimum(c[:, 1], 1.0 - c[:, 1]))
    
    # Pairwise distances
    diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Combine boundary clearances and half-pairwise clearances (upper triangle only)
    return np.concatenate([b, dists[np.triu_indices(n, k=1)] / 2.0])

def smooth_obj(params, k):
    """
    Objective function: negative smooth approximation of the minimum clearance.
    Minimizing this maximizes the minimum clearance.
    """
    c = params.reshape(-1, 2)
    vals = compute_clearance(c)
    
    # Stable log-sum-exp approximation of the minimum
    m = np.min(vals)
    diff = vals - m
    exp_sum = np.sum(np.exp(-k * diff))
    smooth_min_val = -np.log(exp_sum) / k - m
    
    return -smooth_min_val

def run_packing():

    bounds = [(0.0, 1.0)] * (2 * n)

        dy = np.sqrt(3.0) * r_est

            x = r_est + shift

                x += 2.0 * r_est

        pts = np.array(pts[:n])

        configs.append(np.random.uniform(0.1, 0.9, (n, 2)))

    c = best_c
