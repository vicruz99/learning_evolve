import numpy as np
from scipy.optimize import minimize

def compute_penalty_objective(vars, n, w):
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary penalties: circles must stay within [0, 1]
    pen += np.sum(np.maximum(0.0, r - c[:,0])**2)
    pen += np.sum(np.maximum(0.0, r + c[:,0] - 1.0)**2)
    pen += np.sum(np.maximum(0.0, r - c[:,1])**2)
    pen += np.sum(np.maximum(0.0, r + c[:,1] - 1.0)**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    np.fill_diagonal(dists, np.inf)
    
    r_sums = r[:, np.newaxis] + r[np.newaxis, :]
    tri_idx = np.triu_indices(n, k=1)
    violations = np.maximum(0.0, r_sums[tri_idx] - dists[tri_idx])
    pen += np.sum(violations**2)
    
    return obj + w * pen

def compute_slsqp_objective(vars):
    return -np.sum(vars[2::3])

def compute_slsqp_constraints(vars, n):
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    con = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con.append(c[:,0] - r)
    con.append(1.0 - c[:,0] - r)
    con.append(c[:,1] - r)
    con.append(1.0 - c[:,1] - r)
    
    # Pairwise non-overlap constraints: dist(i,j) >= r_i + r_j
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    r_sums = r[:, np.newaxis] + r[np.newaxis, :]
    tri_idx = np.triu_indices(n, k=1)
    con.append(dists[tri_idx] - r_sums[tri_idx])
    
    return np.concatenate(con)
