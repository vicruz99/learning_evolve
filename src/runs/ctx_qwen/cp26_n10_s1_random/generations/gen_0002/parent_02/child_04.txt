import numpy as np
from scipy.optimize import minimize

def compute_penalty(vars_flat, r, n):
    """
    Computes the squared penalty for boundary and overlap violations
    for a given fixed radius r.
    """
    c = vars_flat.reshape(n, 2)
    x, y = c[:, 0], c[:, 1]
    
    # Boundary penalties: circle must be within [r, 1-r]
    p = np.sum(np.maximum(0.0, r - x)**2)
    p += np.sum(np.maximum(0.0, r - (1.0 - x))**2)
    p += np.sum(np.maximum(0.0, r - y)**2)
    p += np.sum(np.maximum(0.0, r - (1.0 - y))**2)
    
    # Overlap penalties: pairwise distance must be >= 2*r
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    
    overlaps = np.maximum(0.0, 2.0 * r - dists)
    p += np.sum(overlaps**2)
    
    return p

def run_packing():
    n = 26
    np.random.seed(42)
    
    # Generate diverse initial hexagonal configurations
    configs = []
    patterns = [[6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
                [5, 5, 6, 6, 4], [6, 5, 5, 6, 4], [5, 6, 6, 5, 4]]
    
    for pat in patterns:
        pts = []
        r_init = 0.10
        y = r_init
        row = 0
        for cnt in pat:
            shift = r_init if row % 2 == 1 else 0.0
            x = r_init + shift
            for _ in range(cnt):
                pts.append([x, y])
                x += 2 * r_init
            y += np.sqrt(3) * r_init
            row += 1
        
        cfg = np.array(pts[:n])
        # Normalize to fit comfortably in [0, 1]
        cfg = (cfg - cfg.min(axis=0)) / (cfg.max(axis=0) - cfg.min(axis=0)) * 0.8 + 0.1
        configs.append(cfg)
        
        # Add perturbed versions
        for _ in range(2):
            p = cfg + np.random.uniform(-0.02, 0.02, cfg.shape)
            p = np.clip(p, 0.05, 0.95)
            configs.append(p)

    best_r = 0.08
    best_centers = configs[0].copy()
    
    low, high = 0.08, 0.12
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # Bisection search on radius to find maximum feasible r
    for _ in range(40):
        if high - low < 1e-7:
            break
        mid = (low + high) / 2.0
        
        found_valid = False
        best_p = np.inf
        best_c = None
        
        # Shuffle to try different starting points
        indices = np.random.permutation(len(configs))
        
        for idx in indices[:10]:
            cfg = configs[idx]
            res = minimize(compute_penalty, cfg.flatten(), args=(mid, n),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 5000, 'ftol': 1e-15})
            
            if res.fun < best_p:
                best_p = res.fun
                best_c = res.x.reshape(n, 2)
                
            # If penalty is sufficiently small, consider r feasible
            if res.fun < 1e-5:
                found_valid = True
                best_c = res.x.reshape(n, 2)
                break
        
        if found_valid:
            low = mid
            best_r = mid
            best_centers = best_c.copy()
            # Prioritize successful configurations
            configs.insert(0, best_c)
        else:
            high = mid
            if best_c is not None:
                configs.insert(0, best_c)

    # Final precise clearance calculation to ensure strict validity
    c = best_centers
    clearances = np.zeros(n)
    for i in range(n):
        # Distance to boundaries
        d_min = min(c[i, 0], 1.0 - c[i, 0], c[i, 1], 1.0 - c[i, 1])
        # Half distance to other circles
        for j in range(n):
            if i != j:
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = np.sqrt(dx*dx + dy*dy)
                if d < d_min:
                    d_min = d
        clearances[i] = d_min / 2.0
        
    # Use the minimum clearance as the guaranteed safe radius
    final_r = np.min(clearances)
    # Apply a tiny margin to strictly satisfy the 1e-12 tolerance in validation
    final_r *= 0.99999
    
    radii = np.full(n, final_r)
    sum_radii = float(np.sum(radii))
    
    return best_centers, radii, sum_radii
