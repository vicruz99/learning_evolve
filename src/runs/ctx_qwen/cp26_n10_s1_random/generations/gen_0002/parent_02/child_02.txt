from scipy.optimize import minimize

N = 26

def objective(vars_flat):
    """Objective: minimize negative radius (equivalent to maximizing radius)"""
    return -vars_flat[-1]

def constraints(vars_flat):
    """
    Inequality constraints >= 0 for valid equal-radius packing:
    1. Boundary: r <= x <= 1-r, r <= y <= 1-r
    2. Non-overlap: dist(i,j)^2 >= 4*r^2
    """
    x = vars_flat[0::2]
    y = vars_flat[1::2]
    r = vars_flat[-1]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    np.fill_diagonal(dist_sq, np.inf)
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(dist_sq[mask] - 4.0 * r**2)
    
    return np.concatenate(c)

def run_packing():
    bounds = [(0, 1)] * (2 * N) + [(0.05, 0.15)]
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_r = 0.0
    best_centers = None
    np.random.seed(42)
    
    # Generate diverse hexagonal initial configurations
    configs = []
    for row_counts in [[5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [4,6,6,6,4]]:

        pts = []

                    pts.append([x, y])
                x += 2 * r_init
            y += np.sqrt(3) * r_init
            row_idx += 1
        configs.append(np.array(pts[:N]))

        x0[1::2] = cfg[:, 1]
        x0[-1] = 0.095

        radii[i] = min_d / 2.0

    return best_centers, radii, np.sum(radii)
