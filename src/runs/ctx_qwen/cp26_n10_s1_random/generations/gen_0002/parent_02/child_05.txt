import numpy as np
import scipy.optimize as opt


def objective_func(vars):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2::3])


def constraint_func(vars, n):
    """Inequality constraints >= 0 for valid packing."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]

    c_list = []
    # Boundary constraints
    c_list.append(x - r)
    c_list.append(1.0 - x - r)
    c_list.append(y - r)
    c_list.append(1.0 - y - r)

    # Pairwise non-overlap
    rows, cols = np.triu_indices(n, k=1)
    x_diff = x[rows] - x[cols]
    y_diff = y[rows] - y[cols]
    dists = np.sqrt(x_diff ** 2 + y_diff ** 2)
    r_sums = r[rows] + r[cols]
    c_list.append(dists - r_sums)

    return np.concatenate(c_list)


def penalty_func(vars, n, mu):
    """Penalty-based objective for L-BFGS-B."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]

    obj = -np.sum(r)

    # Boundary penalties
    penalty = 0.0
    v1 = x - r
    v2 = 1.0 - x - r
    v3 = y - r
    v4 = 1.0 - y - r
    penalty += np.sum(np.maximum(0, -v1) ** 2)
    penalty += np.sum(np.maximum(0, -v2) ** 2)
    penalty += np.sum(np.maximum(0, -v3) ** 2)
    penalty += np.sum(np.maximum(0, -v4) ** 2)

    # Overlap penalties
    x_diff = x[:, np.newaxis] - x[np.newaxis, :]
    y_diff = y[:, np.newaxis] - y[np.newaxis, :]
    dists = np.sqrt(x_diff ** 2 + y_diff ** 2)
    np.fill_diagonal(dists, np.inf)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    viol = r_sum - dists
    penalty += np.sum(np.maximum(0, viol) ** 2)

    return obj + mu * penalty


def generate_hex_config(n, row_dists, r_init):
    """Generate hexagonal lattice configuration."""
    y = r_init
    row_idx = 0
    pts = []
    for count in row_dists:
        shift = r_init if row_idx % 2 == 1 else 0.0
        x = r_init + shift
        for _ in range(count):
            if len(pts) < n:
                pts.append([x, y])
            x += 2 * r_init
        y += np.sqrt(3) * r_init
        row_idx += 1
    return np.array(pts[:n])


def validate_solution(centers, radii, n):
    """Strict validation of packing."""
    x_opt = centers[:, 0]
    y_opt = centers[:, 1]
    r_opt = radii

    if np.any(r_opt < 1e-10):
        return False

    if np.any(x_opt - r_opt < -1e-10) or np.any(x_opt + r_opt > 1 + 1e-10):
        return False
    if np.any(y_opt - r_opt < -1e-10) or np.any(y_opt + r_opt > 1 + 1e-10):
        return False

    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    r_sums = r_opt[:, np.newaxis] + r_opt[np.newaxis, :]
    rows, cols = np.triu_indices(n, k=1)
    if np.any(dists[rows, cols] < r_sums[rows, cols] - 1e-10):
        return False

    return True
