import numpy as np
from scipy.optimize import minimize

def compute_clearances(centers, n):
    """Computes all relevant clearances: boundaries and half-pairwise distances."""
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Distance to nearest boundary
    b = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Half pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    p = dists[np.triu_indices(n, k=1)] / 2.0
    
    return np.concatenate([b, p])

def smooth_min(vals, k):
    """Smooth approximation of the minimum value using log-sum-exp."""
    max_val = np.max(vals)
    exp_vals = np.exp(-k * (vals - max_val))
    return -np.log(np.sum(exp_vals)) / k - max_val

def objective(params, n, k):
    """Objective to maximize the smooth minimum clearance."""
    c = params.reshape(n, 2)
    vals = compute_clearances(c, n)
    return -smooth_min(vals, k)

def get_hex_init(n, r_est, seed=None):
    """Generates a hexagonal lattice initialization, optionally perturbed."""
    pts = []
    y = r_est
    row = 0
    while len(pts) < n:
        shift = r_est if row % 2 == 1 else 0.0
        x = r_est + shift
        while x + r_est <= 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2 * r_est
        y += r_est * np.sqrt(3)
        row += 1
        
    if seed is not None:
        np.random.seed(seed)
        pts = np.array(pts[:n])
        pts += np.random.uniform(-0.015, 0.015, pts.shape)
        pts = np.clip(pts, 0.05, 0.95)
        return pts
    return np.array(pts[:n])

def run_packing():
    n = 26
    best_val = -np.inf
    best_centers = None
    
    # Generate diverse starting configurations
    starts = []
    starts.append(get_hex_init(n, 0.095))
    for i in range(15):
        starts.append(get_hex_init(n, 0.095, seed=42 + i))
        
    for r_init in [0.09, 0.10]:
        starts.append(get_hex_init(n, r_init))
        for i in range(5):
            starts.append(get_hex_init(n, r_init, seed=200 + i))
            
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # Optimize each configuration with annealing sharpness
    for cfg in starts:
        current_x = cfg.flatten()
        for stage_k in [40.0, 100.0, 250.0, 500.0]:
            try:
                res = minimize(objective, current_x, args=(n, stage_k), method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': 4000, 'ftol': 1e-14})
                current_x = res.x
            except Exception:
                pass
                
        c_opt = current_x.reshape(n, 2)
        min_clr = np.min(compute_clearances(c_opt, n))
        
        if min_clr > best_val:
            best_val = min_clr
            best_centers = c_opt.copy()
            
    # High-precision polishing on the best found configuration
    if best_centers is not None:
        best_x = best_centers.flatten()
        for stage_k in [800.0, 1500.0]:
            try:
                res = minimize(objective, best_x, args=(n, stage_k), method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-15})
                best_centers = res.x.reshape(n, 2)
            except Exception:
                pass
    else:
        best_centers = get_hex_init(n, 0.095)

    # Compute exact maximum feasible equal radius and apply safety margin
    final_r = np.min(compute_clearances(best_centers, n)) * 0.99999
    final_radii = np.full(n, final_r)
    
    return best_centers, final_radii, float(np.sum(final_radii))
