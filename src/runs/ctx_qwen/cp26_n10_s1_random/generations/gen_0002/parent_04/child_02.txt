import numpy as np
from scipy.optimize import minimize

def obj_eq(vars_array):
    """Objective for equal-radius phase: minimize negative radius R"""
    return -vars_array[-1]

def con_eq(vars_array, n):
    """Constraints for equal-radius phase: boundaries and pairwise non-overlap"""
    R = vars_array[-1]
    c = vars_array[:2*n].reshape(n, 2)
    con = []
    
    # Boundary constraints
    con.extend(c[:, 0] - R)
    con.extend(1.0 - c[:, 0] - R)
    con.extend(c[:, 1] - R)
    con.extend(1.0 - c[:, 1] - R)
    
    # Pairwise squared distance constraints
    dx = c[:, 0:1] - c[:, 0:1].T
    dy = c[:, 1:2] - c[:, 1:2].T
    d2 = dx**2 + dy**2
    np.fill_diagonal(d2, 1.0)  # Self-distance doesn't matter
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    con.extend(d2[mask] - 4.0 * R * R)
    return np.array(con)

def obj_uneq(vars_array, n):
    """Objective for unequal-radius phase: minimize negative sum of radii"""
    return -np.sum(vars_array[:n])

def con_uneq(vars_array, n, triu_idx):
    """Constraints for unequal-radius phase: pairwise non-overlap only (boundaries handled by parameterization)"""
    r = vars_array[:n]
    u = vars_array[n:2*n]
    v = vars_array[2*n:3*n]
    
    # Parameterized coordinates guarantee r <= x <= 1-r and r <= y <= 1-r
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    rs2 = rs**2
    
    return d2[triu_idx] - rs2[triu_idx]

def get_hex_init(n, r_init, rot=0.0):
    """Generates a hexagonal lattice initialization, optionally rotated"""
    pts = []
    y = r_init
    row = 0
    while len(pts) < n + 10:  # Generate extra points to fill square after rotation
        shift = r_init if row % 2 == 1 else 0.0
        x = r_init + shift
        while x + r_init <= 1.0:
            pts.append([x, y])
            x += 2.0 * r_init
        y += np.sqrt(3) * r_init
        row += 1
    pts = np.array(pts[:n])
    
    if rot != 0.0:
        cos_t, sin_t = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[cos_t, -sin_t], [sin_t, cos_t]])
        # Re-center and scale to fit within [0.1, 0.9] safely
        pts -= pts.min(axis=0)
        pts /= pts.max(axis=0)
        pts = pts * 0.8 + 0.1
        
    return pts

def run_packing():
    n = 26
    triu_idx = np.triu_indices(n, k=1)
    
    # Phase 1: Find optimal center layout using equal-radius optimization
    best_eq_r = 0.0
    best_eq_centers = None
    
    # Diverse initializations to escape local minima
    inits = [
        get_hex_init(n, 0.10, rot=0.0),
        get_hex_init(n, 0.10, rot=0.2),
        get_hex_init(n, 0.10, rot=0.4),
        get_hex_init(n, 0.09, rot=0.0),
    ]
    
    for cfg in inits:
        x0 = np.concatenate([cfg.flatten(), [0.09]])
        bounds_eq = [(0.0, 1.0)] * (2 * n) + [(0.05, 0.15)]
        
        try:
            res = minimize(obj_eq, x0, method='SLSQP', bounds=bounds_eq,
                           constraints={'type': 'ineq', 'fun': con_eq, 'args': (n,)},
                           options={'maxiter': 2000, 'ftol': 1e-12})
            if np.isfinite(res.fun) and -res.fun > best_eq_r:
                best_eq_r = -res.fun
                best_eq_centers = res.x[:2*n].reshape(n, 2)
        except Exception:
            pass
            
    if best_eq_centers is None:
        best_eq_centers = inits[0]
        best_eq_r = 0.10
        
    # Phase 2: Convert to parameterized form and optimize unequal radii
    # x = r + (1-2r)u  =>  u = (x-r)/(1-2r)
    r0 = np.full(n, best_eq_r)
    u0 = (best_eq_centers[:, 0] - r0) / (1.0 - 2.0 * r0)

    centers_out = np.column_stack((x_out, y_out))

    c_vals = con_uneq(final_best_vars, n, triu_idx)

        y_out = r_out + (1.0 - 2.0 * r_out) * v_out
        centers_out = np.column_stack((x_out, y_out))
        
    return centers_out, r_out, float(np.sum(r_out))
