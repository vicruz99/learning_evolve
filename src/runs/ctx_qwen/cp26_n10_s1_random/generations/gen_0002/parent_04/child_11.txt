import numpy as np
from scipy.optimize import minimize

def objective(vars_flat, n):
    """Objective: maximize sum of radii => minimize negative sum."""
    r = vars_flat[:n]
    return -np.sum(r)

def constraint_overlap(vars_flat, n, triu_i, triu_j):
    """
    Computes pairwise non-overlap constraints:
    dist_sq(i,j) - (r_i + r_j)^2 >= 0
    Returns array of constraint values (all must be >= 0).
    """
    r = vars_flat[:n]
    x = vars_flat[n:2*n]
    y = vars_flat[2*n:3*n]
    
    # Vectorized squared distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    # Squared sum of radii
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Return only upper triangular constraints (i < j)
    return dist_sq[triu_i, triu_j] - r_sum_sq[triu_i, triu_j]
