import numpy as np
from scipy.optimize import minimize

def eq_objective(v):
    """Objective for Phase 1: maximize equal radius t => minimize -t"""
    return -v[-1]

def eq_constraint(v, n):
    """Constraints for Phase 1: boundary and pairwise distances >= 2t"""
    xs = v[0::2]
    ys = v[1::2]
    t = v[-1]
    con = []
    # Boundary constraints
    con.extend(xs - t)
    con.extend(1.0 - xs - t)
    con.extend(ys - t)
    con.extend(1.0 - ys - t)
    # Pairwise constraints
    dx = xs[:, np.newaxis] - xs[np.newaxis, :]
    dy = ys[:, np.newaxis] - ys[np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    con.extend(dists[dists < np.inf] - 2.0 * t)
    return np.array(con)

def var_objective(v, n):
    """Objective for Phase 2: maximize sum of radii => minimize -sum(r)"""
    return -np.sum(v[:n])

def var_constraint(v, triu_idx):
    """Constraints for Phase 2: pairwise non-overlap. Boundaries are implicit."""
    n = triu_idx[0].shape[0]
    r = v[:n]
    u = v[n:2*n]
    vt = v[2*n:3*n]
    # Parameterization guarantees x in [r, 1-r] and y in [r, 1-y]
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * vt
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[triu_idx] - r_sum[triu_idx]**2

def run_packing():
    n = 26
    triu_idx = np.triu_indices(n, k=1)
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    np.random.seed(42)
    starts = []
    
    # Generate diverse hexagonal initializations
    for shift_x in [0.0, 0.05]:
        for shift_y in [0.0, 0.05]:
            pts = []
            r_est = 0.09
            dy = r_est * np.sqrt(3)
            dx = 2.0 * r_est
            y = r_est + shift_y
            row = 0
            while y < 1.0 - r_est:
                x = r_est + shift_x + (row % 2) * (dx / 2)
                while x < 1.0 - r_est:
                    pts.append([x, y])
                    x += dx
                y += dy
                row += 1
            while len(pts) < n:
                pts.append(np.random.uniform(r_est, 1.0 - r_est, 2))
            starts.append(np.array(pts[:n]))
            
    # Add random perturbations to escape symmetry traps
    for _ in range(4):
        p = starts[0].copy()
        p += np.random.uniform(-0.04, 0.04, p.shape)
        p = np.clip(p, 0.05, 0.95)
        starts.append(p)
        
    # --- Phase 1: Equal Radius Optimization ---
    eq_bounds = [(0.0, 1.0)] * (2 * n) + [(0.05, 0.12)]
    best_eq_t = 0.0
    best_eq_centers = None
    
    for pt in starts:
        v0 = np.concatenate([pt.flatten(), [0.09]])
        try:
            res = minimize(eq_objective, v0, method='SLSQP', bounds=eq_bounds,
                           constraints={'type': 'ineq', 'fun': eq_constraint, 'args': (n,)},
                           options={'maxiter': 15000, 'ftol': 1e-13})
            if res.x[-1] > best_eq_t:
                best_eq_t = res.x[-1]
                best_eq_centers = res.x[:-1].reshape(n, 2)
        except Exception:
            pass
            
    # --- Phase 2: Variable Radius Optimization ---
    var_bounds = [(1e-4, 0.5)] * n + [(0.0, 1.0)] * n + [(0.0, 1.0)] * n
    phase2_starts = []
    
    if best_eq_centers is not None:
        # Convert best equal-radius config to parameterized form
        r_init = best_eq_t * np.ones(n)
        denom = 1.0 - 2.0 * r_init
        u_init = (best_eq_centers[:, 0] - r_init) / denom
        v_init = (best_eq_centers[:, 1] - r_init) / denom
        v0_base = np.concatenate([r_init, u_init, v_init])
        phase2_starts.append(v0_base)
        
        # Perturb base to explore nearby variable-radius optima
        for _ in range(3):
            v0_pert = v0_base.copy()
            v0_pert[:n] += np.random.uniform(-0.005, 0.005, n)
            v0_pert[n:] += np.random.uniform(-0.05, 0.05, 2 * n)
            v0_pert[n:] = np.clip(v0_pert[n:], 0.0, 1.0)
            phase2_starts.append(v0_pert)
    else:
        # Fallback starts from hex grids
        for pt in starts[:3]:
            r_pt = np.full(n, 0.085)
            denom = 1.0 - 2.0 * r_pt
            u_pt = (pt[:, 0] - r_pt) / denom
            v_pt = (pt[:, 1] - r_pt) / denom
            phase2_starts.append(np.concatenate([r_pt, u_pt, v_pt]))
            
    for v0 in phase2_starts:
        try:
            res = minimize(var_objective, v0, args=(n,), method='SLSQP', bounds=var_bounds,
                           constraints={'type': 'ineq', 'fun': var_constraint, 'args': (triu_idx,)},
                           options={'maxiter': 20000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_vals = var_constraint(res.x, triu_idx)
                if np.min(c_vals) >= -1e-7:
                    s = np.sum(res.x[:n])
                    if s > best_sum:
                        best_sum = s
                        r = res.x[:n]
                        u = res.x[n:2*n]
                        vt = res.x[2*n:3*n]
                        x = r + (1.0 - 2.0 * r) * u
                        y = r + (1.0 - 2.0 * r) * vt
                        best_centers = np.column_stack((x, y))
                        best_radii = r
        except Exception:
            pass
            
    # Fallback if all optimizations fail unexpectedly
    if best_centers is None:
        best_centers = starts[0]
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    # Final geometric safeguard to guarantee strict validator compliance
    scale = 1.0
    for i in range(n):
        xi, yi = best_centers[i]
        ri = best_radii[i]
        if ri > 1e-9:
            scale = min(scale, xi / ri, (1.0 - xi) / ri, yi / ri, (1.0 - yi) / ri)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    if scale < 1.0:
        best_radii *= scale * 0.99999
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
