import numpy as np
from scipy.optimize import minimize, linprog

def get_constraints(vars, n):
    """Computes inequality constraints >= 0 for valid packing."""
    cx = vars[0::3]
    cy = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: circles inside [0,1]
    b1 = cx - r
    b2 = 1.0 - cx - r
    b3 = cy - r
    b4 = 1.0 - cy - r
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    cx_mat = cx[:, None] - cx[None, :]
    cy_mat = cy[:, None] - cy[None, :]
    dr_mat = r[:, None] + r[None, :]
    
    dist_sq = cx_mat**2 + cy_mat**2
    dr_sq = dr_mat**2
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    overlap = dist_sq[i_idx, j_idx] - dr_sq[i_idx, j_idx]
    
    return np.concatenate([b1, b2, b3, b4, overlap])

def objective(vars):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars[2::3])

def refine_radii_lp(centers, n):
    """
    For fixed centers, solve an LP to find the maximum sum of radii
    subject to non-overlap and boundary constraints.
    """
    cx, cy = centers[:, 0], centers[:, 1]
    
    # Maximum radius allowed by boundaries for each circle
    b_u = np.minimum(np.minimum(cx, 1.0 - cx), np.minimum(cy, 1.0 - cy))
    
    # Pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dist_mat = np.sqrt(np.sum(diff**2, axis=2))
    
    # LP: maximize sum(r) => minimize -sum(r)
    c_obj = -np.ones(n)
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    n_pairs = len(i_idx)
    
    # Build constraint matrix A_ub @ r <= b_ub
    A_ub = np.zeros((n + n_pairs, n))
    b_ub = np.zeros(n + n_pairs)
    
    # Boundary constraints: r_i <= b_u_i
    A_ub[:n, :] = np.eye(n)
    b_ub[:n] = b_u
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    A_ub[n:n+n_pairs, i_idx] = 1.0
    A_ub[n:n+n_pairs, j_idx] = 1.0
    b_ub[n:] = dist_mat[i_idx, j_idx]
    
    bounds_r = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if res.success:
        return res.x
    else:
        return np.full(n, 0.05)

    return get_constraints(vars, n)
