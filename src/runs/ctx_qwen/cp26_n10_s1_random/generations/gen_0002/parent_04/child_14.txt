def penalty_objective(params, n, mu):
    """Computes the negative sum of radii plus penalty for constraint violations."""
    r = params[:n]
    x = params[n:2*n]
    y = params[2*n:3*n]
    
    obj = -np.sum(r)
    
    # Boundary penalty: circles must stay within [0,1]
    min_dist_boundary = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    pen_boundary = np.sum(np.maximum(0.0, r - min_dist_boundary)**2)
    
    # Overlap penalty: pairwise distances must be >= r_i + r_j
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dist = np.sqrt(dx**2 + dy**2 + 1e-12)
    r_sum = r[i_idx] + r[j_idx]
    pen_overlap = np.sum(np.maximum(0.0, r_sum - dist)**2)
    
    return obj + mu * (pen_boundary + pen_overlap)

def slsqp_objective(params, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(params[:n])

def slsqp_constraints(params, n):
    """Computes exact inequality constraints for valid packing."""
    r = params[:n]
    x = params[n:2*n]
    y = params[2*n:3*n]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[i_idx] + r[j_idx]
    c.append(dist - r_sum)
    
    return np.concatenate(c)
