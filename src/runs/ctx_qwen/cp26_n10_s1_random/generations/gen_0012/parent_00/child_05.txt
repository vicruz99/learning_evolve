import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIU_I, TRIU_J = np.triu_indices(N_CIRCLES, k=1)
M = len(TRIU_I)

# Precompute constant inequality matrix for LP
A_ub_global = np.zeros((M, N_CIRCLES))
A_ub_global[np.arange(M), TRIU_I] = 1.0
A_ub_global[np.arange(M), TRIU_J] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[TRIU_I, TRIU_J]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_global, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def objective_func(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def constraint_func(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    c1 = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[TRIU_I] - cx[TRIU_J]
    dy = cy[TRIU_I] - cy[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    
    c2 = dx**2 + dy**2 - dr**2
    return np.concatenate([c1, c2])

def local_search(centers, rng):
    """Simulated Annealing + Deterministic Hill Climbing on centers."""
    c = centers.copy()
    r, curr_sum = solve_lp(c)
    best_c = c.copy()
    best_r = r.copy()
    best_sum = curr_sum
    
    # Phase 1: Simulated Annealing
    temp = 0.025
    step = 0.035
    for _ in range(8000):
        temp *= 0.9995
        step = max(0.0008, step * 0.9997)
        
        i = rng.integers(N_CIRCLES)
        old_pos = c[i].copy()
        
        c[i] = np.clip(old_pos + rng.uniform(-step, step, 2), 1e-4, 1.0 - 1e-4)
        _, new_sum = solve_lp(c)
        
        # Metropolis criterion
        if new_sum > curr_sum:
            curr_sum = new_sum
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_c = c.copy()
                best_r = solve_lp(c)[0]
        else:
            delta = new_sum - curr_sum
            prob = np.exp(min(0.0, delta / max(temp, 1e-9)))
            if rng.random() < prob:
                curr_sum = new_sum
            else:
                c[i] = old_pos
                
    # Phase 2: Deterministic Local Refinement
    c = best_c.copy()
    curr_sum = best_sum
    r = best_r.copy()
    step = 0.008
    
    for _ in range(18):
        improved = False
        for i in range(N_CIRCLES):
            old_pos = c[i].copy()
            moves = [
                (0, step), (0, -step), (step, 0), (-step, 0),
                (step / 1.414, step / 1.414), (-step / 1.414, step / 1.414),
                (step / 1.414, -step / 1.414), (-step / 1.414, -step / 1.414)
            ]
            for dx, dy in moves:
                c[i] = np.clip(old_pos + np.array([dx, dy]), 1e-4, 1.0 - 1e-4)
                _, s = solve_lp(c)
                if s > curr_sum:
                    curr_sum = s
                    r, _ = solve_lp(c)
                    best_c = c.copy()
                    best_r = r.copy()
                    best_sum = curr_sum
                    improved = True
                    break
            if not improved:
                c[i] = old_pos
        step *= 0.88
        if not improved:
            break
            
    return best_c, best_r, best_sum

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5], 
        [6, 4, 6, 4, 6], [4, 6, 5, 6, 5], [7, 6, 6, 7]
    ]
    
    for pat in patterns:
        if sum(pat) != N_CIRCLES:
            continue
        pts = []
        y = 0.10
        dy = 0.165
        for idx, cnt in enumerate(pat):
            shift = 0.092 if idx % 2 == 1 else 0.0
            x = 0.10 + shift
            for _ in range(cnt):
                if len(pts) >= N_CIRCLES:
                    break
                pts.append([x, y])
                x += 0.184
            y += dy
        inits.append(np.array(pts[:N_CIRCLES]))
        
    # Add randomized dense starts
    for _ in range(4):
        inits.append(rng.uniform(0.12, 0.88, (N_CIRCLES, 2)))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_c = None
    best_r = None
    
    inits = generate_inits(rng)
    
    for cfg in inits:
        # Metaheuristic search on centers
        c, r, s = local_search(cfg, rng)
        if s > best_sum:
            best_sum = s
            best_c = c
            best_r = r
            
        # Phase 3: Joint SLSQP Polish
        x0 = np.concatenate([c.flatten(), r * 0.97])
        bounds_opt = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(1e-6, 0.5)] * N_CIRCLES
        
        try:
            res = minimize(objective_func, x0, args=(N_CIRCLES,), method='SLSQP',
                           bounds=bounds_opt,
                           constraints={'type': 'ineq', 'fun': constraint_func, 'args': (N_CIRCLES,)},
                           options={'maxiter': 8000, 'ftol': 1e-14})
            
            if np.isfinite(res.fun):
                c_pol = np.column_stack((res.x[:N_CIRCLES], res.x[N_CIRCLES:2 * N_CIRCLES]))
                r_pol, s_pol = solve_lp(c_pol)
                if s_pol > best_sum:
                    best_sum = s_pol
                    best_c = c_pol
                    best_r = r_pol
        except Exception:
            pass

    # Phase 4: Strict Safety Scaling
    if best_c is not None:
        scale = 1.0
        for i in range(N_CIRCLES):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
                
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                rs = best_r[i] + best_r[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)
                    
        best_r *= scale * 0.9999999
        best_sum = float(np.sum(best_r))
    else:
        # Strict fallback
        best_c = rng.uniform(0.2, 0.8, (N_CIRCLES, 2))
        best_r = np.full(N_CIRCLES, 0.08)
        best_sum = float(np.sum(best_r))
        
    return best_c, best_r, best_sum
