import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRIU_I)

# Preallocate LP inequality matrix for pairwise constraints
A_PAIRWISE = np.zeros((M_PAIRS, N))
A_PAIRWISE[np.arange(M_PAIRS), TRIU_I] = 1.0
A_PAIRWISE[np.arange(M_PAIRS), TRIU_J] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Boundary limits for each radius
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 0.0)
    bounds = [(0.0, l) for l in lims]
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[TRIU_I, TRIU_J]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_PAIRWISE, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def get_hex_init(r0, pattern):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for idx, cnt in enumerate(pattern):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    # High-quality row distributions summing to 26
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,4,6,4], [5,5,6,5,5], 
        [6,4,6,4,6], [5,6,6,5,4], [6,5,4,6,5], [4,6,6,6,4],
        [7,6,6,7], [6,7,6,7], [5,6,5,6,5,1], [8,5,6,7]
    ]
    
    for pat in patterns:
        if sum(pat) != N: continue
        for r0 in [0.095, 0.105]:
            pts = get_hex_init(r0, pat)
            # Normalize to fit comfortably in [0.05, 0.95]
            mn = pts.min(axis=0)
            mx = pts.max(axis=0)
            span = mx - mn
            if span[0] > 1e-4 and span[1] > 1e-4:
                pts = (pts - mn) / span * 0.85 + 0.075
            inits.append(pts)
            
            # Perturbed versions to break symmetry
            for _ in range(2):
                p = pts + rng.uniform(-0.015, 0.015, pts.shape)
                inits.append(np.clip(p, 0.02, 0.98))
                
    # Random dense starts
    for _ in range(5):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return inits

def hill_climb(centers, rng, max_sweeps=40):
    """Coordinate-wise hill climbing optimized via LP objective."""
    c = centers.copy()
    curr_r, curr_s = solve_lp(c)
    best_s = curr_s
    best_c = c.copy()
    best_r = curr_r.copy()
    
    step = 0.015
    for sweep in range(max_sweeps):
        improved = False
        for i in range(N):
            old_c = c[i].copy()
            best_move = None
            best_move_s = curr_s
            
            # Try random moves in a disk
            n_tries = 8
            angles = rng.uniform(0, 2*np.pi, n_tries)
            dists = rng.uniform(0, step, n_tries)
            moves = np.column_stack([np.cos(angles)*dists, np.sin(angles)*dists])
            
            for mv in moves:
                c[i] = np.clip(old_c + mv, 0.001, 0.999)
                _, s_try = solve_lp(c)
                if s_try > best_move_s + 1e-7:
                    best_move_s = s_try
                    best_move = c[i].copy()
            c[i] = old_c  # Revert immediately
            
            if best_move is not None:
                c[i] = best_move
                curr_r, curr_s = solve_lp(c)
                if curr_s > best_s + 1e-7:
                    best_s = curr_s
                    best_c = c.copy()
                    best_r = curr_r.copy()
                    improved = True
        if not improved:
            break
        step *= 0.94
            
    return best_c, best_r, best_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    inits = generate_inits(rng)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Multi-start Hill Climbing
    for cfg in inits:
        c, r, s = hill_climb(cfg, rng, max_sweeps=35)
        if s > best_sum:
            best_sum = s
            best_centers = c.copy()
            best_radii = r.copy()
            
    # Phase 2: Joint SLSQP Polish
    x0 = np.zeros(3*N)
    x0[:N] = best_centers[:, 0]
    x0[N:2*N] = best_centers[:, 1]
    x0[2*N:] = best_radii * 0.98
    
    bounds_opt = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    
    def obj_joint(v):
        return -np.sum(v[2*N:])
        
    def cons_joint(v):
        cx = v[:N]
        cy = v[N:2*N]
        r = v[2*N:]
        bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
        dx = cx[TRIU_I] - cx[TRIU_J]
        dy = cy[TRIU_I] - cy[TRIU_J]
        dr = r[TRIU_I] + r[TRIU_J]
        pc = dx**2 + dy**2 - dr**2
        return np.concatenate([bc, pc])
        
    try:
        res = minimize(obj_joint, x0, method='SLSQP', bounds=bounds_opt,
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': 4000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            c_pol = np.column_stack((res.x[:N], res.x[N:2*N]))
            r_pol, s_pol = solve_lp(c_pol)
            if s_pol > best_sum:
                best_sum = s_pol
                best_centers = c_pol.copy()
                best_radii = r_pol.copy()
    except Exception:
        pass
        
    # Phase 3: Final aggressive hill climb with controlled restarts
    cur_c = best_centers.copy()
    cur_r = best_radii.copy()
    cur_s = best_sum
    
    for restart in range(3):
        cur_c, cur_r, cur_s = hill_climb(cur_c, rng, max_sweeps=50)
        if cur_s > best_sum:
            best_sum = cur_s
            best_centers = cur_c.copy()
            best_radii = cur_r.copy()
        else:
            # Perturb to escape local minima
            cur_c += rng.uniform(-0.004, 0.004, cur_c.shape)
            cur_c = np.clip(cur_c, 0.02, 0.98)
            
    # Phase 4: Strict numerical safety scaling
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i,0], best_centers[i,1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    diff = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_pair = best_radii[:, np.newaxis] + best_radii[np.newaxis, :]
    valid_pairs = r_pair > 1e-12
    if np.any(valid_pairs):
        scale = min(scale, np.min(dists[valid_pairs] / r_pair[valid_pairs]))
        
    best_radii *= scale * 0.999999
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
