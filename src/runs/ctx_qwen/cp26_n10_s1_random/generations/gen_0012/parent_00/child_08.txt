import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIU_I, TRIU_J = np.triu_indices(N_CIRCLES, k=1)

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    m = len(TRIU_I)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), TRIU_I] = 1.0
    A_ub[np.arange(m), TRIU_J] = 1.0
    b_ub = dists[TRIU_I, TRIU_J]
    
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 1e-9

def obj_lp_centers(v):
    """Objective for center optimization: minimize negative LP sum of radii."""
    c = v.reshape(N_CIRCLES, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    _, s = solve_lp_radii(c)
    return -s

def joint_obj(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N_CIRCLES:])

def joint_cons(v):
    """Inequality constraints >= 0 for valid joint packing."""
    x = v[:N_CIRCLES]
    y = v[N_CIRCLES:2*N_CIRCLES]
    r = np.maximum(v[2*N_CIRCLES:], 1e-9)
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[TRIU_I] - x[TRIU_J]
    dy = y[TRIU_I] - y[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [6,6,4,6,4], 
        [5,5,6,5,5], [6,4,6,4,6], [5,6,6,5,4], [6,5,4,6,5],
        [5,5,5,5,6], [6,6,6,4,4], [4,5,6,5,6], [5,4,6,5,6],
        [7,6,6,7], [6,7,6,7], [5,6,5,6,5,1], [5,5,5,6,5],
        [8,5,6,7], [7,7,6,6], [9,5,6,6], [8,6,6,6]
    ]
    for pat in patterns:
        if sum(pat) != N_CIRCLES: continue
        pts = []
        r0 = 0.098
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx%2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N_CIRCLES: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:N_CIRCLES])
        inits.append(pts)
        for _ in range(3):
            p = pts + rng.uniform(-0.025, 0.025, pts.shape)
            inits.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(10):
        inits.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)))
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = generate_inits(rng)
    bounds_centers = [(0.0, 1.0)] * (2 * N_CIRCLES)
    
    for cfg in inits:
        cfg = np.clip(cfg, 0.01, 0.99)
        
        # Phase 1: L-BFGS-B on centers maximizing LP sum
        try:
            res1 = minimize(obj_lp_centers, cfg.flatten(), method='L-BFGS-B', 
                            bounds=bounds_centers, options={'maxiter': 1000, 'ftol': 1e-13, 'disp': False})
            c1 = res1.x.reshape(N_CIRCLES, 2)
        except Exception:
            c1 = cfg
            
        # Phase 2: Coordinate descent hill-climbing
        curr_c = c1.copy()
        _, curr_s = solve_lp_radii(curr_c)
        step = 0.018
        for _ in range(250):
            improved = False
            for i in range(N_CIRCLES):
                old = curr_c[i].copy()
                best_move = None
                best_s = curr_s
                for _ in range(8):
                    move = rng.uniform(-step, step, 2)
                    nc = np.clip(curr_c[i] + move, 1e-4, 0.999)
                    curr_c[i] = nc
                    _, s_try = solve_lp_radii(curr_c)
                    if s_try > best_s + 1e-9:
                        best_s = s_try
                        best_move = nc.copy()
                    curr_c[i] = old
                if best_move is not None:
                    curr_c[i] = best_move
                    curr_s = best_s
                    improved = True
            if not improved:
                break
            step *= 0.985
            
        # Phase 3: Joint SLSQP polish
        r_curr, _ = solve_lp_radii(curr_c)
        x0 = np.concatenate([curr_c.flatten(), r_curr * 0.98])
        bounds_joint = [(0.0, 1.0)] * (2*N_CIRCLES) + [(1e-6, 0.5)] * N_CIRCLES
        try:
            res2 = minimize(joint_obj, x0, method='SLSQP', bounds=bounds_joint,
                            constraints={'type': 'ineq', 'fun': joint_cons},
                            options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if np.isfinite(res2.fun):
                c2 = np.column_stack((res2.x[:N_CIRCLES], res2.x[N_CIRCLES:2*N_CIRCLES]))
                r2, s2 = solve_lp_radii(c2)
                if s2 > curr_s:
                    curr_c = c2
                    curr_s = s2
        except Exception:
            pass
            
        if curr_s > best_sum:
            best_sum = curr_s
            best_centers = curr_c.copy()
            best_radii, _ = solve_lp_radii(best_centers)
            
    # Phase 4: Strict Safety Scaling to guarantee numerical validity
    if best_centers is not None:
        scale = 1.0
        for i in range(N_CIRCLES):
            x, y, r = best_centers[i,0], best_centers[i,1], best_radii[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
        for i in range(N_CIRCLES):
            for j in range(i+1, N_CIRCLES):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-12:
                    scale = min(scale, d/rs)
        best_radii *= scale * 0.999999
        best_sum = float(np.sum(best_radii))
    else:
        best_centers = np.random.uniform(0.2, 0.8, (N_CIRCLES, 2))
        best_radii = np.full(N_CIRCLES, 0.05)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
