from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M = len(TRIU_I)
A_ub_struct = np.zeros((M, N))
A_ub_struct[np.arange(M), TRIU_I] = 1.0
A_ub_struct[np.arange(M), TRIU_J] = 1.0

def solve_lp_and_grad(centers):
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    wall_lim = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                          np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(w, 1e-9)) for w in wall_lim]
    b_ub = dists[TRIU_I, TRIU_J]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_struct, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None
    except Exception:
        return None, None, None
        
    r = res.x
    s = -res.fun
    grad = np.zeros((n, 2))
    
    try:
        lams = np.asarray(res.ineqlin.marginals)
        active_lams = np.maximum(lams, 0.0)
        mask = active_lams > 1e-6
        if np.any(mask):
            k = np.where(mask)[0]
            ii = TRIU_I[k]
            jj = TRIU_J[k]
            d = dists[ii, jj]
            d = np.where(d < 1e-9, 1.0, d)
            diff_k = centers[ii] - centers[jj]
            factors = active_lams[k][:, np.newaxis] / d[:, np.newaxis]
            np.add.at(grad, ii, diff_k * factors)
            np.add.at(grad, jj, -diff_k * factors)
    except Exception:
        pass
        
    return r, s, grad

def objective_joint(v, n):
    return -np.sum(v[2*n:])

def constraints_joint(v, n):
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    dx = x[TRIU_I] - x[TRIU_J]
    dy = y[TRIU_I] - y[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)
