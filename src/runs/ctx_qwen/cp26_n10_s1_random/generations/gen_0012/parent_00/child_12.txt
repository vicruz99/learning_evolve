

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRIU_I)

# Precompute constant A_ub matrix for LP
A_PAIRS = np.zeros((M_PAIRS, N))
A_PAIRS[np.arange(M_PAIRS), TRIU_I] = 1.0
A_PAIRS[np.arange(M_PAIRS), TRIU_J] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, marginals."""
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[TRIU_I, TRIU_J]
    
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    res = linprog(-np.ones(N), A_ub=A_PAIRS, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun, res.ineqlin.marginals
    return np.full(N, 1e-9), 0.0, np.zeros(M_PAIRS)

def get_obj(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def get_constraints(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    dx = x[triu_i] - x[triu_j]
    dy = y[triu_i] - y[triu_j]
    dr = r[triu_i] + r[triu_j]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def rotate_points(pts, angle):
    """Rotates point cloud around its centroid."""
    cos_a, sin_a = np.cos(angle), np.sin(angle)
    center = pts.mean(axis=0)
    pts_shifted = pts - center
    rot_pts = np.column_stack((pts_shifted[:, 0]*cos_a - pts_shifted[:, 1]*sin_a,
                               pts_shifted[:, 0]*sin_a + pts_shifted[:, 1]*cos_a))
    rot_pts += center
    return rot_pts

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [6,6,4,6,4], 
        [5,5,6,5,5], [6,4,6,4,6], [5,6,6,5,4], [6,5,4,6,5],
        [5,5,5,5,6], [6,6,6,4,4], [4,5,6,5,6], [5,4,6,5,6],
        [7,6,6,7], [6,7,6,7], [5,6,5,6,5,1], [8,5,6,7], [7,7,6,6]
    ]
    
    for pat in patterns:
        if sum(pat) != N: continue
        pts = []
        r0 = 0.098
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx%2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:N])
        
        # Normalize to fit in [0,1] with padding
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = np.where(mx - mn > 1e-4, mx - mn, 1.0)
        pts_norm = (pts - mn) / span * 0.75 + 0.125
        inits.append(pts_norm)
        
        # Perturbed versions
        for _ in range(3):
            p = pts_norm + rng.uniform(-0.02, 0.02, pts_norm.shape)
            inits.append(np.clip(p, 0.05, 0.95))
            
        # Rotated versions
        for ang in [np.pi/12, np.pi/6, np.pi/4]:
            rot = rotate_points(pts_norm.copy(), ang)
            mn_r = rot.min(axis=0)
            mx_r = rot.max(axis=0)
            span_r = np.where(mx_r - mn_r > 1e-4, mx_r - mn_r, 1.0)
            rot_norm = (rot - mn_r) / span_r * 0.75 + 0.125
            inits.append(np.clip(rot_norm, 0.05, 0.95))
            
    # Random starts
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_vars = [(0.0, 1.0)]*(2*N) + [(1e-6, 0.5)]*N
    cons = {'type': 'ineq', 'fun': get_constraints, 'args': (N, TRIU_I, TRIU_J)}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = generate_inits(rng)
    
    # Phase 1: Multi-start SLSQP Joint Optimization
    for cfg in inits:
        x0 = np.zeros(3*N)
        x0[:N] = cfg[:,0]
        x0[N:2*N] = cfg[:,1]
        
        # Estimate initial radii safely
        r_init = np.full(N, 0.08)
        for i in range(N):
            dw = min(cfg[i,0], 1.0-cfg[i,0], cfg[i,1], 1.0-cfg[i,1])
            dn = np.min(np.sqrt(np.sum((cfg - cfg[i])**2, axis=1)) * (np.arange(N) != i))
            r_init[i] = min(dw, dn/2.0) * 0.8
        x0[2*N:] = r_init
        
        try:
            res = minimize(get_obj, x0, args=(N,), method='SLSQP', bounds=bounds_vars,
                           constraints=cons, options={'maxiter': 12000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_mat = np.column_stack((res.x[:N], res.x[N:2*N]))
                r_lp, s_lp, _ = solve_lp(c_mat)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c_mat.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass

    if best_centers is None:
        best_centers = inits[0]
        best_radii, best_sum, _ = solve_lp(best_centers)

    # Phase 2: Gradient Ascent on Centers using LP Duals
    curr_c = best_centers.copy()
    curr_r, curr_s, curr_marg = solve_lp(curr_c)
    
    step = 0.015
    for sweep in range(20):
        grad = np.zeros((N, 2))
        diff = curr_c[TRIU_I] - curr_c[TRIU_J]
        dists = np.linalg.norm(diff, axis=1)
        safe_d = np.where(dists > 1e-8, dists, 1.0)
        weights = curr_marg / safe_d
        
        np.add.at(grad, TRIU_I, diff * weights[:, None])
        np.add.at(grad, TRIU_J, -diff * weights[:, None])
        
        # Wall repulsion to prevent sticking to boundaries
        wall_rep = np.zeros((N, 2))
        wall_rep[:, 0] += np.clip(0.08 - curr_c[:, 0], 0, None) * 5.0
        wall_rep[:, 0] -= np.clip(0.08 - (1.0 - curr_c[:, 0]), 0, None) * 5.0
        wall_rep[:, 1] += np.clip(0.08 - curr_c[:, 1], 0, None) * 5.0
        wall_rep[:, 1] -= np.clip(0.08 - (1.0 - curr_c[:, 1]), 0, None) * 5.0
        grad += wall_rep
        
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-6:
            grad_dir = grad / g_norm
            new_c = curr_c + step * grad_dir
            new_c = np.clip(new_c, 0.02, 0.98)
            r_new, s_new, m_new = solve_lp(new_c)
            if s_new > curr_s + 1e-8:
                curr_c, curr_r, curr_s, curr_marg = new_c, r_new, s_new, m_new
                step = min(step * 1.05, 0.04)
            else:
                step *= 0.85
        else:
            break
            
    best_centers = curr_c
    best_radii = curr_r
    best_sum = curr_s

    # Phase 3: Coordinate-wise Random Hill Climbing
    step = 0.01
    for sweep in range(10):
        improved = False
        for i in range(N):
            old = curr_c[i].copy()
            best_move_s = curr_s
            best_move = None
            for _ in range(12):
                move = rng.uniform(-step, step, 2)
                curr_c[i] = np.clip(old + move, 0.02, 0.98)
                r_try, s_try, _ = solve_lp(curr_c)
                if s_try > best_move_s:
                    best_move_s = s_try
                    best_move = curr_c[i].copy()
                else:
                    curr_c[i] = old
            if best_move is not None:
                curr_c[i] = best_move
                curr_r, curr_s, curr_marg = solve_lp(curr_c)
                if curr_s > best_sum + 1e-8:
                    best_sum = curr_s
                    best_centers = curr_c.copy()
                    best_radii = curr_r.copy()
                    improved = True
            else:
                curr_c[i] = old
        if not improved:
            break
        step *= 0.7
        
    # Phase 4: Final SLSQP Polish
    x0 = np.zeros(3*N)
    x0[:N] = best_centers[:,0]
    x0[N:2*N] = best_centers[:,1]
    x0[2*N:] = best_radii * 0.98
    
    try:
        res = minimize(get_obj, x0, args=(N,), method='SLSQP', bounds=bounds_vars,
                       constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            c_mat = np.column_stack((res.x[:N], res.x[N:2*N]))
            r_lp, s_lp, _ = solve_lp(c_mat)
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = c_mat.copy()
                best_radii = r_lp.copy()
    except Exception:
        pass

    # Safety scaling
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i,0], best_centers[i,1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
```
