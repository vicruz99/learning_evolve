import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N_CIRCLES = 26
TRIU_I, TRIU_J = np.triu_indices(N_CIRCLES, k=1)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]),
                      np.minimum(centers[:,1], 1.0-centers[:,1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    b_ub = dists[TRIU_I, TRIU_J]
    
    m = len(TRIU_I)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), TRIU_I] = 1.0
    A_ub[np.arange(m), TRIU_J] = 1.0
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0, None

def get_lp_gradient(centers, res):
    """Computes gradient of sum(radii) w.r.t. centers using LP dual variables."""
    if res is None or not hasattr(res, 'ineqlin'):
        return np.zeros((N_CIRCLES, 2))
    try:
        lams = np.abs(res.ineqlin.marginals)
    except Exception:
        return np.zeros((N_CIRCLES, 2))
        
    valid = lams > 1e-7
    if not np.any(valid):
        return np.zeros((N_CIRCLES, 2))
        
    idx_k = np.where(valid)[0]
    ii = TRIU_I[idx_k]
    jj = TRIU_J[idx_k]
    diff_k = centers[ii] - centers[jj]
    dists_k = np.sqrt(np.sum(diff_k**2, axis=1))
    dists_k = np.maximum(dists_k, 1e-9)
    
    grad = np.zeros((N_CIRCLES, 2))
    factors = (lams[valid] / dists_k)[:, np.newaxis]
    np.add.at(grad, ii, diff_k * factors)
    np.add.at(grad, jj, -diff_k * factors)
    return grad

def hex_init(n, rows, r0, rng):
    """Generates hexagonal lattice initial positions."""
    pts = []
    y = r0
    for i, cnt in enumerate(rows):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:n])

def obj_joint(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def con_joint(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
    
    dx = cx[TRIU_I] - cx[TRIU_J]
    dy = cy[TRIU_I] - cy[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    # Diverse row patterns for hexagonal packing
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,4,6,4], [4,6,5,6,5], 
        [5,5,6,5,5], [6,4,6,4,6], [5,6,6,5,4], [6,5,4,6,5],
        [7,6,6,7], [6,7,6,7], [5,6,5,6,5,1], [5,5,5,6,5],
        [8,5,6,7], [7,7,6,6], [9,5,6,6], [8,6,6,6], [6,6,6,5,3]
    ]
    
    configs = []
    for pat in patterns:
        if sum(pat) != N_CIRCLES: continue
        for r0 in [0.085, 0.090, 0.095, 0.100]:
            c = hex_init(N_CIRCLES, pat, r0, rng)
            mn = c.min(axis=0)
            mx = c.max(axis=0)
            span = mx - mn

                c = (c - mn) / span * 0.7 + 0.15
            configs.append(c.copy())
            for _ in range(2):
                p = c + rng.uniform(-0.01, 0.01, c.shape)
                configs.append(np.clip(p, 0.05, 0.95))
                
    for _ in range(8):
        configs.append(rng.uniform(0.1, 0.9, (N_CIRCLES, 2)))

            if np.isfinite(res.fun):
                c_opt = res.x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)

                    best_sum = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            pass

    curr_c = best_c.copy()
    curr_r, curr_s, res_lp = solve_lp(curr_c)
    best_sum = curr_s
    best_c = curr_c.copy()
    best_r = curr_r.copy()

                    best_c = curr_c.copy()
                    best_r = curr_r.copy()

                curr_c[i] = old

        if gn > 1e-8:
            c_new = curr_c + step_g * (g / gn)

                curr_r = r_new
                curr_s = s_new

                    best_sum = curr_s
                    best_c = curr_c.copy()
                    best_r = curr_r.copy()

            step_g *= 0.95

        pass

    # Phase 5: Strict Safety Scaling
    scale = 1.0
    for i in range(N_CIRCLES):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

    best_r *= scale * 0.999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
