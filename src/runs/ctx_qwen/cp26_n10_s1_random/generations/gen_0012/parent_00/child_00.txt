import numpy as np
from scipy.optimize import linprog, minimize
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRIU_I)

# Precompute constant LP matrix for pairwise constraints: r_i + r_j <= d_ij
A_UUB = np.zeros((M_PAIRS, N))
A_UUB[np.arange(M_PAIRS), TRIU_I] = 1.0
A_UUB[np.arange(M_PAIRS), TRIU_J] = 1.0

def solve_lp_detailed(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, and dual marginals."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[TRIU_I, TRIU_J]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_UUB, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            r = res.x
            s = -res.fun
            try:
                lams = np.abs(res.ineqlin.marginals)
            except Exception:
                lams = np.zeros(M_PAIRS)
            return r, s, lams
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0, np.zeros(M_PAIRS)

def gradient_ascent(centers, max_iter=400, init_step=0.004):
    """Optimizes centers to maximize the LP-computed sum of radii using exact dual gradients."""
    c = centers.copy()
    r, best_s, lams = solve_lp_detailed(c)
    step = init_step
    
    for it in range(max_iter):
        grad = np.zeros_like(c)
        active = lams > 1e-6
        if not np.any(active):
            break
            
        ii = TRIU_I[active]
        jj = TRIU_J[active]
        lam_v = lams[active]
        
        diff = c[ii] - c[jj]
        dists = np.linalg.norm(diff, axis=1)
        inv_d = np.where(dists > 1e-8, 1.0 / dists, 0.0)
        factors = lam_v * inv_d
        
        np.add.at(grad, ii, diff * factors[:, None])
        np.add.at(grad, jj, -diff * factors[:, None])
        
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-8:
            break
            
        # Propose move
        c_new = c + step * grad
        c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
        
        r_new, s_new, lams_new = solve_lp_detailed(c_new)
        
        if s_new > best_s + 1e-9:
            c = c_new
            best_s = s_new
            r = r_new
            lams = lams_new
            step = min(step * 1.08, 0.025)
        else:
            step *= 0.75
            
        # Occasional micro-shake to escape flat plateaus
        if it > 0 and it % 30 == 0 and step < 1e-4:
            c += np.random.uniform(-0.002, 0.002, c.shape)
            c = np.clip(c, 1e-4, 1.0 - 1e-4)
            r, best_s, lams = solve_lp_detailed(c)
            step = 0.002
            
    return c, r, best_s

def generate_inits(rng):
    """Generates diverse initial center configurations preserving aspect ratios."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [6,6,4,6,4],
        [5,5,6,5,5], [6,4,6,4,6], [5,6,6,5,4], [6,5,4,6,5],
        [5,5,5,5,6], [6,6,6,4,4], [4,5,6,5,6], [5,4,6,5,6],
        [7,6,6,7], [6,7,6,7], [5,6,5,6,5,1], [5,5,5,6,5],
        [8,5,6,7], [7,7,6,6], [9,5,6,6], [8,6,6,6]
    ]
    
    for pat in patterns:
        if sum(pat) != N: continue
        pts = []
        r0 = 0.098
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:N])
        
        # Scale uniformly to fit well within [0,1]x[0,1] while preserving aspect ratio
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        mx_span = max(span[0], span[1])
        if mx_span > 1e-4:
            pts = (pts - mn) / mx_span * 0.75 + 0.125
        else:
            pts = np.full((N, 2), 0.5)
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
        # Perturbed versions
        for _ in range(2):
            p = pts + rng.uniform(-0.015, 0.015, pts.shape)
            inits.append(np.clip(p, 0.05, 0.95))
            
    # Pure random starts
    for _ in range(6):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return inits

def joint_objective(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def joint_constraints(v, n):
    """Inequality constraints >= 0 for valid packing."""
    x = v[:n]
    y = v[n:2 * n]
    r = v[2 * n:]
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    dx = x[TRIU_I] - x[TRIU_J]
    dy = y[TRIU_I] - y[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = generate_inits(rng)
    
    # Phase 1: Gradient Ascent on Centers from diverse starts
    for cfg in inits:
        c_opt, r_opt, s_opt = gradient_ascent(cfg.copy())
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    # Phase 2: Shake & Refine (Local search with re-optimization)
    curr_c = best_centers.copy()
    curr_r = best_radii.copy()
    curr_s = best_sum
    
    for sweep in range(4):
        temp = 0.018 * (0.65 ** sweep)
        improved = False
        for i in range(N):
            best_move = None
            best_move_s = curr_s
            old = curr_c[i].copy()
            
            for _ in range(8):
                move = rng.uniform(-temp, temp, 2)
                new_c = np.clip(curr_c[i] + move, 1e-4, 0.999)
                curr_c[i] = new_c
                
                # Quick local refinement from perturbed position
                c_tmp, r_tmp, s_tmp = gradient_ascent(curr_c.copy(), max_iter=60, init_step=0.002)
                
                if s_tmp > best_move_s:
                    best_move_s = s_tmp
                    best_move = c_tmp.copy()
                else:
                    curr_c[i] = old
                    
            if best_move is not None:
                curr_c = best_move.copy()
                curr_r, curr_s, _ = solve_lp_detailed(curr_c)
                if curr_s > best_sum + 1e-8:
                    best_sum = curr_s
                    best_centers = curr_c.copy()
                    best_radii = curr_r.copy()
                    improved = True
            else:
                curr_c[i] = old
        if not improved:
            break
            
    # Phase 3: Joint SLSQP Polish
    x0 = np.zeros(3 * N)
    x0[:N] = best_centers[:, 0]
    x0[N:2 * N] = best_centers[:, 1]
    x0[2 * N:] = best_radii * 0.975
    
    bounds_vars = [(0.0, 1.0)] * (2 * N) + [(1e-7, 0.5)] * N
    try:
        res = minimize(joint_objective, x0, args=(N,), method='SLSQP', bounds=bounds_vars,
                       constraints={'type': 'ineq', 'fun': joint_constraints, 'args': (N,)},
                       options={'maxiter': 12000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            c_mat = np.column_stack((res.x[:N], res.x[N:2 * N]))
            r_lp, s_lp, _ = solve_lp_detailed(c_mat)
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = c_mat.copy()
                best_radii = r_lp.copy()
    except Exception:
        pass
        
    # Phase 4: Strict Safety Scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
