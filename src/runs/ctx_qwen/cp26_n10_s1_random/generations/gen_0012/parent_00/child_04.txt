import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRIU_I)

# Precompute the constant constraint matrix for the LP radius solver
A_ub_lp = np.zeros((M_PAIRS, N))
A_ub_lp[np.arange(M_PAIRS), TRIU_I] = 1.0
A_ub_lp[np.arange(M_PAIRS), TRIU_J] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[TRIU_I, TRIU_J]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_lp, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-9), 0.0

def force_sim(centers, steps=500):
    """Force-directed simulation to tightly pack circles."""
    c = centers.copy()
    n = len(c)
    vel = np.zeros_like(c)
    dt = 0.004
    damp = 0.85
    
    for step in range(steps):
        target = 0.04 + 0.065 * (step / steps)
        
        diff = c[:, None, :] - c[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dist, 1e9)
        
        overlap = np.maximum(0.0, 2.0 * target - dist)
        strength = overlap * 100.0 / (dist + 1e-12)
        f_pair = np.sum(diff * strength[:, :, None], axis=1)
        
        f_wall = np.zeros_like(c)
        f_wall[:, 0] += np.clip(target - c[:, 0], 0, None) * 200.0
        f_wall[:, 0] -= np.clip(c[:, 0] - (1.0 - target), 0, None) * 200.0
        f_wall[:, 1] += np.clip(target - c[:, 1], 0, None) * 200.0
        f_wall[:, 1] -= np.clip(c[:, 1] - (1.0 - target), 0, None) * 200.0
        
        vel = vel * damp + (f_pair + f_wall) * dt
        c += vel
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
    return c

def slsqp_opt(c0, r0):
    """Joint SLSQP optimization of centers and radii."""
    n = len(r0)
    x0 = np.concatenate([c0[:, 0], c0[:, 1], r0])
    bounds = [(0.0, 1.0)]*(2*n) + [(1e-7, 0.5)]*n
    
    def obj(v):
        return -np.sum(v[2*n:])
        
    def cons(v):
        cx, cy, r = v[:n], v[n:2*n], v[2*n:]
        c_list = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
        dx = cx[TRIU_I] - cx[TRIU_J]
        dy = cy[TRIU_I] - cy[TRIU_J]
        dr = r[TRIU_I] + r[TRIU_J]
        c_list.append(dx**2 + dy**2 - dr**2)
        return np.concatenate(c_list)
        
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 5000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            c_out = np.column_stack((res.x[:n], res.x[n:2*n]))
            r_out = res.x[2*n:]
            return c_out, r_out
    except Exception:
        pass
    return None, None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = -1.0

    best_r = None

        [7,6,6,7], [6,7,6,7], [5,6,5,6,5,1], [5,5,5,6,5],
        [8,5,6,7], [7,7,6,6], [9,5,6,6], [8,6,6,6]

            best_r = r_pol_lp.copy()

    for i in range(N):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

        for j in range(i+1, N):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
