import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26
TRIU_I, TRIU_J = np.triu_indices(N_CIRCLES, k=1)
M_PAIRS = len(TRIU_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]),
                      np.minimum(centers[:,1], 1.0 - centers[:,1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    A = np.zeros((M_PAIRS, n))
    A[np.arange(M_PAIRS), TRIU_I] = 1.0
    A[np.arange(M_PAIRS), TRIU_J] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b = dists[TRIU_I, TRIU_J]
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res.ineqlin.marginals
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0, None

def compute_gradient(centers, duals, dists):
    """Computes gradient of sum of radii w.r.t centers using LP duals."""
    grad = np.zeros_like(centers)
    if duals is None:
        return grad
        
    # duals are <= 0 for minimization, so -duals >= 0 represents active constraint strength
    weights = -duals
    active = weights > 1e-6
    
    if not np.any(active):
        return grad
        
    w = weights[active]
    ii = TRIU_I[active]
    jj = TRIU_J[active]
    d = dists[ii, jj]
    d = np.where(d > 1e-8, d, 1.0)
    
    direction = (centers[ii] - centers[jj]) / d[:, np.newaxis]
    np.add.at(grad, ii, direction * w[:, np.newaxis])
    np.add.at(grad, jj, -direction * w[:, np.newaxis])
    
    return grad

def get_hex_init(r0, rng):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N_CIRCLES:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x + r0 <= 1.0 and len(pts) < N_CIRCLES:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
        row += 1
    return np.array(pts[:N_CIRCLES])

                x += 2.0 * r0

            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            configs.append(np.clip(p, 0.02, 0.98))

        dy = y[TRIU_I] - y[TRIU_J]
        dr = r[TRIU_I] + r[TRIU_J]
        pc = dx**2 + dy**2 - dr**2

                best_centers = c_pol

        pass

        scale = 1.0

            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

        np.fill_diagonal(dists, np.inf)
        rs = best_radii[:, np.newaxis] + best_radii[np.newaxis, :]
        
        pair_scales = dists[TRIU_I, TRIU_J] / rs[TRIU_I, TRIU_J]
        scale = min(scale, np.min(pair_scales))
        
        best_radii *= scale * 0.9999995
        best_sum = float(np.sum(best_radii))
    else:
        # Fallback (should not be reached)
        best_centers = get_hex_init(0.09, rng)
        best_radii, best_sum, _ = solve_lp(best_centers)
        
    return best_centers, best_radii, best_sum
