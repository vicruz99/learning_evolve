TRIU_I, TRIU_J = np.triu_indices(N, k=1)

A_LIN = np.zeros((M, N))

A_LIN[np.arange(M), TRIU_J] = 1.0

    n = centers.shape[0]

    except Exception:
        pass
    return np.full(n, 1e-7), 1e-7

    c = c_flat.reshape(N, 2)
