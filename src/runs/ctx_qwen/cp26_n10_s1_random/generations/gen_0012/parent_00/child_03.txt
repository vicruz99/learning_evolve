import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRIU_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[TRIU_I, TRIU_J]
    
    A_ub = np.zeros((M_PAIRS, n))
    A_ub[np.arange(M_PAIRS), TRIU_I] = 1.0
    A_ub[np.arange(M_PAIRS), TRIU_J] = 1.0
    
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-7), 0.0

def joint_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def joint_cons(v):
    """Inequality constraints >= 0 for valid packing."""
    cx, cy, r = v[:N], v[N:2*N], v[2*N:]
    c = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
    dx = cx[TRIU_I] - cx[TRIU_J]
    dy = cy[TRIU_I] - cy[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def hex_init(rows, r0):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for i, cnt in enumerate(rows):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = -1.0
    best_c = None
    best_r = None

        [4,5,6,5,6], [5,4,6,5,6], [5,5,5,5,6], [6,6,6,4,4]
    ]
    
    configs = []
    for pat in patterns:
        if sum(pat) != N: continue
        for r0 in [0.085, 0.092, 0.098]:
            c = hex_init(pat, r0)
            configs.append(c)
            for _ in range(4):
                cfg = c + rng.uniform(-0.025, 0.025, c.shape)
                configs.append(np.clip(cfg, 0.02, 0.98))
                
    for _ in range(12):
        configs.append(rng.uniform(0.12, 0.88, (N, 2)))
        
    bounds_vars = [(0.0, 1.0)]*(2*N) + [(1e-6, 0.5)]*N
    
    # Phase 1: SLSQP Multi-start Optimization
    for cfg in configs:

        try:
            res = minimize(joint_obj, x0, method='SLSQP', bounds=bounds_vars,
                           constraints={'type': 'ineq', 'fun': joint_cons},
                           options={'maxiter': 6000, 'ftol': 1e-13})
            if np.isfinite(res.fun):

                    best_c = c_opt.copy()
                    best_r = r_opt.copy()

    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
