

<strategy>
The strategy focuses on a hybrid approach combining force-directed relaxation, exact linear programming (LP) for radius optimization, and local hill-climbing for center refinement.

1. **Force-Directed Relaxation**: Start with diverse initial configurations (hexagonal lattices with varying densities and random placements). Run a physics-based simulation where circles grow uniformly over time and repel each other when they overlap. This naturally drives the system towards a dense, valid packing configuration.
2. **Exact LP Radius Calculation**: For any fixed set of centers, the problem of maximizing the sum of radii subject to non-overlap and boundary constraints is a Linear Program. Using `scipy.optimize.linprog` with the `highs` solver gives the exact optimal radii for those centers very quickly.
3. **Local Hill-Climbing on Centers**: Force simulation can get stuck in local minima. We perform iterative perturbations of circle centers. After each perturbation, we re-evaluate the maximal sum of radii via LP. If the sum increases, we accept the move. This effectively climbs the objective landscape in the space of centers.
4. **Multi-Restart & Escaping Local Minima**: We run the process from multiple starts. If the local search stagnates, we inject random noise to escape plateaus.
5. **Strict Numerical Safety**: Finally, we compute a global scaling factor to ensure all constraints are satisfied within the validator's $10^{-12}$ tolerance, guaranteeing a valid submission.

This method leverages the global exploration of force dynamics, the exactness of LP, and the precision of local search to push the sum of radii beyond the current best.

import numpy as np
from scipy.optimize import linprog

N_CIRCLES = 26

# Precompute pairwise indices for efficiency
TRIU_I, TRIU_J = np.triu_indices(N_CIRCLES, k=1)

def compute_max_radii_lp(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns (radii, sum_radii) or (zeros, 0) on failure.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Bounds from walls: r <= min(x, 1-x, y, 1-y)
    wall_lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                           np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-12)) for lim in wall_lims]
    
    # Pairwise distances
    dx = centers[TRIU_I, 0] - centers[TRIU_J, 0]
    dy = centers[TRIU_I, 1] - centers[TRIU_J, 1]
    dists = np.hypot(dx, dy)
    
    m = len(TRIU_I)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), TRIU_I] = 1.0
    A_ub[np.arange(m), TRIU_J] = 1.0
    b_ub = dists
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-9), 1e-9

def force_simulate(centers, radii, steps=2500, dt=0.008, damp=0.85):
    """
    Vectorized force-directed simulation to pack circles densely.
    Grows radii slowly while resolving overlaps via repulsion.
    """
    c = centers.copy()
    r = radii.copy()
    n = len(r)
    
    for _ in range(steps):
        r *= 1.00015  # Gradual growth
        
        # Pairwise repulsion
        diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dist, 1e9)
        
        overlap = np.maximum(0.0, r[:, np.newaxis] + r[np.newaxis, :] - dist)
        rep_mag = overlap * 200.0 / (dist + 1e-9)
        forces = np.sum(diff * rep_mag[:, :, np.newaxis], axis=1)
        
        # Wall repulsion
        forces[:, 0] += np.clip(r - c[:, 0], -1, 0) * 300.0
        forces[:, 0] -= np.clip(c[:, 0] + r - 1.0, -1, 0) * 300.0
        forces[:, 1] += np.clip(r - c[:, 1], -1, 0) * 300.0
        forces[:, 1] -= np.clip(c[:, 1] + r - 1.0, -1, 0) * 300.0
        
        # Integrate velocity (implicit velocity storage for simplicity)
        c += forces * dt
        c = np.clip(c, 0.005, 0.995)
        r = np.maximum(r, 1e-5)
        
    return c, r

def generate_initial_configs(rng):
    """Generates a diverse set of initial center configurations."""
    configs = []
    
    # Hexagonal patterns
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,6,4], [6,6,5,5,4],
        [5,5,6,5,5], [6,4,6,4,6], [5,6,6,5,4], [7,6,6,7],
        [6,7,6,7], [5,6,5,6,5,1], [8,5,6,7], [5,5,5,6,5]
    ]
    
    for pat in patterns:
        if sum(pat) != N_CIRCLES: continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N_CIRCLES: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
            
        pts = np.array(pts[:N_CIRCLES])
        configs.append(pts)
        
        # Perturbed versions
        for _ in range(3):
            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    # Random starts
    for _ in range(6):
        configs.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = generate_initial_configs(rng)
    
    # Phase 1: Force Simulation + LP Evaluation from diverse starts
    for cfg in configs:
        c_sim, r_sim = force_simulate(cfg, np.full(N_CIRCLES, 0.04), steps=2000)
        r_lp, s_lp = compute_max_radii_lp(c_sim)
        if s_lp > best_sum:
            best_sum = s_lp
            best_centers = c_sim.copy()
            best_radii = r_lp.copy()
            
    # Phase 2: Local Hill-Climbing on Centers (evaluated via LP)
    curr_c = best_centers.copy()
    curr_r, curr_s = compute_max_radii_lp(curr_c)
    best_sum = curr_s
    best_centers = curr_c.copy()
    best_radii = curr_r.copy()
    
    step_size = 0.015
    for sweep in range(8):
        improved = False
        # Try random perturbations
        for _ in range(1500):
            idx = rng.integers(N_CIRCLES)
            old_pos = curr_c[idx].copy()
            
            # Perturb
            move = rng.uniform(-step_size, step_size, 2)
            new_pos = np.clip(curr_c[idx] + move, 0.01, 0.99)
            curr_c[idx] = new_pos
            
            r_try, s_try = compute_max_radii_lp(curr_c)
            if s_try > curr_s + 1e-7:
                curr_s = s_try
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_centers = curr_c.copy()
                    best_radii = curr_r.copy() # Update radii reference
                    improved = True
                # Note: best_radii will be recomputed from best_centers later
            else:
                curr_c[idx] = old_pos
                
        step_size *= 0.85
        if not improved:
            # Escape local minimum
            curr_c += rng.uniform(-0.005, 0.005, curr_c.shape)
            curr_c = np.clip(curr_c, 0.05, 0.95)
            
    # Re-evaluate LP on best centers found
    best_radii, best_sum = compute_max_radii_lp(best_centers)
    
    # Phase 3: Final Safety Scaling
    scale = 1.0
    for i in range(N_CIRCLES):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
