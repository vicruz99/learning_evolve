import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26
TRIU_I, TRIU_J = np.triu_indices(N_CIRCLES, k=1)
M_PAIRS = len(TRIU_I)
A_PAIRS = np.zeros((M_PAIRS, N_CIRCLES))
A_PAIRS[np.arange(M_PAIRS), TRIU_I] = 1.0
A_PAIRS[np.arange(M_PAIRS), TRIU_J] = 1.0

# Global RNG for reproducibility
RNG = np.random.default_rng(42)

def solve_lp_and_grad(centers):
    """Solves LP to maximize sum of radii and computes gradient w.r.t centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[TRIU_I, TRIU_J]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_PAIRS, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None
    except Exception:
        return None, None, None
        
    r = res.x
    s = -res.fun
    grad = np.zeros((n, 2))
    
    try:
        marg = np.abs(res.ineqlin.marginals)
        active = marg > 1e-6
        if np.any(active):
            lam = marg[active]
            ii = TRIU_I[active]
            jj = TRIU_J[active]
            d = dists[ii, jj]
            d = np.where(d < 1e-12, 1e-12, d)
            
            dx = centers[ii, 0] - centers[jj, 0]
            dy = centers[ii, 1] - centers[jj, 1]
            
            fx = lam * dx / d
            fy = lam * dy / d
            
            np.add.at(grad[:, 0], ii, fx)
            np.add.at(grad[:, 0], jj, -fx)
            np.add.at(grad[:, 1], ii, fy)
            np.add.at(grad[:, 1], jj, -fy)
    except Exception:
        pass
        
    return r, s, grad

def run_gradient_ascent(centers, max_iter=4000):
    """Optimizes centers using LP-gradient ascent with adaptive steps."""
    n = centers.shape[0]
    c = centers.copy()
    r, s, grad = solve_lp_and_grad(c)
    if r is None:
        return c
    best_s = s
    best_c = c.copy()
    
    step = 0.015
    for it in range(max_iter):
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-10:
            break
            
        grad_dir = grad / g_norm
        improved = False
        
        for alpha in [step, step*0.5, step*0.2]:
            c_new = c + alpha * grad_dir
            c_new = np.clip(c_new, 0.02, 0.98)
            r_new, s_new, grad_new = solve_lp_and_grad(c_new)
            if r_new is not None and s_new > best_s + 1e-7:
                c = c_new
                grad = grad_new
                best_s = s_new
                best_c = c.copy()
                improved = True
                step = min(step * 1.05, 0.04)
                break
        if not improved:
            step *= 0.85
            
        # Occasional jump to escape local minima
        if it > 500 and it % 200 == 0:
            idx = RNG.integers(n)
            c_jump = c.copy()
            c_jump[idx] += RNG.uniform(-0.02, 0.02, 2)
            c_jump = np.clip(c_jump, 0.02, 0.98)
            r_j, s_j, g_j = solve_lp_and_grad(c_jump)
            if r_j is not None and s_j > best_s + 1e-7:
                c = c_jump
                grad = g_j
                best_s = s_j
                best_c = c.copy()
                step = 0.015
                
    return best_c

def generate_configs(rng):
    """Generates diverse initial center configurations."""
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [6,6,4,6,4],
        [5,5,6,5,5], [6,4,6,4,6], [5,6,6,5,4], [6,5,4,6,5],
        [5,5,5,5,6], [6,6,6,4,4], [7,5,5,5,4], [5,5,5,6,5],
        [8,6,6,6], [7,7,6,6], [9,5,6,6]
    ]
    
    for pat in patterns:
        if sum(pat) != N_CIRCLES: continue
        pts = []
        r0 = 0.10
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N_CIRCLES: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:N_CIRCLES])
        
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        if span[0] > 1e-4 and span[1] > 1e-4:
            pts = (pts - mn) / span * 0.76 + 0.12
        configs.append(pts)
        for _ in range(2):
            p = pts + rng.uniform(-0.015, 0.015, pts.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(8):
        configs.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)))
        
    return configs

def joint_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N_CIRCLES:])

def joint_cons(v):
    """Inequality constraints >= 0 for valid packing."""
    x = v[:N_CIRCLES]
    y = v[N_CIRCLES:2*N_CIRCLES]
    r = v[2*N_CIRCLES:]
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    dx = x[TRIU_I] - x[TRIU_J]
    dy = y[TRIU_I] - y[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    configs = generate_configs(RNG)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_vars = [(0.0, 1.0)]*(2*N_CIRCLES) + [(1e-6, 0.5)]*N_CIRCLES
    
    for cfg in configs:

        if r_opt is None: continue
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
        # Phase 2: SLSQP Joint Polish
        x0 = np.zeros(3*N_CIRCLES)
        x0[:N_CIRCLES] = c_opt[:, 0]
        x0[N_CIRCLES:2*N_CIRCLES] = c_opt[:, 1]
        x0[2*N_CIRCLES:] = r_opt * 0.98
        
        try:
            res = minimize(joint_obj, x0, method='SLSQP', bounds=bounds_vars,
                           constraints={'type': 'ineq', 'fun': joint_cons},
                           options={'maxiter': 5000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_mat = np.column_stack((res.x[:N_CIRCLES], res.x[N_CIRCLES:2*N_CIRCLES]))
                r_lp, s_lp, _ = solve_lp_and_grad(c_mat)
                if r_lp is not None and s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c_mat.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass

        # Phase 3: Coordinate Hill-Climbing
        curr_c = best_centers.copy()
        curr_r, curr_s, _ = solve_lp_and_grad(curr_c)
        for sweep in range(5):
            step = 0.015 * (0.7 ** sweep)
            for i in range(N_CIRCLES):
                old = curr_c[i].copy()
                best_move = None
                best_move_s = curr_s
                for _ in range(6):
                    move = RNG.uniform(-step, step, 2)
                    new_c = np.clip(curr_c[i] + move, 0.01, 0.99)
                    curr_c[i] = new_c
                    r_try, s_try, _ = solve_lp_and_grad(curr_c)
                    if r_try is not None and s_try > best_move_s:
                        best_move_s = s_try
                        best_move = new_c.copy()
                    else:
                        curr_c[i] = old
                if best_move is not None:
                    curr_c[i] = best_move
                    curr_r, curr_s, _ = solve_lp_and_grad(curr_c)
                    if curr_s > best_sum + 1e-8:
                        best_sum = curr_s
                        best_centers = curr_c.copy()
                        best_radii = curr_r.copy()
                else:
                    curr_c[i] = old
                    
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum, _ = solve_lp_and_grad(best_centers)

    scale = 1.0
    for i in range(N_CIRCLES):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]

            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
