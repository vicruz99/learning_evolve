    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:

    return -np.sum(v[2 * n:])

def joint_constraints(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    cons = []
    # Boundary constraints
    cons.append(cx - r)
    cons.append(1.0 - cx - r)
    cons.append(cy - r)
    cons.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    dx = cx[_TRUU_I] - cx[_TRUU_J]
    dy = cy[_TRUU_I] - cy[_TRUU_J]
    dr = r[_TRUU_I] + r[_TRUU_J]
    dist = np.sqrt(dx**2 + dy**2 + 1e-16)
    cons.append(dist - dr)
    
    return np.concatenate(cons)
