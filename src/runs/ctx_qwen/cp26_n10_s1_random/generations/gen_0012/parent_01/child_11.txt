# Precompute triangular indices for pairwise constraints to avoid repeated allocation
_TRUU_I, _TRUU_J = np.triu_indices(26, k=1)
_NUM_PAIRS = len(_TRUU_I)

def build_lp_components(centers):
    """Builds LP matrices and bounds for a given set of centers."""
    n = centers.shape[0]
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    A_ub = np.zeros((_NUM_PAIRS, n))
    A_ub[np.arange(_NUM_PAIRS), _TRUU_I] = 1.0
    A_ub[np.arange(_NUM_PAIRS), _TRUU_J] = 1.0
    b_ub = dists[_TRUU_I, _TRUU_J]
    
    bounds = [(0.0, lims[i]) for i in range(n)]
    return A_ub, b_ub, bounds, dists[_TRUU_I, _TRUU_J]

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    A_ub, b_ub, bounds, _ = build_lp_components(centers)
    try:
        res = linprog(-np.ones(26), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(26, 1e-9), 0.0

        res = linprog(-np.ones(26), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

            return None, None, None

        s = -res.fun

            np.add.at(grads, jj, -diffs * factors)
            
        return r, s, grads
    except Exception:
        return None, None, None

def objective_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[52:])

def constraints_joint(v):

    cx = v[:26]
    cy = v[26:52]
    r = v[52:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[_TRUU_I] - cx[_TRUU_J]
    dy = cy[_TRUU_I] - cy[_TRUU_J]
    dr = r[_TRUU_I] + r[_TRUU_J]
    c = np.concatenate([c, np.sqrt(dx**2 + dy**2 + 1e-16) - dr])
    return c

def force_simulate(pts, steps=600):
    """Vectorized force-directed simulation to spread circles and grow radii."""
    n = len(pts)
    c = pts.copy()
    r = np.full(n, 0.04)
    vel = np.zeros_like(c)
    dt = 0.004

        vel = vel * 0.88 + (force + wall_f) * dt
        c += vel
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
    return c, r
