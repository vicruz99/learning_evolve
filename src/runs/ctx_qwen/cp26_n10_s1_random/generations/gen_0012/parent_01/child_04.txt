import numpy as np
from scipy.optimize import linprog, minimize

_N = 26
_TRUU_I, _TRUU_J = np.triu_indices(_N, k=1)

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = len(_TRUU_I)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), _TRUU_I] = 1.0
    A_ub[np.arange(m), _TRUU_J] = 1.0
    b_ub = dists[_TRUU_I, _TRUU_J]
    
    bounds = [(0.0, lims[i]) for i in range(n)]
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def force_simulate(centers, radii, steps=1200):
    """Vectorized force-directed simulation to pack circles tightly."""
    n = len(radii)
    centers = centers.copy()
    radii = radii.copy()
    vel = np.zeros_like(centers)
    dt = 0.004
    damp = 0.86
    
    for _ in range(steps):
        radii *= 1.00007
        
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dist, 1e9)
        
        overlap = np.maximum(0.0, radii[:, np.newaxis] + radii[np.newaxis, :] - dist)
        strength = overlap * 180.0 / (dist + 1e-9)
        f = np.sum(diff * strength[:, :, np.newaxis], axis=1)
        
        w = np.zeros_like(centers)
        mask_l = centers[:, 0] < radii
        w[mask_l, 0] += (radii[mask_l] - centers[mask_l, 0]) * 250.0
        mask_r = centers[:, 0] > 1.0 - radii
        w[mask_r, 0] -= (centers[mask_r, 0] + radii[mask_r] - 1.0) * 250.0
        mask_b = centers[:, 1] < radii
        w[mask_b, 1] += (radii[mask_b] - centers[mask_b, 1]) * 250.0
        mask_t = centers[:, 1] > 1.0 - radii
        w[mask_t, 1] -= (centers[mask_t, 1] + radii[mask_t] - 1.0) * 250.0
        
        vel = damp * vel + (w + f) * dt
        centers += vel
        centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
        
    return centers, radii

def sa_optimize(centers, radii, rng, steps=4000):
    """Simulated Annealing on centers evaluated via LP."""
    n = _N
    curr_centers = centers.copy()
    r_test, curr_sum = solve_lp_radii(curr_centers)
    
    best_centers = curr_centers.copy()
    best_radii = r_test.copy()
    best_sum = curr_sum
    
    T = 0.015
    step_size = 0.025
    
    for step in range(steps):
        idx = rng.integers(n)
        old_pos = curr_centers[idx].copy()
        
        curr_centers[idx] += rng.uniform(-step_size, step_size, 2)
        curr_centers[idx] = np.clip(curr_centers[idx], 1e-4, 1.0 - 1e-4)
        
        r_new, s_new = solve_lp_radii(curr_centers)
        
        delta = s_new - curr_sum
        prob = min(np.exp(delta / T), 1.0)
        if delta > 0 or rng.random() < prob:
            curr_sum = s_new
        else:
            curr_centers[idx] = old_pos
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = curr_centers.copy()
            best_radii = r_new.copy()
            
        step_size *= 0.9992
        T *= 0.998
        
    return best_centers, best_radii, best_sum

def slsqp_polish(centers, radii):
    """Gradient-based joint optimization of centers and radii."""
    n = _N
    v0 = np.zeros(3 * n)
    v0[:n] = centers[:, 0]
    v0[n:2*n] = centers[:, 1]
    v0[2*n:] = radii * 0.98
    
    def obj(v):
        return -np.sum(v[2*n:])
        
    def cons(v):
        cx = v[:n]
        cy = v[n:2*n]
        r = v[2*n:]
        c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
        dx = cx[_TRUU_I] - cx[_TRUU_J]
        dy = cy[_TRUU_I] - cy[_TRUU_J]
        dr = r[_TRUU_I] + r[_TRUU_J]
        c = np.concatenate([c, dx**2 + dy**2 - dr**2])
        return c
        
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 8000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            c_opt = np.column_stack((res.x[:n], res.x[n:2*n]))
            r_opt, s_opt = solve_lp_radii(c_opt)
            return c_opt, r_opt, s_opt
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = _N
    rng = np.random.default_rng(42)
    
    pats = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],

        [9, 5, 6, 6], [10, 8, 8], [6, 5, 5, 6, 4], [5, 7, 5, 6, 3]
    ]
    
    starts = []
    for pat in pats:
        if sum(pat) != n: continue
        pts = []
        r0 = 0.095
        y = r0

            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:n])
        starts.append(pts)
        for _ in range(3):
            p = pts + rng.uniform(-0.025, 0.025, pts.shape)
            starts.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(6):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    for cfg in starts:
        c_sim, _ = force_simulate(cfg.copy(), np.full(n, 0.05), steps=1000)
        r_lp, s_lp = solve_lp_radii(c_sim)
        if s_lp > best_sum:
            best_sum = s_lp
            best_centers = c_sim.copy()
            best_radii = r_lp.copy()
            
    if best_centers is None:
        best_centers = starts[0]
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    for _ in range(3):
        best_centers, best_radii, best_sum = sa_optimize(best_centers, best_radii, rng, steps=3000)
        
    best_centers, best_radii, best_sum = slsqp_polish(best_centers, best_radii)
    
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
