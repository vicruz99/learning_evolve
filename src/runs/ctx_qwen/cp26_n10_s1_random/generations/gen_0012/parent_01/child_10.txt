import numpy as np
from scipy.optimize import linprog, minimize

# Precompute triangular indices for pairwise constraints
TRUU_I, TRUU_J = np.triu_indices(26, k=1)

def compute_lp_and_grad(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns optimal radii, sum of radii, and gradient of sum w.r.t centers.
    """
    n = centers.shape[0]
    
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = len(TRUU_I)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), TRUU_I] = 1.0
    A_ub[np.arange(m), TRUU_J] = 1.0
    b_ub = dists[TRUU_I, TRUU_J]
    
    bounds = [(0.0, lims[i]) for i in range(n)]
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, None, None
    except Exception:
        return None, None, None
        
    r = res.x
    s = -res.fun
    
    # Compute gradient of sum of radii w.r.t centers using LP dual variables
    grad = np.zeros_like(centers)
    if hasattr(res, 'ineqlin') and res.ineqlin is not None and hasattr(res.ineqlin, 'marginals'):
        lams = np.abs(res.ineqlin.marginals)
        valid = lams > 1e-8
        if np.any(valid):
            v_i = TRUU_I[valid]
            v_j = TRUU_J[valid]
            lams_v = lams[valid]
            d_v = dists[v_i, v_j]
            safe_d = np.where(d_v > 1e-10, d_v, 1.0)
            dir_vec = (centers[v_i] - centers[v_j]) / safe_d[:, None]
            
            np.add.at(grad, v_i, dir_vec * lams_v[:, None])
            np.add.at(grad, v_j, -dir_vec * lams_v[:, None])
            
    return r, s, grad

def objective_joint(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def constraints_joint(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    cons = []
    # Boundary constraints
    cons.append(cx - r)
    cons.append(1.0 - cx - r)
    cons.append(cy - r)
    cons.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    dx = cx[TRUU_I] - cx[TRUU_J]
    dy = cy[TRUU_I] - cy[TRUU_J]
    dr = r[TRUU_I] + r[TRUU_J]
    dist = np.sqrt(dx**2 + dy**2 + 1e-16)
    cons.append(dist - dr)
    
    return np.concatenate(cons)

def generate_hex_init(n, rows, r0, rng):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for i, cnt in enumerate(rows):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:n])

    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
