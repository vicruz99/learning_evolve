This gradient points in the direction that increases the sum of radii by pushing tightly constrained circles apart.
2. **L-BFGS-B Optimization**: We feed this objective and gradient to `L-BFGS-B`, which efficiently handles box constraints and uses quasi-Newton updates to converge to high-quality local optima for the center positions.

import numpy as np
from scipy.optimize import minimize, linprog

# Precompute triangular indices for pairwise constraints
_N = 26
_TRIU_I, _TRIU_J = np.triu_indices(_N, k=1)
_M_PAIRS = len(_TRIU_I)

def solve_lp_and_grad(centers):
    """Solves LP to maximize sum of radii for fixed centers and returns gradient w.r.t centers."""
    n = centers.shape[0]
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    np.fill_diagonal(dists, np.inf)
    
    # Construct A_ub and b_ub
    # Rows 0..M-1: pairwise r_i + r_j <= dists[i,j]
    # Rows M..M+N-1: boundary r_i <= lims[i]
    A_ub = np.zeros((_M_PAIRS + n, n))
    b_ub = np.zeros(_M_PAIRS + n)
    
    A_ub[:_M_PAIRS, _TRIU_I] = 1.0
    A_ub[:_M_PAIRS, _TRIU_J] = 1.0
    b_ub[:_M_PAIRS] = dists[_TRIU_I, _TRIU_J]
    
    A_ub[_M_PAIRS:, np.arange(n)] = 1.0
    b_ub[_M_PAIRS:] = lims
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lims[i]) for i in range(n)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return np.full(n, 1e-9), 0.0, np.zeros((n, 2))
        
        r = res.x
        s = -res.fun
        
        # Dual variables for pairwise constraints
        if res.ineqlin.marginals is not None:
            lams = np.asarray(res.ineqlin.marginals).ravel()[:_M_PAIRS]
        else:
            lams = np.zeros(_M_PAIRS)
        
        grad = np.zeros((n, 2))
        valid = lams > 1e-7
        if np.any(valid):
            vi = _TRIU_I[valid]
            vj = _TRIU_J[valid]
            vlam = lams[valid]
            vd = dists[vi, vj]
            vd_safe = np.where(vd > 1e-8, vd, 1.0)
            diff_xy = centers[vi] - centers[vj]
            factors = (vlam / vd_safe)[:, None]
            
            grad[vi] += diff_xy * factors
            grad[vj] -= diff_xy * factors
            
    except Exception:
        return np.full(n, 1e-9), 0.0, np.zeros((n, 2))
        
    return r, s, grad

def obj_centers(c_flat):
    c = c_flat.reshape(-1, 2)
    _, s, _ = solve_lp_and_grad(c)
    return -s

def jac_centers(c_flat):
    c = c_flat.reshape(-1, 2)
    _, _, g = solve_lp_and_grad(c)
    return -g.flatten()

def force_relax(centers, radii, steps=400):
    """Quick force-directed relaxation to tighten packing and remove overlaps."""
    n = centers.shape[0]
    c = centers.copy()
    r = radii.copy()
    dt = 0.004
    for _ in range(steps):
        r *= 1.00015
        diff = c[:, None, :] - c[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
        np.fill_diagonal(dist, 1e9)
        
        overlap = np.maximum(0.0, r[:, None] + r[None, :] - dist)
        force = diff * (overlap / (dist**2 + 1e-9))[:, :, None]
        f = np.sum(force, axis=1)
        
        # Wall repulsion
        f[:, 0] += np.clip(r - c[:, 0], 0, None) * 20.0
        f[:, 0] -= np.clip(c[:, 0] + r - 1.0, 0, None) * 20.0
        f[:, 1] += np.clip(r - c[:, 1], 0, None) * 20.0
        f[:, 1] -= np.clip(c[:, 1] + r - 1.0, 0, None) * 20.0
        
        c += f * dt
        c = np.clip(c, 1e-3, 1.0 - 1e-3)
    return c, r

def generate_starts(n, rng):
    starts = []
    # Diverse hexagonal row distributions summing to 26
    pats = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,4,6,4], [5,6,6,5,4], [6,5,4,6,5], [7,5,5,5,4],
        [4,7,5,5,5], [6,6,6,4,4], [5,5,5,5,6], [8,6,6,6],
        [9,5,6,6], [10,8,8], [9,9,8], [7,7,6,6], [8,8,5,5]
    ]
    for pat in pats:
        if sum(pat) != n: continue
        pts = []
        r0 = 0.09
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:n])
        # Normalize and center configuration
        mn, mx = pts.min(0), pts.max(0)
        span = mx - mn
        if span.max() > 0:
            pts = (pts - mn) / span.max() * 0.78 + 0.11
        starts.append(pts)
        
        # Perturbed variants to break symmetry
        for _ in range(4):
            p = pts + rng.uniform(-0.025, 0.025, pts.shape)
            starts.append(np.clip(p, 0.05, 0.95))
            
    # Random placements
    for _ in range(12):
        starts.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    starts = generate_starts(n, rng)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_c = [(1e-3, 1.0 - 1e-3)] * (2 * n)
    
    # Phase 1: Multi-start Gradient Optimization
    for cfg in starts:
        c_rel, r_rel = force_relax(cfg, np.full(n, 0.07))
        
        try:
            res = minimize(obj_centers, c_rel.flatten(), jac=jac_centers,
                           method='L-BFGS-B', bounds=bounds_c,
                           options={'maxiter': 3000, 'ftol': 1e-13})
            c_opt = res.x.reshape(n, 2)
            r_opt, s_opt, _ = solve_lp_and_grad(c_opt)
            
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Joint SLSQP Polish
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii])
        
        def joint_cons(v):
            cx = v[:n]
            cy = v[n:2*n]
            r = v[2*n:]
            con = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
            dx = cx[_TRIU_I] - cx[_TRIU_J]
            dy = cy[_TRIU_I] - cy[_TRIU_J]
            dr = r[_TRIU_I] + r[_TRIU_J]
            con = np.concatenate([con, np.sqrt(dx**2 + dy**2 + 1e-16) - dr])
            return con
            
        def joint_obj(v):
            return -np.sum(v[2*n:])
            
        bounds_j = bounds_c + [(1e-6, 0.5)] * n
        try:
            res_j = minimize(joint_obj, x0, method='SLSQP', bounds=bounds_j,
                             constraints={'type': 'ineq', 'fun': joint_cons},
                             options={'maxiter': 15000, 'ftol': 1e-14})
            if np.isfinite(res_j.fun):
                c_j = res_j.x[:2*n].reshape(n, 2)
                r_j, s_j, _ = solve_lp_and_grad(c_j)
                if s_j > best_sum:
                    best_sum = s_j
                    best_centers = c_j.copy()
                    best_radii = r_j.copy()
        except Exception:
            pass

    # Phase 3: Stochastic Hill-Climbing on Centers
    if best_centers is not None:
        step = 0.012
        for _ in range(2000):
            idx = rng.integers(n)
            old = best_centers[idx].copy()
            best_centers[idx] += rng.uniform(-step, step, 2)
            best_centers[idx] = np.clip(best_centers[idx], 1e-3, 1.0 - 1e-3)
            
            r_try, s_try, _ = solve_lp_and_grad(best_centers)
            if s_try > best_sum + 1e-9:
                best_sum = s_try
                best_radii = r_try.copy()
                step = min(step * 1.005, 0.025)
            else:
                best_centers[idx] = old
                step *= 0.997

    # Phase 4: Strict Safety Scaling for Validator Tolerance
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-12:
                    scale = min(scale, d/rs)
        best_radii *= scale * 0.999999
        best_sum = float(np.sum(best_radii))
    else:
        best_centers = rng.uniform(0.1, 0.9, (n, 2))
        best_radii, best_sum, _ = solve_lp_and_grad(best_centers)
        best_sum = float(best_sum)
        
    return best_centers, best_radii, best_sum
