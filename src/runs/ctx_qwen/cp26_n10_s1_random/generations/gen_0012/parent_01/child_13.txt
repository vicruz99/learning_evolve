_M = len(_TRUU_I)
_A_ub = np.zeros((_M, N))
_A_ub[np.arange(_M), _TRUU_I] = 1.0
_A_ub[np.arange(_M), _TRUU_J] = 1.0

def solve_lp_radii(centers):

    lims = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))

    dy = y[:, None] - y[None, :]

        res = linprog(-np.ones(N), A_ub=_A_ub, b_ub=b_ub, bounds=bounds, method='highs')

            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-9), 0.0
