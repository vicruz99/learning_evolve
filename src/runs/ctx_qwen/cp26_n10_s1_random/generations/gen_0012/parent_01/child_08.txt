1. **Force-Directed Relaxation**: Start from diverse hexagonal lattice patterns and random seeds. Run a molecular dynamics-style simulation that slowly expands circle radii while applying strong repulsive forces for overlaps and boundary violations. This finds globally tight structural arrangements.
2. **LP-Guided Gradient Ascent**: For any fixed set of centers, the optimal radii are given by a Linear Program. The dual variables (shadow prices) of the LP's pairwise distance constraints directly give the gradient of the sum of radii with respect to the center positions. Moving centers along this gradient efficiently climbs the objective landscape.
3. **Joint SLSQP Polish**: Use the gradient-ascent result as a high-quality starting point for `scipy.optimize.minimize` with SLSQP to simultaneously adjust centers and radii under smooth squared-distance constraints, capturing fine local optima.

We iterate through multiple initial configurations, applying Force Sim -> Gradient Ascent -> SLSQP -> Gradient Ascent. The best result is strictly scaled to satisfy the `1e-12` numerical tolerance of the validator. This hybrid approach balances global structure discovery with exact local sensitivity analysis.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

# Precompute indices for pairwise constraints
_TRUU_I, _TRUU_J = np.triu_indices(26, k=1)

def compute_lp_and_grad(centers):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient w.r.t centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    diff = centers[:, None] - centers[None, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    np.fill_diagonal(dists, np.inf)
    
    m = len(_TRUU_I)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), _TRUU_I] = 1.0
    A_ub[np.arange(m), _TRUU_J] = 1.0
    b_ub = dists[_TRUU_I, _TRUU_J]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, None, None
            
        r = res.x
        s = -res.fun
        
        grad = np.zeros((n, 2))
        if hasattr(res, 'ineqlin') and res.ineqlin is not None:
            lams = getattr(res.ineqlin, 'marginals', None)
            if lams is not None:
                lams = np.abs(lams)
                active = lams > 1e-7
                if np.any(active):
                    k_idx = np.where(active)[0]
                    lam_v = lams[k_idx]
                    i_v = _TRUU_I[k_idx]
                    j_v = _TRUU_J[k_idx]
                    d_v = dists[i_v, j_v]
                    inv_d = np.where(d_v > 1e-8, 1.0 / d_v, 0.0)
                    factors = lam_v * inv_d
                    diff_vecs = diff[i_v, j_v]
                    np.add.at(grad, i_v, diff_vecs * factors[:, None])
                    np.add.at(grad, j_v, -diff_vecs * factors[:, None])
                
        return r, s, grad
    except Exception:
        return None, None, None

def force_simulate(centers, steps=1800, dt=0.004, damp=0.88, k_rep=700.0):
    """Force-directed simulation to find dense packing configurations."""
    n = centers.shape[0]
    c = centers.copy()
    vel = np.zeros_like(c)
    r = np.full(n, 0.085)
    
    for step in range(steps):
        # Slowly expand circles to fill space
        r *= (1.0 + 6e-6)
        
        diff = c[:, None] - c[None, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dist, 1e9)
        
        r_sum = r[:, None] + r[None, :]
        overlap = np.maximum(0.0, r_sum - dist)
        force_mag = overlap * k_rep / (dist + 1e-9)
        
        f = np.zeros_like(c)
        f[:, 0] = np.sum(diff[:, :, 0] * force_mag, axis=1)
        f[:, 1] = np.sum(diff[:, :, 1] * force_mag, axis=1)
        
        # Boundary repulsion
        for dim in [0, 1]:
            inside_left = c[:, dim] < r
            inside_right = c[:, dim] > 1.0 - r
            f[inside_left, dim] += (r[inside_left] - c[inside_left, dim]) * 1500.0
            f[inside_right, dim] -= (c[inside_right, dim] - (1.0 - r[inside_right])) * 1500.0
            
        vel = vel * damp + f * dt
        c += vel
        c = np.clip(c, 0.002, 0.998)
        
    return c, r

def gradient_ascent(centers, max_iter=400, step=0.007):
    """Optimizes centers by following the LP dual gradient."""
    c = centers.copy()
    for _ in range(max_iter):
        r, s, g = compute_lp_and_grad(c)
        if r is None: break
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-8: break
        
        c_new = c + step * (g / g_norm)
        c_new = np.clip(c_new, 0.005, 0.995)
        
        r_new, s_new, _ = compute_lp_and_grad(c_new)
        if r_new is not None and s_new > s + 1e-9:
            c = c_new
            step = min(step * 1.015, 0.02)
        else:
            step *= 0.92
            
    r_final, s_final, _ = compute_lp_and_grad(c)
    return c, r_final, s_final

def joint_objective(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def joint_constraints(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    cons = []
    cons.append(cx - r)
    cons.append(1.0 - cx - r)
    cons.append(cy - r)
    cons.append(1.0 - cy - r)
    
    dx = cx[_TRUU_I] - cx[_TRUU_J]
    dy = cy[_TRUU_I] - cy[_TRUU_J]
    dr = r[_TRUU_I] + r[_TRUU_J]
    dist = np.sqrt(dx**2 + dy**2 + 1e-16)
    cons.append(dist - dr)
    
    return np.concatenate(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    # 1. Generate diverse structural initializations
    configs = []
    pats = [[5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], 
            [6,6,4,6,4], [7,5,5,5,4], [8,6,6,6], [6,7,6,7], [5,6,5,5,5]]
    
    for pat in pats:
        if sum(pat) != n: continue
        pts = []
        y = 0.1
        for i, cnt in enumerate(pat):
            shift = 0.1 if i % 2 == 1 else 0.0
            x = 0.1 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 0.2
            y += 0.1 * np.sqrt(3)
        configs.append(np.array(pts[:n]))
        
    # Perturb top lattice patterns to break symmetry
    for cfg in configs[:3]:
        for _ in range(4):
            cfg_p = cfg + rng.uniform(-0.04, 0.04, (n, 2))
            configs.append(np.clip(cfg_p, 0.05, 0.95))
            
    # Add purely random starts
    for _ in range(6):
        configs.append(rng.uniform(0.2, 0.8, (n, 2)))
        
    bounds_vars = [(0.001, 0.999)] * (2 * n) + [(1e-6, 0.5)] * n
    cons_dict = {'type': 'ineq', 'fun': joint_constraints, 'args': (n,)}
    
    # 2. Multi-phase optimization pipeline
    for cfg in configs:
        # Phase A: Force simulation to find tight structural layout
        c_sim, _ = force_simulate(cfg.copy(), steps=1600, dt=0.004, damp=0.88, k_rep=700.0)
        
        # Phase B: LP-Guided Gradient Ascent on centers
        c_asc, r_asc, s_asc = gradient_ascent(c_sim, max_iter=350, step=0.007)
        
        if s_asc > best_sum:
            best_sum = s_asc
            best_c = c_asc.copy()
            best_r = r_asc.copy()
            
        # Phase C: Joint SLSQP Polish
        x0 = np.concatenate([c_asc.flatten(), r_asc * 0.96])
        try:
            res = minimize(joint_objective, x0, args=(n,), method='SLSQP', bounds=bounds_vars,
                           constraints=cons_dict, options={'maxiter': 6000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_pol = res.x[:2 * n].reshape(n, 2)
                r_lp, s_lp, _ = compute_lp_and_grad(c_pol)
                if r_lp is not None and s_lp > best_sum:
                    best_sum = s_lp
                    best_c = c_pol.copy()
                    best_r = r_lp.copy()
        except Exception:
            pass
            
        # Phase D: Second Gradient Ascent from polished result
        c_asc2, r_asc2, s_asc2 = gradient_ascent(best_c.copy(), max_iter=200, step=0.005)
        if s_asc2 > best_sum:
            best_sum = s_asc2
            best_c = c_asc2.copy()
            best_r = r_asc2.copy()

    # 3. Strict Safety Scaling to guarantee numerical validity (1e-12 tolerance)
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_r *= scale * 0.999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
