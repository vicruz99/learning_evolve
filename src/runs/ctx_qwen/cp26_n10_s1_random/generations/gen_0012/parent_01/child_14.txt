import numpy as np
from scipy.optimize import minimize, linprog

# Precompute triangular indices for pairwise constraints
_TRIU_I, _TRIU_J = np.triu_indices(26, k=1)

def build_lp_problem(centers):
    """Constructs LP matrices for maximizing sum of radii."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = len(_TRIU_I)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), _TRIU_I] = 1.0
    A_ub[np.arange(m), _TRIU_J] = 1.0
    b_ub = dists[_TRIU_I, _TRIU_J]
    
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lims[i]) for i in range(n)]
    
    return A_ub, b_ub, bounds, dists

def solve_lp_and_grad(centers):
    """Solves LP for radii and computes gradient of sum of radii w.r.t centers."""
    n = centers.shape[0]
    A_ub, b_ub, bounds, dists = build_lp_problem(centers)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, None, None
    except Exception:
        return None, None, None
        
    r = res.x
    s = -res.fun

    grad = np.zeros((n, 2))
    try:
        duals = res.ineqlin.marginals
        active = duals > 1e-6

            np.add.at(grad, jj, -dirs * lams[:, None])
    except Exception:
        pass
        
    return r, s, grad

def gradient_ascent(centers, rng, max_iter=250):

def s_obj(v, n):
    return -np.sum(v[2*n:])

def s_cons(v, n):
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[_TRIU_I] - cx[_TRIU_J]
    dy = cy[_TRIU_I] - cy[_TRIU_J]
    dr = r[_TRIU_I] + r[_TRIU_J]
    dist = np.sqrt(dx**2 + dy**2 + 1e-16)
    c = np.concatenate([c, dist - dr])
    return c
