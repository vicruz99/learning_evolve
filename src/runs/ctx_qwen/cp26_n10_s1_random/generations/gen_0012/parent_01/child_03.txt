import numpy as np
from scipy.optimize import minimize, linprog

# Precompute triangular indices for pairwise constraints
_TRUU_I, _TRUU_J = np.triu_indices(26, k=1)

def compute_lp_and_grad(centers):
    """Solves LP to maximize sum of radii and computes gradient w.r.t centers."""
    n = centers.shape[0]
    
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    np.fill_diagonal(dists, 1e9)
    
    m = len(_TRUU_I)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), _TRUU_I] = 1.0
    A_ub[np.arange(m), _TRUU_J] = 1.0
    b_ub = dists[_TRUU_I, _TRUU_J]
    
    bounds = [(0.0, lims[i]) for i in range(n)]
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, None, None
            
        r = res.x
        s = -res.fun
        
        # Gradient calculation from dual variables (marginals)
        grad = np.zeros((n, 2))
        if hasattr(res, 'ineqlin') and res.ineqlin.marginals is not None:
            lams = np.abs(res.ineqlin.marginals)
            valid = lams > 1e-7
            if np.any(valid):
                idx_k = np.where(valid)[0]
                lams_v = lams[valid]
                ii = _TRUU_I[idx_k]
                jj = _TRUU_J[idx_k]
                ds = dists[ii, jj]
                ds = np.where(ds > 1e-9, ds, 1.0)
                vecs = (centers[ii] - centers[jj]) / ds[:, None]
                # Accumulate gradient: grad_i += lambda * (c_i - c_j)/d_ij
                np.add.at(grad, ii, vecs * lams_v[:, None])
                np.add.at(grad, jj, -vecs * lams_v[:, None])
        return r, s, grad
    except Exception:
        return None, None, None

def lp_gradient_ascent(centers, n, rng, iterations=600):
    """Optimizes centers by following the LP dual gradient."""
    best_r = None
    best_s = -1.0
    best_c = centers.copy()
    step = 0.006
    
    for it in range(iterations):
        r, s, grad = compute_lp_and_grad(centers)
        if r is None:
            break
            
        if s > best_s + 1e-8:
            best_s = s
            best_r = r.copy()
            best_c = centers.copy()
            
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-7:
            # Plateau detected: perturb to escape local minimum
            centers += rng.uniform(-0.008, 0.008, centers.shape)
            centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
            step = 0.006
        else:
            step = min(0.025, step * 1.003)
            move = step * grad / g_norm
            centers += 0.6 * move  # Damping for stability
            centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
            
            # Gradually decay step to converge
            if it % 60 == 0 and it > 0:
                step *= 0.85
                
    return best_c, best_r, best_s

def objective_joint(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def constraints_joint(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    dx = cx[_TRUU_I] - cx[_TRUU_J]
    dy = cy[_TRUU_I] - cy[_TRUU_J]
    dist = np.sqrt(dx**2 + dy**2 + 1e-16)
    dr = r[_TRUU_I] + r[_TRUU_J]
    c = np.concatenate([c, dist - dr])
    return c

def generate_hex_configs(n, rng):
    """Generates diverse hexagonal lattice configurations summing to n circles."""
    configs = []
    # Row patterns that sum to 26, mimicking hexagonal close packing
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], [4, 6, 6, 6, 4],
        [7, 5, 5, 5, 4], [5, 5, 6, 5, 5], [6, 5, 4, 6, 5], [5, 4, 6, 6, 5],
        [8, 5, 6, 5, 2], [4, 5, 7, 5, 5], [5, 7, 5, 5, 4], [6, 6, 6, 4, 4]
    ]
    
    for pat in patterns:
        if sum(pat) != n:
            continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n:
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
            
        base = np.array(pts[:n])
        # Normalize and center
        mn = base.min(axis=0)
        mx = base.max(axis=0)
        span = mx - mn
        if span[0] > 1e-4 and span[1] > 1e-4:
            scale = 0.84 / max(span[0] / (1.0 - 2 * r0), span[1] / (1.0 - 2 * r0))
            base = (base - mn) * scale + 0.5 * (1.0 - scale)
        configs.append(np.clip(base, 0.05, 0.95))
        
        # Add perturbed variants
        for _ in range(3):
            p = base + rng.uniform(-0.025, 0.025, base.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    # Random starts
    for _ in range(10):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    configs = generate_hex_configs(n, rng)
    
    # Phase 1: LP Gradient Ascent from diverse starts
    for cfg in configs:
        c_opt, r_opt, s_opt = lp_gradient_ascent(cfg.copy(), n, rng, iterations=500)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            best_radii = r_opt
            
    # Phase 2: Joint SLSQP Polish
    if best_centers is not None:
        x0 = np.zeros(3 * n)
        x0[:n] = best_centers[:, 0]
        x0[n:2 * n] = best_centers[:, 1]
        x0[2 * n:] = np.maximum(best_radii, 1e-6) * 0.96
        
        bounds_opt = [(1e-4, 1.0 - 1e-4)] * (2 * n) + [(1e-6, 0.5)] * n
        cons_dict = {'type': 'ineq', 'fun': constraints_joint, 'args': (n,)}
        
        try:
            res = minimize(objective_joint, x0, args=(n,), method='SLSQP', 
                           bounds=bounds_opt, constraints=cons_dict,
                           options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            if np.isfinite(res.fun):
                c_mat = np.column_stack((res.x[:n], res.x[n:2 * n]))
                r_lp, s_lp, _ = compute_lp_and_grad(c_mat)
                if r_lp is not None and s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c_mat.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass

    # Fallback if optimization fails completely
    if best_centers is None:
        best_centers = rng.uniform(0.2, 0.8, (n, 2))
        best_radii, best_sum, _ = compute_lp_and_grad(best_centers)
        
    # Phase 3: Strict Safety Scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    diff = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    np.fill_diagonal(dists, 1e9)
    r_pair = best_radii[:, np.newaxis] + best_radii[np.newaxis, :]
    
    for i in range(n):
        for j in range(i + 1, n):
            d = dists[i, j]
            rs = r_pair[i, j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
