import numpy as np
from scipy.optimize import linprog

_TRUU_I, _TRUU_J = np.triu_indices(26, k=1)

def solve_lp_and_grad(centers):
    """Solves LP to maximize sum of radii for fixed centers and returns dual variables."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = len(_TRUU_I)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), _TRUU_I] = 1.0
    A_ub[np.arange(m), _TRUU_J] = 1.0
    b_ub = dists[_TRUU_I, _TRUU_J]
    
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, None, None
        r = res.x
        s = -res.fun
        lams = np.abs(res.ineqlin.marginals)
        return r, s, lams
    except Exception:
        return None, None, None

def compute_gradient_vec(centers, lams):
    """Computes gradient of sum of radii w.r.t centers using LP dual variables."""
    n = centers.shape[0]
    grad = np.zeros((n, 2))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    dists_safe = np.where(dists > 1e-9, dists, 1.0)
    
    active = lams > 1e-7
    lam_a = lams[active]
    ii = _TRUU_I[active]
    jj = _TRUU_J[active]
    
    di = dists_safe[ii, jj]
    factors = lam_a / di
    diffs = centers[ii] - centers[jj]
    
    np.add.at(grad, ii, diffs * factors[:, None])
    np.add.at(grad, jj, -diffs * factors[:, None])
    return grad

def gradient_optimize(centers, rng, max_iter=1500):
    """Gradient ascent on centers to maximize sum of radii."""
    n = centers.shape[0]
    c = centers.copy()
    _, s, lams = solve_lp_and_grad(c)
    if s is None:
        return c, 0.0
    best_c = c.copy()
    best_s = s
    
    step = 0.005
    for it in range(max_iter):
        r, s, lams = solve_lp_and_grad(c)
        if r is None:
            break
        grad = compute_gradient_vec(c, lams)
        gn = np.linalg.norm(grad)
        
        if gn > 1e-9:
            new_c = c + step * grad / gn
            new_c = np.clip(new_c, 1e-4, 1.0 - 1e-4)
            
            r_new, s_new, _ = solve_lp_and_grad(new_c)
            if r_new is not None and s_new > best_s + 1e-9:
                best_c = new_c.copy()
                best_s = s_new
                c = new_c
                step = min(step * 1.02, 0.04)
            else:
                step *= 0.85
        else:
            # Random jitter to escape flat regions
            if rng.random() < 0.3:
                c += rng.uniform(-step * 2, step * 2, c.shape)
                c = np.clip(c, 1e-4, 1.0 - 1e-4)
            step *= 0.95
            
        if step < 1e-6:
            break
    return best_c, best_s

def generate_starts(n, rng):
    """Generates diverse initial configurations."""
    starts = []
    pats = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,4,6,4], [5,6,6,5,4], [5,4,6,5,6], [6,5,4,6,5],
        [7,5,5,5,4], [4,7,5,5,5], [6,6,6,4,4], [5,5,5,5,6]
    ]
    for pat in pats:
        if sum(pat) != n:
            continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n:
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:n])
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        if span.max() > 1e-4:
            scale = 0.75 / span.max()
            pts = (pts - mn) * scale + 0.5 * (1.0 - scale)
        starts.append(pts)
        for _ in range(2):
            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            starts.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(5):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)))
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    best_c = None
    best_s = -1.0
    best_r = None
    
    starts = generate_starts(n, rng)
    
    for cfg in starts:
        c_opt, s_opt = gradient_optimize(cfg.copy(), rng, max_iter=1500)
        if s_opt is not None and s_opt > best_s:
            best_s = s_opt
            r_temp, _, _ = solve_lp_and_grad(c_opt)
            best_r = r_temp.copy()
            best_c = c_opt.copy()
            
    # Fallback if optimization unexpectedly fails
    if best_c is None:
        best_c = np.column_stack(np.meshgrid(np.linspace(0.1, 0.9, 5), np.linspace(0.1, 0.9, 5))).reshape(-1, 2)
        best_c = best_c[:n]
        best_r, best_s, _ = solve_lp_and_grad(best_c)
        
    # Local perturbation search to squeeze out final gains
    if best_c is not None:
        step = 0.008
        for _ in range(2000):
            idx = rng.integers(n)
            old = best_c[idx].copy()
            best_c[idx] += rng.uniform(-step, step, 2)
            best_c[idx] = np.clip(best_c[idx], 1e-4, 1.0 - 1e-4)
            
            r_try, s_try, _ = solve_lp_and_grad(best_c)
            if r_try is not None and s_try > best_s + 1e-9:
                best_s = s_try
                best_r = r_try.copy()
            else:
                best_c[idx] = old
            step *= 0.9995
            
    # Strict safety scaling to guarantee numerical validity (1e-12 tolerance)
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_r *= scale * 0.999999
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
