import numpy as np
from scipy.optimize import linprog, minimize

# Precompute triangular indices for pairwise constraints
_TRUU_I, _TRUU_J = np.triu_indices(26, k=1)

def get_lp_solution(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns optimal radii, sum of radii, and gradient of sum w.r.t centers.
    """
    n = centers.shape[0]
    
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    np.fill_diagonal(dists, np.inf)
    
    # LP Setup: max sum(r) s.t. r_i + r_j <= d_ij, 0 <= r_i <= lims_i
    m = len(_TRUU_I)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), _TRUU_I] = 1.0
    A_ub[np.arange(m), _TRUU_J] = 1.0
    b_ub = dists[_TRUU_I, _TRUU_J]
    
    bounds = [(0.0, lims[i]) for i in range(n)]
    c_obj = -np.ones(n)  # linprog minimizes
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, None, None
            
        radii = res.x
        s = -res.fun
        
        # Compute gradient from dual variables (marginals)
        # Duals correspond to constraints r_i + r_j <= d_ij
        # Increasing d_ij by delta increases objective by lambda_ij * delta
        grad = np.zeros((n, 2))
        lams = np.abs(res.ineqlin.marginals)
        
        active = lams > 1e-7
        if np.any(active):
            ii = _TRUU_I[active]
            jj = _TRUU_J[active]
            d = dists[ii, jj]
            d_safe = np.where(d > 1e-9, d, 1.0)
            
            # Direction from j to i normalized
            dirs = (centers[ii] - centers[jj]) / d_safe[:, None]
            forces = lams[active][:, None] * dirs
            
            np.add.at(grad, ii, forces)
            np.add.at(grad, jj, -forces)
            
        return radii, s, grad
    except Exception:
        return None, None, None

def run_gradient_ascent(c0, rng, steps=250, lr_init=0.005):
    """Performs gradient ascent on centers using LP dual gradients."""
    c = c0.copy()
    best_c = c.copy()
    best_s = -1.0
    
    r, s, g = get_lp_solution(c)
    if s is None:
        return c0, -1.0
    best_s = s
    
    for i in range(steps):
        # Decaying learning rate
        lr = lr_init * (1.0 - i / steps)**0.6
        
        # Gradient step
        c_new = c + lr * g
        c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
        
        r_new, s_new, g_new = get_lp_solution(c_new)
        if s_new is None:
            break
            
        if s_new > best_s + 1e-10:
            best_s = s_new
            best_c = c_new.copy()
            c = c_new
            g = g_new
        else:
            # Stochastic escape: occasionally perturb if stuck
            if i % 40 == 0:
                c[:] = best_c + rng.normal(0, 0.004, best_c.shape)
                c = np.clip(c, 1e-4, 1.0 - 1e-4)
                _, s, g = get_lp_solution(c)
                if s is None: break
                
    return best_c, best_s

def local_hill_climb(c0, rng, iterations=1200):
    """Refines centers via random perturbations evaluated by LP."""
    c = c0.copy()
    r, best_s, _ = get_lp_solution(c)
    if best_s is None: return c0
    
    step = 0.012

        idx = rng.integers(26)

            best_s = s_new
            step = min(step * 1.01, 0.025)
        else:
            c[idx] = old
            step *= 0.998
            
    return c
