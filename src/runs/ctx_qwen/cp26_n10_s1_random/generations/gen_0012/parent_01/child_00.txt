import numpy as np
from scipy.optimize import linprog, minimize

# Precompute triangular indices for pairwise constraints
_TRUU_I, _TRUU_J = np.triu_indices(26, k=1)
_M = len(_TRUU_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, and dual marginals."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    np.fill_diagonal(dists, np.inf)
    
    A_ub = np.zeros((_M, n))
    A_ub[np.arange(_M), _TRUU_I] = 1.0
    A_ub[np.arange(_M), _TRUU_J] = 1.0
    b_ub = dists[_TRUU_I, _TRUU_J]
    
    bounds = [(0.0, lims[i]) for i in range(n)]
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            r = res.x
            s = -res.fun
            lams = np.abs(res.ineqlin.marginals)
            return r, s, lams
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0, np.zeros(_M)

def grad_sum_radii(centers, lams):
    """Computes gradient of sum of radii w.r.t centers using LP marginals."""
    n = centers.shape[0]
    grad = np.zeros((n, 2))
    diff = centers[_TRUU_I] - centers[_TRUU_J]
    dists = np.sqrt(np.sum(diff**2, axis=1) + 1e-16)
    factors = (lams[:, None] / dists[:, None]) * diff
    np.add.at(grad, _TRUU_I, factors)
    np.add.at(grad, _TRUU_J, -factors)
    return grad

def s_obj(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def s_cons(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    con = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[_TRUU_I] - cx[_TRUU_J]
    dy = cy[_TRUU_I] - cy[_TRUU_J]
    dr = r[_TRUU_I] + r[_TRUU_J]
    dist = np.sqrt(dx**2 + dy**2 + 1e-16)
    con = np.concatenate([con, dist - dr])
    return con

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    # Generate diverse initial configurations
    configs = []
    pats = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [6, 6, 4, 6, 4], [5, 6, 6, 5, 4], [7, 5, 5, 5, 4], [6, 6, 6, 4, 4],
        [8, 6, 6, 6], [9, 5, 6, 6], [10, 8, 8], [9, 9, 8]
    ]
    
    for pat in pats:
        if sum(pat) != n: continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:n])
        configs.append(pts)
        
        # Perturbed variants to break symmetry traps
        for _ in range(3):
            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(6):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = -1.0
    best_c = None
    best_r = None

            if s > best_sum:
                best_sum = s
                best_c = c.copy()
                best_r = r.copy()

            best_c[idx] = old

    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
