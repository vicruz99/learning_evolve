import numpy as np
from scipy.optimize import minimize, linprog

# Precompute triangular indices for pairwise constraints to avoid repeated allocation
_N = 26
_TRIU_I, _TRIU_J = np.triu_indices(_N, k=1)

def compute_lp_and_grad(centers):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient."""
    n = centers.shape[0]
    
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    np.fill_diagonal(dists, 0.0)
    
    m = len(_TRIU_I)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), _TRIU_I] = 1.0
    A_ub[np.arange(m), _TRIU_J] = 1.0
    b_ub = dists[_TRIU_I, _TRIU_J]
    
    bounds = [(0.0, lims[i]) for i in range(n)]
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return np.full(n, 0.0), 0.0, np.zeros((n, 2))
    except Exception:
        return np.full(n, 0.0), 0.0, np.zeros((n, 2))
        
    r = res.x
    s = -res.fun
    
    # Compute gradient of sum of radii w.r.t. centers using LP dual variables
    grad = np.zeros((n, 2))
    try:
        marginals = np.asarray(res.ineqlin.marginals, dtype=float)
    except Exception:
        marginals = np.zeros(m)
        
    for k in range(m):
        mu = marginals[k]
        if mu > 1e-7:
            i, j = _TRIU_I[k], _TRIU_J[k]
            d = dists[i, j]
            if d > 1e-9:
                factor = mu / d
                diff_ij = centers[i] - centers[j]
                grad[i] += factor * diff_ij
                grad[j] -= factor * diff_ij
                
    return r, s, grad

def joint_obj(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def joint_con(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    con = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[_TRIU_I] - cx[_TRIU_J]
    dy = cy[_TRIU_I] - cy[_TRIU_J]
    dr = r[_TRIU_I] + r[_TRIU_J]
    dist = np.sqrt(dx**2 + dy**2 + 1e-16)
    con = np.concatenate([con, dist - dr])
    
    return con

def optimize_config(centers_init, rng):
    """Runs LP-guided gradient ascent with jitter to escape local minima."""
    centers = centers_init.copy()
    r, s, g = compute_lp_and_grad(centers)
    best_sum = s
    best_c = centers.copy()
    best_r = r.copy()
    
    step = 0.025
    for it in range(800):
        gn = np.linalg.norm(g)
        if gn > 1e-10:
            dir_vec = g / gn
        else:
            dir_vec = rng.normal(size=(26, 2))
            dir_vec /= np.linalg.norm(dir_vec) + 1e-9
            
        # Gradient ascent step
        c_new = centers + step * dir_vec
        c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
        r_new, s_new, g_new = compute_lp_and_grad(c_new)
        
        accepted = False
        if s_new > best_sum + 1e-8:
            best_sum = s_new
            best_c = c_new.copy()
            best_r = r_new.copy()
            centers = c_new
            g = g_new
            step = min(step * 1.03, 0.05)
            accepted = True
        elif rng.random() < 0.12: # Random jitter to escape plateaus
            c_jig = centers.copy()
            idx = rng.integers(26, size=rng.integers(2, 5))
            c_jig[idx] += rng.uniform(-0.015, 0.015, (len(idx), 2))
            c_jig = np.clip(c_jig, 1e-4, 1.0 - 1e-4)
            r_j, s_j, g_j = compute_lp_and_grad(c_jig)

                best_sum = s_j
                best_c = c_jig.copy()
                best_r = r_j.copy()
                centers = c_jig
                g = g_j
                step = 0.03
                accepted = True
                
        if not accepted:
            step *= 0.96
            centers = c_new
            g = g_new
            
    return best_c, best_r, best_sum

    return best_centers, best_radii, best_sum
