

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we leverage the mathematical property that for fixed centers, the optimal radii are given by a linear program (LP). The LP constraints are `r_i + r_j <= dist(i,j)` and `r_i <= boundary_dist(i)`. 

The key insight is that the dual variables (marginals) of this LP provide the exact analytical gradient of the maximum achievable sum of radii with respect to the centers. This allows us to use efficient gradient-based optimizers (like L-BFGS-B) to move centers in directions that increase the objective.

However, the objective landscape is highly non-convex with many local minima. To escape them, we implement a multi-phase strategy:
1. **Diverse Initialization**: Hexagonal lattices with varying row distributions and dense random starts.
2. **Gradient Ascent**: Use L-BFGS-B with the LP dual gradient to climb to local optima.
3. **Simulated Annealing / Basin Hopping**: Systematically perturb random subsets of circles, re-optimize locally, and probabilistically accept moves to escape local minima. The "temperature" decays over time to refine the packing.
4. **Joint SLSQP Polish**: Simultaneously optimize centers and radii using SLSQP to smooth out any remaining constraint violations or numerical tightness.
5. **Strict Safety Scaling**: Apply a final global scaling factor to guarantee strict feasibility against the validator's `1e-12` tolerance.

This approach combines the speed of LP-based gradient ascent with the global exploration capabilities of simulated annealing, consistently pushing past standard local optima.

import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrices(n):
    """Precomputes the constant inequality matrix A_ub for the radius LP."""
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    n_pairs = len(pairs)
    m = 4 * n + n_pairs
    A_ub = np.zeros((m, n))
    
    for i in range(n):
        A_ub[i, i] = 1.0
        A_ub[n + i, i] = 1.0
        A_ub[2 * n + i, i] = 1.0
        A_ub[3 * n + i, i] = 1.0
        
    for k, (i, j) in enumerate(pairs):
        A_ub[4 * n + k, i] = 1.0
        A_ub[4 * n + k, j] = 1.0
        
    return A_ub, pairs, n_pairs, m

def solve_lp_and_grad(centers, A_ub, pairs, n, m):
    """Solves LP for radii and computes gradient of optimal sum w.r.t centers using duals."""
    b = np.zeros(m)
    for i in range(n):
        x, y = centers[i]
        b[i] = x
        b[n + i] = 1.0 - x
        b[2 * n + i] = y
        b[3 * n + i] = 1.0 - y
        
    for k, (i, j) in enumerate(pairs):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b[4 * n + k] = np.hypot(dx, dy)
        
    bounds_r = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=bounds_r, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, 0.0, None
            
        r = res.x
        s = -res.fun
        grad = np.zeros((n, 2))
        
        # Extract marginals robustly across scipy versions
        lam = None
        if hasattr(res, 'marginals'):
            lam = res.marginals.ineqlin
        elif hasattr(res, 'ineqlin'):
            lam = res.ineqlin.marginals
        else:
            lam = np.zeros(m)
            
        if lam is not None:
            # Boundary gradients: d(r_i)/d(x) = mu_left - mu_right
            for i in range(n):
                grad[i, 0] += lam[i] - lam[n + i]
                grad[i, 1] += lam[2 * n + i] - lam[3 * n + i]
                
            # Pairwise distance gradients: d(r_i)/d(x_i) = sum(lambda_k * (dx_ij / d_ij))
            for k, (i, j) in enumerate(pairs):
                val = lam[4 * n + k]
                if val > 1e-9:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = b[4 * n + k]
                    d_safe = max(d, 1e-9)
                    fx = val * dx / d_safe
                    fy = val * dy / d_safe
                    grad[i, 0] += fx
                    grad[i, 1] += fy
                    grad[j, 0] -= fx
                    grad[j, 1] -= fy
                    
        return r, s, grad
    except Exception:
        return None, 0.0, None

def obj_grad_wrap(x_flat, n, A_ub, pairs, m):
    """Wrapper for scipy.minimize: returns objective and gradient."""
    centers = x_flat.reshape(n, 2)
    r, s, grad = solve_lp_and_grad(centers, A_ub, pairs, n, m)
    if r is None:
        return 1e6, np.zeros_like(x_flat)
    # minimize expects negative gradient for maximization
    return -s, -grad.flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    A_ub, pairs, n_pairs, m = build_lp_matrices(n)
    rng = np.random.default_rng(42)
    
    best_s = -1.0
    best_c = None
    best_r = None
    
    # --- Phase 1: Diverse Initializations ---
    configs = []
    # Hexagonal row patterns summing to ~26
    pats = [[6,5,6,5,4], [5,6,5,6,4], [7,6,6,7], [6,6,5,5,4], [5,5,6,5,5], [6,5,5,6,4]]
    for pat in pats:
        if sum(pat) < n: continue
        pts = []
        y = 0.1
        for ri, cnt in enumerate(pat):
            shift = 0.1 if ri % 2 == 1 else 0.0
            x = 0.1 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 0.2
            y += 0.1 * np.sqrt(3)
        configs.append(np.array(pts[:n]))
        
    # Dense random starts
    for _ in range(20):
        configs.append(rng.uniform(0.05, 0.95, (n, 2)))
        
    bounds_c = [(0.005, 0.995)] * (2 * n)
    
    for cfg in configs:
        c0 = np.clip(cfg, 0.01, 0.99)
        try:
            res = minimize(obj_grad_wrap, c0.flatten(), args=(n, A_ub, pairs, m),
                           method='L-BFGS-B', jac=True, bounds=bounds_c,
                           options={'maxiter': 5000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(n, 2)
                r_opt, s_opt, _ = solve_lp_and_grad(c_opt, A_ub, pairs, n, m)
                if r_opt is not None and s_opt > best_s:
                    best_s = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            continue

    # --- Phase 2: Simulated Annealing / Basin Hopping ---
    c_cur = best_c.copy()
    r_cur = best_r.copy()
    s_cur = best_s
    
    for step in range(2500):
        # Cooling schedule
        temp = 0.015 * max(0.1, 1.0 - step / 2500.0)
        
        # Perturb 1 to 3 random circles
        k = rng.integers(1, 4)
        idx = rng.choice(n, k, replace=False)
        old_c = c_cur.copy()
        
        c_cur[idx] += rng.uniform(-temp, temp, (k, 2))
        c_cur = np.clip(c_cur, 0.01, 0.99)
        
        # Local optimization from perturbed state
        try:
            res = minimize(obj_grad_wrap, c_cur.flatten(), args=(n, A_ub, pairs, m),
                           method='L-BFGS-B', jac=True, bounds=bounds_c,
                           options={'maxiter': 800, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_new = res.x.reshape(n, 2)
                r_new, s_new, _ = solve_lp_and_grad(c_new, A_ub, pairs, n, m)
                if r_new is not None:
                    # Accept if improvement or probabilistically based on temperature
                    if s_new > s_cur + 1e-9 or rng.random() < np.exp((s_new - s_cur) / max(temp, 1e-4)):
                        c_cur = c_new
                        r_cur = r_new
                        s_cur = s_new
                        if s_cur > best_s:
                            best_s = s_cur
                            best_c = c_cur.copy()
                            best_r = r_cur.copy()
                    else:
                        c_cur = old_c
                else:
                    c_cur = old_c
        except Exception:
            c_cur = old_c

    # --- Phase 3: Joint SLSQP Polish ---
    def joint_obj(v):
        return -np.sum(v[2 * n:])
        
    def joint_cons(v):
        cx, cy, r = v[:n], v[n:2 * n], v[2 * n:]
        cons = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
        idx = np.triu_indices(n, 1)
        dx = cx[idx[0]] - cx[idx[1]]
        dy = cy[idx[0]] - cy[idx[1]]
        d2 = dx**2 + dy**2
        rs = r[idx[0]] + r[idx[1]]
        cons = np.concatenate([cons, d2 - rs**2])
        return cons

    v0 = np.concatenate([best_c[:, 0], best_c[:, 1], best_r * 0.999])
    bounds_slqp = [(0, 1)] * (2 * n) + [(1e-6, 0.5)] * n
    
    for _ in range(3):
        try:
            res = minimize(joint_obj, v0, args=(), method='SLSQP', bounds=bounds_slqp,
                           constraints={'type': 'ineq', 'fun': joint_cons},
                           options={'maxiter': 3000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_p = np.column_stack((res.x[:n], res.x[n:2 * n]))
                r_p, s_p, _ = solve_lp_and_grad(c_p, A_ub, pairs, n, m)
                if r_p is not None and s_p > best_s:
                    best_s = s_p
                    best_c = c_p.copy()
                    best_r = r_p.copy()
        except Exception:
            pass

    # --- Phase 4: Strict Safety Scaling ---
    if best_c is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                rs = best_r[i] + best_r[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)
        best_r *= scale * 0.9999999
        best_s = float(np.sum(best_r))
    else:
        best_c = rng.uniform(0.1, 0.9, (n, 2))
        best_r = np.full(n, 0.05)
        best_s = float(np.sum(best_r))
        
    return best_c, best_r, best_s
