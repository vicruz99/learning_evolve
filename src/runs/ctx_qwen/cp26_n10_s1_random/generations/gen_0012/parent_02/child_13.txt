from scipy.optimize import minimize, linprog

def build_lp_matrices(n):
    m = 4*n + n*(n-1)//2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k+=1
        A[k, i] = 1.0; k+=1
        A[k, i] = 1.0; k+=1
        A[k, i] = 1.0; k+=1
    for i in range(n):
        for j in range(i+1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k+=1
    return A

def solve_lp_radii(centers, n, A_ub):

    k = 0
    for i in range(n):

        b[k] = x; k+=1
        b[k] = 1.0 - x; k+=1
        b[k] = y; k+=1
        b[k] = 1.0 - y; k+=1
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b[k] = np.hypot(dx, dy); k+=1

    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_joint(v, n):
    return -np.sum(v[2*n:])

def cons_joint(v, n):

    dy = cy[idx[0]] - cy[idx[1]]
    dr = r[idx[0]] + r[idx[1]]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
