import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_and_grad(centers, n, pairs, A_ub, m_con):
    """Solves LP for radii and computes gradient of sum of radii w.r.t centers."""
    b = np.empty(m_con)
    for i in range(n):
        x, y = centers[i]
        b[i] = x
        b[n + i] = 1.0 - x
        b[2 * n + i] = y
        b[3 * n + i] = 1.0 - y
    for k, (i, j) in enumerate(pairs):
        b[4 * n + k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            return np.full(n, 0.0), 0.0, None
    except Exception:
        return np.full(n, 0.0), 0.0, None
        
    r = res.x
    s = -res.fun
    grad = np.zeros((n, 2))
    try:
        marg = res.ineqlin.marginals
        for i in range(n):
            # Gradient from wall constraints: dS/dx = lambda_right - lambda_left
            grad[i, 0] += marg[n + i] - marg[i]
            grad[i, 1] += marg[3 * n + i] - marg[2 * n + i]
        for k, (i, j) in enumerate(pairs):
            lam = marg[4 * n + k]
            if lam > 1e-8:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)
                if d < 1e-9: continue
                inv_d = lam / d
                grad[i, 0] += dx * inv_d
                grad[i, 1] += dy * inv_d
                grad[j, 0] -= dx * inv_d
                grad[j, 1] -= dy * inv_d
    except Exception:
        pass
    return r, s, grad

def objective_grad(x_flat, n, pairs, A_ub, m_con):
    """Objective and gradient for L-BFGS-B: minimizes negative sum of radii."""
    centers = x_flat.reshape(n, 2)
    _, s, grad = solve_lp_and_grad(centers, n, pairs, A_ub, m_con)
    return -s, -grad.flatten()

def run_packing():
    n = 26
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    n_pairs = len(pairs)
    m_con = 4 * n + n_pairs
    
    # Precompute constant part of LP constraint matrix
    A_ub = np.zeros((m_con, n))
    for i in range(n):
        A_ub[i, i] = 1.0
        A_ub[n + i, i] = 1.0
        A_ub[2 * n + i, i] = 1.0
        A_ub[3 * n + i, i] = 1.0
    for k, (i, j) in enumerate(pairs):
        A_ub[4 * n + k, i] = 1.0
        A_ub[4 * n + k, j] = 1.0
        
    rng = np.random.default_rng(42)
    best_s = 0.0
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    configs = []
    hex_patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], 
        [4,6,6,6,4], [7,6,5,5,3], [8,5,6,5], [6,7,5,5,3], [5,4,6,6,5]
    ]
    for pat in hex_patterns:
        if sum(pat) < n: continue
        r0 = 0.10
        pts = []
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        configs.append(np.array(pts[:n]))
        
    # Random dense starts
    for _ in range(15):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Regular grid + center
    grid = np.array([(i*0.2+0.1, j*0.2+0.1) for j in range(5) for i in range(5)])
    grid = np.vstack([grid, [0.5, 0.5]])
    configs.append(grid)
    
    # Corner-focused starts to exploit boundary slack
    for _ in range(5):
        pts = [rng.uniform(0.05, 0.25, 2), rng.uniform(0.75, 0.95, 2),
               rng.uniform(0.05, 0.25, 2), rng.uniform(0.75, 0.95, 2)]
        pts += list(rng.uniform(0.2, 0.8, (n-4, 2)))
        configs.append(np.array(pts))

    bounds_c = [(0.005, 0.995)] * (2 * n)
    
    # Phase 1: Multi-start gradient ascent on centers
    for cfg in configs:
        x0 = np.clip(cfg, 0.01, 0.99).flatten()
        try:
            res = minimize(objective_grad, x0, jac=True, method='L-BFGS-B', 
                           args=(n, pairs, A_ub, m_con), bounds=bounds_c,
                           options={'maxiter': 2500, 'ftol': 1e-13, 'gtol': 1e-12})
            if np.isfinite(res.fun):
                s = -res.fun
                centers_opt = res.x.reshape(n, 2)
                r_opt, _, _ = solve_lp_and_grad(centers_opt, n, pairs, A_ub, m_con)
                if s > best_s:
                    best_s = s
                    best_c = centers_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Iterative perturbation & local refinement on best configuration
    if best_c is not None:
        curr_c = best_c.copy()
        curr_r = best_r.copy()
        curr_s = best_s
        step = 0.025
        
        for epoch in range(12):
            improved = False
            for _ in range(150):
                num_perturb = rng.integers(1, n//2 + 2)
                idx = rng.choice(n, num_perturb, replace=False)
                old_c = curr_c[idx].copy()
                curr_c[idx] += rng.uniform(-step, step, idx.shape)
                curr_c = np.clip(curr_c, 0.01, 0.99)
                
                try:
                    # Short local optimization from perturbed state
                    res_p = minimize(objective_grad, curr_c.flatten(), jac=True, method='L-BFGS-B', 
                                     args=(n, pairs, A_ub, m_con), bounds=bounds_c,
                                     options={'maxiter': 400, 'ftol': 1e-12})
                    if np.isfinite(res_p.fun):
                        s_p = -res_p.fun
                        if s_p > curr_s + 1e-8:
                            curr_c = res_p.x.reshape(n, 2)
                            curr_r, _, _ = solve_lp_and_grad(curr_c, n, pairs, A_ub, m_con)
                            curr_s = s_p
                            improved = True
                            step = min(step * 1.03, 0.05)
                        else:
                            curr_c[idx] = old_c
                            step *= 0.995
                except Exception:
                    curr_c[idx] = old_c
                    
            if not improved:
                step *= 0.5
                
        best_c = curr_c
        best_r = curr_r
        best_s = curr_s
    else:
        # Fallback
        best_c = rng.uniform(0.2, 0.8, (n, 2))
        best_r, best_s, _ = solve_lp_and_grad(best_c, n, pairs, A_ub, m_con)

    # Phase 3: Strict numerical safety scaling
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
    best_r *= scale * 0.9999999
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
