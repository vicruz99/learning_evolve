import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_structure(n):
    """Precomputes constant matrices for the LP radius solver."""
    m_boundary = 4 * n
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    n_pairs = len(pairs)
    m_total = m_boundary + n_pairs
    
    A_ub = np.zeros((m_total, n))
    for i in range(n):
        A_ub[i, i] = 1.0
        A_ub[n + i, i] = 1.0
        A_ub[2 * n + i, i] = 1.0
        A_ub[3 * n + i, i] = 1.0
    for k, (i, j) in enumerate(pairs):
        A_ub[m_boundary + k, i] = 1.0
        A_ub[m_boundary + k, j] = 1.0
        
    return A_ub, pairs, m_boundary, n_pairs

def solve_lp_and_grad(centers, A_ub, pairs, m_boundary, n_pairs, n):
    """Solves LP for radii and computes subgradient w.r.t centers."""
    m_total = A_ub.shape[0]
    b = np.zeros(m_total)
    
    # Boundary bounds
    for i in range(n):
        x, y = centers[i]
        b[i] = x
        b[n + i] = 1.0 - x
        b[2 * n + i] = y
        b[3 * n + i] = 1.0 - y
        
    # Pairwise distance bounds
    for k, (i, j) in enumerate(pairs):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b[m_boundary + k] = np.hypot(dx, dy)
        
    bounds_r = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=bounds_r, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, 0.0, None
            
        r = res.x
        s = -res.fun
        grad_c = np.zeros((n, 2))
        
        marg = None
        try:
            marg = res.ineqlin.marginals
        except AttributeError:
            pass
            
        if marg is not None:
            for i in range(n):
                grad_c[i, 0] += marg[i] - marg[n + i]
                grad_c[i, 1] += marg[2 * n + i] - marg[3 * n + i]
                
            for k, (i, j) in enumerate(pairs):
                lam = marg[m_boundary + k]
                if lam > 1e-9:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = b[m_boundary + k]
                    d_safe = max(d, 1e-9)
                    fx = lam * dx / d_safe
                    fy = lam * dy / d_safe
                    grad_c[i, 0] += fx
                    grad_c[i, 1] += fy
                    grad_c[j, 0] -= fx
                    grad_c[j, 1] -= fy
                    
        return r, s, -grad_c.flatten() # Negative because we minimize -sum(r)
    except Exception:
        return None, 0.0, None

def obj_lp_only(c_flat, args):
    n, A_ub, pairs, m_b, n_p = args
    centers = c_flat.reshape(n, 2)
    _, s, _ = solve_lp_and_grad(centers, A_ub, pairs, m_b, n_p, n)
    return -s if s > 0 else 1e6
