import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix_and_pairs(n):
    """Precomputes the constant LP constraint matrix and pair indices."""
    num_pairs = n * (n - 1) // 2
    pairs = np.zeros((num_pairs, 2), dtype=int)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            pairs[k] = (i, j)
            k += 1
            
    m = 4 * n + num_pairs
    A = np.zeros((m, n))
    for i in range(n):
        A[i, i] = 1.0
        A[n + i, i] = 1.0
        A[2 * n + i, i] = 1.0
        A[3 * n + i, i] = 1.0
    for k, (i, j) in enumerate(pairs):
        A[4 * n + k, i] = 1.0
        A[4 * n + k, j] = 1.0
    return A, pairs

def solve_lp(centers, A_ub, n, pairs):
    """Solves LP to maximize sum of radii for fixed centers (vectorized)."""
    m = A_ub.shape[0]
    b = np.zeros(m)
    
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    x = centers[:, 0]
    y = centers[:, 1]
    b[:n] = x
    b[n:2*n] = 1.0 - x
    b[2*n:3*n] = y
    b[3*n:4*n] = 1.0 - y
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[pairs[:, 0], 0] - centers[pairs[:, 1], 0]
    dy = centers[pairs[:, 0], 1] - centers[pairs[:, 1], 1]
    b[4*n:] = np.hypot(dx, dy)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-6)

def objective_joint(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def constraints_joint(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    con = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dist_sq = dx**2 + dy**2
    r_sum = r[i_idx] + r[j_idx]
    con = np.concatenate([con, dist_sq - r_sum**2])
    return con

def simulate_pack(centers, radii, steps=800):
    """Force-directed simulation to densely pack circles."""
    n = len(radii)
    vel = np.zeros_like(centers)
    dt = 0.004
    damp = 0.85
    for step in range(steps):
        radii *= 1.00008
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dists, np.inf)
        r_sums = radii[:, None] + radii[None, :]
        overlaps = np.maximum(0.0, r_sums - dists)
        inv_dists = 1.0 / dists
        force_mag = overlaps * 400.0 * inv_dists
        forces = np.sum(diff * force_mag[:, :, None], axis=1)
        
        forces[:, 0] += np.clip(radii - centers[:, 0], 0, None) * 400.0
        forces[:, 0] -= np.clip(centers[:, 0] + radii - 1.0, 0, None) * 400.0
        forces[:, 1] += np.clip(radii - centers[:, 1], 0, None) * 400.0
        forces[:, 1] -= np.clip(centers[:, 1] + radii - 1.0, 0, None) * 400.0
        
        vel = damp * vel + forces * dt
        centers += vel
        centers = np.clip(centers, 0.001, 0.999)
    return centers, radii
