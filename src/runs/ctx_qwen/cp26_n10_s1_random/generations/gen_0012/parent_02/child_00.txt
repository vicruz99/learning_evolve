2. **Dual Gradient Ascent**: The gradient of the LP objective (sum of radii) with respect to the centers can be computed exactly using the dual variables (marginals/Lagrange multipliers) from the LP solution. import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_data(n):
    """Builds the constant inequality matrix for the LP radius solver."""
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    n_pairs = len(pairs)
    m_con = 4 * n + n_pairs
    A_ub = np.zeros((m_con, n))
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        A_ub[i, i] = 1.0
        A_ub[n + i, i] = 1.0
        A_ub[2 * n + i, i] = 1.0
        A_ub[3 * n + i, i] = 1.0
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for k, (i, j) in enumerate(pairs):
        A_ub[4 * n + k, i] = 1.0
        A_ub[4 * n + k, j] = 1.0
    return pairs, m_con, A_ub

def solve_lp(centers, n, pairs, m_con, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    b = np.zeros(m_con)
    for i in range(n):
        x, y = centers[i]
        b[i] = x
        b[n + i] = 1.0 - x
        b[2 * n + i] = y
        b[3 * n + i] = 1.0 - y
        
    for k, (i, j) in enumerate(pairs):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b[4 * n + k] = np.hypot(dx, dy)
        
    bounds_r = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=bounds_r, method='highs')
        if res.success and np.isfinite(res.fun):
            marg = None
            try:
                marg = res.ineqlin.marginals
            except AttributeError:
                try:
                    marg = res.marginals.ineqlin
                except AttributeError:
                    pass
            return res.x, -res.fun, marg
    except Exception:
        pass
    return np.zeros(n), 0.0, None

def lp_objective_and_grad(centers_flat, n, pairs, m_con, A_ub):
    """Objective and gradient for L-BFGS-B: minimizes negative sum of radii."""
    centers = centers_flat.reshape(n, 2)
    r, s, marg = solve_lp(centers, n, pairs, m_con, A_ub)
    
    if s == 0.0:
        return 1e6, np.zeros_like(centers_flat)
        
    grad = np.zeros((n, 2))
    if marg is not None:
        # Boundary gradient contributions
        for i in range(n):
            grad[i, 0] += marg[i] - marg[n + i]
            grad[i, 1] += marg[2 * n + i] - marg[3 * n + i]
            
        # Pairwise gradient contributions
        for k, (i, j) in enumerate(pairs):
            lam = marg[4 * n + k]
            if lam > 1e-8:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)
                if d < 1e-9:
                    continue
                fx, fy = lam * dx / d, lam * dy / d
                grad[i, 0] += fx
                grad[i, 1] += fy
                grad[j, 0] -= fx
                grad[j, 1] -= fy
                
    # Return negative sum and negative gradient for minimization
    return -s, -grad.flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    pairs, m_con, A_ub = build_lp_data(n)
    rng = np.random.default_rng(42)
    
    best_s = 0.0
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    configs = []
    pats = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
        [4, 6, 6, 6, 4], [5, 5, 6, 5, 5], [7, 6, 6, 7], 
        [5, 7, 5, 5, 4], [6, 5, 5, 6, 4], [8, 6, 6, 6]
    ]
    for pat in pats:
        if sum(pat) < n:
            continue
        pts = []
        r0 = 0.10
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n:
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        configs.append(np.array(pts[:n]))
        
    for _ in range(25):
        configs.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    bounds_c = [(0.001, 0.999)] * (2 * n)
    
    # Phase 1: Local optimization from diverse starts
    for cfg in configs:
        c0 = np.clip(cfg, 0.005, 0.995)
        try:
            res = minimize(lp_objective_and_grad, c0.flatten(), args=(n, pairs, m_con, A_ub),
                           method='L-BFGS-B', jac=True, bounds=bounds_c,
                           options={'maxiter': 8000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(n, 2)
                r_opt, s_opt, _ = solve_lp(c_opt, n, pairs, m_con, A_ub)
                if s_opt > best_s:
                    best_s = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            continue
            
    # Phase 2: Perturbation & Annealing to escape local minima
    if best_c is not None:
        step = 0.025
        no_improve = 0
        for _ in range(600):
            num_pert = rng.integers(1, 5)
            idx = rng.choice(n, num_pert, replace=False)
            old_c = best_c.copy()
            
            best_c[idx] += rng.uniform(-step, step, (num_pert, 2))
            best_c = np.clip(best_c, 0.002, 0.998)
            
            try:
                res_p = minimize(lp_objective_and_grad, best_c.flatten(), args=(n, pairs, m_con, A_ub),
                                 method='L-BFGS-B', jac=True, bounds=bounds_c,
                                 options={'maxiter': 1500, 'ftol': 1e-13})
                if np.isfinite(res_p.fun):
                    c_p = res_p.x.reshape(n, 2)
                    r_p, s_p, _ = solve_lp(c_p, n, pairs, m_con, A_ub)
                    if s_p > best_s + 1e-8:
                        best_s = s_p
                        best_c = c_p.copy()
                        best_r = r_p.copy()
                        step = min(step * 1.02, 0.06)
                        no_improve = 0
                    else:
                        best_c = old_c.copy()
                        step *= 0.995
                        no_improve += 1
            except Exception:
                best_c = old_c.copy()
                
            if no_improve > 150:
                break
                
    # Phase 3: Strict numerical safety scaling
    if best_c is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
                
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                rs = best_r[i] + best_r[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)
                    
        best_r *= scale * 0.9999999
        best_s = float(np.sum(best_r))
    else:
        # Fallback
        best_c = np.random.uniform(0.1, 0.9, (n, 2))
        best_r = np.full(n, 0.05)
        best_s = float(np.sum(best_r))
        
    return best_c, best_r, best_s
