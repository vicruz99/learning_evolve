def solve_lp(centers, n, A_ub, m_con, triu_idx):

    b = np.zeros(m_con)
    for i in range(n):
        x, y = centers[i]
        b[i] = x
        b[n + i] = 1.0 - x
        b[2*n + i] = y
        b[3*n + i] = 1.0 - y

    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.sqrt(dx**2 + dy**2)

            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def joint_obj(v, n):

    return -np.sum(v[2*n:])

    dy = cy[triu_idx[0]] - cy[triu_idx[1]]
    rs = r[triu_idx[0]] + r[triu_idx[1]]
