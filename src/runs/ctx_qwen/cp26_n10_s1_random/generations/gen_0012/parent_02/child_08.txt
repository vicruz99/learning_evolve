import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers, n, pairs, A_ub, m_con):
    """Solves LP to maximize sum of radii for fixed centers."""
    b = np.empty(m_con)
    for i in range(n):
        x, y = centers[i]
        b[i] = x
        b[n + i] = 1.0 - x
        b[2 * n + i] = y
        b[3 * n + i] = 1.0 - y
    for k, (i, j) in enumerate(pairs):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b[4 * n + k] = np.hypot(dx, dy)
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_i(p, i, centers, n, pairs, A_ub, m_con):
    """Objective for coordinate descent: minimize negative sum of radii."""
    centers[i] = np.clip(p, 1e-5, 1.0 - 1e-5)
    r, s = solve_lp(centers, n, pairs, A_ub, m_con)
    if r is None:
        return 1e6
    return -s

def joint_obj(v, n):
    return -np.sum(v[2 * n:])

def joint_cons(v, n):
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    cons = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    return np.concatenate([cons, dx**2 + dy**2 - dr**2])
