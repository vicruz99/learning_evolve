import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, A_ub, idx_wall, idx_pair, pairs, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    m_total = 4 * n + len(pairs)
    b_ub = np.zeros(m_total)
    
    # Boundary bounds
    for i in range(n):
        x, y = centers[i]
        b_ub[idx_wall[(i, 'x')]] = x
        b_ub[idx_wall[(i, 'X')]] = 1.0 - x
        b_ub[idx_wall[(i, 'y')]] = y
        b_ub[idx_wall[(i, 'Y')]] = 1.0 - y
        
    # Pairwise distance bounds (vectorized)
    dist_matrix = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=2))
    triu_idx = np.triu_indices(n, k=1)
    b_ub[idx_pair[(i, j)] for i, j in pairs] = dist_matrix[triu_idx]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def obj_slqp(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def cons_slqp(v, n):
    """Inequality constraints >= 0 for valid packing in SLSQP."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    # Boundary constraints
    cons = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap constraints (squared)
    triu = np.triu_indices(n, k=1)
    dx = cx[triu[0]] - cx[triu[1]]
    dy = cy[triu[0]] - cy[triu[1]]
    d2 = dx**2 + dy**2
    rs = r[triu[0]] + r[triu[1]]
    cons = np.concatenate([cons, d2 - rs**2])
    return cons
