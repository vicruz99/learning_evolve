import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers, A_ub, m_con, n, pairs):
    """Solves LP to maximize sum of radii for fixed centers."""
    b = np.zeros(m_con)
    for i in range(n):
        x, y = centers[i]
        b[i] = x
        b[n + i] = 1.0 - x
        b[2 * n + i] = y
        b[3 * n + i] = 1.0 - y
        
    for k, (i, j) in enumerate(pairs):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b[4 * n + k] = np.hypot(dx, dy)
        
    bounds_r = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=bounds_r, method='highs')
        if res.success and np.isfinite(res.fun):
            marg = None
            if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
                marg = res.ineqlin.marginals
            elif hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
                marg = res.marginals.ineqlin
            return res.x, -res.fun, marg
    except Exception:
        pass
    return None, 0.0, None

def optimize_centers(c_init, n, pairs, A_ub, m_con, rng):
    """Gradient ascent on centers using LP duals with line search."""
    cur_c = np.clip(c_init, 0.005, 0.995).copy()
    step = 0.02
    for it in range(500):
        r, s, marg = solve_lp(cur_c, A_ub, m_con, n, pairs)
        if r is None:
            break
            
        grad = np.zeros((n, 2))
        if marg is not None:
            for i in range(n):
                grad[i, 0] += marg[i] - marg[n + i]
                grad[i, 1] += marg[2 * n + i] - marg[3 * n + i]
            for k, (i, j) in enumerate(pairs):
                lam = marg[4 * n + k]
                if lam > 1e-8:
                    d = np.hypot(cur_c[i, 0] - cur_c[j, 0], cur_c[i, 1] - cur_c[j, 1])
                    if d > 1e-9:
                        dx = cur_c[i, 0] - cur_c[j, 0]
                        dy = cur_c[i, 1] - cur_c[j, 1]
                        fx = lam * dx / d
                        fy = lam * dy / d
                        grad[i, 0] += fx
                        grad[i, 1] += fy
                        grad[j, 0] -= fx
                        grad[j, 1] -= fy
                        
        if np.linalg.norm(grad) < 1e-9:
            break
            
        # Adaptive line search
        alpha = step
        accepted = False
        for _ in range(5):
            new_c = cur_c + alpha * grad
            new_c = np.clip(new_c, 0.005, 0.995)
            _, s_new, _ = solve_lp(new_c, A_ub, m_con, n, pairs)
            if s_new > s - 1e-10:
                cur_c = new_c
                accepted = True
                step = min(alpha * 1.15, 0.04)
                break
            alpha *= 0.5
        if not accepted:
            step *= 0.5
            if step < 1e-7:
                break
                
        # Periodic jitter to escape local minima
        if it % 150 == 0 and it > 0:
            cur_c += rng.uniform(-0.003, 0.003, (n, 2))
            cur_c = np.clip(cur_c, 0.005, 0.995)
            
    return cur_c

def joint_obj(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def joint_cons(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    cons = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    idx = np.triu_indices(n, k=1)
    dx = cx[idx[0]] - cx[idx[1]]
    dy = cy[idx[0]] - cy[idx[1]]
    rs = r[idx[0]] + r[idx[1]]
    return np.concatenate([cons, dx**2 + dy**2 - rs**2])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    n_pairs = len(pairs)
    m_con = 4 * n + n_pairs
    A_ub = np.zeros((m_con, n))
    
    for i in range(n):
        A_ub[i, i] = 1.0
        A_ub[n + i, i] = 1.0
        A_ub[2 * n + i, i] = 1.0
        A_ub[3 * n + i, i] = 1.0
    for k, (i, j) in enumerate(pairs):
        A_ub[4 * n + k, i] = 1.0
        A_ub[4 * n + k, j] = 1.0
        
    rng = np.random.default_rng(42)
    best_s = -1.0
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    configs = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],

        [6, 5, 5, 6, 4], [7, 6, 6, 7]
    ]
    
    for pat in patterns:
        if sum(pat) < n:
            continue
        r0 = 0.10
        pts = []
        y = r0

            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n:
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        configs.append(np.array(pts[:n]))
        
    for _ in range(12):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))

            best_c = c_opt.copy()
            best_r = r_opt.copy()

            if r_p is not None and s_p > best_s + 1e-9:
                best_s = s_p
                best_r = r_p.copy()
            else:
                best_c[i] = old
            step *= 0.9995

                res = minimize(joint_obj, v0, args=(n,), method='SLSQP',

                               constraints={'type': 'ineq', 'fun': joint_cons, 'args': (n,)},
                               options={'maxiter': 4000, 'ftol': 1e-13})

                    c_j = np.column_stack((res.x[:n], res.x[n:2 * n]))
                    r_j, s_j, _ = solve_lp(c_j, A_ub, m_con, n, pairs)
                    if r_j is not None and s_j > best_s:
                        best_s = s_j
                        best_c = c_j.copy()
                        best_r = r_j.copy()
                    v0 = np.concatenate([best_c[:, 0], best_c[:, 1], best_r * 0.998])
            except Exception:
                pass

    # Phase 4: Strict numerical safety scaling
    if best_c is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                rs = best_r[i] + best_r[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)
        best_r *= scale * 0.9999999
        best_s = float(np.sum(best_r))
    else:
        best_c = rng.uniform(0.1, 0.9, (n, 2))
        best_r = np.full(n, 0.05)
        best_s = float(np.sum(best_r))
        
    return best_c, best_r, best_s
