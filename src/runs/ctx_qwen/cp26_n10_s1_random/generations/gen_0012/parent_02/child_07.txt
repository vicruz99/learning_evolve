import numpy as np
from scipy.optimize import linprog, minimize

def get_lp_matrix(n):
    """Precomputes the constant inequality matrix for the LP radius solver."""
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    m = 4 * n + len(pairs)
    A_ub = np.zeros((m, n))
    for i in range(n):
        A_ub[i, i] = 1.0
        A_ub[n + i, i] = 1.0
        A_ub[2 * n + i, i] = 1.0
        A_ub[3 * n + i, i] = 1.0
    for k, (i, j) in enumerate(pairs):
        A_ub[4 * n + k, i] = 1.0
        A_ub[4 * n + k, j] = 1.0
    return A_ub, pairs

def solve_lp(centers, n, A_ub, pairs, b_buffer):
    """Solves LP to maximize sum of radii for fixed centers."""
    for i in range(n):
        x, y = centers[i]
        b_buffer[i] = x
        b_buffer[n + i] = 1.0 - x
        b_buffer[2 * n + i] = y
        b_buffer[3 * n + i] = 1.0 - y
    for k, (i, j) in enumerate(pairs):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b_buffer[4 * n + k] = np.hypot(dx, dy)
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_buffer, bounds=(0, None), method='highs')
    if res.success and np.isfinite(res.fun):
        marg = None
        try:
            marg = res.ineqlin.marginals
        except AttributeError:
            try:
                marg = res.marginals.ineqlin
            except AttributeError:
                pass
        return res.x, -res.fun, marg
    return None, 0.0, None

def compute_obj_grad(x_flat, n, A_ub, pairs, b_buffer):
    """Objective and gradient for L-BFGS-B: minimizes negative sum of radii."""
    centers = x_flat.reshape(n, 2)
    r, s, marg = solve_lp(centers, n, A_ub, pairs, b_buffer)
    if r is None:
        return 1e6, np.zeros_like(x_flat)
        
    grad = np.zeros((n, 2))
    if marg is not None:
        for i in range(n):
            grad[i, 0] += marg[i] - marg[n + i]
            grad[i, 1] += marg[2 * n + i] - marg[3 * n + i]
        for k, (i, j) in enumerate(pairs):
            lam = marg[4 * n + k]
            if lam > 1e-8:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy) + 1e-9
                fx = lam * dx / d
                fy = lam * dy / d
                grad[i, 0] += fx
                grad[i, 1] += fy
                grad[j, 0] -= fx
                grad[j, 1] -= fy
    return -s, -grad.flatten()

def joint_obj(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def joint_cons(v, n):
    """Inequality constraints >= 0 for valid packing in SLSQP."""
    cx_v = v[:n]
    cy_v = v[n:2 * n]
    r_v = v[2 * n:]
    cons = np.concatenate([cx_v - r_v, 1.0 - cx_v - r_v, cy_v - r_v, 1.0 - cy_v - r_v])
    idx = np.triu_indices(n, k=1)
    dx = cx_v[idx[0]] - cx_v[idx[1]]
    dy = cy_v[idx[0]] - cy_v[idx[1]]
    dr = r_v[idx[0]] + r_v[idx[1]]
    cons = np.concatenate([cons, dx**2 + dy**2 - dr**2])
    return cons

def run_packing():
    n = 26
    A_ub, pairs = get_lp_matrix(n)
    m = A_ub.shape[0]
    b_buffer = np.zeros(m)
    rng = np.random.default_rng(42)
    
    best_s = -1.0
    best_c = None
    best_r = None
    
    bounds_c = [(0.005, 0.995)] * (2 * n)
    
    # Generate diverse initial configurations
    configs = []
    row_pats = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 6, 6, 4], 
        [5, 5, 6, 5, 5], [6, 6, 5, 5, 4], [5, 4, 6, 6, 5],
        [7, 5, 5, 5, 4], [5, 6, 5, 5, 5], [6, 5, 5, 6, 4]
    ]
    
    for pat in row_pats:
        if sum(pat) < n: continue
        r0 = 0.10
        pts = []
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):

                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        pts = np.array(pts[:n])

            configs.append(np.clip(pts + rng.uniform(-0.02, 0.02, pts.shape), 0.05, 0.95))

        configs.append(rng.uniform(0.15, 0.85, (n, 2)))

    for cfg in configs:

                           args=(n, A_ub, pairs, b_buffer),

                           options={'maxiter': 1500, 'ftol': 1e-12})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(n, 2)
                r_opt, s_opt, _ = solve_lp(c_opt, n, A_ub, pairs, b_buffer)

            v_start = v0.copy()

                res_j = minimize(joint_obj, v_start, args=(n,), method='SLSQP', bounds=bounds_slqp,
                                 constraints={'type': 'ineq', 'fun': joint_cons, 'args': (n,)},
                                 options={'maxiter': 3000, 'ftol': 1e-13})
                if np.isfinite(res_j.fun):
                    c_j = np.column_stack((res_j.x[:n], res_j.x[n:2 * n]))
                    r_j, s_j, _ = solve_lp(c_j, n, A_ub, pairs, b_buffer)
                    if r_j is not None and s_j > best_s:
                        best_s = s_j
                        best_c = c_j.copy()

            except Exception:
                pass

    # Final strict safety scaling to guarantee numerical validity
    if best_c is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                rs = best_r[i] + best_r[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)
        best_r *= scale * 0.9999999
        best_s = float(np.sum(best_r))
