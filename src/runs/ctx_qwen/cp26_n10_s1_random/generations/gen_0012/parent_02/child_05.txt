import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_structure(n):
    """Precomputes the constant inequality matrix structure for the LP radius solver."""
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    n_pairs = len(pairs)
    m = 4 * n + n_pairs
    
    A_ub = np.zeros((m, n))
    for i in range(n):
        A_ub[i, i] = 1.0
        A_ub[n + i, i] = 1.0
        A_ub[2 * n + i, i] = 1.0
        A_ub[3 * n + i, i] = 1.0
    for k, (i, j) in enumerate(pairs):
        A_ub[4 * n + k, i] = 1.0
        A_ub[4 * n + k, j] = 1.0
        
    return A_ub, pairs

def solve_lp_and_grad(centers, A_ub, pairs, n):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns optimal radii, sum of radii, and gradient of sum w.r.t centers using LP duals.
    """
    m = A_ub.shape[0]
    b = np.empty(m)
    
    # Boundary bounds: r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n):
        x, y = centers[i]
        b[i] = x
        b[n + i] = 1.0 - x
        b[2 * n + i] = y
        b[3 * n + i] = 1.0 - y
        
    # Pairwise distance bounds: r_i + r_j <= dist(i,j)
    for k, (i, j) in enumerate(pairs):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b[4 * n + k] = np.hypot(dx, dy)
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None
            
        radii = res.x
        sum_r = -res.fun
        
        # Compute gradient using dual variables (marginals)
        grad = np.zeros((n, 2))
        try:
            marg = res.ineqlin.marginals
        except AttributeError:
            try:
                marg = res.marginals.ineqlin
            except AttributeError:
                marg = None
                
        if marg is not None:
            # Boundary contributions
            for i in range(n):
                mu_x = marg[i]
                mu_1x = marg[n + i]
                mu_y = marg[2 * n + i]
                mu_1y = marg[3 * n + i]
                grad[i, 0] += mu_x - mu_1x
                grad[i, 1] += mu_y - mu_1y
                
            # Pairwise distance contributions
            for k, (i, j) in enumerate(pairs):
                lam = marg[4 * n + k]
                if lam > 1e-9:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = b[4 * n + k]
                    if d > 1e-9:
                        fx = lam * dx / d
                        fy = lam * dy / d
                        grad[i, 0] += fx
                        grad[i, 1] += fy
                        grad[j, 0] -= fx
                        grad[j, 1] -= fy
                        
        return radii, sum_r, grad
    except Exception:
        return None, None, None

def objective_lp(v, args):
    """Objective and Jacobian for L-BFGS-B: minimizes negative sum of radii."""
    n, A_ub, pairs = args
    centers = v.reshape(n, 2)
    _, s, g = solve_lp_and_grad(centers, A_ub, pairs, n)
    if s is None:
        return 1e6, np.zeros_like(v)
    return -s, -g.flatten()
