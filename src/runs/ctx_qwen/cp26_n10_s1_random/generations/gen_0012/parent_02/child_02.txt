import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_data(n):
    """Precompute LP structure for n circles."""
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    n_pairs = len(pairs)
    m_con = 4 * n + n_pairs
    A_ub = np.zeros((m_con, n))
    for i in range(n):
        A_ub[i, i] = 1.0
        A_ub[n + i, i] = 1.0
        A_ub[2 * n + i, i] = 1.0
        A_ub[3 * n + i, i] = 1.0
    for k, (i, j) in enumerate(pairs):
        A_ub[4 * n + k, i] = 1.0
        A_ub[4 * n + k, j] = 1.0
    return A_ub, pairs, m_con

def solve_lp_and_grad(c_flat, n, pairs, A_ub, m_con):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns objective value (-sum_r) and gradient w.r.t centers for minimization.
    """
    c = c_flat.reshape(n, 2)
    b = np.zeros(m_con)
    for i in range(n):
        x, y = c[i]
        b[i] = x
        b[n + i] = 1.0 - x
        b[2 * n + i] = y
        b[3 * n + i] = 1.0 - y
    for k, (i, j) in enumerate(pairs):
        b[4 * n + k] = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0.0, None), method='highs')
    if not res.success or not np.isfinite(res.fun):
        return 1e6, np.zeros_like(c_flat)
        
    s = -res.fun
    grad = np.zeros((n, 2))
    
    try:
        marg = res.ineqlin.marginals
    except AttributeError:
        try:
            marg = res.marginals.ineqlin
        except AttributeError:
            marg = None

                fy = lam * dy / d
                grad[i, 0] += fx
                grad[i, 1] += fy
                grad[j, 0] -= fx
                grad[j, 1] -= fy
                
    return -s, -grad.flatten()

def solve_lp_only(c, n, pairs, A_ub, m_con):
    """Solves LP and returns radii and sum, or None if failed."""
    b = np.zeros(m_con)
    for i in range(n):
        x, y = c[i]
        b[i] = x
        b[n + i] = 1.0 - x
        b[2 * n + i] = y
        b[3 * n + i] = 1.0 - y
    for k, (i, j) in enumerate(pairs):
        b[4 * n + k] = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0.0, None), method='highs')
    if not res.success or not np.isfinite(res.fun):
        return None, 0.0
    return res.x, -res.fun

def hex_init(n, rows, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:n])

def joint_obj(v, n):
    return -np.sum(v[2 * n:])

def joint_cons(v, n):
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    cons = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    triu_idx = np.triu_indices(n, k=1)
    dx = cx[triu_idx[0]] - cx[triu_idx[1]]
    dy = cy[triu_idx[0]] - cy[triu_idx[1]]
    d2 = dx ** 2 + dy ** 2
    rs = r[triu_idx[0]] + r[triu_idx[1]]
    return np.concatenate([cons, d2 - rs ** 2])
