import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]), 
                      np.minimum(centers[:,1], 1.0-centers[:,1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    m = len(i_idx)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), i_idx] = 1.0
    A_ub[np.arange(m), j_idx] = 1.0
    b_ub = dists[i_idx, j_idx]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def force_simulate(centers, steps=1200):
    """Force-directed simulation to separate circles and find dense layouts."""
    n = centers.shape[0]
    c = centers.copy()
    r = np.full(n, 0.05)
    vel = np.zeros_like(c)
    dt = 0.005
    for _ in range(steps):
        r *= 1.00005
        diff = c[:, None, :] - c[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dist, 1e9)
        overlap = np.maximum(0.0, r[:, None] + r[None, :] - dist)
        f_pair = overlap * 150.0 / (dist + 1e-9)
        forces = np.sum(diff * f_pair[:, :, None], axis=1)
        
        forces[:, 0] += np.clip(r - c[:, 0], 0, None) * 300.0
        forces[:, 0] -= np.clip(c[:, 0] + r - 1.0, 0, None) * 300.0
        forces[:, 1] += np.clip(r - c[:, 1], 0, None) * 300.0
        forces[:, 1] -= np.clip(c[:, 1] + r - 1.0, 0, None) * 300.0
        
        vel = vel * 0.82 + forces * dt
        c += vel
        c = np.clip(c, 0.002, 0.998)
    return c, r

def hill_climb(centers, radii, current_sum, rng, steps=3000):
    """Iterative perturbation of centers optimized by LP."""
    n = len(centers)
    best_c = centers.copy()
    best_r = radii.copy()
    best_s = current_sum
    step = 0.025
    
    for _ in range(steps):
        i = rng.integers(n)
        old_c = best_c[i].copy()
        best_c[i] += rng.uniform(-step, step, 2)
        best_c[i] = np.clip(best_c[i], 0.01, 0.99)
        
        new_r, new_s = solve_lp(best_c)
        if new_s > best_s + 1e-8:
            best_s = new_s
            best_r = new_r.copy()
            step = min(step * 1.01, 0.05)
        else:
            best_c[i] = old_c
            step *= 0.998
    return best_c, best_r, best_s

def obj_joint(v, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_joint(v, n):
    """Inequality constraints for joint optimization."""
    cx, cy, r = v[:n], v[n:2*n], v[2*n:]
    c1 = cx - r
    c2 = 1.0 - cx - r
    c3 = cy - r
    c4 = 1.0 - cy - r
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dr = r[i_idx] + r[j_idx]
    c5 = dx**2 + dy**2 - dr**2
    return np.concatenate([c1, c2, c3, c4, c5])

def joint_slqp(centers, radii, n):
    """Joint SLSQP optimization of centers and radii."""
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)]*(2*n) + [(1e-6, 0.5)]*n
    try:
        res = minimize(obj_joint, x0, args=(n,), method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': cons_joint, 'args': (n,)},
                       options={'maxiter': 2500, 'ftol': 1e-14})
        if res.success or np.isfinite(res.fun):
            cx = res.x[:n]
            cy = res.x[n:2*n]
            c_mat = np.column_stack((cx, cy))
            r_opt, s_opt = solve_lp(c_mat)
            return c_mat, r_opt, s_opt
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    rng = np.random.default_rng(42)

    ]
    
    configs = []

        if sum(pat) != n: 
            continue

        configs.append(np.array(pts[:n]))

            configs.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(4):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))

            best_radii = r_lp.copy()

            best_sum = s_slqp

    scale = 1.0

    for i in range(n):

            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]

                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
