import numpy as np
from scipy.optimize import linprog

N_CIRCLES = 26
TRIU_I, TRIU_J = np.triu_indices(N_CIRCLES, k=1)
N_PAIRS = len(TRIU_I)

def solve_lp_fast(centers, A_ub_pair, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = N_CIRCLES
    # Bounds: 0 <= r_i <= distance to nearest wall
    walls = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(w, 1e-12)) for w in walls]
    
    # Pairwise distances
    diff = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    b_ub = dists
    
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub_pair, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def generate_hex(row_counts, r0):
    """Generates a hexagonal lattice configuration with specified row counts."""
    n = N_CIRCLES
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        width = (cnt - 1) * 2 * r0
        x_start = 0.5 - width / 2.0 + shift
        for k in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x_start + k * 2 * r0, y])
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
