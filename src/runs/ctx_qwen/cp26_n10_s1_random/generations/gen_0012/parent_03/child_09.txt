import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIU_I, TRIU_J = np.triu_indices(N_CIRCLES, k=1)
N_PAIRS = len(TRIU_I)

# Precompute constant inequality matrix for pairwise radius constraints
A_LIN = np.zeros((N_PAIRS, N_CIRCLES))
A_LIN[np.arange(N_PAIRS), TRIU_I] = 1.0
A_LIN[np.arange(N_PAIRS), TRIU_J] = 1.0

def solve_lp_fast(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    
    # Wall constraints limit for each circle
    lims = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]),
                      np.minimum(centers[:,1], 1.0 - centers[:,1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    # Pairwise distances
    diff = centers[TRIU_I] - centers[TRIU_J]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    
    try:
        res = linprog(c_obj, A_ub=A_LIN, b_ub=dists, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= -1e-7):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def obj_neg_sum_radii(centers_flat, n):
    """Objective for Nelder-Mead: minimize negative sum of radii via LP."""
    centers = centers_flat.reshape(n, 2)
    centers = np.clip(centers, 0.01, 0.99)
    _, s = solve_lp_fast(centers)
    # Penalize configurations that yield negligible radii
    if s < 1e-4:
        return 100.0
    return -s

def cons_slqp(v, n):
    """Inequality constraints >= 0 for SLSQP."""
    x = v[:n]
    y = v[n:2*n]
    rad = v[2*n:]
    
    # Boundary constraints
    con = np.concatenate([x - rad, 1.0 - x - rad, y - rad, 1.0 - y - rad])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[TRIU_I] - x[TRIU_J]
    dy = y[TRIU_I] - y[TRIU_J]
    dr = rad[TRIU_I] + rad[TRIU_J]
    con = np.concatenate([con, dx**2 + dy**2 - dr**2])
    return con

def obj_slqp(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def generate_hex_init(row_counts, r0, rng):
    """Generates a hexagonal lattice configuration with specified row counts."""
    n = N_CIRCLES
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        width = (cnt - 1) * 2 * r0
        x_start = 0.5 - width / 2.0 + shift
        for k in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x_start + k * 2 * r0, y])
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Diverse hexagonal patterns summing to 26
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [6,5,4,6,5], [5,4,6,6,5], [7,5,5,5,4],
        [5,6,6,5,4], [5,5,5,5,6], [4,5,6,6,5], [6,4,6,6,4]
    ]
    
    starts = []
    for pat in patterns:
        if sum(pat) != n:
            continue
        for r0 in [0.08, 0.09, 0.10, 0.11]:
            pts = generate_hex_init(pat, r0, rng)
            starts.append(pts)
            # Controlled perturbations to break symmetry
            for _ in range(2):
                p = pts + rng.uniform(-0.015, 0.015, pts.shape)
                p = np.clip(p, 0.05, 0.95)
                starts.append(p)
                
    # Add purely random dense starts
    for _ in range(8):
        starts.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # Phase 1: Nelder-Mead on centers guided by exact LP radii
    for cfg in starts:
        try:
            res = minimize(obj_neg_sum_radii, cfg.flatten(), args=(n,), method='Nelder-Mead',
                           options={'maxiter': 4000, 'xatol': 1e-6, 'fatol': 1e-8})
            if res.fun < -2.0:
                c_opt = res.x.reshape(n, 2)
                r_opt, s_opt = solve_lp_fast(c_opt)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass

    # Phase 2: SLSQP joint polish to fine-tune centers & radii simultaneously
    if best_centers is not None:
        c = best_centers.copy()
        r = best_radii.copy()
        x0 = np.concatenate([c[:,0], c[:,1], r])
        bounds_slqp = [(0.0, 1.0)]*(2*n) + [(1e-6, 0.5)]*n
        
        try:
            res2 = minimize(obj_slqp, x0, args=(n,), method='SLSQP', bounds=bounds_slqp,
                            constraints={'type': 'ineq', 'fun': cons_slqp, 'args': (n,)},
                            options={'maxiter': 5000, 'ftol': 1e-14})
            if np.isfinite(res2.fun):
                c_pol = res2.x[:2*n].reshape(n, 2)
                r_pol, s_pol = solve_lp_fast(c_pol)
                if s_pol > best_sum:
                    best_sum = s_pol
                    best_centers = c_pol.copy()
                    best_radii = r_pol.copy()
        except Exception:
            pass
            
    # Phase 3: Local perturbation hill-climbing on centers
    if best_centers is not None:
        for step in np.linspace(0.008, 0.0005, 25):
            improved = False
            for _ in range(40):
                idx = rng.integers(n)
                old = best_centers[idx].copy()
                best_centers[idx] += rng.uniform(-step, step, 2)
                best_centers[idx] = np.clip(best_centers[idx], 0.01, 0.99)
                
                r_new, s_new = solve_lp_fast(best_centers)
                if s_new > best_sum + 1e-8:
                    best_sum = s_new
                    best_radii = r_new.copy()
                    improved = True
                else:
                    best_centers[idx] = old
            if not improved:
                break
                
    # Phase 4: Strict numerical safety scaling
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i,0], best_centers[i,1], best_radii[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], 
                             best_centers[i,1]-best_centers[j,1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-12:
                    scale = min(scale, d/rs)
        best_radii *= scale * 0.999999
        best_sum = float(np.sum(best_radii))
    else:
        # Fallback configuration
        best_centers = generate_hex_init([6,5,6,5,4], 0.09, rng)
        best_radii, best_sum = solve_lp_fast(best_centers)
        
    return best_centers, best_radii, best_sum
