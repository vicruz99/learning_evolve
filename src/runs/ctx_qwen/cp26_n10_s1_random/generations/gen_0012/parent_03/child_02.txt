import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]),
                      np.minimum(centers[:,1], 1.0-centers[:,1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    A_ub = np.zeros((len(TRIU_I), n))
    A_ub[np.arange(len(TRIU_I)), TRIU_I] = 1.0
    A_ub[np.arange(len(TRIU_I)), TRIU_J] = 1.0
    diff = centers[TRIU_I] - centers[TRIU_J]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    b_ub = dists
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def obj_centers(centers_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 1e-4, 1.0-1e-4)
    _, s = solve_lp(centers)
    return -s

def get_hex_init(r0, row_counts):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def force_relax(centers, steps=600):
    """Vectorized force-directed simulation to spread circles and grow radii."""
    c = centers.copy()
    r_est = np.full(N, 0.085)
    dt = 0.006
    for _ in range(steps):
        r_est *= 1.0008
        diff = c[:, None, :] - c[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dist, 1e9)
        
        overlap = np.maximum(0.0, r_est[:, None] + r_est[None, :] - dist)
        strength = overlap * 120.0 / (dist + 1e-9)
        f = np.sum(diff * strength[:, :, None], axis=1)
        
        # Wall repulsion
        mask_l = c[:, 0] < r_est
        mask_r = c[:, 0] > 1.0 - r_est
        mask_b = c[:, 1] < r_est
        mask_t = c[:, 1] > 1.0 - r_est
        forces = np.zeros_like(c)
        forces[mask_l, 0] += (r_est[mask_l] - c[mask_l, 0]) * 250.0
        forces[mask_r, 0] -= (c[mask_r, 0] + r_est[mask_r] - 1.0) * 250.0
        forces[mask_b, 1] += (r_est[mask_b] - c[mask_b, 1]) * 250.0
        forces[mask_t, 1] -= (c[mask_t, 1] + r_est[mask_t] - 1.0) * 250.0
        
        f += forces
        c += f * dt
        c = np.clip(c, 1e-4, 1.0-1e-4)
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_c = None
    best_r = None
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [5,4,6,6,5], [7,5,5,5,4], [5,6,6,5,4],
        [5,5,5,5,6], [6,4,6,6,4], [4,7,6,5,4], [5,5,7,4,5]
    ]
    
    starts = []
    for pat in patterns:
        if sum(pat) == N:
            for r0 in [0.095, 0.10, 0.105]:
                starts.append(get_hex_init(r0, pat))
                
    for _ in range(12):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    for cfg in starts:
        # Phase 1: Force relaxation to find dense feasible layout
        c_relaxed = force_relax(cfg)
        
        # Phase 2: Powell optimization on centers
        try:
            res = minimize(obj_centers, c_relaxed.flatten(), method='Powell', 
                           options={'maxiter': 4000, 'ftol': 1e-9, 'xtol': 1e-8})
            c_opt = res.x.reshape(N, 2)
            c_opt = np.clip(c_opt, 1e-4, 1.0-1e-4)
        except Exception:
            c_opt = c_relaxed
            
        # Phase 3: Coordinate Descent Refinement
        c_cd = c_opt.copy()
        for _ in range(6):
            for i in range(N):
                def obj_x(xv):
                    c_cd[i, 0] = xv[0]
                    _, s = solve_lp(c_cd)
                    return -s
                res_x = minimize(obj_x, [c_cd[i, 0]], method='Nelder-Mead', 
                                 options={'maxiter': 600, 'xatol': 1e-6})
                c_cd[i, 0] = res_x.x[0]
                
                def obj_y(yv):
                    c_cd[i, 1] = yv[0]
                    _, s = solve_lp(c_cd)
                    return -s
                res_y = minimize(obj_y, [c_cd[i, 1]], method='Nelder-Mead', 
                                 options={'maxiter': 600, 'xatol': 1e-6})
                c_cd[i, 1] = res_y.x[0]
                
        r_cd, s_cd = solve_lp(c_cd)
        if s_cd > best_sum:
            best_sum = s_cd
            best_c = c_cd.copy()
            best_r = r_cd.copy()
            
        # Phase 4: Random shakes to escape local minima
        for _ in range(25):
            pert = c_cd + rng.uniform(-0.006, 0.006, (N, 2))
            pert = np.clip(pert, 0.02, 0.98)
            r_p, s_p = solve_lp(pert)
            if s_p > best_sum:
                best_sum = s_p
                best_c = pert.copy()
                best_r = r_p.copy()
                
    # Fallback
    if best_c is None:
        best_c = starts[0]
        best_r, best_sum = solve_lp(best_c)
        
    # Phase 5: Strict Safety Scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
