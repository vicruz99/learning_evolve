import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
N_PAIRS = len(TRIU_I)

# Precompute constant inequality matrix for pairwise constraints: r_i + r_j <= d_ij
A_ub = np.zeros((N_PAIRS, N))
for k in range(N_PAIRS):
    A_ub[k, TRIU_I[k]] = 1.0
    A_ub[k, TRIU_J[k]] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for max sum of radii and computes gradient w.r.t centers using duals."""
    n = N
    x, y = centers[:, 0], centers[:, 1]
    lims = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    # Pairwise distances
    dx = centers[:, 0:1] - centers[:, 0:1].T
    dy = centers[:, 1:2] - centers[:, 1:2].T
    dists = np.sqrt(dx**2 + dy**2 + 1e-15)
    
    b_ub = dists[TRIU_I, TRIU_J]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success or not np.isfinite(res.fun):
        return None, None, None
        
    r = np.maximum(res.x, 0.0)
    s = -res.fun
    
    # Extract dual variables (marginals) for pairwise constraints
    # These indicate how much sum(r) increases per unit increase in distance d_ij
    lams = np.zeros(N_PAIRS)
    if hasattr(res, 'ineqlin') and res.ineqlin is not None:
        m = res.ineqlin.marginals
        if m is not None:
            lams = np.abs(m)
            
    grad = np.zeros((n, 2))
    valid = lams > 1e-7
    if np.any(valid):
        idx_k = np.where(valid)[0]
        lams_v = lams[idx_k]
        ii = TRIU_I[idx_k]
        jj = TRIU_J[idx_k]
        
        d_v = dists[ii, jj]
        d_safe = np.where(d_v > 1e-8, d_v, 1.0)
        
        diffs = centers[ii] - centers[jj]
        # grad_i += lam * (c_i - c_j) / d_ij
        np.add.at(grad, ii, diffs * (lams_v[:, None] / d_safe[:, None]))
        # grad_j -= lam * (c_i - c_j) / d_ij  => grad_j += lam * (c_j - c_i) / d_ij
        np.add.at(grad, jj, -diffs * (lams_v[:, None] / d_safe[:, None]))
        
    return r, s, grad

def generate_hex_init(r0):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0
    row = 0
    while y + r0 < 1.0 and len(pts) < N + 5:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x + r0 < 1.0 and len(pts) < N + 5:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_c = None
    best_r = None
    
    # Diverse starting configurations
    starts = []
    for r0 in [0.085, 0.090, 0.095, 0.100, 0.105]:
        pts = generate_hex_init(r0)
        starts.append(pts)
        # Perturbed versions
        for _ in range(3):
            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            starts.append(np.clip(p, 0.05, 0.95))
            
    # Random dense starts
    for _ in range(8):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    for cfg in starts:
        c = cfg.copy()
        c += rng.uniform(-0.005, 0.005, c.shape)
        c = np.clip(c, 0.05, 0.95)
        
        # Phase 1: Gradient Ascent on Centers
        step = 0.025
        for it in range(3500):
            r, s, g = solve_lp_and_grad(c)
            if r is None:
                break
                
            gn = np.linalg.norm(g)
            if gn > 1e-6:
                move = step * (g / gn)
            else:
                move = np.zeros_like(c)
                
            c_new = c + move
            c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
            
            r_new, s_new, _ = solve_lp_and_grad(c_new)
            if r_new is not None and s_new > s + 1e-9:
                c = c_new
                step = min(step * 1.008, 0.045)
            else:
                step *= 0.985
                
            # Periodic shake to escape plateaus
            if it % 600 == 0 and rng.random() < 0.25:
                c_shake = c + rng.uniform(-0.025, 0.025, c.shape)
                c_shake = np.clip(c_shake, 0.05, 0.95)
                r_sh, s_sh, _ = solve_lp_and_grad(c_shake)
                if r_sh is not None and s_sh > s + 1e-8:
                    c = c_shake
                    step = 0.025
                    
        r, s, _ = solve_lp_and_grad(c)
        if r is not None and s > best_sum:
            best_sum = s
            best_c = c.copy()
            best_r = r.copy()
            
    # Phase 2: SLSQP Joint Polish
    if best_c is not None:
        def obj(v):
            return -np.sum(v[2*N:])
            
        def cons(v):
            cx, cy = v[:N], v[N:2*N]
            r = v[2*N:]
            c1 = cx - r
            c2 = 1.0 - cx - r
            c3 = cy - r
            c4 = 1.0 - cy - r
            dx = cx[TRIU_I] - cx[TRIU_J]
            dy = cy[TRIU_I] - cy[TRIU_J]
            dr = r[TRIU_I] + r[TRIU_J]
            c5 = dx**2 + dy**2 - dr**2
            return np.concatenate([c1, c2, c3, c4, c5])
            
        x0 = np.concatenate([best_c[:,0], best_c[:,1], best_r * 0.98])
        bounds_slqp = [(0.0, 1.0)]*(2*N) + [(1e-6, 0.5)]*N
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints={'type': 'ineq', 'fun': cons},
                           options={'maxiter': 5000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_pol = np.column_stack((res.x[:N], res.x[N:2*N]))
                r_pol, s_pol, _ = solve_lp_and_grad(c_pol)
                if r_pol is not None and s_pol > best_sum:
                    best_sum = s_pol
                    best_c = c_pol
                    best_r = r_pol
        except Exception:
            pass
            
    # Phase 3: Strict Safety Scaling
    if best_c is not None:
        scale = 1.0
        for i in range(N):
            x, y, r = best_c[i,0], best_c[i,1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
                
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                rs = best_r[i] + best_r[j]
                if rs > 1e-12:
                    scale = min(scale, d/rs)
                    
        best_r *= scale * 0.9999995
        best_sum = float(np.sum(best_r))
    else:
        # Fallback
        best_c = generate_hex_init(0.09)
        best_r, best_sum = solve_lp_and_grad(best_c)
        best_sum = float(np.sum(best_r))
        
    return best_c, best_r, best_sum
