import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
N_PAIRS = len(TRIU_I)

# Precompute constant LP matrix for efficiency
A_LP = np.zeros((N_PAIRS, N))
A_LP[np.arange(N_PAIRS), TRIU_I] = 1.0
A_LP[np.arange(N_PAIRS), TRIU_J] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    bounds = [(0.0, lim) for lim in limits]
    
    diff = centers[TRIU_I] - centers[TRIU_J]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=dists, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0

def force_simulate(centers, radii, steps=1200):
    """Vectorized force-directed expanding circles simulation."""
    c = centers.copy()
    r = radii.copy()

    dt = 0.005
    damp = 0.85
    grow = 1.0006

        wf[:, 0] += np.clip(r - c[:, 0], 0, None) * 500.0
        wf[:, 0] -= np.clip(c[:, 0] + r - 1.0, 0, None) * 500.0

        vel = damp * vel + (forces + wf) * dt
        c += vel
        c = np.clip(c, 0.001, 0.999)
        
    return c, r

def obj_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_slqp(v):
    """Inequality constraints >= 0 for SLSQP."""
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    
    con = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[TRIU_I] - cx[TRIU_J]
    dy = cy[TRIU_I] - cy[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    con = np.concatenate([con, dx**2 + dy**2 - dr**2])
    return con

def polish_slqp(centers, radii):
    """Joint SLSQP optimization of centers and radii."""
    x0 = np.concatenate([centers[:, 0], centers[:, 1], radii])
    bounds = [(0.0, 1.0)]*(2*N) + [(1e-6, 0.5)]*N
    try:
        res = minimize(obj_slqp, x0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons_slqp},
                       options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
        if np.isfinite(res.fun):
            cx = res.x[:N]
            cy = res.x[N:2*N]
            c_opt = np.column_stack((cx, cy))
            r_opt, s_opt = solve_lp(c_opt)
            return c_opt, r_opt, s_opt
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def local_hill_climb(centers, radii, sum_r, rng):
    """Iterative local search: perturb centers, re-optimize radii via LP."""
    cur_c = centers.copy()
    cur_r = radii.copy()
    cur_s = sum_r
    step = 0.015
    
    for _ in range(2500):
        i = rng.integers(N)
        old_c = cur_c[i].copy()
        
        improved = False
        for _ in range(4):
            delta = rng.uniform(-step, step, 2)
            cand = np.clip(old_c + delta, 0.01, 0.99)
            cur_c[i] = cand
            r_new, s_new = solve_lp(cur_c)
            if s_new > cur_s + 1e-8:
                cur_r = r_new.copy()
                cur_s = s_new
                improved = True
                break
        if not improved:
            cur_c[i] = old_c
            
        step *= 0.9995
        
    return cur_c, cur_r, cur_s
