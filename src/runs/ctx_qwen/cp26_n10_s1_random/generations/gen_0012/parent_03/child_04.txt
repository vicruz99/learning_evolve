2. **Gradient Extraction**: The optimal value of this LP as a function of centers, $V(c)$, has a gradient given by the LP dual variables (shadow prices).    - Introduce random perturbations when the gradient vanishes to escape local minima.

import numpy as np
from scipy.optimize import linprog

N_CIRCLES = 26
TRIU_I, TRIU_J = np.triu_indices(N_CIRCLES, k=1)
N_PAIRS = len(TRIU_I)

def solve_lp_and_grad(centers):
    n = N_CIRCLES
    # Pairwise constraints matrix: r_i + r_j <= dist_ij
    A_pair = np.zeros((N_PAIRS, n))
    A_pair[np.arange(N_PAIRS), TRIU_I] = 1.0
    A_pair[np.arange(N_PAIRS), TRIU_J] = 1.0
    
    diff = centers[TRIU_I] - centers[TRIU_J]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    b_pair = dists
    
    # Boundary constraints matrix: r_i <= wall_dist
    A_bound = np.zeros((4 * n, n))
    for i in range(n):
        A_bound[4 * i, i] = 1.0
        A_bound[4 * i + 1, i] = 1.0
        A_bound[4 * i + 2, i] = 1.0
        A_bound[4 * i + 3, i] = 1.0
        
    wall_vals = np.zeros(4 * n)
    for i in range(n):
        wall_vals[4 * i] = centers[i, 0]
        wall_vals[4 * i + 1] = 1.0 - centers[i, 0]
        wall_vals[4 * i + 2] = centers[i, 1]
        wall_vals[4 * i + 3] = 1.0 - centers[i, 1]
        
    A_ub = np.vstack([A_pair, A_bound])
    b_ub = np.concatenate([b_pair, wall_vals])
    bounds_r = [(0.0, None)] * n
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None
            
        r = res.x
        s = -res.fun
        lams = np.abs(res.ineqlin.marginals)
        
        g = np.zeros((n, 2))
        # Pairwise gradient contributions
        lam_p = lams[:N_PAIRS]
        valid = lam_p > 1e-7
        if np.any(valid):
            ii = TRIU_I[valid]
            jj = TRIU_J[valid]
            lams_v = lam_p[valid]
            diffs = centers[ii] - centers[jj]
            dists_v = dists[valid]
            dists_v = np.where(dists_v < 1e-9, 1e-9, dists_v)
            factors = (lams_v / dists_v)[:, np.newaxis]
            np.add.at(g, ii, diffs * factors)
            np.add.at(g, jj, -diffs * factors)
            
        # Boundary gradient contributions
        lam_b = lams[N_PAIRS:]
        dir_x = np.zeros(n)
        dir_y = np.zeros(n)
        for i in range(n):
            l, r_val, b, t = lam_b[4*i], lam_b[4*i+1], lam_b[4*i+2], lam_b[4*i+3]
            dir_x[i] += l - r_val
            dir_y[i] += b - t
        g[:, 0] += dir_x
        g[:, 1] += dir_y
        
        return r, s, g
    except Exception:
        return None, None, None

def generate_hex_config(row_counts, rng):
    n = N_CIRCLES
    pts = []
    y = 0.1
    for i, cnt in enumerate(row_counts):
        shift = 0.1 if i % 2 == 1 else 0.0
        width = (cnt - 1) * 0.2
        x_start = 0.5 - width / 2.0 + shift
        for k in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x_start + k * 0.2, y])
        y += 0.1 * np.sqrt(3)
    while len(pts) < n:
        pts.append([0.5, 0.5])
    pts_arr = np.array(pts[:n])
    pts_arr += rng.uniform(-0.005, 0.005, pts_arr.shape)
    return np.clip(pts_arr, 0.05, 0.95)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_c = None
    best_r = None
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [6,5,4,6,5], [5,4,6,6,5], [7,5,5,5,4],
        [5,6,6,5,4], [5,5,5,5,6], [6,6,6,4,4], [4,5,6,5,6]
    ]
    
    configs = []
    for pat in patterns:
        if sum(pat) == n:
            configs.append(generate_hex_config(pat, rng))
            
    for _ in range(10):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    for cfg in configs:
        cur_c = cfg.copy()
        step = 0.08
        
        for _ in range(400):
            r, s, g = solve_lp_and_grad(cur_c)
            if r is None:
                step *= 0.5
                continue
                
            if s > best_sum:
                best_sum = s
                best_c = cur_c.copy()
                best_r = r.copy()
                
            gn = np.linalg.norm(g)
            if gn < 1e-8:
                # Escape local minimum
                cur_c += rng.uniform(-0.02, 0.02, cur_c.shape)
                step = 0.05
                continue
                
            dir_vec = g / gn
            c_new = cur_c + step * dir_vec
            c_new = np.clip(c_new, 0.02, 0.98)
            
            # Evaluate new point
            r_new, s_new, _ = solve_lp_and_grad(c_new)
            if r_new is not None and s_new > s + 1e-7:
                cur_c = c_new
                step = min(step * 1.3, 0.25)
            else:
                step *= 0.5
                if step < 1e-5:
                    # Perturb
                    cur_c += rng.uniform(-0.01, 0.01, cur_c.shape)
                    cur_c = np.clip(cur_c, 0.02, 0.98)
                    step = 0.04
                    
    # Fallback if optimization fails
    if best_c is None:
        best_c = generate_hex_config([6,5,6,5,4], rng)
        best_r, best_sum, _ = solve_lp_and_grad(best_c)
        
    # Final safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y = best_c[i]
        r_val = best_r[i]
        if r_val > 1e-12:
            scale = min(scale, x/r_val, (1.0-x)/r_val, y/r_val, (1.0-y)/r_val)
            
    diff = best_c[:, np.newaxis, :] - best_c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_pair = best_r[:, np.newaxis] + best_r[np.newaxis, :]
    
    min_ratio = np.min(dists / np.maximum(r_pair, 1e-12))
    scale = min(scale, min_ratio)
    
    best_r *= scale * 0.9999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
