import numpy as np
from scipy.optimize import linprog

# Global constants for efficiency
N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
NP = len(TRIU_I)

# Precompute the constant structure of the LP inequality matrix
# Constraints order:
# 0..N-1       : r_i <= x_i          (Left boundary)
# N..2N-1      : r_i <= 1 - x_i      (Right boundary)
# 2N..3N-1     : r_i <= y_i          (Bottom boundary)
# 3N..4N-1     : r_i <= 1 - y_i      (Top boundary)
# 4N..4N+NP-1  : r_i + r_j <= d_ij   (Pairwise non-overlap)
A_ub_const = np.zeros((4 * N + NP, N))
for i in range(N):
    A_ub_const[i, i] = 1.0
    A_ub_const[N + i, i] = 1.0
    A_ub_const[2 * N + i, i] = 1.0
    A_ub_const[3 * N + i, i] = 1.0

for k in range(NP):
    i, j = TRIU_I[k], TRIU_J[k]
    A_ub_const[4 * N + k, i] = 1.0
    A_ub_const[4 * N + k, j] = 1.0


def solve_lp_and_grad(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns radii, sum of radii, and gradient of sum w.r.t centers.
    """
    n = N
    # Build RHS vector b_ub dynamically
    b_ub = np.zeros(4 * n + NP)
    for i in range(n):
        b_ub[i] = centers[i, 0]
        b_ub[n + i] = 1.0 - centers[i, 0]
        b_ub[2 * n + i] = centers[i, 1]
        b_ub[3 * n + i] = 1.0 - centers[i, 1]
        
    dx = centers[TRIU_I, 0] - centers[TRIU_J, 0]
    dy = centers[TRIU_I, 1] - centers[TRIU_J, 1]
    dists = np.hypot(dx, dy)
    b_ub[4 * n:] = dists
    
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_const, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, 0.0, None
            
        r = np.maximum(res.x, 0.0)
        sum_r = -res.fun
        
        # Compute gradient from dual variables (marginals)
        grad = np.zeros((n, 2))
        duals = np.abs(res.ineqlin.marginals) if hasattr(res, 'ineqlin') and res.ineqlin is not None else np.zeros(4*n+NP)
        
        # Boundary contributions
        grad[:, 0] += duals[:n] - duals[n:2*n]
        grad[:, 1] += duals[2*n:3*n] - duals[3*n:4*n]
        
        # Pairwise contributions
        p_duals = duals[4*n:]
        safe_dists = np.where(dists > 1e-9, dists, 1.0)
        dir_vecs = (centers[TRIU_I] - centers[TRIU_J]) / safe_dists[:, None]
        forces = p_duals[:, None] * dir_vecs
        
        np.add.at(grad, TRIU_I, forces)
        np.add.at(grad, TRIU_J, -forces)
        
        return r, sum_r, grad
    except Exception:
        return None, 0.0, None


def run_optimization(cfg, rng, steps=1000):
    """Runs gradient ascent on centers guided by LP duals."""
    c = cfg.copy()
    best_sum = 0.0
    best_r = None
    step_size = 0.008
    
    for _ in range(steps):
        r, s, g = solve_lp_and_grad(c)
        if r is None:
            break
            
        if s > best_sum:
            best_sum = s
            best_r = r.copy()
            
        gn = np.linalg.norm(g)
        if gn > 1e-8:
            c_new = c + step_size * (g / gn)
            c_new = np.clip(c_new, 0.005, 0.995)
            
            r_new, s_new, _ = solve_lp_and_grad(c_new)
            if r_new is not None and s_new >= s - 1e-7:
                c = c_new
                step_size = min(step_size * 1.003, 0.025)
            else:
                step_size *= 0.996
        else:
            step_size *= 0.995
            
        # Occasional random shake to escape plateaus
        if rng.random() < 0.025:
            c += rng.uniform(-0.01, 0.01, c.shape)
            c = np.clip(c, 0.005, 0.995)
            
    return c, best_r, best_sum


def generate_configs(rng):
    """Generates diverse initial configurations."""
    configs = []
    
    # 1. Hexagonal lattice patterns with varying row distributions
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [5,4,6,6,5], [6,5,4,6,5], [7,5,5,5,4],
        [5,6,6,5,4], [5,5,5,5,6]
    ]
    
    for pat in patterns:
        if sum(pat) != N:
            continue
        pts = []
        y = 0.09
        row_idx = 0
        while len(pts) < N and y < 1.0:
            shift = 0.09 if row_idx % 2 == 1 else 0.0
            x = 0.09 + shift
            cnt = pat[row_idx] if row_idx < len(pat) else 0
            for _ in range(cnt):
                if len(pts) >= N:
                    break
                if x + 0.09 <= 1.0:
                    pts.append([x, y])
                x += 0.18
            y += 0.09 * np.sqrt(3)
            row_idx += 1
            
        while len(pts) < N:
            pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
            
        configs.append(np.array(pts[:N]))
        configs.append(np.clip(np.array(pts[:N]) + rng.uniform(-0.02, 0.02, (N, 2)), 0.05, 0.95))
        
    # 2. Grid-based starts
    grid = np.array([(0.1 + i*0.18, 0.1 + j*0.18) for j in range(5) for i in range(5)])
    grid = np.vstack([grid, [0.5, 0.5]])
    configs.append(grid)
    configs.append(np.clip(grid + rng.uniform(-0.03, 0.03, grid.shape), 0.05, 0.95))
    
    # 3. Pure random starts
    for _ in range(4):
        configs.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return configs


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = generate_configs(rng)

            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    # Fallback if optimization fails
    if best_centers is None:
        best_centers = generate_configs(rng)[0]
        best_radii, best_sum = solve_lp_and_grad(best_centers)[0:2]
        if best_radii is None:
            best_radii = np.full(N, 0.05)
            best_sum = np.sum(best_radii)
            
    # Phase 2: Strict numerical safety scaling to guarantee validator tolerance
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
