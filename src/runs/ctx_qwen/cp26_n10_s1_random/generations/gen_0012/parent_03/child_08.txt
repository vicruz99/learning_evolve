import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
N_PAIRS = len(TRIU_I)

# Precompute constant LP constraint matrix for speed
A_LP = np.zeros((N_PAIRS, N))
A_LP[np.arange(N_PAIRS), TRIU_I] = 1.0
A_LP[np.arange(N_PAIRS), TRIU_J] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    diff = centers[TRIU_I] - centers[TRIU_J]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    b_ub = dists
    
    try:
        res = linprog(c_obj, A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= -1e-7):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def md_pack(centers, steps=1800, max_r=0.106):
    """Force-directed simulation to pack circles densely."""
    n = centers.shape[0]
    vel = np.zeros_like(centers)
    dt = 0.003
    damp = 0.86
    
    for t in range(steps):
        r = max_r * (t / steps)
        forces = np.zeros_like(centers)
        
        diff = centers[:, None, :] - centers[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist, 1e9)
        
        overlap = np.maximum(0.0, 2.0*r - dist)
        inv_d = np.where(dist > 1e-8, 1.0/dist, 0.0)
        rep = overlap * inv_d * 70.0
        forces += np.sum(diff * rep[:,:,None], axis=1)
        
        # Boundary repulsion
        dx_l = centers[:,0] - r
        dx_r = 1.0 - centers[:,0] - r
        dy_b = centers[:,1] - r
        dy_t = 1.0 - centers[:,1] - r
        forces[:,0] += np.where(dx_l < 0, dx_l * 250.0, 0.0) + np.where(dx_r < 0, -dx_r * 250.0, 0.0)
        forces[:,1] += np.where(dy_b < 0, dy_b * 250.0, 0.0) + np.where(dy_t < 0, -dy_t * 250.0, 0.0)
        
        vel = vel * damp + forces * dt
        centers += vel
        centers = np.clip(centers, 0.005, 0.995)
        
    return centers

def obj_slqp(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_slqp(v, n):
    """Inequality constraints >= 0 for SLSQP."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    
    dx = x[TRIU_I] - x[TRIU_J]
    dy = y[TRIU_I] - y[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    c5 = dx**2 + dy**2 - dr**2
    
    return np.concatenate([c1, c2, c3, c4, c5])

def generate_hex_patterns(rng):
    """Generates diverse hexagonal lattice configurations."""
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [7,5,5,5,4], [5,4,6,6,5], [6,5,4,6,5],
        [5,6,6,4,5], [4,5,6,6,5], [5,5,5,5,6]
    ]
    configs = []
    for pat in patterns:
        if sum(pat) != N: continue
        pts = []
        y = 0.10
        for i, cnt in enumerate(pat):
            shift = 0.10 if i%2==1 else 0.0
            width = (cnt-1)*0.20
            x_start = 0.5 - width/2.0 + shift
            for k in range(cnt):
                if len(pts) >= N: break
                pts.append([x_start + k*0.20, y])
            y += np.sqrt(3)*0.10
        while len(pts) < N: pts.append([0.5, 0.5])
        base = np.array(pts[:N])
        configs.append(base)
        for _ in range(3):
            configs.append(np.clip(base + rng.uniform(-0.02, 0.02, base.shape), 0.05, 0.95))
            
    # Add random dense starts
    for _ in range(6):
        configs.append(rng.uniform(0.12, 0.88, (N, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_c = None
    best_r = None
    
    bounds_vars = [(0.0, 1.0)]*(2*N) + [(1e-6, 0.5)]*N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp, 'args': (N,)}
    
    configs = generate_hex_patterns(rng)
    
    for cfg in configs:
        # Phase 1: MD simulation to settle into a dense basin
        c_md = md_pack(cfg.copy(), steps=1500, max_r=0.106)
        
        # Phase 2: LP refinement for optimal radii
        r_lp, s_lp = solve_lp(c_md)
        if s_lp > best_sum:
            best_sum = s_lp
            best_c = c_md.copy()
            best_r = r_lp.copy()
            
        # Phase 3: Joint SLSQP polish
        x0 = np.concatenate([c_md[:,0], c_md[:,1], r_lp * 0.96])
        try:
            res = minimize(obj_slqp, x0, args=(N,), method='SLSQP', bounds=bounds_vars,
                           constraints=cons_dict, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if np.isfinite(res.fun):
                c_opt = np.column_stack((res.x[:N], res.x[N:2*N]))
                r_opt, s_opt = solve_lp(c_opt)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            pass
            
        # Phase 4: Iterative local hill-climbing on centers
        c_cur = best_c.copy()
        r_cur = best_r.copy()
        s_cur = best_sum
        step = 0.008
        no_improve = 0
        
        for _ in range(2500):
            idx = rng.integers(N)
            old = c_cur[idx].copy()
            
            # Try a random direction
            angle = rng.uniform(0, 2*np.pi)
            delta = step * np.array([np.cos(angle), np.sin(angle)])
            c_cur[idx] = old + delta
            c_cur[idx] = np.clip(c_cur[idx], 0.01, 0.99)
            
            r_new, s_new = solve_lp(c_cur)
            if s_new > s_cur + 1e-9:
                s_cur = s_new
                r_cur = r_new
                best_c = c_cur.copy()
                best_r = r_cur.copy()
                best_sum = s_cur
                step = min(step * 1.01, 0.025)
                no_improve = 0
            else:
                c_cur[idx] = old
                no_improve += 1
                
            # Anneal step size and occasionally shake to escape plateaus
            if no_improve > 300:
                c_cur += rng.uniform(-0.005, 0.005, c_cur.shape)
                c_cur = np.clip(c_cur, 0.05, 0.95)
                step = 0.01
                no_improve = 0
            else:
                step *= 0.998
                
        if best_sum > 2.62:
            break # Early exit if we found a very good packing
            
    # Fallback if optimization fails unexpectedly
    if best_c is None:
        base = configs[0]
        r_lp, s_lp = solve_lp(base)
        best_c, best_r, best_sum = base, r_lp, s_lp
        
    # Phase 5: Strict Safety Scaling to guarantee validator tolerance
    scale = 1.0
    for i in range(N):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    dists = np.sqrt(np.sum((best_c[:,None,:] - best_c[None,:,:])**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    rs = best_r[:,None] + best_r[None,:]
    pair_scales = dists[TRIU_I, TRIU_J] / np.maximum(rs[TRIU_I, TRIU_J], 1e-12)
    scale = min(scale, np.min(pair_scales))
    
    best_r *= scale * 0.9999994
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
