import numpy as np
from scipy.optimize import linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
N_PAIRS = len(TRIU_I)

# Precompute constant constraint matrix for LP
# Constraints: r_i + r_j <= d_ij  (pairwise)
#              r_i <= x_i, 1-x_i, y_i, 1-y_i  (boundary)
A_PAIR = np.zeros((N_PAIRS, N))
A_PAIR[np.arange(N_PAIRS), TRIU_I] = 1.0
A_PAIR[np.arange(N_PAIRS), TRIU_J] = 1.0

A_BOUND = np.zeros((4 * N, N))
for i in range(N):
    A_BOUND[4*i, i] = 1.0
    A_BOUND[4*i+1, i] = 1.0
    A_BOUND[4*i+2, i] = 1.0
    A_BOUND[4*i+3, i] = 1.0

A_UB = np.vstack([A_PAIR, A_BOUND])

def solve_lp_and_grad(centers):
    """Solves LP to maximize sum of radii and returns subgradient w.r.t centers."""
    n = centers.shape[0]
    b = np.zeros(A_UB.shape[0])
    
    # Pairwise distances
    diff = centers[TRIU_I] - centers[TRIU_J]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    b[:N_PAIRS] = dists
    
    # Boundary limits
    for i in range(n):
        x, y = centers[i]
        b[N_PAIRS + 4*i] = x
        b[N_PAIRS + 4*i + 1] = 1.0 - x
        b[N_PAIRS + 4*i + 2] = y
        b[N_PAIRS + 4*i + 3] = 1.0 - y
        
    try:
        res = linprog(-np.ones(n), A_ub=A_UB, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            r = res.x
            s = -res.fun
            grad = np.zeros((n, 2))
            lams = np.maximum(res.ineqlin.marginals, 0.0)  # Ensure non-negative marginals
            
            # Pairwise gradient contributions: d(sum_r)/dC_i = sum_j lam_ij * (C_i - C_j)/d_ij
            valid = lams[:N_PAIRS] > 1e-9
            if np.any(valid):
                lam_v = lams[:N_PAIRS][valid]
                ii = TRIU_I[valid]
                jj = TRIU_J[valid]
                d_v = dists[valid]
                d_safe = np.where(d_v > 1e-12, d_v, 1.0)
                factors = (lam_v / d_safe)[:, None]
                diff_v = diff[valid]
                np.add.at(grad, ii, diff_v * factors)
                np.add.at(grad, jj, -diff_v * factors)
                
            # Boundary gradient contributions
            lams_b = lams[N_PAIRS:]
            for i in range(n):
                base = 4 * i
                grad[i, 0] += lams_b[base]     # r <= x  => grad_x += lam
                grad[i, 0] -= lams_b[base + 1] # r <= 1-x => grad_x -= lam
                grad[i, 1] += lams_b[base + 2] # r <= y  => grad_y += lam
                grad[i, 1] -= lams_b[base + 3] # r <= 1-y => grad_y -= lam
                
            return r, s, grad
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0, np.zeros((n, 2))

def generate_hex(row_counts, r0):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        width = (cnt - 1) * 2 * r0
        x_start = 0.5 - width / 2.0 + shift
        for k in range(cnt):
            if len(pts) >= N: break
            pts.append([x_start + k * 2 * r0, y])
        y += np.sqrt(3) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Diverse structural patterns summing to 26
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [6,5,4,6,5], [5,4,6,6,5], [7,5,5,5,4],
        [5,6,6,5,4], [5,5,5,5,6], [6,6,4,6,4], [5,6,4,6,5]
    ]
    
    starts = []
    for pat in patterns:
        if sum(pat) != N: continue
        c = generate_hex(pat, 0.10)
        starts.append(c)
        for _ in range(3):
            p = c + rng.uniform(-0.02, 0.02, c.shape)
            starts.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(6):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # Phase 1: LP-Gradient Hill Climbing from diverse starts
    for cfg in starts:
        curr_c = cfg.copy()
        curr_r, curr_s, _ = solve_lp_and_grad(curr_c)
        
        step = 0.008
        for it in range(2500):
            r, s, g = solve_lp_and_grad(curr_c)
            
            if s > best_sum + 1e-9:
                best_sum = s
                best_centers = curr_c.copy()
                best_radii = r.copy()
                
            gn = np.linalg.norm(g)
            if gn > 1e-7:
                new_c = curr_c + step * g / gn
                new_c = np.clip(new_c, 1e-4, 0.999)
                r_new, s_new, _ = solve_lp_and_grad(new_c)
                
                if s_new > s + 1e-9:
                    curr_c = new_c
                    curr_s = s_new
                    step = min(step * 1.002, 0.02)
                else:
                    step *= 0.96
            else:
                # Escape plateaus with random shake
                if rng.random() < 0.05:
                    shake = curr_c + rng.uniform(-0.02, 0.02, curr_c.shape)
                    shake = np.clip(shake, 0.05, 0.95)
                    r_sh, s_sh, _ = solve_lp_and_grad(shake)
                    if s_sh > s + 1e-9:
                        curr_c = shake
                        curr_s = s_sh
                        step = 0.008
                        
        # Local refinement around discovered best
        if best_sum > 2.62:
            pert = best_centers + rng.uniform(-0.008, 0.008, best_centers.shape)
            pert = np.clip(pert, 0.05, 0.95)
            curr_c = pert
            curr_r, curr_s, _ = solve_lp_and_grad(curr_c)
            step = 0.004
            for _ in range(800):
                r, s, g = solve_lp_and_grad(curr_c)
                gn = np.linalg.norm(g)
                if gn > 1e-7:
                    new_c = curr_c + step * g / gn
                    new_c = np.clip(new_c, 1e-4, 0.999)
                    r_new, s_new, _ = solve_lp_and_grad(new_c)
                    if s_new > s:
                        curr_c = new_c
                        curr_s = s_new
                        step = min(step * 1.01, 0.015)
                    else:
                        step *= 0.97
                    if s_new > best_sum + 1e-9:
                        best_sum = s_new
                        best_centers = curr_c.copy()
                        best_radii = r_new.copy()
                    
    if best_centers is None:
        best_centers = generate_hex([6,5,6,5,4], 0.09)
        best_radii, best_sum, _ = solve_lp_and_grad(best_centers)
        
    # Final LP evaluation to ensure radii are optimal for the chosen centers
    final_r, final_s, _ = solve_lp_and_grad(best_centers)
    if final_s > best_sum:
        best_sum = final_s
        best_radii = final_r
        
    # Phase 2: Strict Safety Scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    diff = best_centers[:, None, :] - best_centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_pair = best_radii[:, None] + best_radii[None, :]
    min_ratio = np.min(dists[TRIU_I, TRIU_J] / np.maximum(r_pair[TRIU_I, TRIU_J], 1e-12))
    scale = min(scale, min_ratio)
    
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
