N_CIRCLES = 26
TRIU_I, TRIU_J = np.triu_indices(N_CIRCLES, k=1)
N_PAIRS = len(TRIU_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, duals, and gradient."""

def obj_slsqp(v):

    return -np.sum(v[2 * N_CIRCLES:])

def cons_slsqp(v):

def generate_hex(row_counts, r0):
    """Generates a hexagonal lattice configuration with specified row counts."""

    pts = []
    y = r0

        width = (cnt - 1) * 2 * r0

            if len(pts) >= n:
                break
            pts.append([x_start + k * 2 * r0, y])
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
