import numpy as np
from scipy.optimize import linprog, minimize
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
N_PAIRS = len(TRIU_I)

def compute_dists(centers):
    """Computes pairwise Euclidean distances for upper triangular pairs."""
    diff = centers[TRIU_I] - centers[TRIU_J]
    return np.sqrt(np.sum(diff**2, axis=1))

def solve_lp_and_grad(centers):
    """Solves LP for max sum of radii and computes gradient w.r.t centers using dual variables."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    diff = centers[TRIU_I] - centers[TRIU_J]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    
    A_ub = np.zeros((N_PAIRS, N))
    A_ub[np.arange(N_PAIRS), TRIU_I] = 1.0
    A_ub[np.arange(N_PAIRS), TRIU_J] = 1.0
    b_ub = dists
    
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, None, None
    except Exception:
        return None, None, None
        
    r = np.maximum(res.x, 0.0)
    s = -res.fun
    
    grad = np.zeros((N, 2))
    try:
        if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            lams = np.abs(res.ineqlin.marginals)
            valid = lams > 1e-7

                k = np.where(valid)[0]
                ii = TRIU_I[k]
                jj = TRIU_J[k]
                lam_v = lams[k]
                d_v = dists[k]
                d_v_safe = np.where(d_v > 1e-9, d_v, 1.0)
                dx = centers[ii, 0] - centers[jj, 0]
                dy = centers[ii, 1] - centers[jj, 1]
                
                fx = lam_v * dx / d_v_safe
                fy = lam_v * dy / d_v_safe
                
                np.add.at(grad[:, 0], ii, fx)
                np.add.at(grad[:, 1], ii, fy)
                np.add.at(grad[:, 0], jj, -fx)
                np.add.at(grad[:, 1], jj, -fy)
    except Exception:
        pass
    return r, s, grad

def force_simulate(centers, radii, steps=800):
    """Vectorized force-directed simulation to pack circles tightly."""
    vel = np.zeros_like(centers)
    dt = 0.005
    damp = 0.85
    for _ in range(steps):
        radii *= 1.00005
        diff = centers[:, None, :] - centers[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dist, 1e9)
        
        overlap = np.maximum(0.0, radii[:, None] + radii[None, :] - dist)
        force_mag = overlap * 200.0 / (dist + 1e-9)
        f = np.sum(diff * force_mag[:, :, None], axis=1)
        
        wall_f = np.zeros_like(centers)
        wall_f[:, 0] += np.clip(radii - centers[:, 0], 0, None) * 500.0
        wall_f[:, 0] -= np.clip(centers[:, 0] + radii - 1.0, 0, None) * 500.0
        wall_f[:, 1] += np.clip(radii - centers[:, 1], 0, None) * 500.0
        wall_f[:, 1] -= np.clip(centers[:, 1] + radii - 1.0, 0, None) * 500.0
        f += wall_f
        
        vel = damp * vel + f * dt
        centers += vel
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    return centers, radii

def generate_hex_init(rows, r0):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r0
    for i, cnt in enumerate(rows):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def obj_slsqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_slsqp(v):
    """Inequality constraints >= 0 for SLSQP."""
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    con = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[TRIU_I] - cx[TRIU_J]
    dy = cy[TRIU_I] - cy[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    con = np.concatenate([con, dx**2 + dy**2 - dr**2])
    return con
