from scipy.optimize import linprog, minimize

N = 26
TRIU_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(TRIU_IDX[0])

# Precompute A_ub for LP to speed up repeated calls
A_LP = np.zeros((N_PAIRS, N))
A_LP[np.arange(N_PAIRS), TRIU_IDX[0]] = 1.0
A_LP[np.arange(N_PAIRS), TRIU_IDX[1]] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, and gradient."""
    n = N
    # Wall constraints limits
    lims = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]),
                      np.minimum(centers[:,1], 1.0-centers[:,1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    # Pairwise distances
    diff = centers[TRIU_IDX[0]] - centers[TRIU_IDX[1]]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    b_ub = dists
    
    try:
        res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            r = np.maximum(res.x, 0.0)
            s = -res.fun
            
            # Compute gradient of sum of radii wrt centers using LP duals (shadow prices)
            grad = np.zeros((n, 2))
            if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
                lam = np.abs(res.ineqlin.marginals)
                active = lam > 1e-7
                if np.any(active):
                    ii = TRIU_IDX[0][active]
                    jj = TRIU_IDX[1][active]

                    d_a = dists[active]

                        grad[ii[k]] += dirs[k] * lam_a[k]
                        grad[jj[k]] -= dirs[k] * lam_a[k]
            return r, s, grad
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0, np.zeros((n, 2))

def force_sim(centers, steps=1200):

    c = centers.copy()
    dt = 0.003
    for step in range(steps):
        target_r = 0.03 + 0.07 * (step / steps)

        np.fill_diagonal(dist, np.inf)

        forces[mask_r, 0] -= (c[mask_r,0] - (1.0 - target_r)) * 800.0
        mask_b = c[:,1] < target_r
        forces[mask_b, 1] += (target_r - c[mask_b,1]) * 800.0
        mask_t = c[:,1] > 1.0 - target_r
        forces[mask_t, 1] -= (c[mask_t,1] - (1.0 - target_r)) * 800.0
        
        c += forces * dt
        c = np.clip(c, 1e-4, 1.0-1e-4)
    return c

def hex_init(row_counts, r0):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        width = (cnt - 1) * 2 * r0
        x_start = 0.5 - width/2.0 + shift
        for k in range(cnt):
            if len(pts) >= N: break
            pts.append([x_start + k * 2 * r0, y])
        y += np.sqrt(3) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])
