

```python
import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_U, J_U = np.triu_indices(N, k=1)
M = len(I_U)

def lp_solve(c):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient."""
    lims = np.minimum(np.minimum(c[:,0], 1.0-c[:,0]), np.minimum(c[:,1], 1.0-c[:,1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    diff = c[I_U] - c[J_U]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    
    A = np.zeros((M, N))
    A[np.arange(M), I_U] = 1.0
    A[np.arange(M), J_U] = 1.0
    
    try:
        res = linprog(-np.ones(N), A_ub=A, b_ub=dists, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            r = np.maximum(res.x, 0.0)
            s = -res.fun
            
            grad = np.zeros((N, 2))
            try:
                lams = np.abs(res.ineqlin.marginals)
            except AttributeError:
                lams = np.zeros(M)
                
            mask = lams > 1e-7
            if np.any(mask):
                ii = I_U[mask]
                jj = J_U[mask]
                lam = lams[mask]
                d = np.sqrt(np.sum((c[ii]-c[jj])**2, axis=1))
                d = np.where(d > 1e-9, d, 1.0)
                diff_m = c[ii] - c[jj]
                factor = lam[:, None] / d[:, None]
                np.add.at(grad, ii, diff_m * factor)
                np.add.at(grad, jj, -diff_m * factor)
            return r, s, grad
        return np.full(N, 1e-5), 0.0, np.zeros((N,2))
    except Exception:
        return np.full(N, 1e-5), 0.0, np.zeros((N,2))

def obj_j(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def con_j(v):
    """Inequality constraints for SLSQP >= 0."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    
    dx = x[I_U] - x[J_U]
    dy = y[I_U] - y[J_U]
    dr = r[I_U] + r[J_U]
    c5 = dx**2 + dy**2 - dr**2
    
    return np.concatenate([c1, c2, c3, c4, c5])

def generate_hex_patterns():
    """Generates diverse hexagonal lattice configurations."""
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [7,5,5,5,4], [5,6,6,5,4], [5,5,5,5,6],
        [6,5,5,5,5], [4,5,6,6,5], [5,4,6,6,5]
    ]
    starts = []
    for p in pats:
        pts = []
        y = 0.10
        for i, cnt in enumerate(p):
            shift = 0.10 if i % 2 == 1 else 0.0
            w = (cnt - 1) * 0.20
            x0 = 0.5 - w / 2.0 + shift
            for k in range(cnt):
                pts.append([x0 + k * 0.20, y])
            y += 0.10 * np.sqrt(3)
        if len(pts) >= N:
            starts.append(np.array(pts[:N]))
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_s = -1.0
    best_c = None
    best_r = None
    
    # 1. Generate initial configurations
    starts = generate_hex_patterns()
    for _ in range(15):
        starts.append(np.random.uniform(0.15, 0.85, (N, 2)))
        
    # 2. LP Gradient Ascent & Local Search
    for c0 in starts:
        c = c0.copy()
        r, s, g = lp_solve(c)
        
        step = 0.05
        for it in range(1500):
            gn = np.linalg.norm(g)
            if gn < 1e-8:
                break
            cn = c + step * g / gn
            cn = np.clip(cn, 0.02, 0.98)
            rn, sn, gn = lp_solve(cn)
            
            if sn > s + 1e-8:
                c = cn
                r = rn
                s = sn
                g = gn
                step = min(step * 1.015, 0.18)
            else:
                step *= 0.94
                
        if s > best_s:
            best_s = s
            best_c = c.copy()
            best_r = r.copy()
            
        # Local random perturbations to escape plateaus
        for _ in range(6):
            cp = best_c + np.random.uniform(-0.025, 0.025, (N, 2))
            cp = np.clip(cp, 0.05, 0.95)
            rp, sp, gp = lp_solve(cp)
            
            step_p = 0.04
            for _ in range(800):
                gn_p = np.linalg.norm(gp)
                if gn_p < 1e-8:
                    break
                cpn = cp + step_p * gp / gn_p
                cpn = np.clip(cpn, 0.02, 0.98)
                rpn, spn, gpn = lp_solve(cpn)
                
                if spn > sp + 1e-8:
                    cp = cpn
                    rp = rpn
                    sp = spn
                    gp = gpn
                    step_p = min(step_p * 1.015, 0.18)
                else:
                    step_p *= 0.94
                    
            if sp > best_s:
                best_s = sp
                best_c = cp.copy()
                best_r = rp.copy()

    # 3. SLSQP Joint Polish
    if best_c is not None:
        x0 = np.concatenate([best_c[:,0], best_c[:,1], best_r * 0.995])
        bnds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
        try:
            res = minimize(obj_j, x0, method='SLSQP', bounds=bnds,
                          constraints={'type': 'ineq', 'fun': con_j},
                          options={'maxiter': 8000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                cx = res.x[:N]
                cy = res.x[N:2*N]
                c_pol = np.column_stack((cx, cy))
                r_pol, s_pol, _ = lp_solve(c_pol)
                if s_pol > best_s:
                    best_s = s_pol
                    best_c = c_pol.copy()
                    best_r = r_pol.copy()
        except Exception:
            pass

    # Fallback
    if best_c is None:
        best_c = starts[0]
        best_r, best_s, _ = lp_solve(best_c)

    # 4. Strict Safety Scaling
    scale = 1.0
    for i in range(N):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.999999
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
```
