import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26
TRIU_I, TRIU_J = np.triu_indices(N_CIRCLES, k=1)
N_PAIRS = len(TRIU_I)

def build_A_ub():
    A = np.zeros((N_PAIRS, N_CIRCLES))
    A[np.arange(N_PAIRS), TRIU_I] = 1.0
    A[np.arange(N_PAIRS), TRIU_J] = 1.0
    return A

A_Ub = build_A_ub()

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, dual magnitudes, distances."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    diff = centers[TRIU_I] - centers[TRIU_J]
    dists = np.sqrt(np.sum(diff**2, axis=1) + 1e-12)
    b_ub = dists
    
    try:
        res = linprog(c_obj, A_ub=A_Ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            r = np.maximum(res.x, 0.0)
            s = -res.fun
            mags = np.zeros(N_PAIRS)
            if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
                mags = np.abs(res.ineqlin.marginals)
            return r, s, mags, dists
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0, np.zeros(N_PAIRS), np.ones(N_PAIRS)

def obj_slsqp(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N_CIRCLES:])

def cons_slsqp(v):
    """Inequality constraints >= 0 for SLSQP."""
    n = N_CIRCLES
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    
    dx = x[TRIU_I] - x[TRIU_J]
    dy = y[TRIU_I] - y[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    c5 = dx**2 + dy**2 - dr**2
    
    return np.concatenate([c1, c2, c3, c4, c5])

def optimize_centers_lp(centers, rng, max_iters=600):

    n = N_CIRCLES
    best_c = centers.copy()
    best_r, best_s, mags, dists = solve_lp(best_c)
    step = 0.045
    
    for it in range(max_iters):
        # Periodic shake to escape symmetric plateaus
        if it > 0 and it % 60 == 0:
            best_c += rng.uniform(-0.004, 0.004, best_c.shape)
            best_c = np.clip(best_c, 1e-4, 1.0-1e-4)
            
        r, s, mags, dists = solve_lp(best_c)
        if s <= best_s - 1e-8:
            break
            
        grad = np.zeros((n, 2))
        active = mags > 1e-5
        if np.any(active):
            ii = TRIU_I[active]
            jj = TRIU_J[active]
            mm = mags[active]
            d = dists[active]
            diff = best_c[ii] - best_c[jj]
            factors = (mm / d)[:, None]
            np.add.at(grad, ii, diff * factors)
            np.add.at(grad, jj, -diff * factors)
            
        # Strong wall repulsion to maintain feasibility
        w = 15.0
        if best_c[:,0] < 0.015: grad[:,0] -= w
        if best_c[:,0] > 0.985: grad[:,0] += w
        if best_c[:,1] < 0.015: grad[:,1] -= w
        if best_c[:,1] > 0.985: grad[:,1] += w

            else:
                step *= 0.945
        else:
            step *= 0.95
            
        if step < 1e-7:
            break
            
    return best_c, best_s

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [6,5,4,6,5], [5,4,6,6,5], [7,5,5,5,4],
        [5,6,6,5,4], [5,5,5,5,6], [4,5,6,5,6], [5,6,4,5,6]
    ]
    
    for pat in patterns:
        if sum(pat) != N_CIRCLES: continue
        pts = []
        y = 0.10
        for i, cnt in enumerate(pat):
            shift = 0.10 if i % 2 == 1 else 0.0
            width = (cnt - 1) * 0.20
            x_start = 0.5 - width / 2.0 + shift
            for k in range(cnt):
                if len(pts) >= N_CIRCLES: break
                pts.append([x_start + k * 0.20, y])
            y += np.sqrt(3) * 0.10
        while len(pts) < N_CIRCLES:
            pts.append([0.5, 0.5])
        base = np.array(pts[:N_CIRCLES])
        starts.append(base)
        
        for _ in range(3):
            p = base + rng.uniform(-0.03, 0.03, base.shape)
            p = np.clip(p, 0.05, 0.95)
            starts.append(p)
            
    for _ in range(8):
        starts.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)))
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

                        best_sum = s_pol
                        best_centers = c_pol.copy()

            except Exception:
                pass

            best_centers = c_opt.copy()
            best_radii, best_sum, _, _ = solve_lp(best_centers)

    # Phase 3: Strict numerical safety scaling
    if best_centers is not None:
        scale = 1.0
        for i in range(N_CIRCLES):
            x, y, r = best_centers[i,0], best_centers[i,1], best_radii[i]

                rs = best_radii[i] + best_radii[j]

        best_sum = float(np.sum(best_radii))
    else:
        # Strict fallback
        best_centers = starts[0]
        best_radii, best_sum, _, _ = solve_lp(best_centers)
        best_radii *= 0.9999999
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
