import numpy as np
from scipy.optimize import minimize, linprog

def get_triu_indices(n):
    return np.triu_indices(n, k=1)

def build_lp_A(n):
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp(centers, A_ub, n):
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b[k] = np.hypot(dx, dy)
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-7), 0.0

def force_simulate(centers, steps=1200):

        force_mag = overlap / dists

        f_wall[:, 1] -= np.clip(c[:, 1] + target_r - 1.0, 0, None) * 150.0
        
        f = f_pair + f_wall
        c += f * dt
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
        
    return c

def joint_obj(v, n):
    return -np.sum(v[2*n:])

def joint_cons(v, n, idx_i, idx_j):
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    con = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    r_sum = r[idx_i] + r[idx_j]
    con = np.concatenate([con, dx**2 + dy**2 - r_sum**2])
    return con
