import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')
np.seterr(all='ignore')

def build_lp_matrix(n):
    """Precomputes the constant inequality matrix structure for the LP solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        for _ in range(4):
            A[k, i] = 1.0
            k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp_and_grad(centers, n, A_ub, triu_i, triu_j):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns optimal radii, sum of radii, and gradient of sum w.r.t centers using LP duals.
    """
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b[4 * n + idx] = d
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success and np.isfinite(res.fun):
            r = res.x
            obj = -res.fun
            
            # Compute gradient using dual variables (shadow prices)
            grad = np.zeros((n, 2))
            if hasattr(res, 'ineqlin') and res.ineqlin is not None and res.ineqlin.marginals is not None:
                duals = np.abs(res.ineqlin.marginals)
                
                # Pairwise constraints gradient
                for k in range(len(triu_i)):
                    lam = duals[4 * n + k]
                    if lam > 1e-9:
                        i, j = triu_i[k], triu_j[k]
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        d = np.hypot(dx, dy)
                        if d > 1e-9:
                            factor = lam / d
                            grad[i, 0] += dx * factor
                            grad[i, 1] += dy * factor
                            grad[j, 0] -= dx * factor
                            grad[j, 1] -= dy * factor
                            
                # Wall constraints gradient
                for i in range(n):
                    grad[i, 0] += duals[4 * i] - duals[4 * i + 1]
                    grad[i, 1] += duals[4 * i + 2] - duals[4 * i + 3]
                    
            return r, obj, grad
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0, np.zeros((n, 2))

def s_obj(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def s_cons(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing (squared distances for smooth gradients)."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[triu_i] - cx[triu_j]
    dy = cy[triu_i] - cy[triu_j]
    dr = r[triu_i] + r[triu_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def generate_hex(n, rows, r0, rng):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    rng = np.random.default_rng(42)
    A_ub = build_lp_matrix(n)

        [5,6,6,4,5], [6,5,5,6,4], [5,5,5,6,5], [6,5,6,4,5]

                    curr_c = shake

                    c_pol = np.column_stack((res.x[:n], res.x[n:2*n]))

            except Exception:
                pass

    # Final strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    c = best_centers
    r = best_radii
    for i in range(n):
        x, y = c[i]
        ri = r[i]
        if ri > 1e-12:
            scale = min(scale, x / ri, (1.0 - x) / ri, y / ri, (1.0 - y) / ri)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
            rs = r[i] + r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    r *= scale * 0.9999999
    final_sum = float(np.sum(r))
    
    return c, r, final_sum
