1. **LP-based radius evaluation is exact and fast**: For fixed centers, maximizing $\sum r_i$ is a linear program. We can use this as a highly accurate objective function.
2. **Simulated Annealing (SA) escapes local minima better than random restarts**: By treating center coordinates as the primary variables and evaluating radii via LP, SA can explore the configuration space effectively. We use a cooling schedule that allows early exploration and late fine-tuning.
3. **SLSQP Joint Polish**: After SA finds a promising basin, a joint SLSQP optimization on both centers and radii handles the non-linear constraints smoothly to squeeze out the final milliradians.
4. **Hill-Climbing Refinement**: A coordinate-descent style perturbation with decreasing step sizes ensures we don't miss tiny improvements near the optimum.
5. **Structural Diversity**: We initialize with multiple hexagonal row distributions (which are known to be dense) plus random starts to cover different symmetry classes.

import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Precomputes the constant inequality matrix structure for the LP solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        for _ in range(4):
            A[k, i] = 1.0
            k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp(centers, A_ub, n, triu_idx):
    """Solves LP to maximize sum of radii for fixed centers."""
    b_ub = np.empty(4 * n + n*(n-1)//2)
    b_ub[:n] = centers[:, 0]
    b_ub[n:2*n] = 1.0 - centers[:, 0]
    b_ub[2*n:3*n] = centers[:, 1]
    b_ub[3*n:4*n] = 1.0 - centers[:, 1]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(diffs[:,:,0]**2 + diffs[:,:,1]**2)
    np.fill_diagonal(dists, np.inf)
    b_ub[4*n:] = dists[triu_idx]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def joint_obj(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def joint_cons(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[triu_i] - cx[triu_j]
    dy = cy[triu_i] - cy[triu_j]
    dist = np.hypot(dx, dy)
    r_sum = r[triu_i] + r[triu_j]
    c = np.concatenate([c, dist - r_sum])
    return c

def run_sa(centers, n, A_ub, triu_idx, rng, steps):
    """Simulated Annealing on centers, evaluating radii via LP."""
    c = centers.copy()
    r, s = solve_lp(c, A_ub, n, triu_idx)
    best_c_sa, best_r_sa, best_s_sa = c.copy(), r.copy(), s
    
    for i in range(steps):
        progress = i / steps
        temp = 0.008 * (1.0 - progress)
        step = 0.02 * (1.0 - progress) + 0.0005
        idx = rng.integers(n)
        
        c_new = c.copy()
        c_new[idx] += rng.uniform(-step, step, 2)
        c_new[idx] = np.clip(c_new[idx], 0.002, 0.998)
        
        r_new, s_new = solve_lp(c_new, A_ub, n, triu_idx)
        delta = s_new - s
        
        if delta > 0 or rng.random() < np.exp(delta / max(temp, 1e-6)):
            c = c_new
            r = r_new
            s = s_new
            if s > best_s_sa:
                best_s_sa = s
                best_c_sa = c.copy()
                best_r_sa = r.copy()
                
    return best_c_sa, best_r_sa, best_s_sa

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.seterr(all='ignore')
    n = 26
    rng = np.random.default_rng(42)
    A_ub = build_lp_matrix(n)
    triu_i, triu_j = np.triu_indices(n, k=1)
    triu_idx = (triu_i, triu_j)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    inits = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], 
                [4,6,6,6,4], [7,5,5,5,4], [5,7,4,5,5], [6,4,6,6,4]]
    
    for pat in patterns:
        pts = []
        y = 0.095
        row = 0
        for cnt in pat:
            shift = 0.095 if row % 2 == 1 else 0.0
            x = 0.095 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 0.19
            y += 0.095 * np.sqrt(3)
            row += 1
        while len(pts) < n: pts.append([0.5, 0.5])
        inits.append(np.array(pts[:n]))
        
    for _ in range(5):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # Phase 1 & 2: SA + SLSQP Polish for each initialization
    for cfg in inits:
        c_sa, r_sa, s_sa = run_sa(cfg, n, A_ub, triu_idx, rng, steps=10000)
        if s_sa > best_sum:
            best_sum = s_sa
            best_centers = c_sa
            best_radii = r_sa
            
        v0 = np.concatenate([c_sa[:,0], c_sa[:,1], r_sa])
        bounds_v = [(0.0, 1.0)]*(2*n) + [(1e-6, 0.5)]*n
        try:
            res = minimize(joint_obj, v0, args=(n,), method='SLSQP',
                           bounds=bounds_v,
                           constraints={'type': 'ineq', 'fun': joint_cons, 'args': (n, triu_i, triu_j)},
                           options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
            if np.isfinite(res.fun):
                c_pol = np.column_stack((res.x[:n], res.x[n:2*n]))
                r_pol, s_pol = solve_lp(c_pol, A_ub, n, triu_idx)
                if s_pol > best_sum:
                    best_sum = s_pol
                    best_centers = c_pol
                    best_radii = r_pol
        except Exception:
            pass

    # Phase 3: Fine Hill-Climbing Refinement
    curr_c = best_centers.copy()
    curr_r, curr_s = solve_lp(curr_c, A_ub, n, triu_idx)
    step = 0.008
    for _ in range(5000):
        idx = rng.integers(n)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.uniform(-step, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.002, 0.998)
        
        r_try, s_try = solve_lp(curr_c, A_ub, n, triu_idx)
        if s_try > curr_s + 1e-9:
            curr_s = s_try
            curr_r = r_try
            if curr_s > best_sum:
                best_sum = curr_s
                best_centers = curr_c.copy()
                best_radii = curr_r.copy()
        else:
            curr_c[idx] = old
        step *= 0.9995
        
    # Phase 4: Strict numerical safety scaling
    scale = 1.0
    c = best_centers
    r = best_radii
    for i in range(n):
        x, y = c[i]
        ri = r[i]
        if ri > 1e-12:
            scale = min(scale, x/ri, (1.0-x)/ri, y/ri, (1.0-y)/ri)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
            rs = r[i] + r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    r *= scale * 0.9999999
    final_sum = float(np.sum(r))
    
    return c, r, final_sum
