import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

def build_lp_matrix(n):
    """Precomputes the constant inequality matrix structure for the LP solver."""
    m = n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp_and_forces(centers, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers and returns dual forces."""
    n = centers.shape[0]
    walls = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]),
                       np.minimum(centers[:,1], 1.0 - centers[:,1]))
    bounds = [(0.0, max(w, 1e-12)) for w in walls]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return np.full(n, 1e-6), 0.0, np.zeros((n, 2))
    except Exception:
        return np.full(n, 1e-6), 0.0, np.zeros((n, 2))
        
    r = res.x
    sum_r = -res.fun
    
    forces = np.zeros((n, 2))
    try:
        lams = np.abs(res.ineqlin.marginals)
        valid = lams > 1e-7
        if np.any(valid):
            lam_v = lams[valid]
            ii = idx_i[valid]
            jj = idx_j[valid]
            d_v = dists[ii, jj]
            d_safe = np.maximum(d_v, 1e-9)
            diff_v = centers[ii] - centers[jj]
            f_vec = (lam_v / d_safe)[:, None] * diff_v
            np.add.at(forces, ii, f_vec)
            np.add.at(forces, jj, -f_vec)
    except Exception:
        pass
        
    return r, sum_r, forces

def hex_init(n, rows, r0):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def joint_obj(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def joint_cons(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[triu_i] - cx[triu_j]
    dy = cy[triu_i] - cy[triu_j]
    dist = np.hypot(dx, dy)
    r_sum = r[triu_i] + r[triu_j]
    c = np.concatenate([c, dist - r_sum])
    return c

def optimize_config(cfg, n, A_ub, triu_i, triu_j, rng):
    """Runs multi-phase optimization on a single configuration."""
    centers = cfg.copy()
    step = 0.05
    
    # Phase 1: LP-guided force ascent
    for _ in range(120):
        r, s, forces = solve_lp_and_forces(centers, A_ub, triu_i, triu_j)
        f_norm = np.linalg.norm(forces)
        if f_norm < 1e-6:
            break
        centers += step * forces / f_norm
        centers = np.clip(centers, 1e-4, 0.999)
        step *= 0.98
        
    # Phase 2: SLSQP joint polish
    r_lp, _, _ = solve_lp_and_forces(centers, A_ub, triu_i, triu_j)
    v0 = np.concatenate([centers[:,0], centers[:,1], r_lp * 0.95])
    bounds_v = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    try:
        res = minimize(joint_obj, v0, args=(n,), method='SLSQP',
                       bounds=bounds_v,
                       constraints={'type': 'ineq', 'fun': joint_cons, 'args': (n, triu_i, triu_j)},
                       options={'maxiter': 2000, 'ftol': 1e-14, 'disp': False})
        if np.isfinite(res.fun):
            centers = np.column_stack((res.x[:n], res.x[n:2*n]))
    except Exception:
        pass
        
    r_opt, sum_opt, _ = solve_lp_and_forces(centers, A_ub, triu_i, triu_j)
    
    # Phase 3: Adaptive Hill Climbing on centers
    curr_c = centers.copy()
    curr_r = r_opt.copy()
    curr_s = sum_opt
    step = 0.012
    
    for it in range(600):
        i = rng.integers(n)
        old = curr_c[i].copy()
        curr_c[i] += rng.uniform(-step, step, 2)
        curr_c[i] = np.clip(curr_c[i], 0.005, 0.995)
        
        r_try, s_try, _ = solve_lp_and_forces(curr_c, A_ub, triu_i, triu_j)
        if s_try > curr_s + 1e-8:
            curr_s = s_try
            curr_r = r_try.copy()
        else:
            curr_c[i] = old
        step *= 0.998
        
    return curr_c, curr_r, curr_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    A_ub = build_lp_matrix(n)
    triu_i, triu_j = np.triu_indices(n, k=1)

    ]
    
    configs = []
    for pat in patterns:

            c = hex_init(n, pat, 0.095)
            configs.append(c)

                cp = c + rng.uniform(-0.025, 0.025, (n, 2))
                configs.append(np.clip(cp, 0.05, 0.95))

            best_sum = s_res
            best_centers = c_res.copy()
            best_radii = r_res.copy()
            
    # Final strict safety scaling to guarantee numerical validity
    scale = 1.0
    c = best_centers
    r = best_radii
    for i in range(n):
        x, y = c[i]
        ri = r[i]
        if ri > 1e-12:
            scale = min(scale, x / ri, (1.0 - x) / ri, y / ri, (1.0 - y) / ri)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(c[i,0] - c[j,0], c[i,1] - c[j,1])
            rs = r[i] + r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    r *= scale * 0.9999999
    final_sum = float(np.sum(r))
    
    return c, r, final_sum
