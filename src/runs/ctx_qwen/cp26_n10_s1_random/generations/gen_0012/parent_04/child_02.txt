import numpy as np
from scipy.optimize import linprog, minimize

def build_lp_data(n):
    """Precomputes constant LP matrix and pairwise indices."""
    idx_i, idx_j = np.triu_indices(n, k=1)
    m_pairs = len(idx_i)
    A_ub = np.zeros((4 * n + m_pairs, n))
    for i in range(n):
        A_ub[4 * i, i] = 1.0
        A_ub[4 * i + 1, i] = 1.0
        A_ub[4 * i + 2, i] = 1.0
        A_ub[4 * i + 3, i] = 1.0
    for k, (i, j) in enumerate(zip(idx_i, idx_j)):
        A_ub[4 * n + k, i] = 1.0
        A_ub[4 * n + k, j] = 1.0
    return A_ub, idx_i, idx_j

def solve_lp_and_grad(centers, A_ub, idx_i, idx_j):
    """Solves LP for radii and computes gradient of sum radii w.r.t centers."""
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    
    b_ub = np.empty(4 * n + len(idx_i))
    b_ub[:n] = x
    b_ub[n:2 * n] = 1.0 - x
    b_ub[2 * n:3 * n] = y
    b_ub[3 * n:4 * n] = 1.0 - y
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, 0.0)
    b_ub[4 * n:] = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success or not np.isfinite(res.fun):
            return np.full(n, 1e-6), 0.0, np.zeros((n, 2))
    except Exception:
        return np.full(n, 1e-6), 0.0, np.zeros((n, 2))
        
    r = res.x
    s = -res.fun
    
    grad = np.zeros((n, 2))
    # Marginals for pairwise constraints. 
    # For min problem, active constraints typically have negative marginals.
    # We take -marginals to get positive sensitivity (how much sum_r improves per unit distance increase).
    lams = np.maximum(-res.ineqlin.marginals[4 * n:], 0.0)
    active = lams > 1e-7
    if np.any(active):
        k = np.where(active)[0]
        lam_v = lams[k]
        ii = idx_i[k]
        jj = idx_j[k]
        
        dxx = centers[ii, 0] - centers[jj, 0]
        dyy = centers[ii, 1] - centers[jj, 1]
        ds = np.hypot(dxx, dyy)
        ds = np.where(ds > 1e-9, ds, 1.0)
        
        factor = lam_v / ds
        grad[ii, 0] += factor * dxx
        grad[ii, 1] += factor * dyy
        grad[jj, 0] -= factor * dxx
        grad[jj, 1] -= factor * dyy
        
    return r, s, grad

def optimize_centers(c0, A_ub, idx_i, idx_j, rng):
    """Gradient ascent on centers using LP sensitivities."""
    c = c0.copy()
    best_s = -1.0
    best_c = None
    best_r = None
    
    step = 0.025
    for it in range(3000):
        r, s, g = solve_lp_and_grad(c, A_ub, idx_i, idx_j)
        if s > best_s:
            best_s = s
            best_c = c.copy()
            best_r = r.copy()
            
        gn = np.linalg.norm(g)
        if gn < 1e-8:
            break
            
        c_new = c + step * g / gn
        c_new = np.clip(c_new, 0.005, 0.995)
        
        r_new, s_new, _ = solve_lp_and_grad(c_new, A_ub, idx_i, idx_j)
        if s_new > s + 1e-9:
            c = c_new
            step = min(step * 1.005, 0.05)
        else:
            step *= 0.985
            
        # Periodic shake to escape local contact-graph minima
        if it % 150 == 0 and it > 0:
            c_shake = c + rng.uniform(-0.008, 0.008, (26, 2))
            c_shake = np.clip(c_shake, 0.02, 0.98)
            r_sh, s_sh, _ = solve_lp_and_grad(c_shake, A_ub, idx_i, idx_j)
            if s_sh > best_s:
                c = c_shake
                step = 0.025
                
    return best_c, best_r, best_s

def hex_init(n, rows, r0):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def obj_joint(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_joint(v, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid packing."""
    cx, cy, r = v[:n], v[n:2*n], v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dist = np.hypot(dx, dy)
    rs = r[idx_i] + r[idx_j]
    return np.concatenate([c, dist - rs])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.seterr(all='ignore')
    n = 26
    rng = np.random.default_rng(42)
    A_ub, idx_i, idx_j = build_lp_data(n)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Diverse row distributions summing to >= 26
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5],
        [4,6,6,6,4], [7,5,5,5,4], [5,7,5,5,4], [6,4,6,6,4],
        [5,6,6,4,5], [6,5,5,6,4], [5,5,5,6,5], [6,6,4,6,4]
    ]
    
    configs = []
    for pat in patterns:
        if sum(pat) >= n:
            c = hex_init(n, pat, 0.095)
            configs.append(c)
            # Add perturbed variants to break symmetry
            for _ in range(4):
                cp = c + rng.uniform(-0.015, 0.015, (n, 2))
                configs.append(np.clip(cp, 0.05, 0.95))
                
    # Add dense random starts
    for _ in range(8):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # Phase 1: LP-Gradient Ascent from diverse starts
    for cfg in configs:
        c_opt, r_opt, s_opt = optimize_centers(cfg, A_ub, idx_i, idx_j, rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            best_radii = r_opt
            
    # Phase 2: SLSQP Joint Polish to fine-tune centers & radii simultaneously
    if best_centers is not None:
        v0 = np.concatenate([best_centers[:, 0], best_centers[:, 1], best_radii])
        bounds_v = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
        try:
            res = minimize(obj_joint, v0, args=(n,), method='SLSQP', bounds=bounds_v,
                           constraints={'type': 'ineq', 'fun': cons_joint, 'args': (n, idx_i, idx_j)},
                           options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if np.isfinite(res.fun):
                c_pol = np.column_stack((res.x[:n], res.x[n:2*n]))
                r_pol, s_pol, _ = solve_lp_and_grad(c_pol, A_ub, idx_i, idx_j)
                if s_pol > best_sum:
                    best_sum = s_pol
                    best_centers = c_pol
                    best_radii = r_pol
        except Exception:
            pass

    # Phase 3: Strict safety scaling to guarantee numerical validity
    scale = 1.0
    c = best_centers
    r = best_radii
    for i in range(n):
        x, y = c[i]
        ri = r[i]
        if ri > 1e-12:
            scale = min(scale, x/ri, (1.0-x)/ri, y/ri, (1.0-y)/ri)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
            rs = r[i] + r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    r *= scale * 0.9999995
    final_sum = float(np.sum(r))
    
    return c, r, final_sum
