import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Precomputes the constant inequality matrix structure for the LP solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        for _ in range(4):
            A[k, i] = 1.0
            k += 1
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[4*n + idx, i] = 1.0
            A[4*n + idx, j] = 1.0
            idx += 1
    return A

def solve_lp(centers, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            b[4*n + idx] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            idx += 1
            
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun, res
    return None, 0.0, None

def compute_center_gradient(res, centers, n, triu_i, triu_j):
    """Computes gradient of LP objective w.r.t center positions using dual variables."""
    grad = np.zeros((n, 2))
    if res is None:
        return grad
        
    lams = None
    if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
        lams = res.ineqlin.marginals
    elif hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
        lams = res.marginals.ineqlin
        
    if lams is None:
        return grad

    # Pairwise constraints start at index 4*n
    pair_lams = lams[4*n:]
    active = pair_lams > 1e-6
    if not np.any(active):
        return grad
        
    ii = triu_i[active]
    jj = triu_j[active]
    ll = pair_lams[active]
    
    diff = centers[ii] - centers[jj]
    dists = np.linalg.norm(diff, axis=1)
    safe_dists = np.where(dists > 1e-9, dists, 1.0)
    factors = (ll / safe_dists)[:, None]
    
    np.add.at(grad, ii, diff * factors)
    np.add.at(grad, jj, -diff * factors)
    return grad

def joint_obj(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def joint_cons(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[triu_i] - cx[triu_j]
    dy = cy[triu_i] - cy[triu_j]
    dist_sq = dx**2 + dy**2
    r_sum = r[triu_i] + r[triu_j]
    c = np.concatenate([c, dist_sq - r_sum**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.seterr(all='ignore')
    n = 26
    rng = np.random.default_rng(42)
    A_ub = build_lp_matrix(n)
    triu_i, triu_j = np.triu_indices(n, k=1)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5],
        [4,6,6,6,4], [7,5,5,5,4], [5,7,4,5,5], [6,4,6,6,4],
        [5,6,6,4,5], [6,5,5,6,4], [5,5,5,6,5], [6,5,6,4,5],

                pass

            best_sum = s_best_lp
            best_centers = curr_c.copy()
            best_radii = r_best_lp.copy()

    scale = 1.0
    c = best_centers
    r = best_radii
    for i in range(n):
        x, y = c[i]
        ri = r[i]
        if ri > 1e-12:
            scale = min(scale, x/ri, (1.0-x)/ri, y/ri, (1.0-y)/ri)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
            rs = r[i] + r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    r *= scale * 0.9999999
    final_sum = float(np.sum(r))
    
    return c, r, final_sum
