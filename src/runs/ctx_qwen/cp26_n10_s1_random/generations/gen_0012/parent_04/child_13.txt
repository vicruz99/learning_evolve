import numpy as np
from scipy.optimize import linprog, minimize

def hex_init(rows, r0, n):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def get_dists(c, triu_i, triu_j):
    """Computes pairwise Euclidean distances for upper triangular indices."""
    diff = c[:, None, :] - c[None, :, :]
    return np.sqrt(np.sum(diff**2, axis=2))[triu_i, triu_j]

def solve_lp_grad(c, n, A_ub, dists, triu_i, triu_j):
    """Solves LP for radii and computes gradient of sum of radii w.r.t centers using duals."""
    limits = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                        np.minimum(c[:, 1], 1.0 - c[:, 1]))
    bounds = [(0.0, max(l, 1e-12)) for l in limits]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if res.success:
            r = res.x
            s = -res.fun
            
            # Extract dual variables (marginals) for distance constraints
            try:
                margs = np.asarray(res.ineqlin.marginals)
                lams = np.maximum(-margs, 0.0)  # Negate because linprog minimizes
            except Exception:
                lams = np.zeros(len(dists))
                
            grad = np.zeros((n, 2))
            active = lams > 1e-7
            if np.any(active):
                ai = np.where(active)[0]
                lams_a = lams[active]
                i_a = triu_i[ai]
                j_a = triu_j[ai]
                d_a = np.maximum(dists[active], 1e-9)
                diff = c[i_a] - c[j_a]
                w = (lams_a / d_a)[:, None]
                contrib = diff * w
                np.add.at(grad, i_a, contrib)
                np.add.at(grad, j_a, -contrib)
            return r, s, grad
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0, np.zeros((n, 2))

def joint_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def joint_cons(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid joint packing."""
    cx = v[:n]
    cy = v[n:2*n]
    rr = v[2*n:]
    c1 = np.concatenate([cx - rr, 1.0 - cx - rr, cy - rr, 1.0 - cy - rr])
    dx = cx[triu_i] - cx[triu_j]
    dy = cy[triu_i] - cy[triu_j]
    c2 = dx**2 + dy**2 - (rr[triu_i] + rr[triu_j])**2
    return np.concatenate([c1, c2])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.seterr(all='ignore')
    n = 26
    rng = np.random.default_rng(42)
    
    triu_i, triu_j = np.triu_indices(n, k=1)
    m = len(triu_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), triu_i] = 1.0
    A_ub[np.arange(m), triu_j] = 1.0
    
    # 1. Generate diverse initial configurations
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5],
        [4,6,6,6,4], [7,5,5,5,4], [5,7,4,5,5], [6,4,6,6,4]
    ]
    for pat in patterns:
        if sum(pat) >= n:
            configs.append(hex_init(pat, 0.095, n))
            configs.append(hex_init(pat, 0.105, n))
    for _ in range(10):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # 2. Gradient Ascent on Centers
    for cfg in configs:

        dists = get_dists(c, triu_i, triu_j)
        r, s, g = solve_lp_grad(c, n, A_ub, dists, triu_i, triu_j)
        step = 0.025
        
        for it in range(600):
            gn = np.linalg.norm(g)

                    c = c_new
                    r = r_new
                    s = s_new

                    s = s_new

                    step *= 0.975

                    best_r = r_sl
                    best_sum = s_sl
        except Exception:
            pass
            
    # 5. Strict safety scaling to guarantee numerical validity
    scale = 1.0
    c = best_c
    r = best_r
    for i in range(n):
        x, y = c[i]
        ri = r[i]
        if ri > 1e-12:
            scale = min(scale, x / ri, (1.0 - x) / ri, y / ri, (1.0 - y) / ri)
            
    dists = get_dists(c, triu_i, triu_j)
    r_pair = r[triu_i] + r[triu_j]
    scale = min(scale, np.min(dists / np.maximum(r_pair, 1e-12)))
    
    r *= scale * 0.9999995
    final_sum = float(np.sum(r))
    
    return c, r, final_sum
