

<strategy>
The current solution uses SLSQP joint optimization and random hill-climbing. While functional, it can get trapped in local minima and lacks a principled way to move centers to maximize the sum of radii. 

Key improvements:
1. **LP Duality for Gradient Ascent**: The problem of maximizing radii for fixed centers is an LP. By LP duality, the sensitivity of the optimal sum of radii to changes in center positions is given by the dual variables (shadow prices) of the LP constraints. This provides an exact, analytically derived gradient direction to move centers to increase the total radius sum. This is far more efficient and directed than random hill-climbing.
2. **Dual-Guided Ascent Loop**: We implement a gradient ascent on centers using the dual variables from `scipy.optimize.linprog` (which uses the HiGHS solver). The duals for pairwise distance constraints push circles apart proportionally to their "binding" strength, while wall constraint duals push circles optimally relative to boundaries.
3. **Adaptive Step Size & Perturbation**: The ascent uses an adaptive step size that increases when progress is made and decreases otherwise. Random perturbations are injected when the gradient vanishes to escape plateaus.
4. **SLSQP Polishing**: After the dual-guided ascent converges, a joint SLSQP optimization fine-tunes both centers and radii simultaneously to squeeze out remaining gains and handle non-smooth transitions.
5. **Diverse Initializations**: Multiple hexagonal lattice patterns with varying row distributions and random placements are used to ensure broad exploration of the configuration space.

This approach leverages the mathematical structure of the packing problem, turning it into a fast, gradient-driven optimization that consistently outperforms generic heuristics.

import numpy as np
from scipy.optimize import linprog, minimize

def build_lp_data(n):
    """Precomputes the constant inequality matrix structure for the LP solver."""
    m_pairs = n * (n - 1) // 2
    A_ub = np.zeros((4 * n + m_pairs, n))
    for i in range(n):
        A_ub[4*i, i] = 1.0
        A_ub[4*i+1, i] = 1.0
        A_ub[4*i+2, i] = 1.0
        A_ub[4*i+3, i] = 1.0
        
    idx = 0
    triu_i, triu_j = [], []
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[4*n + idx, i] = 1.0
            A_ub[4*n + idx, j] = 1.0
            idx += 1
            triu_i.append(i)
            triu_j.append(j)
            
    return A_ub, np.array(triu_i), np.array(triu_j)

def solve_lp_duals(centers, A_ub, triu_i, triu_j):
    """Solves LP to maximize sum of radii for fixed centers and returns radii, sum, and gradient w.r.t centers."""
    n = centers.shape[0]
    b_ub = np.zeros(4 * n + len(triu_i))
    k = 0
    for i in range(n):
        x, y = centers[i]
        b_ub[k] = x; k += 1
        b_ub[k] = 1.0 - x; k += 1
        b_ub[k] = y; k += 1
        b_ub[k] = 1.0 - y; k += 1
        
    diff = centers[triu_i] - centers[triu_j]
    b_ub[4*n:] = np.sqrt(np.sum(diff**2, axis=1))
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            r = res.x
            s = -res.fun
            # Marginals correspond to constraints A_ub @ r <= b_ub.
            # For min problem, marginals <= 0. Sensitivity of max sum is -marginals.
            sens = -res.ineqlin.marginals
            
            grad = np.zeros((n, 2))
            # Wall sensitivities: dS/dx_i = sens_left - sens_right, dS/dy_i = sens_bottom - sens_top
            grad[:, 0] = sens[0::4] - sens[1::4]
            grad[:, 1] = sens[2::4] - sens[3::4]
            
            # Pairwise sensitivities push centers apart
            pair_sens = sens[4*n:]
            dists = b_ub[4*n:]
            dists_safe = np.where(dists > 1e-12, dists, 1.0)
            forces = pair_sens[:, None] * (diff / dists_safe[:, None])
            
            grad[triu_i] += forces
            grad[triu_j] -= forces
            
            return r, s, grad
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0, np.zeros((n, 2))

def obj_slqp(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_slqp(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing."""
    cx, cy, r = v[:n], v[n:2*n], v[2*n:]
    c_mat = np.column_stack((cx, cy))
    diff = c_mat[:, None, :] - c_mat[None, :, :]
    d = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(d, np.inf)
    return np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r, 
                           d[triu_i, triu_j] - (r[triu_i] + r[triu_j])])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.seterr(all='ignore')
    n = 26
    rng = np.random.default_rng(42)
    A_ub, triu_i, triu_j = build_lp_data(n)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Diverse hexagonal row distributions summing to >= 26
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], 
        [4,6,6,6,4], [7,5,5,5,4], [5,6,6,4,5], [6,5,5,6,4]
    ]
    configs = []
    for pat in patterns:
        if sum(pat) >= n:
            pts = []
            y = 0.10
            for ri, cnt in enumerate(pat):
                shift = 0.10 if ri % 2 == 1 else 0.0
                x = 0.10 + shift
                for _ in range(cnt):
                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2.0 * 0.10
                y += np.sqrt(3.0) * 0.10
            while len(pts) < n:
                pts.append([0.5, 0.5])
            c = np.array(pts[:n])
            configs.append(c)
            for _ in range(3):
                configs.append(np.clip(c + rng.uniform(-0.02, 0.02, (n,2)), 0.05, 0.95))
                
    for _ in range(6):
        configs.append(rng.uniform(0.15, 0.85, (n,2)))

    bounds_v = [(0.0, 1.0)]*(2*n) + [(1e-6, 0.5)]*n

    for cfg in configs:
        c = np.clip(cfg.copy(), 0.02, 0.98)
        step = 0.02
        curr_r, curr_s, _ = solve_lp_duals(c, A_ub, triu_i, triu_j)
        
        # Phase 1: Dual-Guided Gradient Ascent on Centers
        for it in range(1200):
            r, s, grad = solve_lp_duals(c, A_ub, triu_i, triu_j)
            if s < 1e-6: break
            
            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-8:
                c_new = c + step * grad / g_norm
                c_new = np.clip(c_new, 0.001, 0.999)
                r_new, s_new, _ = solve_lp_duals(c_new, A_ub, triu_i, triu_j)
                
                if s_new > s + 1e-7:
                    c = c_new
                    curr_r, curr_s = r_new, s_new
                    step = min(step * 1.01, 0.05)
                else:
                    step *= 0.95
            else:
                # Random perturbation to escape flat regions
                if rng.random() < 0.05:
                    c_pert = c + rng.uniform(-0.01, 0.01, (n,2))
                    c_pert = np.clip(c_pert, 0.01, 0.99)
                    r_p, s_p, _ = solve_lp_duals(c_pert, A_ub, triu_i, triu_j)
                    if s_p > curr_s:
                        c = c_pert
                        curr_r, curr_s = r_p, s_p
                        step = 0.02
                        
        if curr_s > best_sum:
            best_sum = curr_s
            best_centers = c.copy()
            best_radii = curr_r.copy()
            
        # Phase 2: SLSQP Joint Polish
        x0 = np.concatenate([c.flatten(), curr_r])
        try:
            res = minimize(obj_slqp, x0, args=(n,), method='SLSQP', bounds=bounds_v,
                           constraints={'type': 'ineq', 'fun': cons_slqp, 'args': (n, triu_i, triu_j)},
                           options={'maxiter': 2000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_pol = np.column_stack((res.x[:n], res.x[n:2*n]))
                r_pol, s_pol, _ = solve_lp_duals(c_pol, A_ub, triu_i, triu_j)
                if s_pol > best_sum:
                    best_sum = s_pol
                    best_centers = c_pol.copy()
                    best_radii = r_pol.copy()
        except Exception:
            pass

    # Phase 3: Strict numerical safety scaling
    scale = 1.0
    c, r = best_centers, best_radii
    for i in range(n):
        x, y = c[i]
        ri = r[i]
        if ri > 1e-12:
            scale = min(scale, x/ri, (1.0-x)/ri, y/ri, (1.0-y)/ri)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
            rs = r[i] + r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    r *= scale * 0.9999999
    return c, r, float(np.sum(r))
