import numpy as np
from scipy.optimize import minimize, linprog
import math

def solve_radii_lp(centers):
    """
    Solves the LP to find the maximum sum of radii for fixed centers.
    Maximize sum(r_i) subject to:
        0 <= r_i <= distance to boundary
        r_i + r_j <= distance(i, j)
    """
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]), 
                      np.minimum(centers[:,1], 1.0 - centers[:,1]))
    lims = np.maximum(lims, 0.0)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    c = np.ones(n) * -1
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = [(0.0, lims[i]) for i in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and not np.any(np.isnan(res.x)):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
        
    # Fallback heuristic if LP fails
    safe_r = np.minimum(lims, np.min(dists, axis=1) / 2.0)
    return np.maximum(safe_r, 0.0)

def objective(vars_array, n, mu):
    """
    Objective function for L-BFGS-B: minimize negative sum of radii + penalty for violations.
    """
    c = vars_array[:2*n].reshape(n, 2)
    r = vars_array[2*n:]
    
    obj = -np.sum(r)
    pen = 0.0
    
    # Overlap penalty
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((c[i] - c[j])**2))
            viol = r[i] + r[j] - d
            if viol > 0:
                pen += viol**2
                
    # Boundary penalty
    for i in range(n):
        x, y = c[i]
        ri = r[i]
        if x < ri: 
            pen += (x - ri)**2
        if x > 1.0 - ri: 
            pen += (x - (1.0 - ri))**2
        if y < ri: 
            pen += (y - ri)**2
        if y > 1.0 - ri: 
            pen += (y - (1.0 - ri))**2
            
    return obj + mu * pen

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    best_sum = 0.0
    best_res = None
    
    # Generate multiple high-quality initial configurations
    inits = []
    
    # 1. Hexagonal lattice initialization
    hex_pts = []
    r_start = 0.10
    dy = r_start * math.sqrt(3)
    for row in range(6):
        shift = r_start if row % 2 == 1 else 0.0
        y = r_start + row * dy
        x = r_start + shift
        while x + r_start <= 1.0:
            if len(hex_pts) < n:
                hex_pts.append([x, y])
            x += 2.0 * r_start
    while len(hex_pts) < n:
        hex_pts.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
    inits.append(np.array(hex_pts[:n]))
    
    # 2-5. Perturbed hexagonal configurations to escape local minima
    for _ in range(4):
        p = hex_pts.copy()
        np.random.shuffle(p)
        noise = np.random.normal(0, 0.02, size=(n, 2))
        p = np.array(p) + noise
        p = np.clip(p, 0.05, 0.95)
        inits.append(p)
        
    # Alternate between center optimization and exact LP radius calculation
    for init_c in inits:
        centers = init_c.copy()
        radii = np.full(n, 0.08)
        
        for _ in range(3):
            x0 = np.concatenate([centers.flatten(), radii])
            bounds_list = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]
            
            res = minimize(objective, x0, args=(n, 5000), method='L-BFGS-B',
                           bounds=bounds_list,
                           options={'maxiter': 3000, 'ftol': 1e-12})
            centers = res.x[:2*n].reshape(n, 2)
            radii = solve_radii_lp(centers)
            
        # Final exact radius calculation
        radii = solve_radii_lp(centers)
        s = np.sum(radii)
        
        if s > best_sum:
            best_sum = s
            best_res = (centers, radii)
            
    centers, radii = best_res
    
    # Apply a tiny safety margin to guarantee strict validity against checker tolerances
    radii *= 0.999995
    
    return centers, radii, float(np.sum(radii))
