import numpy as np
from scipy.optimize import minimize

def compute_objective(vars_arr, n):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(vars_arr[2::3])

def compute_constraints(vars_arr, n):
    """Constraint function: returns array of values >= 0 for valid packing"""
    cx = vars_arr[0::3]
    cy = vars_arr[1::3]
    r = vars_arr[2::3]
    
    con = []
    # Boundary constraints: circle inside [0,1]x[0,1]
    for i in range(n):
        con.append(cx[i] - r[i])          # x - r >= 0
        con.append(1.0 - cx[i] - r[i])    # 1 - x - r >= 0
        con.append(cy[i] - r[i])          # y - r >= 0
        con.append(1.0 - cy[i] - r[i])    # 1 - y - r >= 0
        
    # Pairwise non-overlap constraints (squared distance)
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            dist_sq = dx*dx + dy*dy
            r_sum = r[i] + r[j]
            con.append(dist_sq - r_sum * r_sum)
            
    return np.array(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    best_sum = -1.0
    best_vars = None
    
    # Generate diverse initial configurations to escape local minima
    inits = []
    
    # 1. Hexagonal lattice approximation
    pts_hex = []
    r_h = 0.101
    row = 0
    y = r_h
    while len(pts_hex) < n and y + r_h <= 1.0:
        shift = r_h if row % 2 == 1 else 0.0
        x = r_h + shift
        while len(pts_hex) < n and x + r_h <= 1.0:
            pts_hex.append([x, y])
            x += 2.0 * r_h
        y += np.sqrt(3) * r_h
        row += 1
    while len(pts_hex) < n:
        pts_hex.append([0.5, 0.5])
    inits.append(np.array(pts_hex[:n]))
    
    # 2. Grid 5x5 + 1 in a gap
    pts_grid = []
    for i in range(5):
        for j in range(5):
            pts_grid.append([0.1 + i*0.2, 0.1 + j*0.2])
    pts_grid.append([0.3, 0.3])
    inits.append(np.array(pts_grid))
    
    # 3. 5-6-5-6-4 staggered hex pattern (known dense layout for N=26)
    pts_stag = []
    counts = [5, 6, 5, 6, 4]
    y_pos = 0.101
    row_h = 0.101 * np.sqrt(3)
    for idx, cnt in enumerate(counts):
        shift = 0.101 if idx % 2 == 1 else 0.0
        x_start = shift + 0.101
        x_end = 1.0 - 0.101
        x_vals = np.linspace(x_start, x_end, cnt) if cnt > 1 else [0.5]
        for xv in x_vals:
            pts_stag.append([xv, y_pos])
        y_pos += row_h
        if len(pts_stag) >= n: break
    inits.append(np.array(pts_stag[:n]))
    
    # 4. Perturbed hexagonal
    np.random.seed(42)
    inits.append(np.clip(inits[0] + np.random.normal(0, 0.015, size=inits[0].shape), 0.05, 0.95))
    
    # 5. Random feasible spread
    pts_rand = np.random.uniform(0.15, 0.85, (n, 2))
    inits.append(pts_rand)
    
    # 6. Perturbed grid
    inits.append(np.clip(inits[1] + np.random.normal(0, 0.02, size=inits[1].shape), 0.05, 0.95))

    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    for cfg in inits:
        x0 = np.empty(3 * n)
        x0[0::3] = cfg[:, 0]
        x0[1::3] = cfg[:, 1]
        x0[2::3] = 0.05  # Safe initial radius to ensure feasible start
        
        try:
            res = minimize(
                compute_objective,
                x0,
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': compute_constraints, 'args': (n,)},
                options={'maxiter': 5000, 'ftol': 1e-13}
            )
            
            if res.success:
                # Strict verification of constraints
                c_vals = compute_constraints(res.x, n)
                if np.min(c_vals) >= -1e-8:
                    s = np.sum(res.x[2::3])
                    if s > best_sum:
                        best_sum = s
                        best_vars = res.x
        except Exception:
            pass
            
    if best_vars is not None:
        centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
        radii = best_vars[2::3]
        return centers, radii, float(np.sum(radii))
        
    # Fallback to a known valid configuration
    centers = np.array([(0.1 + i*0.2, 0.1 + j*0.2) for j in range(5) for i in range(5)] + [[0.3, 0.3]])
    radii = np.full(26, 0.09)
    return centers, radii, float(np.sum(radii))
