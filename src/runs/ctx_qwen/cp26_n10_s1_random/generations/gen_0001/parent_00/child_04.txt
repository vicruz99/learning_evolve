import numpy as np
from scipy.optimize import minimize, linprog

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def objective_equal(vars, n):
    """Objective function: minimize negative radius to maximize it."""
    return -vars[2 * n]

def constraints_equal(vars, n):
    """Constraint function: returns array of values >= 0 for valid packing."""
    centers = vars[:2 * n].reshape(n, 2)
    r = vars[2 * n]
    cons = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        cons.append(x - r)          # Left
        cons.append(1.0 - x - r)    # Right
        cons.append(y - r)          # Bottom
        cons.append(1.0 - y - r)    # Top
        
    # Pairwise non-overlap constraints: dist^2 >= (2r)^2
    for i in range(n):
        for j in range(i + 1, n):
            d2 = np.sum((centers[i] - centers[j]) ** 2)
            cons.append(d2 - (2.0 * r) ** 2)
            
    return np.array(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_r = 0.0
    best_centers = None
    
    # Define various hexagonal row configurations to explore layout space
    row_configs = [
        [5, 5, 5, 5, 6],
        [6, 5, 5, 5, 5],
        [5, 6, 5, 5, 5],
        [5, 5, 6, 5, 5],
        [5, 5, 5, 6, 5],
        [5, 5, 5, 5, 5, 1]
    ]
    
    configs = []
    for rc in row_configs:
        pts = []
        r_approx = 0.1
        y = r_approx
        h = r_approx * np.sqrt(3)
        for row_idx, count in enumerate(rc):
            shift = r_approx if row_idx % 2 == 1 else 0.0
            x_start = r_approx + shift
            for col in range(count):
                x = x_start + col * 2.0 * r_approx
                pts.append([x, y])
            y += h
            if len(pts) >= n:
                break
        if len(pts) < n:
            for _ in range(n - len(pts)):
                pts.append([0.5, 0.5])
        configs.append(np.array(pts[:n]))
        
    # Add perturbed square grids for diversity
    np.random.seed(42)
    for _ in range(5):
        grid = np.array([[i*0.2+0.1, j*0.2+0.1] for j in range(5) for i in range(5)] + [[0.5, 0.5]])
        noise = np.random.uniform(-0.03, 0.03, grid.shape)
        configs.append(np.clip(grid + noise, 0.05, 0.95))

    # Optimize each configuration
    for cfg in configs:
        cfg = cfg.copy()
        # Normalize to fit tightly in [0,1] with margin
        cfg -= cfg.min(axis=0)
        cfg /= cfg.max(axis=0)
        cfg *= 0.9
        cfg += 0.05
        
        # Compute safe initial radius
        min_dist = 1.0
        for i in range(n):
            min_dist = min(min_dist, cfg[i,0], 1-cfg[i,0], cfg[i,1], 1-cfg[i,1])
            for j in range(i+1, n):
                d = np.linalg.norm(cfg[i] - cfg[j])
                if d < min_dist: min_dist = d
        r_init = min_dist * 0.45
        
        x0 = np.concatenate([cfg.flatten(), [r_init]])
        bounds = [(0.0, 1.0)]*(2*n) + [(0.0, 0.5)]
        
        try:
            res = minimize(objective_equal, x0, args=(n,), method='SLSQP', 
                           bounds=bounds, 
                           constraints={'type': 'ineq', 'fun': constraints_equal, 'args': (n,)},

            A_ub[idx, j] = 1.0

            idx += 1

    try:
        res_lp = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res_lp.success and np.sum(res_lp.x) > np.sum(radii_final) * 0.99:
            radii_final = res_lp.x
    except Exception:
        pass

    return best_centers, radii_final, float(np.sum(radii_final))
