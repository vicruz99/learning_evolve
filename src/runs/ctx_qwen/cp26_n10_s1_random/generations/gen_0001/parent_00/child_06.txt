import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def objective(x):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(x[2::3])

def get_constraints(x):
    """
    Computes inequality constraints >= 0 for valid packing.
    Constraints:
    1. Boundary: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    2. Non-overlap: (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    """
    n = 26
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    b1 = cx - r
    b2 = 1.0 - cx - r
    b3 = cy - r
    b4 = 1.0 - cy - r
    
    # Pairwise non-overlap constraints (vectorized)
    cx_mat = cx[:, None] - cx[None, :]
    cy_mat = cy[:, None] - cy[None, :]
    dr_mat = r[:, None] + r[None, :]
    
    dist_sq = cx_mat**2 + cy_mat**2
    dr_sq = dr_mat**2
    
    # Upper triangular mask to avoid duplicates and self-interaction
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlap_cons = (dist_sq - dr_sq)[mask]
    
    return np.concatenate([b1, b2, b3, b4, overlap_cons])

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds for [x0, y0, r0, x1, y1, r1, ...]
    bounds = [(0, 1)] * (2 * n) + [(1e-6, 0.5)] * n
    
    # Generate diverse initial configurations
    configs = []
    
    # 1. Hexagonal lattice initialization
    pts = []
    r_est = 0.1
    for row in range(6):
        y = r_est + row * np.sqrt(3) * r_est
        shift = r_est if row % 2 == 1 else 0.0
        x = r_est + shift
        while x + r_est <= 1.0 and len(pts) < 26:
            pts.append([x, y])
            x += 2 * r_est
    while len(pts) < 26:
        pts.append([0.5, 0.5])
    configs.append(np.array(pts[:26]))
    
    # 2. Regular grid initialization (5x5 + 1)
    pts2 = []
    step = 0.2
    for i in range(5):
        for j in range(5):
            pts2.append([step * (i + 0.5), step * (j + 0.5)])
    pts2.append([0.15, 0.15])
    configs.append(np.array(pts2[:26]))
    
    # 3-5. Random perturbations of hexagonal grid
    np.random.seed(42)
    for _ in range(3):
        cfg = configs[0].copy()
        cfg += np.random.uniform(-0.04, 0.04, cfg.shape)
        cfg = np.clip(cfg, 0.1, 0.9)
        configs.append(cfg)
        
    # Optimize each configuration
    for cfg in configs:
        x0 = np.zeros(3 * n)
        x0[0::3] = cfg[:, 0]
        x0[1::3] = cfg[:, 1]
        x0[2::3] = 0.09  # Start with a feasible small radius
        
        try:
            res = minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': get_constraints},
                options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
            )
            
            if res.success:
                c = np.column_stack((res.x[0::3], res.x[1::3]))
                r = res.x[2::3]
                
                # Strict validation check
                if validate_packing(c, r):
                    s = np.sum(r)
                    if s > best_sum:
                        best_sum = s
                        best_centers = c.copy()
                        best_radii = r.copy()
        except Exception:
            continue
            
    # Fallback configuration if optimization fails unexpectedly
    if best_centers is None:
        grid_r = 0.09
        best_centers = np.array([(grid_r + i * 2 * grid_r, grid_r + j * 2 * grid_r) 
                                 for j in range(5) for i in range(5)] + [[0.55, 0.55]])
        best_radii = np.full(26, grid_r)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
