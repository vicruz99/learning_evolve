import numpy as np
from scipy.optimize import minimize, linprog

def penalty_func(v, n, mu):
    """Computes objective + penalty for constraint violations."""
    c_x = v[0::3]
    c_y = v[1::3]
    r = v[2::3]
    
    obj = -np.sum(r)
    p = 0.0
    
    # Boundary penalties: x-r>=0, 1-x-r>=0, y-r>=0, 1-y-r>=0
    p += np.sum(np.maximum(0.0, r - c_x)**2)
    p += np.sum(np.maximum(0.0, r - (1.0 - c_x))**2)
    p += np.sum(np.maximum(0.0, r - c_y)**2)
    p += np.sum(np.maximum(0.0, r - (1.0 - c_y))**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    diff_x = c_x[:, None] - c_x[None, :]
    diff_y = c_y[:, None] - c_y[None, :]
    dist = np.sqrt(diff_x**2 + diff_y**2)
    np.fill_diagonal(dist, np.inf)
    
    r_sum = r[:, None] + r[None, :]
    triu_idx = np.triu_indices(n, k=1)
    overlaps = np.maximum(0.0, r_sum[triu_idx] - dist[triu_idx])
    p += np.sum(overlaps**2)
    
    return obj + mu * p

def solve_radii_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    limits = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    c = -np.ones(n)
    bounds = [(0.0, lim) for lim in limits]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    n_cons = n * (n - 1) // 2
    A_ub = np.zeros((n_cons, n))
    b_ub = np.zeros(n_cons)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    # Fallback: safe heuristic
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    safe_r = np.minimum(limits, min_dists / 2.0)
    return np.maximum(safe_r, 0.0)

            best_result = (final_c, final_r)
