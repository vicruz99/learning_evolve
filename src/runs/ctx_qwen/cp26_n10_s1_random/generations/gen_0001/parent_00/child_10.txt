import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """
    Solves an LP to maximize sum(r_i) for fixed center positions.
    Constraints: r_i >= 0, r_i <= dist_to_wall, r_i + r_j <= dist(i,j)
    """
    n = centers.shape[0]
    c_obj = np.ones(n) * -1.0  # Minimize negative sum => maximize sum
    bounds = [(0, None)] * n
    
    x = centers[:, 0]
    y = centers[:, 1]
    walls = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    A_rows = [np.eye(n)]
    b_vals = [walls]
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_rows.append(row)
            b_vals.append(dists[i, j])
            
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    # Try fast method first, fallback to others
    for method in ['highs', 'interior-point', 'simplex']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success:
                return res.x
        except Exception:
            continue
            
    # Fallback heuristic if LP fails
    min_dists = np.min(dists, axis=1)
    return np.maximum(np.minimum(walls, min_dists / 2.0), 0.0)

def packing_constraint(vars, n):
    """Computes constraint values >= 0 for valid packing."""
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]
    
    x, y = centers[:, 0], centers[:, 1]
    cons = []
    
    # Boundary constraints
    cons.append(x - radii)
    cons.append(1.0 - x - radii)
    cons.append(y - radii)
    cons.append(1.0 - y - radii)
    
    # Overlap constraints: dist^2 - (r1+r2)^2 >= 0
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum_sq = (radii[:, np.newaxis] + radii[np.newaxis, :])**2
    tri = np.tril_indices(n, -1)
    cons.append(dist_sq[tri] - r_sum_sq[tri])
    
    return np.concatenate(cons)

def constraint_wrapper(v):
    return packing_constraint(v, 26)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    configs = []

            x = x_start
            while x + r_base <= 1.0 and len(pts) < n:
                pts.append([x, y])
                x += 2.0 * r_base
            y += np.sqrt(3) * r_base
            row += 1
        configs.append(np.array(pts[:n]))
        
    # 2. Perturbed hex grid and random valid starts
    for _ in range(4):
        base = configs[0].copy()
        base += np.random.uniform(-0.03, 0.03, base.shape)
        base = np.clip(base, 0.05, 0.95)
        configs.append(base)

    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    for cfg in configs:
        # Estimate feasible initial radii
        dists = np.sqrt(((cfg[:, np.newaxis] - cfg[np.newaxis, :])**2).sum(axis=2))
        np.fill_diagonal(dists, np.inf)

        r0 = np.minimum(min_d / 2.0, walls)
        r0 = np.maximum(r0, 0.02)
        
        x0 = np.concatenate([cfg.flatten(), r0])

            res = minimize(
                lambda v: -np.sum(v[2*n:]),
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': constraint_wrapper},
                options={'maxiter': 2000, 'ftol': 1e-12}

            lp_r = solve_lp_radii(c_opt)

                best_centers = c_opt
