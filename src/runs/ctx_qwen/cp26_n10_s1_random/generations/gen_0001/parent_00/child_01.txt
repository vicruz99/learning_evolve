import numpy as np
from scipy.optimize import minimize, linprog

def compute_smooth_clearance(centers, k=60.0):
    """
    Computes a smooth approximation of the minimum clearance.
    Clearance is the minimum distance to boundaries and half pairwise distances.
    """
    n = centers.shape[0]
    
    # Distances to four boundaries
    b_min = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Pairwise distances / 2
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    p_dists = dists[np.triu_indices(n, k=1)] / 2.0
    
    all_clearances = np.concatenate([b_min, p_dists])
    
    # Numerically stable smooth min: min(x) ≈ shift + log(sum(exp(k*(x-shift))))/k
    shift = np.min(all_clearances)
    # Clip to prevent overflow in exp, though values should be reasonable
    vals = np.clip(all_clearances - shift, -10.0, 10.0)
    sm = shift + np.log(np.sum(np.exp(k * vals))) / k
    return sm

def get_objective(centers_flat, k):
    """Objective function for scipy minimize: maximizes smooth clearance."""
    n = len(centers_flat) // 2
    centers = centers_flat.reshape(n, 2)
    return -compute_smooth_clearance(centers, k)

def solve_radii_lp(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Max sum(r_i) s.t. r_i + r_j <= dist(i,j) and 0 <= r_i <= boundary_dist(i)
    """
    n = centers.shape[0]
    
    # Boundary limits
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 0.0)
    
    # Pairwise constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    upper_idx, lower_idx = np.triu_indices(n, k=1)
    pairwise_dists = dists[upper_idx, lower_idx]
    
    # LP setup: minimize -sum(r)
    c = np.ones(n) * -1.0
    A_ub = np.zeros((len(pairwise_dists), n))
    A_ub[np.arange(len(pairwise_dists)), upper_idx] = 1.0
    A_ub[np.arange(len(pairwise_dists)), lower_idx] = 1.0
    b_ub = pairwise_dists
    
    bounds = [(0.0, lim) for lim in lims]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    
    # Fallback heuristic if LP fails
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    return np.minimum(lims, min_dists / 2.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_centers = None
    best_clearance = -np.inf
    
    # 1. Initialization: Hexagonal lattice pattern
    r_est = 0.101
    pts = []
    row = 0
    while len(pts) < n + 10:
        y = r_est + row * r_est * np.sqrt(3.0)
        shift = r_est if row % 2 == 1 else 0.0
        x = r_est + shift
        while x + r_est <= 1.0 and len(pts) < n + 10:
            pts.append([x, y])
            x += 2.0 * r_est
        row += 1
        if y + r_est > 1.0:
            break
            
    pts = np.array(pts[:n])
    
    # 2. Optimization with multiple restarts
    rng = np.random.default_rng(42)
    for trial in range(8):
        # Perturb initial points to explore landscape
        current_pts = pts.copy() + rng.uniform(-0.025, 0.025, size=pts.shape)
        current_pts = np.clip(current_pts, 0.05, 0.95)
        
        res = minimize(get_objective, current_pts.flatten(), 
                       args=(60.0,),
                       method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * (2 * n),
                       options={'maxiter': 3000, 'ftol': 1e-13})
        
        curr_clear = -res.fun
        if curr_clear > best_clearance:
            best_clearance = curr_clear
            best_centers = res.x.reshape(n, 2)
            
    # 3. Compute exact maximal radii for the best centers
    final_radii = solve_radii_lp(best_centers)
    final_radii = np.maximum(final_radii, 1e-12) # Ensure strictly positive
    
    # Ensure centers are strictly within bounds for safety
    final_centers = np.clip(best_centers, 0.0, 1.0)
    
    total_sum = np.sum(final_radii)
    return final_centers, final_radii, total_sum
