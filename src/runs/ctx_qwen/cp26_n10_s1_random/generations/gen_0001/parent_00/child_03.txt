import numpy as np
from scipy.optimize import minimize, linprog

def compute_penalty(centers, n, r_target):
    """Computes the squared penalty for boundary and overlap violations."""
    pen = 0.0
    for i in range(n):
        x, y = centers[i]
        for d in (x, 1.0 - x, y, 1.0 - y):
            if d < r_target:
                pen += (r_target - d) ** 2
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < 2.0 * r_target:
                pen += (2.0 * r_target - dist) ** 2
    return pen

def compute_grad(centers, n, r_target):
    """Computes the gradient of the penalty function w.r.t. centers."""
    grad = np.zeros_like(centers)
    for i in range(n):
        x, y = centers[i]
        if x < r_target:
            grad[i, 0] -= 2.0 * (r_target - x)
        if 1.0 - x < r_target:
            grad[i, 0] += 2.0 * (r_target - (1.0 - x))
        if y < r_target:
            grad[i, 1] -= 2.0 * (r_target - y)
        if 1.0 - y < r_target:
            grad[i, 1] += 2.0 * (r_target - (1.0 - y))
            
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff)
            if dist < 2.0 * r_target and dist > 1e-9:
                factor = 2.0 * (2.0 * r_target - dist) / dist
                grad[i] += factor * diff
                grad[j] -= factor * diff
    return grad.flatten()

def solve_lp_for_radii(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = centers[i]
        limits = [x, 1.0 - x, y, 1.0 - y]
        for lim in limits:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(lim)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c = np.ones(n) * -1.0
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_centers = None
    best_lp_sum = -1.0
    
    # Initialize with a dense hexagonal lattice
    r_init = 0.101
    init_pts = []
    for row in range(7):
        y = r_init + row * r_init * np.sqrt(3)
        if y + r_init > 1.05:
            break
        shift = r_init if row % 2 == 1 else 0.0
        x = r_init + shift
        while x + r_init <= 1.05:
            init_pts.append([x, y])
            x += 2.0 * r_init
            
    # Ensure we have exactly n points
    while len(init_pts) < n:
        init_pts.append([np.random.rand(), np.random.rand()])
    init_pts = np.array(init_pts[:n])
    
    np.random.seed(42)
    r_target = 0.1014  # Slightly above known optimal equal radius to push boundaries
    
    for trial in range(10):
        # Perturb initial configuration
        centers0 = init_pts.copy() + np.random.normal(0, 0.008, init_pts.shape)
        centers0 = np.clip(centers0, 0.01, 0.99)
        
        # Optimize positions to minimize overlap/boundary penalties
        res = minimize(
            compute_penalty,
            centers0.flatten(),
            jac=compute_grad,
            args=(n, r_target),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * (2 * n),

        radii_opt = solve_lp_for_radii(centers_opt, n)
        current_sum = np.sum(radii_opt)

            best_lp_sum = current_sum
            best_centers = centers_opt.copy()
