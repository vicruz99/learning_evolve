import numpy as np
from scipy.optimize import minimize, linprog

def compute_objective(vars_array, n):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(vars_array[2::3])

def compute_constraints(vars_array, n):
    """Constraint function: returns array of values >= 0 for valid packing"""
    xs = vars_array[0::3]
    ys = vars_array[1::3]
    rs = vars_array[2::3]
    cons = []
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    for i in range(n):
        cons.append(xs[i] - rs[i])          # Left: x - r >= 0
        cons.append(1.0 - xs[i] - rs[i])    # Right: 1 - x - r >= 0
        cons.append(ys[i] - rs[i])          # Bottom: y - r >= 0
        cons.append(1.0 - ys[i] - rs[i])    # Top: 1 - y - r >= 0
        
    # Pairwise non-overlap constraints: dist(i,j) >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dist = np.sqrt(dx*dx + dy*dy)
            cons.append(dist - rs[i] - rs[j])
            
    return np.array(cons)

def solve_radii_lp(centers, n):
    """
    Solves the Linear Programming problem to find the maximum sum of radii
    for a fixed set of centers.
    """
    # Distance to nearest boundary for each circle
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 0.0)
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c = -np.ones(n)
    bounds = [(0.0, lim) for lim in limits]
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Try modern 'highs' method, fallback to 'simplex'
    for method in ['highs', 'simplex']:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success:
                return res.x
        except ValueError:
            continue
            
    # Fallback if LP fails
    return np.maximum(np.full(n, 0.0), 0.0)

def neg_lp_sum(centers_flat, n):
    """Objective for Nelder-Mead: negative of the maximum LP sum of radii"""
    centers = centers_flat.reshape(n, 2)
    radii = solve_radii_lp(centers, n)
    return -np.sum(radii)

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    
    # --- 1. Initialization: Hexagonal Lattice ---
    centers_hex = []
    r_init = 0.09
    for row in range(6):
        y = r_init + row * r_init * np.sqrt(3)
        shift = r_init if row % 2 == 1 else 0.0
        x = r_init + shift
        while x + r_init <= 1.0:
            centers_hex.append([x, y])
            x += 2 * r_init
        if len(centers_hex) >= 26:
            break
            
    centers_hex = np.array(centers_hex[:26])
    # Normalize to center nicely in the unit square
    min_c = centers_hex.min(axis=0)
    max_c = centers_hex.max(axis=0)
    centers_hex = (centers_hex - min_c) / (max_c - min_c) * 0.86 + 0.07
    
    # Create multiple starting points
    inits = [centers_hex]
    np.random.seed(42)
    for _ in range(4):
        pert = centers_hex + np.random.normal(0, 0.02, centers_hex.shape)
        pert = np.clip(pert, 0.05, 0.95)
        inits.append(pert)
        
    bounds_list = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    # --- 2. SLSQP Joint Optimization ---
    for c_init in inits:
        x0 = np.concatenate([c_init.flatten(), np.full(n, 0.08)])
        try:
            res = minimize(compute_objective, x0, args=(n,), method='SLSQP',
                           bounds=bounds_list,
                           constraints={'type': 'ineq', 'fun': compute_constraints, 'args': (n,)},
                           options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                s = np.sum(res.x[2::3])
                if s > best_sum:
                    best_sum = s
                    best_centers = res.x[:2 * n].reshape(n, 2)
        except Exception:
            continue
            
    if best_centers is None:
        best_centers = inits[0]
        
    # --- 3. Nelder-Mead Center Refinement via LP Objective ---
    # This step optimizes centers to maximize the exact LP-computed radii sum,
    # bypassing non-convex constraint gradients and finding tighter packings.
    best_centers = np.clip(best_centers, 0.01, 0.99)
    try:
        res_nm = minimize(neg_lp_sum, best_centers.flatten(), args=(n,), method='Nelder-Mead', 
                          options={'maxiter': 3000, 'xatol': 1e-7, 'fatol': 1e-9})
        best_centers = res_nm.x.reshape(n, 2)
    except Exception:
        pass
        
    # --- 4. Final LP Radius Assignment ---
    final_radii = solve_radii_lp(best_centers, n)
    final_radii = np.maximum(final_radii, 0.0)
    final_sum = float(np.sum(final_radii))
    
    return best_centers, final_radii, final_sum
