import numpy as np
from scipy.optimize import minimize

def compute_objective(params, n, mu):
    """
    Computes the penalized objective: -sum(r) + mu * penalty.
    params: 1D array of length 3*n [x0, y0, r0, x1, y1, r1, ...]
    """
    cx = params[0::3]
    cy = params[1::3]
    r = params[2::3]
    
    # Objective: maximize sum of radii => minimize negative sum
    obj = -np.sum(r)
    
    # Penalty accumulation
    penalty = 0.0
    
    # Boundary constraints: r <= coord <= 1-r
    # Violations: r - coord > 0 or r - (1-coord) > 0
    for coord in [cx, cy]:
        viol = np.maximum(0.0, r - coord)
        penalty += np.sum(viol**2)
        viol = np.maximum(0.0, r - (1.0 - coord))
        penalty += np.sum(viol**2)
        
    # Pairwise non-overlap: dist(i, j) >= r_i + r_j
    # Violations: r_i + r_j - dist > 0
    # Vectorized distance computation
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf) # Ignore self-distance
    
    r_sum = r[:, None] + r[None, :]
    viol_pair = np.maximum(0.0, r_sum - dist)
    
    # Sum over all pairs, divide by 2 to avoid double counting
    penalty += 0.5 * np.sum(viol_pair**2)
    
    return obj + mu * penalty

def run_packing() -> tuple:
    n = 26
    mu = 8000.0 # Penalty weight for constraint violations
    
    # Bounds for [x, y, r] for each circle
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-5, 0.5)] * n
    
    # Generate high-quality initial configurations
    configs = []
    
    # 1. Hexagonal lattice pattern (optimal for dense packing)
    r_init = 0.085
    points_hex = []
    row_y = r_init
    row_idx = 0
    while row_y + r_init <= 1.0 and len(points_hex) < n + 10:
        shift = r_init if row_idx % 2 == 1 else 0.0
        x = r_init + shift
        while x + r_init <= 1.0:
            points_hex.append([x, row_y])
            x += 2.0 * r_init
        row_y += r_init * np.sqrt(3)
        row_idx += 1
        
    # Take exactly n points, normalize to fit well within [0,1] with margin
    pts = np.array(points_hex[:n])
    pts = (pts - pts.min(axis=0)) / (pts.max(axis=0) - pts.min(axis=0)) * 0.85 + 0.075
    configs.append(pts)

    grid_full = np.vstack([grid_pts, [[0.5, 0.5]]])
    configs.append(grid_full)
    
    # 3. Randomized perturbations to escape symmetric local minima
    rng = np.random.default_rng(42)
    for _ in range(8):
        noise = rng.uniform(-0.04, 0.04, size=(n, 2))
        cfg = np.clip(pts + noise, 0.05, 0.95)
        configs.append(cfg)
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Optimization loop over all configurations
    for centers_init in configs:
        x0 = np.empty(3 * n)
        x0[0::3] = centers_init[:, 0]
        x0[1::3] = centers_init[:, 1]
        x0[2::3] = 0.08 # Start with small valid radius to ensure feasibility
        
        try:
            res = minimize(
                compute_objective,
                x0,
                args=(n, mu),
                method='L-BFGS-B',
