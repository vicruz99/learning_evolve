import numpy as np
from scipy.optimize import minimize

def compute_objective(params, n):
    """Objective function for L-BFGS-B: minimizes negative sum of radii + penalty for violations."""
    c = params.reshape(n, 3)
    cx, cy, r = c[:, 0], c[:, 1], c[:, 2]
    
    # Primary objective: maximize sum of radii
    obj = -np.sum(r)
    penalty = 0.0
    
    # Boundary penalties: circle must be within [0,1]x[0,1]
    wall_viol = np.maximum(0, r - cx) + np.maximum(0, r - (1.0 - cx)) + \
                np.maximum(0, r - cy) + np.maximum(0, r - (1.0 - cy))
    penalty += 1e5 * np.sum(wall_viol**2)
    
    # Overlap penalties: circles must not overlap
    diff = c[:, np.newaxis, :2] - c[np.newaxis, :, :2]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    overlap = np.maximum(0, r_sum - dists)
    # Sum only upper triangle to avoid double counting
    penalty += 1e5 * np.sum(np.triu(overlap**2, 1))
    
    return obj + penalty

def get_max_equal_radius(centers):
    """Computes the largest equal radius that fits the given center configuration."""
    n = centers.shape[0]
    min_val = 1.0
    
    # Distance to boundaries
    min_val = min(min_val, np.min(centers[:, 0]))
    min_val = min(min_val, np.min(1.0 - centers[:, 0]))
    min_val = min(min_val, np.min(centers[:, 1]))
    min_val = min(min_val, np.min(1.0 - centers[:, 1]))
    
    # Half of minimum pairwise distance
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pair_dist = np.min(dists)
    min_val = min(min_val, min_pair_dist / 2.0)
    
    return min_val

def run_packing() -> tuple:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Bounds: x, y in [0, 1], r in [1e-4, 0.5]
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(1e-4, 0.5)] * n

    # Multi-restart optimization
    for seed in range(15):
        rng = np.random.default_rng(seed)
        
        # 1. Hexagonal lattice initialization
        pts = []
        r_init = 0.10
        dy = r_init * np.sqrt(3)
        y = r_init
        row = 0
        while y < 1.0 and len(pts) < 30:
            x_start = r_init if row % 2 == 0 else 2 * r_init
            x = x_start
            while x < 1.0 and len(pts) < 30:
                pts.append([x, y])
                x += 2 * r_init
            y += dy
            row += 1
            
        pts = np.array(pts[:n])
        
        # 2. Random perturbation to break symmetry
        pts += rng.uniform(-0.03, 0.03, size=pts.shape)
        pts = np.clip(pts, 0.05, 0.95)
        
        # 3. Form initial parameters [x0, y0, r0, x1, y1, r1, ...]
        params = np.zeros(3 * n)
        params[0::3] = pts[:, 0]
        params[1::3] = pts[:, 1]
        params[2::3] = 0.095  # Start slightly below expected optimum to ensure feasibility
        
        try:
            # 4. Optimize centers using L-BFGS-B
            res = minimize(compute_objective, params, args=(n,), method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 8000, 'ftol': 1e-13})
            
            c_opt = res.x.reshape(n, 3)
            centers_opt = c_opt[:, :2]
            
            # 5. Extract exact max equal radius for the optimized geometry
            r_eq = get_max_equal_radius(centers_opt)
            if r_eq > 0:
                sum_r = 26 * r_eq
                if sum_r > best_sum:
                    best_sum = sum_r
                    best_centers = centers_opt
                    # Shrink slightly to guarantee strict validation compliance
                    best_radii = np.full(n, r_eq * 0.999999)
        except Exception:
            pass

    # Fallback configuration if optimization fails unexpectedly
    if best_centers is None or best_sum < 2.0:
        cx_fb = np.linspace(0.0833, 0.9167, 6)
        cy_fb = np.linspace(0.0833, 0.9167, 5)
        fb_centers = []
        for i, y in enumerate(cy_fb):
            for j, x in enumerate(cx_fb):
                if len(fb_centers) >= n: break
                if i % 2 == 1:
                    x += 0.0416
                fb_centers.append([x, y])
            if len(fb_centers) >= n: break
        best_centers = np.array(fb_centers[:n])
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
