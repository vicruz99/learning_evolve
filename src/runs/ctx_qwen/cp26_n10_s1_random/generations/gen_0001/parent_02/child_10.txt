If we define $t$ as this minimum clearance, then each circle can have radius $r = t/2$, and the sum of radii is $N \times t/2$.

import numpy as np
from scipy.optimize import minimize

_N = 26
_TRIU_INDICES = np.triu_indices(_N, k=1)

def objective(vars_array):
    """Objective: maximize t (last element of vars_array)"""
    return -vars_array[-1]

def constraints(vars_array):
    """Constraint function: returns array of values >= 0 for valid packing"""
    centers = vars_array[:2*_N].reshape(_N, 2)
    t = vars_array[-1]
    
    # Boundary constraints: x >= t/2, 1-x >= t/2, y >= t/2, 1-y >= t/2
    xb = np.concatenate([centers[:, 0] - t/2, 1.0 - centers[:, 0] - t/2])
    yb = np.concatenate([centers[:, 1] - t/2, 1.0 - centers[:, 1] - t/2])
    
    # Pairwise non-overlap constraints: dist(i,j)^2 >= t^2
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    p_cons = dist_sq[_TRIU_INDICES] - t**2
    
    return np.concatenate([xb, yb, p_cons])

def generate_hex_init(scale=1.0):
    """Generates an initial hexagonal lattice configuration"""
    pts = []
    r = 0.09
    dy = np.sqrt(3) * r
    y = r
    row = 0
    while y < 1.0:
        x = r
        shift = r if row % 2 == 1 else 0.0
        while x < 1.0:
            pts.append([x + shift, y])
            x += 2 * r
        y += dy
        row += 1
    pts = np.array(pts[:_N])
    cx, cy = pts.mean(axis=0)
    pts = (pts - [cx, cy]) * scale + [0.5, 0.5]
    return np.clip(pts, 0.05, 0.95)

def run_packing():
    bounds = [(0.0, 1.0)] * (2*_N) + [(0.1, 0.22)]
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_t = 0.0
    best_x = None
    
    np.random.seed(42)
    for trial in range(15):
        if trial == 0:
            c_init = generate_hex_init(1.1)
        elif trial == 1:
            cx = np.linspace(0.15, 0.85, 5)
            cy = np.linspace(0.15, 0.85, 5)
            grid = np.array([[x, y] for y in cy for x in cx]).flatten()[:_N]
            c_init = grid.reshape(_N, 2) + np.random.uniform(-0.03, 0.03, (_N, 2))
            c_init = np.clip(c_init, 0.1, 0.9)
        else:
            c_init = np.random.uniform(0.2, 0.8, (_N, 2))
            
        x0 = np.concatenate([c_init.flatten(), [0.195]])
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 1500, 'ftol': 1e-12})
            if -res.fun > best_t:
                best_t = -res.fun
                best_x = res.x.copy()
        except Exception:
            pass
            
    # Fallback to a valid grid if optimization fails entirely
    if best_x is None:
        cx = np.linspace(0.1, 0.9, 5)
        cy = np.linspace(0.1, 0.9, 5)
        pts = np.array([[x, y] for y in cy for x in cx]).flatten()[:_N].reshape(_N, 2)
        best_t = 0.18
        best_x = np.concatenate([pts.flatten(), [best_t]])
        
    centers = best_x[:2*_N].reshape(_N, 2)
    radii = np.full(_N, best_t / 2.0)
    return centers, radii, best_t * _N / 2.0
