import numpy as np

def simulate_packing(centers_init, n, seed=None):
    """Runs force-directed expansion to densely pack circles."""
    if seed is not None:
        np.random.seed(seed)
        
    centers = centers_init.copy()
    r = 0.03
    v = np.zeros_like(centers)
    damping = 0.95
    
    # Force constants
    k_wall = 100.0
    k_rep = 40.0
    
    for step in range(20000):
        # Slowly increase target radius
        r *= 1.00002
        if step > 10000:
            r *= 1.000005
            
        # Adaptive timestep: larger early, smaller for fine-tuning
        dt = 0.00005 * (1.0 - step / 20000.0)
        if dt < 0.000005:
            dt = 0.000005
            
        f = np.zeros_like(centers)
        
        # Boundary forces
        for i in range(n):
            if centers[i, 0] < r:
                f[i, 0] += k_wall * (r - centers[i, 0])
            if centers[i, 0] > 1.0 - r:
                f[i, 0] -= k_wall * (centers[i, 0] - (1.0 - r))
            if centers[i, 1] < r:
                f[i, 1] += k_wall * (r - centers[i, 1])
            if centers[i, 1] > 1.0 - r:
                f[i, 1] -= k_wall * (centers[i, 1] - (1.0 - r))
                
        # Pairwise repulsive forces
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                req = 2.0 * r
                
                if dist < req and dist > 1e-9:
                    overlap = req - dist
                    force_mag = k_rep * overlap
                    fx = (dx / dist) * force_mag
                    fy = (dy / dist) * force_mag
                    f[i, 0] += fx
                    f[i, 1] += fy
                    f[j, 0] -= fx
                    f[j, 1] -= fy
                    
        # Update velocities and positions
        v += f * dt
        v *= damping
        centers += v
        centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
        
    # Compute exact maximal radii for the final configuration
    radii = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
        d_min = d_wall
        for j in range(n):
            if i == j:
                continue
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            if d < d_min:
                d_min = d
        radii[i] = d_min / 2.0
        
    return centers, radii, np.sum(radii)
