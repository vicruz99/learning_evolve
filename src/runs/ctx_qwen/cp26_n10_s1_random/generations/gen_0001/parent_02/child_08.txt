import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple:
    n = 26
    
    def objective(vars):
        # vars structure: [x0, y0, ..., x25, y25, r]
        # We maximize r by minimizing -r
        return -vars[-1]
        
    def constraints(vars):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[-1]
        c = []
        # Boundary constraints: circle must be inside [0,1]x[0,1]
        c.extend(x - r)
        c.extend(1.0 - x - r)
        c.extend(y - r)
        c.extend(1.0 - y - r)
        
        # Pairwise non-overlap constraints: dist(i,j) >= 2r
        centers = np.column_stack((x, y))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                # Add small epsilon to avoid singularity in gradient computation
                dist = np.sqrt(dx*dx + dy*dy + 1e-14)
                c.append(dist - 2.0 * r)
        return np.array(c)

    # Bounds for variables: x,y in [0,1], r in [0.01, 0.25]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.01, 0.25)]
    
    best_r = 0.0
    best_centers = None
    
    # Generate multiple high-quality initial configurations (hexagonal lattices with shifts)
    configs = []
    for sx in [0.0, 0.05, 0.1, 0.15]:
        for sy in [0.0, 0.05, 0.1, 0.15]:
            pts = []
            r_init = 0.09
            dy = r_init * np.sqrt(3)
            y = r_init + sy
            row = 0
            while len(pts) < n:
                shift = r_init if row % 2 == 1 else 0.0
                x = r_init + shift + sx
                while x < 1.0 and len(pts) < n:
                    pts.append([x, y])
                    x += 2.0 * r_init
                y += dy
                row += 1
            # Ensure points are strictly inside the square for initialization
            configs.append(np.clip(np.array(pts[:n]), 0.05, 0.95))
            
    # Add a standard grid fallback
    grid_x = np.linspace(0.15, 0.85, 5)
    grid_y = np.linspace(0.15, 0.85, 5)
    grid_pts = []
    for y in grid_y:
        for x in grid_x:
            grid_pts.append([x, y])
    grid_pts.append([0.5, 0.5])
    configs.append(np.array(grid_pts[:n]))
    
    # Optimize from each configuration
    for cfg in configs:
        # Compute a strictly feasible initial radius for this configuration
        min_clear = 1.0
        min_clear = min(min_clear, np.min(cfg), np.min(1.0 - cfg))
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(cfg[i] - cfg[j])
                if d < min_clear:
                    min_clear = d
        r_init = 0.5 * min_clear * 0.95 # 5% margin for safety
        
        x0 = np.concatenate([cfg.flatten(), [r_init]])
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraints},
                           options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            
            if np.isfinite(res.fun):
                r_cand = res.x[-1]
                # Quick feasibility check
                cons_val = constraints(res.x)
                if np.min(cons_val) >= -1e-7:
                    if r_cand > best_r:
                        best_r = r_cand
                        best_centers = res.x[:2*n].reshape(n, 2)
        except Exception:
            continue
            
    # Fallback if optimization fails
    if best_centers is None:
        cx = np.linspace(0.1, 0.9, 6)
        cy = np.linspace(0.1, 0.9, 5)
        pts = []
        for i, y in enumerate(cy):
            for j, x in enumerate(cx):
                if i % 2 == 1: x += 0.05
                pts.append([x, y])
                if len(pts) == n: break
            if len(pts) == n: break
        best_centers = np.array(pts)
        best_r = 0.08
        
    # Apply a tiny safety margin to guarantee validation passes with 1e-12 tolerance
    radii = np.full(n, best_r * 0.9999999)
    sum_radii = float(np.sum(radii))
    
    return best_centers, radii, sum_radii
