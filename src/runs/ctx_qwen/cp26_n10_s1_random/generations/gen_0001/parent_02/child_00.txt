import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def compute_penalty(vars_arr, n, mu):
    """
    Computes -sum(radii) + mu * penalty for violations.
    Designed for L-BFGS-B optimization.
    """
    cx = vars_arr[0::3]
    cy = vars_arr[1::3]
    rad = vars_arr[2::3]
    
    obj = -np.sum(rad)
    pen = 0.0
    
    # Boundary penalties
    for i in range(n):
        if cx[i] < rad[i]:
            pen += (rad[i] - cx[i])**2
        if cx[i] > 1.0 - rad[i]:
            pen += (cx[i] - (1.0 - rad[i]))**2
        if cy[i] < rad[i]:
            pen += (rad[i] - cy[i])**2
        if cy[i] > 1.0 - rad[i]:
            pen += (cy[i] - (1.0 - rad[i]))**2
            
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            dist = np.hypot(dx, dy)
            min_d = rad[i] + rad[j]
            if dist < min_d:
                pen += (min_d - dist)**2
                
    return obj + mu * pen
