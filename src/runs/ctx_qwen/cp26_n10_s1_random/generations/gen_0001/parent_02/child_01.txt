import numpy as np
from scipy.optimize import minimize

def smooth_min_clearance(centers_flat, n, alpha):
    """
    Computes a smooth approximation of the minimum clearance to boundaries and other circles.
    """
    centers = centers_flat.reshape(n, 2)
    vals = []
    # Distance to boundaries
    vals.append(centers[:, 0])
    vals.append(1.0 - centers[:, 0])
    vals.append(centers[:, 1])
    vals.append(1.0 - centers[:, 1])
    
    # Half pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    vals.append(dists[dists < np.inf] / 2.0)
    
    vals = np.array(vals)
    # Log-sum-exp approximation of the minimum
    return -np.log(np.sum(np.exp(-alpha * vals))) / alpha

def neg_smooth_min(centers_flat, n, alpha):
    """Objective to minimize: negative smooth min clearance."""
    return -smooth_min_clearance(centers_flat, n, alpha)

def run_packing() -> tuple:
    n = 26
    best_r = 0.0
    best_centers = None
    
    inits = []
    # 1. Hexagonal grid initialization
    pts = []
    r0 = 0.09
    y = r0
    row = 0
    while len(pts) < n:
        x = r0
        if row % 2 == 1: x += r0
        while len(pts) < n and x + r0 <= 1.0:
            pts.append([x, y])
            x += 2 * r0
        y += r0 * np.sqrt(3)
        row += 1
    inits.append(np.array(pts[:n]))
    
    # 2. Perturbed hex grids to escape symmetry/local minima
    np.random.seed(42)
    for _ in range(5):
        noise = np.random.normal(0, 0.02, size=(n, 2))
        cfg = inits[0] + noise
        cfg = np.clip(cfg, 0.05, 0.95)
        inits.append(cfg)
        
    # 3. Regular grid initialization
    grid_x = np.linspace(0.1, 0.9, 6)
    grid_y = np.linspace(0.1, 0.9, 5)
    g_pts = []
    for yy in grid_y:
        for xx in grid_x:
            if len(g_pts) < n:
                g_pts.append([xx, yy])
    inits.append(np.array(g_pts[:n]))
            
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # Optimize from each initial configuration
    for init_c in inits:
        curr_c = init_c.flatten()
        # Continuation method: increase sharpness of the min approximation
        for alpha in [10, 20, 50, 100, 200]:
            res = minimize(
                neg_smooth_min,
                curr_c,
                args=(n, alpha),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-10}
            )
            curr_c = res.x
            
        # Evaluate exact min clearance for this configuration
        c_opt = curr_c.reshape(n, 2)
        vals = []
        vals.append(c_opt[:, 0])
        vals.append(1.0 - c_opt[:, 0])
        vals.append(c_opt[:, 1])
        vals.append(1.0 - c_opt[:, 1])
        
        diffs = c_opt[:, np.newaxis, :] - c_opt[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        vals.append(dists[dists < np.inf] / 2.0)
        
        min_clr = np.min(vals)
        if min_clr > best_r:
            best_r = min_clr
            best_centers = c_opt.copy()
            
    # Fallback safety net
    if best_centers is None:
        best_centers = inits[0]
        vals = []
        vals.append(best_centers[:, 0])
        vals.append(1.0 - best_centers[:, 0])
        vals.append(best_centers[:, 1])
        vals.append(1.0 - best_centers[:, 1])
        diffs = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        vals.append(dists[dists < np.inf] / 2.0)
        best_r = np.min(vals)
        
    # Assign equal radii, slightly shrunk for numerical safety margin
    radii = np.full(n, best_r * 0.999999)
    return best_centers, radii, np.sum(radii)
