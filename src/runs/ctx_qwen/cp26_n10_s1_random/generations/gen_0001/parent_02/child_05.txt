import numpy as np
from scipy.optimize import minimize

def get_centers_and_radii(vars_array, n):
    """Decodes optimization variables into centers and radii."""
    r = vars_array[0::3]
    u = vars_array[1::3]
    v = vars_array[2::3]
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    centers = np.column_stack((x, y))
    return centers, r

def objective_penalty(vars_array, n, mu):
    """Objective with overlap penalty for L-BFGS-B."""
    centers, r = get_centers_and_radii(vars_array, n)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    viol = np.maximum(0.0, r_sum - dist)
    penalty = mu * np.sum(viol**2)
    return -np.sum(r) + penalty

def constraint_function(vars_array, n):
    """Returns pairwise non-overlap constraints for SLSQP."""
    centers, r = get_centers_and_radii(vars_array, n)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    cons = dist - r_sum
    return cons[np.triu_indices(n, k=1)]

def objective_sum_radii(vars_array):
    """Standard objective for SLSQP refinement."""
    return -np.sum(vars_array[0::3])

def make_init(points, r_init, n):
    """Creates initial optimization vector from geometric points."""
    x0 = np.empty(3 * n)
    x0[0::3] = r_init
    denom = 1.0 - 2.0 * r_init
    u = (points[:, 0] - r_init) / denom
    v = (points[:, 1] - r_init) / denom
    x0[1::3] = u
    x0[2::3] = v
    return x0

def run_packing() -> tuple:
    n = 26
    best_sum = -np.inf
    best_vars = None
    
    # Bounds: r in [epsilon, 0.5], u, v in [0, 1]
    bounds = [(1e-6, 0.5)] * n + [(0.0, 1.0)] * n + [(0.0, 1.0)] * n

    configs = []

    pts = []
    r_est = 0.095

    y = r_est

        x = r_est + shift
        while x < 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r_est
        y += dy
        row += 1
    configs.append(np.array(pts[:n]))
    
    # 2. Regular grid
    cx = np.linspace(0.15, 0.85, 6)
    cy = np.linspace(0.15, 0.85, 5)
    pts_grid = np.array([[x, y] for y in cy for x in cx])[:n]
    configs.append(pts_grid)
    
    # 3. Random perturbations of hex pattern
    np.random.seed(42)
    for _ in range(5):
        noise = np.random.uniform(-0.03, 0.03, size=(n, 2))
        pts_pert = np.clip(np.array(pts[:n]) + noise, 0.1, 0.9)
        configs.append(pts_pert)

    # Stage 1: Fast penalty-based optimization
    mu = 2000.0
    for pts in configs:
        x0 = make_init(pts, 0.085, n)
        try:
            res = minimize(objective_penalty, x0, args=(n, mu), method='L-BFGS-B',

            if np.isfinite(res.fun):

            fb.append([x, y])
        if len(fb) >= n: break
    c_fb = np.array(fb[:n])
    r_fb = np.full(n, 0.08)
    return c_fb, r_fb, float(np.sum(r_fb))
