import numpy as np
from scipy.optimize import minimize

def objective_function(vars_arr, n):
    """Objective: Maximize the shared radius R (minimize -R)"""
    return -vars_arr[-1]

def constraint_function(vars_arr, n):
    """Constraints: Circles must be inside [0,1]^2 and non-overlapping"""
    centers = vars_arr[:2 * n].reshape(n, 2)
    R = vars_arr[-1]
    
    constraints = []
    
    # Boundary constraints: distance to each wall >= R
    for i in range(n):
        x, y = centers[i]
        constraints.append(x - R)          # Left wall
        constraints.append(1.0 - x - R)    # Right wall
        constraints.append(y - R)          # Bottom wall
        constraints.append(1.0 - y - R)    # Top wall
        
    # Pairwise non-overlap constraints: dist(i,j) >= 2R  <=>  dist^2 >= 4R^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            constraints.append(dist_sq - 4.0 * R * R)
            
    return np.array(constraints)

def run_packing() -> tuple:
    n = 26
    
    # Variable bounds: centers in [0, 1], radius in [small, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)]
    
    constraint_dict = {
        'type': 'ineq',
        'fun': constraint_function,
        'args': (n,)
    }
    
    # --- Generate Initial Configurations ---
    initial_configs = []
    
    # 1. Hexagonal Lattice (Optimal baseline)
    r_hex = 0.08
    hex_pts = []
    y = r_hex + 0.02
    row = 0
    while len(hex_pts) < n and y < 1.0:
        shift = r_hex if row % 2 == 1 else 0.0
        x = r_hex + shift + 0.02
        while len(hex_pts) < n and x < 1.0:
            hex_pts.append([x, y])
            x += 2 * r_hex
        y += np.sqrt(3) * r_hex
        row += 1
    initial_configs.append(np.array(hex_pts[:n]))
    
    # 2. Uniform Grid
    grid_size = 6
    step = 1.0 / (grid_size + 1)
    grid_pts = []
    for i in range(1, grid_size + 1):
        for j in range(1, grid_size + 1):
            if len(grid_pts) < n:
                grid_pts.append([i * step, j * step])
    initial_configs.append(np.array(grid_pts[:n]))
    
    # 3. Perturbed Hexagonal (to escape local minima)
    for seed in range(5):
        rng = np.random.RandomState(seed)
        p = initial_configs[0].copy()
        p += rng.uniform(-0.04, 0.04, size=p.shape)
        p = np.clip(p, 0.05, 0.95)
        initial_configs.append(p)
        
    # --- Optimization Loop ---
    best_R = 0.0
    best_centers = None
    
    for cfg in initial_configs:
        # Initial guess: [x0, y0, ..., x25, y25, R]
        x0 = np.concatenate([cfg.flatten(), [0.08]])
        
        try:
            res = minimize(
                objective_function,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint_dict,
                args=(n,),
                options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
            )
            
            if res.success or (np.isfinite(res.fun) and -res.fun > best_R):
                R_opt = res.x[-1]
                if R_opt > best_R:
                    best_R = R_opt
                    best_centers = res.x[:2 * n].reshape(n, 2)
        except Exception:
            continue
            
    # Fallback to a safe configuration if optimization fails
    if best_centers is None:
        best_centers = initial_configs[0]
        best_R = 0.08
        
    # --- Post-Processing & Strict Validation ---
    # Compute the exact maximum feasible equal radius for the optimized centers.
    # This guarantees validity and handles any tiny numerical violations from SLSQP.
    min_clearance = 1.0
    
    # Check boundaries
    for i in range(n):
        x, y = best_centers[i]
        min_clearance = min(min_clearance, x, 1.0 - x, y, 1.0 - y)
        
    # Check pairwise distances
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            min_clearance = min(min_clearance, d)
            
    # Radius is half the minimum clearance
    final_R = max(min_clearance / 2.0, 1e-6)
    final_radii = np.full(n, final_R)
    sum_radii = float(np.sum(final_radii))
    
    return best_centers, final_radii, sum_radii
