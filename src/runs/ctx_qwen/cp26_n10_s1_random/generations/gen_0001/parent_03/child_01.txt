To maximize the sum of radii for 26 circles in a unit square, we separate the problem into two stages:
1. **Radius Optimization (LP)**: For any fixed set of centers, finding the optimal radii that maximize the sum is a Linear Programming problem. The constraints are linear: each radius must be non-negative, bounded by the distance to the nearest wall, and satisfy $r_i + r_j \le \text{dist}(c_i, c_j)$ for all pairs.
2. **Center Optimization**: We treat the maximum sum of radii (from the LP) as an objective function over the center coordinates. Since this objective is non-smooth (active constraints change), we use derivative-free optimizers (`Powell` for broad search, `Nelder-Mead` for refinement).

This alternating approach is highly effective for circle packing. We initialize with multiple configurations (hexagonal, grid, random, perturbed) to avoid local minima. The LP solver (`scipy.optimize.linprog` with the `highs` method) is extremely fast for $N=26$, making this strategy computationally feasible and accurate. Finally, we return the exact LP-computed radii for the best centers, which strictly satisfy all constraints up to solver precision.

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
pair_indices = [(i, j) for i in range(N) for j in range(i + 1, N)]
NUM_PAIRS = len(pair_indices)

# Precompute the static constraint matrix for r_i + r_j <= dist(i, j)
A_ub = np.zeros((NUM_PAIRS, N))
for k, (i, j) in enumerate(pair_indices):
    A_ub[k, i] = 1.0
    A_ub[k, j] = 1.0

# Objective for LP: minimize -sum(r)
c_obj = np.ones(N) * -1.0

def solve_radii_lp(centers):
    """Solves the LP to find optimal radii for fixed centers."""
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Boundary constraints: 0 <= r_i <= min(x, 1-x, y, 1-y)
    upper_bounds = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    bounds = [(0.0, ub) for ub in upper_bounds]

    # Pairwise constraints RHS: dist(i, j)
    b_ub = np.zeros(NUM_PAIRS)
    for k, (i, j) in enumerate(pair_indices):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b_ub[k] = np.hypot(dx, dy)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    else:
        # Fallback if LP unexpectedly fails
        r = np.full(N, 0.01)
        return r, -np.sum(r)

def objective(centers_flat):
    """Objective function for center optimization: minimize negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    # Keep centers strictly inside to ensure positive feasible radii
    centers = np.clip(centers, 0.005, 0.995)
    _, s = solve_radii_lp(centers)
    return -s

def run_packing():
    best_val = np.inf
    best_centers = None

    inits = []
    
    # 1. Hexagonal lattice initialization
    r_init = 0.09
    pts = []
    row = 0
    y = r_init
    while y + r_init <= 1.0:
        shift = r_init if row % 2 == 1 else 0.0
        x = r_init + shift
        while x + r_init <= 1.0:
            pts.append([x, y])
            if len(pts) >= N: break
            x += 2 * r_init
        if len(pts) >= N: break
        y += np.sqrt(3) * r_init
        row += 1
    while len(pts) < N:
        pts.append([0.5, 0.5])
    inits.append(np.array(pts[:N]))

    # 2. Regular grid + center initialization
    grid = np.linspace(0.1, 0.9, 5)
    pts2 = [[gx, gy] for gx in grid for gy in grid]
    pts2.append([0.5, 0.5])
    inits.append(np.array(pts2[:N]))

    # 3. Random and perturbed initializations to escape local minima
    np.random.seed(42)
    for _ in range(4):
        inits.append(np.random.uniform(0.05, 0.95, size=(N, 2)))
    for _ in range(4):
        inits.append(np.clip(inits[0] + np.random.normal(0, 0.02, size=(N, 2)), 0.05, 0.95))

    # First pass: Broad search with Powell method
    for init in inits:
        try:
            res = minimize(objective, init.flatten(), method='Powell',
                           options={'maxiter': 500, 'xtol': 1e-6, 'ftol': 1e-7})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass

    # Refinement pass: High-precision local search with Nelder-Mead
    if best_centers is not None:
        try:
            res = minimize(objective, best_centers.flatten(), method='Nelder-Mead',
                           options={'maxiter': 2000, 'xatol': 1e-7, 'fatol': 1e-9, 'adaptive': True})
            best_centers = res.x.reshape(N, 2)
        except Exception:
            pass

    # Final exact radius calculation for the best centers found
    best_radii, final_sum = solve_radii_lp(best_centers)
    return best_centers, best_radii, final_sum
```
