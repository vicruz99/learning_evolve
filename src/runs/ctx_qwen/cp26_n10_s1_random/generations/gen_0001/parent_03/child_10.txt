import numpy as np
from scipy.optimize import minimize

def objective_penalty(vars, n, mu):
    """
    Computes the penalized objective for circle packing.
    Minimizes negative sum of radii while penalizing boundary and overlap violations.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Base objective: maximize sum of radii
    obj = -np.sum(r)
    
    # Boundary violations: circle must be inside [0,1]x[0,1]
    v = np.vstack([r - x, r - (1 - x), r - y, r - (1 - y)])
    v = np.maximum(0, v)
    obj += mu * np.sum(v**2)
    
    # Overlap violations: dist(i,j) >= r_i + r_j
    centers = np.column_stack((x, y))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    viol = np.maximum(0, r_sum - dist)
    
    # Only consider upper triangle to avoid double counting
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    obj += mu * np.sum(viol[mask]**2)
    
    return obj
