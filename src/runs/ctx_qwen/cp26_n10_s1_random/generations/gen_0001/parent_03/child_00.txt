import numpy as np
from scipy.optimize import minimize

def obj(vars_arr):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_arr[2::3])

def cons(vars_arr, idx_i, idx_j):
    """
    Constraint function: returns array of values that must be >= 0.
    Includes boundary constraints and pairwise non-overlap constraints.
    Vectorized for performance.
    """
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    
    # Pairwise differences using precomputed upper-triangle indices
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    
    return np.concatenate([
        x - r,                  # Left boundary: x >= r
        1.0 - x - r,            # Right boundary: x + r <= 1
        y - r,                  # Bottom boundary: y >= r
        1.0 - y - r,            # Top boundary: y + r <= 1
        dx**2 + dy**2 - dr**2   # Non-overlap: dist^2 >= (r_i + r_j)^2
    ])

def run_packing():
    n = 26
    idx_i, idx_j = np.triu_indices(n, k=1)
    
    # Bounds for [x, y, r] for each circle
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(1e-5, 0.5)] * n
    
    best_x = None
    best_obj = np.inf
    
    configs = []
    
    # 1. Hexagonal lattice initialization
    r_hex = 0.095
    pts_hex = []
    dy = r_hex * np.sqrt(3)
    for row in range(7):
        y = r_hex + row * dy
        shift = r_hex if row % 2 == 1 else 0.0
        x = r_hex + shift
        while x + r_hex <= 1.0 + 1e-6 and len(pts_hex) < n:
            pts_hex.append([x, y])
            x += 2 * r_hex
        if len(pts_hex) >= n: break
    while len(pts_hex) < n:
        pts_hex.append([0.5, 0.5])
    configs.append(np.array(pts_hex[:n]))
    
    # 2. Regular grid initialization (5x5 + 1)
    g = np.linspace(0.12, 0.88, 5)
    pts_grid = [[x, y] for y in g for x in g]
    pts_grid.append([0.5, 0.5])
    configs.append(np.array(pts_grid[:n]))

    np.random.seed(42)

        cfg = configs[idx].copy()

        cfg = np.clip(cfg, 0.05, 0.95)
        configs.append(cfg)
        
    # Optimization loop over configurations
    for cfg in configs:
        x0 = np.zeros(3*n)
        x0[0::3] = cfg[:, 0]
        x0[1::3] = cfg[:, 1]
        x0[2::3] = 0.08  # Start with feasible radii to help SLSQP
        
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': cons, 'args': (idx_i, idx_j)},

            if res.fun < best_obj:
                best_obj = res.fun
                best_x = res.x
        except Exception:
            pass
            
    # Fallback if optimization fails completely
    if best_x is None:
        x0 = np.zeros(3*n)
        x0[0::3] = np.linspace(0.1, 0.9, n)
        x0[1::3] = 0.5
        x0[2::3] = 0.05
        best_x = minimize(obj, x0, method='SLSQP', bounds=bounds,
