import numpy as np
from scipy.optimize import minimize

N = 26

def eval_constraints(vars):
    """
    Vectorized constraint evaluation for SLSQP.
    Returns an array of values that must be >= 0 for a valid packing.
    Constraints:
      1. Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
      2. Pairwise: dist(i,j)^2 >= (r_i + r_j)^2
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints (4*N)
    b_cons = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise constraints (N*(N-1)/2) using broadcasting
    diff_x = x[:, np.newaxis] - x[np.newaxis, :]
    diff_y = y[:, np.newaxis] - y[np.newaxis, :]
    sum_r = r[:, np.newaxis] + r[np.newaxis, :]
    
    dist_sq = diff_x**2 + diff_y**2
    r_sum_sq = sum_r**2
    
    # Extract upper triangle (i < j)
    idx = np.triu_indices(N, k=1)
    p_cons = dist_sq[idx] - r_sum_sq[idx]
    
    return np.concatenate([b_cons, p_cons])

def obj(vars):
    """Objective: maximize sum of radii -> minimize negative sum"""
    return -np.sum(vars[2::3])

def run_packing():
    # Bounds: x, y in [0, 1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    constraint_def = {'type': 'ineq', 'fun': eval_constraints}
    
    best_sum = -np.inf
    best_vars = None
    
    # Generate diverse starting configurations
    starts = []
    
    # 1. Hexagonal lattice
    r_h = 0.10
    pts_hex = []
    y = r_h
    row = 0
    while len(pts_hex) < N:
        x = r_h if row % 2 == 0 else 2 * r_h
        while x + r_h <= 1.0 and len(pts_hex) < N:
            pts_hex.append([x, y])
            x += 2 * r_h
        y += np.sqrt(3) * r_h
        row += 1
    starts.append(np.array(pts_hex[:N]))
    
    # 2. Uniform grid
    pts_grid = []
    for i in range(1, 6):
        for j in range(1, 6):
            if len(pts_grid) < N:
                pts_grid.append([0.2 * i, 0.2 * j])
    starts.append(np.array(pts_grid))
    
    # 3-8. Perturbed versions of hex and grid
    np.random.seed(42)
    for _ in range(6):
        base = starts[np.random.randint(2)].copy()
        noise = np.random.uniform(-0.04, 0.04, size=base.shape)
        starts.append(np.clip(base + noise, 0.05, 0.95))
        
    # 9-10. Pure random feasible starts
    for _ in range(2):
        cx = np.random.uniform(0.15, 0.85, N)
        cy = np.random.uniform(0.15, 0.85, N)
        starts.append(np.column_stack((cx, cy)))
        
    # Optimize each start
    for init_centers in starts:
        x0 = np.zeros(3 * N)
        x0[0::3] = init_centers[:, 0]
        x0[1::3] = init_centers[:, 1]
        x0[2::3] = 0.065  # Feasible initial radius
        
        try:
            res = minimize(
                obj,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint_def,
                options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False}
            )
            
            if np.isfinite(res.fun):
                current_sum = -res.fun
                if current_sum > best_sum:
                    # Quick internal validation
                    c = np.column_stack((res.x[0::3], res.x[1::3]))
                    r = res.x[2::3]
                    
                    valid = True
                    # Check boundaries
                    if np.any(r < 0) or np.any(c[:, 0] < r) or np.any(c[:, 0] > 1 - r) or \
                       np.any(c[:, 1] < r) or np.any(c[:, 1] > 1 - r):
                        valid = False
                    
                    # Check overlaps
                    if valid:
                        dists = np.linalg.norm(c[:, np.newaxis, :] - c[np.newaxis, :, :], axis=2)
                        r_sums = r[:, np.newaxis] + r[np.newaxis, :]
                        if np.any(dists[np.triu_indices(N, k=1)] < r_sums[np.triu_indices(N, k=1)] - 1e-9):
                            valid = False
                            
                    if valid:
                        best_sum = current_sum
                        best_vars = res.x.copy()
                        
        except Exception:
            continue
            
    # Fallback if optimization fails unexpectedly
    if best_vars is None:
        best_vars = np.zeros(3 * N)
        best_vars[0::3] = 0.5
        best_vars[1::3] = 0.5
        best_vars[2::3] = 0.02
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    # Final safety clamp to guarantee validation passes
    radii = np.maximum(radii, 1e-9)
    centers = np.clip(centers, radii, 1.0 - radii)
    
    # Slight uniform shrink to account for float precision in validator
    # Only if constraints are extremely tight
    cons_vals = eval_constraints(best_vars)
    if np.min(cons_vals) < -1e-7:
        scale = 0.9995
        radii *= scale
        centers = np.clip(centers, radii, 1.0 - radii)
        
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
