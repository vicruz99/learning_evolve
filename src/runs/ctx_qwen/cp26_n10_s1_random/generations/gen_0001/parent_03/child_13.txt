def compute_constraints(x, n):

    Computes inequality constraints for the circle packing problem.
    Returns an array of values that must be >= 0 for a valid packing.
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    # x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
