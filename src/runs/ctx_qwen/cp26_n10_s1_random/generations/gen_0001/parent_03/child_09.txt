import numpy as np
from scipy.optimize import minimize

def objective(vars_arr, n):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(vars_arr[2::3])

def compute_constraints(vars_arr, n):
    """Constraint function: returns array of values >= 0 for valid packing"""
    # Preallocate constraint array: 4 boundary per circle + pairwise overlaps
    cons = np.zeros(4*n + n*(n-1)//2)
    idx = 0
    
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    
    # Boundary constraints: x-r>=0, 1-x-r>=0, y-r>=0, 1-y-r>=0
    for i in range(n):
        cons[idx] = x[i] - r[i]; idx += 1
        cons[idx] = 1.0 - x[i] - r[i]; idx += 1
        cons[idx] = y[i] - r[i]; idx += 1
        cons[idx] = 1.0 - y[i] - r[i]; idx += 1
        
    # Overlap constraints: dist^2 - (r_i+r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            cons[idx] = dx**2 + dy**2 - (r[i] + r[j])**2
            idx += 1
            
    return cons

def force_expand(centers, r_init, n, steps=600):
    """Force-directed expansion to find a dense initial configuration"""
    r = r_init
    dt = 0.05
    stiffness = 50.0
    wall_stiffness = 100.0
    
    for _ in range(steps):
        r *= 1.0008
        forces = np.zeros_like(centers)
        
        for i in range(n):
            # Wall repulsion
            if centers[i, 0] - r < 0: forces[i, 0] += wall_stiffness * (r - centers[i, 0])
            if centers[i, 0] + r > 1: forces[i, 0] -= wall_stiffness * (centers[i, 0] + r - 1)
            if centers[i, 1] - r < 0: forces[i, 1] += wall_stiffness * (r - centers[i, 1])
            if centers[i, 1] + r > 1: forces[i, 1] -= wall_stiffness * (centers[i, 1] + r - 1)
            
            # Circle-circle repulsion
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < 2 * r and dist > 1e-8:
                    overlap = 2 * r - dist
                    f = stiffness * overlap / dist
                    forces[i, 0] += dx * f
                    forces[i, 1] += dy * f
                    forces[j, 0] -= dx * f
                    forces[j, 1] -= dy * f
                    
        centers += forces * dt
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
        
    return centers, r

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    configs = []
    
    # 1. Hexagonal lattice
    pts = []
    row, y = 0, 0.09
    while y + 0.09 <= 1.0:
        shift = 0.09 if row % 2 == 1 else 0.0
        x = 0.09 + shift
        while x + 0.09 <= 1.0:
            pts.append([x, y])
            x += 0.18
        y += 0.09 * 1.732
        row += 1
    while len(pts) < n: pts.append([0.5, 0.5])
    configs.append(np.array(pts[:n]))
    
    # 2. Regular grid
    pts2 = [[0.05 + i*0.16, 0.1 + j*0.2] for i in range(6) for j in range(5)]
    configs.append(np.array(pts2[:n]))
    
    # 3. Random dense configurations
    np.random.seed(42)
    for _ in range(3):
        configs.append(np.random.uniform(0.1, 0.9, size=(n, 2)))

    bounds = [(0.0, 1.0)] * (2*n) + [(1e-5, 0.5)] * n
    cons_args = (n,)
    
    for cfg in configs:
        # Force expand to get a tight packing
        centers_exp, r_exp = force_expand(cfg, 0.04, n, steps=600)
        
        # Ensure strict feasibility for SLSQP start
        centers_start = centers_exp * 0.95 + 0.025
        centers_start = np.clip(centers_start, 0.1, 0.9)
        r_start = r_exp * 0.8
        
        x0 = np.zeros(3 * n)
        x0[0::3] = centers_start[:, 0]
        x0[1::3] = centers_start[:, 1]
        x0[2::3] = r_start
        
        # Add small jitter to break symmetry and help optimizer
        x0 += np.random.uniform(-1e-4, 1e-4, size=x0.shape)
        
        try:
            res = minimize(objective, x0, args=(n,), method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': compute_constraints, 'args': cons_args},
                           options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
            
            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
            r_opt = res.x[2::3]
            
            # Strict validation check
            valid = True
            for i in range(n):
                x, y = c_opt[i]
                r = r_opt[i]
                if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
                    valid = False; break
            if valid:
                for i in range(n):
                    for j in range(i+1, n):
                        dist = np.sqrt(np.sum((c_opt[i]-c_opt[j])**2))
                        if dist < r_opt[i] + r_opt[j] - 1e-9:
                            valid = False; break
                    if not valid: break
                    
            if valid:
                s = np.sum(r_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = c_opt
                    best_radii = r_opt
        except Exception:
            pass
            
    # Fallback to a safe small packing if optimization fails
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.full(n, 0.04)
        for i in range(n):
            best_centers[i] = [0.1 + (i%6)*0.15, 0.1 + (i//6)*0.15]
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
