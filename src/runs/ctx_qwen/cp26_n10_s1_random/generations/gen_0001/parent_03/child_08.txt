To achieve this high sum, the strategy focuses on:
1. **High-Quality Initialization**: Generating a base hexagonal lattice that tightly fits 26 circles. Multiple perturbed versions of this lattice are created to help the optimizer escape local minima.
2. **Explicit Constraint Optimization**: Using `scipy.optimize.minimize` with the SLSQP algorithm, which natively handles non-linear inequality constraints. All boundary and non-overlap constraints are formulated into a single vector-valued function for computational efficiency and robustness.
3. **Joint Position-Radius Optimization**: Optimizing centers $(x_i, y_i)$ and radii $r_i$ simultaneously to maximize $\sum r_i$. The constraints naturally push circles to boundaries and each other, maximizing space utilization.

import numpy as np
from scipy.optimize import minimize

def compute_constraints(vars, n):
    """Computes all inequality constraints for the circle packing problem."""
    # Total constraints: 4 per circle (boundaries) + n*(n-1)/2 (non-overlap)
    c = np.zeros(n * 4 + n * (n - 1) // 2)
    idx = 0
    
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    for i in range(n):
        c[idx] = vars[3*i] - vars[3*i+2]; idx += 1          # x_i - r_i >= 0
        c[idx] = 1.0 - vars[3*i] - vars[3*i+2]; idx += 1    # 1 - x_i - r_i >= 0
        c[idx] = vars[3*i+1] - vars[3*i+2]; idx += 1        # y_i - r_i >= 0
        c[idx] = 1.0 - vars[3*i+1] - vars[3*i+2]; idx += 1  # 1 - y_i - r_i >= 0
        
    # Non-overlap constraints: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dx = vars[3*i] - vars[3*j]
            dy = vars[3*i+1] - vars[3*j+1]
            dr = vars[3*i+2] + vars[3*j+2]
            c[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
            
    return c

def obj(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def run_packing():
    n = 26
    
    # Bounds: centers in [0, 1], radii in [1e-6, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    # Constraint definition using a single function call for efficiency
    cons = {'type': 'ineq', 'fun': compute_constraints, 'args': (n,)}
    
    best_sum = 0.0
    best_vars = None
    
    np.random.seed(42)
    
    # Generate initial configurations
    inits = []
    
    # 1. Base hexagonal grid (known to be near-optimal for dense packing)
    r_hex = 0.095
    dy = r_hex * np.sqrt(3)
    pts = []
    for row in range(6):
        y = r_hex + row * dy
        if y + r_hex > 1.05: break
        shift = r_hex if row % 2 == 1 else 0.0
        x = r_hex + shift
        while x + r_hex <= 1.05:
            pts.append([x, y])
            x += 2 * r_hex
            if len(pts) >= 26: break
        if len(pts) >= 26: break
    base_init = np.array(pts[:26])
    inits.append(base_init)
    
    # 2. Perturbed hex grids to escape local minima
    for _ in range(8):
        cfg = base_init.copy()
        cfg += np.random.normal(0, 0.01, cfg.shape)
        cfg = np.clip(cfg, 0.05, 0.95)
        inits.append(cfg)
        
    # 3. Dense grid fallback
    cfg_grid = np.array([[0.08 + 0.16*i, 0.08 + 0.16*j] for j in range(6) for i in range(5)])
    inits.append(cfg_grid[:26])
    
    # Optimization loop over initializations
    for cfg in inits:
        x0 = np.zeros(3 * n)
        x0[0::3] = cfg[:, 0]
        x0[1::3] = cfg[:, 1]
        x0[2::3] = 0.09  # Initial radii guess
        
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-11})
            if not np.isnan(res.x).any():
                r_opt = res.x[2::3]
                s = np.sum(r_opt)
                if s > best_sum:
                    best_sum = s
                    best_vars = res.x
        except Exception:
            continue
            
    # Fallback in case optimization fails entirely
    if best_vars is None:
        best_vars = np.zeros(3 * n)
        best_vars[0::3] = base_init[:, 0]
        best_vars[1::3] = base_init[:, 1]
        best_vars[2::3] = 0.09
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    return centers, radii, best_sum
