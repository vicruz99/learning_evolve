import numpy as np
from scipy.optimize import minimize

def decode(vars, n):
    """Decodes optimization variables into centers and radii using boundary-safe parameterization."""
    r = vars[0::3]
    u = vars[1::3]
    v = vars[2::3]
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    return np.column_stack((x, y)), r

def compute_penalty(vars, n, mu):
    """Objective: maximize sum of radii with quadratic penalty for overlaps."""
    centers, r = decode(vars, n)
    obj = -np.sum(r)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    violation = np.maximum(0.0, r_sum - dist)
    penalty = np.sum(violation**2)
    
    return obj + mu * penalty

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds: r in (1e-5, 0.45), u, v in [0, 1]
    bounds = [(1e-5, 0.45)] * n + [(0.0, 1.0)] * n + [(0.0, 1.0)] * n
    
    np.random.seed(42)
    
    inits = []
    
    # 1. Random initializations
    for _ in range(15):
        v0 = np.random.rand(3 * n)
        v0[0::3] = np.random.uniform(0.04, 0.1, n)
        inits.append(v0)

    hex_pts = []
    r_h = 0.09
    y_h = r_h
    row = 0
    while y_h <= 1.0 and len(hex_pts) < n:
        shift = 0.0 if row % 2 == 0 else r_h
        x_h = r_h + shift
        while x_h <= 1.0 and len(hex_pts) < n:
            hex_pts.append([x_h, y_h])
            x_h += 2.0 * r_h
        y_h += np.sqrt(3.0) * r_h
        row += 1
        
    while len(hex_pts) < n:
        hex_pts.append([0.5, 0.5])
        
    hex_centers = np.array(hex_pts[:n])
    r_h_init = 0.09
    u_h = (hex_centers[:, 0] - r_h_init) / (1.0 - 2.0 * r_h_init)
    v_h = (hex_centers[:, 1] - r_h_init) / (1.0 - 2.0 * r_h_init)
    u_h = np.clip(u_h, 0.0, 1.0)
    v_h = np.clip(v_h, 0.0, 1.0)
    
    v0_hex = np.zeros(3 * n)
    v0_hex[0::3] = r_h_init
    v0_hex[1::3] = u_h
    v0_hex[2::3] = v_h
    inits.append(v0_hex)
    
    # Optimize each initialization
    for v0 in inits:
        v_curr = v0.copy()
        mu = 50.0
        
        # Annealing loop: gradually increase penalty weight
        for step in range(40):
            res = minimize(compute_penalty, v_curr, args=(n, mu), 
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-12})
            v_curr = res.x
            mu *= 2.0
            
        centers, radii = decode(v_curr, n)
        
        # Validate and repair slight numerical violations
