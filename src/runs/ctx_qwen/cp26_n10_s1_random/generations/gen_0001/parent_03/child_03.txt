import numpy as np
from scipy.optimize import minimize, linprog

def objective(vars):
    """Objective: maximize the shared radius r (minimize -r)"""
    return -vars[-1]

def constraint_boundary(vars, idx, n):
    """Ensures circle idx stays within [0,1]x[0,1] given radius r"""
    r = vars[2*n]
    xi = vars[2*idx]
    yi = vars[2*idx + 1]
    # Return minimum clearance to any boundary
    return min(xi - r, 1 - xi - r, yi - r, 1 - yi - r)

def constraint_overlap(vars, i, j, n):
    """Ensures circles i and j do not overlap given radius r"""
    xi, yi = vars[2*i], vars[2*i+1]
    xj, yj = vars[2*j], vars[2*j+1]
    r = vars[2*n]
    # Squared distance minus squared sum of radii (2r)^2
    return (xi - xj)**2 + (yi - yj)**2 - 4.0 * (r**2)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    constraints = []
    
    # Define constraints without lambda functions
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': constraint_boundary, 'args': (i, n)})
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap, 'args': (i, j, n)})
            
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)]
    
    best_r = 0.0
    best_centers = None
    
    # Generate a base hexagonal lattice for initialization
    r_init = 0.085
    dy = np.sqrt(3) * r_init
    dx = 2 * r_init
    pts = []
    y = r_init
    row = 0
    while y + r_init <= 1.0:
        x = r_init if row % 2 == 0 else 2 * r_init
        while x + r_init <= 1.0:
            pts.append([x, y])
            x += dx
        y += dy
        row += 1
        
    while len(pts) < 30:
        pts.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
        
    np.random.seed(42)
    # Multiple restarts to escape local minima
    for _ in range(20):
        cfg_pts = np.array(pts[:26]) + np.random.uniform(-0.03, 0.03, (26, 2))
        cfg_pts = np.clip(cfg_pts, 0.05, 0.95)
        cfg = np.hstack([cfg_pts.flatten(), [r_init]])
        
        try:
            res = minimize(objective, cfg, method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 5000, 'ftol': 1e-10})
            if res.success:
                r_opt = -res.fun
                if r_opt > best_r:
                    best_r = r_opt
                    best_centers = res.x[:2*n].reshape(n, 2)
        except Exception:
            pass
            
    if best_centers is None:
        best_centers = np.array(pts[:26])
        best_r = 0.085
        
    # Phase 2: Optimize individual radii via Linear Programming for fixed centers
    c = np.ones(n) * -1  # Maximize sum(r) => minimize -sum(r)
    
    # Boundary limits for each circle
    b_lim = np.minimum(np.minimum(best_centers[:, 0], 1 - best_centers[:, 0]),
                       np.minimum(best_centers[:, 1], 1 - best_centers[:, 1]))
    bounds_lp = [(0.0, lim) for lim in b_lim]
    
    # Pairwise distance constraints: r_i + r_j <= dist(i,j)
    diff = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    n_constraints = n * (n - 1) // 2
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    radii = np.full(n, best_r)  # Fallback
    
    try:
        res_lp = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res_lp.success:
            radii = res_lp.x
    except Exception:
        try:
            res_lp = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='simplex')
            if res_lp.success:
                radii = res_lp.x
        except Exception:
            pass
            
    # Safety scaling to guarantee strict validity within numerical tolerance
    scale = 1.0
    for i in range(n):
        x, y = best_centers[i]
        r = radii[i]
        if r < 1e-12: continue
        ratio = min(x/r, (1-x)/r, y/r, (1-y)/r)
        if ratio < scale: scale = ratio
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(best_centers[i] - best_centers[j])
            r_sum = radii[i] + radii[j]
            if r_sum > 1e-12:
                ratio = dist / r_sum
                if ratio < scale: scale = ratio
                
    # Apply scale with a tiny margin for numerical safety
    radii *= scale * 0.99999
    
    sum_radii = float(np.sum(radii))
    return best_centers, radii, sum_radii
