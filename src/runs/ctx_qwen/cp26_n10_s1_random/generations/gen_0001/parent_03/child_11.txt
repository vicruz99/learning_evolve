import numpy as np
from scipy.optimize import minimize

def evaluate_constraints(v, n):
    """Evaluates all boundary and non-overlap constraints for the packing."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: circles must be inside [0,1]x[0,1]
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    b_cons = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: dist(i,j)^2 >= (r_i + r_j)^2
    # Vectorized pairwise distance squared computation
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    diff = dist_sq - r_sum_sq
    
    # Extract upper triangle to avoid duplicates and self-checks
    triu_idx = np.triu_indices(n, k=1)
    o_cons = diff[triu_idx]
    
    return np.concatenate([b_cons, o_cons])

def constraint_func(v):
    """Top-level constraint function for scipy optimizer."""
    return evaluate_constraints(v, 26)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def generate_hex_init(r_init=0.101):
    """Generates a centered hexagonal lattice initialization for 26 circles."""
    pts = []
    # Optimal row structure for 26 circles in hex packing
    rows = [5, 6, 5, 6, 4]
    # Center the entire structure vertically
    y = 0.5 - (len(rows) - 1) * r_init * np.sqrt(3) / 2.0
    
    for idx, count in enumerate(rows):
        # Shift odd rows horizontally by r_init for hexagonal nesting
        shift = r_init if idx % 2 == 1 else 0.0
        total_w = (count - 1) * 2 * r_init
        x_start = 0.5 - total_w / 2.0 + shift
        
        for k in range(count):
            if len(pts) < 26:
                x = x_start + k * 2 * r_init
                pts.append([x, y])
        y += r_init * np.sqrt(3)
        
    # Fallback padding if structure yields fewer than 26
    while len(pts) < 26:
        pts.append([0.5, 0.5])
        
    return np.array(pts[:26])

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Variable bounds: x,y in [0,1], r in [1e-6, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
        
    constraint = {'type': 'ineq', 'fun': constraint_func}
    
    # Prepare diverse initial configurations
    configs = []
    cfg_hex = generate_hex_init(0.101)
    configs.append(cfg_hex)
    
    # Add perturbed hexagonal configurations to escape local minima
    for _ in range(5):
        cfg = cfg_hex + np.random.normal(0, 0.015, cfg_hex.shape)
        cfg = np.clip(cfg, 0.05, 0.95)
        configs.append(cfg)
        
    # Add a dense grid fallback
    grid_pts = np.array([[i*0.2+0.1, j*0.2+0.1] for j in range(5) for i in range(5)])
    configs.append(np.vstack([grid_pts, [0.5, 0.5]]))
    
    # Optimize from each configuration
    for cfg in configs:
        v0 = np.zeros(3 * n)
        v0[0::3] = cfg[:, 0]
        v0[1::3] = cfg[:, 1]
        v0[2::3] = 0.095
        
        try:
            res = minimize(
                objective, v0, method='SLSQP', bounds=bounds, 
                constraints=constraint, 
                options={'maxiter': 5000, 'ftol': 1e-12}
            )
            
            if np.isfinite(res.fun):
                c_opt = res.x[:2*n].reshape(n, 2)
                r_opt = res.x[2*n:]
                
                # Verify constraints are satisfied within tolerance
                cons_val = evaluate_constraints(res.x, n)
                if np.all(cons_val >= -1e-8):
                    s = np.sum(r_opt)
                    if s > best_sum:
                        best_sum = s
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Fallback if optimization fails completely
    if best_centers is None:
        best_centers = generate_hex_init(0.09)
        best_radii = np.full(n, 0.09)
        best_sum = np.sum(best_radii)

    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        best_centers[i, 0] = np.clip(x, r, 1.0 - r)
        best_centers[i, 1] = np.clip(y, r, 1.0 - r)

                    best_radii[i] *= 0.9995
                    best_radii[j] *= 0.9995
                    valid = False
        if valid: break
        
    best_sum = np.sum(best_radii)
    return best_centers, best_radii, best_sum
