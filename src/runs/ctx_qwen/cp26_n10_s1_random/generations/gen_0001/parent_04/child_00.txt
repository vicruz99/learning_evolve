import numpy as np
from scipy.optimize import minimize

def compute_objective(vars_arr):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)."""
    return -np.sum(vars_arr[2::3])

def compute_constraints(vars_arr, n):
    """Compute inequality constraints >= 0 for boundaries and non-overlaps."""
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    b1 = x - r                # x >= r
    b2 = 1.0 - x - r          # x <= 1 - r
    b3 = y - r                # y >= r
    b4 = 1.0 - y - r          # y <= 1 - r
    
    # Pairwise non-overlap constraints (vectorized)
    # dist_sq(i,j) - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    r_sum_sq = r_sum**2
    
    # Extract upper triangle to avoid duplicate constraints
    i_tri, j_tri = np.triu_indices(n, k=1)
    overlap = dist_sq[i_tri, j_tri] - r_sum_sq[i_tri, j_tri]
    
    return np.concatenate([b1, b2, b3, b4, overlap])

def run_packing():
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    best_sum = -1.0
    best_x = None
    
    configs = []
    
    # 1. Hexagonal lattice initializations
    r_hex = 0.085
    pts = []
    row = 0
    y = r_hex
    while y < 1.0:
        x_start = r_hex if row % 2 == 0 else 2 * r_hex
        x = x_start
        while x < 1.0:
            if x + r_hex <= 1.0 and y + r_hex <= 1.0:
                pts.append([x, y])
            x += 2 * r_hex
        y += np.sqrt(3) * r_hex
        row += 1
    pts = np.array(pts[:n])
    
    for seed in range(8):
        np.random.seed(seed)
        cfg = pts.copy()
        cfg += np.random.uniform(-0.02, 0.02, cfg.shape)
        cfg = np.clip(cfg, 0.05, 0.95)
        configs.append(cfg)
        
    # 2. Grid pattern initializations
    grid = np.zeros((n, 2))
    xs = np.linspace(0.12, 0.88, 6)
    ys = np.linspace(0.12, 0.88, 5)
    idx = 0
    for yy in ys:
        for xx in xs:
            if idx < n:
                grid[idx] = [xx, yy]
                idx += 1
    while idx < n:
        grid[idx] = [0.5, 0.5]
        idx += 1
    for _ in range(4):
        cfg = grid.copy()
        cfg += np.random.uniform(-0.015, 0.015, cfg.shape)
        cfg = np.clip(cfg, 0.05, 0.95)
        configs.append(cfg)
        
    # Optimization loop over multiple starts
    constraint_def = {'type': 'ineq', 'fun': compute_constraints, 'args': (n,)}
    
    for centers_init in configs:
        x0 = np.zeros(3 * n)
        x0[0::3] = centers_init[:, 0]
        x0[1::3] = centers_init[:, 1]
        
        # Compute feasible initial radii based on positions
        r0 = np.minimum(centers_init[:, 0], 1.0 - centers_init[:, 0])
        r0 = np.minimum(r0, np.minimum(centers_init[:, 1], 1.0 - centers_init[:, 1]))
        # Shrink slightly to guarantee initial feasibility
        r0 = np.clip(r0 * 0.85, 0.04, 0.12)
        x0[2::3] = r0
        
        try:
            res = minimize(
                compute_objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint_def,
                options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
            )
            
            if np.isfinite(res.fun):
                cur_sum = np.sum(res.x[2::3])
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_x = res.x.copy()
        except Exception:
            continue
            
    if best_x is None:
        # Fallback safe configuration
        centers = np.random.uniform(0.2, 0.8, (n, 2))
        radii = np.full(n, 0.05)
        return centers, radii, np.sum(radii)
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final adjustment to guarantee strict validity within tolerance
    max_viol = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            v = (radii[i] + radii[j]) - d
            if v > max_viol: max_viol = v
        x, y = centers[i]
        r = radii[i]
        if x - r < 0: max_viol = max(max_viol, -(x - r))
        if x + r > 1: max_viol = max(max_viol, x + r - 1)
        if y - r < 0: max_viol = max(max_viol, -(y - r))
        if y + r > 1: max_viol = max(max_viol, y + r - 1)
        
    if max_viol > 1e-12:
        shrink = max_viol * 0.5 + 1e-10
        radii = np.maximum(radii - shrink, 1e-9)
        
    return centers, radii, np.sum(radii)
