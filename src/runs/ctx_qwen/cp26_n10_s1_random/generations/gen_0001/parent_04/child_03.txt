def compute_optimal_radii(centers, n):
    """
    Computes the maximum feasible radii for fixed centers such that 
    circles do not overlap and stay within [0,1]x[0,1].
    Solves: r_i + r_j <= dist(i,j) and r_i <= boundary(i)
    """
    # Initialize with boundary limits
    r = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    
    # Iterative relaxation to satisfy pairwise constraints
    for _ in range(300):
        r_new = r.copy()
        for i in range(n):
            limit = r[i]
            for j in range(n):
                if i == j: 
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                val = dist - r[j]
                if val < limit:
                    limit = val
            r_new[i] = limit
        r = r_new
        
    return np.maximum(r, 1e-9)
