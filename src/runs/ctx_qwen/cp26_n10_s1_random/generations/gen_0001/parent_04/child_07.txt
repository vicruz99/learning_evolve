from scipy.optimize import minimize

def _obj(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def _constraints(v, n, mask_indices):
    """
    Constraint function: returns array of values >= 0 for valid packing.
    Includes boundary and pairwise non-overlap constraints.
    """
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    b = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = x[mask_indices[0]] - x[mask_indices[1]]
    dy = y[mask_indices[0]] - y[mask_indices[1]]
    dr = r[mask_indices[0]] + r[mask_indices[1]]
    p = dx**2 + dy**2 - dr**2
    
    return np.concatenate([b, p])
