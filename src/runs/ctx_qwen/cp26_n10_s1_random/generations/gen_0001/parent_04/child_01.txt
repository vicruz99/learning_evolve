The constraints are expressed as quadratic inequalities: `(x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2`.
2. **Solver Choice**: We use `scipy.optimize.minimize` with the `SLSQP` method, which handles bound and general inequality constraints efficiently. The constraint function is fully vectorized using NumPy to evaluate all ~429 constraints rapidly per iteration.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_constraints(vars_flat):
    """
    Computes all boundary and pairwise non-overlap constraints.
    Returns a 1D array where each element >= 0 indicates a satisfied constraint.
    """
    n = N_CIRCLES
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    cons = []
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    cons.extend(x - r)
    cons.extend(1.0 - x - r)
    cons.extend(y - r)
    cons.extend(1.0 - y - r)
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # Vectorized computation for efficiency
    X = x[:, None] - x[None, :]
    Y = y[:, None] - y[None, :]
    R_sum = r[:, None] + r[None, :]
    D2 = X**2 + Y**2
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    cons.extend(D2[i_idx, j_idx] - R_sum[i_idx, j_idx]**2)
    
    return np.array(cons)

def compute_objective(vars_flat):
    """Objective: minimize negative sum of radii (maximize sum of radii)"""
    return -np.sum(vars_flat[2::3])

def generate_initial_config(rng):
    """Generates an initial hexagonal lattice configuration with random perturbations."""
    n = N_CIRCLES
    r_guess = 0.09 + rng.uniform(-0.005, 0.015)
    perturb = 0.01 + rng.uniform(0, 0.02)
    
    pts = []
    dy = r_guess * np.sqrt(3)
    row = 0
    y = r_guess
    while y < 1.0 - r_guess:
        x = r_guess
        shift = r_guess if row % 2 == 1 else 0.0
        while x < 1.0 - r_guess:
            pts.append([x + shift, y])
            x += 2 * r_guess
        y += dy
        row += 1
        
    pts = np.array(pts[:n])
    if pts.shape[0] < n:
        extra = rng.uniform(0.2, 0.8, size=(n - pts.shape[0], 2))
        pts = np.vstack([pts, extra])
        
    pts += rng.normal(0, perturb, pts.shape)
    pts = np.clip(pts, 0.1, 0.9)
    
    vars_flat = np.zeros(3 * n)
    vars_flat[0::3] = pts[:, 0]
    vars_flat[1::3] = pts[:, 1]
    vars_flat[2::3] = r_guess * 0.95
    return vars_flat

def run_packing():
    n = N_CIRCLES
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Variable bounds: centers in [0,1], radii in [1e-5, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-5, 0.5)] * n
    constraint_dict = {'type': 'ineq', 'fun': compute_constraints}
    
    rng = np.random.default_rng(42)
    
    # Multiple restarts to avoid local minima
    for _ in range(15):
        x0 = generate_initial_config(rng)
        try:
            res = minimize(
                compute_objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint_dict,
                options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False}
            )
            
            if np.isfinite(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = res.x[2::3]
                
                # Quick internal validity check
                valid = True
                if np.any(r_opt < 0):
                    valid = False
                elif np.any(r_opt > c_opt[:, 0]) or np.any(r_opt > 1.0 - c_opt[:, 0]) or \
                     np.any(r_opt > c_opt[:, 1]) or np.any(r_opt > 1.0 - c_opt[:, 1]):
                    valid = False
                else:
                    X = c_opt[:, 0][:, None] - c_opt[:, 0][None, :]
                    Y = c_opt[:, 1][:, None] - c_opt[:, 1][None, :]
                    D = np.sqrt(X**2 + Y**2)
                    R_Sum = r_opt[:, None] + r_opt[None, :]
                    i_idx, j_idx = np.triu_indices(n, k=1)
                    if np.any(D[i_idx, j_idx] < R_Sum[i_idx, j_idx] - 1e-9):
                        valid = False
                        
                if valid:
                    s = np.sum(r_opt)
                    if s > best_sum:
                        best_sum = s
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
        except Exception:
            continue
            
    # Post-processing safety: shrink radii slightly if any numerical overlap persists
    if best_centers is not None:
        max_overlap = 0.0
        X = best_centers[:, 0][:, None] - best_centers[:, 0][None, :]
        Y = best_centers[:, 1][:, None] - best_centers[:, 1][None, :]
        D = np.sqrt(X**2 + Y**2)
        R_Sum = best_radii[:, None] + best_radii[None, :]
        i_idx, j_idx = np.triu_indices(n, k=1)
        overlaps = R_Sum[i_idx, j_idx] - D[i_idx, j_idx]
        if np.any(overlaps > 1e-10):
            max_overlap = np.max(overlaps)
            best_radii -= max_overlap / 2.0
            best_radii = np.maximum(best_radii, 1e-6)
        best_sum = np.sum(best_radii)

    # Fallback configuration if optimization fails completely
    if best_centers is None:
        step = 0.2
        centers = []
        for i in range(5):
            for j in range(5):
                if len(centers) < 26:
                    centers.append([0.1 + i * step, 0.1 + j * step])
        centers.append([0.5, 0.5])
        best_centers = np.array(centers)
        best_radii = np.full(26, 0.08)
        best_sum = 26 * 0.08

    return best_centers, best_radii, best_sum
