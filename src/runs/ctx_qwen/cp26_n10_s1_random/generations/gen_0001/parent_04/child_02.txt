from scipy.optimize import minimize

def compute_constraints(vars_flat, n):
    """Computes inequality constraints >= 0 for SLSQP."""
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    boundary_cons = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    pair_cons = []
    for i in range(n):
        xi, yi, ri = x[i], y[i], r[i]
        for j in range(i + 1, n):
            dx = xi - x[j]
            dy = yi - y[j]
            r_sum = ri + r[j]
            pair_cons.append(dx*dx + dy*dy - r_sum*r_sum)
            
    return np.concatenate([boundary_cons, pair_cons])

def objective_function(vars_flat):

    return -np.sum(vars_flat[2::3])

                        centers[i, 1] -= uy * move
