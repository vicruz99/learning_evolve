import numpy as np
from scipy.optimize import minimize

def compute_objective(vars_flat, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def compute_constraints(vars_flat, n):
    """Returns array of constraint values that must be >= 0."""
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    cons = np.empty(4 * n + n * (n - 1) // 2)
    idx = 0
    
    # Boundary constraints: x-r>=0, 1-x-r>=0, y-r>=0, 1-y-r>=0
    for i in range(n):
        cons[idx]   = x[i] - r[i]
        cons[idx+1] = 1.0 - x[i] - r[i]
        cons[idx+2] = y[i] - r[i]
        cons[idx+3] = 1.0 - y[i] - r[i]
        idx += 4
        
    # Pairwise non-overlap: dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist_sq = dx * dx + dy * dy
            r_sum = r[i] + r[j]
            cons[idx] = dist_sq - r_sum * r_sum
            idx += 1
            
    return cons

def generate_hex_init(n, r_init, seed):
    """Generates a hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    y = r_init
    row = 0
    while y + r_init <= 1.0 and len(pts) < n:
        shift = r_init if row % 2 == 1 else 0.0
        x = r_init + shift
        while x + r_init <= 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r_init
        y += np.sqrt(3) * r_init
        row += 1
        
    # Pad with center points if lattice doesn't yield enough
    while len(pts) < n:
        pts.append([0.5 + np.random.uniform(-0.1, 0.1), 
                    0.5 + np.random.uniform(-0.1, 0.1)])
                    
    return np.array(pts[:n])

def run_packing():

    best_sum = -1.0

        'fun': compute_constraints,
        'args': (n,)
    }

    initial_guesses = []

    for r_base in [0.08, 0.09, 0.095, 0.10]:

        pert = grid + np.random.uniform(-0.04, 0.04, grid.shape)
        pert = np.clip(pert, 0.05, 0.95)
        initial_guesses.append(pert)

    # Optimization loop
    for centers_init in initial_guesses:
        x0 = np.empty(3 * n)
        x0[0::3] = centers_init[:, 0]
        x0[1::3] = centers_init[:, 1]
        x0[2::3] = 0.07  # Start with strictly feasible radius
        
        try:
            res = minimize(
                compute_objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint_dict,
                args=(n,),
                options={'maxiter': 2000, 'ftol': 1e-13, 'iprint': -1}

        best_radii = np.maximum(best_radii - shrink, 0.0)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
