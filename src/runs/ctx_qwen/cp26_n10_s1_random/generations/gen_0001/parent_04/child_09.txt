import numpy as np
from scipy.optimize import minimize

def compute_penalty(vars_array, n, mu):
    """
    Computes the objective function: -sum(radii) + penalty for constraint violations.
    """
    centers = vars_array[:2*n].reshape(n, 2)
    radii = vars_array[2*n:]
    
    penalty = 0.0
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # Violations are positive when constraints are broken
    v1 = np.maximum(0, radii - centers[:, 0])
    v2 = np.maximum(0, centers[:, 0] + radii - 1.0)
    v3 = np.maximum(0, radii - centers[:, 1])
    v4 = np.maximum(0, centers[:, 1] + radii - 1.0)
    penalty += np.sum(v1**2 + v2**2 + v3**2 + v4**2)
    
    # Overlap constraints: dist_ij >= r_i + r_j
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlaps = radii_sum - dists
    overlaps = np.maximum(0, overlaps)
    # Multiply by 0.5 because each pair is counted twice in the full matrix
    penalty += 0.5 * np.sum(overlaps**2)
    
    # Objective: maximize sum(r) -> minimize -sum(r) + mu * penalty
    return -np.sum(radii) + mu * penalty

def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    rng = np.random.RandomState(42)
    configs = []
    
    # 1. Hexagonal lattice initialization
    r_init = 0.1
    pts = []
    row = 0
    y = r_init
    while y < 1.0:
        shift = (row % 2) * r_init
        x = r_init + shift
        while x < 1.0:
            pts.append([x, y])
            x += 2 * r_init
        y += r_init * np.sqrt(3)
        row += 1
    pts = np.array(pts[:n])
    pts = (pts - pts.min(axis=0)) / (pts.max(axis=0) - pts.min(axis=0))
    pts = pts * 0.8 + 0.1
    configs.append((pts, np.full(n, 0.05)))
    
    # 2. Grid with noise
    grid_x = np.linspace(0.12, 0.88, 6)
    grid_y = np.linspace(0.12, 0.88, 5)
    pts2 = np.array([[x, y] for y in grid_y for x in grid_x][:n])
    pts2 += rng.uniform(-0.03, 0.03, pts2.shape)
    pts2 = np.clip(pts2, 0.05, 0.95)
    configs.append((pts2, np.full(n, 0.05)))
    
    # 3. Multiple random starts to escape poor local minima
    for _ in range(5):
        pts3 = rng.uniform(0.15, 0.85, (n, 2))
        configs.append((pts3, np.full(n, 0.04)))
        
    # Optimization loop over all configurations
    for centers_init, radii_init in configs:
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        bounds = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
        
        x_curr = x0.copy()
        # Progressive penalty weight increases to smoothly enforce constraints
        for mu in [200, 800, 2000, 4000]:
            res = minimize(
                lambda v, m=mu: compute_penalty(v, n, m),
                x_curr,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1500, 'ftol': 1e-12, 'gtol': 1e-10}
            )
            x_curr = res.x
            
        c_opt = x_curr[:2*n].reshape(n, 2)
        r_opt = x_curr[2*n:]
        s = np.sum(r_opt)
        
        # Strict validity check matching the grader's tolerance
        valid = True
        for i in range(n):
            x, y = c_opt[i]
            r = r_opt[i]
            if x < r - 1e-12 or x > 1 - r + 1e-12 or y < r - 1e-12 or y > 1 - r + 1e-12:
                valid = False; break
        if valid:
            for i in range(n):
                for j in range(i+1, n):
                    d = np.sqrt(np.sum((c_opt[i]-c_opt[j])**2))
                    if d < r_opt[i] + r_opt[j] - 1e-12:
                        valid = False; break
                if not valid: break
                
        if valid and s > best_sum:
            best_sum = s
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    # Fallback repair step to guarantee strict validity if optimization leaves tiny overlaps
    if best_centers is None:
        best_centers = x_curr[:2*n].reshape(n, 2)
        best_radii = x_curr[2*n:]
        
        for _ in range(100):
            changed = False
            for i in range(n):
                x, y = best_centers[i]
                r = best_radii[i]
                if r > x: best_radii[i] = x; changed = True
                if r > 1-x: best_radii[i] = 1-x; changed = True
                if r > y: best_radii[i] = y; changed = True
                if r > 1-y: best_radii[i] = 1-y; changed = True
            for i in range(n):
                for j in range(i+1, n):
                    d = np.sqrt(np.sum((best_centers[i]-best_centers[j])**2))
                    r_sum = best_radii[i] + best_radii[j]
                    if r_sum > d:
                        excess = r_sum - d
                        best_radii[i] -= excess/2
                        best_radii[j] -= excess/2
                        changed = True
            if not changed:
                break
        best_radii = np.maximum(best_radii, 1e-9)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
