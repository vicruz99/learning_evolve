import numpy as np
from scipy.optimize import minimize

def obj_func(x_vars, n):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)"""
    return -np.sum(x_vars[2::3])

def constr_func(x_vars, n):
    """
    Computes all inequality constraints (must be >= 0):
    1. Boundary constraints: circle inside [0,1]x[0,1]
    2. Non-overlap constraints: squared distance >= squared sum of radii
    """
    x = x_vars[0::3]
    y = x_vars[1::3]
    r = x_vars[2::3]
    
    # Boundary constraints
    b_cons = np.concatenate([
        x - r,          # x - r >= 0
        1.0 - x - r,    # 1 - x - r >= 0
        y - r,          # y - r >= 0
        1.0 - y - r     # 1 - y - r >= 0
    ])
    
    # Pairwise non-overlap constraints
    # Using broadcasting for efficiency
    cx = x[:, None] - x[None, :]
    cy = y[:, None] - y[None, :]
    r_s = r[:, None] + r[None, :]
    
    d2 = cx**2 + cy**2
    rs2 = r_s**2
    
    # Extract upper triangle to avoid duplicates and self-comparisons
    o_cons = np.triu(d2 - rs2, k=1).flatten()
    
    return np.concatenate([b_cons, o_cons])

def run_packing():
    n = 26
    
    # Variable bounds: x,y in [0,1], r in [tiny, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice patterns with different base radii
    for r_hex in [0.08, 0.09, 0.10, 0.11]:
        pts_hex = []
        y_h = r_hex
        row_h = 0
        while len(pts_hex) < 30:
            shift = r_hex if row_h % 2 == 1 else 0.0
            x_h = r_hex + shift
            while x_h + r_hex <= 1.0 and len(pts_hex) < 30:
                pts_hex.append([x_h, y_h])
                x_h += 2 * r_hex
            y_h += r_hex * np.sqrt(3)
            row_h += 1
        if len(pts_hex) >= n:
            inits.append(np.array(pts_hex[:n]))
            
    # 2. Uniform grid pattern
    pts_grid = []
    step = 1.0 / 6.0
    for i in range(6):
        for j in range(6):
            if len(pts_grid) < n:
                pts_grid.append([step * (i + 0.5), step * (j + 0.5)])
    if len(pts_grid) >= n:
        inits.append(np.array(pts_grid))
        
    best_res = None
    best_sum = -np.inf
    
    np.random.seed(42)
    for cfg in inits:
        # Multiple perturbations per configuration to escape local minima
        for _ in range(5):
            cfg_p = cfg + np.random.uniform(-0.03, 0.03, cfg.shape)
            # Clip to ensure strict feasibility for boundaries with initial radius
            cfg_p = np.clip(cfg_p, 0.12, 0.88)
            
            # Start with a small feasible radius to let optimizer expand
            r_init = np.full(n, 0.05)
            
            x0 = np.empty(3 * n)
            x0[0::3] = cfg_p[:, 0]
            x0[1::3] = cfg_p[:, 1]
            x0[2::3] = r_init
            
            try:
                res = minimize(
                    obj_func, x0, args=(n,),
                    method='SLSQP', bounds=bounds,
                    constraints={'type': 'ineq', 'fun': constr_func, 'args': (n,)},
                    options={'maxiter': 3000, 'ftol': 1e-9, 'disp': False}
                )
                if res.success:
                    s = np.sum(res.x[2::3])
                    if s > best_sum:
                        best_sum = s
                        best_res = res
            except Exception:
                pass
                
    # Fallback if optimization fails (highly unlikely with given starts)
    if best_res is None:
        centers = inits[0]
        radii = np.full(n, 0.05)
        return centers, radii, np.sum(radii)
        
    c_opt = np.column_stack((best_res.x[0::3], best_res.x[1::3]))
    r_opt = best_res.x[2::3]
    
    return c_opt, r_opt, float(best_sum)
