import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    Maximize sum(r_i) s.t. r_i + r_j <= dist(i,j), r_i <= boundary_dist(i), r_i >= 0
    """
    n = len(centers)
    c_obj = -np.ones(n)  # Minimize negative sum => maximize sum
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        bound = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(bound)
        
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, 0.5) for _ in range(n)]
    
    try:
        # 'highs' is the default and fastest method in modern scipy
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    # Fallback to safe small radii if LP unexpectedly fails
    return np.full(n, 0.01)


def run_packing() -> tuple:
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Objective for SLSQP: minimize negative sum of radii
    def objective(v):
        return -np.sum(v[2::3])
        
    # Constraints for SLSQP: return array of values >= 0
    def constraints(v):
        c = []
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        
        # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
        for i in range(n):
            c.extend([x[i] - r[i], 1.0 - x[i] - r[i], y[i] - r[i], 1.0 - y[i] - r[i]])
            
        # Non-overlap constraints: ||x_i - x_j||^2 >= (r_i + r_j)^2
        for i in range(n):
            for j in range(i + 1, n):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                c.append(dx*dx + dy*dy - (r[i] + r[j])**2)
        return np.array(c)

    bounds_vars = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]
    
    # --- Generate Diverse Initial Configurations ---
    inits = []
    rng = np.random.default_rng(42)
    
    # 1. Hexagonal lattices with varying densities
    for r_est in [0.085, 0.09, 0.095, 0.10, 0.105]:
        pts = []
        y = r_est
        row = 0
        while y < 1.0 - r_est:
            x = r_est
            offset = r_est if row % 2 == 1 else 0.0
            x += offset
            while x < 1.0 - r_est:
                pts.append([x, y])
                x += 2 * r_est
            y += r_est * np.sqrt(3)
            row += 1
        while len(pts) < n:
            pts.append([0.5, 0.5])
        inits.append(np.array(pts[:n]))
        
    # 2. Square grid base
    grid_x = np.linspace(0.1, 0.9, 5)
    grid_y = np.linspace(0.1, 0.9, 5)
    pts_grid = []
    for y in grid_y:
        for x in grid_x:
            pts_grid.append([x, y])
    pts_grid.extend([[0.5, 0.5]])
    inits.append(np.array(pts_grid[:n]))
    
    # 3. Perturbed versions
    for _ in range(4):
        base = inits[0].copy()
        base += rng.uniform(-0.02, 0.02, base.shape)
        base = np.clip(base, 0.05, 0.95)
        inits.append(base)

    # --- Optimization Loop ---
    for centers_init in inits:
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers_init[i, 0]
            x0[3*i+1] = centers_init[i, 1]
            x0[3*i+2] = 0.08  # Start with small valid radii
            
        try:
            res = minimize(
                objective, x0, method='SLSQP', bounds=bounds_vars,
                constraints={'type': 'ineq', 'fun': constraints},
                options={'maxiter': 1500, 'ftol': 1e-10, 'iprint': -1}
            )
            
            if res.success or res.fun < 0:
                opt_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                
                # Phase 2: Exact LP radius maximization for fixed centers
                opt_radii = solve_radii_lp(opt_centers)
                current_sum = np.sum(opt_radii)
                
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = opt_centers.copy()
                    best_radii = opt_radii.copy()
        except Exception:
            continue
            
    # Safety fallback
    if best_centers is None or best_sum < 1.0:
        best_centers = np.random.uniform(0.2, 0.8, (n, 2))
        best_radii = np.full(n, 0.05)
        best_sum = np.sum(best_radii)
        
    # Minor numerical safety shrink to guarantee validator tolerance
    best_radii *= 0.9999999
    
    return best_centers, best_radii, best_sum
