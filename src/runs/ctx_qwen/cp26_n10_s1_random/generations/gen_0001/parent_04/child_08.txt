import numpy as np
from scipy.optimize import minimize

def compute_objective(z, n, mu):
    """
    Computes the objective function: -sum(radii) + penalty for constraint violations.
    z: flattened array [x0, y0, r0, x1, y1, r1, ...]
    n: number of circles
    mu: penalty weight
    """
    x = z[0::3]
    y = z[1::3]
    r = z[2::3]
    
    # Objective: maximize sum of radii -> minimize negative sum
    cost = -np.sum(r)
    
    # Boundary constraint penalties: circles must be inside [0, 1]x[0, 1]
    # Violations occur when r > x, r > 1-x, r > y, or r > 1-y
    v_boundary = np.zeros(n)
    v_boundary += np.maximum(0, r - x)
    v_boundary += np.maximum(0, r - (1.0 - x))
    v_boundary += np.maximum(0, r - y)
    v_boundary += np.maximum(0, r - (1.0 - y))
    cost += mu * np.sum(v_boundary**2)
    
    # Pairwise non-overlap constraint penalties: dist(i,j) >= r_i + r_j
    # Violation occurs when r_i + r_j > dist(i,j)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    overlap = r_sum - dist
    # Only consider upper triangle to avoid double counting and diagonal self-interaction
    overlap[np.tril_indices(n)] = 0.0
    
    v_overlap = np.maximum(0, overlap)
    cost += mu * np.sum(v_overlap**2)
    
    return cost

def run_packing():
    n = 26
    np.random.seed(42)
    
    # Variable bounds: x, y in [0, 1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    best_z = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations to escape poor local minima
    initial_centers = []
    
    # 1. Hexagonal lattice pattern (densest packing structure)
    hex_pts = []
    for i in range(10):
        for j in range(10):
            x = j * 0.2 + (i % 2) * 0.1
            y = i * 0.1732 # approx sqrt(3)/2 * 0.2
            if x < 1.0 and y < 1.0:
                hex_pts.append([x, y])
    initial_centers.append(np.array(hex_pts[:n]))
    
    # 2. Regular grid pattern
    grid_x = np.linspace(0.12, 0.88, 6)
    grid_y = np.linspace(0.12, 0.88, 5)
    grid_pts = []
    for gy in grid_y:
        for gx in grid_x:
            if len(grid_pts) < n:
                grid_pts.append([gx, gy])
    initial_centers.append(np.array(grid_pts))
    
    # 3. Perturbed hexagonal (breaks symmetry, allows new local optima)
    init_pert = hex_pts[:n] + np.random.uniform(-0.04, 0.04, size=(n, 2))
    initial_centers.append(np.clip(init_pert, 0.05, 0.95))
    
    # 4. Randomized dense start
    initial_centers.append(np.random.uniform(0.15, 0.85, size=(n, 2)))

    # Penalty parameter continuation schedule: gradually enforce constraints
    mu_schedule = [100, 1000, 10000, 50000, 100000, 200000]
    
    for pts in initial_centers:
        # Ensure initial points are strictly inside the square
        pts = np.clip(pts, 1e-3, 1.0 - 1e-3)
        
        z0 = np.zeros(3 * n)
        z0[0::3] = pts[:, 0]
        z0[1::3] = pts[:, 1]
        z0[2::3] = 0.08  # Start with reasonable radii that allow movement
        
        z_curr = z0.copy()
        
        # Iterate through increasing penalty weights to drive solution to feasibility
        for mu in mu_schedule:
            res = minimize(
                compute_objective,
                z_curr,
                args=(n, mu),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False}
            )
            z_curr = res.x
            
        # Extract radii and compute sum
        r_final = z_curr[2::3]
        current_sum = np.sum(r_final)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_z = z_curr.copy()
            
    # Extract final centers and radii
    centers = np.column_stack((best_z[0::3], best_z[1::3]))
    radii = best_z[2::3]
    
    # Final validation and strict feasibility enforcement
    # Calculate maximum violation to ensure we pass the 1e-12 checker tolerance
    x, y, r = centers[:, 0], centers[:, 1], radii
    max_violation = 0.0
    
    # Boundary checks
    for i in range(n):
        max_violation = max(max_violation, 
                           r[i] - x[i], r[i] - (1.0 - x[i]), 
                           r[i] - y[i], r[i] - (1.0 - y[i]))

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(x[i] - x[j], y[i] - y[j])
            max_violation = max(max_violation, r[i] + r[j] - dist)
            
    # If any violation exceeds tolerance, shrink radii uniformly to guarantee validity
    if max_violation > 1e-9:
        shrink_amount = max_violation + 1e-6
        radii -= shrink_amount
        radii = np.maximum(radii, 1e-6)
        
    final_sum = np.sum(radii)
    return centers, radii, final_sum
