import numpy as np
from scipy.optimize import minimize

# Constants for the optimization
N_CIRCLES = 26
TRIU_IDX = np.triu_indices(N_CIRCLES, k=1)

def evaluate_packing(vars_arr):
    """Decodes optimization variables into centers and radii."""
    n = N_CIRCLES
    r = vars_arr[0::3]
    u = vars_arr[1::3]
    v = vars_arr[2::3]
    x = r + (1 - 2 * r) * u
    y = r + (1 - 2 * r) * v
    centers = np.column_stack((x, y))
    return centers, r, np.sum(r)

def constraints(vars_arr):
    """Computes pairwise non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0."""
    r = vars_arr[0::3]
    u = vars_arr[1::3]
    v = vars_arr[2::3]
    x = r + (1 - 2 * r) * u
    y = r + (1 - 2 * r) * v
    
    dx = x[TRIU_IDX[0]] - x[TRIU_IDX[1]]
    dy = y[TRIU_IDX[0]] - y[TRIU_IDX[1]]
    dist_sq = dx**2 + dy**2
    r_sum = r[TRIU_IDX[0]] + r[TRIU_IDX[1]]
    return dist_sq - r_sum**2

def objective(vars_arr):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_arr[0::3])
