import numpy as np
from scipy.optimize import minimize

N = 26

def penalty_objective(vars_flat):
    n = N
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    obj = -np.sum(r)
    w = 5000.0
    
    # Boundary penalties
    viol_l = np.maximum(0, r - x)
    obj += w * np.sum(viol_l**2)
    viol_r = np.maximum(0, x + r - 1.0)
    obj += w * np.sum(viol_r**2)
    viol_b = np.maximum(0, r - y)
    obj += w * np.sum(viol_b**2)
    viol_t = np.maximum(0, y + r - 1.0)
    obj += w * np.sum(viol_t**2)
    
    # Overlap penalties
    x_diff = x[:, None] - x[None, :]
    y_diff = y[:, None] - y[None, :]
    dists = np.sqrt(x_diff**2 + y_diff**2)
    r_sum = r[:, None] + r[None, :]
    
    rows, cols = np.triu_indices(n, k=1)
    dists_trip = dists[rows, cols]
    r_sum_trip = r_sum[rows, cols]
    
    viol_overlap = np.maximum(0, r_sum_trip - dists_trip)
    obj += w * np.sum(viol_overlap**2)
    
    return obj

def constraint_func(vars_flat):
    n = N
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    c = []
    
    # Boundary constraints
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Overlap constraints
    x_diff = x[:, None] - x[None, :]
    y_diff = y[:, None] - y[None, :]
    dists = np.sqrt(x_diff**2 + y_diff**2)
    r_sum = r[:, None] + r[None, :]
    
    rows, cols = np.triu_indices(n, k=1)
    c.append(dists[rows, cols] - r_sum[rows, cols])
    
    return np.concatenate(c)

def run_packing():
    n = N
    best_sum = -1.0
    best_centers = None
    best_radii = None

    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    
    # Generate initial configurations
    configs = []
    r_init = 0.09
    pts = []
    y = r_init
    counts = [5, 6, 5, 6, 4]
    for i, count in enumerate(counts):
        shift = r_init if i % 2 == 1 else 0.0
        row_w = (count - 1) * 2 * r_init
        start_x = (1.0 - row_w) / 2.0 + shift
        for j in range(count):
            pts.append([start_x + j * 2 * r_init, y])
        y += r_init * np.sqrt(3)
    pts = np.array(pts[:n])
    
    v0 = np.zeros(3 * n)
    v0[0::3] = pts[:, 0]
    v0[1::3] = pts[:, 1]
    v0[2::3] = r_init
    configs.append(v0)
    
    # Perturbed hexagonal starts
    for _ in range(5):
        v = v0.copy()
        v[0::3] += np.random.uniform(-0.03, 0.03, n)
        v[1::3] += np.random.uniform(-0.03, 0.03, n)
        v[0::3] = np.clip(v[0::3], 0.05, 0.95)
        v[1::3] = np.clip(v[1::3], 0.05, 0.95)
        configs.append(v)
        
    # Grid start
    grid_v = np.zeros(3 * n)
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx < n:
                grid_v[3*idx] = 0.1 + j * 0.18
                grid_v[3*idx+1] = 0.1 + i * 0.166
                grid_v[3*idx+2] = 0.08
                idx += 1
    configs.append(grid_v)

    mask = np.triu(np.ones((n, n), dtype=bool), k=1)

    # Stage 1: L-BFGS-B with penalty for fast convergence
    for cfg in configs:
        try:
            res = minimize(penalty_objective, cfg, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 5000, 'ftol': 1e-12})
            x_opt = res.x[0::3]
            y_opt = res.x[1::3]
            r_opt = res.x[2::3]
            
            valid = True
            if np.any(r_opt < -1e-9): valid = False
            if np.any(x_opt - r_opt < -1e-9) or np.any(x_opt + r_opt > 1 + 1e-9): valid = False
            if np.any(y_opt - r_opt < -1e-9) or np.any(y_opt + r_opt > 1 + 1e-9): valid = False
            
            dists = np.linalg.norm(np.column_stack((x_opt, y_opt))[:, None, :] - np.column_stack((x_opt, y_opt))[None, :, :], axis=2)
            r_sums = r_opt[:, None] + r_opt[None, :]
            if np.any(dists[mask] < r_sums[mask] - 1e-9):
                valid = False
                
            if valid:
                s = np.sum(r_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = np.column_stack((x_opt, y_opt))
                    best_radii = r_opt
        except Exception:
            continue

    # Stage 2: Refine best with SLSQP (exact constraints) to hit boundary precisely
    if best_centers is not None:
        v_refine = np.zeros(3 * n)
        v_refine[0::3] = best_centers[:, 0]
        v_refine[1::3] = best_centers[:, 1]
        v_refine[2::3] = best_radii
        
        cons = {'type': 'ineq', 'fun': constraint_func}
        try:
            res2 = minimize(lambda v: -np.sum(v[2::3]), v_refine, method='SLSQP', bounds=bounds, constraints=cons,
                            options={'maxiter': 3000, 'ftol': 1e-11})

            y2 = res2.x[1::3]
            r2 = res2.x[2::3]

            if np.any(y2 - r2 < -1e-9) or np.any(y2 + r2 > 1 + 1e-9): valid2 = False
            
            dists2 = np.linalg.norm(np.column_stack((x2, y2))[:, None, :] - np.column_stack((x2, y2))[None, :, :], axis=2)
            r_sums2 = r2[:, None] + r2[None, :]
            if np.any(dists2[mask] < r_sums2[mask] - 1e-9): valid2 = False
            
            if valid2:
                s2 = np.sum(r2)
                if s2 > best_sum:
                    best_sum = s2
                    best_centers = np.column_stack((x2, y2))
                    best_radii = r2
        except Exception:
            pass

    # Fallback geometric construction
    if best_centers is None:
        best_centers = pts
        min_d = 1.0
        for i in range(n):
            min_d = min(min_d, best_centers[i,0], 1-best_centers[i,0], best_centers[i,1], 1-best_centers[i,1])
            for j in range(i+1, n):
                d = np.linalg.norm(best_centers[i] - best_centers[j])
                if d < min_d: min_d = d
        best_radii = np.full(n, min_d / 2.0)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
