import numpy as np
from scipy.optimize import minimize

def get_clearances(centers, n):
    """Compute clearances to boundaries and half-pairwise distances."""
    # Distance to each boundary
    b_left = centers[:, 0]
    b_right = 1.0 - centers[:, 0]
    b_bottom = centers[:, 1]
    b_top = 1.0 - centers[:, 1]
    b = np.minimum(np.minimum(b_left, b_right), np.minimum(b_bottom, b_top))
    
    # Half distance to all other circles (upper triangle only)
    i_idx, j_idx = np.triu_indices(n, k=1)
    diffs = centers[i_idx] - centers[j_idx]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    p = dists / 2.0
    
    return np.concatenate([b, p])

def compute_smooth_min(centers, n, k):
    """Smooth approximation of the minimum clearance using log-sum-exp."""
    c = get_clearances(centers, n)
    max_c = np.max(c)
    # Stable computation: min(c) approx -1/k * log(sum(exp(-k*(c - max_c)))) - max_c
    exp_vals = np.exp(-k * (c - max_c))
    return -np.log(np.sum(exp_vals)) / k - max_c

def objective(vars_flat, n, k):
    """Objective to minimize: negative smooth min clearance."""
    return -compute_smooth_min(vars_flat.reshape(n, 2), n, k)

def run_packing():
    n = 26
    k = 60.0  # Sharpness of the smooth minimum approximation
    
    bounds = [(0.0, 1.0)] * (2 * n)
    
    best_val = np.inf
    best_centers = None
    
    starts = []
    
    # 1. Hexagonal lattice initialization (optimal for density)
    r0 = 0.1
    dy = np.sqrt(3) * r0
    dx = 2 * r0
    pts = []
    y = r0
    row = 0
    while len(pts) < n:
        shift = (row % 2) * r0
        x = r0 + shift
        while x + r0 <= 1.0 and len(pts) < n:
            pts.append([x, y])
            x += dx
        y += dy
        row += 1
    starts.append(np.array(pts[:n]))
    
    # 2. Randomly perturbed hexagonal starts to escape local minima
    rng = np.random.default_rng(42)
    for _ in range(6):
        noise = rng.uniform(-0.03, 0.03, (n, 2))
        starts.append(np.clip(starts[0] + noise, 0.05, 0.95))
        
    # 3. Dense uniform grid fallback
    grid_pts = []
    for i in range(6):
        for j in range(5):
            if len(grid_pts) < n:
                grid_pts.append([0.083 + j*0.183, 0.083 + i*0.183])
    starts.append(np.array(grid_pts[:n]))

    # Optimize from each starting configuration
    for pt in starts:
        try:
            res = minimize(objective, pt.flatten(), args=(n, k), method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 10000, 'ftol': 1e-14, 'gtol': 1e-12})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(n, 2)
        except Exception:
            continue
            
    # Fallback in case all optimizations fail
    if best_centers is None:
        best_centers = starts[0]
        
    # Compute exact maximum feasible radius for the best layout
    c = best_centers
    clearances = get_clearances(c, n)
    r_max = np.min(clearances)
    
    # Apply a tiny safety margin to strictly satisfy the 1e-12 tolerance in validation
    r_max = r_max * 0.99999
    radii = np.full(n, r_max)
    sum_r = float(np.sum(radii))
    
    return c, radii, sum_r
