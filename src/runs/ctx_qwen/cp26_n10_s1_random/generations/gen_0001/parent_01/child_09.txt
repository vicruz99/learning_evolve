import numpy as np
from scipy.optimize import minimize

def objective(vars_arr):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2::3])

def constraint_func(vars_arr, rows, cols):
    """Computes boundary and non-overlap constraints."""
    n = len(vars_arr) // 3
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    
    c = []
    # Boundary constraints: circles must stay inside [0,1]x[0,1]
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Overlap constraints: dist(i,j) >= r_i + r_j + margin
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    # Margin ensures strict feasibility well beyond the 1e-12 validation tolerance
    c.append(dist[rows, cols] - r_sum[rows, cols] - 1e-6)
    
    return np.concatenate(c)

def make_config(pts, n, r_init):
    """Formats initial positions and radii into the optimization vector."""
    v = np.zeros(3 * n)
    v[0::3] = pts[:, 0]
    v[1::3] = pts[:, 1]
    v[2::3] = r_init * np.ones(n)
    return v

def run_packing():
    n = 26
    # Precompute indices for upper triangle to speed up constraint evaluation
    rows, cols = np.triu_indices(n, k=1)
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    
    configs = []
    np.random.seed(42)
    
    # 1. Hexagonal lattice initialization (dense packing structure)
    hex_pts = []
    y = 0.095
    row = 0
    while len(hex_pts) < n:
        shift = 0.095 if row % 2 == 1 else 0.0
        x = 0.095 + shift
        while x <= 0.905 and len(hex_pts) < n:
            hex_pts.append([x, y])
            x += 0.19
        y += 0.095 * np.sqrt(3)
        row += 1
    hex_pts = np.array(hex_pts[:n])
    configs.append(make_config(hex_pts, n, 0.08))
    
    # 2. Perturbed hexagonal lattices to escape local minima
    for _ in range(12):
        p = hex_pts + np.random.uniform(-0.03, 0.03, hex_pts.shape)
        p = np.clip(p, 0.05, 0.95)
        configs.append(make_config(p, n, 0.075))
        
    # 3. Regular grid initialization
    grid_pts = []
    for i in range(5):
        for j in range(5):
            grid_pts.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    grid_pts.append([0.5, 0.5])
    configs.append(make_config(np.array(grid_pts), n, 0.06))
    
    best_sum = -1.0
    best_c, best_r = None, None
    
    for cfg in configs:
        try:
            res = minimize(
                objective, 
                cfg, 
                method='SLSQP', 
                bounds=bounds, 
                constraints={'type': 'ineq', 'fun': constraint_func, 'args': (rows, cols)},
                options={'maxiter': 3000, 'ftol': 1e-13}
            )
            if np.isfinite(res.fun):
                c = np.column_stack((res.x[0::3], res.x[1::3]))
                r = res.x[2::3]
                s = np.sum(r)
                if s > best_sum:
                    best_sum = s
                    best_c, best_r = c, r
        except Exception:
            continue
            
    # Fallback if optimization didn't converge to a valid configuration
    if best_c is None:
        best_c = hex_pts
        best_r = np.full(n, 0.08)
        best_sum = np.sum(best_r)
        
    return best_c, best_r, best_sum
