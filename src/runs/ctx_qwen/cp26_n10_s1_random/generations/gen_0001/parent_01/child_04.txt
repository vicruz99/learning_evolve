import numpy as np
from scipy.optimize import minimize

def compute_objective_and_penalty(vars_arr, n, weight):
    """
    Computes the objective (-sum of radii) plus a weighted penalty for 
    boundary violations and circle overlaps.
    """
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    
    # Objective: maximize sum of radii => minimize negative sum
    obj = -np.sum(r)
    
    # Boundary penalties: circle must be inside [0,1]x[0,1]
    # Violation occurs if r > x, r > 1-x, r > y, or r > 1-y
    b_pen = np.sum(np.maximum(0, r - x)**2)
    b_pen += np.sum(np.maximum(0, r - (1.0 - x))**2)
    b_pen += np.sum(np.maximum(0, r - y)**2)
    b_pen += np.sum(np.maximum(0, r - (1.0 - y))**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)  # Ignore self-distances
    
    r_sum = r[:, None] + r[None, :]
    overlap = np.maximum(0, r_sum - dist)
    
    # Only count each pair once (upper triangle)
    triu_idx = np.triu_indices(n, k=1)
    o_pen = np.sum(overlap[triu_idx]**2)
    
    return obj + weight * (b_pen + o_pen)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    np.random.seed(42)
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Prepare diverse initial configurations
    init_configs = []
    
    # 1. Hexagonal lattice initialization (best for dense packing)
    pts = []
    r_init = 0.085
    y = r_init
    row = 0
    while len(pts) < n and y + r_init < 1.0:
        shift = r_init if row % 2 == 1 else 0.0
        x = r_init + shift
        while x + r_init < 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r_init
        y += np.sqrt(3) * r_init
        row += 1
    while len(pts) < n:
        pts.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
    init_configs.append(np.array(pts[:n]))
    
    # 2. Perturbed grid initialization
    pts2 = []
    for i in range(1, 7):
        for j in range(1, 7):
            if len(pts2) >= n: break
            pts2.append([i/7.0, j/7.0])
    pts2 = np.array(pts2[:n])
    pts2 += np.random.uniform(-0.03, 0.03, pts2.shape)
    init_configs.append(pts2)
    
    # 3. Random initialization
    init_configs.append(np.random.uniform(0.1, 0.9, size=(n, 2)))
    
    # Bounds for [x, y, r]
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * n
    
    # Optimization settings
    penalty_weight = 5000.0
    
    for init_centers in init_configs:
        for trial in range(4):
            centers_curr = init_centers.copy()
            if trial > 0:
                centers_curr += np.random.uniform(-0.02, 0.02, centers_curr.shape)
                centers_curr = np.clip(centers_curr, 0.05, 0.95)
                
            # Flatten to [x0, y0, r0, x1, y1, r1, ...]
            x0 = np.zeros(3 * n)
            x0[0::3] = centers_curr[:, 0]
            x0[1::3] = centers_curr[:, 1]
            x0[2::3] = 0.08  # Start with a feasible radius
            
            res = minimize(
                compute_objective_and_penalty,
                x0,
                args=(n, penalty_weight),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 10000, 'ftol': 1e-13}
            )
            
            x_opt = res.x[0::3]
            y_opt = res.x[1::3]
            r_opt = res.x[2::3]
            
            # Repair step: ensure strict validity by scaling down radii if needed
            k = 1.0
            for i in range(n):
                # Boundary constraints
                k = min(k, x_opt[i]/r_opt[i], (1.0-x_opt[i])/r_opt[i],
                        y_opt[i]/r_opt[i], (1.0-y_opt[i])/r_opt[i])
                # Pairwise constraints
                for j in range(i+1, n):
                    d = np.sqrt((x_opt[i]-x_opt[j])**2 + (y_opt[i]-y_opt[j])**2)
                    if r_opt[i] + r_opt[j] > 1e-9:
                        k = min(k, d / (r_opt[i] + r_opt[j]))
                        
            # Apply scaling with tiny safety margin for numerical tolerance
            r_final = r_opt * k * 0.99999
            current_sum = np.sum(r_final)
            
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = np.column_stack((x_opt, y_opt))
                best_radii = r_final
                
    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = np.random.uniform(0.1, 0.9, size=(n, 2))
        best_radii = np.full(n, 0.05)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
