import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

def build_lp_matrix(n):
    """Precomputes the constraint matrix A and indices for the LP."""
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A = np.zeros((m, n))
    A[np.arange(m), idx_i] = 1.0
    A[np.arange(m), idx_j] = 1.0
    return A, idx_i, idx_j

def solve_lp(centers, A, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    wall = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]), 
                      np.minimum(centers[:,1], 1.0 - centers[:,1]))
    bounds = [(0.0, max(w, 1e-12)) for w in wall]
    
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    diff = centers[idx_i] - centers[idx_j]
    b_ub = np.sqrt(np.sum(diff**2, axis=1))
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            r = np.maximum(res.x, 0.0)
            return r, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def get_sum_radii(centers, A, idx_i, idx_j):
    """Returns the maximum sum of radii for given centers."""
    _, s = solve_lp(centers, A, idx_i, idx_j)
    return s

def neg_sum_radii(v, n, A, idx_i, idx_j):
    """Objective for center optimization: minimize negative sum of radii."""
    c = v.reshape(n, 2)
    # Keep centers strictly inside to avoid degenerate LP bounds
    c = np.clip(c, 1e-4, 1.0 - 1e-4)
    return -get_sum_radii(c, A, idx_i, idx_j)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    A, idx_i, idx_j = build_lp_matrix(n)
    rng = np.random.default_rng(42)
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], 
        [4,6,6,6,4], [7,5,5,5,4], [5,7,4,5,5], [6,4,6,5,5],
        [5,5,5,5,6], [6,5,5,5,5]
    ]
    
    for pat in patterns:
        if sum(pat) < n: continue
        pts = []
        y = 0.10
        for i, cnt in enumerate(pat):
            shift = 0.10 if i % 2 == 1 else 0.0
            x = 0.10 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 0.20
            y += 0.10 * np.sqrt(3)
        c = np.array(pts[:n])
        configs.append(c)
        
        # Add controlled perturbations to break symmetry
        for _ in range(3):
            cp = c + rng.uniform(-0.03, 0.03, c.shape)
            configs.append(np.clip(cp, 0.05, 0.95))
            
    # Add purely random starts
    for _ in range(8):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Phase 1: Nelder-Mead optimization on centers directly targeting sum of radii
    for cfg in configs:
        try:
            res = minimize(neg_sum_radii, cfg.flatten(), args=(n, A, idx_i, idx_j), 
                           method='Nelder-Mead', options={'maxiter': 1200, 'xatol': 1e-9, 'fatol': 1e-9})
            c_opt = res.x.reshape(n, 2)
            c_opt = np.clip(c_opt, 1e-4, 1.0 - 1e-4)
            r_lp, s_lp = solve_lp(c_opt, A, idx_i, idx_j)
            if s_lp > best_sum:
                best_sum = s_lp
                best_c = c_opt.copy()
                best_r = r_lp.copy()
        except Exception:
            pass
            
    # Phase 2: Multi-stage Hill Climbing with Adaptive Steps
    if best_c is not None:
        curr_c = best_c.copy()
        curr_r = best_r.copy()
        curr_s = best_sum

                r_new, s_new = solve_lp(curr_c, A, idx_i, idx_j)

                    curr_r = r_new.copy()

                            c_pol = res_loc.x.reshape(n, 2)

                            r_pol, s_pol = solve_lp(c_pol, A, idx_i, idx_j)

                                curr_r = r_pol.copy()

                        except Exception:

                else:

        best_r = curr_r
        best_sum = curr_s

    if best_c is None:
        best_c = configs[0]
        best_r = np.full(n, 0.085)
        best_sum = np.sum(best_r)
        
    # Phase 3: Strict Numerical Safety Scaling
    # Ensures all constraints are satisfied within the validator's 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0 - x)/r, y/r, (1.0 - y)/r)
            
    diff = best_c[:, None, :] - best_c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_pair = best_r[:, None] + best_r[None, :]
    min_ratio = np.min(dists[idx_i, idx_j] / np.maximum(r_pair[idx_i, idx_j], 1e-12))
    scale = min(scale, min_ratio)
    
    # Apply scale with a tiny margin to avoid boundary floating-point issues
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
