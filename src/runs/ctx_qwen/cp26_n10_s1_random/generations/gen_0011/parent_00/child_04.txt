import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

# Precompute pairwise indices for 26 circles
N_CIRCLES = 26
IDX_I, IDX_J = np.triu_indices(N_CIRCLES, k=1)
PAIR_IDX = np.arange(len(IDX_I))

def build_lp_matrix():
    """Constructs the constraint matrix A for the LP: r_i + r_j <= dist_ij."""
    A = np.zeros((len(IDX_I), N_CIRCLES))
    A[PAIR_IDX, IDX_I] = 1.0
    A[PAIR_IDX, IDX_J] = 1.0
    return A

A_LP = build_lp_matrix()

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(w, 1e-12)) for w in wall]
    
    diff = centers[IDX_I] - centers[IDX_J]
    b_ub = np.sqrt(np.sum(diff**2, axis=1))
    
    try:
        res = linprog(-np.ones(N_CIRCLES), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            r = np.maximum(res.x, 0.0)
            return r, -res.fun
    except Exception:
        pass
    return np.full(N_CIRCLES, 1e-6), 0.0

def slsqp_objective(v):
    """Objective for SLSQP: minimize negative common radius R."""
    return -v[-1]

def slsqp_constraints(v):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    cx = v[:N_CIRCLES]
    cy = v[N_CIRCLES:2*N_CIRCLES]
    R = v[-1]
    
    # Boundary constraints
    con = np.concatenate([
        cx - R,
        1.0 - cx - R,
        cy - R,
        1.0 - cy - R
    ])
    
    # Pairwise non-overlap: dist^2 >= 4*R^2
    dx = cx[IDX_I] - cx[IDX_J]
    dy = cy[IDX_I] - cy[IDX_J]
    con = np.concatenate([con, dx**2 + dy**2 - 4.0*R**2])
    return con

def clearance_objective(v):
    """Objective for Nelder-Mead/SA: maximize minimum clearance."""
    c = v.reshape(N_CIRCLES, 2)
    d_wall = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                        np.minimum(c[:, 1], 1.0 - c[:, 1]))
    
    diff = c[:, None, :] - c[None, :, :]
    d_pair = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(d_pair, np.inf)
    
    # Minimum clearance is limited by walls and half the pairwise distances
    min_d = np.min(np.concatenate([d_wall, d_pair[IDX_I, IDX_J] / 2.0]))
    return -min_d  # Minimize negative clearance

def generate_hex_configs(rng):
    """Generates diverse hexagonal initial configurations."""
    configs = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [7, 5, 5, 5, 4],
        [5, 7, 4, 5, 5], [6, 4, 6, 5, 5], [5, 6, 6, 5, 4]
    ]
    
    for pat in patterns:
        if sum(pat) < N_CIRCLES:
            continue
        pts = []
        y = 0.101
        for i, cnt in enumerate(pat):
            shift = 0.101 if i % 2 == 1 else 0.0
            x = 0.101 + shift
            for _ in range(cnt):
                if len(pts) >= N_CIRCLES:
                    break
                pts.append([x, y])
                x += 0.202
            y += 0.101 * np.sqrt(3)
        c = np.array(pts[:N_CIRCLES])
        # Normalize to fit tightly in [0,1]
        c = (c - c.min(axis=0)) / (c.max(axis=0) - c.min(axis=0)) * 0.92 + 0.04
        configs.append(c)
        
        # Add controlled perturbations
        for _ in range(3):
            cp = c + rng.uniform(-0.02, 0.02, c.shape)
            configs.append(np.clip(cp, 0.05, 0.95))
            
    # Add purely random starts
    for _ in range(8):
        configs.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    rng = np.random.default_rng(42)
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    configs = generate_hex_configs(rng)
    
    # Phase 1: SLSQP Joint Optimization (Equal Radius Focus)
    bounds_slqp = [(0.0, 1.0)] * (2 * n) + [(0.09, 0.11)]
    cons_slqp = {'type': 'ineq', 'fun': slsqp_constraints}
    
    for cfg in configs:
        x0 = np.concatenate([cfg.flatten(), [0.100]])
        try:
            res = minimize(slsqp_objective, x0, method='SLSQP', bounds=bounds_slqp, 
                          constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-14})
            if np.isfinite(res.fun) and res.x[-1] > 0.099:
                c_opt = res.x[:2*n].reshape(n, 2)
                r_lp, s_lp = solve_lp(c_opt)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_c = c_opt.copy()
                    best_r = r_lp.copy()
        except Exception:
            pass

    # Phase 2: Simulated Annealing on Centers (Maximize Min-Clearance)
    if best_c is not None:
        curr_c = best_c.copy()
        curr_score = clearance_objective(curr_c.flatten())
        temp = 0.005
        
        for it in range(3000):
            temp *= 0.9995
            idx = rng.choice(n, size=rng.choice([1, 2]), replace=False)
            old = curr_c[idx].copy()
            
            step = 0.006 * (1.0 + temp)
            curr_c[idx] += rng.uniform(-step, step, (len(idx), 2))
            curr_c[idx] = np.clip(curr_c[idx], 1e-4, 1.0 - 1e-4)
            
            new_score = clearance_objective(curr_c.flatten())
            delta = new_score - curr_score
            
            # Accept if better, or with probability exp(delta/temp)
            if delta > 0 or rng.random() < np.exp(delta / max(temp, 1e-6)):
                curr_score = new_score
                # Occasionally run LP to check sum improvement
                if it % 50 == 0:
                    r_lp, s_lp = solve_lp(curr_c)
                    if s_lp > best_sum:
                        best_sum = s_lp
                        best_c = curr_c.copy()
                        best_r = r_lp.copy()
            else:
                curr_c[idx] = old
                
        best_c = curr_c
        best_r, best_sum = solve_lp(best_c)

    # Phase 3: Multi-Circle Hill Climbing with Local Polish
    curr_c = best_c.copy()
    curr_r = best_r.copy()
    curr_s = best_sum
    
    for it in range(1500):
        scale = 0.008 * (1.0 - it / 1500.0)**0.4
        num_pert = rng.choice([1, 2], p=[0.7, 0.3])
        idx = rng.choice(n, size=num_pert, replace=False)
        old = curr_c[idx].copy()
        
        curr_c[idx] += rng.uniform(-scale, scale, (num_pert, 2))
        curr_c[idx] = np.clip(curr_c[idx], 1e-4, 1.0 - 1e-4)
        
        r_new, s_new = solve_lp(curr_c)
        if s_new > curr_s + 1e-7:
            curr_s = s_new
            curr_r = r_new.copy()
            
            # Local Nelder-Mead polish
            try:
                res_loc = minimize(clearance_objective, curr_c.flatten(), method='Nelder-Mead',
                                   options={'maxiter': 400, 'xatol': 1e-9, 'fatol': 1e-10})
                c_pol = np.clip(res_loc.x.reshape(n, 2), 0.001, 0.999)
                r_pol, s_pol = solve_lp(c_pol)
                if s_pol > curr_s:
                    curr_s = s_pol
                    curr_r = r_pol.copy()
                    curr_c = c_pol.copy()
            except Exception:
                pass
        else:
            curr_c[idx] = old
            
    best_c = curr_c
    best_r = curr_r
    best_sum = curr_s
    
    # Phase 4: Strict Numerical Safety Scaling
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    diff = best_c[:, None, :] - best_c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_pair = best_r[:, None] + best_r[None, :]
    min_ratio = np.min(dists[IDX_I, IDX_J] / np.maximum(r_pair[IDX_I, IDX_J], 1e-12))
    scale = min(scale, min_ratio)
    
    # Apply minimal safety margin to strictly satisfy 1e-12 tolerance
    best_r *= scale * (1.0 - 1e-7)
    best_sum = float(np.sum(best_r))
    
    # Fallback safety net
    if best_c is None or best_sum < 1.0:
        best_c = configs[0]
        best_r = np.full(n, 0.095)
        best_sum = float(np.sum(best_r))
        
    return best_c, best_r, best_sum
