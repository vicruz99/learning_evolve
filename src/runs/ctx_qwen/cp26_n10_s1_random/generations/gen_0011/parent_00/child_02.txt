import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

def solve_lp_and_grad(centers, idx_i, idx_j):
    n = centers.shape[0]
    M = len(idx_i)
    wall = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]), 
                      np.minimum(centers[:,1], 1.0 - centers[:,1]))
    wall = np.maximum(wall, 1e-9)
    
    A_pair = np.zeros((M, n))
    A_pair[np.arange(M), idx_i] = 1.0
    A_pair[np.arange(M), idx_j] = 1.0
    A_ub = np.vstack([A_pair, np.eye(n)])
    
    diff = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    b_ub = np.concatenate([dists, wall])
    bounds = [(0.0, 0.5)] * n
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            r = np.maximum(res.x, 0.0)
            # Marginals may be negative depending on scipy version/convention; take absolute value
            lam = np.abs(res.ineqlin.marginals)
            return r, -res.fun, lam, dists, wall
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0, np.zeros(M+n), dists, wall

def compute_gradient(centers, lam, dists, idx_i, idx_j):
    n = centers.shape[0]
    M = len(idx_i)
    grad = np.zeros_like(centers)
    
    pair_lam = lam[:M]
    for k in range(M):
        if pair_lam[k] > 1e-10:
            i, j = idx_i[k], idx_j[k]
            d = dists[k]
            if d < 1e-12: 
                continue
            factor = pair_lam[k] / d
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            grad[i, 0] += factor * dx
            grad[i, 1] += factor * dy
            grad[j, 0] -= factor * dx
            grad[j, 1] -= factor * dy
            
    bound_lam = lam[M:]
    for i in range(n):
        if bound_lam[i] > 1e-10:
            # Push towards center of square to relax boundary constraints
            grad[i, 0] += bound_lam[i] * (0.5 - centers[i, 0])
            grad[i, 1] += bound_lam[i] * (0.5 - centers[i, 1])
    return grad

def gradient_ascent(centers, idx_i, idx_j, max_iter=1000, lr=0.0015):
    n = centers.shape[0]
    curr_c = centers.copy()
    best_c = curr_c.copy()
    best_s = 0.0
    _, init_s, _, _, _ = solve_lp_and_grad(curr_c, idx_i, idx_j)
    best_s = init_s
    
    for it in range(max_iter):
        r, s, lam, dists, wall = solve_lp_and_grad(curr_c, idx_i, idx_j)
        if s > best_s:
            best_s = s
            best_c = curr_c.copy()
            
        grad = compute_gradient(curr_c, lam, dists, idx_i, idx_j)
        step = lr * (1.0 - it / max_iter)**0.5
        curr_c += step * grad
        curr_c = np.clip(curr_c, 0.02, 0.98)
        
    return best_c, best_s

def clearance_obj(x_flat, n):
    c = x_flat.reshape(n, 2)
    d_wall = np.minimum(np.minimum(c[:,0], 1.0 - c[:,0]), 
                        np.minimum(c[:,1], 1.0 - c[:,1]))
    diff = c[:, None, :] - c[None, :, :]
    d_pair = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(d_pair, np.inf)
    min_d = np.min(np.concatenate([d_wall, d_pair.min(axis=1) / 2.0]))
    return -min_d

def joint_obj(v, n):
    return -np.sum(v[2*n:])

def joint_cons(v, n, idx_i, idx_j):
    cx, cy, r = v[:n], v[n:2*n], v[2*n:]
    con = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
