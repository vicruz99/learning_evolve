import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Maximum radius limited by distance to walls
    w = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    w = np.maximum(w, 1e-9)
    
    # Pairwise Euclidean distances
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    
    # LP constraints: r_i + r_j <= dist(i, j)
    m = n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            b[k] = dists[i, j]
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, 
                      bounds=[(0.0, w_i) for w_i in w], method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def neg_sum_obj(x_flat):
    """Objective for optimizer: minimize negative LP sum of radii."""
    n = len(x_flat) // 2
    c = x_flat.reshape(n, 2)
    c = np.clip(c, 1e-4, 0.999)
    _, s = solve_lp(c)
    return -s

def generate_hex_config(pat, r0):
    """Generates a hexagonal lattice configuration with specified row counts."""
    n = 26
    pts = []
    y = r0
    for idx, cnt in enumerate(pat):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Generate diverse initial configurations
    starts = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,5,5], [5,5,6,5,5], 
        [6,6,5,5,4], [6,5,4,6,5], [5,4,6,6,5], [7,5,5,5,4]
    ]
    
    for pat in patterns:
        if sum(pat) != n: 
            continue
        pts = generate_hex_config(pat, 0.10)
        starts.append(pts)
        # Add controlled perturbations to break symmetry
        for _ in range(3):
            p = pts + rng.uniform(-0.025, 0.025, pts.shape)
            p = np.clip(p, 0.05, 0.95)
            starts.append(p)
            
    # Add random dense starts
    for _ in range(10):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Phase 2: Multi-start local optimization
    for cfg in starts:
        # Nelder-Mead
        res_nm = minimize(neg_sum_obj, cfg.flatten(), method='Nelder-Mead', 
                          options={'maxiter': 3000, 'xatol': 1e-8, 'fatol': 1e-10})
        c_nm = np.clip(res_nm.x.reshape(n, 2), 1e-4, 0.999)
        r_nm, s_nm = solve_lp(c_nm)
        if s_nm > best_sum:
            best_sum = s_nm
            best_centers = c_nm.copy()
            best_radii = r_nm.copy()
            
        # Powell refinement from NM result
        res_pow = minimize(neg_sum_obj, c_nm.flatten(), method='Powell', 
                           options={'maxiter': 1000, 'xtol': 1e-8, 'ftol': 1e-10})
        c_pow = np.clip(res_pow.x.reshape(n, 2), 1e-4, 0.999)
        r_pow, s_pow = solve_lp(c_pow)
        if s_pow > best_sum:
            best_sum = s_pow
            best_centers = c_pow.copy()
            best_radii = r_pow.copy()
            
    # Phase 3: Systematic perturbation with cooling to escape local minima
    curr_centers = best_centers.copy()
    temp = 0.04
    for _ in range(60):
        noise = rng.uniform(-temp, temp, curr_centers.shape)
        pert_centers = np.clip(curr_centers + noise, 0.05, 0.95)
        
        res = minimize(neg_sum_obj, pert_centers.flatten(), method='Nelder-Mead',
                       options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10})
        c_opt = np.clip(res.x.reshape(n, 2), 1e-4, 0.999)
        r_opt, s_opt = solve_lp(c_opt)
        
        # Powell polish if promising
        if s_opt > best_sum - 1e-4:
            res_p = minimize(neg_sum_obj, c_opt.flatten(), method='Powell',
                             options={'maxiter': 800, 'xtol': 1e-8, 'ftol': 1e-10})
            c_opt = np.clip(res_p.x.reshape(n, 2), 1e-4, 0.999)
            r_opt, s_opt = solve_lp(c_opt)
            
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            curr_centers = c_opt.copy()
            
        temp *= 0.96  # Cool down schedule
        
    # Fallback if optimization unexpectedly fails
    if best_centers is None:
        grid = np.array([(i * 0.18 + 0.1, j * 0.18 + 0.1) for j in range(5) for i in range(5)] + [[0.5, 0.5]])
        best_centers = grid[:n]
        best_radii, best_sum = solve_lp(best_centers)

    # Phase 4: Strict Safety Scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i,0] - best_centers[j,0], 
                         best_centers[i,1] - best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
