2. **Analytical Gradient via LP Duals**: The gradient of the LP objective w.r.t. circle centers can be computed exactly using the dual variables (marginals) of the active constraints. This allows `L-BFGS-B` to take large, informed steps towards denser packings.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_and_grad(centers):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient."""
    n = centers.shape[0]
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in limits]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, 0.0, None
        
    r = res.x
    s = -res.fun
    
    grad = np.zeros_like(centers)
    try:
        lams = np.zeros(m)
        if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            lams = res.ineqlin.marginals
        elif hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
            lams = res.marginals.ineqlin
            
        mask = lams > 1e-7
        if np.any(mask):
            k = np.where(mask)[0]
            i_a, j_a = idx_i[k], idx_j[k]
            lam = lams[k]
            d_a = b_ub[k]
            d_safe = np.maximum(d_a, 1e-12)
            diff_vec = centers[i_a] - centers[j_a]
            factors = lam[:, None] / d_safe[:, None]
            contrib = diff_vec * factors
            np.add.at(grad, i_a, contrib)
            np.add.at(grad, j_a, -contrib)
    except Exception:
        pass
        
    return r, s, grad

def obj_c(x_flat):
    """Objective for L-BFGS-B: minimize negative LP sum."""
    c = x_flat.reshape(N, 2)
    _, s, _ = solve_lp_and_grad(c)
    return -s if s > 0 else 1e6

def grad_c(x_flat):
    """Gradient for L-BFGS-B."""
    c = x_flat.reshape(N, 2)
    _, _, g = solve_lp_and_grad(c)
    return -g.flatten() if g is not None else np.zeros_like(x_flat)

def obj_single(x, idx, centers_buf):
    """Objective for coordinate descent: optimize one circle at a time."""
    temp = centers_buf.copy()
    temp[idx] = x
    _, s, _ = solve_lp_and_grad(temp)
    return -s if s > 0 else 1e6

def obj_slsqp(vars_arr):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(vars_arr[2*N:])

def cons_slsqp(vars_arr):
    """Constraints for SLSQP: boundary and pairwise non-overlap >= 0."""
    x = vars_arr[:N]
    y = vars_arr[N:2*N]
    r = vars_arr[2*N:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    idx_i, idx_j = np.triu_indices(N, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,5,5], [5,5,6,5,5], 
        [6,6,5,5,4], [6,5,4,6,5], [5,4,6,6,5], [7,5,5,5,4],
        [5,6,6,4,5], [4,5,6,6,5], [6,4,6,6,4], [5,5,5,5,6]
    ]
    
    starts = []
    for pat in patterns:
        if sum(pat) != N: continue
        pts = []
        y = 0.10
        for idx, cnt in enumerate(pat):
            shift = 0.10 if idx % 2 == 1 else 0.0
            x = 0.10 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 0.20
            y += 0.10 * np.sqrt(3)
        pts = np.array(pts[:N])
        mn, mx = pts.min(axis=0), pts.max(axis=0)
        span = mx - mn
        if span[0] > 0: pts[:, 0] = (pts[:, 0] - mn[0]) / span[0] * 0.88 + 0.06
        if span[1] > 0: pts[:, 1] = (pts[:, 1] - mn[1]) / span[1] * 0.88 + 0.06
        starts.append(pts)
        
        for _ in range(3):
            p = pts + rng.uniform(-0.025, 0.025, pts.shape)
            starts.append(np.clip(p, 0.04, 0.96))
            
    for _ in range(10):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    bounds_c = [(0.001, 0.999)] * (2 * N)
    
    for cfg in starts:
        # Phase 1: Gradient-based center optimization
        x0 = cfg.flatten()
        try:
            res = minimize(obj_c, x0, method='L-BFGS-B', jac=grad_c, 
                           bounds=bounds_c, options={'maxiter': 1500, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(N, 2)
                r_opt, s_opt, _ = solve_lp_and_grad(c_opt)
                if r_opt is not None and s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass
            
        # Phase 2: Coordinate descent to tighten gaps
        if best_centers is not None:
            curr = best_centers.copy()
            for _ in range(4):
                improved = False
                for i in range(N):
                    old = curr[i].copy()
                    try:
                        res_cd = minimize(obj_single, old, args=(i, curr), method='Nelder-Mead',
                                          options={'maxiter': 500, 'xatol': 1e-7, 'fatol': 1e-9})
                        if np.isfinite(res_cd.fun) and res_cd.fun < -best_sum - 1e-6:
                            curr[i] = np.clip(res_cd.x, 0.001, 0.999)
                            improved = True
                    except Exception:
                        pass
                if improved:
                    r_opt, s_opt, _ = solve_lp_and_grad(curr)
                    if r_opt is not None and s_opt > best_sum:
                        best_sum = s_opt
                        best_centers = curr.copy()
                        best_radii = r_opt.copy()
                else:
                    break
                    
        # Phase 3: SLSQP joint polish
        if best_centers is not None:
            x0_slqp = np.zeros(3 * N)
            x0_slqp[:N] = best_centers[:, 0]
            x0_slqp[N:2*N] = best_centers[:, 1]
            x0_slqp[2*N:] = best_radii * 0.99
            
            bounds_slqp = [(0.0, 1.0)] * (2*N) + [(1e-7, 0.5)] * N
            try:
                res_slqp = minimize(obj_slsqp, x0_slqp, method='SLSQP', bounds=bounds_slqp,
                                    constraints={'type': 'ineq', 'fun': cons_slsqp},
                                    options={'maxiter': 4000, 'ftol': 1e-14})
                if np.isfinite(res_slqp.fun):
                    cx = res_slqp.x[:N]
                    cy = res_slqp.x[N:2*N]
                    c_mat = np.column_stack((cx, cy))
                    r_lp, s_lp, _ = solve_lp_and_grad(c_mat)
                    if r_lp is not None and s_lp > best_sum:
                        best_sum = s_lp
                        best_centers = c_mat.copy()
                        best_radii = r_lp.copy()
            except Exception:
                pass
                
    # Fallback
    if best_centers is None:
        best_centers = starts[0]
        best_radii, best_sum, _ = solve_lp_and_grad(best_centers)
        
    # Phase 4: Strict safety scaling
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i,0] - best_centers[j,0], 
                         best_centers[i,1] - best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
