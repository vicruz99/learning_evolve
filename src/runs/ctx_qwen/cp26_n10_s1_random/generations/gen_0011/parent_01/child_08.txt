import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient."""
    n = centers.shape[0]
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-12)
    bounds = [(0.0, l) for l in limits]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, None, None
        
    r = res.x
    s = -res.fun
    
    grad = np.zeros((n, 2))
    try:
        lams = None
        if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            lams = res.ineqlin.marginals
        elif hasattr(res, 'marginals'):
            lams = getattr(res.marginals, 'ineqlin', None)
            
        if lams is not None:
            active = lams > 1e-6
            if np.any(active):
                k = np.where(active)[0]
                i_a, j_a = idx_i[k], idx_j[k]
                lam = lams[k]
                d = b_ub[k]
                d_safe = np.maximum(d, 1e-12)
                diff_vec = centers[i_a] - centers[j_a]
                factor = (lam[:, None] / d_safe[:, None])
                contrib = diff_vec * factor
                np.add.at(grad, i_a, contrib)
                np.add.at(grad, j_a, -contrib)
    except Exception:
        pass
        
    return r, s, grad

def obj_lp(x):
    """Objective and gradient for L-BFGS-B: maximize LP sum of radii."""
    n = len(x) // 2
    c = x.reshape(n, 2)
    _, s, g = solve_lp(c)
    if s is None:
        return 1e6, np.zeros_like(x)
    return -s, -g.flatten()

def get_constraints_slsqp(v, n):
    """Computes inequality constraints for SLSQP: must be >= 0."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def constraint_wrapper(v):
    """Wrapper to avoid lambda/closure restrictions."""
    return get_constraints_slsqp(v, N)

def obj_slsqp(v):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(v[2*N:])

def obj_cd(pos, c_fixed, idx):
    """Objective for coordinate descent: maximize LP sum given fixed others."""
    c_temp = c_fixed.copy()
    c_temp[idx] = pos
    _, s, _ = solve_lp(c_temp)
    return -s if s is not None else 1e6

def generate_hex_config(pat, r0):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0
    for idx, cnt in enumerate(pat):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    rng = np.random.default_rng(42)

    best_radii = None

    ]
