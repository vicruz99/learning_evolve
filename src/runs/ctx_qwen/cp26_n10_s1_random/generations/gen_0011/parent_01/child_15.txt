import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
N_PAIRS = len(TRIU_I)

def get_clearance(centers):
    """Computes the minimum clearance to walls and other circles."""
    x = centers[:, 0]
    y = centers[:, 1]
    d_wall = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    d_pair = np.min(dists, axis=1) / 2.0
    return np.minimum(d_wall, d_pair).min()

def neg_clearance(x_flat):
    """Objective for Nelder-Mead: maximize clearance."""
    return -get_clearance(x_flat.reshape(N, 2))

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[TRIU_I, TRIU_J]
    
    A_ub = np.zeros((N_PAIRS, N))
    A_ub[np.arange(N_PAIRS), TRIU_I] = 1.0
    A_ub[np.arange(N_PAIRS), TRIU_J] = 1.0
    
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun
    return np.zeros(N), 0.0

def generate_hex_config(pat, r0):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0
    for idx, cnt in enumerate(pat):
        shift = r0 if idx % 2 == 1 else 0.0
        row_w = (cnt - 1) * 2.0 * r0
        x_start = 0.5 - row_w / 2.0 + shift
        for k in range(cnt):
            pts.append([x_start + k * 2.0 * r0, y])
        y += np.sqrt(3.0) * r0
    return np.array(pts[:N])

def get_constraints_slsqp(vars_arr):
    """Computes inequality constraints >= 0 for SLSQP."""
    x = vars_arr[:N]
    y = vars_arr[N:2*N]
    r = vars_arr[2*N:]
    
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    dx = x[TRIU_I] - x[TRIU_J]
    dy = y[TRIU_I] - y[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def obj_slsqp(vars_arr):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*N:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = -1.0
    best_centers = None
    best_radii = None

        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,5,5], [5,5,6,5,5], 
        [6,6,5,5,4], [6,5,4,6,5], [5,4,6,6,5], [7,5,5,5,4],
        [6,5,5,6,4], [5,6,4,6,5]
    ]
    
    starts = []
    for pat in patterns:
        if sum(pat) != N: continue
        for r0 in [0.095, 0.10, 0.105]:

            mn, mx = pts.min(axis=0), pts.max(axis=0)
            span = mx - mn
            if span[0] > 0: pts[:, 0] = (pts[:, 0] - mn[0]) / span[0] * 0.8 + 0.1
            if span[1] > 0: pts[:, 1] = (pts[:, 1] - mn[1]) / span[1] * 0.8 + 0.1
            starts.append(pts)
            
            # Perturbed variants
            for _ in range(3):
                p = pts + rng.uniform(-0.02, 0.02, pts.shape)
                starts.append(np.clip(p, 0.05, 0.95))
                
    for _ in range(8):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # Phase 1: Clearance optimization + LP + Local Search
    for cfg in starts:
        # Maximize clearance
        res = minimize(neg_clearance, cfg.flatten(), method='Nelder-Mead',
                       options={'maxiter': 25000, 'xatol': 1e-10, 'fatol': 1e-12})
        c_opt = np.clip(res.x.reshape(N, 2), 1e-4, 1.0 - 1e-4)
        
        r_opt, s_opt = solve_lp(c_opt)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()

        curr_c = c_opt.copy()
        curr_r = r_opt.copy()

            best_radii = curr_r.copy()

    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):

            d = np.hypot(best_centers[i,0]-best_centers[j,0], 
                         best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
