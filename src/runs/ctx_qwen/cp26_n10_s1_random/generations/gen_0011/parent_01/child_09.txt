from scipy.optimize import linprog, minimize

def compute_lp(centers):
    n = centers.shape[0]
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-9)) for lim in limits]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun, res
    return None, 0.0, None

def compute_gradient(centers, res, idx_i, idx_j, b_ub):

def get_slsqp_obj(v):
    n = len(v) // 3
    return -np.sum(v[2*n:])

def get_slsqp_cons(v):
    n = len(v) // 3
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
