import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_and_grad(centers, n):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient."""
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, None, None
        
    r = res.x
    s = -res.fun
    
    grad = np.zeros((n, 2))
    try:
        lams = None
        if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            lams = res.ineqlin.marginals
        elif hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
            lams = res.marginals.ineqlin
            
        if lams is not None:
            active = lams > 1e-6
            if np.any(active):
                k_idx = np.where(active)[0]
                i_a = idx_i[k_idx]
                j_a = idx_j[k_idx]
                d_a = b_ub[k_idx]
                lam = lams[k_idx]
                d_safe = np.maximum(d_a, 1e-12)
                factors = lam[:, np.newaxis] / d_safe[:, np.newaxis]
                diffs = centers[i_a] - centers[j_a]
                contribs = diffs * factors
                
                np.add.at(grad, i_a, contribs)
                np.add.at(grad, j_a, -contribs)
                
        # Boundary gradient approximation
        for i in range(n):
            if r[i] > 0.998 * lims[i]:
                x, y = centers[i]
                if x - r[i] < 1e-4: grad[i, 0] += 1.0
                elif x + r[i] > 1.0 - 1e-4: grad[i, 0] -= 1.0
                if y - r[i] < 1e-4: grad[i, 1] += 1.0
                elif y + r[i] > 1.0 - 1e-4: grad[i, 1] -= 1.0
    except Exception:
        pass
        
    return r, s, grad

def lp_obj_and_grad(x_flat, n):
    """Objective and gradient for L-BFGS-B: maximize LP sum of radii."""
    c = x_flat.reshape(n, 2)
    r, s, g = solve_lp_and_grad(c, n)
    if r is None:
        return 1e6, np.full_like(x_flat, 0.0)
    return -s, -g.flatten()

def refine_centers(centers, n, max_iters=500):
    """Refines centers using L-BFGS-B to maximize LP radii sum."""
    x0 = centers.flatten()
    bounds = [(0.001, 0.999)] * (2 * n)
    try:
        res = minimize(lp_obj_and_grad, x0, args=(n,), method='L-BFGS-B', jac=True,
                       bounds=bounds, options={'maxiter': max_iters, 'ftol': 1e-13})
        if np.isfinite(res.fun):
            c_opt = res.x.reshape(n, 2)
            r_opt, s_opt, _ = solve_lp_and_grad(c_opt, n)
            if r_opt is not None:
                return c_opt, r_opt, s_opt
    except Exception:
        pass
    return centers, None, -1.0

def get_constraints(vars_arr, n):
    """Computes inequality constraints for SLSQP: must be >= 0."""
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c_list = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c_list.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c_list)

def slqp_objective(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def slqp_constraints(v, n):
    """Wrapper for SLSQP constraints."""
    return get_constraints(v, n)
