import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    idx_i, idx_j = np.triu_indices(n, k=1)
    d_ij = dists[idx_i, idx_j]
    
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=d_ij, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def neg_lp_obj(x_flat):
    """Objective for center optimization: minimize negative LP sum of radii."""
    c = x_flat.reshape(N, 2)
    _, s = solve_lp_radii(c)
    return -s

def get_joint_constraints(v):
    """Computes inequality constraints >= 0 for joint SLSQP optimization."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    idx_i, idx_j = np.triu_indices(N, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def joint_obj(v):
    """Objective for joint SLSQP: maximize sum of radii."""
    return -np.sum(v[2*N:])

def generate_hex_config(pat, r0):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0
    for idx, cnt in enumerate(pat):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], 
        [4,6,6,6,4], [5,6,6,5,4], [6,5,4,6,5], [7,5,5,5,4]
    ]
    
    starts = []
    for pat in patterns:
        if sum(pat) != N: continue
        cfg = generate_hex_config(pat, 0.101)
        # Normalize to [0.05, 0.95] to give optimizer room
        mn, mx = cfg.min(axis=0), cfg.max(axis=0)
        span = mx - mn
        if span[0] > 0: cfg[:, 0] = (cfg[:, 0] - mn[0]) / span[0] * 0.9 + 0.05
        if span[1] > 0: cfg[:, 1] = (cfg[:, 1] - mn[1]) / span[1] * 0.9 + 0.05
        starts.append(cfg)
        for _ in range(3):
            p = cfg + rng.uniform(-0.025, 0.025, cfg.shape)
            starts.append(np.clip(p, 0.02, 0.98))
            
    for _ in range(10):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    best_sum = 0.0
    best_c = None
    best_r = None
    
    bounds_vars = [(0.0, 1.0)]*(2*N) + [(1e-6, 0.5)]*N
    cons = {'type': 'ineq', 'fun': get_joint_constraints}
    
    # Phase 1: Joint SLSQP optimization from diverse starts
    for cfg in starts:
        r_init = np.full(N, 0.08)
        for i in range(N):
            dw = min(cfg[i,0], 1.0-cfg[i,0], cfg[i,1], 1.0-cfg[i,1])
            d_nn = np.min(np.linalg.norm(cfg - cfg[i], axis=1))
            d_nn = d_nn if d_nn > 1e-6 else 0.1
            r_init[i] = min(dw, d_nn / 2.0) * 0.85
            
        x0 = np.zeros(3*N)
        x0[:N] = cfg[:,0]
        x0[N:2*N] = cfg[:,1]
        x0[2*N:] = r_init
        
        try:
            res = minimize(joint_obj, x0, method='SLSQP', bounds=bounds_vars, 
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                cx = res.x[:N]
                cy = res.x[N:2*N]
                c_mat = np.column_stack((cx, cy))
                r_lp, s_lp = solve_lp_radii(c_mat)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_c = c_mat.copy()
                    best_r = r_lp.copy()
        except Exception:
            pass

    # Phase 2: LP Center Refinement via Powell & Nelder-Mead
    if best_c is not None:
        x_curr = best_c.flatten()
        for _ in range(4):
            try:
                res = minimize(neg_lp_obj, x_curr, method='Powell', 
                               options={'maxiter': 3000, 'xtol': 1e-10, 'ftol': 1e-12})
                if np.isfinite(res.fun):
                    c_opt = np.clip(res.x.reshape(N, 2), 0.001, 0.999)
                    r_opt, s_opt = solve_lp_radii(c_opt)
                    if s_opt > best_sum:
                        best_sum = s_opt
                        best_c = c_opt.copy()
                        best_r = r_opt.copy()
                        x_curr = res.x
            except Exception:
                pass
                
        # Phase 3: Local random restarts around best configuration
        for _ in range(20):
            x_pert = best_c.flatten() + rng.uniform(-0.015, 0.015, 2*N)
            x_pert = np.clip(x_pert, 0.001, 0.999)
            try:
                res = minimize(neg_lp_obj, x_pert, method='Nelder-Mead',
                               options={'maxiter': 1500, 'xatol': 1e-9, 'fatol': 1e-10})
                if np.isfinite(res.fun):
                    c_opt = np.clip(res.x.reshape(N, 2), 0.001, 0.999)
                    r_opt, s_opt = solve_lp_radii(c_opt)
                    if s_opt > best_sum:
                        best_sum = s_opt
                        best_c = c_opt.copy()
                        best_r = r_opt.copy()
            except Exception:
                pass

    # Fallback to safe grid if optimization unexpectedly fails
    if best_c is None:
        grid = np.array([(i * 0.18 + 0.1, j * 0.18 + 0.1) for j in range(5) for i in range(5)] + [[0.5, 0.5]])
        best_c = grid[:N]
        best_r, best_sum = solve_lp_radii(best_c)

    # Final strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N):
        x, y = best_c[i]
        r = best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_c[i,0] - best_c[j,0], best_c[i,1] - best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
