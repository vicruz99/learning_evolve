import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient via duals."""
    n = centers.shape[0]
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-9)) for lim in limits]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return np.full(n, 1e-9), 0.0, np.zeros((n, 2))
        
    r = np.maximum(res.x, 0.0)
    s = -res.fun
    grad = np.zeros((n, 2))
    
    try:
        lams = np.zeros(m)
        if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            lams = res.ineqlin.marginals
        elif hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
            lams = res.marginals.ineqlin
            
        valid = lams > 1e-8
        if np.any(valid):
            k_idx = np.where(valid)[0]
            i_a = idx_i[k_idx]
            j_a = idx_j[k_idx]
            lam = lams[k_idx]
            d_vec = centers[i_a] - centers[j_a]
            d_norm = np.linalg.norm(d_vec, axis=1)
            d_norm_safe = np.maximum(d_norm, 1e-12)
            factors = lam / d_norm_safe
            np.add.at(grad, i_a, d_vec * factors[:, np.newaxis])
            np.add.at(grad, j_a, -d_vec * factors[:, np.newaxis])
    except Exception:
        pass
    return r, s, grad

def lp_obj_only(x_flat):
    """Scalar objective for derivative-free optimizers."""
    c = x_flat.reshape(N_CIRCLES, 2)
    c = np.clip(c, 1e-5, 1.0 - 1e-5)
    return -solve_lp_radii(c)[1]

def lp_obj_and_grad(x_flat):
    """Objective and gradient for L-BFGS-B."""
    c = x_flat.reshape(N_CIRCLES, 2)
    c = np.clip(c, 1e-5, 1.0 - 1e-5)
    _, s, g = solve_lp_radii(c)
    return -s, -g.flatten()

def get_constraints_slsqp(vars_arr):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    n = N_CIRCLES
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def obj_slsqp(vars_arr):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*N_CIRCLES:])

def generate_starts():
    """Generates diverse initial center configurations."""
    starts = []
    rng = np.random.default_rng(42)
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,6,5,5], [5,5,6,5,5], [6,6,5,5,4], [5,4,6,6,5]]
    
    for pat in patterns:
        pts = []
        y = 0.10
        for idx, cnt in enumerate(pat):
            shift = 0.10 if idx % 2 == 1 else 0.0
            x = 0.10 + shift
            for _ in range(cnt):
                if len(pts) >= N_CIRCLES: break
                pts.append([x, y])
                x += 0.20
            y += 0.10 * np.sqrt(3)
        pts = np.array(pts[:N_CIRCLES])
        mn, mx = pts.min(axis=0), pts.max(axis=0)
        span = mx - mn
        if span[0] > 0: pts[:,0] = (pts[:,0]-mn[0])/span[0]*0.9 + 0.05
        if span[1] > 0: pts[:,1] = (pts[:,1]-mn[1])/span[1]*0.9 + 0.05
        starts.append(pts)
        for _ in range(4):
            starts.append(np.clip(pts + rng.uniform(-0.03, 0.03, pts.shape), 0.02, 0.98))
            
    for _ in range(10):
        starts.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)))
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -1.0
    best_c = None
    best_r = None
    
    starts = generate_starts()
    
    for cfg in starts:
        c = cfg.copy()
        
        # Phase 1: Nelder-Mead exploration
        res_nm = minimize(lp_obj_only, c.flatten(), method='Nelder-Mead', 
                          options={'maxiter': 3000, 'xatol': 1e-7, 'fatol': 1e-10})
        c_nm = np.clip(res_nm.x.reshape(N_CIRCLES, 2), 1e-4, 1.0-1e-4)
        r_nm, s_nm, _ = solve_lp_radii(c_nm)
        if s_nm > best_sum:
            best_sum = s_nm
            best_c = c_nm.copy()
            best_r = r_nm.copy()
            
        # Phase 2: L-BFGS-B precise local optimization
        try:
            res_lb = minimize(lp_obj_and_grad, c_nm.flatten(), jac=True, method='L-BFGS-B',
                              bounds=[(0.001, 0.999)]*(2*N_CIRCLES), 
                              options={'maxiter': 4000, 'ftol': 1e-13})
            c_lb = res_lb.x.reshape(N_CIRCLES, 2)
            r_lb, s_lb, _ = solve_lp_radii(c_lb)
            if s_lb > best_sum:
                best_sum = s_lb
                best_c = c_lb.copy()
                best_r = r_lb.copy()
        except Exception:
            pass
            
        # Phase 3: SLSQP joint polish of centers and radii
        if best_c is not None:
            x0 = np.concatenate([best_c[:,0], best_c[:,1], best_r*0.99])
            bounds_s = [(0.0, 1.0)]*(2*N_CIRCLES) + [(1e-6, 0.5)]*N_CIRCLES
            try:
                res_sl = minimize(obj_slsqp, x0, method='SLSQP', bounds=bounds_s,
                                  constraints={'type': 'ineq', 'fun': get_constraints_slsqp},
                                  options={'maxiter': 5000, 'ftol': 1e-14})
                if np.isfinite(res_sl.fun):
                    cx, cy, r_sl = res_sl.x[:N_CIRCLES], res_sl.x[N_CIRCLES:2*N_CIRCLES], res_sl.x[2*N_CIRCLES:]
                    c_sl = np.column_stack((cx, cy))
                    r_final, s_final, _ = solve_lp_radii(c_sl)
                    if s_final > best_sum:
                        best_sum = s_final
                        best_c = c_sl.copy()
                        best_r = r_final.copy()
            except Exception:
                pass

    # Phase 4: Systematic coordinate descent perturbation
    if best_c is not None:
        rng = np.random.default_rng(123)
        curr_c, curr_r, curr_s = best_c.copy(), best_r.copy(), best_sum
        for step in [0.02, 0.01, 0.005, 0.002, 0.001]:
            for _ in range(150):
                idx = rng.integers(N_CIRCLES)
                old_pos = curr_c[idx].copy()
                curr_c[idx] = np.clip(old_pos + rng.uniform(-step, step, 2), 0.01, 0.99)
                r_t, s_t, _ = solve_lp_radii(curr_c)
                if s_t > curr_s + 1e-9:
                    curr_r = r_t.copy()
                    curr_s = s_t
                else:
                    curr_c[idx] = old_pos
        best_c, best_r, best_sum = curr_c, curr_r, curr_s

    # Phase 5: Strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N_CIRCLES):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
