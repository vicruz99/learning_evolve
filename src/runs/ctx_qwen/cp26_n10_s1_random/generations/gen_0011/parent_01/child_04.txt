import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers and computes gradient."""
    n = centers.shape[0]
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in limits]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, None, None
        
    r = res.x
    s = -res.fun
    
    grad = np.zeros((n, 2))
    try:
        lams = None
        if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            lams = res.ineqlin.marginals
        elif hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
            lams = res.marginals.ineqlin
            
        if lams is not None:
            valid = lams > 1e-6
            if np.any(valid):
                k_idx = np.where(valid)[0]
                i_a = idx_i[k_idx]
                j_a = idx_j[k_idx]
                d_a = b_ub[k_idx]
                lam = lams[k_idx]
                d_safe = np.maximum(d_a, 1e-12)
                diff_vec = centers[i_a] - centers[j_a]
                factors = lam[:, np.newaxis] / d_safe[:, np.newaxis]
                grad_contrib = diff_vec * factors
                np.add.at(grad, i_a, grad_contrib)
                np.add.at(grad, j_a, -grad_contrib)
    except Exception:
        pass
        
    return r, s, grad

def lp_obj_and_grad(x_flat, n):
    """Objective and gradient for L-BFGS-B: maximize LP sum of radii."""
    c = x_flat.reshape(n, 2)
    r, s, g = solve_lp_radii(c)
    if r is None:
        return 1e6, np.full_like(x_flat, 0.0)
    return -s, -g.flatten()

def get_constraints(vars_arr, n):
    """Computes inequality constraints for SLSQP: must be >= 0."""
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c_list = []
    c_list.append(x - r)
    c_list.append(1.0 - x - r)
    c_list.append(y - r)
    c_list.append(1.0 - y - r)
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c_list.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c_list)

def constraint_wrapper(vars_arr, n):
    """Wrapper to avoid lambda/closure restrictions."""
    return get_constraints(vars_arr, n)

def get_objective(vars_arr, n):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars_arr[2*n:])

def generate_hex_config(pat, r0):
    """Generates a hexagonal lattice configuration."""
    n = 26
    pts = []
    y = r0
    for idx, cnt in enumerate(pat):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    return np.array(pts[:n])

def obj_coord_desc(v, idx, centers_ref):
    """Objective for coordinate descent: optimize one circle's position."""
    c_temp = centers_ref.copy()
    c_temp[idx] = v
    _, s, _ = solve_lp_radii(c_temp)
    return -s if s is not None else 1e6

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,5,5], [5,5,6,5,5], 
        [6,6,5,5,4], [6,5,4,6,5], [5,4,6,6,5], [7,5,5,5,4],
        [5,5,5,5,6], [6,5,5,6,4], [5,6,6,4,5]
    ]
    
    starts = []
    for pat in patterns:
        if sum(pat) != n: continue
        pts = generate_hex_config(pat, 0.10)
        starts.append(pts)
        for _ in range(4):
            p = pts + rng.uniform(-0.03, 0.03, pts.shape)
            p = np.clip(p, 0.02, 0.98)
            starts.append(p)
            
    for _ in range(10):
        starts.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    bounds_vars = [(0.001, 0.999)]*(2*n) + [(1e-5, 0.4)]*n
    cons = {'type': 'ineq', 'fun': constraint_wrapper, 'args': (n,)}
    
    # Phase 1: SLSQP Joint Optimization
    for cfg in starts:
        r_init = np.full(n, 0.08)
        for i in range(n):
            dw = min(cfg[i,0], 1.0-cfg[i,0], cfg[i,1], 1.0-cfg[i,1])
            dists_i = np.linalg.norm(cfg - cfg[i], axis=1)
            dists_i[i] = np.inf
            dn = np.min(dists_i)
            r_init[i] = min(dw, dn / 2.0) * 0.85
            
        x0 = np.zeros(3*n)
        x0[:n] = cfg[:,0]
        x0[n:2*n] = cfg[:,1]
        x0[2*n:] = r_init
        
        try:
            res = minimize(get_objective, x0, args=(n,), method='SLSQP', bounds=bounds_vars, 
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                cx = res.x[:n]
                cy = res.x[n:2*n]
                c_mat = np.column_stack((cx, cy))
                r_lp, s_lp, _ = solve_lp_radii(c_mat)

                    best_sum = s_lp
                    best_centers = c_mat.copy()

                    if np.isfinite(res.fun):

                        r_opt, s_opt, _ = solve_lp_radii(c_opt)

                            best_sum = s_opt
                            best_centers = c_opt.copy()
                            best_radii = r_opt.copy()
                            x_curr = res.x
                except Exception:
                    pass

        # Phase 3: Coordinate Descent (fine-grained local packing)
        for iteration in range(6):
            improved = False
            for i in range(n):
                old_pos = best_centers[i].copy()
                bounds_2d = [(0.001, 0.999), (0.001, 0.999)]

                    if np.isfinite(res_2d.fun) and res_2d.fun < -best_sum - 1e-9:
                        best_centers[i] = res_2d.x
                        r_opt, s_opt, _ = solve_lp_radii(best_centers)
                        if s_opt > best_sum:
                            best_sum = s_opt
                            best_radii = r_opt.copy()
                            improved = True
                except Exception:
                    pass
            if not improved:
                break

    # Fallback
    if best_centers is None:
        grid = np.array([(i * 0.18 + 0.1, j * 0.18 + 0.1) for j in range(5) for i in range(5)] + [[0.5, 0.5]])
        best_centers = grid[:n]

    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
