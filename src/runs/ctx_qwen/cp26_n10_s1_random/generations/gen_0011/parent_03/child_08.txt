import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, l) for l in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun
    return np.full(n, 1e-6), 0.0

def obj_equal(v, n):
    """Objective for equal-radius optimization: minimize negative shared radius t."""
    return -v[-1]

def cons_equal(v, n):
    """Inequality constraints >= 0 for equal-radius packing."""
    xs = v[:n]
    ys = v[n:2*n]
    r = v[-1]
    
    con = np.concatenate([xs - r, 1.0 - xs - r, ys - r, 1.0 - ys - r])
    
    dx = xs[:, np.newaxis] - xs[np.newaxis, :]
    dy = ys[:, np.newaxis] - ys[np.newaxis, :]
    d2 = dx**2 + dy**2
    idx = np.triu_indices(n, k=1)
    con = np.concatenate([con, d2[idx] - 4.0 * r**2])
    return con

def obj_unequal(v, n):
    """Objective for unequal radii: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_unequal(v, n):
    """Inequality constraints >= 0 for valid packing in SLSQP."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    con = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = (r[:, np.newaxis] + r[np.newaxis, :])**2
    idx = np.triu_indices(n, k=1)
    con = np.concatenate([con, d2[idx] - rs[idx]])
    return con

def generate_hex(row_counts, r0, n):
    """Generates a hexagonal lattice initialization with specified row counts."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        width = (cnt - 1) * 2 * r0
        x_start = 0.5 - width / 2.0 + shift
        for k in range(cnt):
            if len(pts) >= n: break
            pts.append([x_start + k * 2 * r0, y])
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

    best_sum = -1.0

    best_r = None

            if np.isfinite(res.fun):

    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
