import numpy as np
from scipy.optimize import minimize, linprog

def compute_lp_radii(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.sqrt(dx**2 + dy**2)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A = np.zeros((m, n))
    A[np.arange(m), idx_i] = 1.0
    A[np.arange(m), idx_j] = 1.0
    b = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def neg_lp_obj(c_flat, n):
    """Objective for center optimization: negative max sum of radii."""
    c = c_flat.reshape(n, 2)
    _, s = compute_lp_radii(c, n)
    return -s

def obj_joint(v, n):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_joint(v, n):

    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]

    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
