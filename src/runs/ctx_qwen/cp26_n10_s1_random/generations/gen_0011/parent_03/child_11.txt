import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_for_centers(centers, A_ub, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    n_pairs = len(A_ub) - 4 * n
    b_ub = np.empty(len(A_ub))
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    b_ub[:n] = centers[:, 0]
    b_ub[n:2*n] = 1.0 - centers[:, 0]
    b_ub[2*n:3*n] = centers[:, 1]
    b_ub[3*n:4*n] = 1.0 - centers[:, 1]
    
    # Pairwise constraints: r_i + r_j <= d_ij
    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b_ub[idx] = np.hypot(dx, dy)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return None, 0.0, None

def compute_lp_gradient(centers, r, res_obj, n, A_ub):

    grad = np.zeros((n, 2))

            mu = res_obj.marginals.ineqlin
        elif hasattr(res_obj, 'ineqlin') and hasattr(res_obj.ineqlin, 'marginals'):
            mu = res_obj.ineqlin.marginals
        else:
            return None
            
        if mu is None:
            return None
            
        # Boundary duals contribution
        for i in range(n):
            mu_x = mu[i]
            mu_1x = mu[n + i]
            mu_y = mu[2*n + i]
            mu_1y = mu[3*n + i]
            grad[i, 0] += mu_x - mu_1x
            grad[i, 1] += mu_y - mu_1y
            
        # Pairwise duals contribution
        offset = 4 * n
        for i in range(n):
            for j in range(i + 1, n):
                lam = mu[offset]

                        fy = lam * dy / d
                        grad[i, 0] += fx
                        grad[i, 1] += fy
                        grad[j, 0] -= fx
                        grad[j, 1] -= fy
                offset += 1
    except Exception:
        return None
        
    return grad

def obj_and_grad(c_flat, n, A_ub):

def joint_obj(v, n):
    return -np.sum(v[2*n:])

def joint_cons(v, n):
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    triu_i, triu_j = np.triu_indices(n, k=1)
    dx = cx[triu_i] - cx[triu_j]
    dy = cy[triu_i] - cy[triu_j]
    d2 = dx**2 + dy**2
    rs = r[triu_i] + r[triu_j]
    c = np.concatenate([c, d2 - rs**2])
    return c

def generate_initial_configs(n, rng):
    """Generates diverse starting configurations."""
    configs = []
    
    # Hexagonal patterns with various row counts
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [5, 7, 5, 5, 4],
        [6, 5, 5, 6, 4], [5, 6, 6, 4, 5], [7, 6, 6, 7], [8, 6, 6, 6]
    ]
    
    for pat in patterns:
        if sum(pat) < n:
            continue
        r0 = 0.10
        pts = []
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n:
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        pts = np.array(pts[:n])
        # Add slight noise to break symmetry
        pts += rng.uniform(-0.01, 0.01, pts.shape)
        configs.append(np.clip(pts, 0.01, 0.99))
        
    # Random starts
    for _ in range(12):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # Grid starts
    for _ in range(4):
        grid = np.array([(i * 0.18 + 0.05, j * 0.18 + 0.05) 
                         for j in range(6) for i in range(5)] + [[0.5, 0.5]])
        configs.append(np.clip(grid[:n] + rng.uniform(-0.02, 0.02, (n, 2)), 0.01, 0.99))
        
    return configs
