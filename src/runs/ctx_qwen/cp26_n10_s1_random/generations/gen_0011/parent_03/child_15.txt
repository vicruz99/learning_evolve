import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
N_PAIRS = len(TRIU_I)

def slsqp_objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def slsqp_constraints(v):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    
    c = np.empty(4*N + N_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[TRIU_I] - cx[TRIU_J]
    dy = cy[TRIU_I] - cy[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    c[4*N:] = dx**2 + dy**2 - dr**2
    return c

def generate_hex_configs(rng):
    """Generates diverse hexagonal and random initial configurations."""
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5],
        [4,6,6,6,4], [5,7,5,5,4], [6,5,5,6,4], [7,6,6,7],
        [5,5,5,5,6], [6,5,6,4,5], [5,6,6,5,4], [4,5,6,5,6]
    ]
    for pat in patterns:
        if sum(pat) < N: continue
        pts = []
        y = 0.10
        for ri, cnt in enumerate(pat):
            shift = 0.10 if ri % 2 == 1 else 0.0
            x = 0.10 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 0.20
            y += 0.10 * np.sqrt(3.0)
        pts = np.array(pts[:N])
        configs.append(pts)
        
    for _ in range(15):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return configs

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    A_ub = np.zeros((4*N + N_PAIRS, N))
    b_ub = np.zeros(4*N + N_PAIRS)
    
    for i in range(N):
        A_ub[i, i] = 1.0
        A_ub[N+i, i] = 1.0
        A_ub[2*N+i, i] = 1.0
        A_ub[3*N+i, i] = 1.0
        b_ub[i] = centers[i, 0]
        b_ub[N+i] = 1.0 - centers[i, 0]
        b_ub[2*N+i] = centers[i, 1]
        b_ub[3*N+i] = 1.0 - centers[i, 1]
        
    for k in range(N_PAIRS):
        i, j = TRIU_I[k], TRIU_J[k]
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        d = np.hypot(dx, dy)
        A_ub[4*N+k, i] = 1.0
        A_ub[4*N+k, j] = 1.0
        b_ub[4*N+k] = d
        
    bounds = [(0.0, l) for l in lims]
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-7), 0.0

def physics_relaxation(centers, radii, iters=250):
    """Force-directed relaxation to tighten packing."""
    c = centers.copy()
    r = radii.copy()
    dt = 0.002
    
    for _ in range(iters):
        forces = np.zeros_like(c)
        
        for i in range(N):
            # Boundary repulsion
            if c[i,0] < r[i] + 0.01: forces[i,0] += 5.0 * (r[i] + 0.01 - c[i,0])
            elif c[i,0] > 1.0 - r[i] - 0.01: forces[i,0] -= 5.0 * (c[i,0] - (1.0 - r[i] - 0.01))
            if c[i,1] < r[i] + 0.01: forces[i,1] += 5.0 * (r[i] + 0.01 - c[i,1])
            elif c[i,1] > 1.0 - r[i] - 0.01: forces[i,1] -= 5.0 * (c[i,1] - (1.0 - r[i] - 0.01))
            
        for k in range(N_PAIRS):
            i, j = TRIU_I[k], TRIU_J[k]
            dx = c[i,0] - c[j,0]
            dy = c[i,1] - c[j,1]
            d = np.hypot(dx, dy)
            req = r[i] + r[j] + 0.005
            if d < req and d > 1e-6:
                f = (req - d) * 5.0 / d
                fx = dx * f
                fy = dy * f
                forces[i,0] += fx
                forces[i,1] += fy
                forces[j,0] -= fx
                forces[j,1] -= fy
                
        c += forces * dt
        c = np.clip(c, 0.001, 0.999)
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    configs = generate_hex_configs(rng)
    
    best_s = -1.0

    best_r = None

                           constraints={'type': 'ineq', 'fun': slsqp_constraints},
                           options={'maxiter': 5000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_tmp = np.column_stack((res.x[:N], res.x[N:2*N]))
                r_tmp, s_tmp = solve_lp_radii(c_tmp)
                if s_tmp > best_s:
                    best_s = s_tmp
                    best_c = c_tmp.copy()
                    best_r = r_tmp.copy()
        except Exception:
            continue
            
    # Phase 2: Physics relaxation + LP refinement
    if best_c is not None:
        c_phys = physics_relaxation(best_c, best_r, iters=250)
        r_phys, s_phys = solve_lp_radii(c_phys)
        if s_phys > best_s:
            best_s = s_phys
            best_c = c_phys.copy()
            best_r = r_phys.copy()
            
    # Phase 3: LP-based Local Perturbation Search
    if best_c is not None:
        c_curr = best_c.copy()
        r_curr = best_r.copy()
        s_curr = best_s

            idx = rng.integers(N)
            old = c_curr[idx].copy()
            step = 0.006 * (0.99 ** (it / 25.0))

            c_curr[idx] = np.clip(c_curr[idx], 0.005, 0.995)
            
            r_try, s_try = solve_lp_radii(c_curr)
            if s_try > s_curr + 1e-8:
                s_curr = s_try
                r_curr = r_try.copy()
            else:
                c_curr[idx] = old
                
        best_c = c_curr
        best_r = r_curr
        best_s = s_curr

    # Phase 4: Final SLSQP polish
    if best_c is not None:
        v0 = np.concatenate([best_c[:,0], best_c[:,1], best_r * 0.998])
        try:
            res_f = minimize(slsqp_objective, v0, method='SLSQP', bounds=bounds_opt,
                             constraints={'type': 'ineq', 'fun': slsqp_constraints},
                             options={'maxiter': 3000, 'ftol': 1e-13})
            if np.isfinite(res_f.fun):

                r_f, s_f = solve_lp_radii(c_f)
                if s_f > best_s:
                    best_s = s_f
                    best_c = c_f.copy()
                    best_r = r_f.copy()
        except Exception:
            pass

    # Safety scaling to guarantee strict numerical validity
    if best_c is not None:
        scale = 1.0
        for i in range(N):
            x, y, r = best_c[i,0], best_c[i,1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                rs = best_r[i] + best_r[j]
                if rs > 1e-12:
                    scale = min(scale, d/rs)
        best_r *= scale * 0.9999995
        best_s = float(np.sum(best_r))
    else:
        best_c = rng.uniform(0.1, 0.9, (N, 2))
        best_r = np.full(N, 0.05)
        best_s = float(np.sum(best_r))
        
    return best_c, best_r, best_s
