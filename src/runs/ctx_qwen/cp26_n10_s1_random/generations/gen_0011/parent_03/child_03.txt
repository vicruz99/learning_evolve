from scipy.optimize import minimize, linprog

def get_lp_data(n):
    """Precompute LP constraint matrix structure for pairwise distances."""
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    m_pairs = len(pairs)
    A_ub = np.zeros((m_pairs, n))
    for k, (i, j) in enumerate(pairs):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
    return A_ub, pairs, m_pairs

def solve_lp_only(centers, n, A_ub, m_pairs):
    """Solves LP to maximize sum of radii for fixed centers (fast version)."""
    x = centers[:, 0]
    y = centers[:, 1]
    limits = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    bounds = [(0.0, max(l, 1e-9)) for l in limits]
    
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.sqrt(dx**2 + dy**2)
    b_ub = dists[np.triu_indices(n, k=1)]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def solve_lp_with_grad(centers, n, A_ub, m_pairs, pairs):

    y = centers[:, 1]

    b_ub = dists[np.triu_indices(n, k=1)]

        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

            return None, 0.0, None
            
        r = res.x
        s = -res.fun
        grad = np.zeros((n, 2))

        try:
            marg = res.ineqlin.marginals
        except AttributeError:
            try:
                marg = res.marginals.ineqlin
            except AttributeError:
                marg = None

            for k, (i, j) in enumerate(pairs):

                    dyc = centers[i, 1] - centers[j, 1]

                        fy = lam * dyc / d
                        grad[i, 0] += fx
                        grad[i, 1] += fy
                        grad[j, 0] -= fx
                        grad[j, 1] -= fy
                        
        return r, s, grad
    except Exception:
        return None, 0.0, None

def obj_func_lp(c_flat, n, A_ub, m_pairs, pairs):
    """Objective for L-BFGS-B: negative sum of radii and its gradient."""
    c = c_flat.reshape(n, 2)
    r, s, g = solve_lp_with_grad(c, n, A_ub, m_pairs, pairs)
    if r is None:
        return 1e6, np.zeros_like(c_flat)
    return -s, -g.flatten()

def joint_obj(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def joint_cons(v, n):
    """Inequality constraints >= 0 for valid packing in SLSQP."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    con = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    triu_idx = np.triu_indices(n, k=1)
    dx = cx[triu_idx[0]] - cx[triu_idx[1]]
    dy = cy[triu_idx[0]] - cy[triu_idx[1]]
    rs = r[triu_idx[0]] + r[triu_idx[1]]
    con = np.concatenate([con, dx**2 + dy**2 - rs**2])
    return con
