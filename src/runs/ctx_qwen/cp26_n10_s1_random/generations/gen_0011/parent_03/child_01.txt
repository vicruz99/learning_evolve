from scipy.optimize import minimize

def obj_joint(v):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(v[2::3])

def cons_joint(v, n):
    """Inequality constraints >= 0 for valid packing (joint optimization)."""
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    m = 4 * n + n * (n - 1) // 2
    con = np.empty(m)
    
    # Boundary constraints
    con[:n] = xs - rs
    con[n:2*n] = 1.0 - xs - rs
    con[2*n:3*n] = ys - rs
    con[3*n:4*n] = 1.0 - ys - rs
    
    # Pairwise non-overlap constraints
    triu_i, triu_j = np.triu_indices(n, k=1)
    dx = xs[triu_i] - xs[triu_j]
    dy = ys[triu_i] - ys[triu_j]
    r_sum = rs[triu_i] + rs[triu_j]
    con[4*n:] = dx**2 + dy**2 - r_sum**2
    return con

def obj_coord(v):
    """Objective for coordinate descent: maximize current circle's radius."""
    return -v[2]

def cons_coord(v, n, idx, centers, radii):
    """Constraints for optimizing a single circle given fixed others."""
    x, y, r = v
    con = np.empty(n + 4)
    
    con[0] = x - r
    con[1] = 1.0 - x - r
    con[2] = y - r
    con[3] = 1.0 - y - r
    
    dx = centers[:, 0] - x
    dy = centers[:, 1] - y
    r_sum = radii + r
    
    con[4:] = dx**2 + dy**2 - r_sum**2
    con[4 + idx] = 1e9  # Ignore self-interaction
    return con

def generate_hex_init(row_counts, r0, n):
    """Generates an initial hexagonal lattice configuration."""
    pts = []
    y = r0
    for ri, cnt in enumerate(row_counts):
        shift = r0 if ri % 2 == 1 else 0.0
        width = (cnt - 1) * 2 * r0
        x_start = 0.5 - width / 2.0 + shift
        for k in range(cnt):
            if len(pts) >= n:
                break
            x = x_start + k * 2 * r0
            pts.append([x, y])
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
