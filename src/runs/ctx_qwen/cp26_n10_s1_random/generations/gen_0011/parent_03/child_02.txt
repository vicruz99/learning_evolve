import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers, n, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    b = np.empty(4*n + n*(n-1)//2)
    b[:n] = centers[:, 0]
    b[n:2*n] = 1.0 - centers[:, 0]
    b[2*n:3*n] = centers[:, 1]
    b[3*n:4*n] = 1.0 - centers[:, 1]
    idx = 4*n
    for i in range(n):
        for j in range(i+1, n):
            b[idx] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            idx += 1
    bounds = [(0.0, None)]*n
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun
    return np.zeros(n), 0.0

def obj_centers(c_flat, n, A_ub):
    """Objective for center optimization: negative sum of optimal radii."""
    c = c_flat.reshape(n, 2)
    r, s = solve_lp(c, n, A_ub)
    return -s

def joint_obj(v, n):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def joint_cons(v, n):
    """Inequality constraints >= 0 for valid packing in SLSQP."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    con = np.concatenate([cx-r, 1.0-cx-r, cy-r, 1.0-cy-r])
    triu_i, triu_j = np.triu_indices(n, 1)
    dx = cx[triu_i] - cx[triu_j]
    dy = cy[triu_i] - cy[triu_j]
    dr = r[triu_i] + r[triu_j]
    return np.concatenate([con, dx**2 + dy**2 - dr**2])

def run_packing():
    n = 26
    m_con = 4*n + n*(n-1)//2
    A_ub = np.zeros((m_con, n))
    for i in range(n):
        A_ub[i, i] = 1.0
        A_ub[n+i, i] = 1.0
        A_ub[2*n+i, i] = 1.0
        A_ub[3*n+i, i] = 1.0
    idx = 4*n
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1

    rng = np.random.default_rng(42)
    best_s = -1.0
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    configs = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4], 
                [6,6,5,5,4], [5,6,6,5,4], [7,6,6,7], [5,5,5,5,6]]
    for pat in patterns:
        pts = []
        y = 0.10
        for ri, cnt in enumerate(pat):
            shift = 0.10 if ri % 2 == 1 else 0.0
            x = 0.10 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 0.20
            y += 0.10 * np.sqrt(3)
        configs.append(np.array(pts[:n]))
        
    for _ in range(20):
        configs.append(rng.uniform(0.05, 0.95, (n, 2)))
        
    for _ in range(10):
        c = rng.uniform(0.05, 0.95, (n, 2))
        corners = [[0.1,0.1], [0.9,0.1], [0.1,0.9], [0.9,0.9]]
        c[:4] = corners + rng.uniform(-0.02, 0.02, (4,2))
        configs.append(np.clip(c, 0.01, 0.99))

    bounds_c = [(0.01, 0.99)] * (2*n)
    
    # Phase 1: Optimize centers from diverse starts
    for cfg in configs:
        c0 = np.clip(cfg, 0.02, 0.98)
        res = minimize(obj_centers, c0, args=(n, A_ub), method='L-BFGS-B', bounds=bounds_c,
                       options={'maxiter': 2500, 'ftol': 1e-13})
        if np.isfinite(res.fun):

            r_opt, s_opt = solve_lp(c_opt, n, A_ub)

                best_c = c_opt.copy()
                best_r = r_opt.copy()

                    r_j, s_j = solve_lp(c_j, n, A_ub)
                    if s_j > best_s:
                        best_s = s_j
                        best_c = c_j.copy()
                        best_r = r_j.copy()
            except Exception:
                pass

    # Phase 4: Strict safety scaling to guarantee numerical validity
    if best_c is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_c[i,0], best_c[i,1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                rs = best_r[i] + best_r[j]
                if rs > 1e-12:
                    scale = min(scale, d/rs)
        best_r *= scale * 0.9999999
        best_s = float(np.sum(best_r))
    else:
        best_c = np.random.uniform(0.1, 0.9, (n, 2))
        best_r = np.full(n, 0.05)
        best_s = float(np.sum(best_r))
        
    return best_c, best_r, best_s
