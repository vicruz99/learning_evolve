from scipy.optimize import linprog, minimize

def solve_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    m_b = 4 * n
    m_p = n * (n - 1) // 2
    A = np.zeros((m_b + m_p, n))
    b = np.zeros(m_b + m_p)
    idx = 0
    
    for i in range(n):
        x, y = centers[i]
        lims = (x, 1.0 - x, y, 1.0 - y)
        for lim in lims:
            A[idx, i] = 1.0
            b[idx] = lim
            idx += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback to geometrically valid radii if LP unexpectedly fails
    lims = np.min(np.stack([centers[:,0], 1.0-centers[:,0], centers[:,1], 1.0-centers[:,1]], axis=1), axis=1)
    diffs = centers[:, None, :] - centers[None, :, :]

    np.fill_diagonal(dists, np.inf)

    r_fall = np.minimum(np.maximum(lims, 1e-9), np.maximum(min_d, 1e-9)) * 0.9

def penalty_func(centers_flat, radii, n):

    pen = 0.0

    pen += np.sum(np.maximum(0.0, radii - (1.0 - x))**2)

    pen += np.sum(np.maximum(0.0, radii - (1.0 - y))**2)

    r_sum = radii[:, None] + radii[None, :]
    overlaps = np.maximum(0.0, r_sum - dists)
    pen += np.sum(overlaps**2)
    return pen

def relax_centers(centers, radii, n):
    """Minimize penalty function to resolve overlaps for given radii."""
    bounds = [(0.0, 1.0)] * (2 * n)
    try:
        res = minimize(penalty_func, centers.flatten(), args=(radii, n),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-12})
        return np.clip(res.x.reshape(n, 2), 1e-9, 1.0 - 1e-9)
    except Exception:
        return centers

def s_obj(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def s_cons(v, n, ii, jj):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[ii] - x[jj]
    dy = y[ii] - y[jj]
    dr = r[ii] + r[jj]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
