def solve_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    m = n * (n - 1) // 2
    A = np.zeros((4 * n + m, n))
    b = np.zeros(4 * n + m)
    idx = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_i = (x, 1.0 - x, y, 1.0 - y)
        for lim in bounds_i:
            A[idx, i] = 1.0
            b[idx] = max(lim, 1e-9)
            idx += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(d, 1e-9)
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    lims = np.min(np.stack([centers[:, 0], 1.0 - centers[:, 0], centers[:, 1], 1.0 - centers[:, 1]], axis=1), axis=1)

def obj_slsqp(v, n, ii, jj):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def cons_slsqp(v, n, ii, jj):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x = v[:n]
    y = v[n:2 * n]
    r = v[2 * n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[ii] - x[jj]
    dy = y[ii] - y[jj]
    dr = r[ii] + r[jj]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def penalty_obj(v, n, r_vec):
    """Smooth penalty function for center optimization with fixed radii."""
    c = v.reshape(n, 2)
    pen = 0.0
    diffs = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_sum = r_vec[:, None] + r_vec[None, :]
    overlap = np.maximum(0, r_sum - dists)
    pen += np.sum(overlap**2)
    wall_viol = np.maximum(0, r_vec - np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                                                  np.minimum(c[:, 1], 1.0 - c[:, 1])))
    pen += np.sum(wall_viol**2)
    return pen
