import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Maximum radius allowed by boundaries for each circle
    lims = np.min(np.stack([centers[:,0], 1.0-centers[:,0], centers[:,1], 1.0-centers[:,1]], axis=1), axis=1)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Construct LP: maximize sum(r) s.t. r_i <= lims[i], r_i + r_j <= dists[i,j], r_i >= 0
    m = n + n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    
    idx = 0
    for i in range(n):
        A[idx, i] = 1.0
        b[idx] = lims[i]
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success:
            r = np.maximum(res.x, 0.0)
            return r, np.sum(r)
    except Exception:
        pass
        
    # Fallback geometric radii if LP fails
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_f = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists_f, np.inf)
    min_d = np.min(dists_f, axis=1) / 2.0
    r_fall = np.minimum(lims, min_d) * 0.95
    return r_fall, np.sum(r_fall)

def local_search(centers, n, rng, iterations=2500):
    """Optimizes centers using coordinate ascent with Simulated Annealing."""
    curr_c = centers.copy()
    best_c = curr_c.copy()
    _, best_sum = solve_lp(curr_c)
    curr_sum = best_sum
    
    temp = 0.06
    step = 0.035
    
    for it in range(iterations):
        # Cooling schedule
        temp = 0.06 * (0.997 ** it)
        idx = rng.integers(n)
        old = curr_c[idx].copy()
        
        best_move = old
        best_val = curr_sum
        
        # Evaluate a few candidates to find a good move
        for _ in range(5):
            trial = old + rng.uniform(-step, step, 2)
            trial = np.clip(trial, 1e-4, 1.0 - 1e-4)
            curr_c[idx] = trial
            r_try, s_try = solve_lp(curr_c)
            if s_try > best_val:
                best_val = s_try
                best_move = trial.copy()
                
        curr_c[idx] = best_move
        delta = best_val - curr_sum
        
        # Simulated Annealing acceptance
        if delta > 0 or rng.random() < np.exp(delta / max(temp, 1e-9)):
            curr_sum = best_val
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_c = curr_c.copy()
        else:
            curr_c[idx] = old
            
        # Adaptive step decay
        step = max(0.002, step * 0.9992)
        
    return best_c, best_sum
