from scipy.optimize import minimize, linprog

def solve_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]),
                      np.minimum(centers[:,1], 1.0 - centers[:,1]))
    lims = np.maximum(lims, 0.0)
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    m = n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dists[i, j]
            idx += 1
            
    bounds = [(0.0, lims[i]) for i in range(n)]
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)) and np.all(res.x >= -1e-9):
            r = np.maximum(res.x, 0.0)
            return r, np.sum(r)
    except Exception:
        pass
    return None, -1.0

def obj_slsqp(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_slsqp(v, n, ii, jj):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[ii] - x[jj]
    dy = y[ii] - y[jj]
    dr = r[ii] + r[jj]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
