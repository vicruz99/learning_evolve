import numpy as np
from scipy.optimize import minimize, linprog

def compute_clearance(centers):
    """Computes the maximum feasible equal radius for given centers."""
    d_bound = np.min(np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                                np.minimum(centers[:, 1], 1.0 - centers[:, 1])))
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    d_pair = np.min(dists) / 2.0
    return min(d_bound, d_pair)

def neg_clearance(v):
    """Objective for center optimization: maximize min clearance."""
    n = 26
    centers = v.reshape(n, 2)
    return -compute_clearance(centers)

def solve_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    idx_i, idx_j = np.triu_indices(n, k=1)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    m_b = 4 * n
    m_p = n * (n - 1) // 2
    A = np.zeros((m_b + m_p, n))
    b = np.zeros(m_b + m_p)
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lims = np.array([x, 1.0 - x, y, 1.0 - y])
        A[4*i : 4*i+4, i] = 1.0
        b[4*i : 4*i+4] = lims
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    A[m_b:, idx_i] = 1.0
    A[m_b:, idx_j] = 1.0
    b[m_b:] = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback to geometrically valid radii if LP unexpectedly fails
    lims = np.min(np.stack([centers[:,0], 1.0-centers[:,0], centers[:,1], 1.0-centers[:,1]], axis=1), axis=1)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_mat = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists_mat, np.inf)
    min_d = np.min(dists_mat, axis=1) / 2.0
    r_fall = np.minimum(lims, min_d) * 0.9
    return r_fall, np.sum(r_fall)

def cons_slsqp(v, n):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c = np.concatenate([c, (dx**2 + dy**2 - dr**2)[mask].flatten()])
    return c

def obj_slsqp(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])
