import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.min(np.stack([centers[:,0], 1.0-centers[:,0], centers[:,1], 1.0-centers[:,1]], axis=1), axis=1)
    lims = np.maximum(lims, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in lims]
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    A_ub[:, idx_i] = 1.0
    A_ub[:, idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    r_fall = lims * 0.4
    return r_fall, np.sum(r_fall)

def sa_coordinate_ascent(centers, radii, n, rng, iters=2500):
    """Simulated annealing coordinate ascent with LP radius oracle."""
    curr_c = centers.copy()
    curr_r = radii.copy()
    curr_s = np.sum(curr_r)
    best_c = curr_c.copy()
    best_r = curr_r.copy()
    best_s = curr_s
    temp = 0.005
    
    for i in range(iters):
        idx = rng.integers(n)
        old = curr_c[idx].copy()
        step = max(0.0005, 0.04 * (1.0 - i / iters))
        trial = old + rng.uniform(-step, step, 2)
        trial = np.clip(trial, 0.01, 0.99)
        curr_c[idx] = trial
        
        r_try, s_try = solve_lp(curr_c, n)
        if r_try is None:
            curr_c[idx] = old
            continue
            
        delta = s_try - curr_s
        if delta > 0:
            accept = True
        else:
            accept = rng.random() < np.exp(delta / temp)

            curr_r = r_try.copy()
            if curr_s > best_s:
                best_s = curr_s
                best_c = curr_c.copy()
                best_r = curr_r.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9995
    return best_c, best_r, best_s

def obj_joint(v, n):
    return -np.sum(v[2*n:])

def cons_joint(v, n, ii, jj):
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[ii] - x[jj]
    dy = y[ii] - y[jj]
    dr = r[ii] + r[jj]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
