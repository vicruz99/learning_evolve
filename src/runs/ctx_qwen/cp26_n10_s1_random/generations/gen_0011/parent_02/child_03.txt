import numpy as np
from scipy.optimize import linprog, minimize

# Global constants for vectorized operations
N = 26
idx_i, idx_j = np.triu_indices(N, k=1)
M = len(idx_i)
A_ub_global = np.zeros((M, N))
A_ub_global[np.arange(M), idx_i] = 1.0
A_ub_global[np.arange(M), idx_j] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.min(np.stack([centers[:,0], 1.0-centers[:,0], centers[:,1], 1.0-centers[:,1]], axis=1), axis=1)
    lims = np.maximum(lims, 1e-9)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    bounds = [(0.0, l) for l in lims]
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_global, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    # Fallback geometric radii if LP fails
    min_d = np.min(dists, axis=1) / 2.0
    r_fall = np.minimum(lims, min_d) * 0.95
    return r_fall, np.sum(r_fall)

def generate_hex_configs(n, r0, row_counts):
    """Generates hexagonal lattice initial configurations."""
    configs = []
    for pattern in row_counts:
        pts = []
        y = r0
        for idx, cnt in enumerate(pattern):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        while len(pts) < n:
            pts.append([0.5, 0.5])
            
        cfg = np.array(pts[:n])
        # Normalize to comfortably fit inside [0.1, 0.9]
        mn, mx = cfg.min(axis=0), cfg.max(axis=0)
        span = mx - mn
        if span[0] > 1e-5: cfg[:, 0] = (cfg[:, 0] - mn[0]) / span[0] * 0.7 + 0.15
        if span[1] > 1e-5: cfg[:, 1] = (cfg[:, 1] - mn[1]) / span[1] * 0.7 + 0.15
        configs.append(cfg)
    return configs

def local_search_lp(centers, rng, max_iter=2500, init_step=0.035):
    """Greedy coordinate ascent with LP oracle."""
    curr_c = centers.copy()
    curr_r, curr_s = solve_lp(curr_c)
    best_c, best_r, best_s = curr_c.copy(), curr_r.copy(), curr_s
    
    step = init_step
    for it in range(max_iter):
        i = rng.integers(N)
        old_pos = curr_c[i].copy()
        best_move = old_pos
        best_val = curr_s
        
        # Sample multiple moves for current circle
        for _ in range(5):
            trial = old_pos + rng.uniform(-step, step, 2)
            trial = np.clip(trial, 0.02, 0.98)
            curr_c[i] = trial
            r_t, s_t = solve_lp(curr_c)
            if s_t > best_val:
                best_val = s_t
                best_move = trial.copy()
                
        curr_c[i] = best_move
        curr_r, curr_s = solve_lp(curr_c)
        
        if curr_s > best_s + 1e-10:
            best_c, best_r, best_s = curr_c.copy(), curr_r.copy(), curr_s
            step = max(step * 0.9992, 0.0008)
        else:
            # Occasional acceptance of worse moves to escape plateaus
            if rng.random() < 0.02:
                best_c, best_r, best_s = curr_c.copy(), curr_r.copy(), curr_s
            else:
                curr_c[i] = old_pos
        step *= 0.9996
    return best_c, best_r, best_s

def s_obj(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def s_cons(v):
    """Inequality constraints >= 0 for SLSQP."""
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    c = np.concatenate([cx-r, 1.0-cx-r, cy-r, 1.0-cy-r])
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def polish_slsqp(centers, radii):
    """Joint optimization of centers and radii using SLSQP."""
    v0 = np.concatenate([centers[:,0], centers[:,1], radii])
    bounds = [(0.0, 1.0)]*(2*N) + [(1e-7, 0.5)]*N
    try:
        res = minimize(s_obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': s_cons},
                       options={'maxiter': 3000, 'ftol': 1e-14})
        if res.success or np.isfinite(res.fun):
            cx, cy, r = res.x[:N], res.x[N:2*N], res.x[2*N:]
            return np.column_stack((cx, cy)), np.maximum(r, 0.0)
    except Exception:
        pass
    return centers, radii

def run_packing():

        [7,6,6,7], [8,6,6,6], [5,5,5,5,5,1], [6,5,5,6,4]
    ]
    
    inits = generate_hex_configs(N, 0.10, row_patterns)
    
    # Add rotated variants to break axis-aligned traps
    for base in inits[:5]:
        for angle in [0.08, -0.08, 0.15, -0.15]:

            cos_a, sin_a = np.cos(angle), np.sin(angle)
            c_rot = np.column_stack([c[:,0]*cos_a - c[:,1]*sin_a, c[:,0]*sin_a + c[:,1]*cos_a]) + 0.5
            c_rot = np.clip(c_rot, 0.05, 0.95)
            inits.append(c_rot)
            
    # Random uniform starts
    for _ in range(10):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    best_sum = -1.0
    best_c, best_r = None, None
    
    for cfg in inits:
        # Phase 1: LP-guided coordinate ascent
        c_opt, r_opt, s_opt = local_search_lp(cfg, rng, max_iter=2000, init_step=0.03)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c, best_r = c_opt, r_opt
            
        # Phase 2: SLSQP joint polish
        c_pol, r_pol = polish_slsqp(best_c, best_r)
        # LP can often extract slightly larger radii from SLSQP centers
        _, s_pol = solve_lp(c_pol)
        if s_pol > best_sum:
            best_sum = s_pol
            best_c, best_r = c_pol, solve_lp(c_pol)[0]
            
    # Fallback safety net
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum = solve_lp(best_c)
        
    # Phase 3: Strict numerical safety scaling
    scale = 1.0
    for i in range(N):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999995
    final_sum = float(np.sum(best_r))
    
    return best_c, best_r, final_sum
