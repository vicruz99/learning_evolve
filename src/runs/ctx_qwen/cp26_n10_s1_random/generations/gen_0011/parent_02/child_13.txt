import numpy as np
from scipy.optimize import linprog

def solve_lp_radii(centers, n, ii, jj):
    """Solves LP to maximize sum of radii for fixed centers."""
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 0.0)
    
    m_p = len(ii)
    A_ub = np.zeros((m_p + n, n))
    b_ub = np.zeros(m_p + n)
    
    A_ub[:m_p, ii] = 1.0
    A_ub[:m_p, jj] = 1.0
    b_ub[:m_p] = dists[ii, jj]
    
    A_ub[m_p:, :] = np.eye(n)
    b_ub[m_p:] = lims
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
        
    lims_min = np.minimum(lims, np.min(dists, axis=1) / 2.0)
    return lims_min * 0.9, np.sum(lims_min * 0.9)

    ]
    
    for pat in patterns:

            continue

                pts.append([x, y])
                x += 2.0 * r0

        if span[1] > 1e-6:

        configs.append(cfg)

        for _ in range(3):
            p = cfg + rng.uniform(-0.02, 0.02, cfg.shape)
            configs.append(np.clip(p, 0.03, 0.97))
