import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, and dual marginals."""
    m_b = 4 * n
    m_p = n * (n - 1) // 2
    A = np.zeros((m_b + m_p, n))
    b = np.zeros(m_b + m_p)
    idx = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        lims = (x, 1.0 - x, y, 1.0 - y)
        for lim in lims:
            A[idx, i] = 1.0
            b[idx] = lim
            idx += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success and np.isfinite(res.fun):
            r = res.x
            s = -res.fun
            marg = np.zeros(m_b + m_p)
            if hasattr(res, 'ineqlin') and res.ineqlin is not None and res.ineqlin.marginals is not None:
                marg = res.ineqlin.marginals.copy()
            # Ensure non-negative marginals despite numerical noise
            marg = np.maximum(marg, 0.0)
            return r, s, marg
    except Exception:
        pass
        
    # Fallback to geometrically valid radii if LP unexpectedly fails
    lims = np.min(np.stack([centers[:,0], 1.0-centers[:,0], centers[:,1], 1.0-centers[:,1]], axis=1), axis=1)
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_d = np.min(dists, axis=1) / 2.0
    r_fall = np.minimum(lims, min_d) * 0.9
    return r_fall, np.sum(r_fall), np.zeros(m_b + m_p)

def compute_grad(centers, marg, n):
    """Computes gradient of sum of radii w.r.t centers using LP dual variables."""
    grad = np.zeros((n, 2))
    idx = 0
    for i in range(n):
        # Boundary contributions: shadow prices indicate push directions
        grad[i, 0] += marg[4*i] - marg[4*i+1]
        grad[i, 1] += marg[4*i+2] - marg[4*i+3]
        
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.hypot(dx, dy)
            mu = marg[4*n + idx]
            if mu > 1e-12 and d > 1e-12:
                # Pairwise constraint gradient: pushes centers apart proportional to tightness
                factor = mu / d
                grad[i, 0] += factor * dx
                grad[i, 1] += factor * dy
                grad[j, 0] -= factor * dx
                grad[j, 1] -= factor * dy
            idx += 1
    return grad

def obj_slsqp(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_slsqp(v, n, ii, jj):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[ii] - x[jj]
    dy = y[ii] - y[jj]
    dr = r[ii] + r[jj]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
