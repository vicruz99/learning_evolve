import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    m_b = 4 * n
    m_p = n * (n - 1) // 2
    A = np.zeros((m_b + m_p, n))
    b = np.zeros(m_b + m_p)
    idx = 0
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lims = (x, 1.0 - x, y, 1.0 - y)
        for lim in lims:
            A[idx, i] = 1.0
            b[idx] = lim
            idx += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback to geometrically valid radii if LP unexpectedly fails
    lims = np.min(np.stack([centers[:,0], 1.0-centers[:,0], centers[:,1], 1.0-centers[:,1]], axis=1), axis=1)
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_d = np.min(dists, axis=1) / 2.0
    r_fall = np.minimum(lims, min_d) * 0.9
    return r_fall, np.sum(r_fall)

def obj_eq(v, n):
    """Objective for equal-radius optimization: minimize negative shared radius t."""
    return -v[-1]

def cons_eq(v, n):
    """Inequality constraints >= 0 for equal-radius packing."""
    t = v[-1]
    cx = v[:n]
    cy = v[n:2*n]
    
    # Boundary constraints
    c = np.concatenate([cx - t, 1.0 - cx - t, cy - t, 1.0 - cy - t])
    
    # Pairwise non-overlap: dist(i,j) >= 2t
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(d, np.inf)
    triu_idx = np.triu_indices(n, k=1)
    return np.concatenate([c, d[triu_idx] - 2.0 * t])

def obj_centers_lp(c_flat, n):
    """Objective for center optimization: minimize negative LP sum of radii."""
    c = c_flat.reshape(n, 2)
    c = np.clip(c, 1e-4, 1.0 - 1e-4)
    _, s = solve_lp(c, n)
    return -s

            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)

    return best_c, best_r, float(np.sum(best_r))

def get_current_lp_sum(centers, n):
    """Helper to quickly evaluate LP sum during coordinate ascent."""
    _, s = solve_lp(centers, n)
    return s
