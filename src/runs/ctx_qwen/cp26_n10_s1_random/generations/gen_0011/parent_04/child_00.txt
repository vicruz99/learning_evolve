import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A = np.zeros((m, n))
    A[np.arange(m), idx_i] = 1.0
    A[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b = dists[idx_i, idx_j]
    
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def obj_func(x_flat):
    """Objective function for center optimization: negative sum of LP radii."""
    c = x_flat.reshape(N_CIRCLES, 2)
    _, s = solve_lp(c)
    return -s

def generate_configs(rng):
    """Generates diverse hexagonal and random initial configurations."""
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,5,5], [5,5,6,5,5],
        [6,6,5,5,4], [5,6,6,5,4], [5,5,5,5,6], [6,5,5,5,5],
        [7,6,6,7], [6,7,6,7], [5,6,5,6,5,1], [4,6,5,6,5],
        [8,5,6,7], [9,4,6,7], [8,6,6,6], [7,7,6,6]
    ]
    
    for pat in patterns:
        if sum(pat) != N_CIRCLES: continue
        pts = []
        r0 = 0.098
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N_CIRCLES: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:N_CIRCLES])
        
        # Center and scale to fit comfortably inside [0,1]
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        if span[0] > 1e-4 and span[1] > 1e-4:
            pts = (pts - mn) / span * 0.85 + 0.075
        else:
            pts = np.full((N_CIRCLES, 2), 0.5)
            
        configs.append(pts)
        for _ in range(2):
            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(5):
        configs.append(rng.uniform(0.2, 0.8, (N_CIRCLES, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = generate_configs(rng)
    
    # Phase 1: Multi-start Nelder-Mead optimization on centers
    for cfg in inits:
        x0 = cfg.flatten()
        try:
            res = minimize(obj_func, x0, method='Nelder-Mead', 
                           options={'maxiter': 2500, 'xatol': 1e-8, 'fatol': 1e-10})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(N_CIRCLES, 2)
                r_opt, s_opt = solve_lp(c_opt)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass

    if best_centers is None:
        best_centers = inits[0]
        best_radii, best_sum = solve_lp(best_centers)

    # Phase 2: Local perturbation hill-climbing to polish
    curr_c = best_centers.copy()
    curr_r, curr_s = solve_lp(curr_c)
    step = 0.015
    
    for it in range(800):
        idx = rng.integers(N_CIRCLES)
        old = curr_c[idx].copy()
        
        best_move_s = curr_s
        best_move = None
        
        for _ in range(4):
            move = rng.uniform(-step, step, 2)
            new_pos = curr_c[idx] + move
            new_pos = np.clip(new_pos, 1e-4, 1.0 - 1e-4)
            curr_c[idx] = new_pos
            r_try, s_try = solve_lp(curr_c)
            if s_try > best_move_s:
                best_move_s = s_try
                best_move = new_pos.copy()
                
        if best_move is not None:
            curr_c[idx] = best_move
            curr_r, curr_s = solve_lp(curr_c)
            if curr_s > best_sum:
                best_sum = curr_s
                best_centers = curr_c.copy()
                best_radii = curr_r.copy()
        else:
            curr_c[idx] = old
            
        step *= 0.998
        
    # Phase 3: Strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N_CIRCLES):
        x, y, r = best_centers[i,0], best_centers[i,1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
