import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def objective_func(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N_CIRCLES:])

def constraint_func(v):
    """Inequality constraints >= 0 for valid packing."""
    x = v[:N_CIRCLES]
    y = v[N_CIRCLES:2*N_CIRCLES]
    r = v[2*N_CIRCLES:]
    
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    idx_i, idx_j = np.triu_indices(N_CIRCLES, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    
    # Squared distance constraint avoids sqrt and is smooth
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def relax_centers(centers, radii, steps=150):
    """Force-directed relaxation to spread circles apart."""
    n = centers.shape[0]
    pos = centers.copy()
    
    for _ in range(steps):
        forces = np.zeros_like(pos)
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = pos[i] - pos[j]
                dist = np.hypot(diff[0], diff[1])
                if dist < 1e-9:
                    dist = 1e-9
                    diff = np.array([1e-6, 1e-6])
                
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    strength = overlap * 15.0 / dist
                    forces[i] += diff * strength
                    forces[j] -= diff * strength
                    
        # Boundary repulsion
        for i in range(n):
            r = radii[i]
            if pos[i, 0] < r:
                forces[i, 0] += (r - pos[i, 0]) * 15.0
            if pos[i, 0] > 1.0 - r:
                forces[i, 0] -= (pos[i, 0] - (1.0 - r)) * 15.0
            if pos[i, 1] < r:
                forces[i, 1] += (r - pos[i, 1]) * 15.0
            if pos[i, 1] > 1.0 - r:
                forces[i, 1] -= (pos[i, 1] - (1.0 - r)) * 15.0
                
        pos += forces * 0.005
        pos = np.clip(pos, radii, 1.0 - radii)
        
    return pos

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [6,6,4,6,4], 
        [5,5,6,5,5], [6,4,6,4,6], [5,6,6,5,4], [6,5,4,6,5],
        [5,5,5,5,6], [6,6,6,4,4], [4,5,6,5,6], [5,4,6,5,6],
        [7,6,6,7], [6,7,6,7], [5,6,5,6,5,1], [5,5,5,6,5],
        [8,5,6,7], [7,7,6,6], [9,5,6,6], [8,6,6,6],
        [5,5,5,5,5,1], [6,5,5,5,5], [5,6,4,6,5], [4,7,6,5,4]
    ]
    
    for pat in patterns:
        if sum(pat) != N_CIRCLES: continue
        pts = []
        r0 = 0.098
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N_CIRCLES: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:N_CIRCLES])
        
        # Center and scale to fit comfortably inside [0.05, 0.95]
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        if span[0] > 1e-4 and span[1] > 1e-4:
            pts = (pts - mn) / span * 0.80 + 0.10
        else:
            pts = np.full((N_CIRCLES, 2), 0.5)
            
        inits.append(pts)
        for _ in range(3):
            p = pts + rng.uniform(-0.012, 0.012, pts.shape)
            inits.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    bounds_vars = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(1e-6, 0.5)] * N_CIRCLES
    constraints = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = generate_inits(rng)
    
    # Phase 1: Force Relaxation + SLSQP Multi-start
    for cfg in inits:
        # Estimate initial radii for relaxation
        r_est = np.full(N_CIRCLES, 0.09)
        for i in range(N_CIRCLES):
            dw = min(cfg[i,0], 1.0-cfg[i,0], cfg[i,1], 1.0-cfg[i,1])
            dists_i = np.sqrt(np.sum((cfg - cfg[i])**2, axis=1))
            dists_i[i] = np.inf
            dn = np.min(dists_i)
            r_est[i] = min(dw, dn/2.0) * 0.8
            
        # Relax to resolve overlaps
        relaxed_cfg = relax_centers(cfg, r_est, steps=100)
        relaxed_cfg = np.clip(relaxed_cfg, 1e-4, 1.0 - 1e-4)

        x0 = np.zeros(3 * N_CIRCLES)
        x0[:N_CIRCLES] = relaxed_cfg[:, 0]
        x0[N_CIRCLES:2*N_CIRCLES] = relaxed_cfg[:, 1]

            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_vars,
                           constraints=constraints, options={'maxiter': 5000, 'ftol': 1e-13})

                r_lp, s_lp = solve_lp(c_mat)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c_mat.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass

        pass

            rs = best_radii[i] + best_radii[j]

                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999998
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
