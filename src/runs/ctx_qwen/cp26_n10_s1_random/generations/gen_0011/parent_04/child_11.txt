import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution
import math

N_CIRCLES = 26

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-7), 0.0

def eval_sum_radii(centers_flat):
    """Objective for center optimization: negative sum of radii from LP."""
    centers = centers_flat.reshape(-1, 2)
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    _, s = solve_lp(centers)
    return -s

def cons_slqp(v):
    """Inequality constraints >= 0 for valid packing."""
    n = N_CIRCLES
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def obj_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N_CIRCLES:])

def generate_inits(rng, n):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5],
        [4,6,6,5,5], [6,5,4,6,5], [5,7,4,5,5], [7,5,5,5,4],
        [5,5,5,5,6], [6,6,6,4,4], [4,5,6,5,6], [5,4,6,5,6],
        [8,5,6,7], [7,7,6,6], [9,5,6,6], [8,6,6,6],
        [6,6,5,4,5], [5,5,5,6,5]
    ]
    for pat in patterns:
        if sum(pat) != n: continue
        pts = []
        r0 = 0.098
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:n])
        
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        if span[0] > 1e-4 and span[1] > 1e-4:
            pts = (pts - mn) / span * 0.7 + 0.15
        else:
            pts = np.full((n, 2), 0.5)
        inits.append(pts)
        
    for _ in range(10):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    rng = np.random.default_rng(123)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    inits = generate_inits(rng, n)
    
    # Phase 1: Evaluate all inits with LP
    for cfg in inits:
        r, s = solve_lp(cfg)
        if s > best_sum:
            best_sum = s
            best_centers = cfg.copy()
            best_radii = r.copy()
            
    # Phase 2: Nelder-Mead optimization on centers
    for cfg in inits:
        try:
            res = minimize(eval_sum_radii, cfg.flatten(), method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-8, 'fatol': 1e-9})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(n, 2)
                c_opt = np.clip(c_opt, 1e-5, 1.0 - 1e-5)
                r_opt, s_opt = solve_lp(c_opt)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Phase 3: Differential Evolution for global search
    try:
        res_de = differential_evolution(eval_sum_radii, bounds=[(0.05, 0.95)]*(2*n),
                                        popsize=12, maxiter=30, mutation=(0.5, 1.0),
                                        recombination=0.9, seed=42, polish=False, workers=1)
        if np.isfinite(res_de.fun):
            c_de = res_de.x.reshape(n, 2)
            c_de = np.clip(c_de, 1e-5, 1.0 - 1e-5)
            r_de, s_de = solve_lp(c_de)
            if s_de > best_sum:
                best_sum = s_de
                best_centers = c_de.copy()
                best_radii = r_de.copy()
    except Exception:
        pass

    # Phase 4: Local Hill Climbing / Simulated Annealing on centers
    curr_c = best_centers.copy()
    curr_r, curr_s = solve_lp(curr_c)
    step = 0.02
    
    for it in range(2500):
        idx = rng.integers(n)
        old = curr_c[idx].copy()
        
        best_move = None
        best_move_s = curr_s
        
        for _ in range(5):
            move = rng.uniform(-step, step, 2)
            new_c = curr_c[idx] + move
            new_c = np.clip(new_c, 1e-4, 1.0 - 1e-4)
            curr_c[idx] = new_c
            
            r_try, s_try = solve_lp(curr_c)
            if s_try > best_move_s or rng.random() < math.exp((s_try - best_move_s) / max(step, 1e-6)):
                best_move_s = s_try
                best_move = new_c.copy()
                
        if best_move is not None:
            curr_c[idx] = best_move
            curr_r, curr_s = solve_lp(curr_c)
            if curr_s > best_sum:
                best_sum = curr_s
                best_centers = curr_c.copy()
                best_radii = curr_r.copy()
        else:
            curr_c[idx] = old
            
        step *= 0.9992
        
    # Phase 5: Joint SLSQP Polish
    bounds_vars = [(0.0, 1.0)]*(2*n) + [(1e-7, 0.5)]*n
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    x0 = np.zeros(3*n)
    x0[:n] = best_centers[:,0]
    x0[n:2*n] = best_centers[:,1]
    x0[2*n:] = best_radii * 0.95
    
    try:
        res = minimize(obj_slqp, x0, method='SLSQP', bounds=bounds_vars,
                       constraints=cons_dict, options={'maxiter': 8000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            c_mat = np.column_stack((res.x[:n], res.x[n:2*n]))
            r_lp, s_lp = solve_lp(c_mat)
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = c_mat.copy()
                best_radii = r_lp.copy()
    except Exception:
        pass

    # Phase 6: Safety Scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i,0], best_centers[i,1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
