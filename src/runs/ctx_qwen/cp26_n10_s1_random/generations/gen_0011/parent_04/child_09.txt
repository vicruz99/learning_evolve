import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def obj_equal(v):
    """Objective: minimize negative shared radius t."""
    return -v[-1]

def cons_equal(v, n):
    """Inequality constraints >= 0 for valid equal-radius packing."""
    t = v[-1]
    x = v[:n]
    y = v[n:2*n]
    
    c = np.concatenate([x - t, 1.0 - x - t, y - t, 1.0 - y - t])
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - 4.0 * t**2])
    return c

def generate_hex(row_counts, r_guess, n):
    """Generates an initial hexagonal grid with specified row counts."""
    pts = []
    y = r_guess
    for i, cnt in enumerate(row_counts):
        shift = r_guess if i % 2 == 1 else 0.0
        width = (cnt - 1) * 2 * r_guess
        x_start = 0.5 - width / 2.0 + shift
        for k in range(cnt):
            if len(pts) >= n: break
            pts.append([x_start + k * 2 * r_guess, y])
        y += np.sqrt(3) * r_guess
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Diverse row distributions summing to 26
    row_pats = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 4, 6, 4, 6], [5, 6, 6, 5, 4], [6, 5, 4, 6, 5],
        [7, 6, 6, 7], [6, 7, 6, 7], [5, 5, 5, 6, 5], [6, 6, 4, 6, 4]
    ]
    
    inits = []
    for pat in row_pats:
        if sum(pat) != N: continue
        c = generate_hex(pat, 0.095, N)
        inits.append(c)
        for _ in range(3):
            p = c + rng.uniform(-0.02, 0.02, c.shape)
            inits.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(8):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    bounds_eq = [(0.0, 1.0)] * (2 * N) + [(0.08, 0.15)]
    
    # Phase 1: Optimize equal radius t
    for cfg in inits:
        x0 = np.concatenate([cfg.flatten(), [0.095]])
        try:
            res = minimize(obj_equal, x0, args=(N,), method='SLSQP', bounds=bounds_eq,
                           constraints={'type': 'ineq', 'fun': cons_equal, 'args': (N,)},
                           options={'maxiter': 15000, 'ftol': 1e-14})
            if np.isfinite(res.fun) and res.x[-1] > 0.09:
                c_opt = res.x[:2 * N].reshape(N, 2)
                r_lp, s_lp = solve_lp(c_opt)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_c = c_opt.copy()
                    best_r = r_lp.copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum = solve_lp(best_c)
        
    # Phase 2: Local hill-climbing on centers evaluated via LP
    curr_c = best_c.copy()
    curr_r, curr_s = solve_lp(curr_c)
    step = 0.015
    
    for it in range(2000):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        best_move_s = curr_s
        best_move = None
        
        for _ in range(4):
            move = rng.uniform(-step, step, 2)
            new_pos = curr_c[idx] + move
            new_pos = np.clip(new_pos, 1e-4, 1.0 - 1e-4)
            curr_c[idx] = new_pos
            
            r_try, s_try = solve_lp(curr_c)
            if s_try > best_move_s:
                best_move_s = s_try
                best_move = new_pos.copy()
                
        if best_move is not None:
            curr_c[idx] = best_move
            curr_r, curr_s = solve_lp(curr_c)
            if curr_s > best_sum:
                best_sum = curr_s
                best_c = curr_c.copy()
                best_r = curr_r.copy()
        else:
            curr_c[idx] = old
            
        step *= 0.9992
        
    # Phase 3: Joint SLSQP Polish on centers and radii
    x0 = np.zeros(3 * N)
    x0[:N] = best_c[:, 0]
    x0[N:2 * N] = best_c[:, 1]
    x0[2 * N:] = best_r * 0.995
    
    def obj_joint(v):
        return -np.sum(v[2 * N:])

    def cons_joint(v, n):
        x = v[:n]
        y = v[n:2 * n]
        r = v[2 * n:]
        c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
        idx_i, idx_j = np.triu_indices(n, k=1)
        dx = x[idx_i] - x[idx_j]
        dy = y[idx_i] - y[idx_j]
        dr = r[idx_i] + r[idx_j]
        c = np.concatenate([c, dx**2 + dy**2 - dr**2])
        return c

    try:
        res = minimize(obj_joint, x0, args=(N,), method='SLSQP', 
                       bounds=[(0.0, 1.0)]*(2*N) + [(1e-7, 0.5)]*N,
                       constraints={'type': 'ineq', 'fun': cons_joint, 'args': (N,)},
                       options={'maxiter': 10000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            c_mat = np.column_stack((res.x[:N], res.x[N:2*N]))
            r_lp, s_lp = solve_lp(c_mat)
            if s_lp > best_sum:
                best_sum = s_lp
                best_c = c_mat.copy()
                best_r = r_lp.copy()
    except Exception:
        pass

    # Phase 4: Strict Safety Scaling
    scale = 1.0
    for i in range(N):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_r *= scale * 0.999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
