import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, l) for l in lims]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def obj_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_slqp(v):
    """Inequality constraints >= 0 for valid packing."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    idx_i, idx_j = np.triu_indices(N, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def generate_hex(row_counts, r0, n):
    """Generates a hexagonal lattice initialization with specified row counts."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        width = (cnt - 1) * 2 * r0
        x_start = 0.5 - width / 2.0 + shift
        for k in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x_start + k * 2 * r0, y])
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def get_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [5,6,6,5,4], [5,5,5,5,6], [6,5,5,5,5],
        [5,5,5,6,5], [6,5,4,6,5], [4,5,6,5,6], [5,4,6,5,6],
        [7,6,6,7], [6,7,6,7], [5,6,5,6,5,1]
    ]
    for pat in patterns:
        if sum(pat) != N:
            continue
        pts = generate_hex(pat, 0.098, N)
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        if span[0] > 1e-4 and span[1] > 1e-4:
            pts = (pts - mn) / span * 0.75 + 0.125
        inits.append(pts)
        for _ in range(3):
            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            inits.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(5):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
    return inits

def run_sa_lp(centers, rng, iterations=8000):
    """Simulated Annealing on centers evaluated via LP."""
    curr_c = centers.copy()
    curr_r, curr_s = solve_lp(curr_c)
    best_c, best_r, best_s = curr_c.copy(), curr_r.copy(), curr_s
    
    temp = 0.04
    step = 0.035
    
    for it in range(iterations):
        decay = 0.9996
        temp *= decay
        step *= decay
        
        idx = rng.integers(N)
        old_c = curr_c[idx].copy()
        
        best_move_c = old_c
        best_move_s = curr_s
        
        for _ in range(4):
            move = rng.uniform(-step, step, 2)
            cand_c = old_c + move
            cand_c = np.clip(cand_c, 1e-4, 1.0 - 1e-4)
            curr_c[idx] = cand_c
            
            r_try, s_try = solve_lp(curr_c)
            if s_try > best_move_s:
                best_move_s = s_try
                best_move_c = cand_c.copy()
            elif rng.random() < np.exp((s_try - curr_s) / max(temp, 1e-9)):
                best_move_c = cand_c.copy()
                best_move_s = s_try
                
        curr_c[idx] = best_move_c
        curr_r, curr_s = solve_lp(curr_c)
        
        if curr_s > best_s:
            best_s = curr_s
            best_c = curr_c.copy()
            best_r = curr_r.copy()
            
    return best_c, best_r, best_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_vars = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = get_inits(rng)
    
    # Phase 1: SLSQP Joint Optimization from diverse starts
    for cfg in inits:
        x0 = np.zeros(3 * N)
        x0[:N] = cfg[:, 0]
        x0[N:2*N] = cfg[:, 1]
        
        r_lp, s_lp = solve_lp(cfg)
        x0[2*N:] = np.maximum(r_lp * 0.95, 1e-4)
        
        try:
            res = minimize(obj_slqp, x0, method='SLSQP', bounds=bounds_vars,
                           constraints={'type': 'ineq', 'fun': cons_slqp},
                           options={'maxiter': 4000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_mat = np.column_stack((res.x[:N], res.x[N:2*N]))
                r_pol, s_pol = solve_lp(c_mat)
                if s_pol > best_sum:
                    best_sum = s_pol
                    best_centers = c_mat.copy()
                    best_radii = r_pol.copy()
        except Exception:
            pass
            
    if best_centers is None:
        best_centers = inits[0]
        best_radii, best_sum = solve_lp(best_centers)
        
    # Phase 2: Simulated Annealing on Centers with LP evaluation
    curr_c, curr_r, curr_s = run_sa_lp(best_centers, rng, iterations=8000)
    if curr_s > best_sum:
        best_sum = curr_s
        best_centers = curr_c.copy()
        best_radii = curr_r.copy()
        
    # Phase 3: Final Joint SLSQP Polish from SA best
    x0 = np.zeros(3 * N)
    x0[:N] = best_centers[:, 0]
    x0[N:2*N] = best_centers[:, 1]
    x0[2*N:] = np.maximum(best_radii * 0.98, 1e-4)
    
    try:
        res = minimize(obj_slqp, x0, method='SLSQP', bounds=bounds_vars,
                       constraints={'type': 'ineq', 'fun': cons_slqp},
                       options={'maxiter': 8000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            c_mat = np.column_stack((res.x[:N], res.x[N:2*N]))
            r_pol, s_pol = solve_lp(c_mat)
            if s_pol > best_sum:
                best_sum = s_pol
                best_centers = c_mat.copy()
                best_radii = r_pol.copy()
    except Exception:
        pass
        
    # Phase 4: Strict Safety Scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], 
                         best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
