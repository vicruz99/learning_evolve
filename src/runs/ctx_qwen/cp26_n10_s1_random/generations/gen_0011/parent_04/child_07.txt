import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]),
                      np.minimum(centers[:,1], 1.0 - centers[:,1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def compute_res_eq(v):
    """Computes inequality constraints for equal-radius optimization."""
    n = N_CIRCLES
    x = v[:n]
    y = v[n:2*n]
    t = v[2*n]
    res = np.empty(4*n + n*(n-1)//2)
    res[:n] = x - t
    res[n:2*n] = 1.0 - x - t
    res[2*n:3*n] = y - t
    res[3*n:4*n] = 1.0 - y - t
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    idx = np.triu_indices(n, k=1)
    res[4*n:] = dists[idx] - 2.0*t
    return res

def obj_eq(v):
    """Objective for equal-radius optimization: maximize t."""
    return -v[2*N_CIRCLES]

def cons_j(v):
    """Inequality constraints for joint center-radius optimization."""
    n = N_CIRCLES
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def obj_j(v):
    """Objective for joint optimization: maximize sum of radii."""
    return -np.sum(v[2*N_CIRCLES:])

def generate_inits(rng):
    """Generates diverse hexagonal initial configurations."""
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4], 
            [6,6,5,5,4], [5,6,6,5,4], [5,5,5,5,6], [6,5,5,5,5],
            [7,5,6,8], [6,6,6,8], [8,6,6,6], [9,5,6,6], [5,7,5,6,3]]
    for pat in pats:
        if sum(pat) != N_CIRCLES: continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx%2==1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N_CIRCLES: break
                pts.append([x, y])
                x += 2.0*r0
            y += np.sqrt(3)*r0
        pts = np.array(pts[:N_CIRCLES])
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        if span[0]>1e-4 and span[1]>1e-4:
            pts = (pts - mn) / span * 0.82 + 0.09
        inits.append(pts)
        for _ in range(2):
            p = pts + rng.uniform(-0.012, 0.012, pts.shape)
            inits.append(np.clip(p, 0.05, 0.95))
    for _ in range(5):
        inits.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)))
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    inits = generate_inits(rng)
    
    best_t = 0.0
    best_c_eq = None
    
    bounds_eq = [(0.0, 1.0)]*(2*N_CIRCLES) + [(0.08, 0.11)]
    cons_eq = {'type': 'ineq', 'fun': compute_res_eq}
    
    # Phase 1: Optimize geometry using equal-radius assumption
    for cfg in inits:
        x0 = np.concatenate([cfg.flatten(), [0.09]])
        try:
            res = minimize(obj_eq, x0, method='SLSQP', bounds=bounds_eq,
                           constraints=cons_eq, options={'maxiter': 8000, 'ftol': 1e-14})
            if np.isfinite(res.fun) and -res.fun > best_t:
                best_t = -res.fun
                best_c_eq = res.x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
        except Exception:
            continue
            
    if best_c_eq is None:
        best_c_eq = inits[0]
        
    # Phase 2: Extract maximum sum radii via LP
    curr_c = best_c_eq
    curr_r, curr_s = solve_lp(curr_c)
    
    bounds_j = [(0.0, 1.0)]*(2*N_CIRCLES) + [(1e-6, 0.5)]*N_CIRCLES
    cons_j_dict = {'type': 'ineq', 'fun': cons_j}
    
    # Phase 3: Iterative joint optimization and LP refinement
    for iteration in range(3):
        x0 = np.concatenate([curr_c.flatten(), curr_r * 0.96])
        try:
            res_j = minimize(obj_j, x0, method='SLSQP', bounds=bounds_j,
                             constraints=cons_j_dict, options={'maxiter': 6000, 'ftol': 1e-13})
            if np.isfinite(res_j.fun):
                new_c = np.column_stack((res_j.x[:N_CIRCLES], res_j.x[N_CIRCLES:2*N_CIRCLES]))
                new_r, new_s = solve_lp(new_c)
                if new_s > curr_s:
                    curr_c = new_c
                    curr_r = new_r
                    curr_s = new_s
        except Exception:
            pass
            
    best_centers = curr_c
    best_radii = curr_r
    best_sum = curr_s
    
    # Phase 4: Strict Safety Scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N_CIRCLES):
        x, y, r = best_centers[i,0], best_centers[i,1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
