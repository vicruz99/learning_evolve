from scipy.optimize import minimize, linprog
from operator import itemgetter

N_CIRCLES = 26
IDX_I, IDX_J = np.triu_indices(N_CIRCLES, k=1)
M_PAIR = len(IDX_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]),
                      np.minimum(centers[:,1], 1.0 - centers[:,1]))
    lims = np.maximum(lims, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, l) for l in lims]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    A_ub = np.zeros((M_PAIR, n))
    A_ub[np.arange(M_PAIR), IDX_I] = 1.0
    A_ub[np.arange(M_PAIR), IDX_J] = 1.0
    b_ub = dists[IDX_I, IDX_J]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def obj_centers_lp(v):
    """Objective for center optimization: minimize negative sum of LP radii."""
    c = v.reshape(N_CIRCLES, 2)
    c = np.clip(c, 1e-4, 1.0 - 1e-4)
    _, s = solve_lp(c)
    return -s

def obj_slqp(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N_CIRCLES:])

def cons_slqp(v):
    """Inequality constraints >= 0 for valid joint packing."""
    x = v[:N_CIRCLES]
    y = v[N_CIRCLES:2 * N_CIRCLES]
    r = v[2 * N_CIRCLES:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[IDX_I] - x[IDX_J]
    dy = y[IDX_I] - y[IDX_J]
    dr = r[IDX_I] + r[IDX_J]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
