import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_constraints_eq(vars_arr):
    """Computes inequality constraints >= 0 for equal-radius packing."""
    n = N
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    t = vars_arr[2*n]
    
    c = []
    # Boundary constraints
    c.extend(x - t)
    c.extend(1.0 - x - t)
    c.extend(y - t)
    c.extend(1.0 - y - t)
    
    # Pairwise non-overlap constraints: dist^2 >= 4t^2
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    c.extend(dx**2 + dy**2 - 4.0 * t**2)
    
    return np.array(c)

def obj_eq(vars_arr):
    """Objective: maximize shared radius t."""
    return -vars_arr[-1]

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        lim = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def generate_hex_inits(rng):
    """Generates diverse hexagonal initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [6,6,4,6,4], 
        [5,5,6,5,5], [6,4,6,4,6], [5,6,6,5,4], [6,5,4,6,5],

        [7,6,6,7], [6,7,6,7], [5,6,5,6,5,1], [5,5,5,6,5],
        [8,5,6,7], [7,7,6,6], [9,5,6,6], [8,6,6,6]
    ]
    
    for pat in patterns:
        if sum(pat) != N: 
            continue
        pts = []
        r0 = 0.098
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: 
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:N])
        
        # Center and scale to fit comfortably inside [0.1, 0.9]
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        if span[0] > 1e-4 and span[1] > 1e-4:
            pts = (pts - mn) / span * 0.78 + 0.11
        else:
            pts = np.full((N, 2), 0.5)
            
        inits.append(pts)
        # Add small perturbations to each pattern
        for _ in range(3):
            p = pts + rng.uniform(-0.012, 0.012, pts.shape)
            inits.append(np.clip(p, 0.05, 0.95))
            
    # Add purely random starts
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

            pass

    best_sum = float(np.sum(best_r))
