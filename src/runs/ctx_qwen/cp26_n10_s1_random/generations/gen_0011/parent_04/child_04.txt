import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def penalty_and_grad(v, R):
    """Computes overlap/boundary penalty and gradient for fixed equal radius R."""
    n = N_CIRCLES
    c = v.reshape(-1, 2)
    penalty = 0.0
    grad = np.zeros_like(c)
    
    # Boundary penalties
    for i in range(n):
        # Left wall: x >= R
        v_x = R - c[i, 0]
        if v_x > 0:
            penalty += v_x**2
            grad[i, 0] -= 2.0 * v_x
        # Right wall: x + R <= 1
        v_x = c[i, 0] + R - 1.0
        if v_x > 0:
            penalty += v_x**2
            grad[i, 0] += 2.0 * v_x
        # Bottom wall: y >= R
        v_y = R - c[i, 1]
        if v_y > 0:
            penalty += v_y**2
            grad[i, 1] -= 2.0 * v_y
        # Top wall: y + R <= 1
        v_y = c[i, 1] + R - 1.0
        if v_y > 0:
            penalty += v_y**2
            grad[i, 1] += 2.0 * v_y
            
    # Pairwise overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            diff = c[i] - c[j]
            dist = np.sqrt(np.sum(diff**2))
            if dist < 1e-10:
                dist = 1e-10
                diff = np.array([1e-6, 1e-6])
            overlap = 2.0 * R - dist
            if overlap > 0:
                penalty += overlap**2
                factor = 2.0 * overlap / dist
                grad[i] -= factor * diff
                grad[j] += factor * diff
    return penalty, grad.flatten()

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def generate_hex_configs(n, rng):
    """Generates diverse hexagonal lattice initial configurations."""
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,6,6,4], [6,6,5,5,4],
        [5,5,6,5,5], [5,6,6,5,4], [6,5,4,6,5], [5,7,4,5,5],
        [7,5,5,5,4], [5,5,7,5,4], [6,7,6,7], [5,6,5,6,5,1],
        [8,5,6,7], [7,7,6,6]
    ]
    for pat in patterns:
        if sum(pat) != n: continue
        pts = []
        r0 = 0.1
        y = r0
        for idx_r, cnt in enumerate(pat):
            shift = r0 if idx_r % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        base = np.array(pts[:n])
        # Normalize to fit comfortably in [0.05, 0.95]
        mn = base.min(axis=0)
        mx = base.max(axis=0)
        span = mx - mn
        if span[0] > 1e-4 and span[1] > 1e-4:
            base = (base - mn) / span * 0.85 + 0.075
        configs.append(base)
        
        for _ in range(4):
            p = base + rng.uniform(-0.015, 0.015, base.shape)
            configs.append(np.clip(p, 0.02, 0.98))
            
    for _ in range(8):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_centers = None
    best_sum = -1.0
    best_radii = None
    
    configs = generate_hex_configs(N_CIRCLES, rng)
    
    # Phase 1: Find best equal-radius packing via bisection + L-BFGS-B
    best_R = 0.0
    best_cfg = None
    best_pen = np.inf
    
    for cfg in configs:
        low, high = 0.08, 0.105
        for _ in range(14):
            mid = (low + high) / 2.0
            min_pen = np.inf
            best_res = None
            
            # Multiple restarts to escape local minima
            for _ in range(3):
                c0 = cfg + rng.uniform(-0.005, 0.005, cfg.shape)
                c0 = np.clip(c0, mid, 1.0 - mid)
                bounds_opt = [(mid, 1.0-mid)] * (2 * N_CIRCLES)
                res = minimize(penalty_and_grad, c0.flatten(), jac=True, args=(mid,),
                               method='L-BFGS-B', bounds=bounds_opt,
                               options={'maxiter': 2000, 'ftol': 1e-12})
                if res.fun < min_pen:
                    min_pen = res.fun
                    best_res = res
                    
            if min_pen < 1e-5:
                low = mid
                if mid > best_R - 1e-5:
                    if min_pen < best_pen or mid > best_R:
                        best_cfg = best_res.x.reshape(-1, 2).copy()
                        best_R = mid
                        best_pen = min_pen
            else:
                high = mid
                
    if best_cfg is not None:
        r_lp, s_lp = solve_lp(best_cfg)
        best_centers = best_cfg
        best_sum = s_lp
        best_radii = r_lp
        
    # Phase 2: Local Hill-Climbing on Centers evaluated via LP
    if best_centers is not None:
        curr_c = best_centers.copy()
        curr_r, curr_s = solve_lp(curr_c)
        step = 0.015
        
        for it in range(800):
            idx = rng.integers(N_CIRCLES)
            old = curr_c[idx].copy()
            
            best_move = None
            best_move_s = curr_s
            
            # Try multiple perturbations for this circle
            for _ in range(4):
                move = rng.uniform(-step, step, 2)
                new_c = curr_c[idx] + move
                new_c = np.clip(new_c, 1e-4, 1.0 - 1e-4)
                curr_c[idx] = new_c
                
                r_try, s_try = solve_lp(curr_c)
                if s_try > best_move_s:
                    best_move_s = s_try
                    best_move = new_c.copy()
                    
            if best_move is not None:
                curr_c[idx] = best_move
                curr_r, curr_s = solve_lp(curr_c)
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_centers = curr_c.copy()
                    best_radii = curr_r.copy()
            else:
                curr_c[idx] = old
                
            step *= 0.9985
            
    # Fallback safety net
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum = solve_lp(best_centers)
        
    # Phase 3: Strict Safety Scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N_CIRCLES):
        x, y, r = best_centers[i,0], best_centers[i,1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
