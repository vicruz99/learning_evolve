import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """
    Given fixed centers, solves the LP to maximize sum of radii.
    Constraints: r_i <= distance to boundary and r_i + r_j <= dist(i, j).
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, 0.5)] * n
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(lim)
        
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def lp_objective(centers_flat, n):
    """Objective function for center optimization: negative LP sum of radii."""
    centers = centers_flat.reshape(n, 2)

    if s is None or not np.isfinite(s):
        return 100.0  # Penalty for infeasible center configurations
    return -s

def coord_opt_func(x, all_centers, idx, n):
    """Wrapper for coordinate-wise optimization without closures."""
    c = all_centers.copy()
    c[idx] = x
    return lp_objective(c.flatten(), n)

def optimize_single_center(all_centers, idx, n):
    """Optimizes the position of a single circle to maximize LP sum."""
    x0 = all_centers[idx].copy()

                   options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-9})
    return res.x, -res.fun

def full_opt_wrapper(x, n):

    return lp_objective(x, n)

def apply_safety_scale(centers, radii, n):

    scale = 1.0

        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if r > 1e-9:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            rs = radii[i] + radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    return radii * scale * 0.99999999

def make_hex(rows_pattern, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
