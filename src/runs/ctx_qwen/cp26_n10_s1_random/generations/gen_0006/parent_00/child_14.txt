The LP maximizes $\sum r_i$ subject to $r_i \le \text{dist}(c_i, \text{boundary})$ and $r_i + r_j \le \|c_i - c_j\|$.

import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """
    Given fixed centers, solves the LP to maximize sum of radii.
    Constraints: r_i <= dist(i, boundary) and r_i + r_j <= dist(i, j)
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, 0.5)] * n
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(lim)
        
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def get_lp_objective(centers_flat):
    """Objective function for center optimization: minimize negative LP sum of radii."""
    n = 26
    centers = centers_flat.reshape(n, 2)
    _, s = solve_lp_radii(centers)
    return -s

def make_hex(rows_pattern, r0=0.09):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)
    
    # Diverse hexagonal row patterns known to yield high-density packings
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [6, 5, 5, 6, 4], [5, 5, 4, 6, 6], [4, 5, 6, 5, 6], [5, 4, 6, 5, 6]
    ]
    
    inits = []
    for p in patterns:
        if sum(p) < n: continue
        # Scale lattices to cover different density regimes
        for scale in [0.88, 0.92, 0.96, 1.02]:
            pts = make_hex(p, r0=0.09 * scale)
            inits.append(pts)
            # Add perturbed variants to break symmetry and escape traps
            pts_p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            inits.append(np.clip(pts_p, 0.05, 0.95))
            
    # Random dense configurations
    for _ in range(10):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Structured grid fallback
    g = np.linspace(0.15, 0.85, 5)
    grid = np.array([[x, y] for y in g for x in g])
    grid = np.vstack([grid, [0.5, 0.5]])
    inits.append(grid)
    
    # Phase 1: Global search over diverse initializations
    for cfg in inits:
        try:
            res = minimize(get_lp_objective, cfg.flatten(), method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-7, 'fatol': 1e-10})
            c_opt = res.x.reshape(n, 2)
            r_lp, s_lp = solve_lp_radii(c_opt)
            if r_lp is not None and s_lp > best_sum:
                best_sum = s_lp
                best_centers = c_opt.copy()
                best_radii = r_lp.copy()
        except Exception:
            continue

    # Phase 2: Intensive local refinement around the best configuration found
    if best_centers is not None:
        for trial in range(20):
            # Gradually shrinking perturbation scale
            pert_scale = 0.008 * (1.0 - 0.04 * trial)
            c_start = best_centers + rng.normal(0, pert_scale, best_centers.shape)
            c_start = np.clip(c_start, 0.02, 0.98)
            
            try:
                res = minimize(get_lp_objective, c_start.flatten(), method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-11})
                c_opt = res.x.reshape(n, 2)
                r_lp, s_lp = solve_lp_radii(c_opt)
                if r_lp is not None and s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c_opt.copy()
                    best_radii = r_lp.copy()
            except Exception:
                continue

    # Phase 3: Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-12:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
                
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)
                    
        best_radii *= scale * 0.9999995
        best_sum = np.sum(best_radii)
    else:
        # Fallback configuration
        best_centers = make_hex([5, 6, 5, 6, 4])
        best_radii = np.full(n, 0.09)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
