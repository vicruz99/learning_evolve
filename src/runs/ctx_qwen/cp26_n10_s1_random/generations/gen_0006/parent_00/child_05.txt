import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """
    Given fixed centers, solves the LP to maximize sum of radii.
    Constraints: r_i + r_j <= dist(i, j) and r_i <= distance to boundary.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, 0.5)] * n
    
    A_ub, b_ub = [], []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(lim)
        
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def penalty_centers(centers_flat, radii, n):
    """
    Computes squared penalty for boundary violations and circle overlaps.
    Used to push centers apart to satisfy packing constraints for inflated radii.
    """
    c = centers_flat.reshape(n, 2)
    p = 0.0
    
    # Boundary violations
    for i in range(n):
        x, y = c[i]
        r = radii[i]
        v = min(0.0, x - r)
        p += v * v
        v = min(0.0, 1.0 - x - r)
        p += v * v
        v = min(0.0, y - r)
        p += v * v
        v = min(0.0, 1.0 - y - r)
        p += v * v
        
    # Pairwise overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
            v = max(0.0, radii[i] + radii[j] - d)
            p += v * v
    return p

def make_hex(rows, r0=0.09):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    rng = np.random.default_rng(12345)

    # Diverse initial configurations
    inits = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [6, 5, 5, 6, 4], [5, 5, 4, 6, 6], [4, 5, 6, 5, 6], [5, 4, 6, 5, 6],
        [6, 6, 6, 4, 4], [5, 6, 6, 5, 4], [4, 5, 5, 6, 6]
    ]
    for p in patterns:
        if sum(p) < n:
            continue
        pts = make_hex(p, 0.095)
        inits.append(pts)
        for _ in range(5):
            pts_p = pts + rng.uniform(-0.03, 0.03, pts.shape)
            inits.append(np.clip(pts_p, 0.05, 0.95))

    for _ in range(15):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))

    # Optimization loop over all starts
    for cfg in inits:
        centers = cfg.copy()
        r, s = solve_lp_radii(centers)
        if r is None:
            continue

        # Inflate and relax phase
        for step in range(30):
            r_inflated = r * 1.007
            res = minimize(penalty_centers, centers.flatten(), args=(r_inflated, n),
                           method='L-BFGS-B', bounds=[(0.0, 1.0)] * (2 * n),
                           options={'maxiter': 800, 'ftol': 1e-13})
            
            if res.success or res.fun < 1e-7:
                centers = res.x.reshape(n, 2)
                r_new, s_new = solve_lp_radii(centers)
                if r_new is not None and np.sum(r_new) > np.sum(r) + 1e-8:
                    r = r_new
                    s = s_new
                else:
                    break
            else:
                break
                
            # Periodic perturbation to escape local minima
            if step % 8 == 7:
                centers += rng.uniform(-0.002, 0.002, centers.shape)
                centers = np.clip(centers, 0.01, 0.99)

        # Final safety scaling to guarantee strict validity within tolerance
        scale = 1.0
        for i in range(n):
            x, y, rd = centers[i, 0], centers[i, 1], r[i]
            if rd > 1e-9:
                scale = min(scale, x / rd, (1.0 - x) / rd, y / rd, (1.0 - y) / rd)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                rs = r[i] + r[j]
                if rs > 1e-9:
                    scale = min(scale, d / rs)
                    
        r_final = r * scale * 0.999999
        s_final = np.sum(r_final)
        if s_final > best_sum:
            best_sum = s_final
            best_centers = centers.copy()
            best_radii = r_final.copy()

    # Fallback configuration
    if best_centers is None:
        best_centers = make_hex([5, 6, 5, 6, 4])
        r_fb, _ = solve_lp_radii(best_centers)
        best_radii = r_fb if r_fb is not None else np.full(n, 0.09)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
