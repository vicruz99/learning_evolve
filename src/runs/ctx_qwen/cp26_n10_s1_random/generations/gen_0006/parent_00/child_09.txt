import numpy as np
from scipy.optimize import minimize, linprog

def compute_clearances(centers):
    """Computes the minimum clearance (equal radius case) for given centers."""
    n = centers.shape[0]
    cb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    cp = np.min(dists) / 2.0
    return min(np.min(cb), cp)

def max_clearance_obj(vars, n):
    """Objective for Nelder-Mead: maximize minimum clearance."""
    centers = vars[:2*n].reshape(n, 2)
    return -compute_clearances(centers)

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, 0.5)] * n
    A_ub = []
    b_ub = []
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(lim)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def make_hex(rows_pattern, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def obj_joint(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_joint(v, n):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.empty(4*n + n*(n-1)//2)
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    idx = np.triu_indices(n, k=1)
    dx = cx[idx[0]] - cx[idx[1]]
    dy = cy[idx[0]] - cy[idx[1]]
    dr = r[idx[0]] + r[idx[1]]
    c[4*n:] = dx**2 + dy**2 - dr**2
    return c

def apply_safety_scale(centers, radii, n):
    """Computes and applies a safety scale to guarantee strict validity."""
    scale = 1.0
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            rs = radii[i] + radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
    return radii * scale * 0.9999999

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    rng = np.random.default_rng(42)

    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],

        [5, 6, 5, 5, 5], [6, 4, 6, 5, 5], [5, 5, 5, 6, 5], [4, 6, 5, 6, 5]

    configs = []
    for p in patterns:

        c = make_hex(p)
        configs.append(c)

            cp = c + rng.uniform(-0.025, 0.025, c.shape)

        configs.append(rng.uniform(0.1, 0.9, (n, 2)))

    # Phase 1: Optimize centers for maximum equal-radius clearance

        try:

                r_final = apply_safety_scale(c_opt, r_lp, n)
                s_final = np.sum(r_final)

                    best_radii = r_final.copy()

            continue
