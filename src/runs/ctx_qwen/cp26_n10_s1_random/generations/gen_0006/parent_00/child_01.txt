import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """
    Given fixed centers, solves the LP to maximize sum of radii.
    Constraints: r_i + r_j <= dist(i, j) and r_i <= distance to boundary.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, 0.5)] * n
    
    # Pre-allocate constraint matrix for speed
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs + n, n))
    b_ub = np.zeros(n_pairs + n)
    
    idx = 0
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[idx, i] = 1.0
        b_ub[idx] = lim
        idx += 1
        
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_lp(centers_flat, n):
    """Objective for Nelder-Mead: minimize negative sum of LP-optimized radii."""
    c = centers_flat.reshape(n, 2)
    r, s = solve_lp_radii(c)
    if r is None or s <= 1e-9:
        return 1e6  # Heavy penalty for invalid/degenerate configurations
    return -s

def constraint_func_joint(vars_array, n):
    """
    Computes inequality constraints >= 0 for valid packing.
    Uses squared distance to avoid square roots during SLSQP optimization.
    """
    cx = vars_array[:n]
    cy = vars_array[n:2 * n]
    r = vars_array[2 * n:]
    
    # Boundary constraints
    b = np.empty(4 * n)
    b[:n] = cx - r
    b[n:2 * n] = 1.0 - cx - r
    b[2 * n:3 * n] = cy - r
    b[3 * n:4 * n] = 1.0 - cy - r
    
    # Pairwise non-overlap constraints: dist^2 >= (ri + rj)^2
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    
    d2 = cx_m**2 + cy_m**2
    rs2 = r_m**2
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    p = d2[idx_i, idx_j] - rs2[idx_i, idx_j]
    
    return np.concatenate([b, p])

def objective_joint(vars_array, n):
    """Objective for SLSQP: minimize negative sum of radii"""
    return -np.sum(vars_array[2 * n:])

def apply_safety_scale(centers, radii, n):
    """Computes and applies a safety scale to guarantee strict validity."""
    scale = 1.0
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if r > 1e-9:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            rs = radii[i] + radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    return radii * scale * 0.9999995

def make_hex(rows_pattern, r0=0.09):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
