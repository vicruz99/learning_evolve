import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """
    Given fixed centers, solves the LP to maximize sum of radii.
    Constraints: r_i + r_j <= dist(i, j) and r_i <= distance to boundary.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, 0.5)] * n
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(max(lim, 1e-9))
        
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(d)
            
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_lp(centers_flat, n):
    """Objective for center optimization: negative sum of optimal radii."""
    c = centers_flat.reshape(n, 2)
    if np.any(c < 0) or np.any(c > 1):
        return 1e6
    _, s = solve_lp_radii(c)
    return -s if s > 0 else 1e6

def make_hex_centered(rows_pattern, r0=0.10):
    """Generates initial positions on a centered hexagonal lattice."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        row_width = (cnt - 1) * 2 * r0
        x_start = 0.5 - row_width / 2.0 + shift
        for c_idx in range(cnt):
            if len(pts) >= n: break
            x = x_start + c_idx * 2 * r0
            pts.append([x, y])
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
