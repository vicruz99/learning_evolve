import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """
    Given fixed centers, solves the LP to maximize sum of radii.
    Constraints: r_i <= dist to boundary, r_i + r_j <= dist(i, j), r_i >= 0.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, 0.5)] * n
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(max(lim, 1e-9)) # Ensure feasible LP
        
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-6

def get_lp_sum(centers):
    """Wrapper to safely get LP sum."""
    _, s = solve_lp_radii(centers)
    return s

def local_search(centers, rng):
    """
    Performs robust local optimization on centers to maximize LP sum of radii.
    Uses Powell, Nelder-Mead, and adaptive shaking.
    """
    n = centers.shape[0]
    best_c = centers.copy()
    best_s = get_lp_sum(best_c)
    
    def neg_lp_obj(x):
        return -get_lp_sum(x.reshape(n, 2))
    
    # Powell method: efficient for non-smooth landscapes
    try:
        res = minimize(neg_lp_obj, best_c.flatten(), method='Powell', 
                       options={'maxiter': 2000, 'xtol': 1e-9, 'ftol': 1e-12})
        if res.fun < -best_s - 1e-10:
            best_c = res.x.reshape(n, 2)
            best_s = -res.fun
    except Exception:
        pass
        
    # Nelder-Mead: fine-grained convergence
    try:
        res = minimize(neg_lp_obj, best_c.flatten(), method='Nelder-Mead', 
                       options={'maxiter': 8000, 'xatol': 1e-9, 'fatol': 1e-12})
        if res.fun < -best_s - 1e-10:
            best_c = res.x.reshape(n, 2)
            best_s = -res.fun
    except Exception:
        pass
        
    # Adaptive shake to escape local minima
    shake_scales = [0.008, 0.004, 0.002, 0.001]
    for scale in shake_scales:
        for _ in range(6):
            c_shake = best_c + rng.uniform(-scale, scale, best_c.shape)
            c_shake = np.clip(c_shake, 0.05, 0.95)
            s_shake = get_lp_sum(c_shake)
            
            if s_shake > best_s:
                try:
                    res = minimize(neg_lp_obj, c_shake.flatten(), method='Powell',
                                   options={'maxiter': 1000, 'xtol': 1e-9, 'ftol': 1e-12})
                    if res.fun < -best_s - 1e-10:
                        best_c = res.x.reshape(n, 2)
                        best_s = -res.fun
                except Exception:
                    pass
    return best_c, best_s

def make_hex(rows_pattern, r0=0.095):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
