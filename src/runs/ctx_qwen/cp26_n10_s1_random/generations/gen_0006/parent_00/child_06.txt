import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i <= distance to boundary, r_i + r_j <= distance between centers.
    """
    c_obj = -np.ones(n)
    bounds = [(0.0, 0.5)] * n
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(max(lim, 0.0))
        
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub_arr = np.array(A_ub)
    b_ub_arr = np.array(b_ub)
    
    try:
        res = linprog(c_obj, A_ub=A_ub_arr, b_ub=b_ub_arr, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def apply_safety_scale(centers, radii, n):
    """
    Computes the maximum valid scaling factor and applies a tiny safety margin
    to guarantee strict compliance with the 1e-12 validation tolerance.
    """
    scale = 1.0
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if r > 1e-9:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            rs = radii[i] + radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    return radii * scale * 0.999999

def make_hex(rows_pattern, r0=0.095, n=26):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def lp_obj_wrapper(centers_flat, n):
    """Objective function for center optimization: minimize negative LP sum."""
    centers = centers_flat.reshape(n, 2)
    # Keep centers safely inside the square to avoid degenerate LP bounds
    centers = np.clip(centers, 0.002, 0.998)
    _, s = solve_lp_radii(centers, n)
    return -s if s > 0 else -1e5

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)
    
    # Diverse hexagonal row distributions that sum to >= 26
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [6, 5, 5, 6, 4], [5, 5, 4, 6, 6], [4, 5, 6, 5, 6], [5, 4, 6, 5, 6],
        [6, 6, 6, 4, 4], [5, 6, 6, 4, 5], [4, 6, 5, 6, 5], [5, 5, 5, 6, 5]
    ]
    
    inits = []
    for p in patterns:
        if sum(p) < n: continue
        pts = make_hex(p)
        inits.append(pts)
        for _ in range(6):
            pts_p = pts + rng.uniform(-0.025, 0.025, pts.shape)
            inits.append(np.clip(pts_p, 0.05, 0.95))
            
    for _ in range(10):
        inits.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    # Phase 1: Initial screening with LP refinement
    for cfg in inits:
        r_lp, s_lp = solve_lp_radii(cfg, n)
        if r_lp is not None:
            r_final = apply_safety_scale(cfg, r_lp, n)
            s_final = np.sum(r_final)
            if s_final > best_sum:
                best_sum = s_final
                best_centers = cfg.copy()
                best_radii = r_final.copy()

    # Phase 2: Aggressive local search on centers to maximize LP sum
    if best_centers is not None:
        for trial in range(80):
            # Adaptive perturbation scale: starts broad, shrinks to fine-tune
            scale = 0.012 * (1.0 - trial * 0.012)
            if scale < 0.001: scale = 0.001
            
            c_start = best_centers + rng.normal(0, scale, best_centers.shape)
            c_start = np.clip(c_start, 0.02, 0.98)
            
            try:
                res_loc = minimize(lp_obj_wrapper, c_start.flatten(), args=(n,),
                                   method='Nelder-Mead',
                                   options={'maxiter': 3000, 'xatol': 1e-8, 'fatol': 1e-9})
                c_opt = res_loc.x.reshape(n, 2)
                
                r_lp, s_lp = solve_lp_radii(c_opt, n)
                if r_lp is not None:
                    r_final = apply_safety_scale(c_opt, r_lp, n)
                    s_final = np.sum(r_final)
                    
                    if s_final > best_sum:
                        best_sum = s_final
                        best_centers = c_opt.copy()
                        best_radii = r_final.copy()
            except Exception:
                continue

    # Fallback configuration (should rarely trigger)
    if best_centers is None:
        best_centers = make_hex([5, 6, 5, 6, 4])
        r_fb, _ = solve_lp_radii(best_centers, n)
        if r_fb is not None:
            best_radii = apply_safety_scale(best_centers, r_fb, n)
        else:
            best_radii = np.full(n, 0.09)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
