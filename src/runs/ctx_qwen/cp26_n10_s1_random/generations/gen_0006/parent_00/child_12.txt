import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """
    Given fixed centers, solves the LP to maximize sum of radii.
    Constraints: r_i + r_j <= dist(i, j) and r_i <= distance to boundary.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, 0.5)] * n
    
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(lim)
        
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(d)
            
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def lp_sum_radii(centers):
    """Returns the maximum sum of radii for fixed centers."""
    _, s = solve_lp_radii(centers)
    return s

def lp_objective(centers_flat, n):
    """Objective function for center optimization: minimize negative LP sum."""
    centers = centers_flat.reshape(n, 2)
    s = lp_sum_radii(centers)
    if s <= 0:
        return 100.0  # Heavy penalty for infeasible configurations
    return -s

def refine_centers(centers_init, n, rng):
    """
    Refines center positions to maximize the LP sum of radii using 
    Nelder-Mead, Powell, and Simulated Annealing.
    """
    best_centers = centers_init.copy()
    best_val = lp_sum_radii(best_centers)
    if best_val <= 0:
        best_val = 0.0
        
    # Phase 1: Nelder-Mead
    try:
        res = minimize(lp_objective, best_centers.flatten(), args=(n,), method='Nelder-Mead',
                       options={'maxiter': 1500, 'xatol': 1e-9, 'fatol': 1e-10})
        val = -res.fun
        if val > best_val:
            best_val = val
            best_centers = res.x.reshape(n, 2)
    except Exception:
        pass

    # Phase 2: Powell
    try:
        res = minimize(lp_objective, best_centers.flatten(), args=(n,), method='Powell',
                       options={'maxiter': 1500, 'xatol': 1e-9, 'ftol': 1e-10})
        val = -res.fun
        if val > best_val:
            best_val = val
            best_centers = res.x.reshape(n, 2)
    except Exception:
        pass
        
    # Phase 3: Simulated Annealing / Random Hill Climbing
    current = best_centers.copy()
    current_val = best_val
    T = 0.008
    step_size = 0.006
    
    for _ in range(2500):
        # Move a random subset of circles
        k = rng.choice([1, 2, 3, 4])
        idxs = rng.choice(n, k, replace=False)
        moves = rng.normal(0, step_size, (k, 2))
        new_c = current.copy()
        for i, idx in enumerate(idxs):
            new_c[idx] = np.clip(new_c[idx] + moves[i], 0.0, 1.0)
            
        new_val = lp_sum_radii(new_c)
        if new_val <= 0:
            continue
            
        delta = new_val - current_val
        if delta > 0 or rng.random() < np.exp(delta / T):
            current = new_c
            current_val = new_val
            if current_val > best_val:
                best_val = current_val
                best_centers = current.copy()
        T *= 0.9995
        
    return best_centers, best_val

def make_hex(rows_pattern, r0=0.09):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def apply_safety_scale(centers, radii, n):
    """Computes and applies a safety scale to guarantee strict validity."""
    scale = 1.0
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if r > 1e-9:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            rs = radii[i] + radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    return radii * scale * 0.999999

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    
    rng = np.random.default_rng(42)
    
    # Diverse hexagonal row distributions
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 5, 4, 6, 6], [6, 4, 6, 5, 5], [5, 6, 6, 4, 5], [4, 5, 6, 6, 5]
    ]
    
    inits = []
    for p in patterns:
        if sum(p) < n: continue
        pts = make_hex(p)
        inits.append(pts)
        for _ in range(5):
            pts_p = pts + rng.uniform(-0.035, 0.035, pts.shape)
            inits.append(np.clip(pts_p, 0.02, 0.98))
            
    # Add fully random starts
    for _ in range(6):
        inits.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    # Optimize each starting configuration
    for cfg in inits:
        refined_c, val = refine_centers(cfg, n, rng)
        if val > best_sum:
            best_sum = val
            best_centers = refined_c.copy()
            
    # Fallback if all optimizations fail
    if best_centers is None:
        best_centers = make_hex([5, 6, 5, 6, 4])
        
    # Compute optimal radii for the best centers
    r_lp, _ = solve_lp_radii(best_centers)
    if r_lp is not None:
        best_radii = apply_safety_scale(best_centers, r_lp, n)
    else:
        best_radii = np.full(n, 0.09)
        
    final_sum = np.sum(best_radii)
    return best_centers, best_radii, float(final_sum)
