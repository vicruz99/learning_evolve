import numpy as np
from scipy.optimize import minimize, linprog

def compute_penalty_and_grad(centers, r):
    """Computes squared overlap/boundary penalty and its gradient for fixed radius r."""
    n = centers.shape[0]
    pen = 0.0
    grad = np.zeros_like(centers)
    
    # Boundary penalties
    viol_left = np.maximum(0.0, r - centers[:, 0])
    pen += np.sum(viol_left**2)
    grad[:, 0] -= 2.0 * viol_left
    
    viol_right = np.maximum(0.0, centers[:, 0] - (1.0 - r))
    pen += np.sum(viol_right**2)
    grad[:, 0] += 2.0 * viol_right
    
    viol_bottom = np.maximum(0.0, r - centers[:, 1])
    pen += np.sum(viol_bottom**2)
    grad[:, 1] -= 2.0 * viol_bottom
    
    viol_top = np.maximum(0.0, centers[:, 1] - (1.0 - r))
    pen += np.sum(viol_top**2)
    grad[:, 1] += 2.0 * viol_top
    
    # Pairwise overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            d = np.hypot(diff[0], diff[1])
            if d < 1e-12:
                d = 1e-12
                diff = np.array([1.0, 0.0])
            viol = np.maximum(0.0, 2.0 * r - d)
            if viol > 0.0:
                pen += viol**2
                factor = 2.0 * viol / d
                grad[i] -= factor * diff
                grad[j] += factor * diff
    return pen, grad

def optimize_centers_fixed_r(centers0, r, n, max_iter=2000):
    """Optimizes center positions to minimize overlap/boundary violations for a fixed radius r."""
    def obj(x):
        c = x.reshape(n, 2)
        p, _ = compute_penalty_and_grad(c, r)
        return p
    def jac(x):
        c = x.reshape(n, 2)
        _, g = compute_penalty_and_grad(c, r)
        return g.flatten()
        
    res = minimize(obj, centers0.flatten(), jac=jac, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * (2 * n), options={'maxiter': max_iter, 'ftol': 1e-15})
    return res.x.reshape(n, 2), res.fun

def solve_lp_radii(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 0.0)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def hex_init(row_counts, r_guess):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r_guess
    for ri, cnt in enumerate(row_counts):
        shift = r_guess if ri % 2 == 1 else 0.0
        x = r_guess + shift
        for _ in range(cnt):
            if len(pts) >= 26: break
            pts.append([x, y])
            x += 2.0 * r_guess
        y += np.sqrt(3) * r_guess
    while len(pts) < 26:
        pts.append([0.5, 0.5])
    return np.array(pts[:26])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5], [5, 5, 6, 4, 6],
        [5, 6, 5, 5, 5], [5, 5, 5, 6, 5], [5, 5, 5, 5, 6], [6, 5, 5, 5, 5]
    ]
    
    # Phase 1: Bisection to find maximum feasible equal radius, optimizing centers at each step
    low, high = 0.090, 0.105
    for _ in range(25):
        mid = (low + high) * 0.5
        feasible = False
        for p in patterns:
            c0 = hex_init(p, mid)
            mn, mx = c0.min(axis=0), c0.max(axis=0)
            c0 = (c0 - mn) * 0.92 + 0.04
            c0 = np.clip(c0, mid, 1.0 - mid)
            
            opt_c, pen = optimize_centers_fixed_r(c0, mid, n)
            if pen < 1e-6:
                feasible = True
                # Perturb to escape shallow local minima
                for _ in range(3):
                    c_pert = np.clip(opt_c + np.random.normal(0, 0.002, opt_c.shape), mid, 1.0 - mid)
                    opt_c2, pen2 = optimize_centers_fixed_r(c_pert, mid, n)
                    if pen2 < pen:
                        opt_c, pen = opt_c2, pen2
                        
                if pen < 1e-6:
                    r_lp, s_lp = solve_lp_radii(opt_c, n)
                    if r_lp is not None and s_lp > best_sum:
                        best_sum = s_lp
                        best_centers = opt_c.copy()
                        best_radii = r_lp.copy()
                break
        if feasible:
            low = mid
        else:
            high = mid
            
    # Phase 2: Iterative relaxation & LP refinement
    if best_centers is not None:
        for _ in range(12):
            cur_r = best_radii * (0.88 + 0.06 * np.random.rand())
            r_target = np.mean(cur_r)
            c_pert = np.clip(best_centers + np.random.uniform(-0.012, 0.012, (n, 2)), r_target, 1.0 - r_target)
            
            opt_c, pen = optimize_centers_fixed_r(c_pert, r_target, n, max_iter=3000)
            if pen < 1e-5:
                r_lp, s_lp = solve_lp_radii(opt_c, n)
                if r_lp is not None and s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = opt_c.copy()
                    best_radii = r_lp.copy()
                    
    # Fallback configuration
    if best_centers is None:
        best_centers = hex_init([5, 6, 5, 6, 4], 0.09)
        best_radii = np.full(n, 0.085)
        best_sum = np.sum(best_radii)
        
    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
