from scipy.optimize import minimize, linprog

def compute_clearance(centers, n):
    """Computes the maximum feasible equal radius for given centers."""
    d_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    min_bound = np.min(d_bound)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pairwise = np.min(dists) / 2.0
    
    return min(min_bound, min_pairwise)

def obj_equal(vars_arr, n):
    """Objective: maximize r => minimize -r"""
    return -vars_arr[2 * n]

def cons_equal(vars_arr, n):
    """Inequality constraints >= 0 for equal radius packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap: dist >= 2r
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    idx = np.triu_indices(n, k=1)
    dists = np.sqrt(dx[idx]**2 + dy[idx]**2)
    c = np.concatenate([c, dists - 2.0 * r])
    
    return c

def obj_joint(vars_arr, n):
    """Objective: maximize sum(r) => minimize -sum(r)"""
    return -np.sum(vars_arr[2*n:])

        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0

        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def make_hex_init(n, rows_pattern, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append(rng.uniform(r0, 1-r0, 2).tolist())
    return np.array(pts[:n])
