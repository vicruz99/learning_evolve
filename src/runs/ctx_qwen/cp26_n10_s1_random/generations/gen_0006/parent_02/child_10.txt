import numpy as np
from scipy.optimize import minimize, linprog

def objective_slsqp(vars_arr, n):
    """Minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def constraints_slsqp(vars_arr, n):
    """Compute inequality constraints >= 0 for valid packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap constraints (squared)
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dr = r[i_idx] + r[j_idx]
    
    c = np.concatenate([c, (dx**2 + dy**2) - dr**2])
    return c

def solve_lp_radii(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def energy_for_equal(centers_flat, R, n):
    """Smooth penalty energy for equal-circle packing."""
    c = centers_flat.reshape(n, 2)
    E = 0.0
    
    # Boundary violations
    vx = np.minimum(c[:,0], 1.0 - c[:,0])
    vy = np.minimum(c[:,1], 1.0 - c[:,1])
    viol_bound = np.maximum(0.0, R - np.minimum(vx, vy))
    E += np.sum(viol_bound**2)
    
    # Pairwise overlap violations
    diff = c[:, None, :] - c[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, 2.0*R)
    viol_pair = np.maximum(0.0, 2.0*R - dist)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    E += np.sum((viol_pair * mask)**2)
    
    return E

def find_equal_packing(n, rng):
    """Find dense equal-circle packing via bisection and energy minimization."""
    low, high = 0.08, 0.105
    best_c = None
    best_R = low
    
    # Generate a stable hexagonal initialization
    pts = []
    y = 0.09
    row = 0
    while len(pts) < n:
        shift = 0.09 if row % 2 == 1 else 0.0
        x = 0.09 + shift
        while x + 0.09 <= 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 0.18
        y += 0.09 * np.sqrt(3)
        row += 1
    init_c = np.array(pts[:n])
    
    for _ in range(25):
        mid = (low + high) / 2.0
        min_eng = 1e9
        best_pos = None
        
        # Multi-start optimization for positions
        for _ in range(4):
            c0 = init_c + rng.uniform(-0.005, 0.005, init_c.shape)
            c0 = np.clip(c0, 0.01, 0.99)
            res = minimize(energy_for_equal, c0.flatten(), args=(mid, n),
                           method='L-BFGS-B', bounds=[(0.01, 0.99)]*(2*n),
                           options={'maxiter': 3000, 'ftol': 1e-13})
            if res.fun < min_eng:
                min_eng = res.fun
                best_pos = res.x.reshape(n, 2)
                
        if min_eng < 1e-7:
            best_R = mid
            best_c = best_pos
            low = mid
        else:
            high = mid
            
    return best_c, best_R

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    # Phase 1: Equal circle packing via energy minimization
    eq_centers, eq_r = find_equal_packing(n, rng)
    
    radii_lp, s_lp = solve_lp_radii(eq_centers, n)
    if radii_lp is not None:
        best_centers = eq_centers
        best_radii = radii_lp
        best_sum = s_lp
    else:
        best_centers = eq_centers
        best_radii = np.full(n, eq_r)
        best_sum = 26.0 * eq_r
        
    bounds_vars = [(0.0, 1.0)] * (2 * n) + [(1e-7, 0.5)] * n
    cons_dict = {'type': 'ineq', 'fun': constraints_slsqp, 'args': (n,)}
    
    def run_slsqp(cfg, r_init=None, max_iter=6000):
        x0 = np.zeros(3 * n)
        x0[:n] = cfg[:, 0]
        x0[n:2*n] = cfg[:, 1]
        x0[2*n:] = r_init if r_init is not None else 0.06
        
        try:
            res = minimize(objective_slsqp, x0, args=(n,), method='SLSQP', 
                           bounds=bounds_vars, constraints=cons_dict,
                           options={'maxiter': max_iter, 'ftol': 1e-13, 'disp': False})
            if np.isfinite(res.fun):
                cx, cy, r = res.x[:n], res.x[n:2*n], res.x[2*n:]
                centers = np.column_stack((cx, cy))
                return centers, r
        except Exception:
            pass
        return None, None

    # Phase 2: Joint SLSQP refinement
    c_opt, r_opt = run_slsqp(best_centers, best_radii)
    if c_opt is not None:
        r_lp, s_lp = solve_lp_radii(c_opt, n)
        if r_lp is not None and s_lp > best_sum:
            best_centers = c_opt
            best_radii = r_lp
            best_sum = s_lp
            
    # Phase 3: Perturbation & Escape Loop

        pert_c = best_centers.copy()
        pert_r = best_radii.copy()
        
        idx_move = rng.choice(n, size=rng.integers(2, 5), replace=False)
        pert_r[idx_move] *= 0.5
        pert_c[idx_move] = rng.uniform(0.1, 0.9, size=(len(idx_move), 2))
        
        c_tmp, r_tmp = run_slsqp(pert_c, pert_r, max_iter=5000)
        if c_tmp is not None:
            r_lp, s_lp = solve_lp_radii(c_tmp, n)
            if r_lp is not None and s_lp > best_sum:
                best_centers = c_tmp
                best_radii = r_lp
                best_sum = s_lp
                
    # Phase 4: Nelder-Mead on centers to maximize LP sum
    def nm_obj(centers_flat):
        c = centers_flat.reshape(n, 2)
        r_lp, s = solve_lp_radii(c, n)
        return -s if r_lp is not None else -1.0

    x0_nm = best_centers.flatten()
    try:
        res_nm = minimize(nm_obj, x0_nm, method='Nelder-Mead', 
                          options={'maxiter': 4000, 'xatol': 1e-8, 'fatol': 1e-10})
        c_nm = res_nm.x.reshape(n, 2)
        r_lp, s_lp = solve_lp_radii(c_nm, n)
        if r_lp is not None and s_lp > best_sum:
            best_centers = c_nm
            best_radii = r_lp
            best_sum = s_lp
    except Exception:
        pass
        
    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
