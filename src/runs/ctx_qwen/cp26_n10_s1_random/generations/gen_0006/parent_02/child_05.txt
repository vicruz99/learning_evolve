import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    # Boundary limits for each circle
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def lp_objective(x, n):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = x.reshape(n, 2)
    # Keep centers safely inside the square to ensure LP feasibility
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, s = solve_lp(centers, n)
    return -s

def make_hex_init(n, rows_pattern, r0, rng):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append(rng.uniform(r0, 1 - r0, 2).tolist())
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)
    inits = []

        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 5, 5, 6, 5], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [4, 5, 6, 5, 6]
    ]
    for p in patterns:
        if sum(p) < n:
            continue
        for r0 in [0.085, 0.095, 0.105]:
            c = make_hex_init(n, p, r0, rng)
            inits.append(c)
            # Perturbed variants to break symmetry
            for _ in range(3):
                inits.append(np.clip(c + rng.uniform(-0.015, 0.015, c.shape), 0.02, 0.98))
                
    # 2. Dense random starts
    for _ in range(12):
        inits.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    # Phase 1: Multi-start Nelder-Mead optimization
    for init_c in inits:
        try:
            res = minimize(lp_objective, init_c.flatten(), args=(n,), method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-11, 'adaptive': True})
            c_opt = res.x.reshape(n, 2)
            r_opt, s_opt = solve_lp(c_opt, n)
            if r_opt is not None and s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            continue
            
    # Phase 2: Iterative perturbation search around the best configuration
    if best_centers is not None:
        step = 0.012
        for _ in range(150):
            idx = rng.integers(n)

            best_centers[idx] = np.clip(best_centers[idx] + rng.uniform(-step, step, 2), 0.02, 0.98)

                best_radii = r_new.copy()

            rs = best_radii[i] + best_radii[j]

    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
