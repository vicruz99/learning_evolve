import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26

def get_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Upper bounds for each radius based on distance to walls
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 0.0)) for l in lims]
    
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    i_idx, j_idx = np.triu_indices(n, k=1)
    m = len(i_idx)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), i_idx] = 1.0
    A_ub[np.arange(m), j_idx] = 1.0
    
    diffs = centers[i_idx] - centers[j_idx]
    b_ub = np.sqrt(np.sum(diffs**2, axis=1))
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def objective_slsqp(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def constraints_slsqp(v, n):
    """Inequality constraints >= 0 for SLSQP."""
    cx, cy, r = v[:n], v[n:2*n], v[2*n:]
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dr = r[i_idx] + r[j_idx]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def hill_climb_lp(init_centers, rng, max_steps=800):
    """Local search optimizing centers by evaluating LP-optimal sum of radii."""
    n = N_CIRCLES
    curr = init_centers.copy()
    radii, best_sum = get_lp_radii(curr)
    if radii is None:
        return curr, np.full(n, 0.0), 0.0
        
    step = 0.04
    for _ in range(max_steps):
        idx = rng.integers(0, n)
        old_pos = curr[idx].copy()
        move = rng.uniform(-step, step, 2)
        curr[idx] = np.clip(curr[idx] + move, 1e-5, 1.0 - 1e-5)
        
        r_new, s_new = get_lp_radii(curr)
        if r_new is not None and s_new > best_sum + 1e-8:
            best_sum = s_new
            radii = r_new
        else:
            curr[idx] = old_pos
            
        if _ % 100 == 0:
            step *= 0.92
            
    return curr, radii, best_sum

def slosqp_polish(centers, radii, rng):
    """SLSQP joint refinement of centers and radii."""
    n = N_CIRCLES
    x0 = np.zeros(3 * n)
    x0[:n] = centers[:, 0]
    x0[n:2*n] = centers[:, 1]
    x0[2*n:] = radii
    
    # Small perturbation to escape flat regions
    x0[:2*n] += rng.uniform(-0.002, 0.002, 2*n)
    x0[:2*n] = np.clip(x0[:2*n], 0.0, 1.0)
    
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons_dict = {'type': 'ineq', 'fun': constraints_slsqp, 'args': (n,)}
    
    try:
        res = minimize(objective_slsqp, x0, args=(n,), method='SLSQP', bounds=bounds, 
                       constraints=cons_dict, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
        if np.isfinite(res.fun):
            c = np.column_stack((res.x[:n], res.x[n:2*n]))
            r = res.x[2*n:]
            # Strict validity check
            valid = True
            if np.any(r < -1e-10) or np.any(c[:,0] < r - 1e-9) or np.any(c[:,0] > 1-r + 1e-9) or \
               np.any(c[:,1] < r - 1e-9) or np.any(c[:,1] > 1-r + 1e-9):
                valid = False
            if valid:
                i_idx, j_idx = np.triu_indices(n, 1)
                dx = c[i_idx, 0] - c[j_idx, 0]
                dy = c[i_idx, 1] - c[j_idx, 1]
                d2 = dx**2 + dy**2
                rs = r[i_idx] + r[j_idx]
                if np.any(d2 < rs**2 - 1e-8):
                    valid = False
            if valid:
                # Re-evaluate exact radii for optimized centers
                r_lp, s_lp = get_lp_radii(c)
                if r_lp is not None:
                    return c, r_lp, s_lp
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def make_hex_init(n, pattern, r0, rng):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for ri, cnt in enumerate(pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append(rng.uniform(r0, 1-r0, 2).tolist())
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Diverse row patterns for hexagonal initialization
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5], [5, 5, 5, 5, 6],
        [5, 5, 6, 4, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5]
    ]
    
    configs = []
    for p in patterns:
        if sum(p) < n: continue
        for r0 in [0.08, 0.09, 0.10, 0.11]:
            configs.append(make_hex_init(n, p, r0, rng))
            
    # Grid-based starts
    g = np.linspace(0.15, 0.85, 5)
    grid = np.array([[x, y] for y in g for x in g])
    for ex in [[0.5, 0.5], [0.1, 0.9], [0.9, 0.1], [0.5, 0.1]]:
        configs.append(np.vstack([grid, ex]))
        
    # Random dense starts
    for _ in range(5):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Optimization pipeline
    for cfg in configs:
        # Phase 1: Hill-climbing on LP value
        c1, r1, s1 = hill_climb_lp(cfg, rng, max_steps=800)
        if s1 > best_sum:
            best_sum = s1
            best_centers = c1
            best_radii = r1
            
        # Phase 2: SLSQP polish
        c2, r2, s2 = slosqp_polish(best_centers, best_radii, rng)
        if s2 > best_sum:
            best_sum = s2
            best_centers = c2
            best_radii = r2
            
    # Final safety scaling to guarantee strict numerical validity
    if best_centers is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
        i_idx, j_idx = np.triu_indices(n, 1)
        for k in range(len(i_idx)):
            i, j = i_idx[k], j_idx[k]
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
        best_radii *= scale * 0.999999
        best_sum = np.sum(best_radii)
    else:
        fallback = make_hex_init(n, [5, 6, 5, 6, 4], 0.085, rng)
        best_centers = fallback
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
