

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we decouple the problem into two stages:
1. **Center Optimization**: Find center coordinates that maximize the minimum clearance (distance to boundaries and half the distance to nearest neighbor). This proxy objective effectively pushes circles apart and into corners, creating a configuration with high packing density. We use derivative-free optimization (`Powell` method) which handles the non-smooth `min` function robustly.
2. **Radius Assignment (LP)**: For fixed centers, the problem of maximizing $\sum r_i$ subject to $r_i + r_j \le d_{ij}$ and boundary constraints is a Linear Program. We solve this exactly using `scipy.optimize.linprog`. This guarantees optimal radii for any given center layout.

We employ a multi-start strategy with diverse hexagonal lattice patterns and random initializations. After an initial sweep, we perform iterative perturbation around the best configuration found to escape local minima. Finally, we apply a strict safety scaling to ensure numerical compliance with the validation checker's $10^{-12}$ tolerance. This hybrid geometric + LP approach consistently outperforms direct non-convex optimization on $(x,y,r)$ variables.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def get_pairwise_dists(centers):
    """Computes upper triangular pairwise distances between centers."""
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    idx = np.triu_indices(N_CIRCLES, k=1)
    return dists[idx], idx

def compute_clearance(centers):
    """Computes the maximum feasible equal radius for the given centers."""
    x = centers[:, 0]
    y = centers[:, 1]
    # Distance to boundaries
    d_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Distance to other circles
    dists, _ = get_pairwise_dists(centers)
    d_min_pair = np.min(dists)
    
    return min(np.min(d_bound), d_min_pair / 2.0)

def negative_clearance(centers_flat):
    """Objective for center optimization: minimize negative clearance."""
    centers = centers_flat.reshape(N_CIRCLES, 2)
    # Soft clip to keep points strictly inside for numerical stability
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    return -compute_clearance(centers)

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Upper bounds from boundaries
    bounds_lim = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    bounds_lim = np.maximum(bounds_lim, 0.0)
    bounds = [(0.0, b) for b in bounds_lim]
    
    # Pairwise distance constraints
    dists, idx = get_pairwise_dists(centers)
    m_pairs = len(idx[0])
    
    # Constraint matrix: A_ub @ r <= b_ub
    # Rows 0..m_pairs-1: r_i + r_j <= dist_ij
    # Rows m_pairs..m_pairs+N-1: r_i <= bound_i
    A_ub = np.zeros((m_pairs + N_CIRCLES, N_CIRCLES))
    b_ub = np.zeros(m_pairs + N_CIRCLES)
    
    for k in range(m_pairs):
        i, j = idx[0][k], idx[1][k]
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[k]
        
    for i in range(N_CIRCLES):
        A_ub[m_pairs + i, i] = 1.0
        b_ub[m_pairs + i] = bounds_lim[i]
        
    c_obj = -np.ones(N_CIRCLES)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def generate_initial_configs(rng):
    """Generates diverse starting configurations."""
    configs = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 6, 4], [4, 6, 6, 6, 4],
        [6, 6, 5, 5, 4], [5, 7, 5, 5, 4], [5, 5, 5, 6, 5], [5, 6, 4, 6, 5],
        [6, 4, 6, 5, 5], [5, 5, 5, 5, 6], [4, 5, 6, 5, 6], [5, 6, 5, 5, 5]
    ]
    
    for r0 in [0.082, 0.092, 0.102, 0.112]:
        for pat in patterns:
            pts = []
            y = r0
            for ri, cnt in enumerate(pat):
                shift = r0 if ri % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= N_CIRCLES: break
                    if x + r0 <= 1.0:
                        pts.append([x, y])
                    x += 2 * r0
                y += np.sqrt(3) * r0
            while len(pts) < N_CIRCLES:
                pts.append([0.5, 0.5])
            configs.append(np.array(pts[:N_CIRCLES]))
            
            # Perturbed variants
            for _ in range(2):
                p = pts[:N_CIRCLES].copy()
                p += rng.uniform(-0.025, 0.025, p.shape)
                configs.append(np.clip(p, 0.02, 0.98))
                
    # Dense grid fallbacks
    g = np.linspace(0.12, 0.88, 5)
    grid = np.array([[cx, cy] for cy in g for cx in g])
    for extra in [[0.5, 0.5], [0.5, 0.08], [0.08, 0.5], [0.92, 0.5]]:
        configs.append(np.vstack([grid, extra]))
        
    # Random starts
    for _ in range(8):
        configs.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    configs = generate_initial_configs(rng)
    
    # Phase 1: Broad search optimizing centers for max clearance, then LP for radii
    for cfg in configs:
        try:
            res = minimize(negative_clearance, cfg.flatten(), method='Powell',
                           options={'maxiter': 2000, 'ftol': 1e-10, 'xatol': 1e-10})
            opt_centers = res.x.reshape(N_CIRCLES, 2)
            
            radii, s_lp = solve_lp_radii(opt_centers)
            if radii is not None and s_lp > best_sum:
                best_sum = s_lp
                best_centers = opt_centers.copy()
                best_radii = radii.copy()
        except Exception:
            continue
            
    # Phase 2: Iterative perturbation to escape local minima
    if best_centers is not None:
        for _ in range(60):
            pert = rng.uniform(-0.008, 0.008, (N_CIRCLES, 2))
            new_centers = np.clip(best_centers + pert, 0.02, 0.98)
            
            try:
                res = minimize(negative_clearance, new_centers.flatten(), method='Powell',
                               options={'maxiter': 1500, 'ftol': 1e-10, 'xatol': 1e-10})
                opt_c = res.x.reshape(N_CIRCLES, 2)
                
                radii, s_lp = solve_lp_radii(opt_c)
                if radii is not None and s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = opt_c.copy()
                    best_radii = radii.copy()
            except Exception:
                continue
                
    # Fallback if optimization unexpectedly fails
    if best_centers is None:
        best_centers = configs[0]
        best_radii, _ = solve_lp_radii(best_centers)
        if best_radii is None:
            best_radii = np.full(N_CIRCLES, 0.085)
        best_sum = np.sum(best_radii)

    # Final safety scaling to guarantee strict numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(N_CIRCLES):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    idx = np.triu_indices(N_CIRCLES, k=1)
    for k in range(len(idx[0])):
        i, j = idx[0][k], idx[1][k]
        d = np.linalg.norm(best_centers[i] - best_centers[j])
        rs = best_radii[i] + best_radii[j]
        if rs > 1e-12:
            scale = min(scale, d / rs)
            
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
