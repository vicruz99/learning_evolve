To maximize the sum of radii for 26 circles in a unit square, we leverage the fact that the optimal packing for moderate n is typically a slightly distorted hexagonal lattice. The known optimal equal-radius configuration for n=26 yields a radius of ~0.1014, giving a sum of ~2.6364. Our strategy:
1. **Equal-Radius Phase**: Use Nelder-Mead to optimize centers for maximum clearance (min distance to boundaries and other circles). This efficiently converges to the densest lattice arrangement without getting stuck in non-convex radius optimization basins.
2. **LP Refinement**: For any fixed centers, the optimal radii are found by solving a Linear Program (maximize sum(r_i) subject to r_i + r_j <= dist(i,j) and boundary limits). This guarantees locally optimal radii for any center configuration.
3. **Joint Optimization**: Use SLSQP to jointly optimize centers and radii, starting from the best equal-radius configuration. This allows unequal radii to squeeze out marginal gains.
4. **Perturbation Search**: Systematically perturb the best configuration and re-optimize to escape local minima, gradually reducing perturbation magnitude.
5. **Safety Scaling**: Apply a final multiplicative scale factor to guarantee strict numerical validity against the checker's 1e-12 tolerance.

All helper functions are defined at the top level, no closures or lambdas are used, and scipy/numpy are utilized efficiently.

import numpy as np
from scipy.optimize import minimize, linprog

def compute_clearance(x, n):
    """Objective for equal-radius phase: maximize minimum clearance."""
    c = x.reshape(n, 2)
    d_bound = np.min(np.minimum(c, 1.0 - c))
    d_pair = np.inf
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(c[i] - c[j])
            if d < d_pair:
                d_pair = d
    return -min(d_bound, d_pair / 2.0)

def solve_lp_radii(centers, n):
    """Solves the LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 0.0)) for l in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def objective_joint(vars_arr, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def constraints_joint(vars_arr, n):
    """Inequality constraints >= 0 for valid packing (joint centers + radii)."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    cx_m = cx[:, None]
    cy_m = cy[:, None]
    r_m = r[:, None]
    
    dx = cx_m - cx_m.T
    dy = cy_m - cy_m.T
    dr = r_m + r_m.T
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    dist_sq = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    r_sum_sq = dr[i_idx, j_idx]**2
    
    return np.concatenate([c, dist_sq - r_sum_sq])

def get_hex_centers(n, rows_pattern, r0):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(123)
    
    # Phase 1: Optimize equal radii configurations to find dense lattice
    patterns = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4]]
    for p in patterns:
        if sum(p) < n:
            continue
        for r0 in [0.09, 0.095, 0.10, 0.101]:
            centers = get_hex_centers(n, p, r0)
            res = minimize(compute_clearance, centers.flatten(), args=(n,), method='Nelder-Mead',
                           options={'maxiter': 8000, 'xatol': 1e-9, 'fatol': 1e-12})
            opt_centers = res.x.reshape(n, 2)
            
            radii, s = solve_lp_radii(opt_centers, n)
            if radii is not None and s > best_sum:
                best_sum = s
                best_centers = opt_centers.copy()
                best_radii = radii.copy()

    # Phase 2: Joint optimization & perturbation search
    if best_centers is not None:
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        cons = {'type': 'ineq', 'fun': constraints_joint, 'args': (n,)}
        
        x0 = np.zeros(3*n)
        x0[:n] = best_centers[:, 0]
        x0[n:2*n] = best_centers[:, 1]
        x0[2*n:] = best_radii
        
        # Direct joint refinement
        res = minimize(objective_joint, x0, args=(n,), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14})
        
        if np.isfinite(res.fun):
            cx, cy, r = res.x[:n], res.x[n:2*n], res.x[2*n:]
            centers = np.column_stack((cx, cy))
            radii_lp, s_lp = solve_lp_radii(centers, n)
            if radii_lp is not None and s_lp > best_sum:
                best_sum = s_lp
                best_centers = centers
                best_radii = radii_lp
                
        # Perturbation search to escape local minima
        for iter_num in range(25):
            scale = 0.96 - 0.8 * (iter_num / 25.0)
            pert = rng.uniform(-0.006 * scale, 0.006 * scale, (n, 2))
            c_pert = np.clip(best_centers + pert, 0.01, 0.99)
            
            x0_pert = np.zeros(3*n)
            x0_pert[:n] = c_pert[:, 0]
            x0_pert[n:2*n] = c_pert[:, 1]
            x0_pert[2*n:] = best_radii * 0.94
            
            res_pert = minimize(objective_joint, x0_pert, args=(n,), method='SLSQP', bounds=bounds,
                                constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
            
            if np.isfinite(res_pert.fun):
                cx, cy, r = res_pert.x[:n], res_pert.x[n:2*n], res_pert.x[2*n:]
                centers_p = np.column_stack((cx, cy))
                radii_lp, s_lp = solve_lp_radii(centers_p, n)
                if radii_lp is not None and s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = centers_p
                    best_radii = radii_lp

    # Fallback configuration
    if best_centers is None:
        best_centers = get_hex_centers(n, [5, 6, 5, 6, 4], 0.09)
        best_radii = np.full(n, 0.09)
        best_sum = np.sum(best_radii)

    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
