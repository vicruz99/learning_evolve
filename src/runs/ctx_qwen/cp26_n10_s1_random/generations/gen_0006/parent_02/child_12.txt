import numpy as np
from scipy.optimize import minimize, linprog

def compute_clearance(centers):
    """Computes the maximum feasible equal radius for the given centers."""
    n = centers.shape[0]
    # Distance to boundaries
    dw = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    min_dw = np.min(dw)
    
    # Pairwise distances
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, 1.0)  # Ignore self-distance
    min_dij = np.min(dists) / 2.0
    
    return min(min_dw, min_dij)

def neg_clearance(centers_flat):
    """Objective for Nelder-Mead: maximize clearance => minimize negative clearance."""
    centers = centers_flat.reshape(26, 2)
    return -compute_clearance(centers)

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    for k, (i, j) in enumerate(zip(idx_i, idx_j)):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def make_hex(n, rows, r0):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_clearance = 0.0
    best_centers_eq = None

    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 6, 6, 5, 4], [6, 6, 5, 5, 4], [5, 5, 5, 5, 6], [5, 7, 5, 5, 4],
        [4, 5, 6, 5, 6], [6, 4, 6, 5, 5], [5, 5, 6, 4, 6], [5, 6, 4, 6, 5]
    ]
    
    inits = []
    for pat in row_patterns:
        if sum(pat) < n: continue
        for r0 in [0.08, 0.09, 0.10, 0.11]:
            c = make_hex(n, pat, r0)
            inits.append(c)
            for _ in range(3):
                cp = c + rng.uniform(-0.02, 0.02, c.shape)
                cp = np.clip(cp, 0.05, 0.95)
                inits.append(cp)
                
    for _ in range(10):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # Phase 1: Optimize centers for equal radii using derivative-free method
    for cfg in inits:
        try:
            res = minimize(neg_clearance, cfg.flatten(), method='Nelder-Mead',
                          options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-10})
            c_opt = res.x.reshape(n, 2)
            cl = compute_clearance(c_opt)
            if cl > best_clearance:
                best_clearance = cl
                best_centers_eq = c_opt.copy()
        except Exception:
            continue
            
    # Phase 2: LP refinement for variable radii
    # Solving LP on the best centers found usually extracts the optimal sum
    radii_lp, sum_lp = solve_lp_radii(best_centers_eq)
    best_centers = best_centers_eq
    best_radii = radii_lp if radii_lp is not None else np.full(n, best_clearance)
    best_sum = sum_lp if radii_lp is not None else n * best_clearance
    
    # Phase 3: Safety scaling to guarantee strict numerical validity for checker
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
