

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we target the known near-optimal configuration which closely resembles a hexagonal lattice with row counts like [5, 6, 5, 6, 4]. The optimal equal-radius packing for N=26 yields r ≈ 0.10139, giving a sum ≈ 2.636. 

My strategy involves a two-phase optimization pipeline:
1. **Equal-Radius Phase**: Optimize centers and a single radius `r` to maximize `r`. This focuses the search on the densest possible geometric arrangement without worrying about radius inequalities yet. We use SLSQP with multiple high-quality hexagonal starts and perturbations.
2. **Joint Phase**: Starting from the best equal-radius configuration, optimize both centers and individual radii to maximize `sum(r_i)`. This allows radii to vary if it yields a better sum (though equal radii are typically optimal for this objective).
3. **LP Refinement**: For the best center configuration found, solve a Linear Program to find the exact maximal radii satisfying `r_i + r_j <= dist(i,j)` and boundary constraints. This step is exact and often squeezes out tiny improvements.
4. **Safety Scaling**: Apply a conservative scaling factor to guarantee strict validity against the `1e-12` tolerance in the checker.

Key implementation details:
- Vectorized constraint evaluation for speed and numerical stability.
- Robust multi-start initialization from diverse hexagonal patterns.
- Strict feasibility enforcement via bounds and constraint formulation.
- No closures or lambdas per instructions.

import numpy as np
from scipy.optimize import minimize, linprog

def objective_equal(v, N):
    """Objective: maximize equal radius r => minimize -v[-1]"""
    return -v[2 * N]

def constraints_equal(v, N):
    """Inequality constraints >= 0 for equal radius packing."""
    cx = v[0::2]
    cy = v[1::2]
    r = v[-1]
    
    # Boundary constraints: 4N
    c = np.empty(4 * N + N * (N - 1) // 2)
    idx = 0
    c[idx:idx + N] = cx - r; idx += N
    c[idx:idx + N] = 1.0 - cx - r; idx += N
    c[idx:idx + N] = cy - r; idx += N
    c[idx:idx + N] = 1.0 - cy - r; idx += N
    
    # Pairwise constraints: dist^2 >= 4r^2
    cx_mat = cx[:, None]
    cy_mat = cy[:, None]
    dx = cx_mat - cx_mat.T
    dy = cy_mat - cy_mat.T
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c[idx:] = (dx**2 + dy**2)[mask] - 4.0 * r**2
    
    return c

def objective_joint(v, N):
    """Objective: maximize sum of radii => minimize -sum(v[2N:])"""
    return -np.sum(v[2 * N:])

def constraints_joint(v, N):
    """Inequality constraints >= 0 for joint center/radius optimization."""
    cx = v[:N]
    cy = v[N:2 * N]
    r = v[2 * N:]
    
    c = np.empty(4 * N + N * (N - 1) // 2)
    idx = 0
    c[idx:idx + N] = cx - r; idx += N
    c[idx:idx + N] = 1.0 - cx - r; idx += N
    c[idx:idx + N] = cy - r; idx += N
    c[idx:idx + N] = 1.0 - cy - r; idx += N
    
    # Pairwise constraints: dist^2 >= (r_i + r_j)^2
    cx_mat = cx[:, None]
    cy_mat = cy[:, None]
    r_mat = r[:, None]
    dx = cx_mat - cx_mat.T
    dy = cy_mat - cy_mat.T
    dr = r_mat + r_mat.T
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c[idx:] = (dx**2 + dy**2)[mask] - dr[mask]**2
    
    return c

def solve_lp_radii(centers, N):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 0.0)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(N, k=1)
    m = N * (N - 1) // 2
    A_ub = np.zeros((m, N))
    b_ub = dists[idx]
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def make_hex_init(N, rows_pattern, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < N:
        if rng is not None:
            pts.append(rng.uniform(r0, 1.0 - r0, 2).tolist())
        else:
            pts.append([0.5, 0.5])
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_vars = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons_equal = {'type': 'ineq', 'fun': constraints_equal, 'args': (N,)}
    cons_joint = {'type': 'ineq', 'fun': constraints_joint, 'args': (N,)}
    
    rng = np.random.default_rng(42)
    inits = []
    
    # Diverse hexagonal row distributions summing to >= 26
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], 
        [6, 6, 5, 5, 4], [5, 6, 4, 6, 5], [5, 5, 5, 6, 5]
    ]
    
    for p in patterns:
        if sum(p) < N: continue
        base = make_hex_init(N, p, 0.085, rng)
        inits.append(base)
        for _ in range(3):
            pert = base + rng.uniform(-0.015, 0.015, base.shape)
            inits.append(np.clip(pert, 0.05, 0.95))
            
    # Add fully random dense starts
    for _ in range(6):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # Phase 1: Optimize for maximal equal radius
    for cfg in inits:
        x0 = np.zeros(2 * N + 1)
        x0[0::2] = cfg[:, 0]
        x0[1::2] = cfg[:, 1]
        x0[-1] = 0.075  # Feasible initial radius
        
        try:
            res = minimize(objective_equal, x0, args=(N,), method='SLSQP',
                           bounds=[(0.0, 1.0)]*(2*N) + [(0.05, 0.15)],
                           constraints=cons_equal,
                           options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
            if np.isfinite(res.fun):
                r_opt = res.x[-1]
                centers_opt = np.column_stack((res.x[0::2], res.x[1::2]))
                if r_opt > 0.095: # Keep only promising results
                    # Phase 2: Joint refinement from equal-radius optimum
                    x0_j = np.zeros(3 * N)
                    x0_j[:N] = centers_opt[:, 0]
                    x0_j[N:2*N] = centers_opt[:, 1]
                    x0_j[2*N:] = r_opt * 0.98
                    
                    res_j = minimize(objective_joint, x0_j, args=(N,), method='SLSQP',
                                     bounds=bounds_vars, constraints=cons_joint,
                                     options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
                    
                    if np.isfinite(res_j.fun):
                        c_j = np.column_stack((res_j.x[:N], res_j.x[N:2*N]))
                        r_j = res_j.x[2*N:]
                        s_j = np.sum(r_j)
                        
                        # LP refinement for fixed centers
                        r_lp, s_lp = solve_lp_radii(c_j, N)
                        if r_lp is not None and s_lp > best_sum:
                            best_sum = s_lp
                            best_centers = c_j
                            best_radii = r_lp
        except Exception:
            continue

    # Fallback if optimization failed
    if best_centers is None:
        fallback = make_hex_init(N, [5, 6, 5, 6, 4], 0.085, rng)
        r_fb, _ = solve_lp_radii(fallback, N)
        best_centers = fallback
        best_radii = r_fb if r_fb is not None else np.full(N, 0.08)
        best_sum = np.sum(best_radii)
        
    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(N):
        for j in range(i + 1, N):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
