import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n, idx_i, idx_j):
    """Solves the LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 0.0)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m_pairs = len(idx_i)
    A_ub_list = []
    b_ub_list = []
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for k in range(m_pairs):
        i, j = idx_i[k], idx_j[k]
        row = np.zeros(n)
        row[i] = 1.0
        row[j] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(dists[i, j])
        
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        for lim in [centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1]]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(lim)
            
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def obj_equal_R(vars, n):
    """Objective: minimize negative R => maximize R"""
    return -vars[-1]

def cons_equal_R(vars, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid equal-radius packing."""
    cx = vars[:n]
    cy = vars[n:2*n]
    R = vars[-1]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([cx - R, 1.0 - cx - R, cy - R, 1.0 - cy - R])
    
    # Pairwise non-overlap: dist^2 >= (2R)^2 = 4R^2
    cx_m = cx[:, None]
    cy_m = cy[:, None]
    dx = cx_m - cx_m.T
    dy = cy_m - cy_m.T
    dist_sq = dx[idx_i, idx_j]**2 + dy[idx_i, idx_j]**2
    c = np.concatenate([c, dist_sq - 4.0 * R**2])
    return c

def make_hex_init(n, rows_pattern, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append(rng.uniform(r0, 1.0 - r0, 2).tolist())
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    idx_i, idx_j = np.triu_indices(n, k=1)
    
    bounds_vars = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)]
    cons_dict = {'type': 'ineq', 'fun': cons_equal_R, 'args': (n, idx_i, idx_j)}
    
    # Diverse row distributions summing to 26
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [6, 6, 5, 5, 4], [5, 7, 5, 5, 4], [5, 5, 5, 6, 5], [5, 6, 4, 6, 5],
        [6, 4, 6, 5, 5], [5, 5, 5, 5, 6], [5, 6, 5, 5, 5], [4, 5, 6, 5, 6]
    ]
    
    inits = []
    for p in patterns:
        if sum(p) < n: continue
        inits.append(make_hex_init(n, p, 0.082, rng))
        inits.append(make_hex_init(n, p, 0.092, rng))
        
    # Add random dense starts
    for _ in range(8):
        inits.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Optimize equal radius R for diverse starts
    for cfg in inits:
        x0 = np.concatenate([cfg.flatten(), [0.085]])
        try:
            res = minimize(obj_equal_R, x0, args=(n,), method='SLSQP', bounds=bounds_vars,
                           constraints=cons_dict, options={'maxiter': 8000, 'ftol': 1e-12})
            if np.isfinite(res.fun):
                centers_opt = res.x[:2*n].reshape(n, 2)
                R_eq = res.x[-1]
                if R_eq > 0.05:
                    r_lp, s_lp = solve_lp_radii(centers_opt, n, idx_i, idx_j)
                    if r_lp is not None and s_lp > best_sum:
                        best_sum = s_lp
                        best_centers = centers_opt.copy()
                        best_radii = r_lp.copy()
        except Exception:
            continue
            
    # Phase 2: Coordinate ascent local search on centers
    if best_centers is not None:
        step = 0.006
        for iteration in range(600):
            idx = rng.integers(n)
            old_pos = best_centers[idx].copy()
            best_centers[idx] += rng.uniform(-step, step, 2)
            best_centers[idx] = np.clip(best_centers[idx], 0.01, 0.99)
            
            r_new, s_new = solve_lp_radii(best_centers, n, idx_i, idx_j)
            if r_new is not None and s_new > best_sum + 1e-9:
                best_sum = s_new
                best_radii = r_new.copy()
            else:
                best_centers[idx] = old_pos
                
            if iteration % 100 == 0 and iteration > 0:
                step *= 0.85  # Gradually shrink step size

    # Fallback if optimization fails
    if best_centers is None:
        fallback = make_hex_init(n, [5, 6, 5, 6, 4], 0.085, rng)
        best_centers = fallback
        radii_fb, _ = solve_lp_radii(fallback, n, idx_i, idx_j)
        best_radii = radii_fb if radii_fb is not None else np.full(n, 0.08)
        best_sum = np.sum(best_radii)

    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for k in range(len(idx_i)):
        i, j = idx_i[k], idx_j[k]
        d = np.linalg.norm(best_centers[i] - best_centers[j])
        rs = best_radii[i] + best_radii[j]
        if rs > 1e-12:
            scale = min(scale, d / rs)
            
    best_radii *= scale * 0.9999995
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
