import numpy as np
from scipy.optimize import minimize, linprog

def compute_clearance(centers, n):
    """Computes the minimum clearance (distance to boundaries and other centers)."""
    # Distance to boundaries
    d_bound = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    min_bound = np.min(d_bound)
    
    # Distance to other centers
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pairwise = np.min(dists) / 2.0
    
    return min(min_bound, min_pairwise)

def neg_clearance_obj(x, n):
    """Objective for center optimization: maximize minimum clearance."""
    centers = x.reshape(n, 2)
    return -compute_clearance(centers, n)

def solve_lp_radii(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    lims = np.maximum(lims, 0.0)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def make_hex_init(n, rows_pattern, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append(rng.uniform(r0, 1 - r0, 2).tolist())
    return np.array(pts[:n])

def optimize_centers_from(x0, n, maxiter=2000):
    """Run local optimization on centers to maximize clearance."""
    try:
        res = minimize(neg_clearance_obj, x0, args=(n,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * (2 * n),
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return res.x.reshape(n, 2)
    except Exception:
        return x0.reshape(n, 2)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 5, 6, 4, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [4, 5, 6, 5, 6]
    ]
    
    inits = []
    for p in patterns:
        if sum(p) < n: continue
        inits.append(make_hex_init(n, p, 0.085, rng))
        inits.append(make_hex_init(n, p, 0.095, rng))
        inits.append(make_hex_init(n, p, 0.105, rng))
        
    # Add random dense starts
    for _ in range(10):
        inits.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    # Phase 1: Multi-start local optimization
    for cfg in inits:
        opt_c = optimize_centers_from(cfg.flatten(), n, maxiter=3000)
        radii, s = solve_lp_radii(opt_c, n)
        if radii is not None and s > best_sum:
            best_sum = s
            best_centers = opt_c.copy()
            best_radii = radii.copy()
            
    # Phase 2: Coordinate Descent / Perturbation Search
    if best_centers is not None:
        for _ in range(150):
            idx = rng.integers(n)
            pert = best_centers.copy()
            pert[idx] += rng.uniform(-0.02, 0.02, 2)
            pert[idx] = np.clip(pert[idx], 0.01, 0.99)
            
            opt_c = optimize_centers_from(pert.flatten(), n, maxiter=1500)
            radii, s = solve_lp_radii(opt_c, n)
            
            if radii is not None and s > best_sum + 1e-9:
                best_sum = s
                best_centers = opt_c.copy()
                best_radii = radii.copy()
                
    # Fallback
    if best_centers is None:
        best_centers = make_hex_init(n, [5, 6, 5, 6, 4], 0.09, rng)
        radii, _ = solve_lp_radii(best_centers, n)
        best_radii = radii if radii is not None else np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    # Final strict validity scaling
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
