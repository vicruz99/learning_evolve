

<strategy>
1. **Two-Phase Optimization**: First, optimize circle centers assuming equal radii to maximize the minimum clearance (maximin problem). This avoids the non-convex coupling of different radii during the initial geometry search and leverages the robustness of SLSQP for distance constraints.
2. **LP Refinement**: For each optimized center layout, solve a Linear Program (LP) to maximize the sum of radii. This step exploits any asymmetrical gaps by allowing radii to vary, which is often necessary to squeeze out the last bits of sum.
3. **Joint Fine-Tuning**: Start a joint optimization (centers + radii) from the best LP-refined configurations using SLSQP. This allows simultaneous adjustments to break symmetry traps and further increase the objective.
4. **Diverse Initialization**: Use multiple hexagonal lattice patterns known to be near-optimal for N=26 (e.g., row counts [5,6,5,6,4]), perturbed and scaled. Add random dense starts to cover the search space.
5. **Strict Validation & Scaling**: Apply a final deterministic scaling factor to guarantee all constraints are satisfied within the checker's 1e-12 tolerance, preventing numerical failures.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 0.0)
    bounds = [(0.0, lim) for lim in lims]
    
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, 1e9)
    
    idx = np.triu_indices(n, k=1)
    m = len(idx[0])
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx[0]] = 1.0
    A_ub[np.arange(m), idx[1]] = 1.0
    b_ub = dists[idx]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return -res.fun, np.maximum(res.x, 0.0)
    except Exception:
        pass
    return 0.0, np.zeros(n)

def objective_equal(vars_arr, n):
    return -vars_arr[2*n]

def constraints_equal(vars_arr, n):
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    R = vars_arr[2*n]
    c = np.concatenate([cx - R, 1.0 - cx - R, cy - R, 1.0 - cy - R])
    cx_m = cx[:, None]
    cy_m = cy[:, None]
    d = np.hypot(cx_m - cx_m.T, cy_m - cy_m.T)
    idx = np.triu_indices(n, k=1)
    c = np.concatenate([c, d[idx] - 2.0 * R])
    return c

def objective_joint(vars_arr, n):
    return -np.sum(vars_arr[2*n:])

def constraints_joint(vars_arr, n):
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    cx_m = cx[:, None]
    cy_m = cy[:, None]
    d = np.hypot(cx_m - cx_m.T, cy_m - cy_m.T)
    idx = np.triu_indices(n, k=1)
    
    r_sum = r[:, None] + r[None, :]
    c = np.concatenate([c, d[idx] - r_sum[idx]])
    return c

def make_hex(n, rows, R0, rng=None):
    pts = []
    y = R0
    for ri, cnt in enumerate(rows):
        shift = R0 if ri % 2 == 1 else 0.0
        x = R0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2 * R0
        y += np.sqrt(3) * R0
    while len(pts) < n:
        if rng:
            pts.append(rng.uniform(R0, 1-R0, 2).tolist())
        else:
            pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N
    rng = np.random.default_rng(42)
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [5, 7, 5, 5, 4], [6, 6, 5, 5, 4],
        [5, 6, 4, 6, 5], [5, 5, 5, 5, 6], [6, 5, 5, 5, 5]
    ]
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    starts = []
    for p in patterns:
        if sum(p) < n: continue
        for R0 in [0.08, 0.09, 0.10, 0.11]:
            starts.append(make_hex(n, p, R0, rng))
    for _ in range(6):
        starts.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    bounds_eq = [(0.0, 1.0)]*(2*n) + [(0.0, 0.25)]
    cons_eq = {'type': 'ineq', 'fun': constraints_equal, 'args': (n,)}
    bounds_jnt = [(0.0, 1.0)]*(2*n) + [(1e-7, 0.5)]*n
    cons_jnt = {'type': 'ineq', 'fun': constraints_joint, 'args': (n,)}
    
    # Phase 1: Equal radius optimization -> LP refinement
    for cfg in starts:
        x0_eq = np.concatenate([cfg.flatten(), [0.05]])
        try:
            res = minimize(objective_equal, x0_eq, args=(n,), method='SLSQP',
                           bounds=bounds_eq, constraints=cons_eq,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if np.isfinite(res.fun):
                centers_eq = res.x[:2*n].reshape(n, 2)
                s_lp, r_lp = solve_lp_radii(centers_eq)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = centers_eq.copy()
                    best_radii = r_lp.copy()
        except Exception:
            continue
            
    # Phase 2: Joint optimization from best found + perturbations
    if best_centers is not None:
        for _ in range(12):
            pert_c = best_centers + rng.uniform(-0.004, 0.004, (n, 2))
            pert_c = np.clip(pert_c, 0.02, 0.98)
            x0_jnt = np.concatenate([pert_c.flatten(), best_radii * 0.97])
            
            try:
                res = minimize(objective_joint, x0_jnt, args=(n,), method='SLSQP',
                               bounds=bounds_jnt, constraints=cons_jnt,
                               options={'maxiter': 7000, 'ftol': 1e-13, 'disp': False})
                if np.isfinite(res.fun):
                    c_jnt = res.x[:2*n].reshape(n, 2)
                    r_jnt = res.x[2*n:]
                    
                    # LP refine on joint centers to guarantee max sum for geometry
                    s_lp, r_lp = solve_lp_radii(c_jnt)
                    if s_lp > best_sum:
                        best_sum = s_lp
                        best_centers = c_jnt.copy()
                        best_radii = r_lp.copy()
            except Exception:
                continue
                
    # Fallback
    if best_centers is None:
        best_centers = make_hex(n, [5, 6, 5, 6, 4], 0.09, rng)
        best_radii = np.full(n, 0.085)
        best_sum = np.sum(best_radii)
        
    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
