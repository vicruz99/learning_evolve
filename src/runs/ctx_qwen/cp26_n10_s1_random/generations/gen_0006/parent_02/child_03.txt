import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRI_I, TRI_J = np.triu_indices(N_CIRCLES, k=1)

def objective_func(vars_array):
    """Objective: minimize negative sum of radii => maximize sum of radii"""
    return -np.sum(vars_array[2 * N_CIRCLES:])

def constraint_func(vars_array):
    """
    Computes inequality constraints >= 0 for valid packing.
    Layout: [x0, y0, ..., x25, y25, r0, ..., r25]
    """
    cx = vars_array[:N_CIRCLES]
    cy = vars_array[N_CIRCLES:2*N_CIRCLES]
    r = vars_array[2*N_CIRCLES:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[TRI_I] - cx[TRI_J]
    dy = cy[TRI_I] - cy[TRI_J]
    dr = r[TRI_I] + r[TRI_J]
    c_pair = (dx**2 + dy**2) - dr**2
    
    return np.concatenate([c, c_pair])

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Maximum radius allowed by boundaries
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 0.0)
    bounds = [(0.0, lim) for lim in lims]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def make_hex_init(rows_pattern, r0, rng=None):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows_pattern):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N_CIRCLES: break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    while len(pts) < N_CIRCLES:
        if rng is not None:
            pts.append(rng.uniform(r0, 1-r0, 2).tolist())
        else:
            pts.append([0.5, 0.5])
    return np.array(pts[:N_CIRCLES])

def run_slsqp(cfg):
    """Runs SLSQP optimization from a given configuration."""
    n = N_CIRCLES
    bounds_vars = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    x0 = np.zeros(3 * n)
    x0[:n] = cfg[:, 0]
    x0[n:2*n] = cfg[:, 1]
    x0[2*n:] = 0.06  # Feasible small initial radii
    
    try:
        res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_vars,
                       constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
        if np.isfinite(res.fun):
            cx, cy, r = res.x[:n], res.x[n:2*n], res.x[2*n:]
            centers = np.column_stack((cx, cy))
            return centers, r
    except Exception:
        pass
    return None, None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    rng = np.random.default_rng(42)
    
    # Diverse row distributions known to be near-optimal for n=26
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 5, 5, 6, 5], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 5, 6, 4, 6],
        [5, 6, 5, 5, 5], [4, 5, 6, 5, 6], [5, 5, 5, 5, 4, 2]
    ]
    
    inits = []
    for p in patterns:
        if sum(p) < n: continue
        for r0 in [0.085, 0.095, 0.105]:
            inits.append(make_hex_init(p, r0, rng))
            
    # Add random dense starts
    for _ in range(15):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: SLSQP from diverse starts + LP refinement
    for cfg in inits:
        c, r = run_slsqp(cfg)
        if c is not None:
            r_lp, s_lp = solve_lp_radii(c)
            if r_lp is not None:
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c.copy()
                    best_radii = r_lp.copy()
                    
    # Phase 2: Random walk refinement on centers to escape local minima
    if best_centers is not None:
        step = 0.018
        for _ in range(2500):
            idx = rng.integers(n)
            old_pos = best_centers[idx].copy()
            
            # Perturb one center
            best_centers[idx] += rng.uniform(-step, step, 2)
            best_centers[idx] = np.clip(best_centers[idx], 0.02, 0.98)
            
            r_new, s_new = solve_lp_radii(best_centers)
            if r_new is not None and s_new > best_sum + 1e-9:
                best_sum = s_new
                best_radii = r_new.copy()
            else:
                best_centers[idx] = old_pos
                
            # Anneal step size
            step *= 0.9975
            
    # Fallback configuration if optimization fails unexpectedly
    if best_centers is None:
        best_centers = make_hex_init([5, 6, 5, 6, 4], 0.085, rng)
        r_fb, _ = solve_lp_radii(best_centers)
        best_radii = r_fb if r_fb is not None else np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    # Final safety scaling to guarantee strict numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]

            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)

            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
