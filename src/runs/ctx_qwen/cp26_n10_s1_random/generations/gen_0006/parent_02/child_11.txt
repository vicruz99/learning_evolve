import numpy as np
from scipy.optimize import minimize, linprog

def compute_min_clearance(centers):
    """Computes the maximum feasible equal radius for the given centers."""
    n = centers.shape[0]
    # Boundary clearances
    left = centers[:, 0]
    right = 1.0 - centers[:, 0]
    bottom = centers[:, 1]
    top = 1.0 - centers[:, 1]
    clearances = np.minimum(np.minimum(left, right), np.minimum(bottom, top))
    min_c = np.min(clearances)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_c = min(min_c, np.min(dists) / 2.0)
    return min_c

def neg_min_clearance(x):
    """Objective for optimizer: minimize negative min clearance."""
    return -compute_min_clearance(x.reshape(26, 2))

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 0.0)) for l in lims]
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = dists[idx]
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def make_hex(n, rows, r0):
    """Generates a centered hexagonal lattice initialization."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        row_w = (cnt - 1) * 2 * r0
        x_start = 0.5 - row_w / 2.0 + shift
        for c in range(cnt):
            if len(pts) >= n: break
            pts.append([x_start + c * 2 * r0, y])
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
