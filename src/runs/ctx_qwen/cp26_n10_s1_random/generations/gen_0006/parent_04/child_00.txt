import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_lp_objective(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = len(centers)
    bnds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bnds.append((0.0, max(mx, 1e-9)))
        
    m = n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bnds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def neg_sum_radii(centers_flat):
    """Objective for center optimization: minimize negative LP sum of radii."""
    c = centers_flat.reshape(N, 2)
    return -compute_lp_objective(c)[1]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    
    inits = []
    
    # Hexagonal lattice patterns known to be near-optimal for n=26
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 6, 4],
        [4, 6, 6, 6, 4], [5, 6, 6, 5, 4], [6, 6, 5, 5, 4],
        [5, 5, 5, 5, 6], [6, 4, 6, 5, 5]
    ]
    
    for pat in patterns:
        pts = []
        r0 = 0.10
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N:
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        if len(pts) == N:
            pts_arr = np.array(pts)
            inits.append(pts_arr)
            # Add perturbations to escape symmetry traps
            for _ in range(2):
                p = pts_arr + rng.uniform(-0.025, 0.025, (N, 2))
                inits.append(np.clip(p, 0.05, 0.95))
                
    # Random dense starts
    for _ in range(4):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Phase 1: Derivative-free optimization on centers
    for cfg in inits:
        cfg = np.clip(cfg, 1e-4, 1.0 - 1e-4)
        
        try:
            res_pow = minimize(neg_sum_radii, cfg.flatten(), method='Powell',
                               options={'maxiter': 300, 'xtol': 1e-7, 'ftol': 1e-7})
            if -res_pow.fun > best_sum:
                best_sum = -res_pow.fun
                best_centers = res_pow.x.reshape(N, 2)
                best_radii, _ = compute_lp_objective(best_centers)
        except Exception:
            pass
            
        try:
            res_nm = minimize(neg_sum_radii, cfg.flatten(), method='Nelder-Mead',
                              options={'maxiter': 400, 'xatol': 1e-7, 'fatol': 1e-7})
            if -res_nm.fun > best_sum:
                best_sum = -res_nm.fun
                best_centers = res_nm.x.reshape(N, 2)
                best_radii, _ = compute_lp_objective(best_centers)
        except Exception:
            pass
            
    # Phase 2: Local stochastic refinement
    if best_centers is not None:
        current_sum = best_sum
        current_centers = best_centers.copy()
        current_radii = best_radii.copy()
        
        for step in range(25):
            improved = False
            step_size = 0.004 * (0.9 ** step)
            for i in range(N):
                best_loc_sum = current_sum
                best_loc_c = current_centers
                best_loc_r = current_radii
                
                for _ in range(4):
                    trial = current_centers.copy()
                    trial[i] += rng.uniform(-step_size, step_size, 2)
                    trial[i] = np.clip(trial[i], 1e-4, 1.0 - 1e-4)
                    r_t, s_t = compute_lp_objective(trial)
                    if s_t > best_loc_sum:
                        best_loc_sum = s_t
                        best_loc_c = trial
                        best_loc_r = r_t
                        
                if best_loc_sum > current_sum:
                    current_sum = best_loc_sum
                    current_centers = best_loc_c
                    current_radii = best_loc_r
                    improved = True
            if not improved:
                break
                
        best_centers = current_centers
        best_radii = current_radii
        best_sum = current_sum

    # Phase 3: Final exact LP solve and safety scaling
    r_final, s_final = compute_lp_objective(best_centers)
    best_radii = r_final
    best_sum = s_final
    
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if r < 1e-9:
            continue
        scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
        
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs < 1e-9:
                continue
            scale = min(scale, d / rs)
            
    best_radii *= scale * 0.9999995
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(final_sum)
