import numpy as np
from scipy.optimize import linprog, minimize

N = 26
M = N * (N - 1) // 2

# Precompute constant A_ub matrix structure for LP constraints: r_i + r_j <= dist(i,j)
A_ub = np.zeros((M, N))
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        idx += 1

def lp_max_sum(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = N
    c_obj = -np.ones(n)
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        w = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(w, 1e-9)))
        
    # Compute pairwise distances efficiently
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    b_ub = dists[np.triu_indices(n, k=1)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def neg_lp_sum(centers_flat):
    """Objective for optimization: maximize sum of radii via LP."""
    centers = np.clip(centers_flat.reshape(N, 2), 1e-5, 1.0 - 1e-5)
    _, s = lp_max_sum(centers)
    return -s

def make_hex(rows, r):
    """Generates a centered hexagonal lattice configuration."""
    pts = []
    y = r
    for ri, cnt in enumerate(rows):
        shift = r if ri % 2 == 1 else 0.0
        row_w = (cnt - 1) * 2.0 * r
        x0 = 0.5 - row_w / 2.0 + shift
        for c in range(cnt):
            pts.append([x0 + c * 2.0 * r, y])
        y += np.sqrt(3) * r
    return np.array(pts[:N])

def penalty_obj(v):
    """Smooth penalty objective for joint center/radius optimization."""
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    c = np.column_stack((cx, cy))
    
    wall = np.minimum(np.minimum(cx, 1.0 - cx), np.minimum(cy, 1.0 - cy))
    wp = np.sum(np.maximum(0.0, r - wall)**2)
    
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    np.fill_diagonal(dists, np.inf)
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    op = np.sum(np.maximum(0.0, rs - dists)**2)
    
    return -np.sum(r) + 5000.0 * (wp + op)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    
    # Diverse row distributions known to yield dense packings for N=26
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4],
        [6, 4, 6, 6, 4], [5, 6, 6, 5, 4], [7, 5, 6, 5, 3],
        [6, 6, 6, 4, 4], [5, 5, 5, 5, 6]
    ]
    
    inits = []
    for pat in patterns:
        if sum(pat) < N: continue
        for r0 in [0.095, 0.10, 0.105, 0.11]:
            c = make_hex(pat, r0)
            inits.append(c)
            for _ in range(4):
                cp = c + rng.uniform(-0.015, 0.015, c.shape)
                inits.append(cp)
                
    for _ in range(12):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Multi-start Nelder-Mead optimization on centers
    for cfg in inits:
        try:
            res = minimize(neg_lp_sum, cfg.flatten(), method='Nelder-Mead',
                           options={'maxiter': 4000, 'xatol': 1e-9, 'fatol': 1e-12})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt = lp_max_sum(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Coordinate-wise hill climbing refinement
    if best_centers is not None:
        current_c = best_centers.copy()
        current_r = best_radii.copy()
        current_s = best_sum
        
        for step in range(800):
            idx = rng.integers(N)
            old_pos = current_c[idx].copy()
            step_sz = 0.025 * (0.985 ** step)
            current_c[idx] += rng.uniform(-step_sz, step_sz, 2)
            current_c[idx] = np.clip(current_c[idx], 0.01, 0.99)
            
            r_try, s_try = lp_max_sum(current_c)
            if s_try > current_s + 1e-7:
                current_s = s_try
                current_r = r_try
                best_sum = current_s
                best_centers = current_c.copy()
                best_radii = current_r.copy()
            else:
                current_c[idx] = old_pos
                
        # Phase 3: Penalty-based L-BFGS-B polish to escape discrete traps
        x0_pen = np.concatenate([best_centers[:, 0], best_centers[:, 1], best_radii])
        bounds_pen = [(0.05, 0.95)] * (2 * N) + [(0.01, 0.4)] * N
        
        try:
            res_pen = minimize(penalty_obj, x0_pen, method='L-BFGS-B', bounds=bounds_pen,
                               options={'maxiter': 3000, 'ftol': 1e-14})
            c_pen = np.column_stack((res_pen.x[:N], res_pen.x[N:2*N]))
            r_pen, s_pen = lp_max_sum(c_pen)
            if s_pen > best_sum:
                best_centers = c_pen
                best_radii = r_pen
                best_sum = s_pen
        except Exception:
            pass
            
    # Fallback configuration
    if best_centers is None:
        best_centers = make_hex([5, 6, 5, 6, 4], 0.1)
        best_radii, best_sum = lp_max_sum(best_centers)
        
    # Safety scaling to guarantee strict numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if r < 1e-9: continue
        scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
        
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs < 1e-9: continue
            scale = min(scale, d / rs)
            
    best_radii *= scale * 0.9999995
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(final_sum)
