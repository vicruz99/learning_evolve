The goal is to pack 26 circles in a unit square to maximize the sum of radii. The known optimal configuration for N=26 corresponds to an equal-radius packing with radius r ≈ 0.10138, yielding a sum of radii ≈ 2.63588. To reach this target robustly, I will use a **bisection search on the minimum clearance D** (where radius r = D). For each candidate D, I solve a feasibility problem: find centers such that all pairwise distances are ≥ 2D and all points are at least D from the boundaries.

This feasibility problem is optimized using a **smooth penalty function** with **analytical gradients**, optimized via `L-BFGS-B`. The penalty sums squared violations of wall and pairwise constraints. Multiple restarts (hexagonal lattice, perturbed hex, random) ensure we escape local minima. The bisection tightly brackets the maximum feasible D.

After finding the optimal centers for the maximum equal D, I run an **LP refinement** to allow unequal radii, which can sometimes squeeze out marginal gains. Finally, a strict safety scaling step guarantees numerical validity against the checker's tolerances. This approach is mathematically sound, highly stable, and avoids the constraint-handling pitfalls of direct SLSQP optimization on thousands of inequality constraints.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def penalty_and_grad(x, D):
    """Computes penalty and analytical gradient for a given minimum clearance D."""
    c = x.reshape(N, 2)
    pen = 0.0
    grad = np.zeros((N, 2))
    
    # Wall constraints: x >= D, 1-x >= D, y >= D, 1-y >= D
    # Left wall violation
    vL = D - c[:, 0]
    m = vL > 0
    if np.any(m):
        pen += np.sum(vL[m]**2)
        grad[m, 0] -= 2.0 * vL[m]
        
    # Right wall violation
    vR = c[:, 0] + D - 1.0
    m = vR > 0
    if np.any(m):
        pen += np.sum(vR[m]**2)
        grad[m, 0] += 2.0 * vR[m]
        
    # Bottom wall violation
    vB = D - c[:, 1]
    m = vB > 0
    if np.any(m):
        pen += np.sum(vB[m]**2)
        grad[m, 1] -= 2.0 * vB[m]
        
    # Top wall violation
    vT = c[:, 1] + D - 1.0
    m = vT > 0
    if np.any(m):
        pen += np.sum(vT[m]**2)
        grad[m, 1] += 2.0 * vT[m]
        
    # Pairwise constraints: dist(i,j) >= 2D
    idx_i, idx_j = np.triu_indices(N, k=1)
    diff = c[idx_i] - c[idx_j]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    dists_safe = np.maximum(dists, 1e-12)
    
    v_p = 2.0 * D - dists
    m_p = v_p > 0
    if np.any(m_p):
        pen += np.sum(v_p[m_p]**2)
        # Gradient of (2D - d)^2 wrt positions pushes circles apart
        factor = -2.0 * v_p[m_p] / dists_safe[m_p]
        grad_c = np.zeros((N, 2))
        grad_c[idx_i[m_p]] += factor[:, None] * diff[m_p]
        grad_c[idx_j[m_p]] -= factor[:, None] * diff[m_p]
        grad += grad_c
        
    return pen, grad.flatten()

def objective_func(x, D):
    return penalty_and_grad(x, D)[0]

def gradient_func(x, D):
    return penalty_and_grad(x, D)[1]

def get_hex_init(D):
    """Generates a hexagonal lattice initialization tuned for clearance D."""
    pts = []
    y = D
    row = 0
    while y + D < 1.0 and len(pts) < N:
        shift = D if row % 2 == 1 else 0.0
        x = D + shift
        while x + D < 1.0 and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * D
        y += np.sqrt(3) * D
        row += 1
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        w = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(w, 1e-9)))
        
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = d
            k += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 0.0), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    low, high = 0.095, 0.105
    best_D = low
    best_centers = get_hex_init(best_D)
    
    bounds_opt = [(0.0, 1.0)] * (2 * N)
    
    # Bisection search for maximum feasible minimum clearance D
    for _ in range(25):
        mid = (low + high) / 2.0
        found = False
        for start_idx in range(4):
            if start_idx == 0:
                init_c = get_hex_init(mid)
            elif start_idx == 1:
                init_c = get_hex_init(mid) + rng.uniform(-0.02, 0.02, (N, 2))
                init_c = np.clip(init_c, 0.01, 0.99)
            else:
                init_c = rng.uniform(mid + 0.01, 1.0 - mid - 0.01, (N, 2))
                
            try:
                res = minimize(objective_func, init_c.flatten(), jac=gradient_func, args=(mid,),
                               method='L-BFGS-B', bounds=bounds_opt,
                               options={'maxiter': 5000, 'ftol': 1e-14})
                if res.fun < 1e-6:
                    best_centers = res.x.reshape(N, 2)
                    found = True
                    break
            except Exception:
                continue
                
        if found:
            low = mid
            best_D = mid
        else:
            high = mid
            
    # Start with equal radii from bisection
    radii_eq = np.full(N, best_D)
    current_sum = np.sum(radii_eq)
    
    # LP refinement to allow unequal radii and potentially increase sum
    r_lp, s_lp = solve_lp_radii(best_centers)
    if s_lp > current_sum:
        radii_eq = r_lp
        current_sum = s_lp
        
    # Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = radii_eq[i]
        if r < 1e-9: continue
        scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
        
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = radii_eq[i] + radii_eq[j]
            if rs < 1e-9: continue
            scale = min(scale, d/rs)
            
    radii_eq *= scale * 0.999999
    final_sum = np.sum(radii_eq)
    
    return best_centers, radii_eq, float(final_sum)
