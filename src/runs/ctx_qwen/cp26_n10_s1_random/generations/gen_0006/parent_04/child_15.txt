import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def compute_lp_sum_radii(centers, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers. Returns the sum."""
    n = centers.shape[0]
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 0.0)) for l in limits]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun
    except Exception:
        pass
    return 0.0

def neg_lp_sum(x, A_ub, idx_i, idx_j):
    """Objective for optimization: minimize negative sum of radii."""
    centers = x.reshape(N_CIRCLES, 2)
    # Clamp to prevent LP infeasibility from out-of-bounds centers
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    return -compute_lp_sum_radii(centers, A_ub, idx_i, idx_j)

def hex_init(row_counts, r0=0.10):
    """Generates a centered hexagonal lattice configuration."""
    pts = []
    y = r0
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N_CIRCLES:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < N_CIRCLES:
        pts.append([0.5, 0.5])
    return np.array(pts[:N_CIRCLES])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    idx_i, idx_j = np.triu_indices(N_CIRCLES, k=1)
    
    # Precompute constant LP constraint matrix structure
    A_ub = np.zeros((N_CIRCLES * (N_CIRCLES - 1) // 2, N_CIRCLES))
    k = 0
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            k += 1

    best_sum = -1.0
    best_centers = None

    # Generate diverse initial configurations
    starts = []
    hex_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 6, 5, 4]
    ]
    
    for pat in hex_patterns:
        c = hex_init(pat, r0=0.095)
        # Scale to tightly fit the unit square to improve starting density
        mn, mx = c.min(axis=0), c.max(axis=0)
        scale = 0.92 / np.max(mx - mn)
        c_scaled = (c - mn) * scale + 0.5 * (1.0 - scale)
        starts.append(c_scaled.flatten())
        
        # Add perturbed variants to break symmetry and explore nearby basins
        for _ in range(3):
            cp = c_scaled + rng.uniform(-0.02, 0.02, c_scaled.shape)
            starts.append(np.clip(cp, 0.05, 0.95).flatten())
            
    # Add fully random dense starts
    for _ in range(5):
        starts.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)).flatten())

    # Phase 1: Multi-start Nelder-Mead optimization
    for x0 in starts:
        try:
            res = minimize(
                neg_lp_sum, x0, args=(A_ub, idx_i, idx_j),
                method='Nelder-Mead',
                options={'maxiter': 8000, 'xatol': 1e-10, 'fatol': 1e-12}
            )
            if np.isfinite(res.fun):
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_centers = res.x.reshape(N_CIRCLES, 2)
        except Exception:
            continue

    if best_centers is None:
        best_centers = hex_init([5, 6, 5, 6, 4]).flatten().reshape(N_CIRCLES, 2)
        best_sum = compute_lp_sum_radii(best_centers, A_ub, idx_i, idx_j)

    # Phase 2: Coordinate ascent / Hill climbing to escape LP constraint plateaus
    current_centers = best_centers.copy()
    current_sum = best_sum
    step = 0.015
    for _ in range(600):
        i = rng.integers(N_CIRCLES)
        old_pos = current_centers[i].copy()
        
        # Random perturbation
        current_centers[i] += rng.uniform(-step, step, 2)
        current_centers[i] = np.clip(current_centers[i], 0.02, 0.98)
        
        s_new = compute_lp_sum_radii(current_centers, A_ub, idx_i, idx_j)
        if s_new > current_sum + 1e-9:
            current_sum = s_new
            step *= 1.005  # Slight increase on success
        else:
            current_centers[i] = old_pos
            step *= 0.996  # Decay on failure

    best_centers = current_centers
    best_sum = current_sum

    # Phase 3: Final exact LP extraction
    diffs = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    limits = np.minimum(np.minimum(best_centers[:, 0], 1.0 - best_centers[:, 0]),
                        np.minimum(best_centers[:, 1], 1.0 - best_centers[:, 1]))
    bounds_r = [(0.0, max(l, 0.0)) for l in limits]
    
    final_radii = np.full(N_CIRCLES, 1e-6)
    try:
        res = linprog(-np.ones(N_CIRCLES), A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            final_radii = res.x
    except Exception:
        pass

    # Phase 4: Strict safety scaling to guarantee checker tolerance compliance
    scale = 1.0
    for i in range(N_CIRCLES):
        x, y = best_centers[i]
        r = final_radii[i]
        if r < 1e-12:
            continue
        scale = min(scale, x/r, (1.0 - x)/r, y/r, (1.0 - y)/r)
        
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
            rs = final_radii[i] + final_radii[j]
            if rs < 1e-12:
                continue
            scale = min(scale, d / rs)
            
    # Apply tight scaling with minimal numerical margin
    final_radii *= scale * 0.99999995
    final_sum = float(np.sum(final_radii))
    
    return best_centers, final_radii, final_sum
