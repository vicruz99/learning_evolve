To maximize the sum of radii for 26 circles in a unit square, we aim for a configuration very close to the known optimal equal-radius packing (radius ≈ 0.101382, sum ≈ 2.6359). The strategy combines several powerful techniques:
1. **Boundary-Free Parameterization**: We parameterize centers as $x_i = r_i + (1-2r_i)u_i$, $y_i = r_i + (1-2r_i)v_i$ with $u_i, v_i \in [0,1]$. This automatically satisfies wall constraints, reducing the optimization to only handling pairwise non-overlap constraints.
2. **Diverse Hexagonal Initializations**: We generate multiple hexagonal lattice starts with varying row distributions (e.g., 5-6-5-6-4) known to approximate optimal packings, plus perturbed variants.
3. **SLSQP Refinement**: We run `SLSQP` from each start to quickly converge to a local optimum in the parameterized space.
4. **Exact LP Radii Extraction**: For fixed centers, we solve a Linear Program to find the exact maximum sum of radii. This is highly accurate and serves as the ground truth objective.
5. **Adaptive Hill Climbing**: We perform a stochastic local search on centers, evaluated via the LP, with an adaptive step size to escape shallow local minima.
6. **Equal-Radius Push**: Since equal radii often yield the global optimum for dense square packings, we run a Nelder-Mead optimization maximizing the minimum clearance (equivalent to maximizing equal radius), then re-evaluate radii via LP. This helps bridge the final gap to the target.
7. **Safety Scaling**: A final deterministic scaling ensures strict numerical validity against the checker's tolerance.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute LP constraint matrix for pairwise distances
A_ub_pairs = np.zeros((N * (N - 1) // 2, N))
k = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub_pairs[k, i] = 1.0
        A_ub_pairs[k, j] = 1.0
        k += 1

def solve_radii_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        w = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(w, 1e-9)))
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub.append(dist)
            
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c_obj, A_ub=A_ub_pairs, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def compute_constraints(vars_array, triu_i, triu_j):
    """Parameterization automatically satisfies boundary constraints."""
    r = vars_array[:N]
    u = vars_array[N:2*N]
    v = vars_array[2*N:3*N]
    
    # Numerical safety clipping
    r = np.clip(r, 0.0, 0.499999)
    u = np.clip(u, 0.0, 1.0)
    v = np.clip(v, 0.0, 1.0)
    
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return d2[triu_i, triu_j] - rs[triu_i, triu_j]**2

def objective(vars_array):
    return -np.sum(vars_array[:N])

def generate_hex_init(row_counts, r_scale=0.1):
    pts = []
    y = r_scale
    row_idx = 0
    for count in row_counts:
        shift = r_scale if row_idx % 2 == 1 else 0.0
        x = r_scale + shift
        for _ in range(count):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r_scale
        y += np.sqrt(3) * r_scale
        row_idx += 1
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def min_clearance_obj(cf, n):
    """Objective for equal-radius optimization: maximize minimum clearance."""
    c = cf.reshape(n, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    
    # Distance to walls
    wb = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), np.minimum(c[:, 1], 1.0 - c[:, 1]))
    
    # Pairwise distances
    diff = c[:, None, :] - c[None, :, :]
    d = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(d, 1.0) # Ignore self-distance
    
    # Max equal radius is limited by walls and half pairwise distances
    r_eq = np.min(np.minimum(wb, np.min(d, axis=1) / 2.0))
    return -r_eq * n

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    triu_i, triu_j = np.triu_indices(N, k=1)
    
    bounds_vars = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints, 'args': (triu_i, triu_j)}
    
    best_sum = -1.0
    best_vars = None
    
    # Diverse initial configurations based on known dense packings
    row_configs = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 6, 5, 4],
        [6, 4, 6, 6, 4], [5, 6, 4, 6, 5], [5, 5, 6, 6, 4]
    ]
    
    inits = []
    for rc in row_configs:
        c = generate_hex_init(rc, 0.095)
        inits.append(c)
        for _ in range(2):
            cp = c + rng.uniform(-0.025, 0.025, c.shape)
            inits.append(np.clip(cp, 0.05, 0.95))
            
    # Random dense starts
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # Phase 1: SLSQP optimization from diverse starts
    for cfg in inits:
        r0 = np.full(N, 0.09)
        denom = 1.0 - 2.0 * r0
        u0 = np.clip((cfg[:, 0] - r0) / denom, 0.0, 1.0)
        v0 = np.clip((cfg[:, 1] - r0) / denom, 0.0, 1.0)
        x0 = np.concatenate([r0, u0, v0])
        
        for _ in range(2):
            xp = x0.copy()
            xp[:N] *= rng.uniform(0.96, 1.04, N)
            xp[N:] += rng.uniform(-0.03, 0.03, 2*N)
            xp = np.clip(xp, [1e-6]*N + [0.0]*(2*N), [0.49]*N + [1.0]*(2*N))
            
            try:
                res = minimize(objective, xp, method='SLSQP', bounds=bounds_vars,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14})
                if np.isfinite(res.fun):
                    c_vals = compute_constraints(res.x, triu_i, triu_j)
                    if np.min(c_vals) > -1e-6:
                        s = -res.fun
                        if s > best_sum:
                            best_sum = s
                            best_vars = res.x.copy()
            except Exception:
                pass
                
    if best_vars is None:
        r0 = np.full(N, 0.09)
        best_vars = np.concatenate([r0, rng.rand(N), rng.rand(N)])
        best_sum = np.sum(r0)

    # Decode SLSQP results to centers
    r_slsqp = best_vars[:N]
    u_out = best_vars[N:2*N]
    v_out = best_vars[2*N:3*N]
    x_out = r_slsqp + (1.0 - 2.0 * r_slsqp) * u_out
    y_out = r_slsqp + (1.0 - 2.0 * r_slsqp) * v_out
    centers_opt = np.column_stack((x_out, y_out))
    
    # Phase 2: Exact LP refinement
    r_lp, sum_lp = solve_radii_lp(centers_opt)
    best_radii = r_lp
    best_sum_lp = sum_lp
    best_centers = centers_opt.copy()
    
    # Phase 3: Adaptive Hill Climbing on centers
    current_centers = best_centers.copy()
    current_radii = r_lp.copy()
    current_sum = sum_lp
    
    step = 0.012
    for iteration in range(1500):
        i = rng.integers(N)
        old_c = current_centers[i].copy()
        current_centers[i] += rng.uniform(-step, step, 2)
        current_centers[i] = np.clip(current_centers[i], 1e-4, 1.0 - 1e-4)
        
        r_try, s_try = solve_radii_lp(current_centers)
        if s_try > current_sum + 1e-8:
            current_sum = s_try
            current_radii = r_try
            step = min(0.02, step * 1.002)
        else:
            current_centers[i] = old_c
            step *= 0.998
            
    best_centers = current_centers
    best_radii = current_radii
    best_sum = current_sum
    
    # Phase 4: Equal-radius push to target known optimal configuration
    try:
        res_eq = minimize(min_clearance_obj, best_centers.flatten(), args=(N,),
                          method='Nelder-Mead', options={'maxiter': 2500, 'xatol': 1e-8, 'fatol': 1e-9})
        c_eq = np.clip(res_eq.x.reshape(N, 2), 1e-6, 1.0 - 1e-6)
        r_eq, s_eq = solve_radii_lp(c_eq)
        if s_eq > best_sum:
            best_centers = c_eq
            best_radii = r_eq
            best_sum = s_eq
    except Exception:
        pass
        
    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if r < 1e-12: continue
        scale = min(scale, x/r, (1.0 - x)/r, y/r, (1.0 - y)/r)
        
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs < 1e-12: continue
            scale = min(scale, d / rs)
            
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
