import numpy as np
from scipy.optimize import linprog

N = 26

# Precompute constant pairwise constraint matrix for the LP
# r_i + r_j <= dist(i, j)  =>  A_ub @ r <= b_ub
_PAIR_A = np.zeros((N * (N - 1) // 2, N))
_k = 0
for _i in range(N):
    for _j in range(_i + 1, N):
        _PAIR_A[_k, _i] = 1.0
        _PAIR_A[_k, _j] = 1.0
        _k += 1
_TRUU_I, _TRUU_J = np.triu_indices(N, k=1)


def solve_lp(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i <= min(x, 1-x, y, 1-y) and r_i + r_j <= dist(i,j)
    """
    # Wall constraints limit each radius individually
    lims = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    # Pairwise distance constraints
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    b_ub = dists[_TRUU_I, _TRUU_J]
    
    try:
        res = linprog(
            -np.ones(N), A_ub=_PAIR_A, b_ub=b_ub,
            bounds=bounds, method='highs'
        )
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0


def get_hex_config(rows_pattern, r_scale=0.095):
    """Generates a hexagonal lattice configuration based on row counts."""
    pts = []
    y = r_scale
    for idx, cnt in enumerate(rows_pattern):
        shift = r_scale if idx % 2 == 1 else 0.0
        x = r_scale + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r_scale
        y += np.sqrt(3) * r_scale
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

        curr_sum, _ = solve_lp(curr_centers)
