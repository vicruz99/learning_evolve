import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_lp_sum(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns (sum_radii, radii_array).
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    A_ub_rows = []
    b_ub_vals = []
    
    # Boundary constraints handled efficiently via variable bounds
    for i in range(n):
        x, y = centers[i]
        w = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(w, 1e-9)))
        
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(d)
            
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def obj_func(centers_flat):
    """Objective for scipy optimizer: minimize negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    s, _ = compute_lp_sum(centers)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    
    inits = []
    
    # 1. Hexagonal lattice patterns with various row distributions
    row_dists = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 6, 5, 4],
        [5, 5, 5, 6, 5], [6, 4, 6, 5, 5], [5, 6, 5, 5, 5]
    ]
    
    for rd in row_dists:
        pts = []
        y = 0.09
        row_idx = 0
        for count in rd:
            shift = 0.09 if row_idx % 2 == 1 else 0.0
            x = 0.09 + shift
            for _ in range(count):
                if len(pts) < N:
                    pts.append([x, y])
                x += 0.18
            y += 0.155
            row_idx += 1
        if len(pts) == N:
            inits.append(np.array(pts))
            
    # 2. Grid patterns
    for s in range(4, 8):
        step = 1.0 / (s + 1)
        pts = []
        for i in range(1, s+1):
            for j in range(1, s+1):
                if len(pts) < N:
                    pts.append([i*step, j*step])
        if len(pts) == N:
            inits.append(np.array(pts))
            
    # 3. Random dense configurations to escape structural biases
    for _ in range(15):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Multi-start Powell optimization
    for cfg in inits:
        x0 = cfg.flatten()
        try:
            res = minimize(obj_func, x0, method='Powell', 
                          options={'maxiter': 500, 'ftol': 1e-13, 'xtol': 1e-13})
            if np.isfinite(res.fun):
                centers = res.x.reshape(N, 2)
                s_lp, r_lp = compute_lp_sum(centers)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = centers.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass
            
    # Phase 2: Nelder-Mead refinement on the best configuration
    if best_centers is not None:
        try:
            res_nm = minimize(obj_func, best_centers.flatten(), method='Nelder-Mead',
                             options={'maxiter': 1500, 'fatol': 1e-13, 'xatol': 1e-13})
            if np.isfinite(res_nm.fun):
                centers = res_nm.x.reshape(N, 2)
                s_lp, r_lp = compute_lp_sum(centers)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = centers.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass
            
        # Phase 3: Adaptive Hill Climbing on centers
        current_centers = best_centers.copy()
        current_radii = best_radii.copy()
        current_sum = best_sum
        
        for step in range(800):
            idx = rng.integers(N)
            old_pos = current_centers[idx].copy()
            # Decay step size to focus on fine tuning
            step_size = 0.015 * (0.997 ** step)
            current_centers[idx] += rng.uniform(-step_size, step_size, 2)
            current_centers[idx] = np.clip(current_centers[idx], 1e-4, 1.0 - 1e-4)
            
            s_try, r_try = compute_lp_sum(current_centers)
            if s_try > current_sum:
                current_sum = s_try
                current_radii = r_try
            else:
                current_centers[idx] = old_pos
                
        best_centers = current_centers
        best_radii = current_radii
        best_sum = current_sum
        
    # Phase 4: Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    if best_centers is not None:
        scale = 1.0
        for i in range(N):
            x, y = best_centers[i]
            r = best_radii[i]
            if r < 1e-9: continue
            scale = min(scale, x/r, (1.0 - x)/r, y/r, (1.0 - y)/r)
            
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
                rs = best_radii[i] + best_radii[j]
                if rs < 1e-9: continue
                scale = min(scale, d / rs)
                
        best_radii *= scale * 0.99999999
        best_sum = np.sum(best_radii)
    else:
        # Fallback configuration
        best_centers = rng.uniform(0.15, 0.85, (N, 2))
        best_radii = np.full(N, 0.05)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
