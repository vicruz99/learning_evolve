import numpy as np
from scipy.optimize import minimize, linprog

N = 26
M = N * (N - 1) // 2

# Precompute constraint matrix structure for pairwise distances (r_i + r_j <= d_ij)
A_ub_pairs = np.zeros((M, N))
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub_pairs[idx, i] = 1.0
        A_ub_pairs[idx, j] = 1.0
        idx += 1

def solve_lp_for_centers(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Upper bounds from walls: r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(N):
        x, y = centers[i]
        w = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(w, 1e-9)))
        
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-14)
    np.fill_diagonal(dists, np.inf)
    b_ub = dists[np.triu_indices(N, k=1)]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_pairs, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(N)

def negative_lp_sum(x_flat):
    """Objective function for center optimization: minimize negative LP sum."""
    centers = x_flat.reshape(N, 2)
    # Keep centers strictly inside to avoid degenerate LP bounds
    centers = np.clip(centers, 0.02, 0.98)
    s, _ = solve_lp_for_centers(centers)
    return -s

def generate_hex_centers(counts, r_guess):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r_guess
    for row_idx, cnt in enumerate(counts):
        shift = r_guess if row_idx % 2 == 1 else 0.0
        x = r_guess + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r_guess
        y += np.sqrt(3.0) * r_guess
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    
    best_sum = 0.0
    best_centers = None
    
    # Diverse initial configurations
    inits = []
    
    # Hexagonal patterns with varying row distributions
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [5, 7, 5, 5, 4], [6, 6, 5, 5, 4]
    ]
    for p in patterns:
        c = generate_hex_centers(p, 0.095)
        inits.append(c)
        inits.append(c + rng.uniform(-0.02, 0.02, c.shape))
        
    # Grid patterns
    g = np.linspace(0.12, 0.88, 5)
    grid_pts = np.array([[x, y] for y in g for x in g])
    grid_pts = np.vstack([grid_pts, [0.5, 0.5]])
    inits.append(grid_pts)
    inits.append(grid_pts + rng.uniform(-0.015, 0.015, (N, 2)))
    
    # Random dense placements
    for _ in range(8):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # Phase 1: Local optimization from diverse starts using derivative-free methods
    for cfg in inits:
        cfg_clipped = np.clip(cfg, 0.05, 0.95)
        
        # Try Powell method
        try:
            res = minimize(negative_lp_sum, cfg_clipped.flatten(), method='Powell', 
                          options={'maxiter': 800, 'xtol': 1e-7, 'ftol': 1e-7})
            if np.isfinite(res.fun):
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_centers = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
        # Try Nelder-Mead method
        try:
            res_nm = minimize(negative_lp_sum, cfg_clipped.flatten(), method='Nelder-Mead',
                             options={'maxiter': 1000, 'xatol': 1e-7, 'fatol': 1e-7})
            if np.isfinite(res_nm.fun):
                s_nm = -res_nm.fun
                if s_nm > best_sum:
                    best_sum = s_nm
                    best_centers = res_nm.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    # Phase 2: Random perturbation & local search on the best configuration found
    if best_centers is not None:
        for _ in range(15):
            pert = best_centers.copy()
            pert += rng.uniform(-0.008, 0.008, pert.shape)
            pert = np.clip(pert, 0.05, 0.95)
            
            try:
                res_p = minimize(negative_lp_sum, pert.flatten(), method='Powell',
                                options={'maxiter': 600, 'xtol': 1e-7})
                if np.isfinite(res_p.fun):
                    s_p = -res_p.fun
                    if s_p > best_sum:
                        best_sum = s_p
                        best_centers = res_p.x.reshape(N, 2).copy()
            except Exception:
                pass
                
    # Get final radii via LP on the optimized centers
    if best_centers is None:
        best_centers = inits[0]
        
    _, best_radii = solve_lp_for_centers(best_centers)
    
    # Phase 3: Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if r < 1e-9: 
            continue
        scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
        
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs < 1e-9: 
                continue
            scale = min(scale, d / rs)
            
    best_radii *= max(scale * 0.9999995, 0.0)
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
