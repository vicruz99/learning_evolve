import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def get_lp_components(centers):
    """Prepares components for linear programming radius optimization."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(lim, 1e-9)))
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.hypot(dx, dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(d)
            
    return c_obj, np.array(A_ub_list), np.array(b_ub_list), bounds

def solve_radii_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    c_obj, A_ub, b_ub, bounds = get_lp_components(centers)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    # Fallback for infeasible configurations
    return np.full(N_CIRCLES, 1e-6), 0.0

def compute_min_clearance(centers):
    """Computes maximum feasible equal radius for given centers."""
    n = centers.shape[0]
    d_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    m_bound = np.min(d_bound)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    np.fill_diagonal(dists, np.inf)
    m_pair = np.min(dists) / 2.0
    
    return min(m_bound, m_pair)

def neg_clearance_obj(vars_flat):
    """Objective for Nelder-Mead: maximize minimum clearance."""
    centers = vars_flat.reshape(N_CIRCLES, 2)
    return -compute_min_clearance(centers)

def neg_lp_sum_obj(vars_flat):
    """Objective for Nelder-Mead: maximize sum of radii via LP."""
    centers = vars_flat.reshape(N_CIRCLES, 2)
    _, s = solve_radii_lp(centers)
    return -s

def hex_init(row_counts, r0, angle=0.0):
    """Generates a hexagonal lattice configuration with optional rotation."""
    pts = []
    y = r0
    row_idx = 0
    for cnt in row_counts:
        shift = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N_CIRCLES:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row_idx += 1
        
    pts = np.array(pts[:N_CIRCLES])
    if pts.shape[0] < N_CIRCLES:
        filler = np.random.uniform(0.2, 0.8, (N_CIRCLES - pts.shape[0], 2))
        pts = np.vstack([pts, filler])
        
    # Center at (0.5, 0.5), rotate, then shift back to [0,1]^2
    center = pts.mean(axis=0)
    pts_centered = pts - center
    pts_centered[0] += 0.5 # Shift one point to break symmetry
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        rot_mat = np.array([[c, -s], [s, c]])
        pts_centered = pts_centered @ rot_mat.T
        
    # Scale and translate to fit tightly in [0,1]^2
    mn = pts_centered.min(axis=0)
    mx = pts_centered.max(axis=0)
    span = mx - mn
    if span[0] > 0 and span[1] > 0:
        scale = 0.92 / max(span[0], span[1])
        pts_centered *= scale
        pts_centered += (1.0 - 0.92) / 2.0 + (mn - mx) * scale + 0.5 * (1.0 - 0.92)
        
    return np.clip(pts_centered, 0.01, 0.99)

def compute_constraints(vars_array, triu_i, triu_j):
    """Computes pairwise non-overlap constraints for SLSQP with boundary parameterization."""
    r = vars_array[:N_CIRCLES]
    u = vars_array[N_CIRCLES:2*N_CIRCLES]
    v = vars_array[2*N_CIRCLES:3*N_CIRCLES]
    
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return d2[triu_i, triu_j] - rs[triu_i, triu_j]**2

def objective_vars(vars_array):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars_array[:N_CIRCLES])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    triu_i, triu_j = np.triu_indices(N_CIRCLES, k=1)

    best_radii = None

    ]
    
    inits = []
    for rc in row_configs:
        inits.append(hex_init(rc, 0.10, angle=0.0))
        inits.append(hex_init(rc, 0.10, angle=0.05))
        inits.append(hex_init(rc, 0.10, angle=0.15))

    for _ in range(6):
        pts = rng.uniform(0.15, 0.85, (N_CIRCLES, 2))
        inits.append(pts)
        
    # Phase 2: Optimize for equal-radius clearance (Nelder-Mead)
    for cfg in inits:
        try:
            res_nm = minimize(neg_clearance_obj, cfg.flatten(), method='Nelder-Mead',
                              options={'maxiter': 8000, 'xatol': 1e-10, 'fatol': 1e-12})
            centers_opt = res_nm.x.reshape(N_CIRCLES, 2)
            c_val = compute_min_clearance(centers_opt)
            
            # If we found a tight packing, extract unequal radii via LP
            r_lp, s_lp = solve_radii_lp(centers_opt)
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = centers_opt.copy()
                best_radii = r_lp.copy()
        except Exception:
            continue
            
    # Phase 3: Refine centers by maximizing LP sum directly
    if best_centers is not None:
        for _ in range(3):
            # Slight perturbation
            c_pert = best_centers + rng.uniform(-0.005, 0.005, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            try:
                res_lp_opt = minimize(neg_lp_sum_obj, c_pert.flatten(), method='Nelder-Mead',
                                      options={'maxiter': 12000, 'xatol': 1e-10, 'fatol': 1e-12})
                centers_lp = res_lp_opt.x.reshape(N_CIRCLES, 2)
                r_final, s_final = solve_radii_lp(centers_lp)
                if s_final > best_sum:
                    best_sum = s_final
                    best_centers = centers_lp.copy()
                    best_radii = r_final.copy()
            except Exception:
                continue
                
    # Phase 4: Joint SLSQP polish with boundary-enforcing parameterization
    if best_centers is not None:
        r0 = best_radii
        denom = 1.0 - 2.0 * r0
        u0 = np.clip((best_centers[:, 0] - r0) / denom, 0.0, 1.0)
        v0 = np.clip((best_centers[:, 1] - r0) / denom, 0.0, 1.0)
        x0_slqp = np.concatenate([r0, u0, v0])
        
        bounds_vars = [(1e-6, 0.5)] * N_CIRCLES + [(0.0, 1.0)] * N_CIRCLES + [(0.0, 1.0)] * N_CIRCLES

        try:

                    # Final LP extraction to ensure radii are optimally distributed for these centers

    scale = 1.0
    for i in range(N_CIRCLES):
        x, y = best_centers[i]
        r = best_radii[i]
        if r < 1e-12:
            continue
        scale = min(scale, x/r, (1.0 - x)/r, y/r, (1.0 - y)/r)

            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs < 1e-12:
                continue
            scale = min(scale, d / rs)
            
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
