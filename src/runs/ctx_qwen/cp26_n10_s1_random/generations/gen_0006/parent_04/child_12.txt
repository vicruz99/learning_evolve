2. **LP refinement**: Allowing radii to vary slightly via Linear Programming (LP) on fixed centers typically extracts marginal gains by filling microscopic gaps near boundaries or between tightly packed rows.

- **Bisection on equal radius**: We perform a binary search on $r$. - **LP Maximization**: Once the best equal-radius configuration is found, we fix the centers and solve an LP to maximize $\sum r_i$ subject to $r_i + r_j \le \|c_i - c_j\|$ and boundary constraints. This automatically handles unequal radii optimally for those centers.
- **Stochastic Coordinate Ascent**: We iteratively perturb individual centers and re-solve the LP, accepting moves that increase the sum of radii. This escapes local optima in the center space.
- **Safety Scaling**: Finally, we compute a global scaling factor to strictly satisfy the $10^{-12}$ tolerance in the validator, ensuring no numerical false negatives.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def penalty(centers_flat, r):
    """Smooth penalty function for fixed radius r."""
    centers = centers_flat.reshape(N, 2)
    x, y = centers[:, 0], centers[:, 1]
    
    p = np.sum(np.maximum(0.0, r - x)**2)
    p += np.sum(np.maximum(0.0, x - (1.0 - r))**2)
    p += np.sum(np.maximum(0.0, r - y)**2)
    p += np.sum(np.maximum(0.0, y - (1.0 - r))**2)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d = np.sqrt(dx**2 + dy**2)
    d[np.arange(N), np.arange(N)] = 2.0 * r
    
    v = np.maximum(0.0, 2.0 * r - d)
    m = np.triu(np.ones((N, N), dtype=bool), k=1)
    p += np.sum(v[m]**2)
    return p

def optimize_centers_for_r(r, init_centers):
    """Optimize centers to minimize overlap/boundary penalty for fixed r."""
    bounds = [(0.0, 1.0)] * (2 * N)
    res = minimize(penalty, init_centers.flatten(), args=(r,), 
                   method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 1500, 'ftol': 1e-12})
    return res.x.reshape(N, 2), res.fun

def solve_lp(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    A_ub = np.zeros((N * (N - 1) // 2, N))
    b_ub = np.zeros(N * (N - 1) // 2)
    k = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            k += 1
            
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0

def get_hex(rows, r):
    """Generate hexagonal lattice initialization."""
    pts = []
    y = r
    for ri, cnt in enumerate(rows):
        shift = r if ri % 2 == 1 else 0.0
        x = r + shift
        for _ in range(cnt):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3) * r
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_centers = None
    best_r_eq = 0.09
    
    # Diverse row patterns known to approach optimal density for n=26
    pats = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [5, 6, 6, 5, 4]]
    
    # Phase 1: Bisection to find max feasible equal radius
    lo, hi = 0.095, 0.105
    for _ in range(32):
        if hi - lo < 1e-7: break
        mid = (lo + hi) / 2.0
        found = False
        
        for pat in pats:
            c_init = get_hex(pat, mid)
            c_pert = c_init + rng.uniform(-0.002, 0.002, c_init.shape)
            c_pert = np.clip(c_pert, 0.05, 0.95)
            
            c_opt, pen = optimize_centers_for_r(mid, c_pert)
            if pen < 1e-4:
                found = True
                best_centers = c_opt
                best_r_eq = mid
                break
        if found:
            lo = mid
        else:
            hi = mid
            
    if best_centers is None:
        best_centers = get_hex([5, 6, 5, 6, 4], 0.10)
        
    # Phase 2: Exact LP refinement on optimized centers
    radii_lp, sum_lp = solve_lp(best_centers)
    best_centers = best_centers.copy()
    current_radii = radii_lp.copy()
    current_sum = sum_lp
    
    # Phase 3: Stochastic coordinate ascent to refine centers further
    for step in range(150):
        i = rng.integers(N)
        old = best_centers[i].copy()
        s = 0.012 * (0.985 ** step)
        best_centers[i] += rng.uniform(-s, s, 2)
        best_centers[i] = np.clip(best_centers[i], 1e-4, 1.0 - 1e-4)
        
        r_try, s_try = solve_lp(best_centers)
        if s_try > current_sum:
            current_sum = s_try
            current_radii = r_try
        else:
            best_centers[i] = old
            
    # Phase 4: Strict safety scaling for numerical validity
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = current_radii[i]
        if r < 1e-9: continue
        scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
        
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
            rs = current_radii[i] + current_radii[j]
            if rs < 1e-9: continue
            scale = min(scale, d / rs)
            
    current_radii *= scale * 0.999999
    final_sum = np.sum(current_radii)
    
    return best_centers, current_radii, float(final_sum)
