import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
triu_i, triu_j = np.triu_indices(N, k=1)
n_pairs = N * (N - 1) // 2

# Precompute LP constraint matrix structure (constant across calls)
A_LP = np.zeros((n_pairs, N))
_k = 0
for _i in range(N):
    for _j in range(_i + 1, N):
        A_LP[_k, _i] = 1.0
        A_LP[_k, _j] = 1.0
        _k += 1

def solve_lp(centers, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(N)
    bounds = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(N):
        x, y = centers[i]
        w = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(w, 1e-9)))
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub.append(d)
            
    b_ub = np.array(b_ub)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-9), 0.0

def min_clearance(centers):
    """Computes maximum feasible equal radius for given centers."""
    d_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    np.fill_diagonal(dists, np.inf)
    return min(np.min(d_bound), np.min(dists) / 2.0)

def neg_clearance(x_flat):
    """Objective for Nelder-Mead: maximize minimum clearance."""
    return -min_clearance(x_flat.reshape(N, 2))

def hex_init(row_counts, r0=0.09):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0
    for ri, cnt in enumerate(row_counts):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def compute_constraints_slqp(vars_arr, triu_i, triu_j):
    """Inequality constraints >= 0 for SLSQP joint optimization."""
    r = vars_arr[:N]
    u = vars_arr[N:2*N]
    v = vars_arr[2*N:3*N]
    
    # Reparameterization automatically satisfies boundary constraints:
    # x = r + (1 - 2r)u  =>  r <= x <= 1-r for u in [0,1]
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[triu_i, triu_j] - rs[triu_i, triu_j]**2

def obj_slqp(vars_arr):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars_arr[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Equal radius optimization via Nelder-Mead
    configs = []
    row_dists = [[5,6,5,6,4], [6,5,6,5,4], [5,5,6,6,4], [4,6,6,6,4], [5,5,5,5,6]]
    for rd in row_dists:
        c = hex_init(rd)
        configs.append(c)
        # Perturbed variants to escape symmetry traps
        for _ in range(3):
            cp = c.copy()
            cp += rng.uniform(-0.025, 0.025, c.shape)
            configs.append(np.clip(cp, 0.05, 0.95))
            
    # Random dense configurations
    for _ in range(5):
        configs.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    for cfg in configs:
        res = minimize(neg_clearance, cfg.flatten(), method='Nelder-Mead',
                       options={'maxiter': 15000, 'xatol': 1e-9, 'fatol': 1e-12})
        c_opt = res.x.reshape(N, 2)
        r_lp, s_lp = solve_lp(c_opt, A_LP)
        if s_lp > best_sum:
            best_sum = s_lp
            best_centers = c_opt.copy()
            best_radii = r_lp.copy()
            
    # Phase 2: Joint optimization using SLSQP with boundary-safe reparameterization
    r0 = best_radii.copy() * 0.999  # Slight shrink to ensure initial feasibility
    denom = 1.0 - 2.0 * r0
    u0 = np.clip((best_centers[:, 0] - r0) / denom, 0.0, 1.0)
    v0 = np.clip((best_centers[:, 1] - r0) / denom, 0.0, 1.0)
    x0 = np.concatenate([r0, u0, v0])
    
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': compute_constraints_slqp, 'args': (triu_i, triu_j)}
    
    try:
        res_slqp = minimize(obj_slqp, x0, method='SLSQP', bounds=bounds_slqp,
                            constraints=cons_slqp, options={'maxiter': 8000, 'ftol': 1e-14})
        if np.isfinite(res_slqp.fun):
            r_s = res_slqp.x[:N]
            u_s = res_slqp.x[N:2*N]
            v_s = res_slqp.x[2*N:3*N]
            c_s = np.column_stack([r_s + (1.0 - 2.0 * r_s) * u_s, 
                                   r_s + (1.0 - 2.0 * r_s) * v_s])
            r_lp_s, s_lp_s = solve_lp(c_s, A_LP)
            if s_lp_s > best_sum:
                best_sum = s_lp_s
                best_centers = c_s
                best_radii = r_lp_s
    except Exception:
        pass
        
    # Phase 3: Stochastic hill climbing on centers evaluated via LP
    step = 0.025
    for _ in range(600):
        i = rng.integers(N)
        old = best_centers[i].copy()
        best_centers[i] += rng.uniform(-step, step, 2)
        best_centers[i] = np.clip(best_centers[i], 1e-4, 1.0 - 1e-4)
        
        r_try, s_try = solve_lp(best_centers, A_LP)
        if s_try > best_sum:
            best_sum = s_try
            best_radii = r_try
        else:
            best_centers[i] = old
        step *= 0.995
        
    # Phase 4: Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if r < 1e-9:
            continue
        scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
        
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs < 1e-9:
                continue
            scale = min(scale, d/rs)
            
    best_radii *= scale * 0.9999995
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(final_sum)
