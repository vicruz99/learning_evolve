2. **Derivative-Free Optimization**: Since the LP objective is piecewise linear and non-differentiable with respect to centers, we use `Powell` and `Nelder-Mead` (multi-start) to maximize this objective. These methods handle non-smooth landscapes well.
3. **Stochastic Local Search**: After deterministic optimization, we run a hill-climbing phase. Randomly perturbing individual circle centers and accepting improvements allows us to escape shallow local minima and fine-tune the configuration.

import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii_sum(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns (sum_radii, radii_array)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    
    A_ub = np.zeros((m_bound + m_pair, n))
    b_ub = np.zeros(m_bound + m_pair)
    bounds = []
    
    idx = 0
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        lims = [x, 1.0 - x, y, 1.0 - y]
        mx = max(0.0, min(lims))
        bounds.append((0.0, mx))
        for lim in lims:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def objective_centers(vars_arr):
    """Objective for center optimization: maximize LP sum of radii."""
    n = 26
    centers = vars_arr.reshape(n, 2)
    centers = np.clip(centers, 0.005, 0.995)
    s, _ = solve_lp_radii_sum(centers)
    return -s

def generate_hex_init(n, row_counts, r0=0.1):
    """Generates initial positions on a hexagonal lattice, scaled to unit square."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        
    while len(pts) < n:
        pts.append([0.5, 0.5])
        
    pts = np.array(pts[:n])
    mn = pts.min(axis=0)
    mx = pts.max(axis=0)
    span = mx - mn
    if span.max() > 1e-6:
        scale = 0.8 / span.max()
        pts = (pts - mn) * scale + 0.5 * (1.0 - scale)
    return np.clip(pts, 0.05, 0.95)

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    row_counts_options = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [5, 7, 5, 5, 4],
        [6, 4, 6, 5, 5], [5, 5, 5, 5, 6], [7, 6, 6, 7],
        [6, 7, 6, 7], [5, 6, 5, 6, 5, 1]
    ]
    
    inits = []
    for rc in row_counts_options:
        if sum(rc) >= n:
            base = generate_hex_init(n, rc)
            inits.append(base)
            for _ in range(2):
                pert = base + rng.uniform(-0.025, 0.025, (n, 2))
                inits.append(np.clip(pert, 0.05, 0.95))
                
    for _ in range(5):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = 0.0
    best_centers = None
    
    # Phase 1: Deterministic multi-start optimization
    for cfg in inits:
        try:
            res = minimize(objective_centers, cfg.flatten(), method='Powell',
                           options={'maxiter': 3000, 'xtol': 1e-7, 'ftol': 1e-12})
            if not np.isfinite(res.fun):
                continue
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape(n, 2).copy()
        except Exception:
            continue
            
    if best_centers is None:
        best_centers = inits[0]
        
    # Phase 2: Stochastic local search (Hill Climbing) to polish
    current_centers = best_centers.copy()
    current_sum = best_sum
    step_size = 0.015
    
    def get_sum(c):
        s, _ = solve_lp_radii_sum(c)
        return s
        
    for _ in range(3000):
        idx = rng.integers(n)
        old_pos = current_centers[idx].copy()
        delta = rng.uniform(-step_size, step_size, 2)
        new_pos = np.clip(old_pos + delta, 0.005, 0.995)
        current_centers[idx] = new_pos
        
        new_sum = get_sum(current_centers)
        if new_sum > current_sum + 1e-9:
            current_sum = new_sum
            best_sum = new_sum
            best_centers = current_centers.copy()
        else:
            current_centers[idx] = old_pos
            
        if _ % 300 == 299 and _ > 0:
            step_size *= 0.95
            
    # Phase 3: Final exact LP extraction
    final_sum, radii = solve_lp_radii_sum(best_centers)
    
    # Safety scaling to guarantee strict numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = radii[i] + radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    radii *= scale * 0.9999995
    final_sum = np.sum(radii)
    
    return best_centers, radii, float(final_sum)
