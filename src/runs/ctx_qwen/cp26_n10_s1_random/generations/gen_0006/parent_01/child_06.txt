import numpy as np
from scipy.optimize import linprog
import math

def get_lp_sum(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns (radii_array, sum_radii) or (None, 0.0) if infeasible.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(dist to walls)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(lim, 1e-9)))
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse hexagonal lattice initializations
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], 
        [5,5,6,5,5], [4,6,6,6,4], [5,6,6,4,5], [5,5,5,5,6]
    ]
    
    for pat in patterns:
        pts = []
        y = 0.095
        for ri, cnt in enumerate(pat):
            shift = 0.095 if ri % 2 == 1 else 0.0
            x = 0.095 + shift
            for _ in range(cnt):
                if len(pts) >= n:
                    break
                pts.append([x, y])
                x += 0.19
            y += 0.095 * math.sqrt(3)
        while len(pts) < n:
            pts.append([0.5, 0.5])
        inits.append(np.array(pts[:n]))
        
    # Add purely random starts to cover unexpected optimal topologies
    for _ in range(8):
        inits.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    # Main optimization loop: LP evaluation + annealed hill climbing on centers
    for cfg in inits:
        c_curr = cfg.copy()
        
        # Initial LP evaluation
        r_lp, s_lp = get_lp_sum(c_curr)
        if r_lp is not None and s_lp > best_sum:
            best_sum = s_lp
            best_centers = c_curr.copy()
            best_radii = r_lp.copy()
            
        # Hill climbing phase
        step = 0.04
        for _ in range(600):
            idx = rng.integers(n)
            old_pos = c_curr[idx].copy()
            
            # Perturb one circle
            c_curr[idx] += rng.uniform(-step, step, 2)
            c_curr[idx] = np.clip(c_curr[idx], 0.02, 0.98)
            
            # Evaluate new LP sum
            r_new, s_new = get_lp_sum(c_curr)
            
            if r_new is not None and s_new > best_sum + 1e-12:
                best_sum = s_new
                best_centers = c_curr.copy()
                best_radii = r_new.copy()
            else:
                # Revert move if it didn't improve
                c_curr[idx] = old_pos
                
            # Anneal step size
            step *= 0.996
            
    # Fallback configuration (should rarely be triggered)
    if best_centers is None:
        best_centers = inits[0]
        best_radii = np.full(n, 0.08)
        best_sum = float(np.sum(best_radii))

    if best_centers is not None:
        scale = 1.0

            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if r > 1e-12:

                rs = best_radii[i] + best_radii[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)
                    
        best_radii *= scale * 0.9999995
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
