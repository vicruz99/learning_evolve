import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRI_IDX = np.triu_indices(N_CIRCLES, k=1)

def compute_penalty(params, n):
    """Computes the objective (negative sum of radii) plus smooth penalty terms."""
    cx = params[:n]
    cy = params[n:2*n]
    r = params[2*n:]
    
    # Primary objective
    obj = -np.sum(r)
    
    # Penalty weight
    mu = 8000.0
    pen = 0.0
    
    # Boundary penalties: circles must be within [0,1]x[0,1]
    pen += mu * np.sum(np.maximum(0.0, r - cx)**2)
    pen += mu * np.sum(np.maximum(0.0, r + cx - 1.0)**2)
    pen += mu * np.sum(np.maximum(0.0, r - cy)**2)
    pen += mu * np.sum(np.maximum(0.0, r + cy - 1.0)**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    d = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(d, 1e9)  # Avoid zero diagonal issues
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    viol = np.maximum(0.0, r_sum[TRI_IDX] - d[TRI_IDX])
    pen += mu * np.sum(viol**2)
    
    return obj + pen

def solve_lp_radii(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    bounds = []
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints for radii: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        mx = max(0.0, min(x, 1.0 - x, y, 1.0 - y))
        bounds.append((0.0, mx))
        for lim in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(lim)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i, j in zip(*TRI_IDX):
        d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        row = np.zeros(n)
        row[i] = 1.0
        row[j] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(d)
        
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub_list), b_ub=np.array(b_ub_list), 
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def generate_hex_init(n, row_counts):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    r0 = 0.08
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        
    while len(pts) < n:
        pts.append([0.5, 0.5])
        
    pts = np.array(pts[:n])
    # Center the configuration in the unit square
    cx_mean, cy_mean = pts.mean(axis=0)
    pts -= np.array([cx_mean - 0.5, cy_mean - 0.5])
    return np.clip(pts, 0.05, 0.95)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Diverse row distributions for hexagonal lattices
    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [5, 6, 6, 4, 5],
        [7, 6, 6, 7], [6, 7, 6, 7], [5, 6, 5, 6, 5, 1]
    ]
    
    inits = []
    for rc in row_patterns:
        if sum(rc) < n:
            continue
        base = generate_hex_init(n, rc)
        inits.append(base)
        # Add perturbations to break symmetry and escape local minima
        for _ in range(4):
            pert = base + rng.uniform(-0.025, 0.025, (n, 2))
            inits.append(np.clip(pert, 0.05, 0.95))
            
    # Random dense starts
    for _ in range(6):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    bounds_opt = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.25)] * n
    
    # Phase 1: Multi-start penalty optimization
    for cfg in inits:
        x0 = np.zeros(3 * n)
        x0[:n] = cfg[:, 0]
        x0[n:2*n] = cfg[:, 1]
        x0[2*n:] = 0.05  # Start with small radii to avoid massive initial penalties
        
        res = minimize(compute_penalty, x0, args=(n,), method='L-BFGS-B', 
                       bounds=bounds_opt, options={'maxiter': 6000, 'ftol': 1e-15})
                       
        if not np.isfinite(res.fun):
            continue
            
        c_opt = res.x[:2*n].reshape(n, 2)
        r_lp = solve_lp_radii(c_opt, n)
        if r_lp is not None:
            s = np.sum(r_lp)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = r_lp.copy()

    if best_centers is not None:
        for _ in range(12):
            pert = best_centers + rng.normal(0, 0.004, best_centers.shape)

            x0 = np.zeros(3 * n)

            r_lp = solve_lp_radii(c_opt, n)
            if r_lp is not None:
                s = np.sum(r_lp)
                if s > best_sum:
                    best_sum = s
                    best_centers = c_opt.copy()
                    best_radii = r_lp.copy()

    # Fallback configuration if optimization unexpectedly fails
    if best_centers is None:
        best_centers = generate_hex_init(n, [6, 5, 6, 5, 4])
        best_radii = np.full(n, 0.085)
        best_sum = np.sum(best_radii)

    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]

            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)

        d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                     best_centers[i, 1] - best_centers[j, 1])
        rs = best_radii[i] + best_radii[j]
        if rs > 1e-12:
            scale = min(scale, d / rs)
            
    best_radii *= scale * 0.9999999
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
