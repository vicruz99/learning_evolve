import numpy as np
from scipy.optimize import minimize, linprog

def compute_min_clearance(centers):
    n = centers.shape[0]
    d_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    min_bound = np.min(d_bound)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pairwise = np.min(dists) / 2.0
    return min(min_bound, min_pairwise)

def obj_equal_radius(params, n):
    r = params[2*n]
    if r < 0:
        return 1e6
    return -r

def cons_equal_radius(params, n):
    centers = np.column_stack((params[:n], params[n:2*n]))
    r = params[2*n]
    c = []
    c.extend(centers[:, 0] - r)
    c.extend(1.0 - centers[:, 0] - r)
    c.extend(centers[:, 1] - r)
    c.extend(1.0 - centers[:, 1] - r)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_sq = np.sum(diff**2, axis=2)
    idx = np.triu_indices(n, k=1)
    c.extend(dists_sq[idx] - 4.0 * r**2)
    return np.asarray(c)

def solve_lp_radii(centers):
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    A_ub_list = []
    b_ub_list = []
    
    for i in range(n):
        x, y = centers[i]
        mx = max(0.0, min(x, 1.0 - x, y, 1.0 - y))
        bounds.append((0.0, mx))
        for lim in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(lim)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(d)
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub_list), b_ub=np.array(b_ub_list), 
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def generate_hex_centers(n, row_counts, r0=0.1):
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    pts = np.array(pts[:n])
    pts -= pts.min(axis=0)
    pts /= (pts.max(axis=0) - pts.min(axis=0)) * 0.8 + 1e-9
    pts *= 0.8
    pts += 0.1
    return np.clip(pts, 0.05, 0.95)

def obj_joint(params, n):
    return -np.sum(params[2*n:])

def cons_joint(params, n):
    cx = params[:n]
    cy = params[n:2*n]
    r = params[2*n:]
    c = []
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    cx_m = cx[:, np.newaxis] - cx[np.newaxis, :]
    cy_m = cy[:, np.newaxis] - cy[np.newaxis, :]
    r_m = r[:, np.newaxis] + r[np.newaxis, :]
    dists_sq = cx_m**2 + cy_m**2
    idx = np.triu_indices(n, k=1)
    c.extend(dists_sq[idx] - r_m[idx]**2)
    return np.asarray(c)

    best_sum = float(np.sum(best_radii))
