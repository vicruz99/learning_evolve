import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def hex_init(row_counts):
    """Generates a hexagonal lattice initialization based on row counts."""
    pts = []
    y = 0.0
    r0 = 0.1
    for i, cnt in enumerate(row_counts):
        if len(pts) >= N_CIRCLES:
            break
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N_CIRCLES:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        
    while len(pts) < N_CIRCLES:
        pts.append([0.5, 0.5])
        
    pts = np.array(pts[:N_CIRCLES])
    # Center and scale to safely fit within [0.1, 0.9] initially
    mn = pts.min(axis=0)
    mx = pts.max(axis=0)
    span = mx - mn
    if span[0] > 1e-6 and span[1] > 1e-6:
        pts = (pts - mn) / span * 0.8 + 0.1
    return pts

def compute_clearance(centers):
    """Computes the minimum clearance (wall distance and half pairwise distance)."""
    n = centers.shape[0]
    d_wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    min_wall = np.min(d_wall)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pair = np.min(dists) / 2.0
    
    return min(min_wall, min_pair)

def neg_clearance_obj(x):
    """Objective for clearance maximization: minimize negative clearance."""
    centers = x.reshape(N_CIRCLES, 2)
    return -compute_clearance(centers)

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = max(0.0, min(x, 1.0 - x, y, 1.0 - y))
        bounds.append((0.0, lim))
        for val in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(val)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(d)
            
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def joint_obj(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2 * N_CIRCLES:])

def joint_cons(v):
    """Inequality constraints >= 0 for joint center/radius optimization."""
    n = N_CIRCLES
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    c = []
    
    # Boundary constraints
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    # Pairwise non-overlap constraints
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    dists = np.sqrt(cx_m**2 + cy_m**2)
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    c.extend(dists[idx] - r_m[idx])
    return np.array(c)
