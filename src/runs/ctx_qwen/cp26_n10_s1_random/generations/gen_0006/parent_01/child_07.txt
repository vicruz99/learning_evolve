from scipy.optimize import linprog, minimize

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    m_total = m_bound + m_pair
    
    A_ub = np.zeros((m_total, n))
    b_ub = np.zeros(m_total)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        if mx <= 1e-9:
            return None
        bounds.append((0.0, mx))
        for lim in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def obj_slsqp(v, n):
    """Objective function for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_slsqp(v, n):
    """Inequality constraints >= 0 for SLSQP."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.empty(4 * n + n * (n - 1) // 2)
    
    # Boundary constraints
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    
    # Pairwise non-overlap constraints
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c[4*n:] = np.sqrt(dx**2 + dy**2) - dr
    return c
