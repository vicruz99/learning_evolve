import numpy as np
from scipy.optimize import minimize, linprog

def get_hex_init(n, row_counts, r0):
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def solve_lp_radii(centers):
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 0.0)) for l in lims]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    m_pairs = n * (n - 1) // 2
    A_ub = np.zeros((m_pairs + n, n))
    b_ub = np.zeros(m_pairs + n)
    
    k = 0
    for i, j in zip(idx[0], idx[1]):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]
        k += 1
    for i in range(n):
        A_ub[m_pairs + i, i] = 1.0
        b_ub[m_pairs + i] = lims[i]
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def obj_equal(vars_arr):
    return -vars_arr[-1]

def cons_equal(vars_arr, n):
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    R = vars_arr[2*n]
    m_pairs = n * (n - 1) // 2
    c = np.empty(4*n + m_pairs)
    c[:n] = cx - R
    c[n:2*n] = 1.0 - cx - R
    c[2*n:3*n] = cy - R
    c[3*n:4*n] = 1.0 - cy - R
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    d2 = cx_m**2 + cy_m**2
    idx = np.triu_indices(n, k=1)
    c[4*n:] = d2[idx] - 4.0*R**2
    return c

def obj_joint(vars_arr, n):
    return -np.sum(vars_arr[2*n:])

def cons_joint(vars_arr, n):
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    m_pairs = n * (n - 1) // 2
    c = np.empty(4*n + m_pairs)
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_sum = r[:, None] + r[None, :]
    d2 = cx_m**2 + cy_m**2
    idx = np.triu_indices(n, k=1)
    c[4*n:] = d2[idx] - r_sum[idx]**2
    return c

def run_packing():
    n = 26
    np.random.seed(42)
    
    row_configs = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [5, 6, 6, 4, 5],
        [6, 5, 5, 6, 4], [5, 6, 5, 5, 5, 0]
    ]
    
    inits = []
    for rc in row_configs:
        if sum(rc) < n:
            continue
        base = get_hex_init(n, rc, 0.09)
        inits.append(base)
        for _ in range(4):
            p = base + np.random.uniform(-0.02, 0.02, base.shape)
            inits.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(10):
        inits.append(np.random.uniform(0.1, 0.9, (n, 2)))
        
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    bounds_eq = [(0.0, 1.0)] * (2*n) + [(0.01, 0.12)]
    
    for cfg in inits:
        x0 = np.zeros(2*n + 1)
        x0[:n] = cfg[:, 0]
        x0[n:2*n] = cfg[:, 1]
        x0[2*n] = 0.09
        
        try:

                           constraints={'type': 'ineq', 'fun': cons_equal, 'args': (n,)},
                           options={'maxiter': 5000, 'ftol': 1e-12})
            if res.success and res.x[-1] > 0.101:
                c_eq = res.x[:2*n].reshape(n, 2)
                d_bound = np.min(np.minimum(c_eq[:, 0], 1.0 - c_eq[:, 0]), 
                                 np.minimum(c_eq[:, 1], 1.0 - c_eq[:, 1]))

                if R_valid > best_sum/n:

                    if r_opt is not None:

                            best_radii = r_opt
                            best_sum = s_opt
            except Exception:

    if best_centers is None:
        best_centers = inits[0]
        best_radii = np.full(n, 0.09)
        best_sum = 2.34

    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
