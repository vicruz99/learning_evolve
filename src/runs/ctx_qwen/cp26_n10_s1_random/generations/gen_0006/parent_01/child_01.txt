1. **Structured Initialization**: We generate diverse initial configurations based on hexagonal lattice patterns with row counts known to yield high densities for N=26 (e.g., `[6,5,6,5,4]`, `[5,6,5,6,4]`). These are scaled to fit tightly within the square and perturbed to break symmetries.
2. **Phase 1 - Center Optimization (Nelder-Mead)**: Instead of optimizing radii and centers simultaneously initially, we fix the objective to maximizing the *minimum clearance* (distance to boundaries and between circles). Nelder-Mead is highly effective at finding well-separated center configurations even from rough starts.
3. **Phase 2 - Radius Extraction (Linear Programming)**: For each optimized center configuration, we solve a fast Linear Program to maximally expand the radii. This decouples the non-convex distance constraints into a convex problem for fixed centers, yielding precise optimal radii.
4. **Phase 3 - Joint Refinement (SLSQP)**: Using the best configuration from Phase 1/2 as a seed, we run SLSQP to jointly tweak centers and radii, allowing minor boundary/overlap relaxations to squeeze out additional sum of radii. LP refinement is applied again to guarantee optimality for the new centers.
5. **Safety Scaling**: A final multiplicative scaling factor ensures strict compliance with the $10^{-12}$ tolerance in the validation checker.

This pipeline leverages the strengths of derivative-free global search, convex optimization, and constrained gradient-based refinement, reliably surpassing previous baselines.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def get_hex_init(n, row_counts, r_guess=0.10):
    pts = []
    y = r_guess
    for i, cnt in enumerate(row_counts):
        shift = r_guess if i % 2 == 1 else 0.0
        x = r_guess + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r_guess
        y += np.sqrt(3) * r_guess
    while len(pts) < n:
        pts.append([0.5, 0.5])
    pts = np.array(pts[:n])
    cx, cy = pts.mean(axis=0)
    pts -= np.array([cx, cy])
    scale = 0.45 / max(np.abs(pts).max(), 1e-9)
    pts *= scale
    pts += 0.5
    return pts

def compute_min_dist(centers, n):
    d_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    min_b = np.min(d_bound)
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_p = np.min(dists) / 2.0
    return min(min_b, min_p)

def obj_min_dist(x, n):
    centers = x.reshape(n, 2)
    return -compute_min_dist(centers, n)

def solve_lp_radii(centers):
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    A_ub = np.zeros((m_bound + m_pair, n))
    b_ub = np.zeros(m_bound + m_pair)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lim = max(1e-9, min(x, 1.0 - x, y, 1.0 - y))
        bounds.append((0.0, lim))
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except:
        pass
    return None, 0.0

def objective_sum_radii(v, n):
    return -np.sum(v[2*n:])

def constraints_slqp(v, n):
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = []
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, 1e9)
    r_sum = r[:, None] + r[None, :]
    idx = np.triu_indices(n, k=1)
    c.extend(dists[idx] - r_sum[idx])
    return np.array(c)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    rng = np.random.default_rng(42)
    
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [6, 5, 5, 6, 4],
        [5, 6, 6, 4, 5], [7, 5, 5, 5, 4], [5, 5, 5, 6, 5],
        [6, 6, 6, 4, 4], [5, 7, 5, 5, 4], [6, 4, 6, 5, 5]
    ]
    
    configs = []
    for p in patterns:
        if sum(p) < n: continue
        c = get_hex_init(n, p)
        configs.append(c)
        for _ in range(4):
            cp = c + rng.uniform(-0.03, 0.03, (n, 2))
            configs.append(np.clip(cp, 0.05, 0.95))
            
    for _ in range(10):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Optimize centers for max equal radius using Nelder-Mead
    for cfg in configs:
        try:
            res = minimize(obj_min_dist, cfg.flatten(), args=(n,), method='Nelder-Mead',
                          options={'maxiter': 10000, 'xatol': 1e-10, 'fatol': 1e-12})
            c_opt = res.x.reshape(n, 2)
            r_lp, s_lp = solve_lp_radii(c_opt)
            if r_lp is not None and s_lp > best_sum:
                best_sum = s_lp
                best_centers = c_opt.copy()
                best_radii = r_lp.copy()
        except:
            continue

    # Phase 2: Refine best configuration using SLSQP on (x, y, r)
    if best_centers is not None:
        for trial in range(15):
            pert = best_centers + rng.normal(0, 0.005, best_centers.shape)
            pert = np.clip(pert, 0.02, 0.98)
            
            x0 = np.zeros(3 * n)
            x0[:n] = pert[:, 0]
            x0[n:2*n] = pert[:, 1]
            x0[2*n:] = best_radii * 0.95
            
            bounds_opt = [(0.001, 0.999)] * (2*n) + [(1e-5, 0.25)] * n
            cons_dict = {'type': 'ineq', 'fun': constraints_slqp, 'args': (n,)}
                
            try:
                res = minimize(objective_sum_radii, x0, args=(n,), method='SLSQP', bounds=bounds_opt,
                              constraints=cons_dict, options={'maxiter': 8000, 'ftol': 1e-13})
                if np.isfinite(res.fun):
                    c_s = res.x[:2*n].reshape(n, 2)
                    r_lp, s_lp = solve_lp_radii(c_s)
                    if r_lp is not None and s_lp > best_sum:
                        best_sum = s_lp
                        best_centers = c_s.copy()
                        best_radii = r_lp.copy()
            except:
                continue
                
    if best_centers is None:
        best_centers = get_hex_init(n, [6, 5, 6, 5, 4])
        best_radii = np.full(n, 0.09)
        best_sum = np.sum(best_radii)
        
    # Safety scaling to guarantee strict numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(best_centers[i]-best_centers[j])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= max(scale * 0.999999, 0.0)
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
