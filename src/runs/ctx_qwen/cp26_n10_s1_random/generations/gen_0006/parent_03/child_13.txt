import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub_rows = []
    b_ub_vals = []
    bounds = []
    
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        lims = [x, 1.0-x, y, 1.0-y]
        upper = max(lims)
        bounds.append((0.0, max(upper, 1e-9)))
        for lim in lims:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(lim)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(d)
            
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub_rows), b_ub=np.array(b_ub_vals),
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-6

def obj_slqp(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_slqp(v, n):
    """Inequality constraints >= 0 for valid packing."""
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    c = []
    # Boundary clearances
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2  => dist >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    idx = np.triu_indices(n, k=1)
    c.append(dists[idx] - r_sum[idx])
    return np.concatenate(c)

def gen_hex(n, r0, rows):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        sx = r0 if ri % 2 == 1 else 0.0
        x = r0 + sx
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Diverse row distributions that naturally fit 26 circles
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [5,5,5,5,6], [4,6,6,6,4], [6,6,5,5,4]]
    inits = []
    
    for p in patterns:
        if sum(p) < n: continue
        for r0 in [0.09, 0.095, 0.10, 0.102]:
            cfg = gen_hex(n, r0, p)
            inits.append(cfg)
            for _ in range(3):
                cfg_p = cfg + rng.uniform(-0.015, 0.015, (n, 2))
                inits.append(np.clip(cfg_p, 0.02, 0.98))
                
    # Add fully random starts for diversity
    for _ in range(5):
        inits.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    bounds_slqp = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    # Phase 1: Joint SLSQP optimization from diverse starts
    for cfg in inits:
        x0 = np.zeros(3*n)
        x0[:n] = cfg[:, 0]
        x0[n:2*n] = cfg[:, 1]
        
        # Feasible initial radii estimation
        dists_wall = np.minimum(np.minimum(cfg[:, 0], 1.0-cfg[:, 0]), 
                                np.minimum(cfg[:, 1], 1.0-cfg[:, 1]))
        diffs = cfg[:, None, :] - cfg[None, :, :]
        dists_nn = np.sqrt(np.sum(diffs**2, axis=2))
        np.fill_diagonal(dists_nn, np.inf)
        min_nn = np.min(dists_nn, axis=1)
        r_init = np.minimum(dists_wall, min_nn / 2.0)
        x0[2*n:] = np.maximum(r_init, 0.01)
        
        try:
            res = minimize(obj_slqp, x0, args=(n,), method='SLSQP',
                          bounds=bounds_slqp, 
                          constraints={'type': 'ineq', 'fun': cons_slqp, 'args': (n,)},
                          options={'maxiter': 2000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_opt = np.column_stack((res.x[:n], res.x[n:2*n]))
                r_opt, s_opt = solve_lp_radii(c_opt)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Coordinate ascent refinement on centers using LP evaluation
    if best_c is not None:
        curr_c = best_c.copy()
        curr_r, curr_s = solve_lp_radii(curr_c)
        
        for step in range(3000):
            i = rng.integers(n)
            old = curr_c[i].copy()
            best_move_s = curr_s
            best_move = None
            
            # Try multiple random perturbations for the selected circle
            for _ in range(10):
                cand = curr_c[i] + rng.uniform(-0.025, 0.025, 2)
                cand = np.clip(cand, 0.01, 0.99)
                curr_c[i] = cand
                _, s = solve_lp_radii(curr_c)
                if s > best_move_s:
                    best_move_s = s
                    best_move = cand.copy()
                    
            curr_c[i] = best_move if best_move is not None else old
            curr_s = best_move_s
            
            r_tmp, s_tmp = solve_lp_radii(curr_c)
            if s_tmp > curr_s:
                curr_s = s_tmp
                curr_r = r_tmp
                
            if curr_s > best_sum:
                best_sum = curr_s
                best_c = curr_c.copy()
                best_r = curr_r.copy()
                
    # Phase 3: Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y = best_c[i]
        r = best_r[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
    best_r *= scale * 0.999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
