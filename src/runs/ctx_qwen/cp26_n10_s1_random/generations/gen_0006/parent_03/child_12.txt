2. **Two-phase optimization**: First, we optimize centers to maximize the *minimum clearance* (equal-radius approximation) using Nelder-Mead. import numpy as np
from scipy.optimize import minimize, linprog

def compute_min_clearance(centers):
    """Computes the maximum feasible equal radius for given centers."""
    n = centers.shape[0]
    d_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pairwise = np.min(dists) / 2.0
    return min(np.min(d_bound), min_pairwise)

def neg_clearance(x):
    """Objective for Nelder-Mead: maximize minimum clearance."""
    return -compute_min_clearance(x.reshape(26, 2))

def solve_lp_radii(centers, A_ub):
    """Solves the LP to maximize sum of radii for fixed centers using precomputed A_ub."""
    n = centers.shape[0]
    b_ub = np.zeros(A_ub.shape[0])
    
    for i in range(n):
        x, y = centers[i]
        b_ub[4*i] = x
        b_ub[4*i+1] = 1.0 - x
        b_ub[4*i+2] = y
        b_ub[4*i+3] = 1.0 - y
        
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    idx = np.triu_indices(n, k=1)
    b_ub[4*n:] = dists[idx]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            r_out = np.maximum(res.x, 0.0)
            return r_out, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    # Precompute LP constraint matrix structure (constant for fixed n)
    A_ub = np.zeros((4*n + n*(n-1)//2, n))
    for i in range(n):
        A_ub[4*i, i] = 1.0
        A_ub[4*i+1, i] = 1.0
        A_ub[4*i+2, i] = 1.0
        A_ub[4*i+3, i] = 1.0
        
    idx = 4*n
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
            
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = []
    
    # Hexagonal patterns known to be near-optimal for n=26
    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 6, 4],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4],
        [5, 5, 5, 5, 6], [6, 5, 5, 6, 4]
    ]
    
    # Generate diverse lattice starts with varying densities
    for r0 in [0.085, 0.095, 0.105, 0.115]:
        for pattern in row_patterns:
            if sum(pattern) < n: 
                continue
            pts = []
            y = r0
            row_idx = 0
            for count in pattern:
                shift = r0 if row_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(count):
                    if len(pts) >= n: 
                        break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
                row_idx += 1
            if len(pts) >= n:
                configs.append(np.array(pts[:n]))
                # Perturbed variants to break symmetry
                for _ in range(3):
                    p = np.array(pts[:n])
                    p += rng.uniform(-0.02, 0.02, (n, 2))
                    configs.append(np.clip(p, 0.05, 0.95))
                    
    # Add random dense starts
    for _ in range(12):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # Phase 1: Maximize min-clearance to find dense center layouts
    for cfg in configs:
        try:
            res = minimize(neg_clearance, cfg.flatten(), method='Nelder-Mead',
                           options={'maxiter': 15000, 'xatol': 1e-9, 'fatol': 1e-12})
            c_opt = res.x.reshape(n, 2)
            r_lp, s_lp = solve_lp_radii(c_opt, A_ub)
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = c_opt.copy()
                best_radii = r_lp.copy()
        except Exception:
            continue
            
    # Phase 2: Coordinate ascent on centers to directly maximize LP sum
    if best_centers is not None:
        curr_centers = best_centers.copy()
        curr_radii, curr_sum = solve_lp_radii(curr_centers, A_ub)
        best_sum = curr_sum
        best_radii = curr_radii
        best_centers = curr_centers

                        best_local_c = new_c.copy()

                    curr_sum = best_local_sum
                    curr_radii = solve_lp_radii(curr_centers, A_ub)[0]

                step *= 0.85

        best_sum = curr_sum
        best_centers = curr_centers
        best_radii = curr_radii
        
    # Phase 3: Strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
