from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

def get_equal_radius_constraints(vars_arr, n):
    """Constraints for maximizing a single equal radius R."""
    x = vars_arr[:2*n:2]
    y = vars_arr[1:2*n:2]
    R = vars_arr[-1]
    
    c = []
    # Boundary constraints: circle i must be at least R away from walls
    c.append(x - R)
    c.append(1.0 - x - R)
    c.append(y - R)
    c.append(1.0 - y - R)
    
    # Pairwise constraints: distance between centers >= 2R
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    idx = np.triu_indices(n, k=1)
    c.append(dists[idx] - 2.0*R)
    
    return np.concatenate(c)

def obj_equal_radius(vars_arr):
    """Objective: maximize R => minimize -R"""
    return -vars_arr[-1]

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = centers[i]
        lims = (x, 1.0 - x, y, 1.0 - y)
        for lim in lims:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(lim)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(d)
            
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0.0, None), method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def get_joint_constraints(vars_arr, n):
    """Constraints for joint center and radius optimization."""
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    idx = np.triu_indices(n, k=1)
    c.append(dists[idx] - r_sum[idx])
    return np.concatenate(c)

def obj_joint(vars_arr):
    """Objective: maximize sum of radii => minimize -sum(r)"""
    return -np.sum(vars_arr[2::3])
