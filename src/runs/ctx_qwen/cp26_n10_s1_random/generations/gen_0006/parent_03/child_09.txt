import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    m = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    # Working copy to safely clip coordinates for feasibility
    c_work = centers.copy()
    idx = 0
    
    for i in range(n):
        x, y = c_work[i]
        x = max(1e-6, min(1.0 - 1e-6, x))
        y = max(1e-6, min(1.0 - 1e-6, y))
        c_work[i] = (x, y)
        
        lims = [x, 1.0 - x, y, 1.0 - y]
        for lim in lims:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(c_work[i, 0] - c_work[j, 0], c_work[i, 1] - c_work[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0
