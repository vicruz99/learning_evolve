import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        lims = (x, 1-x, y, 1-y)
        for lim in lims:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(lim)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0.0, None), method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def lp_objective(centers_flat):
    """Objective for Nelder-Mead: maximize LP sum of radii."""
    centers = centers_flat.reshape(26, 2)
    _, s = solve_lp(centers)
    return -s

def get_constraints(vars_arr, n):
    """Computes inequality constraints >= 0 for SLSQP."""
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    c = []
    # Boundary
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    # Pairwise squared (smooth)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    r_sum = r[:, None] + r[None, :]
    idx = np.triu_indices(n, k=1)
    c.append((dx[idx]**2 + dy[idx]**2) - r_sum[idx]**2)
    return np.concatenate(c)

def objective_func(vars_arr):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(vars_arr[2::3])

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    bounds_slqp = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * n
    cons_slqp = {'type': 'ineq', 'fun': get_constraints, 'args': (n,)}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    inits = []
    
    # Generate hexagonal initializations with varying row structures
    for n_rows in [4, 5, 6, 7]:
        for st in [0, 1]:
            pts = []
            y = 0.09
            for row in range(n_rows):
                shift = 0.09 if (row + st) % 2 == 1 else 0.0
                x = 0.09 + shift
                while x + 0.09 <= 1.0 and len(pts) < n:
                    pts.append([x, y])
                    x += 0.18
                y += 0.156
            if len(pts) >= n:
                inits.append(np.array(pts[:n]))
                
    # Rotated hex grids to explore tilted packings
    for ang in [0.05, -0.05, 0.1, -0.1]:
        base = inits[0].copy() if inits else rng.uniform(0.1, 0.9, (n, 2))
        center_pt = np.array([0.5, 0.5])
        rot = base - center_pt
        ca, sa = np.cos(ang), np.sin(ang)
        br = rot @ np.array([[ca, -sa], [sa, ca]]) + center_pt
        mn, mx = br.min(0), br.max(0)
        sc = 0.85 / np.max(mx - mn)
        br = (br - mn) * sc + 0.075
        inits.append(br)
        
    # Random placements
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Phase 1: SLSQP from diverse starts
    for cfg in inits:
        for _ in range(4):
            pert = np.clip(cfg + rng.uniform(-0.015, 0.015, (n, 2)), 0.05, 0.95)
            x0 = np.zeros(3*n)
            x0[0::3] = pert[:, 0]
            x0[1::3] = pert[:, 1]
            x0[2::3] = 0.075
            try:
                res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_slqp,
                               constraints=cons_slqp, options={'maxiter': 6000, 'ftol': 1e-14})
                if np.isfinite(res.fun):
                    c_s = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_lp, s_lp = solve_lp(c_s)
                    if s_lp > best_sum:
                        best_sum = s_lp
                        best_centers = c_s.copy()
                        best_radii = r_lp.copy()
            except Exception:
                pass

                          options={'maxiter': 10000, 'xatol': 1e-9, 'fatol': 1e-12})
        c_nm = res_nm.x.reshape(n, 2)
        r_nm, s_nm = solve_lp(c_nm)
        if s_nm > best_sum:
            best_sum = s_nm
            best_centers = c_nm
            best_radii = r_nm

            i = rng.integers(n)

            if s_new > curr_s:
                curr_s, curr_r = s_new, r_new
                if s_new > best_sum:
                    best_sum = s_new
                    best_radii = curr_r.copy()
                    best_centers = curr_c.copy()
            else:
                curr_c[i] = old
            step *= 0.999
            
    # Phase 3: Force-directed relaxation to relieve tight constraints
    if best_centers is not None:
        curr_c = best_centers.copy()
        for _ in range(300):
            curr_r, _ = solve_lp(curr_c)
            forces = np.zeros_like(curr_c)
            for i in range(n):
                x, y = curr_c[i]
                r = curr_r[i]
                if x - r < 0.001: forces[i, 0] += 0.1
                if 1 - x - r < 0.001: forces[i, 0] -= 0.1
                if y - r < 0.001: forces[i, 1] += 0.1
                if 1 - y - r < 0.001: forces[i, 1] -= 0.1
                for j in range(i+1, n):
                    dx = curr_c[i,0] - curr_c[j,0]
                    dy = curr_c[i,1] - curr_c[j,1]
                    d = np.sqrt(dx**2 + dy**2)
                    if d < 1e-4: continue
                    if d - curr_r[i] - curr_r[j] < 0.002:
                        f = 0.02 / (d + 1e-3)
                        forces[i, 0] += f * dx
                        forces[i, 1] += f * dy
                        forces[j, 0] -= f * dx
                        forces[j, 1] -= f * dy
            curr_c += forces * 0.02
            curr_c = np.clip(curr_c, 0.01, 0.99)
            r_new, s_new = solve_lp(curr_c)
            if s_new > best_sum:
                best_sum = s_new
                best_radii = r_new
                best_centers = curr_c.copy()
                
    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt((best_centers[i,0]-best_centers[j,0])**2 + (best_centers[i,1]-best_centers[j,1])**2)
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
