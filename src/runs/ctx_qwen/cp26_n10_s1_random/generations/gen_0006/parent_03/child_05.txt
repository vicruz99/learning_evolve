import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def build_A_ub():
    """Precomputes the constant inequality constraint matrix for the radii LP."""
    A = np.zeros((4*N + N*(N-1)//2, N))
    row = 0
    for i in range(N):
        # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
        A[row, i] = 1.0; row += 1
        A[row, i] = 1.0; row += 1
        A[row, i] = 1.0; row += 1
        A[row, i] = 1.0; row += 1
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(N):
        for j in range(i+1, N):
            A[row, i] = 1.0
            A[row, j] = 1.0
            row += 1
    return A

A_ub_global = build_A_ub()
TRI_IDX = np.triu_indices(N, k=1)

def compute_b_ub(centers):
    """Computes the right-hand side vector for the radii LP given fixed centers."""
    b = np.empty(4*N + N*(N-1)//2)
    b[:N] = centers[:, 0]
    b[N:2*N] = 1.0 - centers[:, 0]
    b[2*N:3*N] = centers[:, 1]
    b[3*N:4*N] = 1.0 - centers[:, 1]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    b[4*N:] = dists[TRI_IDX]
    return b

def lp_max_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    b = compute_b_ub(centers)
    bounds = [(0.0, 0.5)] * N
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_global, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0

def objective_clearance(x):
    """Objective for Nelder-Mead: maximize minimum clearance (equal radius packing)."""
    centers = np.clip(x.reshape(N, 2), 0.001, 0.999)
    d_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    min_bound = np.min(d_bound)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pairwise = np.min(dists) / 2.0
    
    return -min(min_bound, min_pairwise)

def run_packing():
    rng = np.random.default_rng(42)
    inits = []
    
    # Phase 1: Diverse Initialization
    # Hexagonal patterns covering known optimal topologies for N=26
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [5, 6, 6, 5, 4], [6, 6, 5, 5, 4],
        [5, 5, 5, 5, 6], [6, 5, 5, 6, 4], [5, 6, 5, 5, 5]
    ]
    for r0 in [0.09, 0.095, 0.10, 0.105]:
        for pat in patterns:
            pts = []
            y = r0
            ridx = 0
            for cnt in pat:
                shift = r0 if ridx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= N: break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
                ridx += 1
            if len(pts) >= N:
                inits.append(np.array(pts[:N]))
                
    # Regular grid fallback
    gx = np.linspace(0.125, 0.875, 5)
    gy = np.linspace(0.125, 0.875, 5)
    pts = np.array([(x,y) for y in gy for x in gx])
    pts = np.vstack([pts, [0.5, 0.5]])
    inits.append(pts)
    
    # Random starts to escape symmetry traps
    for _ in range(5):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 2: Clearance Optimization + LP Refinement
    for cfg in inits:
        cfg_p = cfg + rng.uniform(-0.005, 0.005, cfg.shape)
        cfg_p = np.clip(cfg_p, 0.05, 0.95)
        
        res = minimize(objective_clearance, cfg_p.flatten(), method='Nelder-Mead',
                       options={'maxiter': 8000, 'xatol': 1e-8, 'fatol': 1e-12})
        c_opt = res.x.reshape(N, 2)
        
        r_lp, s_lp = lp_max_radii(c_opt)
        if s_lp > best_sum:
            best_sum = s_lp
            best_centers = c_opt.copy()
            best_radii = r_lp.copy()
            
    # Phase 3: Simulated Annealing on Centers to maximize LP sum
    if best_centers is not None:
        curr_c = best_centers.copy()
        curr_r, curr_s = lp_max_radii(curr_c)
        best_c, best_r, best_s = curr_c.copy(), curr_r.copy(), curr_s
        
        T = 0.02
        for step in range(2500):
            i = rng.integers(N)
            old_pos = curr_c[i].copy()
            step_size = 0.012 * max(0.05, (2500 - step)/2500)
            curr_c[i] += rng.uniform(-step_size, step_size, 2)
            curr_c[i] = np.clip(curr_c[i], 0.02, 0.98)
            
            new_r, new_s = lp_max_radii(curr_c)
            
            if new_s > curr_s:
                delta = new_s - curr_s
                curr_s = new_s
                curr_r = new_r
                if new_s > best_s:
                    best_s = new_s
                    best_c = curr_c.copy()
                    best_r = curr_r.copy()
            else:
                delta = new_s - curr_s
                if rng.random() < np.exp(delta / (T + 1e-8)):
                    curr_s = new_s
                    curr_r = new_r
                else:
                    curr_c[i] = old_pos
            T *= 0.9995
            
    # Phase 4: Strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(N):
        x, y = best_c[i]
        r = best_r[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999995
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
