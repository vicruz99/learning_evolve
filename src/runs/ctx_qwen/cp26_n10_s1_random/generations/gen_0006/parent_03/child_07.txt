import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    centers = np.clip(centers, 0.001, 0.999)
    
    c_obj = -np.ones(n)
    
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    A_bound = np.repeat(np.eye(n), 4, axis=0)
    b_bound = np.repeat(lims, 4)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    idx_i, idx_j = np.triu_indices(n, k=1)
    diff = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    m = len(idx_i)
    A_pair = np.zeros((m, n))
    A_pair[np.arange(m), idx_i] = 1.0
    A_pair[np.arange(m), idx_j] = 1.0
    
    A_ub = np.vstack([A_bound, A_pair])
    b_ub = np.concatenate([b_bound, dists])
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def neg_lp_sum(centers_flat, n):
    """Objective for optimizer: maximize LP sum of radii."""
    centers = centers_flat.reshape(n, 2)
    centers = np.clip(centers, 0.001, 0.999)
    _, s = solve_lp(centers)
    return -s

def get_hex_centers(n, row_counts, r0):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    row_idx = 0
    for count in row_counts:
        shift = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(count):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row_idx += 1
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
