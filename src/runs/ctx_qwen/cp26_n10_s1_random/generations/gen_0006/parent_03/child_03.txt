import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub_rows = []
    b_ub_vals = []
    bounds = []
    
    # Wall constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        x, y = centers[i]
        lims = [x, 1.0 - x, y, 1.0 - y]
        mx = max(min(lims), 1e-9)
        bounds.append((0.0, mx))
        for lim in lims:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(lim)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(d)
            
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    # Fallback to small valid radii if LP fails
    return np.full(n, 1e-5), 0.0

def lp_objective(vars_arr, n):
    """Objective for center optimization: minimize negative LP sum of radii."""
    centers = np.clip(vars_arr.reshape(n, 2), 0.0, 1.0)
    _, s = solve_lp_radii(centers)
    return -s

def obj_joint(v, n):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def cons_joint(v, n):
    """Inequality constraints >= 0 for joint SLSQP."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    idx = np.triu_indices(n, k=1)
    c = np.concatenate([c, dists[idx] - r_sum[idx]])
    return c

    ]

            y = r0

                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
                row_idx += 1
            if len(pts) >= n:
                inits.append(np.array(pts[:n]))
                
    # Grid + center
    gx = np.linspace(0.12, 0.88, 5)
    gy = np.linspace(0.12, 0.88, 5)

    pts = np.vstack([pts, [0.5, 0.5]])

                s_j = np.sum(r_j)
                if s_j > best_sum:
                    best_centers = c_j
                    best_radii = r_j
                    best_sum = s_j

            pass

    if best_centers is not None:

            x, y = best_centers[i]
            r = best_radii[i]
            if r > 1e-9:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                rs = best_radii[i] + best_radii[j]
                if rs > 1e-9:
                    scale = min(scale, d / rs)
        best_radii *= scale * 0.9999995
        best_sum = float(np.sum(best_radii))
    else:
        # Absolute fallback
        best_centers = inits[0]
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    return best_centers, best_radii, best_sum
