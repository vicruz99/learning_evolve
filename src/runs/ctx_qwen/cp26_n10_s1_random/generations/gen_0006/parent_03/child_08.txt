from scipy.optimize import minimize, linprog

def solve_lp(centers):

    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub_list = []
    b_ub_list = []
    bounds = [(0, None)] * n
    
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        x, y = centers[i]
        lims = [x, 1.0 - x, y, 1.0 - y]
        for lim in lims:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(lim)

    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(d)
            
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def obj_equal_r(vars_arr, n):
    """Objective for equal-radius phase: maximize r"""
    return -vars_arr[2 * n]

def cons_equal_r(vars_arr, n):
    """Inequality constraints >= 0 for equal-radius phase"""
    x = vars_arr[:n]
    y = vars_arr[n:2 * n]
    r = vars_arr[2 * n]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]

    idx = np.triu_indices(n, k=1)
    c.append(dists[idx] - 2.0 * r)
    return np.concatenate(c)

def obj_joint(vars_arr, n):
    """Objective for joint phase: maximize sum(r)"""
    return -np.sum(vars_arr[2 * n:])

def cons_joint(vars_arr, n):
    """Inequality constraints >= 0 for joint center/radius optimization"""
    x = vars_arr[:n]
    y = vars_arr[n:2 * n]
    r = vars_arr[2 * n:]

    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)

    dy = y[:, np.newaxis] - y[np.newaxis, :]

    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    idx = np.triu_indices(n, k=1)
    c.append(dists[idx] - r_sum[idx])
    return np.concatenate(c)
