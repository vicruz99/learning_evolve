import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    x = centers[:, 0]
    y = centers[:, 1]
    lims = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m_pairs = n * (n - 1) // 2
    m_total = n + m_pairs
    
    A_ub = np.zeros((m_total, n))
    b_ub = np.zeros(m_total)
    
    # Wall constraints: r_i <= min(x, 1-x, y, 1-y)
    A_ub[:n, np.arange(n)] = 1.0
    b_ub[:n] = lims
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    idx = np.triu_indices(n, k=1)
    i_idx, j_idx = idx
    A_ub[n:, i_idx] = 1.0
    A_ub[n:, j_idx] = 1.0
    b_ub[n:] = dists[idx]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-6

def neg_lp_sum(centers_flat):
    """Objective for optimizer: negative of max sum of radii."""
    n = 26
    centers = centers_flat.reshape(n, 2)
    # Keep centers safely inside the square during optimization
    centers = np.clip(centers, 0.005, 0.995)
    _, s = solve_lp(centers)
    return -s

def generate_hex_init(pattern, r0):
    """Generates a hexagonal lattice configuration based on row counts."""
    n = 26
    pts = []
    y = r0
    row_idx = 0
    for count in pattern:
        shift = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(count):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row_idx += 1
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
