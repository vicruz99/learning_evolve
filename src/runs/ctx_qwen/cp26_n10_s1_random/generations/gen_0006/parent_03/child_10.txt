import numpy as np
from scipy.optimize import minimize, linprog

def compute_equal_radius(centers):
    """Computes the maximum feasible equal radius for given centers."""
    n = centers.shape[0]
    d_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    r_bound = np.min(d_bound)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_pair = np.min(dists) / 2.0
    
    return min(r_bound, r_pair)

def objective_equal_radius(x, n):
    """Objective to maximize equal radius (minimize negative)."""
    centers = x.reshape(n, 2)
    return -compute_equal_radius(centers)

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        x, y = centers[i]
        lims = [x, 1.0 - x, y, 1.0 - y]
        for lim in lims:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(lim)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, 0.5)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def generate_hex_centers(n, r0, shift_x=0.0, shift_y=0.0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0 + shift_y
    row = 0
    while len(pts) < n:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift + shift_x
        while x + r0 <= 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = []
    
    # 1. Hexagonal lattices with various base radii and shifts
    for r0 in [0.095, 0.100, 0.101, 0.102, 0.105]:
        for sx in np.linspace(-0.02, 0.02, 5):
            for sy in np.linspace(-0.02, 0.02, 5):
                c = generate_hex_centers(n, r0, sx, sy)
                configs.append(c)
                
    # 2. Specific row patterns known to be near-optimal for N=26
    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 6, 5, 4]
    ]
    for pattern in row_patterns:
        r0 = 0.101
        pts = []
        y = r0
        row_idx = 0
        for count in pattern:
            shift = r0 if row_idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(count):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row_idx += 1
        if len(pts) == n:
            configs.append(np.array(pts))
            
    # 3. Perturbed versions to break symmetry
    for c in configs[:25]:
        for _ in range(3):
            p = c + rng.uniform(-0.015, 0.015, c.shape)
            p = np.clip(p, 0.02, 0.98)
            configs.append(p)
            
    # 4. Random dense placements
    for _ in range(10):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # Phase 1: Optimize centers for maximum equal radius
    for cfg in configs:
        res = minimize(objective_equal_radius, cfg.flatten(), args=(n,),
                       method='Nelder-Mead', 
                       options={'maxiter': 5000, 'xatol': 1e-9, 'fatol': 1e-12})
        centers_opt = res.x.reshape(n, 2)
        
        # Phase 2: Exact LP refinement on fixed centers to allow variable radii
        r_lp, s_lp = solve_lp_radii(centers_opt)
        if s_lp > best_sum:
            best_sum = s_lp
            best_centers = centers_opt.copy()
            best_radii = r_lp.copy()
            
    # Phase 3: Stochastic local refinement of centers using LP evaluation
    if best_centers is not None:
        curr_centers = best_centers.copy()
        curr_radii, curr_sum = solve_lp_radii(curr_centers)
        
        step = 0.025
        for _ in range(4000):
            idx = rng.integers(n)
            old_pos = curr_centers[idx].copy()
            curr_centers[idx] += rng.uniform(-step, step, 2)
            curr_centers[idx] = np.clip(curr_centers[idx], 0.01, 0.99)
            
            r_new, s_new = solve_lp_radii(curr_centers)
            if s_new > curr_sum + 1e-10:
                curr_sum = s_new
                curr_radii = r_new
                best_sum = curr_sum
                best_centers = curr_centers.copy()
                best_radii = curr_radii.copy()
            else:
                curr_centers[idx] = old_pos
            step *= 0.9995
            
    # Phase 4: Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.99999999
    best_sum = float(np.sum(best_radii))
    
    # Fallback safety net
    if best_centers is None:
        best_centers = generate_hex_centers(n, 0.10)
        best_radii = np.full(n, 0.09)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
