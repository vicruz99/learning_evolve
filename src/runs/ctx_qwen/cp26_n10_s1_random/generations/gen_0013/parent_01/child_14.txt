

<strategy>
The current solution relies heavily on joint SLSQP optimization and gradient ascent using LP duals. While effective, joint optimization of centers and radii simultaneously introduces a highly non-smooth, dense constraint landscape (325 pairwise distance constraints + boundary constraints) that easily traps gradient-based methods in local minima.

To break through the 2.626 barrier and target 2.636+, I will decouple center positioning from radius calculation:
1. **LP-Driven Objective**: For any fixed center configuration, the optimal radii and their sum are found exactly and efficiently via Linear Programming. This makes the objective function `f(C) = LP_sum(C)` well-defined and fast to evaluate.
2. **Stochastic Coordinate Ascent (Hill Climbing)**: Instead of gradient descent on a non-smooth landscape, I will use a robust stochastic local search on centers. Each iteration picks a random circle, samples multiple move directions, evaluates the LP objective, and accepts improvements. This naturally handles the piecewise-linear nature of the packing constraints and escapes shallow local minima.
3. **Diverse Initializations**: I will seed the search with multiple hexagonal lattice patterns (varying row distributions), rotated variants, and dense random clusters to cover different topological basins.
4. **Derivative-Free Polish**: After hill climbing converges, I will apply `L-BFGS-B` directly on the centers using finite differences over the LP objective. This provides a fine-grained local search without the numerical instability of explicit gradients or constraint Jacobians.
5. **Strict Safety Scaling**: A final deterministic scaling step guarantees all circles satisfy the `1e-12` tolerance bounds for validation.

This decoupled approach is computationally efficient, avoids closure/nesting restrictions, and consistently outperforms joint optimization methods in dense packing problems.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

# Precompute constraint matrix structure for LP to avoid recreation overhead
A_LINPROG = np.zeros((M_PAIRS, N))
A_LINPROG[np.arange(M_PAIRS), PAIR_I] = 1.0
A_LINPROG[np.arange(M_PAIRS), PAIR_J] = 1.0

def compute_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diff = centers[PAIR_I] - centers[PAIR_J]
    b_ub = np.hypot(diff[:, 0], diff[:, 1])
    
    try:
        res = linprog(-np.ones(N), A_ub=A_LINPROG, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-9), 0.0

def hill_climb(centers, rng, max_iter=4000, init_step=0.025):
    """Stochastic coordinate ascent on centers to maximize LP sum of radii."""
    n = centers.shape[0]
    c = centers.copy()
    r_best, s_best = compute_lp(c)
    step = init_step
    
    for _ in range(max_iter):
        i = rng.integers(n)
        old_c = c[i].copy()
        best_move = None
        best_s = s_best
        
        # Sample multiple directions per iteration for better landscape exploration
        moves = rng.uniform(-step, step, (15, 2))
        for m in moves:
            c[i] = np.clip(old_c + m, 1e-4, 1.0 - 1e-4)
            _, s_try = compute_lp(c)
            if s_try > best_s:
                best_s = s_try
                best_move = c[i].copy()
            c[i] = old_c
            
        if best_move is not None:
            c[i] = best_move
            r_best, s_best = compute_lp(c)
            step *= 0.998
        else:
            step *= 0.96
            
    return c, r_best, s_best

def polish_lbfgsb_obj(v):
    """Objective function for L-BFGS-B polish: negative LP sum of radii."""
    c = v.reshape(N, 2)
    c = np.clip(c, 1e-4, 1.0 - 1e-4)
    _, s = compute_lp(c)
    return -s

def polish_lbfgsb(centers):
    """Derivative-free local polish using L-BFGS-B on centers."""
    bounds = [(1e-4, 1.0 - 1e-4)] * (2 * N)
    try:
        res = minimize(polish_lbfgsb_obj, centers.flatten(), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 1500, 'ftol': 1e-12})
        return res.x.reshape(N, 2)
    except Exception:
        return centers

def generate_inits(rng, num_starts=20):
    """Generates diverse initial configurations (hexagonal, perturbed, random)."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [6,6,4,6,4], 
        [5,5,6,5,5], [6,4,6,4,6], [5,6,6,5,4], [6,5,4,6,5],
        [5,5,5,5,6], [6,6,6,4,4], [4,5,6,5,6], [5,4,6,5,6]
    ]
    for pat in patterns:
        if sum(pat) != N: continue
        pts = []
        r0 = 0.098
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:N])
        inits.append(pts)
        for _ in range(2):
            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            inits.append(np.clip(p, 0.05, 0.95))
            
    while len(inits) < num_starts:
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = generate_inits(rng, num_starts=20)
    
    # Phase 1: Robust stochastic hill climbing on diverse starts
    # Decoupling center search from radius calculation avoids non-smooth constraint traps
    for cfg in inits:
        c_opt, r_opt, s_opt = hill_climb(cfg, rng, max_iter=3500, init_step=0.025)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    if best_centers is None:
        best_centers = inits[0]
        best_radii, best_sum = compute_lp(best_centers)
        best_sum = np.sum(best_radii)
        
    # Phase 2: Gradient-based local polish to escape coarse local minima
    best_centers = polish_lbfgsb(best_centers)
    best_radii, best_sum = compute_lp(best_centers)
    best_sum = np.sum(best_radii)
    
    # Phase 3: Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
