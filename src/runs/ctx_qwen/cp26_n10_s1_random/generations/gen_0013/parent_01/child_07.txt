import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M = len(PAIR_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    A = np.zeros((M + 4*n, n))
    b = np.zeros(M + 4*n)
    
    A[:M, PAIR_I] = 1.0
    A[:M, PAIR_J] = 1.0
    b[:M] = dists[PAIR_I, PAIR_J]
    
    b[M:2*n] = centers[:, 0]
    A[M:2*n, np.arange(n)] = 1.0
    b[2*n:3*n] = 1.0 - centers[:, 0]
    A[2*n:3*n, np.arange(n)] = 1.0
    b[3*n:4*n] = centers[:, 1]
    A[3*n:4*n, np.arange(n)] = 1.0
    b[4*n:5*n] = 1.0 - centers[:, 1]
    A[4*n:5*n, np.arange(n)] = 1.0

    bounds = [(0, 1.0)] * n

        r = np.maximum(res.x, 1e-9)

        except AttributeError:
            pass

                marg = getattr(getattr(res, 'ineqlin', None), 'marginals', None)
            except AttributeError:

            marg = np.zeros(M + 4*n)
        return r, -res.fun, marg
    except Exception:

def get_grad(centers, marg):
    """Computes gradient of sum of radii w.r.t centers using LP duals."""
    n = centers.shape[0]
    grad = np.zeros((n, 2))
    
    active = marg[:M] > 1e-8
    if np.any(active):
        idx = np.where(active)[0]
        lam = marg[:M][idx]
        ii = PAIR_I[idx]
        jj = PAIR_J[idx]
        d = np.hypot(centers[ii, 0] - centers[jj, 0], centers[ii, 1] - centers[jj, 1])
        d = np.where(d > 1e-9, d, 1.0)
        vec = (centers[ii] - centers[jj]) / d[:, np.newaxis]
        f = lam[:, np.newaxis] * vec
        np.add.at(grad, ii, f)
        np.add.at(grad, jj, -f)
        
    grad[:, 0] += marg[M:2*n] - marg[2*n:3*n]
    grad[:, 1] += marg[3*n:4*n] - marg[4*n:5*n]
    return grad

def grad_ascent(c0):
    """Gradient ascent on centers using LP dual information."""
    c = c0.copy()
    r, s, marg = solve_lp(c)
    if s <= 0: return c, r
    step = 0.006
    for _ in range(250):
        g = get_grad(c, marg)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-7: break
        g = g / g_norm

        r_new, s_new, marg_new = solve_lp(c_new)

            step = min(step * 1.05, 0.025)
        else:
            step *= 0.6
            if step < 1e-5: break
    return c, r

def hill_climb(c0, rng):

def obj_joint(v):
    return -np.sum(v[2*N:])

def cons_joint(v):
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def joint_slsqp(c0, r0):
    """Joint optimization of centers and radii using SLSQP."""
    v0 = np.concatenate([c0[:, 0], c0[:, 1], r0])
    bounds = [(0, 1)] * (2*N) + [(1e-6, 0.5)] * N
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': 6000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            c_opt = np.column_stack((res.x[:N], res.x[N:2*N]))
            r_lp, _, _ = solve_lp(c_opt)
            return c_opt, r_lp
    except Exception:
        pass
    return c0, r0

def make_hex(rows, r0):
    pts = []
    y = r0
    for idx, cnt in enumerate(rows):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(pts[:N])

            inits.append(np.clip(pr, 0.05, 0.95))
            
    for _ in range(8):
        inits.append(rng.uniform(0.12, 0.88, (N, 2)))
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    inits = generate_inits(rng)

    return best_c, best_r, best_s
