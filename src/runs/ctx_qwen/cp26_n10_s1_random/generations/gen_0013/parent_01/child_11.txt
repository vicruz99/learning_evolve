import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M = len(PAIR_I)

def compute_lp_and_grad(centers):
    """Solves LP for max sum of radii given centers, returns radii, sum, and gradient w.r.t centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Boundary limits: distance to nearest wall
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[PAIR_I, PAIR_J]
    
    A_ub = np.zeros((M, n))
    A_ub[np.arange(M), PAIR_I] = 1.0
    A_ub[np.arange(M), PAIR_J] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, 0.0, np.zeros((n, 2))
            
        r = res.x
        s = -res.fun
        grad = np.zeros((n, 2))
        
        # Extract dual variables (marginals) for sensitivity analysis
        marg = None
        try:
            marg = res.marginals.ineqlin
        except AttributeError:
            try:
                marg = res.ineqlin.marginals
            except AttributeError:
                marg = None
                
        if marg is not None:
            lam = np.maximum(np.asarray(marg).flatten(), 0.0)
            mask = lam > 1e-7
            if np.any(mask):
                idx = np.where(mask)[0]
                ii = PAIR_I[idx]
                jj = PAIR_J[idx]
                g = lam[idx]
                d = dists[ii, jj]
                d = np.where(d > 1e-9, d, 1.0)
                vec = g[:, np.newaxis] * (centers[ii] - centers[jj]) / d[:, np.newaxis]
                np.add.at(grad, ii, vec)
                np.add.at(grad, jj, -vec)
                
        return r, s, grad
    except Exception:
        return None, 0.0, np.zeros((n, 2))

def obj_and_jac(x_flat):
    """Objective and Jacobian for L-BFGS-B center optimization."""
    c = x_flat.reshape(N, 2)
    _, s, g = compute_lp_and_grad(c)
    if g is None or np.all(g == 0.0):
        return -s, np.zeros_like(x_flat)
    return -s, -g.flatten()

def obj_only(x_flat):
    return obj_and_jac(x_flat)[0]

def jac_only(x_flat):
    return obj_and_jac(x_flat)[1]

def joint_obj(v):
    """Objective for joint SLSQP optimization."""
    return -np.sum(v[2*N:])

def joint_cons(v):
    """Inequality constraints >= 0 for joint SLSQP optimization."""
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c_list = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    c_list.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c_list)

def run_packing():
    rng = np.random.default_rng(42)
    best_s = -1.0
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [5,6,6,4,5], [7,5,5,5,4], [6,5,5,6,4],
        [5,4,6,5,6], [6,4,5,6,4], [5,6,5,5,5], [4,5,7,5,4]
    ]
    
    for pat in patterns:
        if sum(pat) < N: continue
        pts = []
        y = 0.1
        for ri, cnt in enumerate(pat):
            shift = 0.1 if ri % 2 == 1 else 0.0
            x = 0.1 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 0.2
            y += 0.1732
        pts = np.array(pts[:N])
        # Normalize to fit comfortably in [0.1, 0.9]
        mn, mx = pts.min(axis=0), pts.max(axis=0)
        pts = (pts - mn) / (mx - mn + 1e-9) * 0.8 + 0.1
        inits.append(pts)
        for _ in range(3):
            p = pts + rng.uniform(-0.02, 0.02, (N, 2))
            inits.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(10):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    bounds_c = [(0.001, 0.999)] * (2 * N)
    
    # Phase 1: LP Gradient Ascent via L-BFGS-B
    for cfg in inits:
        try:
            res = minimize(obj_only, cfg.flatten(), jac=jac_only,
                          method='L-BFGS-B', bounds=bounds_c,
                          options={'maxiter': 2500, 'ftol': 1e-12})
            c_opt = np.clip(res.x.reshape(N, 2), 0.001, 0.999)
            r_opt, s_opt, _ = compute_lp_and_grad(c_opt)
            if r_opt is not None and s_opt > best_s:
                best_s = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            continue
            
    if best_c is None:
        best_c = inits[0]
        best_r, best_s, _ = compute_lp_and_grad(best_c)
        
    # Phase 2: Coordinate-wise Hill Climbing to escape local traps
    curr_c = best_c.copy()
    curr_r, curr_s, _ = compute_lp_and_grad(curr_c)
    
    for sweep in range(25):
        step = 0.035 * (0.87 ** sweep)
        improved = False
        for i in range(N):
            old = curr_c[i].copy()
            best_move_s = curr_s
            best_move = None
            
            dirs = np.array([
                [step, 0], [-step, 0], [0, step], [0, -step],
                [step*0.7071, step*0.7071], [-step*0.7071, step*0.7071],
                [step*0.7071, -step*0.7071], [-step*0.7071, -step*0.7071],
                rng.uniform(-step, step, 2),
                rng.uniform(-step, step, 2)
            ])
            
            for d in dirs:
                new_pos = np.clip(curr_c[i] + d, 0.001, 0.999)
                curr_c[i] = new_pos
                r_try, s_try, _ = compute_lp_and_grad(curr_c)
                if r_try is not None and s_try > best_move_s + 1e-9:
                    best_move_s = s_try
                    best_move = new_pos.copy()
                else:
                    curr_c[i] = old
                    
            if best_move is not None:
                curr_c[i] = best_move
                curr_r, curr_s, _ = compute_lp_and_grad(curr_c)
                if curr_s > best_s + 1e-8:
                    best_s = curr_s
                    best_c = curr_c.copy()
                    best_r = curr_r.copy()
                    improved = True
            else:
                curr_c[i] = old
        if not improved:
            break
            
    # Phase 3: Joint SLSQP Polish for precise boundary/overlap handling
    x0 = np.zeros(3 * N)
    x0[:N] = best_c[:, 0]
    x0[N:2*N] = best_c[:, 1]
    x0[2*N:] = best_r * 0.98
    
    try:
        res_j = minimize(joint_obj, x0, method='SLSQP',
                         bounds=[(0.0, 1.0)]*(2*N) + [(1e-6, 0.5)]*N,
                         constraints={'type': 'ineq', 'fun': joint_cons},
                         options={'maxiter': 6000, 'ftol': 1e-13})
        if np.isfinite(res_j.fun):
            c_j = np.column_stack((res_j.x[:N], res_j.x[N:2*N]))
            r_j, s_j, _ = compute_lp_and_grad(c_j)
            if r_j is not None and s_j > best_s:
                best_s = s_j
                best_c = c_j.copy()
                best_r = r_j.copy()
    except Exception:
        pass
        
    # Phase 4: Strict safety scaling to guarantee numerical validity against 1e-12 tolerance
    scale = 1.0
    c = best_c
    r = best_r
    for i in range(N):
        if r[i] > 1e-12:
            scale = min(scale, c[i,0]/r[i], (1.0-c[i,0])/r[i], c[i,1]/r[i], (1.0-c[i,1])/r[i])
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
            rs = r[i] + r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999995
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
