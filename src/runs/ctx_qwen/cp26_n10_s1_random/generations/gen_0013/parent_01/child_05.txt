import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
M = len(PAIRS_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    A_ub = np.zeros((M, n))
    A_ub[np.arange(M), PAIRS_I] = 1.0
    A_ub[np.arange(M), PAIRS_J] = 1.0
    b_ub = dists[PAIRS_I, PAIRS_J]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return np.full(n, 1e-7), 0.0, None

def compute_grad(centers, res):
    """Compute gradient of LP objective w.r.t centers using dual variables."""
    n = centers.shape[0]
    grad = np.zeros((n, 2))
    if res is None:
        return grad
    try:
        # Pairwise constraints marginals
        m_pairs = getattr(res.ineqlin, 'marginals', None)
        if m_pairs is not None:
            gains = np.maximum(-m_pairs, 0.0)
            mask = gains > 1e-8
            if np.any(mask):
                ii = PAIRS_I[mask]
                jj = PAIRS_J[mask]
                g = gains[mask]
                d_ij = np.hypot(centers[ii, 0] - centers[jj, 0], 
                                centers[ii, 1] - centers[jj, 1])
                d_safe = np.where(d_ij > 1e-9, d_ij, 1.0)
                dx = (centers[ii, 0] - centers[jj, 0]) / d_safe
                dy = (centers[ii, 1] - centers[jj, 1]) / d_safe
                np.add.at(grad, ii, g[:, None] * np.stack([dx, dy], axis=1))
                np.add.at(grad, jj, -g[:, None] * np.stack([dx, dy], axis=1))
                
        # Boundary constraints marginals (upper bounds on r_i)
        m_up = getattr(res.marginals, 'upper', None)
        if m_up is not None:
            gains_b = np.maximum(-m_up, 0.0)
            for i in range(n):
                if gains_b[i] > 1e-8:
                    x, y = centers[i]
                    vals = [x, 1.0-x, y, 1.0-y]
                    min_idx = np.argmin(vals)
                    # Gradient points in direction that increases the limiting boundary distance
                    if min_idx == 0: grad[i, 0] += gains_b[i]
                    elif min_idx == 1: grad[i, 0] -= gains_b[i]
                    elif min_idx == 2: grad[i, 1] += gains_b[i]
                    else: grad[i, 1] -= gains_b[i]
    except Exception:
        pass
    return grad

def gradient_ascent_centers(centers, max_iter=500, lr=0.005):
    """Performs gradient ascent on centers to maximize LP sum of radii."""
    c = centers.copy()
    best_c, best_r, best_s = c.copy(), None, -1.0
    
    for _ in range(max_iter):
        r, s, res = solve_lp(c)
        if s > best_s:
            best_s, best_r, best_c = s, r.copy(), c.copy()
            
        grad = compute_grad(c, res)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-10:
            break
            
        c += lr * (grad / g_norm)
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
        lr *= 0.996
    return best_c, best_r, best_s

def joint_obj(v):
    return -np.sum(v[2*N:])

def joint_cons(v):
    x, y, r = v[:N], v[N:2*N], v[2*N:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c = np.concatenate([c, d2[PAIRS_I, PAIRS_J] - rs[PAIRS_I, PAIRS_J]**2])
    return c
