import numpy as np
from scipy.optimize import linprog

def solve_lp_and_grad(centers, pair_i, pair_j, m_pairs):
    """Solves LP for max sum of radii given centers, returns radii, sum, and gradient w.r.t centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    A_ub = np.zeros((m_pairs, n))
    A_ub[:, pair_i] = 1.0
    A_ub[:, pair_j] = 1.0
    b_ub = dists[pair_i, pair_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, 0.0, np.zeros((n, 2))
            
        r = res.x
        s = -res.fun
        
        grad = np.zeros((n, 2))
        marg = None
        try:
            marg = res.marginals.ineqlin
        except AttributeError:
            try:
                marg = res.ineqlin.marginals
            except AttributeError:
                pass
                
        if marg is not None:
            active_lam = -marg[:m_pairs]
            mask = active_lam > 1e-7
            if np.any(mask):
                idx = np.where(mask)[0]
                lam = active_lam[idx]
                ii = pair_i[idx]
                jj = pair_j[idx]
                d = dists[ii, jj]
                d = np.where(d > 1e-9, d, 1.0)
                vec = lam[:, np.newaxis] * (centers[ii] - centers[jj]) / d[:, np.newaxis]
                np.add.at(grad, ii, vec)
                np.add.at(grad, jj, -vec)
                
        return r, s, grad
    except Exception:
        return None, 0.0, np.zeros((n, 2))

def optimize_centers(c0, pair_i, pair_j, m_pairs, lr=0.008, max_iter=600):
    """Performs gradient ascent on centers to maximize LP sum of radii."""
    c = c0.copy()
    r_best, s_best, _ = solve_lp_and_grad(c, pair_i, pair_j, m_pairs)
    if r_best is None:
        return c, s_best
        
    vel = np.zeros_like(c)
    momentum = 0.5
    step = lr
    
    for _ in range(max_iter):
        r, s, g = solve_lp_and_grad(c, pair_i, pair_j, m_pairs)
        if r is None:
            break
            
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-8:
            break
            
        g_dir = g / g_norm
        vel = momentum * vel + step * g_dir
        c_new = c + vel
        c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
        
        r_new, s_new, _ = solve_lp_and_grad(c_new, pair_i, pair_j, m_pairs)
        if r_new is not None and s_new > s_best + 1e-9:
            c = c_new
            r_best, s_best = r_new, s_new
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.85
            vel *= 0.3
            
        if step < 1e-6:
            break
            
    return c, s_best
