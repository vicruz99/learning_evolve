

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)
A_PAIRS = np.zeros((M_PAIRS, N))
A_PAIRS[np.arange(M_PAIRS), PAIR_I] = 1.0
A_PAIRS[np.arange(M_PAIRS), PAIR_J] = 1.0

def solve_lp(centers):
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[PAIR_I, PAIR_J]
    
    try:
        res = linprog(c_obj, A_ub=A_PAIRS, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-7), 0.0

def lp_objective(v):
    return -solve_lp(v.reshape(N, 2))[1]

def joint_objective(v):
    return -np.sum(v[2*N:])

def joint_constraints(v):
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    c_list = []
    c_list.append(x - r)
    c_list.append(1.0 - x - r)
    c_list.append(y - r)
    c_list.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    idx = np.triu_indices(N, k=1)
    c_list.append(d2[idx] - rs[idx]**2)
    return np.concatenate(c_list)

def run_packing():
    rng = np.random.default_rng(42)
    best_s = -1.0
    best_c = None
    best_r = None
    
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4], 
            [6,6,5,5,4], [5,6,4,6,5], [6,5,5,6,4], [5,5,5,5,6],
            [5,7,5,5,4], [7,5,5,5,4], [6,4,6,5,5]]
            
    for pat in pats:
        if sum(pat) != N: continue
        pts = []
        r0 = 0.10
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx%2 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:N])
        mn, mx = pts.min(0), pts.max(0)
        span = mx - mn + 1e-9
        pts = (pts - mn) / span * 0.82 + 0.09
        
        for angle in [0.0, 0.05, -0.05, 0.15, -0.15]:
            rot_pts = pts.copy()
            if angle != 0.0:
                c = np.array([0.5, 0.5])
                rot_pts = rot_pts - c
                cos_a, sin_a = np.cos(angle), np.sin(angle)
                rot_pts = np.dot(rot_pts, [[cos_a, -sin_a], [sin_a, cos_a]])
                rot_pts += c
                rot_pts = np.clip(rot_pts, 0.01, 0.99)
            inits.append(rot_pts)
            
            for _ in range(2):
                pert = np.clip(rot_pts + rng.uniform(-0.015, 0.015, pts.shape), 0.01, 0.99)
                inits.append(pert)
                
    for _ in range(5):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    bounds_c = [(0.005, 0.995)] * (2*N)
    
    for cfg in inits:
        try:
            res = minimize(lp_objective, cfg.flatten(), method='L-BFGS-B', bounds=bounds_c, 
                           options={'maxiter': 2000, 'ftol': 1e-13, 'gtol': 1e-9})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(N, 2)
                r_opt, s_opt = solve_lp(c_opt)
                if s_opt > best_s:
                    best_s = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            pass
            
    if best_c is not None:
        curr_c = best_c.copy()
        curr_r, curr_s = solve_lp(curr_c)
        for sweep in range(12):
            step = 0.025 * (0.85 ** sweep)
            for i in range(N):
                old = curr_c[i].copy()
                best_move = None
                best_move_s = curr_s
                for _ in range(15):
                    d = rng.uniform(-step, step, 2)
                    new_pos = np.clip(curr_c[i] + d, 0.005, 0.995)
                    curr_c[i] = new_pos
                    r_try, s_try = solve_lp(curr_c)
                    if s_try > best_move_s:
                        best_move_s = s_try
                        best_move = new_pos.copy()
                    curr_c[i] = old
                if best_move is not None:
                    curr_c[i] = best_move
                    curr_r, curr_s = solve_lp(curr_c)
                    if curr_s > best_s:
                        best_s = curr_s
                        best_c = curr_c.copy()
                        best_r = curr_r.copy()
                        
        x0 = np.concatenate([best_c.flatten(), best_r])
        bounds_slqp = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
        try:
            res = minimize(joint_objective, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints={'type': 'ineq', 'fun': joint_constraints},
                           options={'maxiter': 6000, 'ftol': 1e-14})
            if np.isfinite(res.fun):
                c_slqp = res.x[:2*N].reshape(N, 2)
                r_slqp, s_slqp = solve_lp(c_slqp)
                if s_slqp > best_s:
                    best_s = s_slqp
                    best_c = c_slqp.copy()
                    best_r = r_slqp.copy()
        except Exception:
            pass

    scale = 1.0
    for i in range(N):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
    best_r *= scale * 0.9999999
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
```
