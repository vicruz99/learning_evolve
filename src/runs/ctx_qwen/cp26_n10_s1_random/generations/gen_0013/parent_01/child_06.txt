import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
N_PAIRS = len(PAIR_I)

def compute_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    A_ub = np.zeros((N_PAIRS, n))
    A_ub[np.arange(N_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(N_PAIRS), PAIR_J] = 1.0
    b_ub = dists[PAIR_I, PAIR_J]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0, None

def compute_forces_from_lp(centers):
    """Computes exact repulsive forces based on LP dual variables."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),

    A_ub = np.zeros((N_PAIRS, n))
    A_ub[np.arange(N_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(N_PAIRS), PAIR_J] = 1.0
    b_ub = dists[PAIR_I, PAIR_J]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return np.zeros((n, 2)), None
            
        r = res.x
        forces = np.zeros((n, 2))
        
        # Extract marginals for pairwise constraints
        try:
            marginals = res.marginals.ineqlin
        except AttributeError:
            try:
                marginals = res.ineqlin.marginals
            except AttributeError:
                return forces, r
                
        if marginals is not None:

                idx_active = np.where(active)[0]

                jj = PAIR_J[idx_active]

                d_safe = np.where(d_ij > 1e-9, d_ij, 1.0)
                vec = centers[ii] - centers[jj]
                factors = (lam / d_safe)[:, np.newaxis]
                np.add.at(forces, ii, vec * factors)
                np.add.at(forces, jj, -vec * factors)
                
        return forces, r
    except Exception:
        return np.zeros((n, 2)), None

def gradient_ascent(centers, max_iter=200, lr_init=0.005):
    """Moves centers using LP-derived forces to maximize sum of radii."""
    c = centers.copy()
    forces, r = compute_forces_from_lp(c)
    if r is None:
        r = np.full(N, 0.09)
        
    lr = lr_init
    for _ in range(max_iter):
        forces, r = compute_forces_from_lp(c)
        if r is None:
            break
            
        f_norm = np.linalg.norm(forces)
        if f_norm < 1e-8:
            break
            
        c = c + lr * forces
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
        lr *= 0.995
        
    return c, r

def coordinate_search(centers, max_sweeps=10):
    """Fine-tunes positions by testing individual circle moves."""
    c = centers.copy()
    r, s, _ = compute_lp_radii(c)
    if r is None:
        return c, np.full(N, 0.09)
        
    step = 0.02

        improved = False
        for i in range(N):
            old_pos = c[i].copy()
            best_pos = old_pos
            best_s = s
            
            dirs = np.array([
                [step, 0], [-step, 0], [0, step], [0, -step],
                [step, step], [-step, step], [step, -step], [-step, -step]
            ])
            
            for d in dirs:
                c[i] = np.clip(old_pos + d, 1e-4, 1.0 - 1e-4)
                _, curr_s, _ = compute_lp_radii(c)

                    best_s = curr_s
                    best_pos = c[i].copy()
            c[i] = best_pos

                s = best_s
                r, _, _ = compute_lp_radii(c)
                improved = True
                
        if not improved:
            step *= 0.5
        else:
            step *= 0.9
            
    return c, r

def joint_slsqp(centers, radii):
    """Polishes packing using constrained nonlinear optimization."""
    v0 = np.concatenate([centers.flatten(), radii])
    
    def obj(v):
        return -np.sum(v[2*N:])
        
    def con(v):
        x, y, r = v[:N], v[N:2*N], v[2*N:]
        cons = np.concatenate([
            x - r, 1.0 - x - r, y - r, 1.0 - y - r
        ])
        dx = x[PAIR_I] - x[PAIR_J]
        dy = y[PAIR_I] - y[PAIR_J]
        dr = r[PAIR_I] + r[PAIR_J]
        cons = np.concatenate([cons, dx**2 + dy**2 - dr**2])
        return cons
        
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': con},
                       options={'maxiter': 5000, 'ftol': 1e-13})
        if np.isfinite(res.fun):
            c_opt = np.column_stack((res.x[:N], res.x[N:2*N]))
            r_opt, _, _ = compute_lp_radii(c_opt)
            return c_opt, r_opt
    except Exception:
        pass
    return centers, radii

def generate_hex_config(row_counts, r0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])
