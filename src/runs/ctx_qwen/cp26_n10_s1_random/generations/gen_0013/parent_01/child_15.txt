from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[PAIR_I, PAIR_J]
    
    A_ub = np.zeros((M_PAIRS, n))
    A_ub[np.arange(M_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(M_PAIRS), PAIR_J] = 1.0
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def joint_obj(v):
    """Objective for joint optimization: maximize sum of radii."""
    return -np.sum(v[2 * N:])

def joint_cons(v):
    """Inequality constraints >= 0 for valid packing."""
    x, y, r = v[:N], v[N:2 * N], v[2 * N:]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
