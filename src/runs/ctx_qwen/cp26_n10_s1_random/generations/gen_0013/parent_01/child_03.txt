from scipy.optimize import minimize, linprog

N = 26
PAIRS = np.array([(i, j) for i in range(N) for j in range(i + 1, N)])
NP = len(PAIRS)

def solve_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    A_ub = np.zeros((NP, N))
    A_ub[np.arange(NP), PAIRS[:, 0]] = 1.0
    A_ub[np.arange(NP), PAIRS[:, 1]] = 1.0
    b_ub = dists[PAIRS[:, 0], PAIRS[:, 1]]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-7), 1e-7 * N

def obj_neg_lp(v):
    """Objective for scipy: minimize negative LP sum of radii."""
    _, s = solve_radii(v.reshape(N, 2))
    return -s

def make_hex(rows, r0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    for idx, cnt in enumerate(rows):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    return np.array(pts[:N])

def joint_obj(v):
    """Objective for joint SLSQP: maximize sum of radii."""
    return -np.sum(v[2 * N:])

def joint_cons(v):

    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
