import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIU_I, TRIU_J = np.triu_indices(N_CIRCLES, k=1)
M_PAIRS = len(TRIU_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, l) for l in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub = dists[TRIU_I, TRIU_J]
    
    A_ub = np.zeros((M_PAIRS, n))
    A_ub[np.arange(M_PAIRS), TRIU_I] = 1.0
    A_ub[np.arange(M_PAIRS), TRIU_J] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return lims, np.sum(lims), None

def force_relaxation(centers, max_iter=80, init_step=0.02):
    """Moves centers using LP dual marginals as repulsive forces."""
    c = centers.copy()
    step = init_step
    for _ in range(max_iter):
        _, _, res_lp = solve_lp(c)
        if res_lp is None:
            break
            
        marginals = None
        try:
            marginals = res_lp.marginals.ineqlin
        except AttributeError:
            try:
                marginals = res_lp.ineqlin.marginals
            except AttributeError:
                break
                
        if marginals is None:
            break
            
        forces = np.zeros((N_CIRCLES, 2))
        # Marginals are <= 0 for active constraints in minimization.
        # Gain per unit distance increase is -marginal.
        gains = np.maximum(-marginals[:M_PAIRS], 0.0)
        
        active = gains > 1e-6
        if np.any(active):
            ii = TRIU_I[active]
            jj = TRIU_J[active]
            g = gains[active]
            d_ij = np.sqrt(np.sum((c[ii] - c[jj])**2, axis=1))
            d_safe = np.where(d_ij > 1e-8, d_ij, 1.0)
            vecs = (c[ii] - c[jj]) / d_safe[:, np.newaxis]
            
            np.add.at(forces, ii, vecs * g[:, np.newaxis])
            np.add.at(forces, jj, -vecs * g[:, np.newaxis])
            
        c = c + step * forces
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
        step *= 0.985
        
    return c

def joint_obj(v):
    return -np.sum(v[2*N_CIRCLES:])

def joint_cons(v):
    x = v[:N_CIRCLES]
    y = v[N_CIRCLES:2*N_CIRCLES]
    r = v[2*N_CIRCLES:]
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    dx = x[TRIU_I] - x[TRIU_J]
    dy = y[TRIU_I] - y[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_c = None
    best_r = None
    
    # 1. Generate diverse initial configurations
    starts = []
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], 
        [6,6,5,5,4], [5,4,6,5,6], [6,5,5,6,4], [5,6,4,6,5]
    ]
    for pat in patterns:
        if sum(pat) < N_CIRCLES:
            continue
        pts = []
        y = 0.10
        for ri, cnt in enumerate(pat):
            shift = 0.10 if ri % 2 == 1 else 0.0
            x = 0.10 + shift
            for _ in range(cnt):
                if len(pts) >= N_CIRCLES:
                    break
                pts.append([x, y])
                x += 0.20
            y += 0.10 * np.sqrt(3)
        starts.append(np.array(pts[:N_CIRCLES]))
        
    for _ in range(6):
        starts.append(rng.uniform(0.15, 0.85, (N_CIRCLES, 2)))
        
    # 2. Force-directed relaxation from each start
    for cfg in starts:
        c_relaxed = force_relaxation(np.clip(cfg, 0.02, 0.98), max_iter=150, init_step=0.025)
        r_rel, s_rel, _ = solve_lp(c_relaxed)
        if s_rel > best_sum:
            best_sum = s_rel
            best_c = c_relaxed.copy()
            best_r = r_rel.copy()
            
    if best_c is None:
        best_c = starts[0]
        best_r, best_sum, _ = solve_lp(best_c)
        
    # 3. Simulated Annealing with Shaking + Force Relaxation
    curr_c = best_c.copy()
    curr_s = best_sum
    temp = 0.025
    
    for step_idx in range(400):
        # Shake all centers slightly
        shake_mag = temp * (0.996 ** step_idx)
        c_try = curr_c + rng.uniform(-shake_mag, shake_mag, (N_CIRCLES, 2))
        c_try = np.clip(c_try, 0.01, 0.99)
        
        # Quick force relaxation to find local basin bottom
        c_try = force_relaxation(c_try, max_iter=40, init_step=0.015)
        _, s_try, _ = solve_lp(c_try)
        
        # Accept if better or with SA probability
        if s_try > curr_s or rng.random() < np.exp((s_try - curr_s) / (temp + 1e-6)):
            curr_c = c_try
            curr_s = s_try
            if s_try > best_sum:
                best_sum = s_try
                best_c = c_try.copy()
                best_r, _, _ = solve_lp(c_try)
                
        temp *= 0.997
        
    # 4. Joint SLSQP Polish
    x0 = np.concatenate([best_c.flatten(), best_r * 0.995])
    bounds_j = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(1e-6, 0.5)] * N_CIRCLES
    
    try:
        res = minimize(joint_obj, x0, method='SLSQP', bounds=bounds_j,
                       constraints={'type': 'ineq', 'fun': joint_cons},
                       options={'maxiter': 6000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            c_pol = np.column_stack((res.x[:N_CIRCLES], res.x[N_CIRCLES:2*N_CIRCLES]))
            r_pol, s_pol, _ = solve_lp(c_pol)
            if s_pol > best_sum:
                best_sum = s_pol
                best_c = c_pol.copy()
                best_r = r_pol.copy()
    except Exception:
        pass
        
    # 5. Strict Safety Scaling
    scale = 1.0
    for i in range(N_CIRCLES):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_r *= scale * 0.999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
