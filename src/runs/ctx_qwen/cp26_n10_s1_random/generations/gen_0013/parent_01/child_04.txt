import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M = len(TRIU_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    A_ub = np.zeros((M + n, n))
    b_ub = np.zeros(M + n)
    
    A_ub[:M, TRIU_I] = 1.0
    A_ub[:M, TRIU_J] = 1.0
    b_ub[:M] = dists[TRIU_I, TRIU_J]
    
    A_ub[M:, np.arange(n)] = 1.0
    b_ub[M:] = lims
    
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0, None

def hex_init(pattern, r0, rng):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    for idx, cnt in enumerate(pattern):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    pts = np.array(pts[:N])
    mn, mx = pts.min(axis=0), pts.max(axis=0)
    span = mx - mn
    if span.max() > 1e-6:
        pts = (pts - mn) / span * 0.82 + 0.09
    return pts

def generate_configs(rng):
    """Generates diverse initial center configurations."""
    configs = []
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], [5,5,6,5,5],
        [6,6,5,5,4], [7,6,6,7], [8,6,6,6], [9,5,6,6],
        [5,7,5,5,4], [6,4,6,6,4], [5,6,4,6,5], [6,5,4,5,6]
    ]
    for pat in patterns:
        if sum(pat) < N:
            continue
        for r0 in [0.08, 0.09, 0.10]:
            cfg = hex_init(pat, r0, rng)
            configs.append(cfg)
            for _ in range(3):
                p = cfg + rng.uniform(-0.03, 0.03, cfg.shape)
                configs.append(np.clip(p, 0.05, 0.95))
                
        base = hex_init(pat, 0.095, rng)
        for angle in [0.15, -0.15, 0.25, -0.25, 0.35, -0.35]:
            cos_a, sin_a = np.cos(angle), np.sin(angle)
            rot = np.array([[cos_a, -sin_a], [sin_a, cos_a]])
            cfg_rot = base @ rot
            mn, mx = cfg_rot.min(axis=0), cfg_rot.max(axis=0)
            span = mx - mn
            if span.max() > 1e-6:
                cfg_rot = (cfg_rot - mn) / span * 0.82 + 0.09
            configs.append(np.clip(cfg_rot, 0.05, 0.95))

    for _ in range(10):
        configs.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return configs

def gradient_ascent_lp(centers, rng, max_iter=600, lr=0.003):
    """Gradient ascent on centers using LP dual variables."""
    c = centers.copy()
    r, s, _ = solve_lp(c)
    best_c, best_r, best_s = c.copy(), r.copy(), s
    
    for it in range(max_iter):
        r, s, res = solve_lp(c)
        if res is None:
            break
            
        grad = np.zeros((N, 2))
        try:
            marg = getattr(res.marginals, 'ineqlin', None)
            if marg is None:
                continue
                
            lam = np.maximum(marg[:M], 0.0)
            active = lam > 1e-7
            if np.any(active):
                ii = TRIU_I[active]
                jj = TRIU_J[active]
                g = lam[active]
                d_ij = np.sqrt(np.sum((c[ii] - c[jj])**2, axis=1))
                d_ij = np.maximum(d_ij, 1e-9)
                diff = c[ii] - c[jj]
                factors = (g / d_ij)[:, None]
                np.add.at(grad, ii, diff * factors)
                np.add.at(grad, jj, -diff * factors)
        except:
            pass
            
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-8:
            break
            
        step = lr * (0.992 ** (it // 15))
        c_new = c + step * grad
        c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
        
        r_new, s_new, _ = solve_lp(c_new)
        if s_new > best_s + 1e-9:
            c = c_new
            best_c, best_r, best_s = c.copy(), r_new.copy(), s_new
        else:
            c += rng.uniform(-0.002, 0.002, (N, 2))
            c = np.clip(c, 1e-4, 1.0 - 1e-4)
            
    return best_c, best_r, best_s

def joint_obj(v):
    return -np.sum(v[2*N:])

def joint_cons(v):
    x, y, r = v[:N], v[N:2*N], v[2*N:]
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    dx = x[TRIU_I] - x[TRIU_J]
    dy = y[TRIU_I] - y[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def slsqp_polish(centers, radii):
    """Joint SLSQP optimization."""
    x0 = np.concatenate([centers[:, 0], centers[:, 1], radii * 0.98])
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    
    try:
        res = minimize(joint_obj, x0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': joint_cons},
                       options={'maxiter': 6000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            c_mat = np.column_stack((res.x[:N], res.x[N:2*N]))
            r_lp, s_lp, _ = solve_lp(c_mat)
            return c_mat, r_lp, s_lp
    except:
        pass
    return centers, radii, np.sum(radii)

def coordinate_hill_climb(centers, radii, rng):
    """Iterative local search to fill gaps."""
    c = centers.copy()
    r, s, _ = solve_lp(c)
    best_c, best_r, best_s = c.copy(), r.copy(), s
    
    for sweep in range(12):
        step = 0.02 * (0.85 ** sweep)
        for i in range(N):
            old = c[i].copy()
            best_move_s = s
            best_move = None
            
            dirs = np.array([
                [step, 0], [-step, 0], [0, step], [0, -step],
                [step, step], [-step, step], [step, -step], [-step, -step]
            ])
            dirs = np.vstack([dirs, rng.uniform(-step, step, (4, 2))])
            
            for d in dirs:
                new_pos = np.clip(c[i] + d, 1e-4, 0.999)
                c[i] = new_pos
                r_try, s_try, _ = solve_lp(c)
                if s_try > best_move_s:
                    best_move_s = s_try
                    best_move = new_pos.copy()
                else:
                    c[i] = old
                    
            if best_move is not None:
                c[i] = best_move
                r, s, _ = solve_lp(c)
                if s > best_s + 1e-9:
                    best_c, best_r, best_s = c.copy(), r.copy(), s
            else:
                c[i] = old
    return best_c, best_r, best_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    configs = generate_configs(rng)

        c_opt, r_opt, s_opt = gradient_ascent_lp(cfg, rng, max_iter=500, lr=0.004)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()

    if best_centers is not None:
        c_p, r_p, s_p = slsqp_polish(best_centers, best_radii)
        if s_p > best_sum:
            best_sum = s_p
            best_centers = c_p.copy()

            best_sum = s_h

    if best_centers is None:
        best_centers = hex_init([5,6,5,6,4], 0.09, rng)
        best_radii, best_sum, _ = solve_lp(best_centers)
        
    # Final LP squeeze
    r_final, s_final, _ = solve_lp(best_centers)
    if r_final is not None and s_final > best_sum:
        best_radii = r_final
        best_sum = s_final
        
    # Strict safety scaling
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
