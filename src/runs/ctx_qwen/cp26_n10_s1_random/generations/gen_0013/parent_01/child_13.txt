from scipy.optimize import linprog, minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

def solve_lp_and_grad(centers):
    """Solves LP for max sum of radii given centers, returns radii, sum, and gradient w.r.t centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[PAIR_I] - centers[PAIR_J]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    
    A_ub = np.zeros((M_PAIRS, n))
    A_ub[np.arange(M_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(M_PAIRS), PAIR_J] = 1.0
    b_ub = dists
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, 0.0, np.zeros((n, 2))
            
        r = res.x
        s = -res.fun
        
        grad = np.zeros((n, 2))
        try:
            marg = res.marginals.ineqlin
        except AttributeError:
            try:
                marg = res.ineqlin.marginals
            except AttributeError:
                marg = None
                
        if marg is not None:
            gains = np.maximum(-marg, 0.0)
            mask = gains > 1e-7
            if np.any(mask):

                jj = PAIR_J[idx]

                vec = (g / d)[:, np.newaxis] * (centers[ii] - centers[jj])
                np.add.at(grad, ii, vec)
                np.add.at(grad, jj, -vec)
                
        return r, s, grad
    except Exception:
        return None, 0.0, np.zeros((n, 2))

def gradient_ascent_with_momentum(c0, max_iter=600, init_step=0.012):
    """Performs gradient ascent on centers using momentum and adaptive step size."""
    c = c0.copy()
    r_best, s_best, _ = solve_lp_and_grad(c)
    if r_best is None:
        return c, s_best

        r, s, g = solve_lp_and_grad(c)

            break

        if g_norm < 1e-8:
            break

        c_new = c + vel

        r_new, s_new, _ = solve_lp_and_grad(c_new)
        if r_new is not None and s_new > s_best + 1e-9:
            c = c_new
            r_best, s_best = r_new, s_new
            step = min(step * 1.03, 0.04)
            vel = 0.8 * vel  # dampen velocity on success to prevent overshooting
        else:
            step *= 0.80
            vel *= 0.5
            
        if step < 1e-6:
            break
            
    return c, s_best

def joint_obj(v):
    return -np.sum(v[2*N:])

def joint_cons(v):
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:]
    
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[PAIR_I] - x[PAIR_J]
    dy = y[PAIR_I] - y[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def generate_initial_configs(rng):
    """Generates diverse initial center configurations."""
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],

                        break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            cfg = np.array(pts[:N])

            cfg += rng.uniform(-0.005, 0.005, cfg.shape)
            cfg = np.clip(cfg, 0.04, 0.96)
            configs.append(cfg)

            cfg[k*N//4] = corners[k] + np.random.uniform(-0.02, 0.02, 2)
        configs.append(np.clip(cfg, 0.04, 0.96))
        
    return configs
