import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRIU_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-9)) for lim in lims]
    
    A_ub = np.zeros((M_PAIRS, n))
    A_ub[np.arange(M_PAIRS), TRIU_I] = 1.0
    A_ub[np.arange(M_PAIRS), TRIU_J] = 1.0
    
    dx = centers[TRIU_I, 0] - centers[TRIU_J, 0]
    dy = centers[TRIU_I, 1] - centers[TRIU_J, 1]
    b_ub = np.sqrt(dx**2 + dy**2)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0, None

def compute_grad_and_obj(centers):
    """Returns (-sum_radii, gradient_vector) for minimization routines."""
    r, s, res = solve_lp(centers)
    if res is None:
        return -s, np.zeros_like(centers)
        
    grad = np.zeros_like(centers)
    dx = centers[TRIU_I, 0] - centers[TRIU_J, 0]
    dy = centers[TRIU_I, 1] - centers[TRIU_J, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Extract LP marginals (dual variables for pairwise constraints)
    try:
        marg = getattr(res.marginals, 'ineqlin', None)
        if marg is None:
            marg = getattr(res.ineqlin, 'marginals', None)
    except:
        marg = None
        
    if marg is not None:
        # Marginals are for minimization of -sum(r). 
        # Increasing distance relaxes constraint -> decreases optimal value -> marginal <= 0.
        # Gain in sum(r) per unit distance increase is -marginal.
        gains = np.maximum(-marg, 0.0)
        active = gains > 1e-8
        if np.any(active):
            lam = gains[active]
            d = np.where(dists[active] > 1e-9, dists[active], 1.0)
            factors = (lam / d)[:, np.newaxis]
            diff_vec = (centers[TRIU_I[active]] - centers[TRIU_J[active]]) * factors
            np.add.at(grad, TRIU_I[active], diff_vec)
            np.add.at(grad, TRIU_J[active], -diff_vec)
            
    # Heuristic boundary gradient: push circles towards walls if limited by boundary
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    slack = lims - r
    bound_active = slack < 1e-6
    if np.any(bound_active):
        idxs = np.where(bound_active)[0]
        for i in idxs:
            if centers[i, 0] < 0.5 - 1e-6: grad[i, 0] += 1.0
            elif centers[i, 0] > 0.5 + 1e-6: grad[i, 0] -= 1.0
            if centers[i, 1] < 0.5 - 1e-6: grad[i, 1] += 1.0
            elif centers[i, 1] > 0.5 + 1e-6: grad[i, 1] -= 1.0
            
    # Return negative because scipy.minimize minimizes
    return -s, -grad.flatten()

def obj_func(v):
    return compute_grad_and_obj(v.reshape(N, 2))[0]

def jac_func(v):
    return compute_grad_and_obj(v.reshape(N, 2))[1]

def optimize_centers(c0):
    """Gradient ascent on centers using LP duals + L-BFGS-B."""
    bounds = [(0.005, 0.995)] * (2 * N)
    try:
        res = minimize(obj_func, c0.flatten(), jac=jac_func, method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 2500, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            return res.x.reshape(N, 2), -res.fun
    except:
        pass
    return c0, 0.0

def joint_obj(v):
    return -np.sum(v[2*N:])

def joint_cons(v):
    x, y, r = v[:N], v[N:2*N], v[2*N:]
    # Boundary constraints
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[TRIU_I] - x[TRIU_J]
    dy = y[TRIU_I] - y[TRIU_J]
    dr = r[TRIU_I] + r[TRIU_J]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def optimize_joint(c0, r0):
    """Joint optimization of centers and radii using SLSQP."""
    v0 = np.concatenate([c0.flatten(), np.maximum(r0 * 0.995, 1e-6)])
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    try:
        res = minimize(joint_obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': joint_cons},
                       options={'maxiter': 6000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            c_opt = res.x[:2*N].reshape(N, 2)
            # Re-solve LP for exact optimal radii given the converged centers
            r_lp, s_lp, _ = solve_lp(c_opt)
            return c_opt, r_lp, s_lp
    except:
        pass
    return c0, r0, 0.0

def run_packing():

    scale = 1.0
    for i in range(N):

    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
