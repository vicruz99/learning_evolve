from scipy.optimize import minimize, linprog

def wall_limits(centers):
    """Computes maximum allowed radius for each circle based on square boundaries."""
    return np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))

def solve_lp(centers, n, pair_i, pair_j, limits):
    """Solves LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in limits]
    A_ub = np.zeros((len(pair_i), n))
    A_ub[np.arange(len(pair_i)), pair_i] = 1.0
    A_ub[np.arange(len(pair_i)), pair_j] = 1.0
    dx = centers[pair_i, 0] - centers[pair_j, 0]
    dy = centers[pair_i, 1] - centers[pair_j, 1]
    b_ub = np.hypot(dx, dy)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun, res
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0, None

def compute_grad(centers, n, pair_i, pair_j, limits):

                elif abs(r_i - (1.0 - y)) < 1e-7: grad[i, 1] -= 1.0
                
    return r, s, grad

def obj_grad(v, n, pair_i, pair_j):
    """Objective and gradient for L-BFGS-B: maximizes sum of radii."""
    c = v.reshape(n, 2)
    lims = wall_limits(c)
    r, s, g = compute_grad(c, n, pair_i, pair_j, lims)
    if r is None:
        return 1e6, np.zeros_like(v)
    return -s, -g.flatten()

def joint_obj(v, n):
    """Objective for SLSQP: minimizes negative sum of radii."""
    return -np.sum(v[2*n:])

def joint_cons(v, n):
    """Inequality constraints >= 0 for valid joint packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    cons = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    d2 = dx**2 + dy**2
    rs = r[idx_i] + r[idx_j]
    cons = np.concatenate([cons, d2 - rs**2])
    return cons
