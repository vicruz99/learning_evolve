import numpy as np
from scipy.optimize import minimize, linprog

def make_hex_pattern(n, rows, r0, angle=0.0):
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    pts = np.array(pts[:n])
    
    if angle != 0.0:
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        rot = np.array([[cos_a, -sin_a], [sin_a, cos_a]])
        pts = (pts - pts.mean(axis=0)) @ rot + pts.mean(axis=0)
        
    mn = pts.min(axis=0)
    mx = pts.max(axis=0)
    span = mx - mn
    if span.max() > 1e-9:
        scale = 0.85 / span.max()
        pts = (pts - mn) * scale + 0.5 * (1.0 - scale)
        
    return np.clip(pts, 0.02, 0.98)

def obj_equal(v, n):
    return -v[-1]

def cons_equal(v, n, idx_i, idx_j):
    cx = v[:n]
    cy = v[n:2*n]
    t = v[2*n]
    c = np.concatenate([cx - t, 1.0 - cx - t, cy - t, 1.0 - cy - t])
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    d2 = dx**2 + dy**2
    c = np.concatenate([c, d2 - 4.0 * t**2])
    return c

def obj_unequal(v, n):
    return -np.sum(v[2*n:])

def cons_unequal(v, n, idx_i, idx_j):
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    d2 = dx**2 + dy**2
    rs = r[idx_i] + r[idx_j]
    c = np.concatenate([c, d2 - rs**2])
    return c

def solve_lp_radii(centers, n, idx_i, idx_j):
    lims = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in lims]
    
    dx = centers[idx_i, 0] - centers[idx_j, 0]
    dy = centers[idx_i, 1] - centers[idx_j, 1]
    dists = np.hypot(dx, dy)
    
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    b_ub = dists
    
    for k, (i, j) in enumerate(zip(idx_i, idx_j)):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def run_packing():
