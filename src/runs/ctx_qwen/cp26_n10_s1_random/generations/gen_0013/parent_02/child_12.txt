import numpy as np
from scipy.optimize import linprog, minimize

# Precompute constant indices and matrix structure for the LP
N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
M_PAIRS = len(IDX_I)

# Constraint matrix structure: r_i + r_j <= d_ij  and  r_i <= lim_i
A_ub_const = np.zeros((M_PAIRS + N, N))
A_ub_const[:M_PAIRS, IDX_I] = 1.0
A_ub_const[:M_PAIRS, IDX_J] = 1.0
A_ub_const[M_PAIRS:, np.arange(N)] = 1.0

def solve_lp_fast(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    
    # Maximum radius allowed by boundaries for each circle
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    # Right-hand side vector: pairwise distances and boundary limits
    b_ub = np.empty(M_PAIRS + n)
    dx = centers[IDX_I, 0] - centers[IDX_J, 0]
    dy = centers[IDX_I, 1] - centers[IDX_J, 1]
    b_ub[:M_PAIRS] = np.sqrt(dx*dx + dy*dy)
    b_ub[M_PAIRS:] = lims
    
    bounds = [(0.0, lim) for lim in lims]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_const, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def hill_climb(centers, rng, iters=2500):
    """Adaptive coordinate-wise hill climbing to optimize center positions."""
    n = centers.shape[0]
    curr = centers.copy()
    best_r, best_s = solve_lp_fast(curr)
    step = 0.025
    
    for _ in range(iters):
        i = rng.integers(n)
        old = curr[i].copy()
        curr[i] += rng.uniform(-step, step, 2)
        curr[i] = np.clip(curr[i], 1e-4, 1.0 - 1e-4)
        
        r, s = solve_lp_fast(curr)
        if r is not None and s > best_s + 1e-8:
            best_s = s
            best_r = r
            step = min(step * 1.03, 0.06)
        else:
            curr[i] = old
            if rng.random() < 0.03:
                step *= 0.96
    return curr, best_r, best_s

def joint_obj(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def joint_cons(v, n):
    """Inequality constraints >= 0 for valid packing in SLSQP."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    
    # Boundary constraints
    cons = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[IDX_I] - cx[IDX_J]
    dy = cy[IDX_I] - cy[IDX_J]
    d2 = dx*dx + dy*dy
    rs = r[IDX_I] + r[IDX_J]
    cons = np.concatenate([cons, d2 - rs*rs])
    return cons

def run_packing():
    n = N
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    # 1. Generate diverse initial configurations
    configs = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], [4,6,6,6,4],
        [5,7,5,5,4], [6,5,5,6,4], [5,6,6,4,5], [7,6,6,7], [8,6,6,6],
        [5,5,5,5,6], [6,5,5,5,5], [5,5,5,6,5], [5,5,6,6,4], [4,5,7,5,5]
    ]
                
    for pat in patterns:
        if sum(pat) < n: continue
        r0 = 0.10
        pts = []
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        pts = np.array(pts[:n])
        # Normalize to fit comfortably in [0.05, 0.95]
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn + 1e-9
        pts = (pts - mn) / span * 0.85 + 0.075
        configs.append(pts)
        
    # Add random dense starts
    for _ in range(12):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # 2. Phase 1: Hill climbing from diverse starts
    for cfg in configs:
        c_opt, r_opt, s_opt = hill_climb(cfg, rng, iters=3000)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            best_r = r_opt.copy()
            
    # 3. Phase 2: Multi-scale perturbation search to escape local minima
    if best_c is not None:
        for scale in [0.012, 0.006, 0.0025]:
            for _ in range(15):
                c_pert = best_c + rng.uniform(-scale, scale, (n, 2))
                c_pert = np.clip(c_pert, 1e-4, 1.0 - 1e-4)
                c_opt, r_opt, s_opt = hill_climb(c_pert, rng, iters=1500)

                    best_sum = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
                    
    # 4. Phase 3: Joint SLSQP polish for precise boundary/overlap handling
    if best_c is not None:

        bounds_slqp = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
        try:

                             options={'maxiter': 6000, 'ftol': 1e-14})
            if np.isfinite(res_j.fun):
                c_j = np.column_stack((res_j.x[:n], res_j.x[n:2*n]))
                r_j, s_j = solve_lp_fast(c_j)
                if r_j is not None and s_j > best_sum:
                    best_sum = s_j
                    best_c = c_j.copy()
                    best_r = r_j.copy()
        except Exception:
            pass

    # 5. Final strict safety scaling to guarantee numerical validity
    if best_c is not None:
        scale = 1.0
        for i in range(n):
            x, y, r = best_c[i,0], best_c[i,1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
        for i in range(n):
            for j in range(i+1, n):
                d = np.sqrt((best_c[i,0]-best_c[j,0])**2 + (best_c[i,1]-best_c[j,1])**2)
                rs = best_r[i] + best_r[j]
                if rs > 1e-12:
                    scale = min(scale, d/rs)
        best_r *= scale * 0.9999998
        best_sum = float(np.sum(best_r))
    else:
        best_c = np.random.uniform(0.1, 0.9, (n, 2))
        best_r = np.full(n, 0.05)
        best_sum = float(np.sum(best_r))
        
    return best_c, best_r, best_sum
