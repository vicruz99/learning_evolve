import numpy as np
from scipy.optimize import minimize, linprog

def get_pair_indices(n):
    return np.triu_indices(n, k=1)

def solve_lp_radii(centers, n, idx_i, idx_j):
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    A_ub = np.zeros((len(idx_i), n))
    A_ub[np.arange(len(idx_i)), idx_i] = 1.0
    A_ub[np.arange(len(idx_i)), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def obj_func(v, n):
    return -np.sum(v[2*n:])

def con_func(v, n, idx_i, idx_j):
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.empty(4*n + n*(n-1)//2)
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    d2 = dx**2 + dy**2
    rs = r[idx_i] + r[idx_j]
    c[4*n:] = d2 - rs**2
    return c

def make_hex(rows, r0, n):
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing():
    n = 26
    idx_i, idx_j = get_pair_indices(n)
    rng = np.random.default_rng(42)
    
    best_s = -1.0
    best_c = None
    best_r = None
    
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], [4,6,6,6,4],
                [5,7,5,5,4], [6,5,5,6,4], [5,6,6,4,5], [7,6,6,7], [8,6,6,6],
                [5,5,5,5,6], [6,5,5,5,5], [5,5,5,6,5], [5,5,6,6,4], [9,5,6,5], [8,5,5,6]]
                
    configs = []
    for pat in patterns:
        if sum(pat) < n: continue
        pts = make_hex(pat, 0.095, n)
        mn, mx = pts.min(axis=0), pts.max(axis=0)
        span = mx - mn + 1e-9
        pts = (pts - mn) / span * 0.72 + 0.14
        configs.append(pts)
        for _ in range(3):
            p = pts + rng.uniform(-0.025, 0.025, pts.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(15):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    bounds_vars = [(1e-4, 1.0 - 1e-4)] * (2 * n) + [(1e-5, 0.5)] * n
    
    # Phase 1: SLSQP on diverse starts
    for cfg in configs:
        r0, _ = solve_lp_radii(cfg, n, idx_i, idx_j)
        x0 = np.concatenate([cfg.flatten(), r0])
        try:
            res = minimize(obj_func, x0, args=(n,), method='SLSQP', bounds=bounds_vars,
                          constraints={'type': 'ineq', 'fun': con_func, 'args': (n, idx_i, idx_j)},
                          options={'maxiter': 4000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_opt = res.x[:2*n].reshape(n, 2)
                r_opt, s_opt = solve_lp_radii(c_opt, n, idx_i, idx_j)
                if s_opt > best_s:
                    best_s = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            continue
            
    # Phase 2: Coordinate Ascent / Local Search
    if best_c is not None:
        curr_c = best_c.copy()
        curr_r, curr_s = solve_lp_radii(curr_c, n, idx_i, idx_j)
        step = 0.015
        for _ in range(1200):
            i = rng.integers(n)
            old = curr_c[i].copy()
            moved = False
            dirs = [(step,0), (-step,0), (0,step), (0,-step),
                    (step,step), (-step,step), (step,-step), (-step,-step)]
            for dx, dy in dirs:
                curr_c[i] = np.clip(old + [dx, dy], 1e-4, 1.0 - 1e-4)
                r_new, s_new = solve_lp_radii(curr_c, n, idx_i, idx_j)
                if s_new > curr_s + 1e-9:
                    curr_s = s_new
                    curr_r = r_new
                    moved = True
                    break
            if not moved:
                curr_c[i] = old
            step *= 0.9995
        best_c = curr_c
        best_r = curr_r
        best_s = curr_s
        
        # Phase 3: Perturbation + SLSQP polish
        for _ in range(40):
            c_pert = best_c + rng.uniform(-0.004, 0.004, (n, 2))
            c_pert = np.clip(c_pert, 1e-4, 1.0 - 1e-4)
            r_p, _ = solve_lp_radii(c_pert, n, idx_i, idx_j)
            x0 = np.concatenate([c_pert.flatten(), r_p])
            try:
                res = minimize(obj_func, x0, args=(n,), method='SLSQP', bounds=bounds_vars,
                              constraints={'type': 'ineq', 'fun': con_func, 'args': (n, idx_i, idx_j)},
                              options={'maxiter': 2000, 'ftol': 1e-13})
                if np.isfinite(res.fun):
                    c_opt = res.x[:2*n].reshape(n, 2)
                    r_opt, s_opt = solve_lp_radii(c_opt, n, idx_i, idx_j)
                    if s_opt > best_s:
                        best_s = s_opt
                        best_c = c_opt.copy()
                        best_r = r_opt.copy()
            except Exception:
                continue
                
    # Fallback
    if best_c is None:
        best_c = np.tile(np.linspace(0.15, 0.85, 5), 5).reshape(25, 1)
        best_c = np.hstack([best_c, np.repeat(np.linspace(0.15, 0.85, 5), 5).reshape(25, 1)])
        best_c = np.vstack([best_c, [[0.5, 0.5]]])
        best_r, best_s = solve_lp_radii(best_c, n, idx_i, idx_j)
        
    # Final strict safety scaling
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999999
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
