from scipy.optimize import minimize, linprog

def solve_lp(centers, n, m_con, A_ub, pairs_i, pairs_j):
    b = np.empty(m_con)
    x, y = centers[:, 0], centers[:, 1]
    b[:n] = x
    b[n:2*n] = 1.0 - x
    b[2*n:3*n] = y
    b[3*n:4*n] = 1.0 - y
    dx = x[pairs_i] - x[pairs_j]
    dy = y[pairs_i] - y[pairs_j]
    b[4*n:] = np.hypot(dx, dy)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if res.success and np.isfinite(res.fun):
            r = res.x
            s = -res.fun
            try:
                marg = np.asarray(res.ineqlin.marginals)
            except AttributeError:
                try:
                    marg = np.asarray(res.marginals.ineqlin)
                except AttributeError:
                    marg = None
            return r, s, marg
    except Exception:
        pass
    return None, 0.0, None

def obj_and_grad(c_flat, n, m_con, A_ub, pairs_i, pairs_j):

    r, s, marg = solve_lp(c, n, m_con, A_ub, pairs_i, pairs_j)
    if r is None:
        return 1e6, np.ones_like(c_flat)
    grad = np.zeros((n, 2))
    if marg is not None:
        x, y = c[:, 0], c[:, 1]
        for i in range(n):
            grad[i, 0] += marg[i] - marg[n+i]
            grad[i, 1] += marg[2*n+i] - marg[3*n+i]

        dy = y[pairs_i] - y[pairs_j]
        d = np.hypot(dx, dy)
        d = np.where(d < 1e-12, 1e-12, d)
        lam = marg[4*n:]

                fi = pairs_i[k]
                fj = pairs_j[k]

                grad[fi, 0] += fx
                grad[fi, 1] += fy
                grad[fj, 0] -= fx
                grad[fj, 1] -= fy
    return -s, -grad.flatten()

def joint_obj(v, n):
    return -np.sum(v[2*n:])

def joint_cons(v, n):
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    triu = np.triu_indices(n, k=1)
    c = np.concatenate([c, d2[triu] - rs[triu]**2])
    return c
