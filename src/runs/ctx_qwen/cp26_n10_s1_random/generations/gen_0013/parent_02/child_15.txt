1. **LP Formulation**: For fixed centers, finding the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints is a linear program. We solve this efficiently using `scipy.optimize.linprog` with the `highs` solver.
2. **Gradient Computation via Duals**: The gradient of the LP objective w.r.t. center positions is derived from the Karush-Kuhn-Tucker (KKT) conditions. import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_and_grad(centers, N, pair_i, pair_j, A_ub):
    """Solves LP for radii given centers and computes gradient of sum of radii w.r.t centers."""
    M = len(pair_i)
    b = np.empty(4 * N + M)
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(N):
        x, y = centers[i]
        b[i] = x
        b[N + i] = 1.0 - x
        b[2 * N + i] = y
        b[3 * N + i] = 1.0 - y
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for k, (i, j) in enumerate(zip(pair_i, pair_j)):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b[4 * N + k] = np.hypot(dx, dy)
        
    bounds_r = [(0.0, None)] * N
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b, bounds=bounds_r, method='highs')
        if res.success and np.isfinite(res.fun):
            r = res.x
            s = -res.fun
            grad_c = np.zeros((N, 2))
            
            # Extract marginals robustly across scipy versions
            marg = None
            if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
                marg = np.asarray(res.ineqlin.marginals)
            elif hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
                marg = np.asarray(res.marginals.ineqlin)
                
            if marg is not None and len(marg) > 0:
                # Boundary gradients
                for i in range(N):
                    grad_c[i, 0] += marg[i] - marg[N + i]
                    grad_c[i, 1] += marg[2 * N + i] - marg[3 * N + i]
                    
                # Pairwise repulsion gradients
                for k, (i, j) in enumerate(zip(pair_i, pair_j)):
                    lam = marg[4 * N + k]
                    if lam > 1e-9:
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        d = np.hypot(dx, dy)
                        if d < 1e-12: d = 1e-12
                        fx = lam * dx / d
                        fy = lam * dy / d
                        grad_c[i, 0] += fx
                        grad_c[i, 1] += fy
                        grad_c[j, 0] -= fx
                        grad_c[j, 1] -= fy
            return r, s, grad_c.flatten()
    except Exception:
        pass
    return None, 0.0, np.zeros(2 * N)

def obj_and_grad(c_flat, N, pair_i, pair_j, A_ub):
    """Objective function and its gradient for L-BFGS-B."""
    c = c_flat.reshape(N, 2)
    r, s, g = solve_lp_and_grad(c, N, pair_i, pair_j, A_ub)
    if r is None:
        return 1e6, np.zeros_like(c_flat)
    return -s, -g

def joint_obj(v, N):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def joint_cons(v, N, pair_i, pair_j):
    """Inequality constraints >= 0 for valid packing in SLSQP."""
    cx = v[:N]
    cy = v[N:2 * N]
    r = v[2 * N:]
    
    cons = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[pair_i] - cx[pair_j]
    dy = cy[pair_i] - cy[pair_j]
    d2 = dx**2 + dy**2
    rs = r[pair_i] + r[pair_j]
    cons = np.concatenate([cons, d2 - rs**2])
    return cons

def generate_configs(N, rng):
    """Generates diverse initial center configurations."""
    configs = []
    
    # Hexagonal lattice patterns
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [5, 7, 5, 5, 4],
        [6, 5, 5, 6, 4], [5, 6, 6, 4, 5], [7, 6, 6, 7], [8, 6, 6, 6]
    ]
    
    for pat in patterns:
        if sum(pat) < N: continue
        r0 = 0.10
        pts = []
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        pts = np.array(pts[:N])
        configs.append(pts)
        
        # Perturbed versions
        for _ in range(3):
            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    # Corner-biased configurations
    for _ in range(6):
        c = rng.uniform(0.1, 0.9, (N, 2))
        # Push 4 circles towards corners
        corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        for i, cor in enumerate(corners):
            c[i] = cor + rng.uniform(-0.05, 0.05, 2)
        c = np.clip(c, 0.05, 0.95)
        configs.append(c)
        
    # Random dense starts
    for _ in range(15):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return configs

def run_packing():
    N = 26
    pair_i, pair_j = np.triu_indices(N, k=1)
    M = len(pair_i)
    
    # Precompute A_ub for LP
    A_ub = np.zeros((4 * N + M, N))
    for i in range(N):
        A_ub[i, i] = 1.0
        A_ub[N + i, i] = 1.0
        A_ub[2 * N + i, i] = 1.0
        A_ub[3 * N + i, i] = 1.0
    for k, (i, j) in enumerate(zip(pair_i, pair_j)):
        A_ub[4 * N + k, i] = 1.0
        A_ub[4 * N + k, j] = 1.0
        
    rng = np.random.default_rng(42)
    best_s = -1.0
    best_c = None
    best_r = None
    
    bounds_c = [(1e-4, 1.0 - 1e-4)] * (2 * N)
    configs = generate_configs(N, rng)
    
    # Phase 1: Gradient ascent on centers using LP duals from diverse starts
    for cfg in configs:
        c0 = np.clip(cfg, 1e-4, 1.0 - 1e-4)
        try:
            res = minimize(obj_and_grad, c0.flatten(), 
                           args=(N, pair_i, pair_j, A_ub),
                           method='L-BFGS-B', jac=True, bounds=bounds_c,
                           options={'maxiter': 5000, 'ftol': 1e-14, 'gtol': 1e-10})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(N, 2)
                r_opt, s_opt, _ = solve_lp_and_grad(c_opt, N, pair_i, pair_j, A_ub)
                if r_opt is not None and s_opt > best_s:
                    best_s = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            continue
            
    # Phase 2: Perturbation-optimization loop to escape local minima
    if best_c is not None:
        scales = [0.005, 0.002, 0.0005]
        for scale in scales:
            for _ in range(40):
                c_pert = best_c + rng.uniform(-scale, scale, (N, 2))
                c_pert = np.clip(c_pert, 1e-4, 1.0 - 1e-4)
                try:
                    res_p = minimize(obj_and_grad, c_pert.flatten(),
                                     args=(N, pair_i, pair_j, A_ub),
                                     method='L-BFGS-B', jac=True, bounds=bounds_c,
                                     options={'maxiter': 2000, 'ftol': 1e-13})
                    if np.isfinite(res_p.fun):
                        c_p = res_p.x.reshape(N, 2)
                        r_p, s_p, _ = solve_lp_and_grad(c_p, N, pair_i, pair_j, A_ub)
                        if r_p is not None and s_p > best_s:
                            best_s = s_p
                            best_c = c_p.copy()
                            best_r = r_p.copy()
                except Exception:
                    continue
                    
    # Phase 3: Joint SLSQP polish for precise constraint handling
    if best_c is not None:
        v0 = np.concatenate([best_c[:, 0], best_c[:, 1], best_r * 0.995])
        bounds_slqp = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
        try:
            res_j = minimize(joint_obj, v0, args=(N,),
                             method='SLSQP', bounds=bounds_slqp,
                             constraints={'type': 'ineq', 'fun': joint_cons, 'args': (N, pair_i, pair_j)},
                             options={'maxiter': 5000, 'ftol': 1e-14})
            if np.isfinite(res_j.fun):
                c_j = np.column_stack((res_j.x[:N], res_j.x[N:2 * N]))
                r_j, s_j, _ = solve_lp_and_grad(c_j, N, pair_i, pair_j, A_ub)
                if r_j is not None and s_j > best_s:
                    best_s = s_j
                    best_c = c_j.copy()
                    best_r = r_j.copy()
        except Exception:
            pass

    # Final strict safety scaling to guarantee numerical validity
    if best_c is not None:
        scale = 1.0
        for i in range(N):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            if r > 1e-12:
                scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
                
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                rs = best_r[i] + best_r[j]
                if rs > 1e-12:
                    scale = min(scale, d / rs)
                    
        best_r *= scale * 0.9999999
        best_s = float(np.sum(best_r))
    else:
        # Fallback
        best_c = np.random.uniform(0.1, 0.9, (N, 2))
        best_r = np.full(N, 0.05)
        best_s = float(np.sum(best_r))
        
    return best_c, best_r, best_s
