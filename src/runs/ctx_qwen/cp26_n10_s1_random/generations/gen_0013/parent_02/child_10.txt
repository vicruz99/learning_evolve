import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_and_grad(centers, n, idx_i, idx_j):
    """Solves LP for radii given centers and computes gradient of sum of radii w.r.t centers."""
    # Explicit boundary constraints for accurate duals/gradients
    lims_x = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    lims_y = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    lims = np.minimum(lims_x, lims_y)
    lims = np.maximum(lims, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    
    m = len(idx_i)
    total_con = 4 * n + m
    A = np.zeros((total_con, n))
    b = np.zeros(total_con)
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    b[:n] = centers[:, 0]
    b[n:2*n] = 1.0 - centers[:, 0]
    b[2*n:3*n] = centers[:, 1]
    b[3*n:4*n] = 1.0 - centers[:, 1]
    A[:4*n, np.arange(n)] = 1.0
    
    # Pair constraints: r_i + r_j <= dist(i,j)
    A[4*n:, idx_i] = 1.0
    A[4*n:, idx_j] = 1.0
    b[4*n:] = dists
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, 0.0, None
            
        r = res.x
        s = -res.fun
        
        grad = np.zeros((n, 2))
        marg = None
        try:
            marg = np.asarray(res.marginals.ineqlin)
        except AttributeError:
            try:
                marg = np.asarray(res.ineqlin.marginals)
            except:
                pass
                
        if marg is not None and len(marg) > 0:
            # Boundary gradients
            for i in range(n):
                grad[i, 0] += marg[i] - marg[n + i]
                grad[i, 1] += marg[2*n + i] - marg[3*n + i]
                
            # Pairwise repulsion gradients
            for k in range(m):
                lam = marg[4*n + k]
                if lam > 1e-9:
                    i, j = idx_i[k], idx_j[k]
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = np.hypot(dx, dy)
                    if d < 1e-12: d = 1e-12
                    fx = lam * dx / d
                    fy = lam * dy / d
                    grad[i, 0] += fx
                    grad[i, 1] += fy
                    grad[j, 0] -= fx
                    grad[j, 1] -= fy
        return r, s, grad.flatten()
    except Exception:
        return None, 0.0, None

def objective(v, n, idx_i, idx_j):
    """Objective function for L-BFGS-B: negative sum of radii and its gradient."""
    r, s, g = solve_lp_and_grad(v.reshape(n, 2), n, idx_i, idx_j)
    if r is None:
        return 1e6, np.zeros_like(v)
    return -s, -g

def run_packing():

    best_r = None

                [5,7,5,5,4], [6,5,5,6,4], [5,6,6,4,5], [7,6,6,7], [8,6,6,6],
                [5,5,5,5,6], [6,5,5,5,5], [5,5,5,6,5], [5,5,6,6,4], [5,6,4,6,5]]

        r0 = 0.10

        configs.append(pts)

            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
