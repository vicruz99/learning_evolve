import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Maximum radius allowed by boundaries for each circle
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    wall = np.maximum(wall, 1e-12)
    
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # LP formulation: min -sum(r) s.t. r_i + r_j <= dists[i,j] and 0 <= r_i <= wall[i]
    c = -np.ones(n)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = [(0.0, w) for w in wall]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
        
    # Fallback: geometrically valid radii if LP fails unexpectedly
    min_pair = dists.min(axis=1)
    min_clr = np.minimum(wall, min_pair / 2.0)
    return np.maximum(min_clr * 0.995, 1e-9)

def get_hex_config(n, row_counts, r0):
    """Generates a hexagonal lattice initialization with specified row counts."""
    pts = []
    y = r0
    for ri, cnt in enumerate(row_counts):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
        if len(pts) >= n:
            break
    return np.array(pts[:n])

def normalize_and_fit(pts):
    """Normalizes points to fit comfortably inside [0.02, 0.98]."""
    mn = pts.min(axis=0)
    mx = pts.max(axis=0)
    span = mx - mn
    span[span < 1e-9] = 1.0
    pts = (pts - mn) / span * 0.85 + 0.075
    return np.clip(pts, 0.02, 0.98)

def get_constraints(vars_array, n):
    """Computes inequality constraints >= 0 for valid packing in SLSQP."""
    cx = vars_array[:n]
    cy = vars_array[n:2 * n]
    r = vars_array[2 * n:]
    
    cons = []
    # Boundary constraints
    cons.extend(cx - r)
    cons.extend(1.0 - cx - r)
    cons.extend(cy - r)
    cons.extend(1.0 - cy - r)
    
    # Pairwise non-overlap constraints
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    
    d2 = cx_m**2 + cy_m**2
    rs2 = r_m**2
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cons.extend((d2 - rs2)[mask])
    return np.array(cons)
