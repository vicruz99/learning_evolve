import numpy as np
from scipy.optimize import minimize, linprog

def get_lp_sum_and_radii(centers, n, idx_i, idx_j, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    wall = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]), 
                      np.minimum(centers[:,1], 1.0 - centers[:,1]))
    wall = np.maximum(wall, 1e-12)
    
    dx = centers[idx_i, 0] - centers[idx_j, 0]
    dy = centers[idx_i, 1] - centers[idx_j, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = [(0.0, w) for w in wall]
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
        
    # Fallback: geometric minimum clearance
    dists = np.hypot(dx, dy)
    min_d = np.minimum(np.min(dists) / 2.0, np.min(wall))
    val = n * max(0.0, min_d - 1e-7)
    return val, np.full(n, max(0.0, min_d - 1e-7))

def objective(centers_flat, n, idx_i, idx_j, A_ub):
    """Objective for center optimization: minimize negative LP sum."""
    centers = centers_flat.reshape(n, 2)
    s, _ = get_lp_sum_and_radii(centers, n, idx_i, idx_j, A_ub)
    return -s

def generate_hex_config(n, row_counts, r0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    for ri, cnt in enumerate(row_counts):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        if len(pts) >= n: break
    return np.array(pts[:n])

    n_pairs = len(idx_i)

        cfg_n = (cfg - mn) / span * 0.8 + 0.1
        configs.append(cfg_n)

            mn, mx = cfg_r.min(axis=0), cfg_r.max(axis=0)

            configs.append(cfg_n2)

    return best_c, best_r, best_s
