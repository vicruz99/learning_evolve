1. **Diverse Initialization**: Generate multiple starting configurations using hexagonal lattice patterns with various row distributions, rotated versions of these lattices to align optimally with the square boundaries, force-directed layouts to naturally spread circles, and random dense starts. This ensures coverage of different topological basins.

2. **Joint SLSQP Optimization**: For each initial configuration, compute LP-optimal radii and use them to initialize a joint SLSQP optimization over both centers and radii. This phase respects all non-overlap and boundary constraints explicitly, refining the geometry significantly.

3. **LP Oracle Evaluation**: After joint optimization, solve a Linear Program exactly to determine the maximum possible sum of radii for the optimized centers. This decouples the radius optimization from the center optimization, providing an exact objective value.

4. **Coordinate Ascent on Centers**: Perform iterative local search on the center coordinates. In each step, a random circle is moved within a local neighborhood, and the new sum of radii is evaluated using the LP oracle. This effectively climbs the non-smooth objective landscape of the center-only optimization problem.

5. **Multi-Circle Perturbation & Refinement**: Randomly perturb groups of 2-5 circles simultaneously to break symmetries that single-circle moves cannot escape, followed by rapid local refinement.

6. **Strict Safety Scaling**: Finally, compute the exact scaling factor required to satisfy all constraints within the validator's `1e-12` tolerance, ensuring the solution is strictly valid without sacrificing meaningful area.

import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-12)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx = np.triu_indices(n, k=1)
    m = len(idx[0])
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx[0]] = 1.0
    A_ub[np.arange(m), idx[1]] = 1.0
    b_ub = dists[idx]
    bounds = [(0.0, l) for l in lims]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-6)

def hex_init(row_counts, r0, n):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    for ri, cnt in enumerate(row_counts):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(pts[:n])

def force_layout_init(n, seed):
    """Generates a dense layout via iterative force expansion."""
    rng = np.random.default_rng(seed)
    centers = rng.uniform(0.2, 0.8, (n, 2))
    radii = np.full(n, 0.01)
    dt = 0.01
    for _ in range(1500):
        radii += 0.00005
        forces = np.zeros_like(centers)
        for d in range(2):
            v_l = np.maximum(0.0, radii - centers[:, d])
            v_r = np.maximum(0.0, centers[:, d] + radii - 1.0)
            forces[:, d] += (v_r - v_l) * 100.0
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dists, 0.0)
        req = radii[:, None] + radii[None, :]
        overlap = np.maximum(0.0, req - dists)
        dir_u = diff / (dists[:, :, None] + 1e-12)
        f = overlap[:, :, None] * dir_u * 50.0
        forces += np.sum(f, axis=1)
        centers += forces * dt
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    return centers

def obj_j(v, n):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def con_j(v, n):
    """Inequality constraints >= 0 for valid joint packing."""
    cx, cy, r = v[:n], v[n:2*n], v[2*n:]
    bc = np.concatenate([cx-r, 1-cx-r, cy-r, 1-cy-r])
    idx = np.triu_indices(n, k=1)
    dx, dy = cx[idx[0]]-cx[idx[1]], cy[idx[0]]-cy[idx[1]]
    d2 = dx**2 + dy**2
    rs = r[idx[0]] + r[idx[1]]
    return np.concatenate([bc, d2 - rs**2])

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    configs = []
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [5,6,4,6,5], [7,6,6,7], [8,6,6,6],
        [9,5,6,6], [5,5,5,5,6], [6,5,5,5,5]
    ]
    for pat in patterns:
        if sum(pat) < n: continue
        pts = hex_init(pat, 0.1, n)
        mn, mx = pts.min(axis=0), pts.max(axis=0)
        span = mx - mn
        span[span < 1e-9] = 1.0
        pts_n = (pts - mn) / span * 0.8 + 0.1
        configs.append(pts_n)
        for _ in range(2):
            configs.append(np.clip(pts_n + rng.uniform(-0.02, 0.02, pts_n.shape), 0.05, 0.95))
        for ang in [0.05, -0.05, 0.15, -0.15]:
            c, s = np.cos(ang), np.sin(ang)
            rot = np.array([[c, -s], [s, c]])
            rp = pts_n @ rot
            rpn = (rp - rp.min(axis=0)) / (rp.max(axis=0) - rp.min(axis=0)) * 0.8 + 0.1
            configs.append(rpn)
            
    for s in range(5):
        configs.append(force_layout_init(n, s))
        
    for _ in range(8):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = -1.0
    best_c = None
    best_r = None
    
    bounds_v = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    cons_d = {'type': 'ineq', 'fun': con_j, 'args': (n,)}
    
    for cfg in configs:
        r0 = solve_lp(cfg)
        x0 = np.concatenate([cfg[:,0], cfg[:,1], r0])
        try:
            res = minimize(obj_j, x0, args=(n,), method='SLSQP', bounds=bounds_v, constraints=cons_d,
                           options={'maxiter': 12000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                cx, cy = res.x[:n], res.x[n:2*n]
                r = np.maximum(res.x[2*n:], 1e-9)
                co = np.column_stack((cx, cy))
                rl = solve_lp(co)
                s = np.sum(rl)
                if s > best_sum:
                    best_sum = s
                    best_c = co.copy()
                    best_r = rl.copy()
        except Exception:
            pass
            
    if best_c is not None:
        cc = best_c.copy()
        cs = best_sum
        step = 0.025
        for _ in range(2):
            for _ in range(2000):
                idx = rng.integers(n)
                old = cc[idx].copy()
                bm, bv = old, cs
                for _ in range(5):
                    tr = old + rng.uniform(-step, step, 2)
                    tr = np.clip(tr, 0.01, 0.99)
                    cc[idx] = tr
                    rl = solve_lp(cc)
                    st = np.sum(rl)
                    if st > bv:
                        bv = st
                        bm = tr.copy()
                cc[idx] = bm
                cs = bv
                if cs > best_sum + 1e-9:
                    best_sum = cs
                    best_c = cc.copy()
                    best_r = solve_lp(best_c).copy()
                step *= 0.999
                step = max(step, 0.005)
            step = max(step, 0.015)
            
        for _ in range(40):
            cp = best_c.copy()
            k = rng.integers(2, 6)
            ids = rng.choice(n, k, replace=False)
            cp[ids] += rng.uniform(-0.02, 0.02, (k, 2))
            cp = np.clip(cp, 0.05, 0.95)
            rl = solve_lp(cp)
            st = np.sum(rl)
            if st > best_sum:
                best_sum = st
                best_c = cp.copy()
                best_r = rl.copy()
                for _ in range(600):
                    i = rng.integers(n)
                    old = best_c[i].copy()
                    best_c[i] += rng.uniform(-0.012, 0.012, 2)
                    best_c[i] = np.clip(best_c[i], 0.01, 0.99)
                    rt = solve_lp(best_c)
                    st = np.sum(rt)
                    if st > best_sum:
                        best_sum = st
                        best_r = rt.copy()
                    else:
                        best_c[i] = old
                        break
                        
    if best_c is None:
        best_c = configs[0]
        best_r = solve_lp(best_c)
        best_sum = np.sum(best_r)
        
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
    best_r *= scale * 0.9999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
