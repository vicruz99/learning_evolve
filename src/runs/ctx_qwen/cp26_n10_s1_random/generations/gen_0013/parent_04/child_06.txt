import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, A_ub, triu_i, triu_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    wall_limits = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    wall_limits = np.maximum(wall_limits, 1e-12)
    
    dx = centers[triu_i, 0] - centers[triu_j, 0]
    dy = centers[triu_i, 1] - centers[triu_j, 1]
    pair_dists = np.hypot(dx, dy)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lim) for lim in wall_limits]
    b_ub = pair_dists
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-7)

def get_A_ub_and_indices(n):
    """Precomputes constant LP constraint matrix and pair indices."""
    triu_i, triu_j = np.triu_indices(n, k=1)
    m = len(triu_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), triu_i] = 1.0
    A_ub[np.arange(m), triu_j] = 1.0
    return A_ub, triu_i, triu_j

def joint_objective(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def joint_constraints(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid joint packing."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    cons = []
    cons.append(cx - r)
    cons.append(1.0 - cx - r)
    cons.append(cy - r)
    cons.append(1.0 - cy - r)
    
    dx = cx[triu_i] - cx[triu_j]
    dy = cy[triu_i] - cy[triu_j]
    d2 = dx**2 + dy**2
    rs = r[triu_i] + r[triu_j]
    cons.append(d2 - rs**2)
    
    return np.concatenate(cons)

def coordinate_ascent(centers, n, A_ub, triu_i, triu_j, rng, steps=3000):
    """Local search on centers using LP evaluation for precise objective."""
    best_c = centers.copy()
    curr_r = solve_lp_radii(best_c, A_ub, triu_i, triu_j)
    best_s = np.sum(curr_r)
    
    step = 0.02
    for _ in range(steps):
        idx = rng.integers(n)
        old = best_c[idx].copy()
        best_move = old
        best_val = best_s
        
        # Local sampling
        for _ in range(10):
            trial = old + rng.uniform(-step, step, 2)
            trial = np.clip(trial, 1e-5, 1.0 - 1e-5)
            best_c[idx] = trial
            r_try = solve_lp_radii(best_c, A_ub, triu_i, triu_j)
            s_try = np.sum(r_try)
            if s_try > best_val:
                best_val = s_try
                best_move = trial.copy()
                
        best_c[idx] = best_move
        curr_r = solve_lp_radii(best_c, A_ub, triu_i, triu_j)
        curr_s = np.sum(curr_r)
        
        if curr_s > best_s:
            best_s = curr_s
            
        step *= 0.9985
        step = max(step, 5e-4)
        
    return best_c, solve_lp_radii(best_c, A_ub, triu_i, triu_j), best_s

def generate_seeds(n, rng):
    """Generates diverse hexagonal lattice and random configurations."""
    seeds = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 5, 5, 5, 5]
    ]
    
    for pat in patterns:
        pts = []
        y = 0.09
        row = 0
        for cnt in pat:
            shift = 0.09 if row % 2 == 1 else 0.0
            x = 0.09 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 0.18
            y += 0.155
            row += 1
            if len(pts) >= n: break
        pts = np.array(pts[:n])
        
        mn, mx = pts.min(axis=0), pts.max(axis=0)
        span = mx - mn
        span[span < 1e-9] = 1.0
        pts = (pts - mn) / span * 0.8 + 0.1
        seeds.append(pts)
        
        for _ in range(3):
            p = pts + rng.uniform(-0.025, 0.025, pts.shape)
            seeds.append(np.clip(p, 0.05, 0.95))
            
    for _ in range(6):
        seeds.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return seeds

            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_r *= scale * 0.9999999
    final_s = float(np.sum(best_r))
    
    return best_c, best_r, final_s
