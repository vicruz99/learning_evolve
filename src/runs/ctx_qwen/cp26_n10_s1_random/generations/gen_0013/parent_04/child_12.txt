import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Distance to nearest boundary for each circle
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    wall_dists = np.maximum(wall_dists, 1e-9)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # LP formulation: min -sum(r) s.t. r_i + r_j <= dists[i,j] and 0 <= r_i <= wall_dists[i]
    c = -np.ones(n)
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = [(0.0, w) for w in wall_dists]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x
    except Exception:
        pass
    return wall_dists * 0.85

def hill_climb_lp(centers, n, rng, steps=4000):
    """Coordinate ascent on centers evaluated by exact LP radii solution."""
    best_centers = centers.copy()
    best_radii = solve_radii_lp(best_centers)
    best_sum = np.sum(best_radii)
    
    step_size = 0.025
    no_improve = 0
    
    for _ in range(steps):
        i = rng.integers(n)
        old = best_centers[i].copy()
        
        # Try multiple moves for the selected circle
        best_trial = old
        best_trial_sum = best_sum
        
        for _ in range(4):
            trial = old + rng.uniform(-step_size, step_size, 2)
            trial = np.clip(trial, 1e-5, 1.0 - 1e-5)
            best_centers[i] = trial
            
            new_radii = solve_radii_lp(best_centers)
            new_sum = np.sum(new_radii)
            
            if new_sum > best_trial_sum:
                best_trial_sum = new_sum
                best_trial = trial.copy()
                
        best_centers[i] = best_trial
        
        if best_trial_sum > best_sum + 1e-9:
            best_sum = best_trial_sum
            best_radii = solve_radii_lp(best_centers)
            step_size *= 1.001
            no_improve = 0
        else:
            best_centers[i] = old
            step_size *= 0.997
            no_improve += 1
            
        step_size = np.clip(step_size, 1e-6, 0.04)
        if no_improve > 600:
            break
            
    return best_centers, best_radii, best_sum

def multi_jitter_opt(centers, n, rng, rounds=50):
    """Escape local minima by perturbing subsets of circles and re-optimizing."""
    best_centers = centers.copy()
    best_radii = solve_radii_lp(best_centers)
    best_sum = np.sum(best_radii)
    
    for _ in range(rounds):
        k = rng.integers(2, 6)
        idxs = rng.choice(n, size=k, replace=False)
        old_positions = best_centers[idxs].copy()
        
        best_centers[idxs] += rng.uniform(-0.015, 0.015, (k, 2))
        best_centers[idxs] = np.clip(best_centers[idxs], 1e-5, 1.0 - 1e-5)
        
        new_radii = solve_radii_lp(best_centers)
        new_sum = np.sum(new_radii)
        
        if new_sum > best_sum + 1e-9:
            best_sum = new_sum
            best_radii = new_radii
            # Quick local refinement from new state
            best_centers, best_radii, best_sum = hill_climb_lp(best_centers, n, rng, steps=500)
        else:
            best_centers[idxs] = old_positions
            
    return best_centers, best_radii, best_sum

def joint_slsqp_polish(centers, n):
    """Jointly optimize centers and radii using SLSQP for precise constraint handling."""
    r0 = solve_radii_lp(centers)
    x0 = np.concatenate([centers.flatten(), r0])
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-8, 0.5)] * n
    
    def obj(v):
        return -np.sum(v[2*n:])
        
    def con(v):
        cx = v[:n]
        cy = v[n:2*n]
        r = v[2*n:]
        # Boundary constraints
        c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
        # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
        cx_m = cx[:, None] - cx[None, :]
        cy_m = cy[:, None] - cy[None, :]
        r_m = r[:, None] + r[None, :]
        d2 = cx_m**2 + cy_m**2
        rs2 = r_m**2
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        c = np.concatenate([c, (d2 - rs2)[mask]])
        return c
        
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': con},
                       options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
        if np.isfinite(res.fun):
            cx = res.x[:n]
            cy = res.x[n:2*n]
            c_opt = np.column_stack((cx, cy))
            # Re-solve LP on optimized centers to get exact maximal radii
            r_lp = solve_radii_lp(c_opt)
            return c_opt, r_lp, np.sum(r_lp)
    except Exception:
        pass
    return centers, r0, np.sum(r0)

def generate_hex_config(n, row_counts, r0):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r0
    for ri, cnt in enumerate(row_counts):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        if len(pts) >= n:
            break
            
    return np.array(pts[:n])
