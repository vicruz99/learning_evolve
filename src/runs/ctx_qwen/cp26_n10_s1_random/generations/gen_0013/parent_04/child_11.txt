import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    walls = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    walls = np.maximum(walls, 1e-9)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dists[i, j]
            idx += 1
            
    bounds = [(0.0, w) for w in walls]
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-7)

def obj_slsqp(v, n):
    return -np.sum(v[2 * n:])

def get_constraints_slsqp(v, n):
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    c = []
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    cx_m = cx[:, None] - cx[None, :]
    cy_m = cy[:, None] - cy[None, :]
    r_m = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.extend((cx_m**2 + cy_m**2 - r_m**2)[mask])
    return np.array(c)

def obj_nm(v, n):
    c = v.reshape(n, 2)
    c = np.clip(c, 1e-5, 1.0 - 1e-5)
    r = solve_radii_lp(c)
    return -np.sum(r)
