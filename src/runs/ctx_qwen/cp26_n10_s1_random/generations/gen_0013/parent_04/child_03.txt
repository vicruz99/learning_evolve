import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers, n, A_ub, triu_i, triu_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Distance to nearest boundary
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    wall = np.maximum(wall, 1e-12)
    
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = dists[triu_i, triu_j]
    bounds = ((0.0,) * n, wall)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    # Fallback
    return wall * 0.1, np.sum(wall * 0.1)

def generate_hex_configs(n, rng):
    """Generates diverse hexagonal lattice initial configurations."""
    configs = []
    row_pats = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4]
    ]
    
    for pat in row_pats:
        if sum(pat) < n:
            continue
            
        r0 = 0.095
        pts = []
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n:
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            
        pts = np.array(pts[:n])
        
        # Normalize to [0.05, 0.95]
        mn, mx = pts.min(axis=0), pts.max(axis=0)
        span = mx - mn
        span[span < 1e-9] = 1.0
        norm_pts = (pts - mn) / span * 0.8 + 0.1
        configs.append(norm_pts)
        
        # Rotated variants to break symmetries
        for ang in [0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.25]:
            c, s = np.cos(ang), np.sin(ang)
            rot = np.array([[c, -s], [s, c]])
            center = pts.mean(axis=0)
            rot_pts = (pts - center) @ rot.T + center
            mn, mx = rot_pts.min(axis=0), rot_pts.max(axis=0)
            span = mx - mn
            span[span < 1e-9] = 1.0
            norm_rot = (rot_pts - mn) / span * 0.8 + 0.1
            configs.append(norm_rot)
            
    # Random dense starts
    for _ in range(6):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    # Precompute LP structure
    triu_i, triu_j = np.triu_indices(n, k=1)
    m = len(triu_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), triu_i] = 1.0
    A_ub[np.arange(m), triu_j] = 1.0
    
    # Phase 1: Initialization & Best Selection
    configs = generate_hex_configs(n, rng)
    best_centers, best_radii, best_sum = None, None, -1.0
    
    for cfg in configs:
        r, s = solve_lp(cfg, n, A_ub, triu_i, triu_j)
        if s > best_sum:
            best_sum = s
            best_centers = cfg.copy()
            best_radii = r.copy()
            
    if best_centers is None:
        best_centers = rng.uniform(0.2, 0.8, (n, 2))
        best_radii, best_sum = solve_lp(best_centers, n, A_ub, triu_i, triu_j)

    # Phase 2: Coordinate Ascent with Adaptive Jitter
    # Optimizes one circle at a time to maximize LP sum. Highly effective for packing.
    for epoch in range(200):
        # Jitter to escape local minima (decaying magnitude)
        if epoch % 25 == 0 and epoch > 0:
            jitter_mag = 0.018 * np.exp(-epoch / 80.0)
            best_centers += rng.uniform(-jitter_mag, jitter_mag, (n, 2))
            best_centers = np.clip(best_centers, 0.01, 0.99)
            best_radii, best_sum = solve_lp(best_centers, n, A_ub, triu_i, triu_j)
            
        # Randomize processing order to avoid bias
        order = rng.permutation(n)
        for k in order:
            old_pos = best_centers[k].copy()
            
            # 2D objective: minimize negative sum of radii
            def obj_2d(pos):
                best_centers[k] = np.clip(pos, 0.01, 0.99)
                _, s = solve_lp(best_centers, n, A_ub, triu_i, triu_j)
                return -s
                
            try:
                res = minimize(obj_2d, old_pos, method='Nelder-Mead',
                              options={'xatol': 1e-7, 'fatol': 1e-9, 'maxiter': 100, 'adaptive': True})
                
                if -res.fun > best_sum + 1e-7:
                    best_sum = -res.fun
                    best_centers[k] = res.x
                    best_radii, best_sum = solve_lp(best_centers, n, A_ub, triu_i, triu_j)
                else:
                    best_centers[k] = old_pos
            except Exception:
                best_centers[k] = old_pos

    # Phase 3: Final LP Solve & Strict Safety Scaling
    best_radii, best_sum = solve_lp(best_centers, n, A_ub, triu_i, triu_j)
    
    # Compute exact maximum scaling factor to satisfy 1e-12 validator tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
