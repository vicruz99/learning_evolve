import numpy as np
from scipy.optimize import minimize, linprog

def setup_lp_matrices(n):
    num_pairs = n * (n - 1) // 2
    A = np.zeros((num_pairs, n))
    idx_i, idx_j = np.triu_indices(n, k=1)
    A[np.arange(num_pairs), idx_i] = 1.0
    A[np.arange(num_pairs), idx_j] = 1.0
    return A, idx_i, idx_j

def solve_radii_lp(centers, A_ub, idx_i, idx_j):
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    wall_lim = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    wall_lim = np.maximum(wall_lim, 1e-12)
    
    dists = np.hypot(centers[idx_i, 0] - centers[idx_j, 0], 
                     centers[idx_i, 1] - centers[idx_j, 1])
    
    bounds = [(0.0, w) for w in wall_lim]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def generate_configs(n, rng):
    configs = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4]
    ]
    
    for pat in patterns:
        if sum(pat) < n: continue
        r0 = 0.095
        pts = []
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
        pts = np.array(pts[:n])
        
        mn, mx = pts.min(axis=0), pts.max(axis=0)
        span = mx - mn
        span[span < 1e-9] = 1.0
        pts = (pts - mn) / span * 0.9 + 0.05
        configs.append(pts)
        
        for angle in [0.1, -0.1, 0.2, -0.2, 0.05]:
            c, s = np.cos(angle), np.sin(angle)
            rot = np.array([[c, -s], [s, c]])
            p_rot = pts @ rot
            mn, mx = p_rot.min(axis=0), p_rot.max(axis=0)
            span = mx - mn
            span[span < 1e-9] = 1.0
            p_rot = (p_rot - mn) / span * 0.85 + 0.075
            configs.append(p_rot)
            
        for _ in range(3):
            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            configs.append(np.clip(p, 0.05, 0.95))

    for _ in range(5):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return configs

def coordinate_ascent(centers, A_ub, idx_i, idx_j, n, rng, passes=3):
    best_centers = centers.copy()
    curr_radii, curr_sum = solve_radii_lp(best_centers, A_ub, idx_i, idx_j)
    
    for p in range(passes):
        improved = False
        order = rng.permutation(n)
        for i in order:
            old_pos = best_centers[i].copy()
            
            def obj_2d(pos):
                best_centers[i] = np.clip(pos, 1e-4, 1.0 - 1e-4)
                _, s = solve_radii_lp(best_centers, A_ub, idx_i, idx_j)
                return -s
                
            try:
                res = minimize(obj_2d, old_pos, method='Nelder-Mead', 
                               options={'xatol': 1e-7, 'fatol': 1e-9, 'maxiter': 60})
                if res.fun < obj_2d(old_pos) - 1e-10:
                    best_centers[i] = np.clip(res.x, 1e-4, 1.0 - 1e-4)
                    _, new_sum = solve_radii_lp(best_centers, A_ub, idx_i, idx_j)
                    if new_sum > curr_sum + 1e-12:
                        curr_sum = new_sum
                        curr_radii = solve_radii_lp(best_centers, A_ub, idx_i, idx_j)[0]
                        improved = True
                    else:
                        best_centers[i] = old_pos
                else:
                    best_centers[i] = old_pos
            except Exception:
                best_centers[i] = old_pos
                
        if not improved:
            break
            
        if p % 2 == 1:
            idx = rng.integers(n)
            best_centers[idx] += rng.uniform(-0.01, 0.01, 2)
            best_centers[idx] = np.clip(best_centers[idx], 0.05, 0.95)
            _, new_sum = solve_radii_lp(best_centers, A_ub, idx_i, idx_j)
            if new_sum > curr_sum:
                curr_sum = new_sum
                curr_radii = solve_radii_lp(best_centers, A_ub, idx_i, idx_j)[0]
                
    return best_centers, curr_radii, curr_sum

def run_packing():

            best_centers = c_opt

            return -np.sum(v[2*n:])

            dy = cy[:, None] - cy[None, :]
            d2 = dx**2 + dy**2

            return c

        try:
            res = minimize(joint_obj, v0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': joint_cons},

            if np.isfinite(res.fun):

    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
