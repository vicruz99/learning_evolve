import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    wall = np.maximum(wall, 1e-12)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    c = -np.ones(n)
    idx_i, idx_j = np.triu_indices(n, k=1)
    A_ub = np.zeros((len(idx_i), n))
    A_ub[np.arange(len(idx_i)), idx_i] = 1.0
    A_ub[np.arange(len(idx_i)), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    bounds = [(0.0, w) for w in wall]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x
    except Exception:
        pass
        
    # Geometric fallback if LP fails
    min_d = np.min(dists, axis=1) / 2.0
    return np.minimum(wall, min_d) * 0.95

def generate_configs(n):
    """Generates diverse high-quality initial configurations."""
    configs = []
    rng = np.random.default_rng(42)
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 7, 5, 5, 4]
    ]
    for pat in patterns:
        if sum(pat) < n: continue
        pts = []
        y = 0.095
        for ri, cnt in enumerate(pat):
            shift = 0.095 if ri % 2 == 1 else 0.0
            x = 0.095 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 0.19
            y += 0.164
        pts = np.array(pts[:n])
        mn, mx = pts.min(0), pts.max(0)
        span = mx - mn
        span[span < 1e-9] = 1.0
        pts_norm = (pts - mn) / span * 0.9 + 0.05
        configs.append(pts_norm)
        for _ in range(3):
            p = pts_norm + rng.uniform(-0.02, 0.02, (n, 2))
            configs.append(np.clip(p, 0.05, 0.95))
            
    # Corner/Edge focused layout
    cfg_ce = np.array([
        [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9],
        [0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5],
        [0.3, 0.3], [0.7, 0.3], [0.3, 0.7], [0.7, 0.7],
        [0.5, 0.5], [0.25, 0.5], [0.75, 0.5], [0.5, 0.25], [0.5, 0.75]
    ])
    remaining = n - len(cfg_ce)
    cfg_ce = np.vstack([cfg_ce, rng.uniform(0.2, 0.8, (remaining, 2))])
    configs.append(cfg_ce)
    
    for _ in range(5):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return configs

def joint_obj(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def joint_cons(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx, cy, r = v[:n], v[n:2*n], v[2*n:]
    cons = []
    cons.append(cx - r)
    cons.append(1.0 - cx - r)
    cons.append(cy - r)
    cons.append(1.0 - cy - r)
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    d2 = dx**2 + dy**2
    rs = r[idx_i] + r[idx_j]
    cons.append(d2 - rs**2)
    return np.concatenate(cons)

def coordinate_ascent(centers, n, rng, iters=1200):
    """Local search on centers using LP evaluation for precise objective."""
    best_c = centers.copy()
    best_r = solve_lp_radii(best_c)
    best_s = np.sum(best_r)
    
    step = 0.025
    for _ in range(iters):
        i = rng.integers(n)
        old = best_c[i].copy()
        best_move = old
        cur_s = best_s
        
        for _ in range(10):
            trial = old + rng.uniform(-step, step, 2)
            trial = np.clip(trial, 1e-4, 1.0 - 1e-4)
            best_c[i] = trial
            r_t = solve_lp_radii(best_c)
            s_t = np.sum(r_t)
            if s_t > cur_s:
                cur_s = s_t
                best_move = trial.copy()
                
        best_c[i] = best_move
        best_r = solve_lp_radii(best_c)
        best_s = np.sum(best_r)
        step *= 0.9985
        step = max(step, 1e-5)
        
    return best_c, best_r, best_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(123)
    
    configs = generate_configs(n)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Coordinate Ascent on diverse starts
    for cfg in configs:
        c_curr, r_curr, s_curr = coordinate_ascent(cfg, n, rng, iters=1000)
        if s_curr > best_sum:
            best_sum = s_curr
            best_centers = c_curr.copy()
            best_radii = r_curr.copy()
            
    # Phase 2: SLSQP Joint Polish
    if best_centers is not None:
        bounds = [(0.0, 1.0)] * (2 * n) + [(1e-7, 0.5)] * n
        x0 = np.concatenate([best_centers[:, 0], best_centers[:, 1], best_radii * 0.995])
        
        for _ in range(3):
            x0_p = x0 + rng.uniform(-0.004, 0.004, 3*n)
            x0_p = np.clip(x0_p, [0.0]*2*n + [1e-7]*n, [1.0]*2*n + [0.5]*n)
            try:
                res = minimize(joint_obj, x0_p, args=(n,), method='SLSQP',
                              bounds=bounds,
                              constraints={'type': 'ineq', 'fun': joint_cons, 'args': (n,)},
                              options={'maxiter': 6000, 'ftol': 1e-14})
                if np.isfinite(res.fun):
                    cx = res.x[:n]
                    cy = res.x[n:2*n]
                    r = np.maximum(res.x[2*n:], 1e-9)
                    centers_s = np.column_stack((cx, cy))
                    radii_s = r
                    
                    # Strict feasibility check
                    valid = True
                    if np.any(centers_s[:, 0] < radii_s - 1e-12) or np.any(centers_s[:, 0] > 1.0 - radii_s + 1e-12) or \
                       np.any(centers_s[:, 1] < radii_s - 1e-12) or np.any(centers_s[:, 1] > 1.0 - radii_s + 1e-12):
                        valid = False
                    if valid:
                        diff = centers_s[:, None, :] - centers_s[None, :, :]
                        dists = np.sqrt(np.sum(diff**2, axis=2))
                        np.fill_diagonal(dists, np.inf)
                        if np.any(dists < radii_s[:, None] + radii_s[None, :] - 1e-12):
                            valid = False
                            
                    if valid:
                        r_lp = solve_lp_radii(centers_s)
                        s_lp = np.sum(r_lp)
                        if s_lp > best_sum:
                            best_sum = s_lp
                            best_centers = centers_s.copy()
                            best_radii = r_lp.copy()
            except Exception:

        # Phase 3: Final Coordinate Ascent from polished state
        best_centers, best_radii, best_sum = coordinate_ascent(best_centers, n, rng, iters=800)

    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
