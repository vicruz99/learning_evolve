import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    wall = np.maximum(wall, 1e-12)
    
    c = -np.ones(n)
    bounds = [(0.0, w) for w in wall]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def obj_smooth(centers_flat, k):
    """Objective: minimize negative smooth minimum clearance."""
    c = centers_flat.reshape(-1, 2)
    wall = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                      np.minimum(c[:, 1], 1.0 - c[:, 1]))
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    np.fill_diagonal(dists, np.inf)
    clearances = np.concatenate([wall, dists / 2.0])
    m = np.max(clearances)
    return -(m - np.log(np.sum(np.exp(-k * (clearances - m)))) / k)

def generate_inits(n, rng):
    """Generates diverse initial configurations."""
    configs = []
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], [6,6,5,5,4],
        [5,7,5,5,4], [6,5,5,6,4], [5,6,4,6,5], [7,6,6,7], [8,6,6,6]
    ]
    for pat in patterns:
        if sum(pat) < n: continue
        pts = []
        y = 0.09
        for ri, cnt in enumerate(pat):
            shift = 0.09 if ri % 2 == 1 else 0.0
            x = 0.09 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 0.18
            y += np.sqrt(3) * 0.09
        pts = np.array(pts[:n])
        mn, mx = pts.min(axis=0), pts.max(axis=0)
        span = mx - mn
        span[span < 1e-9] = 1.0
        pts_norm = (pts - mn) / span * 0.8 + 0.1
        configs.append(pts_norm)
        
        for _ in range(3):
            p = pts_norm + rng.uniform(-0.02, 0.02, pts_norm.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
        for angle in [0.05, -0.05, 0.1, -0.1]:
            cos_t, sin_t = np.cos(angle), np.sin(angle)
            rot = np.array([[cos_t, -sin_t], [sin_t, cos_t]])
            pts_r = pts @ rot
            mn_r, mx_r = pts_r.min(axis=0), pts_r.max(axis=0)
            span_r = mx_r - mn_r
            span_r[span_r < 1e-9] = 1.0
            pts_r = (pts_r - mn_r) / span_r * 0.8 + 0.1
            configs.append(pts_r)

    for seed in range(8):
        rng_local = np.random.default_rng(seed + 100)
        c = rng_local.uniform(0.2, 0.8, (n, 2))
        r = np.full(n, 0.01)
        for _ in range(800):
            r += 0.00006
            forces = np.zeros_like(c)
            for d in range(2):
                viol_l = np.maximum(0.0, r - c[:, d])
                viol_r = np.maximum(0.0, c[:, d] + r - 1.0)
                forces[:, d] += 500.0 * (viol_r - viol_l)
            diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
            np.fill_diagonal(dists, 0.0)
            req = r[:, np.newaxis] + r[np.newaxis, :]
            overlap = np.maximum(0.0, req - dists)
            dir_unit = diff / (dists[:, :, np.newaxis] + 1e-12)
            f_pair = overlap[:, :, np.newaxis] * dir_unit * 100.0
            forces += np.sum(f_pair, axis=1)
            c += forces * 0.0015
            c = np.clip(c, 1e-4, 1.0 - 1e-4)
        configs.append(c)
        
    for _ in range(5):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return configs

def obj_slsqp(v, n):
    return -np.sum(v[2*n:])

def cons_slsqp(v, n):
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    c_col = np.column_stack((cx, cy))
    diff = c_col[:, np.newaxis, :] - c_col[np.newaxis, :, :]
    d2 = np.sum(diff**2, axis=2)
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pc = d2[mask] - rs[mask]**2
    return np.concatenate([bc, pc])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = generate_inits(n, rng)
    
    # Phase 1: Smooth optimization + LP refinement
    bounds_c = [(0.01, 0.99)] * (2 * n)
    for cfg in configs:
        res = minimize(obj_smooth, cfg.flatten(), args=(50.0,), method='L-BFGS-B',
                       bounds=bounds_c, options={'maxiter': 5000, 'ftol': 1e-14})
        c_opt = res.x.reshape(n, 2)
        r_lp, s_lp = solve_lp_radii(c_opt)
        if s_lp > best_sum:
            best_sum = s_lp
            best_centers = c_opt.copy()
            best_radii = r_lp.copy()
            
    # Phase 2: Hill climbing with LP oracle
    if best_centers is not None:
        step = 0.025
        for _ in range(2500):
            idx = rng.integers(n)
            old = best_centers[idx].copy()
            best_centers[idx] = old + rng.uniform(-step, step, 2)
            best_centers[idx] = np.clip(best_centers[idx], 1e-4, 1.0 - 1e-4)
            r_new, s_new = solve_lp_radii(best_centers)
            if s_new > best_sum:
                best_sum = s_new
                best_radii = r_new.copy()
            else:
                best_centers[idx] = old
            step *= 0.999
            step = max(step, 0.001)
            
    # Phase 3: Multi-circle perturbation to break symmetry
    if best_centers is not None:
        for _ in range(50):
            c_pert = best_centers.copy()
            k = rng.integers(2, 6)
            idxs = rng.choice(n, size=k, replace=False)
            c_pert[idxs] += rng.uniform(-0.015, 0.015, (k, 2))
            c_pert = np.clip(c_pert, 0.05, 0.95)
            r_try, s_try = solve_lp_radii(c_pert)
            if r_try is not None and s_try > best_sum:
                best_sum = s_try
                best_centers = c_pert.copy()
                best_radii = r_try.copy()
                for _ in range(500):
                    i = rng.integers(n)
                    old = best_centers[i].copy()
                    best_centers[i] += rng.uniform(-0.01, 0.01, 2)
                    best_centers[i] = np.clip(best_centers[i], 1e-4, 1.0 - 1e-4)
                    r_t, s_t = solve_lp_radii(best_centers)
                    if r_t is not None and s_t > best_sum:
                        best_sum = s_t
                        best_radii = r_t.copy()
                    else:
                        best_centers[i] = old
                        break

    # Phase 4: Joint SLSQP polish
    if best_centers is not None:
        v0 = np.concatenate([best_centers[:, 0], best_centers[:, 1], best_radii * 0.995])
        bounds_v = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
        
        try:
            res = minimize(obj_slsqp, v0, args=(n,), method='SLSQP',
                           bounds=bounds_v, constraints={'type': 'ineq', 'fun': cons_slsqp, 'args': (n,)},
                           options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            if np.isfinite(res.fun):
                c_s = np.column_stack((res.x[:n], res.x[n:2*n]))
                r_s, s_s = solve_lp_radii(c_s)
                if r_s is not None and s_s > best_sum:
                    best_sum = s_s
                    best_centers = c_s.copy()
                    best_radii = r_s.copy()
        except Exception:
            pass

    # Safety scaling
    scale = 1.0
    c, r = best_centers, best_radii
    for i in range(n):
        if r[i] > 1e-12:
            scale = min(scale, c[i,0]/r[i], (1.0-c[i,0])/r[i], c[i,1]/r[i], (1.0-c[i,1])/r[i])
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
            rs = r[i] + r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
