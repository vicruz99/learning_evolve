from scipy.optimize import minimize, linprog

def solve_lp(centers, n, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    # Distance to nearest boundary
    wall = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    bounds = [(0.0, max(0.0, w)) for w in wall]
    
    # Pairwise distances
    dx = centers[idx_i, 0] - centers[idx_j, 0]
    dy = centers[idx_i, 1] - centers[idx_j, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    b_ub = dists
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x
    except Exception:
        pass
    # Fallback: half of min clearance
    lim = np.minimum(wall, np.min(np.sqrt((centers[:,None,:]-centers[None,:,:])**2).sum(axis=2), axis=1)/2.0)
    return np.maximum(lim * 0.9, 0.0)

def joint_obj(vars, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2*n:])

def joint_cons(vars, n, idx_i, idx_j):
    """Inequality constraints >= 0."""
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    cons = []
    # Boundary
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap (squared distances to maintain smoothness)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    d2 = dx**2 + dy**2
    rs = r[idx_i] + r[idx_j]
    cons.append(d2 - rs**2)
    
    return np.concatenate(cons)

def generate_hex_configs(n, rng):
    """Generates diverse hexagonal lattice configurations."""
    configs = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4]
    ]

        if sum(pat) < n: continue
        pts = []
        y = 0.0
        r0 = 0.10
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            
        pts = np.array(pts[:n])
        # Normalize to [0.1, 0.9]
        mn, mx = pts.min(axis=0), pts.max(axis=0)
        span = mx - mn
        span[span < 1e-9] = 1.0
        pts_norm = (pts - mn) / span * 0.8 + 0.1
        configs.append(pts_norm)
        
        # Add rotations to explore alignment with square boundaries
        for ang in rng.uniform(-0.08, 0.08, 3):
            c, s = np.cos(ang), np.sin(ang)
            rot = np.array([[c, -s], [s, c]])
            pts_r = (pts_norm - pts_norm.mean(axis=0)) @ rot
            mn_r, mx_r = pts_r.min(axis=0), pts_r.max(axis=0)
            sp = mx_r - mn_r
            sp[sp < 1e-9] = 1.0
            pts_s = (pts_r - mn_r) / sp * 0.8 + 0.1
            configs.append(np.clip(pts_s, 0.05, 0.95))
            
    # Random dense starts
    for _ in range(8):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    return configs
