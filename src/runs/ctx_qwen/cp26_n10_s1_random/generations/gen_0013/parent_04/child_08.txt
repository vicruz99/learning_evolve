from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    wall_dists = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    wall_dists = np.maximum(wall_dists, 1e-12)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    triu_i, triu_j = np.triu_indices(n, k=1)
    m = len(triu_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), triu_i] = 1.0
    A_ub[np.arange(m), triu_j] = 1.0
    b_ub = dists[triu_i, triu_j]
    
    bounds = [(0.0, w) for w in wall_dists]
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback geometric radii
    r_fall = np.minimum(wall_dists, np.min(dists, axis=1) / 2.0) * 0.95
    return r_fall, np.sum(r_fall)

def get_joint_obj(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def get_joint_cons(v, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    cons = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[triu_i] - cx[triu_j]
    dy = cy[triu_i] - cy[triu_j]
    rs = r[triu_i] + r[triu_j]
    
    cons = np.concatenate([cons, dx**2 + dy**2 - rs**2])
    return cons

def generate_hex(row_counts, r0):
    """Generates hexagonal lattice points."""
    pts = []
    y = r0
    for ri, cnt in enumerate(row_counts):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= 26:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(pts[:26])
