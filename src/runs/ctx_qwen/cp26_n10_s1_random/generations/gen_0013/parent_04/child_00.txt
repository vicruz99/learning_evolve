import numpy as np
from scipy.optimize import minimize, linprog


def solve_lp_radii(centers, n, ii, jj):
    """Solves LP to maximize sum of radii for fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    lims = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    c_obj = -np.ones(n)
    bounds = [(0.0, max(l, 1e-12)) for l in lims]

    dx = centers[ii, 0] - centers[jj, 0]
    dy = centers[ii, 1] - centers[jj, 1]
    b_ub = np.hypot(dx, dy)

    m = len(ii)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), ii] = 1.0
    A_ub[np.arange(m), jj] = 1.0

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0


def joint_obj(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])


def joint_cons(v, n, ii, jj):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]

    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[ii] - cx[jj]
    dy = cy[ii] - cy[jj]
    dr = r[ii] + r[jj]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c


def generate_hex_config(rd, n, rng, r0):
    """Generate a hexagonal lattice configuration."""
    pts = []
    y = r0
    for idx, cnt in enumerate(rd):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(pts[:n])


def force_directed_init(n, seed, rng):
    """Generate configuration via force-directed expansion."""
    rng_local = np.random.default_rng(seed)
    centers = rng_local.uniform(0.25, 0.75, (n, 2))
    radii = np.full(n, 0.015)

    dt = 0.015
    for _ in range(1800):
        radii += 0.00004

        forces = np.zeros_like(centers)
        for d in range(2):
            viol_left = np.maximum(0.0, radii - centers[:, d])
            viol_right = np.maximum(0.0, centers[:, d] + radii - 1.0)
            forces[:, d] += (viol_right - viol_left) * 600.0

        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dists, 0.0)

        required = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlap = np.maximum(0.0, required - dists)

        dir_unit = diff / (dists[:, :, np.newaxis] + 1e-12)
        forces_add = overlap[:, :, np.newaxis] * dir_unit * 120.0
        forces += np.sum(forces_add, axis=1)

        centers += forces * dt
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)

    return centers
