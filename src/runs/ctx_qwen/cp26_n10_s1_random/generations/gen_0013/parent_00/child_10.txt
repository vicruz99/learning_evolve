import numpy as np
from scipy.optimize import linprog, minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

def compute_lp_and_grad(centers):
    """Solves LP for max sum of radii given centers, returns radii, sum, and gradient w.r.t centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, lim) for lim in lims]
    
    diffs = centers[PAIR_I] - centers[PAIR_J]
    dists = np.sqrt(np.sum(diffs**2, axis=1) + 1e-16)
    
    # Constraints: M_PAIRS + 4*N
    n_constraints = M_PAIRS + 4 * N
    A_ub = np.zeros((n_constraints, N))
    b_ub = np.zeros(n_constraints)
    
    # Pairwise: r_i + r_j <= dist_ij
    A_ub[:M_PAIRS, PAIR_I] = 1.0
    A_ub[:M_PAIRS, PAIR_J] = 1.0
    b_ub[:M_PAIRS] = dists
    
    # Boundary: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    idx_offset = M_PAIRS
    A_ub[idx_offset + np.arange(N), np.arange(N)] = 1.0
    b_ub[idx_offset + np.arange(N)] = centers[:, 0]
    
    idx_offset += N
    A_ub[idx_offset + np.arange(N), np.arange(N)] = 1.0
    b_ub[idx_offset + np.arange(N)] = 1.0 - centers[:, 0]
    
    idx_offset += N
    A_ub[idx_offset + np.arange(N), np.arange(N)] = 1.0
    b_ub[idx_offset + np.arange(N)] = centers[:, 1]
    
    idx_offset += N
    A_ub[idx_offset + np.arange(N), np.arange(N)] = 1.0
    b_ub[idx_offset + np.arange(N)] = 1.0 - centers[:, 1]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, 0.0, np.zeros((N, 2))
            
        r = np.maximum(res.x, 0.0)
        s = -res.fun
        
        grad = np.zeros((N, 2))
        try:
            marg = res.marginals.ineqlin
        except AttributeError:
            try:
                marg = res.ineqlin.marginals
            except AttributeError:
                marg = None
                
        if marg is not None:
            # Pairwise gradients
            active_lam = marg[:M_PAIRS]
            mask = active_lam > 1e-7
            if np.any(mask):
                idx = np.where(mask)[0]
                lam = active_lam[idx]
                ii = PAIR_I[idx]
                jj = PAIR_J[idx]
                d = dists[idx]
                d = np.where(d > 1e-9, d, 1.0)
                vec = lam[:, np.newaxis] * (centers[ii] - centers[jj]) / d[:, np.newaxis]
                grad[ii] += vec
                grad[jj] -= vec
                
            # Boundary gradients
            lam_x = marg[M_PAIRS : M_PAIRS + N]
            lam_1x = marg[M_PAIRS + N : M_PAIRS + 2*N]
            lam_y = marg[M_PAIRS + 2*N : M_PAIRS + 3*N]
            lam_1y = marg[M_PAIRS + 3*N : M_PAIRS + 4*N]
            
            mask_x = lam_x > 1e-7
            if np.any(mask_x):
                grad[mask_x, 0] -= lam_x[mask_x]
                
            mask_1x = lam_1x > 1e-7
            if np.any(mask_1x):
                grad[mask_1x, 0] += lam_1x[mask_1x]
                
            mask_y = lam_y > 1e-7
            if np.any(mask_y):
                grad[mask_y, 1] -= lam_y[mask_y]
                
            mask_1y = lam_1y > 1e-7
            if np.any(mask_1y):
                grad[mask_1y, 1] += lam_1y[mask_1y]
                
        return r, s, grad
    except Exception:
        return None, 0.0, np.zeros((N, 2))

def force_sim(centers, steps=3000):
    """Vectorized force-directed simulation to find dense packings."""
    c = centers.copy()
    r = np.full(N, 0.035)
    vel = np.zeros_like(c)
    
    for step in range(steps):
        dt = 0.008 * (1.0 - 0.6 * step / steps) + 0.001
        damp = 0.82
        growth = 1.00015 * (1.0 - 0.4 * step / steps)
        k = 300.0
        
        r *= growth
        
        diff = c[:, None, :] - c[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dist, 1e9)
        
        overlap = np.maximum(0.0, r[:, None] + r[None, :] - dist)
        inv_d = 1.0 / (dist + 1e-9)
        rep = overlap * inv_d * k
        f_rep = np.sum(diff * rep[:, :, None], axis=1)
        
        f_wall = np.zeros_like(c)
        f_wall[:, 0] += np.clip(r - c[:, 0], 0, None) * k
        f_wall[:, 0] -= np.clip(c[:, 0] + r - 1.0, 0, None) * k
        f_wall[:, 1] += np.clip(r - c[:, 1], 0, None) * k
        f_wall[:, 1] -= np.clip(c[:, 1] + r - 1.0, 0, None) * k
        
        vel = vel * damp + (f_rep + f_wall) * dt
        c += vel
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
        
    return c, r

def slqp_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def slqp_cons(v):
    """Inequality constraints >= 0 using squared distances for smooth gradients."""
    cx = v[:N]
    cy = v[N:2 * N]
    r = v[2 * N:]
    
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[PAIR_I] - cx[PAIR_J]
    dy = cy[PAIR_I] - cy[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def _obj_and_grad(v):
    """Objective and gradient for L-BFGS-B using LP duals."""
    c = v.reshape(N, 2)
    r, s, g = compute_lp_and_grad(c)
    if r is None:
        return 0.0, np.zeros_like(v)
    return -s, -g.flatten()

def lbfgs_optimize(c0):
    """Optimize centers using L-BFGS-B with analytical LP gradient."""
    res = minimize(_obj_and_grad, c0.flatten(), method='L-BFGS-B', 
                   bounds=[(1e-4, 0.999)] * (2 * N),
                   options={'maxiter': 2500, 'ftol': 1e-12})
    return res.x.reshape(N, 2), -res.fun
