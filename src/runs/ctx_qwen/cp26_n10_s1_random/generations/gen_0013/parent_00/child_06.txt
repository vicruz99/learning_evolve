import numpy as np
from scipy.optimize import linprog, minimize

_N = 26
_TRU_I, _TRU_J = np.triu_indices(_N, k=1)
_N_PAIRS = len(_TRU_I)

def _solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    diffs = centers[_TRU_I] - centers[_TRU_J]
    dists = np.sqrt(np.sum(diffs**2, axis=1) + 1e-16)
    
    A_ub = np.zeros((_N_PAIRS, _N))
    A_ub[np.arange(_N_PAIRS), _TRU_I] = 1.0
    A_ub[np.arange(_N_PAIRS), _TRU_J] = 1.0
    b_ub = dists
    
    c_obj = -np.ones(_N)
    bounds = [(0.0, 0.5)] * _N
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            r = np.maximum(res.x, 0.0)
            return r, np.sum(r)
    except Exception:
        pass
    return np.full(_N, 1e-6), 0.0

def _solve_lp_and_grad(centers):

    dists = np.sqrt(np.sum(diffs**2, axis=1) + 1e-16)

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

            return np.full(_N, 1e-6), 0.0, np.zeros((_N, 2))
    except Exception:
        return np.full(_N, 1e-6), 0.0, np.zeros((_N, 2))

    s = np.sum(r)

    try:
        marg = res.marginals.ineqlin

            grad[i, 1] -= np.clip(centers[i, 1] + r[i] - 1.0, 0, None) * 200.0
            
    return r, s, grad

def _gradient_ascent(c0, max_iter=700):

    c = c0.copy()
    r, s, g = _solve_lp_and_grad(c)
    best_s, best_c, best_r = s, c.copy(), r.copy()

            break

            c = c_new
        else:
            lr *= 0.93

    return best_c, best_r, best_s

def _local_search(c0, r0, s0, max_iter=4000, rng=None):

            s = s_try
            r = r_try.copy()
            step = min(step * 1.009, 0.045)
        else:
            c[i] = old
            step *= 0.995
            
    return c, r, s

def _slqp_polish(c0, r0):

        return -np.sum(v[2*_N:])
        
    def cons(v):

        dx = cx[_TRU_I] - cx[_TRU_J]
        dy = cy[_TRU_I] - cy[_TRU_J]
        dr = r[_TRU_I] + r[_TRU_J]
        c = np.concatenate([c, dx**2 + dy**2 - dr**2])
        return c

            return np.column_stack((cx, cy)), np.maximum(r, 0.0)
    except Exception:

    return c0, r0
