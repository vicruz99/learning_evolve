import numpy as np
from scipy.optimize import linprog, minimize
import warnings
warnings.filterwarnings('ignore')

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

def solve_lp_and_grad(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, and gradient w.r.t centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    diffs = centers[PAIR_I] - centers[PAIR_J]
    dists = np.sqrt(np.sum(diffs**2, axis=1) + 1e-16)
    
    A_ub = np.zeros((M_PAIRS + 4 * n, n))
    b_ub = np.zeros(M_PAIRS + 4 * n)
    
    A_ub[:M_PAIRS, PAIR_I] = 1.0
    A_ub[:M_PAIRS, PAIR_J] = 1.0
    b_ub[:M_PAIRS] = dists
    
    idx = M_PAIRS
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    bounds = [(0.0, lims[i]) for i in range(n)]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, 0.0, np.zeros((n, 2))
            
        r = np.maximum(res.x, 0.0)
        s = -res.fun
        
        grad = np.zeros((n, 2))
        try:
            marg = res.marginals.ineqlin
        except AttributeError:
            try:
                marg = res.ineqlin.marginals
            except AttributeError:
                marg = None
                
        if marg is not None:
            active_lam = marg[:M_PAIRS]
            mask = active_lam > 1e-7
            if np.any(mask):
                idx_active = np.where(mask)[0]
                lam = active_lam[mask]
                ii = PAIR_I[idx_active]
                jj = PAIR_J[idx_active]
                d = dists[idx_active]
                d = np.where(d > 1e-9, d, 1.0)
                vec = lam[:, np.newaxis] * (centers[ii] - centers[jj]) / d[:, np.newaxis]
                np.add.at(grad, ii, vec)
                np.add.at(grad, jj, -vec)

                base = M_PAIRS + 4 * i
                grad[i, 0] += marg[base] - marg[base+1]
                grad[i, 1] += marg[base+2] - marg[base+3]
                
        return r, s, grad
    except Exception:
        return None, 0.0, np.zeros((n, 2))

def gradient_ascent(c0, steps=500):
    """Performs gradient ascent on centers to maximize LP sum of radii."""
    c = c0.copy()
    r_best, s_best, _ = solve_lp_and_grad(c)
    if r_best is None:
        return c, s_best
        
    step = 0.025
    vel = np.zeros_like(c)
    momentum = 0.65
    
    for _ in range(steps):
        r, s, g = solve_lp_and_grad(c)
        if r is None:
            break
            
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-6:
            break
            
        g_dir = g / g_norm
        vel = momentum * vel + step * g_dir
        c_new = c + vel
        c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
        
        r_new, s_new, _ = solve_lp_and_grad(c_new)
        if r_new is not None and s_new > s_best + 1e-8:
            c = c_new
            r_best, s_best = r_new, s_new
            step = min(step * 1.03, 0.06)
        else:
            step *= 0.88
            vel *= 0.4
            
        if step < 1e-5:
            break
            
    return c, s_best

def obj_slqp(v, n):
    return -np.sum(v[2 * n:])

def cons_slqp(v, n):
    cx = v[:n]
    cy = v[n:2 * n]
    r = v[2 * n:]
    
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[PAIR_I] - cx[PAIR_J]
    dy = cy[PAIR_I] - cy[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def slqp_polish(c0, r0, n):
    """Refine centers and radii simultaneously using SLSQP."""
    v0 = np.concatenate([c0[:, 0], c0[:, 1], r0])
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    cons = {'type': 'ineq', 'fun': cons_slqp, 'args': (n,)}
    
    try:
        res = minimize(obj_slqp, v0, args=(n,), method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            return res.x[:n], res.x[n:2*n], res.x[2*n:]
    except Exception:
        pass
    return c0[:, 0], c0[:, 1], r0

def generate_configs(rng, n):

    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4],

    ]
    
    for pat in patterns:

                    if len(pts) >= n: break
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            cfg = np.array(pts[:n])
            
            mn, mx = cfg.min(axis=0), cfg.max(axis=0)
            span = mx - mn + 1e-9
            cfg = (cfg - mn) / span * 0.82 + 0.09
            
            for _ in range(2):
                cfg_pert = cfg + rng.uniform(-0.015, 0.015, cfg.shape)
                configs.append(np.clip(cfg_pert, 0.05, 0.95))
                
            # Rotated variants
            for angle in [0.05, -0.05, 0.1]:
                c, s_rot = np.cos(angle), np.sin(angle)

                cfg_rot = cfg @ rot_mat
                cfg_rot = (cfg_rot - cfg_rot.min(axis=0)) / (cfg_rot.max(axis=0) - cfg_rot.min(axis=0)) * 0.82 + 0.09
                configs.append(np.clip(cfg_rot, 0.05, 0.95))

    for _ in range(12):
        configs.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    return configs
