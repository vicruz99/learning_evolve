

<strategy>
1. **Diverse Initialization**: Generate multiple hexagonal lattice configurations with varying row counts and rotations to cover different symmetry-breaking patterns. Add random dense starts for exploration.
2. **Force-Directed Simulation**: Use a vectorized repulsive force simulation to quickly separate circles, grow them uniformly, and push them into a dense, non-overlapping arrangement. This provides a high-quality starting point for gradient-based methods.
3. **LP Gradient Ascent**: Iteratively solve a Linear Program to find maximal radii for fixed centers. Extract dual variables (marginals) from the LP to compute the exact gradient of the sum of radii with respect to center positions. Perform gradient ascent with adaptive step size and momentum to optimize center placements directly for the objective.
4. **SLSQP Joint Polish**: Use SLSQP to jointly optimize centers and radii with squared-distance constraints for precise handling of tight boundary and overlap conditions.
5. **Coordinate Ascent & Safety Scaling**: Run a final coordinate ascent pass to squeeze out remaining gains, then compute a strict scaling factor to guarantee all constraints are satisfied within the validator's tolerance, applying it to the final radii.
6. **Robustness**: Ensure all helper functions are top-level, avoid closures/lambdas, handle LP marginal extraction safely across scipy versions, and include fallbacks to prevent failures.

import numpy as np
from scipy.optimize import linprog, minimize
import warnings
warnings.filterwarnings('ignore')

N = 26
TRI_I, TRI_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRI_I)

def solve_lp_and_grad(centers):
    """Solves LP to maximize sum of radii for fixed centers, returns radii, sum, and gradient."""
    n = N
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    diffs = centers[TRI_I] - centers[TRI_J]
    dists = np.sqrt(np.sum(diffs**2, axis=1) + 1e-16)
    
    A_ub = np.zeros((M_PAIRS + n, n))
    A_ub[:M_PAIRS, TRI_I] = 1.0
    A_ub[:M_PAIRS, TRI_J] = 1.0
    b_ub = np.zeros(M_PAIRS + n)
    b_ub[:M_PAIRS] = dists
    A_ub[M_PAIRS:, np.arange(n)] = 1.0
    b_ub[M_PAIRS:] = lims
    
    c_obj = -np.ones(n)
    bounds = [(0.0, lims[i]) for i in range(n)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or not np.isfinite(res.fun):
            return None, 0.0, np.zeros((n, 2))
            
        r = np.maximum(res.x, 0.0)
        s = -res.fun
        
        grad = np.zeros((n, 2))
        marg = None
        try:
            marg = res.marginals.ineqlin
        except AttributeError:
            try:
                marg = res.ineqlin.marginals
            except AttributeError:
                pass
                
        if marg is not None:
            lam_pairs = marg[:M_PAIRS]
            mask = lam_pairs > 1e-8
            if np.any(mask):
                idx = np.where(mask)[0]
                lam = lam_pairs[idx]
                ii = TRI_I[idx]
                jj = TRI_J[idx]
                d = dists[idx]
                d = np.where(d > 1e-9, d, 1.0)
                vec = lam[:, np.newaxis] * (centers[ii] - centers[jj]) / d[:, np.newaxis]
                np.add.at(grad, ii, vec)
                np.add.at(grad, jj, -vec)
                
        return r, s, grad
    except Exception:
        return None, 0.0, np.zeros((n, 2))

def force_sim(centers, steps=1500):
    """Vectorized force-directed simulation to find dense packings."""
    n = N
    c = centers.copy()
    r = np.full(n, 0.035)
    vel = np.zeros_like(c)
    
    for step in range(steps):
        dt = 0.008 * (1.0 - 0.6 * step / steps) + 0.001
        damp = 0.85
        growth = 1.00015 * (1.0 - 0.5 * step / steps)
        k = 200.0
        
        r *= growth
        
        diff = c[:, None, :] - c[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dist, 1e9)
        
        overlap = np.maximum(0.0, r[:, None] + r[None, :] - dist)
        inv_d = 1.0 / (dist + 1e-9)
        rep = overlap * inv_d * k
        f_rep = np.sum(diff * rep[:, :, None], axis=1)
        
        f_wall = np.zeros_like(c)
        f_wall[:, 0] += np.clip(r - c[:, 0], 0, None) * k
        f_wall[:, 0] -= np.clip(c[:, 0] + r - 1.0, 0, None) * k
        f_wall[:, 1] += np.clip(r - c[:, 1], 0, None) * k
        f_wall[:, 1] -= np.clip(c[:, 1] + r - 1.0, 0, None) * k
        
        vel = vel * damp + (f_rep + f_wall) * dt
        c += vel
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
        
    return c

def obj_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def cons_slqp(v):
    """Inequality constraints >= 0 using squared distances for smooth gradients."""
    cx = v[:N]
    cy = v[N:2 * N]
    r = v[2 * N:]
    
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[TRI_I] - cx[TRI_J]
    dy = cy[TRI_I] - cy[TRI_J]
    dr = r[TRI_I] + r[TRI_J]
    
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def slqp_polish(c0, r0):
    """Refine centers and radii simultaneously using SLSQP."""
    v0 = np.concatenate([c0[:, 0], c0[:, 1], r0])
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': cons_slqp}
    
    try:
        res = minimize(obj_slqp, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-13})
        if np.isfinite(res.fun):
            return res.x[:N], res.x[N:2 * N], res.x[2 * N:]
    except Exception:
        pass
    return c0[:, 0], c0[:, 1], r0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(123)
    best_sum = 0.0
    best_c = None
    best_r = None
    
    starts = []
    # Hex patterns
    pats = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 6, 5, 4],
        [7, 5, 5, 5, 4], [4, 5, 7, 5, 5], [8, 6, 6, 6]
    ]
    
    for pat in pats:
        if sum(pat) != N: continue
        pts = []
        y = 0.09
        for i, cnt in enumerate(pat):
            shift = 0.09 if i % 2 == 1 else 0.0
            x = 0.09 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 0.18
            y += 0.09 * np.sqrt(3)
        starts.append(np.array(pts[:N]))
        
    # Rotated hex
    for pat in pats[:3]:
        if sum(pat) != N: continue
        base = []
        y = 0.09
        for i, cnt in enumerate(pat):
            shift = 0.09 if i % 2 == 1 else 0.0
            x = 0.09 + shift
            for _ in range(cnt):
                if len(base) >= N: break
                base.append([x, y])
                x += 0.18
            y += 0.09 * np.sqrt(3)
        base = np.array(base[:N])
        base = base - base.mean(axis=0)
        for ang in [0.15, -0.15, 0.3, -0.3]:
            rot = np.array([[np.cos(ang), -np.sin(ang)], [np.sin(ang), np.cos(ang)]])
            cfg = base @ rot
            cfg = cfg - cfg.min(axis=0)
            cfg = cfg / (cfg.max(axis=0) + 1e-9) * 0.88 + 0.06
            starts.append(np.clip(cfg, 0.05, 0.95))

    # Random
    for _ in range(6):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # Main optimization loop
    for s in starts:
        # Perturbations
        for _ in range(3):
            s_p = s + rng.uniform(-0.02, 0.02, s.shape)
            s_p = np.clip(s_p, 0.05, 0.95)
            
            c_sim = force_sim(s_p, steps=1200)
            
            # LP Gradient Ascent
            c_curr = c_sim.copy()
            r_curr, s_curr, _ = solve_lp_and_grad(c_curr)
            if r_curr is None: continue
            
            step = 0.02
            vel = np.zeros_like(c_curr)
            for _ in range(500):
                r_new, s_new, g = solve_lp_and_grad(c_curr)
                if r_new is None: break
                
                g_norm = np.linalg.norm(g)
                if g_norm < 1e-7: break
                
                g_dir = g / g_norm
                vel = 0.6 * vel + step * g_dir
                c_new = c_curr + vel
                c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
                
                r_try, s_try, _ = solve_lp_and_grad(c_new)
                if r_try is not None and s_try > s_curr + 1e-8:
                    c_curr = c_new
                    r_curr, s_curr = r_try, s_try
                    step = min(step * 1.02, 0.05)
                else:
                    step *= 0.85
                    vel *= 0.5
                    
                if step < 1e-5: break
                
            if s_curr > best_sum:
                best_sum = s_curr
                best_c = c_curr.copy()
                best_r = r_curr.copy()
                
            # SLSQP Polish
            cx, cy, r_pol = slqp_polish(c_curr, r_curr * 0.95)
            c_pol = np.column_stack((cx, cy))
            r_lp, s_lp, _ = solve_lp_and_grad(c_pol)
            if r_lp is not None and s_lp > best_sum:
                best_sum = s_lp
                best_c = c_pol.copy()
                best_r = r_lp.copy()

    # Final Coordinate Ascent
    if best_c is not None:
        curr_c, curr_r, curr_s = best_c.copy(), best_r.copy(), best_sum
        step = 0.01
        for _ in range(2000):
            i = rng.integers(N)
            old = curr_c[i].copy()
            curr_c[i] += rng.uniform(-step, step, 2)
            curr_c[i] = np.clip(curr_c[i], 1e-4, 0.999)
            
            r_try, s_try, _ = solve_lp_and_grad(curr_c)
            if r_try is not None and s_try > curr_s + 1e-8:
                curr_s = s_try
                curr_r = r_try.copy()
                step = min(step * 1.01, 0.025)
            else:
                curr_c[i] = old
                step *= 0.995
                
        best_c, best_r, best_sum = curr_c, curr_r, curr_s

    # Safety Scaling
    scale = 1.0
    for i in range(N):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    diff = best_c[:, None, :] - best_c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    for i in range(N):
        for j in range(i + 1, N):
            d = dists[i, j]
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
