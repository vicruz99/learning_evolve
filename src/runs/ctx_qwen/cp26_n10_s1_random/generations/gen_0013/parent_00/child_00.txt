import numpy as np
from scipy.optimize import minimize, linprog

# Module-level constants for constraints
N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(IDX_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    A_ub = np.zeros((NUM_PAIRS, N))
    A_ub[np.arange(NUM_PAIRS), IDX_I] = 1.0
    A_ub[np.arange(NUM_PAIRS), IDX_J] = 1.0
    b_ub = dists[IDX_I, IDX_J]
    
    bounds = [(0.0, lims[i]) for i in range(N)]
    c_obj = -np.ones(N)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0

def force_relax(centers, steps=600):
    """Vectorized force-directed simulation to separate overlapping circles."""
    c = centers.copy()
    r = np.full(N, 0.045)
    vel = np.zeros_like(c)
    
    for _ in range(steps):
        diff = c[:, None, :] - c[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dist, 1e9)
        
        rep = np.maximum(0.0, 2.0 * r[:, None] - dist) / (dist + 1e-9)
        f = np.sum(diff * rep[:, :, None], axis=1)
        
        vel = 0.85 * vel + f * 0.04
        c += vel
        c = np.clip(c, 1e-3, 0.999)
    return c

def slqp_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def slqp_con(v):
    """Inequality constraints >= 0 using squared distances for smooth gradients."""
    cx, cy, r = v[:N], v[N:2 * N], v[2 * N:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[IDX_I] - cx[IDX_J]
    dy = cy[IDX_I] - cy[IDX_J]
    dr = r[IDX_I] + r[IDX_J]
    
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

        [7, 5, 5, 5, 4], [4, 5, 7, 5, 5], [6, 5, 5, 6, 4]
    ]

            continue

            x = 0.1 + shift

                if len(pts) >= N: 
                    break
                pts.append([x, y])
                x += 0.2
            y += 0.1 * np.sqrt(3)
        starts.append(np.array(pts[:N]))
        
    # Rotated variants to break symmetry

                scale = min(scale, d / rs)
                
    best_r *= scale * 0.9999998
    best_s = float(np.sum(best_r))
    
    return best_c, best_r, best_s
