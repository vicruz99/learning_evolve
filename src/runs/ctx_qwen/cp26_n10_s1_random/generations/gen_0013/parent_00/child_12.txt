import numpy as np
from scipy.optimize import minimize, linprog

_N = 26
_TRIU_I, _TRIU_J = np.triu_indices(_N, k=1)
_NUM_PAIRS = len(_TRIU_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = _N
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    np.fill_diagonal(dists, 1e9)
    
    A_ub = np.zeros((_NUM_PAIRS, n))
    A_ub[np.arange(_NUM_PAIRS), _TRIU_I] = 1.0
    A_ub[np.arange(_NUM_PAIRS), _TRIU_J] = 1.0
    b_ub = dists[_TRIU_I, _TRIU_J]
    
    bounds = [(0.0, lims[i]) for i in range(n)]
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= -1e-7):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def force_sim(centers, steps=1500):
    """Vectorized force-directed simulation to find dense packings."""
    n = _N
    c = centers.copy()
    r = np.full(n, 0.035)
    vel = np.zeros_like(c)
    
    for step in range(steps):
        dt = 0.008 * (1.0 - 0.7 * step / steps) + 0.0005
        damp = 0.85
        growth = 1.0002 * (1.0 - 0.5 * step / steps)
        k = 300.0
        
        r *= growth
        
        diff = c[:, None, :] - c[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dist, 1e9)
        
        overlap = np.maximum(0.0, r[:, None] + r[None, :] - dist)
        inv_d = 1.0 / (dist + 1e-9)
        rep = overlap * inv_d * k
        f_rep = np.sum(diff * rep[:, :, None], axis=1)
        
        f_wall = np.zeros_like(c)
        f_wall[:, 0] += np.clip(r - c[:, 0], 0, None) * k
        f_wall[:, 0] -= np.clip(c[:, 0] + r - 1.0, 0, None) * k
        f_wall[:, 1] += np.clip(r - c[:, 1], 0, None) * k
        f_wall[:, 1] -= np.clip(c[:, 1] + r - 1.0, 0, None) * k
        
        vel = vel * damp + (f_rep + f_wall) * dt
        c += vel
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
        
    return c, r

def obj_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * _N:])

def cons_slqp(v):
    """Inequality constraints >= 0 using squared distances for smooth gradients."""
    cx = v[:_N]
    cy = v[_N:2 * _N]
    r = v[2 * _N:]
    
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[_TRIU_I] - cx[_TRIU_J]
    dy = cy[_TRIU_I] - cy[_TRIU_J]
    dr = r[_TRIU_I] + r[_TRIU_J]
    
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def slqp_polish(c0, r0):
    """Refine centers and radii simultaneously using SLSQP."""
    v0 = np.concatenate([c0[:, 0], c0[:, 1], r0])
    bounds = [(0.0, 1.0)] * (2 * _N) + [(1e-6, 0.5)] * _N
    cons = {'type': 'ineq', 'fun': cons_slqp}
    
    try:
        res = minimize(obj_slqp, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            return res.x[:_N], res.x[_N:2 * _N], res.x[2 * _N:]
    except Exception:
        pass
    return c0[:, 0], c0[:, 1], r0

def center_relaxation(centers, radii, steps=400):
    """Optimize centers for fixed radii to maximize clearance using penalty L-BFGS-B."""
    n = _N
    c = centers.copy()
    v = c.flatten()
    
    def penalty(v_arr):
        ca = v_arr.reshape(n, 2)
        p = 0.0
        dx = ca[:, None, 0] - ca[None, :, 0]
        dy = ca[:, None, 1] - ca[None, :, 1]
        dist = np.sqrt(dx**2 + dy**2 + 1e-12)
        np.fill_diagonal(dist, 1e9)
        overlap = np.maximum(0.0, radii[:, None] + radii[None, :] - dist)
        p += 100.0 * np.sum(overlap**2)
        
        p += 100.0 * np.sum(np.maximum(0.0, radii - ca[:, 0])**2)
        p += 100.0 * np.sum(np.maximum(0.0, radii - (1.0 - ca[:, 0]))**2)
        p += 100.0 * np.sum(np.maximum(0.0, radii - ca[:, 1])**2)
        p += 100.0 * np.sum(np.maximum(0.0, radii - (1.0 - ca[:, 1]))**2)
        return p

    bounds = [(1e-4, 1.0 - 1e-4)] * (2 * n)
    try:
        res = minimize(penalty, v, method='L-BFGS-B', bounds=bounds, options={'maxiter': steps, 'ftol': 1e-15})
        return res.x.reshape(n, 2)
    except Exception:
        return c
