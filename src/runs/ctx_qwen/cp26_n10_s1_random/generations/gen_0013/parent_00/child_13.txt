from scipy.optimize import linprog, minimize
import warnings
warnings.filterwarnings('ignore')

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

def compute_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = N
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-16)
    
    A_ub = np.zeros((M_PAIRS, n))
    A_ub[np.arange(M_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(M_PAIRS), PAIR_J] = 1.0
    b_ub = dists[PAIR_I, PAIR_J]
    
    bounds = [(0.0, lims[i]) for i in range(n)]
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            r = np.maximum(res.x, 0.0)
            try:
                marg = res.marginals.ineqlin
            except AttributeError:
                try:
                    marg = res.ineqlin.marginals
                except AttributeError:
                    marg = None
            return r, -res.fun, marg
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0, None

def gradient_ascent(c0, max_iter=600):

        return c, s

    no_improve = 0

        if marg is None:
            break

            np.add.at(grad, ii, forces)
            np.add.at(grad, jj, -forces)

        if g_norm < 1e-9:
            no_improve += 1
            if no_improve > 50: 
                break

        vel = mom * vel + step * g_dir
        c_new = c + vel
        c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)

        if s_new > best_s + 1e-9:
            c = c_new
            r, s, marg = r_new, s_new, marg_new

            step = min(step * 1.06, 0.06)
            no_improve = 0
        else:
            step *= 0.88
            vel *= 0.5
            no_improve += 1
        if step < 1e-6:
            break
    return best_c, best_s

def obj_slqp(v):
    return -np.sum(v[2*N:])

def cons_slqp(v):
    cx, cy, r = v[:N], v[N:2*N], v[2*N:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[PAIR_I] - cx[PAIR_J]
    dy = cy[PAIR_I] - cy[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def joint_slqp(c0, r0):
    """Refines centers and radii simultaneously using SLSQP."""
    v0 = np.concatenate([c0[:, 0], c0[:, 1], r0 * 0.95])
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    
    try:
        res = minimize(obj_slqp, v0, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': cons_slqp},
                       options={'maxiter': 5000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            return np.column_stack((res.x[:N], res.x[N:2*N])), res.x[2*N:]
    except Exception:
        pass
    return c0, r0

def coordinate_ascent(c0, r0, s0, iters=2500):
    """Hill-climbing by moving one circle at a time."""
    rng = np.random.default_rng(42)
    c, r, s = c0.copy(), r0.copy(), s0
    step = 0.015
    for _ in range(iters):
        i = rng.integers(N)
        old = c[i].copy()
        c[i] += rng.uniform(-step, step, 2)
        c[i] = np.clip(c[i], 1e-4, 1.0 - 1e-4)
        r_new, s_new, _ = compute_lp(c)
        if s_new > s + 1e-9:
            s = s_new
            r = r_new.copy()
            step = min(step * 1.02, 0.035)
        else:
            c[i] = old
            step *= 0.996
    return c, r, s
