from scipy.optimize import minimize, linprog

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M_PAIRS = len(PAIR_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-7)
    
    diff = centers[PAIR_I] - centers[PAIR_J]
    dists = np.sqrt(np.sum(diff**2, axis=1) + 1e-16)

    A_ub[np.arange(M_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(M_PAIRS), PAIR_J] = 1.0
    b_ub = dists

    c_obj = -np.ones(N)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0

def joint_obj(v):
    return -np.sum(v[2 * N:])

def joint_cons(v):
    cx = v[:N]
    cy = v[N:2 * N]
    r = v[2 * N:]

    dx = cx[PAIR_I] - cx[PAIR_J]
    dy = cy[PAIR_I] - cy[PAIR_J]
    dr = r[PAIR_I] + r[PAIR_J]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def slqp_polish(c0, r0):
    """Refine centers and radii simultaneously using SLSQP."""
    v0 = np.concatenate([c0[:, 0], c0[:, 1], r0])
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': joint_cons}
    try:
        res = minimize(joint_obj, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            return res.x[:N], res.x[N:2 * N], res.x[2 * N:]
    except Exception:
        pass
    return c0[:, 0], c0[:, 1], r0
