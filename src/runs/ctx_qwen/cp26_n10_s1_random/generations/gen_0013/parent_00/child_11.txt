import numpy as np
from scipy.optimize import linprog, minimize
import warnings
warnings.filterwarnings('ignore')

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

# Precompute constant structure of pairwise constraints matrix
A_STRUC = np.zeros((N_PAIRS, N))
A_STRUC[np.arange(N_PAIRS), I_IDX] = 1.0
A_STRUC[np.arange(N_PAIRS), J_IDX] = 1.0

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    diff = centers[I_IDX] - centers[J_IDX]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    
    bounds = [(0.0, lims[i]) for i in range(N)]
    res = linprog(-np.ones(N), A_ub=A_STRUC, b_ub=dists, bounds=bounds, method='highs')
    
    if res.success and np.isfinite(res.fun):
        return np.maximum(res.x, 0.0), -res.fun
    return np.full(N, 1e-7), 0.0

def local_hill_climb(centers, rng, max_iters=1200):
    """Adaptive coordinate ascent to maximize LP sum of radii."""
    c = centers.copy()
    r, s = solve_lp_radii(c)
    step = 0.025
    
    for _ in range(max_iters):
        i = rng.integers(N)
        old = c[i].copy()
        c[i] += rng.uniform(-step, step, 2)
        c[i] = np.clip(c[i], 1e-4, 0.999)
        
        r_new, s_new = solve_lp_radii(c)
        if s_new > s + 1e-8:
            s = s_new
            r = r_new
            step = min(step * 1.015, 0.05)
        else:
            c[i] = old
            step *= 0.996
    return c, r, s

def obj_slqp(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_slqp(v):
    """Inequality constraints >= 0 for valid packing."""
    cx, cy, r = v[:N], v[N:2*N], v[2*N:]
    c_bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    dr = r[I_IDX] + r[J_IDX]
    
    return np.concatenate([c_bc, dx**2 + dy**2 - dr**2])

    ]

        best_r = np.full(N, 0.085)
        best_s = float(np.sum(best_r))
        
    return best_c, best_r, best_s
