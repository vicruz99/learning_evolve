import numpy as np
from scipy.optimize import minimize, linprog

_N = 26
_TRIU_I, _TRIU_J = np.triu_indices(_N, k=1)
_NUM_PAIRS = len(_TRIU_I)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, and gradient w.r.t centers."""
    n = _N
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    np.fill_diagonal(dists, 1e9)
    
    A_ub = np.zeros((_NUM_PAIRS, n))
    A_ub[np.arange(_NUM_PAIRS), _TRIU_I] = 1.0
    A_ub[np.arange(_NUM_PAIRS), _TRIU_J] = 1.0
    b_ub = dists[_TRIU_I, _TRIU_J]
    
    bounds = [(0.0, lims[i]) for i in range(n)]
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= -1e-7):
            r = np.maximum(res.x, 0.0)
            s = np.sum(r)
            grad = np.zeros((n, 2))
            try:
                marg = res.marginals.ineqlin
                if marg is not None:
                    lam = marg[:_NUM_PAIRS]
                    active = lam > 1e-7
                    if np.any(active):
                        idx = np.where(active)[0]
                        l = lam[idx]
                        i, j = _TRIU_I[idx], _TRIU_J[idx]
                        d = dists[idx] + 1e-9
                        vec = l[:, None] * (centers[i] - centers[j]) / d[:, None]
                        np.add.at(grad, i, vec)
                        np.add.at(grad, j, -vec)
            except Exception:
                pass
            return r, s, grad
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0, np.zeros((n, 2))

def grad_ascent(c0, steps=500):
    """Gradient ascent on centers to maximize LP sum of radii."""
    c = c0.copy()
    r, s_best, _ = solve_lp(c)
    vel = np.zeros_like(c)
    lr = 0.008
    
    for _ in range(steps):
        r, s, g = solve_lp(c)
        if s > s_best: s_best = s
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-8:
            g_dir = g / g_norm
            vel = 0.75 * vel + lr * g_dir
            c_new = c + vel
            c_new = np.clip(c_new, 2e-4, 1.0 - 2e-4)
            
            r_new, s_new, _ = solve_lp(c_new)
            if s_new > s_best + 1e-7:
                c = c_new
                s_best = s_new
                lr = min(lr * 1.03, 0.025)
            else:
                lr *= 0.92
                vel *= 0.6
        if lr < 1e-6: break
    return c, s_best

def obj_slqp(v):
    return -np.sum(v[2 * _N:])

def cons_slqp(v):
    cx = v[:_N]
    cy = v[_N:2 * _N]
    r = v[2 * _N:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[_TRIU_I] - cx[_TRIU_J]
    dy = cy[_TRIU_I] - cy[_TRIU_J]
    dr = r[_TRIU_I] + r[_TRIU_J]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def slqp_polish(c0, r0):
    """Refine centers and radii simultaneously using SLSQP."""
    v0 = np.concatenate([c0[:, 0], c0[:, 1], r0 * 0.97])
    bounds = [(0.0, 1.0)] * (2 * _N) + [(1e-6, 0.5)] * _N
    cons = {'type': 'ineq', 'fun': cons_slqp}
    try:
        res = minimize(obj_slqp, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 4000, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            cx = res.x[:_N]
            cy = res.x[_N:2 * _N]
            r = np.maximum(res.x[2 * _N:], 1e-6)
            return np.column_stack((cx, cy)), r
    except Exception:
        pass
    return c0, r0

def make_hex(rows, r0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    for i, cnt in enumerate(rows):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= _N: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(pts[:_N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_r = None

    ]
    
    for pat in pats:

            span = mx - mn + 1e-9
            rb = (rb - mn) / span * 0.75 + 0.125
            starts.append(np.clip(rb, 0.05, 0.95))

        starts.append(rng.uniform(0.15, 0.85, (_N, 2)))

    # Phase 2: SLSQP Polish

            best_c = c_pol.copy()

            best_sum = s_lp
            best_c = c_pol.copy()

    scale = 1.0
    for i in range(_N):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)

    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    for i in range(_N):
        for j in range(i + 1, _N):
            d = dists[i, j]
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
