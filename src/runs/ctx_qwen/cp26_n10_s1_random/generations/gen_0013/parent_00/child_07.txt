import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

_N = 26
_TRIU_I, _TRIU_J = np.triu_indices(_N, k=1)
_NUM_PAIRS = len(_TRIU_I)

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    A = np.zeros((_NUM_PAIRS, _N))
    A[np.arange(_NUM_PAIRS), _TRIU_I] = 1.0
    A[np.arange(_NUM_PAIRS), _TRIU_J] = 1.0
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b = dists[_TRIU_I, _TRIU_J]
    
    bounds = [(0.0, lims[i]) for i in range(_N)]
    c_obj = -np.ones(_N)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= -1e-7):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(_N, 1e-5), 0.0

def obj_slsqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * _N:])

def cons_slsqp(v):
    """Inequality constraints >= 0 using squared distances for smooth gradients."""
    cx = v[:_N]
    cy = v[_N:2 * _N]
    r = v[2 * _N:]
    
    bc = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[_TRIU_I] - cx[_TRIU_J]
    dy = cy[_TRIU_I] - cy[_TRIU_J]
    dr = r[_TRIU_I] + r[_TRIU_J]
    pc = dx**2 + dy**2 - dr**2
    return np.concatenate([bc, pc])

def run_slsqp(c0, r0):
    """Refine centers and radii simultaneously using SLSQP."""
    v0 = np.concatenate([c0[:, 0], c0[:, 1], r0])
    bounds = [(0.0, 1.0)] * (2 * _N) + [(1e-6, 0.5)] * _N
    cons = {'type': 'ineq', 'fun': cons_slsqp}
    try:
        res = minimize(obj_slsqp, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 6000, 'ftol': 1e-14})
        if np.isfinite(res.fun) and np.min(cons_slsqp(res.x)) > -1e-6:
            return res.x[:_N], res.x[_N:2 * _N], res.x[2 * _N:]
    except Exception:
        pass
    return c0[:, 0], c0[:, 1], r0

def local_search(c_init, r_init, n_iters=1500):
    """Adaptive coordinate search on centers maximizing LP sum of radii."""
    c = c_init.copy()
    r = r_init.copy()
    s_best = np.sum(r)
    step = 0.012
    
    for it in range(n_iters):
        k = np.random.randint(1, 4)
        idx = np.random.choice(_N, k, replace=False)
        old_c = c[idx].copy()
        
        c[idx] += np.random.uniform(-step, step, (k, 2))
        c[idx] = np.clip(c[idx], 1e-4, 0.999)
        
        r_new, s_new = solve_lp_radii(c)
        if s_new > s_best + 1e-8:
            s_best = s_new
            r = r_new
            step = min(step * 1.015, 0.035)
        else:
            c[idx] = old_c
            step *= 0.9985
            
    return c, r, s_best

def generate_starts(rng):

    starts = []
    pats = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 6, 5, 4],

    ]

        starts.append(cfg)
        
    for _ in range(5):
        starts.append(rng.uniform(0.15, 0.85, (_N, 2)))
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
