def compute_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = _N

                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))

    A_ub[np.arange(_NUM_PAIRS), _TRIU_I] = 1.0
    A_ub[np.arange(_NUM_PAIRS), _TRIU_J] = 1.0

        if res.success and np.all(res.x >= -1e-7):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def obj_slqp(v):

    return -np.sum(v[2 * _N:])

def cons_slqp(v):

    dr = r[_TRIU_I] + r[_TRIU_J]

def slqp_polish(c0, r0):

            return res.x[:_N], res.x[_N:2 * _N], res.x[2 * _N:]
    except Exception:
        pass
    return c0[:, 0], c0[:, 1], r0

                else:
                    c[i] = c_old[i]

            _, s = compute_lp(c)
            step *= 0.95 # Cool down
            
    return c, s
