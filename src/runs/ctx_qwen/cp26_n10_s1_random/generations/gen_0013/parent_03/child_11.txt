import numpy as np
from scipy.optimize import linprog, minimize

def build_lp_A(n):
    """Builds the constant inequality matrix for the LP radius solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_radii_lp(centers, n, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0
    x, y = centers[:, 0], centers[:, 1]
    
    b[k:k+n] = x; k += n
    b[k:k+n] = 1.0 - x; k += n
    b[k:k+n] = y; k += n
    b[k:k+n] = 1.0 - y; k += n
    
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    b[k:] = np.hypot(dx, dy)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.05)

def force_sim(centers, radii, steps=300, rng=None):
    """Force-directed simulation to relax overlaps and pack tightly."""
    n = len(radii)
    vel = np.zeros_like(centers)
    dt = 0.005
    for _ in range(steps):
        radii *= 1.0005
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, 1e9)
        r_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlaps = np.maximum(0.0, r_sums - dists)
        inv_dists = np.where(dists > 1e-9, 1.0 / dists, 0.0)

        forces = np.sum(diff * rep[:, :, np.newaxis], axis=1)
        
        wall = np.zeros_like(centers)
        wall[:, 0] += np.clip(radii - centers[:, 0], 0, None) * 50.0
        wall[:, 0] -= np.clip(centers[:, 0] + radii - 1.0, 0, None) * 50.0
        wall[:, 1] += np.clip(radii - centers[:, 1], 0, None) * 50.0
        wall[:, 1] -= np.clip(centers[:, 1] + radii - 1.0, 0, None) * 50.0
        forces += wall
        
        vel = vel * 0.8 + forces * dt
        centers += vel
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    return centers, radii

def coord_ascent(centers, n, idx_i, idx_j, A_ub, rng, max_iter=2000):
    """Hill climbing with LP oracle to maximize sum of radii."""
    r = solve_radii_lp(centers, n, A_ub, idx_i, idx_j)
    if np.all(r < 1e-7):
        return centers, r, 0.0
    best_sum = np.sum(r)
    step = 0.04
    for it in range(max_iter):
        i = rng.integers(n)
        old_pos = centers[i].copy()
        improved = False
        for _ in range(5):
            trial = old_pos + rng.uniform(-step, step, 2)
            trial = np.clip(trial, 1e-4, 1.0 - 1e-4)
            centers[i] = trial
            r_new = solve_radii_lp(centers, n, A_ub, idx_i, idx_j)
            if np.all(r_new > 0):
                s_new = np.sum(r_new)
                if s_new > best_sum + 1e-10:
                    best_sum = s_new
                    r = r_new
                    improved = True
                    break
        if not improved:
            centers[i] = old_pos
            step *= 0.996
        else:
            step = min(step * 1.02, 0.05)
    return centers, r, best_sum

def swap_perturb(centers, n, idx_i, idx_j, A_ub, rng, trials=50):
    """Randomly swaps circle positions to escape local minima."""
    r = solve_radii_lp(centers, n, A_ub, idx_i, idx_j)
    best_sum = np.sum(r)
    for _ in range(trials):
        i, j = rng.choice(n, 2, replace=False)
        centers[i], centers[j] = centers[j].copy(), centers[i].copy()
        r_new = solve_radii_lp(centers, n, A_ub, idx_i, idx_j)
        if np.all(r_new > 0) and np.sum(r_new) > best_sum + 1e-8:
            best_sum = np.sum(r_new)
            r = r_new
    return centers, r, best_sum

def obj_slsqp(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_slsqp(v, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid packing in SLSQP."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    d2 = dx**2 + dy**2
    rs = r[idx_i] + r[idx_j]
    c = np.concatenate([c, d2 - rs**2])
    return c
