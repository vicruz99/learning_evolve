import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Builds the constant inequality matrix for the LP radius solver."""
    M = 4 * n + n * (n - 1) // 2
    A = np.zeros((M, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp(centers, A_ub, b_template, pair_i, pair_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    b = b_template.copy()
    b[:n] = centers[:, 0]
    b[n:2*n] = 1.0 - centers[:, 0]
    b[2*n:3*n] = centers[:, 1]
    b[3*n:4*n] = 1.0 - centers[:, 1]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b[4*n:] = dists[pair_i, pair_j]
    
    bounds_r = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=bounds_r, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-7)

def neg_lp_sum(c_flat, n, A_ub, b_template, pair_i, pair_j):
    """Objective for center optimization: minimize negative LP sum of radii."""
    centers = c_flat.reshape(n, 2)
    r = solve_lp(centers, A_ub, b_template, pair_i, pair_j)
    return -np.sum(r)

def local_center_obj(x, centers, idx, A_ub, b_template, pair_i, pair_j):
    """Objective for coordinate ascent: optimizes single circle position."""
    centers[idx] = np.asarray(x).flatten()
    r = solve_lp(centers, A_ub, b_template, pair_i, pair_j)
    return -np.sum(r)

def get_hex_init(n, row_counts, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(row_counts):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    pts = np.array(pts[:n])
    if len(pts) > 0:
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        span = mx - mn
        if span.max() > 1e-9:
            pts = (pts - mn) / span * 0.7 + 0.15
    return pts

def obj_joint(v, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_joint(v, n):
    """Inequality constraints >= 0 for valid packing in SLSQP."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    d2 = dx**2 + dy**2
    rs = r[i_idx] + r[j_idx]
    return np.concatenate([c, d2 - rs**2])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    pair_i, pair_j = np.triu_indices(n, k=1)
    A_ub = build_lp_matrix(n)
    b_template = np.zeros(4*n + n*(n-1)//2)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    configs = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [7, 6, 6, 7],
        [5, 5, 5, 5, 6], [6, 5, 5, 6, 4]
    ]
    
    for pat in patterns:
        if sum(pat) < n: continue
        base = get_hex_init(n, pat, 0.10, rng)
        configs.append(base)
        for _ in range(3):
            p = base + rng.uniform(-0.03, 0.03, base.shape)
            configs.append(np.clip(p, 0.05, 0.95))

        configs.append(rng.uniform(0.15, 0.85, (n, 2)))

                           method='L-BFGS-B', bounds=bounds_centers,

            if np.isfinite(res.fun):
                c_opt = res.x.reshape(n, 2)
                r_lp = solve_lp(c_opt, A_ub, b_template, pair_i, pair_j)

                    best_sum = s_lp
                    best_centers = c_opt.copy()
                    best_radii = r_lp.copy()

                best_centers = curr_c.copy()
                best_radii = curr_r.copy()

                curr_c[i] = old_pos

                r_s = res_slqp.x[2*n:]
                r_lp = solve_lp(c_s, A_ub, b_template, pair_i, pair_j)
                s_lp = np.sum(r_lp)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c_s.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass
            
    # Fallback configuration
    if best_centers is None:
        best_centers = get_hex_init(n, [6, 5, 6, 5, 4], 0.10, rng)
        best_radii = solve_lp(best_centers, A_ub, b_template, pair_i, pair_j)
        best_sum = np.sum(best_radii)
        
    # Final strict numerical scaling to guarantee validator tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
