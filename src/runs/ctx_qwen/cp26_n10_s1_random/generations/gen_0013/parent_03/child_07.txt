from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Precomputes the constant inequality matrix for the LP radius solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp(centers, A_ub):

    n = centers.shape[0]
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0

        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1

            b[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            k += 1

        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return res.x
    except Exception:
        pass

def objective_joint(v, n):

def constraints_joint(v, n, i_idx, j_idx):

    c = v[:2 * n].reshape(n, 2)

    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])

    con = np.concatenate([con, dist - r[i_idx] - r[j_idx]])
    return con
