import numpy as np
from scipy.optimize import linprog

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
N_PAIRS = len(IDX_I)

# Precompute the constant inequality matrix for the radius LP
# Constraints: A @ r <= b
# Rows 0..N-1:      r_i <= x_i
# Rows N..2N-1:     r_i <= 1 - x_i
# Rows 2N..3N-1:    r_i <= y_i
# Rows 3N..4N-1:    r_i <= 1 - y_i
# Rows 4N..end:     r_i + r_j <= dist(i, j)
A_LUB = np.zeros((4 * N + N_PAIRS, N))
for i in range(N):
    A_LUB[i, i] = 1.0
    A_LUB[N + i, i] = 1.0
    A_LUB[2 * N + i, i] = 1.0
    A_LUB[3 * N + i, i] = 1.0
for k, (i, j) in enumerate(zip(IDX_I, IDX_J)):
    A_LUB[4 * N + k, i] = 1.0
    A_LUB[4 * N + k, j] = 1.0

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Boundary limits
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)

    b_ub[N:2*N] = 1.0 - centers[:, 0]
    b_ub[2*N:3*N] = centers[:, 1]
    b_ub[3*N:4*N] = 1.0 - centers[:, 1]
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub[4*N:] = dists[IDX_I, IDX_J]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_LUB, b_ub=b_ub, bounds=(0, None), method='highs')

            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0
