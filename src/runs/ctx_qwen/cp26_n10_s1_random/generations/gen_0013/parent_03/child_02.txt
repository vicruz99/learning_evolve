from scipy.optimize import minimize, linprog

def solve_lp(centers, A_ub, m_boundary):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    m = A_ub.shape[0]
    b = np.zeros(m)
    
    # Boundary constraints RHS
    b[:m_boundary:4] = centers[:, 0]
    b[1:m_boundary:4] = 1.0 - centers[:, 0]
    b[2:m_boundary:4] = centers[:, 1]
    b[3:m_boundary:4] = 1.0 - centers[:, 1]
    
    # Pairwise distance constraints RHS
    idx_i, idx_j = np.triu_indices(n, k=1)
    diffs = centers[idx_i] - centers[idx_j]
    b[m_boundary:] = np.sqrt(np.sum(diffs**2, axis=1))
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def obj_slsqp(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def cons_slsqp(v, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    
    con = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[idx_i] + r[idx_j]
    con = np.concatenate([con, dist - r_sum])
    return con

def hex_init(n, r0):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    row = 0
    while len(pts) < n:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x <= 1.0 - r0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    return np.array(pts[:n])
