import numpy as np
from scipy.optimize import minimize, linprog

# Global constants for LP setup (N=26)
N = 26
M_BOUND = 4 * N
M_PAIR = N * (N - 1) // 2
M = M_BOUND + M_PAIR
I_IDX, J_IDX = np.triu_indices(N, k=1)

# Precompute constant inequality matrix A_ub for LP: r_i + r_j <= dist_ij
A_UB = np.zeros((M, N))
for i in range(N):
    A_UB[i, i] = 1.0
    A_UB[N + i, i] = 1.0
    A_UB[2 * N + i, i] = 1.0
    A_UB[3 * N + i, i] = 1.0
A_UB[M_BOUND:, I_IDX] = 1.0
A_UB[M_BOUND:, J_IDX] = 1.0

BOUNDS_R = [(0.0, None)] * N

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    b = np.empty(M)
    b[:N] = centers[:, 0]
    b[N:2 * N] = 1.0 - centers[:, 0]
    b[2 * N:3 * N] = centers[:, 1]
    b[3 * N:] = 1.0 - centers[:, 1]
    
    dx = centers[I_IDX, 0] - centers[J_IDX, 0]
    dy = centers[I_IDX, 1] - centers[J_IDX, 1]
    b[M_BOUND:] = np.hypot(dx, dy)
    
    try:
        res = linprog(-np.ones(N), A_ub=A_UB, b_ub=b, bounds=BOUNDS_R, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0

def obj_func(x):
    """Objective for center optimization: minimize negative sum of radii."""
    return -solve_lp(x.reshape(N, 2))[1]

def make_hex(rows, r0):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for row_i, cnt in enumerate(rows):
        shift = r0 if row_i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    
    inits = []
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [5,6,4,6,5], [5,4,6,5,6], [6,5,5,6,4],
        [7,6,6,7], [8,6,6,6], [9,5,6,6], [6,5,6,4,5],
        [5,7,5,6,3], [4,5,6,5,6], [6,6,6,5,3], [7,7,6,6]
    ]
    
    for p in patterns:
        if sum(p) < N: continue
        base = make_hex(p, 0.095)
        mn = base.min(axis=0)
        mx = base.max(axis=0)
        span = mx - mn
        if span[0] > 1e-5: base[:, 0] = (base[:, 0] - mn[0]) / span[0] * 0.8 + 0.1
        if span[1] > 1e-5: base[:, 1] = (base[:, 1] - mn[1]) / span[1] * 0.8 + 0.1
        inits.append(base)
        
        for _ in range(3):
            pert = base + rng.uniform(-0.03, 0.03, (N, 2))
            inits.append(np.clip(pert, 0.05, 0.95))
            
    for _ in range(10):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    best_sum = -1.0
    best_c = None
    best_r = None
    
    bounds_c = [(0.0, 1.0)] * (2 * N)
    
    # Phase 1: L-BFGS-B center optimization from diverse starts
    for cfg in inits:
        cfg = np.clip(cfg, 0.005, 0.995)
        try:
            res = minimize(obj_func, cfg.flatten(), method='L-BFGS-B', bounds=bounds_c,
                           options={'maxiter': 1500, 'ftol': 1e-14})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt = solve_lp(c_opt)
            
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            continue
            
    # Phase 2: Coordinate Ascent Hill-Climbing to escape local minima & refine boundaries
    if best_c is not None:
        curr_c = best_c.copy()
        curr_r = best_r.copy()
        curr_s = best_sum
        step = 0.015
        for _ in range(2500):
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            improved = False
            for _ in range(6):
                trial = old + rng.uniform(-step, step, 2)
                trial = np.clip(trial, 0.005, 0.995)
                curr_c[idx] = trial
                r_try, s_try = solve_lp(curr_c)
                if s_try > curr_s + 1e-8:
                    curr_s = s_try
                    curr_r = r_try.copy()
                    improved = True
                    break
            if not improved:
                curr_c[idx] = old
                step *= 0.9985
                if step < 1e-4: step = 1e-4
            else:
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_c = curr_c.copy()
                    best_r = curr_r.copy()
                    
    # Phase 3: Strict numerical safety scaling
    scale = 1.0
    for i in range(N):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_r *= scale * 0.9999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
