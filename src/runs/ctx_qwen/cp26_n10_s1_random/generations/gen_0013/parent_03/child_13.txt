def build_A_ub(n):
    """Builds the constant inequality matrix for the LP radius solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp_radii(centers, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    b = np.zeros(A_ub.shape[0])
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            b[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    # Fallback
    r_fall = np.full(n, 1e-7)
    return r_fall, 0.0

def obj_joint(v, n):

    rs = r[idx[0]] + r[idx[1]]
    c = np.concatenate([c, d2 - rs**2])
