import numpy as np
from scipy.optimize import minimize, linprog

def build_A_ub(n):
    """Builds the constant inequality matrix for the LP radius solver."""
    m_pairs = n * (n - 1) // 2
    m = 4 * n + m_pairs
    A = np.zeros((m, n))
    idx = 0
    for i in range(n):
        A[idx, i] = 1.0; idx += 1
        A[idx, i] = 1.0; idx += 1
        A[idx, i] = 1.0; idx += 1
        A[idx, i] = 1.0; idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
    return A

def solve_lp(centers, A_ub, n, i_idx, j_idx):
    """Solves LP to maximize sum of radii for fixed centers using vectorized b construction."""
    m = A_ub.shape[0]
    b = np.zeros(m)
    b[:n] = centers[:, 0]
    b[n:2*n] = 1.0 - centers[:, 0]
    b[2*n:3*n] = centers[:, 1]
    b[3*n:4*n] = 1.0 - centers[:, 1]
    diffs = centers[i_idx] - centers[j_idx]
    b[4*n:] = np.hypot(diffs[:, 0], diffs[:, 1])
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def obj_func(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def con_func(v, n, i_idx, j_idx):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    d2 = dx**2 + dy**2
    rs = r[i_idx] + r[j_idx]
    c = np.concatenate([c, d2 - rs**2])
    return c
