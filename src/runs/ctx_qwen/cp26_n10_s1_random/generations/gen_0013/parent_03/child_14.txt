import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Builds the constant inequality matrix for the LP radius solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_radii(centers, n, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
    for k_i, (i, j) in enumerate(zip(idx_i, idx_j)):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b[4 * n + k_i] = np.hypot(dx, dy)
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-7)

def force_relax(centers, n, steps=400):
    """Force-directed simulation to resolve overlaps and create a valid initial packing."""
    r = np.full(n, 0.06)
    vel = np.zeros_like(centers)
    dt = 0.004
    for _ in range(steps):
        r *= 1.0015
        forces = np.zeros_like(centers)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, 1e9)
        r_sums = r[:, np.newaxis] + r[np.newaxis, :]
        overlap = np.maximum(0.0, r_sums - dists)
        inv_d = np.where(dists > 1e-9, 1.0 / dists, 0.0)
        rep = overlap * inv_d * 100.0
        forces += np.sum(diff * rep[:, :, np.newaxis], axis=1)
        
        wf = np.zeros_like(centers)
        wf[:, 0] += np.clip(r - centers[:, 0], 0, None) * 50.0
        wf[:, 0] -= np.clip(centers[:, 0] + r - 1.0, 0, None) * 50.0
        wf[:, 1] += np.clip(r - centers[:, 1], 0, None) * 50.0
        wf[:, 1] -= np.clip(centers[:, 1] + r - 1.0, 0, None) * 50.0
        forces += wf
        
        vel = vel * 0.85 + forces * dt
        centers += vel
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    return centers, r

def obj_slqp(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def con_slqp(v, n, idx_i, idx_j):
    """Inequality constraints >= 0 for valid packing."""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    d2 = dx**2 + dy**2
    rs = r[idx_i] + r[idx_j]
    c = np.concatenate([c, d2 - rs**2])
    return c

def local_search(centers, radii, n, A_ub, idx_i, idx_j, rng):
    """Coordinate ascent on centers using LP oracle for radius evaluation."""
    curr_c = centers.copy()
    curr_r = radii.copy()
    curr_sum = np.sum(curr_r)
    step = 0.025
    for _ in range(1800):
        i = rng.integers(n)
        old_c = curr_c[i].copy()
        move = rng.uniform(-step, step, 2)
        new_c = np.clip(old_c + move, 1e-4, 1.0 - 1e-4)
        curr_c[i] = new_c
        new_r = solve_radii(curr_c, n, A_ub, idx_i, idx_j)
        new_sum = np.sum(new_r)
        if new_sum > curr_sum + 1e-8:
            curr_sum = new_sum
            curr_r = new_r
        else:
            curr_c[i] = old_c
        step *= 0.9995
    return curr_c, curr_r, curr_sum

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    A_ub = build_lp_matrix(n)
    idx_i, idx_j = np.triu_indices(n, k=1)
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    configs = []
    # Hexagonal patterns with varying row distributions
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], [7,6,6,7], [8,6,6,6], [6,6,5,5,4], [5,6,4,6,5]]
    for pat in patterns:
        pts = []
        y = 0.09
        for ri, cnt in enumerate(pat):
            shift = 0.09 if ri % 2 else 0.0
            x = 0.09 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 0.18
            y += 0.155
        configs.append(np.array(pts[:n]))
        
    # Corner/Edge biased starts
    for _ in range(6):
        c = rng.uniform(0.15, 0.85, (n, 2))
        c[:4] = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        configs.append(c)
        
    # Pure random starts
    for _ in range(8):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    bounds_v = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    
    for cfg in configs:
        # Phase 1: Force relaxation to ensure feasibility
        c_rel, r_rel = force_relax(cfg.copy(), n)
        
        # Phase 2: Joint SLSQP optimization
        v0 = np.concatenate([c_rel.flatten(), r_rel * 0.95])
        try:
            res = minimize(obj_slqp, v0, args=(n,), method='SLSQP', bounds=bounds_v,
                          constraints={'type': 'ineq', 'fun': con_slqp, 'args': (n, idx_i, idx_j)},
                          options={'maxiter': 7000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_opt = np.column_stack((res.x[:n], res.x[n:2*n]))
                r_opt = np.maximum(res.x[2*n:], 1e-7)
                
                # Phase 3: Exact LP refinement
                r_lp = solve_radii(c_opt, n, A_ub, idx_i, idx_j)
                s_lp = np.sum(r_lp)
                
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_c = c_opt.copy()
                    best_r = r_lp.copy()
                    
                # Phase 4: Local search with LP oracle
                c_ls, r_ls, s_ls = local_search(c_opt.copy(), r_opt.copy(), n, A_ub, idx_i, idx_j, rng)
                if s_ls > best_sum:
                    best_sum = s_ls
                    best_c = c_ls.copy()
                    best_r = r_ls.copy()
        except Exception:
            continue
            
    # Fallback if all optimizations fail
    if best_c is None:
        best_c = configs[0]
        best_r = solve_radii(best_c, n, A_ub, idx_i, idx_j)
        best_sum = np.sum(best_r)
        
    # Final strict numerical scaling to guarantee validator tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999999
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
