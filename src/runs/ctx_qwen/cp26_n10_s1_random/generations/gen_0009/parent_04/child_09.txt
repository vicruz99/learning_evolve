import numpy as np
from scipy.optimize import minimize, linprog

def get_lp_sum(centers, n, i_idx, j_idx, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    bounds = [(0.0, lim) for lim in limits]
    
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def obj_centers(centers_flat, n, i_idx, j_idx, A_ub):
    """Objective for center optimization: maximize LP sum of radii."""
    centers = centers_flat.reshape(n, 2)
    _, s = get_lp_sum(centers, n, i_idx, j_idx, A_ub)
    return -s

def obj_joint(vars_arr, n):
    """Objective for joint SLSQP: maximize sum of radii."""
    return -np.sum(vars_arr[2*n:])

def cons_joint(vars_arr, n, i_idx, j_idx):
    """Inequality constraints for joint SLSQP: >= 0."""
    cx, cy, r = vars_arr[:n], vars_arr[n:2*n], vars_arr[2*n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dr = r[i_idx] + r[j_idx]
    return np.concatenate([c, dx**2 + dy**2 - dr**2])

def gen_hex(n, rows, r0):
    """Generates hexagonal lattice initialization."""
    pts = []
    y = r0
    for i, cnt in enumerate(rows):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) < n:
                pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(pts[:n])
