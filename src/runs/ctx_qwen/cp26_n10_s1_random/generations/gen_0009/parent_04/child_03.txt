import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

def solve_lp(centers, n, A_ub, i_idx, j_idx):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-12)) for l in lims]
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            return res.x, -res.fun
    except Exception:
        pass
    return np.maximum(lims, 0.0) * 0.95, np.sum(np.maximum(lims, 0.0) * 0.95)

def force_sim(c, n, steps=2500):
    """Force-directed simulation to spread and pack circles tightly."""
    c = c.copy()
    r = np.full(n, 0.025)
    v = np.zeros_like(c)
    dt = 0.0012
    damp = 0.80
    k_rep = 120.0
    k_wall = 300.0
    
    for _ in range(steps):
        r *= 1.00015
        f = np.zeros_like(c)
        
        for i in range(n):
            for j in range(i + 1, n):
                d = c[i] - c[j]
                dist = np.linalg.norm(d)
                req = r[i] + r[j]
                if dist < req and dist > 1e-9:
                    fm = k_rep * (req - dist) / dist
                    f[i] += fm * d
                    f[j] -= fm * d
                    
            for k in range(2):
                if c[i, k] < r[i]:
                    f[i, k] += k_wall * (r[i] - c[i, k])
                elif c[i, k] > 1.0 - r[i]:
                    f[i, k] -= k_wall * (c[i, k] - (1.0 - r[i]))
                    
        v = damp * v + f * dt
        c += v
        c = np.clip(c, 0.0, 1.0)
    return c, r

def nm_obj(x, n, A_ub, i_idx, j_idx):
    """Objective for Nelder-Mead: maximize LP sum of radii."""
    c = np.clip(x.reshape(n, 2), 0.002, 0.998)
    _, s = solve_lp(c, n, A_ub, i_idx, j_idx)
    return -s

def obj_slsqp(v, n):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(v[2 * n:])

def con_slsqp(v, n, i_idx, j_idx):
    """Constraints for SLSQP: boundary and non-overlap."""
    cx, cy, rr = v[:n], v[n:2 * n], v[2 * n:]
    c_list = []
    c_list.append(cx - rr)
    c_list.append(1.0 - cx - rr)
    c_list.append(cy - rr)
    c_list.append(1.0 - cy - rr)
    
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dr = rr[i_idx] + rr[j_idx]
    c_list.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c_list)

def run_packing():
    n = 26
    i_idx, j_idx = np.triu_indices(n, k=1)
    m = len(i_idx)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), i_idx] = 1.0
    A_ub[np.arange(m), j_idx] = 1.0
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    inits = []
    # Hexagonal row distributions
    rows_list = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4], 
        [6, 6, 6, 4, 4], [5, 5, 5, 5, 6]
    ]
    for rows in rows_list:
        pts = []
        y = 0.085
        for idx, cnt in enumerate(rows):
            shift = 0.095 if idx % 2 == 1 else 0.0
            w = (cnt - 1) * 0.175
            x0 = 0.5 - w / 2.0 + shift
            for k in range(cnt):
                if len(pts) < n:
                    pts.append([x0 + k * 0.175, y])
            y += 0.152
        inits.append(np.array(pts[:n]))
        
    # Random starts
    rng = np.random.default_rng(42)
    for _ in range(12):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))

        if s_lp > best_sum:
            best_sum = s_lp
            best_c = c_f.copy()
            best_r = r_lp.copy()

        try:
            res_nm = minimize(nm_obj, c_f.flatten(), args=(n, A_ub, i_idx, j_idx), method='Nelder-Mead',
                              options={'maxiter': 2500, 'xatol': 1e-7, 'fatol': 1e-9})
            c_nm = np.clip(res_nm.x.reshape(n, 2), 0.002, 0.998)

                best_sum = s_nm

                best_r = r_nm.copy()

            pass

                r_s, s_s = solve_lp(c_s, n, A_ub, i_idx, j_idx)

                    best_c = c_s.copy()
                    best_r = r_s.copy()
        except Exception:
            pass

    # Fallback safety
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum = solve_lp(best_c, n, A_ub, i_idx, j_idx)
        
    # Final strict scaling to guarantee validator tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
        if r > 1e-12:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    dx = best_c[i_idx, 0] - best_c[j_idx, 0]
    dy = best_c[i_idx, 1] - best_c[j_idx, 1]
    d = np.sqrt(dx**2 + dy**2)
    rs = best_r[i_idx] + best_r[j_idx]
    if np.any(rs > 1e-12):
        scale = min(scale, np.min(d / np.maximum(rs, 1e-12)))
        
    best_r *= scale * 0.9999995
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
