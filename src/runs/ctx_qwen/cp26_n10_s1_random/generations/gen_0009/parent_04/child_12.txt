import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

def get_hex_config(n, r0, seed_shift=0.0):
    """Generate a hexagonal lattice configuration scaled to fit the square."""
    pts = []
    y = r0
    row = 0
    while len(pts) < n:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift + seed_shift
        while x + r0 <= 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    return np.array(pts[:n])

def force_relax(centers, radii, n, steps=1000):
    """Force-directed relaxation to resolve overlaps and pack tightly."""
    c = centers.copy()
    r = radii.copy()
    dt = 0.008
    k_rep = 120.0
    k_wall = 250.0
    vel = np.zeros_like(c)
    damping = 0.80
    
    for step in range(steps):
        # Gradually expand radii to push boundaries
        r *= 1.0 + 0.00005 * (1.0 - step/steps)
        forces = np.zeros_like(c)
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = c[i] - c[j]
                dist = np.linalg.norm(diff)
                req = r[i] + r[j]
                if dist < req and dist > 1e-7:
                    overlap = req - dist
                    f = k_rep * overlap / dist
                    forces[i] += f * diff
                    forces[j] -= f * diff
                    
        # Wall repulsion
        for i in range(n):
            for d in [0, 1]:
                if c[i, d] - r[i] < 0:
                    forces[i, d] += k_wall * (r[i] - c[i, d])
                if c[i, d] + r[i] > 1.0:
                    forces[i, d] -= k_wall * (c[i, d] + r[i] - 1.0)
                    
        vel = damping * vel + forces * dt
        c += vel
        c = np.clip(c, 0.0, 1.0)
    return c, r

def solve_lp_radii(centers, n, i_idx, j_idx):
    """Solves LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(lim, 1e-9)) for lim in limits]
    
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    m = len(i_idx)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), i_idx] = 1.0
    A_ub[np.arange(m), j_idx] = 1.0
    b_ub = dists
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun) and np.all(np.isfinite(res.x)):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def joint_obj(vars_arr, n):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars_arr[2*n:])

def joint_cons(vars_arr, n, i_idx, j_idx):
    """Computes inequality constraints >= 0 for valid packing."""
    xs = vars_arr[:n]
    ys = vars_arr[n:2*n]
    rs = vars_arr[2*n:]
    
    # Boundary constraints
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = xs[i_idx] - xs[j_idx]
    dy = ys[i_idx] - ys[j_idx]
    dr = rs[i_idx] + rs[j_idx]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
