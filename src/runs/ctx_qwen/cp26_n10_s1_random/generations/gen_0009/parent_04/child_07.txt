import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, n, i_idx, j_idx):
    """Solves LP to maximize sum of radii for fixed centers."""
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 0.0)
    
    m = len(i_idx)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), i_idx] = 1.0
    A_ub[np.arange(m), j_idx] = 1.0
    
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    dists = np.sqrt(dx**2 + dy**2)
    b_ub = dists
    
    bounds = [(0.0, lim) if lim > 0 else (0.0, 1e-9) for lim in limits]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun) and np.all(np.isfinite(res.x)):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def get_joint_constraints(vars_arr, n, i_idx, j_idx):
    """Computes inequality constraints >= 0 for valid packing."""
    xs = vars_arr[:n]
    ys = vars_arr[n:2*n]
    rs = vars_arr[2*n:]
    
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    dx = xs[i_idx] - xs[j_idx]
    dy = ys[i_idx] - ys[j_idx]
    dr = rs[i_idx] + rs[j_idx]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def objective_joint(vars_arr, n):
    """Objective: maximize sum of radii -> minimize negative sum"""
    return -np.sum(vars_arr[2*n:])

def generate_hex_config(row_counts, r0=0.09, perturb=0.0, rng=None):
    """Generate hexagonal configuration with given row counts."""
    n = 26
    pts = []
    y = r0
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        if len(pts) >= n:
            break
    
    while len(pts) < n:
        pts.append([0.5, 0.5])
    
    cfg = np.array(pts[:n])
    if perturb > 0 and rng is not None:
        cfg += rng.uniform(-perturb, perturb, cfg.shape)
    return np.clip(cfg, 0.01, 0.99)

def hill_clamp_lp(centers, n, i_idx, j_idx, rng, n_iter=1500, step_init=0.05):
    """Hill climbing on centers using LP objective."""
    curr = centers.copy()
    r_best, s_best = solve_lp_radii(curr, n, i_idx, j_idx)
    if r_best is None:
        return centers, solve_lp_radii(centers, n, i_idx, j_idx)
    
    step = step_init
    
    for it in range(n_iter):
        idx = rng.integers(n)
        old = curr[idx].copy()
        best_pos = old.copy()
        best_sum = s_best
        
        # Try multiple random moves
        for _ in range(15):
            new_pos = old + rng.uniform(-step, step, 2)
            new_pos = np.clip(new_pos, 0.005, 0.995)
            test = curr.copy()
            test[idx] = new_pos
            r_t, s_t = solve_lp_radii(test, n, i_idx, j_idx)
            if r_t is not None and s_t > best_sum:
                best_sum = s_t
                best_pos = new_pos.copy()
        
        curr[idx] = best_pos
        if best_sum > s_best:
            r_best, s_best = solve_lp_radii(curr, n, i_idx, j_idx)
        
        # Decay step size
        step *= 0.9975
    
    return curr, r_best, s_best
