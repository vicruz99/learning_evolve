import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    bounds = [(0.0, l) for l in lims]
    
    dx = centers[idx_i, 0] - centers[idx_j, 0]
    dy = centers[idx_i, 1] - centers[idx_j, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    m = len(idx_i)
    A = np.zeros((m, n))
    A[np.arange(m), idx_i] = 1.0
    A[np.arange(m), idx_j] = 1.0
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=dists, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def nelder_objective(x, n, idx_i, idx_j):
    """Objective for Nelder-Mead: minimize negative sum of radii."""
    centers = x.reshape(n, 2)
    _, s = solve_lp(centers, idx_i, idx_j)
    return -s

def get_joint_constraints(vars_arr, n, idx_i, idx_j):
    """Computes inequality constraints >= 0 for valid packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[idx_i] - cx[idx_j]
    dy = cy[idx_i] - cy[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    idx_i, idx_j = np.triu_indices(n, k=1)
    
    rng = np.random.default_rng(42)
    configs = []
    
    # Hexagonal patterns with varied row distributions
    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4]
    ]
    for rd in row_patterns:
        pts = []
        y = 0.095
        for r_idx, cnt in enumerate(rd):
            shift = 0.095 if r_idx % 2 == 1 else 0.0
            x = 0.095 + shift
            for _ in range(cnt):
                if len(pts) < n:
                    pts.append([x, y])
                x += 0.185
            y += 0.16
            if len(pts) >= n: break
        configs.append(np.array(pts[:n]))
        
    # Random dense starts to escape structured minima
    for _ in range(8):
        configs.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    for cfg in configs:
        centers = cfg.copy()
        r, s = solve_lp(centers, idx_i, idx_j)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = r.copy()
            
        curr = centers.copy()
        curr_sum = s
        step = 0.04
        
        # Hill climbing / Simulated Annealing on centers
        for it in range(1500):
            idx = rng.integers(n)
            old = curr[idx].copy()
            
            curr[idx] += rng.uniform(-step, step, 2)
            curr[idx] = np.clip(curr[idx], 1e-4, 0.9999)
            
            r_new, s_new = solve_lp(curr, idx_i, idx_j)
            
            accept = False
            if s_new > curr_sum:
                accept = True
            else:
                temp = max(0.005, 1.0 - it / 1500)
                if rng.random() < np.exp((s_new - curr_sum) / temp):
                    accept = True
                    
            if accept:
                curr_sum = s_new
                if s_new > best_sum:
                    best_sum = s_new
                    best_centers = curr.copy()
                    best_radii = r_new.copy()
            else:
                curr[idx] = old
                
            step *= 0.997
            
        # Local polish with Nelder-Mead on centers
        try:
            res = minimize(nelder_objective, curr.flatten(), args=(n, idx_i, idx_j),
                           method='Nelder-Mead', options={'maxiter': 3000, 'xatol': 1e-8, 'fatol': 1e-10})
            c_opt = res.x.reshape(n, 2)
            r_opt, s_opt = solve_lp(c_opt, idx_i, idx_j)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            pass

    # Final joint SLSQP polish for precision
    if best_centers is not None:
        x0 = np.zeros(3 * n)
        x0[:n] = best_centers[:, 0]
        x0[n:2*n] = best_centers[:, 1]
        x0[2*n:] = best_radii * 0.95
        
        def obj_slq(v):
            return -np.sum(v[2*n:])
            
        bounds_slq = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
        try:
            res = minimize(obj_slq, x0, method='SLSQP', bounds=bounds_slq,
                           constraints={'type': 'ineq', 'fun': get_joint_constraints, 'args': (n, idx_i, idx_j)},
                           options={'maxiter': 10000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_f = np.column_stack((res.x[:n], res.x[n:2*n]))
                r_f, s_f = solve_lp(c_f, idx_i, idx_j)
                if s_f > best_sum:
                    best_sum = s_f
                    best_centers = c_f.copy()
                    best_radii = r_f.copy()
        except Exception:
            pass

    # Fallback if all optimizations fail unexpectedly
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum = solve_lp(best_centers, idx_i, idx_j)
        
    # Strict safety scaling to guarantee numerical validity within 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    dx = best_centers[idx_i, 0] - best_centers[idx_j, 0]
    dy = best_centers[idx_i, 1] - best_centers[idx_j, 1]
    d = np.sqrt(dx**2 + dy**2)
    rs = best_radii[idx_i] + best_radii[idx_j]
    if np.any(rs > 1e-12):
        scale = min(scale, np.min(d / np.maximum(rs, 1e-12)))
        
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
