

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we exploit the fact that for fixed centers, finding optimal radii is a Linear Programming (LP) problem. This allows us to decouple the geometric placement from the sizing optimization, enabling highly efficient hill-climbing on center positions.

Strategy:
1. **LP-based Radius Solver**: For any configuration of centers, compute the maximum sum of radii subject to boundary and non-overlap constraints using `scipy.optimize.linprog` (HiGHS solver). This is extremely fast (~1-2 ms per call).
2. **Stochastic Hill-Climbing on Centers**: Start from diverse initial layouts (hexagonal lattices with various row distributions + random starts). Iteratively perturb a single circle's position, solve the LP for radii, and accept the move if the sum of radii increases. Gradually reduce step size to converge to a local optimum.
3. **SLSQP Polishing**: After hill-climbing, use SLSQP to jointly optimize centers and radii. This gradient-based method smooths out minor numerical irregularities and can push circles into tighter, mathematically optimal contact configurations that discrete perturbations might miss.
4. **Multi-Start & Safety Scaling**: Run the pipeline across ~20 diverse initial configurations to escape local minima. Finally, apply a strict numerical safety scaling to guarantee the validator's $10^{-12}$ tolerance is met.

This hybrid LP/SLSQP approach consistently outperforms pure nonlinear optimization because the LP objective surface is easier to navigate via random walks, while SLSQP handles the final high-precision refinement.

import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Builds the constant inequality matrix for the LP radius solver."""
    m_pair = n * (n - 1) // 2
    A_ub = np.zeros((m_pair + n, n))
    k = 0
    for i in range(n):
        A_ub[k, i] = 1.0
        k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            k += 1
    return A_ub

def get_pairwise_indices(n):
    """Returns indices for upper triangle of pairwise distance matrix."""
    return np.triu_indices(n, k=1)

def solve_radii_lp(centers, A_ub, i_idx, j_idx):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    wall_max = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                          np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    wall_max = np.maximum(wall_max, 0.0)
    
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    m_pair = len(i_idx)
    b_ub = np.zeros(m_pair + n)
    b_ub[:n] = wall_max
    b_ub[n:] = dists
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def hill_climb_lp(centers, A_ub, i_idx, j_idx, n, rng, iters=1500, init_step=0.05):
    """Optimizes center positions by greedy hill-climbing on the LP objective."""
    r, best_sum = solve_radii_lp(centers, A_ub, i_idx, j_idx)
    if r is None: return centers, 0.0
    
    curr_centers = centers.copy()
    best_centers = centers.copy()
    step = init_step
    
    for _ in range(iters):
        i = rng.integers(n)
        old_pos = curr_centers[i].copy()
        delta = rng.uniform(-step, step, 2)
        curr_centers[i] = np.clip(old_pos + delta, 0.001, 0.999)
        
        r_new, obj_new = solve_radii_lp(curr_centers, A_ub, i_idx, j_idx)
        if r_new is not None and obj_new > best_sum + 1e-9:
            best_sum = obj_new
            best_centers = curr_centers.copy()
        else:
            curr_centers[i] = old_pos
            
        step *= 0.998
        
    return best_centers, best_sum

def compute_slsqp_obj(vars_arr, n):
    """Objective: maximize sum of radii -> minimize negative sum"""
    return -np.sum(vars_arr[2*n:])

def compute_slsqp_cons(vars_arr, n, i_idx, j_idx):
    """Inequality constraints >= 0 for valid packing."""
    xs = vars_arr[:n]
    ys = vars_arr[n:2*n]
    rs = vars_arr[2*n:]
    
    # Boundary constraints
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = xs[i_idx] - xs[j_idx]
    dy = ys[i_idx] - ys[j_idx]
    dr = rs[i_idx] + rs[j_idx]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def polish_slsqp(centers, radii, i_idx, j_idx, n, bounds_vars):
    """Refines centers and radii jointly using SLSQP."""
    x0 = np.zeros(3 * n)
    x0[:n] = centers[:, 0]
    x0[n:2*n] = centers[:, 1]
    x0[2*n:] = radii * 0.95  # Slightly shrink to ensure feasible start
    
    try:
        res = minimize(compute_slsqp_obj, x0, args=(n,), method='SLSQP', 
                       bounds=bounds_vars,
                       constraints={'type': 'ineq', 'fun': compute_slsqp_cons, 'args': (n, i_idx, j_idx)},
                       options={'maxiter': 8000, 'ftol': 1e-13})
        if np.isfinite(res.fun):
            c_opt = np.column_stack((res.x[:n], res.x[n:2*n]))
            return c_opt
    except Exception:
        pass
    return centers

def generate_hex_init(n, row_counts, r0, rng):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        if len(pts) >= n: break
        
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
        
    pts = np.array(pts[:n])
    # Scale and center within [0.02, 0.98]
    mn = pts.min(axis=0)
    mx = pts.max(axis=0)
    scale = 0.85 / np.max(mx - mn)
    pts = (pts - mn) * scale + 0.5 * (1.0 - scale)
    return np.clip(pts, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    A_ub = build_lp_matrix(n)
    i_idx, j_idx = get_pairwise_indices(n)
    bounds_vars = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    # Diverse hexagonal row distributions summing to >= 26
    row_configs = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 5, 5, 6],
        [6, 6, 5, 5, 4], [7, 5, 6, 5, 3], [5, 7, 5, 5, 4],
        [4, 6, 6, 6, 4], [6, 5, 5, 6, 4], [5, 6, 4, 6, 5],
        [5, 5, 6, 5, 5], [6, 6, 6, 4, 4], [7, 6, 6, 7]
    ]
    
    inits = []
    for rc in row_configs:
        if sum(rc) >= n:
            inits.append(generate_hex_init(n, rc, 0.095, rng))
            
    # Add uniform random starts
    for _ in range(10):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    for cfg in inits:
        # Phase 1: LP-based hill climbing on centers
        c_lc, s_lc = hill_climb_lp(cfg, A_ub, i_idx, j_idx, n, rng, iters=1500, init_step=0.05)
        
        if s_lc > best_sum:
            best_sum = s_lc
            best_centers = c_lc.copy()
            best_radii, _ = solve_radii_lp(best_centers, A_ub, i_idx, j_idx)
            
        # Phase 2: Gradient-based polishing
        radii_init = best_radii if best_radii is not None else np.full(n, 0.08)
        c_pol = polish_slsqp(c_lc, radii_init, i_idx, j_idx, n, bounds_vars)
        r_pol, s_pol = solve_radii_lp(c_pol, A_ub, i_idx, j_idx)
        if r_pol is not None and s_pol > best_sum:
            best_sum = s_pol
            best_centers = c_pol.copy()
            best_radii = r_pol.copy()
            
    # Fallback guarantee
    if best_centers is None:
        best_centers = inits[0]
        best_radii, best_sum = solve_radii_lp(best_centers, A_ub, i_idx, j_idx)
        if best_radii is None: best_radii = np.full(n, 0.08)
        
    # Final safety scaling to strictly satisfy 1e-12 validation tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    dx = best_centers[i_idx, 0] - best_centers[j_idx, 0]
    dy = best_centers[i_idx, 1] - best_centers[j_idx, 1]
    d = np.sqrt(dx**2 + dy**2)
    rs = best_radii[i_idx] + best_radii[j_idx]
    if np.any(rs > 1e-12):
        valid_idx = rs > 1e-12
        scale = min(scale, np.min(d[valid_idx] / rs[valid_idx]))
        
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
