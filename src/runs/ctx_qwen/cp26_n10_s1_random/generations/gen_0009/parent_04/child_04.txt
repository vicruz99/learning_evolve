import numpy as np
from scipy.optimize import minimize, linprog

def get_lp_solution(centers, n, i_idx, j_idx):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Maximum radius allowed by boundaries for each circle
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    # Pairwise Euclidean distances
    dists_ij = np.sqrt(np.sum((centers[i_idx] - centers[j_idx])**2, axis=1))
    
    m = len(i_idx)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), i_idx] = 1.0
    A_ub[np.arange(m), j_idx] = 1.0
    b_ub = dists_ij
    
    bounds = [(0.0, lims[i]) for i in range(n)]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def force_simulate(centers, n, steps=1800):
    """Force-directed repulsion simulation to spread circles evenly while growing radii."""
    r = np.full(n, 0.035)
    vel = np.zeros_like(centers)
    dt = 0.004
    damping = 0.92
    k_rep = 45.0
    k_wall = 25.0
    
    for _ in range(steps):
        r *= 1.00012  # Slow, steady growth
        
        # Vectorized pairwise repulsion
        diff = centers[:, None, :] - centers[None, :, :]
        dists_sq = np.sum(diff**2, axis=2)
        dists = np.sqrt(np.maximum(dists_sq, 1e-12))
        np.fill_diagonal(dists, np.inf)
        
        req = r[:, None] + r[None, :]
        overlap = np.maximum(0.0, req - dists)
        inv_dists = 1.0 / np.maximum(dists, 1e-9)
        rep = k_rep * overlap * inv_dists
        
        forces = np.zeros_like(centers)
        forces[:, 0] = np.sum(diff[:, :, 0] * rep, axis=1)
        forces[:, 1] = np.sum(diff[:, :, 1] * rep, axis=1)
        
        # Wall repulsion
        for i in range(n):
            c = centers[i]
            ri = r[i]
            if c[0] - ri < 0: forces[i, 0] += k_wall * (ri - c[0])
            if c[0] + ri > 1.0: forces[i, 0] -= k_wall * (c[0] + ri - 1.0)
            if c[1] - ri < 0: forces[i, 1] += k_wall * (ri - c[1])
            if c[1] + ri > 1.0: forces[i, 1] -= k_wall * (c[1] + ri - 1.0)
            
        vel = damping * vel + forces * dt
        centers += vel
        centers = np.clip(centers, 0.0, 1.0)
        
    return centers, r

def coordinate_descent(centers, n, i_idx, j_idx, rng):
    """Optimizes center positions by maximizing the LP-optimal sum of radii."""
    r, s = get_lp_solution(centers, n, i_idx, j_idx)
    if r is None:
        return centers, np.full(n, 0.05), 1.3
    
    for step in range(300):
        idx = rng.integers(n)
        old_c = centers[idx].copy()
        
        # Adaptive step size with decay
        step_size = 0.012 * (0.95 ** (step / 40.0))
        centers[idx] += rng.uniform(-step_size, step_size, 2)
        centers[idx] = np.clip(centers[idx], 1e-4, 1.0 - 1e-4)
        
        r_new, s_new = get_lp_solution(centers, n, i_idx, j_idx)
        if r_new is not None and s_new > s + 1e-7:
            s = s_new
            r = r_new
        else:
            centers[idx] = old_c
            
    return centers, r, s

def joint_obj(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def joint_cons(x, n, i_idx, j_idx):
    """Inequality constraints >= 0 for valid packing."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    
    # Boundary constraints
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dr = r[i_idx] + r[j_idx]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c
