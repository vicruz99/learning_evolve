from scipy.optimize import minimize, linprog

def force_simulate(centers, radii, n, steps=5000, grow_rate=1.00008):
    """Force-directed simulation with growing radii to find good packing."""
    centers = centers.copy().astype(float)
    radii = radii.copy().astype(float)
    
    vel = np.zeros_like(centers)
    dt = 0.005
    damping = 0.90
    k_rep = 80.0
    k_wall = 50.0
    
    for step in range(steps):
        radii *= grow_rate
        
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                req = radii[i] + radii[j]
                if dist < req and dist > 1e-10:
                    overlap = req - dist
                    f = k_rep * overlap / dist
                    forces[i] += f * diff
                    forces[j] -= f * diff
                    vel[i] += forces[i] * dt
                    vel[j] -= forces[i] * dt
        
        for i in range(n):
            # Wall repulsion
            if centers[i, 0] - radii[i] < 0:
                forces[i, 0] += k_wall * (radii[i] - centers[i, 0])
            if centers[i, 0] + radii[i] > 1.0:
                forces[i, 0] -= k_wall * (centers[i, 0] + radii[i] - 1.0)
            if centers[i, 1] - radii[i] < 0:
                forces[i, 1] += k_wall * (radii[i] - centers[i, 1])
            if centers[i, 1] + radii[i] > 1.0:
                forces[i, 1] -= k_wall * (centers[i, 1] + radii[i] - 1.0)
        
        vel = damping * vel + forces * dt
        centers += vel
        centers = np.clip(centers, 1e-8, 1.0 - 1e-8)
        
    return centers, radii

def solve_lp_radii(centers, n, i_idx, j_idx, limits):
    """Solves LP to maximize sum of radii for fixed centers."""
    bounds = [(0.0, max(lim, 1e-12)) for lim in limits]
    
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    m = len(i_idx)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), i_idx] = 1.0
    A_ub[np.arange(m), j_idx] = 1.0
    b_ub = dists
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun) and np.all(np.isfinite(res.x)) and np.all(res.x >= -1e-12):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return None, 0.0

def neg_clearance(x, n):
    """Negative of minimum clearance (for Nelder-Mead maximization)."""
    c = x.reshape(n, 2)
    d_w = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                     np.minimum(c[:, 1], 1.0 - c[:, 1]))
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    min_d = np.min(dist) / 2.0
    return -min(min(np.min(d_w), min_d))

def hex_init(n, row_counts, r0=0.10):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    row_idx = 0
    for cnt in row_counts:
        shift = r0 * np.sqrt(3) / 2.0 if row_idx % 2 == 1 else 0.0
        width = (cnt - 1) * 2.0 * r0 if cnt > 1 else 0.0
        x_start = (1.0 - width) / 2.0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x_start, y])
            x_start += 2.0 * r0
        y += r0 * np.sqrt(3)
        row_idx += 1
    
    while len(pts) < n:
        pts.append([0.5, 0.5])
    
    return np.array(pts[:n])
