import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Precompute the constant inequality matrix for the LP radius solver."""
    m = n + n * (n - 1) // 2
    A = np.zeros((m, n))
    A[:n, :] = np.eye(n)
    k = n
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp_radii(centers, n, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    limits = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    limits = np.maximum(limits, 1e-9)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Build RHS vector for LP
    b_ub = np.zeros(A_ub.shape[0])
    b_ub[:n] = limits
    k = n
    for i in range(n):
        for j in range(i + 1, n):
            b_ub[k] = dists[i, j]
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def lp_objective(pos, idx, fixed_centers, n, A_ub):
    """Objective for coordinate descent: maximize LP sum of radii."""
    c = fixed_centers.copy()
    c[idx] = np.clip(pos, 0.001, 0.999)
    _, s = solve_lp_radii(c, n, A_ub)
    return -s  # Minimize negative sum

    A_ub = build_lp_matrix(n)
    best_sum = -1.0

    best_radii = None

    configs = []
    patterns = [

    ]
    for rc in patterns:
        pts = []
        y = 0.09
        for r_idx, cnt in enumerate(rc):
            shift = 0.095 if r_idx % 2 == 1 else 0.0
            width = (cnt - 1) * 0.19
            x_start = 0.5 - width / 2.0 + shift
            for c in range(cnt):
                if len(pts) < n:
                    pts.append([x_start + c * 0.19, y])
            y += 0.165
            if len(pts) >= n:
                break
        configs.append(np.array(pts[:n]))
        
    rng = np.random.default_rng(42)
    for _ in range(6):
        configs.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # 2. Optimization Pipeline
    for cfg in configs:
        centers = np.clip(cfg, 0.05, 0.95)
        
        # Phase A: Coordinate Descent on LP objective
        # Optimizing one circle at a time is highly effective for escaping local minima in packing
        for _ in range(8):
            for i in range(n):
                fixed = centers.copy()
                res = minimize(lp_objective, centers[i], args=(i, fixed, n, A_ub),
                               method='Nelder-Mead', 
                               options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-8})
                centers[i] = np.clip(res.x, 0.001, 0.999)
                
        _, s_cd = solve_lp_radii(centers, n, A_ub)
        if s_cd > best_sum:
            best_sum = s_cd
            best_centers = centers.copy()
            best_radii, _ = solve_lp_radii(centers, n, A_ub)

            return -np.sum(v[2*n:])

    return best_centers, best_radii, best_sum
