import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_data(n):
    """Precomputes the constant inequality matrix for the LP radius solver."""
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    m = m_bound + m_pair
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        for _ in range(4):
            A[k, i] = 1.0
            k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp(centers, A):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    m = A.shape[0]
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        for _ in range(4):
            b[k] = lim
            k += 1
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b[k] = d
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-7)

def obj_joint(x, n):
    """Objective: minimize negative sum of radii => maximize sum."""
    return -np.sum(x[2 * n:])

def cons_joint(x, n, triu_i, triu_j):
    """Inequality constraints >= 0 for valid joint (center, radius) packing."""
    cx = x[:n]
    cy = x[n:2 * n]
    r = x[2 * n:]
    c = []
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    dx = cx[triu_i] - cx[triu_j]
    dy = cy[triu_i] - cy[triu_j]
    dist = np.hypot(dx, dy)
    r_sum = r[triu_i] + r[triu_j]
    c.append(dist - r_sum)
    return np.concatenate(c)

def force_sim(centers, steps=2500):
    """Force-directed expansion simulation to pack circles tightly."""
    n = centers.shape[0]
    r = np.full(n, 0.07)
    vel = np.zeros_like(centers)
    dt = 0.004
    k_rep = 80.0
    k_wall = 40.0
    damp = 0.85
    
    for _ in range(steps):
        r *= 1.0002
        forces = np.zeros_like(centers)
        
        # Wall repulsion
        for i in range(n):
            for dim in range(2):
                if centers[i, dim] - r[i] < 0:
                    forces[i, dim] += k_wall * (r[i] - centers[i, dim])
                if centers[i, dim] + r[i] > 1.0:
                    forces[i, dim] -= k_wall * (centers[i, dim] + r[i] - 1.0)
                    
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.hypot(diff[0], diff[1])
                req = r[i] + r[j]
                if dist < req and dist > 1e-9:
                    f = k_rep * (req - dist) / dist
                    forces[i] += f * diff
                    forces[j] -= f * diff
                    
        vel = damp * vel + forces * dt
        centers += vel
        centers = np.clip(centers, 0.0, 1.0)
        
    return centers
