import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        for _ in range(4):
            A[k, i] = 1.0
            k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_radii_lp(centers, A_ub):
    n = centers.shape[0]
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            b[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            k += 1
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 1e-9)

def objective_func(x, n):
    return -np.sum(x[2 * n:])

def constraints_func(x, n):
    cx = x[:n]
    cy = x[n:2 * n]
    r = x[2 * n:]
    c_list = []
    c_list.append(cx - r)
    c_list.append(1.0 - cx - r)
    c_list.append(cy - r)
    c_list.append(1.0 - cy - r)
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dr = r[i_idx] + r[j_idx]
    c_list.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c_list)

def generate_hex_config(rows, r0):
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= 26:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < 26:
        pts.append([0.5, 0.5])
    return np.array(pts[:26])

def optimize_candidate(cfg, n, bounds_vars, cons_dict):

    x0[:n] = cfg[:, 0]
    x0[n:2 * n] = cfg[:, 1]

    for i in range(n):

        dn = np.min(np.sqrt(np.sum((cfg - cfg[i])**2, axis=1)))
        min_d[i] = min(dw, dn / 2.0)
    x0[2 * n:] = min_d * 0.95

    try:
        res = minimize(objective_func, x0, args=(n,), method='SLSQP',
                       bounds=bounds_vars, constraints=cons_dict,
                       options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
        if np.isfinite(res.fun):
            cx = res.x[:n]
            cy = res.x[n:2 * n]
            r = np.maximum(res.x[2 * n:], 1e-9)
            centers_opt = np.column_stack((cx, cy))
            radii_opt = r
            c_vals = constraints_func(res.x, n)
            if np.min(c_vals) >= -1e-7:
                return centers_opt, radii_opt, np.sum(r)
    except Exception:
        pass
    return None
