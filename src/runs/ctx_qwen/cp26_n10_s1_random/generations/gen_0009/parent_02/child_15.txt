import numpy as np
from scipy.optimize import linprog, minimize

def build_lp_matrix(n):
    """Builds the constant inequality matrix for the LP radius solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def eval_lp(centers, A_ub, n):
    """Solves LP to maximize sum of radii for fixed centers. Returns (sum_r, radii)."""
    m = A_ub.shape[0]
    b = np.empty(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            b[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    # Fallback if LP fails (rare with valid centers)
    return 0.0, np.full(n, 1e-6)

def objective_func(params, n, A_ub):

    centers = params.reshape(n, 2)
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    s, _ = eval_lp(centers, A_ub, n)
    return -s

def coord_obj(params, idx, centers, n, A_ub):
    """Objective for coordinate-wise optimization: negative LP sum when moving circle idx."""
    c = centers.copy()
    c[idx] = params
    c = np.clip(c, 1e-5, 1.0 - 1e-5)
    s, _ = eval_lp(c, A_ub, n)
    return -s

def generate_hex(rows, r0, rng):

    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:n])
