import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Builds the constant inequality matrix for the LP radius solver."""
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    A = np.zeros((m_bound + m_pair, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp(centers, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            b[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            k += 1
            
    bounds = [(0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= -1e-12):
            return np.maximum(res.x, 0.0), -res.fun
        return np.full(n, 1e-9), 0.0
    except Exception:
        return np.full(n, 1e-9), 0.0

def neg_sum_lp(x_flat, A_ub):
    """Objective for center optimization: negative sum of optimal radii."""
    try:
        n = 26
        centers = x_flat.reshape(n, 2)
        centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
        _, val = solve_lp(centers, A_ub)
        return -val
    except Exception:
        return -0.0

def generate_starts(n, rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [4,6,6,6,4], 
        [6,6,5,5,4], [7,5,6,4,4], [5,6,6,5,4], [5,5,5,5,6]
    ]
    for pat in patterns:
        pts = []
        r0 = 0.095
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        while len(pts) < n:
            pts.append([0.5, 0.5])
        starts.append(np.array(pts[:n]))
        p = np.array(pts[:n]) + rng.uniform(-0.025, 0.025, (n, 2))
        starts.append(np.clip(p, 0.05, 0.95))
        
    for _ in range(12):
        starts.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    A_ub = build_lp_matrix(n)
    
    starts = generate_starts(n, rng)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Nelder-Mead optimization on centers using LP projection
    for cfg in starts:
        x0 = cfg.flatten()
        try:
            res = minimize(neg_sum_lp, x0, args=(A_ub,), method='Nelder-Mead',
                           options={'maxiter': 1500, 'xatol': 1e-7, 'fatol': 1e-8})
            centers_opt = np.clip(res.x.reshape(n, 2), 1e-5, 1.0 - 1e-5)
            r_opt, s_opt = solve_lp(centers_opt, A_ub)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = centers_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            continue
            
    if best_centers is None:
        best_centers = starts[0]
        best_radii, best_sum = solve_lp(best_centers, A_ub)

    # Phase 2: Coordinate descent / hill-climbing refinement
    step_size = 0.015
    for _ in range(1500):
        i = rng.integers(n)
        old_pos = best_centers[i].copy()
        best_centers[i] += rng.uniform(-step_size, step_size, 2)
        best_centers[i] = np.clip(best_centers[i], 1e-5, 1.0 - 1e-5)
        
        r_new, s_new = solve_lp(best_centers, A_ub)
        if s_new > best_sum + 1e-9:
            best_sum = s_new
            best_radii = r_new.copy()
        else:
            best_centers[i] = old_pos
        step_size *= 0.998
        
    # Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-9:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
