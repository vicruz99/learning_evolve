import numpy as np
from scipy.optimize import linprog, minimize

def build_lp_matrix(n):
    """Constructs the inequality matrix A_ub for the radius LP."""
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    m = m_bound + m_pair
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp_radii(centers, A_ub, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b[k] = np.hypot(dx, dy)
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 1e-6)

def joint_objective(vars_flat, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars_flat[2 * n:])

def joint_constraints(vars_flat, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = vars_flat[:n]
    cy = vars_flat[n:2 * n]
    r = vars_flat[2 * n:]
    
    c_list = []
    c_list.append(cx - r)
    c_list.append(1.0 - cx - r)
    c_list.append(cy - r)
    c_list.append(1.0 - cy - r)
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dist = np.hypot(dx, dy)
    r_sum = r[i_idx] + r[j_idx]
    c_list.append(dist - r_sum)
    
    return np.concatenate(c_list)

def run_packing():

    rng = np.random.default_rng(42)
    A_ub = build_lp_matrix(n)

    ]
    
    for pat in patterns:

        y = 0.1
        for ri, cnt in enumerate(pat):
            shift = 0.1 if ri % 2 == 1 else 0.0
            x = 0.1 + shift

                if len(pts) >= n:
                    break
                pts.append([x, y])
                x += 0.2
            y += 0.1 * np.sqrt(3)
        starts.append(np.array(pts[:n]))
        
    for _ in range(8):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)))
        
    # 2. Hill-climbing optimization on centers using LP sum of radii
    for cfg in starts:

        curr_r = solve_lp_radii(curr_c, A_ub, n)
        curr_s = np.sum(curr_r)

            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)

    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
