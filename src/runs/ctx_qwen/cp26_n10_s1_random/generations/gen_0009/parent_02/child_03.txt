import numpy as np
from scipy.optimize import minimize, linprog

def compute_lp_value(centers_flat, n, i_idx, j_idx, A_ub, bounds_lp):
    """
    Computes the negative maximum sum of radii for given centers using an LP solver.
    Acts as the objective function for center optimization.
    """
    centers = np.clip(centers_flat.reshape(n, 2), 1e-5, 1.0 - 1e-5)
    m = 4 * n + len(i_idx)
    b = np.empty(m)
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    b[:n] = centers[:, 0]
    b[n:2*n] = 1.0 - centers[:, 0]
    b[2*n:3*n] = centers[:, 1]
    b[3*n:4*n] = 1.0 - centers[:, 1]
    
    # Pairwise constraints: r_i + r_j <= distance(i, j)
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    b[4*n:] = np.hypot(dx, dy)
    
    try:
        # Maximize sum(r) <=> Minimize -sum(r)
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=bounds_lp, method='highs')
        if res.success:
            return -res.fun
    except Exception:
        pass
    return -1.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    A_ub = np.zeros((m, n))

    A_ub[3*n:4*n, :] = np.eye(n)

                best_sum = -res.fun

    final_sum = float(np.sum(radii))
    
    return best_centers, radii, final_sum
