import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Builds the constant inequality matrix for the LP radius solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_radii_lp(centers, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            b[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 1e-6)

def neg_clearance(x, n):
    """Objective: maximize minimum clearance (for equal circles)."""
    c = x.reshape(n, 2)
    d_w = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                     np.minimum(c[:, 1], 1.0 - c[:, 1]))
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    return -min(np.min(d_w), np.min(dist) / 2.0)

def lp_sum_obj(centers_flat, args_tuple):
    """Objective for Nelder-Mead: maximize LP sum of radii for given centers."""
    n, A_ub = args_tuple
    centers = centers_flat.reshape(n, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    r = solve_radii_lp(centers, A_ub)
    return -np.sum(r)

def joint_obj(vars_flat, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2 * n:])

def joint_cons_sq(vars_flat, n):
    """Inequality constraints >= 0 using squared distances (smoother)."""
    cx = vars_flat[:n]
    cy = vars_flat[n:2 * n]
    r = vars_flat[2 * n:]
    
    c_list = []
    c_list.append(cx - r)
    c_list.append(1.0 - cx - r)
    c_list.append(cy - r)
    c_list.append(1.0 - cy - r)
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dist_sq = dx**2 + dy**2
    r_sum_sq = (r[i_idx] + r[j_idx])**2
    c_list.append(dist_sq - r_sum_sq)
    
    return np.concatenate(c_list)

def generate_hex_init(rows, r0):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def generate_grid_init(n, spacing=0.18, offset=0.1):
    """Generates a grid-based initial configuration."""
    pts = []
    for j in range(6):
        for i in range(5):
            if len(pts) >= n:
                break
            pts.append([offset + i * spacing, offset + j * spacing])
        if len(pts) >= n:
            break
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def generate_corner_init(n, r0=0.10):
    """Places circles in corners and fills remaining with hex pattern."""
    pts = [[r0, r0], [1-r0, r0], [r0, 1-r0], [1-r0, 1-r0]]
    y = r0 * 2.5
    row = 0
    while len(pts) < n:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x <= 1.0 - r0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    return np.array(pts[:n])

def lp_hill_climb(centers, rng, A_ub, steps=150):
    """Local search: perturb one circle at a time and re-solve LP."""
    n = centers.shape[0]
    centers = centers.copy()
    current_r = solve_radii_lp(centers, A_ub)
    current_sum = np.sum(current_r)
    step_size = 0.012
    
    for step in range(steps):
        i = rng.integers(n)
        old_pos = centers[i].copy()
        
        centers[i] += rng.uniform(-step_size, step_size, 2)
        centers[i] = np.clip(centers[i], 1e-4, 1.0 - 1e-4)
        
        new_r = solve_radii_lp(centers, A_ub)
        new_sum = np.sum(new_r)
        
        if new_sum > current_sum + 1e-10:
            current_r = new_r
            current_sum = new_sum
        else:
            centers[i] = old_pos
        
        step_size *= 0.995
    
    return centers, current_r, current_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)

    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Comprehensive hex patterns

        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [6, 6, 6, 4, 4], [4, 5, 6, 5, 6], [5, 4, 6, 5, 6], [5, 6, 6, 5, 4],
        [7, 5, 5, 5, 4], [5, 5, 7, 5, 4], [6, 4, 6, 6, 4], [5, 6, 7, 4, 4],
        [5, 5, 5, 6, 5], [6, 5, 5, 5, 5], [5, 5, 6, 6, 4], [4, 5, 6, 6, 5],
        [6, 6, 4, 5, 5], [5, 6, 5, 5, 5], [4, 4, 6, 6, 6], [6, 6, 5, 6, 3],
        [5, 5, 5, 5, 5, 1], [4, 5, 5, 6, 6], [5, 4, 5, 6, 6]
    ]
    
    inits = []
    for p in patterns:

            for r0 in [0.088, 0.092, 0.096, 0.100]:
                cfg = generate_hex_init(p, r0)

                for _ in range(3):
                    pert = cfg + rng.uniform(-0.018, 0.018, cfg.shape)
                    inits.append(np.clip(pert, 0.05, 0.95))
    
    # Grid-based inits
    for sp in [0.16, 0.18, 0.20]:
        for off in [0.08, 0.10, 0.12]:
            cfg = generate_grid_init(n, sp, off)

            pert = cfg + rng.uniform(-0.015, 0.015, cfg.shape)
            inits.append(np.clip(pert, 0.05, 0.95))

            centers_opt = np.clip(res.x.reshape(n, 2), 1e-6, 1.0 - 1e-6)
