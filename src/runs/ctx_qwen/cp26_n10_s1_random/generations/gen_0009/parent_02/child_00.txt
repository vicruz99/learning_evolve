

<strategy>
To improve the packing sum of radii beyond 2.628, I'll use a multi-phase approach:

1. **Force-directed simulation** with gradually expanding radii - this naturally finds tight packings
2. **Nelder-Mead optimization** on centers to maximize minimum clearance (equal radius proxy)
3. **LP radius extraction** for fixed centers to allow unequal radii
4. **SLSQP joint optimization** for fine-tuning both centers and radii
5. **Iterative perturbation search** with diminishing step sizes to escape local minima
6. **Multiple diverse initial configurations** including various hexagonal row patterns and random starts

Key insights:
- Using `dist - r_sum` (not squared) in constraints provides better gradients
- LP on fixed centers allows circles in less constrained regions to grow larger
- Force simulation with damping helps circles settle into tight configurations
- Multiple restarts from perturbed best solutions help escape local optima
- Very small final perturbations can squeeze out additional radius
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Builds the constant inequality matrix for the LP radius solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_radii_lp(centers, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            b[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success and np.all(res.x >= -1e-10):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 1e-8)

def joint_objective(vars_flat, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2 * n:])

def joint_constraints(vars_flat, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = vars_flat[:n]
    cy = vars_flat[n:2 * n]
    r = vars_flat[2 * n:]
    
    c_list = []
    c_list.append(cx - r)
    c_list.append(1.0 - cx - r)
    c_list.append(cy - r)
    c_list.append(1.0 - cy - r)
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dist = np.hypot(dx, dy)
    r_sum = r[i_idx] + r[j_idx]
    c_list.append(dist - r_sum)
    
    return np.concatenate(c_list)

def force_simulate(centers, radii, n, steps=4000, dt=0.005, damping=0.90):
    """Force-directed repulsion simulation to spread circles evenly."""
    centers = centers.copy()
    radii = radii.copy()
    vel = np.zeros_like(centers)
    k_rep = 80.0
    k_wall = 50.0
    
    for step in range(steps):
        # Gradually expand radii
        radii *= 1.0 + 1e-5
        
        forces = np.zeros_like(centers)
        
        # Pairwise repulsion
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        r_sum = radii[:, None] + radii[None, :]
        
        overlap = np.maximum(0.0, r_sum - dists)
        rep_strength = k_rep * overlap / (dists + 1e-12)
        
        fx = np.sum(diff[:, :, 0] * rep_strength, axis=1)
        fy = np.sum(diff[:, :, 1] * rep_strength, axis=1)
        forces[:, 0] += fx
        forces[:, 1] += fy
        
        # Wall repulsion
        for i in range(n):
            if centers[i, 0] - radii[i] < 0:
                forces[i, 0] += k_wall * (radii[i] - centers[i, 0])
            if centers[i, 0] + radii[i] > 1.0:
                forces[i, 0] -= k_wall * (centers[i, 0] + radii[i] - 1.0)
            if centers[i, 1] - radii[i] < 0:
                forces[i, 1] += k_wall * (radii[i] - centers[i, 1])
            if centers[i, 1] + radii[i] > 1.0:
                forces[i, 1] -= k_wall * (centers[i, 1] + radii[i] - 1.0)
        
        vel = damping * vel + forces * dt
        centers += vel
        centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    
    return centers, radii

def neg_clearance(x, n):
    """Objective for Nelder-Mead: maximize minimum clearance."""
    c = x.reshape(n, 2)
    d_w = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                     np.minimum(c[:, 1], 1.0 - c[:, 1]))
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    return -min(np.min(d_w), np.min(dist) / 2.0)

def generate_hex_init(rows, r0, rng):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:n])

def optimize_stage(cfg, n, A_ub, bounds_vars):
    """Runs LP + SLSQP refinement pipeline and returns (centers, radii, sum_r)."""
    r_lp = solve_radii_lp(cfg, A_ub)
    s_lp = np.sum(r_lp)
    
    x0 = np.zeros(3 * n)
    x0[:n] = cfg[:, 0]
    x0[n:2 * n] = cfg[:, 1]
    x0[2 * n:] = np.maximum(r_lp * 0.96, 1e-9)
    
    try:
        res = minimize(joint_objective, x0, args=(n,), method='SLSQP',
                       bounds=bounds_vars,
                       constraints={'type': 'ineq', 'fun': joint_constraints, 'args': (n,)},
                       options={'maxiter': 10000, 'ftol': 1e-14})
        
        if np.isfinite(res.fun):
            cx = res.x[:n]
            cy = res.x[n:2 * n]
            r = np.maximum(res.x[2 * n:], 1e-9)
            centers_opt = np.column_stack((cx, cy))
            radii_opt = r
            
            # Strict validity verification
            valid = True
            if np.any(centers_opt[:, 0] < radii_opt - 1e-8) or np.any(centers_opt[:, 0] > 1 - radii_opt + 1e-8):
                valid = False
            if np.any(centers_opt[:, 1] < radii_opt - 1e-8) or np.any(centers_opt[:, 1] > 1 - radii_opt + 1e-8):
                valid = False
            
            if valid:
                idx_i, idx_j = np.triu_indices(n, k=1)
                dx = centers_opt[:, 0][idx_i] - centers_opt[:, 0][idx_j]
                dy = centers_opt[:, 1][idx_i] - centers_opt[:, 1][idx_j]
                d = np.hypot(dx, dy)
                rs = radii_opt[idx_i] + radii_opt[idx_j]
                if np.any(d < rs - 1e-8):
                    valid = False
                    
            if valid:
                return centers_opt, radii_opt, np.sum(radii_opt)
                
    except Exception:
        pass
        
    return cfg, r_lp, s_lp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    A_ub = build_lp_matrix(n)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [6, 6, 6, 4, 4], [4, 5, 6, 5, 6], [5, 4, 6, 5, 6], [5, 6, 6, 5, 4],
        [7, 6, 5, 5, 3], [6, 7, 5, 5, 3], [5, 6, 7, 5, 3], [6, 5, 7, 5, 3],
        [5, 5, 5, 7, 4], [5, 5, 7, 5, 4], [7, 5, 5, 5, 4], [6, 6, 6, 5, 3]
    ]
    
    bounds_vars = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    # Phase 0: Force simulation from multiple starts
    all_candidates = []
    
    for p in patterns:
        if sum(p) >= n:
            for r0 in [0.08, 0.085, 0.09, 0.095]:
                cfg = generate_hex_init(p, r0, rng)
                sim_c, sim_r = force_simulate(cfg, np.full(n, r0 * 0.9), n, steps=5000)
                all_candidates.append(sim_c)
                
    for _ in range(8):
        cfg = rng.uniform(0.1, 0.9, (n, 2))
        sim_c, sim_r = force_simulate(cfg, np.full(n, 0.08), n, steps=5000)
        all_candidates.append(sim_c)
    
    # Phase 1: Nelder-Mead optimization for maximum clearance
    for cfg in all_candidates:
        try:
            res = minimize(neg_clearance, cfg.flatten(), args=(n,), 
                          method='Nelder-Mead',
                          options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-12})
            centers_nm = res.x.reshape(n, 2)
            centers_nm = np.clip(centers_nm, 0.001, 0.999)
            
            r_lp = solve_radii_lp(centers_nm, A_ub)
            s = np.sum(r_lp)
            
            if s > best_sum:
                best_sum = s
                best_centers = centers_nm.copy()
                best_radii = r_lp.copy()
        except Exception:
            continue
    
    # Phase 2: SLSQP joint optimization from best Nelder-Mead results
    if best_centers is not None:
        for _ in range(30):
            pert = np.clip(best_centers + rng.uniform(-0.008, 0.008, best_centers.shape), 0.05, 0.95)
            c_res, r_res, s_res = optimize_stage(pert, n, A_ub, bounds_vars)
            if s_res > best_sum:
                best_sum = s_res
                best_centers = c_res.copy()
                best_radii = r_res.copy()
    
    # Also try SLSQP from force-simulated candidates
    for cfg in all_candidates[:12]:
        c_res, r_res, s_res = optimize_stage(cfg, n, A_ub, bounds_vars)
        if s_res > best_sum:
            best_sum = s_res
            best_centers = c_res.copy()
            best_radii = r_res.copy()
    
    # Phase 3: Iterative local search with diminishing steps
    if best_centers is not None:
        step_size = 0.006
        for iteration in range(50):
            improved = False
            for _ in range(20):
                pert = np.clip(best_centers + rng.uniform(-step_size, step_size, best_centers.shape), 0.02, 0.98)
                r_pert = solve_radii_lp(pert, A_ub)
                s_pert = np.sum(r_pert)
                
                if s_pert > best_sum + 1e-8:
                    best_sum = s_pert
                    best_centers = pert.copy()
                    best_radii = r_pert.copy()
                    improved = True
                    break
            
            if improved:
                c_res, r_res, s_res = optimize_stage(best_centers, n, A_ub, bounds_vars)
                if s_res > best_sum:
                    best_sum = s_res
                    best_centers = c_res.copy()
                    best_radii = r_res.copy()
            
            step_size *= 0.94
            
            if step_size < 0.0003:
                break
    
    # Phase 4: Ultra-fine perturbation search
    if best_centers is not None:
        for _ in range(40):
            pert = np.clip(best_centers + rng.uniform(-0.002, 0.002, best_centers.shape), 0.01, 0.99)
            c_res, r_res, s_res = optimize_stage(pert, n, A_ub, bounds_vars)
            if s_res > best_sum:
                best_sum = s_res
                best_centers = c_res.copy()
                best_radii = r_res.copy()
    
    # Fallback if optimization unexpectedly fails
    if best_centers is None:
        best_centers = generate_hex_init([5, 6, 5, 6, 4], 0.09, rng)
        best_radii = solve_radii_lp(best_centers, A_ub)
        best_sum = np.sum(best_radii)

    # Final LP extraction on best centers
    r_final = solve_radii_lp(best_centers, A_ub)
    if np.sum(r_final) > best_sum:
        best_radii = r_final
        best_sum = np.sum(r_final)

    # Final safety scaling to guarantee strict validity within numerical tolerance
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-9:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
