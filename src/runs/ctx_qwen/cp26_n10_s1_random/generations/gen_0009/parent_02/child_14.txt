def make_hex(rows, r0, rng):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:n])

    return -np.sum(vars_arr[2 * n:])

def constraints(vars_arr, n, triu_i, triu_j):

    cy = vars_arr[n:2 * n]
    r = vars_arr[2 * n:]

    c[:n] = cx - r

    c[3 * n:4 * n] = 1.0 - cy - r

    dy = cy[triu_i] - cy[triu_j]
    dr = r[triu_i] + r[triu_j]
    c[4 * n:] = dx**2 + dy**2 - dr**2
    return c

def solve_lp_radii(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        A[k, i] = 1.0; b[k] = x; k += 1
        A[k, i] = 1.0; b[k] = 1.0 - x; k += 1
        A[k, i] = 1.0; b[k] = y; k += 1
        A[k, i] = 1.0; b[k] = 1.0 - y; k += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            b[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0.0, None), method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 1e-9)
