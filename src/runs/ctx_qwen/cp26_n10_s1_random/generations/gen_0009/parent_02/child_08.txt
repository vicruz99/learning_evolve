import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Builds the constant inequality matrix for the LP radius solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_radii_lp(centers, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
        
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    idx_i, idx_j = np.triu_indices(n, k=1)
    b[k:] = dists[idx_i, idx_j]
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-6)

def joint_objective(vars_flat, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2 * n:])

def joint_constraints(vars_flat, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = vars_flat[:n]
    cy = vars_flat[n:2 * n]
    r = vars_flat[2 * n:]
    
    c_list = []
    c_list.append(cx - r)
    c_list.append(1.0 - cx - r)
    c_list.append(cy - r)
    c_list.append(1.0 - cy - r)
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    r_sum = r[i_idx] + r[j_idx]
    c_list.append(dx**2 + dy**2 - r_sum**2)
    
    return np.concatenate(c_list)

def force_simulate(centers, n, steps=800):
    """Vectorized force-directed relaxation to untangle circles and expand radii."""
    centers = centers.copy()
    radii = np.full(n, 0.06)
    vel = np.zeros_like(centers)
    dt = 0.008
    damp = 0.92
    
    for _ in range(steps):
        radii *= 1.0001
        forces = np.zeros_like(centers)
        
        # Wall forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x < r: forces[i, 0] += (r - x) * 100.0
            if x > 1.0 - r: forces[i, 0] -= (x - (1.0 - r)) * 100.0
            if y < r: forces[i, 1] += (r - y) * 100.0
            if y > 1.0 - r: forces[i, 1] -= (y - (1.0 - r)) * 100.0
            
        # Pairwise repulsion
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        req = radii[:, None] + radii[None, :]
        overlap = np.maximum(0.0, req - dists)
        rep = (overlap * 50.0) / dists
        forces[:, 0] += np.sum(diff[:, :, 0] * rep, axis=1)
        forces[:, :, 1] += np.sum(diff[:, :, 1] * rep, axis=1)
        
        vel = damp * vel + forces * dt
        centers += vel
        centers = np.clip(centers, 0.0, 1.0)
        
    return centers

def hill_climb_lp(centers, n, A_ub, rng, max_iter=600):
    """Local search on centers using LP sum of radii as objective."""
    current_r = solve_radii_lp(centers, A_ub)
    current_sum = np.sum(current_r)
    step = 0.02
    
    for _ in range(max_iter):
        i = rng.integers(n)
        old_pos = centers[i].copy()
        centers[i] += rng.uniform(-step, step, 2)
        centers[i] = np.clip(centers[i], 1e-4, 1.0 - 1e-4)
        
        new_r = solve_radii_lp(centers, A_ub)
        new_sum = np.sum(new_r)
        
        if new_sum > current_sum:
            current_r = new_r
            current_sum = new_sum
        else:
            centers[i] = old_pos
            
        step *= 0.996
        
    return centers, current_r, current_sum

def optimize_candidate(cfg, n, A_ub, rng, bounds_vars):

    # Phase 3: SLSQP Polish
    x0 = np.zeros(3 * n)
    x0[:n] = hc_centers[:, 0]
    x0[n:2 * n] = hc_centers[:, 1]
    x0[2 * n:] = hc_radii
    
    try:

                       bounds=bounds_vars,
                       constraints={'type': 'ineq', 'fun': joint_constraints, 'args': (n,)},

        if np.isfinite(res.fun):
            cx = res.x[:n]

            if valid:
                return np.column_stack((cx, cy)), r, np.sum(r)
                
    except Exception:
        pass
        
    return hc_centers, hc_radii, hc_sum

def generate_hex_init(rows, r0, rng):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:n])
