import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Builds the constant inequality matrix for the LP radius solver."""
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_radii_lp(centers, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    m = A_ub.shape[0]
    b = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        b[k] = x; k += 1
        b[k] = 1.0 - x; k += 1
        b[k] = y; k += 1
        b[k] = 1.0 - y; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            b[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            k += 1
            
    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mx, 1e-9)))
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-6)

def joint_obj(vars_flat, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2 * n:])

def joint_cons(vars_flat, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = vars_flat[:n]
    cy = vars_flat[n:2 * n]
    r = vars_flat[2 * n:]
    
    c_list = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dist_sq = dx**2 + dy**2
    r_sum = r[i_idx] + r[j_idx]
    c_list.append(dist_sq - r_sum**2)
    
    return np.concatenate(c_list)

def generate_hex(rows, r0):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= 26:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < 26:
        pts.append([0.5, 0.5])
    return np.array(pts[:26])

def hill_climb(centers, n, A_ub, rng, passes=60):
    """Local search on centers using LP sum of radii as objective."""
    current_r = solve_radii_lp(centers, A_ub)
    current_sum = np.sum(current_r)
    step = 0.02
    
    for _ in range(passes):
        improved = False
        idx = rng.permutation(n)
        for i in idx:
            best_pos = centers[i].copy()
            best_s = current_sum
            for _ in range(10):
                pert = centers[i] + rng.uniform(-step, step, 2)
                pert = np.clip(pert, 1e-4, 1.0 - 1e-4)
                centers[i] = pert
                new_r = solve_radii_lp(centers, A_ub)
                if new_r is not None:
                    new_s = np.sum(new_r)
                    if new_s > best_s:
                        best_s = new_s
                        best_pos = pert.copy()
                centers[i] = best_pos
            if best_s > current_sum:
                current_sum = best_s
                current_r = solve_radii_lp(centers, A_ub)
                improved = True
        if not improved:
            step *= 0.8
    return centers, current_r, current_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    configs = []

        try:

                           bounds=bounds_vars,

        pass

    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]

            rs = best_radii[i] + best_radii[j]
