import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Precompute the constant inequality matrix for the LP radius solver."""
    m_pairs = n * (n - 1) // 2
    m_walls = 4 * n
    A = np.zeros((m_pairs + m_walls, n))
    i_idx, j_idx = np.triu_indices(n, k=1)
    A[:m_pairs, i_idx] = 1.0
    A[:m_pairs, j_idx] = 1.0
    
    # Wall constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for k in range(n):
        A[m_pairs + 4*k, k] = 1.0
        A[m_pairs + 4*k + 1, k] = 1.0
        A[m_pairs + 4*k + 2, k] = 1.0
        A[m_pairs + 4*k + 3, k] = 1.0
    return A, i_idx, j_idx

def solve_lp_for_centers(centers, A, i_idx, j_idx):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    b = np.empty(A.shape[0])
    
    # Pairwise distance limits
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b[:len(i_idx)] = dists[i_idx, j_idx]
    
    # Wall distance limits
    for k in range(n):
        x, y = centers[k]
        b[4*n + 4*k] = x
        b[4*n + 4*k + 1] = 1.0 - x
        b[4*n + 4*k + 2] = y
        b[4*n + 4*k + 3] = 1.0 - y
        
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success and np.isfinite(res.fun):
        return res.x, -res.fun
    return None, 0.0

def force_relax(centers, steps=1500):
    """Physics-based relaxation to tightly pack circles."""
    n = centers.shape[0]
    r = np.full(n, 0.04)
    dt = 0.01
    for _ in range(steps):
        r *= 1.00004  # Slowly expand radii to jam packing
        
        forces = np.zeros_like(centers)

        dy = centers[:, np.newaxis, 1] - centers[np.newaxis, :, 1]

        np.fill_diagonal(dists, np.inf)

        rep = (overlap / dists) * 50.0
        forces[:, 0] += np.sum(dx * rep, axis=1)
        forces[:, 1] += np.sum(dy * rep, axis=1)
        
        # Wall repulsion
        for k in range(n):
            if centers[k, 0] < r[k]:
                forces[k, 0] += 100.0 * (r[k] - centers[k, 0])
            if centers[k, 0] > 1.0 - r[k]:
                forces[k, 0] -= 100.0 * (centers[k, 0] - (1.0 - r[k]))
            if centers[k, 1] < r[k]:
                forces[k, 1] += 100.0 * (r[k] - centers[k, 1])
            if centers[k, 1] > 1.0 - r[k]:
                forces[k, 1] -= 100.0 * (centers[k, 1] - (1.0 - r[k]))
                
        centers += dt * forces
        centers = np.clip(centers, 0.0, 1.0)
    return centers

def hex_init(n, row_counts, r0=0.08):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for idx, cnt in enumerate(row_counts):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def joint_objective(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:])

def joint_constraints(vars_arr, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    
    c_list = []
    c_list.append(cx - r)
    c_list.append(1.0 - cx - r)
    c_list.append(cy - r)
    c_list.append(1.0 - cy - r)
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    r_sum = r[i_idx] + r[j_idx]
    c_list.append(dx**2 + dy**2 - r_sum**2)
    
    return np.concatenate(c_list)
