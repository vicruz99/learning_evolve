import numpy as np
from scipy.optimize import minimize, linprog

def eval_lp(centers, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    wall = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    wall = np.maximum(wall, 1e-9)
    bounds = [(0.0, w) for w in wall]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.full(n, 1e-9)

def obj_centers(x, n, A_ub, idx_i, idx_j):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = x.reshape(n, 2)
    s, _ = eval_lp(centers, A_ub, idx_i, idx_j)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    triu_idx = np.triu_indices(n, k=1)
    idx_i, idx_j = triu_idx
    m = len(idx_i)
    
    # Precompute constant LP constraint matrix A_ub: r_i + r_j <= d_ij
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    rng = np.random.default_rng(42)
    configs = []
    
    # Hexagonal lattice patterns summing to 26
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 4, 6, 4],
        [7, 7, 6, 6], [5, 5, 6, 5, 5], [6, 4, 6, 4, 6],
        [8, 6, 6, 6], [7, 6, 7, 6], [5, 7, 5, 5, 4]
    ]
    
    for pat in patterns:
        if sum(pat) != 26: 
            continue
        pts = []
        y = 0.09
        r0 = 0.09
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: 
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        cfg = np.array(pts[:n])
        
        # Normalize to fit comfortably inside [0.05, 0.95]
        mn = cfg.min(axis=0)
        mx = cfg.max(axis=0)
        scale = 0.85 / np.max(mx - mn)
        cfg = (cfg - mn) * scale + (1.0 - scale) / 2.0
        configs.append(cfg)
        
        # Add perturbed variants to break symmetry
        for _ in range(3):
            p = cfg + rng.uniform(-0.02, 0.02, cfg.shape)
            configs.append(np.clip(p, 0.05, 0.95))
            
    # Add random dense starts
    for _ in range(5):
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    args = (n, A_ub, idx_i, idx_j)
    
    # Phase 1: Multi-start Nelder-Mead optimization on centers
    # Using LP as an exact oracle for radii makes the objective continuous and well-behaved
    for cfg in configs:
        try:
            res = minimize(obj_centers, cfg.flatten(), args=args, method='Nelder-Mead',
                          options={'maxiter': 2500, 'xatol': 1e-9, 'fatol': 1e-10})
            centers_opt = res.x.reshape(n, 2)
            s, r = eval_lp(centers_opt, A_ub, idx_i, idx_j)
            if s > best_sum:
                best_sum = s
                best_centers = centers_opt.copy()
                best_radii = r.copy()
        except Exception:
            pass
            
    # Phase 2: Local refinement to escape local minima
    if best_centers is not None:
        current_centers = best_centers.copy()
        current_radii = best_radii.copy()
        current_sum = best_sum
        
        for _ in range(12):
            pert = current_centers + rng.normal(0, 0.004, current_centers.shape)
            pert = np.clip(pert, 0.01, 0.99)
            try:
                res = minimize(obj_centers, pert.flatten(), args=args, method='Nelder-Mead',
                              options={'maxiter': 1500, 'xatol': 1e-9, 'fatol': 1e-10})
                centers_opt = res.x.reshape(n, 2)
                s, r = eval_lp(centers_opt, A_ub, idx_i, idx_j)
                if s > current_sum + 1e-8:
                    current_sum = s
                    current_centers = centers_opt.copy()
                    current_radii = r.copy()
            except Exception:
                pass
                
        best_centers = current_centers
        best_radii = current_radii
        best_sum = current_sum
        
    # Phase 3: Strict numerical safety scaling
    scale = 1.0
    wall_lim = np.minimum(
        np.minimum(best_centers[:, 0], 1.0 - best_centers[:, 0]),
        np.minimum(best_centers[:, 1], 1.0 - best_centers[:, 1])
    )
    scale = min(scale, np.min(wall_lim / np.maximum(best_radii, 1e-12)))
    
    diff = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_pair = best_radii[:, np.newaxis] + best_radii[np.newaxis, :]
    scale = min(scale, np.min(dists[triu_idx] / np.maximum(r_pair[triu_idx], 1e-12)))
    
    # Apply high-precision shrinkage to strictly satisfy validator tolerances
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    # Fallback if optimization unexpectedly fails
    if best_centers is None:
        grid = np.array([(0.1 + i*0.18, 0.1 + j*0.18) for j in range(5) for i in range(5)])
        best_centers = np.vstack([grid, [[0.55, 0.55]]])
        best_radii = np.full(26, 0.08)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
