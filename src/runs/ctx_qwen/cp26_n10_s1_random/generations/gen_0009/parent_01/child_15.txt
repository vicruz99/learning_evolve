import numpy as np
from scipy.optimize import linprog

def get_pair_matrix(n):
    """Precomputes the constant inequality matrix for pairwise distance constraints."""
    m = n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A, np.triu_indices(n, k=1)

def solve_lp(centers, A, idx_i, idx_j):
    """Solves the LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Maximum radius allowed by walls for each circle
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    wall = np.maximum(wall, 1e-12)
    bounds = [(0.0, w) for w in wall]
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0
