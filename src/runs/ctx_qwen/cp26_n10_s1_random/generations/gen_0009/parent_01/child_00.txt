import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_matrix(n):
    """Constructs the inequality matrix for pairwise distance constraints."""
    m = n * (n - 1) // 2
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_lp(centers, A_ub, wall_dists, dists_ij):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    bounds = [(0.0, max(wd, 1e-12)) for wd in wall_dists]
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=dists_ij, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def obj_equal(x, n):
    """Objective for equal radius optimization: maximize t."""
    return -x[-1]

def cons_equal(x, n):
    """Constraints for equal radius optimization."""
    c = x[:2*n].reshape(n, 2)
    t = x[-1]
    c_list = []
    c_list.append(c[:, 0] - t)
    c_list.append(1.0 - c[:, 0] - t)
    c_list.append(c[:, 1] - t)
    c_list.append(1.0 - c[:, 1] - t)
    diff = c[:, np.newaxis, :] - c[np.newaxis, :]
    d2 = np.sum(diff**2, axis=2)
    idx = np.triu_indices(n, 1)
    c_list.append(d2[idx] - 4.0 * t**2)
    return np.concatenate(c_list)

def optimize_equal(centers0, n):
    """Runs SLSQP to maximize equal radius clearance."""
    x0 = np.concatenate([centers0.flatten(), [0.09]])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.08, 0.15)]
    try:
        res = minimize(obj_equal, x0, args=(n,), method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons_equal, 'args': (n,)},
                       options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if np.isfinite(res.fun) and res.x[-1] > 0.09:
            return res.x[:2*n].reshape(n, 2), res.x[-1]
    except Exception:
        pass
    return centers0, 0.0

def generate_inits(n, rng):
    """Generates diverse hexagonal and random initial configurations."""
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [6, 4, 6, 4, 6],
        [5, 6, 6, 5, 4], [7, 6, 7, 6], [6, 7, 6, 7]
    ]
    inits = []
    for pat in patterns:
        if sum(pat) != n: 
            continue
        pts = []
        y = 0.1
        r0 = 0.09
        for i, cnt in enumerate(pat):
            shift = r0 if i % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: 
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:n])
        # Center and scale to fit comfortably in [0.1, 0.9]
        mn = pts.min(axis=0)
        mx = pts.max(axis=0)
        scale = 0.7 / np.max(mx - mn)
        pts = (pts - mn) * scale + 0.5 * (1.0 - scale)
        inits.append(pts)
        # Add perturbed variants to break symmetry
        for _ in range(4):
            p = pts + rng.uniform(-0.03, 0.03, pts.shape)
            inits.append(np.clip(p, 0.05, 0.95))
    # Add random dense starts
    for _ in range(10):
        inits.append(rng.uniform(0.15, 0.85, (n, 2)))
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    A_ub = build_lp_matrix(n)
    idx_i, idx_j = np.triu_indices(n, 1)

    best_sum = 0.0
    best_centers = None
    best_radii = None

    inits = generate_inits(n, rng)

    # Phase 1: Equal radius optimization from diverse starts
    for cfg in inits:
        c_opt, t_opt = optimize_equal(cfg, n)
        if t_opt > 0.09:
            wall = np.minimum(np.minimum(c_opt[:, 0], 1.0 - c_opt[:, 0]),
                              np.minimum(c_opt[:, 1], 1.0 - c_opt[:, 1]))
            diff = c_opt[:, np.newaxis, :] - c_opt[np.newaxis, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            d_ij = dists[idx_i, idx_j]
            r_lp, s_lp = solve_lp(c_opt, A_ub, wall, d_ij)
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = c_opt.copy()
                best_radii = r_lp.copy()

    # Phase 2: Coordinate ascent local search using LP as oracle
    if best_centers is not None:
        curr_c = best_centers.copy()
        curr_r = best_radii.copy()
        curr_s = best_sum
        step_size = 0.025

        for step in range(6000):
            i = rng.integers(n)
            old_pos = curr_c[i].copy()
            curr_c[i] += rng.uniform(-step_size, step_size, 2)
            curr_c[i] = np.clip(curr_c[i], 1e-4, 1.0 - 1e-4)

            wall = np.minimum(np.minimum(curr_c[:, 0], 1.0 - curr_c[:, 0]),
                              np.minimum(curr_c[:, 1], 1.0 - curr_c[:, 1]))
            diff = curr_c[:, np.newaxis, :] - curr_c[np.newaxis, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            d_ij = dists[idx_i, idx_j]

            r_new, s_new = solve_lp(curr_c, A_ub, wall, d_ij)
            if s_new > curr_s + 1e-9:
                curr_s = s_new
                curr_r = r_new.copy()
                step_size = max(0.005, step_size * 0.9995)
            else:
                curr_c[i] = old_pos
                step_size = min(0.03, step_size * 1.001)

        best_centers = curr_c
        best_radii = curr_r
        best_sum = curr_s

    # Phase 3: Strict numerical safety scaling
    scale = 1.0
    wall_lim = np.minimum(np.minimum(best_centers[:, 0], 1.0 - best_centers[:, 0]),
                          np.minimum(best_centers[:, 1], 1.0 - best_centers[:, 1]))
    scale = min(scale, np.min(wall_lim / np.maximum(best_radii, 1e-12)))

    diff = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_pair = best_radii[:, np.newaxis] + best_radii[np.newaxis, :]
    scale = min(scale, np.min(dists[idx_i, idx_j] / np.maximum(r_pair[idx_i, idx_j], 1e-12)))

    # Apply minimal shrinkage to guarantee strict validity within 1e-12 tolerance
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))

    # Fallback configuration (should rarely be reached)
    if best_centers is None:
        grid = np.array([(0.1 + i*0.18, 0.1 + j*0.18) for j in range(5) for i in range(5)])
        best_centers = np.vstack([grid, [[0.55, 0.55]]])
        best_radii = np.full(26, 0.08)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
