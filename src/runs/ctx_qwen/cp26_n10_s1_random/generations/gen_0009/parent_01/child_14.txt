import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_radii(centers, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Distance to boundaries
    wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    wall = np.maximum(wall, 1e-9)
    bounds = [(0.0, w) for w in wall]
    
    # Pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def obj_single_circle(xy, centers, A_ub, idx_i, idx_j, idx):
    """Objective for optimizing a single circle's position to maximize LP sum."""
    nc = centers.copy()
    nc[idx] = np.clip(xy, 1e-4, 1.0 - 1e-4)
    _, s = solve_lp_radii(nc, A_ub, idx_i, idx_j)
    return -s
