import numpy as np
from scipy.optimize import minimize, linprog

def compute_constraints_slsqp(vars_arr, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = vars_arr[0:n]
    cy = vars_arr[n:2*n]
    r = vars_arr[2*n:3*n]
    
    c_list = []
    c_list.append(cx - r)
    c_list.append(1.0 - cx - r)
    c_list.append(cy - r)
    c_list.append(1.0 - cy - r)
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dr = r[i_idx] + r[j_idx]
    c_list.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(c_list)

def objective_slsqp(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*n:3*n])

def solve_radii_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    wall = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    wall = np.maximum(wall, 1e-12)
    bounds = [(0.0, w) for w in wall]
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[i_idx, j_idx]
    
    m = len(i_idx)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), i_idx] = 1.0
    A_ub[np.arange(m), j_idx] = 1.0
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def lp_sum_func(centers_flat, n):
    """Objective function for center optimization: negative max sum of radii."""
    centers = centers_flat.reshape(n, 2)
    _, s = solve_radii_lp(centers, n)
    return -s

def generate_hex_config(n, row_counts, r0, shift_x=0.0, shift_y=0.0, scale=1.0):
    """Generates initial hexagonal lattice configurations."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
        
    pts = np.array(pts[:n])
    center = np.mean(pts, axis=0)
    pts -= center
    pts *= scale
    pts += 0.5 + np.array([shift_x, shift_y])
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        [6, 7, 6, 7], [5, 6, 5, 5, 5], [6, 5, 4, 6, 5]
    ]
    
    configs = []

            continue
