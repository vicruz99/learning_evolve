import numpy as np
from scipy.optimize import linprog
import math

def max_sum_radii_lp(centers, i_idx, j_idx):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns (optimal_radii, sum_radii).
    """
    n = centers.shape[0]
    
    # Maximum radius allowed by walls for each circle
    wall = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    wall = np.maximum(wall, 0.0)
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    d_ij = dists[i_idx, j_idx]
    
    # Build LP: minimize -sum(r) s.t. r_i <= wall_i, r_i + r_j <= d_ij, r_i >= 0
    M = len(i_idx)
    A_ub = np.zeros((n + M, n))
    b_ub = np.zeros(n + M)
    
    # Wall constraints: r_i <= wall_i
    A_ub[:n] = np.eye(n)
    b_ub[:n] = wall
    
    # Pairwise constraints: r_i + r_j <= d_ij
    A_ub[n:, i_idx] = 1.0
    A_ub[n:, j_idx] = 1.0
    b_ub[n:] = d_ij
    
    bounds = [(0.0, w) for w in wall]
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback: equal small radii
    return np.full(n, 1e-8), 0.0

def sa_optimize(centers0, n, i_idx, j_idx, steps=4000, seed=0):
    """
    Simulated Annealing on centers to maximize the LP-sum of radii.
    """
    rng = np.random.default_rng(seed)
    centers = centers0.copy()
    radii, best_s = max_sum_radii_lp(centers, i_idx, j_idx)
    best_centers = centers.copy()
    best_radii = radii.copy()
    current_s = best_s
    
    T = 0.008
    T_min = 1e-8
    step_size = 0.025
    
    for t in range(steps):
        decay = (1.0 - t / steps) ** 0.4
        cur_step = step_size * decay
        
        # Perturb one circle
        i = rng.integers(n)
        old_pos = centers[i].copy()
        
        centers[i] += rng.uniform(-cur_step, cur_step, 2)
        centers[i] = np.clip(centers[i], 1e-5, 1.0 - 1e-5)
        
        radii, s = max_sum_radii_lp(centers, i_idx, j_idx)
        
        delta = s - current_s
        # Accept if better, or probabilistically if worse
        if delta > 0 or rng.random() < math.exp(min(delta / max(T, T_min), 50.0)):
            current_s = s
            if s > best_s:
                best_s = s
                best_centers = centers.copy()
                best_radii = radii.copy()
        else:
            centers[i] = old_pos
            
        T *= 0.9992
        
    return best_centers, best_radii, best_s

def generate_hex_starts(n, rng):
    """Generates diverse hexagonal lattice initializations."""
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 4, 6, 4],
        [5, 5, 6, 5, 5], [4, 6, 6, 4, 6], [6, 4, 6, 5, 5],
        [5, 7, 5, 5, 4], [7, 5, 5, 5, 4], [6, 5, 5, 6, 4]
    ]
    starts = []
    r0 = 0.102
    
    for pat in patterns:
        if sum(pat) != n:
            continue
        pts = []
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n:
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        
        pts = np.array(pts[:n])
        starts.append(pts)
        
        # Add perturbed variants
        for _ in range(3):
            p = pts + rng.uniform(-0.025, 0.025, pts.shape)
            starts.append(np.clip(p, 0.05, 0.95))
            
    # Add a few random dense starts
    for _ in range(5):
        starts.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    i_idx, j_idx = np.triu_indices(n, k=1)
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    starts = generate_hex_starts(n, rng)
    
    # Phase 1: SA from diverse starts
    for i, start_c in enumerate(starts):
        c, r, s = sa_optimize(start_c, n, i_idx, j_idx, steps=3500, seed=i)
        if s > best_sum:
            best_sum = s
            best_centers = c.copy()
            best_radii = r.copy()
            
    # Phase 2: Final strict numerical safety scaling
    # Ensure all constraints strictly satisfy the 1e-12 validation tolerance
    scale = 1.0
    
    # Wall constraints
    wall_lim = np.minimum(
        np.minimum(best_centers[:, 0], 1.0 - best_centers[:, 0]),
        np.minimum(best_centers[:, 1], 1.0 - best_centers[:, 1])
    )
    wall_ratios = wall_lim / np.maximum(best_radii, 1e-12)
    scale = min(scale, np.min(wall_ratios))
    
    # Pairwise constraints
    diff = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    pair_ratios = dists[i_idx, j_idx] / np.maximum(best_radii[i_idx] + best_radii[j_idx], 1e-12)
    scale = min(scale, np.min(pair_ratios))
    
    # Apply high-precision shrinkage
    safety_factor = 0.9999995
    best_radii *= scale * safety_factor
    best_sum = float(np.sum(best_radii))
    
    # Guaranteed fallback if optimization unexpectedly fails
    if best_centers is None:
        grid = np.array([(0.1 + i*0.18, 0.1 + j*0.18) for j in range(5) for i in range(5)])
        best_centers = np.vstack([grid, [[0.55, 0.55]]])
        best_radii = np.full(26, 0.08)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, best_sum
