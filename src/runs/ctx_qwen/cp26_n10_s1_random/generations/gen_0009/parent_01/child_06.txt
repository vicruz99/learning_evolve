

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we leverage the fact that for fixed center coordinates, finding the optimal radii is a Linear Programming (LP) problem. This allows us to separate the optimization into two phases:
1. **Center Optimization**: We optimize only the $(x, y)$ coordinates of the 26 circles. The objective function evaluates the maximum possible sum of radii via a fast LP solver (`scipy.optimize.linprog` with the `highs` method). This avoids the non-convex overlap constraints during the main search.
2. **Multi-Start & Global Search**: We generate diverse high-quality initial configurations using hexagonal lattice patterns (which are near-optimal for circle packing), random dense clusters, and a brief Differential Evolution run to escape local minima. Each configuration is refined using `Nelder-Mead` optimization.
3. **Fine-Tuning**: A coordinate-wise hill-climbing phase performs greedy local moves on the best configuration found, evaluated exactly via LP.
4. **Validation Safety**: Finally, we compute a strict scaling factor to ensure all circles satisfy the boundary and non-overlap constraints within the $10^{-12}$ numerical tolerance required by the validator.

This approach is highly efficient because the LP solver handles all radius constraints exactly and quickly, allowing gradient-free optimizers to focus solely on arranging centers.
</strategy>

```python
import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

# Global constants
N = 26
M_BOUND = 4 * N
M_PAIR = N * (N - 1) // 2

# Precompute the constant inequality matrix for the LP: A_ub @ r <= b_ub
# Structure: 4 boundary constraints per circle, 1 pairwise constraint per pair
A_ub_global = np.zeros((M_BOUND + M_PAIR, N))
k = 0
for i in range(N):
    for _ in range(4):
        A_ub_global[k, i] = 1.0
        k += 1
for i in range(N):
    for j in range(i + 1, N):
        A_ub_global[k, i] = 1.0
        A_ub_global[k, j] = 1.0
        k += 1

def solve_lp_radii(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns (sum_radii, radii_array).
    """
    b_ub = np.zeros(M_BOUND + M_PAIR)
    k = 0
    for i in range(N):
        x, y = centers[i]
        b_ub[k] = x; k += 1
        b_ub[k] = 1.0 - x; k += 1
        b_ub[k] = y; k += 1
        b_ub[k] = 1.0 - y; k += 1
        
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b_ub[k] = np.hypot(dx, dy)
            k += 1
            
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_global, b_ub=b_ub, 
                      bounds=(0, None), method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(N)

def objective_function(x):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = np.clip(x.reshape(N, 2), 1e-6, 1.0 - 1e-6)
    return -solve_lp_radii(centers)[0]

def generate_hex_config(row_counts, r0=0.09):
    """Generates a hexagonal lattice configuration with specified row counts."""
    pts = []
    y = r0
    for i, cnt in enumerate(row_counts):
        shift = r0 if i % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    pts = np.array(pts[:N])
    
    # Scale to fit comfortably within [0.1, 0.9]
    mn = pts.min()
    mx = pts.max()
    if mx > mn:
        pts = (pts - mn) / (mx - mn) * 0.8 + 0.1
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # 1. Generate diverse initial configurations
    starts = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5],
        [4,6,6,6,4], [6,5,5,6,4], [5,6,6,4,5], [7,6,7,6],
        [6,7,6,7], [5,5,5,5,6], [6,6,4,6,4], [4,4,6,6,6]
    ]
    for pat in patterns:
        if sum(pat) == N:
            starts.append(generate_hex_config(pat))
            
    for _ in range(12):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # 2. Local refinement from each start
    for cfg in starts:
        try:
            res = minimize(objective_function, cfg.flatten(), method='Nelder-Mead', 
                           options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10})
            s = -res.fun
            if s > best_sum:
                c_opt = np.clip(res.x.reshape(N, 2), 1e-5, 1-1e-5)
                _, r_opt = solve_lp_radii(c_opt)
                if np.sum(r_opt) > 0:
                    best_sum = s
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass
            
    # 3. Global search via Differential Evolution
    try:
        bounds = [(0.0, 1.0)] * (2 * N)
        res_de = differential_evolution(objective_function, bounds=bounds, 
                                        maxiter=150, popsize=10, seed=42)
        s_de = -res_de.fun
        if s_de > best_sum:
            c_de = np.clip(res_de.x.reshape(N, 2), 1e-5, 1-1e-5)
            _, r_de = solve_lp_radii(c_de)
            if np.sum(r_de) > 0:
                best_sum = s_de
                best_centers = c_de.copy()
                best_radii = r_de.copy()
                
        # Refine DE result
        res_ref = minimize(objective_function, res_de.x, method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-8, 'fatol': 1e-10})
        s_ref = -res_ref.fun
        if s_ref > best_sum:
            c_ref = np.clip(res_ref.x.reshape(N, 2), 1e-5, 1-1e-5)
            _, r_ref = solve_lp_radii(c_ref)
            if np.sum(r_ref) > 0:
                best_sum = s_ref
                best_centers = c_ref.copy()
                best_radii = r_ref.copy()
    except Exception:
        pass
        
    # 4. Coordinate-wise Hill Climbing for fine-tuning
    if best_centers is not None:
        c_curr = best_centers.copy()
        r_curr = best_radii.copy()
        s_curr = best_sum
        
        for step in range(3000):
            # Adaptive step size decays over iterations
            sz = 0.03 * (1.0 - step / 3000.0)**0.5
            i = rng.integers(N)
            old_pos = c_curr[i].copy()
            
            c_curr[i] += rng.uniform(-sz, sz, 2)
            c_curr[i] = np.clip(c_curr[i], 1e-5, 1-1e-5)
            
            s_new, r_new = solve_lp_radii(c_curr)
            if s_new > s_curr + 1e-8:
                s_curr = s_new
                r_curr = r_new.copy()
                best_centers = c_curr.copy()
                best_radii = r_curr.copy()
                best_sum = s_curr
            else:
                c_curr[i] = old_pos
                
    # 5. Strict numerical safety scaling
    if best_centers is None:
        # Guaranteed fallback
        best_centers = np.column_stack(np.meshgrid(np.linspace(0.1, 0.9, 5), np.linspace(0.1, 0.9, 5))).reshape(-1, 2)
        best_centers = np.vstack([best_centers, [[0.55, 0.55]]])
        _, best_radii = solve_lp_radii(best_centers)
        
    scale = 1.0
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1-x)/r, y/r, (1-y)/r)
            
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)
                
    # Apply high-precision shrinkage to strictly satisfy validator tolerances
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
```
