To maximize the sum of radii for 26 circles in a unit square, we exploit a key mathematical property: for any fixed set of centers, the problem of finding radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints is a Linear Programming (LP) problem. The optimal value of this LP, $V(\mathbf{c})$, is a concave function of the center positions $\mathbf{c}$. 

Therefore, we can decouple the optimization into two stages:
1. **Center Optimization**: Use derivative-free optimization (Powell's method) to maximize $V(\mathbf{c})$ over the center coordinates. This avoids the non-convex pairwise distance constraints inherent in joint non-linear optimization.
2. **Exact Radius Solution**: After finding optimal centers, solve the LP exactly to obtain the maximum feasible radii.
3. **Refinement**: Apply coordinate ascent to escape potential flat regions or local basins in the center space, evaluating the LP objective at each step.
4. **Validation Safety**: Apply a high-precision scaling factor to guarantee strict satisfaction of the $10^{-12}$ tolerance in the validator.

This approach is mathematically robust, highly parallelizable in evaluation, and directly targets the global structure of the sum-of-radii landscape.

import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp_for_radii(centers, n, idx_i, idx_j, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    wall = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    bounds = [(0.0, max(w, 1e-12)) for w in wall]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def lp_objective(centers_flat, n, idx_i, idx_j, A_ub):
    """Objective for center optimization: negative LP optimum value."""
    centers = np.clip(centers_flat.reshape((n, 2)), 1e-5, 1.0 - 1e-5)
    _, val = solve_lp_for_radii(centers, n, idx_i, idx_j, A_ub)
    return -val if val > 0 else 5.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    triu_idx = np.triu_indices(n, k=1)
    idx_i, idx_j = triu_idx
    
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    # Generate diverse high-quality initial configurations
    inits = []
    
    # Hexagonal lattice patterns summing to 26
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5], 
        [5, 5, 6, 5, 5], [6, 6, 4, 6, 4], [6, 4, 6, 4, 6]
    ]
    
    for pat in patterns:
        pts = []
        y_curr = 0.09
        r0 = 0.09
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x_curr = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x_curr, y_curr])
                x_curr += 2.0 * r0
            y_curr += np.sqrt(3.0) * r0
        inits.append(np.array(pts[:n]).flatten())
        
    # Structured grid + center
    grid_pts = []
    for i in range(5):
        for j in range(5):
            grid_pts.append([0.1 + i * 0.18, 0.1 + j * 0.18])
    grid_pts.append([0.5, 0.5])
    inits.append(np.array(grid_pts[:n]).flatten())
    
    # Random starts to escape symmetric basins
    rng = np.random.default_rng(42)
    for _ in range(10):
        inits.append(rng.uniform(0.12, 0.88, (n, 2)).flatten())
        
    best_sum = -1.0
    best_centers = None
    
    # Phase 1: Powell optimization on centers via LP oracle
    for x0 in inits:
        try:
            res = minimize(lp_objective, x0, args=(n, idx_i, idx_j, A_ub), 
                           method='Powell', 
                           options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = np.clip(res.x.reshape((n, 2)), 1e-5, 1.0 - 1e-5)
        except Exception:
            pass

    # Phase 2: Coordinate ascent refinement to polish local optimum
    if best_centers is not None:
        current_centers = best_centers.copy()
        for _ in range(3000):
            i = rng.integers(n)
            old_pos = current_centers[i].copy()
            
            # Perturb one center
            current_centers[i] += rng.uniform(-0.015, 0.015, 2)
            current_centers[i] = np.clip(current_centers[i], 1e-5, 1.0 - 1e-5)
            
            # Evaluate exact LP sum
            _, new_sum = solve_lp_for_radii(current_centers, n, idx_i, idx_j, A_ub)
            
            if new_sum > best_sum + 1e-9:
                best_sum = new_sum
                best_centers = current_centers.copy()
            else:
                current_centers[i] = old_pos
                
        final_centers = best_centers
    else:
        # Fallback to a valid configuration
        final_centers = np.array([[0.2 + i*0.15, 0.2 + j*0.15] for j in range(5) for i in range(5)] + [[0.5, 0.5]])
        final_centers = final_centers[:n]
        
    # Phase 3: Final exact LP solution for radii
    best_radii, best_sum = solve_lp_for_radii(final_centers, n, idx_i, idx_j, A_ub)
    if best_radii is None:
        best_radii = np.full(n, 0.085)
        best_sum = n * 0.085

    # Phase 4: Strict numerical safety scaling
    scale = 1.0
    wall_lim = np.minimum(
        np.minimum(final_centers[:, 0], 1.0 - final_centers[:, 0]),
        np.minimum(final_centers[:, 1], 1.0 - final_centers[:, 1])
    )
    scale = min(scale, np.min(wall_lim / np.maximum(best_radii, 1e-12)))
    
    diff = final_centers[:, None, :] - final_centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_pair = best_radii[:, None] + best_radii[None, :]
    scale = min(scale, np.min(dists[triu_idx] / np.maximum(r_pair[triu_idx], 1e-12)))
    
    # Apply high-precision shrinkage to guarantee validator tolerance
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return final_centers, best_radii, best_sum
