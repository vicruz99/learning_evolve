import numpy as np
from scipy.optimize import minimize, linprog

def get_constraints_slsqp(vars_arr, n, triu_idx):
    """Inequality constraints >= 0 for SLSQP: boundaries and pairwise non-overlap."""
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    
    c_list = []
    # Boundary constraints
    c_list.append(x - r)
    c_list.append(1.0 - x - r)
    c_list.append(y - r)
    c_list.append(1.0 - y - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    c_list.append(dist_sq[triu_idx] - r_sum_sq[triu_idx])
    return np.concatenate(c_list)

def objective_slsqp(vars_arr, n):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars_arr[2::3])

def solve_lp_radii(centers, A_ub, idx_i, idx_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Maximum radius allowed by boundaries
    wall_dists = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    wall_dists = np.maximum(wall_dists, 1e-7)
    bounds = [(0.0, lim) for lim in wall_dists]
    
    # Pairwise distance limits
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def neg_lp_obj(cf, n, A_ub, idx_i, idx_j):
    """Objective for Nelder-Mead: negative max sum of radii for given centers."""
    c = cf.reshape(n, 2)
    # Clip to ensure centers stay valid for LP
    c_clipped = np.clip(c, 1e-4, 0.9999)
    r, s = solve_lp_radii(c_clipped, A_ub, idx_i, idx_j)
    return -s if r is not None else -1.0
