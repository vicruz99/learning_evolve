import numpy as np
from scipy.optimize import minimize, linprog

def build_lp_A_ub(n):
    """Precomputes the constant inequality matrix for the LP radius solver."""
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    m = m_bound + m_pair
    A = np.zeros((m, n))
    k = 0
    for i in range(n):
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
        A[k, i] = 1.0; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            k += 1
    return A

def solve_radii_lp_fast(centers, A_ub):
    """Solves LP to maximize sum of radii for fixed centers using precomputed matrix."""
    n = centers.shape[0]
    m_bound = 4 * n
    m = A_ub.shape[0]
    b = np.zeros(m)
    
    # Boundary limits
    for i in range(n):
        x, y = centers[i]
        b[4*i] = x
        b[4*i+1] = 1.0 - x
        b[4*i+2] = y
        b[4*i+3] = 1.0 - y
        
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b[m_bound:] = dists[np.triu_indices(n, k=1)]
            
    bounds = [(0.0, 0.5)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def force_simulate(centers, steps=600):
    """Force-directed repulsion simulation to spread circles and expand radii."""
    n = centers.shape[0]
    r = np.full(n, 0.08)
    dt = 0.012
    k_rep = 60.0
    k_wall = 25.0
    
    for _ in range(steps):
        forces = np.zeros_like(centers)
        r *= 1.00008
        
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        req = r[:, np.newaxis] + r[np.newaxis, :]
        
        overlap = np.maximum(0.0, req - dists)
        rep = (k_rep * overlap) / np.maximum(dists, 1e-9)
        fx = np.sum(diff[:, :, 0] * rep, axis=1)
        fy = np.sum(diff[:, :, 1] * rep, axis=1)
        forces[:, 0] += fx
        forces[:, 1] += fy
        
        for i in range(n):
            if centers[i, 0] - r[i] < 0: forces[i, 0] += k_wall * (r[i] - centers[i, 0])
            if centers[i, 0] + r[i] > 1.0: forces[i, 0] -= k_wall * (centers[i, 0] + r[i] - 1.0)
            if centers[i, 1] - r[i] < 0: forces[i, 1] += k_wall * (r[i] - centers[i, 1])
            if centers[i, 1] + r[i] > 1.0: forces[i, 1] -= k_wall * (centers[i, 1] + r[i] - 1.0)
            
        centers += forces * dt
        centers = np.clip(centers, 0.001, 0.999)
        
    return centers, r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    A_ub = build_lp_A_ub(n)
    
    best_centers = None
    best_radii = None
