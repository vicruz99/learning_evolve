import numpy as np
from scipy.optimize import minimize, linprog

N = 26
triu_i, triu_j = np.triu_indices(N, k=1)
n_pairs = len(triu_i)

# Precompute constant LP constraint matrix structure
# Constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i, r_i + r_j <= dist(i,j)
A_ub_lp = np.zeros((4 * N + n_pairs, N))
for i in range(N):
    A_ub_lp[4 * i, i] = 1.0
    A_ub_lp[4 * i + 1, i] = 1.0
    A_ub_lp[4 * i + 2, i] = 1.0
    A_ub_lp[4 * i + 3, i] = 1.0
for k, (i, j) in enumerate(zip(triu_i, triu_j)):
    A_ub_lp[4 * N + k, i] = 1.0
    A_ub_lp[4 * N + k, j] = 1.0

def solve_lp_radii(centers, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    b_ub = np.empty(4 * n + n_pairs)
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        b_ub[4 * i] = x
        b_ub[4 * i + 1] = 1.0 - x
        b_ub[4 * i + 2] = y
        b_ub[4 * i + 3] = 1.0 - y
        
    # Pairwise distance constraints
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    b_ub[4 * n:] = dists[triu_i, triu_j]
    
    bounds = [(0.0, 0.5)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 1e-6)

def get_constraints_joint(vars_arr, N, triu_i, triu_j):
    """Inequality constraints >= 0 for valid packing (centers + radii)."""
    cx = vars_arr[0::3]
    cy = vars_arr[1::3]
    r = vars_arr[2::3]

    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)

    dy = cy[:, None] - cy[None, :]
    dist_sq = dx ** 2 + dy ** 2
    r_sum = r[:, None] + r[None, :]
    c.append(dist_sq[triu_i, triu_j] - r_sum[triu_i, triu_j] ** 2)
    
    return np.concatenate(c)

def objective_joint(vars_arr):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2::3])

    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])
