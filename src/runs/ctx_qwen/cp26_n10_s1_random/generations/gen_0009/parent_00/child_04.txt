N = 26

def compute_constraints(vars_arr):
    """Inequality constraints >= 0 for valid packing."""
    x = vars_arr[0:N]
    y = vars_arr[N:2*N]
    r = vars_arr[2*N:]
    c = []
    # Boundary constraints
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    # Pairwise non-overlap constraints
    idx_i, idx_j = np.triu_indices(N, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def objective_func(vars_arr):

    return -np.sum(vars_arr[2*N:])

    n = centers.shape[0]
    diffs = centers[pair_idx[:,0]] - centers[pair_idx[:,1]]

        res = linprog(-np.ones(n), A_ub=A_ub_pair, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:

    except Exception:
        pass
    return np.full(n, 1e-6), 0.0
