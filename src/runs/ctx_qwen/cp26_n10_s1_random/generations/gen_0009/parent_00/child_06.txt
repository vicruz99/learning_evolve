import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_lp_sum(centers, A_full, pair_i, pair_j):
    """Solves LP to maximize sum of radii for fixed centers."""
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-9)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    d_ij = dists[pair_i, pair_j]
    
    b_ub = np.concatenate([lims, d_ij])
    bounds = [(0.0, lim) for lim in lims]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_full, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0

def objective_func(x, A_full, pair_i, pair_j):
    """Objective: minimize negative LP sum of radii."""
    c = np.clip(x.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    _, s = compute_lp_sum(c, A_full, pair_i, pair_j)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    
    # Precompute constant LP constraint structure
    pair_i = []
    pair_j = []
    for i in range(N):
        for j in range(i + 1, N):
            pair_i.append(i)
            pair_j.append(j)
    pair_i = np.array(pair_i)
    pair_j = np.array(pair_j)
    n_pairs = len(pair_i)
    
    A_full = np.zeros((N + n_pairs, N))
    A_full[:N, :] = np.eye(N)
    A_full[N:, pair_i] = 1.0
    A_full[N:, pair_j] = 1.0
    
    # Generate diverse initial configurations
    inits = []
    patterns = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
                [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [7, 6, 7, 6]]
                
    for pat in patterns:
        if sum(pat) < N: continue
        pts = []
        y = 0.12
        for ri, cnt in enumerate(pat):
            shift = 0.06 if ri % 2 == 1 else 0.0
            x = 0.12 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 0.18
            y += 0.15
        pts = np.array(pts[:N])
        
        # Try rotated and normalized versions to fit square boundaries better
        for angle in [0.0, 0.05, -0.05]:

            rot = np.array([[c_ang, -s_ang], [s_ang, c_ang]])
            r_pts = pts @ rot
            r_pts = (r_pts - r_pts.min(0)) / (r_pts.max(0) - r_pts.min(0)) * 0.8 + 0.1
            inits.append(r_pts.flatten())
            
            # Add perturbed variants
            for _ in range(2):
                p = r_pts + rng.uniform(-0.015, 0.015, r_pts.shape)
                inits.append(np.clip(p, 0.05, 0.95).flatten())

    # Add fully random starts
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)).flatten())

    best_c = None
    best_r = None
    best_s = 0.0
    
    # Phase 1: Multi-start derivative-free optimization on centers
    for x0 in inits:
        try:
            res = minimize(objective_func, x0, args=(A_full, pair_i, pair_j), 
                          method='Nelder-Mead',
                          options={'maxiter': 400, 'xatol': 1e-7, 'fatol': 1e-8})
            c_opt = np.clip(res.x.reshape(N, 2), 1e-4, 1.0 - 1e-4)
            r_opt, s_opt = compute_lp_sum(c_opt, A_full, pair_i, pair_j)

                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:

                    best_r = radii.copy()

            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999999
    final_sum = float(np.sum(best_r))
    
    return best_c, best_r, final_sum
