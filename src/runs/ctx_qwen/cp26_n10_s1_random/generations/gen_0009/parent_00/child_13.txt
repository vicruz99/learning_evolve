from scipy.optimize import minimize, linprog

N = 26

def get_constraints(vars_arr, n):
    """Inequality constraints >= 0 for valid packing."""
    x = vars_arr[0::3]
    y = vars_arr[1::3]
    r = vars_arr[2::3]
    cons = []
    # Boundary constraints
    cons.append(x - r)
    cons.append(1.0 - x - r)
    cons.append(y - r)
    cons.append(1.0 - y - r)
    # Pairwise non-overlap constraints (squared form)
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    cons.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(cons)

def objective(vars_arr):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2::3])

def solve_lp_radii(centers, A_ub):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    b_ub = np.zeros(A_ub.shape[0])
    k = 0
    for i in range(n):
        x, y = centers[i]
        b_ub[k] = x; k += 1
        b_ub[k] = 1.0 - x; k += 1
        b_ub[k] = y; k += 1
        b_ub[k] = 1.0 - y; k += 1
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub[k] = d
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 1e-6)
