

<strategy>
The problem is to maximize the sum of radii of 26 circles packed in a unit square. This is equivalent to finding center positions that maximize the optimal value of a linear program (LP) where variables are radii, constrained by pairwise distances and boundary distances.

Key insights for improvement:
1. **LP Oracle for Radii**: For any fixed set of centers, the maximum sum of radii is found exactly by solving a small LP (26 variables, ~351 constraints). This turns the problem into a black-box optimization over center coordinates.
2. **Multi-Start Optimization**: The objective landscape over centers is highly non-convex and non-smooth. Using `scipy.optimize.minimize` with `method='Powell'` (which handles bounds and non-smooth functions well) from diverse initial configurations (hexagonal lattices, perturbed lattices, random starts) helps escape poor local minima.
3. **Intensive Local Search**: After global search, a directed hill-climbing phase perturbs individual circle positions while re-solving the LP. This fine-tunes contacts and exploits local geometric improvements.
4. **Precomputation & Speed**: The LP constraint matrix structure is constant and precomputed. `scipy.optimize.linprog` with the `highs` solver is extremely fast (~0.5ms), allowing thousands of evaluations within time limits.
5. **Numerical Safety**: A final scaling step guarantees strict satisfaction of the validator's `1e-12` tolerance without significantly reducing the objective.

This strategy systematically explores the configuration space, leverages exact LP solutions for radii, and uses robust derivative-free optimization to push the sum of radii beyond 2.636.

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
triu_i, triu_j = np.triu_indices(N, k=1)
n_pairs = len(triu_i)

# Precompute LP constraint matrix structure for pairwise and boundary constraints
A_lp_pairs = np.zeros((n_pairs, N))
for k, (i, j) in enumerate(zip(triu_i, triu_j)):
    A_lp_pairs[k, i] = 1.0
    A_lp_pairs[k, j] = 1.0

A_full = np.vstack([np.eye(N), A_lp_pairs])

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    lims = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]), 
                      np.minimum(centers[:,1], 1.0-centers[:,1]))
    lims = np.maximum(lims, 1e-9)
    
    # Pairwise distances: r_i + r_j <= dist_ij
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_pairs = dists[triu_i, triu_j]
    
    b_full = np.concatenate([lims, b_pairs])
    bounds = [(0, None)] * N
    
    try:
        res = linprog(-np.ones(N), A_ub=A_full, b_ub=b_full, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-6), 0.0

def neg_sum_radii(x):
    """Objective for minimization: negative sum of radii."""
    centers = x.reshape(N, 2)
    _, s = solve_lp(centers)
    return -s

def hex_init(rows, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [6, 6, 6, 4, 4], [4, 5, 6, 5, 6], [5, 4, 6, 5, 6], [5, 6, 6, 5, 4]
    ]
    
    starts = []
    for p in patterns:
        if sum(p) >= N:
            c = hex_init(p)
            starts.append(c.flatten())
            cp = c + rng.uniform(-0.02, 0.02, c.shape)
            cp = np.clip(cp, 0.05, 0.95)
            starts.append(cp.flatten())
            
    for _ in range(15):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)).flatten())
        
    best_sum = 0.0
    best_centers = None
    
    bounds_xy = [(0.0, 1.0) for _ in range(2*N)]
    
    # Phase 1: Multi-start derivative-free optimization
    for s in starts:
        res = minimize(neg_sum_radii, s, method='Powell', 
                       bounds=bounds_xy,
                       options={'maxiter': 400, 'ftol': 1e-13, 'xtol': 1e-10})
        c_opt = res.x.reshape(N, 2)
        r_opt, s_opt = solve_lp(c_opt)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            
    if best_centers is None:
        best_centers = hex_init([5, 6, 5, 6, 4])
        
    # Phase 2: Intensive Local Search (Hill Climbing with LP oracle)
    centers = best_centers.copy()
    radii, current_sum = solve_lp(centers)
    best_c = centers.copy()
    best_r = radii.copy()
    
    step = 0.015
    for it in range(3000):
        idx = rng.integers(N)
        best_move = None
        best_val = current_sum
        
        # Try multiple random displacements for the selected circle
        for _ in range(8):
            trial = centers[idx].copy()
            trial += rng.uniform(-step, step, 2)
            trial = np.clip(trial, 1e-4, 1.0-1e-4)
            centers[idx] = trial
            _, val = solve_lp(centers)
            if val > best_val:
                best_val = val
                best_move = trial.copy()
            centers[idx] = best_c[idx]
            
        if best_move is not None:
            best_c[idx] = best_move
            _, current_sum = solve_lp(best_c)
            if current_sum > best_sum:
                best_sum = current_sum
                best_r, _ = solve_lp(best_c)
                
        step *= 0.9995
        
    # Phase 3: Final safety scaling to guarantee strict numerical validity
    scale = 1.0
    for i in range(N):
        x, y, r = best_c[i,0], best_c[i,1], best_r[i]
        if r > 1e-9:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
            rs = best_r[i] + best_r[j]
            if rs > 1e-9:
                scale = min(scale, d/rs)
                
    best_r *= scale * 0.9999999
    return best_c, best_r, float(np.sum(best_r))
```
