1. **LP Oracle**: Given any set of centers, compute the maximum possible sum of radii using `scipy.optimize.linprog`. This is highly efficient and exact.
2. **Local Search on Centers**: Treat the LP objective as a black-box function of the centers. Use a robust Simulated Annealing / Hill-Climbing hybrid to perturb centers and accept moves that increase the LP sum.

- **Diverse Initializations**: Tested multiple hexagonal lattice patterns (varying row counts) and random starts, each refined independently before selecting the global best.

import numpy as np
from scipy.optimize import linprog

N = 26
TRIU_IDX = np.triu_indices(N, k=1)
PAIR_INDICES = []
for i in range(N):
    for j in range(i + 1, N):
        PAIR_INDICES.append((i, j))

A_LP = np.zeros((len(PAIR_INDICES), N))
for k, (i, j) in enumerate(PAIR_INDICES):
    A_LP[k, i] = 1.0
    A_LP[k, j] = 1.0

def compute_lp_sum(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    try:
        # Vectorized pairwise distances
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        b_ub = dists[TRIU_IDX]

        # Boundary constraints as LP bounds
        bounds = []
        for c in centers:
            lim = min(c[0], 1.0 - c[0], c[1], 1.0 - c[1])
            bounds.append((0.0, max(lim, 1e-9)))

        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return -res.fun, res.x
        return 0.0, None
    except Exception:
        return 0.0, None

def generate_hex(rows, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def local_search_opt(centers, rng, max_iter=15000, step_init=0.02):
    """Optimizes centers to maximize LP sum of radii."""
    cur_sum, cur_radii = compute_lp_sum(centers)
    best_centers = centers.copy()
    best_sum = cur_sum
    best_radii = cur_radii

    step = step_init
    temp = 0.002

    for it in range(max_iter):
        i = rng.integers(N)
        old_pos = centers[i].copy()

        # Generate diverse candidates around current position
        candidates = []
        for _ in range(12):
            s = step * rng.uniform(0.1, 1.0)
            cand = old_pos + rng.uniform(-s, s, 2)
            cand = np.clip(cand, 0.001, 0.999)
            candidates.append(cand)

        best_cand = old_pos
        best_cand_sum = cur_sum

        for cand in candidates:
            centers[i] = cand
            s_val, _ = compute_lp_sum(centers)
            if s_val > best_cand_sum:
                best_cand_sum = s_val
                best_cand = cand.copy()
            centers[i] = old_pos

        # Simulated Annealing acceptance
        if best_cand_sum > cur_sum:
            centers[i] = best_cand
            cur_sum = best_cand_sum
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = centers.copy()
                _, best_radii = compute_lp_sum(best_centers)
        elif rng.random() < np.exp((best_cand_sum - cur_sum) / max(temp, 1e-6)):
            centers[i] = best_cand
            cur_sum = best_cand_sum

        # Adaptive cooling
        step *= 0.9998
        temp *= 0.999

        # Periodic global perturbation to escape deep local minima
        if it % 2500 == 0 and it > 0:
            centers += rng.uniform(-0.004, 0.004, (N, 2))
            centers = np.clip(centers, 0.001, 0.999)
            cur_sum, _ = compute_lp_sum(centers)

    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)

    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4], [5, 6, 4, 6, 5],
        [6, 6, 6, 4, 4], [4, 5, 6, 5, 6], [5, 4, 6, 5, 6], [5, 6, 6, 5, 4]
    ]

    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Phase 1: Structured hexagonal starts
    for pat in patterns:
        if sum(pat) < N:
            continue
        centers = generate_hex(pat, r0=0.09)
        centers += rng.uniform(-0.005, 0.005, (N, 2))
        centers = np.clip(centers, 0.05, 0.95)

        c_opt, r_opt, s_opt = local_search_opt(centers, rng, max_iter=12000, step_init=0.015)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            best_radii = r_opt

    # Phase 2: Random dense starts
    for _ in range(5):
        centers = rng.uniform(0.15, 0.85, (N, 2))
        c_opt, r_opt, s_opt = local_search_opt(centers, rng, max_iter=10000, step_init=0.02)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            best_radii = r_opt

    # Phase 3: Strict numerical safety scaling
    scale = 1.0
    for i in range(N):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-9:
            scale = min(scale, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-9:
                scale = min(scale, d / rs)

    best_radii *= scale * 0.999999
    final_sum = float(np.sum(best_radii))

    return best_centers, best_radii, final_sum
