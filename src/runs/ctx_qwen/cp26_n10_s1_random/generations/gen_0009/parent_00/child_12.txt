import numpy as np
from scipy.optimize import minimize, linprog

N = 26
triu_i, triu_j = np.triu_indices(N, k=1)
num_pairs = len(triu_i)

# Precompute constant LP constraint matrix for pairwise distances: r_i + r_j <= dist_ij
A_ub_lp = np.zeros((num_pairs, N))
for k, (i, j) in enumerate(zip(triu_i, triu_j)):
    A_ub_lp[k, i] = 1.0
    A_ub_lp[k, j] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    centers = np.clip(centers, 0.001, 0.999)
    b_ub = np.empty(num_pairs)
    for k in range(num_pairs):
        i, j = triu_i[k], triu_j[k]
        b_ub[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        
    bounds = []
    for i in range(N):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(lim, 1e-9)))
        
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_lp, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-9), 0.0

def get_hex(rows, r0=0.1):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])

def obj_fn(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * n:])

def cons_fn(v, n, ti, tj):
    """Inequality constraints >= 0 for valid packing."""
    cx, cy, r = v[:n], v[n:2 * n], v[2 * n:]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c = np.concatenate([c, d2[ti, tj] - rs[ti, tj]**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

            configs.append(c.copy())
            # Perturbed variants to break symmetry
            configs.append(c.copy() + rng.uniform(-0.02, 0.02, (N, 2)))
            
    # Random dense starts

        configs.append(rng.uniform(0.1, 0.9, (N, 2)))

                scale = min(scale, d / rs)
