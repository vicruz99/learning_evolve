from scipy.optimize import minimize, linprog

N = 26
triu_i, triu_j = np.triu_indices(N, k=1)
n_pairs = len(triu_i)

# Precompute LP inequality matrix structure for pairwise constraints
A_ub_pairs = np.zeros((n_pairs, N))
for k, (i, j) in enumerate(zip(triu_i, triu_j)):
    A_ub_pairs[k, i] = 1.0
    A_ub_pairs[k, j] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Compute pairwise distances
    dx = centers[:, np.newaxis, 0] - centers[np.newaxis, :, 0]
    dy = centers[:, np.newaxis, 1] - centers[np.newaxis, :, 1]
    d_ij = np.sqrt(dx**2 + dy**2)
    b_pairs = d_ij[triu_i, triu_j]
    
    # Upper bounds for radii based on distance to square boundaries
    bounds = []
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(lim, 1e-9)))
        
    # Solve LP: max sum(r) <=> min -sum(r)
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_pairs, b_ub=b_pairs, 
                      bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def lp_objective(x_flat, clip_low, clip_high):
    """Objective for derivative-free optimization: maximize LP sum."""
    centers = x_flat.reshape(N, 2)
    centers = np.clip(centers, clip_low, clip_high)

    return -s

                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0

        pts.append([0.5, 0.5])
    return np.array(pts[:N])
