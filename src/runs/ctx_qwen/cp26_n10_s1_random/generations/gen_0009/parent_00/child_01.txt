from scipy.optimize import minimize, linprog

N = 26
triu_i, triu_j = np.triu_indices(N, k=1)
N_PAIRS = len(triu_i)

# Precompute LP constraint matrix structure (constant for fixed N)
A_ub_struct = np.zeros((4*N + N_PAIRS, N))
for i in range(N):
    A_ub_struct[i, i] = 1.0
    A_ub_struct[N+i, i] = 1.0
    A_ub_struct[2*N+i, i] = 1.0
    A_ub_struct[3*N+i, i] = 1.0
for k, (i, j) in enumerate(zip(triu_i, triu_j)):
    A_ub_struct[4*N+k, i] = 1.0
    A_ub_struct[4*N+k, j] = 1.0

def compute_constraints(vars_arr):
    """Inequality constraints >= 0 for valid packing."""
    x = vars_arr[:N]
    y = vars_arr[N:2*N]
    r = vars_arr[2*N:3*N]
    
    c = np.empty(4*N + N_PAIRS)
    # Boundary constraints
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    c[4*N:] = dist_sq[triu_i, triu_j] - r_sum[triu_i, triu_j]**2
    
    return c

def objective(vars_arr):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2*N:])

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    lims = np.min(np.column_stack([centers[:, 0], 1.0 - centers[:, 0], 
                                   centers[:, 1], 1.0 - centers[:, 1]]), axis=1)
    lims = np.maximum(lims, 1e-9)
    
    b_ub = np.empty(4*n + N_PAIRS)
    b_ub[:N] = centers[:, 0]
    b_ub[N:2*N] = 1.0 - centers[:, 0]
    b_ub[2*N:3*N] = centers[:, 1]
    b_ub[3*N:4*N] = 1.0 - centers[:, 1]

    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub[4*N:] = dists[triu_i, triu_j]
    
    bounds = [(0.0, lims[i]) for i in range(n)]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_struct, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def make_hex_init(rows, r0=0.095):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])
