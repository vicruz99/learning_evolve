from scipy.optimize import minimize, linprog

N = 26
triu_i, triu_j = np.triu_indices(N, k=1)
PAIRS = list(zip(triu_i, triu_j))

def build_lp_matrix():
    """Builds the constant inequality matrix for the LP radius solver."""
    m_bound = 4 * N
    m_pair = N * (N - 1) // 2
    A = np.zeros((m_bound + m_pair, N))
    k = 0
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(N):
        for _ in range(4):
            A[k, i] = 1.0
            k += 1
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i, j in PAIRS:
        A[k, i] = 1.0
        A[k, j] = 1.0
        k += 1
    return A

LP_A = build_lp_matrix()

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers using precomputed A."""
    b_bound = np.empty(4 * N)
    k = 0
    for i in range(N):
        x, y = centers[i]
        b_bound[k] = x; k += 1
        b_bound[k] = 1.0 - x; k += 1
        b_bound[k] = y; k += 1
        b_bound[k] = 1.0 - y; k += 1
        
    b_pair = np.empty(len(PAIRS))
    for k, (i, j) in enumerate(PAIRS):
        b_pair[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])

    except Exception:
        pass
    return np.full(N, 1e-6), 0.0

def compute_constraints(vars_arr):
    """Inequality constraints >= 0 for valid packing."""
    x = vars_arr[0:N]
    y = vars_arr[N:2*N]
    r = vars_arr[2*N:3*N]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)

    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    c.append(dist_sq[triu_i, triu_j] - r_sum[triu_i, triu_j]**2)
    
    return np.concatenate(c)

def objective(vars_arr):
    return -np.sum(vars_arr[2*N:3*N])

def make_hex(rows, r0=0.095):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])
