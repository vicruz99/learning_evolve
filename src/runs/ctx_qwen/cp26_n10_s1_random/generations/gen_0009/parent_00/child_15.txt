N = 26
triu_i, triu_j = np.triu_indices(N, k=1)
idx_pairs = list(zip(triu_i, triu_j))
A_ub_pre = np.zeros((len(idx_pairs), N))
for k, (i, j) in enumerate(idx_pairs):
    A_ub_pre[k, i] = 1.0

        lim = min(x, 1.0 - x, y, 1.0 - y)

        y += np.sqrt(3.0) * r0

        pts.append([0.5, 0.5])
    return np.array(pts[:N])
