import numpy as np
from scipy.optimize import minimize, linprog

N = 26
triu_i, triu_j = np.triu_indices(N, k=1)
n_pairs = len(triu_i)

def compute_lp_radii(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j), r_i <= min_dist_to_wall
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((n_pairs + 4*n, n))
    b_ub = np.zeros(n_pairs + 4*n)
    
    # Pairwise non-overlap: r_i + r_j <= d_ij
    A_ub[:n_pairs, triu_i] = 1.0
    A_ub[:n_pairs, triu_j] = 1.0
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_ub[:n_pairs] = dists[triu_i, triu_j]
    
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        idx = n_pairs + 4*i
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x
        idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x
        idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y
        idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def s_obj(vars):
    """Objective: minimize negative sum of radii => maximize sum of radii"""
    return -np.sum(vars[2::3])

def s_cons(vars):
    """Inequality constraints >= 0 for SLSQP"""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    c = np.empty(4*N + n_pairs)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r

    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c[4*N:] = dist2[triu_i, triu_j] - rs[triu_i, triu_j]**2
    return c

def make_hex(rows, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
    while len(pts) < N:
        pts.append([0.5, 0.5])
    return np.array(pts[:N])
