import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((5 * n + n_pairs, n))
    b_ub = np.zeros(5 * n + n_pairs)
    bounds = []
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(lim, 1e-9)))
        for val in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[k, i] = 1.0
            b_ub[k] = val
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = d
            k += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.isfinite(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def force_pack(centers, r_init, steps=2000):
    """Force-directed repulsion simulation to spread circles evenly."""
    n = centers.shape[0]
    c = centers.copy()
    r = np.full(n, r_init)
    dt = 0.008
    k_rep = 80.0
    k_wall = 400.0
    
    for _ in range(steps):
        r *= 1.00004
        forces = np.zeros_like(c)
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = c[i] - c[j]
                dist = np.linalg.norm(diff)
                req = r[i] + r[j]
                if dist < req and dist > 1e-6:
                    f = k_rep * (req - dist) / dist
                    forces[i] += f * diff
                    forces[j] -= f * diff
                    
            for d in range(2):
                if c[i, d] - r[i] < 0:
                    forces[i, d] += k_wall * (r[i] - c[i, d])
                if c[i, d] + r[i] > 1.0:
                    forces[i, d] -= k_wall * (c[i, d] + r[i] - 1.0)
                    
        c += forces * dt
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
    return c

def objective(vars_arr, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_arr[2 * n:])

def constraints(vars_arr, n):
    """Inequality constraints >= 0 for valid packing."""
    cx = vars_arr[:n]
    cy = vars_arr[n:2 * n]
    r = vars_arr[2 * n:]
    
    c_list = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dr = r[i_idx] + r[j_idx]
    c_list.append(np.hypot(dx, dy) - dr)
    
    return np.concatenate(c_list)

def make_hex(rows, r0=0.09):
    """Generates initial positions on a hexagonal lattice."""
    n = 26
    pts = []
    y = r0
    for ri, cnt in enumerate(rows):
        shift = r0 if ri % 2 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
