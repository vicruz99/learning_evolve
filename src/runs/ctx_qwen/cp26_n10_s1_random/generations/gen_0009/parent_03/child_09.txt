from scipy.optimize import minimize, linprog

def get_constraints(vars_arr, n):
    """Computes inequality constraints for SLSQP: must be >= 0."""
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c = []
    # Boundary constraints: circle inside [0,1]x[0,1]
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise non-overlap: squared distance >= squared sum of radii
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def get_objective(vars_arr, n):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars_arr[2*n:])

def constraint_wrapper(vars_arr, n):
    """Wrapper to avoid lambda/closure restrictions."""
    return get_constraints(vars_arr, n)

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        # Max radius limited by distance to nearest boundary
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def simulate_forces(cfg, r_init, steps=1500):
    """Force-directed repulsion simulation to spread circles evenly."""
    n = len(cfg)
    c = cfg.copy()
    r = np.full(n, r_init)
    v = np.zeros_like(c)
    dt = 0.002
    damp = 0.92
    
    for _ in range(steps):
        f = np.zeros_like(c)
        
        # Pairwise repulsion (vectorized)
        diff = c[:, None, :] - c[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        min_d = r[:, None] + r[None, :]
        overlap = np.maximum(0.0, min_d - dists)
        force_mat = (overlap / dists) * 100.0
        fx = np.sum(diff[:, :, 0] * force_mat, axis=1)
        fy = np.sum(diff[:, :, 1] * force_mat, axis=1)
        f[:, 0] += fx
        f[:, 1] += fy
        
        # Wall repulsion
        f[:, 0] += np.clip(r - c[:, 0], 0, None) * 50.0
        f[:, 0] -= np.clip(c[:, 0] + r - 1.0, 0, None) * 50.0
        f[:, 1] += np.clip(r - c[:, 1], 0, None) * 50.0
        f[:, 1] -= np.clip(c[:, 1] + r - 1.0, 0, None) * 50.0
        
        v = damp * v + f * dt
        c += v
        c = np.clip(c, 1e-4, 1.0-1e-4)
        # Gradually expand radii to pack tighter, but cap to prevent explosion
        r = np.minimum(r * 1.00002, 0.2)
        
    return c, r
