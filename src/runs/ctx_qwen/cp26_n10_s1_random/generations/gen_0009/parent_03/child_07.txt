import numpy as np
from scipy.optimize import linprog

def setup_lp(n):
    """Precomputes the constant inequality matrix for the LP solver."""
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((4*n + n_pairs, n))
    idx_i, idx_j = np.triu_indices(n, k=1)
    
    # Boundary constraints: r_i <= limit_i
    for i in range(n):
        A_ub[i, i] = 1.0
        
    # Pairwise constraints: r_i + r_j <= dist_ij
    k = 4*n
    for i in range(n):
        for j in range(i+1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            k += 1
            
    return A_ub, idx_i, idx_j

def solve_lp(centers, A_ub, idx_i, idx_j, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    bounds = []
    b_ub = np.zeros(A_ub.shape[0])
    
    # Boundary limits and LP bounds
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0-x, y, 1.0-y)
        b_ub[i] = lim
        bounds.append((0.0, max(lim, 1e-9)))
        
    # Pairwise distances
    diff_x = centers[:, None, 0] - centers[None, :, 0]
    diff_y = centers[:, None, 1] - centers[None, :, 1]
    dists = np.hypot(diff_x, diff_y)
    b_ub[4*n:] = dists[idx_i, idx_j]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def generate_hex(n, rows, r0, rng):
    """Generates initial positions on a hexagonal lattice with specified row distribution."""
    pts = []
    y = r0
    for idx, cnt in enumerate(rows):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    while len(pts) < n:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
    return np.array(pts[:n])

def hill_climb(centers, A_ub, idx_i, idx_j, n, rng, iterations=2000):
    """Local search on centers maximizing the LP-sum-of-radii objective."""
    best_r, best_s = solve_lp(centers, A_ub, idx_i, idx_j, n)
    best_centers = centers.copy()
    step = 0.02
    
    for _ in range(iterations):
        # Randomly perturb 1 or 2 circles to escape local traps
        if rng.random() < 0.7:
            idx = [rng.integers(n)]
        else:
            idx = [rng.integers(n) for _ in range(2)]
            
        old = centers[idx].copy()
        centers[idx] += rng.uniform(-step, step, (len(idx), 2))
        centers[idx] = np.clip(centers[idx], 1e-5, 1.0-1e-5)
        
        r_new, s_new = solve_lp(centers, A_ub, idx_i, idx_j, n)
        if s_new > best_s:
            best_s = s_new
            best_r = r_new.copy()
            best_centers = centers.copy()
        else:
            centers[idx] = old
            
        step *= 0.9993  # Adaptive decay
        
    return best_centers, best_r, best_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    A_ub, idx_i, idx_j = setup_lp(n)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Diverse hexagonal row patterns summing to 26
    row_pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5],
        [6,6,4,6,4], [5,5,6,5,5], [6,4,6,4,6],
        [5,6,6,5,4], [6,5,4,6,5], [5,5,5,5,6]
    ]
    
    starts = []
    for pat in row_pats:
        if sum(pat) != n: continue
        pts = generate_hex(n, pat, 0.095, rng)
        starts.append(pts)
        for _ in range(3):
            p = pts + rng.uniform(-0.02, 0.02, pts.shape)
            starts.append(np.clip(p, 0.05, 0.95))
            
    # Add random dense configurations
    for _ in range(5):
        starts.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    # Multi-start optimization
    for s in starts:
        c = s.copy()
        # Initial spread to avoid overlapping starts
        for _ in range(500):
            i = rng.integers(n)
            c[i] += rng.uniform(-0.01, 0.01, 2)
            c[i] = np.clip(c[i], 0.05, 0.95)
            
        c_res, r_res, s_res = hill_climb(c, A_ub, idx_i, idx_j, n, rng, iterations=2000)
        if s_res > best_sum:
            best_sum = s_res
            best_centers = c_res
            best_radii = r_res
            
    # Final strict safety scaling to guarantee numerical validity within 1e-12 tolerance
    scale = 1.0
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(best_centers[i,0]-best_centers[j,0], 
                         best_centers[i,1]-best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d/rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
