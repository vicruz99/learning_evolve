Key insights:
1. **Equal vs Unequal Radii**: Known optimal packings for N=26 in a square often converge to nearly equal radii ($r \approx 0.10155$, sum $\approx 2.640$). Optimizing for equal radii first provides a strong symmetric baseline that unequal optimization might miss due to symmetry-breaking traps.
2. **Lattice Orientations & Perturbations**: Hexagonal lattices are near-optimal, but slight rotations and row-count variations (e.g., [6,5,6,5,4] vs [5,6,5,6,4]) yield different local optima. We'll generate diverse structured starts and perturb them.
3. **Exact LP Refinement**: For any fixed center configuration, the maximum sum of radii is a linear program: maximize $\sum r_i$ subject to $r_i + r_j \le d_{ij}$ and $r_i \le \text{dist}(\text{center}_i, \text{boundary})$. This gives the exact local optimum for radii instantly.
4. **Coordinate Ascent + SLSQP**: We'll alternate between SLSQP for joint center/radius optimization and LP refinement. A coordinate-ascent perturbation step moves one circle at a time while re-solving the LP, effectively climbing out of shallow local minima.
5. **Safety Margins**: Strict numerical scaling ensures the validator's `1e-12` tolerance is never violated due to float precision.

The implementation will:
- Generate multiple hexagonal and random initial configurations.
- Run SLSQP for both equal and unequal radii objectives from each start.

import numpy as np
from scipy.optimize import minimize, linprog

def constraints_unequal(v, n):
    """Inequality constraints for unequal radii packing: >= 0"""
    cx = v[:n]
    cy = v[n:2*n]
    r = v[2*n:3*n]
    c = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
    i, j = np.triu_indices(n, 1)
    c.append((cx[i]-cx[j])**2 + (cy[i]-cy[j])**2 - (r[i]+r[j])**2)
    return np.concatenate(c)

def objective_unequal(v, n):
    """Minimize negative sum of radii"""
    return -np.sum(v[2*n:3*n])

def constraints_equal(v, n):
    """Inequality constraints for equal radii packing: >= 0"""
    c = v[:2*n].reshape(n, 2)
    t = v[-1]
    cx, cy = c[:, 0], c[:, 1]
    con = [cx - t, 1.0 - cx - t, cy - t, 1.0 - cy - t]
    i, j = np.triu_indices(n, 1)
    dx = cx[i] - cx[j]
    dy = cy[i] - cy[j]
    con.append(dx**2 + dy**2 - 4.0 * t**2)
    return np.concatenate(con)

def objective_equal(v, n):
    """Minimize negative radius t"""
    return -v[-1]

def hex_init(n, counts, r0=0.09):
    """Generates initial positions on a hexagonal lattice"""
    pts = []
    y = r0
    for idx, cnt in enumerate(counts):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers"""
    n = centers.shape[0]
    i, j = np.triu_indices(n, 1)
    m = len(i)
    
    A = np.zeros((m, n))
    A[np.arange(m), i] = 1.0
    A[np.arange(m), j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b = dists[i, j]
    
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(l, 1e-9)) for l in lims]
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)
    
    # Diverse row distributions summing to 26
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,6,6,4], [6,6,5,5,4], 
                [5,5,6,5,5], [6,4,6,4,6], [5,7,5,4,5], [7,5,5,5,4]]
    
    configs = []
    for pat in patterns:
        if sum(pat) < n: continue
        c = hex_init(n, pat, r0=0.095)
        configs.append(c)
        # Perturb to break symmetry and explore neighborhood
        for _ in range(5):
            p = np.clip(c + rng.uniform(-0.025, 0.025, (n, 2)), 0.05, 0.95)
            configs.append(p)
            
    # Add fully random dense starts
    for _ in range(10):
        configs.append(rng.uniform(0.12, 0.88, (n, 2)))
        
    bounds_uneq = [(0.0, 1.0)]*(2*n) + [(1e-6, 0.5)]*n
    bounds_eq = [(0.0, 1.0)]*(2*n) + [(0.06, 0.12)]
    
    # Phase 1: Multi-start SLSQP for both equal and unequal radii
    for cfg in configs:
        # --- Unequal Optimization ---
        x0 = np.zeros(3*n)
        x0[:n] = cfg[:, 0]
        x0[n:2*n] = cfg[:, 1]
        r_est = np.zeros(n)
        for i in range(n):
            dw = min(cfg[i,0], 1.0-cfg[i,0], cfg[i,1], 1.0-cfg[i,1])
            dn = np.min(np.sqrt(np.sum((cfg - cfg[i])**2, axis=1)))
            r_est[i] = min(dw, dn/2.0) * 0.9
        x0[2*n:] = r_est
        
        try:
            res = minimize(objective_unequal, x0, args=(n,), method='SLSQP', bounds=bounds_uneq,
                           constraints={'type': 'ineq', 'fun': constraints_unequal, 'args': (n,)},
                           options={'maxiter': 6000, 'ftol': 1e-13})
            if np.isfinite(res.fun):
                c_val = constraints_unequal(res.x, n)
                if np.min(c_val) >= -1e-5:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_centers = res.x[:2*n].reshape(n, 2).copy()
                        best_radii = res.x[2*n:].copy()
        except Exception:
            pass
            
        # --- Equal Optimization ---
        x0_eq = np.concatenate([cfg.flatten(), [0.092]])
        try:
            res_eq = minimize(objective_equal, x0_eq, args=(n,), method='SLSQP', bounds=bounds_eq,
                              constraints={'type': 'ineq', 'fun': constraints_equal, 'args': (n,)},
                              options={'maxiter': 6000, 'ftol': 1e-13})
            if np.isfinite(res_eq.fun):
                c_val_eq = constraints_equal(res_eq.x, n)
                if np.min(c_val_eq) >= -1e-5:
                    s_eq = res_eq.x[-1] * n
                    if s_eq > best_sum:
                        best_sum = s_eq
                        best_centers = res_eq.x[:2*n].reshape(n, 2).copy()
                        best_radii = np.full(n, res_eq.x[-1])
        except Exception:
            pass

    # Fallback if optimization unexpectedly fails
    if best_centers is None:
        best_centers = configs[0]
        best_radii = np.full(n, 0.085)
        best_sum = np.sum(best_radii)
        
    # Phase 2: Exact LP refinement for fixed centers
    lp_r, lp_s = solve_lp_radii(best_centers)
    if lp_r is not None and lp_s > best_sum:
        best_radii = lp_r
        best_sum = lp_s
        
    # Phase 3: Coordinate ascent perturbation search on centers + LP
    # Moves one circle at a time, re-solves exact radii, accepts if sum increases
    for step in range(300):
        idx = rng.integers(n)
        old_c = best_centers[idx].copy()
        # Adaptive step size
        step_size = 0.008 * (0.95 ** (step / 50.0))
        best_centers[idx] += rng.uniform(-step_size, step_size, 2)
        best_centers[idx] = np.clip(best_centers[idx], 0.002, 0.998)
        
        r_try, s_try = solve_lp_radii(best_centers)
        if r_try is not None and s_try > best_sum:
            best_sum = s_try
            best_radii = r_try.copy()
        else:
            best_centers[idx] = old_c
            
    # Phase 4: Final strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                         best_centers[i, 1] - best_centers[j, 1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
