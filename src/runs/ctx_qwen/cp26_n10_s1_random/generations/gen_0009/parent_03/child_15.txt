import numpy as np
from scipy.optimize import minimize, linprog

def compute_lp_objective(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Boundary limits for each circle: distance to nearest wall
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 1e-10)

    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # LP formulation: max sum(r) s.t. r_i + r_j <= dist_ij, 0 <= r_i <= lims_i
    c_obj = -np.ones(n)
    
    # Precompute upper triangle indices for pairwise constraints
    i_idx, j_idx = np.triu_indices(n, k=1)
    m_pairs = len(i_idx)
    
    A_ub = np.zeros((m_pairs, n))
    A_ub[np.arange(m_pairs), i_idx] = 1.0
    A_ub[np.arange(m_pairs), j_idx] = 1.0
    b_ub = dists[i_idx, j_idx]
    
    bounds = [(0.0, lim) for lim in lims]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.full(n, 1e-9)

def objective_negative_sum(x_flat, n):
    """Objective function for optimizer: minimize negative sum of radii."""
    centers = x_flat.reshape(-1, 2)
    s, _ = compute_lp_objective(centers, n)
    return -s

def generate_hex_starts(n, rng):
    """Generates diverse initial configurations based on hexagonal lattices."""
    starts = []
    row_pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], 
        [6,6,4,6,4], [5,5,6,5,5], [6,4,6,4,6],
        [5,6,6,5,4], [6,5,4,6,5], [5,5,5,5,6]
    ]
    for pat in row_pats:
        if sum(pat) != n: 
            continue
        pts = []
        r0 = 0.095
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: 
                    break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        pts = np.array(pts[:n])
        starts.append(pts)
        
        # Rotated versions to break symmetry and explore different alignments
        for ang in [15, 30, 45]:
            c = np.radians(ang)
            rot = np.array([[np.cos(c), -np.sin(c)], [np.sin(c), np.cos(c)]])
            p = pts @ rot.T
            mn = p.min(axis=0)
            mx = p.max(axis=0)
            p = (p - mn) / (mx - mn + 1e-9) * 0.7 + 0.15
            starts.append(p)

    # Random dense starts
    for _ in range(10):
        starts.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    return starts

def coordinate_descent(centers, n, rng, max_iter=3000, init_step=0.02):
    """Refines centers using stochastic coordinate descent with LP evaluation."""
    best_sum, best_r = compute_lp_objective(centers, n)
    best_c = centers.copy()
    current_c = centers.copy()
    
    for step in range(max_iter):
        i = rng.integers(n)
        old_c = current_c[i].copy()
        
        # Adaptive step size that decays over time for fine-tuning
        cur_step = init_step * (0.98 ** (step / 25.0))
        
        # Perturb position
        current_c[i] = old_c + rng.uniform(-cur_step, cur_step, 2)
        current_c[i] = np.clip(current_c[i], 1e-4, 1.0 - 1e-4)
        
        s_new, r_new = compute_lp_objective(current_c, n)
        if s_new > best_sum + 1e-8:
            best_sum = s_new
            best_r = r_new.copy()
            best_c = current_c.copy()
        else:
            current_c[i] = old_c
            
    return best_c, best_r, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = np.random.default_rng(42)
    
    starts = generate_hex_starts(n, rng)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Powell optimization from diverse starts
    for s in starts:
        x0 = s.flatten()
        try:
            res = minimize(objective_negative_sum, x0, args=(n,), method='Powell', 
                          bounds=[(0.01, 0.99)]*(2*n),
                          options={'maxiter': 400, 'ftol': 1e-12, 'disp': False})
            if np.isfinite(res.fun):
                c_opt = res.x.reshape(n, 2)
                s_opt, r_opt = compute_lp_objective(c_opt, n)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Coordinate descent refinement on best found
    if best_centers is not None:
        c_cd, r_cd, s_cd = coordinate_descent(best_centers, n, rng, max_iter=4000, init_step=0.015)
        if s_cd > best_sum:
            best_sum = s_cd
            best_centers = c_cd
            best_radii = r_cd
            
        # Local restarts to escape shallow local minima
        for _ in range(25):
            pert = best_centers + rng.uniform(-0.025, 0.025, (n, 2))
            pert = np.clip(pert, 0.02, 0.98)
            c_cd, r_cd, s_cd = coordinate_descent(pert, n, rng, max_iter=2000, init_step=0.01)
            if s_cd > best_sum:
                best_sum = s_cd
                best_centers = c_cd
                best_radii = r_cd
                
    # Fallback configuration
    if best_centers is None:
        best_centers = rng.uniform(0.15, 0.85, (n, 2))
        best_radii, best_sum = compute_lp_objective(best_centers, n)
        
    # Phase 3: Final strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i,0] - best_centers[j,0], 
                         best_centers[i,1] - best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.9999995
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
