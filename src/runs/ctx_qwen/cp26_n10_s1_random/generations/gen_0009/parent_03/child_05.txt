import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        lim = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def constraints(vars_arr, n):
    """Inequality constraints for SLSQP: must be >= 0."""
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    r = vars_arr[2*n:]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(c)

def objective(vars_arr, n):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars_arr[2*n:])

def force_layout(n, r0=0.04, steps=4000, seed=0):
    """Force-directed relaxation that grows circles until they jam tightly."""
    rng = np.random.default_rng(seed)
    centers = rng.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, r0)
    
    dt = 0.015
    k_rep = 50.0
    k_wall = 40.0
    growth = 1.00006
    
    for _ in range(steps):
        forces = np.zeros_like(centers)
        
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, 1e9)
        req = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlap = np.maximum(0.0, req - dists)
        rep = k_rep * overlap / (dists + 1e-9)
        fx = np.sum(diff[:, :, 0] * rep, axis=1)
        fy = np.sum(diff[:, :, 1] * rep, axis=1)
        forces[:, 0] += fx
        forces[:, 1] += fy
        
        forces[:, 0] += k_wall * np.clip(radii - centers[:, 0], 0, None)
        forces[:, 0] -= k_wall * np.clip(centers[:, 0] - (1.0 - radii), 0, None)
        forces[:, 1] += k_wall * np.clip(radii - centers[:, 1], 0, None)
        forces[:, 1] -= k_wall * np.clip(centers[:, 1] - (1.0 - radii), 0, None)
        
        centers += forces * dt
        centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
        radii *= growth
        
    return centers, radii

    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
