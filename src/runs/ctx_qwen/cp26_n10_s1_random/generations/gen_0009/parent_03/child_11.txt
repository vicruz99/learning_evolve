To maximize the sum of radii for 26 circles in a unit square, we leverage two key insights:
1. **Equal-Radius Optimization as a Strong Starting Point**: Dense circle packings in squares often converge to configurations where circles are nearly equal in size. Optimizing for a single slack variable $t$ (representing a uniform radius) simplifies the landscape and helps SLSQP find high-clearance center arrangements efficiently. Known theoretical bounds for $n=26$ suggest an optimal equal radius $r \approx 0.1014$, yielding a sum $\approx 2.636$.
2. **LP Refinement for Unequal Radii**: Once optimal centers are found, allowing radii to vary slightly can squeeze out additional space. We fix the optimized centers and solve a Linear Program (LP) to maximize $\sum r_i$ subject to $r_i + r_j \le \text{dist}(i,j)$ and boundary constraints. This is convex and solves instantly.
3. **Iterative Local Search**: We perform a coordinate-wise perturbation search on the centers, re-solving the LP each time. This escapes shallow local minima without the overhead of full non-linear re-optimization.
4. **Robust Initialization**: We generate multiple hexagonal lattice patterns with varying row counts (summing to 26), normalize them to fit comfortably in the square, and perturb them. This ensures diverse basins of attraction for the optimizer.

The pipeline: Generate starts $\rightarrow$ SLSQP equal-radius optimization $\rightarrow$ LP refinement $\rightarrow$ Local center search $\rightarrow$ Strict numerical scaling.

import numpy as np
from scipy.optimize import minimize, linprog

def constraints_equal(vars_arr, n):
    """Inequality constraints for equal-radius optimization: must be >= 0."""
    x = vars_arr[:n]
    y = vars_arr[n:2*n]
    t = vars_arr[2*n]
    c = []
    # Boundary constraints
    c.append(x - t)
    c.append(1.0 - x - t)
    c.append(y - t)
    c.append(1.0 - y - t)
    
    # Pairwise non-overlap: dist^2 >= 4t^2
    idx_i, idx_j = np.triu_indices(n, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    c.append(dx**2 + dy**2 - 4.0*t**2)
    return np.concatenate(c)

def obj_equal(vars_arr, n):
    """Objective: maximize t => minimize -t."""
    return -vars_arr[2*n]

def solve_radii_lp(centers, n):
    """Solves LP to maximize sum of radii for fixed centers."""
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds.append((0.0, max(lim, 1e-9)))
        
    idx_i, idx_j = np.triu_indices(n, k=1)
    m = len(idx_i)
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idx_i] = 1.0
    A_ub[np.arange(m), idx_j] = 1.0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def generate_starts(n, rng):
    """Generates diverse hexagonal lattice initial configurations."""
    starts = []
    row_pats = [
        [6,5,6,5,4], [5,6,5,6,4], [6,6,4,6,4], [5,5,6,5,5],
        [4,6,6,5,5], [6,4,6,5,5], [5,6,4,6,5], [6,5,5,6,4]
    ]
    for pat in row_pats:
        if sum(pat) != n: continue
        pts = []
        r0 = 1.0
        y = r0
        for idx, cnt in enumerate(pat):
            shift = r0 if idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(pts) >= n: break
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        base = np.array(pts[:n])
        # Normalize and scale to fit safely inside [0,1]
        mn = base.min(axis=0)
        mx = base.max(axis=0)
        span = mx - mn + 1e-9
        base = (base - mn) / span * 0.8 + 0.1
        starts.append(base)
        for _ in range(5):
            p = base + rng.uniform(-0.008, 0.008, base.shape)
            starts.append(np.clip(p, 0.05, 0.95))
    return starts

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    starts = generate_starts(n, rng)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Bounds: x,y in [0,1], uniform radius t in [0.01, 0.2]
    bounds_eq = [(0.0, 1.0)]*(2*n) + [(0.01, 0.2)]
    
    # Phase 1: Equal-radius optimization from multiple starts
    for cfg in starts:
        # Compute a strictly feasible initial t
        min_wall = np.min(np.minimum(cfg[:, 0], 1.0 - cfg[:, 0]))
        diff = cfg[:, np.newaxis, :] - cfg[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        min_pair = np.min(dists) / 2.0
        t0 = min(min_wall, min_pair) * 0.85
        
        x0 = np.concatenate([cfg.flatten(), [t0]])
        
        try:
            res = minimize(obj_equal, x0, args=(n,), method='SLSQP',
                           bounds=bounds_eq,
                           constraints={'type': 'ineq', 'fun': constraints_equal, 'args': (n,)},
                           options={'maxiter': 8000, 'ftol': 1e-12})
            
            if np.isfinite(res.fun):
                centers_cand = res.x[:2*n].reshape(n, 2)
                # Phase 1.5: LP refinement allows radii to vary, often increasing sum
                r_lp, s_lp = solve_radii_lp(centers_cand, n)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = centers_cand.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass

    # Phase 2: Stochastic local search on centers evaluated via exact LP
    if best_centers is not None:
        for _ in range(800):
            i = rng.integers(n)
            old = best_centers[i].copy()
            best_centers[i] += rng.uniform(-0.005, 0.005, 2)
            best_centers[i] = np.clip(best_centers[i], 1e-4, 1.0 - 1e-4)
            
            r_try, s_try = solve_radii_lp(best_centers, n)
            if s_try > best_sum:
                best_sum = s_try
                best_radii = r_try.copy()
            else:
                best_centers[i] = old
                
    # Phase 3: Final strict safety scaling to guarantee numerical validity
    scale = 1.0
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if r > 1e-12:
            scale = min(scale, x/r, (1.0-x)/r, y/r, (1.0-y)/r)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(best_centers[i,0] - best_centers[j,0], 
                         best_centers[i,1] - best_centers[j,1])
            rs = best_radii[i] + best_radii[j]
            if rs > 1e-12:
                scale = min(scale, d / rs)
                
    best_radii *= scale * 0.999999
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
