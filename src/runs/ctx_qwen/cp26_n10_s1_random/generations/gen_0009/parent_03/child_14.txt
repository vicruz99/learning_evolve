import numpy as np
from scipy.optimize import linprog

def solve_lp_and_grad(centers, n):
    """
    Solves LP to maximize sum of radii for fixed centers and returns 
    radii, objective value, and gradient of the objective w.r.t centers 
    using LP dual variables (shadow prices).
    """
    x = centers[:, 0]
    y = centers[:, 1]
    
    m_bound = 4 * n
    m_pair = n * (n - 1) // 2
    A_ub = np.zeros((m_bound + m_pair, n))
    b_ub = np.zeros(m_bound + m_pair)
    
    idx = 0
    for i in range(n):
        # r_i <= x_i
        A_ub[idx, i] = 1.0; b_ub[idx] = x[i]; idx += 1
        # r_i <= 1 - x_i
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x[i]; idx += 1
        # r_i <= y_i
        A_ub[idx, i] = 1.0; b_ub[idx] = y[i]; idx += 1
        # r_i <= 1 - y_i
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y[i]; idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(x[i] - x[j], y[i] - y[j])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds = [(0, None)] * n
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return np.full(n, 1e-6), 0.0, np.zeros(n), np.zeros(n)
    except Exception:
        return np.full(n, 1e-6), 0.0, np.zeros(n), np.zeros(n)
        
    r = res.x
    s = -res.fun
    duals = res.ineqlin.marginals
    
    # Gradient from boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    gx = duals[:n] - duals[n:2*n]
    gy = duals[2*n:3*n] - duals[3*n:4*n]
    
    # Gradient from pairwise constraints: r_i + r_j <= d_ij
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[m_bound + idx]
            d = b_ub[m_bound + idx]
            if d > 1e-8:
                fx = lam * (x[i] - x[j]) / d
                fy = lam * (y[i] - y[j]) / d
                gx[i] += fx
                gy[i] += fy
                gx[j] -= fx
                gy[j] -= fy
            idx += 1
            
    return r, s, gx, gy

def hex_init(pat, r0=0.095):
    """Generates initial positions on a staggered hexagonal lattice."""
    n = 26
    pts = []
    y = r0
    for idx, cnt in enumerate(pat):
        shift = r0 if idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(pts) >= n: 
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    while len(pts) < n:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])

def run_packing():
    n = 26
    rng = np.random.default_rng(42)

    pats = [

        [5,5,6,5,5], [4,6,6,6,4], [6,5,5,6,4], [5,6,6,4,5]

    best_sum = -1.0
    best_c = None

                C = C_new
            else:
                # Backtracking line search

                    C_new = C + (eta * (0.5 ** backtrack)) * grad / norm_g
                    C_new = np.clip(C_new, 1e-4, 1.0 - 1e-4)
                    _, s_new, _, _ = solve_lp_and_grad(C_new, n)
                    if s_new > s:
                        C = C_new
                        break
        
        _, s_final, _, _ = solve_lp_and_grad(C, n)

            best_sum = s_final
            best_c = C.copy()
            
    if best_c is None:
        best_c = hex_init([6,5,6,5,4])

    best_r, best_sum, _, _ = solve_lp_and_grad(best_c, n)

    scale = 1.0

        scale = min(scale, best_c[i,0]/best_r[i], (1.0-best_c[i,0])/best_r[i],

    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
