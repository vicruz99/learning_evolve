import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def get_centers_radii(params):
    """Reconstruct centers and radii from (r, u, v) parameters."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraint_func(params):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundary handled by parameterization."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def greedy_init(rng, seed):
    """Generates a packing by sequentially placing circles at the location maximizing radius."""
    rng_g = np.random.RandomState(seed)
    centers = []
    radii = []
    for _ in range(N):
        best_r = -1.0
        best_pos = None
        for _ in range(50):
            pos = rng_g.uniform(0.02, 0.98, 2)
            # Max feasible radius at this position given current circles and boundaries
            r = min(pos[0], 1.0 - pos[0], pos[1], 1.0 - pos[1])
            for c, rr in zip(centers, radii):
                d = np.linalg.norm(pos - c)
                r = min(r, d - rr)
            if r > best_r:
                best_r = r
                best_pos = pos
        if best_r > 1e-4:
            centers.append(best_pos)
            radii.append(best_r)
        else:
            # Fallback if greedy fails
            centers.append(rng_g.uniform(0.1, 0.9, 2))
            radii.append(0.01)
    return np.array(centers), np.array(radii)

def force_init(rng, seed):
    """Spread points using force-directed layout."""
    rng_f = np.random.RandomState(seed)
    pts = rng_f.uniform(0.05, 0.95, (N, 2))
    r_est = 0.06
    for step in range(200):
        forces = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces += np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 30.0 * np.maximum(0, r_est - pts[:, d])
            forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - (1.0 - r_est))
        pts += 0.006 * forces
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def hex_init(rng, pattern, rot, scale, seed):
    """Generates a hexagonal lattice initialization."""
    rng_h = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(pattern):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng_h.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [5,5,6,5,5], [7,5,5,5,4], [4,5,5,5,7], [5,7,5,5,4]
    ]
    # 1. Hexagonal lattices with varied parameters
    for pat in patterns:
        for s in range(4):
            rot = rng.uniform(-0.2, 0.2)
            sc = rng.uniform(0.85, 1.15)
            inits.append(hex_init(rng, pat, rot, sc, s))
            
    # 2. Force-directed layouts
    for s in range(15):
        inits.append(force_init(rng, s))
        
    # 3. Greedy sequential placements
    for s in range(10):
        c, r = greedy_init(rng, s)
        inits.append(c)
        
    # 4. Perturbed grids
    for s in range(10):
        rng_g = np.random.RandomState(s)
        grid = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        grid = np.vstack([grid, [0.5, 0.5]])
        grid += rng_g.uniform(-0.03, 0.03, (N, 2))
        inits.append(np.clip(grid, 0.02, 0.98))
        
    return inits

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    inits = generate_inits(rng)
    
    # Phase 1: Broad SLSQP search from diverse initializations
    for pts in inits:
        r_init, _ = solve_lp_radii(pts)
        r_init = np.clip(r_init * 0.98, 1e-6, 0.49)
        vars0 = get_params(pts, r_init)
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                if np.min(constraint_func(res.x)) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    if best_vars is not None:
        best_c, best_r = get_centers_radii(best_vars)
        best_r, best_sum = solve_lp_radii(best_c)
        
        # Phase 2: LP-driven Simulated Annealing on centers
        curr_c = best_c.copy()
        curr_r, curr_sum = solve_lp_radii(curr_c)
        temp = 0.04
        step = 0.03
        
        for it in range(8000):
            op = rng.choice(['move', 'swap', 'jump'], p=[0.7, 0.2, 0.1])
            
            if op == 'move':
                idx = rng.integers(N)
                old_pos = curr_c[idx].copy()
                curr_c[idx] += rng.normal(0, step, 2)
                curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
            elif op == 'swap':
                i, j = rng.choice(N, 2, replace=False)
                curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
                old_pos = None
            else: # jump
                idx = rng.integers(N)
                old_pos = curr_c[idx].copy()
                curr_c[idx] = rng.uniform(0.05, 0.95, 2)
                
            nr, ns = solve_lp_radii(curr_c)
            delta = ns - curr_sum
            
            if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(delta / temp)):
                curr_sum = ns
                if ns > best_sum:
                    best_sum = ns
                    best_c = curr_c.copy()
                    best_r = nr.copy()
            else:
                if old_pos is not None and op != 'swap':
                    curr_c[idx] = old_pos
                
            temp *= 0.9993
            step = max(0.001, step * 0.9994)
            
        # Update best if SA found better
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_c = curr_c.copy()
            best_r, best_sum = solve_lp_radii(best_c)

        # Phase 3: SLSQP refinement on SA best
        r_init = np.clip(best_r * 0.99, 1e-6, 0.49)
        vars0 = get_params(best_c, r_init)
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds,

            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x.copy()
        except Exception:
            pass
            
        # Phase 4: Perturbation & SLSQP to escape local minima
        for k in range(25):
            r_p = best_r * 0.995
            vars0 = get_params(best_c, r_p)
            vars0[1::3] += rng.uniform(-0.025, 0.025, N)
            vars0[2::3] += rng.uniform(-0.025, 0.025, N)
            vars0[0::3] += rng.uniform(-0.002, 0.002, N)
            vars0[0::3] = np.clip(vars0[0::3], 1e-6, 0.49)
            vars0[1::3] = np.clip(vars0[1::3], 0.0, 1.0)
            vars0[2::3] = np.clip(vars0[2::3], 0.0, 1.0)
            
            try:
                res = minimize(objective, vars0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
            except Exception:
                continue
                
        if best_vars is not None:
            best_c, best_r = get_centers_radii(best_vars)
            best_r, best_sum = solve_lp_radii(best_c)
            
        # Phase 5: High-precision final polish
        r_init = np.clip(best_r * 0.998, 1e-6, 0.49)
        vars0 = get_params(best_c, r_init)
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-9:
                best_vars = res.x
                best_sum = -res.fun
        except Exception:
            pass
            
    # Fallback configuration
    if best_vars is None:
        pts = inits[0]
        r, _ = solve_lp_radii(pts)
        best_vars = get_params(pts, r)
        best_sum = np.sum(best_vars[0::3])
        
    best_c, best_r = get_centers_radii(best_vars)
    # Final LP to ensure radii are optimal for the exact final centers
    best_r, best_sum = solve_lp_radii(best_c)
    
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
