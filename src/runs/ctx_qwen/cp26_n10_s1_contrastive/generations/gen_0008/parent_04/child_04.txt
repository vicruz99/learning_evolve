3. **SLSQP Refinement**: Use a boundary-safe parameterization `(r, u, v)` where `x = r + u*(1-2r)` and `y = r + v*(1-2r)`. This automatically satisfies boundary constraints, leaving only pairwise non-overlap constraints for the optimizer. Run SLSQP from each initialization.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = max(b, 0.0) # Ensure non-negative bounds
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def get_centers_radii(params):
    """Reconstruct centers and radii from (r, u, v) parameters."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraint_func(params):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundary handled by parameterization."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [5,5,6,5,5], [7,5,5,5,4], [4,5,5,5,7], [5,7,5,5,4],
        [6,5,5,5,5], [5,5,5,6,5], [7,6,5,4,4]
    ]
    for pat in patterns:
        for _ in range(3):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            pts = np.array(pts[:N])
            pts -= 0.5
            scale = rng.uniform(0.85, 1.15)
            pts *= scale
            pts += 0.5
            rot = rng.uniform(-0.25, 0.25)
            c, s = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[c, -s], [s, c]])
            pts += rng.uniform(-0.01, 0.01, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    # 2. Force-directed layouts
    for seed in range(12):
        rng_fd = np.random.RandomState(seed)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        r_curr = np.full(N, 0.05)
        for step in range(250):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 20.0 * np.maximum(0, r_curr - pts[:, d])
                forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            step_size = 0.008 * (0.995**step)
            pts += step_size * forces
            pts = np.clip(pts, 0.02, 0.98)
            for i in range(N):
                d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                dists_others = np.linalg.norm(pts[i] - pts, axis=1)
                dists_others[i] = np.inf
                d_pair = np.min(dists_others)
                r_curr[i] = 0.85 * min(d_wall, d_pair/2.0)
        inits.append(pts)
        
    return inits

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    inits = generate_inits(rng)
    
    # Phase 1: Broad SLSQP search from diverse initializations
    for pts in inits:
        r_init, _ = solve_lp(pts)
        r_init = np.clip(r_init * 0.98, 1e-6, 0.49)
        vars0 = get_params(pts, r_init)
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                if np.min(constraint_func(res.x)) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    if best_vars is not None:
        best_c, best_r = get_centers_radii(best_vars)
        best_r, best_sum = solve_lp(best_c)
        
        # Phase 2: LP-driven Local Search on centers
        curr_c = best_c.copy()
        curr_r, curr_sum = solve_lp(curr_c)
        step = 0.025
        
        for it in range(3000):
            idx = rng.integers(N)
            old_pos = curr_c[idx].copy()
            
            # Try moving in a random direction
            direction = rng.standard_normal(2)
            direction /= np.linalg.norm(direction)
            curr_c[idx] = np.clip(old_pos + direction * step, 0.005, 0.995)
            
            nr, ns = solve_lp(curr_c)
            delta = ns - curr_sum
            
            if delta > 0:
                curr_sum = ns
                if ns > best_sum:
                    best_sum = ns
                    best_c = curr_c.copy()
                    best_r = nr.copy()
                step = np.maximum(0.001, step * 0.99)
            else:
                curr_c[idx] = old_pos
                step = np.minimum(0.05, step * 1.01)
                
        # Update best if local search found better
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_c = curr_c.copy()
            best_r, best_sum = solve_lp(best_c)

        # Phase 3: Rotation Escapes
        for _ in range(10):
            rot = rng.uniform(-0.15, 0.15)
            c_val, s_val = np.cos(rot), np.sin(rot)
            mat = np.array([[c_val, -s_val], [s_val, c_val]])
            c_rot = (best_c - 0.5) @ mat.T + 0.5
            c_rot = np.clip(c_rot, 0.01, 0.99)
            
            r_rot, s_rot = solve_lp(c_rot)
            if s_rot > best_sum:
                best_sum = s_rot
                best_c = c_rot.copy()
                best_r = r_rot.copy()

            try:

                    if curr_sum > best_sum:

                        best_vars = res.x.copy()
            except Exception:
                pass
                
        if best_vars is not None:

            best_r, best_sum = solve_lp(best_c)

        for _ in range(12):
            r_p = best_r * 0.995
            vars0 = get_params(best_c, r_p)
            vars0[1::3] += rng.uniform(-0.02, 0.02, N)
            vars0[2::3] += rng.uniform(-0.02, 0.02, N)

            try:

                        best_sum = curr_sum
                        best_vars = res.x.copy()

                continue
                
        if best_vars is not None:
            best_c, best_r = get_centers_radii(best_vars)
            best_r, best_sum = solve_lp(best_c)

        pts = inits[0]
        r, _ = solve_lp(pts)
        best_vars = get_params(pts, r)
        best_sum = np.sum(best_vars[0::3])

    # Final LP to ensure radii are optimal for the exact final centers
    best_r, best_sum = solve_lp(best_c)
    
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
