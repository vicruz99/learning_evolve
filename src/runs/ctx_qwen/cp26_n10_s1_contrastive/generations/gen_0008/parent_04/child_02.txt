import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def get_centers_radii(params):
    """Reconstruct centers and radii from (r, u, v) parameters."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraint_func(params):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundary handled by parameterization."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def force_init(rng, seed):
    """Generates a well-spread initial configuration using repulsion forces."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.rand(N, 2) * 0.8 + 0.1
    r_curr = np.full(N, 0.06)
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 30.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
        step_size = 0.01 * (0.993**step)
        pts += step_size * forces
        pts = np.clip(pts, 0.01, 0.99)
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_others[i] = np.inf
            d_pair = np.min(dists_others)
            r_curr[i] = 0.85 * min(d_wall, d_pair/2.0)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraint_func}
    
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    # Phase 1: Force-directed inits + Simulated Annealing on centers
    for seed in range(15):
        pts = force_init(rng, seed)
        curr_c = pts.copy()
        curr_r, curr_sum = solve_lp(curr_c)
        
        temp = 0.04
        step = 0.035
        
        for it in range(8000):

            old_pos = curr_c[idx].copy()

            delta = ns - curr_sum
            
            if delta > 0 or (temp > 1e-7 and rng.random() < np.exp(delta / temp)):

                if ns > best_sum:
                    best_sum = ns
                    best_c = curr_c.copy()
                    best_r = nr.copy()
            else:
                curr_c[idx] = old_pos
                
            temp *= 0.9993
            step = max(0.0015, step * 0.9995)
            
    if best_c is None:
        best_c = force_init(rng, 0)
        best_r, best_sum = solve_lp(best_c)
        
    # Phase 2: Rotation Optimization to align packing with square boundaries
    best_c_centered = best_c - 0.5
    best_rot_sum = best_sum
    best_rot_c = best_c.copy()
    
    angles = np.linspace(0, np.pi/2, 101)
    for ang in angles:
        c, s = np.cos(ang), np.sin(ang)
        rot_mat = np.array([[c, -s], [s, c]])
        c_rot = (best_c_centered @ rot_mat.T) + 0.5
        c_rot = np.clip(c_rot, 0.01, 0.99)

        if s_rot > best_rot_sum:
            best_rot_sum = s_rot
            best_rot_c = c_rot.copy()

        best_c = best_rot_c

    r_init = np.clip(best_r * 0.995, 1e-6, 0.49)
    vars0 = get_params(best_c, r_init)
    
    try:
        res = minimize(objective, vars0, method='SLSQP', bounds=bounds_slqp,
                       constraints=cons_slqp, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(constraint_func(res.x)) >= -1e-8:

            if curr_sum > best_sum:
                best_sum = curr_sum
                best_c, best_r = get_centers_radii(res.x)
                best_r, best_sum = solve_lp(best_c)
    except Exception:
        pass

        vars0[0::3] = np.clip(vars0[0::3], 1e-6, 0.49)
        vars0[1::3] = np.clip(vars0[1::3], 0.0, 1.0)
        vars0[2::3] = np.clip(vars0[2::3], 0.0, 1.0)
        
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_c, best_r = get_centers_radii(res.x)
                    best_r, best_sum = solve_lp(best_c)
        except Exception:
            continue
            
    # Final LP to ensure radii are strictly optimal for the exact final centers
    best_r, best_sum = solve_lp(best_c)
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
