import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-4

def objective_powell(c_flat):
    """Objective for derivative-free optimization: minimize negative sum of radii."""
    centers = c_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, s = solve_lp_radii(centers)
    return -s

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(p):
    """Reconstruct physical centers and radii from parameters."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraints_slsqp(p):
    """Inequality constraints for SLSQP: pairwise non-overlap."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def objective_slsqp(p):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(p[:N])

def hex_init(rng, row_counts, rot=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(400):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.08 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.92)
        pts += 0.004 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints_slsqp}
    bounds_powell = [(0.05, 0.95)] * (2 * N)
    
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    best_params = None
    
    # Phase 1: Diverse starts + Powell optimization on centers
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7],
        [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5]
    ]
    for p in patterns:
        for _ in range(4):
            inits.append(hex_init(rng, p, rot=rng.uniform(-0.4, 0.4), scale=rng.uniform(0.88, 1.12)))
    for _ in range(8):
        inits.append(force_init(rng))
    for _ in range(5):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    for c0 in inits:
        try:
            res = minimize(objective_powell, c0.flatten(), method='Powell', bounds=bounds_powell,
                           options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False})
            curr_c = res.x.reshape(N, 2)
            curr_r, curr_s = solve_lp_radii(curr_c)
            if curr_s > best_sum:
                best_sum = curr_s
                best_centers = curr_c.copy()
                best_radii = curr_r.copy()
        except Exception:
            continue
            
    # Fallback if Powell failed completely
    if best_centers is None:
        c_f = hex_init(rng, [6,5,6,5,4])
        r_f, s_f = solve_lp_radii(c_f)
        best_centers, best_radii, best_sum = c_f, r_f, s_f

    # Phase 2: Simulated Annealing on Centers
    curr_c = best_centers.copy()
    curr_s = best_sum
    temp = 0.04
    step = 0.04
    
    for it in range(8000):
        move_type = rng.choice(['shift', 'swap', 'rotate', 'scale'])
        old_c = curr_c.copy()
        
        if move_type == 'shift':
            idx = rng.integers(N)
            d = rng.standard_normal(2)
            d /= np.linalg.norm(d)
            curr_c[idx] = np.clip(curr_c[idx] + step * d, 0.02, 0.98)
        elif move_type == 'swap':
            i, j = rng.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        elif move_type == 'rotate':
            angle = rng.uniform(-0.05, 0.05)
            c_mat, s_mat = np.cos(angle), np.sin(angle)
            rot_mat = np.array([[c_mat, -s_mat], [s_mat, c_mat]])
            curr_c = (curr_c - 0.5) @ rot_mat.T + 0.5
            curr_c = np.clip(curr_c, 0.02, 0.98)
        else: # scale
            factor = 1.0 + rng.uniform(-0.02, 0.02)
            curr_c = 0.5 + (curr_c - 0.5) * factor
            curr_c = np.clip(curr_c, 0.02, 0.98)
            
        _, new_s = solve_lp_radii(curr_c)
        delta = new_s - curr_s
        
        if delta > 0 or (temp > 1e-7 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_s = new_s
            if new_s > best_sum:
                best_sum = new_s
                best_centers = curr_c.copy()
                r_lp, _ = solve_lp_radii(curr_c)
                best_radii = r_lp.copy()
        else:
            curr_c = old_c
            
        temp *= 0.9993
        step = max(0.002, step * 0.9995)
        
    # Phase 3: SLSQP Polishing with (r, u, v) parameterization
    r_init = np.maximum(best_radii * 0.992, 1e-6)
    x0 = to_params(best_centers, r_init)
    
    try:
        res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(constraints_slsqp(res.x)) >= -1e-8:
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_params = res.x.copy()
    except Exception:
        pass
        
    # Phase 4: Perturbation & SLSQP restarts to escape local minima
    for _ in range(25):
        if best_params is None:
            best_params = to_params(best_centers, best_radii * 0.995)
        xp = best_params.copy()
        xp[:N] += rng.uniform(-0.0015, 0.0015, N)
        xp[N:3*N] += rng.uniform(-0.012, 0.012, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(objective_slsqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints_slsqp(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            continue
            
    # Phase 5: High-precision final polish
    if best_params is not None:
        try:
            res_f = minimize(objective_slsqp, best_params, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                             options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(constraints_slsqp(res_f.x)) >= -1e-9:
                best_params = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Reconstruct final centers and radii
    if best_params is not None:
        r_opt = best_params[:N]
        u_opt = best_params[N:2*N]
        v_opt = best_params[2*N:3*N]
        x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
        y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
        final_centers = np.column_stack((x_opt, y_opt))
        final_radii = np.maximum(r_opt, 0.0)
    else:
        final_centers = best_centers
        final_radii = best_radii
        best_sum = np.sum(final_radii)
        
    return final_centers, final_radii, float(best_sum)
