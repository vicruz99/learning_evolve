import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-6), 2.6e-5

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def constraints_slqp(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def obj_slqp(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def generate_hex_init(rng, row_counts, rot, scale):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 15.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 15.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.004 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def sa_optimize(centers0, rng, steps=3000):
    """Simulated Annealing on centers with LP radius evaluation."""
    c = centers0.copy()
    r, s = solve_lp(c)
    best_c, best_r, best_s = c.copy(), r.copy(), s
    temp = 0.04
    step = 0.035
    
    for it in range(steps):
        idx = rng.integers(N)
        old = c[idx].copy()
        
        # Propose move
        d = rng.standard_normal(2)
        c[idx] = np.clip(old + d * step, 1e-4, 1.0 - 1e-4)
        
        nr, ns = solve_lp(c)
        delta = ns - s
        
        # Metropolis criterion
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            s = ns
            if s > best_s:
                best_s = s
                best_c = c.copy()
                best_r = nr.copy()
        else:
            c[idx] = old
            
        temp *= 0.9992
        step = max(0.001, step * 0.9994)
        
    return best_c, best_r, best_s

def run_packing():
    rng = np.random.default_rng(42)
    best_c, best_r, best_s = None, None, -np.inf

    # Generate diverse initial configurations
    inits = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [7,5,5,5,4], [5,7,5,5,4]]
    for p in patterns:
        for _ in range(5):
            inits.append(generate_hex_init(rng, p, rng.uniform(-0.3, 0.3), rng.uniform(0.85, 1.15)))
    for _ in range(8):
        inits.append(generate_force_init(rng))

    # Phase 1: LP-Driven Simulated Annealing on centers
    for init_c in inits:
        bc, br, bs = sa_optimize(init_c, rng, steps=3000)
        if bs > best_s:
            best_s = bs
            best_c = bc
            best_r = br

    # Phase 2: SLSQP refinement with perturbations to escape local minima
    if best_c is not None:
        p0 = to_params(best_c, best_r * 0.995)
        bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * (2 * N)
        cons = {'type': 'ineq', 'fun': constraints_slqp}

        for _ in range(12):
            xp = p0.copy()
            xp[:N] += rng.uniform(-0.001, 0.001, N)
            xp[N:] += rng.uniform(-0.012, 0.012, 2*N)
            xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
            xp[N:] = np.clip(xp[N:], 0.0, 1.0)
            
            try:
                res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(constraints_slqp(res.x)) >= -1e-8:
                    curr_s = -res.fun
                    if curr_s > best_s:
                        best_s = curr_s
                        rp = res.x[:N]
                        up = res.x[N:2*N]
                        vp = res.x[2*N:3*N]
                        xp_c = rp + up * (1.0 - 2.0 * rp)
                        yp_c = rp + vp * (1.0 - 2.0 * rp)
                        best_c = np.column_stack((xp_c, yp_c))
                        best_r = rp
                        p0 = res.x.copy()
            except Exception:
                pass

        # Phase 3: High-precision final polish
        try:
            res_f = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(constraints_slqp(res_f.x)) >= -1e-8:
                best_s = -res_f.fun
                rp = res_f.x[:N]
                up = res_f.x[N:2*N]
                vp = res_f.x[2*N:3*N]
                xp_c = rp + up * (1.0 - 2.0 * rp)
                yp_c = rp + vp * (1.0 - 2.0 * rp)
                best_c = np.column_stack((xp_c, yp_c))
                best_r = rp
        except Exception:
            pass

    # Fallback (should rarely be reached)
    if best_c is None:
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        best_r, best_s = solve_lp(pts)
        best_c = pts

    return best_c, np.maximum(best_r, 0.0), float(best_s)
