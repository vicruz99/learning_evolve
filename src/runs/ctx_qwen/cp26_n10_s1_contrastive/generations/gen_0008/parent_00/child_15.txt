Key improvements in this strategy:
1. **LP-Driven Coordinate Ascent**: Instead of relying solely on SLSQP, we use a fast Linear Programming (LP) solver to exactly compute the maximum sum of radii for any fixed set of centers. We then perform coordinate ascent on the centers: iteratively moving one circle at a time, evaluating the LP objective, and accepting moves that improve the sum. This avoids gradient issues and efficiently navigates the center-space landscape.
2. **Precomputed LP Matrix**: The constraint matrix `A_ub` for the LP is constant in structure. We precompute it once and only update the right-hand side `b_ub` (distances and wall bounds) during optimization. This drastically reduces overhead per LP call.
3. **Diverse Initializations**: We generate multiple hexagonal lattice patterns with varied row counts, rotations, and scales, plus force-directed layouts. Each serves as a starting point for coordinate ascent, ensuring we explore different topological arrangements.
4. **SLSQP Polishing**: After coordinate ascent converges, we convert the best centers/radii to a boundary-safe `(r, u, v)` parameterization and run SLSQP. This parameterization automatically satisfies wall constraints, allowing SLSQP to focus on maximizing radii while respecting pairwise non-overlap. We include perturbation-restart loops to escape shallow local minima.
5. **High-Precision Final Polish**: A final SLSQP run with tight tolerances ensures the solution is pushed to numerical limits without violating constraints.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute constant structures for LP to avoid reallocation overhead
A_ub = np.zeros((4*N + N*(N-1)//2, N))
k = 0
for i in range(N):
    for _ in range(4):
        A_ub[k, i] = 1.0
        k += 1
for i in range(N):
    for j in range(i+1, N):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        k += 1

b_ub_storage = np.zeros(A_ub.shape[0])
idx_pair_i, idx_pair_j = np.triu_indices(N, k=1)
c_obj = -np.ones(N)
bounds_lp = [(0.0, None)] * N

def solve_lp_fast(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    k = 0
    for i in range(N):
        x, y = centers[i]
        b_ub_storage[k] = x
        b_ub_storage[k+1] = 1.0 - x
        b_ub_storage[k+2] = y
        b_ub_storage[k+3] = 1.0 - y
        k += 4
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    b_ub_storage[k:] = dists[idx_pair_i, idx_pair_j]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_storage, bounds=bounds_lp, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(N, 1e-6), 0.0

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraint_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return dist2[idx_pair_i, idx_pair_j] - rs[idx_pair_i, idx_pair_j]**2

def objective_func(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7]
    ]
    for pat in patterns:
        for _ in range(2):
            pts = []
            r_est = 0.095
            y = r_est
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.9, 1.1) + 0.5
            rot = rng.uniform(-0.2, 0.2)
            c, s = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pts += rng.uniform(-0.015, 0.015, pts.shape)
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    # Force-directed layouts
    for _ in range(3):
        pts = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(150):
            f = np.zeros_like(pts)
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            d = np.sqrt(np.sum(diff**2, axis=2))
            d = np.maximum(d, 1e-4)
            f += np.sum((1.0/d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
            for dim in range(2):
                f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
                f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
            pts += 0.002 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
    return inits

def run_packing():
    rng = np.random.default_rng(42)
    inits = generate_inits(rng)
    
    best_c = None
    best_s = -np.inf
    
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraint_func}
    
    # Phase 1: LP-Driven Coordinate Ascent from diverse starts
    for init_c in inits:
        curr_c = init_c.copy()
        _, curr_s = solve_lp_fast(curr_c)
        step = 0.025
        
        for _ in range(2000):
            idx = rng.integers(N)
            d = rng.standard_normal(2)
            d /= np.linalg.norm(d)
            nc = curr_c.copy()
            nc[idx] = np.clip(nc[idx] + step * d, 0.02, 0.98)
            _, ns = solve_lp_fast(nc)
            
            if ns > curr_s + 1e-7:
                curr_c = nc
                curr_s = ns
                step = max(0.0005, step * 0.96)
            else:
                step = min(0.04, step * 1.002)
                
        if curr_s > best_s:
            best_s = curr_s
            best_c = curr_c.copy()
            
    # Phase 2: SLSQP Polish with boundary-safe parameterization
    r_best, _ = solve_lp_fast(best_c)
    x0 = to_params(best_c, np.clip(r_best * 0.99, 1e-6, 0.49))
    
    try:
        res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(constraint_func(res.x)) >= -1e-8:
            s = -res.fun
            if s > best_s:
                best_s = s
                best_c, r_best = from_params(res.x)
    except Exception:
        pass
        
    # Phase 3: Perturbation & SLSQP to escape local minima
    for _ in range(25):
        xp = to_params(best_c, np.clip(r_best * 0.995, 1e-6, 0.49))
        xp[N:3*N] += rng.uniform(-0.012, 0.012, 2*N)
        xp[:N] += rng.uniform(-0.0015, 0.0015, N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(objective_func, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                s = -res.fun
                if s > best_s:
                    best_s = s
                    best_c, r_best = from_params(res.x)
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(objective_func, to_params(best_c, r_best * 0.999), method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                         options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(constraint_func(res_f.x)) >= -1e-8:
            best_c, r_best = from_params(res_f.x)
            best_s = -res_f.fun
    except Exception:
        pass
        
    return best_c, np.maximum(r_best, 0.0), float(best_s)
