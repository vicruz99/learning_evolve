import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        x, y = centers[i]
        bounds_vals = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_vals:
            A_ub[k, i] = 1.0
            b_ub[k] = max(0.0, b)
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def objective_func(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraint_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    # Constraint: dist^2 >= (r_i + r_j)^2  =>  dist2 - rs^2 >= 0
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def hex_grid_init(rng, pattern, rot, scale):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(400):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 30.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 30.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.004 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_sa(centers, rng, n_iters=2000):
    """LP-Driven Simulated Annealing on centers."""
    curr_c = centers.copy()
    _, curr_sum = solve_lp(curr_c)
    best_c = curr_c.copy()
    best_sum = curr_sum
    
    temp = 0.04
    step = 0.025
    
    for i in range(n_iters):
        idx = rng.integers(N)
        old_c = curr_c[idx].copy()
        
        # Occasional large random jump to escape deep local minima
        if rng.random() < 0.03:
            curr_c[idx] = rng.uniform(0.1, 0.9, 2)
        else:
            direction = rng.standard_normal(2)
            direction /= np.linalg.norm(direction)
            curr_c[idx] += step * direction
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
        _, new_sum = solve_lp(curr_c)
        delta = new_sum - curr_sum
        
        # Accept if improvement or probabilistically if worse
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = curr_c.copy()
        else:
            curr_c[idx] = old_c
            
        temp *= 0.9992
        step = max(0.0012, step * 0.9993)
        
    return best_c, best_sum

def run_packing():
    rng = np.random.default_rng(42)

    cons = {'type': 'ineq', 'fun': constraint_func}

    best_p = None
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7],
        [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5]

    sa_results = []

    for p in patterns:

            c_init = hex_grid_init(rng, p, rot, sc)
            c_sa, s_sa = run_sa(c_init, rng, n_iters=2000)
            sa_results.append((c_sa, s_sa))
            
    for _ in range(5):
        c_init = force_init(rng)
        c_sa, s_sa = run_sa(c_init, rng, n_iters=2000)
        sa_results.append((c_sa, s_sa))
        
    # Sort results by sum of radii (descending)
    sa_results.sort(key=lambda x: x[1], reverse=True)
    
    # Keep top candidates for intensive polishing
    top_candidates = sa_results[:12]
    
    # Phase 2: SLSQP polishing on best candidates
    for c_cand, s_cand in top_candidates:
        r_cand, _ = solve_lp(c_cand)
        r_cand = np.clip(r_cand * 0.996, 1e-6, 0.49)
        x0 = to_params(c_cand, r_cand)
        
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                s = -res.fun
                if s > best_sum:
                    best_sum = s

        except Exception:

    # Phase 3: Perturbation & Restart to escape local minima
    for _ in range(25):
        xp = best_p.copy()
        xp[:N] += rng.uniform(-0.001, 0.001, N)
        xp[N:3*N] += rng.uniform(-0.01, 0.01, 2*N)

    # Phase 4: High-precision final polish
    try:

        if res_f.success and np.min(constraint_func(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_p)
    return centers, np.maximum(radii, 0.0), float(best_sum)
