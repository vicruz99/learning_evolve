import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            r_opt = res.x
            # Ensure strict non-negativity
            r_opt = np.maximum(r_opt, 0.0)
            return r_opt, np.sum(r_opt)
    except Exception:
        pass
        
    return np.full(n, 1e-6), 2.6e-5

def phys_to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def param_to_phys(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def objective_params(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints_params(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_hex_inits(rng, count):
    """Generates diverse hexagonal lattice initializations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7],
        [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5],
        [8,5,5,5,3], [5,8,5,5,3], [4,4,5,6,7], [7,6,5,4,4]
    ]
    for _ in range(count):
        pat = patterns[rng.integers(0, len(patterns))]
        pts = []
        r_est = 0.095
        y = r_est
        row = 0
        for cnt in pat:
            shift = (row % 2) * r_est
            x = r_est + shift
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
        pts = np.array(pts[:N])
        pts = (pts - 0.5) * rng.uniform(0.85, 1.15) + 0.5
        rot = rng.uniform(-0.25, 0.25)
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        pts += rng.uniform(-0.015, 0.015, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
    return inits

def generate_force_inits(rng, count):
    """Generates force-directed layout initializations."""
    inits = []
    for _ in range(count):
        pts = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(200):
            f = np.zeros_like(pts)
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            d = np.sqrt(np.sum(diff**2, axis=2))
            d = np.maximum(d, 1e-4)
            f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
            for dim in range(2):
                f[:, dim] += 15.0 * np.maximum(0, 0.1 - pts[:, dim])
                f[:, dim] -= 15.0 * np.maximum(0, pts[:, dim] - 0.9)
            pts += 0.002 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
    return inits

def simulated_annealing(centers, rng):
    """Optimizes centers using Simulated Annealing with LP oracle."""
    curr_c = centers.copy()
    _, curr_s = solve_lp_radii(curr_c)
    best_c = curr_c.copy()
    best_s = curr_s
    
    temp = 0.04
    step = 0.035
    
    for it in range(12000):
        idx = rng.integers(N)
        direction = rng.standard_normal(2)
        direction /= np.linalg.norm(direction)
        
        # Occasional large jump
        current_step = step * (6.0 if rng.random() < 0.03 else 1.0)
        
        old = curr_c[idx].copy()
        curr_c[idx] = np.clip(curr_c[idx] + current_step * direction, 0.01, 0.99)
        
        _, new_s = solve_lp_radii(curr_c)
        delta = new_s - curr_s
        
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
            curr_s = new_s
            if new_s > best_s:
                best_s = new_s
                best_c = curr_c.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9993
        step = max(0.001, step * 0.9995)
        
    return best_c, best_s

def nelder_mead_optimize(centers):
    """Optimizes centers using Nelder-Mead with LP objective."""
    def neg_sum_radii(flat_c):
        c = np.clip(flat_c.reshape(N, 2), 0.01, 0.99)
        _, s = solve_lp_radii(c)
        return -s
        
    x0 = centers.flatten()
    res = minimize(neg_sum_radii, x0, method='Nelder-Mead',
                   options={'maxiter': 8000, 'xatol': 1e-9, 'fatol': 1e-10, 'disp': False})
    opt_c = np.clip(res.x.reshape(N, 2), 0.01, 0.99)
    _, opt_s = solve_lp_radii(opt_c)
    return opt_c, opt_s

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints_params}
    
    best_sum = -np.inf
    best_p = None
    
    # Generate diverse initial configurations
    all_inits = generate_hex_inits(rng, 60)
    all_inits.extend(generate_force_inits(rng, 20))
    
    # Evaluate inits and run SLSQP on top candidates
    scores = []
    for c in all_inits:
        r, s = solve_lp_radii(c)
        scores.append((s, c, r))
        
    # Sort by sum of radii descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Run SLSQP on top 15
    for i, (s, c, r) in enumerate(scores[:15]):
        r_init = np.clip(r * 0.99, 1e-6, 0.49)
        x0 = phys_to_params(c, r_init)
        
        try:
            res = minimize(objective_params, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints_params(res.x)) >= -1e-8:
                curr_s = -res.fun
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Fallback if SLSQP failed
    if best_p is None:
        c_f = scores[0][1]
        r_f, _ = solve_lp_radii(c_f)
        best_p = phys_to_params(c_f, r_f * 0.99)
        best_sum = -objective_params(best_p)
        
    # Extract best centers for further optimization
    c_best, _ = param_to_phys(best_p)
    
    # Phase 2: Simulated Annealing on centers
    c_sa, s_sa = simulated_annealing(c_best, rng)
    if s_sa > best_sum:
        best_sum = s_sa
        r_sa, _ = solve_lp_radii(c_sa)
        best_p = phys_to_params(c_sa, r_sa * 0.995)
        c_best = c_sa
        
    # Phase 3: Nelder-Mead refinement
    c_nm, s_nm = nelder_mead_optimize(c_best)
    if s_nm > best_sum:
        best_sum = s_nm
        r_nm, _ = solve_lp_radii(c_nm)
        best_p = phys_to_params(c_nm, r_nm * 0.995)
        c_best = c_nm
        
    # Phase 4: Perturbation & SLSQP restarts to escape remaining local minima
    for k in range(25):
        x0 = best_p.copy()
        x0[:N] += rng.uniform(-0.002, 0.002, N)
        x0[N:3*N] += rng.uniform(-0.012, 0.012, 2*N)
        x0[:N] = np.clip(x0[:N], 1e-6, 0.49)
        x0[N:3*N] = np.clip(x0[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(objective_params, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints_params(res.x)) >= -1e-8:
                curr_s = -res.fun
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 5: High-precision final polish
    x0_polish = best_p.copy()
    x0_polish[:N] *= 0.9999  # Slight shrink for strict interior feasibility
    try:
        res_f = minimize(objective_params, x0_polish, method='SLSQP', bounds=bounds_slqp,
                         constraints=cons_slqp,
                         options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(constraints_params(res_f.x)) >= -1e-9:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    # Reconstruct physical centers and radii
    r_opt = best_p[:N]
    u_opt = best_p[N:2*N]
    v_opt = best_p[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    radii = np.maximum(r_opt, 0.0)
    
    # Final validation adjustment (clamp tiny negative/overflow due to float)
    radii = np.clip(radii, 0.0, 0.5)
    centers = np.clip(centers, 1e-12, 1.0 - 1e-12)
    
    return centers, radii, float(best_sum)
