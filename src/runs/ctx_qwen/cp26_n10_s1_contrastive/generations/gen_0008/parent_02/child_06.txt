import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(params):
    """Objective: maximize sum of radii <=> minimize negative sum."""
    return -np.sum(params[:N])

def constraint_func(params):
    """
    Inequality constraints: pairwise non-overlap.
    Boundary constraints are automatically satisfied by parameterization.
    Returns array of values that must be >= 0.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    num_con = 4*n + n*(n-1)//2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i+1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds_r = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def centers_to_params(centers, radii):
    """Converts physical centers to (r, u, v) optimization parameters."""
    r = np.maximum(radii * 0.99, 1e-6)
    x, y = centers[:, 0], centers[:, 1]
    
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((x - r) / denom, 0.0, 1.0)
    v = np.clip((y - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def hex_init(rng, pattern, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(pattern):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
        
    pts += rng.uniform(-0.015, 0.015, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def force_init(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.rand(N, 2) * 0.8 + 0.1
    r_curr = np.full(N, 0.06)
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / (dists**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
        step_size = 0.008 * (0.99**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_others[i] = np.inf
            d_pair = np.min(dists_others)
            r_curr[i] = 0.8 * min(d_wall, d_pair/2.0)
    return pts

def run_packing():
    rng = np.random.default_rng(123)
    bounds = [(1e-6, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_params = None
    best_sum = -np.inf
    
    # Diverse hexagonal patterns summing to 26
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5],
        [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5],
        [5,5,5,6,5], [7,5,5,5,4], [4,5,5,5,7], [5,7,5,5,4], [4,5,5,7,5],
        [6,7,5,4,4], [7,6,5,5,3], [8,5,5,5,3], [5,8,5,5,3], [5,5,8,5,3],
        [9,5,5,5,2], [5,9,5,5,2], [5,5,9,5,2], [5,5,5,9,2], [6,6,6,5,3]
    ]
    
    inits = []
    for p in patterns:
        for _ in range(6):
            rot = rng.uniform(-0.3, 0.3)
            sc = rng.uniform(0.85, 1.15)
            inits.append(hex_init(rng, p, rot, sc))
            
    for s in range(25):
        inits.append(force_init(rng, s))
        
    # Phase 1: Broad SLSQP search from diverse starts
    for pts in inits:
        r_lp, _ = solve_lp(pts)
        p0 = centers_to_params(pts, r_lp)
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            pass
            
    # Fallback if optimization failed
    if best_params is None:
        pts = hex_init(rng, [6,5,6,5,4], 0.0, 1.0)
        r_lp, _ = solve_lp(pts)
        best_params = centers_to_params(pts, r_lp)
        best_sum = np.sum(best_params[:N])
        
    # Phase 2: LP-driven Simulated Annealing on centers
    r_best = best_params[:N]
    u_best = best_params[N:2*N]
    v_best = best_params[2*N:3*N]
    c_best = np.column_stack((r_best + u_best*(1.0-2.0*r_best), r_best + v_best*(1.0-2.0*r_best)))
    
    curr_c = c_best.copy()
    curr_r, curr_sum = solve_lp(curr_c)
    best_lp_c = curr_c.copy()
    best_lp_r = curr_r.copy()
    best_lp_sum = curr_sum
    
    temp = 0.03
    step = 0.035
    for it in range(6000):
        idx = rng.integers(N)
        old_c = curr_c[idx].copy()
        if rng.random() < 0.08:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            direction = rng.standard_normal(2)
            curr_c[idx] = np.clip(old_c + direction * step, 0.01, 0.99)
            
        new_r, new_sum = solve_lp(curr_c)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_sum = new_sum
            curr_r = new_r
            if new_sum > best_lp_sum:
                best_lp_sum = new_sum
                best_lp_c = curr_c.copy()
                best_lp_r = curr_r.copy()
        else:
            curr_c[idx] = old_c
            
        temp *= 0.9993
        step = max(0.0015, step * 0.9995)
        
    # Update best_params if LP-SA found better
    if best_lp_sum > best_sum:
        best_sum = best_lp_sum
        best_params = centers_to_params(best_lp_c, best_lp_r)
        
    # Phase 3: Perturbation & SLSQP restarts to escape local minima
    for k in range(15):
        x0 = best_params.copy()
        x0[:N] += rng.normal(0, 0.002, N)
        x0[N:2*N] += rng.uniform(-0.02, 0.02, N)
        x0[2*N:3*N] += rng.uniform(-0.02, 0.02, N)
        x0[:N] = np.clip(x0[:N], 1e-6, 0.49)
        x0[N:3*N] = np.clip(x0[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            pass
            
    # Phase 4: High-precision final polish
    try:
        res = minimize(objective, best_params, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        if res.success and np.min(constraint_func(res.x)) >= -1e-9:
            best_params = res.x
            best_sum = -res.fun
    except Exception:
        pass
        
    # Reconstruct physical centers and radii
    r_opt = best_params[:N]
    u_opt = best_params[N:2*N]
    v_opt = best_params[2*N:3*N]
    centers = np.column_stack((r_opt + u_opt*(1.0-2.0*r_opt), r_opt + v_opt*(1.0-2.0*r_opt)))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
