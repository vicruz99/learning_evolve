import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    n_con = 4*n + n*(n-1)//2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i+1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)]*n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def obj_slqp(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons_slqp(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Converts physical centers/radii to (r, u, v) optimization parameters."""
    r = np.maximum(radii * 0.995, 1e-6)
    denom = np.clip(1.0 - 2.0*r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r)/denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r)/denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(p):
    """Reconstructs physical centers and radii from parameters."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u*(1.0 - 2.0*r)
    y = r + v*(1.0 - 2.0*r)
    return np.column_stack([x, y]), r

def make_hex(seed, pat, rot, sc):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    re = 0.095
    y = re
    for ri, cnt in enumerate(pat):
        shift = (ri % 2) * re
        x = re + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0*re
        y += np.sqrt(3.0)*re
    pts = np.array(pts[:N])
    pts = (pts - 0.5)*sc + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]).T + 0.5
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)*0.8 + 0.1
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-5
        rep = 1.0/(d**2)
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:,:,None]*diff/d[:,:,None], axis=1)
        for dd in range(2):
            f[:, dd] += 15.0*np.maximum(0, 0.05 - pts[:, dd])
            f[:, dd] -= 15.0*np.maximum(0, pts[:, dd] - 0.95)
        pts += 0.004*f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_grid(seed):
    """Generates a perturbed grid initialization."""
    rng = np.random.RandomState(seed)
    pts = np.array([[0.08 + 0.16*i, 0.08 + 0.16*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)
