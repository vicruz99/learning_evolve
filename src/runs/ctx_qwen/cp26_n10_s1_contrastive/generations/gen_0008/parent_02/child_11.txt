import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Preallocate constraints matrix
    n_constr = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constr, n))
    b_ub = np.zeros(n_constr)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_list = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_list:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds_r = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def constraint_func(params):
    """Inequality constraints: pairwise non-overlap. Boundary handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def objective_func(params):
    """Objective: maximize sum of radii <=> minimize negative sum."""
    return -np.sum(params[:N])

def to_params(centers, radii=None):
    """Converts physical centers to (r, u, v) optimization parameters."""
    if radii is None:
        radii, _ = solve_lp_radii(centers)
    r = np.maximum(radii * 0.995, 1e-6)
    x, y = centers[:, 0], centers[:, 1]
    
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((x - r) / denom, 0.0, 1.0)
    v = np.clip((y - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstructs physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def hex_init(rng, pattern, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(pattern):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def corner_grid_init(rng):
    """Generates a corner-anchored initialization with hex inner packing."""
    pts = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    pts += [[0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5]]
    
    inner_pts = []
    r_est = 0.09
    y = 0.25
    row = 0
    while len(inner_pts) < 18 and y <= 0.75:
        shift = (row % 2) * r_est
        x = 0.25 + shift
        while x <= 0.75 and len(inner_pts) < 18:
            inner_pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts += inner_pts[:18]
    pts = np.array(pts)
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        rep = 1.0 / (dist**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dist[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
            
        pts += 0.005 * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    
    bounds_slqp = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons_slqp = {'type': 'ineq', 'fun': constraint_func}
    
    best_params = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5],
        [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7], [6,6,5,5,4], [5,5,6,5,5],
        [6,5,5,5,5], [5,5,5,6,5], [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3]
    ]
    
    for p in patterns:
        for s in range(3):
            rot = rng.uniform(-0.2, 0.2)
            sc = rng.uniform(0.88, 1.12)
            inits.append(hex_init(rng, p, rot, sc))
            
    for _ in range(8):
        inits.append(corner_grid_init(rng))
    for _ in range(8):
        inits.append(force_init(rng))
        
    # Phase 1: Broad SLSQP search from diverse starts
    for pts in inits:
        x0 = to_params(pts)
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            continue
            
    if best_params is None:
        best_params = to_params(hex_init(rng, [6,5,6,5,4], 0.0, 1.0))
        best_sum = -objective_func(best_params)
        
    # Phase 2: LP-driven SA on centers
    centers, radii = from_params(best_params)
    curr_sum = best_sum
    temp = 0.03
    step = 0.04
    
    for it in range(5000):
        op = rng.choice(['move', 'swap', 'jump'])
        old_c = None
        swap_occurred = False
        
        if op == 'swap':
            i, j = rng.choice(N, 2, replace=False)
            old_c = centers[i].copy()
            centers[i], centers[j] = centers[j].copy(), old_c
            swap_occurred = True
        elif op == 'jump':
            idx_m = rng.integers(N)
            old_c = centers[idx_m].copy()
            centers[idx_m] = rng.uniform(0.05, 0.95, 2)
        else:
            idx_m = rng.integers(N)
            old_c = centers[idx_m].copy()
            dir = rng.standard_normal(2)
            dir /= np.linalg.norm(dir)
            step_cur = step * (4.0 if rng.random() < 0.05 else 1.0)
            centers[idx_m] = np.clip(old_c + dir * step_cur, 0.01, 0.99)
            
        new_r, new_sum = solve_lp_radii(centers)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_sum = new_sum
            radii = new_r
            if new_sum > best_sum:
                best_sum = new_sum
                best_params = to_params(centers, radii)
        else:
            if not swap_occurred:
                centers[idx_m] = old_c
            else:
                centers[i], centers[j] = centers[j].copy(), old_c
                
        temp *= 0.9993
        step = max(0.0015, step * 0.9995)
        
    # Phase 3: Basin Hopping / Perturbation Restarts
    for k in range(10):
        x0 = best_params.copy()
        x0[:N] += rng.uniform(-0.002, 0.002, N)
        x0[N:3*N] += rng.uniform(-0.02, 0.02, 2*N)
        x0[:N] = np.clip(x0[:N], 1e-6, 0.49)
        x0[N:3*N] = np.clip(x0[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            pass
            
    # Phase 4: Rotation Escape & Re-optimization
    centers, radii = from_params(best_params)
    for _ in range(8):
        rot = rng.uniform(-0.15, 0.15)
        c_val, s_val = np.cos(rot), np.sin(rot)
        mat = np.array([[c_val, -s_val], [s_val, c_val]])
        c_rot = (centers - 0.5) @ mat.T + 0.5
        c_rot = np.clip(c_rot, 0.02, 0.98)
        
        r_rot, sum_rot = solve_lp_radii(c_rot)
        if sum_rot > best_sum:
            best_sum = sum_rot
            best_params = to_params(c_rot, r_rot)
            
        x0 = to_params(c_rot, r_rot * 0.995)
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            pass
            
    # Phase 5: High-precision final polish
    try:
        res = minimize(objective_func, best_params, method='SLSQP', bounds=bounds_slqp,
                       constraints=cons_slqp, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        if res.success and np.min(constraint_func(res.x)) >= -1e-9:
            best_params = res.x
            best_sum = -res.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_params)
    
    # Final safety shrink to guarantee strict validity within numerical tolerance
    radii = np.maximum(radii * 0.99999, 0.0)
    
    return centers, radii, float(np.sum(radii))
