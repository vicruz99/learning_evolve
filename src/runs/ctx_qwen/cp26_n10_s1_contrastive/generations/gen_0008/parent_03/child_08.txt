import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 1e-4

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def constraints(v):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2."""
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def centers_to_params(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    v = np.empty(3 * N)
    v[0::3] = r
    v[1::3] = u
    v[2::3] = w
    return v

def params_to_centers(params):
    """Reconstruct centers and radii from parameters."""
    r = params[0::3]
    u = params[1::3]
    w = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def generate_hex_init(rng, seed_idx):
    """Generates a hexagonal lattice initialization."""
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,6,5,5,4], [5,7,5,5,4]]
    pat = pats[seed_idx % len(pats)]
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(pat):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
        
    pts = np.array(pts[:N])
    pts -= 0.5
    scale = rng.uniform(0.80, 1.20)
    pts *= scale
    pts += 0.5
    rot = rng.uniform(-0.35, 0.35)
    c, s = np.cos(rot), np.sin(rot)
    pts = pts @ np.array([[c, -s], [s, c]])
    pts += rng.uniform(-0.02, 0.02, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def generate_force_init(rng, seed_idx):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    r_curr = np.full(N, 0.08)
    for _ in range(300):
        forces = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / (dists**2)
        np.fill_diagonal(rep, 0.0)
        forces += np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        
        for d in range(2):
            forces[:, d] += 40.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 40.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.008 * (0.99**_)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
        
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_others[i] = np.inf
            d_pair = np.min(dists_others)
            r_curr[i] = 0.85 * min(d_wall, d_pair/2.0)
            
    return pts

def run_sa(centers, rng, iterations=5000):
    """Simulated Annealing on centers using LP for evaluation."""
    curr_c = centers.copy()
    curr_r, curr_sum = solve_lp(curr_c)
    best_c = curr_c.copy()
    best_r = curr_r.copy()
    best_sum = curr_sum
    
    temp = 0.06
    step = 0.035
    no_improve = 0
    
    for it in range(iterations):
        idx = rng.integers(N)
        old_pos = curr_c[idx].copy()
        
        # Occasional large random jump to escape deep local minima
        if rng.random() < 0.03:
            curr_c[idx] = rng.uniform(0.02, 0.98, 2)
        else:
            direction = rng.standard_normal(2)
            direction /= np.linalg.norm(direction)
            curr_c[idx] += direction * step
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
        nr, ns = solve_lp(curr_c)
        delta = ns - curr_sum
        
        # Adaptive cooling and step size
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_sum = ns
            curr_r = nr
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
                no_improve = 0
            # Increase step slightly on improvement to explore vicinity
            step = min(0.05, step * 1.02)
        else:
            curr_c[idx] = old_pos
            no_improve += 1
            if no_improve > 400:
                # Escape stagnation
                step = 0.04
                no_improve = 0
                
        temp *= 0.9993
        step = max(0.0015, step * 0.9995)
        
    return best_c, best_r, best_sum

def run_packing():

    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints}
    
    best_vars = None
    best_sum = -np.inf

    best_r = None

    inits = []

    sa_results = []
    for pts in inits:

    # Sort by SA sum descending
    sa_results.sort(key=lambda x: x[2], reverse=True)

    # Take top 8 to balance exploration and computation
    top_sa = sa_results[:8]

        v0 = centers_to_params(c_sa, np.clip(r_sa * 0.99, 1e-6, 0.49))
        try:

            if res.success:
