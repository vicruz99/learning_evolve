def objective_slqp(v):

    return -np.sum(v[0::3])

def constraint_func_slqp(v):
    """
    Inequality constraints: dist_sq >= (r_i + r_j)^2.
    Boundary constraints are automatically satisfied by the (r, u, w) parameterization.
    """
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)

    dy = y[:, np.newaxis] - y[np.newaxis, :]

    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    v = np.empty(3 * N)
    v[0::3] = r
    v[1::3] = u
    v[2::3] = w
    return v

def params_to_centers(v):
    """Reconstruct centers and radii from (r, u, w) parameters."""
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

            x += 2.0 * r_est

    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)
