import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(val)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dists[i, j])
            
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def objective_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def constraints_slqp(v):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundary handled by parameterization."""
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    v = np.empty(3 * N)
    v[0::3] = r
    v[1::3] = u
    v[2::3] = w
    return v

def generate_hex_init(rng, seed):
    """Generates a hexagonal lattice initialization with random perturbation."""
    rng_s = np.random.RandomState(seed)
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,6,5,5,4], [5,7,5,5,4]]
    pat = pats[seed % len(pats)]
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(pat):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:N])
    pts -= 0.5
    scale = rng_s.uniform(0.85, 1.15)
    pts *= scale
    pts += 0.5
    rot = rng_s.uniform(-0.2, 0.2)
    c, s = np.cos(rot), np.sin(rot)
    pts = pts @ np.array([[c, -s], [s, c]])
    pts += rng_s.uniform(-0.015, 0.015, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def generate_force_init(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_s = np.random.RandomState(seed)
    pts = rng_s.uniform(0.1, 0.9, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-5)
        rep = 1.0 / d**2
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 15.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 15.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.006 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_sa(centers, rng, steps=4500):
    """LP-driven Simulated Annealing on centers."""
    r, s = solve_lp(centers)
    best_c = centers.copy()
    best_r = r.copy()
    best_s = s
    curr_c = centers.copy()
    curr_s = s
    temp = 0.05
    step = 0.025
    
    for i in range(steps):
        idx = rng.integers(N)
        old_pos = curr_c[idx].copy()
        
        if rng.random() < 0.04:
            curr_c[idx] = rng.uniform(0.02, 0.98, 2)
        else:
            direction = rng.standard_normal(2)
            direction /= np.linalg.norm(direction)
            curr_c[idx] = np.clip(curr_c[idx] + step * direction, 0.01, 0.99)
            
        nr, ns = solve_lp(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            if ns > best_s:
                best_s = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
        else:
            curr_c[idx] = old_pos
            
        temp *= 0.9994
        step = max(0.0015, step * 0.9996)
        
        # Occasional shake to escape flat regions
        if i % 600 == 0 and i > 0:
            if rng.random() < 0.25:
                curr_c += rng.uniform(-0.015, 0.015, (N, 2))
                curr_c = np.clip(curr_c, 0.02, 0.98)
                _, curr_s = solve_lp(curr_c)
                
    return best_c, best_r, best_s
