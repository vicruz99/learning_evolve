import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def objective_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def constraint_func_slqp(v):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundaries handled by parameterization."""
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def centers_to_params(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    v = np.empty(3 * N)
    v[0::3] = r
    v[1::3] = u
    v[2::3] = w
    return v

def params_to_centers_radii(params):
    """Reconstruct centers and radii from (r, u, w) parameters."""
    r = params[0::3]
    u = params[1::3]
    w = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def generate_hex_inits(rng, count):
    """Generates diverse hexagonal lattice initializations."""
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,6,5,5,4], [5,7,5,5,4]]
    for _ in range(count):
        pat = rng.choice(pats)
        pts = []
        r_est = 0.095
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = (r_idx % 2) * r_est
            x = r_est + shift
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3.0)
        pts = np.array(pts[:N])
        pts -= 0.5
        scale = rng.uniform(0.85, 1.15)
        pts *= scale
        pts += 0.5
        rot = rng.uniform(-0.25, 0.25)
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts += rng.uniform(-0.015, 0.015, (N, 2))
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
    return inits

def de_fitness(centers_flat):
    """Fitness function for Differential Evolution: negative sum of LP-optimal radii."""
    centers = centers_flat.reshape(N, 2)
    _, s = solve_radii_lp(centers)
    return -s

                cv = constraint_func_slqp(res.x)
