import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = N
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 1e-4

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    v = np.empty(3 * N)
    v[0::3] = r
    v[1::3] = u
    v[2::3] = w
    return v

def get_centers_radii(v):
    """Reconstruct centers and radii from (r, u, w) parameters."""
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def slqp_constraints(v):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2."""
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def slqp_objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def hill_climb(centers, rng):
    """Intensive local search on centers using LP evaluation."""
    curr_c = centers.copy()
    curr_r, curr_sum = solve_lp(curr_c)
    best_c, best_r, best_sum = curr_c.copy(), curr_r.copy(), curr_sum
    
    step = 0.025

        idx = rng.integers(N)
        old_pos = curr_c[idx].copy()
        
        # Occasionally jump to a new random location to escape deep basins

            curr_c[idx] = rng.uniform(0.02, 0.98, 2)
        else:
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
        nr, ns = solve_lp(curr_c)
        
        if ns > best_sum + 1e-9:

            curr_c[idx] = old_pos
            if rng.random() < 0.04:
                step = min(0.06, step * 1.15)
                
    return best_c, best_r, best_sum

def generate_inits(rng):

    # Random organic starts
    for _ in range(15):
        pts = rng.uniform(0.05, 0.95, (N, 2))
        inits.append(pts)
