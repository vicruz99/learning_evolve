1. **LP-based Radius Evaluation**: For any fixed set of center coordinates, the optimal radii that maximize the sum while respecting boundary and non-overlap constraints form a Linear Programming problem. Solving this LP is fast and exact, providing a reliable objective function value for any center configuration.
2. **Simulated Annealing on Centers**: We treat the LP-computed sum of radii as an energy function and run Simulated Annealing (SA) on the 52-dimensional center space. SA effectively navigates the rugged landscape of circle packings, escaping local minima through controlled thermal fluctuations and probabilistic acceptance of worse moves.
3. **Diverse Initialization**: We generate multiple distinct starting configurations (hexagonal lattices, square grids, and force-directed layouts) to ensure coverage of different topological packing basins.
4. **Intensive Search & Topology Changes**: The SA routine incorporates random swaps and occasional large jumps to alter the contact graph topology, which is crucial for reaching high-density packings.
5. **SLSQP Joint Polish**: After SA converges, we take the best centers and radii and run a constrained nonlinear optimizer (SLSQP) on all `(x, y, r)` variables jointly. This fine-tunes positions and sizes simultaneously to squeeze out the final precision, respecting explicit boundary and pairwise distance constraints.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4*n + n*(n-1)//2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i+1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)]*n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-4

def generate_inits(rng, count):
    """Generates diverse initial center configurations."""
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,6,5,5,4], [7,5,5,5,4]]
    
    # Hexagonal lattice patterns
    for _ in range(count//2):
        pat = rng.choice(pats)
        pts = []
        r_e = 0.095
        y = r_e
        for r_idx, cnt in enumerate(pat):
            shift = (r_idx%2)*r_e
            x = r_e + shift
            for _ in range(cnt):
                if len(pts) < N: pts.append([x,y])
                x += 2.0*r_e
            y += np.sqrt(3.0)*r_e
        pts = np.array(pts[:N])
        pts = (pts - 0.5)*rng.uniform(0.8, 1.2) + 0.5
        rot = rng.uniform(-0.3, 0.3)
        c,s = np.cos(rot), np.sin(rot)
        pts = (pts-0.5) @ np.array([[c,-s],[s,c]]) + 0.5
        pts += rng.uniform(-0.02, 0.02, (N,2))
        inits.append(np.clip(pts, 0.02, 0.98))
        
    # Square grid + center
    for _ in range(count//4):
        pts = np.array([[0.1+0.2*i, 0.1+0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        pts += rng.uniform(-0.03, 0.03, (N,2))
        inits.append(np.clip(pts, 0.02, 0.98))
        
    # Force-directed organic layouts
    for _ in range(count//4):
        pts = rng.uniform(0.1, 0.9, (N,2))
        for _ in range(80):
            diff = pts[:,None,:] - pts[None,:,:]
            dist = np.sqrt(np.sum(diff**2, axis=2))
            dist = np.maximum(dist, 0.05)
            rep = 1.0/(dist**2)
            np.fill_diagonal(rep, 0.0)
            f = np.sum(rep[:,:,None] * diff/dist[:,:,None], axis=1)
            pts += 0.002*f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    return inits[:count]

def run_sa(centers_init, rng, iters=4000):
    """Simulated Annealing on center positions using LP objective."""
    curr = centers_init.copy()
    curr_r, curr_sum = solve_lp(curr)
    best_c, best_r, best_sum = curr.copy(), curr_r.copy(), curr_sum
    temp = 0.06
    step = 0.05
    
    for i in range(iters):
        prev = curr.copy()
        
        # Occasionally swap two circles to change contact topology
        if rng.random() < 0.2:
            i1, i2 = rng.choice(N, 2, replace=False)
            curr[i1], curr[i2] = curr[i2].copy(), curr[i1].copy()
        else:
            idx = rng.integers(N)
            if rng.random() < 0.05:
                # Large random jump to escape deep local minima
                curr[idx] = rng.uniform(0.02, 0.98, 2)
            else:
                curr[idx] += rng.normal(0, step, 2)
                curr[idx] = np.clip(curr[idx], 0.01, 0.99)
                
        nr, ns = solve_lp(curr)
        delta = ns - curr_sum
        
        # Metropolis criterion
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta/temp)):
            curr_sum = ns
            curr_r = nr
            if ns > best_sum:
                best_sum = ns
                best_c = curr.copy()
                best_r = nr.copy()
        else:
            curr = prev
            
        temp *= 0.9993
        step = max(0.001, step * 0.9995)
        
    return best_c, best_r, best_sum

def slsqp_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def slsqp_cons(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise non-overlap: dist_sq >= (r_i + r_j)^2
    dx = x[:,None] - x[None,:]
    dy = y[:,None] - y[None,:]
    d2 = dx**2 + dy**2
    rs = r[:,None] + r[None,:]
    c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    
    bounds_slqp = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_slqp = {'type': 'ineq', 'fun': slsqp_cons}
    
    best_c, best_r, best_sum = None, None, -np.inf
    
    # Phase 1: Broad search from diverse initializations
    inits = generate_inits(rng, 12)
    for pts in inits:
        c, r, s = run_sa(pts, rng, iters=3000)
        if s > best_sum:
            best_c, best_r, best_sum = c, r, s
            
    # Phase 2: Intensive SA from the best found so far
    c, r, s = run_sa(best_c, rng, iters=8000)
    if s > best_sum:
        best_c, best_r, best_sum = c, r, s
        
    # Phase 3: High-precision SLSQP polish on joint variables
    v0 = np.empty(3*N)
    v0[0::3] = best_c[:, 0]
    v0[1::3] = best_c[:, 1]
    v0[2::3] = best_r * 0.995  # Slight shrink ensures strict interior feasibility for SLSQP start
    
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds_slqp,
                       constraints=cons_slqp, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
        if res.success:
            cv = slsqp_cons(res.x)
            if np.min(cv) >= -1e-7:
                s_polish = -res.fun
                if s_polish > best_sum:
                    best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_r = res.x[2::3]
                    best_sum = s_polish
    except Exception:
        pass
        
    # Final LP evaluation guarantees radii are exactly maximal for the returned centers
    final_r, final_sum = solve_lp(best_c)
    if final_sum > best_sum:
        best_r = final_r
        best_sum = final_sum
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
