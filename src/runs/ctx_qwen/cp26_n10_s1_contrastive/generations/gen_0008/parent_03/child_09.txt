import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((4*n + n*(n-1)//2, n))
    b_ub = np.zeros(4*n + n*(n-1)//2)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 1e-4

def obj_func(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def constr_func(params):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundary handled by parameterization."""
    r = params[0::3]
    u = params[1::3]
    w = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def params_to_cen_rad(params):
    """Reconstruct centers and radii from (r, u, w) parameters."""
    r = params[0::3]
    u = params[1::3]
    w = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def cen_rad_to_params(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    v = np.empty(3 * N)
    v[0::3] = r
    v[1::3] = u
    v[2::3] = w
    return v

def run_local_search(centers, rng):
    """Performs simulated annealing on centers, evaluating LP sum of radii."""
    n = centers.shape[0]
    curr_c = centers.copy()
    curr_r, curr_s = solve_lp(curr_c)
    best_c, best_r, best_s = curr_c.copy(), curr_r.copy(), curr_s
    
    step = 0.025
    temp = 0.06
    
    for it in range(8000):
        idx = rng.integers(n)
        old = curr_c[idx].copy()
        
        # Occasional large jump to escape deep basins
        if rng.random() < 0.03:
            curr_c[idx] = rng.uniform(0.02, 0.98, 2)
        else:
            dir_vec = rng.standard_normal(2)
            dir_vec /= np.linalg.norm(dir_vec)
            curr_c[idx] += dir_vec * step
            curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
            
        nr, ns = solve_lp(curr_c)
        delta = ns - curr_s
        
        # SA acceptance
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            curr_r = nr
            if ns > best_s:
                best_s = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
        else:
            curr_c[idx] = old
            
        # Adaptive step size
        temp *= 0.9992
        step = max(0.001, step * (1.02 if delta > 0 else 0.98))
        
    return best_c, best_r, best_s

def generate_inits(rng, num):
    """Generates diverse hexagonal lattice initializations."""
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], 
            [7,5,5,5,4], [5,7,5,5,4], [6,4,6,5,5], [5,5,6,5,5], [6,6,5,5,4]]
    
    for _ in range(num):
        pat = rng.choice(pats)
        pts = []
        r_est = 0.095
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = (r_idx % 2) * r_est
            x = r_est + shift
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3.0)
            
        pts = np.array(pts[:N])
        pts -= 0.5
        scale = rng.uniform(0.85, 1.15)
        pts *= scale
        pts += 0.5
        
        rot = rng.uniform(-0.3, 0.3)
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
        
        pts += rng.uniform(-0.02, 0.02, (N, 2))
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
    return inits

def run_packing():

        except Exception:
            pass
            
    # Phase 2: Iterative Local Search + SLSQP refinement
