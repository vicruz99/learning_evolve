N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):

    n = centers.shape[0]

    b_ub = np.zeros(num_c)

        x, y = centers[i]

            b_ub[k] = b
            k += 1

        pass
    return np.full(n, 1e-5), 1e-4

def slqp_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def slqp_cons(v):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundary handled by parameterization."""
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

    w = np.clip((c[:, 1] - r) / denom, 0.0, 1.0)
    
    v = np.empty(3 * N)
    v[0::3] = r
    v[1::3] = u
    v[2::3] = w
    return v

def from_params(v):

    u = v[1::3]
    w = v[2::3]
    x = r + u * (1.0 - 2.0 * r)
