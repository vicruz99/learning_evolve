To maximize the sum of radii for 26 circles in a unit square, we employ a hybrid optimization strategy combining dual-gradient ascent on circle centers with a high-precision SLSQP refinement.

1. **Dual-Gradient Ascent**: For fixed centers, the maximum sum of radii is found via Linear Programming (LP). The sensitivity of the objective to center positions is given exactly by the LP dual variables (Lagrange multipliers) of the pairwise distance and boundary constraints. We compute these duals and perform gradient ascent on the centers: `centers += step * gradient`. This efficiently pushes circles apart or towards boundaries to relax constraints and grow radii.
2. **Diverse Initialization**: We seed the gradient ascent from multiple configurations (hexagonal lattices, random placements, and corner/edge-focused layouts) to explore different topological basins of attraction.
3. **SLSQP Polish**: The best configuration from gradient ascent is mapped to a boundary-safe parameterization `(r, u, w)` where `x = r + u*(1-2r)` and `y = r + w*(1-2r)`. This automatically satisfies boundary constraints, allowing SLSQP to focus purely on pairwise non-overlap constraints for a final high-precision refinement.
4. **Robustness**: We include adaptive step sizes, occasional random perturbations to escape local minima, and strict feasibility checks. This approach consistently pushes past local optima that trap pure coordinate descent or basic force-directed methods.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_and_duals(centers):
    """Solves LP for radii and returns optimal radii, objective value, and gradient w.r.t centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_pairs = n * (n - 1) // 2
    num_cons = 4 * n + num_pairs
    A_ub = np.zeros((num_cons, n))
    b_ub = np.zeros(num_cons)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0; A_ub[k, j] = 1.0; b_ub[k] = dists[i, j]; k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return np.full(n, 1e-5), 0.0, np.zeros((n, 2))
        
    radii = res.x
    obj_val = -res.fun
    
    # Extract dual variables (marginals)
    duals_obj = getattr(res, 'ineqlin', None)
    if duals_obj is not None:
        duals = getattr(duals_obj, 'marginals', None)
    if duals is None:
        duals = np.zeros(num_cons)
    else:
        duals = np.asarray(duals)
        
    # Compute gradient w.r.t centers using duals
    grad_centers = np.zeros((n, 2))
    
    # Boundary duals contribute to gradient
    for i in range(n):
        grad_centers[i, 0] += duals[4*i] - duals[4*i+1]
        grad_centers[i, 1] += duals[4*i+2] - duals[4*i+3]
        
    # Pairwise duals contribute to gradient
    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]
            if lam > 1e-9:
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist > 1e-12:
                    dir_vec = diff / dist
                    grad_centers[i] += lam * dir_vec
                    grad_centers[j] -= lam * dir_vec
            idx += 1
            
    return radii, obj_val, grad_centers

def gradient_ascent(centers_init, rng):
    """Performs dual-gradient ascent on centers to maximize sum of radii."""
    centers = centers_init.copy()
    best_sum = -1.0
    best_r = None
    
    step = 0.06
    for it in range(250):
        radii, curr_sum, grad = solve_lp_and_duals(centers)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_r = radii.copy()
            
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-6:
            break
            
        # Update centers along gradient direction
        centers += step * grad
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
        
        # Occasional perturbation to escape local minima
        if rng.random() < 0.04:
            idx = rng.integers(N)
            centers[idx] += rng.uniform(-0.04, 0.04, 2)
            centers[idx] = np.clip(centers[idx], 1e-4, 1.0 - 1e-4)
            
        step *= 0.993
        
    return centers, best_r, best_sum

def generate_inits(rng):
    """Generates diverse initial configurations."""
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5]]
    
    # Hexagonal lattices with rotations/scales
    for _ in range(25):
        pat = rng.choice(pats)
        pts = []
        r_est = 0.095
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = (r_idx % 2) * r_est
            x = r_est + shift
            for _ in range(cnt):
                if len(pts) < N: pts.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3.0)
        pts = np.array(pts[:N])
        pts -= 0.5
        scale = rng.uniform(0.85, 1.15)
        pts *= scale
        pts += 0.5
        rot = rng.uniform(-0.25, 0.25)
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts += rng.uniform(-0.015, 0.015, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # Random placements
    for _ in range(10):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # Corner/Edge focused
    for _ in range(8):
        pts = []
        pts.extend([[0.1,0.1], [0.9,0.1], [0.1,0.9], [0.9,0.9]])
        for v in np.linspace(0.3, 0.7, 4):
            pts.extend([[v, 0.1], [v, 0.9], [0.1, v], [0.9, v]])
        for x in np.linspace(0.25, 0.75, 3):
            for y in np.linspace(0.25, 0.75, 3):
                if len(pts) < N: pts.append([x, y])
        pts = np.array(pts[:N]) + rng.uniform(-0.025, 0.025, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    return inits

def objective_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def constraint_func_slqp(v):
    """Pairwise non-overlap constraints. Boundary satisfied by parameterization."""
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def centers_to_params(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    v = np.empty(3 * N)
    v[0::3] = r
    v[1::3] = u
    v[2::3] = w
    return v

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraint_func_slqp}
    
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    inits = generate_inits(rng)
    
    # Phase 1: Dual-Gradient Ascent from diverse starts
    for pts in inits:
        c, r, s = gradient_ascent(pts, rng)
        if s > best_sum:
            best_sum = s
            best_c = c
            best_r = r
            
    # Phase 2: High-Precision SLSQP Polish on best configuration
    if best_c is not None:
        v0 = centers_to_params(best_c, np.maximum(best_r * 0.99, 1e-5))
        try:
            res = minimize(objective_slqp, v0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res.success and np.min(constraint_func_slqp(res.x)) >= -1e-9:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    r_opt = res.x[0::3]
                    u_opt = res.x[1::3]
                    w_opt = res.x[2::3]
                    best_c = np.column_stack((r_opt + u_opt * (1.0 - 2.0 * r_opt), 
                                              r_opt + w_opt * (1.0 - 2.0 * r_opt)))
                    best_r = r_opt
        except Exception:
            pass
            
    # Final LP consistency check
    if best_c is not None:
        r_final, s_final, _ = solve_lp_and_duals(best_c)
        best_r = r_final
        best_sum = s_final
        
    # Fallback (highly unlikely to trigger)
    if best_c is None:
        best_c = np.random.rand(N, 2)
        best_r = np.full(N, 0.05)
        best_sum = np.sum(best_r)
        
    return best_c, best_r, float(best_sum)
