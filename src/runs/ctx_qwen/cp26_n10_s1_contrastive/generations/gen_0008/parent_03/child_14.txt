import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    x_c = centers[:, 0]
    y_c = centers[:, 1]
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        A_ub[k, i] = 1.0; b_ub[k] = x_c[i]; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x_c[i]; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y_c[i]; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y_c[i]; k += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def objective_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def constraint_func_slqp(v):
    """
    Inequality constraints: dist_sq >= (r_i + r_j)^2.
    Boundary constraints are satisfied by the (r, u, w) parameterization.
    """
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def centers_to_params(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    v = np.empty(3 * N)
    v[0::3] = r
    v[1::3] = u
    v[2::3] = w
    return v

def generate_hex_centers(rng, pat, rot, scale, noise):
    """Generates a hexagonal lattice initialization with specified parameters."""
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(pat):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:N])
    pts -= 0.5
    pts *= scale
    pts += 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        mat = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ mat.T + 0.5
    pts += rng.uniform(-noise, noise, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def run_packing():

    bounds_slqp = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraint_func_slqp}
    
    best_vars = None
    best_sum = -np.inf
    best_c = None
    best_r = None

    for _ in range(35):
        pat = rng.choice(pats)

        pts = generate_hex_centers(rng, pat, rot, sc, 0.015)

        r_lp = np.maximum(r_lp * 0.985, 1e-5)
        v0 = centers_to_params(pts, r_lp)
        try:
            res = minimize(objective_slqp, v0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                cv = constraint_func_slqp(res.x)
