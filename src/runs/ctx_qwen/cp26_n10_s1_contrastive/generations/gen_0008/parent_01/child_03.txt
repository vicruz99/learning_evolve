import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    num_rows = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_rows, n))
    b_ub = np.zeros(num_rows)
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        A_ub[4*i, i] = 1.0
        A_ub[4*i+1, i] = 1.0
        A_ub[4*i+2, i] = 1.0
        A_ub[4*i+3, i] = 1.0
        b_ub[4*i] = max(x, 1e-9)
        b_ub[4*i+1] = max(1.0 - x, 1e-9)
        b_ub[4*i+2] = max(y, 1e-9)
        b_ub[4*i+3] = max(1.0 - y, 1e-9)
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    rows_idx, cols_idx = np.triu_indices(n, k=1)
    
    base_idx = 4 * n
    for k in range(len(rows_idx)):
        i = rows_idx[k]
        j = cols_idx[k]
        A_ub[base_idx + k, i] = 1.0
        A_ub[base_idx + k, j] = 1.0
        b_ub[base_idx + k] = dists[i, j]
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def objective_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    return dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX]

def centers_to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    params = np.empty(3 * N)
    params[:N] = r
    params[N:2*N] = u
    params[2*N:3*N] = v
    return params

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def generate_hex_centers(seed, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_layout_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for step in range(400):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 25.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 25.0 * np.maximum(0, pts[:, d] - 0.95)
            
        step_size = 0.007 * (0.995 ** step)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_sa(centers, n_iter=15000, rng=None):
    """Simulated Annealing on centers maximizing LP sum of radii."""
    if rng is None:
        rng = np.random.default_rng()
    curr_c = centers.copy()
    curr_r, curr_sum = solve_lp_radii(curr_c)
    best_c = curr_c.copy()
    best_r = curr_r.copy()
    best_sum = curr_sum
    
    temp = 0.045
    step = 0.028
    
    for it in range(n_iter):
        # Occasional global shake to escape deep local minima
        if rng.random() < 0.04:
            curr_c += rng.normal(0, step * 0.4, (N, 2))
            curr_c = np.clip(curr_c, 0.01, 0.99)
            is_shake = True
        else:
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            is_shake = False
            
        new_r, new_sum = solve_lp_radii(curr_c)
        delta = new_sum - curr_sum
        
        # Metropolis criterion
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = curr_c.copy()
                best_r = new_r.copy()
        else:
            if not is_shake:
                curr_c[idx] = old
                
        temp *= 0.9993
        step = max(0.0012, step * 0.9994)
        
    return best_c, best_r, best_sum
