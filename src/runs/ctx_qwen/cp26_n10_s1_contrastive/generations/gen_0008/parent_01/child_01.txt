import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_cons = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_cons, n))
    b_ub = np.zeros(num_cons)
    k = 0
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    ii, jj = np.triu_indices(n, k=1)
    A_ub[k:, ii] = 1.0
    A_ub[k:, jj] = 1.0
    b_ub[k:] = dists[ii, jj]
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 0.0

def objective_slsqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints_slsqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def centers_to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    p = np.empty(3 * N)
    p[:N] = r
    p[N:2*N] = u
    p[2*N:3*N] = v
    return p

def generate_hex(rng, seed, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.1, 0.9, (N, 2))
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
        step_size = 0.006 * (0.995 ** step)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_sa(centers, rng):
    """Simulated Annealing on centers, optimizing LP sum of radii."""
    n = centers.shape[0]
    best_c = centers.copy()
    best_r, best_sum = solve_lp(best_c)
    curr_c = best_c.copy()
    curr_sum = best_sum
    temp = 0.05
    step = 0.03
    for it in range(6000):
        idx = rng.integers(n)
        old = curr_c[idx].copy()
        if rng.random() < 0.04:
            curr_c[idx] = rng.uniform(0.01, 0.99, 2)
        else:
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        new_r, new_sum = solve_lp(curr_c)
        delta = new_sum - curr_sum
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = curr_c.copy()
                best_r = new_r.copy()
        else:
            curr_c[idx] = old
        temp *= 0.9992
        step = max(0.0015, step * 0.9995)
    return best_c, best_r, best_sum

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints_slsqp}
    
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [7,5,5,5,4], [4,5,5,5,7],
        [6,6,5,5,4], [5,5,6,5,5], [7,6,5,4,4], [4,6,6,6,4]
    ]
    for pat in patterns:
        for s in range(3):
            rot = rng.uniform(-0.25, 0.25)
            sc = rng.uniform(0.85, 1.15)
            inits.append(generate_hex(rng, s, pat, rot, sc))
            
    for s in range(12):
        inits.append(force_init(rng, s))
        
    # Evaluate inits with LP
    init_scores = []
    for pts in inits:
        _, s_val = solve_lp(pts)
        init_scores.append((s_val, pts))
        
    # Sort descending without lambda
    scores = [s for s, _ in init_scores]
    order = np.argsort(scores)[::-1]
    init_scores = [init_scores[i] for i in order]
    
    best_sum = -np.inf
    best_params = None
    
    # Run SA on top 5 inits
    for i, (s_val, pts) in enumerate(init_scores[:5]):
        c_sa, r_sa, s_sa = run_sa(pts, rng)
        s_sa = max(s_sa, s_val)
        
        # SLSQP refinement
        p0 = centers_to_params(c_sa, np.clip(r_sa * 0.99, 1e-6, 0.49))
        try:
            res = minimize(objective_slsqp, p0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                cv = constraints_slsqp(res.x)
                if np.min(cv) >= -1e-7:
                    s_slsqp = -res.fun
                    if s_slsqp > best_sum:
                        best_sum = s_slsqp
                        best_params = res.x.copy()
        except Exception:
            pass
            
        if s_sa > best_sum:
            best_sum = s_sa
            best_params = centers_to_params(c_sa, np.clip(r_sa * 0.99, 1e-6, 0.49))
            
    # Additional perturbation & polish from best found so far
    if best_params is not None:
        for _ in range(15):
            p0 = best_params.copy()
            p0[:N] += rng.normal(0, 0.002, N)
            p0[N:3*N] += rng.normal(0, 0.01, 2*N)
            p0[:N] = np.clip(p0[:N], 1e-6, 0.49)
            p0[N:3*N] = np.clip(p0[N:3*N], 0.0, 1.0)

                best_sum = -res_f.fun
        except Exception:
            pass

    # Fallback configuration
    if best_params is None:
        pts = inits[0]
        r_f, _ = solve_lp(pts)
        best_params = centers_to_params(pts, r_f)
        best_sum = np.sum(r_f)
        
    # Reconstruct final centers and radii
    r_opt = best_params[:N]
    u_opt = best_params[N:2*N]
    v_opt = best_params[2*N:3*N]
    centers = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
