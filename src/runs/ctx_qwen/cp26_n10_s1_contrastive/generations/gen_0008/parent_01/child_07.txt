import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[k, i] = 1.0
            b_ub[k] = max(b, 0.0)
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds_r = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def objective_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def centers_to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[:N] = r
    params[N:2*N] = u
    params[2*N:3*N] = v
    return params

def generate_hex_centers(seed, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_layout_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
            
        step_size = 0.006 * (0.995 ** step)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_sa(centers_init, rng, steps=10000):
    """LP-driven Simulated Annealing on centers."""
    c = centers_init.copy()
    r, s = solve_lp_radii(c)
    best_c, best_r, best_s = c.copy(), r.copy(), s
    temp = 0.04
    step = 0.025
    
    for it in range(steps):
        mt = rng.random()
        if mt < 0.6:
            idx = rng.integers(N)
            nc = c.copy()
            nc[idx] += rng.normal(0, step, 2)
            nc[idx] = np.clip(nc[idx], 0.01, 0.99)
        elif mt < 0.85:
            i, j = rng.choice(N, 2, replace=False)
            nc = c.copy()
            nc[[i, j]] = nc[[j, i]]
        else:
            rot = rng.uniform(-0.04, 0.04)
            cr, sr = np.cos(rot), np.sin(rot)
            M = np.array([[cr, -sr], [sr, cr]])
            nc = (c - 0.5) @ M.T + 0.5
            nc = np.clip(nc, 0.01, 0.99)
            
        nr, ns = solve_lp_radii(nc)
        delta = ns - s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(min(0.0, delta / temp))):
            c, r, s = nc, nr, ns
            if s > best_s:
                best_c, best_r, best_s = c.copy(), r.copy(), s
                
        temp *= 0.9995
        step = max(0.001, step * 0.9997)
        
    return best_c, best_r, best_s

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints_slqp}
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [7,5,5,5,4], [4,5,5,5,7]
    ]
    
    starts = []
    for pat in patterns:
        for s in range(3):
            rot = rng.uniform(-0.25, 0.25)
            sc = rng.uniform(0.88, 1.12)
            starts.append(generate_hex_centers(s, pat, rot, sc))
            
    for s in range(8):
        starts.append(force_layout_init(s))
        
    candidates = []
    for pts in starts:
        bc, br, bs = run_sa(pts, rng, steps=8000)
        candidates.append((bc, br, bs))
        
    candidates.sort(key=lambda x: x[2], reverse=True)
    top_k = candidates[:6]
    
    best_params = None
    best_sum = -np.inf
    
    for bc, br, bs in top_k:
        r_init = np.clip(br * 0.985, 1e-6, 0.49)
        x0 = centers_to_params(bc, r_init)
        
        try:
            res = minimize(objective_slqp, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints_slqp(res.x)) >= -1e-7:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_params = res.x.copy()
        except Exception:
            continue
            
    if best_params is None:
        bc, br, bs = candidates[0]
        best_params = centers_to_params(bc, np.clip(br * 0.98, 1e-6, 0.49))
        best_sum = bs
        
    # Perturbation refinements to escape local minima
    for k in range(12):
        x0 = best_params.copy()
        x0[:N] += rng.uniform(-0.001, 0.001, N)
        x0[N:3*N] += rng.uniform(-0.012, 0.012, 2*N)
        x0[:N] = np.clip(x0[:N], 1e-6, 0.49)
        x0[N:3*N] = np.clip(x0[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(objective_slqp, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints_slqp(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            pass
            
    # Final high-precision polish
    x0_f = best_params.copy()
    x0_f[:N] *= 0.99995
    try:
        res_f = minimize(objective_slqp, x0_f, method='SLSQP', bounds=bounds_slqp,
                         constraints=cons_slqp, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(constraints_slqp(res_f.x)) >= -1e-8:
            best_params = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    r_opt = best_params[:N]
    u_opt = best_params[N:2*N]
    v_opt = best_params[2*N:3*N]
    centers = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
