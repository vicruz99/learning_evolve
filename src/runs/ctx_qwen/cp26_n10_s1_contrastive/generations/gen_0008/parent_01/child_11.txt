import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(max(b, 0.0))
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i,j])
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)]*n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def constraints_full(params):
    """Explicit inequality constraints for SLSQP on (x, y, r) variables."""
    n = len(params) // 3
    x = params[0::3]
    y = params[1::3]
    r = params[2::3]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    c.append(dist_sq[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def obj_full(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[2::3])

def hex_init(seed, pat, rot, scale):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pat:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
        step_size = 0.006 * (0.995 ** step)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_sa(centers, rng, iters=6000, temp_init=0.05, step_init=0.04):
    """LP-Driven Simulated Annealing on circle centers."""
    curr_c = centers.copy()
    _, curr_sum = solve_lp(curr_c)
    best_c = curr_c.copy()
    best_sum = curr_sum
    temp = temp_init
    step = step_init
    for it in range(iters):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.normal(0, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        _, new_sum = solve_lp(curr_c)
        delta = new_sum - curr_sum
        if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_sum = new_sum
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_c = curr_c.copy()
        else:
            curr_c[idx] = old
        temp *= 0.9995
        step = max(0.0015, step * 0.9996)
    return best_c, best_sum

def run_packing():

        x0[2::3] = r_init
