import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = max(b, 0.0)
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds_r = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 1e-4

def objective_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - (r_sum**2)[I_IDX, J_IDX]

def centers_to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    params = np.empty(3 * N)
    params[:N] = r
    params[N:2*N] = u
    params[2*N:3*N] = v
    return params

def params_to_centers(params):
    """Reconstruct physical centers from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y))

def generate_hex(seed, counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
        
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints_slqp}
    
    best_sum = -np.inf
    best_params = None
    
    # Phase 1: Diverse SLSQP starts from hexagonal lattices
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [7,5,5,5,4], [6,6,5,5,4], [5,5,5,6,5], [6,5,5,6,4]
    ]
    starts = []
    for pat in patterns:
        for s in range(8):
            rot = rng.uniform(-0.4, 0.4)
            sc = rng.uniform(0.80, 1.20)
            starts.append(generate_hex(s, pat, rot, sc))
            
    for pts in starts:
        r_init, _ = solve_lp_radii(pts)
        r_init = np.clip(r_init * 0.98, 1e-6, 0.49)
        x0 = centers_to_params(pts, r_init)
        try:
            res = minimize(objective_slqp, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(constraints_slqp(res.x)) >= -1e-6:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            continue
            
    if best_params is None:
        pts = starts[0]
        r_f, _ = solve_lp_radii(pts)
        best_params = centers_to_params(pts, r_f)
        best_sum = np.sum(r_f)
        
    # Phase 2: Perturbation Local Search to escape local minima
    curr_params = best_params.copy()
    curr_sum = best_sum
    
    for iter_num in range(150):
        n_perturb = rng.integers(1, 4)
        idxs = rng.choice(N, n_perturb, replace=False)
        
        x0_p = curr_params.copy()
        # Perturb u, v (positions) significantly
        x0_p[N + idxs] += rng.normal(0, 0.06, n_perturb)
        x0_p[2*N + idxs] += rng.normal(0, 0.06, n_perturb)
        x0_p[N:3*N] = np.clip(x0_p[N:3*N], 0.0, 1.0)
        
        try:
            res_p = minimize(objective_slqp, x0_p, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res_p.success and np.min(constraints_slqp(res_p.x)) >= -1e-6:
                s_p = -res_p.fun
                if s_p > curr_sum:
                    curr_sum = s_p
                    curr_params = res_p.x.copy()
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_params = curr_params.copy()
        except Exception:
            continue
            
    # Phase 3: LP-driven Simulated Annealing on centers
    c_best = params_to_centers(best_params)
    r_best, lp_sum = solve_lp_radii(c_best)
    best_sum = max(best_sum, lp_sum)
    
    curr_c = c_best.copy()
    curr_r, curr_sum = r_best, best_sum
    temp = 0.06
    step = 0.035
    
    for it in range(5000):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        # Occasional large random jump to break rigid clusters
        if rng.random() < 0.06:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.02, 0.98)
            
        new_r, new_sum = solve_lp_radii(curr_c)
        delta = new_sum - curr_sum
        
        # Metropolis acceptance criterion
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_r = new_r
            curr_sum = new_sum
            if curr_sum > best_sum:
                best_sum = curr_sum
                c_best = curr_c.copy()
                r_best = curr_r.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9991
        step = max(0.0015, step * 0.9994)
        
    best_params = centers_to_params(c_best, np.clip(r_best * 0.995, 1e-6, 0.49))
    
    # Phase 4: High-precision SLSQP polish
    for _ in range(6):
        x0_polish = best_params.copy()
        x0_polish[:N] *= 0.9999
        try:
            res_f = minimize(objective_slqp, x0_polish, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons_slqp, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(constraints_slqp(res_f.x)) >= -1e-8:
                best_params = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Reconstruct final centers and radii
    r_opt = best_params[:N]
    u_opt = best_params[N:2*N]
    v_opt = best_params[2*N:3*N]
    centers = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
