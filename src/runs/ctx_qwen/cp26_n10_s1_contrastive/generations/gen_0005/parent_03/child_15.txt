import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 2.6e-4

def joint_obj(vars_vec):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_vec[2::3])

def joint_cons(vars_vec):
    """Inequality constraints: g(vars_vec) >= 0."""
    x = vars_vec[0::3]
    y = vars_vec[1::3]
    r = vars_vec[2::3]
    
    c = []
    # Boundary containment
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    c.append(dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def gen_hex_init(seed, rot, scale):
    """Generates a hexagonal lattice initialization with rotation and scaling."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.1
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def force_init(seed, steps=400):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    r_curr = np.full(N, 0.05)
    
    for step in range(steps):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        repulsion_mag = 1.0 / dists**2
        np.fill_diagonal(repulsion_mag, 0.0)
        forces = np.sum(repulsion_mag[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 5.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 5.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.01 * (0.995**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            d_pair = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            r_curr[i] = 0.85 * min(d_wall, max(d_pair, 1e-6)/2.0)
            
    return pts

def run_packing():
    np.random.seed(42)
    best_sum = -np.inf
    best_vars = None
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': joint_cons}
    
    inits = []
    # 1. Diverse hexagonal lattices with random rotations/scales
    for s in range(25):
        rot = np.random.uniform(-0.4, 0.4)
        scale = np.random.uniform(0.85, 1.15)
        inits.append(gen_hex_init(s, rot, scale))
        
    # 2. Force-directed layouts for organic spreading
    for s in range(15):
        inits.append(force_init(s))
        
    # Phase 1: Broad search from diverse initializations
    for pts in inits:
        radii, _ = solve_radii_lp(pts)
        # Shrink slightly to ensure strict interior feasibility for SLSQP
        radii = np.maximum(radii * 0.998, 1e-6)
        
        v0 = np.zeros(3 * N)
        v0[0::3] = pts[:, 0]
        v0[1::3] = pts[:, 1]
        v0[2::3] = radii
        
        try:
            res = minimize(joint_obj, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                if np.min(joint_cons(res.x)) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Local perturbation refinement to escape local minima
    if best_vars is not None:
        for _ in range(40):
            vp = best_vars.copy()
            vp += np.random.randn(3 * N) * 0.003
            vp[0::3] = np.clip(vp[0::3], 0.01, 0.99)
            vp[1::3] = np.clip(vp[1::3], 0.01, 0.99)
            vp[2::3] = np.clip(vp[2::3], 1e-6, 0.49)
            
            try:
                res_p = minimize(joint_obj, vp, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 2500, 'ftol': 1e-12, 'disp': False})
                if res_p.success:

            except Exception:
                continue
                
        # Phase 3: High-precision polish
        try:
            res_f = minimize(joint_obj, best_vars, method='SLSQP', bounds=bounds,

            if res_f.success and np.min(joint_cons(res_f.x)) >= -1e-8:
                best_vars = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Fallback (should rarely be reached)
    if best_vars is None:
        pts = gen_hex_init(0, 0.0, 1.0)
        radii, _ = solve_radii_lp(pts)
        best_vars = np.zeros(3 * N)
        best_vars[0::3] = pts[:, 0]
        best_vars[1::3] = pts[:, 1]
        best_vars[2::3] = np.maximum(radii, 1e-6)
        best_sum = np.sum(best_vars[2::3])
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
