import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def compute_constraints(v):
    """Inequality constraints: g(v) >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary containment
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2::3])

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 0.0

def generate_hex_init(seed, rot, scale):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.09
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    # Simple repulsion simulation
    for step in range(400):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                d = dists[i, j]
                if d < 1e-5: d = 1e-5
                f = 1.0 / (d**2)
                forces[i] += (diff[i, j] / d) * f
                forces[j] -= (diff[i, j] / d) * f
        step_size = 0.008 * (0.998 ** step)
        pts += step_size * forces
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    np.random.seed(42)
    best_v = None
    best_sum = -np.inf
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    inits = []
    
    # Phase 1: Diverse initializations
    # Hex lattice variations
    for s in range(35):
        rot = np.random.uniform(-0.35, 0.35)
        scale = np.random.uniform(0.82, 1.18)
        c_init = generate_hex_init(s, rot, scale)
        r_init, _ = solve_lp_radii(c_init)
        r_init = np.maximum(r_init * 0.992, 1e-5)
        v0 = np.zeros(3 * N)
        v0[0::3] = c_init[:, 0]
        v0[1::3] = c_init[:, 1]
        v0[2::3] = r_init
        inits.append(v0)
        
    # Force-directed variations
    for s in range(25):
        c_init = generate_force_init(s)
        r_init, _ = solve_lp_radii(c_init)
        r_init = np.maximum(r_init * 0.992, 1e-5)
        v0 = np.zeros(3 * N)
        v0[0::3] = c_init[:, 0]
        v0[1::3] = c_init[:, 1]
        v0[2::3] = r_init
        inits.append(v0)

    # Phase 2: Broad search from feasible starts
    for v0 in inits:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                if np.min(compute_constraints(res.x)) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_v = res.x.copy()
        except Exception:
            continue
            
    # Phase 3: Perturbation refinement with LP projection
    if best_v is not None:
        for k in range(50):
            v_p = best_v.copy()
            # Jitter positions
            v_p[0::3] += np.random.randn(N) * 0.004
            v_p[1::3] += np.random.randn(N) * 0.004
            v_p[0::3] = np.clip(v_p[0::3], 0.015, 0.985)
            v_p[1::3] = np.clip(v_p[1::3], 0.015, 0.985)
            
            # Project to feasible radii via LP to ensure strict feasibility
            c_p = np.column_stack((v_p[0::3], v_p[1::3]))
            r_lp, _ = solve_lp_radii(c_p)
            v_p[2::3] = np.maximum(r_lp * 0.995, 1e-6)
            
            try:
                res_p = minimize(objective, v_p, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 2500, 'ftol': 1e-13, 'disp': False})
                if res_p.success:
                    if np.min(compute_constraints(res_p.x)) >= -1e-7:
                        s_val = -res_p.fun
                        if s_val > best_sum:
                            best_sum = s_val
                            best_v = res_p.x.copy()
            except Exception:
                continue
                
        # Phase 4: High-precision final polish
        try:
            res_f = minimize(objective, best_v, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(compute_constraints(res_f.x)) >= -1e-8:
                best_v = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Fallback (should rarely be triggered)
    if best_v is None:
        fallback_c = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                      np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        fallback_r, _ = solve_lp_radii(fallback_c)
        best_v = np.zeros(3 * N)
        best_v[0::3] = fallback_c[:, 0]
        best_v[1::3] = fallback_c[:, 1]
        best_v[2::3] = np.maximum(fallback_r, 1e-5)
        best_sum = np.sum(best_v[2::3])
        
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3]
    return centers, radii, float(best_sum)
