The constraints are $r_i \le \text{dist}(C_i, \text{boundary})$ and $r_i + r_j \le \text{dist}(C_i, C_j)$. Solving this LP exactly for given centers is extremely fast and guarantees a feasible, optimal radius assignment.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

# Precompute LP constraint matrix structure for efficiency
A_BOUND = np.zeros((4 * N, N))
for i in range(N):
    base = i * 4
    A_BOUND[base, i] = 1.0
    A_BOUND[base + 1, i] = 1.0
    A_BOUND[base + 2, i] = 1.0
    A_BOUND[base + 3, i] = 1.0

A_PAIR = np.zeros((len(I_IDX), N))
A_PAIR[np.arange(len(I_IDX)), I_IDX] = 1.0
A_PAIR[np.arange(len(I_IDX)), J_IDX] = 1.0

A_LP = np.vstack([A_BOUND, A_PAIR])
C_LP = -np.ones(N)
LP_BOUNDS = [(0.0, None)] * N

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    b_bound = np.column_stack([x, 1.0 - x, y, 1.0 - y]).flatten()
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_pair = dists[I_IDX, J_IDX]
    
    b_ub = np.concatenate([b_bound, b_pair])
    
    try:
        res = linprog(C_LP, A_ub=A_LP, b_ub=b_ub, bounds=LP_BOUNDS, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(N)

def force_layout(seed, steps=200):
    """Generates an initial configuration using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    for step in range(steps):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-10)
        repulsion = 1.0 / (dists**2)
        np.fill_diagonal(repulsion, 0.0)
        forces = np.sum(repulsion[:, :, None] * diff / dists[:, :, None], axis=1)
        
        margin = 0.05
        forces[:, 0] += 10.0 * np.maximum(0, margin - pts[:, 0])
        forces[:, 0] -= 10.0 * np.maximum(0, pts[:, 0] - (1.0 - margin))
        forces[:, 1] += 10.0 * np.maximum(0, margin - pts[:, 1])
        forces[:, 1] -= 10.0 * np.maximum(0, pts[:, 1] - (1.0 - margin))
        
        step_size = 0.015 * (0.99 ** step)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def hex_layout(seed):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def objective_centers(centers_flat):
    """Objective function for center optimization: negative max sum of radii."""
    centers = centers_flat.reshape(-1, 2)
    centers = np.clip(centers, 0.001, 0.999)
    sum_r, _ = solve_radii_lp(centers)
    return -sum_r

def optimize_centers(centers_init):
    """Optimizes center positions to maximize sum of radii."""
    res = minimize(objective_centers, centers_init.flatten(), method='Nelder-Mead',
                   options={'maxiter': 2000, 'xatol': 1e-7, 'fatol': 1e-10})
    best_centers = res.x.reshape(-1, 2)
    best_sum, best_radii = solve_radii_lp(best_centers)
    return best_centers, best_sum, best_radii

def joint_objective(v):
    """Objective for joint polishing: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def joint_constraints(v):
    """Constraints for joint polishing: boundary and non-overlap."""
    cx, cy, r = v[0::3], v[1::3], v[2::3]
    c = []
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def run_packing():
    """Main function to solve the circle packing problem."""
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    inits = []
    for s in range(12):
        inits.append(force_layout(s))
    for s in range(8):
        inits.append(hex_layout(s))
        
    # Phase 1: Global search via center optimization with LP radii solver
    for init_centers in inits:
        curr_centers, curr_sum, curr_radii = optimize_centers(init_centers)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = curr_centers.copy()
            best_radii = curr_radii.copy()
            
    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = np.random.rand(N, 2)
        _, best_radii = solve_radii_lp(best_centers)
        best_sum = np.sum(best_radii)
        
    # Phase 2: High-precision joint polishing with SLSQP
    v0 = np.zeros(3 * N)
    v0[0::3] = best_centers[:, 0]
    v0[1::3] = best_centers[:, 1]
    v0[2::3] = best_radii
    # Ensure strict interior feasibility for SLSQP start
    v0[2::3] *= 0.995
    v0[2::3] = np.maximum(v0[2::3], 1e-6)
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': joint_constraints}
    
    try:
        res = minimize(joint_objective, v0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
        if res.success:
            if np.min(joint_constraints(res.x)) >= -1e-7:
                best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                best_radii = res.x[2::3]
                best_sum = -res.fun
    except Exception:
        pass
        
    return best_centers, best_radii, float(best_sum)
