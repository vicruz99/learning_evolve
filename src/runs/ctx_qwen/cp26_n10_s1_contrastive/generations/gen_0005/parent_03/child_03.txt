import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)."""
    return -np.sum(v[2::3])

def constraints(v):
    """
    Inequality constraints: g(v) >= 0.
    Enforces boundary containment and pairwise non-overlap.
    """
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def get_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i]-centers[j])**2))
                mx = min(mx, d*0.5)
        # 0.85 factor ensures strict feasibility for SLSQP start
        r[i] = max(1e-4, mx * 0.85)
    return r

def make_hex_init(row_counts, rot, scale, jitter, seed):
    """Generate a hexagonal lattice initialization with specified parameters."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    # Center and scale
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force_init(seed):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    r_curr = np.full(N, 0.04)
    
    for step in range(400):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        repulsion_mag = 1.0 / dists**2
        np.fill_diagonal(repulsion_mag, 0.0)
        forces = np.sum(repulsion_mag[:, :, None] * diff / dists[:, :, None], axis=1)
        
        # Wall repulsion to keep points inside
        for d in range(2):
            forces[:, d] += 10.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 10.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.005 * (0.995**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-5, 1.0 - 1e-5)
        
        # Adaptively update radii during simulation
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            d_pair = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            r_curr[i] = 0.9 * min(d_wall, d_pair/2.0)
            
    return pts

def make_vars(centers):
    """Map physical centers to optimization parameters (x, y, r)."""
    r = get_radii(centers)
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = r
    return v

def run_packing():

    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -np.inf

        [8,5,5,5,3], [6,5,5,5,5], [5,6,6,4,5], [7,5,6,4,4],
        [6,6,5,5,4], [5,5,5,6,5], [4,6,6,5,5], [7,7,4,4,4]
    ]

                if np.min(cv) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_v = res.x.copy()
        except Exception:
            pass

            except Exception:
                pass

            if res.success and np.min(constraints(res.x)) >= -1e-8:
                best_v = res.x
                best_sum = -res.fun
        except Exception:
            pass
            
    # Fallback configuration (should rarely be reached)
    if best_v is None:
        fallback_pts = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N],

        fallback_pts[25] = [0.5, 0.5]
        best_v = make_vars(fallback_pts)

    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3]
    return centers, radii, float(np.sum(radii))
