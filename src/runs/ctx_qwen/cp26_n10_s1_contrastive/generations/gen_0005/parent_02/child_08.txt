import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def obj_centers(x_flat):
    """Objective for center optimization: negative sum of optimal radii."""
    centers = np.clip(x_flat.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    _, s = solve_radii_lp(centers)
    return -s

def constraints_full(v):
    """SLSQP inequality constraints: boundary containment and non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def objective_full(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2::3])

def get_hex_init(seed, rot, scale):
    """Generates a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    counts = [6, 5, 6, 5, 4]
    pts = []
    r_est = 0.1
    y = r_est
    for r_idx, cnt in enumerate(counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += np.random.randn(N, 2) * 0.02
    return np.clip(pts, 0.01, 0.99)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds_xy = [(0.0, 1.0)] * (2 * N)
    best_x_flat = None
    best_sum = -np.inf
    
    # Phase 1: Global exploration of center space
    try:
        res_de = differential_evolution(obj_centers, bounds_xy, popsize=12, maxiter=80, 
                                        tol=1e-5, mutation=(0.5, 1.0), recombination=0.7, 
                                        seed=42, polish=False)
        best_x_flat = res_de.x
        best_sum = -res_de.fun
    except Exception:
        pass
        
    # Phase 2: Local refinement from diverse hexagonal starts
    candidates = [best_x_flat]
    for seed in range(20):
        rot = np.random.uniform(-0.3, 0.3)
        scale = np.random.uniform(0.85, 1.15)
        candidates.append(get_hex_init(seed, rot, scale).flatten())
        
    for cand in candidates:
        if cand is None: continue
        try:
            res_nm = minimize(obj_centers, cand, method='Nelder-Mead', 
                              options={'maxiter': 3000, 'xatol': 1e-6, 'fatol': 1e-7})
            if -res_nm.fun > best_sum:
                best_sum = -res_nm.fun
                best_x_flat = res_nm.x
        except Exception:
            pass
            
    # Initialize joint variables
    centers_best = np.clip(best_x_flat.reshape(N, 2), 0.0, 1.0)
    radii_best, sum_best = solve_radii_lp(centers_best)
    
    v0 = np.zeros(3 * N)
    v0[0::3] = centers_best[:, 0]
    v0[1::3] = centers_best[:, 1]
    v0[2::3] = radii_best * 0.99  # Slight shrink for strict feasibility
    
    bounds_full = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_full}
    
    v_opt = v0.copy()
    
    # Phase 3: High-precision joint optimization
    try:
        res_slsqp = minimize(objective_full, v0, method='SLSQP', bounds=bounds_full,
                             constraints=cons, options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
        if res_slsqp.success:
            if np.min(constraints_full(res_slsqp.x)) >= -1e-7:
                v_opt = res_slsqp.x
                sum_best = -res_slsqp.fun
                centers_best = np.column_stack((v_opt[0::3], v_opt[1::3]))
                radii_best = v_opt[2::3]
    except Exception:
        pass
        
    # Phase 4: Perturbation refinement to escape local minima
    for _ in range(50):
        v_p = v_opt.copy()
        v_p += np.random.randn(3 * N) * 0.004
        v_p[0::3] = np.clip(v_p[0::3], 0.01, 0.99)
        v_p[1::3] = np.clip(v_p[1::3], 0.01, 0.99)
        v_p[2::3] = np.clip(v_p[2::3], 1e-6, 0.49)
        
        try:
            res_p = minimize(objective_full, v_p, method='SLSQP', bounds=bounds_full,
                             constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res_p.success and np.min(constraints_full(res_p.x)) >= -1e-7:
                s_p = -res_p.fun
                if s_p > sum_best:
                    sum_best = s_p
                    v_opt = res_p.x
                    centers_best = np.column_stack((v_opt[0::3], v_opt[1::3]))
                    radii_best = v_opt[2::3]
        except Exception:
            pass
            
    # Phase 5: Final exact radius optimization for fixed optimal centers
    radii_final, sum_final = solve_radii_lp(centers_best)
    if sum_final > sum_best:
        radii_best = radii_final
        sum_best = sum_final
        
    return centers_best, radii_best, float(sum_best)
