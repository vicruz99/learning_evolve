import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def compute_pairwise_dists(centers):
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    return dists

def solve_radii_lp(centers):
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    dists = compute_pairwise_dists(centers)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-4

def constraint_func(vars_vec):
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    return dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX]

def objective_func(vars_vec):
    return -np.sum(vars_vec[0::3])

def make_params(centers, radii):
    r = radii.copy()
    d = 1.0 - 2.0 * r
    d = np.clip(d, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def reconstruct(params):
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def get_safe_radii(centers):
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        d_min = np.inf
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if d < d_min:
                    d_min = d
        r[i] = 0.90 * min(d_wall, d_min * 0.5)
    return np.maximum(r, 1e-6)

def generate_hex_init(row_counts, seed, rotation=0.0, scale=1.0, jitter=0.02):
    np.random.seed(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    
    pts = np.array(pts[:N])
    pts = pts - 0.5
    pts = pts * scale
    pts = pts + 0.5
    
    if abs(rotation) > 1e-10:
        c, s = np.cos(rotation), np.sin(rotation)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = pts @ rot_mat.T
        pts = pts - pts.mean(axis=0) + 0.5
    
    pts += np.random.randn(N, 2) * jitter
    pts = np.clip(pts, 0.02, 0.98)
    
    return pts

def force_init(seed, steps=300):
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    r_curr = np.full(N, 0.05)
    
    for step in range(steps):

        for d in range(2):
            forces[:, d] += 10.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 10.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.003 * (0.997**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-5, 1.0 - 1e-5)
        
        for i in range(N):
            d_wall = min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])
            dists_i = np.linalg.norm(pts[i] - pts, axis=1)
            dists_i[i] = np.inf
            d_pair = np.min(dists_i)
            r_curr[i] = 0.85 * min(d_wall, d_pair / 2.0)
            
    r_final = np.zeros(N)
    for i in range(N):
        d_wall = min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])

        dists_i[i] = np.inf
        d_pair = np.min(dists_i)
        r_final[i] = 0.95 * min(d_wall, d_pair / 2.0)

def lp_hill_climbing(centers, radii, best_sum, rng, n_iterations=5000, initial_step=0.025):
    """LP-based hill climbing: perturb centers and recompute optimal radii via LP."""
    current_sum = np.sum(radii)
    step_size = initial_step
    
    for iteration in range(n_iterations):
        idx = rng.randint(N)
        old_pos = centers[idx].copy()
        
        centers[idx] += rng.normal(0, step_size, 2)
        centers[idx] = np.clip(centers[idx], 0.01, 0.99)
        
        new_radii, new_sum = solve_radii_lp(centers)
        
        if new_sum > current_sum:
            current_sum = new_sum
            radii = new_radii.copy()
        else:
            centers[idx] = old_pos
        
        if iteration % 500 == 0 and iteration > 0:
            current_sum_lp, _ = solve_radii_lp(centers)
            if current_sum_lp <= best_sum + 0.0001:
                centers[idx] = old_pos
                break
        
        step_size = max(0.0005, step_size * 0.9985)
    
    final_radii, final_sum = solve_radii_lp(centers)
    return centers, final_radii, final_sum
