import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def compute_lp_sum(centers):
    """
    Solves the LP to find radii that maximize sum(r_i) for fixed centers.
    Returns the maximum sum of radii.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(val)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun
    except Exception:
        pass
        
    return 0.0

def get_lp_radii(centers):
    """
    Solves the LP and returns the optimal radii array.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(val)
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):

            row = np.zeros(n)

        if res.success:
            return res.x
    except Exception:
        pass
        
    return np.full(n, 0.01)

def generate_hex_init(seed, rot, scale):
    """Generates a hexagonal lattice initialization with rotation and scaling."""
    np.random.seed(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
        
    pts += np.random.uniform(-0.012, 0.012, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_init(seed):
    """Generates a force-directed initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    for _ in range(80):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        rep = 1.0 / (dists**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        pts += 0.015 * forces
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None
    rng = np.random.RandomState(42)
    
    # Phase 1: Diverse starts with LP-based hill climbing
    inits = []
    for i in range(25):
        rot = rng.uniform(-0.25, 0.25)
        scl = rng.uniform(0.88, 1.12)
        inits.append(generate_hex_init(i, rot, scl))
        
    for i in range(10):
        inits.append(generate_force_init(i))
        
    for pts in inits:
        curr_centers = pts.copy()
        curr_sum = compute_lp_sum(curr_centers)
        
        step = 0.025
        improved = True
        iter_count = 0
        while improved and iter_count < 300:
            improved = False
            for _ in range(N):
                idx = rng.randint(N)
                old_pos = curr_centers[idx].copy()
                
                curr_centers[idx] += rng.normal(0, step, 2)
                curr_centers[idx] = np.clip(curr_centers[idx], 0.015, 0.985)
                
                new_sum = compute_lp_sum(curr_centers)
                if new_sum > curr_sum + 1e-9:
                    curr_sum = new_sum
                    improved = True
                else:
                    curr_centers[idx] = old_pos
                    
            step *= 0.94
            iter_count += 1
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = curr_centers.copy()
            best_radii = get_lp_radii(best_centers)
            
    # Phase 2: Joint SLSQP polish for high-precision tightening
    if best_centers is not None:
        v = np.zeros(3 * N)
        v[0::3] = best_centers[:, 0]
        v[1::3] = best_centers[:, 1]
        v[2::3] = best_radii * 0.998  # Slight shrink to ensure strict feasibility start
        
        bounds_sl = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
        
        def obj_sl(v):
            return -np.sum(v[2::3])
            
        def cons_sl(v):
            x = v[0::3]
            y = v[1::3]
            r = v[2::3]
            c = []
            c.append(x - r)
            c.append(1.0 - x - r)
            c.append(y - r)
            c.append(1.0 - y - r)
            
            dx = x[:, np.newaxis] - x[np.newaxis, :]
            dy = y[:, np.newaxis] - y[np.newaxis, :]
            d2 = dx**2 + dy**2
            rs = r[:, np.newaxis] + r[np.newaxis, :]
            c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
            return np.concatenate(c)
            
        cons = {'type': 'ineq', 'fun': cons_sl}
        try:
            res = minimize(obj_sl, v, method='SLSQP', bounds=bounds_sl, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-15, 'disp': False})
            if res.success:
                if np.min(cons_sl(res.x)) >= -1e-9:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
        except Exception:
            pass
            
    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        best_centers = np.vstack([best_centers, [0.5, 0.5]])
        best_radii = np.full(N, 0.04)
        best_sum = float(np.sum(best_radii))
        
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
