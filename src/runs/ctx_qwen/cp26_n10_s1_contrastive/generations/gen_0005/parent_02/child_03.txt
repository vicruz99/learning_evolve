import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to maximize sum of radii under non-overlap and boundary constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-6), 1e-5

def lp_objective(centers_flat):
    """Objective for center optimization: maximize LP-derived sum of radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    _, s = solve_radii_lp(centers)
    return -s

def generate_hex_centers(seed, rot, scale):
    """Generates a perturbed hexagonal lattice configuration."""
    np.random.seed(seed)
    rows_counts = [6, 5, 6, 5, 4]
    pts = []
    r_est = 0.1
    y = r_est
    for r_idx, cnt in enumerate(rows_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts = pts - pts.mean(axis=0) + 0.5
        
    pts += np.random.randn(N, 2) * 0.02
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def generate_repelled_random(seed):
    """Generates a force-directed layout with mutual repulsion."""
    np.random.seed(seed)
    pts = np.random.rand(N, 2)
    for _ in range(300):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-6)
        forces = np.zeros_like(pts)
        repulsion = 1.0 / (dists**2)
        np.fill_diagonal(repulsion, 0.0)
        forces += np.sum(repulsion[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        
        for d in range(2):
            forces[:, d] += 10.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 10.0 * np.maximum(0, pts[:, d] - 0.95)
            
        pts += 0.005 * forces
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def compute_full_constraints(v):
    """Computes inequality constraints g(v) >= 0 for SLSQP."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    c.append(dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    return np.concatenate(c)

def full_objective(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2::3])
