import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def constraints(v):
    """Computes inequality constraints g(v) >= 0 for pairwise non-overlap."""
    r = v[0::3]
    u = v[1::3]
    v_param = v[2::3]
    
    # Parameterization ensures r <= x <= 1-r and r <= y <= 1-r automatically
    x = r + u * (1.0 - 2.0 * r)
    y = r + v_param * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def get_safe_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        dists = np.linalg.norm(centers[i] - centers, axis=1)
        dists[i] = np.inf
        d_pair = np.min(dists)
        # Scale down slightly to ensure strict interior feasibility
        r[i] = 0.92 * min(d_wall, d_pair * 0.5)
    return np.maximum(r, 1e-4)

def gen_hex(row_counts, rot=0.0, scale=1.0, seed=0):
    """Generate a hexagonal lattice initialization with specified parameters."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:N])
    
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.randn(N, 2) * 0.025
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    np.random.seed(42)
    bounds = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -np.inf
    best_params = None
    
    # Diverse row counts summing to 26
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5],
        [5,5,5,5,6], [6,4,6,5,5], [4,5,6,5,6],
        [7,5,5,5,4], [4,5,5,5,7], [8,6,6,6], [7,6,6,7],
        [5,7,5,7], [6,6,6,6,2], [7,5,6,6,2], [6,5,5,5,5],
        [9,5,5,5,2], [8,5,6,4,3], [5,8,5,8]
    ]
    
    inits = []
    for pat in patterns:
        for _ in range(4):
            rot = np.random.uniform(-0.3, 0.3)
            scale = np.random.uniform(0.85, 1.15)
            centers = gen_hex(pat, rot=rot, scale=scale, seed=np.random.randint(100))
            r0 = get_safe_radii(centers)
            inits.append(make_params(centers, r0))
            
    # Add random initializations
    for _ in range(20):
        centers = np.random.rand(N, 2)
        centers = np.clip(centers, 0.05, 0.95)
        r0 = get_safe_radii(centers)
        inits.append(make_params(centers, r0))
        
    # Phase 1: Broad search
    for p0 in inits:
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Local perturbation refinement
    if best_params is not None:
        for _ in range(80):
            p0 = best_params.copy()
            p0[0::3] += np.random.uniform(-0.0005, 0.0005, N)
            p0[1::3] += np.random.uniform(-0.015, 0.015, N)
            p0[2::3] += np.random.uniform(-0.015, 0.015, N)
            p0[0::3] = np.clip(p0[0::3], 1e-6, 0.5)
            p0[1::3] = np.clip(p0[1::3], 0.0, 1.0)
            p0[2::3] = np.clip(p0[2::3], 0.0, 1.0)

                    if np.min(c_val) >= -1e-7:
                        s_val = -res.fun
                        if s_val > best_sum:
                            best_sum = s_val
                            best_params = res.x.copy()
            except Exception:
                pass
                
        # Phase 3: High-precision polish
        try:
            res = minimize(objective, best_params, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-8:
                best_params = res.x
                best_sum = -res.fun
        except Exception:
            pass

    # Fallback configuration
    if best_params is None:
        centers_f = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                     np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        radii_f = np.full(N, 0.04)
        return centers_f, radii_f, float(np.sum(radii_f))

    # Reconstruct physical centers from parameters
    r = best_params[0::3]
    u = best_params[1::3]
    v = best_params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    centers = np.column_stack([x, y])
    radii = np.maximum(r, 0.0)
    
    return centers, radii, float(best_sum)
