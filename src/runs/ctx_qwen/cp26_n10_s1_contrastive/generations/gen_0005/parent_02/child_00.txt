import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        x, y = centers[i]
        limits = [x, 1.0 - x, y, 1.0 - y]
        for lim in limits:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(lim)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    dx = centers[:, np.newaxis, 0] - centers[np.newaxis, :, 0]
    dy = centers[:, np.newaxis, 1] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i, j in zip(I_IDX, J_IDX):
        row = np.zeros(n)
        row[i] = 1.0
        row[j] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(dists[i, j])
        
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    r_fallback = np.full(n, 1e-5)
    return r_fallback, 2.6e-4

def lp_obj(centers_flat):
    """Objective for center optimization: negative sum of optimal radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 1e-6, 1.0 - 1e-6)
    _, s = solve_radii_lp(centers)
    return -s

def get_safe_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        d_min = np.inf
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if d < d_min:
                    d_min = d
        r[i] = 0.9 * min(d_wall, d_min * 0.5)
    return np.maximum(r, 1e-6)

def compute_constraints(v):
    """Computes inequality constraints g(v) >= 0 for SLSQP."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def generate_hex_init(seed, rotation=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization with specified parameters."""
    rng = np.random.RandomState(seed)
    rows_counts = [6, 5, 6, 5, 4]
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(rows_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:N])
    
    pts = (pts - 0.5) * scale + 0.5
    if rotation != 0.0:
        c, s = np.cos(rotation), np.sin(rotation)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = pts @ rot_mat.T
        pts = pts - pts.mean(axis=0) + 0.5
        
    pts += rng.randn(N, 2) * 0.01
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # Phase 1: Diverse SLSQP joint optimization
    configs = []
    np.random.seed(42)
    for i in range(60):
        rot = np.random.uniform(-0.25, 0.25)
        scale = np.random.uniform(0.85, 1.15)
        configs.append((i, rot, scale))
        
    for seed, rot, scale in configs:
        centers_init = generate_hex_init(seed, rot, scale)
        radii_init = get_safe_radii(centers_init)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = centers_init[:, 0]
        x0[1::3] = centers_init[:, 1]
        x0[2::3] = radii_init
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                if np.min(compute_constraints(res.x)) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
        except Exception:
            continue

    # Phase 2: Nelder-Mead on centers maximizing LP-derived radius sum
    if best_centers is not None:
        try:
            res_nm = minimize(lp_obj, best_centers.flatten(), method='Nelder-Mead',
                              options={'maxiter': 15000, 'xatol': 1e-6, 'fatol': 1e-8})
            centers_nm = np.clip(res_nm.x.reshape(N, 2), 1e-6, 1.0 - 1e-6)
            radii_nm, sum_nm = solve_radii_lp(centers_nm)
            if sum_nm > best_sum:
                best_sum = sum_nm
                best_centers = centers_nm
                best_radii = radii_nm
        except Exception:
            pass

    # Phase 3: Simulated Annealing on centers with exact LP radius updates
    if best_centers is not None:
        rng = np.random.RandomState(123)
        centers_curr = best_centers.copy()
        radii_curr, curr_sum = solve_radii_lp(centers_curr)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers_curr
            best_radii = radii_curr
            
        temp = 0.04
        for step in range(2500):
            idx = rng.randint(N)
            old_pos = centers_curr[idx].copy()
            
            centers_curr[idx] += rng.normal(0, temp, 2)
            centers_curr[idx] = np.clip(centers_curr[idx], 1e-6, 1.0 - 1e-6)
            
            new_radii, new_sum = solve_radii_lp(centers_curr)
            
            delta = new_sum - curr_sum
            if delta > 0 or rng.rand() < np.exp(delta / max(temp, 1e-12)):
                curr_sum = new_sum
                radii_curr = new_radii
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = centers_curr.copy()
                    best_radii = radii_curr.copy()
            else:
                centers_curr[idx] = old_pos
                
            temp *= 0.998
            
    # Phase 4: Final high-precision SLSQP polish
    if best_centers is not None:
        x_final = np.zeros(3 * N)
        x_final[0::3] = best_centers[:, 0]
        x_final[1::3] = best_centers[:, 1]
        x_final[2::3] = best_radii
        
        try:
            res_final = minimize(objective, x_final, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            if res_final.success:
                if np.min(compute_constraints(res_final.x)) >= -1e-9:
                    final_sum = -res_final.fun
                    if final_sum > best_sum:
                        best_sum = final_sum
                        best_centers = np.column_stack((res_final.x[0::3], res_final.x[1::3]))
                        best_radii = res_final.x[2::3]
        except Exception:
            pass

    # Fallback configuration (should rarely be reached)
    if best_centers is None:
        centers_f = np.zeros((N, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                centers_f[idx] = [0.1 + 0.2 * i, 0.1 + 0.2 * j]
                idx += 1
        centers_f[25] = [0.5, 0.5]
        r_f = np.full(N, 0.05)
        best_centers = centers_f
        best_radii = r_f
        best_sum = np.sum(r_f)

    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
