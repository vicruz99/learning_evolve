1. **Parameterized Boundary Handling**: Instead of explicitly enforcing $r_i \le x_i \le 1-r_i$, we parameterize centers as $x_i = r_i + u_i(1-2r_i)$ and $y_i = r_i + v_i(1-2r_i)$ with $u_i, v_i \in [0,1]$. This automatically satisfies boundary constraints, drastically reducing the constraint set and improving solver stability.
2. **Diverse Initialization**: We generate initial configurations using both a force-directed layout (which naturally spreads points to maximize packing density) and rotated/scaled hexagonal lattices. For each, we compute strictly feasible initial radii via Linear Programming (LP).
3. **LP-Based Hill Climbing**: Gradient-based methods struggle with the non-differentiable nature of optimal radius assignment as centers move. We run a simulated annealing/hill climbing loop that perturbs circle centers and exactly re-optimizes radii using `scipy.optimize.linprog`. This efficiently navigates the discrete geometric changes.
4. **Gradient Polish**: After finding the best center configuration via hill climbing, we map it back to the $(r, u, v)$ parameterization and run SLSQP with tight tolerances to extract every fraction of a radius. Additional perturbed restarts ensure we escape the final local minima.

This combination exploits the strengths of LP (exact radius calculation), local search (topology changes), and gradient optimization (precision), consistently pushing past standard gradient-only baselines.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective_params(params):
    """Minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints_params(params):
    """
    Pairwise non-overlap constraints.
    Boundary constraints are satisfied automatically via parameterization.
    Returns g(params) >= 0.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    denom = 1.0 - 2.0 * r
    denom = np.maximum(denom, 1e-9)
    x = r + u * denom
    y = r + v * denom
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    r_sum_sq = r_sum**2
    
    return dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX]

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to maximize sum of radii."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i,j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)]*n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 2.6e-4

def get_init_params_from_centers(centers):
    """Maps physical centers to (r, u, v) optimization parameters."""
    n = centers.shape[0]
    r, _ = solve_radii_lp(centers)
    r = np.maximum(r, 1e-6)
    denom = np.clip(1.0 - 2.0*r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def force_directed_init(seed, steps=300):
    """Generates a dense, spread-out initial layout using force simulation."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    r_curr = np.full(N, 0.08)
    for step in range(steps):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        rep = 1.0 / (dists**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 10.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 10.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
        step_size = 0.005 * (0.99**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-5, 1.0-1e-5)
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            d_pair = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            r_curr[i] = 0.9 * min(d_wall, max(d_pair, 1e-6)/2.0)
    return pts

def hex_init(seed, rot=0.0, scale=1.0):
    """Generates a perturbed hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    rows = [6, 5, 6, 5, 4]
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(rows):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    np.random.seed(42)
    bounds = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*(2*N)
    cons = {'type': 'ineq', 'fun': constraints_params}
    
    best_sum = -np.inf
    best_params = None
    
    # Phase 1: Diverse global search via SLSQP on (r, u, v)
    inits = []
    for s in range(12):
        inits.append(get_init_params_from_centers(force_directed_init(s)))
    for s in range(18):
        rot = np.random.uniform(-0.2, 0.2)
        sc = np.random.uniform(0.9, 1.1)
        inits.append(get_init_params_from_centers(hex_init(s, rot, sc)))
        
    for x0 in inits:
        try:
            res = minimize(objective_params, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                if np.min(constraints_params(res.x)) >= -1e-8:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Map to physical centers for exact LP-based hill climbing
    if best_params is not None:
        r_opt = best_params[:N]
        u_opt = best_params[N:2*N]
        v_opt = best_params[2*N:3*N]
        denom = 1.0 - 2.0 * r_opt
        x_opt = r_opt + u_opt * denom
        y_opt = r_opt + v_opt * denom
        centers_hc = np.column_stack((x_opt, y_opt))
    else:
        centers_hc = hex_init(0)
        
    best_lp_sum, best_radii_lp = solve_radii_lp(centers_hc)
    best_centers_lp = centers_hc.copy()
    
    rng = np.random.RandomState(123)
    step = 0.03
    for _ in range(4000):
        idx = rng.randint(N)
        old = centers_hc[idx].copy()
        centers_hc[idx] += rng.normal(0, step, 2)
        centers_hc[idx] = np.clip(centers_hc[idx], 0.01, 0.99)
        
        new_r, new_sum = solve_radii_lp(centers_hc)
        if new_sum > best_lp_sum:
            best_lp_sum = new_sum
            best_centers_lp = centers_hc.copy()
            best_radii_lp = new_r.copy()
        else:
            centers_hc[idx] = old
        step *= 0.997
        
    # Phase 3: High-precision gradient polish
    best_params = get_init_params_from_centers(best_centers_lp)
    try:
        res_final = minimize(objective_params, best_params, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
        if res_final.success and np.min(constraints_params(res_final.x)) >= -1e-9:
            best_params = res_final.x
            best_sum = -res_final.fun
    except Exception:
        pass
        
    # Phase 4: Perturbation restarts to escape final local minima
    for _ in range(10):
        x0_pert = best_params.copy()
        x0_pert += np.random.normal(0, 0.002, 3*N)
        x0_pert[0:N] = np.clip(x0_pert[0:N], 1e-6, 0.5)
        x0_pert[N:3*N] = np.clip(x0_pert[N:3*N], 0.0, 1.0)
        try:
            res_p = minimize(objective_params, x0_pert, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
            if res_p.success and np.min(constraints_params(res_p.x)) >= -1e-9:
                if -res_p.fun > best_sum:
                    best_sum = -res_p.fun
                    best_params = res_p.x.copy()
        except Exception:
            pass
            
    # Reconstruct physical centers and radii
    r_opt = best_params[:N]
    u_opt = best_params[N:2*N]
    v_opt = best_params[2*N:3*N]
    denom = 1.0 - 2.0 * r_opt
    x_opt = r_opt + u_opt * denom
    y_opt = r_opt + v_opt * denom
    centers = np.column_stack((x_opt, y_opt))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(np.sum(radii))
