import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective_ruv(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def constraints_ruv(vars_vec):
    """Inequality constraints: pairwise non-overlap. Boundary handled by parameterization."""
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def solve_radii_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= distance to boundaries
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 0.0

def centers_to_ruv(centers):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        d_min = np.inf
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if d < d_min: d_min = d
        r[i] = 0.85 * min(d_wall, d_min * 0.5)
    r = np.maximum(r, 1e-5)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def generate_hex_init(row_counts, seed, rot=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization with specified parameters."""
    np.random.seed(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    center = pts.mean(axis=0)
    pts = pts - center
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = pts @ rot_mat.T
    pts = pts * scale + 0.5
    
    pts += np.random.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    np.random.seed(42)
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    bounds_ruv = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons_ruv = {'type': 'ineq', 'fun': constraints_ruv}
    
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [7, 6, 6, 7], [8, 6, 6, 6], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 5, 6, 5, 6], [5, 7, 5, 7],
        [6, 6, 6, 6, 2], [7, 5, 6, 6, 2], [6, 5, 5, 5, 5]
    ]
    
    inits = []
    for pat in patterns:
        for s in range(4):
            rot = np.random.uniform(-0.2, 0.2)
            scale = np.random.uniform(0.85, 1.15)
            pts = generate_hex_init(pat, seed=s, rot=rot, scale=scale)
            inits.append(centers_to_ruv(pts))
            
    for s in range(10):
        pts = np.random.rand(N, 2)
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(centers_to_ruv(pts))
        
    ruv_best = None
    
    # Phase 1: Broad SLSQP search
    for x0 in inits:
        try:
            res = minimize(objective_ruv, x0, method='SLSQP', bounds=bounds_ruv,
                           constraints=cons_ruv, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = constraints_ruv(res.x)
                if np.min(c_val) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        ruv_best = res.x.copy()
        except Exception:
            continue
            
    if ruv_best is not None:
        r_opt = ruv_best[0::3]
        u_opt = ruv_best[1::3]
        v_opt = ruv_best[2::3]
        x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
        y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
        best_centers = np.column_stack((x_opt, y_opt))
        best_radii = r_opt
        
    # Phase 2: LP-based Hill Climbing on centers
    if best_centers is not None:
        centers_hc = best_centers.copy()
        radii_hc, curr_sum = solve_radii_lp(centers_hc)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers_hc.copy()
            best_radii = radii_hc.copy()

        step_size = 0.025
        for iteration in range(4000):
            idx = rng.randint(N)
            old_pos = centers_hc[idx].copy()

            new_radii, new_sum = solve_radii_lp(centers_hc)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = centers_hc.copy()
                best_radii = new_radii.copy()
            else:
                centers_hc[idx] = old_pos
                
            step_size = max(0.0005, step_size * 0.997)
            
    # Phase 3: High-precision SLSQP polish
    if best_centers is not None:
        x0_final = centers_to_ruv(best_centers)
        try:
            res_final = minimize(objective_ruv, x0_final, method='SLSQP', bounds=bounds_ruv,
                                 constraints=cons_ruv, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if res_final.success:
                c_val = constraints_ruv(res_final.x)
                if np.min(c_val) >= -1e-8:
                    final_sum = -res_final.fun
                    if final_sum > best_sum:
                        best_sum = final_sum
                        r_f = res_final.x[0::3]
                        u_f = res_final.x[1::3]
                        v_f = res_final.x[2::3]
                        x_f = r_f + u_f * (1.0 - 2.0 * r_f)
                        y_f = r_f + v_f * (1.0 - 2.0 * r_f)
                        best_centers = np.column_stack((x_f, y_f))
                        best_radii = r_f
        except Exception:
            pass

    # Fallback configuration
    if best_centers is None:
        centers_f = np.zeros((N, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                centers_f[idx] = [0.1 + 0.2 * i, 0.1 + 0.2 * j]
                idx += 1
        centers_f[25] = [0.5, 0.5]
        r_f, _ = solve_radii_lp(centers_f)
        best_centers = centers_f
        best_radii = r_f
        best_sum = np.sum(r_f)
        
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
