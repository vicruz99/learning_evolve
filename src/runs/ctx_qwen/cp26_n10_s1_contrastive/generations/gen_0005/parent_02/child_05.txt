import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_constraints(v):
    """Computes inequality constraints g(v) >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary containment
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    c.append(dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    
    return np.concatenate(c)

def objective_slsqp(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def solve_radii_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)]*n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def lp_objective(centers_flat):
    """Objective for center optimization: negative sum of LP-optimal radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-6, 1.0-1e-6)
    _, s = solve_radii_lp(centers)
    return -s

def generate_hex_init(seed, rotation=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization with specified parameters."""
    np.random.seed(seed)
    rows_counts = [6, 5, 6, 5, 4]
    pts = []
    r_est = 0.1
    y = r_est
    for r_idx, cnt in enumerate(rows_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:N])
    
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rotation != 0.0:
        c_val, s_val = np.cos(rotation), np.sin(rotation)
        rot = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = pts @ rot.T
        pts -= pts.mean(axis=0) + 0.5
        
    pts += np.random.randn(N, 2) * 0.01
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    bounds_slsqp = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # Phase 1: Diverse SLSQP restarts
    rng_init = np.random.RandomState(42)
    for i in range(60):
        rot = rng_init.uniform(-0.3, 0.3)
        scale = rng_init.uniform(0.85, 1.15)
        centers_init = generate_hex_init(i, rot, scale)
        r_init, _ = solve_radii_lp(centers_init)
        r_init = np.maximum(r_init * 0.98, 1e-5)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = centers_init[:, 0]
        x0[1::3] = centers_init[:, 1]
        x0[2::3] = r_init
        
        try:
            res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds_slsqp,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                if np.min(compute_constraints(res.x)) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
        except Exception:
            pass
            
    # Phase 2: Nelder-Mead on centers using exact LP objective
    if best_centers is not None:
        x0_lm = best_centers.flatten()
        try:
            res_lm = minimize(lp_objective, x0_lm, method='Nelder-Mead', 
                              options={'maxiter': 4000, 'xatol': 1e-8, 'fatol': 1e-9, 'disp': False})
            if res_lm.success:
                curr_centers = np.clip(res_lm.x.reshape(N, 2), 0.001, 0.999)
                r_lm, s_lm = solve_radii_lp(curr_centers)
                if s_lm > best_sum:
                    best_sum = s_lm
                    best_centers = curr_centers
                    best_radii = r_lm
        except Exception:
            pass
            
    # Phase 3: Annealed Hill Climbing on centers
    if best_centers is not None:
        centers_hc = best_centers.copy()
        radii_hc, curr_sum = solve_radii_lp(centers_hc)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_radii = radii_hc
            
        rng_hc = np.random.RandomState(123)
        step = 0.025
        for _ in range(6000):
            idx = rng_hc.randint(N)
            old_pos = centers_hc[idx].copy()
            centers_hc[idx] += rng_hc.normal(0, step, 2)
            centers_hc[idx] = np.clip(centers_hc[idx], 0.005, 0.995)
            
            _, new_sum = solve_radii_lp(centers_hc)
            if new_sum > best_sum + 1e-9:
                best_sum = new_sum
                best_centers = centers_hc.copy()
            else:
                centers_hc[idx] = old_pos
                
            step = max(0.0005, step * 0.995)
            
        radii_hc, best_sum = solve_radii_lp(best_centers)
        best_radii = radii_hc

    # Phase 4: Final high-precision SLSQP polish
    if best_centers is not None:
        x0_final = np.zeros(3 * N)
        x0_final[0::3] = best_centers[:, 0]
        x0_final[1::3] = best_centers[:, 1]
        x0_final[2::3] = best_radii * 0.9995
        
        try:
            res_f = minimize(objective_slsqp, x0_final, method='SLSQP', bounds=bounds_slsqp,
                             constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res_f.success:
                if np.min(compute_constraints(res_f.x)) >= -1e-7:
                    final_sum = -res_f.fun
                    if final_sum > best_sum:
                        best_sum = final_sum
                        best_centers = np.column_stack((res_f.x[0::3], res_f.x[1::3]))
                        best_radii = res_f.x[2::3]
        except Exception:
            pass
            
    # Fallback configuration
    if best_centers is None:
        best_centers = np.tile(np.array([0.5, 0.5]), (N, 1)) + np.random.randn(N, 2)*0.01
        best_radii = np.full(N, 0.01)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
