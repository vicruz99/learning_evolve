import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective_func(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_func(x):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise separation: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 2.6e-4

def generate_hex(row_counts, seed):
    """Generates a hexagonal lattice initialization with jitter."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.098
    y = r_est
    for r_idx, count in enumerate(row_counts):
        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    r_curr = np.full(N, 0.05)
    
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        rep = 1.0 / (dists**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 5.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 5.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        pts += 0.005 * forces
        pts = np.clip(pts, 1e-5, 1.0 - 1e-5)
        
        for i in range(N):
            d_w = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            d_p = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            d_p = max(d_p, 1e-6)
            r_curr[i] = 0.8 * min(d_w, d_p/2.0)
            
    return pts

def hill_climb_centers(centers, steps=1500):
    """Adaptive local search on centers evaluated via LP."""
    best_c = centers.copy()
    _, best_s = solve_lp_radii(best_c)
    step_size = 0.04
    
    for _ in range(steps):
        idx = np.random.randint(0, N)
        dir_vec = np.random.randn(2)
        dir_vec /= np.linalg.norm(dir_vec)
        
        new_c = best_c.copy()
        new_c[idx] = np.clip(best_c[idx] + step_size * dir_vec, 0.01, 0.99)
        
        _, new_s = solve_lp_radii(new_c)
        
        if new_s > best_s + 1e-7:
            best_c = new_c
            best_s = new_s
            step_size = min(step_size * 1.15, 0.1)
        else:
            step_size = max(step_size * 0.92, 0.001)
            
    return best_c, best_s

def run_packing():
    np.random.seed(42)
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_func}
    
    # Diverse hexagonal row patterns
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [4, 6, 5, 6, 5],
        [5, 5, 5, 5, 6], [6, 4, 6, 5, 5], [7, 5, 5, 6, 3],
        [5, 7, 5, 7, 2], [6, 6, 6, 6, 2]
    ]
    
    inits = []
    for pat in patterns:
        for s in range(5):
            inits.append(generate_hex(pat, s))
    for s in range(12):
        inits.append(force_init(s))
        
    # Phase 1: Multi-start local search + SLSQP refinement
    for c_init in inits:
        c_hc, s_hc = hill_climb_centers(c_init, steps=1500)
        
        r_init, _ = solve_lp_radii(c_hc)
        r_init = np.maximum(r_init * 0.995, 1e-6)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = c_hc[:, 0]
        x0[1::3] = c_hc[:, 1]
        x0[2::3] = r_init
        
        try:
            res = minimize(objective_func, x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-13})
            if res.success and np.min(constraints_func(res.x)) >= -1e-7:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_r = res.x[2::3]
        except Exception:
            pass
            
        if s_hc > best_sum:
            best_sum = s_hc
            best_c = c_hc
            best_r, _ = solve_lp_radii(c_hc)
            
    # Phase 2: Iterative perturbation to escape local minima
    if best_c is not None:
        for _ in range(4):
            c_p = best_c.copy()
            c_p += np.random.randn(N, 2) * 0.005
            c_p = np.clip(c_p, 0.01, 0.99)
            c_pol, s_pol = hill_climb_centers(c_p, steps=1000)
            if s_pol > best_sum:
                best_sum = s_pol
                best_c = c_pol
                best_r, _ = solve_lp_radii(c_pol)
                
        # Phase 3: High-precision joint polish
        r_p = np.maximum(best_r * 0.999, 1e-6)
        x0_f = np.zeros(3 * N)
        x0_f[0::3] = best_c[:, 0]
        x0_f[1::3] = best_c[:, 1]
        x0_f[2::3] = r_p
        
        try:
            res_f = minimize(objective_func, x0_f, method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 5000, 'ftol': 1e-14})
            if res_f.success and np.min(constraints_func(res_f.x)) >= -1e-8:
                best_c = np.column_stack((res_f.x[0::3], res_f.x[1::3]))
                best_r = res_f.x[2::3]
                best_sum = -res_f.fun
        except Exception:
            pass

    # Fallback (should rarely trigger)
    if best_c is None:
        best_c = np.random.rand(N, 2)
        best_r, best_sum = solve_lp_radii(best_c)
        
    return best_c, best_r, float(best_sum)
