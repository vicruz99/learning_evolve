

<strategy>
To maximize the sum of radii for 26 circles in a unit square, I will employ a coordinate parameterization that automatically satisfies boundary constraints. By letting:
$x_i = r_i + u_i(1 - 2r_i)$
$y_i = r_i + v_i(1 - 2r_i)$
with $u_i, v_i \in [0, 1]$, the circles are guaranteed to stay inside $[0, 1]^2$ regardless of $r_i$. This removes 104 boundary constraints from the optimization problem, allowing the solver to focus entirely on the 325 pairwise non-overlap constraints: $\text{dist}_{ij}^2 \ge (r_i + r_j)^2$.

The optimization pipeline will be:
1. **Diverse Initialization**: Generate hexagonal lattice patterns with various row distributions, random rotations, and scaling factors. Complement these with force-directed layouts that naturally spread points while respecting rough boundary limits.
2. **Parameter Mapping**: Convert all initial $(x, y, r)$ configurations into the $(r, u, v)$ parameter space, ensuring strict initial feasibility by scaling radii down slightly.
3. **Joint SLSQP Optimization**: Run `scipy.optimize.minimize` with SLSQP to maximize $\sum r_i$ subject to pairwise separation. SLSQP handles the smooth nonlinear constraints efficiently.
4. **Local Perturbation Search**: Systematically perturb the best solution in $(r, u, v)$ space to escape shallow local minima, re-optimizing each perturbation.
5. **High-Precision Polish**: Run a final SLSQP pass with tight tolerances to squeeze out marginal gains and ensure numerical validity.

This strategy avoids the common pitfall of solvers getting stuck satisfying boundary constraints, significantly improves convergence speed, and leverages geometric intuition through structured initializations.

import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(vars_vec):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_vec[:N])

def get_constraints(vars_vec):
    """
    Computes inequality constraints g(vars_vec) >= 0.
    Uses parameterization to automatically satisfy boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = vars_vec[:N]
    u = vars_vec[N:2*N]
    v = vars_vec[2*N:3*N]
    
    # Parameterization maps u, v in [0,1] to x, y such that circle is strictly inside [0,1]^2
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    # Pairwise squared distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    # Squared sum of radii
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX]

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = 1.0 - 2.0 * r
    denom = np.clip(denom, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def get_safe_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        dists = np.linalg.norm(centers[i] - centers, axis=1)
        dists[i] = np.inf
        d_pair = np.min(dists)
        r[i] = max(1e-4, min(d_wall, d_pair * 0.5) * 0.85)
    return r

def gen_hex(row_counts, rot, scale, jitter):
    """Generate a hexagonal lattice initialization with specified parameters."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    # Center and scale
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += np.random.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    r_curr = np.full(N, 0.04)
    
    for _ in range(300):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        repulsion = 0.5 / (dists**2)
        np.fill_diagonal(repulsion, 0.0)
        forces = np.sum(repulsion[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        
        # Wall repulsion to keep points inside
        for d in range(2):
            forces[:, d] += 5.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 5.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        pts += 0.003 * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        # Adaptively update radii during simulation
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists_i = np.linalg.norm(pts[i] - pts, axis=1)
            dists_i[i] = np.inf
            d_pair = np.min(dists_i)
            r_curr[i] = 0.9 * min(d_wall, d_pair * 0.5)
            
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    best_params = None
    best_sum = -np.inf
    
    inits = []
    
    # Diverse hexagonal patterns
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5],
        [7,6,6,7], [8,6,6,6], [5,5,5,5,6],
        [6,4,6,5,5], [4,5,6,5,6], [5,7,5,7],
        [6,6,6,6,2], [7,5,6,6,2], [6,5,5,5,5],
        [5,5,6,5,5], [4,4,6,6,6], [6,7,6,7],
        [5, 6, 7, 8], [4, 7, 7, 8], [5, 5, 6, 7, 3],
        [6, 6, 5, 6, 3], [7, 7, 6, 6], [8, 7, 7, 4]
    ]
    
    for pat in patterns:
        if sum(pat) < N:
            continue
        for _ in range(6):
            rot = np.random.uniform(-0.35, 0.35)
            scale = np.random.uniform(0.85, 1.15)
            jitt = np.random.uniform(0.01, 0.04)
            inits.append(gen_hex(pat, rot=rot, scale=scale, jitter=jitt))
            
    # Random initializations with sorting to encourage grid-like structure
    for _ in range(15):
        pts = np.random.rand(N, 2)
        pts[:, 0] = np.sort(pts[:, 0])
        pts += np.random.uniform(-0.04, 0.04, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # Force-directed initializations
    for s in range(10):
        inits.append(force_init(s))
        
    # Phase 1: Broad search from diverse initializations
    for pts in inits:
        r0 = get_safe_radii(pts)
        p0 = make_params(pts, r0)
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                c_val = get_constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            pass

    # Phase 2: Local perturbation refinement to escape local minima
    if best_params is not None:
        for trial in range(50):
            p0 = best_params.copy()
            p0[:N] += np.random.uniform(-0.0003, 0.0003, N)
            p0[N:3*N] += np.random.uniform(-0.008, 0.008, 2*N)
            p0[:N] = np.clip(p0[:N], 1e-6, 0.5)
            p0[N:3*N] = np.clip(p0[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
                if res.success:
                    c_val = get_constraints(res.x)
                    if np.min(c_val) >= -1e-7:
                        s_val = -res.fun
                        if s_val > best_sum:
                            best_sum = s_val
                            best_params = res.x.copy()
            except Exception:
                pass
                
        # Phase 3: High-precision polish
        try:
            res = minimize(objective, best_params, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-15, 'disp': False})
            if res.success and np.min(get_constraints(res.x)) >= -1e-8:
                best_params = res.x
                best_sum = -res.fun
        except Exception:
            pass

    # Fallback configuration (should rarely be reached)
    if best_params is None:
        pts = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                               np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        r0 = np.full(N, 0.04)
        best_params = make_params(pts, r0)
        
    # Reconstruct centers from optimized parameters
    r = best_params[:N]
    u = best_params[N:2*N]
    v = best_params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    centers = np.column_stack([x, y])
    radii = np.maximum(r, 0.0)
    
    return centers, radii, float(best_sum)
