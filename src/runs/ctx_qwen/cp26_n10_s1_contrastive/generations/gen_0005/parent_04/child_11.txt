1. **Parameterized Joint Optimization**: We parameterize circle positions as `x = r + u*(1-2r)` and `y = r + v*(1-2r)` with `u, v ∈ [0,1]`. This automatically satisfies boundary constraints, reducing the optimization to only pairwise non-overlap constraints. This makes SLSQP much more robust and faster.
2. **Diverse Initialization**: We generate numerous starting configurations using hexagonal lattices with varying row counts, rotations, scales, and jitter, alongside random dense layouts. Each is initialized with strictly feasible radii.
3. **SLSQP Local Refinement**: We run SLSQP from each initialization to push circles into tight local optima. We keep the best result.
4. **Perturbation Search**: We perturb the best parameters (radii and relative positions) and re-run SLSQP to escape shallow local minima.

6. **Final Polish**: We map the improved centers back to the parameterized form and run a high-precision SLSQP polish to ensure exact feasibility and tightness.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def obj_func(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N_CIRCLES])

def constr_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N_CIRCLES]
    u = params[N_CIRCLES:2*N_CIRCLES]
    v = params[2*N_CIRCLES:3*N_CIRCLES]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    d = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def get_safe_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i]-centers[j])**2))
                mx = min(mx, d*0.5)
        r[i] = max(1e-4, mx * 0.85)
    return r

def solve_lp_radii(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    for i in range(n):
        x, y = centers[i]
        bounds_val = np.array([x, 1.0 - x, y, 1.0 - y])
        for b in bounds_val:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 2.6e-4

def generate_hex(n, row_counts, seed, rot, scale, jitter):
    """Generate a hexagonal lattice initialization with specified parameters."""
    rng = np.random.default_rng(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:n])
    pts -= 0.5
    pts *= scale
    pts += 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(1e-5, 0.5)]*N_CIRCLES + [(0.0, 1.0)]*N_CIRCLES + [(0.0, 1.0)]*N_CIRCLES
    cons = {'type': 'ineq', 'fun': constr_func}
    
    best_sum = -np.inf
    best_params = None
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [4, 6, 5, 6, 5], 
        [7, 6, 6, 7], [8, 6, 6, 6], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 5, 6, 5, 6], [5, 7, 5, 7],
        [6, 6, 6, 6, 2], [7, 5, 6, 6, 2], [6, 5, 5, 5, 5],
        [5, 5, 6, 5, 5], [4, 4, 6, 6, 6], [6, 7, 6, 7],
        [5, 5, 5, 6, 5], [6, 5, 5, 5, 5], [7, 7, 6, 6]
    ]
    
    inits = []
    for pat in patterns:
        for _ in range(8):
            rot = rng.uniform(-0.4, 0.4)
            scale = rng.uniform(0.80, 1.25)
            jitt = rng.uniform(0.005, 0.04)
            inits.append(('hex', pat, rot, scale, jitt, rng.integers(10000)))
            
    for s in range(25):
        centers_r = rng.uniform(0.05, 0.95, (N_CIRCLES, 2))
        inits.append(('rand', None, None, None, None, centers_r))
        
    for itype, pat, rot, scale, jitt, seed_or_pts in inits:
        if itype == 'hex':
            centers = generate_hex(N_CIRCLES, pat, seed_or_pts, rot, scale, jitt)
        else:
            centers = seed_or_pts
            
        r0 = get_safe_radii(centers)
        p0 = make_params(centers, r0)
        
        try:
            res = minimize(obj_func, p0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                c_val = constr_func(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            pass
            
    if best_params is not None:
        # Perturbation refinement
        for trial in range(80):
            p0 = best_params.copy()
            p0[:N_CIRCLES] += rng.uniform(-0.0004, 0.0004, N_CIRCLES)
            p0[N_CIRCLES:3*N_CIRCLES] += rng.uniform(-0.006, 0.006, 2*N_CIRCLES)
            p0[:N_CIRCLES] = np.clip(p0[:N_CIRCLES], 1e-5, 0.5)
            p0[N_CIRCLES:3*N_CIRCLES] = np.clip(p0[N_CIRCLES:3*N_CIRCLES], 0.0, 1.0)
            
            try:
                res = minimize(obj_func, p0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
                if res.success:
                    c_val = constr_func(res.x)
                    if np.min(c_val) >= -1e-7:
                        s_val = -res.fun
                        if s_val > best_sum:
                            best_sum = s_val
                            best_params = res.x.copy()
            except Exception:
                pass

        # Extract centers for LP-based hill climbing
        r_b = best_params[:N_CIRCLES]
        u_b = best_params[N_CIRCLES:2*N_CIRCLES]
        v_b = best_params[2*N_CIRCLES:3*N_CIRCLES]
        x_b = r_b + u_b * (1.0 - 2.0 * r_b)
        y_b = r_b + v_b * (1.0 - 2.0 * r_b)
        centers_cur = np.column_stack([x_b, y_b])
        
        # Hill climbing on centers
        step_sizes = [0.03, 0.015, 0.007, 0.003, 0.001]
        for step in step_sizes:
            improved = True
            while improved:
                improved = False
                for _ in range(N_CIRCLES * 15):
                    idx = rng.integers(N_CIRCLES)
                    direction = rng.standard_normal(2)
                    direction /= np.linalg.norm(direction)
                    new_centers = centers_cur.copy()
                    new_centers[idx] += step * direction
                    new_centers[idx] = np.clip(new_centers[idx], 0.01, 0.99)
                    _, new_sum = solve_lp_radii(new_centers)
                    if new_sum > best_sum + 1e-7:
                        centers_cur = new_centers
                        _, best_sum = solve_lp_radii(centers_cur)
                        improved = True
                        break
                        
        # Convert back to params and polish
        r_final = solve_lp_radii(centers_cur)[0]
        best_params = make_params(centers_cur, r_final)
        
        try:
            res_final = minimize(obj_func, best_params, method='SLSQP', bounds=bounds, constraints=cons,
                                 options={'maxiter': 8000, 'ftol': 1e-15, 'disp': False})
            if res_final.success and np.min(constr_func(res_final.x)) >= -1e-8:
                best_params = res_final.x
                best_sum = -res_final.fun
        except Exception:
            pass

    # Fallback
    if best_params is None:
        centers_f = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N_CIRCLES], 
                                     np.tile(np.linspace(0.1, 0.9, 5), 6)[:N_CIRCLES]])
        r_f = np.full(N_CIRCLES, 0.04)
        best_params = make_params(centers_f, r_f)
        best_sum = np.sum(r_f)

    r = best_params[:N_CIRCLES]
    u = best_params[N_CIRCLES:2*N_CIRCLES]
    v = best_params[2*N_CIRCLES:3*N_CIRCLES]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    centers = np.column_stack([x, y])
    radii = np.maximum(r, 0.0)
    
    return centers, radii, float(best_sum)
