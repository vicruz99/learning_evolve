import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

N = 26

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    x, y = centers[:, 0], centers[:, 1]
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        for b in [x[i], 1.0 - x[i], y[i], 1.0 - y[i]]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 2.6e-4

def obj_de(x_flat):
    """Objective function for Differential Evolution."""
    centers = x_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, s = solve_lp_radii(centers)
    return -s

def joint_obj(x):
    """Objective for joint SLSQP optimization."""
    return -np.sum(x[2::3])

def joint_cons(x):
    """Inequality constraints for joint optimization."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    c = []
    # Boundary containment
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    rs = r[i_idx] + r[j_idx]
    c.append(dx*dx + dy*dy - rs*rs)
    return np.concatenate(c)

def run_packing():

    de_bounds = [(0.05, 0.95), (0.05, 0.95)] * N
    
    best_centers = None
    best_sum = -1.0

    for trial in range(3):

            res = differential_evolution(
                obj_de,
                bounds=de_bounds,
                popsize=15,
                maxiter=400,
                tol=1e-8,
                mutation=(0.5, 1.2),
                recombination=0.9,
                seed=42 + trial,
                polishing=False,
                init='sobol'
            )
            centers = res.x.reshape(N, 2)
            _, s = solve_lp_radii(centers)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
        except Exception:
            continue

    # Fallback initialization if DE fails
    if best_centers is None:
        pts = []
        r = 0.09
        y = r
        row = 0
        while len(pts) < N:

            x = x_off

                x += 2.0 * r

            row += 1

    return best_centers, final_radii, float(final_sum)
