import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        bounds_val = np.array([x, 1.0 - x, y, 1.0 - y])
        for b in bounds_val:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback
    return np.full(n, 1e-5), 2.6e-4

def objective(vars_vec):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_vec[:N_CIRCLES])

def constraint_func(vars_vec):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = vars_vec[:N_CIRCLES]
    u = vars_vec[N_CIRCLES:2*N_CIRCLES]
    v = vars_vec[2*N_CIRCLES:3*N_CIRCLES]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    i_idx, j_idx = np.triu_indices(N_CIRCLES, k=1)
    
    return dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx]**2

def hex_init(seed, rot, scale, row_counts):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.098
    y = r_est
    for r_idx, count in enumerate(row_counts):
        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N_CIRCLES])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    
    r_safe = np.zeros(N_CIRCLES)
    for i in range(N_CIRCLES):
        db = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
        r_safe[i] = 0.85 * min(db, dp/2.0)
        
    denom = np.clip(1.0 - 2.0 * r_safe, 1e-6, 1.0)
    u_safe = np.clip((pts[:, 0] - r_safe) / denom, 0.0, 1.0)
    v_safe = np.clip((pts[:, 1] - r_safe) / denom, 0.0, 1.0)
    
    vars0 = np.empty(3 * N_CIRCLES)
    vars0[:N_CIRCLES] = r_safe
    vars0[N_CIRCLES:2*N_CIRCLES] = u_safe
    vars0[2*N_CIRCLES:] = v_safe
    return vars0

def force_init(seed):
    """Spread points using force-directed layout within unit square."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N_CIRCLES, 2)
    r_curr = np.full(N_CIRCLES, 0.05)
    
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        rep_mag = 1.0 / dists**2
        np.fill_diagonal(rep_mag, 0.0)
        forces = np.sum(rep_mag[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 5.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 5.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        pts += 0.005 * (0.995**step) * forces
        pts = np.clip(pts, 1e-5, 1.0 - 1e-5)
        
        for i in range(N_CIRCLES):
            db = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            r_curr[i] = 0.8 * min(db, max(dp, 1e-6)/2.0)
            
    r_safe = np.zeros(N_CIRCLES)
    for i in range(N_CIRCLES):
        db = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        dists_i = np.linalg.norm(pts[i] - pts, axis=1)
        dists_i[i] = np.inf
        dp = np.min(dists_i)
        r_safe[i] = 0.9 * min(db, dp/2.0)
        
    denom = np.clip(1.0 - 2.0 * r_safe, 1e-6, 1.0)
    u_safe = np.clip((pts[:, 0] - r_safe) / denom, 0.0, 1.0)
    v_safe = np.clip((pts[:, 1] - r_safe) / denom, 0.0, 1.0)
    
    vars0 = np.empty(3 * N_CIRCLES)
    vars0[:N_CIRCLES] = r_safe
    vars0[N_CIRCLES:2*N_CIRCLES] = u_safe
    vars0[2*N_CIRCLES:] = v_safe
    return vars0

def run_packing():
    bounds = [(1e-6, 0.5)]*N_CIRCLES + [(0.0, 1.0)]*N_CIRCLES + [(0.0, 1.0)]*N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_sum = -np.inf
    rng = np.random.default_rng(42)
    
    inits = []

    ]
    
    # Generate diverse hexagonal starts
    for pat in patterns:
        for _ in range(6):

    # Generate force-directed starts
    for _ in range(20):
        inits.append(force_init(int(rng.integers(0, 10000))))
        
    # Phase 1: Broad search
    for x0 in inits:
        try:

            if res.success:

                if np.min(c_val) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            pass

    if best_vars is not None:

            x0[:N_CIRCLES] += rng.normal(0, 0.0015, N_CIRCLES)

            x0[:N_CIRCLES] = np.clip(x0[:N_CIRCLES], 1e-6, 0.5)
            x0[N_CIRCLES:3*N_CIRCLES] = np.clip(x0[N_CIRCLES:3*N_CIRCLES], 0.0, 1.0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
                if res.success:
                    c_val = constraint_func(res.x)
                    if np.min(c_val) >= -1e-7:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:

                            best_vars = res.x.copy()
            except Exception:
                pass
                
        # Phase 3: High-precision polish
        try:

            if res.success and np.min(constraint_func(res.x)) >= -1e-8:

                best_sum = -res.fun
        except Exception:
            pass

    # Extract centers from optimized parameters
    r_opt = best_vars[:N_CIRCLES]
    u_opt = best_vars[N_CIRCLES:2*N_CIRCLES]
    v_opt = best_vars[2*N_CIRCLES:]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    # Phase 4: LP polish to extract maximum radii for fixed optimal centers
    radii, final_sum = solve_lp_radii(centers)
    
    return centers, radii, float(final_sum)
