import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, etc.
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 2.6e-4

def joint_constraints(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def obj_func(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    d = 1.0 - 2.0 * r
    d = np.clip(d, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def lp_obj_for_centers(centers_flat):
    """Objective for center-space optimization: negative sum of LP-optimal radii."""
    centers = centers_flat.reshape(N, 2)
    _, s = solve_lp_radii(centers)
    return -s

def generate_inits():
    """Generates diverse initial configurations."""
    inits = []
    np.random.seed(42)
    
    # Hexagonal patterns with varied row counts
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], 
        [6,4,6,5,5], [5,6,4,6,5], [7,5,5,5,4], [5,7,5,6,3]
    ]
    for pat in pats:
        for s in range(8):
            rng = np.random.RandomState(s)
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
                for _ in range(cnt):
                    pts.append([x_start, y])
                    x_start += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                
            pts = np.array(pts[:N])
            angle = rng.uniform(-0.35, 0.35)
            c, s_val = np.cos(angle), np.sin(angle)
            rot = np.array([[c, -s_val], [s_val, c]])
            pts = (pts - 0.5) @ rot.T + 0.5
            scale = rng.uniform(0.85, 1.15)
            pts = (pts - 0.5) * scale + 0.5
            pts += rng.uniform(-0.015, 0.015, pts.shape)
            pts = np.clip(pts, 0.02, 0.98)
            
            r_init = np.full(N, 0.05)
            for i in range(N):
                d_w = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                d_p = np.min(np.linalg.norm(pts[i] - pts, axis=1))
                r_init[i] = 0.85 * min(d_w, d_p/2.0)
            inits.append(make_params(pts, r_init))
            
    # Force-directed layouts
    for s in range(15):
        rng = np.random.RandomState(s)
        pts = rng.rand(N, 2)
        r_curr = np.full(N, 0.05)
        for step in range(300):
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            dists = np.maximum(dists, 1e-5)
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:,:,np.newaxis] * diff / dists[:,:,np.newaxis], axis=1)
            for d in range(2):
                forces[:, d] += 5.0 * np.maximum(0, r_curr - pts[:, d])
                forces[:, d] -= 5.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            pts += 0.005 * forces
            pts = np.clip(pts, 1e-4, 1.0-1e-4)
            for i in range(N):
                d_w = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                d_p = np.min(np.linalg.norm(pts[i] - pts, axis=1))
                r_curr[i] = 0.85 * min(d_w, d_p/2.0)
        inits.append(make_params(pts, r_curr))
        
    return inits

def run_packing():
    bounds = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': joint_constraints}
    
    inits = generate_inits()
    best_params = None
    best_sum = -np.inf
    
    # Phase 1: Joint SLSQP from diverse starts
    for x0 in inits:
        try:
            res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(joint_constraints(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_params = res.x.copy()
        except Exception:
            pass
            
    if best_params is None:
        return np.zeros((N,2)), np.zeros(N), 0.0

    # Extract centers for center-space optimization phase
    r_opt = best_params[:N]
    u_opt = best_params[N:2*N]
    v_opt = best_params[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers_best = np.column_stack((x_opt, y_opt))
    
    # Phase 2: Nelder-Mead on centers to maximize LP sum
    try:
        res_nm = minimize(lp_obj_for_centers, centers_best.flatten(), method='Nelder-Mead',
                          options={'maxiter': 15000, 'xatol': 1e-7, 'fatol': 1e-9, 'disp': False})
        centers_nm = np.clip(res_nm.x.reshape(N, 2), 0.001, 0.999)
        r_nm, s_nm = solve_lp_radii(centers_nm)
        if s_nm > best_sum:
            best_sum = s_nm
            best_params = make_params(centers_nm, r_nm)
    except Exception:
        pass
        
    # Phase 3: Polish with SLSQP from updated centers
    try:
        res = minimize(obj_func, best_params, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(joint_constraints(res.x)) >= -1e-7:
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_params = res.x.copy()
    except Exception:
        pass

    # Phase 4: Perturb & Refine to escape local minima
    np.random.seed(42)
    for _ in range(40):
        x_pert = best_params.copy()
        x_pert[:N] += np.random.uniform(-0.001, 0.001, N)
        x_pert[N:3*N] += np.random.uniform(-0.015, 0.015, 2*N)
        x_pert[:N] = np.clip(x_pert[:N], 1e-6, 0.5)
        x_pert[N:3*N] = np.clip(x_pert[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_func, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(joint_constraints(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_params = res.x.copy()
        except Exception:
            pass

    # Final high-precision polish
    try:
        res = minimize(obj_func, best_params, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 8000, 'ftol': 1e-15, 'disp': False})
        if res.success and np.min(joint_constraints(res.x)) >= -1e-8:
            best_params = res.x
            best_sum = -res.fun
    except Exception:
        pass

    # Reconstruct final centers
    r_opt = best_params[:N]
    u_opt = best_params[N:2*N]
    v_opt = best_params[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    # Final LP solve guarantees exact feasibility and maximal radii for these centers
    final_radii, final_sum = solve_lp_radii(centers)
    
    return centers, final_radii, float(final_sum)
