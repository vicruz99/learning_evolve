import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def solve_lp_radii(centers):
    """
    Given fixed centers, solves the LP to find radii that maximize sum(r_i).
    Returns (radii, sum_radii)
    """
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback
    return np.full(n, 1e-5), 2.6e-4

def obj_joint(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def cons_joint(x):
    """Inequality constraints g(x) >= 0 for SLSQP."""
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    
    c_list = []
    # Boundary containment
    c_list.append(xc - r)
    c_list.append(1.0 - xc - r)
    c_list.append(yc - r)
    c_list.append(1.0 - yc - r)
    
    # Pairwise non-overlap: dist_sq >= (r_i + r_j)^2
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c_list.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c_list)

def generate_hex_init(row_counts, seed, rot, scale):
    """Generate a hexagonal lattice initialization with specified parameters."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.098
    y = r_est
    for r_idx, count in enumerate(row_counts):
        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N_CIRCLES])
    
    # Center, scale, rotate, jitter
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c_rot, s_rot = np.cos(rot), np.sin(rot)
        R = np.array([[c_rot, -s_rot], [s_rot, c_rot]])
        pts = pts @ R.T
        
    pts += rng.uniform(-0.025, 0.025, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    np.random.seed(42)
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    bounds_joint = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N_CIRCLES

            best_centers = final_centers
    except Exception:
        pass
        
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
