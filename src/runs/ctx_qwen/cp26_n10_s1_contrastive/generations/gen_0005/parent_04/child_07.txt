1. **Boundary-Free Parameterization**: We parameterize circle centers as `x_i = r_i + u_i*(1 - 2*r_i)` and `y_i = r_i + v_i*(1 - 2*r_i)` with `u_i, v_i ∈ [0, 1]`. This automatically guarantees `r_i ≤ x_i ≤ 1-r_i` and `r_i ≤ y_i ≤ 1-r_i`, eliminating boundary constraints from the optimizer. The only constraints left are pairwise non-overlap: `dist(i,j)^2 ≥ (r_i + r_j)^2`.

2. **Diverse Initialization**: We generate dozens of starting configurations using hexagonal lattices with varying row counts, rotations, and scales, alongside random layouts. Each is mapped to the `(r, u, v)` parameter space with safely shrunk initial radii to ensure strict feasibility.

3. **Multi-Phase Optimization**:

   - **Phase 3 (LP-Guided Refinement)**: For fixed centers, use an LP solver to compute the exact maximum feasible radii. Map these back to parameters and re-run SLSQP. This bridges the gap between joint optimization and exact radius allocation.
   - **Phase 4 (High-Precision Polish)**: A final SLSQP run with tight tolerances to converge to the numerical optimum.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(vars_vec):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def constraints(vars_vec):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def centers_from_params(params):
    """Reconstruct physical centers from (r, u, v) parameters."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y))

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    d = 1.0 - 2.0 * r
    d = np.clip(d, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def get_safe_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i]-centers[j])**2))
                mx = min(mx, d*0.5)
        r[i] = max(1e-4, mx * 0.85)
    return r

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)]*n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 0.0

def gen_hex(row_counts, rot, scale, jitter, seed):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    np.random.seed(42)
    bounds = [(1e-6, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_params = None
    best_sum = -np.inf
    rng = np.random.RandomState(42)
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5],
        [7,6,6,7], [8,6,6,6], [5,5,5,5,6],
        [6,4,6,5,5], [4,5,6,5,6], [5,7,5,7],
        [6,6,6,6,2], [7,5,6,6,2], [6,5,5,5,5],
        [5,5,6,5,5], [4,4,6,6,6], [6,7,6,7],
        [8,7,7,4], [7,8,6,5], [9,5,6,6], [5,8,7,6]
    ]
    
    inits_centers = []
    for pat in patterns:
        for _ in range(8):
            rot = rng.uniform(-0.4, 0.4)
            scale = rng.uniform(0.8, 1.2)
            jitt = rng.uniform(0.01, 0.04)
            inits_centers.append(gen_hex(pat, rot, scale, jitt, seed=rng.randint(0, 10000)))
            
    for _ in range(25):
        pts = rng.rand(N, 2)
        inits_centers.append(pts)
        
    # Phase 1: Broad search
    for centers_init in inits_centers:
        r_init = get_safe_radii(centers_init)
        p0 = make_params(centers_init, r_init)
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            pass
            
    if best_params is None:
        centers_fb = np.array([[0.1+0.2*i, 0.1+0.2*j] for i in range(5) for j in range(5)])
        centers_fb = np.vstack([centers_fb, [0.5, 0.5]])
        r_fb = get_safe_radii(centers_fb)
        best_params = make_params(centers_fb, r_fb)
        best_sum = np.sum(r_fb)
        
    # Phase 2: Perturbation refinement
    for _ in range(50):
        p0 = best_params.copy()
        p0[0::3] += rng.normal(0, 0.002, N)
        p0[1::3] += rng.normal(0, 0.02, N)
        p0[2::3] += rng.normal(0, 0.02, N)
        p0[0::3] = np.clip(p0[0::3], 1e-6, 0.49)
        p0[1::3] = np.clip(p0[1::3], 0.0, 1.0)
        p0[2::3] = np.clip(p0[2::3], 0.0, 1.0)
        
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: LP-guided refinement loop
    for _ in range(3):
        centers_cur = centers_from_params(best_params)
        r_lp, sum_lp = solve_lp(centers_cur)
        if sum_lp > best_sum:
            best_params = make_params(centers_cur, r_lp)
            best_sum = sum_lp
            
        # Re-optimize jointly after LP update
        p0 = best_params.copy()
        p0[0::3] += rng.normal(0, 0.0005, N)
        p0[1::3] += rng.normal(0, 0.005, N)
        p0[2::3] += rng.normal(0, 0.005, N)
        p0[0::3] = np.clip(p0[0::3], 1e-6, 0.49)
        p0[1::3] = np.clip(p0[1::3], 0.0, 1.0)
        p0[2::3] = np.clip(p0[2::3], 0.0, 1.0)
        
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            pass

    # Phase 4: High-precision polish
    try:
        res = minimize(objective, best_params, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        if res.success and np.min(constraints(res.x)) >= -1e-8:
            best_params = res.x
            best_sum = -res.fun
    except Exception:
        pass
        
    r_opt = best_params[0::3]
    u_opt = best_params[1::3]
    v_opt = best_params[2::3]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
