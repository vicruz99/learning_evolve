import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)
N_CONSTRAINTS = 4 * N + N_PAIRS

# Precompute LP constraint matrix structure (constant across all calls)
A_UB = np.zeros((N_CONSTRAINTS, N))
for i in range(N):
    A_UB[i, i] = 1.0       # r_i <= x_i
    A_UB[N + i, i] = 1.0   # r_i <= 1 - x_i
    A_UB[2*N + i, i] = 1.0 # r_i <= y_i
    A_UB[3*N + i, i] = 1.0 # r_i <= 1 - y_i

for k in range(N_PAIRS):
    A_UB[4*N + k, I_IDX[k]] = 1.0
    A_UB[4*N + k, J_IDX[k]] = 1.0

c_lp = -np.ones(N)
bounds_lp = [(0.0, None)] * N

def solve_lp_radii_fast(centers):
    """Solves LP to find optimal radii for fixed centers using precomputed matrix."""
    x, y = centers[:, 0], centers[:, 1]
    b_ub = np.empty(N_CONSTRAINTS)
    b_ub[:N] = x
    b_ub[N:2*N] = 1.0 - x
    b_ub[2*N:3*N] = y
    b_ub[3*N:4*N] = 1.0 - y
    
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    b_ub[4*N:] = np.sqrt(dx**2 + dy**2)
    
    res = linprog(c_lp, A_ub=A_UB, b_ub=b_ub, bounds=bounds_lp, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(N, 1e-5), 2.6e-4

def generate_configs():
    """Generates diverse initial center configurations."""
    configs = []
    rng = np.random.default_rng(42)
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [4, 6, 5, 6, 5],
        [5, 5, 5, 5, 6], [6, 4, 6, 5, 5], [5, 6, 4, 6, 5],
        [7, 6, 6, 7], [8, 6, 6, 6], [4, 5, 5, 5, 7], [7, 5, 5, 5, 4]
    ]
    
    for pat in patterns:
        for _ in range(3):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, count in enumerate(pat):
                x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
                for _ in range(count):
                    pts.append([x_start, y])
                    x_start += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
            pts = np.array(pts[:N])
            
            angle = rng.uniform(-0.35, 0.35)
            c, s = np.cos(angle), np.sin(angle)
            rot = np.array([[c, -s], [s, c]])
            pts = (pts - 0.5) @ rot.T + 0.5
            
            scale = rng.uniform(0.85, 1.15)
            pts = (pts - 0.5) * scale + 0.5
            
            pts += rng.uniform(-0.02, 0.02, pts.shape)
            pts = np.clip(pts, 0.02, 0.98)
            configs.append(pts)
            
    for _ in range(10):
        pts = rng.rand(N, 2)
        for _ in range(150):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            dists = np.maximum(dists, 1e-5)
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    f = diff[i,j] / (dists[i,j]**2)
                    forces[i] += f
                    forces[j] -= f
            pts += 0.008 * forces
            pts = np.clip(pts, 0.02, 0.98)
        configs.append(pts)
        
    return configs

def hill_climb(centers, iterations=5000):
    """Local search on centers using LP for radius evaluation."""
    best_centers = centers.copy()
    _, best_sum = solve_lp_radii_fast(best_centers)
    curr_centers = best_centers.copy()
    step_size = 0.06
    no_improve = 0
    
    for it in range(iterations):
        idx = np.random.randint(N)
        direction = np.random.randn(2)
        direction /= np.linalg.norm(direction)
        
        new_centers = curr_centers.copy()
        new_centers[idx] += step_size * direction
        new_centers[idx] = np.clip(new_centers[idx], 0.01, 0.99)
        
        _, new_sum = solve_lp_radii_fast(new_centers)
        
        if new_sum > best_sum + 1e-9:
            best_sum = new_sum
            best_centers = new_centers.copy()
            curr_centers = new_centers.copy()
            step_size = min(step_size * 1.15, 0.1)
            no_improve = 0
        else:
            step_size *= 0.96
            no_improve += 1
            
        if no_improve > 600:
            step_size = 0.05
            no_improve = 0
            curr_centers += np.random.randn(N, 2) * 0.005
            curr_centers = np.clip(curr_centers, 0.01, 0.99)
            _, best_sum = solve_lp_radii_fast(curr_centers)
            best_centers = curr_centers.copy()
            
    return best_centers, best_sum

def obj_func(x):
    return -np.sum(x[2::3])

def constr_func(x):
    C = x.reshape(N, 3)
    xc, yc, r = C[:,0], C[:,1], C[:,2]
    c = []
    c.append(xc - r)
    c.append(1.0 - xc - r)
    c.append(yc - r)
    c.append(1.0 - yc - r)
    dx = xc[I_IDX] - xc[J_IDX]
    dy = yc[I_IDX] - yc[J_IDX]
    rs = r[I_IDX] + r[J_IDX]
    c.append(dx**2 + dy**2 - rs**2)
    return np.concatenate(c)

def run_packing():
    np.random.seed(42)
    best_centers = None
    best_sum = -np.inf
    
    configs = generate_configs()
    
    for cfg in configs:
        centers, s = hill_climb(cfg, iterations=4000)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            
    if best_centers is not None:
        r_init, _ = solve_lp_radii_fast(best_centers)
        r_init = np.maximum(r_init, 1e-5) * 0.995
        
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = r_init
        
        bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
        try:
            res = minimize(obj_func, x0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constr_func},
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                if np.min(constr_func(res.x)) >= -1e-8:
                    final_sum = -res.fun
                    if final_sum > best_sum:
                        best_sum = final_sum
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        _, best_sum = solve_lp_radii_fast(best_centers)
        except Exception:
            pass
            
    best_radii, best_sum = solve_lp_radii_fast(best_centers)
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_sum)
