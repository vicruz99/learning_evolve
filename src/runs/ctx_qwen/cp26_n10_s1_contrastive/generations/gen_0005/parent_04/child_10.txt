import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 2.6e-4

def obj_centers(centers_flat):
    """Objective for center optimization: negative sum of optimal LP radii."""
    centers = centers_flat.reshape(N_CIRCLES, 2)
    # Clip to strictly interior to ensure LP feasibility
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    _, s = solve_lp_radii(centers)
    return -s

def make_init_force(n, seed):
    """Force-directed layout to spread points while respecting boundaries."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(n, 2) * 0.8 + 0.1
    v = np.zeros_like(pts)
    
    for _ in range(600):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        mask = d > 1e-4
        sd = np.where(mask, d, 1e-4)
        fm = 0.5 / (sd**2)
        fm = np.clip(fm, 0, 5.0)
        
        for i in range(n):
            for j in range(i + 1, n):
                if mask[i, j]:
                    fv = fm[i, j] * diff[i, j] / sd[i, j]
                    f[i] += fv
                    f[j] -= fv
                    
        for i in range(n):
            x, y = pts[i]
            if x < 0.08: f[i, 0] += 5.0 * (0.08 - x)
            elif x > 0.92: f[i, 0] -= 5.0 * (x - 0.92)
            if y < 0.08: f[i, 1] += 5.0 * (0.08 - y)
            elif y > 0.92: f[i, 1] -= 5.0 * (y - 0.92)
            f[i] -= 0.05 * (pts[i] - 0.5)
            
        v = 0.8 * v + f * 0.008
        pts += v
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def make_init_hex(n, seed):
    """Perturbed hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r = 0.095
    y = r
    row = 0
    while len(pts) < n:
        shift = r if row % 2 == 1 else 0.0
        x = r + shift
        while x <= 1.0 - r and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3.0) * r
        row += 1
        
    pts = np.array(pts[:n])
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    pts = np.clip(pts, 0.03, 0.97)
    return pts

def joint_obj(v):
    """Objective for joint SLSQP polish: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def joint_cons(v):
    """Inequality constraints for joint optimization: g(v) >= 0."""
    xc = v[0::3]
    yc = v[1::3]
    r = v[2::3]
    c = [xc - r, 1.0 - xc - r, yc - r, 1.0 - yc - r]
    dx = xc[I_IDX] - xc[J_IDX]
    dy = yc[I_IDX] - yc[J_IDX]
    rs = r[I_IDX] + r[J_IDX]
    c.append(dx * dx + dy * dy - rs * rs)
    return np.concatenate(c)

def run_packing():
    best_c = None
    best_r = None
    best_s = -np.inf
    
    # Phase 1: Diverse initializations
    inits = []
    for s in range(15):
        inits.append(make_init_force(N_CIRCLES, s))
    for s in range(10):
        inits.append(make_init_hex(N_CIRCLES, s))
        
    # Phase 2: Optimize centers using Nelder-Mead (robust for non-smooth LP objective)
    for pts in inits:
        x0 = pts.flatten()
        try:
            res = minimize(obj_centers, x0, method='Nelder-Mead',
                           options={'maxiter': 1500, 'fatol': 1e-12, 'xatol': 1e-12})
            cc = res.x.reshape(N_CIRCLES, 2)
            rr, ss = solve_lp_radii(cc)
            if ss > best_s:
                best_s = ss
                best_c = cc
                best_r = rr
        except Exception:
            pass
            
    # Phase 3: Perturbation refinement to escape local minima
    rng = np.random.default_rng(42)
    for _ in range(40):
        if best_c is None:
            break
        p = best_c + rng.normal(0, 0.008, best_c.shape)
        p = np.clip(p, 0.03, 0.97)
        try:
            res = minimize(obj_centers, p.flatten(), method='Nelder-Mead',
                           options={'maxiter': 1000, 'fatol': 1e-12, 'xatol': 1e-12})
            cc = res.x.reshape(N_CIRCLES, 2)
            rr, ss = solve_lp_radii(cc)
            if ss > best_s:
                best_s = ss
                best_c = cc
                best_r = rr
        except Exception:
            pass
            
    # Phase 4: Joint SLSQP polish for micro-adjustments
    if best_c is not None:
        rp = np.maximum(best_r, 1e-5) * 0.995
        x0j = np.zeros(3 * N_CIRCLES)
        x0j[0::3] = best_c[:, 0]
        x0j[1::3] = best_c[:, 1]
        x0j[2::3] = rp
        
        bnds_j = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N_CIRCLES
        
        try:
            res_j = minimize(joint_obj, x0j, method='SLSQP', bounds=bnds_j,
                             constraints={'type': 'ineq', 'fun': joint_cons},
                             options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res_j.success and np.min(joint_cons(res_j.x)) >= -1e-8:
                fc = np.column_stack((res_j.x[0::3], res_j.x[1::3]))
                fr, fs = solve_lp_radii(fc)
                if fs > best_s:
                    best_s = fs
                    best_c = fc
                    best_r = fr
        except Exception:
            pass
            
    # Fallback configuration
    if best_c is None:
        best_c = np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_r, best_s = solve_lp_radii(best_c)
        
    return best_c, best_r, float(best_s)
