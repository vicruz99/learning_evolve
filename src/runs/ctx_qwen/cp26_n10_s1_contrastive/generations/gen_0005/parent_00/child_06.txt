import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = N * (N - 1) // 2

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    x_c, y_c = centers[:, 0], centers[:, 1]
    
    c = -np.ones(N)
    n_const = 4 * N + NUM_PAIRS
    A_ub = np.zeros((n_const, N))
    b_ub = np.zeros(n_const)
    
    idx = 0
    for i in range(N):
        A_ub[idx, i] = 1.0; b_ub[idx] = x_c[i]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x_c[i]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y_c[i]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y_c[i]; idx += 1
        
    dx = x_c[:, None] - x_c[None, :]
    dy = y_c[:, None] - y_c[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    
    for k in range(NUM_PAIRS):
        i, j = I_IDX[k], J_IDX[k]
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        b_ub[idx] = dists[i, j]
        idx += 1
        
    bounds = [(0.0, None)] * N
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-5), 2.6e-4

def center_objective(x_flat):
    """Objective for Phase 1: minimize negative sum of radii for fixed centers."""
    centers = x_flat.reshape(N, 2)
    _, s = solve_lp_radii(centers)
    return -s

def generate_force_init(rng):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-6)
        repulsion = 1.0 / (dists**2)
        np.fill_diagonal(repulsion, 0.0)
        forces = np.sum(repulsion[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 5.0 * np.where(pts[:, d] < 0.05, 0.05 - pts[:, d], 0.0)
            forces[:, d] -= 5.0 * np.where(pts[:, d] > 0.95, pts[:, d] - 0.95, 0.0)
            
        pts += 0.004 * forces
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def generate_hex_init(rng, pat, rotation=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization with specified parameters."""
    pts = []
    r_e = 0.095
    y = r_e
    for r_idx, cnt in enumerate(pat):
        shift = (r_idx % 2) * r_e
        x = r_e + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_e
        y += r_e * np.sqrt(3.0)
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rotation != 0.0:
        c, s = np.cos(rotation), np.sin(rotation)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
    pts += rng.normal(0, 0.01, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def centers_to_params(centers, radii):
    """Convert centers and radii to the (r, u, v) parameterization."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    vars0 = np.empty(3 * N)
    vars0[0::3] = r
    vars0[1::3] = u
    vars0[2::3] = v
    return vars0

def joint_objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def joint_constraints(v):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = v[0::3]
    u = v[1::3]
    v_par = v[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v_par * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def run_packing():

    best_sum = -np.inf

    # Phase 2: Optimize centers to find good topologies
    for init_centers in inits:

                             bounds=bounds_center, options={'maxiter': 1500, 'ftol': 1e-12})
            curr_centers = res_c.x.reshape(N, 2)
            radii, curr_sum = solve_lp_radii(curr_centers)
            radii = np.maximum(radii, 1e-5)
            
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = curr_centers.copy()
        except Exception:
            continue
            
    # Phase 3: Joint refinement using (r, u, v) parameterization
    if best_centers is not None:
        radii, _ = solve_lp_radii(best_centers)
        radii = np.maximum(radii, 1e-5)
        x0 = centers_to_params(best_centers, radii * 0.995)
        
        # Fine polish
        try:
            res_j = minimize(joint_objective, x0, method='SLSQP', bounds=bounds_joint,
                             constraints=cons_joint, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res_j.success and np.min(joint_constraints(res_j.x)) >= -1e-7:
                best_vars = res_j.x
                best_sum = -res_j.fun

            pass
            
        # Perturbation search to escape local minima
        if best_vars is not None:

                x_p = best_vars.copy()
                x_p[1::3] += rng.normal(0, 0.04, N)
                x_p[2::3] += rng.normal(0, 0.04, N)

                    if res_p.success and np.min(joint_constraints(res_p.x)) >= -1e-7:

                        if s_val > best_sum:

        # Final high-precision polish
        if best_vars is not None:
            try:

        best_vars = centers_to_params(fallback_centers, fallback_radii)

    u_opt = best_vars[1::3]
    v_opt = best_vars[2::3]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    return centers, r_opt, float(best_sum)
