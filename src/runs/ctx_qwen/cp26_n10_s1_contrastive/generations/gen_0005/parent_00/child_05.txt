import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to maximize sum(r_i) subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def objective_param(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def constraints_param(v):
    """Inequality constraints using parameterization that auto-satisfies boundaries."""
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def objective_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints_joint(v):
    """Explicit constraints for joint (x, y, r) optimization."""
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    c = []
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def centers_to_params(centers, radii):
    """Convert (centers, radii) to (r, u, w) parameterization."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    v = np.empty(3 * N)
    v[0::3] = r
    v[1::3] = u
    v[2::3] = w
    return v

def generate_hex_init(rng, row_counts, rotation=0.0, scale=1.0, jitter_std=0.01):
    """Generate hexagonal lattice with specified row counts, rotation, scale, and jitter."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rotation != 0.0:
        c_val, s_val = np.cos(rotation), np.sin(rotation)
        rot_mat = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ rot_mat + 0.5
    pts += rng.normal(0, jitter_std, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def force_directed_init(seed, steps=400):
    """Generate a layout using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    r = np.full(N, 0.05)
    for step in range(steps):
        forces = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        repulsion = 1.0 / dists**2
        np.fill_diagonal(repulsion, 0.0)
        force_dir = diff / dists[:, :, np.newaxis]
        forces += np.sum(repulsion[:, :, np.newaxis] * force_dir, axis=1)
        for d in range(2):
            forces[:, d] += 10.0 * np.maximum(0, r - pts[:, d])
            forces[:, d] -= 10.0 * np.maximum(0, pts[:, d] - (1.0 - r))
        step_size = 0.008 * (0.996 ** step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        for i in range(N):
            d_wall = min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])
            min_dist = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            min_dist = max(min_dist, 1e-6)
            r[i] = 0.85 * min(d_wall, min_dist / 2.0)
    return pts, r
