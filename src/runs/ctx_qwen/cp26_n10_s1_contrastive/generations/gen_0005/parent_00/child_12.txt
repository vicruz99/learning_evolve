import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to maximize sum of radii subject to boundary and pairwise constraints."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        limits = [x, 1.0 - x, y, 1.0 - y]
        for lim in limits:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(lim)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Geometric fallback
    radii = np.zeros(n)
    for i in range(n):
        min_dist = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i]-centers[j])**2))
                if d/2.0 < min_dist:
                    min_dist = d/2.0
        radii[i] = max(1e-6, min_dist * 0.95)
    return radii, np.sum(radii)

def lp_objective(centers_flat):
    """Objective for center-only optimization: minimize negative LP sum of radii."""
    centers = centers_flat.reshape(N, 2)
    _, sum_r = solve_lp_radii(centers)
    return -sum_r

def generate_force_init(seed):
    """Generate initial centers using repulsive forces to spread points."""
    rng = np.random.RandomState(seed)
    centers = rng.rand(N, 2)
    radii = np.full(N, 0.05)
    
    for step in range(300):
        forces = np.zeros_like(centers)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-6)
        
        rep_mag = 1.0 / (dists**2 + 1e-4)
        np.fill_diagonal(rep_mag, 0.0)
        forces += np.sum(rep_mag[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        
        for d in range(2):
            forces[:, d] += 10.0 * np.maximum(0, radii - centers[:, d])
            forces[:, d] -= 10.0 * np.maximum(0, centers[:, d] - (1.0 - radii))
            
        centers += 0.005 * forces
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
        
    return centers

def get_hex_init(row_counts, seed, rot=0.0, scale=1.0):
    """Generate hexagonal lattice initialization with rotation and scaling."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
        
    centers = np.array(pts[:N])
    centers = (centers - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c, -s], [s, c]])
        centers = (centers - 0.5) @ rot_mat.T + 0.5
        
    centers += rng.uniform(-0.02, 0.02, centers.shape)
    return np.clip(centers, 0.02, 0.98)

def constraint_func(vars_vec):
    """Inequality constraints for SLSQP: pairwise non-overlap only. Boundaries handled by parameterization."""
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def objective(vars_vec):
    """Minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def centers_to_params(centers, radii):
    """Map physical centers/radii to the (r, u, v) parameterization."""
    r = np.maximum(radii, 1e-6)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    vars0 = np.empty(3 * N)
    vars0[0::3] = r
    vars0[1::3] = u
    vars0[2::3] = v
    return vars0

def run_packing():
    np.random.seed(42)
    bounds = [(1e-6, 0.5), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_sum = -np.inf
    
    # Phase 1: Diverse initializations
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [6,6,5,5,4], [5,5,6,5,5]
    ]
    for pat in patterns:
        for s in range(4):
            rot = np.random.uniform(-0.3, 0.3)
            scale = np.random.uniform(0.85, 1.15)
            inits.append(get_hex_init(pat, s, rot, scale))
            
    for s in range(15):
        inits.append(generate_force_init(s))
        
    for s in range(10):
        centers = np.random.rand(N, 2)
        centers = np.clip(centers, 0.05, 0.95)
        inits.append(centers)
        
    for centers_init in inits:
        radii_init, _ = solve_lp_radii(centers_init)
        x0 = centers_to_params(centers_init, radii_init * 0.99)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                if np.min(constraint_func(res.x)) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Center-only optimization on best configuration
    if best_vars is not None:
        best_centers = np.column_stack((best_vars[0::3] + best_vars[1::3]*(1.0-2.0*best_vars[0::3]),
                                        best_vars[0::3] + best_vars[2::3]*(1.0-2.0*best_vars[0::3])))
        
        try:
            res_centers = minimize(lp_objective, best_centers.flatten(), method='Nelder-Mead',
                                   options={'maxiter': 5000, 'xatol': 1e-7, 'fatol': 1e-7})
            improved_centers = res_centers.x.reshape(N, 2)
            imp_radii, imp_sum = solve_lp_radii(improved_centers)
            if imp_sum > best_sum:
                best_sum = imp_sum
                best_vars = centers_to_params(improved_centers, imp_radii * 0.99)
        except Exception:
            pass
            
        try:
            res = minimize(objective, best_vars, method='SLSQP', bounds=bounds,

            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Perturbation & Swap search to escape local minima
    if best_vars is not None:
        rng = np.random.default_rng(42)
        for _ in range(60):
            x0 = best_vars.copy()
            x0[1::3] += rng.normal(0, 0.06, N)
            x0[2::3] += rng.normal(0, 0.06, N)
            x0[0::3] += rng.normal(0, 0.004, N)

                x0[idx_i], x0[idx_j] = x0[idx_j], x0[idx_i]
                idx_i2, idx_j2 = 2 + 3*i, 2 + 3*j
                x0[idx_i2], x0[idx_j2] = x0[idx_j2], x0[idx_i2]
                
            x0[0::3] = np.clip(x0[0::3], 1e-6, 0.5)
            x0[1::3] = np.clip(x0[1::3], 0.0, 1.0)
            x0[2::3] = np.clip(x0[2::3], 0.0, 1.0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                if res.success:
                    if np.min(constraint_func(res.x)) >= -1e-8:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_vars = res.x.copy()
            except Exception:
                continue
                
    # Phase 4: Final high-precision polish
    if best_vars is not None:
        try:
            res = minimize(objective, best_vars, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if np.min(constraint_func(res.x)) >= -1e-8:
                best_vars = res.x
                best_sum = -res.fun
        except Exception:
            pass
            
    # Fallback
    if best_vars is None:
        centers_f = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        r_f = np.full(N, 0.09)
        best_vars = centers_to_params(centers_f, r_f)
        best_sum = np.sum(r_f)

    r_opt = best_vars[0::3]
    u_opt = best_vars[1::3]
    v_opt = best_vars[2::3]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    return centers, r_opt, float(best_sum)
