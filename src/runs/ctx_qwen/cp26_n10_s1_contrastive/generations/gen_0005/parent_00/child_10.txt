

<strategy>
To improve the packing of 26 circles in a unit square, I will employ a two-phase optimization strategy that separates center placement from radius assignment. This avoids the high-dimensional constraint coupling that often traps standard optimizers.

1. **Center Optimization via LP Objective**: For any fixed set of centers, the optimal radii that maximize the sum can be found exactly using a small Linear Program (LP). The LP constraints are simple: each radius is bounded by the distance to the nearest wall, and the sum of any two radii cannot exceed the distance between their centers. I will define an objective function `center_objective` that takes flattened center coordinates, solves this LP, and returns the negative sum of optimal radii.
2. **Diverse Initialization & Local Search**: I will generate 20 diverse initial configurations using:
   - Force-directed layouts with repulsive forces to naturally spread circles.
   - Rotated and scaled hexagonal lattices.
   - Random perturbations.
   Each initialization will be refined using `Nelder-Mead` on the 52-dimensional center space. This method is robust to the non-smooth nature of the LP objective.
3. **Constrained Polish**: The best center configuration will be passed to `SLSQP` optimizing all `(x, y, r)` variables simultaneously with explicit boundary and non-overlap constraints. This phase handles micro-adjustments and ensures strict feasibility.
4. **Perturbation & High-Precision Refinement**: Finally, I will run localized random perturbations around the best solution, followed by a high-precision SLSQP polish to squeeze out the last fractions of radius sum.

This hybrid approach leverages the speed and exactness of LP for global topology search, then switches to gradient-based constrained optimization for local precision.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii_fast(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    max_r_bounds = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                              np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    max_r_bounds = np.clip(max_r_bounds, 1e-6, 0.5)
    bounds = [(0.0, max_r_bounds[i]) for i in range(n)]
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    A_ub = np.zeros((len(I_IDX), n))
    A_ub[np.arange(len(I_IDX)), I_IDX] = 1.0
    A_ub[np.arange(len(I_IDX)), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-3

def center_objective(centers_flat):
    """Objective for center optimization: negative sum of optimal LP radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    _, s = solve_lp_radii_fast(centers)
    return -s

def force_init(seed):
    """Generates a well-spread initial configuration using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    for step in range(300):
        forces = np.zeros((N, 2))
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / (dists**2 + 0.005)
        np.fill_diagonal(rep, 0.0)
        dir_vec = diff / dists[:, :, None]
        forces += np.sum(rep[:, :, None] * dir_vec, axis=1)
        
        # Wall repulsion to keep points inside [0,1]
        for d in range(2):
            forces[:, d] += 5.0 * np.maximum(0, 0.06 - pts[:, d])
            forces[:, d] -= 5.0 * np.maximum(0, pts[:, d] - 0.94)
            
        pts += 0.02 * (0.985**step) * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
    return pts

def hex_init(seed, rotation=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization with rotation and scaling."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rotation != 0.0:
        c, s = np.cos(rotation), np.sin(rotation)
        rot = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ rot + 0.5
    pts += rng.uniform(-0.02, 0.02, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def constraint_func_full(vars_vec):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    x = vars_vec[0::3]
    y = vars_vec[1::3]
    r = vars_vec[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def objective_full(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[2::3])

def run_packing():
    rng = np.random.default_rng(42)
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    inits = []
    for s in range(12):
        inits.append(force_init(s))
    for s in range(8):
        rot = rng.uniform(-0.25, 0.25)
        sc = rng.uniform(0.85, 1.15)
        inits.append(hex_init(s, rot, sc))
        
    # Phase 1: Optimize centers using LP-based objective
    for init_centers in inits:
        try:
            res = minimize(center_objective, init_centers.flatten(), method='Nelder-Mead',
                           options={'maxiter': 1500, 'xatol': 1e-7, 'fatol': 1e-8})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = res.x.reshape(N, 2)
        except Exception:
            continue
            
    # Phase 2: Compute exact radii and prepare for constrained polish
    if best_centers is not None:
        best_radii, best_sum = solve_lp_radii_fast(best_centers)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii * 0.995  # Slightly shrink for strict interior feasibility
        
        bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
        cons = {'type': 'ineq', 'fun': constraint_func_full}
        
        try:
            res = minimize(objective_full, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func_full(res.x)) >= -1e-7:
                best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                best_radii = res.x[2::3]
                best_sum = -res.fun
        except Exception:
            pass
            
        # Phase 3: Local perturbation to escape shallow local minima
        for _ in range(25):
            x0_pert = np.zeros(3 * N)
            x0_pert[0::3] = best_centers[:, 0] + rng.normal(0, 0.004, N)
            x0_pert[1::3] = best_centers[:, 1] + rng.normal(0, 0.004, N)
            x0_pert[2::3] = best_radii + rng.normal(0, 0.0015, N)
            x0_pert[0::3] = np.clip(x0_pert[0::3], 0.01, 0.99)
            x0_pert[1::3] = np.clip(x0_pert[1::3], 0.01, 0.99)
            x0_pert[2::3] = np.maximum(x0_pert[2::3], 1e-5)
            
            try:
                res_p = minimize(objective_full, x0_pert, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if res_p.success and np.min(constraint_func_full(res_p.x)) >= -1e-7:
                    s_val = -res_p.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_centers = np.column_stack((res_p.x[0::3], res_p.x[1::3]))
                        best_radii = res_p.x[2::3]
            except Exception:
                continue
                
        # Phase 4: High-precision final polish
        if best_centers is not None:
            x0_final = np.zeros(3 * N)
            x0_final[0::3] = best_centers[:, 0]
            x0_final[1::3] = best_centers[:, 1]
            x0_final[2::3] = best_radii * 0.999
            try:
                res_f = minimize(objective_full, x0_final, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if res_f.success and np.min(constraint_func_full(res_f.x)) >= -1e-8:
                    best_centers = np.column_stack((res_f.x[0::3], res_f.x[1::3]))
                    best_radii = res_f.x[2::3]
                    best_sum = -res_f.fun
            except Exception:
                pass

    # Fallback valid configuration
    if best_centers is None:
        best_centers = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                        np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        best_radii = np.full(N, 0.09)
        best_sum = float(np.sum(best_radii))
        
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
