import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 2.6e-4

def hex_init(rng, pattern, rot, scale, jitter):
    """Generate a feasible initial center configuration based on a hexagonal lattice."""
    pts = []
    r_est = 0.095
    y = r_est
    
    for r_idx, cnt in enumerate(pattern):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
        
    pts = np.array(pts[:N])
    
    # Center and scale
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ rot_mat.T + 0.5
        
    pts += rng.normal(0, jitter, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def hillclimb_centers(centers, rng, steps=2000, step_size_init=0.015):

    best_centers = centers.copy()
    best_sum, best_radii = solve_lp_radii(centers)
    curr_centers = centers.copy()
    
    for step in range(steps):
        # Occasional global perturbation to escape local minima
        if step > 0 and step % 300 == 0:
            curr_centers += rng.normal(0, 0.008, curr_centers.shape)
            curr_centers = np.clip(curr_centers, 1e-4, 1.0 - 1e-4)
            
        idx = rng.integers(N)
        direction = rng.normal(size=2)
        norm = np.linalg.norm(direction)
        if norm > 1e-8:
            direction /= norm
            
        step_size = step_size_init * max(0.05, 1.0 - step / steps)
        
        new_centers = curr_centers.copy()
        new_centers[idx] += direction * step_size
        new_centers = np.clip(new_centers, 1e-4, 1.0 - 1e-4)
        
        new_sum, new_radii = solve_lp_radii(new_centers)
        
        if new_sum > best_sum:
            best_sum = new_sum
            best_centers = new_centers.copy()
            best_radii = new_radii.copy()
            curr_centers = new_centers
        elif rng.random() < 0.02:  # Simulated annealing acceptance
            curr_centers = new_centers
            
    return best_centers, best_radii, best_sum

def centers_to_params(centers, radii):
    """Convert centers and radii to the (r, u, v) parameterization."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    vars0 = np.empty(3 * N)
    vars0[0::3] = r
    vars0[1::3] = u
    vars0[2::3] = v
    return vars0

def objective_param(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def constraint_param(vars_vec):
    """Computes inequality constraints g(vars_vec) >= 0."""
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    i_idx, j_idx = np.triu_indices(N, k=1)
    return dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx]**2
