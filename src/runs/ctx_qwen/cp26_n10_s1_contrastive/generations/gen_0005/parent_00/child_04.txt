

<strategy>
To improve the sum of radii beyond the current ~2.631, we need a more robust global search strategy that avoids the stiffness of directly optimizing $(x, y, r)$ jointly from scratch. 

Key insights:
1. **LP-based Center Optimization**: For any fixed set of centers, the optimal radii that maximize the sum can be found exactly and efficiently via Linear Programming. This transforms the problem into optimizing only the $2N$ center coordinates with a smooth(er) objective landscape. Derivative-free optimizers like `Powell` work exceptionally well here.
2. **Boundary-free Parameterization for Polish**: While optimizing centers, we handle boundaries via clipping. For the final high-precision polish, we switch to the parameterization $x = r + u(1-2r)$, $y = r + v(1-2r)$ with $u, v \in [0,1]$. This automatically satisfies boundary constraints, allowing SLSQP to focus entirely on pairwise non-overlap and radius expansion.
3. **Diverse Initializations & Perturbation**: We generate random, hexagonal, and grid-based starts. We optimize centers via LP objective, then perturb the best configuration to escape local minima, and finally polish with SLSQP.
4. **Strict Compliance**: All helper functions are defined at the top level. No closures or lambdas are used. The code relies on `scipy.optimize` and `numpy`.

This pipeline reliably finds tight packings by decoupling center placement (global geometry) from radius sizing (local constraints), then jointly refining.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub, b_ub = [], []
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-5

def centers_obj(x):
    """Objective for center optimization: negative sum of optimal radii."""
    c = np.clip(x.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    _, s = solve_lp_radii(c)
    return -s

def get_init_centers(rng, style):
    """Generates diverse initial center configurations."""
    if style == 0:
        # Random uniform
        return rng.uniform(0.05, 0.95, (N, 2)).flatten()
    elif style == 1:
        # Hexagonal lattice with perturbation
        pts = []
        r_est = 0.095
        y = r_est
        row = 0
        while len(pts) < N:
            shift = (row % 2) * r_est
            x = r_est + shift
            while x <= 1.0 - r_est and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3.0)
            row += 1
        pts = np.array(pts[:N])
        pts += rng.normal(0, 0.012, (N, 2))
        return np.clip(pts, 0.02, 0.98).flatten()
    elif style == 2:
        # 5x5 Grid + center with perturbation
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng.normal(0, 0.025, (N, 2))
        return np.clip(pts, 0.02, 0.98).flatten()
    else:
        return rng.uniform(0.05, 0.95, (N, 2)).flatten()

def obj_full(vars_vec):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def con_full(vars_vec):
    """Constraints for SLSQP: pairwise non-overlap only (boundaries handled by parameterization)."""
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def run_packing():
    rng = np.random.default_rng(42)
    best_x = None
    best_sum = -np.inf
    
    # Phase 1: Center optimization via LP objective
    inits = []
    for _ in range(12):
        inits.append(get_init_centers(rng, 0))
    for _ in range(12):
        inits.append(get_init_centers(rng, 1))
    for _ in range(6):
        inits.append(get_init_centers(rng, 2))
        
    for x0 in inits:
        try:
            res = minimize(centers_obj, x0, method='Powell',
                           options={'maxiter': 3000, 'xtol': 1e-8, 'ftol': 1e-12, 'disp': False})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation refinement on centers to escape local minima
    if best_x is not None:
        for _ in range(20):
            x0 = best_x.copy()
            x0 += rng.normal(0, 0.018, x0.shape)
            x0 = np.clip(x0, 0.01, 0.99)
            try:
                res = minimize(centers_obj, x0, method='Powell',
                               options={'maxiter': 2000, 'xtol': 1e-7, 'ftol': 1e-12, 'disp': False})
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
            except Exception:
                pass
                
    # Phase 3: Joint SLSQP polish with boundary-free parameterization
    centers_opt = np.clip(best_x.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    radii_opt, _ = solve_lp_radii(centers_opt)
    radii_opt = np.maximum(radii_opt, 1e-6)
    
    denom = np.clip(1.0 - 2.0 * radii_opt, 1e-6, 1.0)
    u = np.clip((centers_opt[:, 0] - radii_opt) / denom, 0.0, 1.0)
    v = np.clip((centers_opt[:, 1] - radii_opt) / denom, 0.0, 1.0)
    
    x0_full = np.empty(3 * N)
    x0_full[0::3] = radii_opt * 0.995
    x0_full[1::3] = u
    x0_full[2::3] = v
    
    bounds_full = [(1e-7, 0.5), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': con_full}
    
    try:
        res = minimize(obj_full, x0_full, method='SLSQP', bounds=bounds_full,
                       constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        if res.success and np.min(con_full(res.x)) >= -1e-8:
            best_x = res.x
            best_sum = -res.fun
    except Exception:
        pass
        
    # Extract final results
    r_final = best_x[0::3]
    u_final = best_x[1::3]
    v_final = best_x[2::3]
    x_final = r_final + u_final * (1.0 - 2.0 * r_final)
    y_final = r_final + v_final * (1.0 - 2.0 * r_final)
    
    centers = np.column_stack((x_final, y_final))
    radii = np.maximum(r_final, 0.0)
    return centers, radii, float(np.sum(radii))
