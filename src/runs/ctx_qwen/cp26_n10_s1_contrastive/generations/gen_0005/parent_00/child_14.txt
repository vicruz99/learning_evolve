import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def constraints(v):
    """
    Computes inequality constraints g(v) >= 0.
    Uses parameterization to automatically satisfy boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_params(centers, radii):
    """Convert centers and radii to the (r, u, w) parameterization."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    v = np.empty(3 * N)
    v[0::3] = r
    v[1::3] = u
    v[2::3] = w
    return v

def solve_lp_radii(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        bounds_val = np.array([x, 1.0 - x, y, 1.0 - y])
        for b in bounds_val:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    return np.full(n, 0.01)

def run_packing():

    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -np.inf
    
    # Helper to run optimization and update best
    def run_and_update(v0):
        nonlocal best_v, best_sum
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-7:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_v = res.x.copy()
        except Exception:
            pass

    # Phase 1: Hexagonal lattice initializations with rotation and scaling
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [5, 6, 4, 6, 5],
        [6, 4, 6, 5, 5], [4, 5, 6, 5, 6], [5, 5, 5, 5, 6]
    ]
    
    for pat in patterns:
        for _ in range(8):
            r_est = 0.095
            pts = []
            y = r_est
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
                row += 1
                
            pts = np.array(pts[:N])
            pts += rng.normal(0, 0.012, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            
            # Random rotation to break symmetry

            pts = pts @ rot_mat.T
            pts = np.clip(pts, 0.02, 0.98)
            
            # LP radii for tight initial fit
            rad = solve_lp_radii(pts)
            rad = np.maximum(rad, 0.005)
            v0 = make_params(pts, rad * 0.97)
            run_and_update(v0)

            vp[2::3] += rng.normal(0, 0.06, N)
            vp[0::3] += rng.normal(0, 0.002, N)
            
            vp = np.clip(vp, [1e-6, 0.0, 0.0] * N, [0.5, 1.0, 1.0] * N)
            run_and_update(vp)
            
        # Phase 4: High-precision polish
        if best_v is not None:
            try:
                res = minimize(objective, best_v, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
                if np.min(constraints(res.x)) >= -1e-7:
                    best_v = res.x
                    best_sum = -res.fun
            except Exception:
                pass
                
    # Fallback valid configuration (should rarely be reached)
    if best_v is None:
        grid = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        grid = np.vstack([grid, [0.5, 0.5]])
        rad_f = np.full(N, 0.085)

    r_opt = best_v[0::3]
    u_opt = best_v[1::3]
    w_opt = best_v[2::3]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + w_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    return centers, r_opt, float(best_sum)
