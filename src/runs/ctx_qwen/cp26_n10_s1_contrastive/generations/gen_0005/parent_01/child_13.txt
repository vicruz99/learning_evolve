import numpy as np
from scipy.optimize import minimize

N = 26
MASK = np.triu(np.ones((N, N), dtype=bool), k=1)

def objective(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints(params):
    """
    Computes inequality constraints g(params) >= 0.
    Uses (r, u, v) parameterization to automatically satisfy boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization maps u, v in [0,1] to x, y such that circle is strictly inside [0,1]^2
    span = 1.0 - 2.0 * r
    x = r + u * span
    y = r + v * span
    
    # Pairwise squared distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    
    # Squared sum of radii
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist2[MASK] - r_sum[MASK]**2

def centers_to_params(centers):
    """Converts physical centers to strictly feasible (r, u, v) optimization parameters."""
    r = np.zeros(N)
    for i in range(N):
        d_wall = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        dists = np.linalg.norm(centers[i] - centers, axis=1)
        dists[i] = np.inf
        d_pair = np.min(dists)
        r[i] = min(d_wall, d_pair/2.0) * 0.92
        
    span = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / span, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / span, 0.0, 1.0)
    return np.concatenate([r, u, v])

def gen_hex(row_counts, rot, jitter, scale):
    """Generates a hexagonal lattice initialization with specified parameters."""
    pts = []
    r_est = 0.1
    y = r_est
    row = 0
    for cnt in row_counts:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = pts @ rot_mat.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += np.random.uniform(-jitter, jitter, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

        grid = np.clip(grid, 0.05, 0.95)
        inits.append(centers_to_params(grid))

        pts = np.random.rand(N, 2)

        inits.append(centers_to_params(pts))

    y_opt = r_opt + v_opt * span
    centers = np.column_stack((x_opt, y_opt))
    
    return centers, r_opt, float(np.sum(r_opt))
