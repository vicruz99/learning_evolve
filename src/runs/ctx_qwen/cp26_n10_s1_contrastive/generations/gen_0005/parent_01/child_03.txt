import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def force_init(seed):
    """Generates a well-spread configuration using force-directed layout."""
    np.random.seed(seed)
    pts = np.random.rand(N, 2) * 0.8 + 0.1
    
    for step in range(300):
        forces = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-4)
        
        rep_mag = 1.0 / (dists**2)
        np.fill_diagonal(rep_mag, 0.0)
        norm_diff = diff / dists[:, :, None]
        forces += np.sum(rep_mag[:, :, None] * norm_diff, axis=1)
        
        # Wall repulsion
        forces[:, 0] += 20.0 * np.maximum(0, 0.05 - pts[:, 0])
        forces[:, 0] -= 20.0 * np.maximum(0, pts[:, 0] - 0.95)
        forces[:, 1] += 20.0 * np.maximum(0, 0.05 - pts[:, 1])
        forces[:, 1] -= 20.0 * np.maximum(0, pts[:, 1] - 0.95)
        
        step_size = 0.015 * (0.97**step)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def gen_lattice_init(style, seed):
    """Generates hexagonal or grid initializations with rotation and perturbation."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    
    if style == 0:
        r_e = 0.095
        y = r_e
        row = 0
        idx = 0
        while idx < N:
            x_start = r_e if row % 2 == 0 else 2.0 * r_e
            x = x_start
            while x < 1.0 and idx < N:
                pts[idx] = [x, y]
                idx += 1
                x += 2.0 * r_e
            y += np.sqrt(3.0) * r_e
            row += 1
    else:
        idx = 0
        for i in range(5):
            for j in range(5):
                pts[idx] = [0.1 + 0.2*i, 0.1 + 0.2*j]
                idx += 1
        pts[25] = [0.5, 0.5]
        
    # Perturb and rotate
    pts += np.random.uniform(-0.025, 0.025, pts.shape)
    ang = np.random.uniform(-0.25, 0.25)
    c, s = np.cos(ang), np.sin(ang)
    R = np.array([[c, -s], [s, c]])
    pts = (pts - 0.5) @ R.T + 0.5
    return np.clip(pts, 0.03, 0.97)

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    d = 1.0 - 2.0 * r
    d = np.clip(d, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def param_to_phys(params):
    """Map (r, u, v) back to physical centers and radii."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def param_constraints(params):
    """Inequality constraints: pairwise non-overlap. Boundaries are auto-satisfied."""
    centers, r = param_to_phys(params)
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -np.inf
    best_params = None
    curr_params = None
    
    bounds_p = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons_p = {'type': 'ineq', 'fun': param_constraints}
    obj_fn = lambda p: -np.sum(p[:N])
    
    # Phase 1: Diverse global search
    inits = []
    for s in range(20):
        inits.append(force_init(s))
    for s in range(15):
        inits.append(gen_lattice_init(s % 2, s))
        
    for pts in inits:
        r_lp, _ = solve_lp_radii(pts)
        # Shrink slightly for strict interior feasibility
        r_start = np.maximum(r_lp * 0.99, 1e-5)
        p0 = make_params(pts, r_start)
        
        try:
            res = minimize(obj_fn, p0, method='SLSQP', bounds=bounds_p,
                           constraints=cons_p, options={'maxiter': 5000, 'ftol': 1e-12})
            if res.success:
                if np.min(param_constraints(res.x)) >= -1e-7:
                    s_val = np.sum(res.x[:N])
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            continue
            
    if best_params is None:
        # Fallback initialization
        pts_f = force_init(0)
        r_f, _ = solve_lp_radii(pts_f)
        best_params = make_params(pts_f, r_f * 0.95)
        
    curr_params = best_params.copy()
    
    # Phase 2: Aggressive local perturbation refinement
    for k in range(40):
        p0 = curr_params.copy()
        # Perturb u, v significantly to explore topology changes

        p0[2*N:3*N] += np.random.uniform(-0.025, 0.025, N)
        p0[:N] += np.random.uniform(-0.002, 0.002, N)
        
        # Ensure bounds
        p0[:N] = np.clip(p0[:N], 1e-6, 0.5)
        p0[N:] = np.clip(p0[N:], 0.0, 1.0)
        
        try:
            res = minimize(obj_fn, p0, method='SLSQP', bounds=bounds_p,

                        curr_params = res.x.copy()
        except Exception:
            continue
            
    # Phase 3: Permutation to break symmetry/stagnation
    for k in range(15):
        perm = np.random.permutation(N)
        p0 = np.zeros(3 * N)
        for i in range(N):
            p0[3*i : 3*i+3] = curr_params[3*perm[i] : 3*perm[i]+3]
            
        try:
            res = minimize(obj_fn, p0, method='SLSQP', bounds=bounds_p,

        if res_final.success and np.min(param_constraints(res_final.x)) >= -1e-8:
            curr_params = res_final.x
            best_sum = np.sum(curr_params[:N])
    except Exception:
        pass
        
    best_params = curr_params
    centers, radii = param_to_phys(best_params)
    return centers, radii, float(np.sum(radii))
