import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[:N])

def constraints(vars_vec):
    """
    Computes inequality constraints g(vars_vec) >= 0.
    Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    Boundary constraints are handled automatically by the (r, u, v) parameterization.
    """
    r = vars_vec[:N]
    u = vars_vec[N:2*N]
    v = vars_vec[2*N:3*N]
    
    denom = 1.0 - 2.0 * r
    x = r + u * denom
    y = r + v * denom
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_init_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = 1.0 - 2.0 * r
    denom = np.clip(denom, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def gen_hex_init(seed, scale, rot):
    """Generate a hexagonal lattice initialization with specified parameters."""
    np.random.seed(seed)
    pts = []
    r_est = 0.095 * scale
    y = r_est
    row = 0
    while len(pts) < N:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        x = x_start
        while x < 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
        pts = np.clip(pts, 0.02, 0.98)
        
    # Compute feasible radii
    r = np.full(N, r_est * 0.9)
    for i in range(N):
        db = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
        dp = max(dp, 1e-6)
        r[i] = min(db, dp/2.0) * 0.85
        
    # Add slight jitter to u, v to break symmetry
    u_jitter = np.clip(np.random.uniform(-0.03, 0.03, N), -0.05, 0.05)
    v_jitter = np.clip(np.random.uniform(-0.03, 0.03, N), -0.05, 0.05)
    
    denom = 1.0 - 2.0 * r
    denom = np.clip(denom, 1e-6, 1.0)
    u = np.clip((pts[:, 0] - r) / denom + u_jitter, 0.0, 1.0)
    v = np.clip((pts[:, 1] - r) / denom + v_jitter, 0.0, 1.0)
    
    return np.concatenate([r, u, v])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(1e-5, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_vars = None
    best_sum = -np.inf
    
    inits = []
    
    # Diverse hexagonal inits with various scales and rotations
    for s in range(40):
        sc = np.random.uniform(0.80, 1.20)
        rot = np.random.uniform(-0.35, 0.35)
        inits.append(gen_hex_init(s, sc, rot))
        
    # Random inits
    for s in range(20):
        np.random.seed(s + 100)
        pts = np.random.rand(N, 2)
        pts *= 0.85
        pts += 0.075
        r = np.full(N, 0.04)
        for i in range(N):
            db = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            dp = max(dp, 1e-6)
            r[i] = min(db, dp/2.0) * 0.75
        inits.append(make_init_params(pts, r))
        
    # Phase 1: Broad search
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-6:
                    s_val = np.sum(res.x[:N])
                    if s_val > best_sum:
                        best_sum = s_val
                        best_vars = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation and Index Swapping
    if best_vars is not None:

            x0[N:3*N] += np.random.uniform(-0.025, 0.025, 2*N)
            
            x0[:N] = np.clip(x0[:N], 1e-6, 0.49)
            x0[N:2*N] = np.clip(x0[N:2*N], 0.0, 1.0)

            except Exception:
                pass
                
        # Phase 3: High-precision polish

            if res.success and np.min(constraints(res.x)) >= -1e-7:
                best_vars = res.x
                best_sum = np.sum(best_vars[:N])
        except Exception:
            pass

    if best_vars is None:
        centers = np.zeros((N, 2))
        idx = 0
        for i in range(5):

                centers[idx] = [0.1 + 0.2*i, 0.1 + 0.2*j]
                idx += 1

    return centers, radii, float(np.sum(radii))
