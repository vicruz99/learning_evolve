1. **Parameterization**: We parameterize centers as `x = r + u*(1-2r)`, `y = r + v*(1-2r)` with `u,v ∈ [0,1]`. This automatically satisfies boundary constraints, reducing the constraint count and making the feasible region smoother for SLSQP.
2. **Diverse Initializations**: Generate ~80 high-quality starting configurations using hexagonal lattices with various row counts, rotations, and scales. Also include grid-like and corner-focused layouts to cover different topological basins.
3. **Phase 1 - Global Search**: Run SLSQP from each initialization to find strong local optima.
4. **Phase 2 - Perturbation**: Perform randomized perturbations around the best found solution to escape shallow local minima.
5. **Phase 3 - Alternating LP/SLSQP**: This is the key improvement. For fixed centers, solving a Linear Program (LP) exactly maximizes the sum of radii. We alternate between:
   - Running LP to push radii to their theoretical maximum for current centers.
   - Using these expanded radii as a new start point for SLSQP to re-optimize centers.
   This cycle effectively "locks" circles against each other and boundaries, allowing the optimizer to slide them into denser configurations.
6. **Phase 4 - High-Precision Polish**: Final SLSQP run with tight tolerances to squeeze out numerical slack.

This approach leverages the exactness of LP for radii with the flexibility of nonlinear optimization for centers, consistently beating standard SLSQP-only methods on packing problems.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints(params):
    """
    Computes inequality constraints g(params) >= 0.
    Uses parameterization to automatically satisfy boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization maps u, v in [0,1] to x, y such that circle is strictly inside [0,1]^2
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    # Pairwise squared distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    # Squared sum of radii
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX]

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for b_val in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b_val)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds_lp = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    return np.full(n, 1e-4)

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    d = 1.0 - 2.0 * r
    d = np.clip(d, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def get_safe_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i]-centers[j])**2))
                mx = min(mx, d*0.5)
        r[i] = max(1e-4, mx * 0.85)
    return r

def gen_hex(row_counts, rot, scale, jitter):
    """Generate a hexagonal lattice initialization with specified parameters."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    # Center and scale
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        R = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += np.random.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    np.random.seed(42)
    bounds = [(1e-5, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -np.inf
    best_params = None
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5],
        [7,6,6,7], [8,6,6,6], [5,5,5,5,6],
        [6,4,6,5,5], [4,5,6,5,6], [5,7,5,7],
        [6,6,6,6,2], [7,5,6,6,2], [6,5,5,5,5],
        [5,5,6,5,5], [4,4,6,6,6], [6,7,6,7], [5,5,5,6,5]
    ]
    
    inits = []
    for pat in patterns:
        for rot in [0.0, 0.1, -0.1, 0.2, -0.2, np.pi/12, -np.pi/12]:
            for scale in [0.9, 1.0, 1.1]:
                inits.append(gen_hex(pat, rot=rot, scale=scale, jitter=0.02))
                
    # Add random feasible layouts
    for _ in range(20):
        pts = np.random.rand(N, 2) * 0.8 + 0.1
        inits.append(pts)
        
    # Corner-focused layout
    pts_corner = np.zeros((N, 2))
    pts_corner[:4] = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    pts_corner[4:] = np.random.rand(N-4, 2) * 0.6 + 0.2
    inits.append(pts_corner)

    # Phase 1: Broad search from diverse initializations
    for pts in inits:
        r0 = get_safe_radii(pts)
        p0 = make_params(pts, r0)
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 6000, 'ftol': 1e-13})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-6:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            pass

    # Phase 2: Local perturbation refinement
    if best_params is not None:
        for trial in range(40):
            p0 = best_params.copy()
            p0[:N] += np.random.uniform(-0.002, 0.002, N)
            p0[N:3*N] += np.random.uniform(-0.015, 0.015, 2*N)
            p0[:N] = np.clip(p0[:N], 1e-5, 0.49)
            p0[N:3*N] = np.clip(p0[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-13})
                if res.success:
                    c_val = constraints(res.x)
                    if np.min(c_val) >= -1e-6:
                        s_val = -res.fun
                        if s_val > best_sum:
                            best_sum = s_val
                            best_params = res.x.copy()
            except Exception:
                pass
                
        # Phase 3: Alternating LP/SLSQP refinement to escape local minima
        for _ in range(15):
            r_cur = best_params[:N]
            u_cur = best_params[N:2*N]
            v_cur = best_params[2*N:3*N]
            xc = r_cur + u_cur * (1.0 - 2.0 * r_cur)
            yc = r_cur + v_cur * (1.0 - 2.0 * r_cur)
            centers = np.column_stack([xc, yc])
            
            # Exact radii expansion for fixed centers
            r_lp = solve_radii_lp(centers)
            r_lp = np.maximum(r_lp, 1e-5)
            
            # Re-map to parameter space
            d = 1.0 - 2.0 * r_lp
            d = np.clip(d, 1e-6, 1.0)
            u_new = np.clip((xc - r_lp) / d, 0.0, 1.0)
            v_new = np.clip((yc - r_lp) / d, 0.0, 1.0)
            p_new = np.concatenate([r_lp, u_new, v_new])
            
            # SLSQP step to adjust centers around expanded radii
            res = minimize(objective, p_new, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 6000, 'ftol': 1e-13})
            if res.success and np.min(constraints(res.x)) >= -1e-6:
                best_params = res.x
                best_sum = -res.fun

        # Phase 4: High-precision polish
        try:
            res = minimize(objective, best_params, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14})
            if res.success and np.min(constraints(res.x)) >= -1e-7:
                best_params = res.x
                best_sum = -res.fun
        except Exception:
            pass

    # Fallback configuration (should rarely be reached)
    if best_params is None:
        r_f = np.full(N, 0.05)
        u_f = np.linspace(0, 1, N)
        v_f = np.linspace(0, 1, N)
        best_params = np.concatenate([r_f, u_f, v_f])
        
    r_opt = best_params[:N]
    u_opt = best_params[N:2*N]
    v_opt = best_params[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack([x_opt, y_opt])
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(np.sum(radii))
