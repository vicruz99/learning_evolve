1. **Parameterization**: Use a `(r, u, v)` transformation where `x = r + u*(1-2r)` and `y = r + v*(1-2r)` with `u, v ∈ [0, 1]`. This automatically satisfies boundary constraints (`r ≤ x ≤ 1-r`, `r ≤ y ≤ 1-r`), reducing the constraint manifold to only pairwise non-overlap conditions. This significantly improves solver stability and convergence.
2. **Diverse Initializations**: Generate a large pool of starting configurations using:

3. **Multi-Start SLSQP Optimization**: Run `scipy.optimize.minimize` with the SLSQP method from each initialization. SLSQP handles nonlinear constraints well. I'll use tight tolerances (`ftol=1e-13`) and sufficient iterations.
4. **Local Escape & Permutation**: After finding the best local optimum, perform iterative perturbations on `(r, u, v)`. Periodically permute circle indices to break solver stagnation caused by symmetric basins.
5. **High-Precision Polish**: A final optimization pass with maximum iterations and tighter tolerance to extract every bit of possible radius sum.

import numpy as np
from scipy.optimize import minimize

N = 26

def objective(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints(params):
    """
    Computes inequality constraints g(params) >= 0.
    Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    Boundary constraints are automatically satisfied by the (r, u, v) parameterization.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Map normalized u, v to actual coordinates within [r, 1-r]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    # Vectorized pairwise squared distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Only need upper triangular pairs (i < j)
    i_idx, j_idx = np.triu_indices(N, k=1)
    return dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx]**2

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    d = 1.0 - 2.0 * r
    d = np.clip(d, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def hex_init(row_counts, r_est, seed, rot=0.0):
    """Generates a hexagonal lattice initialization with specified parameters."""
    rng = np.random.RandomState(seed)
    pts = []
    y = r_est
    for r_idx, count in enumerate(row_counts):
        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    
    # Estimate safe initial radii
    r_vals = np.zeros(N)
    for i in range(N):
        d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        dists = np.linalg.norm(pts[i] - pts, axis=1)
        dists[i] = np.inf
        d_pair = np.min(dists)
        r_vals[i] = 0.9 * min(d_wall, d_pair/2.0)
        
    return make_params(pts, r_vals)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = [(1e-6, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_params = None
    best_sum = -np.inf
    
    # Diverse hexagonal row patterns
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [7, 6, 6, 7], [8, 6, 6, 6], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 5, 6, 5, 6], [5, 7, 5, 7],
        [6, 6, 6, 6, 2], [7, 5, 6, 6, 2], [6, 5, 5, 5, 5],
        [5, 5, 6, 5, 5], [4, 4, 6, 6, 6], [6, 7, 6, 7],
        [5, 5, 5, 5, 4, 2], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5]
    ]
    
    inits = []
    # Generate hexagonal initializations with varied scales and rotations
    for pat in patterns:
        for r_e in [0.085, 0.09, 0.095, 0.10]:
            for s in range(3):
                rot = np.random.uniform(-0.2, 0.2)
                inits.append(hex_init(pat, r_e, seed=s, rot=rot))
                
    # Generate force-relaxed random initializations
    for s in range(30):
        rng = np.random.RandomState(s)
        pts = rng.rand(N, 2) * 0.8 + 0.1
        # Quick force simulation to spread points
        for _ in range(100):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    diff = pts[i] - pts[j]
                    dist = np.linalg.norm(diff) + 1e-5
                    f = diff / (dist**2)
                    forces[i] += f
                    forces[j] -= f
            pts += 0.003 * forces
            pts = np.clip(pts, 0.05, 0.95)
            
        r_vals = np.zeros(N)
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists = np.linalg.norm(pts[i] - pts, axis=1)
            dists[i] = np.inf
            d_pair = np.min(dists)
            r_vals[i] = 0.85 * min(d_wall, d_pair/2.0)
        inits.append(make_params(pts, r_vals))
        
    # Phase 1: Broad search from diverse initializations
    for p0 in inits:
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = np.sum(res.x[:N])
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Local perturbation & index permutation refinement
    if best_params is not None:
        for k in range(40):
            p0 = best_params.copy()
            p0[:N] += np.random.uniform(-0.001, 0.001, N)
            p0[N:2*N] += np.random.uniform(-0.015, 0.015, N)
            p0[2*N:3*N] += np.random.uniform(-0.015, 0.015, N)
            p0 = np.clip(p0, 0.0, 1.0)
            p0[:N] = np.clip(p0[:N], 1e-6, 0.49)
            
            # Periodically permute circle indices to break symmetry stagnation
            if k % 5 == 0:
                perm = np.random.permutation(N)
                p0_new = np.empty_like(p0)
                p0_new[:N] = p0[:N][perm]
                p0_new[N:2*N] = p0[N:2*N][perm]
                p0_new[2*N:3*N] = p0[2*N:3*N][perm]
                p0 = p0_new
                
            try:
                res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
                if res.success:
                    c_val = constraints(res.x)
                    if np.min(c_val) >= -1e-7:
                        s_val = np.sum(res.x[:N])
                        if s_val > best_sum:
                            best_sum = s_val
                            best_params = res.x.copy()
            except Exception:
                pass
                
        # Phase 3: High-precision polish
        try:
            res = minimize(objective, best_params, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
            if res.success and np.min(constraints(res.x)) >= -1e-8:
                best_params = res.x
        except Exception:
            pass

    # Fallback configuration (should rarely be reached)
    if best_params is None:
        fallback_centers = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                            np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        fallback_radii = np.full(N, 0.05)
        return fallback_centers, fallback_radii, float(np.sum(fallback_radii))
        
    # Reconstruct physical centers from optimized parameters
    r_opt = best_params[:N]
    u_opt = best_params[N:2*N]
    v_opt = best_params[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack([x_opt, y_opt])
    
    return centers, r_opt, float(np.sum(r_opt))
