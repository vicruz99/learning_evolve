import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(z):
    """Minimize negative sum of radii."""
    return -np.sum(z[:N])

def constraints(z):
    """
    Computes pairwise non-overlap constraints g(z) >= 0.
    Boundary constraints are implicitly satisfied by the (r, u, v) parameterization.
    """
    r = z[:N]
    u = z[N:2*N]
    v = z[2*N:3*N]
    
    # Map normalized coordinates to physical space
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    # Vectorized pairwise squared distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    
    # Squared sum of radii
    rs = r[:, None] + r[None, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def get_init_z(centers, safety=0.85):
    """Converts physical centers to a strictly feasible (r, u, v) optimization vector."""
    r = np.zeros(N)
    for i in range(N):
        # Distance to boundaries
        db = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
        # Distance to nearest neighbor
        dists = np.linalg.norm(centers[i] - centers, axis=1)
        dists[i] = np.inf
        dp = np.min(dists)
        # Apply safety margin to ensure strict interior feasibility
        r[i] = safety * min(db, dp / 2.0)
        
    r = np.clip(r, 1e-5, 0.4)
    denom = 1.0 - 2.0 * r
    denom = np.clip(denom, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def generate_hex_centers(rot, scale, jitter):
    """Generates a hexagonal lattice configuration with transformations."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        x = x_start
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    # Center and scale
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    # Rotate
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        
    # Jitter
    pts += np.random.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_spread(pts, steps=80):
    """Spreads points using a simple repulsion simulation."""
    for _ in range(steps):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        # Repulsion force ~ 1/d^2
        forces = np.sum(diff / dists**3, axis=1)
        pts += 0.005 * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    bounds = [(1e-5, 0.4)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraints}

    best_sum = -np.inf
    
    inits = []
