import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
np.seterr(all='ignore')

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons_slqp(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], 
            [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
            [8,5,5,5,3], [5,5,5,5,4,2], [7,6,5,4,3], [6,7,5,5,3], [4,5,6,5,6]]
    for pat in pats:
        for _ in range(3):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.35, 0.35), scale=rng.uniform(0.88, 1.12)))
            
    for _ in range(15):
        inits.append(force_init(rng))
        
    for _ in range(5):
        g_pts = np.array([[0.1 + i*0.2, 0.1 + j*0.2] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        g_pts += rng.uniform(-0.03, 0.03, (N, 2))
        inits.append(np.clip(g_pts, 0.05, 0.95))

    # Phase 1: SLSQP from diverse starts
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)
        r0 = np.clip(r0 * 0.99, 1e-6, 0.49)
        p0 = to_params(c0, r0)
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)
        best_p = to_params(c_f, r_f * 0.99)
        best_sum = -obj_slqp(best_p)
        
    c_best, r_best = from_params(best_p)
    
    # Phase 2: LP-driven Simulated Annealing on centers
    curr_c = c_best.copy()
    curr_r, curr_s = solve_lp_radii(curr_c)
    
    temp = 0.035
    step = 0.03
    
    for it in range(10000):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        if rng.random() < 0.02:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                c_best = curr_c.copy()
                r_best = nr.copy()
                step = min(0.04, step * 1.02)
            else:
                step = max(0.002, step * 0.996)
        else:
            curr_c[idx] = old
            step = max(0.002, step * 0.995)
            
        temp = max(1e-6, temp * 0.9993)
        
    # Update best from SA
    if curr_s > best_sum:
        best_sum = curr_s
        c_best = curr_c.copy()
        r_best, best_sum = solve_lp_radii(c_best)
        
    # Phase 3: SLSQP refinement on SA best
    r_init = np.clip(r_best * 0.995, 1e-6, 0.49)
    p0 = to_params(c_best, r_init)
    try:
        res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                       options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_p = res.x.copy()
                c_best, r_best = from_params(best_p)
    except Exception:
        pass
        
    # Phase 4: Aggressive Perturbation & SLSQP to escape local minima
    for k in range(40):
        xp = to_params(c_best, np.clip(r_best * 0.99, 1e-6, 0.49)).copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.03, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
                    c_best, r_best = from_params(best_p)
        except Exception:
            continue
            
    # Phase 5: Multi-circle shaking + SLSQP
    for _ in range(20):
        shake_c = c_best.copy()
        shake_idx = rng.choice(N, size=4, replace=False)
        shake_c[shake_idx] += rng.uniform(-0.08, 0.08, (4, 2))
        shake_c = np.clip(shake_c, 0.02, 0.98)
        
        r_shake, _ = solve_lp_radii(shake_c)
        p_shake = to_params(shake_c, np.clip(r_shake * 0.995, 1e-6, 0.49))
        
        try:
            res_s = minimize(obj_slqp, p_shake, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                             options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
            if res_s.success and np.min(cons_slqp(res_s.x)) >= -1e-8:
                s_s = -res_s.fun
                if s_s > best_sum:
                    best_sum = s_s
                    best_p = res_s.x.copy()
                    c_best, r_best = from_params(best_p)
        except Exception:
            continue

    # Phase 6: Rotation escapes to find optimal boundary alignments
    for _ in range(10):
        rot = rng.uniform(-0.15, 0.15)
        c_val, s_val = np.cos(rot), np.sin(rot)
        mat = np.array([[c_val, -s_val], [s_val, c_val]])
        c_rot = (c_best - 0.5) @ mat.T + 0.5
        c_rot = np.clip(c_rot, 0.01, 0.99)
        r_rot, _ = solve_lp_radii(c_rot)
        p_rot = to_params(c_rot, np.clip(r_rot * 0.995, 1e-6, 0.49))
        try:
            res_r = minimize(obj_slqp, p_rot, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                             options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
            if res_r.success and np.min(cons_slqp(res_r.x)) >= -1e-8:
                s_r = -res_r.fun
                if s_r > best_sum:
                    best_sum = s_r
                    best_p = res_r.x.copy()
                    c_best, r_best = from_params(best_p)
        except Exception:
            pass
            
    # Phase 7: High-precision final polish
    if best_p is not None:
        try:
            res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                             options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-9:
                best_p = res_f.x
                best_sum = -res_f.fun
                c_best, r_best = from_params(best_p)
        except Exception:
            pass
            
    # Final LP to ensure radii are exactly optimal for centers
    if c_best is not None:
        r_final, s_final = solve_lp_radii(c_best)
        return c_best, np.maximum(r_final, 0.0), float(s_final)
        
    # Fallback (should not be reached)
    return np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N],
