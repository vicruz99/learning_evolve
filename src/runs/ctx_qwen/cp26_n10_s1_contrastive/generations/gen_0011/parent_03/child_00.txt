import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def get_centers_radii(params):
    """Reconstruct centers and radii from (r, u, v) parameters."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraint_func(params):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundary handled by parameterization."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def hex_init(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = 0.0 if row % 2 == 0 else r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(350):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 25.0 * np.maximum(0, 0.08 - pts[:, dim])
            f[:, dim] -= 25.0 * np.maximum(0, pts[:, dim] - 0.92)
        pts += 0.003 * f
        pts = np.clip(pts, 0.04, 0.96)
    return pts

def grid_init(rng):
    """Generates a perturbed grid initialization."""
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-0.025, 0.025, (N, 2))
    return np.clip(pts, 0.05, 0.95)

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': constraint_func}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], 
        [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
        [5,6,4,6,5], [4,5,6,5,6], [6,5,4,6,5], [4,6,4,6,6], [6,4,4,6,6],
        [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3], [5,5,5,5,5,1], [6,5,5,5,5]
    ]
    for pat in pats:
        for _ in range(4):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.88, 1.12)))
    for _ in range(18):
        inits.append(force_init(rng))
    for _ in range(8):
        inits.append(grid_init(rng))
        
    # Phase 1: Evaluate all inits with LP, then SLSQP refine top candidates
    init_scores = []
    for c in inits:
        r, s_val = solve_lp_radii(c)
        init_scores.append((s_val, c, r))
    init_scores.sort(key=lambda x: x[0], reverse=True)
    
    for s0, c0, r0 in init_scores[:15]:
        r_init = np.clip(r0 * 0.985, 1e-6, 0.49)
        p0 = get_params(c0, r_init)
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is not None:
        best_c, best_r = get_centers_radii(best_p)
        best_r, best_sum = solve_lp_radii(best_c)
        
        # Phase 2: LP-driven Simulated Annealing on centers
        curr_c = best_c.copy()
        curr_s = best_sum
        temp = 0.04
        step = 0.035
        
        for it in range(8000):
            move_type = rng.integers(0, 4)
            old = None
            if move_type == 0:
                idx = rng.integers(N)
                old = curr_c[idx].copy()
                if rng.random() < 0.03:
                    curr_c[idx] = rng.uniform(0.05, 0.95, 2)
                else:
                    curr_c[idx] += rng.normal(0, step, 2)
                    curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
            elif move_type == 1:
                idx = rng.integers(N)
                old = curr_c[idx].copy()
                curr_c[idx] = rng.uniform(0.1, 0.9, 2)
            elif move_type == 2:
                i, j = rng.choice(N, 2, replace=False)
                old = curr_c[[i, j]].copy()
                curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
            else:
                idx = rng.integers(N)
                old = curr_c[idx].copy()
                curr_c[idx] += rng.normal(0, step * 0.5, 2)
                curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
                
            nr, ns = solve_lp_radii(curr_c)
            delta = ns - curr_s
            
            if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
                curr_s = ns
                if ns > best_sum:
                    best_sum = ns
                    best_c = curr_c.copy()
                    best_r = nr.copy()
            else:
                if old is not None:
                    if move_type < 2:
                        curr_c[idx] = old
                    else:
                        curr_c[[i, j]] = old
                        
            temp *= 0.9993
            step = max(0.001, step * 0.9994)
            
        if best_c is not None:
            best_r, best_sum = solve_lp_radii(best_c)
            best_p = get_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
        
        # Phase 3: Perturbation & SLSQP restarts
        for k in range(30):
            xp = best_p.copy()
            xp[:N] += rng.normal(0, 0.003, N)
            xp[N:3*N] += rng.normal(0, 0.025, 2*N)
            xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
            xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(objective, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                               options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(constraint_func(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_p = res.x.copy()
            except Exception:
                continue
                
        # Phase 4: Rotation escape strategy
        if best_c is not None:
            for _ in range(15):
                rot = rng.uniform(-0.15, 0.15)
                c_val, s_val = np.cos(rot), np.sin(rot)
                mat = np.array([[c_val, -s_val], [s_val, c_val]])
                c_rot = (best_c - 0.5) @ mat.T + 0.5
                c_rot = np.clip(c_rot, 0.01, 0.99)
                r_rot, _ = solve_lp_radii(c_rot)
                p_rot = get_params(c_rot, np.clip(r_rot * 0.995, 1e-6, 0.49))
                try:
                    res_r = minimize(objective, p_rot, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                                     options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                    if res_r.success and np.min(constraint_func(res_r.x)) >= -1e-7:
                        s_r = -res_r.fun
                        if s_r > best_sum:
                            best_sum = s_r
                            best_p = res_r.copy()
                except Exception:
                    pass
                    
        # Phase 5: High-precision final polish
        if best_p is not None:
            try:
                res_f = minimize(objective, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                                 options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if res_f.success and np.min(constraint_func(res_f.x)) >= -1e-8:
                    best_p = res_f.x
                    best_sum = -res_f.fun
            except Exception:
                pass
                
    # Fallback configuration
    if best_p is None:
        c_f = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        r_f = np.full(N, 0.04)
        best_p = get_params(c_f, r_f)
        best_sum = np.sum(r_f)
        
    centers, radii = get_centers_radii(best_p)
    radii_final, sum_final = solve_lp_radii(centers)
    
    return centers, np.maximum(radii_final, 0.0), float(sum_final)
