To maximize the sum of radii for 26 circles in a unit square, we will combine three powerful techniques that have proven effective in computational geometry:
1. **Implicit Boundary Parameterization**: Instead of enforcing boundary constraints explicitly in the optimizer, we parameterize each circle's center as $x_i = r_i + u_i(1-2r_i)$ and $y_i = r_i + v_i(1-2r_i)$ with $u_i, v_i \in [0, 1]$. This guarantees $r_i \le x_i, y_i \le 1-r_i$ by construction, drastically reducing the search space and numerical instability.
2. **LP-Driven Simulated Annealing (SA)**: For any fixed set of centers, the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints form a Linear Programming problem. We use this LP solver as the objective function for a Simulated Annealing process that explores center configurations. This allows us to efficiently navigate the non-convex landscape and discover high-quality contact topologies.
3. **SLSQP Topology Refinement**: Once a promising configuration is found, we use Sequential Least Squares Programming (SLSQP) to jointly optimize radii and centers (via the $u,v$ parameters) with explicit pairwise non-overlap constraints $d_{ij}^2 \ge (r_i+r_j)^2$. This polishes the packing and resolves tight contact geometries that LP-fixed centers cannot achieve alone.

The algorithm proceeds in phases:
- Generate diverse initial layouts (hexagonal lattices with various row counts/rotations, force-directed spreads, perturbed grids).
- Evaluate each with LP, then refine the top candidates with SLSQP.
- Run LP-based SA on the best centers to explore topology changes.
- Apply SLSQP refinement and perturbation restarts to escape local minima.
- Final high-precision polish and exact LP radius recomputation.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(seed, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = 0.0 if row % 2 == 0 else r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], 
        [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
        [5,4,6,5,6], [4,7,5,5,5], [6,5,4,6,5], [7,6,5,4,4], [8,6,5,4,3]
    ]
    for pat in pats:
        for s in range(4):
            inits.append(hex_init(s, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.9, 1.1)))
    for s in range(20):
        inits.append(force_init(s))
        
    # Evaluate inits with LP to find strong starts
    init_scores = []
    for c in inits:
        r, s_val = solve_lp_radii(c)
        init_scores.append((s_val, c, r))
    init_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Phase 1: SLSQP on top starts
    top_starts = init_scores[:15]
    for s0, c0, r0 in top_starts:
        r0_clipped = np.clip(r0 * 0.995, 1e-6, 0.49)
        p0 = to_params(c0, r0_clipped)
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        c_f, r_f = inits[0], init_scores[0][2]
        best_p = to_params(c_f, np.clip(r_f * 0.99, 1e-6, 0.49))
        best_sum = -obj_slqp(best_p)
        
    # Phase 2: LP-driven Simulated Annealing on centers
    c_curr, r_curr = from_params(best_p)
    c_best_sa = c_curr.copy()
    s_curr = best_sum
    s_best_sa = best_sum
    temp = 0.045
    step = 0.035
    
    for it in range(5000):
        idx = rng.integers(N)
        old = c_curr[idx].copy()
        
        if rng.random() < 0.03:
            c_curr[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            c_curr[idx] += rng.normal(0, step, 2)
            c_curr[idx] = np.clip(c_curr[idx], 0.01, 0.99)
            
        r_new, s_new = solve_lp_radii(c_curr)
        delta = s_new - s_curr
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            s_curr = s_new
            if s_new > s_best_sa:
                s_best_sa = s_new
                c_best_sa = c_curr.copy()
                step = min(0.05, step * 1.02)
            else:
                step = max(0.001, step * 0.998)
        else:
            c_curr[idx] = old
            step = max(0.001, step * 0.995)
            
        temp *= 0.9993
        
    if s_best_sa > best_sum:
        best_sum = s_best_sa
        r_sa, _ = solve_lp_radii(c_best_sa)
        best_p = to_params(c_best_sa, np.clip(r_sa * 0.995, 1e-6, 0.49))
        
    # Phase 3: Perturbation & SLSQP restarts
    for k in range(30):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.003, N)
        xp[N:3*N] += rng.normal(0, 0.025, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-9:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_p)
    # Final LP to ensure radii exactly match centers and are optimal
    radii_final, sum_final = solve_lp_radii(centers)
    
    return centers, np.maximum(radii_final, 0.0), float(sum_final)
