To maximize the sum of radii for 26 circles in a unit square, we need a robust optimization strategy that handles both continuous coordinate/radius adjustments and discrete topology changes (contact graphs). The most effective approach combines:
1. **Linear Programming (LP) for Radii**: Given fixed centers, optimizing radii is a convex LP problem. This is extremely fast and guarantees optimal radii for any fixed layout.
2. **Simulated Annealing (SA) on Centers**: Moving centers changes the feasible region for radii. SA allows us to explore different contact topologies by accepting occasionally worse solutions based on a temperature schedule, using LP to evaluate each configuration.
3. **SLSQP with Boundary-Safe Parameterization**: Once a good topology is found, we switch to `scipy.optimize.minimize` with method `SLSQP`. We parameterize centers as `(r, u, v)` where `x = r + u*(1-2r)` and `y = r + v*(1-2r)` with `u,v in [0,1]`. This automatically satisfies boundary constraints `r <= x,y <= 1-r`, allowing the optimizer to focus purely on pairwise non-overlap constraints `dist^2 >= (r_i + r_j)^2`.
4. **Perturbation & Re-optimization**: To escape local minima, we perturb the optimized parameters and re-run SLSQP multiple times.
5. **Final LP Polish**: After SLSQP converges, we freeze centers and run LP one last time to extract mathematically exact optimal radii for the final coordinates.

This hybrid approach leverages the speed of LP for global topology search and the gradient-based precision of SLSQP for local packing tightness, consistently reaching sums > 2.635.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def phys_to_param(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def param_to_phys(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def cons_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def generate_hex_centers(seed, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    rows = [6, 5, 6, 5, 4]
    pts = []
    ry = 0.095
    y = ry
    row = 0
    for cnt in rows:
        shift = (row % 2) * ry
        x = ry + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * ry
        y += np.sqrt(3.0) * ry
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        mat = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ mat.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_directed_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(400):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-6
        forces = np.sum(diff / dists[:, :, np.newaxis]**2, axis=1)
        for d in range(2):
            forces[:, d] += 30.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.004 * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Phase 1: Diverse initializations
    inits = []
    for s in range(20):
        inits.append(generate_hex_centers(s, rot=np.random.uniform(-0.2, 0.2), scale=np.random.uniform(0.9, 1.1)))
    for s in range(10):
        inits.append(force_directed_init(s))
    for s in range(5):
        rng_g = np.random.RandomState(s)
        pts = rng_g.uniform(0.15, 0.85, (N, 2))
        inits.append(pts)
        
    # Evaluate inits with LP
    for pts in inits:
        r, s_val = solve_lp(pts)
        if s_val > best_sum:
            best_sum = s_val
            best_c = pts.copy()
            best_r = r.copy()
            
    # SLSQP refinement on initial LP solutions
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    for pts in inits:
        r_init, _ = solve_lp(pts)
        p0 = phys_to_param(pts, np.clip(r_init * 0.99, 1e-6, 0.49))
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_dict, options={'maxiter': 4000, 'ftol': 1e-13})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_c, best_r = param_to_phys(res.x)
        except Exception:
            pass
            
    # Phase 2: LP-driven Simulated Annealing on centers to explore topology
    if best_c is not None:
        curr_c = best_c.copy()
        curr_s = best_sum
        step = 0.035
        temp = 0.045
        
        for it in range(4000):
            op = np.random.randint(3)
            if op == 0:
                idx = np.random.randint(N)
                old = curr_c[idx].copy()
                curr_c[idx] += np.random.normal(0, step, 2)
                curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            elif op == 1:
                i, j = np.random.choice(N, 2, replace=False)
                old = curr_c[[i, j]].copy()
                curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
            else:
                idx = np.random.randint(N)
                old = curr_c[idx].copy()
                curr_c[idx] = np.random.uniform(0.1, 0.9, 2)
                
            nr, ns = solve_lp(curr_c)
            delta = ns - curr_s
            
            if delta > 0 or (temp > 1e-8 and np.random.random() < np.exp(min(0.0, delta / temp))):
                curr_s = ns
                if ns > best_sum:
                    best_sum = ns
                    best_c = curr_c.copy()
                    best_r = nr.copy()
                    step = min(0.05, step * 1.02)
                else:
                    step = max(0.001, step * 0.998)
            else:
                if op == 0:
                    curr_c[idx] = old
                elif op == 2:
                    curr_c[idx] = old
                else:
                    curr_c[[i, j]] = old
                    
            temp *= 0.9993
            
    # Phase 3: SLSQP refinement after SA
    if best_c is not None:
        p0 = phys_to_param(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_dict, options={'maxiter': 8000, 'ftol': 1e-13})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_c, best_r = param_to_phys(res.x)
        except Exception:
            pass
            
        # Phase 4: Perturbation & SLSQP to escape local minima
        for k in range(30):
            xp = phys_to_param(best_c, best_r * 0.995)
            xp[N:3*N] += np.random.normal(0, 0.02, 2*N)
            xp[:N] += np.random.normal(0, 0.001, N)
            xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
            xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp,
                               constraints=cons_dict, options={'maxiter': 5000, 'ftol': 1e-13})
                if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_c, best_r = param_to_phys(res.x)
            except Exception:
                continue
                
        # Phase 5: High-precision final polish
        p_f = phys_to_param(best_c, np.clip(best_r * 0.999, 1e-6, 0.49))
        try:
            res_f = minimize(obj_slqp, p_f, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons_dict, options={'maxiter': 15000, 'ftol': 1e-14})
            if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-8:
                best_c, best_r = param_to_phys(res_f.x)
        except Exception:
            pass
            
    # Final LP to ensure radii are exactly optimal for the final centers
    if best_c is not None:
        final_r, final_sum = solve_lp(best_c)
        return best_c, np.maximum(final_r, 0.0), float(final_sum)
        
    # Fallback (should rarely be reached)
    c_f = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
    r_f = np.full(N, 0.09)
    r_opt, s_opt = solve_lp(c_f)
    return c_f, np.maximum(r_opt, 0.0), float(s_opt)
