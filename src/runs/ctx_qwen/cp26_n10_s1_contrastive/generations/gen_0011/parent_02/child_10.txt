import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_list = (x, 1.0 - x, y, 1.0 - y)
        for b in bounds_list:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N_CIRCLES]
    u = params[N_CIRCLES:2*N_CIRCLES]
    v = params[2*N_CIRCLES:3*N_CIRCLES]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N_CIRCLES])

def cons_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N_CIRCLES]
    u = params[N_CIRCLES:2*N_CIRCLES]
    v = params[2*N_CIRCLES:3*N_CIRCLES]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = 0.0 if row % 2 == 0 else r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N_CIRCLES:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N_CIRCLES])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N_CIRCLES, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def grid_init(rng):
    """Generates a perturbed grid initialization."""
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-0.025, 0.025, (N_CIRCLES, 2))
    return np.clip(pts, 0.05, 0.95)

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N_CIRCLES + [(0.0, 1.0)] * N_CIRCLES + [(0.0, 1.0)] * N_CIRCLES
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_lp_c = None
    best_lp_r = None
    best_lp_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], 
            [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
            [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3]]
    for pat in pats:
        for _ in range(4):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.9, 1.1)))
    for _ in range(15):
        inits.append(force_init(rng))
    for _ in range(8):
        inits.append(grid_init(rng))
        
    # Evaluate inits with LP to find the strongest starting topology
    for c in inits:
        r, s_val = solve_lp_radii(c)
        if s_val > best_lp_sum:
            best_lp_sum = s_val
            best_lp_c = c.copy()
            best_lp_r = r.copy()
            
    # Phase 1: LP-driven Simulated Annealing on centers
    curr_c = best_lp_c.copy()
    curr_s = best_lp_sum
    temp = 0.045
    step = 0.035
    
    for it in range(10000):
        idx = rng.integers(N_CIRCLES)
        old = curr_c[idx].copy()
        if rng.random() < 0.05:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_s = ns
            if ns > best_lp_sum:
                best_lp_sum = ns
                best_lp_c = curr_c.copy()
                best_lp_r = nr.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9993
        step = max(0.001, step * 0.9995)
        
    # Phase 2: SLSQP refinement with boundary-safe parameterization
    best_vars = None
    best_sum = best_lp_sum
    
    r_init = np.clip(best_lp_r * 0.995, 1e-6, 0.49)
    p0 = to_params(best_lp_c, r_init)
    
    try:
        res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_vars = res.x.copy()
    except Exception:
        pass
        
    if best_vars is None:
        best_vars = p0.copy()
        
    # Phase 3: Perturbation & SLSQP restarts to escape local minima
    for k in range(30):
        xp = best_vars.copy()
        xp[:N_CIRCLES] += rng.normal(0, 0.002, N_CIRCLES)
        xp[N_CIRCLES:3*N_CIRCLES] += rng.normal(0, 0.025, 2*N_CIRCLES)
        xp[:N_CIRCLES] = np.clip(xp[:N_CIRCLES], 1e-6, 0.49)
        xp[N_CIRCLES:3*N_CIRCLES] = np.clip(xp[N_CIRCLES:3*N_CIRCLES], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 4: Global rotation escape
    c_best, r_best = from_params(best_vars)
    for _ in range(15):
        rot = rng.uniform(-0.1, 0.1)
        c_val, s_val = np.cos(rot), np.sin(rot)
        mat = np.array([[c_val, -s_val], [s_val, c_val]])
        c_rot = (c_best - 0.5) @ mat.T + 0.5
        c_rot = np.clip(c_rot, 0.01, 0.99)
        r_rot, _ = solve_lp_radii(c_rot)
        p_rot = to_params(c_rot, np.clip(r_rot * 0.995, 1e-6, 0.49))
        try:
            res_r = minimize(obj_slqp, p_rot, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                             options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res_r.success and np.min(cons_slqp(res_r.x)) >= -1e-7:
                s_r = -res_r.fun
                if s_r > best_sum:
                    best_sum = s_r
                    best_vars = res_r.x.copy()
                    c_best, r_best = from_params(best_vars)
        except Exception:
            pass
            
    # Phase 5: High-precision final polish
    try:
        res_f = minimize(obj_slqp, best_vars, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-8:
            best_vars = res_f.x
            c_best, r_best = from_params(best_vars)
    except Exception:
        pass
        
    # Final exact LP resolution to guarantee optimality for the precise centers
    final_r, final_sum = solve_lp_radii(c_best)
    
    return c_best, np.maximum(final_r, 0.0), float(final_sum)
