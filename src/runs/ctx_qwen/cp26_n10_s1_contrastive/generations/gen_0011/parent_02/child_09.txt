To improve the packing of 26 circles in a unit square, I will employ a hybrid optimization strategy that separates the discrete/topological placement of centers from the continuous sizing of radii. This approach has proven highly effective in past iterations:

1. **Linear Programming for Radii**: Given fixed centers, the optimal radii that maximize $\sum r_i$ can be found exactly via Linear Programming (LP). The constraints are $r_i \le \text{dist\_to\_boundary}_i$ and $r_i + r_j \le \text{dist}_{ij}$. This is solved efficiently using `scipy.optimize.linprog` with the `highs` solver.
2. **Diverse Initialization**: Generate high-quality starting configurations using rotated/scaled hexagonal lattices, force-directed layouts, and perturbed grids. Evaluate each with LP to find the strongest initial topology.
3. **Simulated Annealing (SA) on Centers**: Perform a stochastic local search on circle centers. At each step, move a random circle, re-solve the LP for radii, and accept/reject the move based on the change in the sum of radii. This explores the contact graph topology effectively.
4. **SLSQP Joint Refinement**: After SA, switch to a nonlinear optimizer (SLSQP) using a boundary-safe parameterization $(r, u, v)$ where $x = r + u(1-2r)$ and $y = r + v(1-2r)$. This automatically satisfies boundary constraints and allows simultaneous joint optimization of centers and radii to squeeze out marginal gains.
5. **Escape & Polish**: Apply rotation escapes, random perturbations, and a final high-precision SLSQP pass followed by one last LP evaluation to guarantee radii are maximized for the final centers.

This hybrid LP-SA-SLSQP pipeline balances global exploration, topological flexibility, and local precision, consistently pushing past local minima that trap pure nonlinear or pure linear methods.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_boundary = 4 * n
    num_pairs = n * (n - 1) // 2
    num_c = num_boundary + num_pairs
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_list = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_list:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    A_ub[k:num_c, I_IDX] = 1.0
    A_ub[k:num_c, J_IDX] = 1.0
    b_ub[k:num_c] = dists[I_IDX, J_IDX]
    
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(seed, row_counts, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += np.random.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Generates a force-directed layout initialization."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7], [6,5,5,5,5],
        [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3], [5,5,5,5,5,1]
    ]
    for pat in pats:
        for s in range(5):
            inits.append(hex_init(s, pat, rot=np.random.uniform(-0.3, 0.3), scale=np.random.uniform(0.85, 1.15)))
    for s in range(20):
        inits.append(force_init(s))
        
    # Evaluate inits with LP to find a strong starting topology
    for pts in inits:
        r, s_val = solve_lp_radii(pts)
        if s_val > best_sum:
            best_sum = s_val
            best_c = pts.copy()
            best_r = r.copy()
            
    # Phase 1: LP-driven Simulated Annealing on centers
    if best_c is not None:
        curr_c = best_c.copy()
        curr_sum = best_sum
        temp = 0.04
        step = 0.035
        
        for it in range(8000):
            idx = np.random.randint(N)
            old = curr_c[idx].copy()
            
            # Occasionally take a large shock to escape deep local minima
            if np.random.rand() < 0.05:
                curr_c[idx] = np.random.uniform(0.05, 0.95, 2)
            else:
                curr_c[idx] += np.random.normal(0, step, 2)
                curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
                
            nr, ns = solve_lp_radii(curr_c)
            delta = ns - curr_sum
            
            # Accept if better, or probabilistically if worse
            if delta > 0 or (temp > 1e-7 and np.random.rand() < np.exp(delta / temp)):
                curr_sum = ns
                if ns > best_sum:
                    best_sum = ns
                    best_c = curr_c.copy()
                    best_r = nr.copy()
            else:
                curr_c[idx] = old
                
            temp *= 0.9993
            step = max(0.001, step * 0.9995)
            
    # Phase 2: SLSQP joint refinement with boundary-safe parameterization
    if best_c is not None:
        bounds_slqp = [(1e-5, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
        cons_slqp_dict = {'type': 'ineq', 'fun': cons_slqp}
        
        r_init = np.maximum(best_r * 0.995, 1e-5)
        p0 = to_params(best_c, r_init)
        
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp_dict,
                           options={'maxiter': 8000, 'ftol': 1e-14})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_c, best_r = from_params(res.x)
        except Exception:
            pass
            
        # Phase 3: Perturbation & SLSQP restarts to escape local minima
        for k in range(30):
            p0 = to_params(best_c, best_r * 0.995)
            p0[N:3*N] += np.random.uniform(-0.02, 0.02, 2*N)
            p0[:N] += np.random.uniform(-0.002, 0.002, N)
            p0[:N] = np.clip(p0[:N], 1e-5, 0.5)
            p0[N:3*N] = np.clip(p0[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp_dict,
                               options={'maxiter': 5000, 'ftol': 1e-14})
                if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_c, best_r = from_params(res.x)
            except Exception:
                pass
                
        # Phase 4: Global rotation escapes
        for _ in range(15):
            rot = np.random.uniform(-0.1, 0.1)
            c_val, s_val = np.cos(rot), np.sin(rot)
            mat = np.array([[c_val, -s_val], [s_val, c_val]])
            c_rot = (best_c - 0.5) @ mat.T + 0.5
            c_rot = np.clip(c_rot, 0.01, 0.99)
            
            r_rot, _ = solve_lp_radii(c_rot)
            p_rot = to_params(c_rot, np.clip(r_rot * 0.995, 1e-5, 0.5))
            
            try:
                res_r = minimize(obj_slqp, p_rot, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp_dict,
                                 options={'maxiter': 4000, 'ftol': 1e-14})
                if res_r.success and np.min(cons_slqp(res_r.x)) >= -1e-7:
                    s_r = -res_r.fun
                    if s_r > best_sum:
                        best_sum = s_r
                        best_c, best_r = from_params(res_r.x)
            except Exception:
                pass
                
        # Phase 5: Final LP polish to guarantee radii are exactly optimal for final centers
        if best_c is not None:
            final_r, final_sum = solve_lp_radii(best_c)
            best_c, best_r, best_sum = best_c, final_r, final_sum
            
    # Fallback configuration (should rarely be reached)
    if best_c is None:
        best_c = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_r, best_sum = solve_lp_radii(best_c)
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
