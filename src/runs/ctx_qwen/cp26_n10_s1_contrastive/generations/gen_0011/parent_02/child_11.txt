import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def get_centers_radii(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N_CIRCLES]
    u = params[N_CIRCLES:2*N_CIRCLES]
    v = params[2*N_CIRCLES:3*N_CIRCLES]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def slqp_constraints(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N_CIRCLES]
    u = params[N_CIRCLES:2*N_CIRCLES]
    v = params[2*N_CIRCLES:3*N_CIRCLES]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    i_idx, j_idx = np.triu_indices(N_CIRCLES, k=1)
    return dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx]**2

def slqp_objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N_CIRCLES])

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7], [6,5,5,5,5]
    ]
    
    for pat in patterns:
        for s in range(5):
            pts = []
            ry = 0.095
            y = ry
            row = 0
            for cnt in pat:
                shift = ry if row % 2 == 1 else 0.0
                x = ry + shift
                for _ in range(cnt):
                    if len(pts) < N_CIRCLES:
                        pts.append([x, y])
                    x += 2.0 * ry
                y += np.sqrt(3.0) * ry
                row += 1
            pts = np.array(pts[:N_CIRCLES])
            pts = (pts - 0.5) * rng.uniform(0.9, 1.1) + 0.5
            rot = rng.uniform(-0.3, 0.3)
            c, s_val = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s_val], [s_val, c]]) + 0.5
            pts += rng.uniform(-0.02, 0.02, pts.shape)
            inits.append(np.clip(pts, 0.02, 0.98))

    # Force-directed layouts
    for s in range(15):
        pts = rng.uniform(0.15, 0.85, (N_CIRCLES, 2))
        for _ in range(300):
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-6
            forces = np.sum(diff / dists[:, :, np.newaxis] / dists[:, :, np.newaxis], axis=1)
            for d in range(2):
                forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
                forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
            pts += 0.015 * forces
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
    return inits

def run_packing():
    rng = np.random.default_rng(42)
    best_c = None
    best_r = None
    best_sum = -np.inf

    # Phase 0: Evaluate diverse initializations
    inits = generate_inits(rng)
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)
        if s0 > best_sum:
            best_sum = s0
            best_c = c0.copy()
            best_r = r0.copy()

    # Phase 1: LP-driven Simulated Annealing on centers
    curr_c = best_c.copy()
    curr_sum = best_sum
    temp = 0.05
    step = 0.04

    for it in range(15000):
        idx = rng.integers(N_CIRCLES)
        old = curr_c[idx].copy()

        # Occasional large jump to escape deep local minima
        if rng.random() < 0.05:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)

        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_sum

        # Acceptance criterion
        if delta > 0 or (temp > 1e-7 and rng.random() < np.exp(min(500.0, delta / temp))):
            curr_sum = ns
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
        else:
            curr_c[idx] = old

        temp *= 0.9993
        step = max(0.001, step * 0.9994)

    # Phase 2: SLSQP Refinement with boundary-safe parameterization
    bounds_slqp = [(1e-6, 0.5)] * N_CIRCLES + [(0.0, 1.0)] * N_CIRCLES + [(0.0, 1.0)] * N_CIRCLES
    cons_slqp = {'type': 'ineq', 'fun': slqp_constraints}

    r_init = np.maximum(best_r * 0.995, 1e-6)
    p0 = get_params(best_c, r_init)

    try:
        res = minimize(slqp_objective, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                       options={'maxiter': 8000, 'ftol': 1e-14})
        if res.success and np.min(slqp_constraints(res.x)) >= -1e-7:
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_c, best_r = get_centers_radii(res.x)
    except Exception:
        pass

    # Phase 3: Perturbation & SLSQP restarts to escape remaining local minima
    for k in range(30):
        p0 = get_params(best_c, best_r * 0.995)
        p0[N_CIRCLES:3*N_CIRCLES] += rng.uniform(-0.025, 0.025, 2*N_CIRCLES)
        p0[:N_CIRCLES] += rng.uniform(-0.002, 0.002, N_CIRCLES)
        p0[:N_CIRCLES] = np.clip(p0[:N_CIRCLES], 1e-6, 0.5)
        p0[N_CIRCLES:3*N_CIRCLES] = np.clip(p0[N_CIRCLES:3*N_CIRCLES], 0.0, 1.0)

        try:
            res = minimize(slqp_objective, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                           options={'maxiter': 6000, 'ftol': 1e-14})
            if res.success and np.min(slqp_constraints(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_c, best_r = get_centers_radii(res.x)
        except Exception:
            continue

    # Phase 4: High-precision final polish
    try:
        p0 = get_params(best_c, best_r * 0.999)
        res_f = minimize(slqp_objective, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                         options={'maxiter': 15000, 'ftol': 1e-14})
        if res_f.success and np.min(slqp_constraints(res_f.x)) >= -1e-8:
            best_c, best_r = get_centers_radii(res_f.x)
            best_sum = -res_f.fun
    except Exception:
        pass

    # Final LP guarantee: ensure radii are exactly optimal for the exact final centers
    if best_c is not None:
        final_r, final_sum = solve_lp_radii(best_c)
        best_c, best_r, best_sum = best_c, final_r, final_sum

    return best_c, np.maximum(best_r, 0.0), float(best_sum)
