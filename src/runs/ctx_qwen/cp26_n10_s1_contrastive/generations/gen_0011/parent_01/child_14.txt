To maximize the sum of radii for 26 circles in a unit square, we need to navigate a highly non-convex optimization landscape. The best known packings for this number of circles closely resemble hexagonal lattices, often with rows of lengths like 6, 5, 6, 5, 4 or 5, 6, 5, 6, 4. 

Our strategy combines three powerful techniques:
1. **Linear Programming (LP) for Radii**: Given any set of centers, finding the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints is a convex LP problem. We solve this efficiently using `scipy.optimize.linprog` with the `highs` method. This gives us an exact upper bound on the sum of radii for any fixed center configuration.
2. **Simulated Annealing (SA) on Centers**: We treat the center coordinates as our search space. By repeatedly perturbing centers and evaluating the resulting LP-optimal sum of radii, we can explore topological changes (which circles touch which) and escape local minima. We use a cooling schedule and accept worse moves probabilistically. To enhance exploration, we occasionally swap circle positions or perturb multiple circles simultaneously.
3. **Sequential Least Squares Programming (SLSQP) Refinement**: The SA phase finds good center topologies, but gradient-based refinement is needed to push circles to their exact limiting contacts. We use a parameterization $(r, u, v)$ where $x = r + u(1-2r)$ and $y = r + v(1-2r)$ with $u,v \in [0,1]$. This parameterization automatically satisfies boundary constraints, allowing SLSQP to focus purely on pairwise non-overlap constraints $d_{ij}^2 \ge (r_i+r_j)^2$. We run SLSQP periodically during SA and extensively at the end for high-precision polishing.

We initialize with diverse hexagonal lattice patterns (rotated and scaled), run the hybrid SA+LP search, and conclude with aggressive gradient-based polishing to squeeze out the maximum possible sum.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        vals = [x, 1.0 - x, y, 1.0 - y]
        for v in vals:
            A_ub[k, i] = 1.0
            b_ub[k] = v
            k += 1
            
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-7), 0.0

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def cons(p):
    """Inequality constraints for SLSQP: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def hex_init(rng, pattern, rot, scale):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.1
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-5:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def run_packing():
    rng = np.random.default_rng(42)
    best_c = None
    best_r = None
    best_sum = -np.inf
    best_p = None
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], 
        [7,5,5,5,4], [6,6,5,5,4], [5,7,5,5,4], [6,5,5,5,5], 
        [5,5,6,5,5], [4,5,5,5,7], [5,5,5,6,5], [6,4,6,5,5]
    ]
    inits = []
    for pat in patterns:
        for _ in range(4):
            inits.append(hex_init(rng, pat, rng.uniform(-0.3, 0.3), rng.uniform(0.9, 1.1)))
            
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': cons}
    
    # Phase 1: Broad SLSQP search from diverse hexagonal starts
    for c0 in inits:
        r0, _ = solve_lp(c0)
        r0 = np.clip(r0 * 0.99, 1e-6, 0.49)
        p0 = to_params(c0, r0)
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        c0 = inits[0]
        r0, _ = solve_lp(c0)
        best_p = to_params(c0, r0 * 0.99)
        best_sum = -obj(best_p)
        
    c_curr, r_curr = from_params(best_p)
    curr_sum = best_sum
    best_c = c_curr.copy()
    best_r = r_curr.copy()
    
    # Phase 2: Simulated Annealing on centers with LP evaluation
    temp = 0.06
    step = 0.04
    for it in range(10000):
        idx = rng.integers(N)
        old_centers = c_curr.copy()
        
        move_type = rng.integers(5)
        if move_type == 0:
            c_curr[idx] += rng.normal(0, step, 2)
        elif move_type == 1:
            c_curr[idx] = rng.uniform(0.05, 0.95, 2)
        elif move_type == 2:
            j = rng.choice([k for k in range(N) if k != idx])
            c_curr[[idx, j]] = c_curr[[j, idx]].copy()
        else:
            # Shake multiple circles to escape deep local minima
            idxs = rng.choice(N, 3, replace=False)
            c_curr[idxs] += rng.normal(0, step * 1.5, (3, 2))
            
        c_curr = np.clip(c_curr, 0.005, 0.995)
        
        nr, ns = solve_lp(c_curr)
        delta = ns - curr_sum
        
        if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_sum = ns
            if ns > best_sum:
                best_sum = ns
                best_c = c_curr.copy()
                best_r = nr.copy()
            if delta > 0:
                step = min(0.07, step * 1.02)
            else:
                step = max(0.001, step * 0.98)
        else:
            c_curr[:] = old_centers
            step = max(0.001, step * 0.95)
            
        temp = max(1e-5, temp * 0.9993)
        
        # Periodic SLSQP polish to lock in improvements
        if it % 1200 == 0 and best_c is not None:
            p_pol = to_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
            try:
                res_p = minimize(obj, p_pol, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                                 options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if res_p.success and np.min(cons(res_p.x)) >= -1e-7:
                    s_p = -res_p.fun
                    if s_p > best_sum:
                        best_sum = s_p
                        best_p = res_p.x.copy()
                        c_curr, r_curr = from_params(best_p)
                        curr_sum = best_sum
                        best_c = c_curr.copy()
                        best_r = r_curr.copy()
            except Exception:
                pass
                
    if best_c is not None:
        c_curr = best_c
        r_curr, curr_sum = solve_lp(c_curr)
        best_p = to_params(c_curr, np.clip(r_curr * 0.995, 1e-6, 0.49))
        
    # Phase 3: Perturbation & SLSQP refinement
    for _ in range(25):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.015, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res_f = minimize(obj, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                             options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res_f.success and np.min(cons(res_f.x)) >= -1e-7:
                s_f = -res_f.fun
                if s_f > best_sum:
                    best_sum = s_f
                    best_p = res_f.x.copy()
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    try:
        res_hp = minimize(obj, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                          options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
        if res_hp.success and np.min(cons(res_hp.x)) >= -1e-8:
            best_p = res_hp.x
            best_sum = -res_hp.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_p)
    # Final LP pass to ensure radii are exactly optimal for the final centers
    radii_final, sum_final = solve_lp(centers)
    
    return centers, radii_final, float(sum_final)
