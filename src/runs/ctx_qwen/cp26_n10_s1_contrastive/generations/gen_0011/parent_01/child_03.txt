<strategy>
1. **More diverse initial configurations**: Add more hexagonal patterns with varied row counts, including asymmetric ones that might pack better in a square. Also add "corner-pinning" initializations where circles are placed in corners.

2. **Improved Simulated Annealing**: 
   - Use adaptive step size based on acceptance rate
   - Add occasional large jumps to escape deep local minima
   - Run longer with better temperature cooling

3. **Better SLSQP perturbation**: 
   - More aggressive perturbations to change contact topology
   - Multiple perturbation magnitudes to explore different scales of rearrangement

4. **Hybrid approach**: After SA finds a good center configuration, do LP for radii, then use SLSQP with the (r,u,v) parameterization for fine-tuning, then extract final radii via LP again.

5. **Corner optimization**: Explicitly try configurations that pin circles to corners, which is often optimal for sum of radii.

6. **Multiple independent runs**: Run several full optimization pipelines with different random seeds to find the best.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-5

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot=0.0, scale=1.0):
    """Generate hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.012, 0.012, pts.shape)
    return np.clip(pts, 0.015, 0.985)

def force_init(rng, seed_val):
    """Force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed_val)
    pts = rng_fd.rand(N, 2) * 0.7 + 0.15
    for _ in range(500):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-4
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 30.0 * np.maximum(0, 0.08 - pts[:, dim])
            f[:, dim] -= 30.0 * np.maximum(0, pts[:, dim] - 0.92)
        pts += 0.003 * f
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def corner_init(rng):
    """Initialize with circles pinned to corners and edges."""
    pts = np.array([
        [0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88],  # corners
        [0.37, 0.12], [0.63, 0.12], [0.37, 0.88], [0.63, 0.88],  # edge mid
        [0.12, 0.37], [0.12, 0.63], [0.88, 0.37], [0.88, 0.63],  # edge mid
        [0.37, 0.37], [0.63, 0.37], [0.37, 0.63], [0.63, 0.63],  # inner grid
        [0.50, 0.12], [0.50, 0.88], [0.12, 0.50], [0.88, 0.50],  # more edge
        [0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75],  # more inner
        [0.50, 0.50], [0.25, 0.50]  # center
    ])
    pts += rng.uniform(-0.02, 0.02, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def generate_inits(rng):
    """Generate diverse initial configurations."""
    inits = []
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [5,5,6,5,5], [7,5,5,5,4], [4,5,5,5,7], [5,7,5,5,4],
        [6,5,5,5,5], [5,5,5,6,5], [7,6,5,4,4], [6,7,5,5,3],
        [8,6,5,4,3], [4,4,6,6,6], [5,4,6,6,5], [6,4,5,6,5],
        [4,7,5,6,4], [5,4,7,5,5], [6,5,4,6,5], [5,6,6,5,4],
        [7,4,6,5,4], [4,6,7,5,4], [6,5,7,4,4], [5,6,4,5,6]
    ]
    
    for pat in patterns:
        for _ in range(4):
            rot = rng.uniform(-0.35, 0.35)
            sc = rng.uniform(0.88, 1.12)
            inits.append(hex_init(rng, pat, rot=rot, scale=sc))
    
    for s in range(25):
        inits.append(force_init(rng, seed_val=s))
    
    for _ in range(8):
        inits.append(corner_init(rng))
    
    return inits

def run_sa_on_centers(centers, rng, n_iter=8000):
    """Simulated annealing on centers with LP evaluation of radii."""
    curr_c = centers.copy()
    curr_r, curr_s = solve_lp_radii(curr_c)
    
    best_c = curr_c.copy()
    best_r = curr_r.copy()
    best_s = curr_s
    
    temp = 0.05
    step = 0.04
    accept_count = 0
    
    for it in range(n_iter):
        move_type = rng.integers(0, 6)
        old = None
        
        if move_type == 0:
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
        elif move_type == 1:
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        elif move_type == 2:
            i, j = rng.choice(N, 2, replace=False)
            old = curr_c[[i, j]].copy()
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        elif move_type == 3:
            # Move two circles together
            i, j = rng.choice(N, 2, replace=False)
            old = curr_c[[i, j]].copy()
            shift = rng.normal(0, step * 0.7, 2)
            curr_c[i] = np.clip(curr_c[i] + shift, 0.005, 0.995)
            curr_c[j] = np.clip(curr_c[j] + shift, 0.005, 0.995)
        elif move_type == 4:
            # Large jump for one circle
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] = rng.uniform(0.02, 0.98, 2)
        else:
            # Random restart of a few circles
            n_restart = rng.integers(2, 5)
            idxs = rng.choice(N, n_restart, replace=False)
            old = curr_c[idxs].copy()
            curr_c[idxs] = rng.uniform(0.05, 0.95, (n_restart, 2))
        
        _, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_s = ns
            accept_count += 1
            if ns > best_s:
                best_s = ns
                best_c = curr_c.copy()
                best_r, _ = solve_lp_radii(best_c)
            step = min(0.06, step * 1.02)
        else:
            if old is not None:
                if move_type < 2:
                    curr_c[idx] = old
                elif move_type == 2:
                    curr_c[[i, j]] = old
                elif move_type == 3:
                    curr_c[[i, j]] = old
                else:
                    curr_c[idxs] = old
            step = max(0.001, step * 0.96)
        
        # Adaptive temperature
        temp = max(1e-6, temp * 0.9993)
        
        # Adaptive step based on acceptance
        if (it + 1) % 200 == 0:
            acc_rate = accept_count / 200.0
            if acc_rate > 0.6:
                step = min(0.06, step * 1.3)
            elif acc_rate < 0.2:
                step = max(0.002, step * 0.6)
            accept_count = 0
    
    return best_c, best_r, best_s

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_p = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = generate_inits(rng)
    
    # Phase 1: Evaluate all inits with LP to find promising starting points
    init_scores = []
    for c in inits:
        r, s_val = solve_lp_radii(c)
        init_scores.append((s_val, c, r))
    init_scores.sort(key=lambda x: x[0], reverse=True)
    
    print(f"Best init score: {init_scores[0][0]:.6f}")
    
    # Phase 2: SLSQP refinement on top initial configurations
    top_starts = init_scores[:15]
    for s0, c0, r0 in top_starts:
        r_safe = np.clip(r0 * 0.995, 1e-6, 0.49)
        p0 = to_params(c0, r_safe)
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
    
    if best_p is None:
        c_f = init_scores[0][1]
        r_f = init_scores[0][2]
        best_p = to_params(c_f, np.clip(r_f * 0.99, 1e-6, 0.49))
        best_sum = -obj(best_p)
    
    c_best, r_best = from_params(best_p)
    
    # Phase 3: SA on centers with LP evaluation
    print(f"Starting SA with sum={best_sum:.6f}")
    sa_best_c, sa_best_r, sa_best_s = run_sa_on_centers(c_best, rng, n_iter=10000)
    
    if sa_best_s > best_sum:
        best_sum = sa_best_s
        c_best = sa_best_c
        r_best = sa_best_r
    
    print(f"After SA: sum={best_sum:.6f}")
    
    # Phase 4: SLSQP refinement on SA result
    r_safe = np.clip(r_best * 0.998, 1e-6, 0.49)
    p_sa = to_params(c_best, r_safe)
    try:
        res = minimize(obj, p_sa, method='SLSQP', bounds=bounds, constraints=cons_dict,
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(cons(res.x)) >= -1e-7:
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_p = res.x.copy()
                c_best, r_best = from_params(best_p)
    except Exception:
        pass
    
    # Phase 5: Iterative perturbation with SLSQP to escape local minima
    for k in range(40):
        if best_p is not None:
            xp = best_p.copy()
            # Varying perturbation magnitudes
            scale_r = 0.001 + 0.003 * (k % 3)
            scale_uv = 0.015 + 0.03 * (k % 5)
            
            xp[:N] += rng.normal(0, scale_r, N)
            xp[N:3*N] += rng.normal(0, scale_uv, 2*N)
            xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
            xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(obj, xp, method='SLSQP', bounds=bounds, constraints=cons_dict,
                               options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(cons(res.x)) >= -1e-7:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_p = res.x.copy()
                        c_best, r_best = from_params(best_p)
            except Exception:
                continue
        else:
            break
    
    # Phase 6: Global rotation/reflection escapes
    for trial in range(15):
        if best_p is None:
            break
        c_cur, r_cur = from_params(best_p)
        
        rot = rng.uniform(-0.15, 0.15)
        c_val, s_val = np.cos(rot), np.sin(rot)
        mat = np.array([[c_val, -s_val], [s_val, c_val]])
        c_rot = (c_cur - 0.5) @ mat.T + 0.5
        c_rot = np.clip(c_rot, 0.01, 0.99)
        
        r_rot, _ = solve_lp_radii(c_rot)
        p_rot = to_params(c_rot, np.clip(r_rot * 0.995, 1e-6, 0.49))
        
        try:
            res_r = minimize(obj, p_rot, method='SLSQP', bounds=bounds, constraints=cons_dict,
                             options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res_r.success and np.min(cons(res_r.x)) >= -1e-7:
                s_r = -res_r.fun
                if s_r > best_sum:
                    best_sum = s_r
                    best_p = res_r.x.copy()
                    c_best, r_best = from_params(best_p)
        except Exception:
            pass
    
    # Phase 7: Run SA again from the improved position
    if best_p is not None:
        c_cur, r_cur = from_params(best_p)
        sa2_c, sa2_r, sa2_s = run_sa_on_centers(c_cur, rng, n_iter=6000)
        if sa2_s > best_sum:
            best_sum = sa2_s
            c_best = sa2_c
            r_best = sa2_r
            best_p = to_params(c_best, np.clip(r_best * 0.998, 1e-6, 0.49))
    
    # Phase 8: Final high-precision polish
    if best_p is not None:
        try:
            res_f = minimize(obj, best_p, method='SLSQP', bounds=bounds, constraints=cons_dict,
                             options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(cons(res_f.x)) >= -1e-8:
                best_p = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass
    
    # Fallback if everything failed
    if best_p is None:
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        r_f = np.full(N, 0.085)
        best_p = to_params(pts, r_f)
        best_sum = np.sum(best_p[:N])
    
    centers, radii = from_params(best_p)
    
    # Final LP to ensure optimal radii for these exact centers
    radii_final, sum_final = solve_lp_radii(centers)
    
    return centers, radii_final, float(sum_final)
