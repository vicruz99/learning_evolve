To maximize the sum of radii for 26 circles in a unit square, we exploit the fact that for fixed centers, the optimal radii assignment is a Linear Programming (LP) problem. The overall problem is non-convex and piecewise-linear with respect to center positions. 

My strategy combines several complementary optimization techniques:
1. **LP-based Objective**: For any set of centers, we solve an LP to find the exact optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints. This gives a precise evaluation of any center configuration.
2. **Diverse Initialization**: Generate multiple hexagonal lattice patterns with varied row counts, rotations, and scales, plus force-directed layouts and perturbed grids. This covers the most promising topological basins.
3. **Simulated Annealing (SA) on Centers**: Use the LP sum as the energy function. SA allows occasional uphill moves to escape local minima and explore new contact graphs. Adaptive step sizing and temperature cooling ensure thorough exploration.
4. **Coordinate Descent (2D Optimization)**: Sequentially optimize each circle's position using Nelder-Mead, holding others fixed. This exploits the local concavity of the LP envelope and rapidly converges to strong local optima.
5. **SLSQP Refinement with Boundary-Safe Parameterization**: Convert $(x, y, r)$ to $(r, u, v)$ where $u, v \in [0,1]$ parameterize the feasible region for centers given a radius. This automatically satisfies boundary constraints, letting SLSQP focus purely on pairwise non-overlap constraints for high-precision polishing.
6. **Final LP Projection**: After all geometric optimizations, project back to exact optimal radii via LP to guarantee strict feasibility and maximal sum.

This pipeline systematically explores the configuration space, refines promising regions, and guarantees valid, high-scoring outputs.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons_slqp(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def optimize_center_2d(pt, centers, idx):
    """Objective for 2D optimization of a single center."""
    pt_clipped = np.clip(pt, 0.001, 0.999)
    tc = centers.copy()
    tc[idx] = pt_clipped
    _, s = solve_lp_radii(tc)
    return -s

def run_coord_descent(centers, rng):
    """Sequentially optimize each center using Nelder-Mead."""
    best_sum = solve_lp_radii(centers)[1]
    for _ in range(6):
        for i in range(N):
            curr_pos = centers[i].copy()
            try:
                res = minimize(optimize_center_2d, curr_pos, args=(centers, i),
                               method='Nelder-Mead', options={'xatol': 1e-6, 'fatol': 1e-9, 'maxiter': 400})
                if res.success:
                    new_pos = res.x
                    temp_c = centers.copy()
                    temp_c[i] = new_pos
                    _, new_sum = solve_lp_radii(temp_c)
                    if new_sum > best_sum + 1e-8:
                        centers[i] = new_pos
                        best_sum = new_sum
            except Exception:
                continue
    return centers

def hex_init(rng, row_counts, rot, scale):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.012, 0.012, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(250):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 15.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 15.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.002 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def grid_init(rng):
    """Generate a perturbed grid initialization."""
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-0.02, 0.02, (N, 2))
    return np.clip(pts, 0.05, 0.95)

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], 
            [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
            [5,5,5,5,4,2], [6,5,5,5,3,2], [5,6,5,4,4,2], [7,6,5,4,3]]
    for pat in pats:
        for _ in range(4):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.35, 0.35), scale=rng.uniform(0.88, 1.12)))
    for _ in range(12):
        inits.append(force_init(rng))
    for _ in range(6):
        inits.append(grid_init(rng))
        
    # Phase 2: SLSQP refinement on initial configs
    for c in inits:
        r, s_val = solve_lp_radii(c)
        r = np.clip(r * 0.992, 1e-6, 0.49)
        p0 = to_params(c, r)
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)
        best_p = to_params(c_f, r_f * 0.99)
        best_sum = -obj_slqp(best_p)
        
    c_best, r_best = from_params(best_p)
    best_c = c_best.copy()
    
    # Phase 3: LP-driven Simulated Annealing on centers
    curr_c = c_best.copy()
    curr_s = best_sum
    step = 0.035
    temp = 0.045
    
    for it in range(6000):
        move_type = rng.integers(0, 3)
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        if move_type == 0:
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        elif move_type == 1:
            curr_c[idx] = rng.uniform(0.15, 0.85, 2)
        else:
            # Perturb multiple circles slightly
            for _ in range(3):
                i = rng.integers(N)
                curr_c[i] += rng.normal(0, step * 0.5, 2)
                curr_c[i] = np.clip(curr_c[i], 0.02, 0.98)
                
        _, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                step = min(0.05, step * 1.06)
            else:
                step = max(0.001, step * 0.996)
        else:
            if move_type < 2:
                curr_c[idx] = old
            else:
                # Revert multi-move is complex, just accept step reduction
                pass
            step = max(0.001, step * 0.96)
            
        temp = max(1e-5, temp * 0.9993)
        
    # Phase 4: Coordinate Descent to polish local optima
    if best_c is not None:
        c_best = best_c
        c_best = run_coord_descent(c_best, rng)
        r_best, best_sum = solve_lp_radii(c_best)
        best_p = to_params(c_best, np.clip(r_best * 0.996, 1e-6, 0.49))
        
    # Phase 5: Perturbation & SLSQP to escape remaining local minima
    for k in range(20):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.018, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue

        rot = rng.uniform(-0.12, 0.12)
        sc = rng.uniform(0.96, 1.04)
        c_val, s_val = np.cos(rot), np.sin(rot)
        mat = np.array([[c_val, -s_val], [s_val, c_val]])
        c_rot = (c_best - 0.5) @ mat.T * sc + 0.5
        c_rot = np.clip(c_rot, 0.02, 0.98)
        
        r_rot, _ = solve_lp_radii(c_rot)
        p_rot = to_params(c_rot, np.clip(r_rot * 0.995, 1e-6, 0.49))
        try:
            res_r = minimize(obj_slqp, p_rot, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                             options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
            if res_r.success and np.min(cons_slqp(res_r.x)) >= -1e-7:
                s_r = -res_r.fun
                if s_r > best_sum:
                    best_sum = s_r
                    best_p = res_r.x.copy()
                    c_best, _ = from_params(best_p)
        except Exception:
            pass
            
    # Phase 7: High-precision final polish
    try:
        res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})

        pass
        
    # Final projection to exact optimal radii
    centers_final, radii_final = from_params(best_p)
    radii_final, sum_final = solve_lp_radii(centers_final)
    
    return centers_final, radii_final, float(sum_final)
