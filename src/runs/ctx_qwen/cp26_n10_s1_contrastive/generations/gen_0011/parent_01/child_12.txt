import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons_slqp(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], 
            [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5]]
    for pat in pats:
        for _ in range(3):
            pts = []
            r_est = 0.095
            y = r_est
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N: pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.9, 1.1) + 0.5
            rot = rng.uniform(-0.3, 0.3)
            c, s = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pts += rng.uniform(-0.015, 0.015, pts.shape)
            inits.append(np.clip(pts, 0.02, 0.98))
            
    # Force-directed layouts
    for _ in range(12):
        pts = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(200):
            f = np.zeros_like(pts)
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            d = np.sqrt(np.sum(diff**2, axis=2))
            d = np.maximum(d, 1e-4)
            f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
            for dim in range(2):
                f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
                f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
            pts += 0.0025 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    return inits

def run_sa(rng, centers, iterations=10000):

    curr_c = centers.copy()
    _, curr_s = solve_lp_radii(curr_c)
    best_c = curr_c.copy()
    best_s = curr_s
    
    temp = 0.045
    step = 0.035
    
    for it in range(iterations):
        move_type = rng.integers(0, 5)
        old = None
        idx = rng.integers(N)
        
        if move_type == 0:
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
        elif move_type == 1:
            old = curr_c[idx].copy()
            curr_c[idx] = rng.uniform(0.1, 0.9, 2)
        elif move_type == 2:
            old = curr_c.copy()
            i, j = rng.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        elif move_type == 3:
            old = curr_c.copy()
            rot = rng.uniform(-0.04, 0.04)
            c, s = np.cos(rot), np.sin(rot)
            mat = np.array([[c, -s], [s, c]])
            curr_c = (curr_c - 0.5) @ mat.T + 0.5
            
        curr_c = np.clip(curr_c, 0.005, 0.995)
        
        _, new_s = solve_lp_radii(curr_c)
        delta = new_s - curr_s

            curr_s = new_s
            if new_s > best_s:
                best_s = new_s
                best_c = curr_c.copy()
                step = min(0.05, step * 1.02)
            else:
                step = max(0.001, step * 0.995)
        else:
            if old is not None:
                curr_c = old
            step = max(0.001, step * 0.95)
            
        temp = max(1e-6, temp * 0.9992)
        
    return best_c, best_s

def run_jiggle(rng, centers, max_iter=120):
    """Greedy coordinate-wise local search to tighten packing."""
    curr_c = centers.copy()
    _, best_s = solve_lp_radii(curr_c)
    improved = True
    
    while improved and max_iter > 0:
        improved = False
        max_iter -= 1
        for i in range(N):
            old = curr_c[i].copy()
            directions = rng.normal(0, 0.006, (8, 2))
            best_dir_s = best_s
            best_dir_c = old
            
            for d in directions:
                curr_c[i] = np.clip(old + d, 0.005, 0.995)
                _, s = solve_lp_radii(curr_c)
                if s > best_dir_s:
                    best_dir_s = s
                    best_dir_c = curr_c[i].copy()
                    
            if best_dir_s > best_s:
                curr_c[i] = best_dir_c
                best_s = best_dir_s
                improved = True
            else:
                curr_c[i] = old
                
    return curr_c, best_s
