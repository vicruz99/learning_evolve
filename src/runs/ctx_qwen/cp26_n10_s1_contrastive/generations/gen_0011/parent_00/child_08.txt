

<strategy>
1. **Diverse Initialization**: Generate a large set of starting configurations using hexagonal lattices (with varied row counts, rotations, and scales), force-directed repulsion layouts, and perturbed grids. This covers multiple topological basins.
2. **LP-Based Filtering**: For each initial configuration, solve a Linear Program (LP) to find the optimal radii. This provides a strict upper bound on the sum of radii for that fixed topology. Keep the top performers.
3. **SLSQP Refinement**: Use the $(r, u, v)$ parameterization to run SLSQP on the best candidates. This parameterization inherently satisfies boundary constraints ($r \le x \le 1-r$), leaving only pairwise non-overlap constraints. This phase simultaneously optimizes positions and radii.
4. **Simulated Annealing (SA)**: Perform SA on the best configuration found. In each step, perturb a random circle's center, re-solve the LP, and accept/reject via the Metropolis criterion. Include occasional "shock" moves that reposition circles to boundaries or swap positions to escape local minima.
5. **Perturbation & Restart Loop**: After SA, repeatedly apply targeted perturbations to the $(r, u, v)$ parameters and run SLSQP again. This explores the feasible region boundaries where the global optimum typically resides.
6. **High-Precision Polish**: Run a final SLSQP with tight tolerances and high iteration limits to squeeze out numerical slack.
7. **Robustness & Efficiency**: Vectorize constraint evaluations, use `method='highs'` for fast LP solves, and ensure all functions are top-level without closures as required.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    # Precompute pairwise distances
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Build LP constraints
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.ones(n) * 1e-6, 1e-4

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints: dist^2 >= (r_i + r_j)^2. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns with varied row counts
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [6,6,5,5,4],
        [5,5,6,5,5], [4,5,5,5,7], [5,6,4,6,5]
    ]
    
    for pat in patterns:
        for s in range(4):
            pts = []
            r_est = 0.095
            y = r_est
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
                
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.85, 1.15) + 0.5
            
            rot = rng.uniform(-0.35, 0.35)
            c_val, s_val = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5
            
            pts += rng.uniform(-0.02, 0.02, pts.shape)
            inits.append(np.clip(pts, 0.02, 0.98))
            
    # 2. Force-directed layouts for organic packings
    for s in range(20):
        rng_fd = np.random.RandomState(s)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        for _ in range(300):
            f = np.zeros_like(pts)
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-4
            f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
            pts += 0.002 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # 3. Perturbed grids
    for s in range(10):
        rng_g = np.random.RandomState(s)
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng_g.uniform(-0.03, 0.03, (N, 2))
        inits.append(np.clip(pts, 0.05, 0.95))
        
    return inits

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    
    inits = generate_inits(rng)
    
    # Evaluate all inits with LP to find the most promising starting topologies
    init_scores = []
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)
        init_scores.append((s0, c0, r0))
    init_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Phase 1: SLSQP refinement on top initial configurations
    top_starts = init_scores[:15]
    for s0, c0, r0 in top_starts:
        p0 = to_params(c0, np.clip(r0 * 0.995, 1e-6, 0.49))
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization failed completely
    if best_p is None:
        best_p = to_params(init_scores[0][1], init_scores[0][2] * 0.99)
        best_sum = -obj_slqp(best_p)
        
    # Phase 2: LP-Driven Simulated Annealing on Centers to explore topology
    curr_c, curr_r = from_params(best_p)
    curr_s = best_sum
    temp = 0.05
    step = 0.04
    
    for it in range(12000):
        # Occasional shock: replace a random circle with a boundary-corner attempt
        if rng.random() < 0.03:
            idx = rng.integers(N)
            corner_choice = rng.integers(4)
            if corner_choice == 0: curr_c[idx] = rng.uniform(0.02, 0.15, 2)
            elif corner_choice == 1: curr_c[idx] = np.array([rng.uniform(0.85, 0.98), rng.uniform(0.02, 0.15)])
            elif corner_choice == 2: curr_c[idx] = np.array([rng.uniform(0.02, 0.15), rng.uniform(0.85, 0.98)])
            else: curr_c[idx] = rng.uniform(0.85, 0.98, 2)
        else:
            idx = rng.integers(N)
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_p = to_params(curr_c, np.clip(nr * 0.995, 1e-6, 0.49))
        else:
            # Revert shock moves completely to maintain SA integrity on continuous space
            pass
            
        temp *= 0.9994
        step = max(0.0005, step * 0.9995)
        
    # Phase 3: Perturbation & SLSQP restarts to escape local minima
    for k in range(50):
        xp = best_p.copy()
        # Perturb radii slightly, but aggressively perturb positions
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.025, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-9:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_p)
    return centers, np.maximum(radii, 0.0), float(best_sum)
```
