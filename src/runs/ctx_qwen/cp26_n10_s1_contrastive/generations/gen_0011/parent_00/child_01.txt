

<strategy>
1. **Parameterization**: Use the $(r, u, v)$ mapping where $x = r + u(1-2r)$ and $y = r + v(1-2r)$ with $u,v \in [0,1]$. This automatically enforces boundary constraints $r \le x,y \le 1-r$, allowing the optimizer to focus purely on non-overlap constraints.
2. **Diverse Initialization**: Generate hexagonal lattice patterns with varied row distributions, rotations, and scales, combined with force-directed layouts. This covers multiple topological arrangements known to be efficient for dense packing.
3. **LP Evaluation**: For fixed centers, solve a Linear Program to find radii that maximize $\sum r_i$. This is fast and exact for the center-only subspace.
4. **Multi-Phase Optimization**:
   - *Phase 1*: SLSQP refinement on the top scoring initial configurations.
   - *Phase 2*: LP-driven Simulated Annealing on centers. This explores topology changes via single-point moves, random jumps, and pairwise swaps, using the LP to evaluate moves efficiently.
   - *Phase 3*: Perturbation restarts with SLSQP to escape shallow local minima found by SA.
   - *Phase 4*: High-precision final polish with tight tolerances.
5. **Final Validation**: Re-run the LP on the final optimized centers to guarantee radii are mathematically optimal for the exact geometry, ensuring maximum sum of radii.
This strategy systematically balances global exploration with local precision, specifically targeting the small gap remaining in the current best solution.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons(params):
    """Inequality constraints: dist^2 >= (r_i + r_j)^2. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [6,6,5,5,4],
        [5,5,6,5,5], [4,5,5,5,7], [5,5,5,6,5], [7,6,5,4,4]
    ]
    for pat in patterns:
        for _ in range(4):
            pts = []
            r_est = 0.095
            y = r_est
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.88, 1.12) + 0.5
            rot = rng.uniform(-0.35, 0.35)
            c_val, s_val = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5
            pts += rng.uniform(-0.02, 0.02, pts.shape)
            inits.append(np.clip(pts, 0.02, 0.98))
            
    for s in range(12):
        rng_fd = np.random.RandomState(s)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        for _ in range(250):
            f = np.zeros_like(pts)
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-4
            f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
            pts += 0.004 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    return inits

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_p = None
    best_sum = -np.inf
    
    inits = generate_inits(rng)
    
    # Evaluate inits with LP to find strong starting topologies
    init_scores = []
    for c0 in inits:
        r0, s0 = solve_lp(c0)
        init_scores.append((s0, c0, r0))
    init_scores.sort(reverse=True)
    
    # Phase 1: SLSQP refinement on top initial configurations
    top_count = min(8, len(init_scores))
    for idx_score in range(top_count):
        s0, c0, r0 = init_scores[idx_score]
        p0 = to_params(c0, np.clip(r0 * 0.995, 1e-6, 0.49))
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-8:
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_p = res.x.copy()
        except Exception:
            pass
            
    if best_p is None:
        s0, c0, r0 = init_scores[0]
        best_p = to_params(c0, r0 * 0.99)
        best_sum = -obj(best_p)
        
    # Phase 2: LP-Driven Simulated Annealing on Centers
    curr_c, curr_r = from_params(best_p)
    curr_s = best_sum
    temp = 0.05
    step = 0.04
    
    for it in range(6000):
        move_type = rng.integers(0, 3)
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        if move_type == 0:
            curr_c[idx] += rng.normal(0, step, 2)
        elif move_type == 1:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            j = rng.integers(0, N)
            while j == idx:
                j = rng.integers(0, N)
            temp_pos = curr_c[idx].copy()
            curr_c[idx] = curr_c[j].copy()
            curr_c[j] = temp_pos
            
        curr_c = np.clip(curr_c, 0.005, 0.995)
        
        nr, ns = solve_lp(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_p = to_params(curr_c, np.clip(nr * 0.995, 1e-6, 0.49))
                step = min(0.05, step * 1.02)
        else:
            if move_type < 2:
                curr_c[idx] = old
            step = max(0.001, step * 0.998)
            
        temp *= 0.9993
        
    # Phase 3: Perturbation & SLSQP restarts to escape local minima
    for k in range(40):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.uniform(-0.025, 0.025, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            pass
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons(res_f.x)) >= -1e-9:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_p)
    # Final LP ensures radii are strictly optimal for the exact final centers
    radii, best_sum = solve_lp(centers)
    
    return centers, np.maximum(radii, 0.0), float(best_sum)
