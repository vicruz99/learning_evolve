

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we employ a hybrid optimization strategy combining Linear Programming (LP), Sequential Least Squares Quadratic Programming (SLSQP), and Simulated Annealing (SA). 

1. **Parameterization**: We use a boundary-safe parameterization $x_i = r_i + u_i(1-2r_i)$ and $y_i = r_i + v_i(1-2r_i)$ where $u_i, v_i \in [0, 1]$. This automatically guarantees that each circle stays within the unit square, reducing the optimization problem to only pairwise non-overlap constraints: $\text{dist}_{ij}^2 \ge (r_i + r_j)^2$.

2. **Linear Programming for Radii**: Given fixed centers, the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints can be found efficiently via LP. This is used to evaluate center configurations during the SA phase.

3. **Diverse Initialization**: We generate multiple starting configurations using perturbed hexagonal lattices (various row counts, rotations, scales), force-directed layouts, and perturbed grids. Each is refined with SLSQP.

4. **Simulated Annealing on Centers**: We run SA on the circle centers, evaluating each move by solving the LP for radii. This helps escape local minima by exploring different contact topologies.

5. **Iterative Refinement**: The best configuration found is polished with multiple rounds of SLSQP with random perturbations to the $(r, u, v)$ parameters, followed by a global rotation escape and a final high-precision SLSQP polish. A final LP ensures radii are exactly optimal for the returned centers.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise distances
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-4

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_func(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def grid_init(rng):
    """Generates a perturbed grid initialization."""
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-0.025, 0.025, (N, 2))
    return np.clip(pts, 0.05, 0.95)

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_func}
    
    best_p = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], 
        [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
        [5,6,4,6,5], [4,5,6,5,6], [6,5,5,5,6], [5,5,5,6,5], [7,6,5,4,3]
    ]
    for pat in pats:
        for _ in range(5):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.9, 1.1)))
    for _ in range(15):
        inits.append(force_init(rng))
    for _ in range(8):
        inits.append(grid_init(rng))
        
    # Phase 1: Broad SLSQP search from diverse initializations
    for c in inits:
        r, s_val = solve_lp_radii(c)
        r = np.clip(r * 0.99, 1e-6, 0.49)
        p0 = to_params(c, r)
        try:
            res = minimize(obj_func, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_func(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)
        best_p = to_params(c_f, r_f * 0.99)
        best_sum = -obj_func(best_p)
        
    c_best, r_best = from_params(best_p)
    best_c = c_best.copy()
    
    # Phase 2: LP-driven Simulated Annealing on centers to explore topologies
    curr_c = c_best.copy()
    curr_s = best_sum
    step = 0.03
    temp = 0.04
    
    for it in range(6000):
        move_type = rng.integers(0, 4)
        if move_type == 0:
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        elif move_type == 1:
            idx = rng.integers(N)
            curr_c[idx] = rng.uniform(0.1, 0.9, 2)
            old = None
        else:
            i, j = rng.choice(N, 2, replace=False)
            old = curr_c[[i, j]].copy()
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
            
        _, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                step = min(0.05, step * 1.05)
            else:
                step = max(0.001, step * 0.995)
        else:
            if old is not None:
                if move_type < 2:
                    curr_c[idx] = old
                else:
                    curr_c[[i, j]] = old
            step = max(0.001, step * 0.95)
            
        temp = max(1e-5, temp * 0.9995)
        
    if best_c is not None:
        c_best = best_c
        r_best, best_sum = solve_lp_radii(c_best)
        best_p = to_params(c_best, np.clip(r_best * 0.995, 1e-6, 0.49))
        
    # Phase 3: Perturbation & SLSQP restarts to escape local minima
    for k in range(30):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.02, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_func, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_func(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 4: Global rotation escapes
    for _ in range(12):
        rot = rng.uniform(-0.1, 0.1)
        c_val, s_val = np.cos(rot), np.sin(rot)
        mat = np.array([[c_val, -s_val], [s_val, c_val]])
        c_rot = (c_best - 0.5) @ mat.T + 0.5
        c_rot = np.clip(c_rot, 0.01, 0.99)
        r_rot, _ = solve_lp_radii(c_rot)
        p_rot = to_params(c_rot, np.clip(r_rot * 0.995, 1e-6, 0.49))
        try:
            res_r = minimize(obj_func, p_rot, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                             options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res_r.success and np.min(cons_func(res_r.x)) >= -1e-7:
                s_r = -res_r.fun
                if s_r > best_sum:
                    best_sum = s_r
                    best_p = res_r.x.copy()
        except Exception:
            pass
            
    # Phase 5: High-precision final polish
    try:
        res_f = minimize(obj_func, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_func(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_p)
    # Final LP to ensure radii are exactly optimal for the exact final centers
    radii_final, sum_final = solve_lp_radii(centers)
    
    return centers, radii_final, float(sum_final)
