1. **LP for Radii**: Given fixed centers, a linear program exactly maximizes sum of radii subject to boundary and pairwise non-overlap constraints. This is much more robust than joint optimization.
2. **Boundary-Safe Parameterization**: Instead of optimizing `(x, y, r)` with explicit boundary constraints, we use `x = r + u(1-2r)`, `y = r + v(1-2r)` with `u, v ∈ [0, 1]`. This automatically guarantees `r ≤ x ≤ 1-r` and `r ≤ y ≤ 1-r`, reducing constraints to only pairwise non-overlap.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def get_params(centers, radii):
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def get_centers_radii(params):
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def cons_func(params):
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def obj_func(params):
    return -np.sum(params[0::3])

def hex_init(rng, row_counts, rot, scale):
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def grid_init(rng):
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-0.025, 0.025, (N, 2))
    return np.clip(pts, 0.05, 0.95)

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_func}

    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None

    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6],
            [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5]]
    for pat in pats:
        for _ in range(5):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.9, 1.1)))
    for _ in range(15):
        inits.append(force_init(rng))
    for _ in range(8):
        inits.append(grid_init(rng))

    # Phase 1: SLSQP on diverse inits
    for c in inits:
        r_init, _ = solve_lp_radii(c)
        r_init = np.clip(r_init * 0.99, 1e-6, 0.49)
        p0 = get_params(c, r_init)
        try:
            res = minimize(obj_func, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_func(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue

    if best_p is not None:
        best_c, best_r = get_centers_radii(best_p)
        best_c = best_c.copy()
        best_r = best_r.copy()
        best_sum = np.sum(best_r)

        # Phase 2: LP-driven Simulated Annealing on centers
        curr_c = best_c.copy()
        curr_s = best_sum
        temp = 0.04
        step = 0.035

        for it in range(6000):
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)

            nr, ns = solve_lp_radii(curr_c)
            delta = ns - curr_s

            if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
                curr_s = ns
                if ns > best_sum:
                    best_sum = ns
                    best_c = curr_c.copy()
                    best_r = nr.copy()
            else:
                curr_c[idx] = old

            temp *= 0.9993
            step = max(0.001, step * 0.9994)

        # Update best_p after SA
        best_r, best_sum = solve_lp_radii(best_c)
        best_p = get_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))

        # Phase 3: Perturbation & SLSQP restarts
        for k in range(30):
            xp = best_p.copy()
            xp[0::3] += rng.normal(0, 0.002, N)
            xp[1::3] += rng.normal(0, 0.02, N)
            xp[2::3] += rng.normal(0, 0.02, N)
            xp[0::3] = np.clip(xp[0::3], 1e-6, 0.49)
            xp[1::3] = np.clip(xp[1::3], 0.0, 1.0)
            xp[2::3] = np.clip(xp[2::3], 0.0, 1.0)

            try:
                res = minimize(obj_func, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                               options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(cons_func(res.x)) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_p = res.x.copy()
                        best_c, best_r = get_centers_radii(best_p)
            except Exception:
                continue

        # Phase 4: High-precision polish
        try:
            res_f = minimize(obj_func, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                             options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(cons_func(res_f.x)) >= -1e-9:
                best_p = res_f.x
                best_c, best_r = get_centers_radii(best_p)
                best_sum = np.sum(best_r)
        except Exception:
            pass

    # Fallback
    if best_c is None:
        best_c = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_r, best_sum = solve_lp_radii(best_c)

    # Final LP to ensure radii are optimal for exact centers
    best_r_final, best_sum_final = solve_lp_radii(best_c)

    return best_c, np.maximum(best_r_final, 0.0), float(best_sum_final)
