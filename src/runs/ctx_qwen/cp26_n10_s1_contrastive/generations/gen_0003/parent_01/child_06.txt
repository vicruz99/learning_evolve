import numpy as np
from scipy.optimize import minimize, differential_evolution

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def penalty_objective(v, weight):
    """Objective for Differential Evolution with quadratic penalty for constraints."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    obj = -np.sum(r)
    
    # Boundary violations: r <= x <= 1-r, r <= y <= 1-r
    v1 = np.minimum(0.0, x - r)
    v2 = np.minimum(0.0, 1.0 - x - r)
    v3 = np.minimum(0.0, y - r)
    v4 = np.minimum(0.0, 1.0 - y - r)
    p_bound = np.sum(v1**2 + v2**2 + v3**2 + v4**2)
    
    # Pairwise separation violations: dist >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    v_pair = np.minimum(0.0, dist - r_sum)
    p_pair = np.sum(v_pair[I_IDX, J_IDX]**2)
    
    return obj + weight * (p_bound + p_pair)

def constraints_slsqp(v):
    """Exact inequality constraints for SLSQP: g(v) >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def objective_slsqp(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def run_de_search(seed, weight):
    """Runs Differential Evolution to find a high-quality starting configuration."""
    bounds = []
    for _ in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.01, 0.5)])
        
    res = differential_evolution(
        func=penalty_objective,
        bounds=bounds,
        args=(weight,),
        popsize=40,
        maxiter=200,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        tol=1e-7,
        polish=False
    )
    return res.x
