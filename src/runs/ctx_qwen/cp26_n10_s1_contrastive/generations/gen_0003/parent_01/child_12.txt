

<strategy>
The problem of maximizing the sum of radii for packed circles in a square can be reformulated elegantly: for any fixed set of centers, the optimal radii are found by solving a linear program (LP). The LP maximizes $\sum r_i$ subject to $r_i + r_j \le \text{dist}(c_i, c_j)$ and $r_i \le \text{dist}(c_i, \text{boundary})$. This separates the combinatorial geometry of placement from the sizing.

Crucially, the objective function $F(c_1, \dots, c_N) = \max \sum r_i$ subject to distance constraints is a concave function of the centers (since distances are concave and the LP maximizes a linear function of concave terms). Maximizing a concave function over a convex domain means local optima are likely global, and derivative-free methods work exceptionally well.

Strategy:
1. **LP Oracle**: Implement a fast LP solver that takes fixed centers and returns the maximum sum of radii and the corresponding radii.
2. **Global & Local Search**: Use `Nelder-Mead` from diverse structured initializations (hexagonal lattices with shifts/rotations) to quickly climb to high-quality local optima.
3. **Differential Evolution**: Run a moderate-population DE to escape any remaining basins and explore the full 52-dimensional center space.
4. **Perturbation Refinement**: Kick the best solution slightly and re-optimize to polish tight contacts.
5. This avoids the non-linear constraint handling pitfalls of SLSQP, leverages the concavity of the problem, and is highly robust.
</strategy>

```python
import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

N = 26
PAIR_IDX = []
for i in range(N):
    for j in range(i + 1, N):
        PAIR_IDX.append((i, j))

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    k = len(PAIR_IDX)
    A_ub = np.zeros((4 * n + k, n))
    b_ub = np.zeros(4 * n + k)

    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        A_ub[4 * i, i] = 1.0
        A_ub[4 * i + 1, i] = 1.0
        A_ub[4 * i + 2, i] = 1.0
        A_ub[4 * i + 3, i] = 1.0
        b_ub[4 * i] = centers[i, 0]
        b_ub[4 * i + 1] = 1.0 - centers[i, 0]
        b_ub[4 * i + 2] = centers[i, 1]
        b_ub[4 * i + 3] = 1.0 - centers[i, 1]

    # Pairwise constraints: r_i + r_j <= dist(c_i, c_j)
    for idx, (i, j) in enumerate(PAIR_IDX):
        row = 4 * n + idx
        A_ub[row, i] = 1.0
        A_ub[row, j] = 1.0
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b_ub[row] = np.sqrt(dx * dx + dy * dy)

    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def neg_sum_lp(centers_flat):
    """Objective for optimizers: negative max sum of radii given centers."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 0.0, 1.0)
    s, _ = solve_lp(centers)
    return -s

def get_hex_init(seed, shift_x=0.0, shift_y=0.0):
    """Generates a hexagonal lattice initialization with controlled shifts."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est + shift_y
    row = 0
    while len(pts) < N:
        x_start = r_est + shift_x + (r_est if row % 2 == 1 else 0.0)
        x = x_start
        while x < 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Phase 1: Structured initializations for fast local climbs
    inits = []
    for s in range(10):
        inits.append(get_hex_init(seed=s))
        inits.append(get_hex_init(seed=s, shift_x=0.04, shift_y=0.04))
        inits.append(get_hex_init(seed=s, shift_x=-0.02, shift_y=0.06))
    for s in range(10):
        rng = np.random.RandomState(s)
        inits.append(np.clip(rng.rand(N, 2), 0.05, 0.95))

    for pts in inits:
        x0 = pts.flatten()
        try:
            res = minimize(neg_sum_lp, x0, method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-10})
            centers_opt = res.x.reshape(N, 2)
            s, r = solve_lp(centers_opt)
            if s > best_sum:
                best_sum = s
                best_centers = centers_opt.copy()
                best_radii = r.copy()
        except Exception:
            pass

    # Phase 2: Differential Evolution for global exploration
    bounds_de = [(0.0, 1.0)] * (2 * N)
    try:
        res_de = differential_evolution(neg_sum_lp, bounds_de, popsize=15, maxiter=150,
                                        tol=1e-7, seed=42, polish=False)
        centers_de = res_de.x.reshape(N, 2)
        s, r = solve_lp(centers_de)
        if s > best_sum:
            best_sum = s
            best_centers = centers_de.copy()
            best_radii = r.copy()
    except Exception:
        pass

    # Phase 3: Perturbation refinement to escape shallow local minima
    if best_centers is not None:
        for _ in range(25):
            rng = np.random.RandomState(None)
            pert = best_centers + rng.normal(0, 0.004, (N, 2))
            pert = np.clip(pert, 0.02, 0.98)
            try:
                res = minimize(neg_sum_lp, pert.flatten(), method='Nelder-Mead',
                               options={'maxiter': 3000, 'xatol': 1e-8, 'fatol': 1e-10})
                centers_opt = res.x.reshape(N, 2)
                s, r = solve_lp(centers_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers_opt.copy()
                    best_radii = r.copy()
            except Exception:
                pass

    # Fallback configuration
    if best_centers is None:
        best_centers = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N],
                                        np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        best_radii, best_sum = np.full(N, 0.01), 0.26

    return best_centers, best_radii, float(best_sum)
```
