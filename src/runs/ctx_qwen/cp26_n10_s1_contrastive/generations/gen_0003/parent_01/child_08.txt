import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_constraints(v):
    """
    Computes inequality constraints g(v) >= 0.
    Includes boundary containment and pairwise non-overlap.
    """
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    # Only upper triangular pairs to avoid duplicates and self-comparison
    i_idx, j_idx = np.triu_indices(N, k=1)
    c.append(dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    
    return np.concatenate(c)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def solve_radii_lp(centers):
    """
    Given fixed centers, solves the LP to find radii that maximize sum(r_i).
    Constraints:
    1. r_i >= 0
    2. r_i <= distance to boundaries
    3. r_i + r_j <= distance(centers[i], centers[j])
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        bounds_val = np.array([x, 1.0 - x, y, 1.0 - y])
        for b in bounds_val:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 2.6e-4

def center_objective(centers_flat):
    """Objective for center-only optimization: maximize LP-computed sum of radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 0.001, 0.999)
    _, s = solve_radii_lp(centers)
    if np.isnan(s) or s < 0:
        return 1e6
    return -s

def generate_hex_init(seed, scale=1.0, rot=0.0):
    """Generate a hexagonal lattice initialization with rotation and jitter."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095 * scale
    y = r_est
    row = 0
    while len(pts) < N and y < 1.0 - r_est:
        x_start = r_est + (row % 2) * r_est
        x = x_start
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    while len(pts) < N:
        pts.append(rng.uniform(r_est, 1.0-r_est, 2).tolist())
        
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    
    # Apply rotation
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ rot_mat.T + 0.5
        
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def force_init(seed):
    """Spread points using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(150):
        forces = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-5)
        f = 1.0 / (dists**2)
        f = np.clip(f, 0, 15.0)
        forces += np.sum(f[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in [0, 1]:
            mask = pts[:, d] < 0.15
            forces[mask, d] += 8.0 * (0.15 - pts[mask, d])
            mask = pts[:, d] > 0.85
            forces[mask, d] -= 8.0 * (pts[mask, d] - 0.85)
            
        pts += 0.005 * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    best_sum = -np.inf
    best_v = None
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    inits = []
    # Hex variations with different scales, rotations, and seeds
    for s in range(25):
        for sc in [0.95, 1.0, 1.05]:
            for rot in [-0.1, 0.0, 0.1]:
                inits.append(generate_hex_init(s, scale=sc, rot=rot))
                
    # Force variations
    for s in range(15):
        inits.append(force_init(s))
        
    # Random spread
    for s in range(10):
        rng = np.random.RandomState(s)
        pts = rng.uniform(0.05, 0.95, (N, 2))
        inits.append(pts)

    # Phase 1: SLSQP on full variables from diverse starts
    for idx, pts in enumerate(inits):
        radii, _ = solve_radii_lp(pts)
        radii = np.maximum(radii, 1e-5) * 0.98
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = radii
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = compute_constraints(res.x)
                if np.min(c_val) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Center-only optimization using Nelder-Mead on best solution
    if best_v is not None:
        best_centers = np.column_stack((best_v[0::3], best_v[1::3]))
        
        # Run Nelder-Mead on centers
        res_nm = minimize(center_objective, best_centers.flatten(), method='Nelder-Mead',
                          options={'maxiter': 5000, 'xatol': 1e-6, 'fatol': 1e-7})
        curr_sum_nm = -res_nm.fun
        if curr_sum_nm > best_sum:
            best_sum = curr_sum_nm
            best_v = np.zeros(3 * N)
            best_v[0::3] = res_nm.x.reshape(N, 2)[:, 0]
            best_v[1::3] = res_nm.x.reshape(N, 2)[:, 1]
            radii_nm, _ = solve_radii_lp(res_nm.x.reshape(N, 2))
            best_v[2::3] = radii_nm
            
        # Phase 3: Refine best V with SLSQP again to ensure strict feasibility & polish
        try:
            # Shrink radii slightly to ensure interior point start for SLSQP
            v_ref = best_v.copy()
            v_ref[2::3] *= 0.995
            res_ref = minimize(objective, v_ref, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
            if res_ref.success:
                c_val = compute_constraints(res_ref.x)
                if np.min(c_val) >= -1e-9:
                    curr_sum = -res_ref.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_v = res_ref.x.copy()
        except Exception:
            pass
            
        # Phase 4: Perturb centers and repeat NM + SLSQP to escape local minima
        for _ in range(20):
            rng = np.random.RandomState(None)
            c_pert = np.column_stack((best_v[0::3], best_v[1::3]))
            c_pert += rng.normal(0, 0.006, c_pert.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            
            res_nm2 = minimize(center_objective, c_pert.flatten(), method='Nelder-Mead',
                               options={'maxiter': 3000, 'xatol': 1e-6, 'fatol': 1e-7})
            curr_sum_nm2 = -res_nm2.fun
            if curr_sum_nm2 > best_sum:
                best_sum = curr_sum_nm2
                best_v = np.zeros(3 * N)
                best_v[0::3] = res_nm2.x.reshape(N, 2)[:, 0]
                best_v[1::3] = res_nm2.x.reshape(N, 2)[:, 1]
                radii_nm2, _ = solve_radii_lp(res_nm2.x.reshape(N, 2))
                best_v[2::3] = radii_nm2
                
                try:
                    v_ref2 = best_v.copy()
                    v_ref2[2::3] *= 0.995
                    res_ref2 = minimize(objective, v_ref2, method='SLSQP', bounds=bounds,
                                        constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14, 'disp': False})
                    if res_ref2.success:
                        c_val = compute_constraints(res_ref2.x)
                        if np.min(c_val) >= -1e-9:
                            curr_sum = -res_ref2.fun
                            if curr_sum > best_sum:
                                best_sum = curr_sum
                                best_v = res_ref2.x.copy()
                except Exception:
                    pass

    # Fallback valid configuration
    if best_v is None:
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        r, _ = solve_radii_lp(pts)
        best_v = np.zeros(3 * N)
        best_v[0::3] = pts[:, 0]
        best_v[1::3] = pts[:, 1]
        best_v[2::3] = r
        best_sum = np.sum(r)
        
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = np.maximum(best_v[2::3], 0.0)
    return centers, radii, float(best_sum)
