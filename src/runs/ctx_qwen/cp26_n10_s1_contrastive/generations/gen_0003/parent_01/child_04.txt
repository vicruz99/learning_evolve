import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26

def solve_lp(centers):
    """
    Given fixed centers, solves the LP to find radii that maximize sum(r_i).
    Constraints: r_i >= 0, r_i <= boundary distances, r_i + r_j <= dist_ij
    """
    n = centers.shape[0]
    c = -np.ones(n)
    
    # Boundary constraints: 4 per circle
    A_bound = np.zeros((4 * n, n))
    b_bound = np.zeros(4 * n)
    x = centers[:, 0]
    y = centers[:, 1]
    for i in range(n):
        A_bound[4 * i, i] = 1.0; b_bound[4 * i] = x[i]
        A_bound[4 * i + 1, i] = 1.0; b_bound[4 * i + 1] = 1.0 - x[i]
        A_bound[4 * i + 2, i] = 1.0; b_bound[4 * i + 2] = y[i]
        A_bound[4 * i + 3, i] = 1.0; b_bound[4 * i + 3] = 1.0 - y[i]

    # Pairwise constraints
    m = n * (n - 1) // 2
    A_pair = np.zeros((m, n))
    b_pair = np.zeros(m)
    idx = 0
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            A_pair[idx, i] = 1.0
            A_pair[idx, j] = 1.0
            b_pair[idx] = dists[i, j]
            idx += 1

    A_ub = np.vstack([A_bound, A_pair])
    b_ub = np.concatenate([b_bound, b_pair])
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def lp_objective(x_flat):
    """Objective for global/local search on centers: minimize negative sum of radii."""
    centers = x_flat.reshape(N, 2)
    _, s = solve_lp(centers)
    return -s

def compute_constraints_joint(v):
    """Joint constraints for SLSQP polish."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary containment
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation: dist^2 >= (r_i + r_j)^2
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    c.append(dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    
    return np.concatenate(c)

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Phase 1: Global search using Differential Evolution
    # We optimize only the centers, using the optimal LP radius sum as the objective.
    de_bounds = [(0.0, 1.0)] * (2 * N)
    try:
        res_de = differential_evolution(
            lp_objective, de_bounds, 
            popsize=25, maxiter=300, tol=1e-6, seed=42, polish=False
        )
        best_centers_de = res_de.x.reshape(N, 2)
    except Exception:
        # Fallback initialization if DE fails
        best_centers_de = np.random.uniform(0.1, 0.9, (N, 2))
        
    # Phase 2: Local refinement using Nelder-Mead
    # Nelder-Mead handles the non-smooth LP landscape robustly
    try:
        res_nm = minimize(lp_objective, res_de.x, method='Nelder-Mead',
                          options={'xatol': 1e-7, 'fatol': 1e-9, 'maxiter': 5000})
        best_centers_loc = res_nm.x.reshape(N, 2)
    except Exception:
        best_centers_loc = best_centers_de

    # Phase 3: Extract optimal radii via LP
    radii, curr_sum = solve_lp(best_centers_loc)
    
    # Phase 4: Joint SLSQP polish
    # Fine-tune centers and radii simultaneously to escape local LP plateaus
    v0 = np.zeros(3 * N)
    v0[0::3] = best_centers_loc[:, 0]
    v0[1::3] = best_centers_loc[:, 1]
    v0[2::3] = radii * 0.995  # Slight shrink to ensure strict feasibility for interior steps
    
    bounds_joint = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons_joint = {'type': 'ineq', 'fun': compute_constraints_joint}
    
    def obj_joint(v):
        return -np.sum(v[2::3])
        
    try:
        res_sl = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,

        if res_sl.success:
            c_val = compute_constraints_joint(res_sl.x)
            if np.min(c_val) >= -1e-8:
                best_centers_loc = np.column_stack((res_sl.x[0::3], res_sl.x[1::3]))
                radii = res_sl.x[2::3]
                curr_sum = -res_sl.fun
    except Exception:
        pass
        
    # Safety fallback if everything fails (highly unlikely)
    if curr_sum <= 0.0 or np.isnan(curr_sum):
        best_centers_loc = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                            np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        radii, curr_sum = solve_lp(best_centers_loc)
        
    best_radii = np.maximum(radii, 0.0)
    return best_centers_loc, best_radii, float(curr_sum)
