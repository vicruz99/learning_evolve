import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_cons = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_cons, n))
    b_ub = np.zeros(n_cons)
    
    x, y = centers[:, 0], centers[:, 1]
    row = 0
    for i in range(n):
        A_ub[row, i] = 1.0; b_ub[row] = x[i]; row += 1
        A_ub[row, i] = 1.0; b_ub[row] = 1.0 - x[i]; row += 1
        A_ub[row, i] = 1.0; b_ub[row] = y[i]; row += 1
        A_ub[row, i] = 1.0; b_ub[row] = 1.0 - y[i]; row += 1
        
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = dists[i, j]
            row += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def generate_inits():
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [5, 5, 5, 5, 6], [6, 4, 6, 5, 5], [5, 6, 4, 6, 5],
        [4, 5, 6, 5, 6], [5, 6, 5, 5, 5], [6, 5, 5, 6, 4]
    ]
    
    for pat in patterns:
        for seed in range(3):
            rng = np.random.RandomState(seed)
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, count in enumerate(pat):
                x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
                for _ in range(count):
                    pts.append([x_start, y])
                    x_start += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
            pts = np.array(pts[:N])
            pts = pts - 0.5
            scale = rng.uniform(0.95, 1.05)
            pts = pts * scale + 0.5
            angle = rng.uniform(-0.2, 0.2)
            c, s = np.cos(angle), np.sin(angle)
            rot = np.array([[c, -s], [s, c]])
            pts = pts @ rot.T
            pts = np.clip(pts + rng.uniform(-0.02, 0.02, (N, 2)), 0.02, 0.98)
            inits.append(pts)
            
    for seed in range(10):
        rng = np.random.RandomState(seed)
        pts = rng.rand(N, 2)
        for _ in range(100):
            forces = np.zeros_like(pts)
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            dists = np.maximum(dists, 1e-5)
            f = 1.0 / (dists**2)
            f = np.clip(f, 0, 10.0)
            forces += np.sum(f[:, :, None] * diff / dists[:, :, None], axis=1)
            pts += 0.005 * forces
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    return inits

def joint_slsqp(x0):
    """Joint SLSQP optimization of centers and radii."""
    def obj(v):
        return -np.sum(v[2::3])
        
    def cons(v):
        xc, yc, r = v[0::3], v[1::3], v[2::3]
        c = [xc - r, 1.0 - xc - r, yc - r, 1.0 - yc - r]
        dx = xc[:, None] - xc[None, :]
        dy = yc[:, None] - yc[None, :]
        d2 = dx**2 + dy**2
        rs = r[:, None] + r[None, :]
        c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
        return np.concatenate(c)
        
    bnds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, 
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 4000, 'ftol': 1e-13})
        if res.success and np.min(cons(res.x)) >= -1e-8:
            return res.x, -res.fun
    except Exception:
        pass
    return None, None

def run_packing():
    np.random.seed(42)
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    inits = generate_inits()
    
    # Phase 1: LP-based evaluation and Hill-Climbing on centers
    for pts in inits:
        radii, s = solve_radii_lp(pts)
        if s > best_sum:
            best_sum = s
            best_centers = pts.copy()
            best_radii = radii.copy()
            
        curr_pts = pts.copy()
        curr_r, curr_s = radii, s
        for _ in range(10):
            improved = False
            for i in range(N):
                best_move_s = curr_s
                best_move_pt = curr_pts[i].copy()
                for _ in range(3):
                    delta = np.random.normal(0, 0.012, 2)
                    new_pt = np.clip(curr_pts[i] + delta, 0.02, 0.98)
                    temp_pts = curr_pts.copy()
                    temp_pts[i] = new_pt
                    _, new_s = solve_radii_lp(temp_pts)
                    if new_s > best_move_s:
                        best_move_s = new_s
                        best_move_pt = new_pt.copy()
                if best_move_s > curr_s + 1e-7:
                    curr_pts[i] = best_move_pt
                    curr_r, curr_s = solve_radii_lp(curr_pts)
                    improved = True
            if not improved:
                break
                
        if curr_s > best_sum:
            best_sum = curr_s
            best_centers = curr_pts.copy()
            best_radii = curr_r.copy()
            
    # Phase 2: Joint SLSQP refinement
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii * 0.995
        
        res_x, res_s = joint_slsqp(x0)
        if res_s is not None and res_s > best_sum:
            best_sum = res_s
            best_centers = np.column_stack((res_x[0::3], res_x[1::3]))
            best_radii = res_x[2::3]
            
        for _ in range(20):
            x_pert = x0.copy()
            x_pert[0::3] += np.random.normal(0, 0.008, N)
            x_pert[1::3] += np.random.normal(0, 0.008, N)
            x_pert[0::3] = np.clip(x_pert[0::3], 0.01, 0.99)
            x_pert[1::3] = np.clip(x_pert[1::3], 0.01, 0.99)
            
            res_x2, res_s2 = joint_slsqp(x_pert)
            if res_s2 is not None and res_s2 > best_sum:
                best_sum = res_s2
                best_centers = np.column_stack((res_x2[0::3], res_x2[1::3]))
                best_radii = res_x2[2::3]
                
        # Final high-precision polish
        x_final = np.zeros(3 * N)
        x_final[0::3] = best_centers[:, 0]
        x_final[1::3] = best_centers[:, 1]
        x_final[2::3] = best_radii * 0.999
        res_x3, res_s3 = joint_slsqp(x_final)
        if res_x3 is not None and res_s3 > best_sum:
            best_sum = res_s3
            best_centers = np.column_stack((res_x3[0::3], res_x3[1::3]))
            best_radii = res_x3[2::3]
            
    # Fallback safety net
    if best_centers is None:
        best_centers = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                        np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        best_radii = np.full(N, 0.01)
        best_sum = np.sum(best_radii)
        
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
