def compute_radii_fast(centers):

    Computes maximal feasible radii for fixed centers using iterative clamping.
    Converges to the exact LP solution efficiently via vectorized operations.
    """

    bounds = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))

    np.fill_diagonal(dists, np.inf)

        r_new = np.maximum(r_new, 1e-9)

        r = r_new

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    c.append(d2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    return np.concatenate(c)

def local_search(init_centers, n_steps=2500, seed=42):

    Greedy hill-climbing on center positions to maximize sum of radii.
    Uses exact radii computation for evaluation.

    best_radii = radii.copy()

                improved = True
                break

                break
    return best_centers, best_radii, best_sum
