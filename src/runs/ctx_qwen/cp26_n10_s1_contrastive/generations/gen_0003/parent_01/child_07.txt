import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_constraints(v):
    """
    Computes inequality constraints g(v) >= 0.
    Includes boundary containment and pairwise non-overlap.
    """
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    # Only upper triangular pairs to avoid duplicates and self-comparison
    i_idx, j_idx = np.triu_indices(N, k=1)
    c.append(dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    
    return np.concatenate(c)

def obj_full(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def solve_radii_lp(centers):
    """
    Given fixed centers, solves the LP to find radii that maximize sum(r_i).
    Constraints:
    1. r_i >= 0
    2. r_i <= distance to boundaries
    3. r_i + r_j <= distance(centers[i], centers[j])
    """
    n = centers.shape[0]
    c = -np.ones(n)
    
    # Preallocate constraint matrix and vector
    # 4 boundary constraints per circle + n*(n-1)/2 pairwise constraints
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    row = 0
    for i in range(n):
        x, y = centers[i]
        # Ensure bounds are non-negative to avoid LP infeasibility from numerical drift
        b_vals = [max(0.0, x), max(0.0, 1.0 - x), max(0.0, y), max(0.0, 1.0 - y)]
        for b in b_vals:
            A_ub[row, i] = 1.0
            b_ub[row] = b
            row += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = d
            row += 1
            
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
        
    return 0.0, np.zeros(n)

def obj_centers(x_flat):
    """Objective for center optimization: minimize negative sum of radii from LP."""
    centers = np.clip(x_flat.reshape(N, 2), 0.01, 0.99)
    s, _ = solve_radii_lp(centers)
    return -s

def get_hex_init(row_counts, seed):
    """Generate a hexagonal lattice initialization with specified row counts."""
    pts = []
    rng = np.random.RandomState(seed)
    r_est = 0.098
    y = r_est
    for r_idx, count in enumerate(row_counts):
        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    bounds_centers = [(0.01, 0.99)] * (2 * N)
    cons = {'type': 'ineq', 'fun': compute_constraints}
    bounds_full = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N

        [5, 6, 5, 6, 4],
        [4, 6, 5, 6, 5],
        [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5],
        [5, 6, 4, 6, 5]
    ]

            inits.append(get_hex_init(pat, s).flatten())

        inits.append(np.clip(rng.rand(2 * N), 0.05, 0.95))

            res = minimize(obj_centers, x0, method='Nelder-Mead',

                best_sum = curr_sum
                c_opt = np.clip(res.x.reshape(N, 2), 0.01, 0.99)
                _, r_opt = solve_radii_lp(c_opt)
                best_centers = c_opt
                best_radii = r_opt
        except Exception:
            continue
