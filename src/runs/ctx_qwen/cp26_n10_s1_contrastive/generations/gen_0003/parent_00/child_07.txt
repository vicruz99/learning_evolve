import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def get_initial_radii_lp(centers):
    """Solves LP to find radii maximizing sum for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = np.zeros((4*n + n*(n-1)//2, n))
    b_ub = np.zeros(4*n + n*(n-1)//2)
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lims = [x, 1.0-x, y, 1.0-y]
        for lim in lims:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.05)

def obj_func(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def cons_func(x, n):
    """Compute inequality constraints: boundary containment and separation."""
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    c_list = []
    c_list.append(xc - r)
    c_list.append(1.0 - xc - r)
    c_list.append(yc - r)
    c_list.append(1.0 - yc - r)
    i_idx, j_idx = np.triu_indices(n, 1)
    dx = xc[i_idx] - xc[j_idx]
    dy = yc[i_idx] - yc[j_idx]
    rs = r[i_idx] + r[j_idx]
    c_list.append(dx**2 + dy**2 - rs**2)
    return np.concatenate(c_list)

def generate_configs(n):
    """Generates diverse initial center configurations."""
    configs = []
    np.random.seed(42)
    
    # 1. Hexagonal lattice variations with controlled jitter
    for k in range(20):
        pts = []
        r_est = 0.09 + 0.01 * np.sin(k)
        dy = r_est * np.sqrt(3.0)
        y = r_est + np.random.uniform(-0.02, 0.02)
        row = 0
        while y < 1.0 - r_est and len(pts) < n:
            shift = (row % 2) * r_est
            x = r_est + shift + np.random.uniform(-0.02, 0.02)
            while x < 1.0 - r_est and len(pts) < n:
                pts.append([x, y])
                x += 2.0 * r_est
            y += dy
            row += 1
        while len(pts) < n:
            pts.append([np.random.uniform(0.05, 0.95), np.random.uniform(0.05, 0.95)])
        configs.append(np.array(pts[:n]))
        
    # 2. Uniform random spreads
    for k in range(15):
        pts = np.random.uniform(0.05, 0.95, (n, 2))
        configs.append(pts)
        
    # 3. Force-directed layouts for natural spacing
    for k in range(10):
        pts = np.random.uniform(0.2, 0.8, (n, 2))
        for _ in range(100):
            forces = np.zeros_like(pts)
            for i in range(n):
                for j in range(i+1, n):
                    diff = pts[i] - pts[j]
                    dist = np.linalg.norm(diff)
                    if dist < 1e-3: dist = 1e-3
                    f = diff / (dist**2)
                    forces[i] += f
                    forces[j] -= f
            margin = 0.05
            for d in range(2):
                below = pts[:, d] < margin
                above = pts[:, d] > 1.0 - margin
                forces[below, d] += 10.0 * (margin - pts[below, d])
                forces[above, d] -= 10.0 * (pts[above, d] - (1.0 - margin))
            pts += 0.005 * forces
            pts = np.clip(pts, 0.02, 0.98)
        configs.append(pts)
        
    return configs

def run_packing():
    n = N_CIRCLES
    best_sum = -np.inf
    best_vars = None
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * n
    
    configs = generate_configs(n)
    
    # Phase 1: Optimize from diverse LP-initialized starts
    for cfg in configs:
        radii_init = get_initial_radii_lp(cfg)
        radii_init = np.maximum(radii_init, 1e-6) * 0.99
        
        x0 = np.zeros(3 * n)
        x0[0::3] = cfg[:, 0]
        x0[1::3] = cfg[:, 1]
        x0[2::3] = radii_init
        
        try:
            res = minimize(obj_func, x0, args=(n,), method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': cons_func, 'args': (n,)},
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                cv = cons_func(res.x, n)
                if np.min(cv) >= -1e-7:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Perturbation refinement to escape local minima
    if best_vars is not None:
        for _ in range(20):
            x_pert = best_vars.copy()
            x_pert[0::3] += np.random.randn(n) * 0.004
            x_pert[1::3] += np.random.randn(n) * 0.004
            x_pert[0::3] = np.clip(x_pert[0::3], 0.01, 0.99)
            x_pert[1::3] = np.clip(x_pert[1::3], 0.01, 0.99)
            x_pert[2::3] = np.clip(x_pert[2::3], 1e-6, 0.49)
            
            try:
                res = minimize(obj_func, x_pert, args=(n,), method='SLSQP', bounds=bounds,
                               constraints={'type': 'ineq', 'fun': cons_func, 'args': (n,)},
                               options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if res.success:
                    cv = cons_func(res.x, n)
                    if np.min(cv) >= -1e-7:
                        s = -res.fun
                        if s > best_sum:
                            best_sum = s
                            best_vars = res.x.copy()
            except Exception:
                continue
                
        # Phase 3: High-precision final polish
        try:
            res = minimize(obj_func, best_vars, args=(n,), method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': cons_func, 'args': (n,)},
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_func(res.x, n)) >= -1e-8:
                best_vars = res.x
                best_sum = -res.fun
        except Exception:
            pass
            
    # Fallback (should not be reached given robust initialization)
    if best_vars is None:
        best_vars = np.zeros(3 * n)
        best_vars[0::3] = np.random.rand(n)
        best_vars[1::3] = np.random.rand(n)
        best_vars[2::3] = 0.05
        best_sum = np.sum(best_vars[2::3])
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
