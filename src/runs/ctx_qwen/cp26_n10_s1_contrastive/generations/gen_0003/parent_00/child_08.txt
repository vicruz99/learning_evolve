import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26

def solve_radii_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 2.6e-4

def lp_objective_flat(centers_flat):
    """Objective for center optimization: minimize negative sum of optimal LP radii."""
    centers = np.clip(centers_flat.reshape(N_CIRCLES, 2), 0.02, 0.98)
    _, sum_r = solve_radii_lp(centers)
    return -sum_r

def compute_objective(x):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def compute_constraints(x):
    """Compute inequality constraints: boundary containment and pairwise separation."""
    n = len(x) // 3
    C = x.reshape(n, 3)
    xc, yc, r = C[:, 0], C[:, 1], C[:, 2]
    
    c_list = []
    # Boundary constraints
    c_list.append(xc - r)
    c_list.append(1.0 - xc - r)
    c_list.append(yc - r)
    c_list.append(1.0 - yc - r)
    
    # Pairwise separation: dist_sq >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = xc[i_idx] - xc[j_idx]
    dy = yc[i_idx] - yc[j_idx]
    r_sum = r[i_idx] + r[j_idx]
    c_list.append(dx*dx + dy*dy - r_sum*r_sum)
    
    return np.concatenate(c_list)

def generate_configs():
    """Generate diverse initial center configurations."""
    configs = []
    
    # Rotated and scaled hexagonal lattices
    for rot in np.linspace(-0.2, 0.2, 9):
        for scale in np.linspace(0.85, 1.15, 5):
            pts = []
            r_est = 0.1
            y = r_est
            row = 0
            while len(pts) < N_CIRCLES and y < 1.0:
                shift = (row % 2) * r_est
                x = r_est + shift
                while x < 1.0 and len(pts) < N_CIRCLES:
                    pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            pts = np.array(pts[:N_CIRCLES])
            
            # Center, scale, and rotate
            pts = pts - 0.5
            pts = pts * scale
            pts = pts + 0.5
            c, s = np.cos(rot), np.sin(rot)
            rot_mat = np.array([[c, -s], [s, c]])
            pts = pts @ rot_mat.T
            pts = np.clip(pts, 0.02, 0.98)
            configs.append(pts.flatten())
            
    # Random configurations
    np.random.seed(42)
    for _ in range(20):
        configs.append(np.random.rand(N_CIRCLES, 2).flatten())
        
    # Force-directed layouts
    for seed in range(10):
        rng = np.random.RandomState(seed)
        pts = rng.rand(N_CIRCLES, 2) * 0.8 + 0.1
        for _ in range(50):
            forces = np.zeros_like(pts)
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            dists = np.maximum(dists, 1e-5)
            f_mag = 1.0 / dists**2
            f_mag = np.clip(f_mag, 0, 10.0)
            forces += np.sum(f_mag[:, :, None] * diff / dists[:, :, None], axis=1)
            pts += 0.005 * forces
            pts = np.clip(pts, 0.02, 0.98)
        configs.append(pts.flatten())
        
    return np.array(configs)

def run_packing():
    configs = generate_configs()
    best_sum = -np.inf
    best_centers = None
    
    # Phase 1: Optimize centers using LP-based objective
    for cfg in configs:
        curr_centers = cfg.reshape(N_CIRCLES, 2)
        _, curr_sum = solve_radii_lp(curr_centers)
        
        # Local derivative-free optimization on centers
        res = minimize(lp_objective_flat, cfg.reshape(-1), method='Powell', 
                       options={'maxiter': 2000, 'xtol': 1e-7, 'ftol': 1e-9})
        opt_sum = -res.fun
        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centers = res.x.reshape(N_CIRCLES, 2)
            
    if best_centers is None:
        best_centers = configs[0].reshape(N_CIRCLES, 2)
        
    # Phase 2: Joint optimization with SLSQP for high-precision polishing
    radii_init, _ = solve_radii_lp(best_centers)
    radii_init = np.maximum(radii_init * 0.99, 1e-6)
    
    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = radii_init
    
    bounds_full = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    try:
        res_slsqp = minimize(compute_objective, x0, method='SLSQP', bounds=bounds_full,
                             constraints=cons, options={'maxiter': 20000, 'ftol': 1e-14})
        if res_slsqp.success:
            cons_val = compute_constraints(res_slsqp.x)
            if np.min(cons_val) >= -1e-8:
                best_centers = res_slsqp.x.reshape(N_CIRCLES, 3)[:, :2]
                radii_final = res_slsqp.x.reshape(N_CIRCLES, 3)[:, 2]
            else:
                radii_final, _ = solve_radii_lp(best_centers)
        else:
            radii_final, _ = solve_radii_lp(best_centers)
    except Exception:
        radii_final, _ = solve_radii_lp(best_centers)
        
    radii_final = np.maximum(radii_final, 1e-7)
    return best_centers, radii_final, float(np.sum(radii_final))
