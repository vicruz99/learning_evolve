import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to maximize sum of radii subject to boundary and pairwise constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Minimize -sum(r) <=> Maximize sum(r)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        vals = [x, 1.0 - x, y, 1.0 - y]
        for v in vals:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(v)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 2.6e-4

def compute_constraints(x):
    """Computes inequality constraints g(x) >= 0: boundary containment and pairwise separation."""
    C = x.reshape(N, 3)
    xc, yc, r = C[:, 0], C[:, 1], C[:, 2]
    
    c_list = []
    # Boundary
    c_list.append(xc - r)
    c_list.append(1.0 - xc - r)
    c_list.append(yc - r)
    c_list.append(1.0 - yc - r)
    
    # Pairwise separation: dist_sq >= (r_i + r_j)^2
    dx = xc[I_IDX] - xc[J_IDX]
    dy = yc[I_IDX] - yc[J_IDX]
    r_sum = r[I_IDX] + r[J_IDX]
    c_list.append(dx*dx + dy*dy - r_sum*r_sum)
    
    return np.concatenate(c_list)

def objective(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def generate_inits(num_inits):
    """Generates diverse initial center configurations."""
    inits = []
    
    # Hexagonal lattice variations
    for k in range(num_inits // 2):
        rng = np.random.RandomState(k)
        rows = rng.choice([4, 5, 6], size=6, p=[0.15, 0.6, 0.25])
        while np.sum(rows) < N:
            rows[rng.randint(len(rows))] += 1
        while np.sum(rows) > N:
            rows[rng.randint(len(rows))] -= 1
            
        pts = []
        y = 0.05
        r_est = 0.09 + rng.uniform(-0.02, 0.02)
        dy = r_est * np.sqrt(3.0)
        row_idx = 0
        for cnt in rows:
            if len(pts) >= N: break
            shift = (row_idx % 2) * r_est
            x = 0.05 + shift
            for _ in range(cnt):
                if len(pts) >= N: break
                pts.append([x, y])
                x += 2.0 * r_est
            y += dy
            row_idx += 1
        pts = np.array(pts[:N])
        
        # Random rotation and jitter
        angle = rng.uniform(-0.2, 0.2)
        c, s = np.cos(angle), np.sin(angle)
        rot = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ rot.T + 0.5
        pts = np.clip(pts + rng.uniform(-0.02, 0.02, pts.shape), 0.02, 0.98)
        inits.append(pts)
        
    # Corner and edge focused layouts
    for k in range(num_inits // 4):
        rng = np.random.RandomState(k + 1000)
        pts = [[0.1, 0.1], [0.9, 0.1], [0.9, 0.9], [0.1, 0.9],
               [0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5]]
        y = 0.25
        r_est = 0.09
        dy = r_est * np.sqrt(3.0)
        while len(pts) < N:
            shift = 0.125 + (len(pts) % 2) * r_est
            x = shift
            while x <= 0.9 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r_est
            y += dy
        pts = np.array(pts[:N])
        pts = pts + rng.uniform(-0.015, 0.015, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)

    # Random spreads
    for k in range(num_inits // 4):
        rng = np.random.RandomState(k + 2000)
        pts = rng.rand(N, 2) * 0.8 + 0.1
        inits.append(pts)
        
    return inits

def run_packing():
    np.random.seed(42)  # For reproducibility of perturbation search
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    inits = generate_inits(50)
    
    # Phase 1: Broad search from diverse initializations
    for idx, centers0 in enumerate(inits):
        r0, _ = solve_radii_lp(centers0)
        r0 = np.maximum(r0, 1e-6) * 0.99  # Ensure strict feasibility
        
        x0 = np.zeros(3 * N)
        x0[0::3] = centers0[:, 0]
        x0[1::3] = centers0[:, 1]
        x0[2::3] = r0
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                c_vals = compute_constraints(res.x)
                if np.min(c_vals) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = res.x[0::3].reshape(N, 2).copy()
                        best_radii = res.x[2::3].copy()
        except Exception:
            continue

    # Phase 2: Iterative perturbation refinement
    if best_centers is not None:
        for iter in range(40):
            centers_p = best_centers.copy()
            noise_scale = 0.006 * (1.0 - iter / 40.0)
            centers_p += np.random.randn(N, 2) * noise_scale
            centers_p = np.clip(centers_p, 0.01, 0.99)
            
            r_p, _ = solve_radii_lp(centers_p)
            r_p = np.maximum(r_p, 1e-6) * 0.995
            
            x0 = np.zeros(3 * N)
            x0[0::3] = centers_p[:, 0]
            x0[1::3] = centers_p[:, 1]
            x0[2::3] = r_p
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                if res.success:
                    c_vals = compute_constraints(res.x)
                    if np.min(c_vals) >= -1e-8:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_centers = res.x[0::3].reshape(N, 2).copy()
                            best_radii = res.x[2::3].copy()
            except Exception:
                continue
                
    # Phase 3: Exact LP polish on best centers
    if best_centers is not None:
        r_final, sum_final = solve_radii_lp(best_centers)
        if sum_final > best_sum:
            best_radii = r_final
            best_sum = sum_final
            
    best_radii = np.maximum(best_radii, 0.0)
    
    # Fallback configuration
    if best_centers is None:
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        r_f, _ = solve_radii_lp(pts)
        return pts, r_f, float(np.sum(r_f))
        
    return best_centers, best_radii, float(best_sum)
