import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """Compute all inequality constraints: boundary containment and pairwise separation."""
    c = v.reshape(N, 3)
    x, y, r = c[:, 0], c[:, 1], c[:, 2]
    
    cons = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    cons.append(x - r)
    cons.append(1.0 - x - r)
    cons.append(y - r)
    cons.append(1.0 - y - r)
    
    # Pairwise separation: dist^2 - (r_i + r_j)^2 >= 0
    i, j = np.triu_indices(N, k=1)
    dx = x[i] - x[j]
    dy = y[i] - y[j]
    rs = r[i] + r[j]
    cons.append(dx*dx + dy*dy - rs*rs)
    
    return np.concatenate(cons)

def force_simulate(n, steps, seed):
    """Spread points using force-directed layout within unit square."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(n, 2) * 0.8 + 0.1
    for step in range(steps):
        forces = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-4)
        f_mag = 1.0 / (dists**2)
        f_mag = np.clip(f_mag, 0, 10.0)
        forces += np.sum(f_mag[:, :, None] * diff / dists[:, :, None], axis=1)
        
        margin = 0.05
        for d in [0, 1]:
            below = pts[:, d] < margin
            above = pts[:, d] > 1.0 - margin
            forces[below, d] += 20.0 * (margin - pts[below, d])
            forces[above, d] -= 20.0 * (pts[above, d] - (1.0 - margin))
            
        forces -= 5.0 * (pts - 0.5)
        pts += 0.002 * forces / (step + 1)
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def get_hex_init(row_counts, seed):
    """Generate a hexagonal lattice initialization with specified row counts."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.09
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def get_initial_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        d_min = np.inf
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if d < d_min:
                    d_min = d
        r[i] = 0.45 * min(d_wall, d_min / 2.0)
    return np.maximum(r, 1e-5)

def run_packing():
    best_sum = -np.inf
    best_x = None
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    inits = []
    
    # 1. Hexagonal variations with different row distributions
    hex_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [5, 5, 5, 5, 6], [6, 4, 6, 5, 5], [5, 6, 4, 6, 5], [4, 5, 6, 5, 6]
    ]
    for pat in hex_patterns:
        for s in range(3):
            inits.append(get_hex_init(pat, s))
            
    # 2. Force simulated layouts for diverse topologies
    for s in range(15):
        inits.append(force_simulate(N, steps=200, seed=s))
        
    # 3. Structured random rows
    for s in range(15):
        rng = np.random.RandomState(s)
        pts = rng.rand(N, 2) * 0.8 + 0.1
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)

    # Run SLSQP from each initialization
    for idx, pts in enumerate(inits):
        r_init = get_initial_radii(pts)
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r_init * 0.98  # Ensure strict feasibility
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'finite_diff_rel_step': 1e-8})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
        except Exception:
            continue

    # Local perturbation search around the best found solution
    if best_x is not None:
        for k in range(40):
            rng = np.random.RandomState(k)
            x_pert = best_x.copy()
            x_pert[0::3] += rng.normal(0, 0.004, N)
            x_pert[1::3] += rng.normal(0, 0.004, N)
            x_pert[2::3] *= np.exp(rng.normal(0, 0.008, N))
            
            x_pert[0::3] = np.clip(x_pert[0::3], 0.01, 0.99)
            x_pert[1::3] = np.clip(x_pert[1::3], 0.01, 0.99)
            x_pert[2::3] = np.clip(x_pert[2::3], 1e-5, 0.49)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'finite_diff_rel_step': 1e-8})
                if res.success:
                    c_val = constraints(res.x)
                    if np.min(c_val) >= -1e-8:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = res.x.copy()
            except Exception:
                continue

    # Final high-precision polish
    if best_x is not None:
        try:
            res_final = minimize(objective, best_x, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'finite_diff_rel_step': 1e-9})
            if res_final.success:
                c_val = constraints(res_final.x)
                if np.min(c_val) >= -1e-8:
                    best_x = res_final.x
                    best_sum = -res_final.fun
        except Exception:
            pass

    # Fallback (should not be reached)
    if best_x is None:
        centers = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        centers = np.vstack([centers, [0.5, 0.5]])
        r_f = np.full(N, 0.09)
        best_x = np.zeros(3 * N)
        best_x[0::3] = centers[:, 0]
        best_x[1::3] = centers[:, 1]
        best_x[2::3] = r_f
        best_sum = np.sum(r_f)

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = np.maximum(best_x[2::3], 0.0)
    return centers, radii, float(best_sum)
