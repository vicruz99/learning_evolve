import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def solve_radii_lp(centers):
    """
    Given fixed centers, solves the LP to find radii that maximize sum(r_i).
    Constraints: r_i >= 0, r_i <= dist_to_boundaries, r_i + r_j <= dist_ij.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    for i in range(n):
        x, y = centers[i]
        bounds_val = np.array([x, 1.0 - x, y, 1.0 - y])
        for b in bounds_val:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def objective_centers(centers_flat):
    """Objective for center optimization: minimize negative sum of LP-optimal radii."""
    centers = centers_flat.reshape(N_CIRCLES, 2)
    _, sum_r = solve_radii_lp(centers)
    return -sum_r

def generate_hex_init(seed, row_counts, jitter=0.02):
    """Generate a feasible initial center configuration based on a hexagonal lattice."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.1
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N_CIRCLES])
    pts += rng.uniform(-jitter, jitter, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def compute_constraints_full(x):
    """Compute all inequality constraints: boundary containment and separation."""
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    c = []
    c.append(xc - r)
    c.append(1.0 - xc - r)
    c.append(yc - r)
    c.append(1.0 - yc - r)
    i_idx, j_idx = np.triu_indices(N_CIRCLES, k=1)
    dx = xc[i_idx] - xc[j_idx]
    dy = yc[i_idx] - yc[j_idx]
    r_s = r[i_idx] + r[j_idx]
    c.append(dx*dx + dy*dy - r_s*r_s)
    return np.concatenate(c)

def objective_full(x):
    """Objective for full optimization: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [4, 6, 5, 6, 5],
        [5, 5, 6, 5, 5], [6, 4, 6, 6, 4], [5, 6, 4, 6, 5]
    ]
    
    # Phase 1: Center optimization via Nelder-Mead from diverse hex starts
    for pat in patterns:
        for seed in range(4):
            centers_init = generate_hex_init(seed, pat, jitter=0.015)
            x0 = centers_init.flatten()
            try:
                res = minimize(objective_centers, x0, method='Nelder-Mead',
                               options={'maxiter': 1500, 'xatol': 1e-7, 'fatol': 1e-9})
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    curr_centers = np.clip(res.x.reshape(N_CIRCLES, 2), 1e-5, 1-1e-5)
                    curr_radii, curr_s = solve_radii_lp(curr_centers)
                    best_sum = curr_s
                    best_centers = curr_centers.copy()
                    best_radii = curr_radii.copy()
            except Exception:
                continue
                
    # Phase 2: Local perturbation + Nelder-Mead refinement
    if best_centers is not None:
        for _ in range(10):
            centers_pert = best_centers + np.random.randn(N_CIRCLES, 2) * 0.005
            centers_pert = np.clip(centers_pert, 0.01, 0.99)
            x0 = centers_pert.flatten()
            try:
                res = minimize(objective_centers, x0, method='Nelder-Mead',
                               options={'maxiter': 1000, 'xatol': 1e-7, 'fatol': 1e-9})
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    curr_centers = np.clip(res.x.reshape(N_CIRCLES, 2), 1e-5, 1-1e-5)
                    curr_radii, curr_s = solve_radii_lp(curr_centers)
                    best_sum = curr_s
                    best_centers = curr_centers.copy()
                    best_radii = curr_radii.copy()
            except Exception:
                continue
                
    # Phase 3: SLSQP high-precision polish on full variables
    if best_centers is not None:
        x0_full = np.zeros(3 * N_CIRCLES)
        x0_full[0::3] = best_centers[:, 0]
        x0_full[1::3] = best_centers[:, 1]
        # Shrink radii slightly to ensure strict feasibility for SLSQP interior steps
        x0_full[2::3] = np.maximum(best_radii, 1e-6) * 0.995
        
        bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N_CIRCLES
        cons = {'type': 'ineq', 'fun': compute_constraints_full}
        try:
            res_full = minimize(objective_full, x0_full, method='SLSQP', bounds=bounds,
                                constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
            if res_full.success:
                c_val = compute_constraints_full(res_full.x)
                if np.min(c_val) >= -1e-7:
                    final_sum = -res_full.fun
                    if final_sum > best_sum:
                        best_sum = final_sum
                        best_centers = np.column_stack((res_full.x[0::3], res_full.x[1::3]))
                        best_radii = res_full.x[2::3]
        except Exception:
            pass
            
    # Fallback to a valid grid packing if optimization fails completely
    if best_centers is None:
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        best_radii, best_sum = solve_radii_lp(pts)
        best_centers = pts
        
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
