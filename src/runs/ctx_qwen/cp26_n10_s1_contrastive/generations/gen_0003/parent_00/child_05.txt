import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """
    Given fixed centers, solves the LP to find radii that maximize sum(r_i).
    Constraints:
    1. r_i >= 0
    2. r_i <= distance to boundaries
    3. r_i + r_j <= distance(centers[i], centers[j])
    """
    n = centers.shape[0]
    c = -np.ones(n)  # Minimize -sum(r_i) <=> Maximize sum(r_i)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        bounds_val = np.array([x, 1.0 - x, y, 1.0 - y])
        for b in bounds_val:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 2.6e-4

def compute_objective(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def compute_constraints(x):
    """Compute all inequality constraints: boundary containment and separation."""
    N = 26
    C = x.reshape(N, 3)
    xc = C[:, 0]
    yc = C[:, 1]
    r = C[:, 2]
    
    c_list = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c_list.append(xc - r)
    c_list.append(1.0 - xc - r)
    c_list.append(yc - r)
    c_list.append(1.0 - yc - r)
    
    # Pairwise separation constraints: dist_sq >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = xc[i_idx] - xc[j_idx]
    dy = yc[i_idx] - yc[j_idx]
    r_sum = r[i_idx] + r[j_idx]
    c_list.append(dx*dx + dy*dy - r_sum*r_sum)
    
    return np.concatenate(c_list)

def generate_hex_init(N, row_counts, seed, rotation=0.0, scale=1.0):
    """Generate a feasible initial center configuration based on a hexagonal lattice."""
    np.random.seed(seed)
    pts = []
    r_est = 0.095
    y = r_est
    dy = r_est * np.sqrt(3.0)
    
    for r_idx, cnt in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += dy
        
    pts = np.array(pts[:N])
    
    # Center and scale the lattice
    pts = pts - 0.5
    pts = pts * scale
    pts = pts + 0.5
    
    # Apply rotation around the center
    if rotation != 0.0:
        c = np.cos(rotation)
        s = np.sin(rotation)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = pts @ rot_mat.T
        pts = pts - pts.mean(axis=0) + 0.5
        
    # Add controlled jitter to escape exact symmetries
    jitter = np.random.randn(N, 2) * 0.012
    pts = pts + jitter
    pts = np.clip(pts, 0.02, 0.98)
    
    return pts

def run_packing():
    N = 26
    best_sum = -np.inf
    best_vars = None
    
    # Variable bounds: x in [0,1], y in [0,1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # Diverse row count patterns that sum to 26
    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], [6, 4, 6, 5, 5],
        [4, 6, 6, 6, 4], [5, 5, 5, 5, 6], [6, 6, 4, 6, 4], [5, 6, 6, 4, 5],
        [5, 7, 5, 5, 4], [7, 5, 5, 5, 4], [5, 5, 5, 7, 4], [6, 6, 6, 5, 3]
    ]
    
    # Phase 1: Structured hexagonal starts with rotations and scaling
    for seed in range(8):
        for pat in row_patterns:
            rot = np.random.uniform(-0.12, 0.12)
            scale = np.random.uniform(0.92, 1.08)
            centers_init = generate_hex_init(N, pat, seed, rot, scale)
            radii_init, _ = solve_radii_lp(centers_init)
            
            # Shrink radii slightly to ensure strict feasibility for SLSQP's interior steps
            radii_init = np.maximum(radii_init, 1e-6) * 0.997
            
            x0 = np.zeros(3 * N)
            x0[0::3] = centers_init[:, 0]
            x0[1::3] = centers_init[:, 1]
            x0[2::3] = radii_init
            
            try:
                res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13})
                
                if res.success:
                    cons_vals = compute_constraints(res.x)
                    if np.min(cons_vals) >= -1e-8:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_vars = res.x.copy()
            except Exception:
                continue
                
    # Phase 2: Local perturbation refinement to escape local minima
    if best_vars is not None:
        for k in range(60):
            x_pert = best_vars.copy()
            # Perturb centers
            x_pert[0::3] += np.random.normal(0, 0.004, N)
            x_pert[1::3] += np.random.normal(0, 0.004, N)
            # Perturb radii slightly
            x_pert[2::3] *= np.random.uniform(0.97, 1.03, N)
            
            # Clip to bounds to maintain feasibility during optimization
            x_pert[0::3] = np.clip(x_pert[0::3], 0.015, 0.985)
            x_pert[1::3] = np.clip(x_pert[1::3], 0.015, 0.985)
            x_pert[2::3] = np.clip(x_pert[2::3], 1e-6, 0.49)
            
            try:
                res = minimize(compute_objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13})
                if res.success:
                    cons_vals = compute_constraints(res.x)
                    if np.min(cons_vals) >= -1e-8:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_vars = res.x.copy()
            except Exception:
                continue
                
        # Phase 3: High-precision polynomial-time polish
        try:
            res_final = minimize(compute_objective, best_vars, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14})
            if res_final.success:
                cons_vals = compute_constraints(res_final.x)
                if np.min(cons_vals) >= -1e-9:
                    best_vars = res_final.x
                    best_sum = -res_final.fun
        except Exception:
            pass

    # Fallback to a valid grid packing if optimization fails completely
    if best_vars is None:
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        r_init, _ = solve_radii_lp(pts)
        best_vars = np.zeros(3 * N)
        best_vars[0::3] = pts[:, 0]
        best_vars[1::3] = pts[:, 1]
        best_vars[2::3] = np.maximum(r_init, 1e-6)
        best_sum = np.sum(best_vars[2::3])
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
