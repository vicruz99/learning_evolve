1. **Global Center Optimization via Differential Evolution (DE) + Linear Programming (LP)**: 
   For any fixed set of centers, the problem of finding optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints is a convex Linear Program. We use `scipy.optimize.differential_evolution` to globally search the 52-dimensional space of circle centers. The DE objective function evaluates the exact maximum sum of radii for each candidate center configuration by solving a tiny LP (26 variables, ~430 constraints). This decouples the highly non-convex geometry search from the radius sizing, guaranteeing that every DE evaluation is geometrically consistent and pushing the population toward dense packings.

2. **Local Joint Refinement via SLSQP**:
   Once DE converges to a high-quality center configuration, we extract the corresponding optimal radii from the LP. We then initialize a constrained nonlinear optimizer (`SLSQP`) on the full 78-dimensional space (x, y, r for each circle). SLSQP uses gradient information to tightly pack circles, resolve boundary tensions, and polish the solution to machine precision. Multiple perturbed restarts around the DE result ensure we don't get trapped in shallow local optima during the final polish.

This approach leverages the global exploration strength of evolutionary algorithms, the exactness of LP for radius sizing, and the local precision of sequential quadratic programming. It reliably surpasses heuristic grid/hexagonal initializations and handles unequal radii optimally.

import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

N = 26

def compute_lp_objective(centers_flat):
    """
    Given flattened center coordinates, solves an LP to find the maximum 
    sum of radii. Returns the negative sum (for minimization).
    """
    centers = centers_flat.reshape(N, 2)
    c_obj = -np.ones(N)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(N):
        x, y = centers[i]
        for lim in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(N)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(lim)
            
    # Pairwise non-overlap constraints: r_i + r_j <= dist(c_i, c_j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(N):
        for j in range(i + 1, N):
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * N
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.fun  # res.fun is -sum(r), minimizing it maximizes sum
    except Exception:
        pass
    return 0.0

def constraints_full(vars_vec):
    """
    Inequality constraints for SLSQP: g(vars) >= 0
    Enforces boundary containment and pairwise separation.
    """
    cx = vars_vec[0::3]
    cy = vars_vec[1::3]
    r = vars_vec[2::3]
    
    c = []
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(dist2[mask] - rs[mask]**2)
    
    return np.concatenate(c)

def objective_full(vars_vec):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_vec[2::3])

def run_packing():
    np.random.seed(42)
    
    # Phase 1: Global optimization of centers using DE + exact LP radius solver
    de_bounds = [(0.0, 1.0)] * (2 * N)
    try:
        de_res = differential_evolution(
            compute_lp_objective, 
            de_bounds, 
            popsize=12, 
            maxiter=600, 
            tol=1e-7, 
            seed=42, 
            polish=False
        )
        best_centers = de_res.x.reshape(N, 2)
    except Exception:
        # Fallback initialization if DE fails
        best_centers = np.random.rand(N, 2)
        
    # Re-solve LP once for the best centers to get precise radii
    c_obj = -np.ones(N)
    A_ub = []
    b_ub = []
    for i in range(N):
        x, y = best_centers[i]
        for lim in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(N)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(lim)
            
    diff = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(N):
        for j in range(i + 1, N):
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    lp_bounds = [(0.0, None)] * N
    try:
        lp_res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), bounds=lp_bounds, method='highs')
        best_radii = lp_res.x
    except Exception:
        best_radii = np.full(N, 0.05)
        
    # Phase 2: Local joint refinement with SLSQP
    bounds_full = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_full}
    
    # Prepare strictly feasible initial guess for SLSQP
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = best_radii * 0.99  # Shrink slightly to guarantee interior point
    
    best_x = x0.copy()
    best_sum = np.sum(x0[2::3])
    
    # Run multiple perturbed SLSQP refinements to escape local minima
    for _ in range(5):
        x_pert = x0.copy()
        x_pert[0::3] += np.random.randn(N) * 0.002
        x_pert[1::3] += np.random.randn(N) * 0.002
        x_pert[2::3] += np.random.randn(N) * 0.001
        x_pert[0::3] = np.clip(x_pert[0::3], 0.01, 0.99)
        x_pert[1::3] = np.clip(x_pert[1::3], 0.01, 0.99)
        x_pert[2::3] = np.clip(x_pert[2::3], 1e-5, 0.49)
        
        try:
            res = minimize(objective_full, x_pert, method='SLSQP', bounds=bounds_full,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13})
            if res.success:
                c_val = constraints_full(res.x)
                if np.min(c_val) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 3: High-precision final polish
    try:
        res_final = minimize(objective_full, best_x, method='SLSQP', bounds=bounds_full,
                             constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14})
        if res_final.success and np.min(constraints_full(res_final.x)) >= -1e-9:
            best_x = res_final.x
            best_sum = -res_final.fun
    except Exception:
        pass
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
