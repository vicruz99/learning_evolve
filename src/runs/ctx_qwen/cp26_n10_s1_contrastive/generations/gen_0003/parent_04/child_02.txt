import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def constraints_full(vars_vec):
    """
    Inequality constraints: boundary containment and pairwise non-overlap.
    Returns array of values that must be >= 0.
    """
    cx = vars_vec[0::3]
    cy = vars_vec[1::3]
    r = vars_vec[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def objective_full(vars_vec):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars_vec[2::3])

def get_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    A_rows = []
    b_rows = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_rows.append(row)
            b_rows.append(b)
            
    # Pairwise constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_rows.append(row)
            b_rows.append(dists[i, j])
            
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def make_hex_init(row_counts, seed=0):
    """Generates a hexagonal lattice initialization scaled to fit in the unit square."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    
    for r_idx, count in enumerate(row_counts):
        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    
    # Center and scale to tightly fit within [0.05, 0.95]
    min_c = pts.min(axis=0)
    max_c = pts.max(axis=0)
    span = max_c - min_c
    span = np.maximum(span, 1e-6)
    pts = (pts - min_c) / span
    pts = pts * 0.85 + 0.075  # Map to [0.075, 0.925]
    
    # Add controlled jitter to break exact symmetry
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    return pts
