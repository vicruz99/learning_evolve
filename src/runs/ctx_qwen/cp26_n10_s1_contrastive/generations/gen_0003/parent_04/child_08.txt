import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(v):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(v[2::3])

def constraints(v):
    """
    Inequality constraints: boundary containment and pairwise non-overlap.
    Returns array of values that must be >= 0.
    """
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def compute_safe_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r_safe = np.zeros(n)
    for i in range(n):
        dw = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        dm = 1.0
        for j in range(n):
            if i != j:
                d = np.linalg.norm(centers[i] - centers[j])
                if d < dm:
                    dm = d
        # 0.85 factor places us close to the active constraints for faster convergence
        r_safe[i] = 0.85 * min(dw, dm / 2.0)
    return r_safe

def make_init_hex(rot, scale):
    """Generates a hexagonal lattice initialization with rotation and scaling."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        x_start = r_est + (row % 2) * r_est
        x = x_start
        while x < 1.0 and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    
    # Center, scale, rotate
    pts -= 0.5
    pts *= scale
    c, s = np.cos(rot), np.sin(rot)
    rot_mat = np.array([[c, -s], [s, c]])
    pts = pts @ rot_mat
    pts += 0.5
    pts = np.clip(pts, 0.05, 0.95)
    
    r_safe = compute_safe_radii(pts)
    v0 = np.zeros(3 * N)
    v0[0::3] = pts[:, 0]
    v0[1::3] = pts[:, 1]
    v0[2::3] = r_safe
    return v0

def make_init_force(seed):
    """Generates initialization via force-directed repulsion."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    
    for _ in range(150):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 0.02)
        f_mag = 1.0 / (dists**2)
        f += np.sum(f_mag[:,:,None] * diff / dists[:,:,None], axis=1)
        
        # Wall repulsion
        for d in [0, 1]:
            below = pts[:, d] < 0.1
            above = pts[:, d] > 0.9
            f[below, d] += 15.0 * (0.1 - pts[below, d])
            f[above, d] -= 15.0 * (pts[above, d] - 0.9)
            
        pts += 0.008 * f
        pts = np.clip(pts, 0.02, 0.98)
        
    r_safe = compute_safe_radii(pts)
    v0 = np.zeros(3 * N)
    v0[0::3] = pts[:, 0]
    v0[1::3] = pts[:, 1]
    v0[2::3] = r_safe
    return v0

def run_packing():

    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N

                if np.min(c_val) >= -1e-7:
                    val = -res.fun

        except Exception:

            pass

        radii *= 0.99999
        best_x[2::3] = radii
        
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
