import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(x):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(x[2::3])

def constraints(x):
    """
    Inequality constraints: boundary containment and pairwise non-overlap.
    Returns array of values that must be >= 0.
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def get_hex_centers(row_counts, seed, rotation=0.0):
    """Generates a hexagonal lattice initialization scaled to fit tightly."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.1
    y = r_est
    for r_idx, count in enumerate(row_counts):
        shift = r_est if r_idx % 2 == 1 else 0.0
        x = r_est + shift
        for _ in range(count):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    
    # Scale to fit tightly in [0.04, 0.96]
    min_pt = pts.min(axis=0)
    max_pt = pts.max(axis=0)
    span = max_pt - min_pt
    target_span = 0.92
    scale = target_span / span.max()
    pts = (pts - min_pt) * scale + 0.04
    
    if rotation != 0.0:
        c, s = np.cos(rotation), np.sin(rotation)
        rot = np.array([[c, -s], [s, c]])
        pts = pts @ rot.T
        # Re-center after rotation to keep within bounds
        pts = pts - pts.mean(axis=0) + 0.5
        
    # Add controlled jitter to escape exact symmetries
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    pts = np.clip(pts, 0.04, 0.96)
    return pts

def make_init(centers, shrink=0.90):
    """Compute strictly feasible initial radii for given centers."""
    r_safe = np.zeros(N)
    for i in range(N):
        d_wall = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        d_min = 1.0
        for j in range(N):
            if i != j:
                d = np.linalg.norm(centers[i] - centers[j])
                if d < d_min: 
                    d_min = d
        # Shrink factor ensures strict feasibility for SLSQP start
        r_safe[i] = shrink * min(d_wall, d_min / 2.0)
        
    x0 = np.zeros(3 * N)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = r_safe
    return x0
