import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def objective(vars_vec):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars_vec[2::3])

def constraint_func(vars_vec):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    cx = vars_vec[0::3]
    cy = vars_vec[1::3]
    r = vars_vec[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def compute_safe_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    dist_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    dist_min = np.min(dists, axis=1)
    # Scale down to 96% to guarantee strict interior feasibility for SLSQP
    return np.minimum(dist_bound, dist_min / 2.0) * 0.96

def generate_initial_configs(num_configs):
    """Generates diverse initial configurations for optimization."""
    configs = []
    rng = np.random.RandomState(2024)
    
    for i in range(num_configs):
        centers = np.zeros((N_CIRCLES, 2))
        mode = i % 6
        
        if mode == 0:  # Hex lattice 1
            r_est = 0.085 + rng.uniform(-0.015, 0.015)
            y = r_est
            row = 0
            idx = 0
            while idx < N_CIRCLES and y < 1.0:
                x = r_est if row % 2 == 0 else 2.0 * r_est
                while idx < N_CIRCLES and x < 1.0:
                    centers[idx] = [x, y]
                    idx += 1
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            centers = centers[:N_CIRCLES]
            
        elif mode == 1:  # Hex lattice 2 (shifted parity)
            r_est = 0.09 + rng.uniform(-0.02, 0.02)
            y = r_est
            row = 0
            idx = 0
            while idx < N_CIRCLES and y < 1.0:
                x = 2.0 * r_est if row % 2 == 0 else r_est
                while idx < N_CIRCLES and x < 1.0:
                    centers[idx] = [x, y]
                    idx += 1
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            centers = centers[:N_CIRCLES]
            
        elif mode == 2:  # Grid 5x5 + 1
            idx = 0
            for r_idx in range(5):
                for c_idx in range(5):
                    centers[idx] = [0.1 + 0.18 * c_idx, 0.1 + 0.18 * r_idx]
                    idx += 1
            centers[25] = [0.5, 0.5]
            
        elif mode == 3:  # Random spread
            centers = rng.rand(N_CIRCLES, 2)
            
        elif mode == 4:  # Force directed layout
            pts = rng.rand(N_CIRCLES, 2)
            for _ in range(100):
                forces = np.zeros_like(pts)
                diff = pts[:, None, :] - pts[None, :, :]
                dists = np.sqrt(np.sum(diff**2, axis=2))
                dists = np.maximum(dists, 1e-5)
                f = 1.0 / (dists**2)
                f = np.clip(f, 0, 8.0)
                forces += np.sum(f[:, :, None] * diff / dists[:, :, None], axis=1)
                pts += 0.003 * forces
                pts = np.clip(pts, 0.05, 0.95)
            centers = pts
            
        else:  # Perturbed hex
            r_est = 0.095
            y = r_est
            row = 0
            idx = 0
            while idx < N_CIRCLES and y < 1.0:

                while idx < N_CIRCLES and x < 1.0:
                    centers[idx] = [x, y]
                    idx += 1
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            centers = centers[:N_CIRCLES] + rng.uniform(-0.04, 0.04, centers.shape)

        v0 = np.zeros(3 * N_CIRCLES)
        v0[0::3] = centers[:, 0]
        v0[1::3] = centers[:, 1]
        v0[2::3] = r_init
        configs.append(v0)

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_val = -np.inf
    best_v = None
    
    # Phase 1: Broad search from diverse initial configurations
    configs = generate_initial_configs(60)
    for v0 in configs:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-9:
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
        except Exception:
            pass

        best_v = configs[0]
        best_val = -objective(best_v)

        v_p[0::3] = np.clip(v_p[0::3], 0.01, 0.99)
        v_p[1::3] = np.clip(v_p[1::3], 0.01, 0.99)
        v_p[2::3] = np.clip(v_p[2::3], 1e-5, 0.49)
        
        try:
            res = minimize(objective, v_p, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 2500, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-9:
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
        except Exception:
            pass

    try:
        res_poly = minimize(objective, best_v, method='SLSQP', bounds=bounds, constraints=cons,

        if res_poly.success and np.min(constraint_func(res_poly.x)) >= -1e-9:
            best_v = res_poly.x
            best_val = -res_poly.fun
    except Exception:
        pass
        
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(best_val)
