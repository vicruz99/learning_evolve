import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def objective(vars_vec):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars_vec[2::3])

def constraints(vars_vec):
    """
    Inequality constraints: boundary containment and pairwise non-overlap.
    Returns array of values that must be >= 0.
    """
    cx = vars_vec[0::3]
    cy = vars_vec[1::3]
    r = vars_vec[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def generate_hex_init(seed, rotation=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N_CIRCLES, 2))
    r_est = 0.095
    y = r_est
    row = 0
    idx = 0
    while idx < N_CIRCLES and y < 1.0 - r_est + 0.01:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        x = x_start
        while idx < N_CIRCLES and x < 1.0 - r_est + 0.01:
            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    if rotation != 0.0:
        c, s = np.cos(rotation), np.sin(rotation)
        rot_mat = np.array([[c, -s], [s, c]])
        centers = centers @ rot_mat.T
        centers -= centers.mean(axis=0)
        centers += 0.5
        
    centers += rng.uniform(-0.02, 0.02, centers.shape)
    centers = np.clip(centers, 0.04, 0.96)
    return centers

def generate_force_init(seed):
    """Generates a force-simulated layout to spread points."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N_CIRCLES, 2)
    for _ in range(600):
        forces = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-5)
        rep = 1.0 / dists**2
        forces += np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)

        for d in [0, 1]:
            below = pts[:, d] < margin
            above = pts[:, d] > 1.0 - margin
            forces[below, d] += 15.0 * (margin - pts[below, d])
            forces[above, d] -= 15.0 * (pts[above, d] - (1.0 - margin))
            
        pts += 0.006 * forces
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def generate_grid_init(seed):
    """Generates a grid-based initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N_CIRCLES, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < N_CIRCLES:
                centers[idx] = [0.1 + 0.2 * i, 0.1 + 0.2 * j]
                idx += 1
    if idx < N_CIRCLES:
        centers[idx] = [0.5, 0.5]
        
    centers += rng.uniform(-0.025, 0.025, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def compute_safe_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    r_safe = np.zeros(N_CIRCLES)
    for i in range(N_CIRCLES):
        d_wall = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
        d_min = 1.0
        for j in range(N_CIRCLES):
            if i != j:
                d = np.linalg.norm(centers[i] - centers[j])
                if d < d_min:
                    d_min = d
        # Scale down significantly to guarantee strict interior feasibility
        r_safe[i] = 0.35 * min(d_wall, d_min)
    return r_safe

def run_packing():

    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N_CIRCLES

                        best_x = res.x.copy()

            pass
