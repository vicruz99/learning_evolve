import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def get_lp_radii_and_sum(centers):
    """Solve LP to find max sum of radii for fixed centers."""
    n = N_CIRCLES
    x, y = centers[:, 0], centers[:, 1]
    wall_bounds = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    m = n + n*(n-1)//2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    # Wall constraints: r_i <= dist_to_wall
    for i in range(n):
        A_ub[i, i] = 1.0
        b_ub[i] = wall_bounds[i]
        
    # Pairwise constraints: r_i + r_j <= dist_ij
    row = n
    for i, j in zip(I_IDX, J_IDX):
        A_ub[row, i] = 1.0
        A_ub[row, j] = 1.0
        b_ub[row] = dists[i, j]
        row += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x, -res.fun
    else:
        # Fallback relaxation if LP fails
        safe_r = np.maximum(wall_bounds * 0.4, 1e-6)
        for _ in range(3):
            for i in range(n):
                for j in range(i+1, n):
                    if safe_r[i] + safe_r[j] > dists[i, j]:
                        s = dists[i, j] / (safe_r[i] + safe_r[j])
                        safe_r[i] *= 0.5 + 0.5*s
                        safe_r[j] *= 0.5 + 0.5*s
        return safe_r, np.sum(safe_r)

def de_obj(x):
    """Objective for Differential Evolution: minimize negative sum of radii."""
    centers = x.reshape(N_CIRCLES, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, s = get_lp_radii_and_sum(centers)
    return -s

def full_obj(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def full_cons(v):
    """Inequality constraints for joint optimization: g(v) >= 0."""
    cx, cy, r = v[0::3], v[1::3], v[2::3]
    c = []
    # Boundary containment
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def run_packing():
    np.random.seed(42)
    bounds_centers = [(0.0, 1.0)] * (2 * N_CIRCLES)
    
    # Phase 1: Global optimization of centers using Differential Evolution
    try:
        res_de = differential_evolution(de_obj, bounds_centers, popsize=12, maxiter=150, 
                                        tol=1e-8, mutation=(0.5, 1.0), recombination=0.7, 
                                        seed=123, polish=False)
        best_centers = res_de.x.reshape(N_CIRCLES, 2)
    except Exception:
        best_centers = np.random.rand(N_CIRCLES, 2) * 0.8 + 0.1
        
    radii, sum_r = get_lp_radii_and_sum(best_centers)
    best_sum = sum_r
    best_vars = np.concatenate([best_centers.flatten(), radii])
    
    bounds_full = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': full_cons}
    
    # Phase 2: Joint optimization of centers and radii using SLSQP
    try:
        res_slsqp = minimize(full_obj, best_vars, method='SLSQP', bounds=bounds_full, 
                             constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
        if res_slsqp.success and np.min(full_cons(res_slsqp.x)) >= -1e-7:
            best_vars = res_slsqp.x
            best_sum = -res_slsqp.fun
    except Exception:
        pass
        
    # Phase 3: Perturbation search to escape local minima
    for _ in range(40):
        x_p = best_vars + np.random.randn(3 * N_CIRCLES) * 0.004
        x_p[0::3] = np.clip(x_p[0::3], 0.01, 0.99)
        x_p[1::3] = np.clip(x_p[1::3], 0.01, 0.99)
        x_p[2::3] = np.clip(x_p[2::3], 1e-5, 0.49)
        
        try:
            res_p = minimize(full_obj, x_p, method='SLSQP', bounds=bounds_full, 
                             constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12})
            if res_p.success and np.min(full_cons(res_p.x)) >= -1e-7:
                val = -res_p.fun
                if val > best_sum:
                    best_sum = val
                    best_vars = res_p.x.copy()
        except Exception:
            pass
            
    # Phase 4: High precision polish
    try:
        res_final = minimize(full_obj, best_vars, method='SLSQP', bounds=bounds_full,
                             constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14})
        if res_final.success and np.min(full_cons(res_final.x)) >= -1e-8:
            best_vars = res_final.x
            best_sum = -res_final.fun
    except Exception:
        pass
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
