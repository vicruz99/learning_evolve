import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b_val in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b_val)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    # Fallback: conservative radii based on nearest neighbor and boundaries
    r_fallback = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        d_min = 1.0
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if d < d_min:
                    d_min = d
        r_fallback[i] = 0.9 * min(d_wall, d_min / 2.0)
    return np.maximum(r_fallback, 1e-6)

def center_objective(centers_flat):
    """Objective for center optimization: minimize negative sum of radii (computed via LP)."""
    centers = centers_flat.reshape(N_CIRCLES, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    r = solve_radii_lp(centers)
    return -np.sum(r)

def joint_objective(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def joint_constraints(v):
    """Inequality constraints for joint optimization: boundary and pairwise separation."""
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def generate_hex_init(seed, rot_range=(-0.3, 0.3)):
    """Generates a rotated hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    r_est = 0.07 + rng.rand() * 0.05
    pts = []
    y = r_est
    row = 0
    while len(pts) < N_CIRCLES:
        x_off = r_est if row % 2 == 1 else 0.0
        x = r_est + x_off
        while x <= 1.0 - r_est and len(pts) < N_CIRCLES:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N_CIRCLES])
    
    # Apply random rotation around center
    angle = rng.uniform(rot_range[0], rot_range[1])
    c, s = np.cos(angle), np.sin(angle)
    rot = np.array([[c, -s], [s, c]])
    pts = (pts - 0.5) @ rot.T + 0.5
    
    # Scale and jitter
    scale = rng.uniform(0.95, 1.05)
    pts = pts * scale + rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    np.random.seed(42)
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Phase 1: Global center search using Powell with LP objective
    inits = [generate_hex_init(s) for s in range(12)]
    
    for init_c in inits:
        try:
            res = minimize(center_objective, init_c.flatten(), method='Powell',
                           options={'maxiter': 300, 'xtol': 1e-7, 'ftol': 1e-9})
            curr_centers = np.clip(res.x.reshape(N_CIRCLES, 2), 1e-5, 1.0 - 1e-5)
            curr_radii = solve_radii_lp(curr_centers)
            curr_sum = np.sum(curr_radii)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = curr_centers.copy()
                best_radii = curr_radii.copy()
        except Exception:
            pass
            
    # Phase 2: Local perturbation search to refine topology
    for _ in range(40):
        c_pert = best_centers.copy()

        c_pert = np.clip(c_pert, 0.02, 0.98)
        r_pert = solve_radii_lp(c_pert)
        if np.sum(r_pert) > best_sum:

            best_radii = r_pert.copy()
            # Greedy improvement step
            for _ in range(8):

                c_pert2[idx2] += np.random.randn(2) * 0.004
                c_pert2 = np.clip(c_pert2, 0.02, 0.98)
                r_pert2 = solve_radii_lp(c_pert2)
                if np.sum(r_pert2) > best_sum:
                    best_sum = np.sum(r_pert2)
                    best_centers = c_pert2.copy()
                    best_radii = r_pert2.copy()

    bounds_full = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N_CIRCLES
    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = np.maximum(best_radii, 1e-6) * 0.995  # Slight shrink for strict feasibility
    
    try:
        res_joint = minimize(joint_objective, x0, method='SLSQP', bounds=bounds_full,
                             constraints={'type': 'ineq', 'fun': joint_constraints},
                             options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
        if res_joint.success:
            c_val = joint_constraints(res_joint.x)
            if np.min(c_val) >= -1e-7:
                new_sum = -res_joint.fun
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = np.column_stack((res_joint.x[0::3], res_joint.x[1::3]))
                    best_radii = res_joint.x[2::3]
    except Exception:
        pass
        
    # Phase 4: High-precision polish
    try:
        res_final = minimize(joint_objective, np.concatenate([best_centers.flatten(), best_radii]), 
                             method='SLSQP', bounds=bounds_full,
                             constraints={'type': 'ineq', 'fun': joint_constraints},
                             options={'maxiter': 2000, 'ftol': 1e-14, 'disp': False})
        if res_final.success and np.min(joint_constraints(res_final.x)) >= -1e-8:
            best_sum = -res_final.fun
            best_centers = np.column_stack((res_final.x[0::3], res_final.x[1::3]))
            best_radii = res_final.x[2::3]
    except Exception:
        pass
        
    # Ensure non-negative radii and basic validity
    best_radii = np.maximum(best_radii, 0.0)
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    # Final safety shrink if any constraint is slightly violated due to numerical noise
    for _ in range(3):
        valid = True
        for i in range(N_CIRCLES):
            x, y = best_centers[i]
            r = best_radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                best_radii[i] = max(0, min(x, 1-x, y, 1-y))
                valid = False
        for i in range(N_CIRCLES):
            for j in range(i+1, N_CIRCLES):
                d = np.sqrt(np.sum((best_centers[i] - best_centers[j])**2))
                if d < best_radii[i] + best_radii[j] - 1e-12:
                    best_radii[i] = max(0, (d - best_radii[j]) * 0.999)
                    best_radii[j] = max(0, (d - best_radii[i]) * 0.999)
                    valid = False
        if valid: break
            
    return best_centers, best_radii, float(np.sum(best_radii))
