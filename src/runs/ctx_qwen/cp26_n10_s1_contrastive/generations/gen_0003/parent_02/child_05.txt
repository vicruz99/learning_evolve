import numpy as np
from scipy.optimize import minimize, linprog

N = 26
_BOUNDS = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """
    Computes inequality constraints: boundary containment and pairwise separation.
    All constraints are formulated as g(v) >= 0.
    """
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    c.append(dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    
    return np.concatenate(c)

def solve_radii_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= distance(centers[i], centers[j])
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 2.6e-4

def generate_init(type_, seed, param):
    """Generates a strictly feasible initial center configuration."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    
    if type_ == 0:
        # Hexagonal lattice
        r_est = 0.095 * param
        y = r_est
        row = 0
        idx = 0
        while y < 1.0 and idx < N:
            x_off = (row % 2) * r_est
            x = r_est + x_off
            while x < 1.0 and idx < N:
                pts[idx] = [x, y]
                idx += 1
                x += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
    elif type_ == 1:
        # Square grid with central extra circle
        idx = 0
        for i in range(5):
            for j in range(5):
                pts[idx] = [0.1 + 0.2*i, 0.1 + 0.2*j]
                idx += 1
        if N > 25:
            pts[25] = [0.5, 0.5]
        pts += np.random.uniform(-0.05, 0.05, pts.shape) * param
    else:
        # Quasi-random spread
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    best_sum = -np.inf
    best_v = None
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Diverse structured starts with LP-initialized radii
    for i in range(70):
        np.random.seed(i)
        t = i % 3
        p = 0.9 + np.random.rand() * 0.2
        centers = generate_init(t, i, p)
        radii, _ = solve_radii_lp(centers)
        radii = np.maximum(radii, 1e-6) * 0.98
        
        v0 = np.zeros(3*N)
        v0[0::3] = centers[:, 0]
        v0[1::3] = centers[:, 1]
        v0[2::3] = radii
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=_BOUNDS, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if np.min(constraints(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass

    # Phase 2: Local perturbation & rotation search to escape local minima
    if best_v is not None:
        for k in range(70):
            np.random.seed(2000 + k)
            v_pert = best_v.copy()
            
            # Random small rotation around center to break symmetry
            if np.random.rand() < 0.5:
                angle = np.random.uniform(-0.12, 0.12)
                c, s = np.cos(angle), np.sin(angle)
                rot = np.array([[c, -s], [s, c]])
                ctr = np.column_stack((v_pert[0::3], v_pert[1::3]))
                ctr = (ctr - 0.5) @ rot.T + 0.5
                v_pert[0::3] = ctr[:, 0]
                v_pert[1::3] = ctr[:, 1]
                
            # Gaussian perturbation of centers
            v_pert[0::3] += np.random.randn(N) * 0.012
            v_pert[1::3] += np.random.randn(N) * 0.012
            v_pert[0::3] = np.clip(v_pert[0::3], 0.01, 0.99)
            v_pert[1::3] = np.clip(v_pert[1::3], 0.01, 0.99)
            
            # Recompute optimal radii via LP for strict feasibility
            c_pert = np.column_stack((v_pert[0::3], v_pert[1::3]))
            r_pert, _ = solve_radii_lp(c_pert)
            v_pert[2::3] = np.maximum(r_pert, 1e-6) * 0.96
            
            try:
                res = minimize(objective, v_pert, method='SLSQP', bounds=_BOUNDS, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                if np.min(constraints(res.x)) >= -1e-7:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_v = res.x.copy()
            except Exception:
                pass

    # Phase 3: High-precision final polish
    if best_v is not None:
        try:
            res_final = minimize(objective, best_v, method='SLSQP', bounds=_BOUNDS, constraints=cons,
                                 options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            if np.min(constraints(res_final.x)) >= -1e-8:
                best_v = res_final.x
                best_sum = -res_final.fun
        except Exception:
            pass
            
    # Fallback safety net
    if best_v is None:
        centers = np.random.uniform(0.2, 0.8, (N, 2))
        radii = np.full(N, 0.05)
        best_v = np.zeros(3*N)
        best_v[0::3] = centers[:,0]
        best_v[1::3] = centers[:,1]
        best_v[2::3] = radii
        best_sum = np.sum(radii)
        
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = np.maximum(best_v[2::3], 0.0)
    return centers, radii, float(np.sum(radii))
