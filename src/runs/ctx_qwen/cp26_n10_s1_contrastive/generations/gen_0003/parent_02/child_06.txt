import numpy as np
from scipy.optimize import minimize, linprog

N = 26


def solve_radii_lp(centers):
    """Given fixed centers, solve LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
    
    # Pairwise constraints: r_i + r_j <= dist(centers[i], centers[j])
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-6), 0.0


def center_objective(centers_flat):
    """Objective: maximize sum of radii for fixed centers (via LP)."""
    centers = centers_flat.reshape(N, 2)
    # Ensure centers are in valid region
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    _, sum_r = solve_radii_lp(centers)
    return -sum_r


def joint_constraints(v):
    """Inequality constraints for joint optimization: g(v) >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation: dist^2 >= (r_i + r_j)^2
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(dist2[mask] - rs[mask]**2)
    
    return np.concatenate(c)


def joint_objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])


def get_safe_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        d_min = np.inf
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if d < d_min:
                    d_min = d
        r[i] = 0.96 * min(d_wall, 0.5 * d_min)
    return np.maximum(r, 1e-6)


def generate_hex_pattern(row_counts, r_est=0.09, scale=1.0, rot=0.0, seed=42):
    """Generate hexagonal lattice with specified row counts."""
    rng = np.random.RandomState(seed)
    pts = []
    y = r_est
    
    for row_idx, count in enumerate(row_counts):
        shift = (row_idx % 2) * r_est
        x = r_est + shift
        for _ in range(count):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    
    pts = np.array(pts[:N])
    
    # Center and scale
    pts = pts - pts.mean(axis=0)
    pts = pts * scale
    pts = pts + 0.5
    
    # Apply rotation around center
    if abs(rot) > 1e-10:
        c, s = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ rot_mat.T + 0.5
    
    # Add jitter
    jitter = rng.randn(N, 2) * 0.02
    pts = pts + jitter
    pts = np.clip(pts, 0.03, 0.97)
    
    return pts


def force_simulate_layout(n_circles, steps, seed):
    """Generate layout using force-directed simulation."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.15, 0.85, (n_circles, 2))
    
    for step in range(steps):
        forces = np.zeros_like(pts)
        
        # Repulsion between points
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 0.01)
        f_mag = 1.0 / (dists**2)
        f_mag = np.clip(f_mag, 0, 10.0)
        forces += np.sum(f_mag[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        
        # Wall repulsion
        margin = 0.08
        for d in [0, 1]:
            below = pts[:, d] < margin
            above = pts[:, d] > 1.0 - margin
            forces[below, d] += 5.0 * (margin - pts[below, d])
            forces[above, d] -= 5.0 * (pts[above, d] - (1.0 - margin))
        
        # Center attraction (weak)
        forces -= 1.0 * (pts - 0.5)
        
        step_size = 0.003 * (1.0 - step / max(steps, 1))
        pts += step_size * forces
        pts = np.clip(pts, 0.03, 0.97)
    
    return pts
