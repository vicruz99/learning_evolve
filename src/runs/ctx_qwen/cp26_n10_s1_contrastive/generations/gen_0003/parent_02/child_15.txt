

<strategy>
The previous solutions relied on SLSQP optimizing all 78 variables (x, y, r) simultaneously. While effective, SLSQP often gets trapped in local minima due to the highly non-linear and rugged constraint landscape. 

A more robust approach for circle packing is to decouple the problem:
1. For any fixed set of centers, the optimal radii are found by solving a Linear Program (LP) that maximizes sum(r_i) subject to boundary and pairwise separation constraints. This LP can be solved extremely quickly (milliseconds) using `scipy.optimize.linprog`.
2. This transforms the problem into a 52-dimensional non-linear optimization over centers only, where the objective function is the negative sum of the LP-optimal radii. The LP acts as an exact, adaptive constraint handler.
3. We optimize this center-only objective using multiple restarts of `Nelder-Mead` (robust to non-smooth landscapes) combined with a perturbation refinement phase to escape local minima. Hexagonal lattice initializations with varying scales/rotations provide excellent starting topologies.
4. This decoupling avoids SLSQP's numerical stagnation, handles constraints exactly via LP, and focuses the global search effort on the geometric arrangement (centers), typically yielding tighter packings.
</strategy>

```python
import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def solve_lp_radii(centers):
    """Solves the LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)

    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        A_ub[i, i] = 1.0
        b_ub[i] = x
        A_ub[n + i, i] = 1.0
        b_ub[n + i] = 1.0 - x
        A_ub[2 * n + i, i] = 1.0
        b_ub[2 * n + i] = y
        A_ub[3 * n + i, i] = 1.0
        b_ub[3 * n + i] = 1.0 - y

    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    i_idx, j_idx = np.triu_indices(n, k=1)
    A_ub[4 * n:, i_idx] = 1.0
    A_ub[4 * n:, j_idx] = 1.0
    b_ub[4 * n:] = dists[i_idx, j_idx]

    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-7)

def objective_centers(c_flat):
    """Objective for center optimization: minimize negative sum of LP-optimal radii."""
    centers = np.clip(c_flat.reshape(N, 2), 0.0, 1.0)
    r = solve_lp_radii(centers)
    return -np.sum(r)

def generate_hex_init(scale=1.0, rot=0.0, seed=0):
    """Generates a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    r_est = 0.09 * scale
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    if rot != 0:
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
    pts = (pts - pts.mean(axis=0)) * 0.85 + 0.5
    pts += np.random.uniform(-0.02, 0.02, pts.shape)
    return pts

def run_packing():
    np.random.seed(42)
    best_obj = np.inf
    best_c_flat = None

    # Phase 1: Diverse structured initializations
    inits = []
    for i in range(10):
        inits.append(generate_hex_init(scale=0.85 + 0.15 * i, rot=0.06 * i - 0.2, seed=i))
    for i in range(5):
        np.random.seed(200 + i)
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))

    for pts in inits:
        c_flat = pts.flatten()
        try:
            res = minimize(objective_centers, c_flat, method='Nelder-Mead',
                           options={'maxiter': 1500, 'xatol': 1e-8, 'fatol': 1e-9})
            if res.fun < best_obj:
                best_obj = res.fun
                best_c_flat = res.x.copy()
        except Exception:
            continue

    # Phase 2: Perturbation refinement to escape local minima
    if best_c_flat is not None:
        for _ in range(25):
            pert = np.random.uniform(-0.006, 0.006, 2 * N)
            c_pert = best_c_flat + pert
            c_pert = np.clip(c_pert, 0.05, 0.95)
            try:
                res = minimize(objective_centers, c_pert, method='Nelder-Mead',
                               options={'maxiter': 1200, 'xatol': 1e-8, 'fatol': 1e-9})
                if res.fun < best_obj:
                    best_obj = res.fun
                    best_c_flat = res.x.copy()
            except Exception:
                continue

    # Fallback safety net
    if best_c_flat is None:
        best_c_flat = inits[0].flatten()

    centers = best_c_flat.reshape(N, 2)
    radii = solve_lp_radii(centers)
    
    # Final projection to ensure strict validity
    centers = np.clip(centers, 0.0, 1.0)
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
```
