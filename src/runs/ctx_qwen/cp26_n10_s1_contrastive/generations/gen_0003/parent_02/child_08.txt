import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26

def solve_radii_lp(centers):
    """
    Given fixed centers, solves the LP to find radii that maximize sum(r_i).
    Constraints:
    1. r_i >= 0
    2. r_i <= distance to boundaries
    3. r_i + r_j <= distance(centers[i], centers[j])
    """
    n = centers.shape[0]
    c = -np.ones(n)
    
    # Preallocate constraint matrix and RHS
    n_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[idx, i] = 1.0
            b_ub[idx] = b
            idx += 1
            
    # Pairwise distance constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-6), 1e-6

def lp_obj(centers_flat):
    """Objective for center optimization: minimize negative sum of optimal radii."""
    centers = np.clip(centers_flat.reshape(N_CIRCLES, 2), 0.0, 1.0)
    _, sum_r = solve_radii_lp(centers)
    return -sum_r

def force_init(seed):
    """Generate a well-spread initial configuration using force-directed layout."""
    np.random.seed(seed)
    pts = np.random.rand(N_CIRCLES, 2)
    for _ in range(400):
        forces = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.where(dists < 1e-5, 1e-5, dists)
        f_mag = np.clip(1.0 / (dists**2), 0, 30.0)
        forces += np.sum(f_mag[:, :, None] * diff / dists[:, :, None], axis=1)
        
        margin = 0.06
        for d in [0, 1]:
            below = pts[:, d] < margin
            above = pts[:, d] > 1.0 - margin
            forces[below, d] += 15.0 * (margin - pts[below, d])
            forces[above, d] -= 15.0 * (pts[above, d] - (1.0 - margin))
            
        pts += 0.004 * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts.flatten()

def hex_init():
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.09
    y = r_est
    row = 0
    while len(pts) < N_CIRCLES:
        x_off = (row % 2) * r_est
        x = r_est + x_off
        while x <= 1.0 - r_est and len(pts) < N_CIRCLES:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    return np.array(pts[:N_CIRCLES]).flatten()

def grid_init():
    """Generate a square grid + center initialization."""
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + 0.2 * i, 0.1 + 0.2 * j])
    pts.append([0.5, 0.5])
    return np.array(pts[:N_CIRCLES]).flatten()

def run_packing():
    best_sum = -np.inf
    best_centers = None
    
    # Phase 1: Diverse starts optimized with Nelder-Mead over centers only
    inits = [grid_init(), hex_init()]
    for s in range(15):
        inits.append(force_init(s))
        
    for init_c in inits:
        try:
            res = minimize(lp_obj, init_c, method='Nelder-Mead', 
                           options={'maxiter': 8000, 'xatol': 1e-8, 'fatol': 1e-9})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = np.clip(res.x.reshape(N_CIRCLES, 2), 0.0, 1.0)
        except Exception:
            continue
            
    # Extract optimal radii for the best centers found
    radii, _ = solve_radii_lp(best_centers)
    
    # Phase 2: High-precision joint polish with SLSQP
    def full_obj(v):
        return -np.sum(v[2::3])
        
    def full_cons(v):
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        c = []
        # Boundary constraints
        c.append(x - r)
        c.append(1.0 - x - r)
        c.append(y - r)
        c.append(1.0 - y - r)
        # Pairwise separation
        dx = x[:, None] - x[None, :]
        dy = y[:, None] - y[None, :]
        dist2 = dx**2 + dy**2
        rs = r[:, None] + r[None, :]
        mask = np.triu(np.ones((N_CIRCLES, N_CIRCLES), dtype=bool), k=1)
        c.append(dist2[mask] - rs[mask]**2)
        return np.concatenate(c)
        
    v0 = np.zeros(3 * N_CIRCLES)
    v0[0::3] = best_centers[:, 0]
    v0[1::3] = best_centers[:, 1]
    v0[2::3] = radii * 0.995  # Slight shrink ensures strict feasibility for interior-point steps
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': full_cons}
    
    try:
        res_polish = minimize(full_obj, v0, method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 6000, 'ftol': 1e-13})
        # Verify feasibility before accepting polish
        if np.min(full_cons(res_polish.x)) >= -1e-7:
            best_centers = np.column_stack((res_polish.x[0::3], res_polish.x[1::3]))
            radii = res_polish.x[2::3]
            best_sum = -res_polish.fun
    except Exception:
        pass
        
    radii = np.maximum(radii, 0.0)
    return best_centers, radii, float(best_sum)
