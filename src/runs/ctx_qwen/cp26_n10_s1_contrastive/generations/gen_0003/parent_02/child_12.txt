N = 26

def compute_pairwise_indices():
    """Precompute upper triangular indices for pairwise constraints."""
    return np.triu_indices(N, k=1)

def objective_full(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints_full(v):
    """Inequality constraints: boundary containment and pairwise non-overlap, g(v) >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    c.append(dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    
    return np.concatenate(c)

def objective_param(v):
    """Objective for parameterized version: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def constraints_param(v):
    """Inequality constraints for parameterized version (boundary-free).
    
    Variables: r[i], u[i], v[i] where:
    x[i] = r[i] + u[i] * (1 - 2*r[i])
    y[i] = r[i] + v[i] * (1 - 2*r[i])
    
    Only pairwise separation constraints needed.
    """
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    
    denom = 1.0 - 2.0 * r
    denom = np.clip(denom, 1e-12, 1.0)
    x = r + u * denom
    y = r + w * denom
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    return dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(max(b, 0.0))
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    for i, j in zip(i_idx, j_idx):
        row = np.zeros(n)
        row[i] = 1.0
        row[j] = 1.0
        A_ub.append(row)
        b_ub.append(dists[i, j])
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    
    return np.full(n, 1e-8), 1e-8

def get_safe_radii(centers, factor=0.98):
    """Compute strictly feasible radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        d_min = np.inf
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i]-centers[j])**2))
                if d < d_min:
                    d_min = d
        r[i] = factor * min(max(d_wall, 0.0), 0.5 * d_min)
    return np.maximum(r, 1e-8)

def force_directed_layout(seed, steps=800):
    """Generate center configuration using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    
    for step in range(steps):
        forces = np.zeros_like(pts)
        
        # Pairwise repulsion (Coulomb-like)
        for i in range(N):
            for j in range(i+1, N):
                diff = pts[i] - pts[j]
                dist_sq = np.sum(diff**2)
                dist = np.sqrt(max(dist_sq, 1e-12))
                f_mag = 1.0 / max(dist_sq, 0.01)
                f_mag = min(f_mag, 20.0)
                f_vec = (diff / dist) * f_mag
                forces[i] += f_vec
                forces[j] -= f_vec
        
        # Wall repulsion
        margin = 0.08
        for i in range(N):
            for d in range(2):
                if pts[i, d] < margin:
                    forces[i, d] += 15.0 * (margin - pts[i, d])
                elif pts[i, d] > 1.0 - margin:
                    forces[i, d] -= 15.0 * (pts[i, d] - (1.0 - margin))
        
        # Center attraction (gentle)
        forces -= 2.0 * (pts - 0.5)
        
        step_size = 0.003 * (1.0 - step / steps) + 0.0005
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
    
    return pts

def hex_lattice_init(row_counts, seed, r_est=0.095):
    """Generate hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    y = r_est
    for row_idx, count in enumerate(row_counts):
        x_start = r_est if row_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            if len(pts) < N:
                pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    
    pts = np.array(pts[:N])
    jitter = rng.uniform(-0.02, 0.02, pts.shape)
    pts = pts + jitter
    pts = np.clip(pts, 0.03, 0.97)
    return pts
