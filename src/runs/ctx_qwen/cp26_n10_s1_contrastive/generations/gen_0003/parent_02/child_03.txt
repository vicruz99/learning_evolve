

<strategy>
The current approach uses direct SLSQP joint optimization of centers and radii, which often gets trapped in local minima due to the highly non-convex constraint landscape. To escape these and approach the theoretical limit, we will employ a two-phase hybrid strategy:

1. **Topology Search via Center Optimization with LP Radii Evaluation**: 
   Instead of optimizing centers and radii jointly from the start, we fix the centers and solve a Linear Program (LP) to find the optimal radii for that specific configuration. This inner LP maximizes $\sum r_i$ subject to boundary and pairwise constraints. We then use this LP's optimal sum as the objective function for a derivative-free optimizer (`Nelder-Mead`) acting only on the centers. This separates the difficult topological arrangement problem from the continuous sizing problem, allowing the optimizer to "feel" the global landscape more effectively.

2. **Diverse Hexagonal & Lattice Initializations**: 
   We generate multiple structured initial configurations (hexagonal lattices with varying row counts, rotations, and scales) to cover different topological basins. Circle packing in squares often converges to slightly rotated or stretched hexagonal patterns.

3. **High-Precision Joint Refinement**: 
   Once the best center configuration is identified, we scale down the LP radii slightly to ensure strict interior feasibility, then run a high-precision SLSQP joint optimization to polish both positions and sizes simultaneously. This captures subtle improvements where center shifts and radius changes co-optimize.

4. **Robustness & Fallbacks**: 
   We include analytical radius fallbacks, strict feasibility checks, and a final perturbation search to guarantee a valid, high-scoring packing regardless of optimizer quirks.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= dist to each wall
    for i in range(n):
        x, y = centers[i]
        bounds_vals = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_vals:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback analytical radii if LP fails
    r_fallback = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        d_min = 1.0
        for j in range(n):
            if i != j:
                d = dists[i, j]
                if d < d_min:
                    d_min = d
        r_fallback[i] = 0.99 * min(d_wall, 0.5 * d_min)
    return r_fallback, np.sum(r_fallback)

def center_objective(x_flat):
    """Objective for center optimization: negative sum of optimal radii."""
    centers = x_flat.reshape(N, 2)
    _, sum_r = solve_lp_radii(centers)
    return -sum_r

def generate_hex_centers(row_counts, jitter_scale=0.0):
    """Generates centers based on hexagonal lattice with specified row counts."""
    pts = []
    r_est = 0.105
    y = r_est
    for r_idx, count in enumerate(row_counts):
        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    centers = np.array(pts[:N])
    
    if jitter_scale > 0:
        centers += np.random.uniform(-jitter_scale, jitter_scale, centers.shape)
        centers = np.clip(centers, 0.02, 0.98)
        
    return centers.flatten()

def joint_constraints(v):
    """Inequality constraints for joint SLSQP optimization."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(dist2[mask] - rs[mask]**2)
    
    return np.concatenate(c)

def joint_objective(v):
    return -np.sum(v[2::3])

def run_packing():
    np.random.seed(42)
    bounds_joint = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_joint = {'type': 'ineq', 'fun': joint_constraints}
    
    best_sum = -np.inf
    best_v = None
    
    # Phase 1: Topology search via LP-augmented center optimization
    # Diverse row counts for hexagonal patterns
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [5, 5, 5, 5, 6], [6, 4, 6, 5, 5], [5, 6, 4, 6, 5],
        [5, 5, 6, 5, 5], [6, 5, 5, 5, 5]
    ]
    
    init_centers_flat = []
    for pat in patterns:
        for jitter in [0.0, 0.01, 0.025]:
            init_centers_flat.append(generate_hex_centers(pat, jitter))
            
    # Add purely random starts
    for _ in range(5):
        pts = np.random.uniform(0.05, 0.95, (N, 2))
        init_centers_flat.append(pts.flatten())
        
    print(f"Running {len(init_centers_flat)} topology optimizations...")
    for idx, x0 in enumerate(init_centers_flat):
        try:
            # Nelder-Mead is robust for non-smooth LP objective
            res = minimize(center_objective, x0, method='Nelder-Mead',
                           options={'maxiter': 1500, 'xatol': 1e-7, 'fatol': 1e-7})
            
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                # Extract centers and get LP radii
                c_opt = res.x.reshape(N, 2)
                r_opt, _ = solve_lp_radii(c_opt)
                
                best_v = np.zeros(3 * N)
                best_v[0::3] = c_opt[:, 0]
                best_v[1::3] = c_opt[:, 1]
                best_v[2::3] = r_opt
        except Exception:
            continue
            
    # Phase 2: Joint SLSQP refinement to polish positions and sizes
    if best_v is not None:
        # Shrink radii slightly to guarantee strict interior start for SLSQP
        refine_v = best_v.copy()
        refine_v[2::3] *= 0.985
        refine_v[0::3] = np.clip(refine_v[0::3], 0.01, 0.99)
        refine_v[1::3] = np.clip(refine_v[1::3], 0.01, 0.99)
        
        try:
            res_joint = minimize(joint_objective, refine_v, method='SLSQP',
                                 bounds=bounds_joint, constraints=cons_joint,
                                 options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res_joint.success:
                c_val = joint_constraints(res_joint.x)
                if np.min(c_val) >= -1e-7:
                    s = -res_joint.fun
                    if s > best_sum:
                        best_sum = s
                        best_v = res_joint.x.copy()
        except Exception:
            pass
            
        # Phase 3: Local perturbation search around best to escape plateaus
        for _ in range(20):
            v_pert = best_v.copy()
            v_pert[0::3] += np.random.uniform(-0.005, 0.005, N)
            v_pert[1::3] += np.random.uniform(-0.005, 0.005, N)
            v_pert[0::3] = np.clip(v_pert[0::3], 0.01, 0.99)
            v_pert[1::3] = np.clip(v_pert[1::3], 0.01, 0.99)
            v_pert[2::3] *= 0.99  # Ensure feasibility
            
            try:
                res_p = minimize(joint_objective, v_pert, method='SLSQP',
                                 bounds=bounds_joint, constraints=cons_joint,
                                 options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                if res_p.success:
                    c_val = joint_constraints(res_p.x)
                    if np.min(c_val) >= -1e-7:
                        s = -res_p.fun
                        if s > best_sum:
                            best_sum = s
                            best_v = res_p.x.copy()
            except Exception:
                continue
                
        # Phase 4: High-precision final polish
        try:
            res_final = minimize(joint_objective, best_v, method='SLSQP',
                                 bounds=bounds_joint, constraints=cons_joint,
                                 options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if res_final.success and np.min(joint_constraints(res_final.x)) >= -1e-8:
                best_v = res_final.x
                best_sum = -res_final.fun
        except Exception:
            pass

    # Fallback safety net
    if best_v is None:
        centers = np.random.uniform(0.2, 0.8, (N, 2))
        radii = np.full(N, 0.05)
        best_v = np.zeros(3 * N)
        best_v[0::3] = centers[:, 0]
        best_v[1::3] = centers[:, 1]
        best_v[2::3] = radii
        best_sum = np.sum(radii)
        
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = np.maximum(best_v[2::3], 0.0)
    return centers, radii, float(best_sum)
