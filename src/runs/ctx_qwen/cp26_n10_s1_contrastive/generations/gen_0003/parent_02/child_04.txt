N = 26

def objective(v):

    return -np.sum(v[2::3])

def constraint_func(v):
    """Inequality constraints: boundary containment and pairwise separation."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)

    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(dist2[mask] - rs[mask]**2)
    
    return np.concatenate(c)

def solve_radii_lp(centers):

            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)

    bounds = [(0.0, None)] * n

def get_safe_radii(centers):
    """Compute strictly feasible radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])

        for j in range(n):

                if d < d_min:
                    d_min = d
        r[i] = 0.97 * min(d_wall, 0.5 * d_min)
    return np.maximum(r, 1e-6)

            x += 2.0 * r_est
        y += dy

    # Jitter

    pts = pts + jitter
    pts = np.clip(pts, 0.02, 0.98)
    
    return pts
