import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_sum_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i) under constraints."""
    n = centers.shape[0]
    c = -np.ones(n)  # Maximize sum(r) <=> Minimize -sum(r)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def objective_nested(x):
    """Objective for outer optimizer: maximize sum of optimal radii for given centers."""
    return -compute_sum_radii(x.reshape(N, 2))[0]

def objective_joint(v):
    """Objective for joint SLSQP polish."""
    return -np.sum(v[2::3])

def constraint_joint(v):
    """Inequality constraints for joint optimization: g(v) >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(dist2[mask] - rs[mask]**2)
    
    return np.concatenate(c)

def generate_hex_init(row_counts, seed):
    """Generates a hexagonal lattice initialization with specified row counts."""
    np.random.seed(seed)
    pts = []
    r_est = 0.105
    dy = np.sqrt(3.0) * r_est
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += dy
    pts = np.array(pts[:N])
    # Add controlled jitter to break exact symmetry
    pts += np.random.uniform(-0.025, 0.025, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts.flatten()

def run_packing():

    # Powell is robust for non-smooth landscapes arising from LP active set changes
    for x0 in inits:

    return best_centers, final_radii, float(final_sum)
