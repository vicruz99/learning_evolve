<strategy>
1. **Use Differential Evolution on center coordinates**: DE is a global optimizer that doesn't get stuck in local minima as easily as SLSQP. By optimizing only centers (52 variables) and computing radii via LP afterward, we avoid the non-convex constraint issues.

2. **LP-based radius computation for fixed centers**: Given centers, finding optimal radii is a linear program (maximize sum of r_i subject to boundary and pairwise constraints). This is exact and fast.

3. **Multi-phase refinement**:
   - Phase 1: DE on centers with fast objective to explore globally
   - Phase 2: Diverse hex lattice initializations + SLSQP refinement
   - Phase 3: Perturbation search around best solution
   - Phase 4: Variable permutation to break symmetry
   - Phase 5: Final high-precision polish

4. **Better initialization diversity**: Multiple hex lattice patterns with various rotations, scalings, and row counts. The rotation parameter is crucial for breaking symmetries that trap local optimizers.

5. **Force-directed layout**: Generate naturally spread configurations that are closer to optimal.

6. **Trust-constr method**: Try scipy's trust-constr as an alternative to SLSQP for better constraint handling.

The key insight is combining global search (DE) with exact LP radius computation and local refinement (SLSQP) to avoid getting trapped in local minima.

import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution
import warnings
warnings.filterwarnings('ignore')

N = 26

def fast_objective(centers_flat):
    """Fast objective for center optimization: minimize negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    d_wall = np.min(np.stack([
        centers[:, 0], 1.0 - centers[:, 0],
        centers[:, 1], 1.0 - centers[:, 1]
    ], axis=1), axis=1)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    d_min = np.min(dists, axis=1)
    r = np.minimum(d_wall, d_min / 2.0)
    return -np.sum(r)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-8), 0.0

def sgs_objective(v):
    return -np.sum(v[2::3])

def sgs_constraints(v):
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(dist2[mask] - rs[mask]**2)
    return np.concatenate(c)

def make_hex_config(pattern, seed, jitter=0.01, rotation=0.0, scale=1.0):
    """Generate hex lattice configuration with specified parameters."""
    np.random.seed(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(pattern):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    pts = pts - 0.5
    pts = pts * scale
    pts = pts + 0.5
    if rotation != 0.0:
        c_val, s_val = np.cos(rotation), np.sin(rotation)
        rot_mat = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = pts @ rot_mat.T
    pts += np.random.uniform(-jitter, jitter, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def force_layout(n, steps, seed):
    """Force-directed layout to spread points."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.15, 0.85, (n, 2))
    for step in range(steps):
        forces = np.zeros_like(centers)
        alpha = 1.0 / (1.0 + step * 0.008)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        f_mag = np.clip(1.0 / np.maximum(dists**2, 0.0001), 0, 20.0)
        forces += np.sum(f_mag[:, :, np.newaxis] * diff / np.maximum(dists[:, :, np.newaxis], 0.01), axis=1)
        margin = 0.08
        for d in [0, 1]:
            below = centers[:, d] < margin
            above = centers[:, d] > 1.0 - margin
            forces[below, d] += 30.0 * (margin - centers[below, d])
            forces[above, d] -= 30.0 * (centers[above, d] - (1.0 - margin))
        centers += 0.005 * alpha * forces
        centers = np.clip(centers, 0.03, 0.97)
    return centers

def make_grid_config(seed):
    """5x5 grid + center with perturbation."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [0.1 + 0.2 * i, 0.1 + 0.2 * j]
            idx += 1
    centers[25] = [0.5, 0.5]
    centers += np.random.uniform(-0.025, 0.025, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    cons = {'type': 'ineq', 'fun': sgs_constraints}
    bounds_3d = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    bounds_2d = [(0.03, 0.97)] * (2 * N)
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [4, 6, 5, 6, 5],
        [5, 5, 5, 5, 6], [6, 4, 6, 5, 5], [5, 6, 4, 6, 5],
        [4, 5, 6, 5, 6], [5, 5, 6, 5, 5], [6, 5, 5, 5, 5],
        [5, 5, 5, 6, 5], [5, 6, 5, 5, 5], [6, 5, 5, 6, 4],
        [5, 4, 5, 6, 6], [6, 6, 5, 5, 4], [4, 4, 6, 6, 6],
    ]

    # Phase 1: Differential Evolution on center coordinates
    np.random.seed(42)
    for de_seed in [42, 123, 456]:
        try:
            res_de = differential_evolution(
                fast_objective,
                bounds_2d,
                popsize=12,
                maxiter=200,
                tol=1e-8,
                mutation=(0.6, 1.0),
                recombination=0.75,
                seed=de_seed,
                workers=1,
                updating='deferred'
            )
            de_centers = res_de.x.reshape(N, 2)
            de_radii, neg_sum = solve_lp_radii(de_centers)
            if -neg_sum > best_sum:
                best_sum = -neg_sum
                best_centers = de_centers.copy()
                best_radii = de_radii.copy()
        except Exception:
            pass

    # Phase 2: Hex lattice initializations with SLSQP
    for pat in patterns:
        for seed in range(8):
            for rot in [0.0, 0.03, -0.03, 0.06, -0.06, 0.1, -0.1]:
                for scale in [0.95, 1.0, 1.05]:
                    centers = make_hex_config(pat, seed, jitter=0.012, rotation=rot, scale=scale)
                    radii, _ = solve_lp_radii(centers)
                    radii_safe = np.maximum(radii, 1e-6) * 0.995
                    v0 = np.zeros(3 * N)
                    v0[0::3] = centers[:, 0]
                    v0[1::3] = centers[:, 1]
                    v0[2::3] = radii_safe
                    try:
                        res = minimize(sgs_objective, v0, method='SLSQP', bounds=bounds_3d,
                                       constraints=cons,
                                       options={'maxiter': 2500, 'ftol': 1e-12, 'disp': False})
                        if np.min(sgs_constraints(res.x)) >= -1e-6:
                            s = -res.fun
                            if s > best_sum:
                                best_sum = s
                                best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                                best_radii = res.x[2::3]
                    except Exception:
                        pass

    # Phase 2b: Force-directed initializations
    for fs in range(10):
        centers = force_layout(N, steps=300, seed=fs * 7 + 3)
        radii, _ = solve_lp_radii(centers)
        radii_safe = np.maximum(radii, 1e-6) * 0.995
        v0 = np.zeros(3 * N)
        v0[0::3] = centers[:, 0]
        v0[1::3] = centers[:, 1]
        v0[2::3] = radii_safe
        try:
            res = minimize(sgs_objective, v0, method='SLSQP', bounds=bounds_3d,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            if np.min(sgs_constraints(res.x)) >= -1e-6:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3]
        except Exception:
            pass

    # Phase 2c: Grid initializations
    for gs in range(8):
        centers = make_grid_config(gs)
        radii, _ = solve_lp_radii(centers)
        radii_safe = np.maximum(radii, 1e-6) * 0.995
        v0 = np.zeros(3 * N)
        v0[0::3] = centers[:, 0]
        v0[1::3] = centers[:, 1]
        v0[2::3] = radii_safe
        try:
            res = minimize(sgs_objective, v0, method='SLSQP', bounds=bounds_3d,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            if np.min(sgs_constraints(res.x)) >= -1e-6:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3]
        except Exception:
            pass

    # Phase 3: Perturbation search around best solution
    if best_centers is not None:
        for pseed in range(40):
            np.random.seed(pseed + 1000)
            v_pert = np.zeros(3 * N)
            c_pert = best_centers.copy()
            c_pert += np.random.uniform(-0.006, 0.006, (N, 2))
            c_pert = np.clip(c_pert, 0.02, 0.98)
            r_pert, _ = solve_lp_radii(c_pert)
            r_pert = np.maximum(r_pert, 1e-6) * 0.995
            v_pert[0::3] = c_pert[:, 0]
            v_pert[1::3] = c_pert[:, 1]
            v_pert[2::3] = r_pert
            try:
                res = minimize(sgs_objective, v_pert, method='SLSQP', bounds=bounds_3d,
                               constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                if np.min(sgs_constraints(res.x)) >= -1e-6:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
            except Exception:
                pass

    # Phase 4: Variable permutation to break solver symmetry
    if best_centers is not None:
        for perm_seed in range(15):
            np.random.seed(perm_seed + 5000)
            perm = np.random.permutation(N)
            v_perm = np.zeros(3 * N)
            for i in range(N):
                v_perm[3 * i] = best_centers[perm[i], 0]
                v_perm[3 * i + 1] = best_centers[perm[i], 1]
                v_perm[3 * i + 2] = best_radii[perm[i]] * 0.995
            v_perm[0::3] = np.clip(v_perm[0::3], 0.01, 0.99)
            v_perm[1::3] = np.clip(v_perm[1::3], 0.01, 0.99)
            try:
                res = minimize(sgs_objective, v_perm, method='SLSQP', bounds=bounds_3d,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if np.min(sgs_constraints(res.x)) >= -1e-6:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
            except Exception:
                pass

    # Phase 5: Center-only Nelder-Mead refinement on best result
    if best_centers is not None:
        centers_flat = best_centers.flatten()
        try:
            res_nm = minimize(fast_objective, centers_flat, method='Nelder-Mead',
                              options={'maxiter': 50000, 'xatol': 1e-8, 'fatol': 1e-10, 'adaptive': True})
            nm_centers = res_nm.x.reshape(N, 2)
            nm_radii, neg_sum = solve_lp_radii(nm_centers)
            if -neg_sum > best_sum:
                best_sum = -neg_sum
                best_centers = nm_centers.copy()
                best_radii = nm_radii.copy()
        except Exception:
            pass

    # Phase 6: Final high-precision SLSQP polish on absolute best
    if best_centers is not None:
        v_final = np.zeros(3 * N)
        v_final[0::3] = best_centers[:, 0]
        v_final[1::3] = best_centers[:, 1]
        r_safe = np.maximum(best_radii, 1e-6) * 0.998
        v_final[2::3] = r_safe
        try:
            res_final = minimize(sgs_objective, v_final, method='SLSQP', bounds=bounds_3d,
                                 constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if np.min(sgs_constraints(res_final.x)) >= -1e-8:
                best_sum = -res_final.fun
                best_centers = np.column_stack((res_final.x[0::3], res_final.x[1::3]))
                best_radii = res_final.x[2::3]
        except Exception:
            pass

    # Fallback safety net
    if best_centers is None:
        centers_fallback = np.zeros((N, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                centers_fallback[idx] = [0.1 + 0.2 * i, 0.1 + 0.2 * j]
                idx += 1
        centers_fallback[25] = [0.5, 0.5]
        radii_fallback = np.full(N, 0.085)
        return centers_fallback, radii_fallback, float(np.sum(radii_fallback))

    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
