import numpy as np
from scipy.optimize import minimize

N = 26
NUM_BOUND = 4 * N
NUM_PAIR = N * (N - 1) // 2
TOTAL_CONST = NUM_BOUND + NUM_PAIR
i_idx, j_idx = np.triu_indices(N, k=1)

def cons_and_jac(vars_vec):
    """Computes constraint values and their analytical Jacobian matrix."""
    xc = vars_vec[0::3]
    yc = vars_vec[1::3]
    r = vars_vec[2::3]
    
    c = np.empty(TOTAL_CONST)
    c[:N] = xc - r
    c[N:2*N] = 1.0 - xc - r
    c[2*N:3*N] = yc - r
    c[3*N:4*N] = 1.0 - yc - r
    
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c[4*N:] = dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2
    
    jac = np.zeros((TOTAL_CONST, 3 * N))
    
    # Boundary constraints jacobian
    for i in range(N):
        jac[i, 3*i] = 1.0
        jac[i, 3*i+2] = -1.0
        jac[N+i, 3*i] = -1.0
        jac[N+i, 3*i+2] = -1.0
        jac[2*N+i, 3*i+1] = 1.0
        jac[2*N+i, 3*i+2] = -1.0
        jac[3*N+i, 3*i+1] = -1.0
        jac[3*N+i, 3*i+2] = -1.0
        
    # Pairwise constraints jacobian
    dx_ij = dx[i_idx, j_idx]
    dy_ij = dy[i_idx, j_idx]
    rs_ij = rs[i_idx, j_idx]
    
    for k in range(NUM_PAIR):
        i = i_idx[k]
        j = j_idx[k]
        row = 4*N + k
        
        jac[row, 3*i] = 2.0 * dx_ij[k]
        jac[row, 3*i+1] = 2.0 * dy_ij[k]
        jac[row, 3*i+2] = -2.0 * rs_ij[k]
        
        jac[row, 3*j] = -2.0 * dx_ij[k]
        jac[row, 3*j+1] = -2.0 * dy_ij[k]
        jac[row, 3*j+2] = -2.0 * rs_ij[k]
        
    return c, jac

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[2::3])

def get_feasible_radii(centers):
    """Computes strictly feasible initial radii for given centers."""
    r = np.zeros(N)
    for i in range(N):
        db = min(centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1])
        diffs = centers[i] - centers
        dists = np.sqrt(np.sum(diffs**2, axis=1))
        dists[i] = np.inf
        dp = np.min(dists)
        r[i] = 0.94 * min(db, dp / 2.0)
    return np.maximum(r, 1e-6)

def generate_inits():
    """Generates diverse initial center configurations."""
    inits = []
    # Hexagonal lattice variations
    for seed in range(25):
        np.random.seed(seed)
        pts = []
        r_est = 0.07 + np.random.rand() * 0.04
        y = r_est
        row = 0
        while len(pts) < N:
            shift = (row % 2) * r_est
            x = r_est + shift
            while x <= 1.0 - r_est and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3) * r_est
            row += 1
        pts = np.array(pts[:N])
        pts += np.random.uniform(-0.02, 0.02, (N, 2))
        pts = np.clip(pts, 0.04, 0.96)
        inits.append(pts)
        
    # Grid-based variations
    for seed in range(15):
        np.random.seed(seed + 100)
        pts = np.array([[0.1 + 0.2*i + np.random.uniform(-0.02, 0.02),
                         0.1 + 0.2*j + np.random.uniform(-0.02, 0.02)]
                        for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        pts = np.clip(pts, 0.04, 0.96)
        inits.append(pts)
        
    # Force-directed layouts
    for seed in range(20):
        np.random.seed(seed + 200)
        pts = np.random.rand(N, 2) * 0.8 + 0.1
        for _ in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            d2 = np.sum(diff**2, axis=2) + 1e-6
            d = np.sqrt(d2)
            f = diff / (d2[:, :, None] * d[:, :, None])
            forces = np.sum(f, axis=0)
            for k in range(2):
                forces[:, k] += 10.0 * (np.maximum(0, 0.05 - pts[:, k]) - np.maximum(0, pts[:, k] - 0.95))
            pts += 0.003 * forces
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
    return inits

def run_packing():

    cons = {'type': 'ineq', 'fun': cons_and_jac}
    
    best_sum = -1.0
    best_vars = None
    
    inits = generate_inits()
    
    # Phase 1: Direct optimization from diverse starts
    for idx, centers_init in enumerate(inits):
        r_init = get_feasible_radii(centers_init)
        x0 = np.zeros(3 * N)
        x0[0::3] = centers_init[:, 0]
        x0[1::3] = centers_init[:, 1]
        x0[2::3] = r_init
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
            if res.success:
                c_val, _ = cons_and_jac(res.x)
                if np.min(c_val) >= -1e-7:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_vars = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation refinement to escape local minima

            try:

                        s = -res.fun
                        if s > best_sum:

            except Exception:
                pass

    if best_vars is None:
        centers = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        r = np.full(N, 0.09)

        best_vars[2::3] = r
        best_sum = np.sum(r)
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(best_sum)
