import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[2::3])

def compute_constraints(vars_vec):
    """
    Computes inequality constraints g(vars_vec) >= 0.
    Includes boundary containment and pairwise non-overlap.
    """
    cx = vars_vec[0::3]
    cy = vars_vec[1::3]
    r = vars_vec[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    d_wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    dists = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_dist = np.min(dists, axis=1)
    r = np.minimum(d_wall, min_dist / 2.0)
    return np.maximum(r * 0.985, 1e-6)

def generate_hex_init(seed, base_r=0.10, rot=0.0):
    """Generates a hexagonal lattice initialization with perturbation."""
    np.random.seed(seed)
    pts = []
    y = base_r
    row = 0
    while len(pts) < N:
        x_start = base_r if row % 2 == 0 else 2.0 * base_r
        x = x_start
        while x <= 1.0 - base_r and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * base_r
        y += np.sqrt(3.0) * base_r
        row += 1
        
    pts = np.array(pts[:N])
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ rot_mat.T + 0.5
        
    pts += np.random.uniform(-0.012, 0.012, pts.shape)
    pts = np.clip(pts, 0.025, 0.975)
    
    r = get_feasible_radii(pts)
    x0 = np.empty(3 * N)
    x0[0::3] = pts[:, 0]
    x0[1::3] = pts[:, 1]
    x0[2::3] = r
    return x0

def generate_force_init(seed):
    """Generates a force-directed layout initialization."""
    np.random.seed(seed)
    pts = np.random.rand(N, 2) * 0.8 + 0.1
    for _ in range(120):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-5)
        forces = np.zeros_like(pts)
        
        f_mag = 1.0 / (dists**2)
        f_mag = np.clip(f_mag, 0, 40.0)
        forces += np.sum(f_mag[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            below = pts[:, d] < 0.12
            above = pts[:, d] > 0.88
            forces[below, d] += 12.0 * (0.12 - pts[below, d])
            forces[above, d] -= 12.0 * (pts[above, d] - 0.88)
            
        pts += 0.006 * forces
        pts = np.clip(pts, 0.03, 0.97)
        
    r = get_feasible_radii(pts)
    x0 = np.empty(3 * N)
    x0[0::3] = pts[:, 0]
    x0[1::3] = pts[:, 1]
    x0[2::3] = r
    return x0

def run_packing():
    np.random.seed(42)
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_x = None
    best_sum = -np.inf
    
    inits = []
    
    # Phase 1: Structured hexagonal initializations
    base_rs = np.linspace(0.098, 0.107, 6)
    rotations = [-0.15, -0.05, 0.05, 0.15]
    for br in base_rs:
        for rot in rotations:
            for s in range(2):
                inits.append(generate_hex_init(s, base_r=br, rot=rot))
                
    # Phase 2: Force-directed initializations
    for s in range(15):
        inits.append(generate_force_init(s))
        
    # Phase 3: Random layered initializations
    for s in range(10):
        np.random.seed(100 + s)
        pts = np.random.rand(N, 2) * 0.82 + 0.09
        pts = pts[np.argsort(pts[:, 1])]
        r = get_feasible_radii(pts)
        x0 = np.empty(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r
        inits.append(x0)

    # Optimization loop over all initializations
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2500, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = compute_constraints(res.x)
                if np.min(c_val) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
        except Exception:
            continue

    # Phase 4: Perturbation refinement to escape local minima
    if best_x is not None:
        for k in range(35):
            x_pert = best_x.copy()
            noise = np.random.randn(3 * N)
            x_pert[0::3] += noise[0::3] * 0.018
            x_pert[1::3] += noise[1::3] * 0.018
            x_pert[2::3] += noise[2::3] * 0.0025
            
            x_pert[0::3] = np.clip(x_pert[0::3], 0.015, 0.985)
            x_pert[1::3] = np.clip(x_pert[1::3], 0.015, 0.985)
            x_pert[2::3] = np.clip(x_pert[2::3], 1e-5, 0.5)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
                if res.success:
                    c_val = compute_constraints(res.x)
                    if np.min(c_val) >= -1e-8:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = res.x.copy()
            except Exception:
                continue
                
        # High-precision polish
        try:
            res_final = minimize(objective, best_x, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if res_final.success and np.min(compute_constraints(res_final.x)) >= -1e-9:
                best_x = res_final.x
                best_sum = -res_final.fun
        except Exception:
            pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = generate_hex_init(0, 0.10)
        best_sum = -objective(best_x)

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    return centers, radii, float(best_sum)
