from scipy.optimize import minimize

N = 26
i_idx, j_idx = np.triu_indices(N, k=1)

def compute_violations(vars_vec):
    """Computes constraint violations (values > 0 indicate violation)."""
    x = vars_vec[0::3]
    y = vars_vec[1::3]
    r = vars_vec[2::3]
    
    # Boundary violations
    v1 = np.maximum(0, r - x)
    v2 = np.maximum(0, r - (1.0 - x))
    v3 = np.maximum(0, r - y)
    v4 = np.maximum(0, r - (1.0 - y))
    
    # Pairwise overlap violations
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    v5 = np.maximum(0, r_sum[i_idx, j_idx] - dist[i_idx, j_idx])
    
    return np.concatenate([v1, v2, v3, v4, v5])

def penalty_obj(vars_vec, mu):
    """Objective with quadratic penalty for constraint violations."""
    v = compute_violations(vars_vec)
    return -np.sum(vars_vec[2::3]) + mu * np.sum(v**2)

def obj_func(vars_vec):
