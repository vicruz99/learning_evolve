import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraint_func(v):
    """
    Computes inequality constraints g(v) >= 0.
    Includes boundary containment and pairwise non-overlap.
    """
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def compute_safe_radii(pts):
    """Computes strictly feasible initial radii for given centers."""
    r = np.zeros(N)
    for i in range(N):
        d_wall = min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])
        d_min = 1.0
        for j in range(N):
            if i != j:
                d = np.sqrt(np.sum((pts[i] - pts[j])**2))
                if d < d_min:
                    d_min = d
        # Conservative scaling to guarantee strict feasibility for SLSQP start
        r[i] = 0.42 * min(d_wall, d_min / 2.0)
    return np.maximum(r, 1e-4)

def generate_hex_init(seed, scale=1.0, rot=0.0):
    """Generates a hexagonal lattice initialization with optional scaling/rotation."""
    np.random.seed(seed)
    pts = []
    base_r = 0.085 * scale
    y = base_r
    row = 0
    while len(pts) < N:
        x_off = (row % 2) * base_r
        x = base_r + x_off
        while x <= 1.0 - base_r and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * base_r
        y += np.sqrt(3.0) * base_r
        row += 1
        
    pts = np.array(pts[:N])
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ rot_mat.T + 0.5
        
    pts += np.random.uniform(-0.025, 0.025, (N, 2))
    return np.clip(pts, 0.03, 0.97)

def generate_boundary_init(seed):
    """Generates configurations biased towards boundaries and corners."""
    np.random.seed(seed)
    pts = np.random.rand(N, 2)
    
    # Simple repulsion to spread, then push to nearest boundary
    for _ in range(150):
        diff = pts[:, None, :] - pts[None, :, :]
        d2 = np.sum(diff**2, axis=2)
        d2 = np.maximum(d2, 1e-6)
        d = np.sqrt(d2)
        forces = np.sum(diff / (d2[:, :, None] * d[:, :, None]), axis=0)
        
        # Boundary attraction
        for k in range(2):
            forces[:, k] += 8.0 * (pts[:, k] - 0.5)
            forces[:, k] += 15.0 * (pts[:, k] < 0.15).astype(float) * (0.15 - pts[:, k])
            forces[:, k] -= 15.0 * (pts[:, k] > 0.85).astype(float) * (pts[:, k] - 0.85)
            
        pts += 0.003 * forces
        pts = np.clip(pts, 0.02, 0.98)
        
    return pts

def run_packing():
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_x = None
    best_val = -np.inf
    
    initializations = []
    
    # Hexagonal variations
    for s in range(25):
        sc = 0.85 + 0.3 * np.random.rand()
        rt = np.random.uniform(-0.25, 0.25)
        initializations.append(generate_hex_init(s, sc, rt))
        
    # Boundary-attracted variations
    for s in range(20):
        initializations.append(generate_boundary_init(s))
        
    # Structured Grid + Center
    grid_pts = np.array([[0.1 + i*0.2, 0.1 + j*0.2] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
    for s in range(5):
        np.random.seed(s + 50)
        initializations.append(np.clip(grid_pts + np.random.uniform(-0.02, 0.02, (N, 2)), 0.03, 0.97))

    # Phase 1: Broad optimization from diverse starts
    for pts in initializations:
        r_init = compute_safe_radii(pts)
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r_init
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            
            if res.success:
                c_val = constraint_func(res.x)
                if np.min(c_val) >= -1e-7:
                    curr_val = -res.fun
                    if curr_val > best_val:
                        best_val = curr_val
                        best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Basin-hopping style local refinement
    if best_x is not None:

            x_p[1::3] = np.clip(x_p[1::3], 0.01, 0.99)

                            best_val = curr_val
                            best_x = res.x.copy()
            except Exception:
                continue
                
    # Phase 3: High-precision polynomial-time polish
    if best_x is not None:
        try:
            res_final = minimize(objective, best_x, method='SLSQP', bounds=bounds,

            if np.min(constraint_func(res_final.x)) >= -1e-7:
                best_x = res_final.x
                best_val = -res_final.fun
        except Exception:
            pass
            
    # Fallback (should rarely be reached)
    if best_x is None:
        centers_f = np.zeros((N, 2))
        idx = 0
        for i in range(5):

                centers_f[idx] = [0.1 + 0.2*i, 0.1 + 0.2*j]
                idx += 1

        best_x[1::3] = centers_f[:, 1]
        best_x[2::3] = r_f
        best_val = np.sum(r_f)

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    return centers, radii, float(best_val)
