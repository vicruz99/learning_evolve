import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[2::3])

def constraint_func(vars_vec):
    """
    Computes inequality constraints g(vars_vec) >= 0.
    Includes boundary containment and pairwise non-overlap.
    """
    xc = vars_vec[0::3]
    yc = vars_vec[1::3]
    r = vars_vec[2::3]
    
    c = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(xc - r)
    c.append(1.0 - xc - r)
    c.append(yc - r)
    c.append(1.0 - yc - r)
    
    # Pairwise separation constraints: dist^2 >= (r_i + r_j)^2
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def get_initial_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    r = np.zeros(N)
    for i in range(N):
        db = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        dm = 2.0
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < dm: dm = d
        r[i] = 0.45 * min(db, dm/2.0)
    return r

def make_hex_init(seed, spacing):
    """Generates a hexagonal lattice initialization with controlled jitter."""
    np.random.seed(seed)
    pts = []
    y = spacing
    row = 0
    while len(pts) < N:
        x_off = 0.0 if row % 2 == 0 else spacing
        x = spacing + x_off
        while x <= 1.0 - spacing and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * spacing
        y += np.sqrt(3.0) * spacing
        row += 1
    pts = np.array(pts[:N])
    pts += np.random.uniform(-0.015, 0.015, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_grid_init(seed):
    """Generates a 5x5 grid + center initialization with jitter."""
    np.random.seed(seed)
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += np.random.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_force_init(seed):
    """Generates a force-directed layout to spread points apart."""
    np.random.seed(seed)
    pts = np.random.rand(N, 2) * 0.8 + 0.1
    for _ in range(150):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        d2 = np.sum(diff**2, axis=2)
        d2 = np.maximum(d2, 1e-6)
        d = np.sqrt(d2)
        mag = 1.0 / (d2 * d)
        mag = np.clip(mag, 0, 15.0)
        f += np.sum(mag[:, :, None] * diff / d[:, :, None], axis=1)
        pts += 0.006 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_x = None
    best_val = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    for s in range(20):
        inits.append(make_hex_init(s, spacing=0.07 + 0.025*(s%3)))
        inits.append(make_grid_init(s))
        inits.append(make_force_init(s))
        
    # Specific strategic placements for the 26th circle relative to a 5x5 grid
    base_grid = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    extra_pos = [[0.05,0.05], [0.95,0.05], [0.05,0.95], [0.95,0.95], 
                 [0.5,0.5], [0.1,0.5], [0.9,0.5], [0.5,0.1], [0.5,0.9],
                 [0.2,0.2], [0.8,0.2], [0.2,0.8], [0.8,0.8], [0.3,0.3], [0.7,0.7]]
    for pos in extra_pos:
        pts = np.vstack([base_grid, pos])
        inits.append(pts)

    # Phase 1: Broad search from diverse initializations
    for pts in inits:
        r_init = get_initial_radii(pts)
        x0 = np.empty(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r_init
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                c_val = constraint_func(res.x)
                if np.min(c_val) >= -1e-7:
                    val = -res.fun
                    if val > best_val:
                        best_val = val
                        best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Aggressive local perturbation search to escape local minima
    if best_x is not None:
        for k in range(80):
            x0 = best_x.copy()

                x0[3*i+2] += np.random.uniform(-0.01, 0.01)
                
            x0[0::3] = np.clip(x0[0::3], 0.01, 0.99)
            x0[1::3] = np.clip(x0[1::3], 0.01, 0.99)
            x0[2::3] = np.clip(x0[2::3], 1e-5, 0.49)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                if res.success:
                    c_val = constraint_func(res.x)
                    if np.min(c_val) >= -1e-7:
                        val = -res.fun
                        if val > best_val:
                            best_val = val
                            best_x = res.x.copy()
            except Exception:
                continue
                
        # Phase 3: High-precision polynomial-time polish
        try:
            res = minimize(objective, best_x, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if np.min(constraint_func(res.x)) >= -1e-7:
                best_x = res.x
                best_val = -res.fun
        except Exception:
            pass

    # Fallback configuration (should not be reached given robust initialization)
    if best_x is None:
        pts = np.vstack([base_grid, [0.5, 0.5]])
        r = np.full(N, 0.09)
        best_x = np.empty(3*N)
        best_x[0::3] = pts[:,0]
        best_x[1::3] = pts[:,1]
        best_x[2::3] = r
        best_val = np.sum(r)
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    return centers, radii, float(best_val)
