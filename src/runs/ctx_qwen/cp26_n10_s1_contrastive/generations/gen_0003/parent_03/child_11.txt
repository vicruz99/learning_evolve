import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def constraint_func(vars_vec):
    """
    Computes inequality constraints g(vars_vec) >= 0.
    Uses parameterization to automatically satisfy boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    # Parameterization maps u, v in [0,1] to x, y such that circle is strictly inside [0,1]^2
    width = 1.0 - 2.0 * r
    x = r + u * width
    y = r + v * width
    
    # Pairwise squared distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    # Squared sum of radii
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX]

def generate_hex_init(seed, rot_angle=0.0):
    """Generates a strictly feasible initial configuration based on a hexagonal lattice."""
    rng = np.random.RandomState(seed)
    rows = [6, 5, 6, 5, 4]
    pts = []
    r_est = 0.095
    y = r_est
    for i, cnt in enumerate(rows):
        shift = (i % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    
    # Center and scale to unit square
    pts = pts - pts.mean(axis=0)
    scale = 0.88 / np.max(np.abs(pts))
    pts = pts * scale + 0.5
    
    # Apply rotation to break symmetry
    if rot_angle != 0.0:
        c, s = np.cos(rot_angle), np.sin(rot_angle)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ rot_mat.T + 0.5
        
    # Add controlled perturbation
    pts += rng.uniform(-0.03, 0.03, (N, 2))
    pts = np.clip(pts, 0.05, 0.95)
    
    # Compute strictly feasible initial radii and parameterized positions
    r0 = np.full(N, 0.06)
    w = 1.0 - 2.0 * r0
    u0 = np.clip((pts[:, 0] - r0) / w, 0.0, 1.0)
    v0 = np.clip((pts[:, 1] - r0) / w, 0.0, 1.0)
    
    vars0 = np.empty(3 * N)
    vars0[0::3] = r0
    vars0[1::3] = u0
    vars0[2::3] = v0
    return vars0

def generate_force_init(seed):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    
    # Force-directed relaxation to spread points
    for _ in range(200):
        forces = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-8)
        f = diff / (dists[:, :, np.newaxis]**2)
        forces += np.sum(f, axis=0)
        
        # Wall repulsion
        margin = 0.1
        for d in range(2):
            below = pts[:, d] < margin
            above = pts[:, d] > 1.0 - margin
            forces[below, d] += 20.0 * (margin - pts[below, d])
            forces[above, d] -= 20.0 * (pts[above, d] - (1.0 - margin))
            
        pts += 0.004 * forces
        pts = np.clip(pts, 0.02, 0.98)
        
    r0 = np.full(N, 0.05)
    w = 1.0 - 2.0 * r0
    u0 = np.clip((pts[:, 0] - r0) / w, 0.0, 1.0)
    v0 = np.clip((pts[:, 1] - r0) / w, 0.0, 1.0)
    
    vars0 = np.empty(3 * N)
    vars0[0::3] = r0
    vars0[1::3] = u0
    vars0[2::3] = v0
    return vars0

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_vars = None
    
    # Phase 1: Broad search from diverse hexagonal initializations
    for seed in range(60):
        rot = np.random.uniform(-0.4, 0.4)
        try:
            x0 = generate_hex_init(seed, rot)
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})

            if np.min(cons_val) >= -1e-7:
                curr_sum = -res.fun

                    best_sum = curr_sum

        except Exception:
            continue

    v_opt = best_vars[2::3]
    width = 1.0 - 2.0 * r_opt
    x_opt = r_opt + u_opt * width
    y_opt = r_opt + v_opt * width
    centers = np.column_stack((x_opt, y_opt))
    
    return centers, r_opt, float(best_sum)
