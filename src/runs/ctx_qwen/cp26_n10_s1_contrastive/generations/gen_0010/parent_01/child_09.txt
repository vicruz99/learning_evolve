import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        A_ub[k:k+4, i] = 1.0
        b_ub[k] = x
        b_ub[k+1] = 1.0 - x
        b_ub[k+2] = y
        b_ub[k+3] = 1.0 - y
        k += 4
        
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constr(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def hex_init(rows, rot, scale, rng):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in rows:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        mag = 1.0 / (d**2 + 0.01)
        f += np.sum(mag[:, :, None] * diff / d[:, :, None], axis=1)
        pts += 0.005 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -np.inf
    best_p = None
    best_c = None
    best_r = None
    
    bounds_slqp = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons_slqp = {'type': 'ineq', 'fun': constr}
    
    # Generate diverse initial configurations
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5],
        [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7], [6,6,5,5,4], [5,5,6,5,5],
        [5,5,5,5,5,1], [4,6,6,5,5], [5,6,6,4,5]
    ]
    for p in patterns:
        for _ in range(4):
            rot = rng.uniform(-0.35, 0.35)
            sc = rng.uniform(0.88, 1.12)
            inits.append(hex_init(p, rot, sc, rng))
    for _ in range(15):
        inits.append(force_init(rng))
        
    # Phase 1: Multi-start SLSQP optimization
    for pts in inits:
        r0, _ = solve_lp_radii(pts)
        r0 = np.clip(r0 * 0.99, 1e-6, 0.49)
        p0 = to_params(pts, r0)
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constr(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            pass
            
    # Fallback if optimization failed completely
    if best_p is None:
        pts = inits[0]
        r0, _ = solve_lp_radii(pts)
        best_p = to_params(pts, np.clip(r0*0.99, 1e-6, 0.49))
        best_sum = -obj(best_p)
        
    best_c, best_r = from_params(best_p)
    curr_c = best_c.copy()
    curr_s = best_sum
    temp = 0.06
    step = 0.04
    
    # Phase 2: LP-driven Simulated Annealing on centers
    for it in range(12000):
        idx = rng.integers(N)
        old_c = curr_c[idx].copy()
        
        # Occasional large jump to escape deep local minima
        if rng.random() < 0.02:
            curr_c[idx] = rng.uniform(0.02, 0.98, 2)
        else:
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-10 and rng.random() < np.exp(delta/temp)):
            curr_s = ns
            # Keep the move
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
        else:
            curr_c[idx] = old_c
            
        temp *= 0.9991
        step = max(0.001, step * 0.9993)
        
    if best_c is not None:
        best_p = to_params(best_c, np.clip(best_r*0.995, 1e-6, 0.49))
        
    # Phase 3: Perturbation & SLSQP restarts to escape local minima
    for k in range(50):
        xp = to_params(best_c, np.clip(best_r*0.99, 1e-6, 0.49))
        xp[:N] += rng.normal(0, 0.0025, N)
        xp[N:3*N] += rng.uniform(-0.025, 0.025, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constr(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
                    best_c, best_r = from_params(best_p)
                    best_r, best_sum = solve_lp_radii(best_c)
        except Exception:
            pass
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(constr(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_c, best_r = from_params(best_p)
            best_r, best_sum = solve_lp_radii(best_c)
    except Exception:
        pass
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
