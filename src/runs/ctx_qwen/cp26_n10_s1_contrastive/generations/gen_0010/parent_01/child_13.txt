import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    n_con = 4*n + n*(n-1)//2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_val = (x, 1.0-x, y, 1.0-y)
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i+1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.ones(n)*1e-6, 0.0

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0*r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u*(1.0 - 2.0*r)
    y = r + v*(1.0 - 2.0*r)
    return np.column_stack([x, y]), r

def obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constr(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u*(1.0 - 2.0*r)
    y = r + v*(1.0 - 2.0*r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(seed, rows, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in rows:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        mag = 1.0 / (d**2 + 0.01)
        f += np.sum(mag[:, :, None] * diff / d[:, :, None], axis=1)
        pts += 0.005 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def corner_edge_init(seed):
    """Generate packings that specifically target corners and edges."""
    rng = np.random.RandomState(seed)
    pts = []
    corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]

        side = rng.randint(4)
        if side == 0: pts.append([rng.uniform(0.2, 0.8), 0.05 + rng.uniform(0, 0.05)])
        elif side == 1: pts.append([rng.uniform(0.2, 0.8), 0.95 - rng.uniform(0, 0.05)])
        elif side == 2: pts.append([0.05 + rng.uniform(0, 0.05), rng.uniform(0.2, 0.8)])
        else: pts.append([0.95 - rng.uniform(0, 0.05), rng.uniform(0.2, 0.8)])
        
    for _ in range(N - len(pts)):
        pts.append(rng.uniform(0.3, 0.7, 2))
        
    pts = np.array(pts[:N])
    for _ in range(100):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        mag = 1.0 / (d**2 + 0.01)
        f += np.sum(mag[:, :, None] * diff / d[:, :, None], axis=1)
        for i in range(N):
            for b in [0.0, 1.0]:
                for dim in range(2):
                    if pts[i, dim] < 0.15: f[i, dim] += 5.0 * (0.15 - pts[i, dim])
                    if pts[i, dim] > 0.85: f[i, dim] -= 5.0 * (pts[i, dim] - 0.85)
        pts += 0.002 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts
