import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4*n + n*(n-1)//2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k+=1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0-x; k+=1
        A_ub[k, i] = 1.0; b_ub[k] = y; k+=1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0-y; k+=1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i+1, n):
            A_ub[k, i] = 1.0; A_ub[k, j] = 1.0; b_ub[k] = dists[i, j]; k+=1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.ones(n)*1e-6, 1e-6

def obj_centers(x_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    return -solve_lp_radii(x_flat.reshape(N, 2))[1]

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0*r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u*(1.0 - 2.0*r)
    y = r + v*(1.0 - 2.0*r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constr_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u*(1.0 - 2.0*r)
    y = r + v*(1.0 - 2.0*r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def run_packing():
    np.random.seed(42)
    bounds_c = [(0.0, 1.0)] * (2 * N)
    
    # Phase 1: Global search on centers using Differential Evolution
    # DE efficiently explores the non-smooth landscape defined by the LP objective
    best_de_x = None
    best_de_val = np.inf
    
    for seed in [42, 123, 456]:
        try:
            res = differential_evolution(
                obj_centers, bounds_c, popsize=12, maxiter=60,
                tol=1e-7, mutation=(0.5, 1.0), recombination=0.8,
                seed=seed, polish=False
            )
            if res.fun < best_de_val:
                best_de_val = res.fun
                best_de_x = res.x.copy()
        except Exception:
            continue
            
    # Fallback to hex grid if DE fails
    if best_de_x is None:
        pts = []
        r_est = 0.095
        y = r_est
        row = 0
        rows = [6,5,6,5,4]
        for cnt in rows:
            shift = (row % 2) * r_est
            x = r_est + shift
            for _ in range(cnt):
                if len(pts) < N: pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
        best_de_x = np.array(pts[:N]).flatten()
        
    centers = best_de_x.reshape(N, 2)
    
    # Phase 2: Simulated Annealing on centers to escape local minima
    curr_c = centers.copy()
    curr_s = solve_lp_radii(curr_c)[1]
    best_s = curr_s
    best_c = curr_c.copy()
    
    temp = 0.03
    step = 0.025
    
    for it in range(15000):
        move_type = np.random.choice(['move', 'swap'], p=[0.75, 0.25])
        if move_type == 'move':
            idx = np.random.randint(N)
            old = curr_c[idx].copy()
            curr_c[idx] += np.random.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
        else:
            i, j = np.random.choice(N, 2, replace=False)
            curr_c[[i, j]] = curr_c[[j, i]]
            
        _, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and np.random.rand() < np.exp(delta/temp)):
            curr_s = ns
            if ns > best_s:
                best_s = ns
                best_c = curr_c.copy()
        else:
            if move_type == 'move':
                curr_c[idx] = old
            else:
                curr_c[[i, j]] = curr_c[[j, i]]
                
        if (it + 1) % 1500 == 0:
            temp *= 0.85
            step *= 0.92
            
    centers = best_c
    
    # Phase 3: SLSQP refinement with (r, u, v) parameterization
    r_init, _ = solve_lp_radii(centers)
    r_init = np.clip(r_init * 0.995, 1e-6, 0.49)
    p0 = to_params(centers, r_init)
    
    bounds_slqp = [(1e-6, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons_slqp = {'type': 'ineq', 'fun': constr_slqp}
    
    best_p = p0
    best_sum = -obj_slqp(p0)
    
    # Run SLSQP multiple times with small perturbations to escape local minima
    for k in range(8):
        xp = best_p.copy()
        xp[:N] += np.random.normal(0, 0.001, N)
        xp[N:3*N] += np.random.uniform(-0.012, 0.012, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-12})
            if res.success and np.min(constr_slqp(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            pass
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp,
                         constraints=cons_slqp, options={'maxiter': 8000, 'ftol': 1e-14})
        if res_f.success and np.min(constr_slqp(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    centers_opt, radii_opt = from_params(best_p)
    # Final LP ensures radii are strictly optimal for the exact final centers
    radii_final, final_sum = solve_lp_radii(centers_opt)
    
    return centers_opt, np.maximum(radii_final, 0.0), float(final_sum)
