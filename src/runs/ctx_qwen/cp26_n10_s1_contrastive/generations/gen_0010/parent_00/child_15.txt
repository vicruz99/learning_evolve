import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

def solve_lp_fast(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_constraints = 4 * n + NUM_PAIRS
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for idx in range(NUM_PAIRS):
        i, j = I_IDX[idx], J_IDX[idx]
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]
        k += 1
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def sa_objective(centers_flat):
    """Objective for Simulated Annealing: negative sum of optimal radii."""
    centers = centers_flat.reshape(N, 2)
    _, s = solve_lp_fast(centers)
    return -s

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns with varied parameters
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [7,5,5,5,4], [5,5,5,5,6], 
        [4,6,5,6,5], [6,4,6,5,5], [5,6,4,6,5]
    ]
    for p in patterns:
        for _ in range(4):
            pts = []
            r_est = 0.095
            y = r_est
            row = 0
            for cnt in p:
                shift = (row % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.9, 1.1) + 0.5
            rot = rng.uniform(-0.3, 0.3)
            c, s = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pts += rng.uniform(-0.02, 0.02, (N, 2))
            inits.append(np.clip(pts, 0.02, 0.98))
            
    # 2. Force-directed layouts
    for s_seed in range(8):
        rng_f = np.random.RandomState(s_seed)
        pts = rng_f.uniform(0.1, 0.9, (N, 2))
        for _ in range(200):
            f = np.zeros_like(pts)
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            d = np.sqrt(np.sum(diff**2, axis=2))
            d = np.maximum(d, 1e-4)
            rep = 1.0 / d**2
            np.fill_diagonal(rep, 0.0)
            f += np.sum(rep[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
            for dim in range(2):
                f[:, dim] += 20.0 * np.maximum(0, 0.05 - pts[:, dim])
                f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.95)
            pts += 0.002 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    return inits

def obj_slqp(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def cons_slqp(v):
    """Inequality constraints for SLSQP: boundaries and non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Use linear distance constraint for better conditioning
    c.append(dists[I_IDX, J_IDX] - rs[I_IDX, J_IDX])
    
    return np.concatenate(c)

def run_packing():

    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).

    inits = generate_inits(rng)
    
    best_c = None
    best_sum = -np.inf
    
    # Phase 1: LP-Driven Simulated Annealing on centers
    for init_c in inits:
        curr_c = init_c.copy()
        curr_sum = -sa_objective(curr_c.flatten())
        
        temp = 0.05
        step = 0.04
        for it in range(2500):
            op = rng.integers(0, 4)
            if op == 0:  # Translate one circle
                idx = rng.integers(N)
                old = curr_c[idx].copy()
                curr_c[idx] += rng.normal(0, step, 2)
                curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            elif op == 1:  # Swap two circles

                curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
            else:  # Random jump or aggressive move

                    idx = rng.integers(N)
                    curr_c[idx] = rng.uniform(0.1, 0.9, 2)
                else:
                    idx = rng.integers(N)
                    old = curr_c[idx].copy()
                    curr_c[idx] += rng.normal(0, step * 2.5, 2)
                    curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
                    
            new_sum = -sa_objective(curr_c.flatten())
            delta = new_sum - curr_sum
            
            # Metropolis criterion
            if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
                curr_sum = new_sum
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_c = curr_c.copy()
            else:
                if op == 0:
                    curr_c[idx] = old
                    
            temp *= 0.9992
            step = max(0.002, step * 0.9995)

    bounds_slqp = [(0.0, 1.0), (0.0, 1.0), (1e-9, 0.5)] * N

        res = minimize(obj_slqp, v0, method='SLSQP', bounds=bounds_slqp,

            best_c = np.column_stack((res.x[0::3], res.x[1::3]))
            r_best = res.x[2::3]
            best_sum = np.sum(r_best)

        pass
        
    # Phase 3: Perturbation & SLSQP to escape local minima

        vp[1::3] += rng.uniform(-0.015, 0.015, N)

        vp[0::3] = np.clip(vp[0::3], 0.01, 0.99)

                         constraints={'type': 'ineq', 'fun': cons_slqp},

    # Final LP ensures radii are strictly optimal for the exact final centers

    radii_final = np.maximum(radii_final, 0.0)
    final_sum = float(np.sum(radii_final))
    
    return best_c, radii_final, final_sum
