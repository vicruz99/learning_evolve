import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def s_obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def s_cons(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    d2 = dx**2 + dy**2
    
    rs = r[I_IDX] + r[J_IDX]
    return d2 - rs**2

def to_params(centers, radii):
    """Convert physical (centers, radii) to optimization parameters (r, u, v)."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Convert optimization parameters (r, u, v) back to physical (centers, radii)."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def de_objective(centers_flat):
    """Objective for Differential Evolution: minimize negative LP sum of radii."""
    centers = centers_flat.reshape(N, 2)
    _, val = solve_lp_radii(centers)
    return -val

def make_hex(seed, pattern, rot, scale):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force(seed):
    """Generate a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        rep = 1.0 / d**2
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def make_seq(seed):
    """Generate a sequential Voronoi-like packing initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    for _ in range(N):
        samples = rng.uniform(0.05, 0.95, (1000, 2))
        d_bound = np.minimum(np.minimum(samples[:, 0], 1.0 - samples[:, 0]),
                             np.minimum(samples[:, 1], 1.0 - samples[:, 1]))
        if len(pts) == 0:
            d_pts = np.full_like(d_bound, 10.0)
        else:
            pts_arr = np.array(pts)
            d_pts = np.min(np.linalg.norm(samples[:, np.newaxis, :] - pts_arr[np.newaxis, :, :], axis=2), axis=1)
        min_d = np.minimum(d_bound, d_pts)
        pts.append(samples[np.argmax(min_d)])
    return np.array(pts)

def run_packing():

    Returns (centers, radii, sum_radii).

    np.random.seed(42)
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * (2 * N)
    cons_dict = {'type': 'ineq', 'fun': s_cons}

    best_sum = -np.inf

        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],

    ]
    for p in patterns:
        for s in range(6):
            rot = np.random.uniform(-0.35, 0.35)
            sc = np.random.uniform(0.85, 1.15)
            inits.append(make_hex(s*10 + np.random.randint(100), p, rot, sc))
    for s in range(8):
        inits.append(make_force(s))
    for s in range(5):

    for c in inits:

        r = np.clip(r * 0.985, 1e-6, 0.49)
        p0 = to_params(c, r)
        try:
            res = minimize(s_obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(s_cons(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)
        best_p = to_params(c_f, r_f * 0.98)

    radii = np.maximum(radii_lp, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
