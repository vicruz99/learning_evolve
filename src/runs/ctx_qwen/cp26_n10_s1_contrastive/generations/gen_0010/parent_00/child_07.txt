import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, 10.0) # Avoid self-constraint issues
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def to_params(centers, radii):
    """Convert physical (centers, radii) to optimization parameters (r, u, v)."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Convert optimization parameters (r, u, v) back to physical (centers, radii)."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def s_obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def s_cons(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    d2 = dx**2 + dy**2
    
    rs = r[I_IDX] + r[J_IDX]
    return d2 - rs**2

def make_hex(seed, pattern, rot, scale):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force(seed):
    """Generate a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(400):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        rep = 1.0 / d**2
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 30.0 * np.maximum(0, 0.02 - pts[:, dim])
            f[:, dim] -= 30.0 * np.maximum(0, pts[:, dim] - 0.98)
        pts += 0.002 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def nelder_search(centers):
    """Local optimization on centers using Nelder-Mead."""
    def neg_sum(centers_flat):
        return -solve_lp(centers_flat.reshape(N, 2))[1]
    
    res = minimize(neg_sum, centers.flatten(), method='Nelder-Mead',
                   options={'maxiter': 3000, 'xatol': 1e-5, 'fatol': 1e-5})
    return res.x.reshape(N, 2), -res.fun

def sa_optimize(centers, iters=15000):
    """Simulated Annealing on centers to escape local minima."""
    curr_c = centers.copy()
    curr_s = solve_lp(curr_c)[1]
    best_c = curr_c.copy()
    best_s = curr_s
    
    temp = 0.06
    step = 0.045
    rng = np.random.default_rng(42)
    
    for it in range(iters):
        move_type = rng.integers(0, 4)
        new_c = curr_c.copy()
        
        if move_type == 0:
            idx = rng.integers(N)
            new_c[idx] += rng.normal(0, step, 2)
            new_c[idx] = np.clip(new_c[idx], 0.01, 0.99)
        elif move_type == 1:
            i, j = rng.choice(N, 2, replace=False)
            new_c[i], new_c[j] = new_c[j].copy(), new_c[i].copy()
        elif move_type == 2:
            i, j = rng.choice(N, 2, replace=False)
            mid = (new_c[i] + new_c[j]) / 2.0
            new_c[i] = mid + rng.normal(0, step*0.5, 2)
            new_c[j] = mid + rng.normal(0, step*0.5, 2)
            new_c[i] = np.clip(new_c[i], 0.01, 0.99)
            new_c[j] = np.clip(new_c[j], 0.01, 0.99)
        else:
            idx = rng.integers(N)
            new_c[idx] = rng.uniform(0.05, 0.95, 2)
            
        new_s = solve_lp(new_c)[1]
        delta = new_s - curr_s
        
        if delta > 0 or (temp > 1e-7 and rng.random() < np.exp(delta / temp)):
            curr_c = new_c
            curr_s = new_s
            if curr_s > best_s:
                best_c = curr_c.copy()
                best_s = curr_s
                
        temp *= 0.9992
        step = max(0.0015, step * 0.9995)
        
    return best_c, best_s

def slqp_polish(centers, radii):
    """High-precision SLSQP refinement using (r, u, v) parameterization."""
    r_init = np.clip(radii * 0.995, 1e-6, 0.49)
    p0 = to_params(centers, r_init)
    
    bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * (2 * N)
    cons_dict = {'type': 'ineq', 'fun': s_cons}
    
    try:
        res = minimize(s_obj, p0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(s_cons(res.x)) >= -1e-7:
            return res.x, -res.fun
    except Exception:
        pass
    return p0, np.sum(radii)

def run_packing():

    Solves the circle packing problem for N=26 in a unit square.

            p_pert = p_best.copy()
