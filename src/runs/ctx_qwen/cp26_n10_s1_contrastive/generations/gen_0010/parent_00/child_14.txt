1. **Linear Programming (LP) for Radius Optimization**: Given fixed centers, the optimal radii maximizing the sum can be found exactly via LP. This is fast and avoids non-smooth penalty functions.
2. **Boundary-Safe Parameterization**: Use $(r, u, v)$ variables where $x = r + u(1-2r)$ and $y = r + v(1-2r)$. This guarantees $r \le x \le 1-r$ and $r \le y \le 1-r$ for $u,v \in [0,1]$, removing boundary constraints from the nonlinear solver and drastically improving SLSQP stability.
3. **LP-Driven Simulated Annealing (SA)**: Evolve centers directly using SA, evaluating the objective via LP. This efficiently explores the non-convex landscape of center positions. I will add topology-swapping moves (randomly exchanging positions of two circles) to escape deep local minima where circles get "locked" in suboptimal contact graphs.
4. **Progressive Refinement**: After SA finds a promising configuration, run SLSQP on the $(r, u, v)$ parameterization to fine-tune positions and sizes simultaneously. Repeat perturbation + SLSQP cycles to polish the solution.
5. **Diverse Initializations**: Generate starts from rotated/scaled hexagonal lattices, force-directed layouts, and sequential greedy packings to cover multiple topological basins.
6. **High-Precision Final Polish**: A final SLSQP run with tight tolerances followed by an exact LP radius recalculation guarantees strict feasibility and maximum sum for the returned centers.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def s_obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def s_cons(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    d2 = dx**2 + dy**2
    
    rs = r[I_IDX] + r[J_IDX]
    return d2 - rs**2

def to_params(centers, radii):
    """Convert physical (centers, radii) to optimization parameters (r, u, v)."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Convert optimization parameters (r, u, v) back to physical (centers, radii)."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def make_hex(seed, pattern, rot, scale):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force(seed):
    """Generate a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        rep = 1.0 / d**2
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * (2 * N)
    cons_dict = {'type': 'ineq', 'fun': s_cons}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Phase 1: Diverse initializations & SLSQP
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], 
        [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7], [6,6,5,5,4],
        [7,6,5,4,4], [6,7,5,5,3], [5,5,5,5,5,1]
    ]
    for p in patterns:
        for s in range(8):
            rot = np.random.uniform(-0.4, 0.4)
            sc = np.random.uniform(0.85, 1.15)
            inits.append(make_hex(s*100 + np.random.randint(1000), p, rot, sc))
    for s in range(10):
        inits.append(make_force(s))
        
    for c in inits:
        r, _ = solve_lp_radii(c)
        r = np.clip(r * 0.985, 1e-6, 0.49)
        p0 = to_params(c, r)
        try:
            res = minimize(s_obj, p0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(s_cons(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
                    best_c, best_r = from_params(res.x)
        except Exception:
            continue
            
    if best_p is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)
        best_p = to_params(c_f, r_f * 0.98)
        best_c, best_r = from_params(best_p)
        best_sum = np.sum(best_r)
        
    # Phase 2: LP-Driven Simulated Annealing on centers
    curr_c = best_c.copy()
    curr_s = best_sum
    step = 0.04
    temp = 0.04
    
    for it in range(15000):
        # Mix of local moves and topology swaps
        if np.random.rand() < 0.15:
            i, j = np.random.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
            is_swap = True
        else:
            idx = np.random.randint(N)
            old = curr_c[idx].copy()
            # Occasional large jump to escape deep basins
            jump_factor = 2.5 if (np.random.rand() < 0.04) else 1.0
            curr_c[idx] += np.random.normal(0, step * jump_factor, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
            is_swap = False
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        # Metropolis criterion
        if delta > 0 or (temp > 1e-8 and np.random.rand() < np.exp(min(0.0, delta / temp))):
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
                best_p = to_params(best_c, best_r * 0.995)
        else:
            if not is_swap:
                curr_c[idx] = old
                
        temp *= 0.9992
        step = max(0.001, step * 0.9995)
        
    # Phase 3: Refine SA best with SLSQP
    p0 = to_params(best_c, best_r * 0.995)
    try:
        res = minimize(s_obj, p0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(s_cons(res.x)) >= -1e-7:
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_p = res.x.copy()
                best_c, best_r = from_params(res.x)
    except Exception:
        pass
        
    # Phase 4: Perturbation & SLSQP cycles to escape remaining local minima
    for k in range(40):
        xp = best_p.copy()
        xp[:N] += np.random.uniform(-0.003, 0.003, N)
        xp[N:3*N] += np.random.uniform(-0.025, 0.025, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:] = np.clip(xp[N:], 0.0, 1.0)
        
        try:
            res = minimize(s_obj, xp, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(s_cons(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
                    best_c, best_r = from_params(res.x)
        except Exception:
            continue
            
    # Phase 5: High-precision final polish
    try:
        res_f = minimize(s_obj, best_p, method='SLSQP', bounds=bounds, constraints=cons_dict,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(s_cons(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_c, best_r = from_params(res_f.x)
            best_sum = np.sum(best_r)
    except Exception:
        pass
        
    # Final LP to guarantee exact feasibility and optimal radii for the final centers
    radii_lp, final_sum_lp = solve_lp_radii(best_c)
    radii = np.maximum(radii_lp, 0.0)
    
    return best_c, radii, float(final_sum_lp)
