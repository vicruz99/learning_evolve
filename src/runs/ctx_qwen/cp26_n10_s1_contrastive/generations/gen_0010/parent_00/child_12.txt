import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def obj_func(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    d2 = dx**2 + dy**2
    
    rs = r[I_IDX] + r[J_IDX]
    return d2 - rs**2

def to_params(centers, radii):
    """Convert physical (centers, radii) to optimization parameters (r, u, v)."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Convert optimization parameters (r, u, v) back to physical (centers, radii)."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def make_hex(seed, pattern, rot, scale, rng):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def make_force(seed, rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        rep = 1.0 / d**2
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 15.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 15.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.003 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    rng = np.random.default_rng(42)

    cons_dict = {'type': 'ineq', 'fun': cons_func}
    
    best_p = None
    best_sum = -np.inf
    
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], 
        [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7], [6,6,5,5,4],
        [7,6,5,4,4], [6,7,5,5,3], [5,5,5,6,5], [6,5,5,6,4]
    ]
    for p in patterns:
        for s in range(5):
            rot = rng.uniform(-0.3, 0.3)
            sc = rng.uniform(0.88, 1.12)
            inits.append(make_hex(s, p, rot, sc, rng))
    for s in range(10):
        inits.append(make_force(s, rng))
        
    # Phase 1: SLSQP from diverse starts
    for c in inits:

            res = minimize(obj_func, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons_func(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)
        best_p = to_params(c_f, r_f * 0.98)
        best_sum = np.sum(best_p[:N])

    c_best, r_best = from_params(best_p)

    curr_s = best_sum

    temp = 0.05

            is_swap = True
        else:
            idx = rng.integers(N)
            old = curr_c[idx].copy()

                r_best = nr.copy()
                best_p = to_params(c_best, r_best * 0.995)

    # Phase 3: Powell refinement on centers (handles non-smooth LP landscape)

        _, s = solve_lp_radii(c)
        return -s

    try:
        res_powell = minimize(neg_lp_sum, c_flat, method='Powell',

        if -res_powell.fun > best_sum:

            c_best = res_powell.x.reshape(N, 2)

        pass

        xp = best_p.copy()

        xp[N:3*N] += rng.uniform(-0.015, 0.015, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)

        try:

            if res.success and np.min(cons_func(res.x)) >= -1e-7:

                if s_val > best_sum:

                    best_p = res.x.copy()

    try:

                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_func(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_p)
    radii_lp, _ = solve_lp_radii(centers)
    radii = np.maximum(radii_lp, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
