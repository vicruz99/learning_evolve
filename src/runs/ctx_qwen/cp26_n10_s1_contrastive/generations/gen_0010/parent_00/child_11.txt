import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def s_obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def s_cons(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    d2 = dx**2 + dy**2
    
    rs = r[I_IDX] + r[J_IDX]
    return d2 - rs**2

def to_params(centers, radii):
    """Convert physical (centers, radii) to optimization parameters (r, u, v)."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Convert optimization parameters (r, u, v) back to physical (centers, radii)."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def make_hex(seed, pattern, rot, scale):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force(seed):
    """Generate a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        rep = 1.0 / d**2
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.002 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def make_corner_bias(seed):
    """Generate a corner/edge biased initialization."""
    rng = np.random.RandomState(seed)
    pts = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    edge_pts = [[0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9],
                [0.1, 0.3], [0.1, 0.5], [0.1, 0.7], [0.9, 0.3], [0.9, 0.5], [0.9, 0.7]]
    for p in edge_pts:
        if len(pts) < N: pts.append(p)
    while len(pts) < N:
        pts.append(rng.uniform(0.2, 0.8, 2))
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def obj_1d_wrapper(pos, centers, idx):
    """Wrapper for coordinate-wise optimization to avoid closures."""
    temp_c = centers.copy()
    temp_c[idx] = pos
    _, s = solve_lp_radii(temp_c)
    return -s

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * (2 * N)
    cons_slqp = {'type': 'ineq', 'fun': s_cons}
    
    inits = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], 
                [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7], [6,6,5,5,4]]
    for p in patterns:
        for s in range(3):
            inits.append(make_hex(s*10, p, float(rng.uniform(-0.25, 0.25)), float(rng.uniform(0.9, 1.1))))
    for s in range(4):
        inits.append(make_force(s))
    for s in range(4):
        inits.append(make_corner_bias(s))
        
    best_p = None
    best_sum = -np.inf
    
    # Phase 1: SLSQP from diverse starts
    for c in inits:
        r, _ = solve_lp_radii(c)
        r = np.clip(r * 0.985, 1e-6, 0.49)
        p0 = to_params(c, r)
        try:
            res = minimize(s_obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                           options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(s_cons(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)
        best_p = to_params(c_f, r_f * 0.98)
        best_sum = np.sum(best_p[:N])
        
    # Phase 2: SA on centers
    c_best, r_best = from_params(best_p)
    curr_c = c_best.copy()
    curr_s = best_sum
    step = 0.03
    temp = 0.04
    
    for it in range(6000):
        if rng.random() < 0.2:
            i, j = rng.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
            is_swap = True
        else:
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
            is_swap = False
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                c_best = curr_c.copy()
                r_best = nr.copy()
                best_p = to_params(c_best, r_best * 0.995)
        else:
            if not is_swap:
                curr_c[idx] = old
                
        temp *= 0.9993
        step = max(0.001, step * 0.9994)
        
    # Phase 3: Coordinate-wise center optimization
    for cycle in range(3):
        for i in range(N):
            old_pos = curr_c[i].copy()
            try:
                res_1d = minimize(obj_1d_wrapper, curr_c[i], args=(curr_c, i), method='Nelder-Mead',
                                  options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-8})
                if res_1d.success:
                    curr_c[i] = res_1d.x
            except Exception:
                curr_c[i] = old_pos
                
        nr, ns = solve_lp_radii(curr_c)
        if ns > best_sum:
            best_sum = ns
            c_best = curr_c.copy()
            r_best = nr.copy()
            best_p = to_params(c_best, r_best * 0.995)
            
    # Phase 4: SLSQP refinement on best SA/Coord result
    p0 = to_params(c_best, r_best * 0.995)
    try:
        res = minimize(s_obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                       options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(s_cons(res.x)) >= -1e-7:
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_p = res.x.copy()
    except Exception:
        pass
        
    # Phase 5: Perturbation & SLSQP
    for k in range(15):
        xp = best_p.copy()
        xp[:N] += rng.uniform(-0.002, 0.002, N)
        xp[N:3*N] += rng.uniform(-0.02, 0.02, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:] = np.clip(xp[N:], 0.0, 1.0)
        try:
            res = minimize(s_obj, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                           options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(s_cons(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 6: High-precision polish
    try:
        res_f = minimize(s_obj, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                         options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(s_cons(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_p)
    # Final LP guarantees exact feasibility and optimal radii for the exact final centers
    radii_lp, _ = solve_lp_radii(centers)
    radii = np.maximum(radii_lp, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
