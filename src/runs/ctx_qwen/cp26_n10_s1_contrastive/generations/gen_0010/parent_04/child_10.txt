

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we need to find a configuration where circles form a rigid contact graph touching each other and the boundaries optimally. The objective function over center coordinates is non-smooth because it depends on the active contact graph. 

Strategy:
1. **LP-based Evaluation**: For any fixed set of centers, the optimal radii can be found exactly and efficiently using Linear Programming (`scipy.optimize.linprog`). This gives a precise objective value for any center configuration.
2. **Simulated Annealing on Centers**: We run a temperature-cooled local search directly on the 2D centers. At each step, we move a random circle, evaluate the new sum of radii via LP, and accept/reject based on the Metropolis criterion. This efficiently explores different contact topologies and escapes shallow local minima.
3. **Diverse Initializations**: We seed the search with multiple hexagonal lattice patterns (varied rows, rotations, scales) and force-directed layouts to ensure coverage of different packing symmetries.
4. **SLSQP Refinement**: Once a strong center configuration is found, we switch to smooth constrained optimization using SLSQP. We parameterize positions as `x = r + u*(1-2r)`, `y = r + v*(1-2r)` with `u,v in [0,1]`, which automatically enforces boundary constraints and keeps the feasible region well-conditioned.
5. **Perturbation & Restart**: We repeatedly perturb the optimized parameters and re-run SLSQP to break out of local basins in the smooth landscape.
6. **High-Precision Polish**: A final long-run SLSQP with tight tolerances followed by an exact LP solve guarantees the returned radii are strictly optimal for the final centers and satisfy all constraints.

This hybrid topology-search + smooth-refinement pipeline consistently pushes past standard local optima in circle packing problems.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and not np.any(np.isnan(res.x)):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng, steps=300):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(steps):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.003 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    
    # Generate diverse initial configurations
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], 
            [5,5,5,5,6], [6,6,5,5,4], [5,5,6,5,5], [5,5,5,5,5,1]]
    for pat in pats:
        for _ in range(3):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.85, 1.15)))
    for _ in range(10):
        inits.append(force_init(rng, steps=250))
        
    best_c = inits[0]
    
    # Phase 1: Simulated Annealing on centers with LP evaluation
    # Explores contact graph topologies efficiently
    for c_init in inits:
        curr_c = c_init.copy()
        _, curr_s = solve_lp_radii(curr_c)
        step = 0.025
        temp = 0.02
        
        for it in range(2000):
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
            
            _, new_s = solve_lp_radii(curr_c)
            delta = new_s - curr_s
            
            if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
                curr_s = new_s
                step = min(0.04, step * 1.02)
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_c = curr_c.copy()
            else:
                curr_c[idx] = old
                step = max(0.0005, step * 0.98)
                
            temp *= 0.9995

    # Phase 2: SLSQP refinement from best center configuration
    if best_c is not None:
        r_best, _ = solve_lp_radii(best_c)
        p0 = get_params(best_c, np.clip(r_best * 0.99, 1e-6, 0.49))
        
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                best_p = res.x
                best_sum = -res.fun
                best_c = np.column_stack((best_p[:N] + best_p[N:2*N]*(1-2*best_p[:N]), 
                                          best_p[:N] + best_p[2*N:3*N]*(1-2*best_p[:N])))
        except Exception:
            pass
            
        # Phase 3: Perturbation & SLSQP restarts to escape local minima
        for k in range(25):
            xp = best_p.copy()
            xp[:N] += rng.normal(0, 0.0015, N)
            xp[N:3*N] += rng.normal(0, 0.012, 2*N)
            xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
            xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(obj, xp, method='SLSQP', bounds=bounds, constraints=cons_dict,
                               options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(cons(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_p = res.x.copy()
                        best_c = np.column_stack((best_p[:N] + best_p[N:2*N]*(1-2*best_p[:N]), 
                                                  best_p[:N] + best_p[2*N:3*N]*(1-2*best_p[:N])))
            except Exception:
                continue
                
        # Phase 4: High-precision final polish
        try:
            res_f = minimize(obj, best_p, method='SLSQP', bounds=bounds, constraints=cons_dict,
                             options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(cons(res_f.x)) >= -1e-8:
                best_p = res_f.x
                best_sum = -res_f.fun
                best_c = np.column_stack((best_p[:N] + best_p[N:2*N]*(1-2*best_p[:N]), 
                                          best_p[:N] + best_p[2*N:3*N]*(1-2*best_p[:N])))
        except Exception:
            pass
            
    # Fallback (should rarely be reached)
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum = solve_lp_radii(best_c)
    else:
        best_r, best_sum = solve_lp_radii(best_c)
        
    return best_c, best_r, float(best_sum)
