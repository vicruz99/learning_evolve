import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj_func(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons_func(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_hex(seed, row_counts, rot, scale):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force(seed):
    """Generate a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(300):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.002 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def lp_sa(centers, rng, steps):
    """LP-driven Simulated Annealing on centers."""
    curr_c = centers.copy()
    _, curr_s = solve_lp_radii(curr_c)
    best_c = curr_c.copy()
    best_s = curr_s
    
    step = 0.03
    temp = 0.04
    
    for it in range(steps):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        if rng.random() < 0.05:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
            
        _, s_new = solve_lp_radii(curr_c)
        delta = s_new - curr_s
        
        if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(delta / max(temp, 1e-6))):
            curr_s = s_new
            if s_new > best_s:
                best_s = s_new
                best_c = curr_c.copy()
            step = min(0.05, step * 1.001)
        else:
            curr_c[idx] = old
            step = max(0.0005, step * 0.998)
            
        temp = max(0.0001, temp * 0.9992)
        
    return best_c, best_s

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_func}
    
    best_p = None
    best_sum = -np.inf
    
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], [6,6,5,5,4], [5,5,6,5,5], [5,5,5,5,5,1]]
    for pat in pats:
        for s in range(5):
            rot = rng.uniform(-0.3, 0.3)
            sc = rng.uniform(0.9, 1.1)
            inits.append(generate_hex(s, pat, rot, sc))
    for s in range(15):
        inits.append(generate_force(s))
        
    sa_results = []
    for c in inits:
        c_sa, s_sa = lp_sa(c, rng, steps=2000)
        sa_results.append((c_sa, s_sa))
        
    for i in range(len(sa_results) - 1):
        for j in range(len(sa_results) - 1 - i):
            if sa_results[j][1] < sa_results[j+1][1]:

    for c, s in sa_results[:20]:
        r, _ = solve_lp_radii(c)
        r = np.clip(r * 0.995, 1e-6, 0.49)
        p0 = get_params(c, r)
        try:
            res = minimize(obj_func, p0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_func(res.x)) >= -1e-7:

        xp[:N] += rng.normal(0, 0.005, N)

        if res_f.success and np.min(cons_func(res_f.x)) >= -1e-8:

    r_opt = best_p[:N]

    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    radii_final, sum_final = solve_lp_radii(centers)
    return centers, radii_final, float(sum_final)
