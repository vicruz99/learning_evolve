import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
PAIRS = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIRS)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    n_con = 4 * n + N_PAIRS
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        bounds_vals = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_vals:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            duals = np.zeros(n_con)
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                duals[:] = res.ineqlin.marginals
            return res.x, -res.fun, duals
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-3, np.zeros(n_con)

def compute_dual_forces(centers, duals):
    """Computes ascent forces on centers based on LP dual variables."""
    n = centers.shape[0]
    forces = np.zeros((n, 2))
    
    # Pairwise constraint duals
    offset = 4 * n
    for k in range(N_PAIRS):
        lam = duals[offset + k]
        if lam > 1e-8:
            i, j = PAIRS[k]
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff)
            if dist > 1e-9:
                dir_ij = diff / dist
                forces[i] += lam * dir_ij
                forces[j] -= lam * dir_ij
                
    # Boundary constraint duals
    # Order per circle: x, 1-x, y, 1-y
    # Gradients w.r.t (x, y): (+lam0, -lam1), (+lam2, -lam3)
    for i in range(n):
        forces[i, 0] += duals[4*i] - duals[4*i + 1]
        forces[i, 1] += duals[4*i + 2] - duals[4*i + 3]
        
    return forces

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.002 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    
    # Generate diverse initial configurations
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], [6,6,5,5,4]]
    for pat in pats:
        for _ in range(3):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.9, 1.1)))
    for _ in range(5):
        inits.append(force_init(rng))
    for _ in range(5):
        pts = rng.uniform(0.1, 0.9, (N, 2))
        inits.append(pts)
        
    best_c = None
    best_sum = -np.inf
    
    # Phase 1: Dual-Guided Simulated Annealing on centers
    for init_c in inits:
        curr_c = init_c.copy()
        curr_r, curr_s, _ = solve_lp(curr_c)
        
        temp = 0.04
        step = 0.035
        best_local_c = curr_c.copy()
        best_local_s = curr_s
        
        for it in range(15000):
            r_curr, s_curr, duals = solve_lp(curr_c)
            forces = compute_dual_forces(curr_c, duals)
            
            # Proposal: mix dual-guided direction, random noise, and occasional swap
            new_c = curr_c.copy()
            move_type = rng.integers(0, 10)
            
            if move_type < 7: # Dual-guided + noise
                idx = rng.integers(N)
                noise = rng.normal(0, 0.5, 2)
                new_c[idx] = curr_c[idx] + step * (noise * 0.4 + forces[idx] * 0.6)
            elif move_type < 9: # Pure random move
                idx = rng.integers(N)
                new_c[idx] = curr_c[idx] + rng.normal(0, step, 2)
            else: # Swap two random circles
                i, j = rng.choice(N, 2, replace=False)
                new_c[i], new_c[j] = new_c[j].copy(), new_c[i].copy()
                
            new_c = np.clip(new_c, 0.005, 0.995)
            
            _, s_new, _ = solve_lp(new_c)
            delta = s_new - curr_s
            
            # Metropolis acceptance
            if delta > 0 or rng.random() < np.exp(delta / max(temp, 1e-6)):
                curr_c = new_c
                curr_s = s_new
                if curr_s > best_local_s:
                    best_local_s = curr_s
                    best_local_c = curr_c.copy()
                    
            temp *= 0.9992
            step = max(0.0008, step * 0.9995)
            
        if best_local_s > best_sum:
            best_sum = best_local_s
            best_c = best_local_c.copy()
            
    # Phase 2: SLSQP refinement from best SA result
    if best_c is not None:
        r_opt, _, _ = solve_lp(best_c)
        r_opt = np.clip(r_opt * 0.99, 1e-6, 0.49)
        p0 = get_params(best_c, r_opt)
        
        bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
        cons_dict = {'type': 'ineq', 'fun': cons}
        
        for _ in range(10):
            try:
                res = minimize(obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                               options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(cons(res.x)) >= -1e-7:
                    if -res.fun > best_sum:
                        best_sum = -res.fun
                        r_best = res.x[:N]
                        u_best = res.x[N:2*N]
                        v_best = res.x[2*N:3*N]
                        best_c = np.column_stack((r_best + u_best * (1 - 2 * r_best), 
                                                  r_best + v_best * (1 - 2 * r_best)))
                        p0 = res.x.copy()
            except Exception:
                pass
                
        # Phase 3: Rotation search
        for angle in np.linspace(-0.4, 0.4, 17):
            c_rot = (best_c - 0.5) @ np.array([[np.cos(angle), -np.sin(angle)], 
                                                [np.sin(angle), np.cos(angle)]]) + 0.5
            c_rot = np.clip(c_rot, 0.01, 0.99)
            
            _, s_rot, _ = solve_lp(c_rot)
            if s_rot > best_sum:
                best_sum = s_rot
                best_c = c_rot.copy()
                
                # Quick polish rotated config
                r_tmp = np.clip(solve_lp(c_rot)[0] * 0.99, 1e-6, 0.49)
                p_rot = get_params(c_rot, r_tmp)
                try:
                    res_r = minimize(obj, p_rot, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                                     options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                    if res_r.success and np.min(cons(res_r.x)) >= -1e-7:
                        if -res_r.fun > best_sum:
                            best_sum = -res_r.fun
                            r_best = res_r.x[:N]
                            u_best = res_r.x[N:2*N]
                            v_best = res_r.x[2*N:3*N]
                            best_c = np.column_stack((r_best + u_best * (1 - 2 * r_best), 
                                                      r_best + v_best * (1 - 2 * r_best)))
                except Exception:
                    pass
                    
    # Final high-precision polish
    r_final, _, _ = solve_lp(best_c)
    r_final = np.clip(r_final * 0.999, 1e-6, 0.49)
    p_final = get_params(best_c, r_final)
    
    try:
        res_f = minimize(obj, p_final, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons(res_f.x)) >= -1e-8:
            r_opt = res_f.x[:N]
            u_opt = res_f.x[N:2*N]
            v_opt = res_f.x[2*N:3*N]
            centers = np.column_stack((r_opt + u_opt * (1 - 2 * r_opt), 
                                       r_opt + v_opt * (1 - 2 * r_opt)))
            radii = r_opt
            best_sum = np.sum(radii)
        else:
            centers = best_c
            radii = r_final
            best_sum = np.sum(radii)
    except Exception:
        centers = best_c
        radii = r_final
        best_sum = np.sum(radii)
        
    # Guarantee strict validity with final LP
    radii_lp, sum_lp, _ = solve_lp(centers)
    
    return centers, np.maximum(radii_lp, 0.0), float(sum_lp)
