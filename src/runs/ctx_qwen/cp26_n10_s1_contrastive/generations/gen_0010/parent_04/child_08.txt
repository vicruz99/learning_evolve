1. **Diverse Initialization**: We generate multiple starting configurations using:
   - Hexagonal lattice patterns with varying row counts, rotations, and scales to capture dense crystalline-like packings.
   - Force-directed layouts that simulate repulsive forces to find organic, space-filling arrangements.
   - Sequential Voronoi-like placement that greedily picks points maximizing minimum distance to boundaries and existing points.

2. **Parameterized SLSQP Optimization**: We optimize using a `(r, u, v)` parameterization where `x = r + u*(1-2r)` and `y = r + v*(1-2r)` with `u, v ∈ [0, 1]`. This automatically satisfies boundary constraints (`r ≤ x ≤ 1-r`, `r ≤ y ≤ 1-r`) and reduces the problem to enforcing pairwise non-overlap `dist² ≥ (r_i + r_j)²`. SLSQP efficiently navigates this constrained space.

3. **LP-Driven Simulated Annealing (SA)**: After an initial SLSQP pass, we run SA directly on the centers. For each center perturbation, we solve a Linear Program to find the globally optimal radii configuration. This decouples position search from size optimization, allowing the algorithm to escape local minima trapped by rigid contact graphs.

4. **Perturbation & Polish Loop**: We repeatedly perturb the best parameters and re-run SLSQP to break remaining symmetries and refine the packing. A final high-tolerance SLSQP run followed by an exact LP solves for the precise radii given the optimized centers.

This pipeline balances exploration (SA, diverse inits) with exploitation (SLSQP, LP), systematically squeezing out every possible fraction of radius sum.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def get_centers_radii(params):
    """Reconstruct centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def seq_init(rng):
    """Generates a Voronoi-like sequential packing initialization."""
    pts = []
    for _ in range(N):
        samples = rng.uniform(0.1, 0.9, (800, 2))
        d_bound = np.minimum(np.minimum(samples[:, 0], 1.0 - samples[:, 0]), 
                             np.minimum(samples[:, 1], 1.0 - samples[:, 1]))
        d_pts = np.full_like(d_bound, 10.0)
        if len(pts) > 0:
            pts_arr = np.array(pts)
            d_pts = np.min(np.linalg.norm(samples[:, np.newaxis, :] - pts_arr[np.newaxis, :, :], axis=2), axis=1)
        min_d = np.minimum(d_bound, d_pts)
        pts.append(samples[np.argmax(min_d)])
    return np.array(pts)

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    
    # Generate diverse initial configurations
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], 
            [6,6,5,5,4], [5,5,6,5,5], [8,5,5,5,3], [5,8,5,5,3], [4,5,5,5,7]]
    for pat in pats:
        for _ in range(3):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.9, 1.1)))
    for _ in range(5):
        inits.append(force_init(rng))
    for _ in range(3):
        inits.append(seq_init(rng))
        
    # Phase 1: Broad SLSQP search from diverse starts
    for c in inits:
        r, _ = solve_lp_radii(c)
        r = np.clip(r * 0.99, 1e-6, 0.49)
        p0 = get_params(c, r)
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
                    best_c, _ = get_centers_radii(best_p)
        except Exception:
            continue
            
    if best_p is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)
        best_p = get_params(c_f, r_f * 0.98)
        best_sum = -obj(best_p)
        best_c = c_f.copy()

    # Phase 2: LP-driven Simulated Annealing on centers
    curr_c = best_c.copy()
    curr_s = best_sum
    step = 0.03
    temp = 0.03
    
    for it in range(4000):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        # Perturb center
        curr_c[idx] += rng.normal(0, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        
        # Occasional large jump to escape deep basins
        if rng.random() < 0.02:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)

        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        # Metropolis criterion
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9994
        step = max(0.0005, step * 0.9995)

    # Update best_p from SA best
    if best_c is not None:
        r_b, _ = solve_lp_radii(best_c)
        best_p = get_params(best_c, np.clip(r_b * 0.995, 1e-6, 0.49))
        best_c, _ = get_centers_radii(best_p)

    # Phase 3: Perturbation & SLSQP refinement to escape local rigidity
    for k in range(20):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.003, N)
        xp[N:3*N] += rng.uniform(-0.02, 0.02, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
                    best_c, _ = get_centers_radii(best_p)
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
            best_c, _ = get_centers_radii(best_p)
    except Exception:
        pass
        
    # Final LP guarantees radii are strictly optimal for the exact final centers
    centers_final = best_c
    radii_final, sum_final = solve_lp_radii(centers_final)
    
    # Ensure non-negative radii
    radii_final = np.maximum(radii_final, 0.0)
    
    return centers_final, radii_final, float(sum_final)
