To maximize the sum of radii for 26 circles in a unit square, I will employ a robust multi-phase optimization strategy:

1. **Diverse Initializations**: Generate multiple starting configurations using:
   - Hexagonal lattices with varied row counts, rotations, and scales.
   - Force-directed layouts where points repel each other and walls to find organic, space-filling arrangements.
   - Perturbed grids to break symmetry.

2. **SLSQP Refinement**: Use `scipy.optimize.minimize` with the `SLSQP` method. To handle boundary constraints automatically and ensure strict feasibility, I'll use a reparameterization `(r, u, v)` where `x = r + u*(1-2r)` and `y = r + v*(1-2r)`. This guarantees `r <= x <= 1-r` and `r <= y <= 1-r`. The only explicit constraints needed are pairwise non-overlap.

3. **Simulated Annealing (SA) on Centers**: After finding the best SLSQP result, run a dedicated SA phase on the centers. At each step, move one circle, solve the LP for radii, and accept/reject based on the Metropolis criterion. This helps escape rigid contact networks that trap gradient-based methods.

4. **Perturbation & Restart**: Aggressively perturb the best solution and re-run SLSQP multiple times to explore nearby basins of attraction.

5. **Shrink & Re-expand**: Scale the configuration slightly towards the center, add noise, and re-optimize. This breaks overly tight constraint topologies that can prevent global improvements.

6. **High-Precision Polish**: Final SLSQP run with tight tolerances and a final LP solve to guarantee optimality of radii for the exact centers.

All helper functions will be top-level to comply with constraints. No closures will be used.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(600):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 60.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 60.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.001 * f
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    
    # Generate diverse initial configurations
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], 
        [5,5,5,5,6], [6,6,5,5,4], [5,5,6,5,5], [8,5,5,5,3],
        [6,7,5,5,3], [5,6,5,5,5]
    ]
    for pat in pats:
        for _ in range(4):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.85, 1.15)))
            
    for _ in range(10):
        inits.append(force_init(rng))
        
    for seed in range(5):
        rng_g = np.random.RandomState(seed)
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng_g.uniform(-0.03, 0.03, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # Phase 1: Broad SLSQP search from diverse starts
    for c in inits:
        r, _ = solve_lp_radii(c)
        r = np.clip(r * 0.985, 1e-6, 0.49)
        p0 = get_params(c, r)
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)

        best_sum = -obj(best_p)
        best_c = c_f.copy()
    else:
        r_opt = best_p[:N]
        u_opt = best_p[N:2*N]
        v_opt = best_p[2*N:3*N]
        best_c = np.column_stack((r_opt + u_opt * (1 - 2 * r_opt),

    # Phase 2: Simulated Annealing on centers
    curr_c = best_c.copy()
    _, curr_s = solve_lp_radii(curr_c)
    step = 0.035
    temp = 0.04
    
    for it in range(15000):
        move_type = rng.integers(0, 3)
        idx = rng.integers(N)
        old_pos = curr_c[idx].copy()
        
        if move_type == 0:
            curr_c[idx] += rng.normal(0, step, 2)
        elif move_type == 1:
            i, j = rng.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
            idx = i
        else:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
            
        curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        
        _, new_s = solve_lp_radii(curr_c)
        delta = new_s - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / max(temp, 1e-4))):
            curr_s = new_s
            if new_s > best_sum:
                best_sum = new_s
                best_c = curr_c.copy()
                best_p = get_params(best_c, np.clip((np.full(N, 0.09)), 1e-6, 0.49))
        else:
            curr_c[idx] = old_pos
            
        temp *= 0.9992
        step = max(0.001, step * 0.9994)
        
    # Update best_p if SA found better
    r_sa, _ = solve_lp_radii(best_c)
    best_p = get_params(best_c, np.clip(r_sa * 0.995, 1e-6, 0.49))
    
    # Phase 3: Perturbation & SLSQP to escape local minima
    for k in range(25):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.0025, N)
        xp[N:3*N] += rng.normal(0, 0.02, 2*N)

        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:

                if s > best_sum:

                    best_p = res.x.copy()

        # Scale towards center
        c_tmp = 0.8 * c_tmp + 0.2 * 0.5
        c_tmp += rng.uniform(-0.02, 0.02, (N, 2))
        c_tmp = np.clip(c_tmp, 0.05, 0.95)

            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            pass

        pass

    u_opt = best_p[N:2*N]
    v_opt = best_p[2*N:3*N]

    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    # Final LP to guarantee radii are optimal for the exact final centers
