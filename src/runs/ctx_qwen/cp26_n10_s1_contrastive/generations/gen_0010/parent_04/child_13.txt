import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def lp_hill_climb(centers, rng, n_steps=6000, init_step=0.025):
    """LP-driven local search on centers to escape local minima."""
    curr_c = centers.copy()
    _, curr_s = solve_lp_radii(curr_c)
    step = init_step
    
    for it in range(n_steps):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.normal(0, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
        
        _, new_s = solve_lp_radii(curr_c)
        if new_s > curr_s:
            curr_s = new_s
            step = min(0.04, step * 1.05)
        else:
            curr_c[idx] = old
            step = max(0.0005, step * 0.96)
            
        # Occasional jump to break rigidity
        if it % 800 == 0:
            idx = rng.integers(N)
            curr_c[idx] = rng.uniform(0.1, 0.9, 2)
            _, curr_s = solve_lp_radii(curr_c)
            
    return curr_c, curr_s

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    
    # Generate diverse initial configurations
    inits = []

            [6,6,5,5,4], [5,5,6,5,5], [7,4,5,5,5], [6,5,5,5,5], [5,6,4,6,5],
            [8,5,5,5,3], [5,5,5,5,5,1], [6,4,5,6,5], [4,5,5,6,6]]
    for pat in pats:
        for _ in range(6):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.35, 0.35), scale=rng.uniform(0.82, 1.18)))
    for _ in range(20):
        inits.append(force_init(rng))
        
    # Phase 1: Broad SLSQP search from diverse starts
    for c in inits:

        r = np.clip(r * 0.99, 1e-6, 0.49)
        p0 = get_params(c, r)
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
                    r_opt = best_p[:N]
                    u_opt = best_p[N:2*N]
                    v_opt = best_p[2*N:3*N]
                    best_c = np.column_stack((r_opt + u_opt * (1 - 2 * r_opt), 
                                              r_opt + v_opt * (1 - 2 * r_opt)))
        except Exception:
            continue
            
    if best_p is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)
        best_p = get_params(c_f, r_f * 0.98)
        best_sum = -obj(best_p)

    # Phase 3: SLSQP refinement & perturbation escapes
    for _ in range(12):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.018, 2*N)

    r_opt = best_p[:N]
    u_opt = best_p[N:2*N]
    v_opt = best_p[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    # Final LP to guarantee radii are strictly optimal for the exact final centers
    radii_final, sum_final = solve_lp_radii(centers)
    
    return centers, radii_final, float(sum_final)
