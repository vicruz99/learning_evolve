import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(p):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.012, 0.012, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 25.0 * np.maximum(0, 0.08 - pts[:, dim])
            f[:, dim] -= 25.0 * np.maximum(0, pts[:, dim] - 0.92)
        pts += 0.002 * f
        pts = np.clip(pts, 0.04, 0.96)
    return pts

def frame_init(rng):
    """Generate a boundary-hugging frame initialization."""
    pts = []
    # Top and bottom rows
    for i in range(7):
        x = 0.1 + i * 0.125
        pts.append([x, 0.09])
        pts.append([x, 0.91])
    # Left and right columns (skip corners)
    for i in range(4):
        y = 0.25 + i * 0.15
        pts.append([0.09, y])
        pts.append([0.91, y])
    # Fill center roughly
    while len(pts) < N:
        pts.append(rng.uniform(0.3, 0.7, 2).tolist())
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():

    rng = np.random.default_rng(42)
    
    bounds = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    
    # Generate diverse initial configurations
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], 
        [5,5,5,5,6], [6,6,5,5,4], [5,5,6,5,5], [8,6,5,4,3],
        [6,5,5,5,5], [5,5,5,6,5], [7,6,5,5,3], [5,7,5,5,4]
    ]
    for pat in pats:
        for _ in range(3):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.2, 0.2), scale=rng.uniform(0.9, 1.1)))
    for _ in range(10):
        inits.append(force_init(rng))
    for _ in range(5):
        inits.append(frame_init(rng))
        
    # Phase 1: Broad SLSQP search from diverse starts
    for c in inits:
        r, _ = solve_lp_radii(c)
        r = np.clip(r * 0.985, 1e-6, 0.49)
        p0 = get_params(c, r)
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
                    best_c, _ = from_params(best_p)
        except Exception:
            continue
            
    if best_p is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)
        best_p = get_params(c_f, r_f * 0.98)
        best_sum = -obj(best_p)
        best_c = c_f.copy()

    # Phase 2: Dual-Guided Local Search on centers
    curr_c = best_c.copy()
    curr_s = best_sum
    step = 0.02
    
    for it in range(1500):
        r_curr, s_curr = solve_lp_radii(curr_c)
        
        # Compute approximate ascent forces based on geometry
        forces = np.zeros((N, 2))

            d_wall = min(curr_c[i,0], 1.0-curr_c[i,0], curr_c[i,1], 1.0-curr_c[i,1])
            for j in range(N):
                if i == j: continue
                diff = curr_c[i] - curr_c[j]
                dist = np.linalg.norm(diff)
                if dist < 1e-8: continue
                # Repulsion proportional to overlap or proximity
                target_dist = r_curr[i] + r_curr[j]
                if dist < target_dist + 0.05:
                    strength = 1.0 / (dist**2 + 1e-8)
                    forces[i] += strength * diff / dist
        forces += 0.5 * forces.max(axis=0, keepdims=True) * np.random.normal(0, 1, (N, 2))

            curr_c[idx] = rng.uniform(0.1, 0.9, 2)

        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj, xp, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 4: High-precision final polish

        res_f = minimize(obj, best_p, method='SLSQP', bounds=bounds, constraints=cons_dict,
                         options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    # Reconstruct physical centers and radii
    centers, radii = from_params(best_p)
    
    # Final LP to guarantee radii are strictly optimal for the exact final centers
    radii_final, sum_final = solve_lp_radii(centers)
    
    return centers, radii_final, float(sum_final)
