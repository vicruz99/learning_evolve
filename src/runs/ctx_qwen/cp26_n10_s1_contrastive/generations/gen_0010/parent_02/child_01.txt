import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def get_lp_sum_radii(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def obj_neg_sum(centers_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    s, _ = get_lp_sum_radii(centers)
    return -s

def generate_hex(row_counts, scale=1.0, rot=0.0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    return np.clip(pts, 0.01, 0.99)

def force_layout(rng):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(400):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        mag = 1.0 / (d**2 + 0.01)
        f += np.sum(mag[:, :, None] * diff / d[:, :, None], axis=1)
        for k in range(2):
            f[:, k] += 10.0 * np.maximum(0, 0.05 - pts[:, k])
            f[:, k] -= 10.0 * np.maximum(0, pts[:, k] - 0.95)
        pts += 0.005 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def cons_slqp(p):
    """Inequality constraints for SLSQP using (r, u, v) parameterization."""
    r_i = p[:N]
    u_i = p[N:2*N]
    v_i = p[2*N:3*N]
    x = r_i + u_i * (1.0 - 2.0 * r_i)
    y = r_i + v_i * (1.0 - 2.0 * r_i)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r_i[:, None] + r_i[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def obj_slqp(p):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(p[:N])

def run_packing():
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5], 
            [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7], [6,6,5,5,4], [5,5,6,5,5]]
    for pat in pats:
        for _ in range(3):
            inits.append(generate_hex(pat, scale=rng.uniform(0.9, 1.1), rot=rng.uniform(-0.3, 0.3)))
    for _ in range(10):
        inits.append(force_layout(rng))
    for _ in range(5):
        inits.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    # Phase 1: Local optimization with Nelder-Mead on centers
    for c0 in inits:
        try:
            res = minimize(obj_neg_sum, c0.flatten(), method='Nelder-Mead', 
                           options={'maxiter': 3000, 'xatol': 1e-6, 'fatol': 1e-6})
            curr_c = res.x.reshape(N, 2)
            curr_s, curr_r = get_lp_sum_radii(curr_c)
            if curr_s > best_sum:
                best_sum = curr_s
                best_c = curr_c.copy()
                best_r = curr_r.copy()
        except Exception:
            continue
            
    # Phase 2: Simulated Annealing on centers
    curr_c = best_c.copy()
    curr_s = best_sum
    curr_r = best_r.copy()
    
    temp = 0.05
    step = 0.03
    
    for it in range(15000):
        idx = rng.integers(N)
        old_c = curr_c[idx].copy()
        
        # Occasional large jump to escape deep local minima
        if rng.random() < 0.02:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            curr_c[idx] += rng.normal(0, step, 2)
            
        curr_c = np.clip(curr_c, 0.005, 0.995)
        
        ns, nr = get_lp_sum_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            curr_r = nr
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
        else:
            curr_c[idx] = old_c
            
        temp *= 0.9996
        step = max(0.0005, step * 0.9998)
        
    # Phase 3: SLSQP refinement on (r, u, v) parameterization
    r = best_r.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((best_c[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((best_c[:, 1] - r) / denom, 0.0, 1.0)
    p0 = np.concatenate([r, u, v])
    
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    
    for _ in range(10):
        xp = p0.copy()
        xp[:N] += rng.normal(0, 0.001, N)
        xp[N:3*N] += rng.uniform(-0.01, 0.01, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, 
                           constraints={'type': 'ineq', 'fun': cons_slqp},
                           options={'maxiter': 4000, 'ftol': 1e-12})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    p0 = res.x.copy()
        except Exception:
            pass
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp,
                         constraints={'type': 'ineq', 'fun': cons_slqp},
                         options={'maxiter': 8000, 'ftol': 1e-14})
        if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-8:
            p0 = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    # Reconstruct centers from optimized parameters
    r_opt = p0[:N]
    u_opt = p0[N:2*N]
    v_opt = p0[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack([x_opt, y_opt])
    
    # Final LP ensures radii are strictly optimal for the exact final centers
    final_r, final_s = get_lp_sum_radii(centers)
    
    return centers, final_r, float(final_s)
