3. **Simulated Annealing (SA) on Centers**: Refine the DE result using SA. This phase uses an adaptive step size, Metropolis acceptance criterion, and includes both single-circle moves and position swaps to escape local minima and rearrange the contact graph topology.
4. **SLSQP Joint Optimization**: Transition to a smooth joint optimization of centers and radii. 5. **Perturbation & Restart**: Run multiple SLSQP restarts from slightly perturbed versions of the best solution to explore adjacent local optima.

import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-6), 1e-5

def de_objective(centers_flat):
    """Objective for Differential Evolution: minimize negative sum of optimal radii."""
    centers = centers_flat.reshape(N, 2)
    _, s = solve_lp_radii(centers)
    return -s

def slsqp_objective(params):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(params[:N])

def slsqp_constraints(params):
    """Inequality constraints: dist^2 >= (r_i + r_j)^2. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': slsqp_constraints}
    
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    # Phase 1: Global search on centers using Differential Evolution
    de_bounds = [(0.02, 0.98)] * (2 * N)
    try:
        res_de = differential_evolution(de_objective, de_bounds, popsize=20, maxiter=300,
                                        mutation=(0.5, 1.0), recombination=0.7, seed=42,
                                        polishing=False, tol=1e-7)
        best_centers = res_de.x.reshape(N, 2)
        best_radii, best_sum = solve_lp_radii(best_centers)
    except Exception:
        pass
        
    # Fallback initialization if DE fails
    if best_centers is None:
        best_centers = rng.uniform(0.15, 0.85, (N, 2))
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    # Phase 2: LP-Driven Simulated Annealing on centers
    curr_c = best_centers.copy()
    curr_s = best_sum
    temp = 0.06
    step = 0.05
    
    for it in range(8000):
        move_type = rng.integers(0, 3)
        old_c = curr_c.copy()
        
        if move_type == 0:
            idx = rng.integers(N)
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        elif move_type == 1:
            i, j = rng.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        else:
            idx = rng.integers(N)
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        # Metropolis criterion
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_centers = curr_c.copy()
                best_radii = nr.copy()
        else:
            curr_c = old_c
            
        temp *= 0.9994
        step = max(0.002, step * 0.9995)
        
    # Phase 3: SLSQP joint optimization of centers and radii
    r_init = np.clip(best_radii * 0.99, 1e-6, 0.49)
    p0 = to_params(best_centers, r_init)
    
    try:
        res = minimize(slsqp_objective, p0, method='SLSQP', bounds=bounds_slqp,
                       constraints=cons_slqp, options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
        if res.success and np.min(slsqp_constraints(res.x)) >= -1e-7:
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_centers, best_radii = from_params(res.x)
    except Exception:
        pass
        
    # Phase 4: Perturbation & SLSQP restarts to escape local minima
    for k in range(40):
        xp = to_params(best_centers, best_radii * 0.995)
        xp[:N] += rng.normal(0, 0.0025, N)
        xp[N:3*N] += rng.uniform(-0.025, 0.025, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(slsqp_objective, xp, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(slsqp_constraints(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_centers, best_radii = from_params(res.x)
        except Exception:
            pass
            
    # Phase 5: High-precision final polish
    p0 = to_params(best_centers, best_radii * 0.999)
    try:
        res_f = minimize(slsqp_objective, p0, method='SLSQP', bounds=bounds_slqp,
                         constraints=cons_slqp, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(slsqp_constraints(res_f.x)) >= -1e-9:
            best_centers, best_radii = from_params(res_f.x)
            best_sum = -res_f.fun
    except Exception:
        pass
        
    # Final LP verification ensures radii are strictly optimal for the exact final centers
    best_radii, best_sum = solve_lp_radii(best_centers)
    
    return best_centers, np.maximum(best_radii, 0.0), float(best_sum)
