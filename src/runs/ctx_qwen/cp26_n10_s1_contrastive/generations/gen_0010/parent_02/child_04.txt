import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.ones(n) * 1e-6, 0.0

def objective(params):
    """Minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints(params):
    """Inequality constraints: dist^2 >= (r_i + r_j)^2. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def hex_init(seed, row_counts, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def corner_init(seed):
    """Generates a corner-focused initialization."""
    rng = np.random.RandomState(seed)
    pts = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
    grid = np.array([[0.25 + i*0.25, 0.25 + j*0.25] for i in range(3) for j in range(3)])
    pts.extend(grid.tolist())
    pts.extend([[0.5, 0.2], [0.5, 0.8], [0.2, 0.5], [0.8, 0.5], [0.5, 0.5]])
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.025, 0.025, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        rep = 1.0 / (dist**2 + 0.02)
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, None] * diff / dist[:, :, None], axis=1)
        for d in range(2):
            f[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            f[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.004 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None

    # Generate diverse initial configurations
    inits = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], 
                [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [4,5,6,5,6], [5,5,6,5,5]]
    for pat in patterns:
        for s in range(6):
            rot = rng.uniform(-0.35, 0.35)
            sc = rng.uniform(0.88, 1.12)
            inits.append(hex_init(s*10+1, pat, rot, sc))
    for s in range(8):
        inits.append(force_init(s))
    for s in range(8):
        inits.append(corner_init(s))
    for s in range(5):
        inits.append(rng.uniform(0.05, 0.95, (N, 2)))

    # Phase 1: Broad SLSQP search from diverse starts
    for c in inits:
        r, _ = solve_lp_radii(c)
        r = np.clip(r * 0.99, 1e-6, 0.49)
        p0 = to_params(c, r)
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue

    # Fallback if optimization failed completely
    if best_p is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)
        best_p = to_params(c_f, r_f * 0.99)
        best_sum = -objective(best_p)

    # Phase 2: Intensive Simulated Annealing on Centers
    curr_c, _ = from_params(best_p)
    curr_r, curr_s = solve_lp_radii(curr_c)
    temp = 0.06
    step = 0.04
    
    for it in range(15000):
        move_type = rng.integers(0, 4)
        old = curr_c.copy()
        
        if move_type == 0: # Single move
            idx = rng.integers(N)
            curr_c[idx] += rng.normal(0, step, 2)
        elif move_type == 1: # Swap
            i, j = rng.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        elif move_type == 2: # Random jump
            idx = rng.integers(N)
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else: # Multi-move
            idxs = rng.choice(N, 3, replace=False)
            curr_c[idxs] += rng.normal(0, step*0.5, (3, 2))
            
        curr_c = np.clip(curr_c, 0.01, 0.99)
        
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_s = ns
            curr_r = nr
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
                best_p = to_params(best_c, best_r * 0.995)
        else:
            curr_c = old
            
        if it % 500 == 0:
            temp *= 0.98
            step = max(0.003, step * 0.995)
            
    if best_c is not None:
        curr_r_s, curr_s_v = solve_lp_radii(best_c)
        if curr_s_v > best_sum:
            best_sum = curr_s_v
            best_r = curr_r_s
            best_p = to_params(best_c, best_r * 0.995)

    # Phase 3: SLSQP refinement from SA best
    if best_p is not None:
        try:
            res = minimize(objective, best_p, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            pass

    # Phase 4: Perturbation & SLSQP restarts
    for k in range(25):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.0025, N)
        xp[N:3*N] += rng.normal(0, 0.018, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(objective, xp, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 5: High-precision final polish
    try:
        res_f = minimize(objective, best_p, method='SLSQP', bounds=bounds, constraints=cons_dict,
                         options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(constraints(res_f.x)) >= -1e-9:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_p)
    final_r, final_sum = solve_lp_radii(centers)
    
    return centers, np.maximum(final_r, 0.0), float(final_sum)
