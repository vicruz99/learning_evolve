2. **Dual-Force Gradient Ascent**: The dual variables (marginals) from the LP indicate which constraints are active. 5. **Diverse Initialization**: We test multiple hexagonal lattice patterns (with random rotations/scales) and force-directed layouts to ensure we don't start in a poor basin.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers. Returns (radii, sum_radii, duals)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            # Duals indicate constraint activity. Absolute value handles solver sign conventions.
            duals = np.abs(np.array(res.ineqlin.marginals).flatten()) if hasattr(res, 'ineqlin') and res.ineqlin is not None else np.zeros(num_c)
            return res.x, -res.fun, duals
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0, np.zeros(num_c)

def compute_forces(centers, duals):
    """Computes ascent forces on centers based on LP dual variables."""
    n = centers.shape[0]
    forces = np.zeros((n, 2))
    
    # Boundary forces: push away from tight walls
    for i in range(n):
        forces[i, 0] += duals[4*i] - duals[4*i+1]
        forces[i, 1] += duals[4*i+2] - duals[4*i+3]
        
    # Pairwise forces: push apart tight circles
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    np.fill_diagonal(dists, 1e-8)
    dirs = diff / dists[:, :, np.newaxis]
    
    idx = 4 * n
    mu_matrix = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            mu_matrix[i, j] = duals[idx]
            mu_matrix[j, i] = duals[idx]
            idx += 1
            
    forces += np.sum(mu_matrix[:, :, np.newaxis] * dirs, axis=1)
    return forces

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj_slqp(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons_slqp(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def run_packing():
    np.random.seed(42)
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    inits = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [7,5,5,5,4]]
    for pat in patterns:
        for s in range(3):
            np.random.seed(s)
            pts = []
            r_est = 0.095
            y = r_est
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * np.random.uniform(0.9, 1.1) + 0.5
            rot = np.random.uniform(-0.3, 0.3)
            c, s_val = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s_val], [s_val, c]]) + 0.5
            pts += np.random.uniform(-0.01, 0.01, (N, 2))
            inits.append(np.clip(pts, 0.02, 0.98))
            
    for s in range(5):
        np.random.seed(s+50)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(300):
            f = np.zeros_like(pts)
            diff = pts[:, None, :] - pts[None, :, :]
            d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            f += np.sum((1.0/d**2)[:, :, None] * diff / d[:, :, None], axis=1)
            pts += 0.003 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)

    bounds_slqp = [(1e-6, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    
    for init_c in inits:
        curr_c = init_c.copy()
        r_curr, s_curr, duals = solve_lp(curr_c)
        step = 0.025
        temp = 0.02
        
        # Phase 1 & 2: Dual-Force Simulated Annealing
        for it in range(1200):
            forces = compute_forces(curr_c, duals)
            f_norm = np.linalg.norm(forces)
            if f_norm > 1e-5:
                forces = forces / f_norm * min(f_norm, 3.0)
                
            noise = np.random.normal(0, step * 0.15, (N, 2))
            new_c = np.clip(curr_c + step * forces + temp * noise, 0.01, 0.99)
            
            r_new, s_new, duals = solve_lp(new_c)
            delta = s_new - s_curr
            
            if delta > 0 or (np.random.rand() < np.exp(delta / max(temp, 1e-5))):
                curr_c = new_c
                s_curr = s_new
                step = max(0.0005, step * 1.001)
            else:
                step = max(0.0005, step * 0.996)
            temp = max(0.0001, temp * 0.999)
            
        if s_curr > best_sum:
            best_sum = s_curr
            best_c = curr_c.copy()
            best_r = r_curr.copy()
            
        # Phase 3: SLSQP Polish
        r_p = np.clip(best_r * 0.995, 1e-6, 0.49)
        p0 = get_params(best_c, r_p)
        for _ in range(2):
            try:
                res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, 
                               constraints={'type': 'ineq', 'fun': cons_slqp},
                               options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        r_opt = res.x[:N]
                        u_opt = res.x[N:2*N]
                        v_opt = res.x[2*N:3*N]
                        best_c = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))
                        best_r = r_opt
                        p0 = res.x.copy()
            except Exception:
                pass

    # Final LP verification to ensure radii exactly match centers
    best_r, best_sum_final, _ = solve_lp(best_c)
    
    return best_c, np.maximum(best_r, 0.0), float(best_sum_final)
