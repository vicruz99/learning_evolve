

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we employ a multi-phase optimization strategy that combines exact linear programming, gradient-based ascent using LP dual variables, and constrained nonlinear refinement.

1. **Exact LP for Radii**: For any fixed set of centers, the optimal radii maximizing `sum(r_i)` subject to boundary and non-overlap constraints form a Linear Program. Solving this LP is extremely fast and gives both the optimal radii and their dual variables (Lagrange multipliers).
2. **Dual-Guided Gradient Ascent**: The dual variables from the LP exactly represent the sensitivity of the objective to constraint relaxation. We use them to compute an ascent direction for the centers: moving centers apart along active contact constraints increases the allowable radii. This acts as a "stress maximization" step that rapidly escapes local minima and pushes circles into tightly packed configurations.
3. **Parameterized SLSQP Refinement**: We convert the best configuration to a boundary-safe `(r, u, v)` parameterization where `x = r + u*(1-2r)` and `y = r + v*(1-2r)`. This automatically satisfies boundary constraints, leaving SLSQP to only handle pairwise non-overlap (`dist^2 >= (r_i+r_j)^2`). This phase polishes the packing to high precision.
4. **Diverse Initialization & Perturbation**: We start from multiple hexagonal lattice patterns with rotations/scaling, force-directed layouts, and sequential gap-filling placements. Periodic random perturbations during ascent prevent the contact graph from becoming rigidly stuck.
5. **Final Polish & Validation**: After convergence, we run a high-iteration SLSQP pass and a final LP to guarantee the radii are strictly optimal for the exact final centers, ensuring the solution passes all validation checks.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            duals = None
            # Handle different scipy versions for dual variables
            if hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
                duals = res.marginals.ineqlin
            elif hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
                duals = res.ineqlin.marginals
                
            return res.x, -res.fun, duals
    except Exception:
        pass
        
    return np.full(n, 1e-5), 0.0, None

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def compute_dual_forces(centers, duals):
    """Computes ascent forces on centers based on LP dual variables."""
    n = centers.shape[0]
    forces = np.zeros((n, 2))
    if duals is None or len(duals) < 4 * n + 1:
        return forces
        
    idx = 4 * n  # Skip boundary constraint duals
    
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]
            idx += 1
            if lam > 1e-7:  # Active constraint
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist > 1e-8:
                    forces[i] += lam * diff / dist
                    forces[j] -= lam * diff / dist
                    
    return forces

def hex_init(rng, row_counts, rot=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(250):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], 
            [5,5,5,5,6], [5,5,6,5,5], [6,6,5,5,4], [5,5,5,5,5,1]]
    for pat in pats:
        for _ in range(6):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.88, 1.12)))
    for _ in range(10):
        inits.append(force_init(rng))
        
    # Phase 1: Broad SLSQP search from diverse starts to find a strong baseline
    for c in inits:
        r, s_val, _ = solve_lp_radii(c)
        if s_val > best_sum:
            best_sum = s_val
            best_c = c.copy()
            best_r = r.copy()
            
    # Convert best LP result to (r, u, v) and run initial SLSQP
    r_init = np.clip(best_r * 0.985, 1e-6, 0.49)
    p0 = get_params(best_c, r_init)
    try:
        res = minimize(obj, p0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                       options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
        if res.success and np.min(cons(res.x)) >= -1e-7:
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_p = res.x.copy()
    except Exception:
        pass
        
    if best_p is None:
        best_p = get_params(best_c, r_init)
        
    # Phase 2: Dual-Guided Local Search (Stress Maximization)
    curr_c = best_c.copy()
    curr_s = best_sum
    step = 0.03
    
    for it in range(15000):
        r_curr, s_curr, duals = solve_lp_radii(curr_c)
        
        if duals is not None:
            forces = compute_dual_forces(curr_c, duals)
            # Normalize forces to prevent explosion and maintain stable step size
            norm_f = np.linalg.norm(forces, axis=1, keepdims=True)
            norm_f = np.maximum(norm_f, 1e-8)
            forces = forces / norm_f * step
            
            new_c = np.clip(curr_c + forces, 0.005, 0.995)
            
            _, s_new, _ = solve_lp_radii(new_c)
            
            if s_new > s_curr + 1e-9:
                curr_c = new_c
                curr_s = s_new
                step = min(0.05, step * 1.05)
                if s_new > best_sum:
                    best_sum = s_new
                    best_c = curr_c.copy()
                    best_r, _ = solve_lp_radii(best_c)
            else:
                step = max(0.001, step * 0.95)
                
        # Occasionally perturb to escape rigidity/local minima
        if it % 500 == 0:
            idx = rng.integers(N)
            curr_c[idx] += rng.normal(0, 0.02, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            _, curr_s, _ = solve_lp_radii(curr_c)

    # Update best_p if dual search found better
    if best_c is not None:
        best_p = get_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
        
    # Phase 3: SLSQP refinement & perturbation to escape remaining local minima
    for k in range(40):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.015, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj, xp, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj, best_p, method='SLSQP', bounds=bounds, constraints=cons_dict,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    # Reconstruct physical centers and radii
    r_opt = best_p[:N]
    u_opt = best_p[N:2*N]
    v_opt = best_p[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    # Final LP to guarantee radii are strictly optimal for the exact final centers
    radii_final, sum_final, _ = solve_lp_radii(centers)
    
    return centers, radii_final, float(sum_final)
