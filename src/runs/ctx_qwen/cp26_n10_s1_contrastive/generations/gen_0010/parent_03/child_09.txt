

<strategy>
The current approach uses a combination of LP for radius optimization, SLSQP for joint center-radius optimization via a boundary-safe parameterization, and Simulated Annealing on centers. A critical bug in the previous code was an incorrect reconstruction of centers from parameters (`1 - 2 * v_b` instead of `1 - 2 * r_b` for the y-coordinate), which likely caused suboptimal convergence.

My strategy to improve the sum of radii:
1. **Fix Parameterization Bug**: Correct the center reconstruction formula to ensure the optimizer correctly maps back to physical coordinates.
2. **Dual-Guided Gradient Ascent**: The LP solver returns dual variables (marginals) that indicate how much the objective (sum of radii) improves if pairwise distances increase. These duals provide exact gradient information for moving centers. I will implement a dual-guided gradient ascent loop where centers are pushed apart along the direction of active constraints, significantly accelerating convergence compared to blind SA.
3. **Adaptive Step Sizes & Random Jumps**: Combine the gradient ascent with an adaptive step size and periodic random perturbations to escape local minima and explore the configuration space effectively.
4. **Robust Refinement**: Follow the global/local search with multiple SLSQP refinements on the `(r, u, v)` parameterization to polish the solution to high precision, ensuring strict feasibility.
5. **Diverse Initializations**: Retain hexagonal and force-directed initializations to provide strong starting points for the optimization pipeline.

This hybrid approach leverages exact sensitivity analysis from the LP to guide geometric movements, leading to tighter packings and a higher sum of radii.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            duals = np.array(res.ineqlin.marginals).flatten() if res.ineqlin is not None else None
            return res.x, -res.fun, duals
    except Exception:
        pass
        
    return np.full(n, 1e-5), 0.0, None

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def compute_dual_forces(centers, duals):
    """Computes ascent forces on centers based on LP dual variables."""
    n = centers.shape[0]
    forces = np.zeros((n, 2))
    idx = 4 * n  # Skip boundary constraint duals
    
    for i in range(n):
        for j in range(i + 1, n):
            # Marginals for <= constraints in minimization are <= 0.
            # We negate to get positive force magnitude.
            lam = -duals[idx]
            idx += 1
            if lam > 1e-7:  
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist > 1e-8:
                    # Force points in direction that increases distance
                    forces[i] += lam * diff / dist
                    forces[j] -= lam * diff / dist
                    
    return forces

def hex_init(rng, row_counts, rot=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(250):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], [5,5,6,5,5], [6,6,5,5,4]]
    for pat in pats:
        for _ in range(3):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.88, 1.12)))
    for _ in range(5):
        inits.append(force_init(rng))
        
    # Phase 1: Broad SLSQP search from diverse starts to find a strong baseline
    for c in inits:
        r, _, _ = solve_lp_radii(c)
        r = np.clip(r * 0.98, 1e-6, 0.49)
        p0 = get_params(c, r)
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization failed completely
    if best_p is None:
        c_f = inits[0]
        r_f, _, _ = solve_lp_radii(c_f)
        best_p = get_params(c_f, r_f * 0.98)
        best_sum = -obj(best_p)
        
    # Extract current best for gradient ascent
    r_b = best_p[:N]
    u_b = best_p[N:2*N]
    v_b = best_p[2*N:3*N]
    # Fixed reconstruction bug: y coordinate depends on r, not v
    c_best = np.column_stack((r_b + u_b * (1 - 2 * r_b), r_b + v_b * (1 - 2 * r_b)))
    
    # Phase 2: Dual-Guided Gradient Ascent + Adaptive SA
    curr_c = c_best.copy()
    curr_s = best_sum
    step = 0.025
    
    for it in range(15000):
        r_curr, s_curr, duals = solve_lp_radii(curr_c)
        
        if duals is not None:
            forces = compute_dual_forces(curr_c, duals)
            # Adaptive step size with slight randomization
            curr_step = step * (0.8 + 0.4 * np.random.rand())
            new_c = np.clip(curr_c + curr_step * forces, 0.005, 0.995)
        else:
            # Fallback to random perturbation if LP fails to return duals
            idx = rng.integers(N)
            new_c = curr_c.copy()
            new_c[idx] = np.clip(new_c[idx] + step * rng.standard_normal(2), 0.02, 0.98)
            
        _, s_new, _ = solve_lp_radii(new_c)
        
        if s_new > s_curr + 1e-9:
            curr_c = new_c
            curr_s = s_new
            step = min(0.05, step * 1.03)
            if s_new > best_sum:
                best_sum = s_new
                best_c = curr_c.copy()
                best_r, _, _ = solve_lp_radii(best_c)
        else:
            step = max(0.001, step * 0.96)
            
        # Occasional random jump to escape local rigidity
        if it % 2000 == 0:
            idx = rng.integers(N)
            curr_c[idx] = rng.uniform(0.1, 0.9, 2)
            _, curr_s, _ = solve_lp_radii(curr_c)

    # Update best_p if SA/Gradient search found better
    if best_c is not None:
        best_p = get_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
        
    # Phase 3: Perturbation & SLSQP to escape local minima
    for k in range(20):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.015, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    # Reconstruct physical centers and radii
    r_opt = best_p[:N]
    u_opt = best_p[N:2*N]
    v_opt = best_p[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
