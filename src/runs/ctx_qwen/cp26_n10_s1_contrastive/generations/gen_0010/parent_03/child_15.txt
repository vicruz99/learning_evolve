Key improvements over previous attempts:
1. **Dual-Guided Local Search**: Instead of purely random moves in Simulated Annealing (SA), I extract dual variables (Lagrange multipliers) from the LP solver. These duals indicate exactly how much the objective (sum of radii) would increase if specific constraints (circle-circle or circle-boundary) were relaxed. I compute ascent forces on circle centers proportional to these duals, directing the search efficiently towards configurations that allow larger radii.
2. **Robust Parameterization for SLSQP**: I use a `(r, u, v)` parameterization where `x = r + u*(1-2r)` and `y = r + v*(1-2r)`. This automatically satisfies boundary constraints `r <= x <= 1-r` and `r <= y <= 1-r`, drastically reducing the constraint burden on the NLP solver and allowing it to focus solely on inter-circle non-overlap.
3. **Hybrid SA Schedule**: The SA phase alternates between dual-guided moves (75% of the time) and random perturbations (25%). 4. **Multi-Restart Refinement**: After SA identifies a strong basin, I run multiple SLSQP optimizations from perturbed versions of the best configuration to squeeze out numerical precision and escape minor local optima within the basin.
5. **Final LP Polish**: Regardless of the NLP output, a final LP step guarantees that radii are mathematically optimal for the exact final centers, ensuring maximum sum of radii and strict feasibility.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers, returns radii, sum_radii, and dual variables."""
    n = centers.shape[0]
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0; A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            duals_raw = getattr(getattr(res, 'ineqlin', None), 'marginals', None)
            duals = np.asarray(duals_raw).flatten() if duals_raw is not None else None
            return res.x, -res.fun, duals
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4, None

def get_dual_forces(centers, duals):
    """Computes ascent directions on centers based on LP dual variables."""
    n = centers.shape[0]
    forces = np.zeros((n, 2))
    if duals is None or len(duals) != 4 * n + n * (n - 1) // 2:
        return forces
        
    # Pairwise constraint forces
    k = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[k]
            k += 1
            if lam > 1e-6:
                diff = centers[i] - centers[j]
                d = np.linalg.norm(diff)
                if d > 1e-8:
                    f = lam * diff / d
                    forces[i] += f
                    forces[j] -= f
                    
    # Boundary constraint forces
    k = 0
    for i in range(n):
        forces[i, 0] += duals[k]       # r <= x
        forces[i, 0] -= duals[k + 1]   # r <= 1-x
        forces[i, 1] += duals[k + 2]   # r <= y
        forces[i, 1] -= duals[k + 3]   # r <= 1-y
        k += 4
    return forces

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, None] * diff / d[:, :, None], axis=1)
        for dim in range(2):
            f[:, dim] += 15.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 15.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.002 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Phase 1: Diverse Initializations & SLSQP Baseline
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], [5,5,6,5,5]]
    for pat in pats:
        for _ in range(4):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.2, 0.2), scale=rng.uniform(0.9, 1.1)))
    for _ in range(6):
        inits.append(force_init(rng))
    for _ in range(4):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    for c in inits:
        r, _, _ = solve_lp(c)
        r = np.clip(r * 0.98, 1e-6, 0.49)
        p0 = get_params(c, r)
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        c_f = inits[0]
        r_f, _, _ = solve_lp(c_f)
        best_p = get_params(c_f, r_f * 0.98)
        best_sum = -obj(best_p)
        
    r_b = best_p[:N]
    u_b = best_p[N:2*N]
    v_b = best_p[2*N:3*N]
    c_best = np.column_stack((r_b + u_b * (1 - 2 * r_b), r_b + v_b * (1 - 2 * v_b)))
    best_c = c_best.copy()
    best_r = r_b.copy()
    
    # Phase 2: Dual-Guided Simulated Annealing on Centers
    curr_c = c_best.copy()
    curr_s = best_sum
    step = 0.035
    temp = 0.04
    
    for it in range(30000):
        move_type = rng.random()
        if move_type < 0.75:
            _, _, duals = solve_lp(curr_c)
            forces = get_dual_forces(curr_c, duals)
            norms = np.linalg.norm(forces, axis=1)
            norms[norms < 1e-6] = 1e-6
            forces = forces / norms[:, None] * step
            nc = curr_c + forces
        else:
            nc = curr_c.copy()
            idx = rng.integers(N)
            nc[idx] += rng.normal(0, step, 2)
            
        nc = np.clip(nc, 0.01, 0.99)
        _, ns, _ = solve_lp(nc)
        delta = ns - curr_s
        
        if delta > 0 or (rng.random() < np.exp(delta / max(temp, 1e-4))):
            curr_c = nc
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_c = nc.copy()
                best_r, _, _ = solve_lp(best_c)
            step = max(0.001, step * 0.9997)
            temp = max(0.0001, temp * 0.9996)
        else:
            step = min(0.05, step * 1.001)
            
    if best_c is not None:
        best_p = get_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
        
    # Phase 3: Perturbation & SLSQP to escape local minima
    for k in range(20):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.015, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    # Reconstruct physical centers and radii
    r_opt = best_p[:N]
    u_opt = best_p[N:2*N]
    v_opt = best_p[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    # Final LP ensures radii are strictly optimal for the exact final centers
    radii_final, sum_final, _ = solve_lp(centers)
    return centers, np.maximum(radii_final, 0.0), float(sum_final)
