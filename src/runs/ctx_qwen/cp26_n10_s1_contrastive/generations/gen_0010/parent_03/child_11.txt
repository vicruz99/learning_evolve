2. **Boundary-Safe SLSQP**: Uses a parameterization `(r, u, v)` where `x = r + u(1-2r)`, `y = r + v(1-2r)` to automatically satisfy boundary constraints, allowing SLSQP to focus on pairwise non-overlap.

5. **Iterative Refinement**: After the local search, we run multiple SLSQP restarts from perturbed states, followed by a high-precision final polish to converge to the local optimum.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            duals = None
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                duals = np.array(res.ineqlin.marginals).flatten()
            return res.x, -res.fun, duals
    except Exception:
        pass
        
    return np.full(n, 1e-5), 0.0, None

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def compute_dual_forces(centers, duals):
    """Computes ascent forces on centers based on LP dual variables."""
    n = centers.shape[0]
    forces = np.zeros((n, 2))
    idx = 4 * n  # Skip boundary constraint duals
    
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]
            idx += 1
            if lam > 1e-6:  # Active constraint
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist > 1e-8:
                    # Force points in direction that increases distance
                    forces[i] += lam * diff / dist
                    forces[j] -= lam * diff / dist
                    
    return forces

def hex_init(rng, row_counts, rot, scale):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.002 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], [6,6,5,5,4], [5,5,6,5,5], [5,5,5,5,5,1]]
    for pat in pats:
        for _ in range(4):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.85, 1.15)))
    for _ in range(10):
        inits.append(force_init(rng))
    for _ in range(5):
        pts = rng.uniform(0.05, 0.95, (N, 2))
        inits.append(pts)
        
    # Phase 1: Broad SLSQP search from diverse starts to find a strong baseline
    for c in inits:
        r, _, _ = solve_lp(c)
        r = np.clip(r * 0.985, 1e-6, 0.49)
        p0 = get_params(c, r)
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization failed completely
    if best_p is None:
        c_f = inits[0]
        r_f, _, _ = solve_lp(c_f)
        best_p = get_params(c_f, r_f * 0.98)
        best_sum = -obj(best_p)
        
    # Extract current best for SA
    r_b = best_p[:N]
    u_b = best_p[N:2*N]
    v_b = best_p[2*N:3*N]
    c_best = np.column_stack((r_b + u_b * (1 - 2 * r_b), r_b + v_b * (1 - 2 * v_b)))
    
    # Phase 2: Dual-Guided Local Search + SA
    curr_c = c_best.copy()
    curr_s = best_sum
    step = 0.025
    temp = 0.03
    
    for it in range(15000):
        # Try dual-guided move
        r_curr, s_curr, duals = solve_lp(curr_c)
        if duals is not None:
            forces = compute_dual_forces(curr_c, duals)
            new_c = np.clip(curr_c + step * forces, 0.005, 0.995)
            _, s_new, _ = solve_lp(new_c)
            
            if s_new > s_curr + 1e-9:
                curr_c = new_c
                curr_s = s_new
                step = min(0.05, step * 1.05)
                if s_new > best_sum:
                    best_sum = s_new
                    best_c = curr_c.copy()
                    best_r, _, _ = solve_lp(best_c)
            else:
                step = max(0.0005, step * 0.95)
        else:
            step *= 0.5
            
        # Occasionally swap two circles to break rigid clusters
        if it % 50 == 0:
            i, j = rng.choice(N, 2, replace=False)
            new_c = curr_c.copy()
            new_c[i], new_c[j] = new_c[j], new_c[i]
            _, s_new, _ = solve_lp(new_c)
            if s_new > curr_s:
                curr_c = new_c
                curr_s = s_new
                if s_new > best_sum:
                    best_sum = s_new
                    best_c = curr_c.copy()
                    best_r, _, _ = solve_lp(best_c)
                    
        # Random jump to escape deep local minima
        if it % 200 == 0:
            idx = rng.integers(N)
            curr_c[idx] = rng.uniform(0.1, 0.9, 2)
            _, curr_s, _ = solve_lp(curr_c)
            step = 0.03
            
        # Standard SA move
        if rng.random() < 0.3:
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step * 0.5, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            _, ns, _ = solve_lp(curr_c)
            delta = ns - curr_s
            if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
                curr_s = ns
                if ns > best_sum:
                    best_sum = ns
                    best_c = curr_c.copy()
                    best_r, _, _ = solve_lp(best_c)
            else:
                curr_c[idx] = old
            temp = max(0.0001, temp * 0.9995)
            
    if best_c is not None:
        best_p = get_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
        
    # Phase 3: Perturbation & SLSQP restarts to escape local minima
    for k in range(20):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.02, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        try:
            res = minimize(obj, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    # Reconstruct physical centers and radii
    r_opt = best_p[:N]
    u_opt = best_p[N:2*N]
    v_opt = best_p[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
