

<strategy>
1. Use the effective `(r, u, v)` parameterization to automatically satisfy boundary constraints, simplifying the SLSQP optimization to only pairwise non-overlap constraints.
2. Implement a fast LP solver (`linprog` with 'highs') to compute optimal radii for fixed centers. Extract dual variables to guide a gradient-like ascent on the center positions during Simulated Annealing.
3. Generate diverse initial configurations using hexagonal lattices (with varied row counts, rotations, scales), force-directed layouts, and sequential Voronoi-like packing.
4. Run SLSQP from each initialization to find strong local optima.
5. Perform LP-driven Simulated Annealing on centers, mixing dual-guided moves (pushing apart tightly constrained circles) with random perturbations to escape local minima.
6. Refine the best SA result with multiple perturbed SLSQP restarts.
7. Apply a high-precision final polish and a final LP call to guarantee radii are strictly optimal for the returned centers.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            duals = None
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                duals = np.asarray(res.ineqlin.marginals).flatten()
            return res.x, -res.fun, duals
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0, None

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def compute_dual_forces(centers, duals):
    """Computes ascent forces on centers based on LP dual variables."""
    n = centers.shape[0]
    forces = np.zeros((n, 2))
    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]
            idx += 1
            if lam > 1e-7:
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist > 1e-8:
                    forces[i] += lam * diff / dist
                    forces[j] -= lam * diff / dist
    return forces

def seq_init(rng):
    """Generates a Voronoi-like sequential packing initialization."""
    pts = []
    for _ in range(N):
        samples = rng.uniform(0.1, 0.9, (1500, 2))
        d_bound = np.minimum(np.minimum(samples[:, 0], 1.0 - samples[:, 0]), 
                             np.minimum(samples[:, 1], 1.0 - samples[:, 1]))
        d_pts = np.full_like(d_bound, 10.0)
        for p in pts:
            d_pts = np.minimum(d_pts, np.linalg.norm(samples - p, axis=1))
        min_d = np.minimum(d_bound, d_pts)
        pts.append(samples[np.argmax(min_d)])
    return np.array(pts)

def hex_init(rng, row_counts, rot=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(250):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    
    # Generate diverse initial configurations
    inits = []
    for _ in range(5):
        inits.append(seq_init(rng))
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], [5,5,6,5,5], [6,4,6,5,5], [6,6,5,5,4]]
    for pat in pats:
        for _ in range(3):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.25, 0.25), scale=rng.uniform(0.9, 1.1)))
    for _ in range(5):
        inits.append(force_init(rng))
        
    # Phase 1: Broad SLSQP search from diverse starts
    for c in inits:
        r, _, _ = solve_lp_radii(c)
        r = np.clip(r * 0.985, 1e-6, 0.49)
        p0 = get_params(c, r)
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization failed completely
    if best_p is None:
        c_f = inits[0]
        r_f, _, _ = solve_lp_radii(c_f)
        best_p = get_params(c_f, r_f * 0.98)
        best_sum = -obj(best_p)
        
    # Extract current best for SA
    r_b = best_p[:N]
    u_b = best_p[N:2*N]
    v_b = best_p[2*N:3*N]
    c_best = np.column_stack((r_b + u_b * (1 - 2 * r_b), r_b + v_b * (1 - 2 * v_b)))
    
    # Phase 2: LP-Driven Simulated Annealing on Centers
    curr_c = c_best.copy()
    curr_s = best_sum
    step = 0.025
    temp = 0.03
    
    for it in range(15000):
        move_type = rng.integers(0, 10)
        if move_type < 6:
            _, _, duals = solve_lp_radii(curr_c)
            if duals is not None:
                forces = compute_dual_forces(curr_c, duals)
                nc = curr_c + step * forces
            else:
                idx = rng.integers(N)
                nc = curr_c.copy()
                nc[idx] += rng.normal(0, step, 2)
        elif move_type < 9:
            idx = rng.integers(N)
            nc = curr_c.copy()
            nc[idx] += rng.normal(0, step, 2)
        else:
            idx = rng.integers(N)
            nc = curr_c.copy()
            nc[idx] = rng.uniform(0.05, 0.95, 2)
            
        nc = np.clip(nc, 0.005, 0.995)
        
        _, ns, _ = solve_lp_radii(nc)
        delta = ns - curr_s
        
        if delta > 0 or (rng.random() < np.exp(delta / max(temp, 1e-6))):
            curr_c = nc
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_c = nc.copy()
            step = max(0.001, step * 0.9995)
            temp = max(0.0001, temp * 0.9995)
        else:
            step = min(0.05, step * 1.001)
            
    # Update best_p if SA found better
    if best_c is not None:
        best_r, _, _ = solve_lp_radii(best_c)
        best_p = get_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
        
    # Phase 3: Perturbation & SLSQP to escape local minima
    for k in range(20):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.0025, N)
        xp[N:3*N] += rng.normal(0, 0.018, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj, xp, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj, best_p, method='SLSQP', bounds=bounds, constraints=cons_dict,
                         options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    # Reconstruct physical centers and radii
    r_opt = best_p[:N]
    u_opt = best_p[N:2*N]
    v_opt = best_p[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    # Final LP to guarantee radii are optimal for the exact final centers
    radii_final, sum_final, _ = solve_lp_radii(centers)
    
    return centers, np.maximum(radii_final, 0.0), float(sum_final)
