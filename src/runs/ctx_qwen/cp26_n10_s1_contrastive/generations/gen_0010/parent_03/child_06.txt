To push the sum of radii beyond 2.635, we need a more robust search strategy that combines:
1. **Dual-Guided Gradient Ascent**: The LP solver provides Lagrange multipliers (duals) for each constraint. These directly indicate how much the objective improves if we increase the distance between specific circle pairs. Using these as forces creates a highly informed search direction, far superior to random walks.
2. **Momentum & Adaptive Steps**: Pure gradient ascent can oscillate or get stuck in flat regions. We add momentum to the forces and adaptively scale the step size based on recent acceptance rates.
3. **Hybrid SA + SLSQP**: We alternate between a fast, dual-guided Simulated Annealing (SA) on centers (which handles the non-smooth objective landscape well) and a high-precision SLSQP polish (which uses the smooth `(r, u, v)` parameterization to satisfy boundaries automatically and finely tune positions/radii simultaneously).
4. **Diverse & Structured Initializations**: We seed the search with multiple hexagonal lattice variants, force-directed layouts, and Voronoi-like sequential packings to cover different topological basins.
5. **Strict Feasibility & Polish**: After the main search, we run an aggressive SLSQP polish and a final LP to guarantee the returned radii are strictly optimal for the final centers, ensuring validity under the checker's tolerance.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to maximize sum(r_i) for fixed centers. Returns radii, sum, duals."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            duals = None
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                duals = np.array(res.ineqlin.marginals).flatten()
            return res.x, -res.fun, duals
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0, None

def compute_dual_forces(centers, duals):
    """Computes ascent forces on centers based on LP dual variables."""
    n = centers.shape[0]
    forces = np.zeros((n, 2))
    if duals is None:
        return forces
        
    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]
            idx += 1
            if lam > 1e-7:
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist > 1e-8:
                    f = lam * diff / dist
                    forces[i] += f
                    forces[j] -= f
    return forces

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons_slqp(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_hex(rng, row_counts, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.012, 0.012, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force(rng):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(250):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], [5,5,6,5,5]]
    for pat in pats:
        for _ in range(3):
            inits.append(generate_hex(rng, pat, rot=rng.uniform(-0.25, 0.25), scale=rng.uniform(0.9, 1.1)))
    for _ in range(8):
        inits.append(generate_force(rng))
        
    # Phase 1: Broad SLSQP search from diverse starts
    for c in inits:
        r, _, _ = solve_lp(c)
        r = np.clip(r * 0.985, 1e-6, 0.49)
        p0 = to_params(c, r)
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        c_f = inits[0]
        r_f, _, _ = solve_lp(c_f)
        best_p = to_params(c_f, r_f * 0.98)
        best_sum = -obj_slqp(best_p)
        
    r_b = best_p[:N]
    u_b = best_p[N:2*N]
    v_b = best_p[2*N:3*N]
    c_best = np.column_stack((r_b + u_b * (1 - 2 * r_b), r_b + v_b * (1 - 2 * v_b)))
    
    # Phase 2: Dual-Guided Simulated Annealing on Centers
    curr_c = c_best.copy()
    curr_s = best_sum
    step = 0.03
    temp = 0.025
    momentum_forces = np.zeros_like(curr_c)
    
    for it in range(15000):
        # Occasional large jump to escape deep basins
        if rng.random() < 0.01:
            idx = rng.integers(N)
            nc = curr_c.copy()
            nc[idx] = rng.uniform(0.05, 0.95, 2)
            move_type = 'jump'
        else:
            _, _, duals = solve_lp(curr_c)
            forces = compute_dual_forces(curr_c, duals)
            momentum_forces = 0.6 * momentum_forces + 0.4 * forces
            
            idx = rng.integers(N)
            nc = curr_c.copy()
            nc[idx] += step * (momentum_forces[idx] + rng.standard_normal(2) * 0.5)
            nc[idx] = np.clip(nc[idx], 0.01, 0.99)
            move_type = 'guided'
            
        _, ns = solve_lp(nc)
        delta = ns - curr_s
        
        accept = delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp))
        
        if accept:
            curr_c = nc
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_c = nc.copy()
                best_r, _, _ = solve_lp(best_c)
            step = min(0.05, step * 1.002)
        else:
            step = max(0.0005, step * 0.98)
            
        temp *= 0.9994
        
    if best_c is not None:
        best_p = to_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
        
    # Phase 3: Perturbation & SLSQP to escape remaining local minima
    for k in range(20):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.015, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    # Reconstruct physical centers and radii
    r_opt = best_p[:N]
    u_opt = best_p[N:2*N]
    v_opt = best_p[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    # Final LP to guarantee radii are optimal for the exact final centers
    radii_final, sum_final, _ = solve_lp(centers)
    
    return centers, np.maximum(radii_final, 0.0), float(sum_final)
