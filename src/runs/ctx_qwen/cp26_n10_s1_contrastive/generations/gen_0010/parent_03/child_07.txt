import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A = np.zeros((n_con, n))
    b = np.zeros(n_con)
    k = 0
    for i in range(n):
        x, y = centers[i]
        vals = [x, 1.0 - x, y, 1.0 - y]
        for v in vals:
            A[k, i] = 1.0
            b[k] = v
            k += 1
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            b[k] = dists[i, j]
            k += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def get_params(centers, radii):
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj_slqp(p):
    return -np.sum(p[:N])

def cons_slqp(p):
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, rows, rot=0.0, scale=1.0):
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(rows):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.012, 0.012, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(200):
        diff = pts[:, None, :] - pts[None, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f = np.sum((1.0 / d**2)[:, :, None] * diff / d[:, :, None], axis=1)
        pts += 0.006 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_sa(centers, rng, n_iter=25000):
    curr = centers.copy()
    _, s_curr = solve_lp(curr)
    best_c = curr.copy()
    best_s = s_curr
    temp = 0.035
    step = 0.045
    
    for it in range(n_iter):
        idx = rng.integers(N)
        new_c = curr.copy()
        new_c[idx] += rng.normal(0, step, 2)
        new_c[idx] = np.clip(new_c[idx], 0.015, 0.985)
        
        _, s_new = solve_lp(new_c)
        delta = s_new - s_curr
        
        accept = False
        if delta > 0:
            accept = True
        elif temp > 1e-8:
            prob = np.exp(delta / max(temp, 1e-8))
            if rng.random() < prob:
                accept = True
                
        if accept:
            curr = new_c
            s_curr = s_new
            if s_new > best_s:
                best_s = s_new
                best_c = curr.copy()
                step = min(0.055, step * 1.04)
            else:
                step = max(0.002, step * 0.992)
            temp *= 0.9993
        else:
            step = min(0.065, step * 1.008)
            
        if it % 600 == 0:
            idxs = rng.choice(N, 2, replace=False)
            new_c = curr.copy()
            new_c[idxs[0]] += rng.normal(0, step * 1.2, 2)
            new_c[idxs[1]] += rng.normal(0, step * 1.2, 2)
            new_c = np.clip(new_c, 0.015, 0.985)
            _, s_new = solve_lp(new_c)
            delta = s_new - s_curr
            if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / max(temp, 1e-8))):
                curr = new_c
                s_curr = s_new
                if s_new > best_s:
                    best_s = s_new
                    best_c = curr.copy()
                    
    return best_c, best_s

def run_packing():

                res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds,

            best_c = np.column_stack((x_opt, y_opt))

            best_sum = np.sum(best_r)
            
    final_r, _ = solve_lp(best_c)
    return best_c, np.maximum(final_r, 0.0), float(np.sum(final_r))
