1. **LP-Based Center Optimization**: For any fixed set of centers, finding the optimal radii that maximize their sum is a Linear Programming (LP) problem. We use `scipy.optimize.linprog` (with the efficient `highs` solver) to solve this exactly. We then treat the negative LP objective as a function of the centers and optimize it using `Nelder-Mead` from multiple diverse initial configurations (hexagonal, perturbed grids). This phase efficiently navigates the non-convex landscape of center positions by leveraging the exact radius solution.
2. **Full Variable Polish**: The best center configuration from Phase 1 is passed to `scipy.optimize.minimize` with `SLSQP` to jointly optimize centers and radii. This handles constraint boundary interactions precisely and polishes the solution to machine tolerance.
3. **Robustness & Diversity**: We generate multiple initial layouts, apply controlled perturbations, and include a fallback deterministic grid. Constraint formulations use squared distances to avoid square roots in gradients, improving SLSQP convergence. The LP phase naturally respects boundary limits by capping individual radii with wall distances, while pairwise constraints are handled via an explicit upper-triangular constraint matrix.

This hybrid approach separates the combinatorial topology search (centers) from the exact sizing problem (radii), significantly reducing the chance of getting trapped in local minima compared to joint optimization alone.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_radii_lp(centers):
    """
    Given fixed centers, solves LP to maximize sum of radii subject to 
    boundary and pairwise non-overlap constraints.
    """
    n = centers.shape[0]
    
    # Distance to each wall
    b_wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Pairwise Euclidean distances
    idx = np.triu_indices(n, k=1)
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))[idx]
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    n_pairs = len(idx[0])
    A_ub = np.zeros((n_pairs, n))
    A_ub[np.arange(n_pairs), idx[0]] = 1.0
    A_ub[np.arange(n_pairs), idx[1]] = 1.0
    b_ub = dists
    
    # Variable bounds: 0 <= r_i <= distance_to_nearest_wall
    bounds = [(0.0, max(0.0, b_wall[i])) for i in range(n)]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback: conservative radii based on walls
    fallback_r = np.maximum(b_wall * 0.15, 0.0)
    return fallback_r, np.sum(fallback_r)

def obj_centers(centers_flat):
    """Objective for center optimization: negative max sum of radii via LP."""
    centers = np.clip(centers_flat.reshape(N, 2), 0.002, 0.998)
    _, s = solve_radii_lp(centers)
    return -s

def generate_initial_centers(seed):
    """Generates diverse initial center configurations."""
    np.random.seed(seed)
    init_type = seed % 3
    pts = []
    
    if init_type == 0:  # Hexagonal lattice
        r_est = 0.1
        y = r_est
        row = 0
        while len(pts) < N:
            x_off = (row % 2) * r_est
            x = r_est + x_off
            while x <= 1.0 - r_est and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
    elif init_type == 1:  # Dense Grid
        step = 0.18
        for i in range(6):
            for j in range(5):
                if len(pts) < N:
                    pts.append([0.05 + j*step, 0.05 + i*step])
    else:  # Concentric/Rings
        angle_step = 2 * np.pi / N
        for i in range(N):
            r = 0.2 + 0.15 * (i // 6)
            theta = i * angle_step
            pts.append([0.5 + r*np.cos(theta), 0.5 + r*np.sin(theta)])
            
    pts = np.array(pts[:N])
    # Add jitter to escape symmetry traps
    pts += np.random.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.01, 0.99)
    return pts.flatten()

def run_packing():
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    # Phase 1: Optimize centers using LP-based objective
    for seed in range(15):
        x0 = generate_initial_centers(seed)
        try:
            res = minimize(obj_centers, x0, method='Nelder-Mead', 
                           options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-8})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = res.x.reshape(N, 2)
        except Exception:
            continue
            
    # Phase 2: Compute exact optimal radii for best centers
    if best_centers is None:
        best_centers = generate_initial_centers(0).reshape(N, 2)
    best_radii, best_sum = solve_radii_lp(best_centers)
    
    # Phase 3: Joint polish with SLSQP
    def objective_full(vars_vec):
        return -np.sum(vars_vec[2::3])

    def constraints_full(vars_vec):
        x_c = vars_vec[0::3]
        y_c = vars_vec[1::3]
        r = vars_vec[2::3]
        c_list = []
        # Boundary constraints
        c_list.append(x_c - r)
        c_list.append(1.0 - x_c - r)
        c_list.append(y_c - r)
        c_list.append(1.0 - y_c - r)
        # Pairwise non-overlap: dist_sq >= (r_i + r_j)^2
        i_idx, j_idx = np.triu_indices(N, k=1)
        dx = x_c[i_idx] - x_c[j_idx]
        dy = y_c[i_idx] - y_c[j_idx]
        r_sum = r[i_idx] + r[j_idx]
        c_list.append(dx*dx + dy*dy - r_sum*r_sum)
        return np.concatenate(c_list)

    x0_full = np.zeros(3 * N)
    x0_full[0::3] = best_centers[:, 0]
    x0_full[1::3] = best_centers[:, 1]
    x0_full[2::3] = best_radii
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_full}
    
    try:
        # Shrink radii slightly to ensure strict interior start for SLSQP
        x0_start = x0_full.copy()
        x0_start[2::3] *= 0.96
        
        res_full = minimize(objective_full, x0_start, method='SLSQP', bounds=bounds,
                            constraints=cons, options={'maxiter': 4000, 'ftol': 1e-14})
        
        if res_full.success:
            # Verify feasibility within tolerance
            cons_val = constraints_full(res_full.x)
            if np.min(cons_val) >= -1e-8:
                best_centers = np.column_stack((res_full.x[0::3], res_full.x[1::3]))
                best_radii = res_full.x[2::3]
                best_sum = np.sum(best_radii)
    except Exception:
        pass
        
    # Final safety: ensure non-negative radii and clip centers if needed
    best_radii = np.maximum(best_radii, 0.0)
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    # Re-solve LP one last time on polished centers to guarantee max sum radii
    final_radii, final_sum = solve_radii_lp(best_centers)
    
    return best_centers, final_radii, float(final_sum)
