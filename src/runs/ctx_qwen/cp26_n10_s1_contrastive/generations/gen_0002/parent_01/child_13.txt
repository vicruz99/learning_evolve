def penalty_obj(x, lam):

    n = len(x) // 3
    centers = x[:2*n].reshape(n, 2)

    p = 0.0

    p += np.sum(np.maximum(0, radii - x_c)**2)
    p += np.sum(np.maximum(0, radii - (1.0 - x_c))**2)
    p += np.sum(np.maximum(0, radii - y_c)**2)
    p += np.sum(np.maximum(0, radii - (1.0 - y_c))**2)
