import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[2::3])

def get_constraints(vars_vec):
    """Compute all inequality constraints: boundary and separation."""
    C = vars_vec.reshape(N, 3)
    x = C[:, 0]
    y = C[:, 1]
    r = C[:, 2]
    
    c = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation constraints: dist_sq >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    r_sum = r[i_idx] + r[j_idx]
    c.append(dx*dx + dy*dy - r_sum*r_sum)
    
    return np.concatenate(c)

def get_safe_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        d_min = 2.0
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if d < d_min:
                    d_min = d
        # Scale down to ensure strict feasibility
        r[i] = 0.4 * min(d_wall, 0.5 * d_min)
    return r

def generate_initializations():
    """Generate a diverse set of initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice (5-6-5-6-4 structure)
    pts = []
    y = 0.05
    rows = [5, 6, 5, 6, 4]
    for r_idx, count in enumerate(rows):
        shift = (r_idx % 2) * 0.1
        x = 0.05 + shift
        for _ in range(count):
            pts.append([x, y])
            x += 0.2
        y += 0.17320508
    pts = np.array(pts[:N])
    inits.append(pts)
    
    # 2. 5x5 Grid + 1 center
    grid = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    grid = np.vstack([grid, [0.5, 0.5]])
    inits.append(grid[:N])
    
    # 3. Corner-focused layout
    corner_pts = np.array([[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]])
    rest = np.random.RandomState(42).rand(N-4, 2) * 0.6 + 0.2
    inits.append(np.vstack([corner_pts, rest]))
    
    # 4. Multiple random feasible layouts
    rng = np.random.RandomState(101)
    for _ in range(5):
        c = rng.rand(N, 2) * 0.8 + 0.1
        inits.append(c)
        
    return inits

def run_packing():
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    best_x = None
    best_sum = -np.inf
    
    inits = generate_initializations()
    
    # Phase 1: Broad search from diverse initializations
    for seed_val in range(35):
        np.random.seed(seed_val)
        base = inits[seed_val % len(inits)]
        
        # Perturb base layout to break symmetry
        pts = base + np.random.normal(0, 0.025, base.shape)
        pts = np.clip(pts, 0.02, 0.98)
        
        r0 = get_safe_radii(pts)
        
        x0 = np.zeros(3 * N)
        for i in range(N):
            x0[3*i] = pts[i, 0]
            x0[3*i+1] = pts[i, 1]
            x0[3*i+2] = r0[i]
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
            
            if res.success:
                cons_val = get_constraints(res.x)
                if np.min(cons_val) >= -1e-9:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Perturbation search around best to escape local minima
    if best_x is not None:
        for k in range(50):
            x_pert = best_x.copy()
            # Perturb centers and radii slightly
            x_pert[:2*N] += np.random.normal(0, 0.004, 2*N)
            x_pert[2*N:] += np.random.normal(0, 0.0015, N)
            
            # Clip to feasible bounds
            x_pert[:2*N] = np.clip(x_pert[:2*N], 0.01, 0.99)
            x_pert[2*N:] = np.clip(x_pert[2*N:], 1e-6, 0.49)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13})
                if res.success:
                    cons_val = get_constraints(res.x)
                    if np.min(cons_val) >= -1e-9:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = res.x.copy()
            except Exception:
                continue
                
    # Phase 3: High-precision polish on the absolute best configuration
    if best_x is not None:
        try:
            res_final = minimize(objective, best_x, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14})
            if res_final.success:
                cons_val = get_constraints(res_final.x)
                if np.min(cons_val) >= -1e-9:
                    best_x = res_final.x
                    best_sum = -res_final.fun
        except Exception:
            pass
            
    # Fallback valid configuration if optimization somehow fails
    if best_x is None:
        best_x = np.zeros(3*N)
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        pts = pts[:N]
        r_f = get_safe_radii(pts)
        for i in range(N):
            best_x[3*i] = pts[i, 0]
            best_x[3*i+1] = pts[i, 1]
            best_x[3*i+2] = r_f[i]
        best_sum = np.sum(r_f)
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
