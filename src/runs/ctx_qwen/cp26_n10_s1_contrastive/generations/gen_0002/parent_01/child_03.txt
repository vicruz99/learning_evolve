The primary bottleneck in previous solutions is the use of SLSQP on a formulation with quadratic non-overlap constraints, which creates a highly non-convex landscape prone to local minima. 

To break through the 2.636 barrier, we switch to a principled two-phase decomposition:
1. **Radius Optimization (LP):** For any fixed set of centers, the problem of maximizing $\sum r_i$ subject to boundary and separation constraints is a *Linear Program*. The constraints $r_i + r_j \leq d_{ij}$ are linear in $r$ when centers are fixed. This can be solved exactly and efficiently using `scipy.optimize.linprog` (HiGHS solver).
2. **Center Optimization (Derivative-Free):** We treat the optimal LP value as a function of the centers $f(\mathbf{C})$. While $f$ is piecewise linear and non-smooth, it is continuous and generally well-behaved. We maximize $f(\mathbf{C})$ using Powell's method, which excels at piecewise-smooth landscapes without requiring gradients.
3. **Initialization Diversity:** We generate multiple structured initial layouts (hexagonal lattices with varying densities, a perturbed square grid) and run local optimization from each. This ensures we explore different topological configurations and escape shallow local optima.

This approach bypasses the non-convex quadratic constraints entirely during the search, allowing the solver to navigate the configuration space more freely and pack circles more densely, particularly by allowing central circles to expand while edge circles shrink appropriately.
</strategy>

```python
import numpy as np
from scipy.optimize import linprog, minimize
import warnings

N_CIRCLES = 26

def compute_max_radii_sum(centers):
    """Solves the LP to find radii that maximize sum(r_i) for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    # Preallocate constraint matrices
    n_pairs = n * (n - 1) // 2
    n_constraints = 4 * n + n_pairs
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = [(0, None)] * n
    
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def objective_centers(x_flat):
    """Objective function for optimizing centers: negative max sum of radii."""
    centers = np.clip(x_flat.reshape(N_CIRCLES, 2), 1e-4, 1.0 - 1e-4)
    val, _ = compute_max_radii_sum(centers)
    return -val

def generate_hex_init(dx, dy):
    """Generates a hexagonal lattice initialization."""
    pts = []
    rows = [6, 5, 6, 5, 4]
    y = 0.05
    for r_idx, count in enumerate(rows):
        x_start = 0.05 + (r_idx % 2) * dx
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * dx
        y += dy
    return np.array(pts[:N_CIRCLES])

def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Diverse initial configurations
    inits = []
    # Hex lattices with varying densities
    inits.append(generate_hex_init(0.17, 0.1472))
    inits.append(generate_hex_init(0.185, 0.1602))
    inits.append(generate_hex_init(0.20, 0.1732))
    
    # Square grid + center
    grid_pts = []
    for i in range(5):
        for j in range(5):
            grid_pts.append([0.1 + 0.2 * i, 0.1 + 0.2 * j])
    grid_pts.append([0.5, 0.5])
    inits.append(np.array(grid_pts[:N_CIRCLES]))
    
    # Perturbed hexagonal to break symmetry
    np.random.seed(42)
    base_hex = generate_hex_init(0.18, 0.1558)
    for _ in range(12):
        pert = base_hex + np.random.normal(0, 0.025, base_hex.shape)
        pert = np.clip(pert, 0.02, 0.98)
        inits.append(pert)

    # Optimize centers using Powell's method
    for init_centers in inits:
        x0 = init_centers.flatten()
        try:
            res = minimize(objective_centers, x0, method='Powell',
                           options={'maxiter': 800, 'ftol': 1e-12, 'xrtol': 1e-10})
            
            # Evaluate best found
            curr_centers = np.clip(res.x.reshape(N_CIRCLES, 2), 1e-6, 1.0 - 1e-6)
            val, radii = compute_max_radii_sum(curr_centers)
            
            if val > best_sum:
                best_sum = val
                best_centers = curr_centers
                best_radii = radii
        except Exception:
            continue
            
    # Fallback if optimization fails completely
    if best_centers is None:
        best_centers = inits[0]
        _, best_radii = compute_max_radii_sum(best_centers)
        best_sum = np.sum(best_radii)
        
    # Ensure strict non-negativity against numerical drift
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_sum)
```
