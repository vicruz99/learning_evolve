

<strategy>
The problem of maximizing the sum of radii for N circles in a unit square can be decoupled into two parts:
1. **Optimize centers**: Find center coordinates that allow for large radii.
2. **Compute optimal radii for fixed centers**: This is a Linear Programming (LP) problem. Given fixed centers, maximizing $\sum r_i$ subject to $r_i \le \text{dist\_to\_wall}_i$ and $r_i + r_j \le \text{dist}(c_i, c_j)$ is convex and can be solved exactly and very quickly using `scipy.optimize.linprog`.

By treating the LP solution as an objective function of the centers, we can use derivative-free optimizers (like Powell or Nelder-Mead) to search for optimal center configurations. This avoids the non-convexity issues of joint $(x, y, r)$ optimization and guarantees feasibility at every step.

**Strategy:**
1. Define a function `solve_lp(centers)` that formulates and solves the LP to return the maximum sum of radii and the corresponding radii vector.
2. Define a wrapper `objective_wrapper` that takes flattened center coordinates, clips them to a safe margin inside $[0,1]^2$, and returns the negative LP sum (for minimization).
3. Generate diverse initial center configurations: hexagonal lattice, square grid + center, and multiple random starts.
4. Run local optimization (`Powell`) from each start to maximize the LP objective.
5. Perform perturbation-based refinement on the best found configuration to escape local minima.
6. Extract final centers and radii, ensuring strict feasibility for the validator.

This approach is computationally efficient (LP solves in milliseconds) and robust against the non-smooth landscape of the center optimization problem.
</strategy>

```python
import numpy as np
import scipy.optimize as opt

def solve_lp(centers):
    """Solves the LP to find radii maximizing sum(r_i) for fixed centers."""
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    c = -np.ones(n)  # Minimize -sum(r_i) <=> Maximize sum(r_i)
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    I = np.eye(n)
    A_ub = np.vstack([I, I, I, I])
    b_ub = np.concatenate([x, 1.0 - x, y, 1.0 - y])
    
    # Pairwise constraints: r_i + r_j <= dist(c_i, c_j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    n_pairs = len(idx_i)
    A_pair = np.zeros((n_pairs, n))
    A_pair[np.arange(n_pairs), idx_i] = 1.0
    A_pair[np.arange(n_pairs), idx_j] = 1.0
    
    A_ub = np.vstack([A_ub, A_pair])
    b_ub = np.concatenate([b_ub, dists[idx_i, idx_j]])
    bounds = [(0, None)] * n
    
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, res.x
    return -1e6, np.zeros(n)

def objective_wrapper(centers_flat):
    """Objective function for center optimization: maximizes LP sum of radii."""
    centers = np.clip(centers_flat.reshape(26, 2), 1e-4, 1.0 - 1e-4)
    val, _ = solve_lp(centers)
    return -val

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    
    inits = []
    # 1. Hexagonal lattice initialization
    r0 = 0.09
    pts = []
    y = r0
    row = 0
    while len(pts) < n:
        x = r0 + (row % 2) * r0
        while x <= 1.0 - r0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    inits.append(np.array(pts[:n]))
    
    # 2. 5x5 Grid + center initialization
    grid = np.array([[0.1 + i*0.2, 0.1 + j*0.2] for i in range(5) for j in range(5)])
    grid = np.vstack([grid, [0.5, 0.5]])
    inits.append(grid)
    
    # 3. Random initializations
    np.random.seed(42)
    for _ in range(4):
        inits.append(np.random.uniform(0.1, 0.9, (n, 2)))
        
    # Phase 1: Local optimization from diverse starts
    for c0 in inits:
        c0 = np.clip(c0, 0.01, 0.99)
        try:
            res = opt.minimize(objective_wrapper, c0.flatten(), method='Powell', 
                               options={'maxiter': 4000, 'ftol': 1e-10})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(n, 2)
        except Exception:
            continue
            
    # Phase 2: Perturbation refinement to escape local minima
    if best_centers is not None:
        for _ in range(15):
            c_pert = best_centers + np.random.uniform(-0.025, 0.025, (n, 2))
            c_pert = np.clip(c_pert, 0.01, 0.99)
            try:
                res = opt.minimize(objective_wrapper, c_pert.flatten(), method='Powell',
                                   options={'maxiter': 3000, 'ftol': 1e-10})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_centers = res.x.reshape(n, 2)
            except Exception:
                continue
                
    # Fallback if optimization somehow fails
    if best_centers is None:
        best_centers = np.random.uniform(0.2, 0.8, (n, 2))
        
    # Final LP solve to get consistent radii
    _, best_radii = solve_lp(best_centers)
    
    return best_centers, best_radii, float(best_sum)
```
