import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[2*N_CIRCLES:])

def constraints(vars_vec):
    """
    Computes inequality constraints g(vars) >= 0.
    Enforces boundary containment and pairwise non-overlap.
    """
    centers = vars_vec[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = vars_vec[2*N_CIRCLES:]
    
    c_list = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c_list.append(centers[:, 0] - radii)
    c_list.append(1.0 - centers[:, 0] - radii)
    c_list.append(centers[:, 1] - radii)
    c_list.append(1.0 - centers[:, 1] - radii)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = centers[:, 0, np.newaxis] - centers[:, 0]
    dy = centers[:, 1, np.newaxis] - centers[:, 1]
    dist_sq = dx**2 + dy**2
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    c_list.append(dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2)
    
    return np.concatenate(c_list)

def generate_hex_init(n, r_guess):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r_guess
    row = 0
    while len(pts) < n:
        x = r_guess + (row % 2) * r_guess
        while x <= 1.0 - r_guess and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r_guess
        y += np.sqrt(3.0) * r_guess
        row += 1
    return np.array(pts[:n])

def generate_grid_init(n, r_guess):
    """Generates a square grid initialization."""
    pts = []
    step = (1.0 - 2.0 * r_guess) / 4.0
    for i in range(5):
        for j in range(5):
            pts.append([r_guess + i*step, r_guess + j*step])
    if n > 25:
        pts.append([0.5, 0.5])
    return np.array(pts[:n])
