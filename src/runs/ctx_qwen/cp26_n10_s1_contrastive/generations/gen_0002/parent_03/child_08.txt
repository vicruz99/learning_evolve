Key insight: For any fixed set of centers, finding the radii that maximize the sum while satisfying non-overlap and boundary constraints is a **Linear Programming (LP)** problem. This is because the constraints $r_i + r_j \le d_{ij}$ and $r_i \le \min(x_i, 1-x_i, y_i, 1-y_i)$ are linear in $r_i$, and the objective $\sum r_i$ is linear. Strategy:
1. **Global Search on Centers**: Use `differential_evolution` to explore the 52-dimensional center space. The objective function evaluates the LP-optimal sum of radii for each candidate center configuration. This guarantees that at every step, we are evaluating the true maximum possible sum for those centers, avoiding the common pitfall of optimizers getting stuck with suboptimal radii.
2. **Local Refinement**: Apply `Nelder-Mead` on the best centers found by DE to fine-tune positions without needing gradients (since the LP active set changes discontinuously).
3. **Joint Polish**: Use `SLSQP` on the full $(x, y, r)$ variable space starting from the refined configuration to squeeze out any remaining numerical improvements and ensure strict constraint satisfaction.
4. This hybrid approach combines global exploration, exact subproblem solving, and local polishing, which is theoretically stronger than joint non-convex optimization alone.

import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def max_sum_radii_lp(centers):
    """Given fixed centers, solve LP to find radii maximizing sum(r_i) subject to constraints."""
    n = centers.shape[0]
    c = -np.ones(n)  # Minimize -sum(r_i) <=> Maximize sum(r_i)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(ub)
        
    # Pairwise non-overlap constraints: r_i + r_j <= distance(i, j)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def objective_centers(centers_flat):
    """Objective for DE/Nelder-Mead: negative of optimal radii sum for given centers."""
    centers = centers_flat.reshape(26, 2)
    s, _ = max_sum_radii_lp(centers)
    return -s

def objective_full(vars_vec):
    """Objective for SLSQP polish: negative sum of radii."""
    n = 26
    r = vars_vec[2*n:]
    return -np.sum(r)

def constraint_full(vars_vec):
    """Inequality constraints >= 0 for SLSQP polish."""
    n = 26
    x = vars_vec[0:n]
    y = vars_vec[n:2*n]
    r = vars_vec[2*n:]
    
    c = []
    # Boundary containment
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation: dist^2 >= (r_i + r_j)^2
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    r_sum_sq = (r[:, np.newaxis] + r[np.newaxis, :])**2
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    c.append(dist_sq[i_idx, j_idx] - r_sum_sq[i_idx, j_idx])
    
    return np.concatenate(c)

def run_packing():
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n)
    best_sum = -np.inf
    best_centers = None
    
    # Phase 1: Global search on centers using Differential Evolution
    # DE robustly navigates the non-convex landscape by evaluating the exact LP-optimal radii at each step.
    for seed in range(5):
        try:
            res = differential_evolution(
                objective_centers, bounds, seed=seed, maxiter=500, popsize=20,
                mutation=(0.5, 1.0), recombination=0.7, tol=1e-7, updating='deferred', polish=False
            )
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = res.x.reshape(n, 2)
        except Exception:
            pass
            
    # Phase 2: Local refinement on centers using derivative-free Nelder-Mead
    if best_centers is not None:
        res_nm = minimize(objective_centers, best_centers.flatten(), method='Nelder-Mead',
                          options={'maxiter': 5000, 'xatol': 1e-7, 'fatol': 1e-7})
        if -res_nm.fun > best_sum:
            best_sum = -res_nm.fun
            best_centers = res_nm.x.reshape(n, 2)
            
    # Compute exact optimal radii for the refined centers
    final_sum, final_radii = max_sum_radii_lp(best_centers)
    
    # Phase 3: Joint SLSQP polish on (x, y, r) to ensure strict feasibility and squeeze last bits
    x0 = np.concatenate([best_centers.flatten(), final_radii])
    x0[2*n:] *= 0.99  # Slight shrinkage guarantees strict interior start for SLSQP
    
    try:
        res_sl = minimize(objective_full, x0, method='SLSQP',
                          bounds=[(0,1)]*(2*n) + [(0, 0.5)]*n,
                          constraints={'type': 'ineq', 'fun': constraint_full},
                          options={'maxiter': 2000, 'ftol': 1e-12})
        if res_sl.success and np.min(constraint_full(res_sl.x)) >= -1e-7:
            sl_sum = -res_sl.fun
            if sl_sum > final_sum:
                final_sum = sl_sum
                best_centers = res_sl.x[:2*n].reshape(n, 2)
                final_radii = res_sl.x[2*n:]
    except Exception:
        pass
        
    return best_centers, final_radii, float(final_sum)
