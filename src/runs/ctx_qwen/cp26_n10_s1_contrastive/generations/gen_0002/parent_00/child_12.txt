import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[2*N:])

def constraints(vars_vec):
    """
    Constraint function: ensures circles are inside the unit square and do not overlap.
    Returns a 1D array of constraint values that must be >= 0.
    """
    centers = vars_vec[:2*N].reshape(N, 2)
    radii = vars_vec[2*N:]
    
    c = []
    # Boundary constraints
    c.append(centers[:, 0] - radii)
    c.append(1.0 - centers[:, 0] - radii)
    c.append(centers[:, 1] - radii)
    c.append(1.0 - centers[:, 1] - radii)
    
    # Pairwise non-overlap constraints
    dx = centers[:, 0, np.newaxis] - centers[:, 0]
    dy = centers[:, 1, np.newaxis] - centers[:, 1]
    dist_sq = dx**2 + dy**2
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    c.append(dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def generate_init(seed, strategy='hex'):
    """Generates a feasible initial configuration based on a given strategy."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    
    if strategy == 'hex':
        r_est = 0.105
        y = r_est
        row = 0
        idx = 0
        while y < 1.0 - r_est and idx < N:
            x_start = r_est if row % 2 == 0 else 2.0 * r_est
            x = x_start
            while x < 1.0 - r_est and idx < N:
                pts[idx] = [x, y]
                idx += 1
                x += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
        if idx < N:
            pts[idx:] = np.random.rand(N - idx, 2) * 0.8 + 0.1
    elif strategy == 'grid':
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts[25] = [0.5, 0.5]
    else:
        pts = np.random.rand(N, 2)
        
    # Perturb to break symmetry and escape degenerate starts
    pts += np.random.uniform(-0.04, 0.04, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    # Compute strictly feasible initial radii
    radii = np.zeros(N)
    for i in range(N):
        d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        d_min = 2.0
        for j in range(N):
            if i != j:
                d = np.sqrt(np.sum((pts[i]-pts[j])**2))
                if d < d_min: 
                    d_min = d
        radii[i] = 0.45 * min(d_wall, 0.5 * d_min)
        
    return np.concatenate([pts.flatten(), radii])

def run_packing():
    """Main function to solve the circle packing problem."""
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_vars = None
    best_sum = -1.0
    
    inits = []
    # Generate diverse initializations to cover different topological basins
    for seed in range(40):
        strat = 'hex' if seed % 3 == 0 else ('grid' if seed % 3 == 1 else 'rand')
        inits.append(generate_init(seed, strat))
        
    # Phase 1: Broad search across multiple initializations
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
            if np.min(constraints(res.x)) >= -1e-8:
                curr_sum = np.sum(res.x[2*N:])
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Local refinement via perturbation to escape local minima
    if best_vars is not None:
        for _ in range(30):
            x_pert = best_vars + np.random.randn(3*N) * 0.005
            x_pert[:2*N] = np.clip(x_pert[:2*N], 0.01, 0.99)
            x_pert[2*N:] = np.clip(x_pert[2*N:], 1e-5, 0.49)
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13})
                if np.min(constraints(res.x)) >= -1e-8:
                    curr_sum = np.sum(res.x[2*N:])
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
            except Exception:
                continue
                
    # Fallback if optimization fails completely (should not happen)
    if best_vars is None:
        best_vars = inits[0]
        
    # Phase 3: High-precision polish on the best configuration found
    res_final = minimize(objective, best_vars, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14})
    if np.min(constraints(res_final.x)) >= -1e-8:
        best_vars = res_final.x
        
    centers = best_vars[:2*N].reshape(N, 2)
    radii = best_vars[2*N:]
    # Ensure non-negative radii against numerical drift
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
