import numpy as np
from scipy.optimize import minimize

N = 26
i_idx, j_idx = np.triu_indices(N, k=1)

def objective(vars_vec):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(vars_vec[2*N:])

def constraints(vars_vec):
    """
    Constraint function: ensures circles are inside the unit square and do not overlap.
    Returns a 1D array of constraint values that must be >= 0.
    """
    centers = vars_vec[:2*N].reshape(N, 2)
    radii = vars_vec[2*N:]
    
    c = []
    # Boundary constraints: circle inside [0,1]^2
    c.append(centers[:, 0] - radii)
    c.append(1.0 - centers[:, 0] - radii)
    c.append(centers[:, 1] - radii)
    c.append(1.0 - centers[:, 1] - radii)
    
    # Pairwise non-overlap constraints: dist_sq >= (r_i + r_j)^2
    dx = centers[:, 0, np.newaxis] - centers[:, 0]
    dy = centers[:, 1, np.newaxis] - centers[:, 1]
    dist_sq = dx**2 + dy**2
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    c.append(dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx]**2)
    
    return np.concatenate(c)

def get_feasible_init(row_counts, r_est, seed, perturb_scale):
    """Generates a strictly feasible initial configuration from a hexagonal lattice."""
    np.random.seed(seed)
    pts = []
    y = r_est
    for k, cnt in enumerate(row_counts):
        x_start = r_est if k % 2 == 0 else 2.0 * r_est
        for m in range(cnt):
            pts.append([x_start + m * 2.0 * r_est, y])
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    if perturb_scale > 0:
        pts += np.random.normal(0, perturb_scale, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    
    # Compute safe radii to guarantee strict feasibility
    dist_b = np.minimum(np.minimum(pts[:, 0], 1.0 - pts[:, 0]), 
                        np.minimum(pts[:, 1], 1.0 - pts[:, 1]))
    diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    # Use 85% of the limiting distance to stay strictly inside feasible region
    radii = np.minimum(dist_b, 0.5 * min_dists) * 0.85
    return np.concatenate([pts.flatten(), radii])

def run_packing():
    # Variable bounds: x, y in [0, 1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_vars = None
    best_sum = -1.0
    
    inits = []
    # Multiple hexagonal row count patterns summing to 26
    row_counts_options = [[6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4]]

    for rc in row_counts_options:
        for seed in range(5):
            for p_scale in [0.0, 0.03]:
                inits.append(get_feasible_init(rc, 0.09, seed, p_scale))
                
    # Add random feasible initializations for diversity
    for seed in range(10):
        np.random.seed(seed + 200)
        pts = np.random.rand(N, 2) * 0.8 + 0.1
        dist_b = np.minimum(np.minimum(pts[:, 0], 1.0 - pts[:, 0]), 
                            np.minimum(pts[:, 1], 1.0 - pts[:, 1]))
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        min_dists = np.min(dists, axis=1)
        radii = np.minimum(dist_b, 0.5 * min_dists) * 0.5
        inits.append(np.concatenate([pts.flatten(), radii]))
        
    # Phase 1: Broad optimization search
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if np.min(constraints(res.x)) >= -1e-8:
                curr_sum = np.sum(res.x[2*N:])
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization fails completely
    if best_vars is None:
        best_vars = get_feasible_init([6, 5, 6, 5, 4], 0.09, 0, 0.0)
        best_sum = np.sum(best_vars[2*N:])

    radii = np.maximum(radii, 0.0)
    centers = np.clip(centers, radii, 1.0 - radii)
    
    return centers, radii, float(np.sum(radii))
