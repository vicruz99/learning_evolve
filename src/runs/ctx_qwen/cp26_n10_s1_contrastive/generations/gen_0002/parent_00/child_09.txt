import numpy as np
from scipy.optimize import minimize, linprog


N = 26
i_idx, j_idx = np.triu_indices(N, k=1)


def compute_pairwise_distances(centers):
    """Compute pairwise distances between centers."""
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    return dists


def get_max_radii_sum_lp(centers):
    """Given fixed centers, solve LP to find radii maximizing sum."""
    n = centers.shape[0]

    # Distance to boundaries
    dist_bound = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )

    # Pairwise distances
    dists = compute_pairwise_distances(centers)
    np.fill_diagonal(dists, np.inf)

    # LP: maximize sum(r_i)
    c_obj = -np.ones(n)
    A_ub_list = []
    b_ub_list = []

    # Boundary constraints: r_i <= dist_bound[i]
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(dist_bound[i])

    # Pairwise constraints: r_i + r_j <= dists[i,j]
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dists[i, j])

    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)


def objective_for_centers(centers_flat):
    """Objective for center optimization: negative of max sum of radii."""
    centers = centers_flat.reshape(N, 2)
    _, radii = get_max_radii_sum_lp(centers)
    return -np.sum(radii)


def get_constraints_full(vars_vec):
    """Constraints for SLSQP: boundary and non-overlap."""
    centers = vars_vec[:2 * N].reshape(N, 2)
    radii = vars_vec[2 * N:]

    c = []
    # Boundary constraints
    c.append(centers[:, 0] - radii)
    c.append(1.0 - centers[:, 0] - radii)
    c.append(centers[:, 1] - radii)
    c.append(1.0 - centers[:, 1] - radii)

    # Pairwise non-overlap: dist_sq >= (r_i + r_j)^2
    dx = centers[:, 0, np.newaxis] - centers[:, 0]
    dy = centers[:, 1, np.newaxis] - centers[:, 1]
    dist_sq = dx ** 2 + dy ** 2

    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    c.append(dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx] ** 2)

    return np.concatenate(c)


def make_hex_layout(r_est, row_counts):
    """Generate hexagonal layout with specified row counts."""
    pts = []
    y = r_est
    for k, cnt in enumerate(row_counts):
        x_start = r_est if k % 2 == 0 else 2.0 * r_est
        for m in range(cnt):
            pts.append([x_start + m * 2.0 * r_est, y])
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    # Normalize to fit in [0.1, 0.9] roughly
    mn = pts.min(axis=0)
    mx = pts.max(axis=0)
    scale = 0.75 / (mx - mn)
    pts_norm = (pts - mn) * scale + (1.0 - 0.75) / 2.0
    return pts_norm


def make_grid_layout():
    """Generate 5x5 grid + center layout."""
    g = []
    for i in range(5):
        for j in range(5):
            g.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    g.append([0.5, 0.5])
    return np.array(g)


def make_corner_layout():
    """Layout that tries to use corners effectively."""
    pts = []
    # 4 corners
    corner_r = 0.08
    for cx, cy in [(corner_r, corner_r), (1 - corner_r, corner_r),
                    (corner_r, 1 - corner_r), (1 - corner_r, 1 - corner_r)]:
        pts.append([cx, cy])

    # Fill remaining with hex pattern
    r_est = 0.075
    y = r_est + 0.05
    row = 0
    while len(pts) < N:
        x_start = r_est + 0.05 if row % 2 == 0 else 2.0 * r_est + 0.05
        x = x_start
        while x <= 1.0 - r_est - 0.05 and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    return np.array(pts[:N])


def make_staggered_rows(row_counts, margin=0.06):
    """Generate staggered rows with specified counts."""
    pts = []
    width = 1.0 - 2 * margin
    height = 1.0 - 2 * margin
    n_rows = len(row_counts)
    dy = height / n_rows
    y = margin

    for k, cnt in enumerate(row_counts):
        if cnt > 0:
            dx = width / (cnt + 1) if cnt > 1 else 0
            x_start = margin + dx / 2 if cnt % 2 == 0 else margin + dx
            for m in range(cnt):
                pts.append([x_start + m * dx, y])
        y += dy
    return np.array(pts[:N])


def get_safe_radii(centers):
    """Compute safe initial radii for given centers."""
    n = centers.shape[0]
    dist_bound = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    dists = compute_pairwise_distances(centers)
    np.fill_diagonal(dists, np.inf)
    min_dist = np.min(dists, axis=1)
    return np.minimum(dist_bound, 0.5 * min_dist) * 0.85


def try_optimization(x0, maxiter_nm=2000, maxiter_slsqp=3000):
    """Run two-phase optimization from given initial point."""
    centers0 = x0[:2 * N].reshape(N, 2)

    # Phase 1: Nelder-Mead on centers
    try:
        res_nm = minimize(objective_for_centers, centers0.flatten(),
                          method='Nelder-Mead',
                          options={'maxiter': maxiter_nm, 'xatol': 1e-8, 'fatol': 1e-10})
        centers_opt = res_nm.x.reshape(N, 2)
        centers_opt = np.clip(centers_opt, 0.001, 0.999)
    except Exception:
        centers_opt = centers0

    # Get radii for these centers
    _, radii_opt = get_max_radii_sum_lp(centers_opt)
    radii_opt = np.maximum(radii_opt, 1e-6)

    # Prepare full vars for SLSQP
    full_vars = np.zeros(3 * N)
    full_vars[:2 * N] = centers_opt.flatten()
    full_vars[2 * N:] = radii_opt * 0.98  # Slightly shrink for feasibility margin

    # Phase 2: SLSQP refinement
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': get_constraints_full}

    try:
        res_slsqp = minimize(
            lambda v: -np.sum(v[2 * N:]),
            full_vars,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter_slsqp, 'ftol': 1e-14, 'disp': False}
        )
        if np.min(get_constraints_full(res_slsqp.x)) >= -1e-9:
            return res_slsqp.x, -res_slsqp.fun
    except Exception:
        pass

    # Fallback to LP radii
    _, radii_fallback = get_max_radii_sum_lp(centers_opt)
    full_fallback = np.zeros(3 * N)
    full_fallback[:2 * N] = centers_opt.flatten()
    full_fallback[2 * N:] = radii_fallback
    return full_fallback, np.sum(radii_fallback)
