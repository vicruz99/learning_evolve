import numpy as np
from scipy.optimize import minimize

N = 26
i_idx, j_idx = np.triu_indices(N, k=1)

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2*N:])

def constraints(vars):
    """
    Computes inequality constraints g(vars) >= 0.
    Ensures circles are inside [0,1]^2 and do not overlap.
    """
    c = vars[:2*N].reshape(N, 2)
    r = vars[2*N:]
    con = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap: dist_sq >= (r_i + r_j)^2
    dx = c[:, 0, np.newaxis] - c[:, 0]
    dy = c[:, 1, np.newaxis] - c[:, 1]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    con.append(d2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    
    return np.concatenate(con)

def get_init(scale=1.0, pert=0.0, seed=0):
    """Generates a strictly feasible initial configuration based on a hexagonal lattice."""
    np.random.seed(seed)
    rows = [6, 5, 6, 5, 4]
    pts = []
    r_est = 0.09 * scale
    y = r_est
    for k, cnt in enumerate(rows):
        x_start = r_est if k % 2 == 0 else 2.0 * r_est
        for m in range(cnt):
            pts.append([x_start + m * 2.0 * r_est, y])
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    
    if pert > 0:
        pts += np.random.normal(0, pert, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    
    # Compute safe initial radii
    r_safe = np.full(N, 0.01)
    for i in range(N):
        d_wall = min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])
        min_d = d_wall
        for j in range(N):
            if i != j:
                d = np.sqrt(np.sum((pts[i] - pts[j])**2))
                if d * 0.5 < min_d:
                    min_d = d * 0.5
        r_safe[i] = min_d * 0.85
        
    vars0 = np.zeros(3 * N)
    for i in range(N):
        vars0[3*i] = pts[i, 0]
        vars0[3*i+1] = pts[i, 1]
        vars0[3*i+2] = r_safe[i]
    return vars0

def run_packing():
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_vars = None
    best_sum = -1.0
    
    # Phase 1: Diverse structured initializations
    inits = []
    for s in range(15):
        scale = 0.8 + 0.4 * (s % 5) / 4.0
        pert = 0.015 * (s // 5)
        inits.append(get_init(scale, pert, seed=s))
        
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
            if np.min(constraints(res.x)) >= -1e-7:
                curr_sum = np.sum(res.x[2*N:])
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Basin hopping to escape local minima
    if best_vars is not None:
        for _ in range(25):
            x_pert = best_vars.copy()
            x_pert[:2*N] += np.random.randn(2*N) * 0.003
            x_pert[2*N:] += np.random.randn(N) * 0.001
            x_pert[:2*N] = np.clip(x_pert[:2*N], 0.01, 0.99)
            x_pert[2*N:] = np.clip(x_pert[2*N:], 1e-6, 0.49)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 2000, 'ftol': 1e-13})
                if np.min(constraints(res.x)) >= -1e-7:
                    curr_sum = np.sum(res.x[2*N:])
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
            except Exception:
                continue
                
    # Phase 3: High-precision final polish
    if best_vars is not None:
        try:
            res_final = minimize(objective, best_vars, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 6000, 'ftol': 1e-15})
            if np.min(constraints(res_final.x)) >= -1e-8:
                best_vars = res_final.x
        except Exception:
            pass
            
    # Fallback
    if best_vars is None:
        best_vars = get_init()
        
    centers = best_vars[:2*N].reshape(N, 2)
    radii = best_vars[2*N:]
    return centers, radii, float(np.sum(radii))
