import numpy as np
from scipy.optimize import minimize, linprog

N = 26
n_pairs = N * (N - 1) // 2
n_constraints = 4 * N + n_pairs

# Precompute constant parts of the LP constraint matrix
# Structure: [x >= r, 1-x >= r, y >= r, 1-y >= r, dist_ij >= r_i + r_j]
A_ub = np.zeros((n_constraints, N))
c_obj = -np.ones(N)

# Boundary constraints: 1*r_i <= bound
for i in range(N):
    A_ub[i, i] = 1.0
    A_ub[N + i, i] = 1.0
    A_ub[2 * N + i, i] = 1.0
    A_ub[3 * N + i, i] = 1.0

# Pairwise constraints: 1*r_i + 1*r_j <= dist_ij
idx = 4 * N
for i in range(N):
    for j in range(i + 1, N):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        idx += 1

def solve_lp(centers):
    """Given centers, solve LP to find max radii and their sum."""
    centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
    b_ub = np.empty(n_constraints)
    b_ub[:N] = centers[:, 0]
    b_ub[N:2*N] = 1.0 - centers[:, 0]
    b_ub[2*N:3*N] = centers[:, 1]
    b_ub[3*N:4*N] = 1.0 - centers[:, 1]
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    iu, ju = np.triu_indices(N, k=1)
    b_ub[4*N:] = dists[iu, ju]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def lp_objective(centers_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    s, _ = solve_lp(centers)
    return -s

def slsqp_objective(vars):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def slsqp_constraints(vars):
    """Inequality constraints for SLSQP: g(vars) >= 0."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(dist2[mask] - rs[mask]**2)
    return np.concatenate(c)

def run_packing():

    best_radii = None

    inits = []

    rows = [6, 5, 6, 5, 4]
    pts = []
    r_est = 0.095
    y = r_est
    for k, cnt in enumerate(rows):
        x_start = r_est if k % 2 == 0 else 2 * r_est
        for m in range(cnt):
            pts.append([x_start + m * 2 * r_est, y])
        y += np.sqrt(3) * r_est
    inits.append(np.array(pts[:N]).flatten())
    
    # Grid 5x5 + center
    g = []
    for i in range(5):
        for j in range(5):
            g.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    g.append([0.5, 0.5])
    inits.append(np.array(g).flatten())
    
    # Random perturbations to escape symmetry
    np.random.seed(42)
    for base in inits[:2]:
        for _ in range(10):
            p = base.reshape(N, 2) + np.random.randn(N, 2) * 0.035
            inits.append(np.clip(p, 0.05, 0.95).flatten())

    # Optimize centers using Nelder-Mead with LP objective
    for x0 in inits:
        res = minimize(lp_objective, x0, method='Nelder-Mead',
                       options={'maxiter': 1200, 'xatol': 1e-7, 'fatol': 1e-9})
        s = -res.fun

            best_sum = s
            best_centers = res.x.reshape(N, 2)

    # Extract optimal radii for the best centers

    vars0[0::3] = best_centers[:, 0]

        best_radii = np.full(N, 0.01)
        best_sum = np.sum(best_radii)

    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
