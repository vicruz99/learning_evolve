import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """
    Computes inequality constraints g(x) >= 0.
    Includes boundary containment and pairwise non-overlap.
    """
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(xc - r)
    c.append(1.0 - xc - r)
    c.append(yc - r)
    c.append(1.0 - yc - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    r_sum_sq = r_sum**2
    
    # Only upper triangular pairs (i < j) to avoid duplicates
    c.append(dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX])
    
    return np.concatenate(c)

def compute_safe_radii(centers):
    """Compute strictly feasible radii for given centers."""
    n = centers.shape[0]
    # Distance to walls
    d_wall = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    d_min = np.min(dists, axis=1)
    
    # Scale down to guarantee strict feasibility for optimizer start
    r = np.minimum(d_wall, 0.5 * d_min) * 0.35
    return np.maximum(r, 1e-5)

def make_hex_init(seed, r_est, shift_x, shift_y):
    """Generates a hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = r_est + shift_y
    row = 0

        x = x_start

            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    # Fill any remaining slots randomly if pattern ran out

        centers[idx] = np.random.uniform(0.15, 0.85, 2)
        idx += 1

    centers = np.clip(centers, 0.05, 0.95)

    x0 = np.zeros(3 * N)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = r
    return x0

def make_grid_init(seed):
    """Generates a perturbed square grid initialization."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    idx = 0
    for i in range(5):
        for j in range(5):

    return x0
