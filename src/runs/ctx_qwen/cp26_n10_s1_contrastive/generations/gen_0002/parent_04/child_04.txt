import numpy as np
from scipy.optimize import minimize

N = 26
# Precompute pairwise indices for efficiency
i_idx, j_idx = np.triu_indices(N, k=1)

def compute_objective(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def compute_constraints(x):
    """
    Computes inequality constraints g(x) >= 0.
    Includes boundary containment and pairwise non-overlap.
    """
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(xc - r)
    c.append(1.0 - xc - r)
    c.append(yc - r)
    c.append(1.0 - yc - r)

    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    # Only upper triangular pairs (i < j)
    c.append(dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx]**2)

    return np.concatenate(c)

def make_init_hex(row_counts, seed, perturb=0.02):
    """Generates a hexagonal lattice initialization with controlled jitter."""
    np.random.seed(seed)
    n = 26
    r_est = 0.095
    dy = np.sqrt(3.0) * r_est
    pts = np.zeros((n, 2))
    y = r_est
    idx = 0
    for r_idx, cnt in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if idx < n:
                pts[idx] = [x, y]
                idx += 1
            x += 2.0 * r_est
        y += dy
        
    pts += np.random.uniform(-perturb, perturb, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    # Compute strictly feasible initial radii
    r0 = np.zeros(n)
    for i in range(n):
        dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        dm = 1.0
        for j in range(n):
            if i != j:
                d = np.hypot(pts[i,0]-pts[j,0], pts[i,1]-pts[j,1])
                if d < dm: dm = d
        r0[i] = min(dw, dm/2.0) * 0.65
        
    x0 = np.zeros(3*n)
    x0[0::3] = pts[:, 0]
    x0[1::3] = pts[:, 1]
    x0[2::3] = r0
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_x = None
    best_sum = -np.inf
    
    inits = []
    # Diverse hexagonal patterns to explore different boundary interactions
    hex_patterns = [[6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,6,4]]
    for pat in hex_patterns:
        for s in range(8):
            inits.append(make_init_hex(pat, s, perturb=0.03))
            
    # Grid-based initializations
    base_grid = np.array([[0.1+0.2*i, 0.1+0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
    for s in range(8):
        np.random.seed(s+50)
        gp = base_grid + np.random.uniform(-0.02, 0.02, (26, 2))
        gp = np.clip(gp, 0.05, 0.95)
        r0 = np.zeros(26)
        for i in range(26):
            dw = min(gp[i,0], 1-gp[i,0], gp[i,1], 1-gp[i,1])
            dm = 1.0
            for j in range(26):
                if i != j:
                    d = np.hypot(gp[i,0]-gp[j,0], gp[i,1]-gp[j,1])
                    if d < dm: dm = d
            r0[i] = min(dw, dm/2.0)*0.65
        x0 = np.zeros(78)
        x0[0::3] = gp[:,0]
        x0[1::3] = gp[:,1]
        x0[2::3] = r0
        inits.append(x0)
        
    # Random initializations for topology diversity
    for s in range(15):
        np.random.seed(s+100)
        pts = np.random.rand(26, 2) * 0.8 + 0.1
        r0 = np.zeros(26)
        for i in range(26):
            dw = min(pts[i,0], 1-pts[i,0], pts[i,1], 1-pts[i,1])
            dm = 1.0
            for j in range(26):
                if i != j:
                    d = np.hypot(pts[i,0]-pts[j,0], pts[i,1]-pts[j,1])
                    if d < dm: dm = d
            r0[i] = min(dw, dm/2.0)*0.5
        x0 = np.zeros(78)
        x0[0::3] = pts[:,0]
        x0[1::3] = pts[:,1]
        x0[2::3] = r0
        inits.append(x0)
        
    # Stage 1: Broad search
    for x0 in inits:
        try:
            res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                cv = compute_constraints(res.x)
                if np.min(cv) >= -1e-7:
                    s_val = np.sum(res.x[2::3])
                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = res.x.copy()
        except Exception:
            continue
            
    # Stage 2: Local perturbation refinement
    if best_x is not None:
        for k in range(25):
            np.random.seed(k+999)
            x_pert = best_x + np.random.uniform(-0.003, 0.003, best_x.shape)
            x_pert[0::3] = np.clip(x_pert[0::3], 0.01, 0.99)
            x_pert[1::3] = np.clip(x_pert[1::3], 0.01, 0.99)
            x_pert[2::3] = np.clip(x_pert[2::3], 1e-6, 0.49)
            try:
                res = minimize(compute_objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
                if res.success:
                    cv = compute_constraints(res.x)
                    if np.min(cv) >= -1e-7:
                        s_val = np.sum(res.x[2::3])
                        if s_val > best_sum:
                            best_sum = s_val
                            best_x = res.x.copy()
            except Exception:
                continue
                
    # Stage 3: High-precision final polish
    if best_x is not None:
        res = minimize(compute_objective, best_x, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
        if res.success:
            cv = compute_constraints(res.x)
            if np.min(cv) >= -1e-8:
                best_x = res.x
                best_sum = np.sum(res.x[2::3])
                
    # Fallback guarantee
    if best_x is None:
        centers = np.zeros((N, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                centers[idx] = [0.1 + 0.2*i, 0.1 + 0.2*j]
                idx += 1
        centers[25] = [0.5, 0.5]
        radii = np.full(N, 0.09)
        return centers, radii, float(np.sum(radii))

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    return centers, radii, float(np.sum(radii))
