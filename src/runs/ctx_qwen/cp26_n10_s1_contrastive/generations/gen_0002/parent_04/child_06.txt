import numpy as np
from scipy.optimize import minimize, differential_evolution

def objective(x):
    """Objective function: minimize negative sum of radii."""
    n = len(x) // 3
    return -np.sum(x[2::3])

def constraints(x):
    """
    Computes inequality constraints g(x) >= 0.
    Includes boundary containment and pairwise non-overlap.
    """
    n = len(x) // 3
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(xc - r)
    c.append(1.0 - xc - r)
    c.append(yc - r)
    c.append(1.0 - yc - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    c.append(dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx]**2)
    
    return np.concatenate(c)

def penalty_objective(x):
    """Objective with heavy penalty for constraint violations, used for DE."""
    n = len(x) // 3
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    
    pen = 0.0
    # Boundary penalties
    pen += np.sum(np.maximum(0, r - xc)**2)
    pen += np.sum(np.maximum(0, r + xc - 1.0)**2)
    pen += np.sum(np.maximum(0, r - yc)**2)
    pen += np.sum(np.maximum(0, r + yc - 1.0)**2)
    
    # Overlap penalties
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    dist = np.sqrt(dx**2 + dy**2 + 1e-15)
    r_sum = r[:, None] + r[None, :]
    
    viol = np.maximum(0, r_sum - dist)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pen += np.sum(viol[mask]**2)
    
    # High penalty weight ensures strict preference for feasibility
    return -np.sum(r) + 1e6 * pen

def generate_hex(n, seed):
    """Generates a hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    r_est = 0.095
    y = r_est
    row = 0
    idx = 0
    while y < 1.0 - r_est + 0.02 and idx < n:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        x = x_start
        while x < 1.0 - r_est + 0.02 and idx < n:
            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    # Controlled jitter to break symmetry
    centers += np.random.uniform(-0.012, 0.012, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    # Compute strictly feasible initial radii
    min_d = 1.0
    for i in range(n):
        d_bound = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        min_d = min(min_d, d_bound)
        for j in range(i+1, n):
            d_pair = np.linalg.norm(centers[i] - centers[j])
            min_d = min(min_d, d_pair)
            
    r0 = min_d * 0.38
    
    vars_init = np.zeros(3 * n)
    vars_init[0::3] = centers[:, 0]
    vars_init[1::3] = centers[:, 1]
    vars_init[2::3] = r0
    return vars_init
