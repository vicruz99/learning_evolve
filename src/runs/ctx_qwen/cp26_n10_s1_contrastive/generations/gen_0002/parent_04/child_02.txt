To improve the packing beyond the ~2.624 sum of radii, we need a more robust global optimization strategy that effectively separates the geometric spreading of centers from the radial sizing. 

Key improvements:
1. **Linear Programming for Radii**: For any fixed set of centers, finding the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints is a Linear Program. This can be solved extremely fast and exactly using `scipy.optimize.linprog` (method 'highs'). This converts the problem into optimizing center coordinates to maximize the LP objective.
2. **Force-Directed Pre-layout**: Start with a hexagonal lattice and apply a repulsive force simulation to spread circles evenly and push them against boundaries, providing a high-quality starting point.
3. **Differential Evolution on Centers**: Use `differential_evolution` to search the 52-dimensional center space. The objective function evaluates the negative sum of optimal radii via the LP. DE handles the non-convex landscape well and avoids local minima that plague gradient-based methods on this problem.
4. **SLSQP Joint Refinement**: After DE finds optimal centers and radii, run `SLSQP` on the full 78-dimensional space (centers + radii) with explicit inequality constraints. This polishes the solution to machine precision and ensures strict feasibility within the validator's tolerance.
5. **Vectorization & Precomputation**: Precompute the constant part of the LP constraint matrix to drastically speed up LP evaluations during DE. Vectorize force calculations to keep the simulation fast.

This hybrid approach (Force Layout -> DE on Centers with LP objective -> SLSQP refinement) is highly effective for circle packing and typically pushes past local optima found by direct constrained optimization.

import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def compute_lp(centers, A_ub_const):
    """Solves LP to find max sum of radii for fixed centers."""
    n = centers.shape[0]
    # Boundary limits for each circle
    b_lim = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))

    # Pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.einsum('ijk,ijk->ij', diff, diff))
    np.fill_diagonal(dists, np.inf)

    # Fill right-hand side of LP constraints
    m = A_ub_const.shape[0]
    b_ub = np.zeros(m)
    b_ub[:n] = b_lim

    idx = n
    for i in range(n):
        for j in range(i + 1, n):
            b_ub[idx] = dists[i, j]
            idx += 1

    c_obj = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    try:
        res = linprog(c_obj, A_ub=A_ub_const, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.fun * -1.0, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def neg_sum_radii(centers_flat, A_ub_const):
    """Objective for DE: minimize negative sum of radii."""
    n = 26
    centers = centers_flat.reshape(n, 2)
    val, _ = compute_lp(centers, A_ub_const)
    return -val

def objective_full(x):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_full(x):
    """Constraints for SLSQP: boundary and pairwise non-overlap."""
    n = len(x) // 3
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    
    c_list = [xc - r, 1.0 - xc - r, yc - r, 1.0 - yc - r]
    
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    c_list.append(dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx]**2)
    
    return np.concatenate(c_list)

def run_packing():
    n = 26
    n_pairs = n * (n - 1) // 2
    
    # Precompute constant structure of LP inequality matrix
    A_ub_const = np.zeros((n + n_pairs, n))
    for i in range(n):
        A_ub_const[i, i] = 1.0
    idx = n
    for i in range(n):
        for j in range(i + 1, n):
            A_ub_const[idx, i] = 1.0
            A_ub_const[idx, j] = 1.0
            idx += 1

    # Phase 1: Hexagonal Lattice Initialization
    centers = []
    r_est = 0.1
    y = r_est
    row = 0
    while len(centers) < n:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        x = x_start
        while x <= 1.0 - r_est and len(centers) < n:
            centers.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    centers = np.array(centers[:n])

    # Phase 2: Force-Directed Relaxation to spread points
    lr = 0.005
    for step in range(1500):
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.einsum('ijk,ijk->ij', diff, diff))
        dists = np.maximum(dists, 1e-5)
        
        # Repulsive force ~ 1/d^2
        rep = 0.005 / (dists**2)
        force_dirs = diff / dists[:, :, None]
        force_mat = rep[:, :, None] * force_dirs
        forces = np.sum(force_mat, axis=1)
        
        # Wall repulsion
        margin = 0.05
        for i in range(n):
            x, y = centers[i]
            if x < margin: forces[i, 0] += 0.1 * (margin - x)
            if x > 1 - margin: forces[i, 0] -= 0.1 * (x - (1 - margin))
            if y < margin: forces[i, 1] += 0.1 * (margin - y)
            if y > 1 - margin: forces[i, 1] -= 0.1 * (y - (1 - margin))
            
        centers += forces * lr
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
        lr *= 0.9995

    # Phase 3: Differential Evolution on Centers
    bounds_de = [(0.0, 1.0)] * (2 * n)
    best_x = None
    best_obj = np.inf

    for seed in range(3):
        try:
            res = differential_evolution(neg_sum_radii, bounds_de, args=(A_ub_const,),
                                        popsize=15, maxiter=400, mutation=(0.5, 1.0),
                                        recombination=0.7, seed=seed + 10, polish=False, tol=1e-7)
            if res.fun < best_obj:
                best_obj = res.fun
                best_x = res.x.copy()
        except Exception:
            continue

    if best_x is None:
        best_x = centers.flatten()

    best_centers = best_x.reshape(n, 2)
    best_sum, best_radii = compute_lp(best_centers, A_ub_const)

    # Phase 4: SLSQP Joint Refinement
    x0 = np.zeros(3 * n)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = np.maximum(best_radii * 0.98, 1e-6)

    try:
        res_sl = minimize(objective_full, x0, method='SLSQP',
                          bounds=[(0, 1), (0, 1), (1e-6, 0.5)] * n,
                          constraints={'type': 'ineq', 'fun': constraints_full},
                          options={'maxiter': 4000, 'ftol': 1e-13, 'iprint': -1})
        if res_sl.success:
            cons_val = constraints_full(res_sl.x)
            if np.min(cons_val) >= -1e-7:
                best_centers = np.column_stack((res_sl.x[0::3], res_sl.x[1::3]))
                best_radii = res_sl.x[2::3]
                best_sum = np.sum(best_radii)
    except Exception:
        pass

    # Final safety adjustments to guarantee strict validity
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if r > max_r:
            best_radii[i] = max_r

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((best_centers[i] - best_centers[j])**2))
            r_sum = best_radii[i] + best_radii[j]
            if r_sum > dist:
                factor = dist / r_sum
                best_radii[i] *= np.sqrt(factor)
                best_radii[j] *= np.sqrt(factor)

    best_sum = float(np.sum(best_radii))
    return best_centers, best_radii, best_sum
