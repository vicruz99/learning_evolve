import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """Inequality constraints: boundary containment and pairwise separation >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(dist2[mask] - rs[mask]**2)
    
    return np.concatenate(c)

def compute_safe_radii(centers):
    """Compute strictly feasible radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                max_r = min(max_r, d * 0.49)
        r[i] = max(max_r, 1e-6)
    return r

def generate_hex_init(r_est, shift_y=0.0, shift_x=0.0):
    """Generate hexagonal lattice initialization."""
    pts = []
    h = r_est * np.sqrt(3.0)
    y = r_est + shift_y
    row = 0
    while len(pts) < N:
        x_off = (row % 2) * r_est
        x = r_est + x_off + shift_x
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += h
        row += 1
    return np.array(pts[:N])

def generate_grid_init(scale=0.9, perturbation=0.0):
    """Generate grid-based initialization."""
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.05 + i * 0.19, 0.05 + j * 0.19])
    pts.append([0.5, 0.5])
    pts = np.array(pts[:N]) * scale
    
    if perturbation > 0:
        pts += np.random.uniform(-perturbation, perturbation, (N, 2))
        pts = np.clip(pts, 0.02, 0.98)
    
    return pts
