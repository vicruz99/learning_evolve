import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """Compute inequality constraints g(v) >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints: circles inside [0,1]^2
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation constraints: dist^2 >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dist2 = dx**2 + dy**2
    
    r_sum = r[i_idx] + r[j_idx]
    c.append(dist2 - r_sum**2)
    
    return np.concatenate(c)

def make_hex_init(r_est):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r_est
    row = 0
    while len(pts) < N:
        x_off = (row % 2) * r_est
        x = r_est + x_off
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    return np.array(pts[:N])

def make_grid_init():
    """Generates a perturbed square grid initialization."""
    pts = np.zeros((N, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            pts[idx] = [0.1 + i * 0.2, 0.1 + j * 0.2]
            idx += 1
    pts[25] = [0.5, 0.5]
    return pts

def make_corner_edge_init():
    """Generates an initialization biased towards corners and edges."""
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    # Explicit corner placement
    pts[0] = [0.05, 0.05]
    pts[1] = [0.95, 0.05]
    pts[2] = [0.05, 0.95]
    pts[3] = [0.95, 0.95]
    # Edge midpoints
    pts[4] = [0.5, 0.05]
    pts[5] = [0.5, 0.95]
    pts[6] = [0.05, 0.5]
    pts[7] = [0.95, 0.5]
    return pts

def compute_safe_radii(pts):
    """Compute strictly feasible initial radii for given centers."""
    r = np.full(N, 0.001)
    for i in range(N):
        # Distance to walls
        mx = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        # Distance to nearest neighbor
        for j in range(N):
            if i == j: continue
            d = np.sqrt(np.sum((pts[i] - pts[j])**2))
            if d * 0.5 < mx:
                mx = d * 0.5
        # Scale down to guarantee strict feasibility for optimizer start
        r[i] = mx * 0.85
    return r

def run_packing():
    best_sum = -np.inf
    best_v = None
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    inits = []
    
    # 1. Hexagonal lattices with varying densities
    for r_est in np.linspace(0.07, 0.13, 7):
        pts = make_hex_init(r_est)
        inits.append(pts)
        
    # 2. Grid based
    inits.append(make_grid_init())
    
    # 3. Corner/Edge biased
    inits.append(make_corner_edge_init())
    
    # 4. Perturbed versions of structured layouts
    for _ in range(20):
        base = inits[np.random.randint(0, len(inits))]
        pert = base + np.random.uniform(-0.04, 0.04, (N, 2))
        inits.append(np.clip(pert, 0.02, 0.98))
        
    # 5. Pure random
    for _ in range(10):
        inits.append(np.random.uniform(0.1, 0.9, (N, 2)))
        
    # Main optimization loop
    for pts in inits:
        r0 = compute_safe_radii(pts)
        v0 = np.zeros(3 * N)
        v0[0::3] = pts[:, 0]
        v0[1::3] = pts[:, 1]
        v0[2::3] = r0
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-9:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_v = res.x.copy()
        except Exception:
            continue

                best_sum = -res_p.fun
        except Exception:
            pass

        for _ in range(15):

        r0 = compute_safe_radii(pts)
        best_v = np.zeros(3*N)
        best_v[0::3] = pts[:, 0]
        best_v[1::3] = pts[:, 1]
        best_v[2::3] = r0
        best_sum = np.sum(r0)

    radii = np.maximum(best_v[2::3], 0.0)

    c_val = constraints(best_v)

        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        best_sum = np.sum(radii)
        
    return centers, radii, float(best_sum)
