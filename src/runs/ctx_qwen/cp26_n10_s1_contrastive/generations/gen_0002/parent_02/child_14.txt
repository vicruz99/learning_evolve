N = 26

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """
    Computes inequality constraints: boundary containment and pairwise separation.
    All constraints are formulated as g(v) >= 0.
    """
    xc = v[0::3]
    yc = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(xc - r)
    c.append(1.0 - xc - r)
    c.append(yc - r)
    c.append(1.0 - yc - r)
    
    # Pairwise separation: dist^2 >= (r_i + r_j)^2

    dx = xc[i_idx] - xc[j_idx]
    dy = yc[i_idx] - yc[j_idx]
    dr = r[i_idx] + r[j_idx]
    c.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(c)
