import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def constraint_func(vars_vec):
    """Computes inequality constraints for pairwise non-overlap."""
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    # Parameterization ensures circles are strictly within [0,1]^2
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    min_dist_sq = r_sum**2
    
    idx_i, idx_j = np.triu_indices(N, k=1)
    return dist_sq[idx_i, idx_j] - min_dist_sq[idx_i, idx_j]

def create_init(row_counts, r_est, perturb_std=0.0):
    """Generates an initial configuration based on hexagonal row counts."""
    pts = []
    y_pos = r_est
    row = 0
    while len(pts) < N:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        x = x_start
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y_pos])
            x += 2.0 * r_est
        y_pos += np.sqrt(3.0) * r_est
        row += 1
    pts = pts[:N]
    
    denom = 1.0 - 2.0 * r_est
    u = np.array([(p[0] - r_est)/denom for p in pts])
    v = np.array([(p[1] - r_est)/denom for p in pts])
    
    if perturb_std > 0:
        u += np.random.normal(0, perturb_std, N)
        v += np.random.normal(0, perturb_std, N)
        u = np.clip(u, 0.0, 1.0)
        v = np.clip(v, 0.0, 1.0)

    x0[1::3] = u
