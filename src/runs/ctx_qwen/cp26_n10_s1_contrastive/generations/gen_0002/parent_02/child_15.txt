import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraint_func(v):
    """
    Computes inequality constraints: boundary containment and pairwise separation.
    All constraints are formulated as g(v) >= 0.
    """
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    # Upper triangular mask to avoid duplicates and self-comparison
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(dist2[mask] - rs[mask]**2)
    
    return np.concatenate(c)

def get_feasible_radii(pts):
    """Compute strictly feasible initial radii for a given set of centers."""
    r = np.zeros(N)
    for i in range(N):
        max_r = min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])
        for j in range(N):
            if i != j:
                d = np.sqrt(np.sum((pts[i] - pts[j])**2))
                max_r = min(max_r, d * 0.5)
        r[i] = max_r * 0.88  # 0.88 ensures strict feasibility for optimizer start
    return r

def make_hex_init(rows, perturb=0.0):
    """Generates a hexagonal lattice initialization with specified row counts."""
    pts = []
    r_est = 0.09
    y = r_est
    row_idx = 0
    while len(pts) < N and row_idx < len(rows):
        count = rows[row_idx]
        x_off = (row_idx % 2) * r_est
        x = r_est + x_off
        for _ in range(count):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row_idx += 1
    pts = np.array(pts[:N])
    if perturb > 0:
        pts += np.random.uniform(-perturb, perturb, pts.shape)
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def make_grid_init(perturb=0.0):
    """Generates a perturbed square grid initialization."""
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    pts.append([0.5, 0.5])
    pts = np.array(pts[:N])
    if perturb > 0:
        pts += np.random.uniform(-perturb, perturb, pts.shape)
        pts = np.clip(pts, 0.01, 0.99)
    return pts

                np.random.seed(s * 100 + len(inits))
                inits.append(make_hex_init(rows, p))
                
    # Grid variants

            inits.append(make_grid_init(p))
