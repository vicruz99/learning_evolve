import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-4

def to_params(centers, radii):
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    return -np.sum(params[:N])

def cons_slqp(params):
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def lp_sum_for_circle_move(centers, idx, new_pos):
    temp_centers = centers.copy()
    temp_centers[idx] = new_pos
    temp_centers = np.clip(temp_centers, 1e-5, 1.0 - 1e-5)
    _, s = solve_lp_radii(temp_centers)
    return -s

def obj_nm(pos, centers, idx):
    return lp_sum_for_circle_move(centers, idx, pos)

def hex_init(rng, row_counts, rot, scale):
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def corner_bias_init(rng):
    pts = rng.uniform(0.05, 0.95, (N, 2))
    for i in range(8):
        side = rng.integers(0, 4)
        if side == 0: pts[i] = [rng.uniform(0.02, 0.2), rng.uniform(0.02, 0.2)]
        elif side == 1: pts[i] = [rng.uniform(0.8, 0.98), rng.uniform(0.02, 0.2)]
        elif side == 2: pts[i] = [rng.uniform(0.02, 0.2), rng.uniform(0.8, 0.98)]
        else: pts[i] = [rng.uniform(0.8, 0.98), rng.uniform(0.8, 0.98)]
    for _ in range(100):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        pts += 0.001 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], 
        [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
        [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3], [5,5,5,5,5,1], [4,5,6,5,6]
    ]
    for pat in pats:
        for _ in range(4):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.9, 1.1)))
    for s in range(20):
        inits.append(force_init(s))
    for _ in range(10):
        inits.append(corner_bias_init(rng))
        
    init_scores = []
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)
        init_scores.append((s0, c0, r0))
    init_scores.sort(reverse=True)
    
    for s0, c0, r0 in init_scores[:15]:
        p0 = to_params(c0, np.clip(r0 * 0.99, 1e-6, 0.49))
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp_dict,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        best_p = to_params(init_scores[0][1], init_scores[0][2] * 0.99)
        best_sum = -obj_slqp(best_p)
        
    curr_c, curr_r = from_params(best_p)
    curr_s = best_sum
    
    # SA Phase
    temp = 0.04
    step = 0.035
    for it in range(15000):
        move_type = rng.integers(0, 4)
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        if move_type == 0:
            curr_c[idx] += rng.normal(0, step, 2)
        elif move_type == 1:
            curr_c[idx] = rng.uniform(0.1, 0.9, 2)
        elif move_type == 2:
            jdx = rng.integers(N)
            while jdx == idx: jdx = rng.integers(N)
            old2 = curr_c[jdx].copy()
            curr_c[[idx, jdx]] += rng.normal(0, step*0.7, (2,2))
        else:
            jdx = rng.integers(N)
            while jdx == idx: jdx = rng.integers(N)
            old2 = curr_c[[idx, jdx]].copy()
            curr_c[idx], curr_c[jdx] = curr_c[jdx].copy(), curr_c[idx].copy()
            
        curr_c = np.clip(curr_c, 0.005, 0.995)
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_p = to_params(curr_c, np.clip(nr * 0.995, 1e-6, 0.49))
        else:
            curr_c[idx] = old
            if move_type == 2: curr_c[jdx] = old2
            elif move_type == 3: curr_c[[idx, jdx]] = old2
            
        temp *= 0.9994
        step = max(0.001, step * 0.9995)
        
    # Coordinate Descent on Centers
    curr_c, curr_r = from_params(best_p)
    curr_s = best_sum
    for cd_iter in range(5):
        for idx in range(N):
            old_pos = curr_c[idx].copy()
            res_nm = minimize(obj_nm, old_pos, args=(curr_c, idx), method='Nelder-Mead', 
                              options={'xatol': 1e-5, 'fatol': 1e-7, 'maxiter': 500})
            if -res_nm.fun > curr_s - 1e-9:
                curr_c[idx] = res_nm.x
                _, ns = solve_lp_radii(curr_c)
                if ns > curr_s:
                    curr_s = ns
            else:
                curr_c[idx] = old_pos
    curr_r, curr_s = solve_lp_radii(curr_c)
    if curr_s > best_sum:
        best_sum = curr_s
        best_p = to_params(curr_c, np.clip(curr_r * 0.995, 1e-6, 0.49))
        
    # Perturbation + SLSQP
    for k in range(50):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.025, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Shrink & Escape Phase
    curr_c_esc, curr_r_esc = from_params(best_p)
    curr_r_esc *= 0.93
    p_esc = to_params(curr_c_esc, np.clip(curr_r_esc, 1e-6, 0.49))
    p_esc[1::3] += rng.uniform(-0.06, 0.06, N)
    p_esc[2::3] += rng.uniform(-0.06, 0.06, N)
    p_esc[1::3] = np.clip(p_esc[1::3], 0.0, 1.0)
    p_esc[2::3] = np.clip(p_esc[2::3], 0.0, 1.0)
    
    try:
        res_esc = minimize(obj_slqp, p_esc, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp_dict,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res_esc.success and np.min(cons_slqp(res_esc.x)) >= -1e-8:
            s_esc = -res_esc.fun
            if s_esc > best_sum:
                best_sum = s_esc
                best_p = res_esc.x.copy()
    except Exception:
        pass
        
    # Global rotation escape
    curr_c_rot, _ = from_params(best_p)
    for _ in range(12):
        rot = rng.uniform(-0.12, 0.12)
        c_val, s_val = np.cos(rot), np.sin(rot)
        mat = np.array([[c_val, -s_val], [s_val, c_val]])
        c_rot = (curr_c_rot - 0.5) @ mat.T + 0.5
        c_rot = np.clip(c_rot, 0.01, 0.99)
        r_rot, _ = solve_lp_radii(c_rot)
        p_rot = to_params(c_rot, np.clip(r_rot * 0.995, 1e-6, 0.49))
        try:
            res_r = minimize(obj_slqp, p_rot, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp_dict,
                             options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res_r.success and np.min(cons_slqp(res_r.x)) >= -1e-8:
                s_r = -res_r.fun
                if s_r > best_sum:
                    best_sum = s_r
                    best_p = res_r.x.copy()
        except Exception:
            pass
            
    # Final Polish
    try:
        res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp_dict,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-9:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_p)
    radii_final, sum_final = solve_lp_radii(centers)
    
    return centers, np.maximum(radii_final, 0.0), float(sum_final)
