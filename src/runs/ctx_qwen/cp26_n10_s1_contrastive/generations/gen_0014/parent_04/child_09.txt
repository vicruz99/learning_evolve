import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to maximize sum(r_i) for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def to_params(centers, radii):
    r = np.clip(radii, 1e-6, 0.49)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    return -np.sum(params[:N])

def cons_slqp(params):
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot, scale):
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.012, 0.012, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng, seed):
    rng_f = np.random.RandomState(seed)
    pts = rng_f.uniform(0.1, 0.9, (N, 2))
    for _ in range(250):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-5
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 10.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 10.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.004 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def corner_init(rng, seed):
    rng_c = np.random.RandomState(seed)
    pts = np.zeros((N, 2))
    pts[0] = [0.1, 0.1]; pts[1] = [0.9, 0.1]
    pts[2] = [0.1, 0.9]; pts[3] = [0.9, 0.9]
    for i in range(4, 12):
        pts[i, 0] = 0.1 + 0.2 * (i-4); pts[i, 1] = 0.1
    for i in range(12, 20):
        pts[i, 0] = 0.1 + 0.2 * (i-12); pts[i, 1] = 0.9
    for i in range(20, 22):
        pts[i, 0] = 0.1; pts[i, 1] = 0.3 + 0.4 * (i-20)
    for i in range(22, 24):
        pts[i, 0] = 0.9; pts[i, 1] = 0.3 + 0.4 * (i-22)
    for i in range(24, N):
        pts[i] = rng_c.uniform(0.25, 0.75, 2)
    pts += rng_c.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def coord_descent(centers, rng, iters=1000):
    best_c = centers.copy()
    best_s = solve_lp_radii(best_c)[1]
    step = 0.045
    for it in range(iters):
        idx = rng.integers(N)
        old = best_c[idx].copy()
        best_c[idx] = np.clip(old + rng.normal(0, step, 2), 0.005, 0.995)
        nr, ns = solve_lp_radii(best_c)
        if ns > best_s:
            best_s = ns
        else:
            best_c[idx] = old
        step *= 0.9996
    return best_c, best_s

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [6,6,5,5,4],
        [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [7,6,5,4,4]
    ]
    for pat in pats:
        for _ in range(4):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.25, 0.25), scale=rng.uniform(0.92, 1.08)))
    for s in range(15):
        inits.append(force_init(rng, s))
    for s in range(10):
        inits.append(corner_init(rng, s))
        
    best_c = None
    best_s = -np.inf
    
    # Phase 1: CD on diverse inits
    for c0 in inits:
        c_cd, s_cd = coord_descent(c0, rng, iters=1200)
        if s_cd > best_s:
            best_s = s_cd
            best_c = c_cd.copy()
            
    # Phase 2: Basin-Hopping SA with CD refinement
    curr_c = best_c.copy()
    curr_s = best_s
    temp = 0.045
    max_sa = 4000
    
    for it in range(max_sa):
        trial_c = curr_c.copy()
        idx = rng.integers(N)
        if rng.random() < 0.06:
            trial_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            trial_c[idx] = np.clip(trial_c[idx] + rng.normal(0, 0.055, 2), 0.01, 0.99)
            
        c_cd, s_cd = coord_descent(trial_c, rng, iters=150)
        delta = s_cd - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_c = c_cd
            curr_s = s_cd
            if s_cd > best_s:
                best_s = s_cd
                best_c = c_cd.copy()
        temp *= 0.9997
        
    # Phase 3: SLSQP Polish
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_r, _ = solve_lp_radii(best_c)
    p0 = to_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
    
    try:
        res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
            val = -res.fun
            if val > best_s:
                best_s = val
                best_c, best_r = from_params(res.x)
    except Exception:
        pass
        
    # Phase 4: Perturbation Restarts
    for k in range(30):
        xp = to_params(best_c, best_r * 0.995)
        xp[N:3*N] += rng.normal(0, 0.025, 2*N)
        xp[:N] += rng.normal(0, 0.0015, N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                val = -res.fun
                if val > best_s:
                    best_s = val
                    best_c, best_r = from_params(res.x)
        except Exception:
            continue
            
    # Phase 5: Final exact LP
    radii_final, sum_final = solve_lp_radii(best_c)
    return best_c, np.maximum(radii_final, 0.0), float(sum_final)
