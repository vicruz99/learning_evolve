

<strategy>
1. **Diverse Initialization**: Generate multiple high-quality starting configurations using hexagonal lattices (with varied row counts, rotations, and scales), force-directed repulsion layouts, and corner/edge-biased placements. Evaluate each with a Linear Program (LP) to maximize the sum of radii given fixed centers, keeping the top performers.
2. **LP-Driven Simulated Annealing (SA)**: Perform a long SA run optimizing center positions. The objective is the LP-computed sum of radii. Use an adaptive step size and temperature schedule. Include "shock" moves (random repositioning, swaps, or subset rotations) to break out of symmetric or dense local traps.
3. **SLSQP Refinement with Boundary-Safe Parameterization**: Convert the best configuration to a parameterization `(r, u, v)` where `x = r + u(1-2r)` and `y = r + v(1-2r)`. This automatically satisfies boundary constraints. Use SLSQP to maximize `sum(r)` subject to pairwise non-overlap constraints (`dist^2 >= (r_i + r_j)^2`).
4. **Aggressive Perturbation & Restarts**: Run SLSQP from many randomly perturbed versions of the best state to explore the landscape around the local optimum. Include targeted perturbations that nudge tightly packed circles apart.
5. **High-Precision Polish**: Run a final SLSQP pass with tight tolerances and high iteration limits. Conclude with a final LP solve to guarantee radii are exactly optimal for the final center positions.
6. **Robustness**: Strictly enforce bounds, handle numerical edge cases, and validate internally before returning.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to maximize sum(r_i) for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = np.clip(radii, 1e-6, 0.49)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_hex_centers(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_centers(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.1, 0.9, (N, 2))
    for _ in range(250):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-4
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 15.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 15.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.003 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def generate_corner_bias_centers(rng, seed):
    """Generates initial configs biased towards corners and edges."""
    rng_c = np.random.RandomState(seed)
    pts = rng_c.uniform(0.0, 1.0, (N, 2))
    pts[:, 0] = np.minimum(pts[:, 0], 1.0 - pts[:, 0]) + rng_c.uniform(0.0, 0.1, N)
    pts[:, 0] = np.where(rng_c.rand(N) > 0.5, pts[:, 0], 1.0 - pts[:, 0])
    pts[:, 1] = np.minimum(pts[:, 1], 1.0 - pts[:, 1]) + rng_c.uniform(0.0, 0.1, N)
    pts[:, 1] = np.where(rng_c.rand(N) > 0.5, pts[:, 1], 1.0 - pts[:, 1])
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Phase 1: Diverse Initializations & LP Evaluation
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [6,6,5,5,4],
        [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,5,6,5],
        [5,6,5,5,5], [7,6,5,4,4], [6,7,5,5,3], [5,5,5,5,5,1]
    ]
    for pat in pats:
        for _ in range(3):
            inits.append(generate_hex_centers(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.9, 1.1)))
    for s in range(15):
        inits.append(generate_force_centers(rng, s))
    for s in range(10):
        inits.append(generate_corner_bias_centers(rng, s))
        
    init_scores = []
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)
        init_scores.append((s0, c0, r0))
    init_scores.sort(key=lambda x: x[0], reverse=True)
    
    # SLSQP on top starts
    for s0, c0, r0 in init_scores[:15]:
        p0 = to_params(c0, np.clip(r0 * 0.99, 1e-6, 0.49))
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                val = -res.fun
                if val > best_sum:
                    best_sum = val
                    best_p = res.x.copy()
        except Exception:
            pass
            
    if best_p is not None:
        best_c, best_r = from_params(best_p)
    else:
        best_c, best_r = init_scores[0][1], init_scores[0][2]
        best_sum = init_scores[0][0]
        
    # Phase 2: LP-driven Simulated Annealing on Centers
    curr_c = best_c.copy()
    curr_r, curr_s = solve_lp_radii(curr_c)
    best_sa_c = curr_c.copy()
    best_sa_r = curr_r.copy()
    best_sa_s = curr_s
    
    temp = 0.045
    step = 0.035
    max_iters = 15000
    
    for it in range(max_iters):
        step_curr = step * (1.0 + 0.5 * np.cos(2.0 * np.pi * it / 3000))
        
        move_type = rng.integers(0, 5)
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        if move_type == 0: # Single move
            if rng.random() < 0.04:
                curr_c[idx] = rng.uniform(0.05, 0.95, 2)
            else:
                curr_c[idx] += rng.normal(0, step_curr, 2)
                curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        elif move_type == 1: # Swap
            j = rng.integers(N)
            while j == idx: j = rng.integers(N)
            curr_c[idx], curr_c[j] = curr_c[j].copy(), curr_c[idx].copy()
            old = curr_c[[idx, j]].copy()
        elif move_type == 2: # Rotate subset
            idxs = rng.choice(N, 5, replace=False)
            center = curr_c[idxs].mean(axis=0)
            rot = rng.uniform(-0.15, 0.15)
            c_val, s_val = np.cos(rot), np.sin(rot)
            mat = np.array([[c_val, -s_val], [s_val, c_val]])
            shifted = curr_c[idxs] - center
            curr_c[idxs] = (shifted @ mat.T) + center
            curr_c[idxs] = np.clip(curr_c[idxs], 0.01, 0.99)
            old = curr_c[idxs].copy()
        else: # Large shock
            curr_c[idx] = rng.uniform(0.0, 1.0, 2)
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            curr_r = nr
            if ns > best_sa_s:
                best_sa_s = ns
                best_sa_c = curr_c.copy()
                best_sa_r = nr.copy()
                step = min(0.045, step * 1.002)
        else:
            if move_type == 0:
                curr_c[idx] = old
            elif move_type == 1:
                curr_c[[idx, j]] = old
            elif move_type == 2:
                curr_c[idxs] = old
            step = max(0.001, step * 0.999)
            
        temp *= 0.9995
        
    if best_sa_s > best_sum:
        best_sum = best_sa_s
        best_c = best_sa_c
        best_r = best_sa_r
        best_p = to_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
    else:
        best_c, best_r = from_params(best_p)
        
    # Phase 3: SLSQP Refinement & Perturbation
    for k in range(45):
        xp = best_p.copy()
        xp[N:3*N] += rng.normal(0, 0.028, 2*N)
        xp[:N] += rng.normal(0, 0.0025, N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                val = -res.fun
                if val > best_sum:
                    best_sum = val
                    best_p = res.x.copy()
                    best_c, best_r = from_params(best_p)
        except Exception:
            continue
            
    # Phase 4: Rotation & Symmetry Breaks
    for _ in range(20):
        rot = rng.uniform(-0.12, 0.12)
        c_val, s_val = np.cos(rot), np.sin(rot)
        mat = np.array([[c_val, -s_val], [s_val, c_val]])
        c_rot = (best_c - 0.5) @ mat.T + 0.5
        c_rot = np.clip(c_rot, 0.01, 0.99)
        r_rot, _ = solve_lp_radii(c_rot)
        p_rot = to_params(c_rot, np.clip(r_rot * 0.995, 1e-6, 0.49))
        
        try:
            res_r = minimize(obj_slqp, p_rot, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                             options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res_r.success and np.min(cons_slqp(res_r.x)) >= -1e-8:
                val_r = -res_r.fun
                if val_r > best_sum:
                    best_sum = val_r
                    best_p = res_r.x.copy()
                    best_c, best_r = from_params(best_p)
        except Exception:
            pass
            
    # Phase 5: Final High-Precision Polish
    try:
        res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 18000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-9:
            best_p = res_f.x
            best_sum = -res_f.fun
            best_c, best_r = from_params(best_p)
    except Exception:
        pass
        
    # Final LP to guarantee optimal radii for exact centers
    radii_final, sum_final = solve_lp_radii(best_c)
    
    # Safety validation & clipping
    radii_final = np.maximum(radii_final, 0.0)
    best_c = np.clip(best_c, 1e-9, 1.0 - 1e-9)
    
    return best_c, radii_final, float(sum_final)
