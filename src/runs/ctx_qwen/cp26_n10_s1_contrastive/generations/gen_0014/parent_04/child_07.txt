

<strategy>
1. **Initialization**: Generate a wide variety of initial center configurations: hexagonal lattices (varied row counts, rotations, scales), force-directed layouts, and explicit boundary-biased configurations. This ensures we explore different topological contact graphs.
2. **LP Radii Solver**: Maintain an efficient Linear Programming solver to compute the exact maximum sum of radii for any fixed set of centers. This provides a precise evaluation metric for the non-smooth objective landscape.
3. **Simulated Annealing (SA)**: Run SA on the center coordinates for 25,000 iterations. Use an adaptive step size and periodic "shocks" to escape local minima. The LP solver evaluates each move, ensuring we always know the true maximum sum for the current geometry.
4. **Hill-Climbing Refinement**: After SA, perform a deterministic local search on centers. Iteratively perturb each circle's position, accept moves that increase the LP-optimal sum of radii. This polishes the configuration to a tight local optimum.
5. **SLSQP Joint Optimization**: Convert the best center/radius configuration to the boundary-safe `(r, u, v)` parameterization. Use SLSQP to jointly optimize centers and radii subject to pairwise non-overlap constraints. Run multiple perturbation restarts to escape local minima in the continuous constraint landscape.
6. **High-Precision Polish**: Finish with a high-iteration, tight-tolerance SLSQP pass, followed by a final LP solve to guarantee radii are exactly optimal for the final centers.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to maximize sum(r_i) for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = np.clip(radii, 1e-6, 0.49)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_hex_centers(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_centers(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.1, 0.9, (N, 2))
    for _ in range(200):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-4
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 15.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 15.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.003 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def generate_boundary_centers(rng, seed):
    """Generates configurations biased towards corners and edges."""
    rng_c = np.random.RandomState(seed)
    pts = rng_c.uniform(0.0, 1.0, (N, 2))
    # Bias towards boundaries
    pts[:, 0] = np.where(rng_c.rand(N) > 0.5, pts[:, 0], 1.0 - pts[:, 0])
    pts[:, 0] = np.minimum(pts[:, 0], 0.35) + rng_c.uniform(0.0, 0.1, N)
    pts[:, 1] = np.where(rng_c.rand(N) > 0.5, pts[:, 1], 1.0 - pts[:, 1])
    pts[:, 1] = np.minimum(pts[:, 1], 0.35) + rng_c.uniform(0.0, 0.1, N)
    
    # Quick repulsion to separate
    for _ in range(100):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-4
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        pts += 0.004 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def hill_climb_centers(centers, rng, n_steps, init_step=0.02):
    """Deterministic local search to polish centers."""
    curr_c = centers.copy()
    _, curr_s = solve_lp_radii(curr_c)
    step = init_step
    
    for _ in range(n_steps):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        best_move = old
        best_val = curr_s
        
        for _ in range(6):
            trial = old + rng.normal(0, step, 2)
            trial = np.clip(trial, 0.005, 0.995)
            curr_c[idx] = trial
            _, val = solve_lp_radii(curr_c)
            if val > best_val:
                best_val = val
                best_move = trial
                
        curr_c[idx] = best_move
        if best_val > curr_s:
            curr_s = best_val
            step = min(0.03, step * 1.05)
        else:
            step = max(0.0005, step * 0.98)
            
    return curr_c, curr_s

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Phase 1: Diverse Initializations
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [6,6,5,5,4],
        [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,5,6,5],
        [5,6,5,5,5], [7,6,5,4,4], [6,7,5,5,3]
    ]
    for pat in pats:
        for _ in range(3):
            inits.append(generate_hex_centers(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.9, 1.1)))
    for s in range(15):
        inits.append(generate_force_centers(rng, s))
    for s in range(10):
        inits.append(generate_boundary_centers(rng, s))
        
    # Evaluate inits with LP
    init_scores = []
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)
        init_scores.append((s0, c0, r0))
    init_scores.sort(key=lambda x: x[0], reverse=True)
    
    # SLSQP on top starts
    for s0, c0, r0 in init_scores[:15]:
        p0 = to_params(c0, np.clip(r0 * 0.99, 1e-6, 0.49))
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                val = -res.fun
                if val > best_sum:
                    best_sum = val
                    best_p = res.x.copy()
        except Exception:
            pass
            
    if best_p is not None:
        best_c, best_r = from_params(best_p)
    else:
        best_c, best_r = init_scores[0][1], init_scores[0][2]
        best_sum = init_scores[0][0]
        
    # Phase 2: LP-driven Simulated Annealing on Centers
    curr_c = best_c.copy()
    curr_r, curr_s = solve_lp_radii(curr_c)
    best_sa_c = curr_c.copy()
    best_sa_r = curr_r.copy()
    best_sa_s = curr_s
    
    temp = 0.045
    step = 0.035
    max_iters = 25000
    
    for it in range(max_iters):
        step_curr = step * (1.0 + 0.5 * np.cos(2.0 * np.pi * it / 3000))
        
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        if rng.random() < 0.025:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            curr_c[idx] += rng.normal(0, step_curr, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            curr_r = nr
            if ns > best_sa_s:
                best_sa_s = ns
                best_sa_c = curr_c.copy()
                best_sa_r = nr.copy()
                if step < 0.045:
                    step = min(0.045, step * 1.01)
        else:
            curr_c[idx] = old
            if step > 0.001:
                step *= 0.9995
                
        temp *= 0.9996
        
    if best_sa_s > best_sum:
        best_sum = best_sa_s
        best_c = best_sa_c
        best_r = best_sa_r
        best_p = to_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
    else:
        best_c, best_r = from_params(best_p)
        
    # Phase 2b: Hill Climbing Refinement
    best_c, best_sum = hill_climb_centers(best_c, rng, n_steps=4000, init_step=0.015)
    best_r, best_sum = solve_lp_radii(best_c)
    best_p = to_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
        
    # Phase 3: SLSQP Refinement & Perturbation
    for k in range(40):
        xp = best_p.copy()
        xp[N:3*N] += rng.normal(0, 0.025, 2*N)
        xp[:N] += rng.normal(0, 0.002, N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                val = -res.fun
                if val > best_sum:
                    best_sum = val
                    best_p = res.x.copy()
                    best_c, best_r = from_params(best_p)
        except Exception:
            continue
            
    # Phase 4: Rotation Escapes
    for _ in range(15):
        rot = rng.uniform(-0.15, 0.15)
        c_val, s_val = np.cos(rot), np.sin(rot)
        mat = np.array([[c_val, -s_val], [s_val, c_val]])
        c_rot = (best_c - 0.5) @ mat.T + 0.5
        c_rot = np.clip(c_rot, 0.01, 0.99)
        r_rot, _ = solve_lp_radii(c_rot)
        p_rot = to_params(c_rot, np.clip(r_rot * 0.995, 1e-6, 0.49))
        
        try:
            res_r = minimize(obj_slqp, p_rot, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                             options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res_r.success and np.min(cons_slqp(res_r.x)) >= -1e-8:
                val_r = -res_r.fun
                if val_r > best_sum:
                    best_sum = val_r
                    best_p = res_r.x.copy()
                    best_c, best_r = from_params(best_p)
        except Exception:
            pass
            
    # Phase 5: Final High-Precision Polish
    try:
        res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-9:
            best_p = res_f.x
            best_sum = -res_f.fun
            best_c, best_r = from_params(best_p)
    except Exception:
        pass
        
    # Final LP to guarantee optimal radii for exact centers
    radii_final, sum_final = solve_lp_radii(best_c)
    
    return best_c, np.maximum(radii_final, 0.0), float(sum_final)
