import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to maximize sum(r_i) for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-4

def to_params(centers, radii):
    r = np.clip(radii, 1e-6, 0.49)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    return -np.sum(params[:N])

def cons_slqp(params):
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_hex_centers(rng, row_counts, rot, scale):
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_centers(rng, seed):
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.1, 0.9, (N, 2))
    r_curr = np.full(N, 0.06)
    for step in range(300):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / d**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            forces[:, dim] += 25.0 * np.maximum(0, r_curr - pts[:, dim])
            forces[:, dim] -= 25.0 * np.maximum(0, pts[:, dim] - (1.0 - r_curr))
        step_size = 0.008 * (0.992**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_others[i] = np.inf
            d_pair = np.min(dists_others)
            r_curr[i] = 0.8 * min(d_wall, d_pair/2.0)
    return pts

def run_packing():

    best_sum = -np.inf

    best_r = None

    inits = []

        [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [7,6,5,4,4],
        [6,7,5,5,3], [8,6,5,4,3], [5,5,5,5,5,1], [4,5,6,5,6],
        [6,5,5,5,5], [5,6,5,5,5], [7,5,4,5,5]
    ]
    for pat in pats:
        for _ in range(4):
            inits.append(generate_hex_centers(rng, pat, rot=rng.uniform(-0.35, 0.35), scale=rng.uniform(0.88, 1.12)))
    for s in range(25):
        inits.append(generate_force_centers(rng, s))
        
    # Evaluate inits with LP
    init_scores = []
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)
        init_scores.append((s0, c0, r0))
    init_scores.sort(key=lambda x: x[0], reverse=True)
    
    # SLSQP on top starts
    for s0, c0, r0 in init_scores[:20]:
        p0 = to_params(c0, np.clip(r0 * 0.99, 1e-6, 0.49))
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:

        elif move_type == 1:

            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        elif move_type == 2:

            curr_c[[i, j]] += rng.normal(0, step_adj, (2, 2))

        else:
            if move_type < 4:

                    curr_c[[i, j]] = old
