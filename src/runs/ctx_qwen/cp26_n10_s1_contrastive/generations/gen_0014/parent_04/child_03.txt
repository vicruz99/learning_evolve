To improve the packing sum of radii from ~2.626 to the target of ~2.636, I will adopt a robust hybrid optimization strategy that combines:
1. **Diverse Topological Initializations**: Generate hexagonal lattices with varied row counts, rotations, and scales, alongside force-directed layouts that explicitly push circles toward boundaries and corners. This increases the chance of starting in the basin of the global optimum.
2. **LP-Driven Simulated Annealing (SA)**: Instead of optimizing radii and centers jointly with a black-box solver, I will fix centers and solve a fast Linear Program (LP) to maximize $\sum r_i$. SA will then perturb centers, accepting moves that improve the LP objective with a Boltzmann probability. This separates the discrete topology search from the continuous radius expansion.
3. **Constraint-Embedded SLSQP Refinement**: Use a parameterization $x_i = r_i + u_i(1-2r_i)$, $y_i = r_i + v_i(1-2r_i)$ with $u_i, v_i \in [0,1]$. This automatically satisfies boundary constraints, leaving only 325 pairwise non-overlap constraints for SLSQP. This makes the feasible region easier to navigate.
4. **Perturbation & Multi-Restart Polishing**: After SA finds a strong configuration, run SLSQP from the best point and multiple perturbed versions to escape local minima. Finish with a high-precision SLSQP run and a final exact LP solve to guarantee radii are optimal for the exact final centers.
5. **Strict Compliance**: All helper functions are top-level with no closures or lambdas, ensuring compatibility with the validation environment.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to maximize sum(r_i) for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = np.clip(radii, 1e-6, 0.49)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_hex_centers(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_centers(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.1, 0.9, (N, 2))
    for _ in range(250):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-4
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 15.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 15.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.003 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def generate_corner_bias_centers(rng, seed):
    """Generates initial configs biased towards corners and edges."""
    rng_c = np.random.RandomState(seed)
    pts = rng_c.uniform(0.0, 1.0, (N, 2))
    # Strongly bias towards boundaries
    pts[:, 0] = np.minimum(pts[:, 0], 1.0 - pts[:, 0]) + rng_c.uniform(0.0, 0.08, N)
    pts[:, 0] = np.where(rng_c.rand(N) > 0.5, pts[:, 0], 1.0 - pts[:, 0])
    pts[:, 1] = np.minimum(pts[:, 1], 1.0 - pts[:, 1]) + rng_c.uniform(0.0, 0.08, N)
    pts[:, 1] = np.where(rng_c.rand(N) > 0.5, pts[:, 1], 1.0 - pts[:, 1])
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Phase 1: Diverse Initializations
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [6,6,5,5,4],
        [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,5,6,5],
        [5,6,5,5,5], [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3]
    ]
    for pat in pats:
        for _ in range(3):
            inits.append(generate_hex_centers(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.9, 1.1)))
    for s in range(15):
        inits.append(generate_force_centers(rng, s))
    for s in range(10):
        inits.append(generate_corner_bias_centers(rng, s))
        
    # Evaluate inits with LP and run SLSQP on top performers
    init_scores = []
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)
        init_scores.append((s0, c0, r0))
    init_scores.sort(key=lambda x: x[0], reverse=True)
    
    for s0, c0, r0 in init_scores[:12]:
        p0 = to_params(c0, np.clip(r0 * 0.98, 1e-6, 0.49))
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                val = -res.fun
                if val > best_sum:
                    best_sum = val
                    best_p = res.x.copy()
        except Exception:
            pass
            
    if best_p is not None:
        best_c, best_r = from_params(best_p)
    else:
        best_c, best_r = init_scores[0][1], init_scores[0][2]
        best_sum = init_scores[0][0]
        
    # Phase 2: LP-driven Simulated Annealing on Centers
    curr_c = best_c.copy()
    curr_r, curr_s = solve_lp_radii(curr_c)
    best_sa_c = curr_c.copy()
    best_sa_r = curr_r.copy()
    best_sa_s = curr_s
    
    temp = 0.05
    step = 0.035
    max_iters = 18000
    
    for it in range(max_iters):
        # Oscillating step size to help escape local minima
        step_curr = step * (1.0 + 0.3 * np.cos(2.0 * np.pi * it / 3000))
        
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        # Occasional large jump
        if rng.random() < 0.025:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            curr_c[idx] += rng.normal(0, step_curr, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        # Acceptance criterion
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            curr_r = nr
            if ns > best_sa_s:
                best_sa_s = ns
                best_sa_c = curr_c.copy()
                best_sa_r = nr.copy()
                if step < 0.04:
                    step = min(0.04, step * 1.005)
        else:
            curr_c[idx] = old
            if step > 0.002:
                step *= 0.9995
                
        temp *= 0.9996
        
    if best_sa_s > best_sum:
        best_sum = best_sa_s
        best_c = best_sa_c
        best_r = best_sa_r
        best_p = to_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
    else:
        best_c, best_r = from_params(best_p)
        
    # Phase 3: SLSQP Refinement & Perturbation
    for k in range(25):
        xp = best_p.copy()
        xp[N:3*N] += rng.normal(0, 0.02, 2*N)
        xp[:N] += rng.normal(0, 0.0015, N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                val = -res.fun
                if val > best_sum:
                    best_sum = val
                    best_p = res.x.copy()
                    best_c, best_r = from_params(best_p)
        except Exception:
            continue
            
    # Phase 4: Global Rotation Escapes
    for _ in range(12):
        rot = rng.uniform(-0.12, 0.12)
        c_val, s_val = np.cos(rot), np.sin(rot)
        mat = np.array([[c_val, -s_val], [s_val, c_val]])
        c_rot = (best_c - 0.5) @ mat.T + 0.5
        c_rot = np.clip(c_rot, 0.01, 0.99)
        r_rot, _ = solve_lp_radii(c_rot)
        p_rot = to_params(c_rot, np.clip(r_rot * 0.995, 1e-6, 0.49))
        
        try:
            res_r = minimize(obj_slqp, p_rot, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                             options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res_r.success and np.min(cons_slqp(res_r.x)) >= -1e-8:
                val_r = -res_r.fun
                if val_r > best_sum:
                    best_sum = val_r
                    best_p = res_r.x.copy()
                    best_c, best_r = from_params(best_p)
        except Exception:
            pass
            
    # Phase 5: Final High-Precision Polish
    try:
        res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-9:
            best_p = res_f.x
            best_sum = -res_f.fun
            best_c, best_r = from_params(best_p)
    except Exception:
        pass
        
    # Final LP to guarantee optimal radii for exact centers
    radii_final, sum_final = solve_lp_radii(best_c)
    
    return best_c, np.maximum(radii_final, 0.0), float(sum_final)
