

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we employ a multi-phase hybrid optimization strategy that combines exact linear programming, stochastic search, and gradient-based refinement:

1. **Decoupled Optimization (LP on Radii)**: For any fixed set of centers, finding the optimal radii is a linear programming problem. Constraints are `r_i >= 0`, `r_i <= distance to walls`, and `r_i + r_j <= distance(i, j)`. We solve this LP exactly at each evaluation step using the fast `highs` solver. This turns the problem into maximizing a piecewise-linear concave function of the centers.

2. **Boundary-Safe Parameterization for SLSQP**: When refining with SLSQP, we parameterize positions as `x = r + u*(1-2r)` and `y = r + v*(1-2r)` with `u, v in [0,1]`. This mathematically guarantees `r <= x <= 1-r` and `r <= y <= 1-r`, automatically satisfying all wall constraints and reducing the constraint set to only pairwise non-overlap conditions.

3. **Diverse Initialization**: We generate ~50 initial configurations using varied hexagonal lattices (different row patterns, rotations, scales), force-directed repulsion layouts, and perturbed grids. These are scored with the LP to identify promising topological starting points.

4. **Simulated Annealing (SA) on Centers**: We run SA on the center coordinates. The "energy" is the negative LP radii sum. SA uses an adaptive step size and exponential cooling. It includes occasional "shake" moves (random repositioning) to escape deep local minima and change the contact graph topology.

5. **Coordinate-wise Hill Climbing**: After SA, we run a deterministic local search. Each iteration picks a random circle, performs a mini random walk around its position, and accepts moves that increase the LP sum. This efficiently tunes individual circle placements.

6. **Multi-Start SLSQP Refinement**: The best configuration from SA/Hill Climbing is converted to `(r, u, v)` parameters. We run SLSQP to polish it, followed by 30 perturbed restarts to escape remaining local optima. A final high-precision SLSQP run ensures numerical tightness.

7. **Final Validation**: We extract centers from the best parameters and run one final exact LP solve to guarantee the returned radii are optimal for those precise centers.

This approach systematically explores global topology changes while precisely optimizing local geometry, targeting the sub-0.002 gap to the target sum.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], 
        [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
        [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3], [5,5,5,5,5,1], [4,5,6,5,6],
        [5,5,5,5,5,1], [6,5,5,5,4,1], [4,6,5,5,6], [5,4,6,6,5]
    ]
    for pat in pats:
        for _ in range(4):
            pts = []
            r_est = 0.095
            y = r_est
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.9, 1.1) + 0.5
            rot = rng.uniform(-0.3, 0.3)
            c_val, s_val = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5
            pts += rng.uniform(-0.015, 0.015, pts.shape)
            inits.append(np.clip(pts, 0.02, 0.98))
            
    # 2. Force-directed layouts
    for s in range(15):
        rng_fd = np.random.RandomState(s)
        pts = rng_fd.uniform(0.15, 0.85, (N, 2))
        for _ in range(300):
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-6
            f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
            for dim in range(2):
                f[:, dim] += 30.0 * np.maximum(0, 0.05 - pts[:, dim])
                f[:, dim] -= 30.0 * np.maximum(0, pts[:, dim] - 0.95)
            pts += 0.003 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # 3. Perturbed grids
    for s in range(10):
        rng_g = np.random.RandomState(s)
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng_g.uniform(-0.03, 0.03, (N, 2))
        inits.append(np.clip(pts, 0.05, 0.95))
        
    return inits

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Generate and score initial configurations
    inits = generate_inits(rng)
    init_scores = []
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)
        init_scores.append((s0, c0, r0))
    init_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Phase 1: SLSQP refinement from top initial configurations
    for s0, c0, r0 in init_scores[:15]:
        p0 = to_params(c0, np.clip(r0 * 0.99, 1e-6, 0.49))
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        best_p = to_params(init_scores[0][1], init_scores[0][2] * 0.99)
        best_sum = -obj_slqp(best_p)
        
    curr_c, curr_r = from_params(best_p)
    curr_s = best_sum
    best_c = curr_c.copy()
    best_r = curr_r.copy()
    
    # Phase 2: LP-Driven Simulated Annealing on Centers
    temp = 0.05
    step = 0.04
    for it in range(6000):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        # Occasional shock to escape deep local minima
        if rng.random() < 0.04:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            curr_c[idx] += rng.normal(0, step, 2)
        curr_c = np.clip(curr_c, 0.01, 0.99)
        
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-7 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
                step = min(0.05, step * 1.02)
            else:
                step = max(0.001, step * 0.995)
        else:
            curr_c[idx] = old
            
        temp *= 0.9995
        step = max(0.001, step * 0.9998)
        
    # Phase 3: Coordinate-wise Hill Climbing
    curr_c = best_c.copy()
    curr_s = best_sum
    step = 0.02
    for _ in range(2000):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        best_local = curr_s
        best_local_pos = old.copy()
        
        for _ in range(6):
            new_pos = old + rng.normal(0, step * 0.5, 2)
            new_pos = np.clip(new_pos, 0.01, 0.99)
            curr_c[idx] = new_pos
            _, score = solve_lp_radii(curr_c)
            if score > best_local:
                best_local = score
                best_local_pos = new_pos.copy()
            curr_c[idx] = best_local_pos
            
        curr_c[idx] = best_local_pos
        if best_local > curr_s:
            curr_s = best_local
            if curr_s > best_sum:
                best_sum = curr_s
                best_c = curr_c.copy()
                best_r, _ = solve_lp_radii(curr_c)
        step = max(0.001, step * 0.999)
        
    # Phase 4: SLSQP refinement on SA/HillClimbing best
    r_safe = np.clip(best_r * 0.995, 1e-6, 0.49)
    p0 = to_params(best_c, r_safe)
    try:
        res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_p = res.x.copy()
    except Exception:
        pass
        
    # Phase 5: Perturbation & SLSQP restarts
    for k in range(35):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.025, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 6: High-precision final polish
    for _ in range(2):
        try:
            res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                             options={'maxiter': 20000, 'ftol': 1e-15, 'disp': False})
            if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-9:
                best_p = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Reconstruct and final exact LP solve
    centers, radii = from_params(best_p)
    radii_final, sum_final = solve_lp_radii(centers)
    
    return centers, np.maximum(radii_final, 0.0), float(sum_final)
