import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n = 26
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    # Define the objective function to minimize (negative sum of radii)
    def objective(vars):
        # vars layout: [x1, y1, r1, x2, y2, r2, ...]
        # We optimize centers and radii simultaneously
        # However, to make constraints easier, we might fix a structure or just let it flow.
        # Here we optimize all 3*n variables.
        radii = vars[2::3]
        return -np.sum(radii)

    # Define constraints
    def constraint_non_overlap(vars):
        # We need to check all pairs. 
        # To make it differentiable and smooth, we use squared distance.
        # dist^2 >= (r1+r2)^2  =>  dist^2 - (r1+r2)^2 >= 0
        # We will enforce this for all pairs i < j
        constraints = []
        for i in range(n):
            xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
            for j in range(i + 1, n):
                xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
                
                # Distance squared
                d2 = (xi - xj)**2 + (yi - yj)**2
                # Sum of radii squared
                r_sum2 = (ri + rj)**2
                
                # Constraint: d2 - r_sum2 >= 0
                # We add a small slack for numerical stability, but optimizer should handle >= 0
                # The constraint function should return >= 0
                constraints.append(d2 - r_sum2)
        return np.array(constraints)

    def constraint_boundary(vars):
        constraints = []
        for i in range(n):
            xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
            # x - r >= 0 => x - r >= 0
            # x + r <= 1 => 1 - x - r >= 0
            # same for y
            constraints.append(xi - ri)
            constraints.append(1.0 - xi - ri)
            constraints.append(yi - ri)
            constraints.append(1.0 - yi - ri)
        return np.array(constraints)

    def constraint_non_negative_r(vars):
        radii = vars[2::3]
        return radii

    # Prepare constraints dict for scipy
    # SLSQP supports equality and inequality constraints.
    # Inequality constraints are fun(x) >= 0.
    
    cons = []
    
    # Non-overlap constraints (approx 325)
    # Creating a function that returns array is efficient
    cons.append({'type': 'ineq', 'fun': constraint_non_overlap})
    
    # Boundary constraints (104)
    cons.append({'type': 'ineq', 'fun': constraint_boundary})
    
    # Non-negative radii (26)
    cons.append({'type': 'ineq', 'fun': constraint_non_negative_r})

    # r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    np.random.seed(42)
    num_attempts = 20

        if strategy == 'grid':
            # 5x5 grid perturbation

            # Let's place 5 rows of 5, and one extra somewhere? 
            # Or just random perturbation of a dense packing.
            # Let's try a 5x5 grid plus one.

            coords = []
            for r in range(5):
                for c in range(5):
                    coords.append([0.1 + 0.2*r, 0.1 + 0.2*c]) # Wait, r is row index? No.
                    # Let's fix coordinates
            # Reset
            coords = []
            for i in range(5):
                for j in range(5):
                    coords.append([0.1 + 0.2*i, 0.1 + 0.2*j])
            # Add 26th circle at center or edge?
            coords.append([0.5, 0.5])
            
            # Shuffle coords to randomize assignment
            np.random.shuffle(coords)

                x, y = coords[i]

                x += np.random.normal(0, 0.01)
                y += np.random.normal(0, 0.01)
                # Clamp
                x = np.clip(x, 0.05, 0.95)
                y = np.clip(y, 0.05, 0.95)
                
                vars[3*i] = x
                vars[3*i+1] = y
                vars[3*i+2] = 0.09 # Initial radius estimate (slightly less than 0.1)
        elif strategy == 'random':
            for i in range(n):
                vars[3*i] = np.random.uniform(0.1, 0.9)

    return best_centers, best_radii, float(best_sum)
