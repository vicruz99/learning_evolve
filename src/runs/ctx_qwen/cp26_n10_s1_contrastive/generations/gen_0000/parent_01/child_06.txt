import numpy as np
from scipy.optimize import minimize

N = 26


def compute_objective_with_penalty(x, penalty_weight):
    centers = x[:2 * N].reshape(N, 2)
    radii = x[2 * N:]
    
    obj = -np.sum(radii)
    
    for i in range(N):
        for dim in range(2):
            val = centers[i, dim] - radii[i]
            if val < 0:
                obj += penalty_weight * val * val
            
            val = centers[i, dim] + radii[i] - 1.0
            if val > 0:
                obj += penalty_weight * val * val
    
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq + 1e-30)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                obj += penalty_weight * overlap * overlap
    
    return obj


def create_hexagonal_initial():
    centers = []
    count = 0
    row = 0
    
    while count < N:
        y = 0.06 + row * 0.16
        if y > 0.94:
            break
        
        col = 0
        x_start = 0.06 if row % 2 == 0 else 0.13
        
        while count < N:
            x = x_start + col * 0.16
            if x > 0.94:
                break
            centers.append([x, y])
            count += 1
            col += 1
        row += 1
    
    centers = np.array(centers[:N])
    radii = np.full(N, 0.06)
    return np.concatenate([centers.flatten(), radii])

    b = []
    for _ in range(N):
        b.append((0.0, 1.0))
        b.append((0.0, 1.0))
        b.append((1e-10, 0.5))
    return b


def force_based_preoptimize(centers, radii, num_steps=3000):
    centers = centers.copy()
    radii = radii.copy()
    
    for step in range(num_steps):
        for i in range(N):
            radii[i] = radii[i] + 1.5e-5
        
        forces = np.zeros((N, 2))
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq + 1e-30)
                
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    if dist > 1e-10:
                        fx = overlap * dx / dist
                        fy = overlap * dy / dist
                        forces[i, 0] += fx
                        forces[i, 1] += fy
                        forces[j, 0] -= fx
                        forces[j, 1] -= fy

                    forces[i, dim] -= (centers[i, dim] + radii[i] - 1.0) * 50.0

        centers = centers + lr * forces

            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    
    return centers, radii


def check_validity(centers, radii):
    for i in range(N):
        for dim in range(2):
            if centers[i, dim] - radii[i] < -1e-11:
                return False
            if centers[i, dim] + radii[i] > 1.0 + 1e-11:
                return False
    
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-11:
                return False
    
    return True

            best_radii[i] = max_r - 2e-13
    
    return best_centers, best_radii, best_sum
