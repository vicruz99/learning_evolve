import numpy as np
from scipy.optimize import minimize


def objective_func(x):
    """Minimize negative sum of radii."""
    n = len(x) // 3
    radii = x[2::3]
    return -np.sum(radii)


def constraint_x_ge_r(x, i):
    """x_i >= r_i"""
    return x[3 * i] - x[3 * i + 2]


def constraint_one_minus_x_ge_r(x, i):
    """1 - x_i >= r_i"""
    return 1.0 - x[3 * i] - x[3 * i + 2]


def constraint_y_ge_r(x, i):
    """y_i >= r_i"""
    return x[3 * i + 1] - x[3 * i + 2]


def constraint_one_minus_y_ge_r(x, i):
    """1 - y_i >= r_i"""
    return 1.0 - x[3 * i + 1] - x[3 * i + 2]


def constraint_r_nonneg(x, i):
    """r_i >= 0"""
    return x[3 * i + 2]


def constraint_no_overlap(x, i, j):
    """Distance between centers >= sum of radii"""
    xi = x[3 * i]
    yi = x[3 * i + 1]
    ri = x[3 * i + 2]
    xj = x[3 * j]
    yj = x[3 * j + 1]
    rj = x[3 * j + 2]
    dx = xi - xj
    dy = yi - yj
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - ri - rj


def generate_hexagonal_config(n, row_counts, seed=0):
    """Generate a hexagonal lattice initial configuration."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    idx = 0
    r_approx = 0.095
    y_spacing = r_approx * np.sqrt(3)
    x_margin = r_approx

    for row_i, count in enumerate(row_counts):
        if count == 0:
            continue
        x_spacing = 1.0 / (count + 1)
        y = r_approx + row_i * y_spacing

        for col_i in range(count):
            x = x_spacing * (col_i + 1)
            if row_i % 2 == 1:
                x += x_spacing / 2.0

            # Add perturbation
            x += rng.uniform(-0.02, 0.02)
            y += rng.uniform(-0.02, 0.02)

            # Clamp to valid range
            x = np.clip(x, 0.05, 0.95)
            y = np.clip(y, 0.05, 0.95)

            centers[idx] = [x, y]
            radii[idx] = min(x, 1.0 - x, y, 1.0 - y) * 0.85
            idx += 1

    return centers, radii


def generate_grid_config(n, seed=0):
    """Generate a grid-based initial configuration."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Try to place in a roughly sqrt(n) x sqrt(n) grid
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))

    x_step = 1.0 / (cols + 1)
    y_step = 1.0 / (rows + 1)

    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n:
                break
            x = x_step * (c + 1)
            y = y_step * (r + 1)

            x += rng.uniform(-0.02, 0.02)
            y += rng.uniform(-0.02, 0.02)

            x = np.clip(x, 0.05, 0.95)
            y = np.clip(y, 0.05, 0.95)

            centers[idx] = [x, y]
            radii[idx] = min(x, 1.0 - x, y, 1.0 - y) * 0.85
            idx += 1

    return centers, radii


def build_constraints(n):
    """Build the constraint list for optimization."""
    cons = []

    # Boundary and non-negativity constraints
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': constraint_x_ge_r, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constraint_one_minus_x_ge_r, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constraint_y_ge_r, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constraint_one_minus_y_ge_r, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constraint_r_nonneg, 'args': (i,)})

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': constraint_no_overlap, 'args': (i, j)})

    return cons


def build_bounds(n):
    """Build bounds for optimization variables."""
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))   # x
        bounds.append((0.0, 1.0))   # y
        bounds.append((0.0, 0.5))   # r
    return bounds


def optimize_from_initial(n, centers, radii):
    """Run optimization from a given initial configuration."""
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]

    cons = build_constraints(n)
    bounds = build_bounds(n)

    result = minimize(
        objective_func,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False}
    )

    co = np.array([[result.x[3 * i], result.x[3 * i + 1]] for i in range(n)])
    ro = np.array([result.x[3 * i + 2] for i in range(n)])
    return co, ro, result
