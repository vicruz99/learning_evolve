from scipy.optimize import minimize

# Global constants
N_CIRCLES = 26
DIM = 2
VAR_PER_CIRCLE = 3 # x, y, r
TOTAL_VARS = N_CIRCLES * VAR_PER_CIRCLE

def get_initial_guess():
    """
    Generates an initial feasible configuration.
    Uses a perturbed grid layout to ensure circles are spread out.
    """
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, 0.01) # Small initial radius to ensure feasibility

    # We will select the first 26 points from this grid and add some noise.

    rows = 5

    points = []

        for x in grid_x:
            points.append([x, y])

def objective(x_vars):
    """
    Objective function: Minimize negative sum of radii.
    x_vars: array of shape (TOTAL_VARS,)
    Structure: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    """
    radii = x_vars[2::3] # Every 3rd element starting from index 2
    return -np.sum(radii)

def compute_constraints(x_vars):
    """
    Computes inequality constraints.
    Returns a numpy array of constraint values. 
    All constraints must be >= 0.
    
    Constraints:
    1. Boundary: x_i - r_i >= 0
    2. Boundary: (1 - x_i) - r_i >= 0
    3. Boundary: y_i - r_i >= 0
    4. Boundary: (1 - y_i) - r_i >= 0
    5. Non-overlap: dist(i,j)^2 - (r_i + r_j)^2 >= 0
    """
    constraints = []
    
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.zeros(N_CIRCLES)
    
    # Unpack variables
    for i in range(N_CIRCLES):
        base_idx = i * 3
        centers[i, 0] = x_vars[base_idx]     # x
        centers[i, 1] = x_vars[base_idx + 1] # y
        radii[i] = x_vars[base_idx + 2]      # r
        
    # 1. Boundary Constraints
    for i in range(N_CIRCLES):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        constraints.append(x - r)       # Left wall
        constraints.append((1 - x) - r) # Right wall
        constraints.append(y - r)       # Bottom wall
        constraints.append((1 - y) - r) # Top wall
        
    # 2. Non-overlap Constraints
    # Check all pairs (i, j) with i < j
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            sum_r = radii[i] + radii[j]
            # Constraint: dist_sq >= sum_r^2  => dist_sq - sum_r^2 >= 0
            constraints.append(dist_sq - sum_r*sum_r)
            
    return np.array(constraints)
