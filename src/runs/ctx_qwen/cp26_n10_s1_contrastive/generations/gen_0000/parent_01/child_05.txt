import numpy as np
from scipy.optimize import minimize
import math

N = 26


def objective(params):
    """Objective: minimize negative sum of radii (i.e., maximize sum of radii)."""
    rs = params[2::3]
    return -np.sum(rs)


def constraint_vector(params):
    """Compute all constraints as a single array (must be >= 0)."""
    xs = params[0::3]
    ys = params[1::3]
    rs = params[2::3]

    # Boundary constraints: x >= r, x + r <= 1, y >= r, y + r <= 1, r >= 0
    vec = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs,
        rs
    ])

    # Overlap constraints: distance(i,j) >= r_i + r_j for all i < j
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = xs[i_idx] - xs[j_idx]
    dy = ys[i_idx] - ys[j_idx]
    dists = np.sqrt(dx * dx + dy * dy)
    vec = np.concatenate([vec, dists - rs[i_idx] - rs[j_idx]])

    return vec


def create_hexagonal_initial():
    """Create initial parameters using hexagonal grid layout."""
    params = np.zeros(3 * N)
    r = 0.075
    idx = 0

    for row in range(7):
        n_in_row = 5 if row % 2 == 0 else 6
        if idx + n_in_row > N:
            n_in_row = N - idx
        for col in range(n_in_row):
            x = 0.12 + col * 0.16 + (row % 2) * 0.08
            y = 0.12 + row * 0.145
            x = max(0.05, min(0.95, x))
            y = max(0.05, min(0.95, y))
            params[3 * idx] = x
            params[3 * idx + 1] = y
            params[3 * idx + 2] = r
            idx += 1
            if idx >= N:
                break
        if idx >= N:
            break

    return params


def create_random_initial():
    """Create initial parameters using random placement with small radii."""
    params = np.zeros(3 * N)
    r = 0.05

    # Place in a perturbed grid
    idx = 0
    for row in range(5):
        for col in range(6):
            if idx >= N:
                break
            x = 0.15 + col * 0.14 + 0.02 * (row % 2)
            y = 0.15 + row * 0.18
            x = max(0.06, min(0.94, x))
            y = max(0.06, min(0.94, y))
            params[3 * idx] = x
            params[3 * idx + 1] = y
            params[3 * idx + 2] = r
            idx += 1

    return params


def force_based_prepacking(params, n_iterations=20000):
    """Use force-based simulation to improve initial packing before SLSQP."""
    xs = params[0::3].copy()
    ys = params[1::3].copy()
    rs = params[2::3].copy()

    dt = 0.05
    grow_rate = 1.0 + (0.15 - 1.0) / n_iterations  # Grow radii by 15% total

    for iteration in range(n_iterations):
        forces_x = np.zeros(N)
        forces_y = np.zeros(N)

        # Repulsion between overlapping circles
        for i in range(N):
            for j in range(i + 1, N):
                dx = xs[j] - xs[i]
                dy = ys[j] - ys[i]
                dist_sq = dx * dx + dy * dy
                dist = math.sqrt(dist_sq) if dist_sq > 0 else 1e-10

                min_dist = rs[i] + rs[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    fx = dx / dist * overlap
                    fy = dy / dist * overlap
                    forces_x[i] -= fx
                    forces_y[i] -= fy
                    forces_x[j] += fx
                    forces_y[j] += fy

        # Boundary repulsion
        for i in range(N):
            r = rs[i]
            if xs[i] - r < 0:
                forces_x[i] += (r - xs[i]) * 100
            if xs[i] + r > 1:
                forces_x[i] -= (xs[i] + r - 1) * 100
            if ys[i] - r < 0:
                forces_y[i] += (r - ys[i]) * 100
            if ys[i] + r > 1:
                forces_y[i] -= (ys[i] + r - 1) * 100

        # Update positions
        xs += forces_x * dt
        ys += forces_y * dt

        # Clamp positions
        xs = np.clip(xs, 0.01, 0.99)
        ys = np.clip(ys, 0.01, 0.99)

        # Grow radii
        if iteration % 50 == 0:
            rs *= grow_rate

    # Reconstruct params
    new_params = np.zeros(3 * N)
    new_params[0::3] = xs
    new_params[1::3] = ys
    new_params[2::3] = rs

    return new_params


def optimize_packing(params, max_iter=3000):
    """Run SLSQP optimization on given parameters."""
    bounds = []
    for i in range(N):
        bounds.append((0.001, 0.999))  # x
        bounds.append((0.001, 0.999))  # y
        bounds.append((0.001, 0.5))    # r

    cons = {'type': 'ineq', 'fun': constraint_vector}

    result = minimize(
        objective,
        params,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={
            'maxiter': max_iter,
            'ftol': 1e-14,
            'disp': False
        }
    )

    return result


def run_packing():
    best_sum = -1.0
    best_centers = np.zeros((N, 2))
    best_radii = np.zeros(N)

    # Strategy 1: Hexagonal initial with multiple perturbations
    for seed in range(8):
        np.random.seed(seed)
        params = create_hexagonal_initial()

        # Add random perturbation
        for i in range(N):
            params[3 * i] += np.random.uniform(-0.03, 0.03)
            params[3 * i + 1] += np.random.uniform(-0.03, 0.03)

        # Force-based pre-optimization
        params = force_based_prepacking(params, n_iterations=8000)

        # SLSQP fine-tuning
        result = optimize_packing(params, max_iter=3000)

        current_sum = -result.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = result.x[:2 * N].reshape(N, 2).copy()
            best_radii = result.x[2 * N:3 * N].copy()

    # Strategy 2: Random initial with multiple perturbations
    for seed in range(8):
        np.random.seed(100 + seed)
        params = create_random_initial()

        for i in range(N):
            params[3 * i] += np.random.uniform(-0.04, 0.04)
            params[3 * i + 1] += np.random.uniform(-0.04, 0.04)

        params = force_based_prepacking(params, n_iterations=8000)

        result = optimize_packing(params, max_iter=3000)

        current_sum = -result.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = result.x[:2 * N].reshape(N, 2).copy()
            best_radii = result.x[2 * N:3 * N].copy()

    # Final validation
    valid = True
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10 or r < 0:
            valid = False
            break

    if valid:
        for i in range(N):
            for j in range(i + 1, N):
                dist = math.sqrt(
                    (best_centers[i, 0] - best_centers[j, 0]) ** 2 +
                    (best_centers[i, 1] - best_centers[j, 1]) ** 2
                )
                if dist < best_radii[i] + best_radii[j] - 1e-10:
                    valid = False
                    break
            if not valid:
                break

    if not valid:
        # Fallback to known valid packing
        best_centers, best_radii, best_sum = simple_valid_packing()

    return best_centers, best_radii, best_sum


def simple_valid_packing():
    """Generate a simple valid packing as fallback."""
    centers = np.zeros((N, 2))
    radii = np.ones(N) * 0.06

    idx = 0
    for row in range(5):
        for col in range(6):
            if idx >= N:
                break
            centers[idx, 0] = 0.15 + col * 0.15
            centers[idx, 1] = 0.15 + row * 0.18
            idx += 1

    return centers, radii, np.sum(radii)
