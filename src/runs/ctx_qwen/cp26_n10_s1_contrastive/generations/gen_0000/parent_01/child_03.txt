import numpy as np
from scipy.optimize import minimize

def _compute_loss(vars, n):
    """
    Computes the objective: -sum(r) + penalty(overlaps) + penalty(walls)
    Uses a smooth max approximation for gradient continuity.
    """
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    # Pairwise distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d_sq = dx**2 + dy**2
    np.fill_diagonal(d_sq, 1.0)
    d = np.sqrt(d_sq)
    
    overlap = r[:, None] + r[None, :] - d
    eps = 1e-4
    # Smooth approximation of max(0, overlap)
    smooth_overlap = (overlap + np.sqrt(overlap**2 + eps**2)) / 2.0
    penalty_pairs = np.sum(smooth_overlap**2)
    
    # Wall constraints: circle i must be within [r_i, 1-r_i] in both axes
    overlap_x = np.maximum(0.0, r - x) + np.maximum(0.0, r - (1.0 - x))
    overlap_y = np.maximum(0.0, r - y) + np.maximum(0.0, r - (1.0 - y))
    penalty_wall = np.sum(overlap_x**2 + overlap_y**2)
    
    lam = 10000.0
    return -np.sum(r) + lam * (penalty_pairs + penalty_wall)

def run_packing():
    n = 26
    rng = np.random.RandomState(42)
    best_loss = 1e9
    best_vars = None
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice initialization
    coords_hex = []
    r_init = 0.08
    y_pos = r_init
    row = 0
    while len(coords_hex) < n:
        x_pos = r_init + (0.5 * r_init if row % 2 == 1 else 0.0)
        while x_pos + r_init <= 1.0 and len(coords_hex) < n:
            coords_hex.append([x_pos, y_pos])
            x_pos += 2.0 * r_init
        y_pos += np.sqrt(3.0) * r_init
        row += 1
    inits.append(np.array(coords_hex))
    
    # 2. Random scattered initialization
    inits.append(np.column_stack([rng.uniform(0.2, 0.8, n), rng.uniform(0.2, 0.8, n)]))
    
    # 3. Grid 5x5 + 1 initialization
    coords_grid = []
    for i in range(5):
        for j in range(5):
            coords_grid.append([0.1 + 0.2 * i, 0.1 + 0.2 * j])
    coords_grid.append([0.5, 0.5])
    inits.append(np.array(coords_grid))
    
    # Optimize from each initialization with random perturbations
    for init_coords in inits:
        for _ in range(8):
            x0 = init_coords[:, 0].copy() + rng.uniform(-0.03, 0.03, n)
            y0 = init_coords[:, 1].copy() + rng.uniform(-0.03, 0.03, n)
            x0 = np.clip(x0, 0.05, 0.95)
            y0 = np.clip(y0, 0.05, 0.95)
            r0 = np.full(n, 0.08) + rng.uniform(0, 0.02, n)
            r0 = np.clip(r0, 0.02, 0.25)
            
            vars0 = np.concatenate([x0, y0, r0])
            bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
            
            res = minimize(_compute_loss, vars0, args=(n,), method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-12})
            
            if res.fun < best_loss:
                best_loss = res.fun
                best_vars = res.x
                
    # Extract and clean results
    x = best_vars[:n]
    y = best_vars[n:2*n]
    r = best_vars[2*n:]
    
    # Enforce hard constraints strictly
    r = np.maximum(r, 1e-5)
    x = np.clip(x, r, 1.0 - r)
    y = np.clip(y, r, 1.0 - r)
    
    centers = np.column_stack([x, y])
    return centers, r, np.sum(r)
