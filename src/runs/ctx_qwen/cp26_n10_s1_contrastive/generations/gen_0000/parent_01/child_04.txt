def compute_penalty(centers_flat, r, n):
    """
    Compute the penalty for a given configuration of centers and radius r.
    Penalty is sum of squared violations of overlap and boundary constraints.
    """
    centers = centers_flat.reshape(-1, 2)
    
    # 1. Inter-circle overlap penalty
    # Calculate pairwise distances
    # diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_sq = np.sum(diff**2, axis=2)
    # Add small epsilon to avoid division by zero or sqrt(0) issues, though max(0) handles 0
    dists = np.sqrt(dists_sq + 1e-12)
    
    # We only care about i < j to avoid double counting and self-distance
    # Use upper triangle mask
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pair_dists = dists[mask]
    
    # Constraint: dist >= 2r  =>  2r - dist <= 0
    # Violation is max(0, 2r - dist)
    # Penalty is violation^2
    overlap_violations = 2.0 * r - pair_dists
    pos_violations = np.maximum(0.0, overlap_violations)
    penalty = np.sum(pos_violations**2)
    
    # 2. Boundary penalty
    # Constraint: r <= x <= 1-r  and  r <= y <= 1-r
    # This implies dist to any wall >= r
    # Walls at x=0, x=1, y=0, y=1
    
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Distances to walls
    dist_left = x
    dist_right = 1.0 - x
    dist_bottom = y
    dist_top = 1.0 - y
    
    # Minimum distance to any wall
    min_dist = np.minimum(np.minimum(dist_left, dist_right), np.minimum(dist_bottom, dist_top))
    
    # Constraint: min_dist >= r  =>  r - min_dist <= 0
    wall_violations = r - min_dist
    pos_wall_violations = np.maximum(0.0, wall_violations)
    penalty += np.sum(pos_wall_violations**2)
    
    return penalty
