def objective(x_vec):
    # x_vec contains [x1, y1, r1, x2, y2, r2, ...]
    # We want to maximize sum(r), so minimize -sum(r)
    n = len(x_vec) // 3
    r_sum = 0.0
    for i in range(n):
        r_sum += x_vec[3*i + 2]
    return -r_sum

def constraint_boundary(x_vec, n):
    # Constraints: 
    # x - r >= 0  => x - r
    # 1 - x - r >= 0 => 1 - x - r
    # y - r >= 0 => y - r
    # 1 - y - r >= 0 => 1 - y - r
    con = np.zeros(4 * n)
    idx = 0
    for i in range(n):
        x = x_vec[3*i]
        y = x_vec[3*i + 1]
        r = x_vec[3*i + 2]
        con[idx] = x - r
        con[idx+1] = 1.0 - x - r
        con[idx+2] = y - r
        con[idx+3] = 1.0 - y - r
        idx += 4
    return con

def constraint_overlap(x_vec, n):
    # Constraints: dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    num_constraints = 0
    for i in range(n):
