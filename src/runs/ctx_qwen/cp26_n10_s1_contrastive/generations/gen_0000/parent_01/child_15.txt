

<strategy>
This problem is a non-convex geometric optimization. Directly optimizing centers and radii jointly with penalty methods often gets trapped in local optima or requires careful tuning. 

A highly effective strategy is to decompose the problem into a bi-level optimization:
1. **Outer Level (Centers)**: Use `Differential Evolution` to globally search for the optimal center coordinates $(x_i, y_i)$. This algorithm is robust for non-convex landscapes and handles the unit square bounds naturally.
2. **Inner Level (Radii)**: For any fixed set of centers, finding the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints is a **Linear Programming** problem. The constraints $r_i + r_j \le \text{dist}(i,j)$ and $r_i \le \text{dist}(i, \text{boundary})$ are linear in $r$. Solving this LP guarantees the best possible radii for any given configuration.
3. **Refinement**: After DE finds a promising configuration, a local `Nelder-Mead` optimization polishes the centers further.

This approach elegantly bypasses penalty parameter tuning and leverages the convexity of the radius subproblem, consistently yielding near-optimal packings.
</strategy>

```python
import numpy as np
from scipy.optimize import differential_evolution, linprog, minimize
import warnings
warnings.filterwarnings('ignore')

def get_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Maximum radius allowed by square boundaries
    b_i = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    b_i = np.maximum(b_i, 0.0)
    
    # Pairwise Euclidean distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, 1.0)  # Prevent self-constraint issues
    
    # LP: Minimize -sum(r_i)  <=>  Maximize sum(r_i)
    c = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    n_pairs = n * (n - 1) // 2
    n_constraints = n_pairs + n
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints: r_i <= b_i
    A_ub[n_pairs:, :] = np.eye(n)
    b_ub[n_pairs:] = b_i
    
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='revised simplex')
        
    if res.success:
        return res.x
    else:
        return np.full(n, 0.01)

def de_objective(centers_flat):
    """Objective for Differential Evolution: negative sum of optimal radii."""
    n = 26
    centers = centers_flat.reshape((n, 2))
    centers = np.clip(centers, 0.0, 1.0)
    radii = get_lp_radii(centers)
    return -np.sum(radii)

def run_packing():
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # 1. Global search for centers using Differential Evolution
    de_result = differential_evolution(
        de_objective, 
        bounds, 
        popsize=20, 
        maxiter=150, 
        tol=1e-8, 
        mutation=(0.5, 1.0), 
        recombination=0.8, 
        seed=42,
        disp=False
    )
    
    best_centers = de_result.x.reshape((n, 2))
    
    # 2. Local refinement to polish the configuration
    local_res = minimize(
        de_objective, 
        de_result.x, 
        method='Nelder-Mead', 
        options={'xatol': 1e-9, 'fatol': 1e-9, 'maxiter': 1000}
    )
    best_centers = local_res.x.reshape((n, 2))
    
    # 3. Compute final optimal radii via LP
    best_radii = get_lp_radii(best_centers)
    
    # Numerical safeguard: slight shrink to guarantee strict inequality for validation
    epsilon = 1e-11
    best_radii = np.maximum(best_radii - epsilon, 0.0)
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    total_sum = np.sum(best_radii)
    
    return (best_centers, best_radii, total_sum)
```
