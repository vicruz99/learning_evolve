from scipy.optimize import minimize


def compute_overlap_penalty(centers, radii, mu):
    n = len(radii)
    penalty = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += mu * (x - r) ** 2
        if x + r > 1:
            penalty += mu * (x + r - 1) ** 2
        if y - r < 0:
            penalty += mu * (y - r) ** 2
        if y + r > 1:
            penalty += mu * (y + r - 1) ** 2
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += mu * (min_dist - dist) ** 2
    return penalty


def force_based_optimization(centers, radii, num_iters=5000):

    centers = centers.copy()
    radii = radii.copy()

    for it in range(num_iters):
        forces = np.zeros((n, 2))
        alpha = max(0.0005, 0.01 / (1 + it * 0.001))

        for i in range(n):
            for d in range(2):
                if centers[i, d] - radii[i] < 0:
                    forces[i, d] += 50 * alpha * (centers[i, d] - radii[i])
                if centers[i, d] + radii[i] > 1:
                    forces[i, d] -= 50 * alpha * (centers[i, d] + radii[i] - 1)

                    forces[j] -= diff * force_mag

        centers += forces

def hexagonal_init(n, r):
    centers = np.zeros((n, 2))
    radii = np.full(n, r)
    row_configs = [6, 5, 6, 5, 4]
    idx = 0
    for row in range(5):
        n_in_row = row_configs[row]
        for col in range(n_in_row):
            if idx >= n:
                break
            x = r + col * 2 * r
            if row % 2 == 1:
                x += r
            y = r + row * r * np.sqrt(3)
            centers[idx] = [x, y]
            radii[idx] = r
            idx += 1
    while idx < n:
        centers[idx] = [0.5, 0.5]
        radii[idx] = r * 0.5
        idx += 1
    return centers, radii


def grid_init(n, r):
    cols = 6
    rows = n // cols + 1
    centers = np.zeros((n, 2))
    radii = np.full(n, r)
    spacing_x = (1 - 2 * r) / (cols - 1) if cols > 1 else 1 - 2 * r
    spacing_y = (1 - 2 * r) / (rows - 1) if rows > 1 else 1 - 2 * r
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = r + col * spacing_x
            y = r + row * spacing_y
            centers[idx] = [x, y]
            radii[idx] = r
            idx += 1
    return centers, radii


def optimize_with_scipy(centers, radii, mu=5000, maxiter=200):
    n = len(radii)

    def objective(params):
        r = params[0]
        c = params[1:].reshape((n, 2))
        penalty = 0.0
        for i in range(n):
            for d in range(2):
                if c[i, d] - r < 0:
                    penalty += mu * (c[i, d] - r) ** 2
                if c[i, d] + r > 1:
                    penalty += mu * (c[i, d] + r - 1) ** 2
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                if dist < 2 * r:
                    penalty += mu * (2 * r - dist) ** 2
        return -r + penalty

    params0 = np.concatenate([[radii[0]], centers.flatten()])
    bounds = [(0.01, 0.3)] + [(0.0, 1.0)] * (2 * n)

    result = minimize(objective, params0, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15})

    r_opt = result.x[0]
    c_opt = result.x[1:].reshape((n, 2))

    radii_opt = np.full(n, r_opt)
    centers_opt = c_opt
    return centers_opt, radii_opt

def scipy_optimize_variable(centers, radii, mu=8000, maxiter=500):

    def objective(params):
        c = params[:2 * n].reshape((n, 2))
        r = params[2 * n:]
        penalty = 0.0
        for i in range(n):
            for d in range(2):
                if c[i, d] - r[i] < 0:
                    penalty += mu * (c[i, d] - r[i]) ** 2
                if c[i, d] + r[i] > 1:
                    penalty += mu * (c[i, d] + r[i] - 1) ** 2
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                if dist < r[i] + r[j]:
                    penalty += mu * (r[i] + r[j] - dist) ** 2
        return -np.sum(r) + penalty

    params0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.001, 0.3)] * n

    result = minimize(objective, params0, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15})

    c_opt = result.x[:2 * n].reshape((n, 2))
    r_opt = result.x[2 * n:]
    return c_opt, r_opt
