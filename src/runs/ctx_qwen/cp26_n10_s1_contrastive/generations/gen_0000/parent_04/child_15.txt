import numpy as np

def compute_forces_and_update(centers, radii, dt, k_rep, k_bound, damping):
    n = len(centers)
    forces = np.zeros_like(centers)
    
    # Pairwise interactions
    diff = centers[:, None, :] - centers[None, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    dist_sq = np.maximum(dist_sq, 1e-12)
    dist = np.sqrt(dist_sq)
    
    radii_sum = radii[:, None] + radii[None, :]
    overlap = radii_sum - dist
    
    # Mask for overlapping pairs, excluding self-interaction
    mask = (overlap > 0) & (~np.eye(n, dtype=bool))
    
    dir_vec = diff / dist[:, :, None]
    force_mag = overlap * k_rep
    
    # Accumulate repulsive forces efficiently
    forces += np.sum((force_mag[:, :, None] * dir_vec) * mask[:, :, None], axis=1)
    
    # Boundary forces
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        if x - r < 0:
            forces[i, 0] += (r - x) * k_bound
        if x + r > 1:
            forces[i, 0] -= (x + r - 1) * k_bound
        if y - r < 0:
            forces[i, 1] += (r - y) * k_bound
        if y + r > 1:
            forces[i, 1] -= (y + r - 1) * k_bound
            
    centers += forces * dt * damping
    centers = np.clip(centers, 0, 1)
    
    return centers
