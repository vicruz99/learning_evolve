import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal lattice initialization and SLSQP optimization.
    """
    n_circles = 26
    
    # ---------------------------------------------------------
    # 1. Initialization: Hexagonal Lattice
    # ---------------------------------------------------------
    # We distribute 26 circles into 5 rows with counts [5, 6, 5, 6, 4]
    # This pattern is dense and fits well within a square.
    row_counts = [5, 6, 5, 6, 4]
    
    # Estimate initial radius based on area density of hexagonal packing
    # Density ~ 0.9069. Area per circle ~ 1/26. 
    # pi * r^2 ~ (1/26) * 0.9069 => r ~ 0.105
    r_init = 0.105
    
    centers = []
    y_current = r_init
    
    for i, count in enumerate(row_counts):
        # Calculate x positions for this row
        # Stagger every other row (index 1, 3, ...) by half the horizontal spacing
        horizontal_spacing = 2 * r_init
        offset = (horizontal_spacing / 2) if (i % 2 == 1) else 0
        
        # Calculate total width of this row to center it
        row_width = (count - 1) * horizontal_spacing
        start_x = (1 - row_width) / 2 + offset
        
        # Clamp start_x to ensure it's positive
        start_x = max(r_init, start_x)
        
        for j in range(count):
            cx = start_x + j * horizontal_spacing
            # Clamp cx to [r_init, 1 - r_init]
            cx = np.clip(cx, r_init, 1 - r_init)
            centers.append([cx, y_current])
        
        # Move to next row
        # Vertical spacing for hexagonal packing is r * sqrt(3)
        y_current += r_init * np.sqrt(3)
        
        # If we run out of vertical space, stop (though row_counts sums to 26)
        if y_current + r_init > 1.0:
            break
            
    # Convert to numpy array
    centers = np.array(centers[:n_circles])
    
    # Initial radii vector
    radii = np.full(n_circles, r_init)
    
    # Combine into a single optimization variable vector: [x1, y1, r1, x2, y2, r2, ...]
    # Order: x, y, r for each circle
    x0 = np.concatenate([centers[:, 0], centers[:, 1], radii])
    
    # ---------------------------------------------------------
    # 2. Optimization Setup
    # ---------------------------------------------------------
    
    # Bounds for variables
    # x in [0, 1], y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n_circles):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((1e-6, 0.5)) # r (small epsilon to avoid zero radius issues)
        
    # Constraint function
    # We need to maximize sum(r), so we minimize -sum(r)
    # Constraints:
    # 1. Non-overlap: dist^2 >= (ri + rj)^2
    # 2. Boundary: r <= x <= 1-r, r <= y <= 1-r
    
    def objective(vars_vec):
        # vars_vec shape: (n_circles * 3,)
        # Extract r's: indices 2, 5, 8, ...
        r = vars_vec[2::3]
        return -np.sum(r)

    def constraint_func(vars_vec):
        # Returns a list of constraint values.
        # For inequality constraints g(x) >= 0, scipy expects 'ineq' type.
        
        constraints_list = []
        
        # Extract variables
        x = vars_vec[0::3]
        y = vars_vec[1::3]
        r = vars_vec[2::3]
        
        # Boundary constraints
        # r <= x <= 1-r  =>  x - r >= 0  AND  1 - r - x >= 0
        constraints_list.extend(x - r)
        constraints_list.extend(1 - r - x)
        
        # r <= y <= 1-r  =>  y - r >= 0  AND  1 - r - y >= 0
        constraints_list.extend(y - r)
        constraints_list.extend(1 - r - y)
        
        # Non-overlap constraints
        # dist^2 >= (ri + rj)^2
        # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dr = r[i] + r[j]
                val = (dx**2 + dy**2) - (dr**2)
                constraints_list.append(val)
                
        return np.array(constraints_list)

    # Define constraints for SLSQP
    # SLSQP handles vectorized constraints well if passed as a dict with 'fun'
    # However, for clarity and stability with many constraints, we can pass them 
    # individually or as a vector. SLSQP 'ineq' expects a vector.
    
    cons = ({'type': 'ineq', 'fun': constraint_func})
    
    # ---------------------------------------------------------
    # 3. Run Optimization
    # ---------------------------------------------------------
    
    # We might need multiple restarts to escape local minima, 
    # but with a good initialization, one run might suffice or get close.
    # Given the complexity, we run one robust optimization.
    
    try:
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons, 
            options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
        )
        
        if res.success:
            # Extract results
            opt_x = res.x[0::3]
            opt_y = res.x[1::3]
            opt_r = res.x[2::3]
            
            # Reconstruct centers and radii
            final_centers = np.column_stack((opt_x, opt_y))
            final_radii = opt_r
            final_sum = np.sum(final_radii)
            
            return final_centers, final_radii, final_sum
        
        else:
            # If optimization fails, return the initial configuration
            # It's better to return a valid packing than fail
            return centers, radii, np.sum(radii)
            
    except Exception as e:
        # Fallback
        return centers, radii, np.sum(radii)
