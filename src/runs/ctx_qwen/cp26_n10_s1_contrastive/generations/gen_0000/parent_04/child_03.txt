import numpy as np

def compute_forces_and_energy(centers, r):
    N = centers.shape[0]
    forces = np.zeros_like(centers)
    energy = 0.0
    
    i_indices, j_indices = np.triu_indices(N, k=1)
    for i, j in zip(i_indices, j_indices):
        diff = centers[i] - centers[j]
        dist_sq = np.sum(diff**2)
        dist = np.sqrt(max(dist_sq, 1e-12))
        overlap = 2.0 * r - dist
        if overlap > 0.0:
            f_mag = overlap / dist
            forces[i] += f_mag * diff
            forces[j] -= f_mag * diff
            energy += overlap * overlap
            
    for i in range(N):
        x, y = centers[i]
        if x < r:
            ov = r - x
            forces[i, 0] += ov
            energy += ov * ov
        elif x > 1.0 - r:
            ov = x - (1.0 - r)
            forces[i, 0] -= ov
            energy += ov * ov
            
        if y < r:
            ov = r - y
            forces[i, 1] += ov
            energy += ov * ov
        elif y > 1.0 - r:
            ov = y - (1.0 - r)
            forces[i, 1] -= ov
            energy += ov * ov
            
    return forces, energy
