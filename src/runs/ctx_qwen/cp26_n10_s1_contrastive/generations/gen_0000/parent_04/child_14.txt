import numpy as np
from scipy.optimize import minimize

def objective(params, n):
    """
    Penalty-based objective to maximize sum of radii.
    params: flattened array of [x1, y1, ..., xn, yn, r1, ..., rn]
    """
    C = params[:2*n].reshape(n, 2)
    R = params[2*n:]
    
    # Compute pairwise distances
    diff = C[:, np.newaxis, :] - C[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Ignore self-distances
    dists[np.arange(n), np.arange(n)] = np.inf
    
    # Overlap penalty: penalize when r_i + r_j > dist(i,j)
    R_sum = R[:, None] + R[None, :]
    overlap_pen = np.sum(np.maximum(R_sum - dists, 0.0)**2)
    
    # Boundary penalty: penalize when circles exceed [0,1] square
    bound_pen = 0.0
    bound_pen += np.sum(np.maximum(R - C[:, 0], 0.0)**2)          # x - r < 0
    bound_pen += np.sum(np.maximum(R - (1.0 - C[:, 0]), 0.0)**2)  # x + r > 1
    bound_pen += np.sum(np.maximum(R - C[:, 1], 0.0)**2)          # y - r < 0
    bound_pen += np.sum(np.maximum(R - (1.0 - C[:, 1]), 0.0)**2)  # y + r > 1
    
    # Large weight ensures constraints are respected relative to radius scale
    return -np.sum(R) + 2000.0 * (overlap_pen + bound_pen)

def run_packing():

    centers = np.zeros((n, 2))

    dy = 0.173205  # ~ sqrt(3)/2 * 0.2
    dx = 0.2
    y = 0.5 - (len(row_counts) - 1) * dy / 2
    
    for i, count in enumerate(row_counts):
        offset = (dx / 2) if i % 2 == 1 else 0
        x_start = 0.5 - (count - 1) * dx / 2 + offset
        for j in range(count):
            centers[idx, 0] = x_start + j * dx
            centers[idx, 1] = y
            idx += 1
        y += dy
        
    # Start with a feasible small radius
    radii = np.full(n, 0.09)
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n

    res = minimize(objective, x0, args=(n,), bounds=bounds, method='L-BFGS-B',

    best_radii = res.x[2*n:]

    best_radii = np.maximum(best_radii, 1e-9)
    best_centers = np.clip(best_centers, 1e-9, 1.0 - 1e-9)
    
    return best_centers, best_radii, float(np.sum(best_radii))
