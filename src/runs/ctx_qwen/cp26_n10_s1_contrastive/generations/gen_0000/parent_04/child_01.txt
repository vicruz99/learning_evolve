import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # 1. Initialization
    # Start with a small radius to ensure a valid initial configuration
    r = 0.01
    
    # Initialize centers on a 6x5 grid to ensure no initial overlaps
    centers = []
    for i in range(6):
        for j in range(5):
            x = (i + 0.5) / 6.0
            y = (j + 0.5) / 5.0
            centers.append([x, y])
    centers = np.array(centers[:n])
    
    # Precompute pairs for the objective function
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

    # 2. Objective Function
    def overlap_penalty(vars, r_val):
        c = vars.reshape(n, 2)
        penalty = 0.0
        for i, j in pairs:
            dist = np.sqrt(np.sum((c[i] - c[j]) ** 2))
            overlap = 2 * r_val - dist
            if overlap > 0:
                penalty += overlap ** 2
        return penalty

    # 3. Iterative Growth and Relaxation
    step = 0.0001
    max_r = r
    
    for _ in range(2000):
        r += step
        
        # Define bounds for centers: [r, 1-r] for both x and y
        bounds = [(r, 1 - r)] * (2 * n)
        
        # Optimize centers to minimize overlap for the current radius
        res = minimize(overlap_penalty, centers.flatten(), args=(r,), method='L-BFGS-B', bounds=bounds)
        
        # Check if the configuration is feasible (overlap is negligible)
        if res.fun < 1e-8:
            centers = res.x.reshape(n, 2)
            max_r = r
        else:
            # If overlaps cannot be resolved, revert the last increase and stop
            r -= step
            break
            
    # 4. Final Output
    # Radii are all equal to the maximum found radius
    radii = np.full(n, max_r)
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
