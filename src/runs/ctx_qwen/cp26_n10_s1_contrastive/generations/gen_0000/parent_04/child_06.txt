import numpy as np
from scipy.optimize import minimize

def compute_obj_and_grad(v, N, mu):
    """
    Computes the objective function and its gradient for the circle packing problem.
    v: 1D array of length 3*N containing [x1, y1, r1, x2, y2, r2, ...]
    N: number of circles
    mu: penalty weight
    """
    obj = -np.sum(v[2::3])
    grad = np.zeros_like(v)
    grad[2::3] = -1.0
    
    # Boundary constraints: 0 <= x - r, x + r <= 1, same for y
    for i in range(N):
        idx_x, idx_y, idx_r = 3*i, 3*i+1, 3*i+2
        xi, yi, ri = v[idx_x], v[idx_y], v[idx_r]
        
        viol = ri - xi
        if viol > 0:
            obj += mu * viol**2
            grad[idx_x] -= 2 * mu * viol
            grad[idx_r] += 2 * mu * viol
            
        viol = ri - (1.0 - xi)
        if viol > 0:
            obj += mu * viol**2
            grad[idx_x] += 2 * mu * viol
            grad[idx_r] += 2 * mu * viol
            
        viol = ri - yi
        if viol > 0:
            obj += mu * viol**2
            grad[idx_y] -= 2 * mu * viol
            grad[idx_r] += 2 * mu * viol
            
        viol = ri - (1.0 - yi)
        if viol > 0:
            obj += mu * viol**2
            grad[idx_y] += 2 * mu * viol
            grad[idx_r] += 2 * mu * viol
            
    # Pairwise non-overlap constraints: dist >= r_i + r_j
    for i in range(N):
        for j in range(i+1, N):
            idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
            idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
            
            dx = v[idx_xi] - v[idx_xj]
            dy = v[idx_yi] - v[idx_yj]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < 1e-8:
                dist = 1e-8
                
            viol = v[idx_ri] + v[idx_rj] - dist
            if viol > 0:
                obj += mu * viol**2
                inv_dist = 1.0 / dist
                grad[idx_xi] -= 2 * mu * viol * dx * inv_dist
                grad[idx_yi] -= 2 * mu * viol * dy * inv_dist
                grad[idx_ri] += 2 * mu * viol
                grad[idx_xj] += 2 * mu * viol * dx * inv_dist
                grad[idx_yj] += 2 * mu * viol * dy * inv_dist
                grad[idx_rj] += 2 * mu * viol
                
    return obj, grad

def get_initial_guess(N, seed):
    """Generates an initial configuration on a hexagonal lattice."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    r_init = 0.04
    k = 0
    spacing = 0.15
    for row in range(7):
        for col in range(6):
            if k >= N: break
            x = 0.1 + col * spacing + (spacing/2 if row % 2 == 1 else 0)
            y = 0.1 + row * spacing * np.sqrt(3)/2
            if x < 0.95 and y < 0.95:
                centers[k] = [x, y]
                k += 1
        if k >= N: break
        
    while k < N:
        centers[k] = [np.random.uniform(0.15, 0.85), np.random.uniform(0.15, 0.85)]
        k += 1
        
    v0 = np.zeros(3*N)
    for i in range(N):
        v0[3*i] = centers[i, 0]
        v0[3*i+1] = centers[i, 1]
        v0[3*i+2] = r_init
    return v0

def run_packing():
    N = 26
    bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(1e-5, 0.5)] * N
    
    best_v = None
    best_sum_r = -1e9
    
    # Multiple restarts to escape local minima
    for seed in [42, 123, 456, 789]:
        v0 = get_initial_guess(N, seed)
        v_current = v0.copy()
        
        # Iteratively increase penalty to enforce constraints strictly
        for mu in [100.0, 500.0, 2000.0, 10000.0]:
            res = minimize(compute_obj_and_grad, v_current, args=(N, mu), method='L-BFGS-B',
                           jac=True, bounds=bounds, options={'maxiter': 4000, 'ftol': 1e-12})
            v_current = res.x
            
        current_sum_r = np.sum(v_current[2::3])
        if current_sum_r > best_sum_r:
            best_sum_r = current_sum_r
            best_v = v_current.copy()
            
    final_centers = np.array([[best_v[3*i], best_v[3*i+1]] for i in range(N)])
    final_radii = np.array([best_v[3*i+2] for i in range(N)])
    
    return final_centers, final_radii, np.sum(final_radii)
