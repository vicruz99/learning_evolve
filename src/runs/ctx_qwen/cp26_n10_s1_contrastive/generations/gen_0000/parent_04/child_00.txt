import numpy as np
from scipy.optimize import minimize

N = 26

def _objective(x):
    """Negative sum of radii (to minimize)."""
    return -np.sum(x[2::3])

def _constraints(x):
    """Vector of inequality constraints >= 0."""
    c = []
    # Boundary constraints: 4 per circle
    for i in range(N):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        c.extend([xi - ri, yi - ri, 1.0 - xi - ri, 1.0 - yi - ri])
        
    # Overlap constraints: |ci - cj|^2 - (ri + rj)^2 >= 0
    for i in range(N):
        for j in range(i+1, N):
            xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
            xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
            dx = xi - xj
            dy = yi - yj
            c.append(dx*dx + dy*dy - (ri + rj)**2)
            
    return np.array(c)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0.0
    best_x = None
    
    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': _constraints}
    
    # Multiple random restarts to escape local optima
    for trial in range(10):
        centers = np.random.rand(N, 2)
        radii = np.full(N, 0.015)
        x0 = np.zeros(3 * N)
        for i in range(N):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
            
        res = minimize(_objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False})
                       
        cur_sum = -res.fun
        # Check constraint satisfaction with tolerance
        c_vals = _constraints(res.x)
        if np.all(c_vals >= -1e-9) and cur_sum > best_sum:
            best_sum = cur_sum
            best_x = res.x.copy()
            
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final safety clamping to handle floating point drift
    radii = np.maximum(radii, 0.0)
    centers = np.clip(centers, 0.0, 1.0)
    
    return centers, radii, float(best_sum)
