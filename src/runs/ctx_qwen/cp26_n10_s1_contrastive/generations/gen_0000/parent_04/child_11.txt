import numpy as np

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses a force-directed grow-and-push algorithm with cooling and perturbation.
    """
    N = 26
    np.random.seed(42)
    
    # Initialize centers in the central region to avoid immediate boundary sticking
    centers = np.random.rand(N, 2) * 0.6 + 0.2
    radii = np.full(N, 0.001)
    dr = 1e-3  # Initial growth step

    for step in range(50000):
        # Grow all circles uniformly
        radii += dr

        # Compute pairwise relative positions and distances
        diffs = centers[:, None, :] - centers[None, :, :]  # Shape (N, N, 2)
        dists = np.sqrt(np.sum(diffs**2, axis=2))           # Shape (N, N)
        np.fill_diagonal(dists, np.inf)                     # Ignore self-distance

        # Calculate overlaps: positive if circles intersect
        overlaps = radii[:, None] + radii[None, :] - dists
        # Only consider upper triangle to avoid double-counting forces
        mask = np.triu(np.ones((N, N), dtype=bool), k=1) & (overlaps > 0)

        if np.any(mask):
            forces = np.zeros((N, 2))
            i_idx, j_idx = np.where(mask)
            for i, j in zip(i_idx, j_idx):
                d = dists[i, j]
                if d < 1e-12:
                    continue
                nx = diffs[i, j, 0] / d
                ny = diffs[i, j, 1] / d
                move = overlaps[i, j] / 2.0
                vec = np.array([nx * move, ny * move])
                forces[i] -= vec
                forces[j] += vec
            centers += forces

        # Enforce boundary constraints strictly
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

        # Simulated annealing: periodic perturbation to escape local minima
        if step % 5000 == 0:
            centers += np.random.randn(N, 2) * 1e-5

        # Cooling schedule: gradually reduce growth step
        if step % 2000 == 0:
            dr *= 0.9
            if dr < 1e-8:
                dr = 1e-8

    return centers, radii, np.sum(radii)
