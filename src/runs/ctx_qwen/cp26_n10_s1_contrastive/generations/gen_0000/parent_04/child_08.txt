**Strategy:**
1. **Formulation:** We frame the problem as minimizing $-\sum r_i$ subject to constraints. Since explicit constraint handling for ~350 pairwise non-overlap conditions can be slow, we use a **penalty method**. The objective function combines the negative sum of radii with a quadratic penalty term that heavily penalizes boundary violations and circle overlaps.
2. **Initialization:** We start with a structured grid placement (5x6 subset) scaled to the center of the square with small radii. This provides a feasible, spread-out starting point that helps the optimizer explore different configurations.
3. **Optimization:** We use `scipy.optimize.minimize` with the `L-BFGS-B` algorithm, which handles bound constraints efficiently. We run multiple restarts with random perturbations to escape local minima. The penalty weight is set high enough to enforce strict constraint satisfaction.
4. **Validation & Selection:** After each optimization run, we strictly validate the configuration against the problem constraints (using the same $10^{-12}$ tolerance as the validator). We keep the best valid solution found across all restarts.

import numpy as np
from scipy.optimize import minimize

def objective_with_penalty(x, n, penalty_weight):
    """Objective function: maximize sum of radii with penalty for constraint violations."""
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    # Primary objective: minimize negative sum of radii
    obj = -np.sum(radii)
    penalty = 0.0
    
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        
        # Violations: radius exceeds distance to any of the 4 walls
        v1 = ri - xi          # Left wall
        v2 = ri - (1.0 - xi)  # Right wall
        v3 = ri - yi          # Bottom wall
        v4 = ri - (1.0 - yi)  # Top wall
        
        penalty += np.maximum(0.0, v1)**2 + np.maximum(0.0, v2)**2 + \
                   np.maximum(0.0, v3)**2 + np.maximum(0.0, v4)**2
        
    # Non-overlap constraints: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i+1, n):
            dist = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            v = radii[i] + radii[j] - dist
            penalty += np.maximum(0.0, v)**2
            
    return obj + penalty_weight * penalty

def get_initial_config(n):
    """Generate a valid initial configuration using a scaled grid."""
    x0 = np.zeros(3*n)
    # Create a 5x6 grid pattern
    xs = np.linspace(0.2, 0.8, 6)
    ys = np.linspace(0.2, 0.8, 5)
    idx = 0
    for y in ys:
        for x in xs:
            if idx < n:
                x0[idx*3] = x
                x0[idx*3+1] = y
                x0[idx*3+2] = 0.03  # Small initial radius ensures validity
                idx += 1
    # Fill any remaining slots if n > 30 (not needed here but safe)
    while idx < n:
        x0[idx*3] = 0.5
        x0[idx*3+1] = 0.5
        x0[idx*3+2] = 0.01
        idx += 1
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to compute the optimal packing."""
    n = 26
    best_x = None
    best_score = -1.0
    best_centers = None
    best_radii = None
    
    # Multiple restarts with different random seeds to explore the landscape
    for seed in range(15):
        rng = np.random.RandomState(seed)
        x0 = get_initial_config(n)
        
        # Add controlled noise to break symmetry and explore different packings
        x0 += rng.randn(3*n) * 0.05
        
        # Enforce bounds for positions and radii
        x0[::3] = np.clip(x0[::3], 0.05, 0.95)   # x coordinates
        x0[1::3] = np.clip(x0[1::3], 0.05, 0.95) # y coordinates
        x0[2::3] = np.clip(x0[2::3], 0.01, 0.4)  # radii
        
        # Optimize
        res = minimize(objective_with_penalty, x0, args=(n, 20000.0), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)]*(2*n) + [(0.0, 0.5)]*n,
                       options={'maxiter': 5000, 'ftol': 1e-12})
        
        centers = res.x[:2*n].reshape(n, 2)
        radii = res.x[2*n:]
        
        # Strict validation matching the provided checker
        valid = True
        for i in range(n):
            if radii[i] < 0: 
                valid = False; break
            if centers[i,0] < 0 or centers[i,0] > 1 or centers[i,1] < 0 or centers[i,1] > 1: 
                valid = False; break
            if centers[i,0] - radii[i] < -1e-12 or centers[i,0] + radii[i] > 1+1e-12: 
                valid = False; break
            if centers[i,1] - radii[i] < -1e-12 or centers[i,1] + radii[i] > 1+1e-12: 
                valid = False; break
                
        for i in range(n):
            for j in range(i+1, n):
                dist = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if dist < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid: break
            
        if valid:
            score = np.sum(radii)
            if score > best_score:
                best_score = score
                best_x = res.x.copy()
                best_centers = centers.copy()
                best_radii = radii.copy()
                
    # Fallback if no valid configuration was found (highly unlikely)
    if best_centers is None:
        init = get_initial_config(n)
        best_centers = init[:2*n].reshape(n, 2)
        best_radii = init[2*n:]
        best_score = np.sum(best_radii)
        
    return best_centers, best_radii, best_score
