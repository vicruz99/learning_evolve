def compute_overlaps_and_boundaries(vars, n=26):
    """
    Computes the constraint violations.
    vars is an array of size 3*n: [x1, y1, r1, x2, y2, r2, ...]
    Returns a vector of constraint values. We want these to be >= 0 for validity.
    Specifically:
    - Boundary: x - r >= 0, x + r <= 1 => 1 - (x + r) >= 0, etc.
    - Overlap: dist^2 - (r1 + r2)^2 >= 0
    """
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        idx = 3 * i
        x, y, r = vars[idx], vars[idx+1], vars[idx+2]
        
        # x - r >= 0
        constraints.append(x - r)
        # x + r <= 1  =>  1 - x - r >= 0
        constraints.append(1 - x - r)
        # y - r >= 0
        constraints.append(y - r)
        # y + r <= 1  =>  1 - y - r >= 0
        constraints.append(1 - y - r)
        
        # r >= 0
        constraints.append(r)

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3 * i
            idx_j = 3 * j
            x_i, y_i, r_i = vars[idx_i], vars[idx_i+1], vars[idx_i+2]

            dy = y_i - y_j

            r_sum = r_i + r_j
            # dist >= r_sum  =>  dist^2 >= r_sum^2  =>  dist^2 - r_sum^2 >= 0
            constraints.append(dist_sq - r_sum*r_sum)
            
    return np.array(constraints)
