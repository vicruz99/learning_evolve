import numpy as np
from scipy.optimize import minimize

def smooth_max(x, eps=1e-6):
    """Smooth approximation of max(0, x) to maintain differentiability."""
    return 0.5 * (x + np.sqrt(x**2 + eps))

def packing_objective(vars, N, C):
    """
    Objective function: maximize sum of radii (minimize negative sum) 
    with penalty for constraint violations.
    """
    centers = vars[:2*N].reshape(N, 2)
    radii = vars[2*N:]
    
    penalty = 0.0
    
    # Boundary penalties: ensure r <= x <= 1-r and r <= y <= 1-r
    # Violations: r - x > 0, x - (1-r) > 0, etc.
    penalty += np.sum(smooth_max(radii - centers[:,0])**2 + smooth_max(centers[:,0] - (1 - radii))**2)
    penalty += np.sum(smooth_max(radii - centers[:,1])**2 + smooth_max(centers[:,1] - (1 - radii))**2)
    
    # Overlap penalties: ensure dist >= r_i + r_j
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Only consider lower triangle pairs (i < j)
    i_idx, j_idx = np.tril_indices(N, -1)
    penalty += C * np.sum(smooth_max(rad_sum[i_idx, j_idx] - dists[i_idx, j_idx])**2)
    
    return -np.sum(radii) + penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    r0 = 0.1
    
    # Generate initial hexagonal grid configuration
    pts = []
    dy = np.sqrt(3)/2 * r0
    y = r0
    row = 0
    while len(pts) < N:
        offset = 0.0
        if row % 2 == 1:
            offset = r0 / 2.0
        x = r0 + offset
        while x + r0 <= 1.0 and len(pts) < N:
            pts.append([x, y])
            x += r0
        y += dy
        row += 1
        
    centers_init = np.array(pts)
    radii_init = np.full(N, r0)
    vars_init = np.concatenate([centers_init.ravel(), radii_init])
    
    # Bounds for optimizer: coordinates in [0,1], radii in [0, 0.5]
    bounds = []
    for _ in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
    bounds.extend([(0.0, 0.5)] * N)
    
    # Iterative penalty annealing to enforce constraints strictly
    current_vars = vars_init.copy()
    C = 50.0
    
    for _ in range(5):
        C *= 5.0
        res = minimize(
            packing_objective, 
            current_vars, 
            args=(N, C), 
            method='L-BFGS-B', 
            bounds=bounds, 
            options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False}
        )
        current_vars = res.x
        
    centers = current_vars[:2*N].reshape(N, 2)
    radii = current_vars[2*N:]
    
    # Ensure non-negative radii
    radii = np.maximum(radii, 0.0)
    
    # Project centers strictly inside valid boundaries to satisfy validation tolerance
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    return centers, radii, float(np.sum(radii))
