import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # 1. Initialization: Hexagonal-like staggered grid
    centers = []
    radii = []
    
    # Estimated max radius for a 26-circle hex packing in a unit square
    r_init = 0.095 
    
    # Row configuration: 6 rows with varying counts to sum to 26
    # Pattern: 5, 4, 5, 4, 5, 3
    row_counts = [5, 4, 5, 4, 5, 3]
    
    idx = 0
    for row_idx, count in enumerate(row_counts):
        # Vertical spacing for hex packing: sqrt(3)/2 * diameter
        # Total height = 2*r + (rows-1) * sqrt(3)*r
        y = r_init + row_idx * r_init * np.sqrt(3)
        
        # Horizontal shift for staggered rows
        shift = r_init if row_idx % 2 == 1 else 0.0
        
        # Calculate x positions centered in the row
        # Width required = (count - 1) * 2r
        row_width = (count - 1) * (2 * r_init)
        start_x = (1.0 - row_width) / 2.0 + shift
        
        for k in range(count):
            x = start_x + k * (2 * r_init)
            # Clamp to [0, 1] just in case of float precision issues
            x = np.clip(x, 0.0, 1.0)
            y = np.clip(y, 0.0, 1.0)
            
            centers.append([x, y])
            radii.append(r_init)
            idx += 1

    centers = np.array(centers)
    radii = np.array(radii)
    
    # Flatten to optimization vector: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
    
    # Bounds: x, y in [0, 1], r in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(3 * n)]
    # Tighten radius lower bound to avoid degenerate circles if needed, but 0 is fine.
    
    # Objective: Minimize negative sum of radii (Maximize sum of radii)
    def objective(vars):
        return -sum(vars[3*i + 2] for i in range(n))
    
    # Constraints
    constraints = []
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
    for i in range(n):
        ix, iy, ir = 3*i, 3*i+1, 3*i+2
        
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[ix] - v[ir]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[ix] - v[ir]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[iy] - v[ir]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[iy] - v[ir]})
        
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            ix_i, iy_i, ir_i = 3*i, 3*i+1, 3*i+2
            ix_j, iy_j, ir_j = 3*j, 3*j+1, 3*j+2
            
            def dist_constraint(v, i=i, j=j):
                dx = v[ix_i] - v[ix_j]
                dy = v[iy_i] - v[iy_j]
                r_sum = v[ir_i] + v[ir_j]
                return (dx**2 + dy**2) - (r_sum**2)
            
            constraints.append({'type': 'ineq', 'fun': dist_constraint})

    # Run optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                      constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-10})
    
    opt_vars = result.x
    
    # Extract results
    final_centers = np.array([[opt_vars[3*i], opt_vars[3*i+1]] for i in range(n)])
    final_radii = np.array([opt_vars[3*i+2] for i in range(n)])
    sum_radii = np.sum(final_radii)
    
    # Validate
    if not validate_packing(final_centers, final_radii):
        # Fallback to initial guess if optimization fails validation
        # (Though with SLSQP and proper bounds, this is unlikely if convergence is good)
        print("Optimization result invalid, falling back to initial configuration.")
        return centers, radii, np.sum(radii)
        
    return final_centers, final_radii, sum_radii

# Validation function (Read-only)
def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0 or np.isnan(radii[i]):
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
