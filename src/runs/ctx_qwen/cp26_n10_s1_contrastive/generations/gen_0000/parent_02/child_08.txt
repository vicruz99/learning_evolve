import numpy as np
from scipy.optimize import minimize

def _objective(x):
    # Maximize sum of radii => minimize negative sum
    return -x[2::3].sum()

def _con_bounds(x):
    n = 26
    v = x.reshape(n, 3)
    # x_i >= r_i  => x_i - r_i >= 0
    # x_i <= 1 - r_i => 1 - x_i - r_i >= 0
    # y_i >= r_i  => y_i - r_i >= 0
    # y_i <= 1 - r_i => 1 - y_i - r_i >= 0
    return np.concatenate([
        v[:, 0] - v[:, 2],
        1.0 - v[:, 0] - v[:, 2],
        v[:, 1] - v[:, 2],
        1.0 - v[:, 1] - v[:, 2]
    ])

def _con_pairs(x):
    n = 26
    v = x.reshape(n, 3)
    # Vectorized pairwise constraints: dist^2 >= (r_i + r_j)^2
    dx = v[:, 0:1, np.newaxis] - v[:, np.newaxis, 0]
    dy = v[:, 1:2, np.newaxis] - v[:, np.newaxis, 1]
    dist_sq = dx**2 + dy**2
    r_sum = v[:, 2:3, np.newaxis] + v[:, np.newaxis, 2]
    idx = np.triu_indices(n, k=1)
    return (dist_sq - r_sum**2)[idx]

def run_packing():
    np.random.seed(42)
    n = 26
    
    # 1. Initialize with a dense hexagonal-like grid
    pts = []
    for i in range(26):
        col = i % 5
        row = i // 5
        # Hexagonal offset for odd rows improves packing density
        x = col * 0.2 + (0.1 if row % 2 == 1 else 0.0)
        y = row * 0.1732
        pts.append([x, y])
    pts = np.array(pts)
    
    # Center and scale to fit comfortably inside [0,1]x[0,1]
    min_c = pts.min(axis=0)
    max_c = pts.max(axis=0)
    span = max_c - min_c
    pts = (pts - min_c) / span * 0.8 + 0.1
    
    # Initial radii: small to ensure feasibility, with perturbation to break symmetry
    radii_init = 0.02 + np.random.rand(n) * 0.005
    x0 = np.hstack([pts.flatten(), radii_init])
    
    # Bounds: coordinates in [0, 1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Constraints
    cons = [
        {'type': 'ineq', 'fun': _con_bounds},
        {'type': 'ineq', 'fun': _con_pairs}
    ]
    
    # Optimization
    res = minimize(_objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 2000, 'ftol': 1e-12})
                   
    centers = res.x[:3 * n].reshape(n, 2)
    radii = res.x[3 * n:]
    
    return centers, radii, np.sum(radii)
