import numpy as np
import scipy.optimize as opt

# Global validation function provided in the prompt
def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_constraints(n):
    """Generate constraints for the optimization problem."""
    constraints = []
    for i in range(n):
        # Lower bound on radius (avoid zero radius)
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*n + i]})
        
        # Boundary constraints: x >= r, x + r <= 1 => x <= 1 - r
        # y >= r, y + r <= 1 => y <= 1 - r
        # Formatted as fun >= 0
        # 1. x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[i] - v[2*n + i]})
        # 2. 1 - (x + r) >= 0 => 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[i] - v[2*n + i]})
        # 3. y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[n + i] - v[2*n + i]})
        # 4. 1 - (y + r) >= 0 => 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[n + i] - v[2*n + i]})

    for i in range(n):
        for j in range(i + 1, n):
            # Non-overlap: dist >= r_i + r_j
            # dist - r_i - r_j >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: np.sqrt((v[i] - v[j])**2 + (v[n+i] - v[n+j])**2) - (v[2*n + i] + v[2*n + j])
            })
    return constraints

def objective(v, n):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*n:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    constraints = get_constraints(n)
    
    best_centers = None
    best_radii = None
    max_sum_r = -1.0
    
    # Generate multiple initial configurations
    initial_guesses = []

    initial_guesses.append(centers_5x5)

    # 2. 5x5 Grid + Random point
    np.random.seed(42)
    centers_5x5_rand = centers_5x5.copy()
    centers_5x5_rand[25] = [np.random.rand(), np.random.rand()]
    initial_guesses.append(centers_5x5_rand)

    # 3. Hexagonal Packing (Staggered Rows)
    # Trying to fit 26 circles
    # 5 rows: 6, 5, 6, 5, 4 -> Total 26
    hex_centers = []
    y = 0.1
    dy = 0.15 # vertical step
    r_est = 0.08
    counts = [6, 5, 6, 5, 4]
    row_idx = 0
    for count in counts:
        width = (count - 1) * 2 * r_est
        start_x = (1.0 - width) / 2
        for k in range(count):
            x = start_x + k * 2 * r_est
            # Offset odd rows
            if row_idx % 2 == 1:
                x += r_est
            hex_centers.append([x, y])
        y += dy * 1.5
        row_idx += 1
    initial_guesses.append(hex_centers)

    # 4. Random initialization
    for _ in range(5):
        random_centers = np.random.rand(n, 2) * 0.8 + 0.1
        initial_guesses.append(random_centers)

    # Optimization Loop
    # We need to convert list of centers to a flat array for scipy
    # Format: [x0, x1, ..., x25, y0, y1, ..., y25, r0, r1, ..., r25]
    
    for i, centers_list in enumerate(initial_guesses):
        centers_list = np.array(centers_list)

        bounds = [(0, 1)] * n + [(0, 1)] * n + [(1e-5, 0.5)] * n
        
        try:
            res = opt.minimize(objective, v0, args=(n,), method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 500})

            centers_default[i] = [0.5, 0.5]
            radii_default[i] = 0.0
        return centers_default, radii_default, 0.0

    return best_centers, best_radii, max_sum_r
