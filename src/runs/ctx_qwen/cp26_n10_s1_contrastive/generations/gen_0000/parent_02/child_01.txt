def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        radii: np.array of shape (26) with radius of each circle.
        sum_radii: The total sum of the radii.

    def compute_overlap_penalty(centers, radii):
        """
        Computes a penalty for overlaps and boundary violations.
        Positive values indicate violations.
        """
        n = centers.shape[0]
        penalty = 0.0
        
        # Boundary constraints: center_i +/- r_i must be in [0, 1]
        # Penalty is proportional to how much it sticks out
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            if x - r < 0:
                penalty += (x - r)**2
            if x + r > 1:
                penalty += (x + r - 1)**2
            if y - r < 0:
                penalty += (y - r)**2
            if y + r > 1:
                penalty += (y + r - 1)**2
                
        # Overlap constraints: dist(i, j) >= r_i + r_j
        # Equivalent to dist^2 >= (r_i + r_j)^2
        # We penalize if dist < r_i + r_j
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((centers[i] - centers[j])**2)
                min_dist = radii[i] + radii[j]
                if dist_sq < min_dist**2:
                    # Penalty is squared difference to make it smooth
                    penalty += (min_dist - np.sqrt(dist_sq))**2
                    
        return penalty

        Objective function to maximize sum of radii.

        radii = initial_radii.copy()

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square.
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
