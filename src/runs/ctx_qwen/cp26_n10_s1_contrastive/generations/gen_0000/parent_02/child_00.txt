import numpy as np
from scipy.optimize import linprog, differential_evolution
import math

def get_boundary_distances(centers):
    """
    Compute the maximum radius each circle can have based on boundaries.
    Returns array of shape (n,)
    """
    x = centers[:, 0]
    y = centers[:, 1]
    # Distance to left, right, bottom, top
    dists = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
    return dists

def solve_radii_lp(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    Maximize sum(r_i)
    Subject to:
    r_i <= boundary_dist_i
    r_i + r_j <= dist(center_i, center_j)
    r_i >= 0
    """
    n = centers.shape[0]
    
    # Coefficients for objective function: minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Inequality constraints matrix A_ub * r <= b_ub
    # We will collect constraints here
    A_ub_rows = []
    b_ub_rows = []
    
    # 1. Boundary constraints: r_i <= dist_boundary
    b_dists = get_boundary_distances(centers)
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_rows.append(b_dists[i])
        
    # 2. Pairwise constraints: r_i + r_j <= dist_ij
    # Calculate pairwise distances
    # Using broadcasting for efficiency
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Only upper triangle to avoid duplicates
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_rows.append(dists[i, j])
            
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)
    
    # Bounds for r_i: [0, None]
    bounds = [(0, None) for _ in range(n)]
    
    try:
        # Solve LP
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            optimal_radii = res.x
            sum_radii = -res.fun # Objective was negative sum
            return sum_radii, optimal_radii
        else:
            # If LP fails, return a very low sum to penalize this configuration
            return -1e9, np.zeros(n)
            
    except Exception:
        return -1e9, np.zeros(n)

def objective_function_flat(x_flat):
    """
    Objective function for differential evolution.
    x_flat is a 1D array of 2*n coordinates.
    Returns negative sum of radii (to minimize).
    """
    n = 26

        return 1e9 # Penalize NaN
    
    sum_r, _ = solve_radii_lp(centers)
    
    if sum_r < 0: # LP failed or invalid
        return 1e9
        
    return -sum_r
