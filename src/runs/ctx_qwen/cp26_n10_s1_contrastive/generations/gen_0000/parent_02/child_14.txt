import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # 1. Initialization: Hexagonal Lattice
    # We generate points on a hexagonal grid and select the first 26.
    # We use a radius estimate slightly larger than 0.1 to push the optimizer,
    # but ensure validity by checking bounds.
    
    centers = []
    r_init = 0.105  # Estimated optimal radius based on density arguments
    
    # Hexagonal lattice spacing
    # Horizontal distance between centers in a row: 2 * r
    # Vertical distance between rows: sqrt(3) * r
    # Rows are shifted horizontally by r
    
    y = r_init
    while len(centers) < n_circles:
        row_centers_x = []
        # Determine x coordinates for this row
        # We can start at r_init and step by 2*r_init
        # Or for shifted rows, start at 2*r_init
        
        if len(centers) % (int(1 / (2 * r_init)) + 1) == 0:
             # Shifted row
             x = 2 * r_init
        else:
             # Aligned row
             x = r_init
            
        while x <= 1 - r_init:
            centers.append([x, y])
            row_centers_x.append(x)
            x += 2 * r_init
        
        if len(centers) >= n_circles:
            break
        y += np.sqrt(3) * r_init

    # If we didn't get 26 points (unlikely with this logic but safe to check),
    # fallback to a grid or fill remaining with small perturbations.
    while len(centers) < n_circles:
        # Add a circle in a gap or just a valid small position
        # This is a safety fallback
        centers.append([0.1 + 0.05 * len(centers), 0.1 + 0.05 * len(centers)])

    centers = np.array(centers[:n_circles])
    
    # Initial radii
    radii = np.full(n_circles, 0.05) # Start small to be valid
    
    # 2. Optimization Setup
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total 3 * n_circles variables
    
    def objective(vars):
        # We want to maximize sum of radii -> minimize negative sum
        radii_part = vars[2::3] # radii are at indices 2, 5, 8, ...
        return -np.sum(radii_part)

    def constraint_boundary(vars):
        # Returns array of constraint values >= 0
        # x_i >= r_i  => x_i - r_i >= 0
        # 1 - x_i >= r_i => 1 - x_i - r_i >= 0
        # Same for y
        constraints = []
        for i in range(n_circles):
            idx = 3 * i
            x, y, r = vars[idx], vars[idx+1], vars[idx+2]
            constraints.append(x - r)
            constraints.append(1 - x - r)
            constraints.append(y - r)
            constraints.append(1 - y - r)
        return np.array(constraints)

    def constraint_overlap(vars):
        # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
        # => (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        constraints = []
        centers_vars = vars[:2*n_circles].reshape(n_circles, 2)
        radii_vars = vars[2::3]
        
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist_sq = np.sum((centers_vars[i] - centers_vars[j])**2)
                sum_r = radii_vars[i] + radii_vars[j]
                constraints.append(dist_sq - sum_r**2)
        return np.array(constraints)

    # Initial guess vector
    x0 = np.zeros(3 * n_circles)
    for i in range(n_circles):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]

    # Constraints dictionary for scipy
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    # Bounds: radii must be non-negative. Coordinates [0, 1].
    # SLSQP requires bounds to be (min, max)
    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 1.0)) # r (radius can't exceed 1)

    # Run optimization
    # Method 'SLSQP' is appropriate for non-linear constraints
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 1000, 'ftol': 1e-9})
    
    # Extract results
    optimal_vars = res.x
    optimal_centers = optimal_vars[:2*n_circles].reshape(n_circles, 2)
    optimal_radii = optimal_vars[2::3]
    
    # Calculate sum of radii
    sum_radii = np.sum(optimal_radii)
    
    # Validation check (internal)
    # Note: The validation function in the prompt is read-only, 
    # but we can perform a quick check to ensure sanity.
    # However, we must return the arrays directly.
    
    return optimal_centers, optimal_radii, sum_radii
