    *   **Boundary Constraints:** Each circle must be inside $[0,1] \times [0,1]$. This implies $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$.
    *   **Non-Overlap Constraints:** For every pair $(i, j)$, the distance between centers must be at least the sum of their radii: $\sqrt{(x_i-x_j)^2 + (y_i-y_j)^2} \ge r_i + r_j$.
3.  **Robustness (Restarts):** To escape local minima, we run the optimizer multiple times (5 restarts) with random perturbations to the initial centers and radii.
4.  **Variable Layout:** We pack variables as $[x_0, y_0, r_0, x_1, y_1, r_1, \dots]$ for easier indexing.
5.  **Constraint Implementation:** We define a single constraint function that returns a vector of all inequality constraints (values must be $\ge 0$). This includes 4 boundary constraints per circle and 1 distance constraint per pair.

import numpy as np
from scipy.optimize import minimize
import math

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    
    # Check for NaN values
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False

    # Check if radii are nonnegative
    for i in range(n):
        if radii[i] < 0:
            return False
        elif np.isnan(radii[i]):
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Use a small epsilon for floating point tolerance
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt((centers[i,0] - centers[j,0])**2 + (centers[i,1] - centers[j,1])**2)
            if dist < radii[i] + radii[j] - 1e-9:
                return False

    return True

def generate_initial_guess(n):
    """
    Generate a valid initial configuration of n circles.
    Uses a grid layout perturbed slightly.
    """
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.04) # Start with small radii
    
    # Distribute circles in a grid pattern
    # 26 circles -> maybe 6 columns, 5 rows (30 slots), take 26
    cols = 6
    rows = 5
    
    count = 0
    for r_idx in range(rows):
        for c_idx in range(cols):
            if count < n:
                # Place in center of grid cells
                x = (c_idx + 0.5) / cols
                y = (r_idx + 0.5) / rows
                centers[count] = [x, y]
                count += 1
            else:
                break
    
    return centers, radii

def objective(vars, n):
    """
    Objective function: Maximize sum of radii -> Minimize -sum(radii)
    vars layout: [x0, y0, r0, x1, y1, r1, ...]
    """
    # Radii are at indices 2, 5, 8, ...
    radii = vars[2::3]
    return -np.sum(radii)

def constraints(vars, n):
    """
    Returns a vector of constraint values. All must be >= 0.
    Includes boundary and pairwise non-overlap constraints.
    """
    cons = []
    
    # Extract variables
    # centers shape (n, 2), radii shape (n,)
    centers = np.array(vars[:2*n]).reshape((n, 2))
    radii = np.array(vars[2*n:])
    
    # 1. Boundary Constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    # Equivalent to: x - r >= 0, 1 - x - r >= 0, etc.
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        cons.append(x - r)
        cons.append(1 - x - r)
        cons.append(y - r)
        cons.append(1 - y - r)
        
    # 2. Pairwise Distance Constraints: dist(i, j) >= r_i + r_j
    # dist - (r_i + r_j) >= 0
    # We compute squared distance to avoid sqrt in loop if possible, 
    # but for constraint formulation dist >= sum is clearer.
    # However, sqrt is monotonic, so we can use dist.
    # To be safe with gradients and numerical stability, let's use dist.
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            cons.append(dist - (radii[i] + radii[j]))
            
    return np.array(cons)

    return best_centers, best_radii, best_sum
