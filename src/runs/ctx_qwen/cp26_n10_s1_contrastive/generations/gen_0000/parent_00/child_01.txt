import numpy as np
from scipy.spatial.distance import pdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses force-directed simulation with adaptive radius growth.
    """
    np.random.seed(42)
    n = 26
    
    # 1. Initial hexagonal layout for high density
    pos = []
    for i in range(6):
        for j in range(5):
            if len(pos) == n: break
            # Stagger x-coordinates for hexagonal arrangement
            x = j * 1.5 + 0.5 if i % 2 == 1 else j + 0.5
            y = i * np.sqrt(3) / 2
            pos.append([x, y])
        if len(pos) == n: break
            
    pos = np.array(pos[:n])
    # Normalize to unit square
    pos -= pos.min(axis=0)
    pos /= (pos.max(axis=0) - pos.min(axis=0) + 1e-10)
    # Add slight random jitter to break symmetry
    pos += np.random.uniform(-0.02, 0.02, size=pos.shape)
    
    r = 0.02
    lr = 0.04
    n_steps = 30000
    
    for step in range(n_steps):
        forces = np.zeros_like(pos)
        
        # Pairwise repulsion forces
        for i in range(n):
            for j in range(i + 1, n):
                diff = pos[i] - pos[j]
                dist = np.sqrt(np.sum(diff**2))
                if dist < 2 * r:
                    if dist < 1e-9:
                        diff = np.random.randn(2) * 1e-4
                        dist = np.sqrt(np.sum(diff**2))
                    overlap = 2 * r - dist
                    f_mag = overlap * 2.5
                    forces[i] += diff / dist * f_mag
                    forces[j] -= diff / dist * f_mag
                    
            # Wall repulsion forces
            for d in range(2):
                if pos[i, d] < r:
                    forces[i, d] += (r - pos[i, d]) * 15.0
                if pos[i, d] > 1.0 - r:
                    forces[i, d] -= (pos[i, d] - (1.0 - r)) * 15.0
                    
        # Update positions
        pos += lr * forces
        # Strictly enforce boundaries relative to current radius
        pos = np.clip(pos, r + 1e-7, 1.0 - r - 1e-7)
        
        # Periodic perturbation to escape local minima
        if step % 2000 == 0:
            pos += np.random.randn(n, 2) * lr * 0.5
            
        # Compute maximum feasible radius for current layout
        min_dist = np.min(pdist(pos))
        min_wall = np.min(np.concatenate([pos, 1.0 - pos], axis=1))
        possible_r = min(min_dist / 2.0, min_wall)
        
        # Grow radius gradually
        if possible_r > r * 1.0001:
            r = min(possible_r, r * 1.002)
            
        # Cool down learning rate
        lr *= 0.9995
        
    centers = pos
    radii = np.full(n, r)
    return centers, radii, float(np.sum(radii))
