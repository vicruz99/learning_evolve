import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, r):
    """Computes sum of squared violations for boundaries and overlaps."""
    n = centers.shape[0]
    # Boundary penalties: max(0, violation)^2
    b = np.maximum(0, r - centers[:, 0])**2
    b += np.maximum(0, r - centers[:, 1])**2
    b += np.maximum(0, centers[:, 0] + r - 1.0)**2
    b += np.maximum(0, centers[:, 1] + r - 1.0)**2
    
    # Overlap penalties
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    o = np.maximum(0, 2.0*r - dist)**2
    # Sum only upper triangle to avoid double counting pairs
    o = np.sum(np.triu(o, k=1))
    
    return np.sum(b) + o

def penalty_wrapper(v, r):
    """Wrapper for scipy minimize to handle 1D vector input."""
    return compute_penalty(v.reshape(26, 2), r)

def run_packing():
    np.random.seed(42)
    n = 26

    centers = []

            x = 0.05 + j * 0.17
            centers.append([x, y])
    centers = np.array(centers)
    
    # 2. Continuation method: gradually increase radius while optimizing positions
    r = 0.05
    v = centers.ravel()

                       bounds=[(0.0, 1.0)]*(n*2),

    v_full = np.concatenate([v, [r]])

        c = v[:r_idx].reshape(n, 2)
        r = v[r_idx]
        pen = compute_penalty(c, r)
        # Maximize r (-r) with heavy penalty for violations
        return -r + 10000.0 * pen

    res_final = minimize(objective_final, v_full, method='L-BFGS-B',
                         bounds=[(0.0, 1.0)]*(n*2) + [(0.01, 0.5)],

    r_opt = res_final.x[r_idx]

    # Shrink slightly to guarantee numerical safety with validator tolerance
    if r_opt > max_possible_r:
        r_opt = max_possible_r * 0.999999999
        
    return c_opt, np.full(n, r_opt), n * r_opt
