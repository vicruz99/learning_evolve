import numpy as np
from scipy.optimize import minimize
from functools import partial
import math


def objective(x, n):
    return -np.sum(x[2 * n:])


def bc_left(x, n, i):
    return x[2 * i] - x[2 * n + i]


def bc_right(x, n, i):
    return 1.0 - x[2 * i] - x[2 * n + i]


def bc_bottom(x, n, i):
    return x[2 * i + 1] - x[2 * n + i]


def bc_top(x, n, i):
    return 1.0 - x[2 * i + 1] - x[2 * n + i]


def noverlap(x, n, i, j):
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist = math.sqrt(dx * dx + dy * dy)
    return dist - x[2 * n + i] - x[2 * n + j]


def r_nonneg(x, n, i):
    return x[2 * n + i]


def make_hex_config(n, row_counts=None, base_r=0.09):
    if row_counts is None:
        row_counts = [5, 5, 5, 5, 5, 1]
    
    centers = np.zeros((n, 2))
    radii = np.full(n, base_r)
    
    total_spacing_y = (1.0 - 2.0 * base_r) / max(len(row_counts) - 1, 1)
    
    idx = 0
    for row_idx, n_in_row in enumerate(row_counts):
        y = base_r + row_idx * total_spacing_y
        if n_in_row == 1:
            x = 0.5
        else:
            spacing_x = (1.0 - 2.0 * base_r) / max(n_in_row - 1, 1)
            for col in range(n_in_row):
                x = base_r + col * spacing_x
                centers[idx] = [x, y]
                idx += 1
            continue
        centers[idx] = [x, y]
        idx += 1
    
    return centers, radii


def make_grid_config(n, base_r=0.09):
    centers = np.zeros((n, 2))
    radii = np.full(n, base_r)
    
    # 5x5 grid + 1 extra
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [0.1 + i * 0.2, 0.1 + j * 0.2]
            idx += 1
    
    # 26th circle in a gap
    centers[25] = [0.5, 0.5]
    radii[25] = 0.02
    
    return centers, radii


def make_perturbed_config(n, seed=0, base_r=0.09):
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, base_r)
    
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [0.1 + i * 0.2, 0.1 + j * 0.2]
            idx += 1
    
    centers[25] = [0.5, 0.5]
    radii[25] = 0.02
    
    # Add small random perturbation
    centers += rng.uniform(-0.01, 0.01, size=centers.shape)
    radii += rng.uniform(-0.005, 0.005, size=n)
    radii = np.maximum(radii, 0.01)
    
    # Clip centers to valid range
    centers = np.clip(centers, base_r, 1.0 - base_r)
    
    return centers, radii


def make_constraints(n):
    cons = []
    
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': partial(bc_left, n=n, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(bc_right, n=n, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(bc_bottom, n=n, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(bc_top, n=n, i=i)})
    
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': partial(noverlap, n=n, i=i, j=j)})
    
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': partial(r_nonneg, n=n, i=i)})
    
    return cons


def optimize_from_initial(x0, n, cons):
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    try:
        result = minimize(
            partial(objective, n=n),
            x0,
            method='SLSQP',
            constraints=cons,
            bounds=bounds,
            options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False}
        )
        return result.x
    except Exception:
        return None


def run_packing():
    n = 26
    
    cons = make_constraints(n)
    
    best_x = None
    best_sum = -np.inf
    
    # Try multiple initial configurations
    initial_configs = []
    
    # Config 1: 5x5 grid + 1 extra
    centers1, radii1 = make_grid_config(n, 0.09)
    initial_configs.append(np.concatenate([centers1.flatten(), radii1]))

    centers2, radii2 = make_hex_config(n, [5, 5, 5, 5, 5, 1], 0.09)
    initial_configs.append(np.concatenate([centers2.flatten(), radii2]))
    
    # Config 3-8: Perturbed grids with different seeds
    for seed in range(5):
        centers_p, radii_p = make_perturbed_config(n, seed=seed, base_r=0.09)
        initial_configs.append(np.concatenate([centers_p.flatten(), radii_p]))
    
    # Config 9: Start with slightly larger radii
    centers_lg, radii_lg = make_grid_config(n, 0.1)
    radii_lg[25] = 0.01
    initial_configs.append(np.concatenate([centers_lg.flatten(), radii_lg]))
    
    for x0 in initial_configs:
        result_x = optimize_from_initial(x0, n, cons)
        if result_x is not None:
            current_sum = np.sum(result_x[2 * n:])
            if current_sum > best_sum:
                best_sum = current_sum
                best_x = result_x.copy()
    
    # If optimization succeeded, refine the best result with another pass
    if best_x is not None:
        # Perturb the best result slightly and re-optimize
        rng = np.random.RandomState(42)
        for trial in range(3):
            x0_refine = best_x.copy()
            x0_refine += rng.uniform(-1e-6, 1e-6, size=len(x0_refine))
            result_x = optimize_from_initial(x0_refine, n, cons)
            if result_x is not None:
                current_sum = np.sum(result_x[2 * n:])
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_x = result_x.copy()
    
    centers_opt = best_x[:2 * n].reshape((n, 2))
    radii_opt = best_x[2 * n:]
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
