import numpy as np
from scipy.optimize import minimize, linprog


def bc_left(x, i, n):
    """Boundary constraint: x_i - r_i >= 0"""
    return x[2*i] - x[2*n + i]


def bc_right(x, i, n):
    """Boundary constraint: 1 - x_i - r_i >= 0"""
    return 1.0 - x[2*i] - x[2*n + i]


def bc_bottom(x, i, n):
    """Boundary constraint: y_i - r_i >= 0"""
    return x[2*i + 1] - x[2*n + i]


def bc_top(x, i, n):
    """Boundary constraint: 1 - y_i - r_i >= 0"""
    return 1.0 - x[2*i + 1] - x[2*n + i]


def nonoverlap(x, i, j, n):
    """Non-overlap constraint: dist(c_i, c_j) >= r_i + r_j"""
    dx = x[2*i] - x[2*j]
    dy = x[2*i + 1] - x[2*j + 1]
    dist = np.sqrt(dx*dx + dy*dy)
    return dist - x[2*n + i] - x[2*n + j]


def objective(x, n):
    """Maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2*n:])


def get_bounds(n):
    """Get variable bounds for optimization"""
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # x coordinate
        bounds.append((0.0, 1.0))  # y coordinate
        bounds.append((0.0, 0.5))  # radius (at most 0.5)
    return bounds


def get_constraints(n):
    """Get all constraints for optimization"""
    constraints = []
    
    # Boundary constraints for each circle
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': bc_left, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': bc_right, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': bc_bottom, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': bc_top, 'args': (i, n)})
    
    # Non-overlap constraints for each pair
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': nonoverlap, 'args': (i, j, n)})
    
    return constraints


def initialize_hexagonal(n, seed=0):
    """Initialize centers in a hexagonal packing pattern"""
    rng = np.random.RandomState(seed)
    
    r_est = 0.105
    h_spacing = 2.0 * r_est
    v_spacing = r_est * np.sqrt(3.0)
    
    centers = []
    row = 0
    while len(centers) < n:
        if row % 2 == 0:
            n_cols = 5
            x_start = r_est
        else:
            n_cols = 4
            x_start = 2.0 * r_est
        
        for col in range(n_cols):
            if len(centers) >= n:
                break
            x = x_start + col * h_spacing
            y = r_est + row * v_spacing
            
            # Add small random perturbation
            x += rng.uniform(-0.005, 0.005)
            y += rng.uniform(-0.005, 0.005)
            
            # Clip to valid range
            x = np.clip(x, 0.01, 0.99)
            y = np.clip(y, 0.01, 0.99)
            
            centers.append([x, y])
        row += 1
    
    return np.array(centers[:n])


def solve_radii_lp(centers, n):
    """Solve LP to find optimal radii given fixed centers"""
    # Maximize sum(r_i) subject to constraints
    c = -np.ones(n)
    
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        A_ub_list.append(np.eye(n)[i])
        b_ub_list.append(max_r)
    
    # Non-overlap: r_i + r_j <= ||c_i - c_j||
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dist)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    bounds = [(0.0, None) for _ in range(n)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success and result.fun is not None:
        return result.x
    else:
        # Fallback: equal radii
        return np.ones(n) * 0.08
