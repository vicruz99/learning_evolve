

<strategy>
To pack 26 circles in a unit square to maximize the sum of radii, we decompose the problem into two stages:
1.  **Optimize Centers (Outer Loop):** We treat the center coordinates $(x_i, y_i)$ as variables. For any fixed set of centers, the maximum possible radii can be determined efficiently.
2.  **Determine Radii via Linear Programming (Inner Step):** Given fixed centers, finding the radii that maximize the sum $\sum r_i$ is a Linear Programming (LP) problem. The constraints are linear: each radius $r_i$ is bounded by the distance to the square boundaries and half the distance to any other center ($r_i + r_j \le d_{ij}$).
3.  **Search Strategy:** We use `scipy.optimize.minimize` with the Nelder-Mead method to optimize the center coordinates. This derivative-free method is robust for the non-smooth objective function resulting from the LP. We initialize the search from multiple configurations (perturbed grids, random seeds) to escape local optima and ensure we find a high-quality packing.
</strategy>

import numpy as np
from scipy.optimize import linprog, minimize
import math

def get_optimal_radii_and_sum(centers):
    """
    Solves the LP to find maximum sum of radii for fixed centers.
    Returns (sum_radii, radii_vector).
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)
    
    # Constraints A_ub @ r <= b_ub
    A_ub = []
    b_ub = []
    
    # 1. Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        
        # Check if center is validly within [0, 1] (allowing tiny epsilon)
        # If center is strictly outside, no positive radius is possible.
        # However, linprog handles r_i <= negative bound with r_i >= 0 by failing.
        # We handle this gracefully.
        
        row = np.zeros(n)
        row[i] = 1.0
        
        # r_i <= x
        if x < -1e-9: return 0.0, np.zeros(n)
        A_ub.append(row.copy())
        b_ub.append(x)
        
        # r_i <= 1-x
        if 1-x < -1e-9: return 0.0, np.zeros(n)
        A_ub.append(row.copy())
        b_ub.append(1.0 - x)
        
        # r_i <= y
        if y < -1e-9: return 0.0, np.zeros(n)
        A_ub.append(row.copy())
        b_ub.append(y)
        
        # r_i <= 1-y
        if 1-y < -1e-9: return 0.0, np.zeros(n)
        A_ub.append(row.copy())
        b_ub.append(1.0 - y)
        
    # 2. Pairwise constraints: r_i + r_j <= dist(i, j)
    # Precompute distances
    # To save time, we can compute on the fly or batch
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.hypot(centers[i, 0] - centers[j, 0], 
                              centers[i, 1] - centers[j, 1])
            
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds: r_i >= 0
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            # linprog minimizes, so fun is -sum(r). Return sum.
            return -res.fun, res.x
        else:
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective_function(centers_flat):
    """
    Objective for the optimizer: minimize negative sum of radii.
    Centers must be in [0, 1]. We clip to ensure feasibility for LP bounds.
    """
    centers = centers_flat.reshape((26, 2))
    # Clip to valid range to avoid infeasible LP bounds
    centers = np.clip(centers, 0.0, 1.0)
    
    sum_r, _ = get_optimal_radii_and_sum(centers)
    return -sum_r

def generate_initialization(seed=0):
    """
    Generates an initial configuration of 26 centers.
    Uses a mix of grid and random perturbation.
    """
    rng = np.random.RandomState(seed)
    n = 26
    
    # Base grid: 5x5 = 25 points, plus one extra
    # Spread from 0.1 to 0.9 to leave room for radii
    xs = np.linspace(0.15, 0.85, 5) 
    ys = np.linspace(0.15, 0.85, 5)
    
    grid_points = []
    for y in ys:
        for x in xs:
            grid_points.append([x, y])
            
    # We have 25 points. Add 26th point.
    # Place it at center of a gap or random
    # Let's just add one at (0.5, 0.5) which is already in grid?
    # No, (0.5, 0.5) is in the 5x5 linspace.
    # Let's shift one row or add a point.
    # Let's use 6x5 grid = 30 points, pick 26.
    
    xs6 = np.linspace(0.1, 0.9, 6)
    ys5 = np.linspace(0.1, 0.9, 5)
    
    points = []
    for y in ys5:
        for x in xs6:
            points.append([x, y])
            
    # Shuffle and pick 26
    rng.shuffle(points)
    init_centers = np.array(points[:26])
    
    # Add small random perturbation
    init_centers += rng.normal(0, 0.02, size=init_centers.shape)
    init_centers = np.clip(init_centers, 0.05, 0.95)
    
    return init_centers.flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Try multiple random seeds to find global optimum
    seeds = [0, 1, 2, 42, 123, 999, 555]
    
    for seed in seeds:
        initial_centers_flat = generate_initialization(seed)
        
        # Optimize centers using Nelder-Mead
        # Nelder-Mead doesn't support bounds, but we clip inside objective.
        # This creates a flat region at boundaries, but it works.
        res = minimize(objective_function, 
                       initial_centers_flat, 
                       method='Nelder-Mead', 
                       options={'maxiter': 5000, 'xatol': 1e-7, 'fatol': 1e-9})
        
        current_sum = -res.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = res.x.reshape((26, 2))
            
            # Retrieve radii for the best centers found
            _, best_radii = get_optimal_radii_and_sum(best_centers)

    # Final validation and cleanup
    # Ensure radii are consistent with centers
    if best_radii is None:
        best_sum, best_radii = get_optimal_radii_and_sum(best_centers)

    return best_centers, best_radii, float(best_sum)

if __name__ == "__main__":
    centers, radii, s_r = run_packing()
    print(f"Sum of radii: {s_r}")
