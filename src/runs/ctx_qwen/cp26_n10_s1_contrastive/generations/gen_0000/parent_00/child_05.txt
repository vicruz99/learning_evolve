import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses SLSQP with multiple restarts to find a high-quality solution.
    """
    n = 26
    best_sum_radii = -1.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    # Number of optimization runs (restarts)
    num_runs = 10

    for run in range(num_runs):
        # --- Initialization ---
        if run == 0:
            # Run 0: Hexagonal-ish grid initialization
            # Trying to fit 26 circles. 6 rows seems reasonable.
            # Rows with 5, 4, 5, 4, 5, 3 circles? Total 26.
            rows = [5, 4, 5, 4, 5, 3]
            centers = np.zeros((n, 2))
            idx = 0
            # Vertical spacing and horizontal spacing estimation
            # Let's aim for r ~ 0.1 initially
            r_est = 0.09
            h_step = 2 * r_est * np.sqrt(3) / 2 # vertical dist between rows for hex
            # Actually for hex, dy = sqrt(3)*r. 
            # But we have varying row lengths.
            # Let's just place them roughly.
            y_pos = r_est
            for i, count in enumerate(rows):
                # Shift odd rows (index 1, 3, 5) by r_est
                x_start = r_est if i % 2 == 1 else r_est
                # If count is small, center them?
                # Width available 1.
                # Spacing dx = 2*r_est
                # Total width = (count-1)*2*r_est + 2*r_est = 2*count*r_est
                # If 2*count*r_est > 1, shrink r_est?
                # Let's just place them evenly in [r_est, 1-r_est]
                if count > 0:
                    if count == 1:
                        x_pos = 0.5
                        centers[idx, 0] = x_pos
                    else:
                        xs = np.linspace(r_est, 1 - r_est, count)
                        for k in range(count):
                            centers[idx, 0] = xs[k]
                            centers[idx, 1] = y_pos
                            idx += 1
                y_pos += h_step
                if idx >= n: break
            radii = np.full(n, r_est)
            
        elif run == 1:
            # Run 1: Random initialization with small radii to ensure feasibility
            centers = np.random.rand(n, 2)
            radii = np.full(n, 0.05) # Small initial radius
            
        else:
            # Other runs: Random centers, varying radii
            centers = np.random.rand(n, 2)
            radii = np.random.uniform(0.05, 0.15, n)

        # --- Optimization ---
        # Variables: [x1, y1, r1, x2, y2, r2, ...]
        # Total 3 * n variables
        x0 = np.concatenate([centers.flatten(), radii])

        def objective(x):
            # Maximize sum of radii -> Minimize negative sum
            r = x[2::3]
            return -np.sum(r)

        # Constraints
        constraints = []
        bounds = []

        # 1. Variable bounds
        for i in range(n):
            # x in [0, 1], y in [0, 1], r in [0, 0.5]
            bounds.append((0, 1)) # x
            bounds.append((0, 1)) # y
            bounds.append((0, 0.5)) # r

        # 2. Boundary constraints (Circle inside square)
        # x - r >= 0  => r - x <= 0
        # x + r <= 1  => x + r - 1 <= 0
        # Same for y
        # SLSQP expects constraints in form fun(x) >= 0 for inequality? 
        # Actually scipy minimize constraints: type 'ineq' means fun(x) >= 0.
        
        # Helper to add constraint
        def make_constraint(idx, axis):
            # axis 0 for x, 1 for y
            # x_i - r_i >= 0
            def fun(x):
                val = x[3*idx + axis] - x[3*idx + 2]
                return val
            return fun

        def make_constraint_upper(idx, axis):
            # 1 - (x_i + r_i) >= 0
            def fun(x):
                val = 1.0 - (x[3*idx + axis] + x[3*idx + 2])
                return val
            return fun

        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': make_constraint(i, 0)}) # x >= r
            constraints.append({'type': 'ineq', 'fun': make_constraint_upper(i, 0)}) # x + r <= 1
            constraints.append({'type': 'ineq', 'fun': make_constraint(i, 1)}) # y >= r
            constraints.append({'type': 'ineq', 'fun': make_constraint_upper(i, 1)}) # y + r <= 1

        # 3. Non-overlap constraints
        # dist(i, j) >= r_i + r_j
        # sqrt((xi-xj)^2 + (yi-yj)^2) - (ri + rj) >= 0
        for i in range(n):
            for j in range(i + 1, n):
                def overlap_constraint(idx_i=i, idx_j=j):
                    def fun(x):
                        xi, yi = x[3*idx_i], x[3*idx_i + 1]
                        xj, yj = x[3*idx_j], x[3*idx_j + 1]
                        ri, rj = x[3*idx_i + 2], x[3*idx_j + 2]
                        
                        dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                        return dist - (ri + rj)
                    return fun
                constraints.append({'type': 'ineq', 'fun': overlap_constraint()})

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False})
            
            if res.success or res.fun < 0: # If we found a solution
                curr_sum = -res.fun
                if curr_sum > best_sum_radii:
                    # Extract solution
                    best_centers = np.array(res.x[0::3], res.x[1::3]).T
                    best_radii = res.x[2::3]
                    best_sum_radii = curr_sum
        except Exception as e:
            # If optimization fails, continue to next run
            pass

    # Final validation and cleaning
    # Ensure no NaNs
    if np.isnan(best_centers).any() or np.isnan(best_radii).any():
        # Fallback to a safe grid
        best_centers, best_radii, _ = safe_grid_packing(n)
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, best_sum_radii

def safe_grid_packing(n):
    """Fallback simple grid packing."""
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    centers = []
    # Adjust spacing to fit
    # Simple grid
    step_x = 1.0 / (cols + 1)
    step_y = 1.0 / (rows + 1)
    
    r = min(step_x, step_y) / 2
    
    cnt = 0
    for r_idx in range(rows):
        for c_idx in range(cols):
            if cnt < n:
                x = (c_idx + 1) * step_x
                y = (r_idx + 1) * step_y
                centers.append([x, y])
                cnt += 1
    
    centers = np.array(centers[:n])
    radii = np.full(n, r)
    return centers, radii, np.sum(radii)

if __name__ == "__main__":
    centers, radii, s_r = run_packing()
    print(f"Sum of radii: {s_r}")
    print(f"Centers:\n{centers}")
    print(f"Radii:\n{radii}")
