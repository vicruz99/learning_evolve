import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the placement and radii of 26 circles in a unit square to maximize sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    N = 26
    np.random.seed(42)

    # 1. Initialization: Hexagonal Staggered Grid
    # We generate rows. If a row is even, we shift it by r to create a hexagonal pattern.
    # We distribute N circles across rows.
    # 26 circles: 5, 5, 5, 5, 5, 1 (Total 6 rows)
    # We start with r=0.1 and try to fit them.
    
    r_init = 0.1
    centers_init = []
    
    # Define row counts to sum to 26
    # 5 rows of 5 = 25, plus 1 circle.
    # To save height, we try to fit the 6th row or squeeze into existing.
    # A hexagonal packing of 5x5 rows uses height ~ 0.89.
    # Let's try 6 rows: 5, 4, 5, 4, 5, 3 (Sum = 26)
    row_counts = [5, 4, 5, 4, 5, 3]
    
    # Calculate y spacing based on hexagonal vertical distance
    dy = r_init * np.sqrt(3)
    # But to be safe and fit in [0,1], we normalize y distribution later or just estimate.
    # Let's place them at estimated positions first.
    
    y = r_init
    for i, count in enumerate(row_counts):
        # Calculate x positions for 'count' circles
        # Available width is 1. Spacing 2*r_init.
        # We center the row of circles in [0, 1].
        total_width_circles = (count - 1) * 2 * r_init
        x_start = r_init + (1 - (total_width_circles + 2*r_init)) / 2
        
        # Stagger even rows (0-indexed)
        if i % 2 == 1:
            shift = r_init
            x_start += shift # Shift right
            
        x_positions = x_start + np.arange(count) * 2 * r_init
        
        # If x_positions go out of bounds, we clamp them for initialization (optimizer will fix)
        # But for a good start, we ensure they are inside.
        
        for x in x_positions:
            centers_init.append([x, y])
        
        y += dy
        
    centers_init = np.array(centers_init)
    
    # Flatten variables for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    # Initial radii
    radii_init = np.full(N, r_init)
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # 2. Optimization Setup
    def objective(vars):
        # Maximize sum of radii => Minimize negative sum
        r_vars = vars[2::3]
        return -np.sum(r_vars)

    def constraint_non_overlap(vars):
        centers = vars.reshape(-1, 3)[:, :2]
        radii = vars.reshape(-1, 3)[:, 2]
        
        violations = []
        for i in range(N):
            for j in range(i + 1, N):
                dist_sq = np.sum((centers[i] - centers[j])**2)
                r_sum = radii[i] + radii[j]
                # Constraint: dist >= r_sum  => dist^2 >= r_sum^2
                violations.append(dist_sq - r_sum**2)
        return violations

    def constraint_boundary_left(vars):
        # x >= r => x - r >= 0
        centers = vars.reshape(-1, 3)[:, :2]
        radii = vars.reshape(-1, 3)[:, 2]
        return centers[:, 0] - radii

    def constraint_boundary_right(vars):
        # x + r <= 1 => 1 - x - r >= 0
        centers = vars.reshape(-1, 3)[:, :2]
        radii = vars.reshape(-1, 3)[:, 2]
        return 1.0 - centers[:, 0] - radii

    def constraint_boundary_bottom(vars):
        # y >= r
        centers = vars.reshape(-1, 3)[:, :2]
        radii = vars.reshape(-1, 3)[:, 2]
        return centers[:, 1] - radii

    def constraint_boundary_top(vars):
        # y + r <= 1
        centers = vars.reshape(-1, 3)[:, :2]
        radii = vars.reshape(-1, 3)[:, 2]
        return 1.0 - centers[:, 1] - radii

    # Constraints dictionary for SLSQP
    constraints = []
    
    # Non-overlap
    constraints.append({
        'type': 'ineq',
        'fun': constraint_non_overlap
    })
    
    # Boundaries
    constraints.append({'type': 'ineq', 'fun': constraint_boundary_left})
    constraints.append({'type': 'ineq', 'fun': constraint_boundary_right})
    constraints.append({'type': 'ineq', 'fun': constraint_boundary_bottom})
    constraints.append({'type': 'ineq', 'fun': constraint_boundary_top})

    # Bounds for variables [x, y, r]
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1) for _ in range(2 * N)] + [(0, 0.5) for _ in range(N)]

    # 3. Run Optimization
    # SLSQP is generally good for constrained non-linear problems
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                      options={'maxiter': 1000, 'ftol': 1e-9})
    
    # 4. Extract and Validate Solution
    opt_vars = result.x
    centers_opt = opt_vars.reshape(-1, 3)[:, :2]
    radii_opt = opt_vars.reshape(-1, 3)[:, 2]
    
    # Post-processing: Ensure strict validity (clip tiny errors)
    # 1. Clip centers to ensure they are not outside [r, 1-r]
    for i in range(N):
        r = radii_opt[i]
        x = centers_opt[i, 0]
        y = centers_opt[i, 1]
        
        # Enforce boundaries
        if x - r < 0: x = r
        if x + r > 1: x = 1 - r
        if y - r < 0: y = r
        if y + r > 1: y = 1 - r
        
        centers_opt[i, 0] = x
        centers_opt[i, 1] = y

    # 2. Verify non-overlap and fix if minor numerical issues persist
    # (Though SLSQP should handle this, robust validation is good)
    for i in range(N):
        for j in range(i + 1, N):
            c1 = centers_opt[i]
            c2 = centers_opt[j]
            r1 = radii_opt[i]
            r2 = radii_opt[j]
            
            dist = np.sqrt(np.sum((c1 - c2)**2))
            min_dist = r1 + r2
            
            if dist < min_dist:
                # Push apart along the vector connecting centers
                if dist > 1e-12:
                    direction = (c2 - c1) / dist
                    # Move both away from center of mass by half the overlap
                    overlap = min_dist - dist
                    centers_opt[i] -= direction * (overlap / 2)
                    centers_opt[j] += direction * (overlap / 2)
                    # After moving, re-check boundaries for these two
                    for k in [i, j]:
                        rk = radii_opt[k]
                        centers_opt[k, 0] = np.clip(centers_opt[k, 0], rk, 1 - rk)
                        centers_opt[k, 1] = np.clip(centers_opt[k, 1], rk, 1 - rk)

    # Calculate final sum
    final_sum = np.sum(radii_opt)
    
    return centers_opt, radii_opt, final_sum

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any(): return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
