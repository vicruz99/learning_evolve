import numpy as np
from scipy.optimize import minimize


N = 26


def objective(x):
    """Maximize sum of radii (minimize negative)."""
    return -np.sum(x[2 * N:])


def objective_grad(x):
    """Analytical gradient of objective."""
    g = np.zeros(3 * N)
    g[2 * N:] = -1.0
    return g


def constraint_func(x):
    """
    Returns array of constraint values (must be >= 0).
    Constraints:
    - x_i - r_i >= 0 (left boundary)
    - 1 - x_i - r_i >= 0 (right boundary)
    - y_i - r_i >= 0 (bottom boundary)
    - 1 - y_i - r_i >= 0 (top boundary)
    - dist(i,j) - r_i - r_j >= 0 (non-overlap)
    """
    centers = x[:2 * N].reshape(N, 2)
    radii = x[2 * N:]

    result = []

    # Boundary constraints
    for i in range(N):
        result.append(centers[i, 0] - radii[i])
        result.append(1.0 - centers[i, 0] - radii[i])
        result.append(centers[i, 1] - radii[i])
        result.append(1.0 - centers[i, 1] - radii[i])

    # Non-overlap constraints
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            result.append(dist - radii[i] - radii[j])

    return np.array(result)


def create_hexagonal_init(r_init, rng_state=None):
    """Create hexagonal initial configuration with optional perturbation."""
    centers_list = []
    rows_config = [5, 6, 5, 6, 4]  # 26 total
    y = r_init

        x_offset = r_init * np.sqrt(3) / 2 if row_idx % 2 == 1 else 0
        x = r_init + x_offset

        for _ in range(count):
            if len(centers_list) >= N:
                break
            centers_list.append([x, y])
            x += 2 * r_init

        y += r_init * np.sqrt(3)

    centers = np.array(centers_list[:N])
    radii = np.full(N, r_init)

    # Ensure initial feasibility
    for i in range(N):
        max_r = min(centers[i, 0], 1 - centers[i, 0],
                     centers[i, 1], 1 - centers[i, 1], 0.48)
        radii[i] = min(radii[i], max_r - 1e-4)

    return centers, radii

        centers += np.random.uniform(-0.005, 0.005, (N, 2))

        for i in range(N):
            max_r = min(centers[i, 0], 1 - centers[i, 0],

            radii[i] = min(radii[i], max_r - 1e-4)
