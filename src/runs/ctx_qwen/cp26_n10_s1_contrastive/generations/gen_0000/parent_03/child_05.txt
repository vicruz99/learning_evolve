import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Hexagonal Grid
    # Estimate initial radius for packing 26 circles
    # A rough estimate allows for spacing. 26 circles implies approx 5x5 or 6x5 density.
    # r approx 1/10 = 0.1.
    r_init = 0.09
    
    centers_list = []
    # Hexagonal packing parameters
    dy = r_init * np.sqrt(3)
    dx = 2 * r_init
    
    # Generate rows
    row = 0
    while len(centers_list) < n:
        y = r_init + row * dy
        # Offset x for odd rows
        if row % 2 == 1:
            x_start = r_init + dx / 2
        else:
            x_start = r_init
            
        x = x_start
        while x <= 1.0 - r_init:
            if len(centers_list) < n:
                centers_list.append([x, y])
            x += dx
        row += 1
        
    centers = np.array(centers_list[:n])
    radii = np.full(n, 0.01) # Start with small valid radii
    
    # Flatten variables for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.hstack([centers.flatten(), radii])
    
    # Define objective function
    def objective(x_vec):
        c = x_vec[:2*n].reshape(n, 2)
        r = x_vec[2*n:]
        
        # 1. Objective: Maximize sum of radii (Minimize negative sum)
        obj_val = -np.sum(r)
        
        # 2. Penalty for boundary violations
        # Constraints: r <= x <= 1-r and r <= y <= 1-r
        # Equivalent to: x - r >= 0, 1 - x - r >= 0, etc.
        penalty = 0.0
        weight = 1000.0 # High penalty weight to enforce constraints
        
        # Boundary penalties
        # Left & Bottom
        val = r - c[:, 0] # r - x >= 0 violation
        penalty += np.sum(np.maximum(0, val)**2)
        val = r - c[:, 1] # r - y >= 0 violation
        penalty += np.sum(np.maximum(0, val)**2)
        
        # Right & Top
        val = r + c[:, 0] - 1.0 # x + r - 1 >= 0 violation
        penalty += np.sum(np.maximum(0, val)**2)
        val = r + c[:, 1] - 1.0 # y + r - 1 >= 0 violation
        penalty += np.sum(np.maximum(0, val)**2)
        
        # 3. Penalty for overlaps
        # Constraint: dist(i, j) >= r_i + r_j  =>  dist - (r_i + r_j) >= 0
        # Vectorized distance calculation
        c_diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dists = np.sqrt(np.sum(c_diff**2, axis=2))
        
        # Lower triangular part (i < j)
        i_indices, j_indices = np.tril_indices(n, -1)
        r_sum = r[i_indices] + r[j_indices]
        violations = r_sum - dists[i_indices, j_indices]
        
        penalty += np.sum(np.maximum(0, violations)**2)
        
        return obj_val + weight * penalty

    # Define bounds
    # Coords in [0, 1], Radii in [0, 0.5] (loose upper bound)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Optimization
    best_res = None
    best_score = float('inf')
    
    # Run optimization from the hex grid initialization
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 1000, 'ftol': 1e-12})
    
    if res.fun < best_score:
        best_score = res.fun
        best_res = res
        
    # Random restarts to avoid local minima
    # Try a few random initial configurations
    for _ in range(5):
        np.random.seed(None)
        random_centers = np.random.rand(n, 2) * 0.8 + 0.1 # Random in [0.1, 0.9]
        random_radii = np.full(n, 0.05)
        x0_rand = np.hstack([random_centers.flatten(), random_radii])
        
        res_rand = minimize(objective, x0_rand, method='L-BFGS-B', bounds=bounds,
                            options={'maxiter': 500})
        
        if res_rand.fun < best_score:
            best_score = res_rand.fun
            best_res = res_rand

    # Extract results
    best_x = best_res.x
    final_centers = best_x[:2*n].reshape(n, 2)
    final_radii = best_x[2*n:]
    
    # Post-processing: Ensure radii are strictly non-negative and clean
    final_radii = np.maximum(final_radii, 0.0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
