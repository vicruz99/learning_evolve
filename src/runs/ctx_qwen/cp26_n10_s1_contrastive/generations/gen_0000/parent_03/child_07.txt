def compute_penalty(centers, radii, penalty_weight):
    """
    Computes the penalty for constraint violations.
    Constraints:
    1. Circles inside unit square: r <= x <= 1-r, r <= y <= 1-r
    2. No overlaps: dist(i, j) >= r_i + r_j
    """

    # x - r >= 0
    vx = centers[:, 0]
    vy = centers[:, 1]

    # Overlap constraints
    # dist(i, j) >= r_i + r_j

    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
