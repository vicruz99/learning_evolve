import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_constraints(v):
    """
    Computes the constraint values for the optimization.
    Returns an array of values that must be >= 0.
    Constraints:
    1. x >= r
    2. 1 - x >= r
    3. y >= r
    4. 1 - y >= r
    5. (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    """
    r = v[0::3]
    x = v[1::3]
    y = v[2::3]
    
    num_constraints = 4 * N_CIRCLES + N_CIRCLES * (N_CIRCLES - 1) // 2
    cons = np.empty(num_constraints)
    idx = 0
    
    # Boundary constraints
    cons[idx:idx+N_CIRCLES] = x - r
    idx += N_CIRCLES
    cons[idx:idx+N_CIRCLES] = 1.0 - x - r
    idx += N_CIRCLES
    cons[idx:idx+N_CIRCLES] = y - r
    idx += N_CIRCLES
    cons[idx:idx+N_CIRCLES] = 1.0 - y - r
    idx += N_CIRCLES
    
    # Pairwise non-overlap constraints
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dr = r[i] + r[j]
            cons[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
            
    return cons

def objective(v):
    """Negative sum of radii to be minimized."""
    return -np.sum(v[0::3])

def get_initial_guess():
    """Generates a feasible initial grid configuration."""
    cols = 5
    rows = 5
    step_x = 1.0 / (cols + 1)
    step_y = 1.0 / (rows + 1)
    
    x0 = np.zeros(3 * N_CIRCLES)
    r_init = 0.04
    
    for i in range(N_CIRCLES):
        c = i % cols
        r_idx = i // cols
        x0[3*i] = r_init
        x0[3*i+1] = step_x * (c + 1)
        x0[3*i+2] = step_y * (r_idx + 1)
        
    return x0

def run_packing():
    # Bounds for r, x, y
    bounds = [(1e-5, 0.5)] * N_CIRCLES + [(1e-5, 1.0-1e-5)] * (2 * N_CIRCLES)
    lower = np.array([b[0] for b in bounds])
    upper = np.array([b[1] for b in bounds])
    
    cons = {'type': 'ineq', 'fun': compute_constraints}
    x0 = get_initial_guess()
    
    best_res = None
    best_val = -np.inf
    
    # Multi-start optimization
    for seed in range(8):
        np.random.seed(seed)
        x0_pert = x0 + np.random.normal(0, 0.008, 3 * N_CIRCLES)
        x0_pert = np.clip(x0_pert, lower, upper)
            
        try:
            res = minimize(objective, x0_pert, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
            if -res.fun > best_val:
                best_val = -res.fun
                best_res = res
        except:
            continue
            
    if best_res is None:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000})
        best_res = res
        
    v = best_res.x
    centers = np.column_stack((v[1::3], v[2::3]))
    radii = v[0::3]
    
    # Post-processing to guarantee strict validity
    cons_vals = compute_constraints(v)
    min_val = np.min(cons_vals)
    
    if min_val < -1e-11:
        # Shrink radii slightly to resolve any numerical constraint violations
        # The violation scales roughly with radius, so a small factor suffices
        shrink_factor = 1.0 - (abs(min_val) + 1e-12) / (2.0 * np.max(radii) + 1e-12)
        radii *= max(0.5, shrink_factor)
        centers = np.column_stack((v[1::3], v[2::3]))
        
    return centers, radii, np.sum(radii)
