

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we treat this as a joint optimization problem over circle centers and radii. The key insight is that for any fixed set of centers, the optimal radii can be found exactly by solving a Linear Programming (LP) problem. This transforms the original non-convex problem into a 52-dimensional optimization over centers only, where the objective function is the LP-optimal sum of radii.

Strategy:
1. **LP Formulation for Radii**: Given centers $\mathbf{c}_i$, maximize $\sum r_i$ subject to:
   - $0 \le r_i \le \text{dist}(\mathbf{c}_i, \text{boundary})$
   - $r_i + r_j \le \|\mathbf{c}_i - \mathbf{c}_j\|$ for all $i \ne j$
   This is a standard LP that can be solved efficiently with HiGHS.
2. **Center Optimization**: Use `scipy.optimize.minimize` with the `Powell` method (derivative-free, robust for non-smooth objectives) to maximize the LP objective over centers.
3. **Initialization & Multi-start**: Initialize centers on a hexagonal lattice (denser than square grid) perturbed slightly. Run multiple optimization seeds to escape local minima.
4. **Safety & Validation**: Clamp results to ensure strict feasibility within numerical tolerances required by `validate_packing`.

This block-coordinate ascent approach is highly effective for circle packing, converging quickly to near-optimal configurations while guaranteeing feasibility at each step.

import numpy as np
from scipy.optimize import minimize, linprog

def get_initial_centers(n=26):
    """Generate a hexagonal lattice initialization packed towards the center."""
    pts = []
    spacing = 0.19
    y = spacing
    row = 0
    while y < 1.0:
        x_start = spacing * 0.5 if (row % 2 == 1) else spacing
        x = x_start
        while x < 1.0:
            pts.append([x, y])
            x += spacing
        y += spacing * np.sqrt(3)/2
        row += 1
    pts = np.array(pts)
    # Sort by distance to square center to keep initial cluster compact
    center = np.array([0.5, 0.5])
    dists = np.linalg.norm(pts - center, axis=1)
    order = np.argsort(dists)
    return pts[order[:n]]

def solve_radii_lp(centers):
    """
    Solve the linear program to find optimal radii for fixed centers.
    Maximize sum(r_i) subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    c = -np.ones(n)  # Minimize -sum(r)
    
    # Upper bounds from distance to square boundaries
    bounds = []
    for i in range(n):
        d_min = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
        bounds.append((0.0, max(0.0, d_min)))
        
    # Pairwise distance constraints: r_i + r_j <= ||c_i - c_j||
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        # Ensure non-negative radii due to potential solver noise
        return np.maximum(0.0, res.x), -res.fun
    return np.zeros(n), 0.0

def objective(centers_flat):
    """Negative sum of radii to be minimized."""
    centers = centers_flat.reshape(-1, 2)
    _, sum_r = solve_radii_lp(centers)
    return -sum_r

def run_packing():
    np.random.seed(42)
    n = 26
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Candidate initial configurations
    candidates = [get_initial_centers(n)]
    
    # Add perturbed versions of the grid initialization
    for _ in range(4):
        base = get_initial_centers(n)
        perturbed = base + np.random.randn(*base.shape) * 0.03
        perturbed = np.clip(perturbed, 0.02, 0.98)
        candidates.append(perturbed)
        
    # Optimize centers for each candidate
    for cand in candidates:
        res = minimize(objective, cand.flatten(), method='Powell', 
                       options={'xtol': 1e-7, 'ftol': 1e-7, 'maxiter': 2000})
        curr_centers = res.x.reshape(-1, 2)
        curr_radii, curr_sum = solve_radii_lp(curr_centers)
        
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = curr_centers
            best_radii = curr_radii
            
    # Return the best found packing
    return best_centers, best_radii, best_sum
