import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers_flat, r, n):
    """Compute sum of squared constraint violations for a fixed radius r."""
    centers = centers_flat.reshape(n, 2)
    penalty = 0.0
    
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        if x < r:
            penalty += (r - x)**2
        elif x > 1.0 - r:
            penalty += (x - (1.0 - r))**2
        if y < r:
            penalty += (r - y)**2
        elif y > 1.0 - r:
            penalty += (y - (1.0 - r))**2
            
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            if dist < 2.0 * r:
                penalty += (2.0 * r - dist)**2
                
    return penalty

def get_initial_centers(n=26):
    """Generate a hexagonal lattice initial guess inside the unit square."""
    centers = []
    r0 = 0.105
    y = r0
    row = 0
    while len(centers) < n:
        x = r0
        if row % 2 == 1:
            x = r0 + r0  # Shift odd rows
        while x <= 1.0 - r0 and len(centers) < n:
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    return np.array(centers[:n])

def run_packing():
    n = 26
    init_centers = get_initial_centers(n)
    
    low = 0.05
    high = 0.11
    best_r = low
    best_centers = init_centers.copy()
    
    # Binary search for the maximum feasible equal radius
    for _ in range(35):
        mid_r = (low + high) / 2.0
        
        # Deterministic perturbation to break symmetry and escape local minima
        pert = init_centers + np.arange(n).reshape(-1, 1) * 1e-5
        pert = np.clip(pert, 0.0, 1.0)
        
        res = minimize(compute_penalty, pert.ravel(), args=(mid_r, n),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * (2 * n),
                       options={'ftol': 1e-12, 'maxiter': 3000, 'disp': False})
        
        # If penalty is sufficiently low, configuration is valid
        if res.fun < 1e-7:
            best_r = mid_r
            best_centers = res.x.reshape(n, 2)
            low = mid_r
        else:
            high = mid_r
            
    # Final high-precision refinement at the best found radius
    final_res = minimize(compute_penalty, best_centers.ravel(), args=(best_r, n),
                         method='L-BFGS-B', bounds=[(0.0, 1.0)] * (2 * n),
                         options={'ftol': 1e-14, 'maxiter': 5000, 'disp': False})
    final_centers = final_res.x.reshape(n, 2)

    final_r = best_r * 0.99999
    for i in range(n):
        final_centers[i, 0] = np.clip(final_centers[i, 0], final_r, 1.0 - final_r)
        final_centers[i, 1] = np.clip(final_centers[i, 1], final_r, 1.0 - final_r)
        
    radii = np.full(n, final_r)
    sum_r = float(np.sum(radii))
    
    return final_centers, radii, sum_r
