import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def obj_func(vars_):
    """Objective: maximize r (minimize -r)"""
    return -vars_[-1]

def constr_boundary(vars_, n):
    """Boundary constraints: each circle must be inside [0,1]x[0,1]"""
    r = vars_[-1]
    coords = vars_[:-1].reshape(-1, 2)
    cons = []
    for c in coords:
        cons.extend([c[0]-r, 1-c[0]-r, c[1]-r, 1-c[1]-r])
    return np.array(cons)

def constr_overlap(vars_, n):
    """Non-overlap constraints: distance between any pair >= 2r"""
    r = vars_[-1]
    coords = vars_[:-1].reshape(-1, 2)
    cons = []
    for i in range(n):
        diff = coords[i+1:] - coords[i]
        d2 = np.sum(diff**2, axis=1)
        cons.extend(d2 - 4*r**2)
    return np.array(cons)

def run_packing():
    n = N_CIRCLES
    
    # Initial feasible guess: hexagonal lattice pattern
    coords = []
    s = 0.24  # spacing large enough to be feasible with r=0.09
    for i in range(6):
        for j in range(6):
            x = 0.1 + j * s + (i % 2) * s / 2
            y = 0.1 + i * s * np.sqrt(3) / 2
            if x < 0.9 and y < 0.9 and len(coords) < n:
                coords.append([x, y])
    while len(coords) < n:
        coords.append([0.5, 0.5])
    coords = np.array(coords[:n])
    
    # Small perturbation to break symmetry and aid optimization
    coords += np.random.default_rng(42).randn(*coords.shape) * 0.005
    
    # Concatenate coordinates and initial radius
    x0 = np.concatenate([coords.flatten(), [0.09]])
    
    constraints = [
        {'type': 'ineq', 'fun': constr_boundary, 'args': (n,)},
        {'type': 'ineq', 'fun': constr_overlap, 'args': (n,)}
    ]
    
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-5, 0.5)]
    
    # Run SLSQP optimizer
    res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                   
    if res.success:
        r_opt = res.x[-1]
        c_opt = res.x[:-1].reshape(-1, 2)
        # Ensure strict boundary satisfaction within tolerance
        c_opt = np.clip(c_opt, r_opt, 1.0 - r_opt)
        radii = np.full(n, r_opt)
        return c_opt, radii, float(26 * r_opt)
    else:
        # Fallback to a valid feasible packing if optimization fails
        r_fall = 0.08
        c_fall = coords.copy()
        c_fall = np.clip(c_fall, r_fall, 1.0 - r_fall)
        return c_fall, np.full(n, r_fall), float(26 * r_fall)
