import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a penalty-based optimization approach starting from a hexagonal grid.
    """
    N = 26
    np.random.seed(42)  # For reproducibility

    # 1. Initialize centers and radii
    # Start with a hexagonal-like grid to ensure a dense packing initially
    # Radius 0.05 is safe and small enough to not overlap
    r_init = 0.05
    centers = []
    
    # Generate hexagonal grid positions
    # Row spacing: sqrt(3) * diameter / 2 = sqrt(3) * r? No.
    # In hex packing of radius r, vertical dist between rows is sqrt(3)*r.
    # Horizontal dist between centers is 2*r.
    # Rows alternate by shift r.
    
    y = r_init
    row_shift = 0.0
    while y + r_init <= 1.0 + 1e-9:
        x = r_init + row_shift
        while x + r_init <= 1.0 + 1e-9:
            if len(centers) < N:
                centers.append([x, y])
            x += 2 * r_init
        y += np.sqrt(3) * r_init
        # Toggle shift: 0 -> r -> 0? 
        # Row 0: x starts at r. (shift 0)
        # Row 1: x should start at 2r? No, for hex packing, centers are at distance 2r.
        # If row 0 has centers at r, 3r, 5r...
        # Row 1 centers should be at 2r, 4r, 6r... to maintain distance 2r from neighbors.
        # So shift is r.
        # But if row 1 starts at 2r, next row (row 2) should start at r again.
        # So shift toggles between 0 and r.
        row_shift = r_init if row_shift == 0 else 0.0

    # If we didn't get 26 centers (unlikely with r=0.05), pad with random valid positions
    while len(centers) < N:
        # Find a spot far from others
        # Simple random retry
        cx, cy = np.random.rand(2)
        # Check distance to existing
        min_d = np.inf
        for c in centers:
            d = np.hypot(cx - c[0], cy - c[1])
            if d < min_d: min_d = d
        if min_d > 2 * r_init: # Should be at least 2r apart
            centers.append([cx, cy])
        else:
            # Just place it, optimizer will move it
            centers.append([cx, cy])

    centers = np.array(centers[:N])
    radii = np.full(N, r_init)

    # 2. Define Optimization Variables
    # Variables: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    # Shape: (3 * N,)
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]

    # Bounds: x, y in [0, 1], r in [0, 0.5] (max possible radius is 0.5)
    bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(0.0, 0.5)] * N

    # 3. Objective Function with Penalty
    # We want to maximize sum(radii), so minimize -sum(radii)
    # Penalty for overlaps and boundary violations
    
    # Penalty weight. Higher is stricter.
    # We can start moderate and the optimizer will handle it.
    # 1000 is usually sufficient for strict constraints.
    PENALTY_WEIGHT = 5000.0 

    def objective(vars):
        c = vars[:2*N].reshape(N, 2)
        r = vars[2*N:]
        
        # Objective: -sum of radii
        val = -np.sum(r)
        
        penalty = 0.0
        
        # Pairwise Overlap Penalty
        # Condition: dist(c_i, c_j) >= r_i + r_j
        # Violation: (r_i + r_j)^2 - dist^2 > 0
        # We penalize the square of the violation to make it smooth and strongly convex-ish locally
        # Violation V = (r_i + r_j)^2 - ((x_i-x_j)^2 + (y_i-y_j)^2)
        # If V > 0, we have overlap.
        
        # Vectorized computation for speed? 
        # N=26 is small, loop is fine.
        for i in range(N):
            ri = r[i]
            xi, yi = c[i]
            
            # Boundary penalties
            # x - r >= 0 -> if x-r < 0, penalty (x-r)^2
            # 1 - x - r >= 0 -> if 1-x-r < 0, penalty (1-x-r)^2
            
            # Left
            gap_l = xi - ri
            if gap_l < 0:
                penalty += gap_l * gap_l
            
            # Right
            gap_r = 1.0 - xi - ri
            if gap_r < 0:
                penalty += gap_r * gap_r
                
            # Bottom
            gap_b = yi - ri
            if gap_b < 0:
                penalty += gap_b * gap_b
                
            # Top
            gap_t = 1.0 - yi - ri
            if gap_t < 0:
                penalty += gap_t * gap_t

            # Pairwise interactions with subsequent circles
            for j in range(i + 1, N):
                rj = r[j]
                xj, yj = c[j]
                
                dx = xi - xj
                dy = yi - yj
                dist_sq = dx*dx + dy*dy
                min_dist = ri + rj
                min_dist_sq = min_dist * min_dist
                
                # Violation if dist_sq < min_dist_sq
                # We penalize (min_dist_sq - dist_sq)^2
                # This is zero when touching or separated, positive when overlapping
                overlap = min_dist_sq - dist_sq
                if overlap > 0:
                    penalty += overlap * overlap

        return val + PENALTY_WEIGHT * penalty

    # 4. Optimization
    # Use L-BFGS-B as it handles bounds well and is efficient
    # We might need multiple restarts or a high ftol
    res = minimize(
        objective, 
        x0, 
        method='L-BFGS-B', 
        bounds=bounds,
        options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-8}
    )
    
    # 5. Extract Result
    best_vars = res.x
    final_centers = best_vars[:2*N].reshape(N, 2)
    final_radii = best_vars[2*N:]
    
    # 6. Validation and Correction
    # Due to numerical precision, slight overlaps might exist.
    # We can perform a simple relaxation or clipping if needed, 
    # but the penalty method usually finds a very good local optimum.
    # Let's check validity. If invalid, we might need to shrink radii slightly.
    
    # Check overlaps
    valid = True
    min_gap = 1.0
    
    for i in range(N):
        x, y = final_centers[i]
        r = final_radii[i]
        # Check boundaries
        if x < r - 1e-9 or x > 1-r + 1e-9 or y < r - 1e-9 or y > 1-r + 1e-9:
             # If boundary violation, shrink radius slightly to fit
             # But optimization should have handled this.
             pass

    for i in range(N):
        for j in range(i + 1, N):
            dist = np.sqrt(np.sum((final_centers[i] - final_centers[j])**2))
            r_sum = final_radii[i] + final_radii[j]
            gap = dist - r_sum
            if gap < -1e-9: # Strict overlap
                valid = False
            
            if gap < min_gap:
                min_gap = gap

    # If there are tiny overlaps, we can scale down radii uniformly just to be safe?
    # Or better, just return. The validate function allows 1e-12 tolerance.
    # But let's ensure we are safe.
    # If min_gap is slightly negative (e.g. -1e-10), it's acceptable.
    
    # Just in case, let's do a small scaling if overlaps are significant
    if valid or min_gap >= -1e-9:
        pass
    else:
        # Reduce radii slightly to clear overlaps
        # Factor to reduce
        scale_factor = 1.0 + min_gap / 2.0 # Conservative
        if scale_factor < 0.9: scale_factor = 0.999
        final_radii *= scale_factor

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
