import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-6), 1e-4

def objective_slqp(vars_vec):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def constraint_func_slqp(vars_vec):
    """
    Inequality constraints: dist_sq >= (r_i + r_j)^2.
    Boundary constraints are satisfied by the (r, u, v) parameterization.
    """
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def make_vars_from_lp(centers):
    """Map physical centers to (r, u, v) optimization parameters using LP radii."""
    r, _ = solve_radii_lp(centers)
    r = np.clip(r, 1e-6, 0.499)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    vars0 = np.empty(3 * N)
    vars0[0::3] = r
    vars0[1::3] = u
    vars0[2::3] = v
    return vars0

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [5,5,6,5,5], [7,5,5,5,4], [4,5,5,5,7], [5,7,5,5,4]
    ]
    for pat in patterns:
        for _ in range(3):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            pts = np.array(pts[:N])
            pts -= 0.5
            scale = rng.uniform(0.85, 1.15)
            pts *= scale
            pts += 0.5
            rot = rng.uniform(-0.3, 0.3)
            c, s = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[c, -s], [s, c]])
            pts += rng.uniform(-0.015, 0.015, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    # Force-directed layouts
    for seed in range(12):
        rng_fd = np.random.RandomState(seed)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        r_curr = np.full(N, 0.05)
        for step in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 20.0 * np.maximum(0, r_curr - pts[:, d])
                forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            pts += 0.006 * (0.995**step) * forces
            pts = np.clip(pts, 0.02, 0.98)
            for i in range(N):
                d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                d_pair = np.min(np.linalg.norm(pts[i] - pts, axis=1) * (1 - np.eye(N)[i]))
                r_curr[i] = 0.9 * min(d_wall, d_pair/2.0)
        inits.append(pts)
        
    # Perturbed grids
    for seed in range(8):
        rng_g = np.random.RandomState(seed)
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng_g.uniform(-0.03, 0.03, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    return inits

def run_sa_lp(centers0, rng):
    """Simulated Annealing on centers maximizing LP sum of radii."""
    curr_c = centers0.copy()
    curr_r, curr_sum = solve_radii_lp(curr_c)
    
    best_c = curr_c.copy()
    best_r = curr_r.copy()
    best_sum = curr_sum
    
    temp = 0.04
    step = 0.04
    n_iter = 1000
    
    for it in range(n_iter):
        idx = rng.integers(N)
        direction = rng.standard_normal(2)
        direction /= np.linalg.norm(direction)
        
        jump = 1.0 + 5.0 * (rng.random() < 0.04)
        new_c = curr_c.copy()
        new_c[idx] = np.clip(curr_c[idx] + step * jump * direction, 0.005, 0.995)
        
        new_r, new_sum = solve_radii_lp(new_c)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-7 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_c = new_c
            curr_r = new_r
            curr_sum = new_sum
            if new_sum > best_sum:
                best_c = curr_c.copy()
                best_r = curr_r.copy()
                best_sum = new_sum
            step = max(0.001, step * 0.985)
        else:
            step = min(0.06, step * 1.01)
        temp *= 0.9993
        
    return best_c, best_r, best_sum

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func_slqp}
    
    best_vars = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    inits = generate_inits(rng)
    
    # Phase 1: SA on centers to maximize LP sum of radii
    for pts in inits:
        c_opt, r_opt, s_opt = run_sa_lp(pts, rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt
            best_r = r_opt
            
    # Phase 2: SLSQP refinement from best LP config
    if best_c is not None:
        vars0 = make_vars_from_lp(best_c)
        vars0[0::3] *= 0.99
        vars0[0::3] = np.clip(vars0[0::3], 1e-6, 0.49)
        
        try:
            res = minimize(objective_slqp, vars0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13})
            if res.success and np.min(constraint_func_slqp(res.x)) >= -1e-8:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Perturbation & SLSQP to escape local minima
    if best_vars is not None:
        for k in range(25):
            vars0 = best_vars.copy()
            vars0[1::3] += rng.normal(0, 0.04, N)
            vars0[2::3] += rng.normal(0, 0.04, N)
            vars0[0::3] += rng.normal(0, 0.001, N)
            vars0[0::3] = np.clip(vars0[0::3], 1e-6, 0.49)
            vars0[1::3] = np.clip(vars0[1::3], 0.0, 1.0)
            vars0[2::3] = np.clip(vars0[2::3], 0.0, 1.0)
            
            try:
                res = minimize(objective_slqp, vars0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
                if res.success and np.min(constraint_func_slqp(res.x)) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
            except Exception:
                continue
                
        # High-precision polish
        try:
            res = minimize(objective_slqp, best_vars, method='SLSQP', bounds=bounds,

            if res.success and np.min(constraint_func_slqp(res.x)) >= -1e-9:

                best_sum = -res.fun
        except Exception:
            pass
            
    # Phase 4: LP Post-Processing
    # Re-extract centers from SLSQP and solve LP for exact radii assignment
    if best_vars is not None:
        r_tmp = best_vars[0::3]
        u_tmp = best_vars[1::3]
        v_tmp = best_vars[2::3]
        x_tmp = r_tmp + u_tmp * (1.0 - 2.0 * r_tmp)

        final_centers = np.column_stack((x_tmp, y_tmp))
        
        lp_r, lp_sum = solve_radii_lp(final_centers)

            best_sum = lp_sum
            best_vars = make_vars_from_lp(final_centers)

        pts = generate_inits(np.random.default_rng(0))[0]

        best_sum = np.sum(best_vars[0::3])
        
    # Reconstruct centers from optimized parameters
    r_opt = best_vars[0::3]
    u_opt = best_vars[1::3]
    v_opt = best_vars[2::3]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    # Ensure radii are strictly positive and valid
    r_final = np.maximum(r_opt, 0.0)
    
    return centers, r_final, float(best_sum)
