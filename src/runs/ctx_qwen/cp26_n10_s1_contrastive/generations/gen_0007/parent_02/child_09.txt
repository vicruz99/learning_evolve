import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A = np.zeros((num_con, n))
    b = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_i = [x, 1.0 - x, y, 1.0 - y]
        for val in bounds_i:
            A[k, i] = 1.0
            b[k] = val
            k += 1
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            b[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def constraints(params):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    # Parameterization guarantees boundary containment
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = np.clip(radii, 1e-6, 0.499)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    p = np.empty(3 * N)
    p[0::3] = r
    p[1::3] = u
    p[2::3] = v
    return p

def from_params(params):
    """Reconstruct centers from (r, u, v) parameters."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y))

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], 
        [5,5,5,5,6], [6,4,6,5,5], [5,6,4,6,5],
        [4,5,6,5,6], [6,6,5,5,4], [5,5,6,5,5], [7,5,5,5,4]
    ]
    
    for pat in patterns:
        for _ in range(6):
            pts = []
            re = 0.095
            y = re
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * re
                x = re + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * re
                y += re * np.sqrt(3.0)
                
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.85, 1.15) + 0.5
            rot = rng.uniform(-0.3, 0.3)
            c, s = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pts += rng.uniform(-0.02, 0.02, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    for seed in range(20):
        rng_fd = np.random.RandomState(seed)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        rc = np.full(N, 0.05)
        
        for step in range(300):
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
            
            for d in range(2):
                forces[:, d] += 20.0 * np.maximum(0, rc - pts[:, d])
                forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - rc))
                
            step_size = 0.006 * (0.995**step)
            pts += step_size * forces
            pts = np.clip(pts, 0.02, 0.98)
            
            for i in range(N):
                dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                dists_to_others = np.linalg.norm(pts[i] - pts, axis=1)
                dists_to_others[i] = np.inf
                dp = np.min(dists_to_others)
                rc[i] = 0.85 * min(dw, dp/2.0)
                
        inits.append(pts)
        
    return inits

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_vars = None
    best_sum = -np.inf
    
    inits = generate_inits(rng)
    
    # Phase 1: Broad search from diverse initializations using SLSQP
    for pts in inits:
        r_init, _ = solve_lp(pts)
        p0 = to_params(pts, r_init * 0.98)
        
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                cons_val = constraints(res.x)
                if np.min(cons_val) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    if best_vars is None:
        pts = inits[0]
        r_init, _ = solve_lp(pts)
        best_vars = to_params(pts, r_init)
        best_sum = np.sum(best_vars[0::3])
        
    # Reconstruct best centers for Phase 2
    best_c = from_params(best_vars)
    best_r, best_sum_lp = solve_lp(best_c)
    best_sum = max(best_sum, best_sum_lp)
    if best_sum_lp > best_sum:
        best_vars = to_params(best_c, best_r)
        
    # Phase 2: LP-Driven Simulated Annealing on Centers
    curr_c = best_c.copy()
    curr_r = best_r.copy()
    curr_s = best_sum
    temp = 0.05
    step = 0.03
    
    for it in range(3000):
        op = rng.choice(['move', 'swap', 'jump'])
        old_c = curr_c.copy()
        
        if op == 'move':
            idx = rng.integers(N)
            d = rng.standard_normal(2)
            d /= np.linalg.norm(d) + 1e-9
            curr_c[idx] = np.clip(curr_c[idx] + step * d, 0.005, 0.995)
        elif op == 'swap':
            i, j = rng.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        else:
            idx = rng.integers(N)
            curr_c[idx] = rng.uniform(0.1, 0.9, 2)
            
        nr, ns = solve_lp(curr_c)
        delta = ns - curr_s
        
        # Acceptance criterion
        accept_prob = np.exp(np.clip(delta / (temp + 1e-12), -500, 500))
        if delta > 0 or (rng.random() < accept_prob):
            # Move accepted
            curr_r = nr.copy()
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                best_r = curr_r.copy()
                best_vars = to_params(best_c, best_r)
        else:
            # Move rejected, revert centers
            curr_c = old_c
            
        temp *= 0.999
        step = max(0.002, step * 0.9995)
        
    # Phase 3: High-precision SLSQP polish on the refined configuration
    p0 = to_params(best_c, best_r * 0.99)
    try:
        res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
        if res.success:
            cons_val = constraints(res.x)
            if np.min(cons_val) >= -1e-9:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x.copy()
    except Exception:
        pass
        
    # Phase 4: Perturbation refinement to escape local minima
    for _ in range(25):
        p0 = best_vars.copy()
        p0[1::3] += rng.uniform(-0.02, 0.02, N)
        p0[2::3] += rng.uniform(-0.02, 0.02, N)
        p0[0::3] += rng.uniform(-0.002, 0.002, N)
        
        p0[0::3] = np.clip(p0[0::3], 1e-6, 0.49)
        p0[1::3] = np.clip(p0[1::3], 0.0, 1.0)
        p0[2::3] = np.clip(p0[2::3], 0.0, 1.0)
        
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                cons_val = constraints(res.x)
                if np.min(cons_val) >= -1e-9:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 5: Final ultra-high-precision polish
    p0 = best_vars.copy()
    p0[0::3] *= 0.9999  # Slight shrink for strict interior feasibility
    try:
        res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res.success:
            cons_val = constraints(res.x)
            if np.min(cons_val) >= -1e-9:
                best_vars = res.x
                best_sum = -res.fun
    except Exception:
        pass
        
    # Reconstruct final centers and radii
    centers = from_params(best_vars)
    radii = best_vars[0::3]
    
    return centers, np.maximum(radii, 0.0), float(best_sum)
