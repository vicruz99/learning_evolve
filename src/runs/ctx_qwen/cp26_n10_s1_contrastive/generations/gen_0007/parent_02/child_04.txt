import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = N * (N - 1) // 2

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_constraints = 4 * n + NUM_PAIRS
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-6), 1e-5

def objective_slsqp(vars_vec):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def constraint_slsqp(vars_vec):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2."""
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    w = vars_vec[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    return dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX]

def centers_to_params(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = w
    return params

def generate_hex_centers(seed, rot, scale, offset_x, offset_y):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    pts += np.array([offset_x, offset_y])
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_layout(seed):
    """Generates an organic layout using force-directed relaxation."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.15, 0.85, (N, 2))
    r_curr = np.full(N, 0.06)
    
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.006 * (0.995**step)
        pts += step_size * forces
        pts = np.clip(pts, 0.03, 0.97)
        
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists_to_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_to_others[i] = np.inf
            d_pair = np.min(dists_to_others)
            r_curr[i] = 0.85 * min(d_wall, d_pair/2.0)
            
    return pts

def optimize_sa(init_centers, rng, n_steps=4000):
    """Simulated Annealing on centers using LP objective."""
    c = init_centers.copy()
    r, s = solve_lp(c)
    best_c, best_r, best_s = c.copy(), r.copy(), s
    
    temp = 0.04
    step = 0.025
    
    for _ in range(n_steps):
        idx = rng.integers(N)
        nc = c.copy()
        nc[idx] += rng.normal(0, step, 2)
        nc[idx] = np.clip(nc[idx], 0.005, 0.995)
        
        nr, ns = solve_lp(nc)
        delta = ns - s
        
        if delta > 0 or (temp > 1e-7 and rng.random() < np.exp(delta / temp)):
            c, r, s = nc, nr, ns
            if ns > best_s:
                best_c, best_r, best_s = c.copy(), r.copy(), ns
                
        temp *= 0.9993
        step = max(0.0015, step * 0.9995)
        
    return best_c, best_r, best_s

def run_slsqp_polish(params0, rng):
    """Runs SLSQP optimization with multiple perturbation attempts."""
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_slsqp}
    
    best_vars = None
    best_sum = -np.inf
    
    # Try base + several perturbations
    attempts = [params0.copy()]
    for _ in range(8):
        pert = params0.copy()
        pert[1::3] += rng.uniform(-0.03, 0.03, N)
        pert[2::3] += rng.uniform(-0.03, 0.03, N)
        pert[0::3] += rng.uniform(-0.001, 0.001, N)
        pert[0::3] = np.clip(pert[0::3], 1e-6, 0.49)
        pert[1::3] = np.clip(pert[1::3], 0.0, 1.0)
        pert[2::3] = np.clip(pert[2::3], 0.0, 1.0)
        attempts.append(pert)
        
    for x0 in attempts:
        try:
            res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                if np.min(constraint_slsqp(res.x)) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    return best_vars, best_sum
