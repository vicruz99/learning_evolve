To maximize the sum of radii for 26 circles in a unit square, we employ a multi-phase optimization strategy that combines global exploration, precise local refinement, and topology-aware local search:

1. **Diverse Initialization**: Generate ~100 starting configurations using perturbed hexagonal lattices (with varied row counts, rotations, scales, and shears), force-directed organic layouts, and perturbed grids. This ensures coverage of multiple contact graph topologies.
2. **LP-Radius Solver**: For any fixed center configuration, the optimal radii are found by solving a Linear Program (LP). This is exact, fast, and naturally handles boundary and pairwise non-overlap constraints.
3. **Joint SLSQP Optimization**: Use the `(r, u, v)` parameterization where `x = r + u(1-2r)` and `y = r + v(1-2r)` with `u,v \in [0,1]`. This automatically satisfies boundary constraints, leaving only pairwise non-overlap constraints for SLSQP. We run SLSQP from the best LP-initialized points to jointly optimize centers and radii.
4. **LP-Driven Coordinate Ascent**: After SLSQP, we perform a hill-climbing phase on the center coordinates. Moving one circle at a time and re-evaluating the LP objective efficiently escapes shallow local minima that gradient-based methods sometimes get trapped in due to the piecewise-linear nature of the LP envelope.
5. **Perturbation & High-Precision Polish**: Randomly perturb the `(u, v)` parameters to change the contact topology, then re-run SLSQP. Finally, a high-precision SLSQP polish tightens the packing to numerical limits.
6. **Robustness**: Strict feasibility checks, careful radius clipping, and fallback configurations ensure validity. The LP uses the `highs` solver for speed and reliability.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    k = 0
    x, y = centers[:, 0], centers[:, 1]
    bounds_arr = np.column_stack([x, 1.0 - x, y, 1.0 - y])
    A_ub[:4 * n, :] = np.repeat(np.eye(n), 4, axis=0)
    b_ub[:4 * n] = bounds_arr.ravel()
    k = 4 * n
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    A_ub[k:, I_IDX] = 1.0
    A_ub[k:, J_IDX] = 1.0
    b_ub[k:] = dists[I_IDX, J_IDX]
    
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 1e-5

def objective(vars_vec):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def constraints(vars_vec):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundaries handled by parameterization."""
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def centers_to_vars(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    vars0 = np.empty(3 * N)
    vars0[0::3] = r
    vars0[1::3] = u
    vars0[2::3] = v
    return vars0

def generate_configs(rng):
    """Generates diverse initial center configurations."""
    configs = []
    
    # 1. Hexagonal lattice patterns with varied row counts, rotations, scales, shears
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [5,5,6,5,5], [7,5,5,5,4], [4,5,5,5,7], [5,7,5,5,4]
    ]
    
    for pat in patterns:
        for _ in range(6):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            pts = np.array(pts[:N])
            
            pts -= 0.5
            scale = rng.uniform(0.85, 1.15)
            pts *= scale
            pts += 0.5
            
            rot = rng.uniform(-0.3, 0.3)
            c, s = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[c, -s], [s, c]])
            
            shear = rng.uniform(-0.1, 0.1)
            pts[:, 0] += shear * pts[:, 1]
            
            pts += rng.uniform(-0.02, 0.02, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            configs.append(pts)
            
    # 2. Force-directed layouts to find organic packings
    for seed in range(20):
        rng_fd = np.random.RandomState(seed)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        r_curr = np.full(N, 0.05)
        for step in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 20.0 * np.maximum(0, r_curr - pts[:, d])
                forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            step_size = 0.006 * (0.995**step)
            pts += step_size * forces
            pts = np.clip(pts, 0.02, 0.98)
            for i in range(N):
                d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                d_pair = np.min(np.linalg.norm(pts[i] - pts, axis=1))
                r_curr[i] = 0.85 * min(d_wall, d_pair/2.0)
        configs.append(pts)
        
    # 3. Perturbed grids
    for seed in range(15):
        rng_g = np.random.RandomState(seed)
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng_g.uniform(-0.04, 0.04, (N, 2))
        pts = np.clip(pts, 0.02, 0.98)
        configs.append(pts)
        
    return configs

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_vars = None
    best_sum = -np.inf
    
    configs = generate_configs(rng)
    
    # Phase 1: Broad search from diverse initializations using SLSQP
    for pts in configs:
        r_init, _ = solve_radii_lp(pts)
        r_init = np.clip(r_init * 0.99, 1e-5, 0.49)
        vars0 = centers_to_vars(pts, r_init)
        
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                cons_val = constraints(res.x)
                if np.min(cons_val) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    if best_vars is not None:
        # Reconstruct best centers for Phase 2
        r_tmp = best_vars[0::3]
        u_tmp = best_vars[1::3]
        v_tmp = best_vars[2::3]
        x_tmp = r_tmp + u_tmp * (1.0 - 2.0 * r_tmp)
        y_tmp = r_tmp + v_tmp * (1.0 - 2.0 * r_tmp)
        best_c = np.column_stack((x_tmp, y_tmp))
        best_r, _ = solve_radii_lp(best_c)
        
        # Phase 2: LP-Driven Coordinate Ascent on Centers
        current_c = best_c.copy()
        current_r, curr_sum = best_r, best_sum
        step = 0.03
        
        for it in range(3000):
            idx = rng.integers(N)
            direction = rng.standard_normal(2)
            direction /= np.linalg.norm(direction)
            
            nc = current_c.copy()
            nc[idx] = np.clip(nc[idx] + step * direction, 0.005, 0.995)
            
            nr, ns = solve_radii_lp(nc)
            if ns > curr_sum + 1e-8:
                current_c = nc
                current_r = nr
                curr_sum = ns
                if ns > best_sum:
                    best_c = current_c.copy()
                    best_sum = ns
                step = max(0.001, step * 0.97)
            else:
                step = min(0.06, step * 1.015)
                
        # Phase 3: Perturbation & SLSQP to escape local minima
        for _ in range(25):
            vars0 = centers_to_vars(best_c, best_vars[0::3] * 0.995)
            vars0[1::3] += rng.normal(0, 0.03, N)
            vars0[2::3] += rng.normal(0, 0.03, N)
            vars0[1::3] = np.clip(vars0[1::3], 0.0, 1.0)
            vars0[2::3] = np.clip(vars0[2::3], 0.0, 1.0)
            vars0[0::3] = np.clip(vars0[0::3], 1e-6, 0.49)
            
            try:
                res = minimize(objective, vars0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
                if res.success:
                    cons_val = constraints(res.x)
                    if np.min(cons_val) >= -1e-8:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_vars = res.x.copy()
            except Exception:
                continue
                
        # Phase 4: High-precision final polish
        vars0 = centers_to_vars(best_c, best_vars[0::3] * 0.999)
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                cons_val = constraints(res.x)
                if np.min(cons_val) >= -1e-9:
                    best_vars = res.x
                    best_sum = -res.fun
        except Exception:
            pass
            
    # Fallback configuration (should rarely be reached)
    if best_vars is None:
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        r_f, _ = solve_radii_lp(pts)
        best_vars = centers_to_vars(pts, r_f)
        best_sum = np.sum(r_f)
        
    # Reconstruct final centers and radii
    r_opt = best_vars[0::3]
    u_opt = best_vars[1::3]
    v_opt = best_vars[2::3]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    # Ensure radii are strictly positive and valid
    r_final = np.maximum(r_opt, 0.0)
    
    return centers, r_final, float(best_sum)
