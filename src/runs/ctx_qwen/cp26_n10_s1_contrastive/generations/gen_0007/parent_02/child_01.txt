import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        vals = [x, 1.0 - x, y, 1.0 - y]
        for v in vals:
            A_ub[k, i] = 1.0
            b_ub[k] = v
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-6), 1e-5

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def make_init_hex(scale, rot, noise):
    """Generates a hexagonal lattice initialization normalized to the unit square."""
    pts = []
    r_est = 0.1
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    # Normalize to [0, 1] to handle arbitrary scales robustly
    min_pts = pts.min(axis=0)
    range_pts = pts.max(axis=0) - min_pts
    pts = (pts - min_pts) / range_pts
    pts = pts * 0.9 + 0.05  # Leave margin for perturbation
    
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ rot_mat.T + 0.5
        
    pts += np.random.uniform(-noise, noise, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def make_init_growth(seed):
    """Generates an initial configuration via circle expansion and overlap resolution."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.15, 0.85, (N, 2))
    radii = np.full(N, 0.02)
    
    for _ in range(300):
        radii += 0.001
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)
                req = radii[i] + radii[j]
                if d < req and d > 1e-9:
                    push = (req - d) * 0.5
                    nx, ny = dx / d, dy / d
                    centers[i, 0] += push * nx
                    centers[i, 1] += push * ny
                    centers[j, 0] -= push * nx
                    centers[j, 1] -= push * ny
                    
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            if x < r: centers[i, 0] = r
            if x > 1.0 - r: centers[i, 0] = 1.0 - r
            if y < r: centers[i, 1] = r
            if y > 1.0 - r: centers[i, 1] = 1.0 - r
            
    return centers, radii
