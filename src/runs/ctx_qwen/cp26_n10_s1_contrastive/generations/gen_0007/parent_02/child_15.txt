3. **Coordinate Ascent on Centers**: Use the LP sum as an objective function to perform local search on centers. Move circles randomly; if the LP sum improves, accept the move. Adapt step sizes dynamically. This effectively navigates the non-smooth landscape of the LP objective.

4. **SLSQP Joint Refinement**: After finding a good center configuration, switch to a joint optimization of centers and radii using SLSQP. The optimizer then focuses solely on pairwise non-overlap constraints $dist^2 \ge (r_i+r_j)^2$, maximizing $\sum r_i$.

5. **Perturbation & Escaping**: Repeat SLSQP from perturbed versions of the best solution to escape shallow local minima. Finally, reconstruct and validate the best solution.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = max(0.0, b)
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = max(0.0, dists[i, j])
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def constraint_func(v):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundary constraints satisfied by parameterization."""
    r = v[0::3]
    u = v[1::3]
    v_par = v[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v_par * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = np.clip(radii, 1e-6, 0.499)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    res = np.empty(3 * N)
    res[0::3] = r
    res[1::3] = u
    res[2::3] = v
    return res

def gen_hex(rng, pat, rot, sc):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(pat):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * sc + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    pts += rng.uniform(-0.01, 0.01, (N, 2))
    return np.clip(pts, 0.02, 0.98)
