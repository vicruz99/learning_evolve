

<strategy>
1. **Diverse Initialization**: Generate a wide variety of starting configurations using hexagonal lattices with different row counts, rotations, and scales, combined with force-directed layouts and random spreads. This ensures coverage of multiple topological basins.
2. **Simulated Annealing on Centers**: Instead of fixed-step coordinate ascent, use Simulated Annealing (SA) to move circle centers. SA probabilistically accepts worse configurations, allowing it to escape local minima that trap greedy methods. The objective for SA is the sum of radii computed via Linear Programming (LP) for fixed centers.
3. **LP for Radii**: For any fixed center configuration, finding optimal radii is a linear program. `scipy.optimize.linprog` with the `highs` solver is extremely fast for N=26, making it practical to call thousands of times during SA.
4. **SLSQP Polishing**: After SA finds a promising rigid packing, use `scipy.optimize.minimize` with the SLSQP method to jointly optimize centers and radii. This handles the non-linear distance constraints smoothly and extracts the last bits of radius sum. Perturbations around the best SA result are fed into SLSQP to catch nearby better optima.
5. **Top-Level Functions & No Closures**: All helper functions are defined at the module level. Global indices `I_IDX, J_IDX` are precomputed to avoid redundant allocations. No lambdas or nested closures are used, complying strictly with the constraints.
6. **Robustness**: Radii are clamped to small positive values, centers are kept strictly inside `[0.005, 0.995]`, and feasibility checks are performed before accepting SLSQP results. A fallback configuration ensures the function always returns valid output.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0; A_ub[k, j] = 1.0; b_ub[k] = dists[i, j]; k += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def obj_slqp(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constr_slqp(v):
    """Inequality constraints for SLSQP: boundaries and pairwise non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    rs = r[:, None] + r[None, :]
    c.append(dist[I_IDX, J_IDX] - rs[I_IDX, J_IDX])
    return np.concatenate(c)

def slsqp_polish(centers, radii):
    """Refine a configuration using SLSQP on joint (x, y, r) variables."""
    v0 = np.zeros(3 * N)
    v0[0::3] = centers[:, 0]
    v0[1::3] = centers[:, 1]
    v0[2::3] = np.maximum(radii * 0.995, 1e-5)
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    try:
        res = minimize(obj_slqp, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constr_slqp},
                       options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
        if res.success:
            c_val = constr_slqp(res.x)
            if np.min(c_val) >= -1e-7:
                return np.column_stack((res.x[0::3], res.x[1::3])), res.x[2::3], -res.fun
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def run_sa(centers_init, rng, max_iter=3500):
    """Simulated Annealing on circle centers to maximize LP sum of radii."""
    current_c = centers_init.copy()
    r_cur, sum_cur = solve_lp(current_c)
    best_c = current_c.copy()
    best_r = r_cur.copy()
    best_sum = sum_cur
    
    temp = 0.04
    step = 0.035
    
    for it in range(max_iter):
        # Periodic multi-point perturbation to break symmetry/escape basins
        if it % 400 == 0 and it > 0:
            idxs = rng.choice(N, size=3, replace=False)
            for idx in idxs:
                move = rng.normal(0, step * 1.5, 2)
                current_c[idx] = np.clip(current_c[idx] + move, 0.005, 0.995)
            r_cur, sum_cur = solve_lp(current_c)
            continue
            
        idx = rng.integers(N)
        old = current_c[idx].copy()
        move = rng.normal(0, step, 2)
        current_c[idx] = np.clip(current_c[idx] + move, 0.005, 0.995)
        
        r_new, sum_new = solve_lp(current_c)
        delta = sum_new - sum_cur
        
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
            sum_cur = sum_new
            if sum_new > best_sum:
                best_sum = sum_new
                best_c = current_c.copy()
                best_r = r_new.copy()
        else:
            current_c[idx] = old
            
        temp *= 0.9994
        step = max(0.001, step * 0.9996)
        
    return best_c, best_r, best_sum

def make_hex_init(rng, row_counts, rot, scale):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force_init(rng):
    """Generate a force-directed organic packing initialization."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    r_curr = np.full(N, 0.05)
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 30.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
        step_size = 0.006 * (0.998**step)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
        for i in range(N):
            dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            r_curr[i] = 0.85 * min(dw, dp/2.0)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4]
    ]
    
    candidates = []
    
    # 1. Hexagonal lattice variants
    for pat in patterns:
        for _ in range(3):
            rot = rng.uniform(-0.3, 0.3)
            sc = rng.uniform(0.90, 1.10)
            candidates.append(make_hex_init(rng, pat, rot, sc))
            
    # 2. Force-directed variants
    for _ in range(8):
        candidates.append(make_force_init(rng))
        
    # 3. Random spreads
    for _ in range(5):
        candidates.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # Phase 1: Simulated Annealing on diverse starts
    for init_c in candidates:
        c, r, s = run_sa(init_c, rng, max_iter=3500)
        if s > best_sum:
            best_sum = s
            best_c = c.copy()
            best_r = r.copy()
            
    # Phase 2: SLSQP Polishing
    if best_c is not None:
        c_pol, r_pol, s_pol = slsqp_polish(best_c, best_r)
        if s_pol > best_sum:
            best_sum = s_pol
            best_c = c_pol
            best_r = r_pol
            
        # Multiple localized restarts to catch nearby optima
        for _ in range(6):
            pert_c = best_c + rng.normal(0, 0.005, (N, 2))
            pert_c = np.clip(pert_c, 0.01, 0.99)
            c2, r2, s2 = slsqp_polish(pert_c, best_r * 0.99)
            if s2 > best_sum:
                best_sum = s2
                best_c = c2
                best_r = r2
                
        # Final high-precision polish
        c_final, r_final, s_final = slsqp_polish(best_c, best_r)
        if s_final > best_sum:
            best_sum = s_final
            best_c = c_final
            best_r = r_final
            
    # Fallback (should rarely be reached)
    if best_c is None:
        best_c = make_hex_init(rng, [6,5,6,5,4], 0.0, 1.0)
        best_r, best_sum = solve_lp(best_c)
        
    # Ensure strict non-negativity
    best_r = np.maximum(best_r, 0.0)
    
    return best_c, best_r, float(best_sum)
```
