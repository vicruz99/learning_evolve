import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    # Precompute pairwise distances efficiently
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        x, y = centers[i]
        bounds_vals = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_vals:
            A_ub[k, i] = 1.0
            b_ub[k] = max(b, 0.0)
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds_r = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def objective_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    return dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX]

def centers_to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    params = np.empty(3 * N)
    params[:N] = r
    params[N:2*N] = u
    params[2*N:3*N] = v
    return params

def generate_hex_centers(seed, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_layout_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
            
        step_size = 0.006 * (0.995 ** step)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints_slqp}
    
    best_sum = -np.inf
    best_params = None
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [7,5,5,5,4], [4,5,5,5,7],
        [6,6,5,5,4], [5,5,6,5,5]
    ]
    
    starts = []
    for pat in patterns:
        for s in range(4):
            rot = rng.uniform(-0.3, 0.3)
            sc = rng.uniform(0.85, 1.15)
            starts.append(generate_hex_centers(s, pat, rot, sc))
            
    for s in range(15):
        starts.append(force_layout_init(s))
        
    # Phase 1: SLSQP search
    for pts in starts:
        radii_init, _ = solve_lp_radii(pts)
        radii_init = np.clip(radii_init * 0.98, 1e-6, 0.49)
        x0 = centers_to_params(pts, radii_init)
        
        try:
            res = minimize(objective_slqp, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                cons_val = constraints_slqp(res.x)
                if np.min(cons_val) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_params = res.x.copy()
        except Exception:
            continue
            
    if best_params is None:
        pts = starts[0]
        radii_f, _ = solve_lp_radii(pts)
        best_params = centers_to_params(pts, radii_f)
        best_sum = np.sum(radii_f)
        
    # Extract best centers for Phase 2
    r_b = best_params[:N]
    u_b = best_params[N:2*N]
    v_b = best_params[2*N:3*N]
    best_c = np.column_stack((r_b + u_b*(1-2*r_b), r_b + v_b*(1-2*r_b)))
    best_r, best_lp_sum = solve_lp_radii(best_c)
    best_sum = max(best_sum, best_lp_sum)
    
    # Phase 2: LP-driven Simulated Annealing on centers
    curr_c = best_c.copy()
    curr_r, curr_sum = best_r, best_lp_sum
    temp = 0.04
    step = 0.025
    
    for it in range(5000):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.normal(0, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        
        new_r, new_sum = solve_lp_radii(curr_c)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-7 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_r = new_r
            curr_sum = new_sum
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_c = curr_c.copy()
                best_r = curr_r.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9993
        step = max(0.0015, step * 0.9995)
        
    # Update best_params to reflect SA results
    best_params = centers_to_params(best_c, np.clip(best_r * 0.99, 1e-6, 0.49))
    
    # Phase 3: Rotation escapes & SLSQP refinement
    for _ in range(12):
        rot = rng.uniform(-0.15, 0.15)
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        c_rot = (best_c - 0.5) @ M.T + 0.5
        c_rot = np.clip(c_rot, 0.01, 0.99)
        
        r_rot, _ = solve_lp_radii(c_rot)
        x0 = centers_to_params(c_rot, np.clip(r_rot * 0.99, 1e-6, 0.49))
        
        try:
            res_r = minimize(objective_slqp, x0, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res_r.success and np.min(constraints_slqp(res_r.x)) >= -1e-7:
                s_r = -res_r.fun
                if s_r > best_sum:
                    best_sum = s_r
                    best_params = res_r.x.copy()
        except Exception:
            pass
            
    # Phase 4: High-precision polish
    if best_params is not None:
        x0_p = best_params.copy()
        x0_p[:N] *= 0.9999
        try:
            res_f = minimize(objective_slqp, x0_p, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons_slqp, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(constraints_slqp(res_f.x)) >= -1e-8:
                best_params = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Reconstruct final
    r_opt = best_params[:N]
    u_opt = best_params[N:2*N]
    v_opt = best_params[2*N:3*N]
    centers = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
