import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective_func(params):
    """Minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraint_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    d2 = dx**2 + dy**2
    
    rs = r[I_IDX] + r[J_IDX]
    return d2 - rs**2

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-4

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def generate_hex(row_counts, rng, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.098
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force(rng):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        f += np.sum((1.0/dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        for d in range(2):
            f[:, d] += 20.0 * np.maximum(0, 0.1 - pts[:, d])
            f[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.9)
        pts += 0.005 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def local_search_centers(centers, rng):
    """LP-driven Simulated Annealing on centers to maximize sum of radii."""
    current_c = centers.copy()
    _, curr_sum = solve_lp_radii(current_c)
    best_c = current_c.copy()
    best_sum = curr_sum
    
    temp = 0.04
    step = 0.035
    
    for it in range(4000):
        idx = rng.integers(N)
        old = current_c[idx].copy()
        
        current_c[idx] += rng.normal(0, step, 2)
        current_c[idx] = np.clip(current_c[idx], 0.02, 0.98)
        
        _, new_sum = solve_lp_radii(current_c)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-6 and rng.rand() < np.exp(delta / temp)):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = current_c.copy()
        else:
            current_c[idx] = old
            
        temp *= 0.9992
        step = max(0.002, step * 0.9994)
        
    return best_c, best_sum

def run_packing():

    best_sum = -np.inf

    ]
    
    inits = []
    for pat in patterns:
        for _ in range(6):
            rot = rng.uniform(-0.3, 0.3)

            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 6000, 'ftol': 1e-13})
            if res.success:
                c_val = constraint_func(res.x)
