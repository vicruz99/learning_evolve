import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-5)

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Convert (r, u, v) parameters back to physical centers and radii."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    rs = r[i_idx] + r[j_idx]
    
    return dx**2 + dy**2 - rs**2

def generate_hex_init(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_init(rng):
    """Spread points using force-directed layout."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dist**2
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, None] * diff / dist[:, :, None], axis=1)
        for d in range(2):
            f[:, d] += 10.0 * np.maximum(0, 0.1 - pts[:, d])
            f[:, d] -= 10.0 * np.maximum(0, pts[:, d] - 0.9)
        pts += 0.005 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -np.inf
    best_params = None
    
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7],
        [6,6,5,5,4], [5,6,4,6,5], [6,5,5,5,5], [5,5,5,6,5],
        [4,5,6,5,6], [6,6,4,5,5], [5,5,4,6,6]
    ]
    
    # Generate diverse initial configurations
    for pat in patterns:
        for _ in range(6):
            rot = rng.uniform(-0.35, 0.35)
            sc = rng.uniform(0.85, 1.15)
            inits.append(generate_hex_init(rng, pat, rot, sc))
            
    for _ in range(20):
        inits.append(generate_force_init(rng))
        
    # Phase 1: Broad SLSQP search from diverse starts
    for c_init in inits:
        r_init = solve_lp_radii(c_init)
        r_init = np.clip(r_init * 0.98, 1e-6, 0.49)
        p0 = to_params(c_init, r_init)
        
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_params = res.x.copy()
        except Exception:
            continue
            
    if best_params is None:
        c_f = generate_hex_init(rng, [6,5,6,5,4], 0.0, 1.0)
        r_f = solve_lp_radii(c_f)
        best_params = to_params(c_f, r_f)
        best_sum = np.sum(r_f)
        
    # Phase 2: LP-driven Simulated Annealing on centers
    curr_c, curr_r = from_params(best_params)
    curr_sum = np.sum(curr_r)
    
    step = 0.035
    temp = 0.025
    for it in range(4000):
        idx = rng.integers(N)
        direction = rng.standard_normal(2)
        direction /= np.linalg.norm(direction)
        nc = curr_c.copy()
        nc[idx] = np.clip(nc[idx] + step * direction, 0.02, 0.98)
        
        nr = solve_lp_radii(nc)
        ns = np.sum(nr)
        delta = ns - curr_sum
        
        if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(delta / temp)):
            curr_c = nc
            curr_r = nr
            curr_sum = ns
            if ns > best_sum:
                best_sum = ns
            step = max(0.001, step * 0.96)
        else:
            step = min(0.05, step * 1.02)

    # Convert best SA result to params for SLSQP refinement
    best_params_lp = to_params(curr_c, curr_r * 0.99)
    try:
        res_lp = minimize(objective, best_params_lp, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13})
        if res_lp.success:
            c_val = constraints(res_lp.x)
            if np.min(c_val) >= -1e-7:
                s = -res_lp.fun
                if s > best_sum:
                    best_sum = s
                    best_params = res_lp.x.copy()
    except Exception:
        pass
        
    # Phase 3: Rotation escapes & Perturbation refinements
    for _ in range(15):
        # Random rotation escape
        c_rot, r_rot = from_params(best_params)
        rot = rng.uniform(-0.12, 0.12)
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        c_rot = (c_rot - 0.5) @ M.T + 0.5
        c_rot = np.clip(c_rot, 0.02, 0.98)
        
        p_rot = to_params(c_rot, r_rot * 0.98)
        try:
            res = minimize(objective, p_rot, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13})

                    best_sum = -res.fun
                    best_params = res.x.copy()
        except Exception:
            pass
            
        # Perturbation refinement
        p_pert = best_params.copy()
        p_pert[:N] += rng.normal(0, 0.002, N)
        p_pert[N:] += rng.uniform(-0.025, 0.025, 2*N)
        p_pert[:N] = np.clip(p_pert[:N], 1e-6, 0.49)
        p_pert[N:] = np.clip(p_pert[N:], 0.0, 1.0)
        
        try:
            res = minimize(objective, p_pert, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
            if res.success and np.min(constraints(res.x)) >= -1e-7:

                    best_sum = -res.fun
                    best_params = res.x.copy()

    p_polish = best_params.copy()
    p_polish[:N] *= 0.9998
    try:
        res_f = minimize(objective, p_polish, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 12000, 'ftol': 1e-14})
        if res_f.success:
            c_val = constraints(res_f.x)
            if np.min(c_val) >= -1e-8:
                best_params = res_f.x.copy()
                best_sum = -res_f.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_params)
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(best_sum)
