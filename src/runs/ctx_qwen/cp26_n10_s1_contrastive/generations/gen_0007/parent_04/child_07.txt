1. **Linear Programming (LP) for Radii**: Given fixed centers, finding the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints is a convex LP. This gives an exact, fast evaluation of any center configuration.
2. **Simulated Annealing (SA) on Centers**: We treat the LP-optimal sum as the objective function over center coordinates. SA explores the rugged landscape effectively, escaping local minima that gradient methods get trapped in. We use adaptive temperature and step-size cooling, plus occasional large jumps for basin hopping.
3. **SLSQP with Boundary-Safe Parameterization**: We parameterize centers as $x_i = r_i + u_i(1-2r_i)$, $y_i = r_i + v_i(1-2r_i)$ with $u_i, v_i \in [0,1]$. This automatically satisfies boundary constraints $r_i \le x_i \le 1-r_i$. SLSQP then optimizes $(r, u, v)$ jointly using only pairwise separation constraints, providing high-precision polishing.

- Generate diverse initial configurations (hexagonal lattices with variations, force-directed repulsion layouts).

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    x, y = centers[:, 0], centers[:, 1]
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        for b in [x[i], 1.0 - x[i], y[i], 1.0 - y[i]]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-5

def slqp_obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def slqp_constraints(params):
    """Inequality constraints: pairwise separation. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    r_sum = r[I_IDX] + r[J_IDX]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dx*dx + dy*dy - r_sum*r_sum

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def generate_starts(rng):
    """Generates diverse initial center configurations."""
    starts = []
    
    # 1. Hexagonal lattice patterns with varied counts, scales, rotations
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [7,5,5,5,4], [4,5,5,5,7],
        [6,6,5,5,4], [5,5,6,5,5], [8,5,5,5,3], [5,7,6,5,3]
    ]
    
    for pat in patterns:
        for _ in range(4):
            pts = []
            r_est = 0.095 + rng.uniform(-0.012, 0.012)
            y = r_est
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
                
            pts = np.array(pts[:N])
            pts -= 0.5
            scale = rng.uniform(0.88, 1.12)
            pts *= scale
            pts += 0.5
            
            rot = rng.uniform(-0.25, 0.25)
            c, s = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[c, -s], [s, c]])
            
            pts += rng.normal(0, 0.015, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            starts.append(pts)
            
    # 2. Force-directed layouts for organic distributions
    for seed in range(20):
        rng_fd = np.random.RandomState(seed)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        for _ in range(200):
            forces = np.zeros_like(pts)
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            
            for d in range(2):
                forces[:, d] += 20.0 * np.maximum(0, 0.1 - pts[:, d])
                forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.9)
                
            pts += 0.005 * forces
            pts = np.clip(pts, 0.05, 0.95)
        starts.append(pts)
        
    return starts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    
    bounds_slqp = [(1e-6, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons_slqp = {'type': 'ineq', 'fun': slqp_constraints}
    
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    starts = generate_starts(rng)
    
    # Phase 1: Broad SLSQP search from diverse initializations
    for pts in starts:
        r_init, _ = solve_lp_radii(pts)
        r_init = np.maximum(r_init * 0.99, 1e-5)
        x0 = to_params(pts, r_init)
        
        try:
            res = minimize(slqp_obj, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = slqp_constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        r_opt = res.x[:N]
                        u_opt = res.x[N:2*N]
                        v_opt = res.x[2*N:3*N]
                        best_c = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))
                        best_r = r_opt
        except Exception:
            continue
            
    # Phase 2: Simulated Annealing on centers using LP evaluation
    if best_c is not None:
        curr_c = best_c.copy()
        curr_r, curr_sum = solve_lp_radii(curr_c)
        best_lp_c = curr_c.copy()
        best_lp_r = curr_r.copy()
        best_lp_sum = curr_sum
        
        temp = 0.045
        step = 0.035
        
        for it in range(8000):
            # Occasionally take a larger jump to escape basins
            current_step = step * (4.0 if rng.random() < 0.04 else 1.0)
            
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, current_step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
            new_r, new_sum = solve_lp_radii(curr_c)
            delta = new_sum - curr_sum
            
            if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(min(0.0, delta / temp))):
                curr_sum = new_sum
                if new_sum > best_lp_sum:
                    best_lp_sum = new_sum
                    best_lp_c = curr_c.copy()
                    best_lp_r = new_r.copy()
            else:
                curr_c[idx] = old
                
            temp *= 0.9991
            step = max(0.0015, step * 0.9993)
            
        if best_lp_sum > best_sum:
            best_sum = best_lp_sum
            best_c = best_lp_c
            best_r = best_lp_r
            
    # Phase 3: SLSQP polish on the SA result
    if best_c is not None:
        r_polish = np.maximum(best_r * 0.995, 1e-5)
        x0 = to_params(best_c, r_polish)
        try:
            res = minimize(slqp_obj, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(slqp_constraints(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    r_opt = res.x[:N]
                    u_opt = res.x[N:2*N]
                    v_opt = res.x[2*N:3*N]
                    best_c = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))
                    best_r = r_opt
        except Exception:
            pass
            
    # Phase 4: Perturbation escapes & final high-precision refinement
    if best_c is not None:
        for k in range(25):
            x0 = to_params(best_c, best_r * 0.99)
            # Perturb u, v more aggressively to change contact topology
            x0[N:3*N] += rng.uniform(-0.025, 0.025, 2*N)
            x0[:N] += rng.normal(0, 0.0015, N)
            
            x0[:N] = np.clip(x0[:N], 1e-6, 0.49)
            x0[N:3*N] = np.clip(x0[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(slqp_obj, x0, method='SLSQP', bounds=bounds_slqp,
                               constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(slqp_constraints(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        r_opt = res.x[:N]
                        u_opt = res.x[N:2*N]
                        v_opt = res.x[2*N:3*N]
                        best_c = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))
                        best_r = r_opt
            except Exception:
                continue
                
    # Fallback configuration (should rarely be reached)
    if best_c is None:
        best_c = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_r = np.full(N, 0.08)
        best_sum = np.sum(best_r)
        
    radii = np.maximum(best_r, 0.0)
    return best_c, radii, float(best_sum)
