import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)]*n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def constraint_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def objective(params):
    """Minimize negative sum of radii."""
    return -np.sum(params[:N])

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def hex_init(seed, scale=1.0, rot=0.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.098 * scale
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(500):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        rep = 1.0 / (dist**2)
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, None] * diff / dist[:, :, None], axis=1)
        for d in range(2):
            f[:, d] += 10.0 * np.maximum(0, 0.1 - pts[:, d])
            f[:, d] -= 10.0 * np.maximum(0, pts[:, d] - 0.9)
        pts += 0.005 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slsqp = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons_slsqp = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    inits = []
    for s in range(30):
        inits.append(hex_init(s, scale=rng.uniform(0.9, 1.1), rot=rng.uniform(-0.3, 0.3)))
    for s in range(20):
        inits.append(force_init(s))
        
    # Phase 1: Broad SLSQP search from diverse starts
    for pts in inits:
        r, s_val = solve_lp_radii(pts)
        if s_val < 1e-3: 
            continue
        x0 = to_params(pts, r * 0.99)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_slsqp, constraints=cons_slsqp,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_centers, best_radii = from_params(res.x)
        except Exception:
            continue

    # Fallback if optimization completely fails
    if best_centers is None:
        pts = hex_init(0)
        r, _ = solve_lp_radii(pts)
        best_centers, best_radii = pts, r
        best_sum = np.sum(r)

    # Phase 2: LP-driven Coordinate Ascent to change contact topology
    curr_c = best_centers.copy()
    curr_r, curr_s = best_radii.copy(), best_sum
    step = 0.03
    for it in range(4000):
        idx = rng.choice(N)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.normal(0, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        
        nr, ns = solve_lp_radii(curr_c)
        if ns > curr_s + 1e-7:
            curr_r, curr_s = nr, ns
            if ns > best_sum:
                best_sum = ns
                best_centers = curr_c.copy()
                best_radii = curr_r.copy()
            step = max(0.001, step * 0.96)
        else:
            curr_c[idx] = old
            step = min(0.05, step * 1.015)
            
    # Phase 3: Refine best configuration with SLSQP
    x0 = to_params(best_centers, best_radii * 0.995)
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds_slsqp, constraints=cons_slsqp,
                       options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        if res.success and np.min(constraint_func(res.x)) >= -1e-9:
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_centers, best_radii = from_params(res.x)
    except Exception:
        pass

    # Phase 4: Perturbation & SLSQP loop to escape remaining local minima
    for _ in range(35):
        xp = to_params(best_centers, best_radii * 0.99)
        xp[:N] += rng.normal(0, 0.003, N)
        xp[N:3*N] += rng.normal(0, 0.02, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.5)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        try:
            res = minimize(objective, xp, method='SLSQP', bounds=bounds_slsqp, constraints=cons_slsqp,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_centers, best_radii = from_params(res.x)
        except Exception:
            pass

    # Phase 5: High-precision final polish
    x0 = to_params(best_centers, best_radii * 0.999)
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds_slsqp, constraints=cons_slsqp,
                       options={'maxiter': 12000, 'ftol': 1e-15, 'disp': False})
        if res.success and np.min(constraint_func(res.x)) >= -1e-9:
            best_centers, best_radii = from_params(res.x)
            best_sum = -res.fun
    except Exception:
        pass

    return best_centers, np.maximum(best_radii, 0.0), float(best_sum)
