import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= dist to walls
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-4)

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Convert (r, u, v) parameters back to centers and radii."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def get_constraints(params):
    """Inequality constraints for SLSQP: pairwise non-overlap (dist^2 >= (r_i+r_j)^2)."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return (dx[I_IDX, J_IDX]**2 + dy[I_IDX, J_IDX]**2) - rs[I_IDX, J_IDX]**2

def get_obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def hex_init(seed, row_counts, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.098
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    for _ in range(400):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        f += np.sum((1.0/dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        for d in range(2):
            f[:, d] += 20.0 * np.maximum(0, 0.1 - pts[:, d])
            f[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.9)
        pts += 0.004 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def grid_init(seed):
    """Generates a perturbed grid initialization."""
    rng = np.random.RandomState(seed)
    pts = np.array([[0.1 + i*0.2, 0.1 + j*0.2] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-0.04, 0.04, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    np.random.seed(42)
    bounds = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    best_sum = -np.inf
    best_x = None
    
    inits = []
    
    # Diverse hexagonal patterns
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5],
        [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3], [5,7,6,5,3], [6,6,6,5,3],
        [5,5,5,6,5], [6,5,5,5,5], [5,5,6,5,5], [4,5,5,5,7], [7,5,5,5,4]
    ]
    idx = 0
    for p in patterns:
        for s in range(4):
            rot = (s % 5 - 2) * 0.1
            sc = 0.9 + (s // 5) * 0.1
            pts = hex_init(idx, p, rot=rot, scale=sc)
            r = solve_lp_radii(pts)
            inits.append(to_params(pts, r * 0.98))
            idx += 1
            
    # Force layouts
    for s in range(15):
        pts = force_init(s)
        r = solve_lp_radii(pts)
        inits.append(to_params(pts, r * 0.98))
        
    # Grid layouts
    for s in range(8):
        pts = grid_init(s)
        r = solve_lp_radii(pts)
        inits.append(to_params(pts, r * 0.98))
        
    # Phase 1: Broad search
    for x0 in inits:
        try:
            res = minimize(get_obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = get_constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = res.x.copy()
        except Exception:
            continue
            
    if best_x is None:
        pts = grid_init(0)
        r = solve_lp_radii(pts)
        best_x = to_params(pts, r * 0.98)
        best_sum = np.sum(best_x[:N])
        
    # Phase 2: Perturbation refinement
    rng = np.random.RandomState(42)
    for _ in range(40):
        x_p = best_x.copy()
        x_p[:N] += rng.uniform(-0.002, 0.002, N)
        x_p[N:3*N] += rng.uniform(-0.015, 0.015, 2*N)
        x_p[:N] = np.clip(x_p[:N], 1e-6, 0.49)
        x_p[N:3*N] = np.clip(x_p[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(get_obj, x_p, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(get_constraints(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: High-precision polish
    try:
        res_f = minimize(get_obj, best_x, method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(get_constraints(res_f.x)) >= -1e-8:
            best_x = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_x)
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(best_sum)
