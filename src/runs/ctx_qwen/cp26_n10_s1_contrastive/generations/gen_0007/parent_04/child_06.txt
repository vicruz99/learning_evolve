import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Precompute pairwise distances
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Build constraint matrices
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    for idx in range(len(I_IDX)):
        i, j = I_IDX[idx], J_IDX[idx]
        A_ub[k, i] = 1.0; A_ub[k, j] = 1.0; b_ub[k] = dists[i, j]; k += 1
        
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 1e-5

def get_constraints(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    d2 = dx**2 + dy**2
    rs2 = r_sum**2
    
    return d2[I_IDX, J_IDX] - rs2[I_IDX, J_IDX]

def get_obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def centers_to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = np.clip(radii, 1e-6, 0.499)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def generate_hex_pattern(row_counts, rot, scale, seed):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        mat = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ mat.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_layout_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    r_curr = np.full(N, 0.05)
    for step in range(400):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
        step_size = 0.008 * (0.995**step)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            d_pair = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            r_curr[i] = 0.8 * min(d_wall, d_pair/2.0)
    return pts

def run_packing():
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    best_sum = -np.inf
    best_vars = None
    
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [7,5,5,5,4], [4,5,5,5,7],
        [6,6,5,5,4], [5,7,5,5,4], [4,5,5,7,5]
    ]
    for pat in patterns:
        for s in range(5):
            rot = (s % 5 - 2) * 0.08
            sc = 0.9 + (s // 5) * 0.1
            inits.append(generate_hex_pattern(pat, rot, sc, s))
            
    for s in range(15):
        inits.append(force_layout_init(s))
        
    # Phase 1: Broad SLSQP search from diverse initializations
    for pts in inits:
        r_init, _ = solve_lp_radii(pts)
        r_init = np.maximum(r_init * 0.98, 1e-6)
        x0 = centers_to_params(pts, r_init)
        
        try:
            res = minimize(get_obj, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
            if res.success:
                if np.min(get_constraints(res.x)) >= -1e-8:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization failed
    if best_vars is None:
        pts = inits[0]
        r_f, _ = solve_lp_radii(pts)
        best_vars = centers_to_params(pts, r_f * 0.99)
        best_sum = np.sum(best_vars[:N])
        
    # Reconstruct best centers for SA
    r_best = best_vars[:N]
    u_best = best_vars[N:2*N]
    v_best = best_vars[2*N:3*N]
    c_best = np.column_stack((r_best + u_best*(1-2*r_best), r_best + v_best*(1-2*v_best)))
    
    # Phase 2: Simulated Annealing on centers with LP radius evaluation
    curr_c = c_best.copy()
    r_curr, curr_sum = solve_lp_radii(curr_c)
    best_c = curr_c.copy()
    best_r = r_curr.copy()
    best_lp_sum = curr_sum
    
    rng = np.random.RandomState(42)
    temp = 0.03
    step = 0.025
    
    for it in range(2000):
        idx = rng.randint(N)
        old_pos = curr_c[idx].copy()
        
        curr_c[idx] += rng.normal(0, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        
        r_new, sum_new = solve_lp_radii(curr_c)
        delta = sum_new - curr_sum
        
        if delta > 0 or (temp > 1e-7 and rng.rand() < np.exp(delta / temp)):
            curr_sum = sum_new
            if sum_new > best_lp_sum:
                best_lp_sum = sum_new
                best_c = curr_c.copy()
                best_r = r_new.copy()
        else:
            curr_c[idx] = old_pos
            
        temp *= 0.9993
        step = max(0.002, step * 0.9995)
        
    # Update global best if SA found something better
    if best_lp_sum > best_sum:
        best_sum = best_lp_sum
        best_vars = centers_to_params(best_c, best_r)
        
    # Phase 3: High-precision SLSQP polish on the best configuration
    x0 = best_vars.copy()
    x0[:N] *= 0.999  # Slight shrink for strict interior feasibility
    try:
        res_f = minimize(get_obj, x0, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
        if res_f.success and np.min(get_constraints(res_f.x)) >= -1e-8:
            s_f = -res_f.fun
            if s_f > best_sum:
                best_sum = s_f
                best_vars = res_f.x.copy()
    except Exception:
        pass
        
    # Reconstruct final centers and radii
    r_opt = best_vars[:N]
    u_opt = best_vars[N:2*N]
    v_opt = best_vars[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
