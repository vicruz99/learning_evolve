import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    m = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def generate_hex(seed, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.02, 0.02, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def generate_force(seed, steps=400):
    """Generates a force-directed spread initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    r_curr = np.full(N, 0.06)
    
    for step in range(steps):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.005 * (0.996**step)
        pts += step_size * forces
        pts = np.clip(pts, 0.01, 0.99)
        
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            d_pair = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            r_curr[i] = 0.85 * min(d_wall, d_pair/2.0)
            
    return pts

def run_sa(centers_init, rng):
    """Optimizes centers using Simulated Annealing with LP radius evaluation."""
    curr_c = centers_init.copy()
    curr_r, curr_sum = solve_lp(curr_c)
    best_c = curr_c.copy()
    best_r = curr_r.copy()
    best_sum = curr_sum
    
    temp = 0.045
    step = 0.065
    
    for _ in range(1800):
        idx = rng.integers(N)
        new_c = curr_c.copy()
        new_c[idx] += rng.normal(0, step, 2)
        new_c[idx] = np.clip(new_c[idx], 0.005, 0.995)
        
        new_r, new_sum = solve_lp(new_c)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-9 and rng.rand() < np.exp(delta / temp)):
            curr_c = new_c
            curr_sum = new_sum
            curr_r = new_r
            if new_sum > best_sum:
                best_c = new_c.copy()
                best_r = new_r.copy()
                best_sum = new_sum
                
        temp *= 0.9993
        step = max(0.001, step * 0.9995)
        
    return best_c, best_r, best_sum

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def constraint_func(params):
    """Inequality constraints for SLSQP using boundary-safe parameterization."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    i_idx, j_idx = np.triu_indices(N, k=1)
    return dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx]**2

def objective_func(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    rng = np.random.default_rng(42)
    np.seterr(all='ignore')
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # Generate diverse initial configurations
    inits = []

    # Phase 1: Global topology search via Simulated Annealing + LP
    for init in inits:
        c, r, s_val = run_sa(init, rng)
        if s_val > best_sum:
            best_sum = s_val
            best_c = c
            best_r = r
            
    if best_c is not None:
        # Phase 2: High-precision analytical polish using SLSQP
        # Shrink radii slightly to guarantee strict interior feasibility for SLSQP start
        safe_r = np.maximum(best_r * 0.992, 1e-6)
        x0 = get_params(best_c, safe_r)
        
        bounds = [(1e-6, 0.5), (0.0, 1.0), (0.0, 1.0)] * N
        cons = {'type': 'ineq', 'fun': constraint_func}
        
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                cv = constraint_func(res.x)
                if np.min(cv) >= -1e-8:
                    r_opt = res.x[0::3]
                    u_opt = res.x[1::3]
                    v_opt = res.x[2::3]
                    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
                    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
                    centers = np.column_stack((x_opt, y_opt))
                    radii = np.maximum(r_opt, 0.0)
                    final_sum = float(np.sum(radii))
                    return centers, radii, final_sum
        except Exception:
            pass
            
        # Fallback to best SA result if SLSQP fails to improve or validate
        return best_c, best_r, float(best_sum)
        
    # Extreme fallback (should not be reached)
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    rad = np.full(N, 0.09)
    return pts, rad, np.sum(rad)
