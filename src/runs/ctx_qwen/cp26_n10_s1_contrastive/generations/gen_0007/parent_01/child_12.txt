

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we need a robust optimization pipeline that escapes local minima and tightly packs circles against boundaries and each other. The strategy combines:

1. **Parameterization**: Use `(r, u, v)` where `x = r + u*(1-2r)` and `y = r + v*(1-2r)`. This automatically satisfies boundary constraints `r <= x <= 1-r` and `r <= y <= 1-r`, leaving only pairwise non-overlap constraints for the optimizer.
2. **LP-Driven Radius Solver**: For fixed centers, optimal radii are found via Linear Programming. This is fast, exact, and handles the non-smooth nature of the packing problem well.
3. **Diverse Initialization**: Generate starts from rotated/scaled hexagonal lattices, force-directed layouts, and perturbed grids.
4. **Multi-Phase Optimization**:
   - *Phase 1*: SLSQP from diverse initial parameters.
   - *Phase 2*: LP-driven coordinate ascent on centers to explore the configuration space without gradient bottlenecks.
   - *Phase 3*: Aggressive perturbation of `(u, v)` parameters followed by SLSQP refinement to change contact topologies.
   - *Phase 4*: High-precision polishing.
5. **Robustness**: Strict clipping, tolerance handling, and fallback configurations ensure validity. The pipeline avoids closures and lambda functions per constraints.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_pair = n * (n - 1) // 2
    A_ub = np.zeros((4 * n + num_pair, n))
    b_ub = np.zeros(4 * n + num_pair)
    
    x, y = centers[:, 0], centers[:, 1]
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        A_ub[i, i] = 1.0; b_ub[i] = x[i]
        A_ub[n + i, i] = 1.0; b_ub[n + i] = 1.0 - x[i]
        A_ub[2*n + i, i] = 1.0; b_ub[2*n + i] = y[i]
        A_ub[3*n + i, i] = 1.0; b_ub[3*n + i] = 1.0 - y[i]
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    k = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[:N])

def constraints(vars_vec):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = vars_vec[:N]
    u = vars_vec[N:2*N]
    v = vars_vec[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(vars_vec):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = vars_vec[:N]
    u = vars_vec[N:2*N]
    v = vars_vec[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def generate_hex_centers(seed, row_counts, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    r_curr = np.full(N, 0.05)
    
    for step in range(500):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.005 * (0.996**step)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
        
        for i in range(N):
            dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            r_curr[i] = 0.85 * min(dw, max(dp, 1e-6)/2.0)
            
    r_final = np.zeros(N)
    for i in range(N):
        dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        dists = np.linalg.norm(pts[i] - pts, axis=1)
        dists[i] = np.inf
        dp = np.min(dists)
        r_final[i] = 0.95 * min(dw, max(dp, 1e-6)/2.0)
    return pts, r_final

def run_packing():
    np.random.seed(42)
    np.seterr(all='ignore')
    
    bounds = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_vars = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7],
        [6,6,5,5,4], [5,5,6,5,5], [8,5,5,5,3], [5,5,5,5,5,1]
    ]
    for pat in patterns:
        for s in range(6):
            rot = np.random.uniform(-0.25, 0.25)
            sc = np.random.uniform(0.90, 1.10)
            pts = generate_hex_centers(s, pat, rot, sc)
            r, _ = solve_lp_radii(pts)
            r = np.clip(r * 0.99, 1e-6, 0.5)
            inits.append(to_params(pts, r))
            
    for s in range(25):
        pts, r = generate_force_init(s)
        r = np.clip(r * 0.99, 1e-6, 0.5)
        inits.append(to_params(pts, r))
        
    # Phase 1: Broad SLSQP search from diverse starts
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                if np.min(constraints(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: LP-Driven Coordinate Ascent on Centers
    if best_vars is not None:
        c_best, r_best = from_params(best_vars)
        curr_c = c_best.copy()
        curr_r, curr_sum = r_best, best_sum
        
        step = 0.025
        rng = np.random.RandomState(123)
        for it in range(4000):
            idx = rng.randint(N)
            old = curr_c[idx].copy()
            direction = rng.standard_normal(2)
            direction /= np.linalg.norm(direction)
            
            curr_c[idx] = np.clip(curr_c[idx] + step * direction, 0.01, 0.99)
            nr, ns = solve_lp_radii(curr_c)
            
            if ns > curr_sum + 1e-7:
                curr_sum = ns
                curr_r = nr
                if ns > best_sum:
                    best_sum = ns
                    c_best = curr_c.copy()
                    r_best = curr_r.copy()
                step = max(0.001, step * 0.96)
            else:
                curr_c[idx] = old
                step = min(0.05, step * 1.01)
                
        # Convert best LP result back to params for SLSQP polish
        best_vars = to_params(c_best, np.clip(r_best * 0.995, 1e-6, 0.5))
        try:
            res = minimize(objective, best_vars, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-7:
                best_vars = res.x
                best_sum = -res.fun
        except Exception:
            pass

    # Phase 3: Perturbation refinement to escape local minima
    if best_vars is not None:
        rng_p = np.random.RandomState(456)
        for k in range(80):
            xp = best_vars.copy()
            # Perturb positions more aggressively to change contact topology
            xp[N:3*N] += rng_p.uniform(-0.025, 0.025, 2*N)
            xp[:N] += rng_p.uniform(-0.002, 0.002, N)
            xp[:N] = np.clip(xp[:N], 1e-6, 0.5)
            xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(objective, xp, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(constraints(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_vars = res.x.copy()
            except Exception:
                continue
                
        # Phase 4: High-precision final polish
        try:
            res = minimize(objective, best_vars, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-15, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-7:
                best_vars = res.x
                best_sum = -res.fun
        except Exception:
            pass

    # Fallback configuration (should rarely be reached)
    if best_vars is None:
        c_f = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                               np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        r_f = np.full(N, 0.09)
        best_vars = to_params(c_f, r_f)
        best_sum = np.sum(r_f)
        
    centers, radii = from_params(best_vars)
    # Ensure non-negative radii
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(best_sum)
