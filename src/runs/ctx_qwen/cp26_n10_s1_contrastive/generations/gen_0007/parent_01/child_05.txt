This transforms the constrained nonlinear optimization into an unconstrained optimization over center coordinates, where the objective function is the LP-optimal sum of radii.

Strategy:
1. **LP-Radii Solver**: For any given set of 26 centers, solve an LP to find radii that maximize their sum while satisfying boundary and pairwise non-overlap constraints. This function is extremely fast for N=26.
2. **Simulated Annealing on Centers**: Use SA to search the space of center configurations. At each step, perturb one circle's position, re-evaluate the LP sum, and accept/reject based on the Metropolis criterion. 3. **Diverse Initialization**: Start SA chains from multiple hexagonal lattice patterns (with random rotations/scales) and force-directed layouts to cover different topological basins.
4. **Deterministic Polish**: After SA, run a few iterations of `Nelder-Mead` on the best configuration to fine-tune center positions without getting stuck in local SA minima.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    x, y = centers[:, 0], centers[:, 1]
    bounds_arr = np.stack([x, 1.0 - x, y, 1.0 - y], axis=1)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    n_constr = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constr, n))
    b_ub = np.zeros(n_constr)
    
    k = 0
    for i in range(n):
        for b in bounds_arr[i]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    # Fallback to tiny radii if LP fails
    return np.full(n, 1e-5), 1e-5

def get_lp_neg_sum(centers_flat):
    """Objective function for optimizers: returns negative LP sum of radii."""
    centers = centers_flat.reshape(N, 2)
    _, s = solve_lp_radii(centers)
    return -s

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [7,5,5,5,4], [5,7,5,5,4], [4,5,5,7,5], [5,5,7,5,4]
    ]
    for pat in patterns:
        for _ in range(3):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            pts = np.array(pts[:N])
            
            # Random rotation, translation, scale, and jitter
            rot = rng.uniform(-0.4, 0.4)
            c, s = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            scale = rng.uniform(0.9, 1.1)
            pts = (pts - 0.5) * scale + 0.5
            pts += rng.uniform(-0.02, 0.02, pts.shape)
            pts = np.clip(pts, 0.05, 0.95)
            inits.append(pts.flatten())
            
    # 2. Force-directed layouts
    for seed in range(10):
        rng_fd = np.random.RandomState(seed)
        pts = rng_fd.rand(N, 2)
        for _ in range(400):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            pts += 0.005 * forces
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts.flatten())
        
    return inits

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    rng = np.random.default_rng(42)
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    inits = generate_inits(rng)
    
    # Phase 1: Simulated Annealing on centers driven by LP radii evaluation
    for init_flat in inits:
        curr_c = init_flat.reshape(N, 2).copy()
        _, curr_sum = solve_lp_radii(curr_c)
        
        step = 0.05
        temp = 0.06
        best_local_sum = curr_sum
        best_local_c = curr_c.copy()
        best_local_r, _ = solve_lp_radii(curr_c)
        
        # SA Loop
        for it in range(4000):
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            
            # Adaptive step with occasional large jumps
            jump_mult = 4.0 if rng.random() < 0.015 else 1.0
            curr_c[idx] += rng.normal(0, step * jump_mult, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
            _, new_sum = solve_lp_radii(curr_c)
            delta = new_sum - curr_sum
            
            # Metropolis criterion
            if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
                curr_sum = new_sum
                if curr_sum > best_local_sum:
                    best_local_sum = curr_sum
                    best_local_c = curr_c.copy()
                    best_local_r, _ = solve_lp_radii(curr_c)
            else:
                curr_c[idx] = old
                
            temp *= 0.9985
            step = max(0.0008, step * 0.9993)
            
        if best_local_sum > best_sum:
            best_sum = best_local_sum
            best_centers = best_local_c
            best_radii = best_local_r
            
    # Phase 2: Deterministic polishing with Nelder-Mead on best configuration
    if best_centers is not None:
        curr_flat = best_centers.flatten()
        for _ in range(3):
            # Small perturbation to escape plateaus
            curr_flat += rng.normal(0, 0.008, N * 2)
            curr_flat = np.clip(curr_flat, 0.01, 0.99)
            
            res = minimize(get_lp_neg_sum, curr_flat, method='Nelder-Mead',
                           options={'maxiter': 1500, 'xatol': 1e-7, 'fatol': 1e-9})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
                best_radii, _ = solve_lp_radii(best_centers)
                
    # Phase 3: Final consistency check & strict boundary enforcement
    if best_centers is not None:
        best_radii, best_sum = solve_lp_radii(best_centers)
        # Clamp radii slightly to guarantee strict interior validity against 1e-12 tolerance
        best_radii = np.maximum(best_radii, 0.0)
        
    # Fallback configuration (should rarely be reached)
    if best_centers is None:
        best_centers = np.column_stack([
            np.repeat(np.linspace(0.1, 0.9, 5), 5),
            np.tile(np.linspace(0.1, 0.9, 5), 5)
        ])
        best_centers = np.vstack([best_centers, [0.5, 0.5]])[:N]
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    return best_centers, best_radii, float(best_sum)
