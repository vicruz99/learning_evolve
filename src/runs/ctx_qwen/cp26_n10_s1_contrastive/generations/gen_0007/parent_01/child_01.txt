Key improvements in this solution:
1. **LP-Driven Simulated Annealing (SA)**: Instead of directly optimizing radii and centers jointly, we separate the problem. For fixed centers, the optimal radii are found via a fast Linear Program (LP). We then run SA on the centers to maximize this LP objective. The LP objective is piecewise-linear and discontinuous at constraint boundaries, making gradient-based methods prone to local minima. SA effectively jumps between different Voronoi/contact topologies.
2. **Diverse High-Quality Initializations**: We generate hexagonal lattices (optimal for density), force-directed layouts (organic packing), and a corner/edge-focused layout. These are scored via LP, and only the top candidates undergo expensive SA refinement.
3. **Boundary-Safe Parameterization for SLSQP**: After SA finds a promising center configuration, we map it to a parameterization `(r, u, v)` where `x = r + u(1-2r)` and `y = r + v(1-2r)`. This automatically satisfies boundary constraints, allowing SLSQP to focus purely on pairwise non-overlap constraints for high-precision polishing.
4. **Perturbation Escapes**: Multiple rounds of targeted perturbations followed by SLSQP runs help escape any remaining local basins.
5. **Efficient LP Solver**: Uses `scipy.optimize.linprog` with the `highs` solver, precomputing constraint structures efficiently for rapid evaluation during SA.

This hybrid approach combines the topology-exploring strength of SA with the precise constraint-handling of SLSQP, typically pushing packing sums significantly higher.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub_rows = []
    b_ub_vals = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(dists[i, j])
            
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.fun is not None:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 2.6e-4

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def params_to_centers(params):
    """Reconstruct physical centers and radii from optimization parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constr(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def run_sa(centers, rng, iters=1500):
    """Simulated Annealing on centers using LP objective."""
    curr_c = centers.copy()
    curr_r, curr_s = solve_lp_radii(curr_c)
    best_c, best_r, best_s = curr_c.copy(), curr_r.copy(), curr_s
    
    temp = 0.04
    step = 0.05
    
    for _ in range(iters):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.normal(0, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        
        new_r, new_s = solve_lp_radii(curr_c)
        delta = new_s - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(min(0.0, delta/temp))):
            curr_s = new_s
            if new_s > best_s:
                best_s = new_s
                best_c = curr_c.copy()
                best_r = new_r.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9993
        step = max(0.001, step * 0.9993)
        
    return best_c, best_r, best_s

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [7,5,5,5,4]]
    for pat in patterns:
        for _ in range(3):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N: pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * 1.73205
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.9, 1.1) + 0.5
            rot = rng.uniform(-0.2, 0.2)
            c, s = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pts = np.clip(pts + rng.uniform(-0.02, 0.02, (N, 2)), 0.05, 0.95)
            inits.append(pts)
            
    # 2. Force-directed layouts
    for seed in range(10):
        rng_f = np.random.RandomState(seed)
        pts = rng_f.rand(N, 2) * 0.8 + 0.1
        r_c = np.full(N, 0.06)
        for step in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 10.0 * np.maximum(0, r_c - pts[:, d])
                forces[:, d] -= 10.0 * np.maximum(0, pts[:, d] - (1.0 - r_c))
            pts += 0.005 * (0.99**step) * forces
            pts = np.clip(pts, 0.02, 0.98)
            for i in range(N):
                dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
                dp = max(dp, 1e-5)
                r_c[i] = 0.85 * min(dw, dp/2.0)
        inits.append(pts)
        
    # 3. Corner/Edge focused layout
    pts_corner = np.array([
        [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9],
        [0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5],
        [0.35, 0.35], [0.65, 0.35], [0.35, 0.65], [0.65, 0.65],
        [0.5, 0.5]
    ])
    rng_f = np.random.RandomState(123)
    extra = rng_f.uniform(0.2, 0.8, (N - len(pts_corner), 2))
    pts_corner = np.vstack([pts_corner, extra])
    inits.append(pts_corner)
        
    return inits

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.5), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constr}
    
    best_vars = None
    best_sum = -np.inf
    
    inits = generate_inits(rng)
    
    # Score inits and keep top candidates for SA
    scored_inits = []
    for pts in inits:
        r, s = solve_lp_radii(pts)
        scored_inits.append((s, pts))
    scored_inits.sort(key=lambda x: x[0], reverse=True)
    
    top_inits = scored_inits[:10]
    
    # Phase 1: SA on centers + SLSQP refinement for top candidates
    for s_score, pts in top_inits:
        c_sa, r_sa, s_sa = run_sa(pts, rng, iters=1500)
        
        if s_sa > best_sum:
            best_sum = s_sa
            best_vars = get_params(c_sa, r_sa)
            
        # SLSQP polish from SA result
        params0 = get_params(c_sa, r_sa)
        params0[:N] *= 0.99
        params0[:N] = np.clip(params0[:N], 1e-6, 0.5)
        
        try:
            res = minimize(obj, params0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                if np.min(constr(res.x)) >= -1e-7:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_vars = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation & SLSQP to escape local minima
    if best_vars is not None:
        for _ in range(50):
            p = best_vars.copy()
            p[:N] += rng.normal(0, 0.001, N)
            p[N:] += rng.normal(0, 0.02, 2*N)
            p[:N] = np.clip(p[:N], 1e-6, 0.5)
            p[N:] = np.clip(p[N:], 0.0, 1.0)
            
            try:
                res = minimize(obj, p, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(constr(res.x)) >= -1e-7:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_vars = res.x.copy()
            except Exception:
                pass
                
        # Phase 3: High-precision final polish
        try:
            res = minimize(obj, best_vars, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if np.min(constr(res.x)) >= -1e-8:
                best_vars = res.x
                best_sum = -res.fun
        except Exception:
            pass
            
    # Fallback (should rarely be reached)
    if best_vars is None:
        pts = inits[0]
        r, _ = solve_lp_radii(pts)
        best_vars = get_params(pts, r)
        best_sum = np.sum(r)
        
    centers, radii = params_to_centers(best_vars)
    return centers, radii, float(best_sum)
