import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        bounds_list = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_list:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 2.6e-4

def constraint_func(vars_vec):
    """Inequality constraints for SLSQP using (r, u, v) parameterization."""
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def objective(vars_vec):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def centers_to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    vars0 = np.empty(3 * N)
    vars0[0::3] = r
    vars0[1::3] = u
    vars0[2::3] = v
    return vars0

def generate_force_init(rng, steps=400):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    r_curr = np.full(N, 0.06)
    
    for step in range(steps):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        repulsion_mag = 1.0 / dists**2
        np.fill_diagonal(repulsion_mag, 0.0)
        forces = np.sum(repulsion_mag[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 5.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 5.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.005 * (0.995**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists = np.linalg.norm(pts[i] - pts, axis=1)
            dists[i] = np.inf
            d_pair = np.min(dists)
            r_curr[i] = 0.85 * min(d_wall, d_pair/2.0)
            
    return pts

def generate_hex_init(rng, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization with rotation and scaling."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    centers = np.array(pts[:N])
    centers = (centers - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c, -s], [s, c]])
        centers = (centers - 0.5) @ rot_mat.T + 0.5
        
    centers += rng.uniform(-0.02, 0.02, (N, 2))
    centers = np.clip(centers, 0.02, 0.98)
    return centers

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    rng = np.random.default_rng(42)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # Generate diverse initial configurations
    inits = []
    for _ in range(5):
        inits.append(generate_force_init(rng))
    for _ in range(5):
        rot = rng.uniform(-0.3, 0.3)
        sc = rng.uniform(0.9, 1.1)
        inits.append(generate_hex_init(rng, rot=rot, scale=sc))
        
    for init_c in inits:
        curr_c = init_c.copy()
        curr_r, curr_sum = solve_lp(curr_c)
        
        if curr_sum > best_sum:
            best_c = curr_c.copy()
            best_r = curr_r.copy()
            best_sum = curr_sum
            
        # Simulated Annealing on centers
        temp = 0.04
        step = 0.045
        sa_c = curr_c.copy()
        sa_sum = curr_sum
        
        for it in range(12000):
            idx = rng.integers(N)
            old = sa_c[idx].copy()
            sa_c[idx] += rng.normal(0, step, 2)
            sa_c[idx] = np.clip(sa_c[idx], 0.015, 0.985)
            
            # Occasionally swap to break symmetry
            do_swap = False
            other_idx = -1
            if rng.random() < 0.05:
                do_swap = True
                other_idx = rng.integers(N)
                if other_idx != idx:
                    sa_c[idx], sa_c[other_idx] = sa_c[other_idx].copy(), sa_c[idx].copy()
                    
            new_r, new_sum = solve_lp(sa_c)
            delta = new_sum - sa_sum
            
            accept = delta > 0 or (temp > 1e-9 and rng.random() < np.exp(min(0.0, delta / temp)))
            
            if accept:
                sa_sum = new_sum
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_c = sa_c.copy()
                    best_r = new_r.copy()
            else:
                if do_swap:
                    sa_c[idx], sa_c[other_idx] = sa_c[other_idx].copy(), sa_c[idx].copy()
                else:
                    sa_c[idx] = old
                    
            temp *= 0.9993
            step = max(0.001, step * 0.9996)
            
        # SLSQP Polish
        vars0 = centers_to_params(best_c, best_r * 0.995)
        bounds = [(1e-6, 0.5), (0.0, 1.0), (0.0, 1.0)] * N
        cons = {'type': 'ineq', 'fun': constraint_func}
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = constraint_func(res.x)
                if np.min(c_val) >= -1e-7:
                    curr_polish_sum = -res.fun
                    if curr_polish_sum > best_sum:
                        best_sum = curr_polish_sum
                        r_opt = res.x[0::3]
                        u_opt = res.x[1::3]
                        v_opt = res.x[2::3]
                        x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
                        y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
                        best_c = np.column_stack((x_opt, y_opt))
                        best_r = r_opt
        except Exception:
            pass
                
    # Fallback configuration
    if best_c is None:
        best_c = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                  np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        best_r, best_sum = solve_lp(best_c)
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
