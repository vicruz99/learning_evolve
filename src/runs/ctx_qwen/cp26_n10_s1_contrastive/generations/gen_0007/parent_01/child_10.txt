

<strategy>
To improve the circle packing sum of radii, I will employ a hybrid optimization strategy that combines the strengths of Linear Programming (LP) and Sequential Least Squares Programming (SLSQP):

1. **LP-Driven Center Optimization**: For any fixed set of centers, finding the optimal radii that maximize the sum is a linear program. This decouples the radius optimization from the non-linear center optimization, creating a much smoother (though non-differentiable) objective landscape for the centers.
2. **Simulated Annealing on Centers**: I will run a robust SA loop that perturbs circle centers, evaluates the LP for radii, and accepts/rejects moves based on temperature. This effectively explores the configuration space and escapes local minima caused by contact topology changes.
3. **SLSQP Parameterization Polish**: Once high-quality centers are found, I will map them to a boundary-safe parameterization `(r, u, v)` where `x = r + u(1-2r)` and `y = r + v(1-2r)`. This automatically satisfies boundary constraints and allows SLSQP to precisely polish the packing with tight tolerances.
4. **Diverse Initializations**: Multiple hexagonal lattice patterns with rotations/scales and force-directed layouts will seed the SA, ensuring coverage of the solution space.
5. **High-Precision Final Polish**: A final SLSQP run with aggressive iteration limits and tight tolerances will squeeze out the last bits of radius sum.

This approach separates the combinatorial/topological search (SA on centers) from the continuous optimization (SLSQP on parameters), which has historically yielded state-of-the-art results for dense packings.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        bounds_list = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_list:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    dists = np.maximum(dists, 1e-12)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-6

def objective_slqp(vars_vec):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def constraints_slqp(vars_vec):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def centers_to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    vars0 = np.empty(3 * N)
    vars0[0::3] = r
    vars0[1::3] = u
    vars0[2::3] = v
    return vars0

def generate_inits():
    """Generates diverse initial center configurations."""
    inits = []
    rng = np.random.RandomState(42)
    
    # 1. Hexagonal lattice patterns
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5]]
    for pat in patterns:
        for _ in range(6):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N: pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            pts = np.array(pts[:N])
            scale = rng.uniform(0.88, 1.12)
            pts = (pts - 0.5) * scale + 0.5
            rot = rng.uniform(-0.35, 0.35)
            c, s = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pts += rng.uniform(-0.025, 0.025, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    # 2. Force-directed layouts
    for s in range(15):
        rng_fd = np.random.RandomState(s)
        pts = rng_fd.rand(N, 2)
        r_c = np.full(N, 0.06)
        for step in range(400):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 25.0 * np.maximum(0, r_c - pts[:, d])
                forces[:, d] -= 25.0 * np.maximum(0, pts[:, d] - (1.0 - r_c))
            pts += 0.006 * (0.996**step) * forces
            pts = np.clip(pts, 1e-4, 1.0-1e-4)
            for i in range(N):
                dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
                r_c[i] = 0.9 * min(dw, dp/2.0)
        inits.append(pts)
        
    return inits

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    bounds_slqp = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints_slqp}
    
    best_centers = None
    best_sum = -1.0
    
    inits = generate_inits()
    rng = np.random.RandomState(123)
    
    # Phase 1: LP-driven Simulated Annealing on centers
    for init_c in inits:
        curr_c = init_c.copy()
        curr_r, curr_s = solve_lp_radii(curr_c)
        temp = 0.04
        step = 0.03
        
        for it in range(2000):
            idx = rng.randint(N)
            old = curr_c[idx].copy()
            
            # Occasionally take a larger jump
            current_step = step * (3.0 if rng.rand() < 0.02 else 1.0)
            curr_c[idx] += rng.normal(0, current_step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
            
            new_r, new_s = solve_lp_radii(curr_c)
            delta = new_s - curr_s
            
            if delta > 0 or (temp > 1e-8 and rng.rand() < np.exp(min(0.0, delta/temp))):
                curr_s = new_s
                if new_s > best_sum:
                    best_sum = new_s
                    best_centers = curr_c.copy()
            else:
                curr_c[idx] = old
                
            temp *= 0.999
            step = max(0.001, step * 0.9993)

    # Phase 2: SLSQP Polish
    if best_centers is not None:
        best_r, _ = solve_lp_radii(best_centers)
        x0 = centers_to_params(best_centers, best_r * 0.99)
        
        try:
            res = minimize(objective_slqp, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 6000, 'ftol': 1e-13})
            if res.success and np.min(constraints_slqp(res.x)) >= -1e-7:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    r_opt = res.x[0::3]
                    u_opt = res.x[1::3]
                    v_opt = res.x[2::3]
                    best_centers = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))
        except Exception:
            pass
            
        # Perturbation & SLSQP to escape remaining local minima
        for _ in range(25):
            x0_p = x0.copy()
            x0_p[1::3] += rng.uniform(-0.025, 0.025, N)
            x0_p[2::3] += rng.uniform(-0.025, 0.025, N)
            x0_p[0::3] += rng.uniform(-0.003, 0.003, N)
            x0_p[0::3] = np.clip(x0_p[0::3], 1e-6, 0.49)
            x0_p[1::3] = np.clip(x0_p[1::3], 0.0, 1.0)
            x0_p[2::3] = np.clip(x0_p[2::3], 0.0, 1.0)
            
            try:
                res = minimize(objective_slqp, x0_p, method='SLSQP', bounds=bounds_slqp,
                               constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-13})
                if res.success and np.min(constraints_slqp(res.x)) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        r_opt = res.x[0::3]
                        u_opt = res.x[1::3]
                        v_opt = res.x[2::3]
                        best_centers = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))
            except Exception:
                continue
                
        # High precision final polish
        x0_f = centers_to_params(best_centers, solve_lp_radii(best_centers)[0] * 0.999)
        try:
            res_f = minimize(objective_slqp, x0_f, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons_slqp, options={'maxiter': 10000, 'ftol': 1e-14})
            if res_f.success and np.min(constraints_slqp(res_f.x)) >= -1e-8:
                r_opt = res_f.x[0::3]
                u_opt = res_f.x[1::3]
                v_opt = res_f.x[2::3]
                best_centers = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))
                best_sum = np.sum(r_opt)
        except Exception:
            pass

    if best_centers is None:
        # Fallback configuration
        centers_f = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_r, best_sum = solve_lp_radii(centers_f)
        best_centers = centers_f
    else:
        best_r, best_sum = solve_lp_radii(best_centers)
        
    return best_centers, best_r, float(best_sum)
