import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
PAIR_COUNT = N * (N - 1) // 2

# Precompute LP constraint matrix structure for maximum speed
A_LP = np.zeros((4 * N + PAIR_COUNT, N))
for i in range(N):
    A_LP[i, i] = 1.0          # r_i <= x_i
    A_LP[N + i, i] = 1.0      # r_i <= 1-x_i
    A_LP[2*N + i, i] = 1.0    # r_i <= y_i
    A_LP[3*N + i, i] = 1.0    # r_i <= 1-y_i

k = 4 * N
for i in range(N):
    for j in range(i + 1, N):
        A_LP[k, i] = 1.0
        A_LP[k, j] = 1.0
        k += 1

def solve_radii_lp(centers):
    """Solves LP to find radii maximizing sum(r_i) for fixed centers."""
    n = centers.shape[0]
    b_ub = np.empty(4 * n + PAIR_COUNT)
    
    for i in range(n):
        x, y = centers[i]
        b_ub[i] = x
        b_ub[n + i] = 1.0 - x
        b_ub[2*n + i] = y
        b_ub[3*n + i] = 1.0 - y

    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)

    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            b_ub[idx] = dists[i, j]
            idx += 1

    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def objective_slqp(vars_vec):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def constraints_slqp(vars_vec):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def centers_to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    vars0 = np.empty(3 * N)
    vars0[0::3] = r
    vars0[1::3] = u
    vars0[2::3] = v
    return vars0

def force_init(seed):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    r_curr = np.full(N, 0.06)
    
    for _ in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        repulsion_mag = 1.0 / dists**2
        np.fill_diagonal(repulsion_mag, 0.0)
        forces = np.sum(repulsion_mag[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 30.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        pts += 0.005 * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            d_pair = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            d_pair = max(d_pair, 1e-6)
            r_curr[i] = 0.85 * min(d_wall, d_pair/2.0)
            
    return pts

def hex_init(seed, rotation=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization with rotation and scaling."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    centers = np.array(pts[:N])
    centers = (centers - 0.5) * scale + 0.5
    
    if rotation != 0.0:
        c, s = np.cos(rotation), np.sin(rotation)
        rot_mat = np.array([[c, -s], [s, c]])
        centers = (centers - 0.5) @ rot_mat.T + 0.5
        
    centers += rng.uniform(-0.015, 0.015, (N, 2))
    return np.clip(centers, 0.02, 0.98)

def run_packing():

    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    best_c = None
    best_r = None

    for c_init in starts:

        if s_val > best_sum:
            best_sum = s_val
            best_c = c_init.copy()
            best_r = r.copy()
            
    # Phase 2: LP-Driven Simulated Annealing on centers
    curr_c = best_c.copy()
    curr_sum = best_sum
    temp = 0.03
    step = 0.025
    
    for it in range(4000):
        idx = rng.randint(N)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.normal(0, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)

        delta = new_sum - curr_sum

            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = curr_c.copy()
                best_r = new_r.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9992
        step = max(0.001, step * 0.9995)
        
    # Phase 3: SLSQP refinement with boundary-safe parameterization
    bounds_slqp = [(1e-6, 0.5), (0.0, 1.0), (0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints_slqp}
    
    # Ensure strict interior feasibility for SLSQP start
    r_init = np.maximum(best_r * 0.99, 1e-5)

        pass

                    u_opt = res.x[1::3]
                    v_opt = res.x[2::3]
                    best_c = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))

        except Exception:
            pass

        best_r = np.full(N, 0.01)
        best_sum = np.sum(best_r)
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
