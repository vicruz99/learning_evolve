import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    A_ub = np.zeros((4 * n, n))
    b_ub = np.zeros(4 * n)
    for i in range(n):
        x, y = centers[i]
        A_ub[4*i, i] = 1.0; b_ub[4*i] = x
        A_ub[4*i+1, i] = 1.0; b_ub[4*i+1] = 1.0 - x
        A_ub[4*i+2, i] = 1.0; b_ub[4*i+2] = y
        A_ub[4*i+3, i] = 1.0; b_ub[4*i+3] = 1.0 - y
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    n_pairs = n * (n - 1) // 2
    A_pair = np.zeros((n_pairs, n))
    b_pair = np.zeros(n_pairs)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_pair[k, i] = 1.0
            A_pair[k, j] = 1.0
            b_pair[k] = dists[i, j]
            k += 1
            
    A_ub_full = np.vstack([A_ub, A_pair])
    b_ub_full = np.concatenate([b_ub, b_pair])
    
    try:
        res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub_full, 
                      bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-4

def centers_to_params(centers, radii):
    """Convert physical centers and radii to (r, u, v) optimization parameters."""
    r = np.clip(radii, 1e-6, 0.499)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def params_to_centers(params):
    """Convert (r, u, v) parameters back to physical centers."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7]
    ]
    for pat in patterns:
        for _ in range(6):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
                
            pts = np.array(pts[:N])
            pts -= 0.5
            scale = rng.uniform(0.85, 1.15)
            pts *= scale
            pts += 0.5
            
            rot = rng.uniform(-0.3, 0.3)
            c, s = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[c, -s], [s, c]])
            pts -= pts.mean(axis=0)
            pts += 0.5
            
            pts += rng.normal(0, 0.01, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    # 2. Force-directed layouts
    for seed in range(20):
        rng_fd = np.random.RandomState(seed)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        r_curr = np.full(N, 0.06)
        
        for step in range(400):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            
            for d in range(2):
                forces[:, d] += 25.0 * np.maximum(0, r_curr - pts[:, d])
                forces[:, d] -= 25.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
                
            step_size = 0.008 * (0.995**step)
            pts += step_size * forces
            pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
            
            for i in range(N):
                d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                dists_to_others = np.linalg.norm(pts[i] - pts, axis=1)
                dists_to_others[i] = np.inf
                d_pair = np.min(dists_to_others)
                r_curr[i] = 0.85 * min(d_wall, d_pair/2.0)
                
        inits.append(pts)
        
    # 3. Corner-anchored configurations
    for seed in range(10):
        rng_c = np.random.RandomState(seed)
        pts = np.array([[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]])
        pts = np.vstack([pts, rng_c.rand(N-4, 2) * 0.6 + 0.2])
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    return inits

def objective(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def constraint_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_sum = -np.inf
    inits = generate_inits(rng)
    
    # Phase 1: LP-Driven Simulated Annealing on Centers
    # Find a strong baseline by optimizing centers directly using LP for radii
    for pts in inits:
        curr_c = pts.copy()
        _, curr_sum = solve_radii_lp(curr_c)
        best_c_local = curr_c.copy()
        best_sum_local = curr_sum
        
        temp = 0.05
        step = 0.04
        for it in range(3000):
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            
            # Adaptive step size
            current_step = step * (3.0 if rng.rand() < 0.05 else 1.0)
            curr_c[idx] += rng.normal(0, current_step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
            _, new_sum = solve_radii_lp(curr_c)
            delta = new_sum - curr_sum
            
            if delta > 0 or (temp > 1e-8 and rng.rand() < np.exp(min(0.0, delta / temp))):
                curr_sum = new_sum
                if new_sum > best_sum_local:
                    best_sum_local = new_sum
                    best_c_local = curr_c.copy()
            else:
                curr_c[idx] = old
                
            temp *= 0.9993
            step = max(0.0015, step * 0.9995)
            
        if best_sum_local > best_sum:
            best_sum = best_sum_local
            # Convert best centers to params for SLSQP
            r_init, _ = solve_radii_lp(best_c_local)
            best_vars = centers_to_params(best_c_local, r_init * 0.98)
            
    # Phase 2: SLSQP Refinement
    if best_vars is not None:
        try:
            res = minimize(objective, best_vars, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-7:
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_vars = res.x.copy()
        except Exception:
            pass
            
        # Phase 3: Perturbation Escapes & Re-optimization
        for k in range(50):
            x0 = best_vars.copy()
            # Perturb u, v more aggressively to change contact topology
            x0[1::3] += rng.normal(0, 0.04, N)
            x0[2::3] += rng.normal(0, 0.04, N)
            x0[0::3] += rng.normal(0, 0.002, N)
            
            x0[0::3] = np.clip(x0[0::3], 1e-6, 0.49)
            x0[1::3] = np.clip(x0[1::3], 0.0, 1.0)
            x0[2::3] = np.clip(x0[2::3], 0.0, 1.0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
                if res.success and np.min(constraint_func(res.x)) >= -1e-7:
                    if -res.fun > best_sum:
                        best_sum = -res.fun
                        best_vars = res.x.copy()
            except Exception:
                continue
                
        # Phase 4: High-precision polish
        if best_vars is not None:
            try:
                res = minimize(objective, best_vars, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
                if np.min(constraint_func(res.x)) >= -1e-7:
                    best_vars = res.x
                    best_sum = -res.fun
            except Exception:
                pass
                
    # Fallback (should rarely be reached)
    if best_vars is None:
        centers_f = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        r_f = np.full(N, 0.09)
        best_vars = centers_to_params(centers_f, r_f)
        best_sum = np.sum(r_f)
        
    # Reconstruct centers and radii
    r_opt = best_vars[0::3]
    u_opt = best_vars[1::3]
    v_opt = best_vars[2::3]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    return centers, np.maximum(r_opt, 0.0), float(best_sum)
