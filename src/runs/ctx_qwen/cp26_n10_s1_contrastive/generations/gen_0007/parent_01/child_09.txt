**Strategy:**
1. **LP-Driven Center Optimization:** Instead of optimizing `(x, y, r)` simultaneously from the start, we decouple the problem. For any fixed set of centers, the optimal radii that maximize the sum can be found exactly and rapidly using Linear Programming (`scipy.optimize.linprog`). This avoids the non-convexity of `dist(i,j) >= r_i + r_j` during the global search phase.
2. **Diverse Initializations:** Generate 30+ high-quality starting configurations using hexagonal lattice patterns with varied rotations/scales, plus force-directed layouts to capture organic packings.
3. **Hill-Climbing / Simulated Annealing on Centers:** Run a local search that moves one circle at a time, re-evaluates the optimal radii via LP, and accepts moves that increase the total sum. Step size is adaptively adjusted. This efficiently navigates the center-space landscape.
4. **SLSQP Fine-Tuning:** Once the best center configuration is found, switch to simultaneous optimization of `(x, y, r)` using `SLSQP` with hard constraints. This handles the coupling between positions and radii precisely and polishes the solution to high accuracy.
5. **Perturbation Escapes:** Run multiple random perturbations of the best solution followed by SLSQP to escape any remaining local minima.

This hybrid approach combines the global exploration strength of stochastic moves with the exact feasibility of LP and the local precision of gradient-based optimization.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
np.seterr(all='ignore')

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-6), 1e-5

def objective_slqp(vars_vec):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_vec[2::3])

def constraints_slqp(vars_vec):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    x = vars_vec[0::3]
    y = vars_vec[1::3]
    r = vars_vec[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def generate_hex_centers(seed, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization with rotation and scaling."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ rot_mat + 0.5
        
    pts += rng.uniform(-0.015, 0.015, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def generate_force_centers(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    for _ in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 15.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 15.0 * np.maximum(0, pts[:, d] - 0.95)
            
        pts += 0.006 * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    np.random.seed(42)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    for s in range(25):
        rot = np.random.uniform(-0.25, 0.25)
        sc = np.random.uniform(0.92, 1.08)
        inits.append(generate_hex_centers(s, rot, sc))
        
    for s in range(15):
        inits.append(generate_force_centers(s))
        
    # Phase 2: LP-Driven Local Search on Centers
    for c0 in inits:
        curr_c = c0.copy()
        curr_r, curr_s = solve_lp_radii(curr_c)
        
        step = 0.035
        for it in range(800):
            idx = np.random.randint(N)
            old = curr_c[idx].copy()
            
            direction = np.random.randn(2)
            direction /= np.linalg.norm(direction)
            
            # Adaptive step: occasionally jump further
            current_step = step * (4.0 if np.random.rand() < 0.03 else 1.0)
            
            curr_c[idx] = np.clip(curr_c[idx] + current_step * direction, 0.02, 0.98)
            
            new_r, new_s = solve_lp_radii(curr_c)
            if new_s > curr_s:
                curr_s = new_s
                curr_r = new_r
                if new_s > best_sum:
                    best_sum = new_s
                    best_c = curr_c.copy()
                    best_r = curr_r.copy()
                step = max(0.001, step * 0.96)
            else:
                curr_c[idx] = old
                step = min(0.05, step * 1.015)
                
    # Phase 3: SLSQP Polish
    if best_c is not None:
        bounds_slqp = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
        cons_slqp = {'type': 'ineq', 'fun': constraints_slqp}
        
        vars0 = np.zeros(3 * N)
        vars0[0::3] = best_c[:, 0]
        vars0[1::3] = best_c[:, 1]
        vars0[2::3] = np.maximum(best_r * 0.995, 1e-6)
        
        try:
            res = minimize(objective_slqp, vars0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints_slqp(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_r = res.x[2::3]
        except Exception:
            pass
            
        # Phase 4: Perturbation Escapes & Re-optimization
        for k in range(40):
            v0 = np.zeros(3 * N)
            v0[0::3] = best_c[:, 0] + np.random.randn(N) * 0.008
            v0[1::3] = best_c[:, 1] + np.random.randn(N) * 0.008
            v0[2::3] = best_r * 0.99
            
            v0[0::3] = np.clip(v0[0::3], 0.02, 0.98)
            v0[1::3] = np.clip(v0[1::3], 0.02, 0.98)
            
            try:
                res = minimize(objective_slqp, v0, method='SLSQP', bounds=bounds_slqp,
                               constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(constraints_slqp(res.x)) >= -1e-8:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_r = res.x[2::3]
            except Exception:
                continue
                
        # Phase 5: High-precision final polish
        try:
            v0 = np.zeros(3 * N)
            v0[0::3] = best_c[:, 0]
            v0[1::3] = best_c[:, 1]
            v0[2::3] = best_r * 0.999
            
            res_f = minimize(objective_slqp, v0, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons_slqp, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(constraints_slqp(res_f.x)) >= -1e-9:
                best_c = np.column_stack((res_f.x[0::3], res_f.x[1::3]))
                best_r = res_f.x[2::3]
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Fallback (highly unlikely)
    if best_c is None:
        best_c = generate_hex_centers(0)
        best_r, best_sum = solve_lp_radii(best_c)
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
