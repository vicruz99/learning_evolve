from scipy.optimize import minimize, linprog

N = 26

def solve_lp(centers):
    """Given fixed centers, solve LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub_list = []
    b_ub_list = []
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(b)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dists[i, j])
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 0.0

def obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constr(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    idx = np.triu_indices(N, k=1)
    return dist2[idx] - rs[idx]**2

def centers_to_params(centers, shrink=0.97):
    """Convert centers to (r, u, v) optimization parameters."""
    r, _ = solve_lp(centers)
    r = r * shrink
    r = np.clip(r, 1e-6, 0.49)
    x, y = centers[:, 0], centers[:, 1]
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((x - r) / denom, 0.0, 1.0)
    v = np.clip((y - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def hex_init(seed, row_counts, rot=0.0, scale=1.0):
    """Hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.1
    y = r_est
    for r_idx, count in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(count):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    pts = np.clip(pts, 0.03, 0.97)
    return pts

def force_init(seed):
    """Force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.7 + 0.15
    r_curr = np.full(N, 0.06)
    for step in range(500):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]

        f += np.sum((1.0 / dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        for d in range(2):
            f[:, d] += 30.0 * np.maximum(0, r_curr - pts[:, d])
            f[:, d] -= 30.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
        step_size = 0.005 * (0.997**step)
        pts += step_size * f
        pts = np.clip(pts, 0.02, 0.98)
        for i in range(N):
            d_wall = min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])
            dists_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_others[i] = np.inf
            d_pair = np.min(dists_others)
            r_curr[i] = 0.8 * min(d_wall, d_pair / 2.0)
    return pts

def grid_init(seed):
