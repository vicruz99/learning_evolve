import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constr(params):
    """
    Inequality constraints: pairwise non-overlap.
    Boundary constraints are automatically satisfied by the (r, u, v) parameterization.
    Returns array of values that must be >= 0.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        for b in [x[i], 1.0-x[i], y[i], 1.0-y[i]]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    for i in range(n):
        for j in range(i+1, n):
            d = dists[i, j]
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds_r = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def centers_to_params(centers):
    """Converts physical centers to (r, u, v) optimization parameters."""
    r, _ = solve_lp(centers)
    # Scale down slightly to ensure strict interior feasibility for SLSQP
    r = np.clip(r * 0.98, 1e-6, 0.49)
    x, y = centers[:, 0], centers[:, 1]
    
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((x - r) / denom, 0.0, 1.0)
    v = np.clip((y - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def hex_init(seed, row_counts, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, count in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(count):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    for _ in range(600):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        # Coulomb-like repulsion with softening
        rep = 1.0 / (dist**2 + 0.02)
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, None] * diff / dist[:, :, None], axis=1)
        
        # Wall repulsion
        for d in range(2):
            f[:, d] += 15.0 * np.maximum(0, 0.05 - pts[:, d])
            f[:, d] -= 15.0 * np.maximum(0, pts[:, d] - 0.95)
            
        pts += 0.005 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():

    bounds = [(1e-6, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constr}
    
    best_sum = -np.inf
    best_x = None
    
    # Generate diverse initial centers
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5],

    for p in patterns:

            sc = 0.9 + (s // 5) * 0.1
            inits.append(hex_init(idx, p, rot=rot, scale=sc))
            idx += 1
            
    for s in range(10):
        inits.append(force_init(s))
        
    # Phase 1: Broad search from diverse initializations using SLSQP
    for c in inits:
        x0 = centers_to_params(c)
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = constr(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization failed entirely
    if best_x is None:
        c_f = hex_init(0, [6,5,6,5,4])
        best_x = centers_to_params(c_f)
        best_sum = -obj(best_x)
        
    # Phase 2: LP-Driven Simulated Annealing on centers to escape local minima
    r_tmp = best_x[:N]
    u_tmp = best_x[N:2*N]
    v_tmp = best_x[2*N:3*N]
    x_tmp = r_tmp + u_tmp * (1.0 - 2.0 * r_tmp)
    y_tmp = r_tmp + v_tmp * (1.0 - 2.0 * r_tmp)
    curr_c = np.column_stack([x_tmp, y_tmp])
    curr_r, curr_sum = solve_lp(curr_c)
    
    r_best = r_tmp.copy()
    x_best = x_tmp.copy()
    y_best = y_tmp.copy()
    
    step = 0.03
    temp = 0.01
    rng = np.random.RandomState(123)
    for it in range(2500):
        idx_move = rng.integers(N)
        old_c = curr_c[idx_move].copy()
        
        # Random directional move
        direction = rng.normal(0, 1, 2)
        direction /= np.linalg.norm(direction)
        curr_c[idx_move] += step * direction
        curr_c[idx_move] = np.clip(curr_c[idx_move], 0.01, 0.99)
        
        new_r, new_sum = solve_lp(curr_c)
        delta = new_sum - curr_sum
        
        # Simulated annealing acceptance criterion
        if delta > 0 or (temp > 1e-8 and rng.rand() < np.exp(delta / temp)):
            curr_r = new_r
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                r_best = curr_r.copy()
                x_best = curr_c[:, 0].copy()
                y_best = curr_c[:, 1].copy()
        else:
            curr_c[idx_move] = old_c
            
        # Annealing schedule
        step = max(0.002, step * 0.9995)
        temp *= 0.9995
        
    # Phase 3: High-precision SLSQP polish on the refined configuration
    best_x = np.empty(3*N)
    best_x[:N] = r_best
    denom = np.clip(1.0 - 2.0 * r_best, 1e-6, 1.0)
    best_x[N:2*N] = np.clip((x_best - r_best) / denom, 0.0, 1.0)
    best_x[2*N:3*N] = np.clip((y_best - r_best) / denom, 0.0, 1.0)
    
    # Slight shrink for strict interior feasibility before final polish
    best_x[:N] *= 0.999
    try:
        res_f = minimize(obj, best_x, method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(constr(res_f.x)) >= -1e-8:
            best_x = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    # Reconstruct physical centers and radii
    r_opt = best_x[:N]
    u_opt = best_x[N:2*N]
    v_opt = best_x[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack([x_opt, y_opt])
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
