N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i

            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)

            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    return np.full(n, 1e-4), 0.0

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints(params):
    """
    Inequality constraints: pairwise non-overlap.
    Boundary constraints are automatically satisfied by parameterization.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def centers_to_params(centers, shrink=0.98):
    """Converts physical centers to (r, u, v) optimization parameters."""
    r, _ = solve_lp_radii(centers)
    r = r * shrink
    x, y = centers[:, 0], centers[:, 1]
    
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((x - r) / denom, 0.0, 1.0)
    v = np.clip((y - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def params_to_centers(params):
    """Converts (r, u, v) parameters back to physical centers and radii."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def coords_obj(x_flat, fixed_centers, idx):
    """Objective function for optimizing a single circle's center."""
    centers = fixed_centers.copy()
    centers[idx] = x_flat
    _, s = solve_lp_radii(centers)
    return -s  # Minimize negative sum

def optimize_single_center(centers, idx):
    """Optimizes the position of circle idx to maximize LP sum."""
    x0 = centers[idx].copy()
    bounds = [(0.01, 0.99), (0.01, 0.99)]
    try:
        res = minimize(coords_obj, x0, args=(centers, idx), method='Nelder-Mead',
                       options={'maxiter': 2000, 'xatol': 1e-7, 'fatol': 1e-9})
        return res.x
    except Exception:
        return x0

def hex_init(seed, row_counts, rot=0.0, scale=1.0):

    pts = []
    r_est = 0.095

        x = r_est + shift
        for _ in range(count):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est

    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:

        pts = (pts - 0.5) @ R.T + 0.5
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.04, 0.96)

def force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    for _ in range(500):
        f = np.zeros_like(pts)

            f[:, d] -= 15.0 * np.maximum(0, pts[:, d] - 0.92)
        pts += 0.003 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts
