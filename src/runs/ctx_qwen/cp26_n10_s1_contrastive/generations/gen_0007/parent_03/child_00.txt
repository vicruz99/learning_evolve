1. **LP-Driven Simulated Annealing (SA)**: For a fixed set of centers, finding the optimal radii that maximize their sum subject to non-overlap and boundary constraints is a linear program. We use `scipy.optimize.linprog` to solve this efficiently. By treating the centers as the decision variables and using the LP solution as the objective value, we run a Simulated Annealing process. This allows us to rapidly explore the center configuration space, accepting occasional worse moves to escape local minima, while gradually cooling the temperature and step size.

2. **Boundary-Safe Parameterization**: For the final polishing phase, we switch to SLSQP. To automatically satisfy boundary constraints (`r <= x <= 1-r`), we parameterize centers as `x = r + u*(1-2r)` and `y = r + v*(1-2r)` where `u, v ∈ [0, 1]`. This reduces the constrained problem to only handling pairwise non-overlap constraints (`dist^2 >= (r_i + r_j)^2`).

3. **Diverse Initialization & Multi-Phase Refinement**: We start with a wide variety of configurations (hexagonal lattices with rotations/scales, force-directed layouts, and perturbed grids). The best is selected via LP, refined with SA, then polished with SLSQP. Multiple rounds of perturbation and re-optimization ensure we converge to a high-quality local optimum near the theoretical limit (~2.636).

This approach avoids the pitfalls of pure gradient-based methods getting trapped in symmetric basins and leverages the exact feasibility guarantees of LP during the exploration phase.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    x, y = centers[:, 0], centers[:, 1]
    bounds_vals = np.stack([x, 1.0 - x, y, 1.0 - y], axis=1)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    for i in range(n):
        for b in bounds_vals[i]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Converts physical centers to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def hex_init(rng, pattern, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, count in enumerate(pattern):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(count):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.rand(N, 2) * 0.8 + 0.1
    for _ in range(400):
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-5)
        rep = 1.0 / (dist**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dist[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 30.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.006 * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints}
    
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5],
        [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5]
    ]
    
    inits = []
    for p in patterns:
        for _ in range(3):
            rot = rng.uniform(-0.2, 0.2)
            sc = rng.uniform(0.90, 1.10)
            inits.append(hex_init(rng, p, rot, sc))
            
    for s in range(15):
        inits.append(force_init(s))
        
    # Grid init
    rng_g = np.random.RandomState(42)
    pts_g = np.array([[0.1 + i*0.2, 0.1 + j*0.2] for i in range(5) for j in range(5)])
    pts_g = np.vstack([pts_g, [0.5, 0.5]])
    pts_g += rng_g.uniform(-0.03, 0.03, pts_g.shape)
    inits.append(np.clip(pts_g, 0.02, 0.98))
        
    # Phase 1: Find best init via LP
    for c in inits:
        r, s_val = solve_lp_radii(c)
        if s_val > best_sum:
            best_sum = s_val
            best_c = c.copy()
            best_r = r.copy()
            
    # Phase 2: LP-driven Simulated Annealing on centers
    curr_c = best_c.copy()
    curr_sum = best_sum
    
    temp = 0.05
    step = 0.04
    
    for it in range(12000):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.normal(0, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.02, 0.98)
        
        new_r, new_sum = solve_lp_radii(curr_c)
        delta = new_sum - curr_sum
        
        accept = False
        if delta > 0:
            accept = True
        elif temp > 1e-7:
            if rng.random() < np.exp(min(0.0, delta / temp)):
                accept = True
                
        if accept:
            curr_r = new_r
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = curr_c.copy()
                best_r = curr_r.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9993
        step = max(0.001, step * 0.9995)
        
    # Phase 3: SLSQP Polish
    r_init = np.maximum(best_r * 0.995, 1e-6)
    x0 = to_params(best_c, r_init)
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds_slqp,
                       constraints=cons_slqp, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
        if res.success:
            c_val = constraints(res.x)
            if np.min(c_val) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_c = np.column_stack((res.x[:N] + res.x[N:2*N]*(1-2*res.x[:N]), 
                                             res.x[:N] + res.x[2*N:3*N]*(1-2*res.x[:N])))
                    best_r = res.x[:N]
    except Exception:
        pass
        
    # Phase 4: Perturbation & SLSQP
    for k in range(20):
        x0_p = to_params(best_c, best_r * 0.99)
        x0_p[N:3*N] += rng.uniform(-0.01, 0.01, 2*N)
        x0_p[:N] += rng.uniform(-0.002, 0.002, N)
        x0_p[:N] = np.clip(x0_p[:N], 1e-6, 0.49)
        x0_p[N:3*N] = np.clip(x0_p[N:3*N], 0.0, 1.0)
        
        try:
            res_p = minimize(objective, x0_p, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res_p.success:
                c_val = constraints(res_p.x)
                if np.min(c_val) >= -1e-7:
                    s_val = -res_p.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_c = np.column_stack((res_p.x[:N] + res_p.x[N:2*N]*(1-2*res_p.x[:N]), 
                                                 res_p.x[:N] + res_p.x[2*N:3*N]*(1-2*res_p.x[:N])))
                        best_r = res_p.x[:N]
        except Exception:
            pass
            
    # Final high-precision polish
    x0_f = to_params(best_c, np.maximum(best_r * 0.999, 1e-6))
    try:
        res_f = minimize(objective, x0_f, method='SLSQP', bounds=bounds_slqp,
                         constraints=cons_slqp, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success:
            c_val = constraints(res_f.x)
            if np.min(c_val) >= -1e-8:
                best_c = np.column_stack((res_f.x[:N] + res_f.x[N:2*N]*(1-2*res_f.x[:N]), 
                                         res_f.x[:N] + res_f.x[2*N:3*N]*(1-2*res_f.x[:N])))
                best_r = res_f.x[:N]
                best_sum = -res_f.fun
    except Exception:
        pass
        
    # Fallback (extremely rare)
    if best_c is None:
        best_c = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        best_c = np.vstack([best_c, [0.5, 0.5]])
        best_r = np.full(N, 0.09)
        best_sum = float(np.sum(best_r))
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
