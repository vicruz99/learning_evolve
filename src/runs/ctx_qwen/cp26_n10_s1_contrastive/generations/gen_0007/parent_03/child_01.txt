import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    for i in range(n):
        for j in range(i + 1, n):
            d = dists[i, j]
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def get_obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def get_constraints(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2  =>  dist^2 - (r_i+r_j)^2 >= 0
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(seed, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.1
    y = r_est
    for r_idx, count in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(count):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    for _ in range(400):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        f += np.sum((1.0/dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        for d in range(2):
            f[:, d] += 15.0 * np.maximum(0, 0.05 - pts[:, d])
            f[:, d] -= 15.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.003 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def grid_init(seed):
    """Generates a perturbed grid initialization."""
    rng = np.random.RandomState(seed)
    pts = np.array([[0.1 + i*0.2, 0.1 + j*0.2] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-0.04, 0.04, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def centers_to_params(centers):
    """Converts physical centers to (r, u, v) optimization parameters."""
    r, _ = solve_lp(centers)
    # Scale down slightly to ensure strict interior feasibility for SLSQP
    r = np.clip(r * 0.98, 1e-6, 0.49)
    x, y = centers[:, 0], centers[:, 1]
    
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((x - r) / denom, 0.0, 1.0)
    v = np.clip((y - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])
