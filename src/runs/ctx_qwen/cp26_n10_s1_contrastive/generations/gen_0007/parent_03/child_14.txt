import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constr = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constr, n))
    b_ub = np.zeros(num_constr)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-5)

def obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constr(params):
    """
    Inequality constraints: pairwise non-overlap.
    Boundary constraints are automatically satisfied by parameterization.
    Returns array of values that must be >= 0.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def centers_to_params(centers):
    """Converts physical centers to (r, u, v) optimization parameters."""
    r = solve_lp_radii(centers)
    r = np.clip(r * 0.995, 1e-6, 0.49)
    x, y = centers[:, 0], centers[:, 1]
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((x - r) / denom, 0.0, 1.0)
    v = np.clip((y - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def hex_init(seed, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, count in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(count):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    pts = np.clip(pts, 0.03, 0.97)
    return pts

def force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    for _ in range(500):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        f += np.sum((1.0 / dist**2)[:, :, np.newaxis] * diff / dist[:, :, np.newaxis], axis=1)
        for d in range(2):
            f[:, d] += 15.0 * np.maximum(0, 0.05 - pts[:, d])
            f[:, d] -= 15.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.003 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def grid_init(seed):
    """Generates a perturbed grid initialization."""
    rng = np.random.RandomState(seed)
    pts = np.array([[0.1 + i * 0.2, 0.1 + j * 0.2] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-0.03, 0.03, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_sa(centers, rng):

    best_c = centers.copy()
    r_best = solve_lp_radii(best_c)
    best_sum = np.sum(r_best)
    curr_c = best_c.copy()
    curr_sum = best_sum
    
    temp = 0.03
    step = 0.035
    
    for it in range(2500):
        idx = rng.randint(0, N)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.normal(0, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        
        r_new = solve_lp_radii(curr_c)
        new_sum = np.sum(r_new)
        delta = new_sum - curr_sum

                best_c = curr_c.copy()
                r_best = r_new.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9992
        step = max(0.0005, step * 0.9996)
        
    return best_c, r_best, best_sum
