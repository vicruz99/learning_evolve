To maximize the sum of radii for 26 circles in a unit square, we combine three powerful techniques:
1. **Linear Programming for Radii**: Given fixed centers, the optimal radii maximizing the sum can be found exactly and efficiently via LP. This decouples the hard combinatorial topology from the convex radius assignment.
2. **Coordinate Ascent on Centers**: We iteratively optimize each circle's 2D position to maximize the LP-computed sum of radii. This is robust against non-smooth objective landscapes.
3. **Simulated Annealing (SA)**: To escape local minima where coordinate ascent gets stuck, we apply SA on centers with a cooling schedule, accepting worse moves probabilistically.
4. **SLSQP Polynomial**: A final gradient-based refinement using a boundary-safe parameterization `(r, u, v)` where `x = r + u*(1-2r)`, `y = r + v*(1-2r)` guarantees boundary constraints while optimizing pairwise non-overlap constraints to high precision.

This pipeline is initialized with diverse hexagonal, force-directed, and perturbed grids to cover the search space thoroughly. The LP solver uses `method='highs'` for speed and stability. All helper functions are top-level to comply with constraints.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_fast(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    ub = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    ub = np.maximum(ub, 0.0)
    bounds = [(0.0, u) for u in ub]
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    k = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def get_sum_r(centers):
    """Wrapper to get sum of radii for fixed centers."""
    _, s = solve_lp_fast(centers)
    return s

def obj_ca(pos, centers, i):
    """Objective for coordinate ascent: minimize negative sum of radii."""
    nc = centers.copy()
    nc[i] = np.clip(pos, 0.0, 1.0)
    return -get_sum_r(nc)

def optimize_center_i(centers, i):
    """Optimizes the position of a single circle to maximize LP sum of radii."""
    x0 = centers[i]
    res = minimize(obj_ca, x0, args=(centers, i), method='Nelder-Mead', 
                   options={'maxiter': 300, 'xatol': 1e-7, 'fatol': 1e-9})
    return res.x

def obj_slqp(x):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[:N])

def cons_slqp(x):
    """Inequality constraints for SLSQP: pairwise non-overlap."""
    r = x[:N]
    u = x[N:2*N]
    v = x[2*N:3*N]
    
    xc = r + u * (1.0 - 2.0 * r)
    yc = r + v * (1.0 - 2.0 * r)
    
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def run_packing():
    rng = np.random.default_rng(42)
    
    inits = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], 
                [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7]]
    
    # 1. Hexagonal lattice patterns with rotations and scales
    for p in patterns:
        for s in range(4):
            pts = []
            r_est = 0.095 + rng.uniform(-0.01, 0.01)
            y = r_est
            for r_idx, cnt in enumerate(p):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N: 
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
            pts = np.array(pts[:N])
            
            rot = rng.uniform(-0.25, 0.25)
            c, s_val = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s_val], [s_val, c]]) + 0.5
            pts += rng.uniform(-0.015, 0.015, pts.shape)
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    # 2. Force-directed layouts
    for s in range(10):
        pts = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(400):
            f = np.zeros_like(pts)
            diff = pts[:, None, :] - pts[None, :, :]
            d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-5
            f += np.sum((1.0/d**2)[:, :, None] * diff / d[:, :, None], axis=1)
            pts += 0.004 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    best_c = None
    best_s = -np.inf
    
    # 3. Optimize each initialization
    for c_init in inits:
        curr_c = c_init.copy()
        curr_s = get_sum_r(curr_c)
        
        # Coordinate Ascent on centers
        for cycle in range(3):
            for i in range(N):
                new_pos = optimize_center_i(curr_c, i)
                nc = curr_c.copy()
                nc[i] = new_pos
                new_s = get_sum_r(nc)
                if new_s > curr_s + 1e-9:
                    curr_c = nc
                    curr_s = new_s
                    
        # Simulated Annealing to escape local minima
        temp = 0.04
        step = 0.025
        for _ in range(1500):
            i = rng.integers(N)
            old = curr_c[i].copy()
            curr_c[i] += rng.normal(0, step, 2)
            curr_c[i] = np.clip(curr_c[i], 0.01, 0.99)
            
            new_s = get_sum_r(curr_c)
            delta = new_s - curr_s
            
            if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(min(0.0, delta / temp))):
                curr_s = new_s
            else:
                curr_c[i] = old
                
            temp *= 0.996
            step = max(0.001, step * 0.998)
            
        if curr_s > best_s:
            best_s = curr_s
            best_c = curr_c.copy()
            
    # 4. High-precision SLSQP polish with boundary-safe parameterization
    r_init, _ = solve_lp_fast(best_c)
    r_init = r_init * 0.985  # Shrink slightly for strict interior feasibility
    denom = np.clip(1.0 - 2.0 * r_init, 1e-6, 1.0)
    u_init = np.clip((best_c[:, 0] - r_init) / denom, 0.0, 1.0)
    v_init = np.clip((best_c[:, 1] - r_init) / denom, 0.0, 1.0)
    x0 = np.concatenate([r_init, u_init, v_init])
    
    bounds_slqp = [(1e-6, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': cons_slqp}
    
    try:
        res = minimize(obj_slqp, x0, method='SLSQP', bounds=bounds_slqp, constraints=cons,
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success:
            c_val = cons_slqp(res.x)
            if np.min(c_val) >= -1e-7:
                s_val = -res.fun
                if s_val > best_s:
                    best_s = s_val
                    r_opt = res.x[:N]
                    u_opt = res.x[N:2*N]
                    v_opt = res.x[2*N:3*N]
                    best_c = np.column_stack([r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)])
    except Exception:
        pass
        
    # Final LP to guarantee strict constraint satisfaction and consistency
    radii, final_sum = solve_lp_fast(best_c)
    
    return best_c, radii, float(final_sum)
