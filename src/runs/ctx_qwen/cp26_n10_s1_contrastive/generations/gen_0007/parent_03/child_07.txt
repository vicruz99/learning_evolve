import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints(params):
    """
    Inequality constraints: pairwise non-overlap.
    Boundary constraints are automatically satisfied by parameterization.
    Returns array of values that must be >= 0.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        for b in [x[i], 1.0-x[i], y[i], 1.0-y[i]]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    for i in range(n):
        for j in range(i+1, n):
            d = dists[i, j]
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds_r = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-4), 0.0

def centers_to_params(centers):
    """Converts physical centers to (r, u, v) optimization parameters."""
    r, _ = solve_lp(centers)
    # Scale down slightly to ensure strict interior feasibility for SLSQP
    r = np.clip(r * 0.99, 1e-6, 0.49)
    x, y = centers[:, 0], centers[:, 1]
    
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((x - r) / denom, 0.0, 1.0)
    v = np.clip((y - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def make_hex(seed, row_counts, rot=0.0, scale=1.0, jitter=0.02):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, count in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(count):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
        
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    for _ in range(500):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        # Coulomb-like repulsion
        f += np.sum((1.0/dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        # Wall repulsion
        for d in range(2):
            f[:, d] += 15.0 * np.maximum(0, 0.08 - pts[:, d])
            f[:, d] -= 15.0 * np.maximum(0, pts[:, d] - 0.92)
        pts += 0.005 * f
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def run_slqp(params):
    """Runs SLSQP optimization from given parameters."""
    bounds = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    try:
        res = minimize(objective, params, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
        if res.success:
            c_val = constraints(res.x)
            if np.min(c_val) >= -1e-7:
                return res.x, -res.fun
    except Exception:
        pass
    return params, -objective(params)

def run_sa_and_ca(centers, rng, max_sa_iters=3500, init_temp=0.045):
    """Runs Simulated Annealing followed by Coordinate Ascent on centers."""
    curr_c = centers.copy()
    r_curr, curr_sum = solve_lp(curr_c)
    best_c = curr_c.copy()
    best_r = r_curr.copy()
    best_sum = curr_sum
    
    temp = init_temp
    step = 0.035
    
    # SA Phase
    for it in range(max_sa_iters):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.normal(0, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.02, 0.98)
        
        r_new, sum_new = solve_lp(curr_c)
        delta = sum_new - curr_sum
        
        if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(delta / temp)):
            curr_sum = sum_new
            if sum_new > best_sum:
                best_sum = sum_new
                best_c = curr_c.copy()
                best_r = r_new.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9990
        step = max(0.001, step * 0.9993)
        
    # Coordinate Ascent Phase
    ca_c = best_c.copy()
    _, ca_sum = solve_lp(ca_c)
    step = 0.025
    for _ in range(1200):
        idx = rng.integers(N)
        best_move_sum = ca_sum
        best_move = None
        
        # Try several random directions
        for _ in range(6):
            direction = rng.normal(0, 1, 2)
            direction /= np.linalg.norm(direction)
            nc = ca_c.copy()
            nc[idx] = np.clip(nc[idx] + step * direction, 0.02, 0.98)
            _, ns = solve_lp(nc)
            if ns > best_move_sum:
                best_move_sum = ns
                best_move = nc.copy()
                
        if best_move is not None:
            ca_c = best_move
            ca_sum = best_move_sum
            if ca_sum > best_sum:
                best_sum = ca_sum
                best_c = ca_c.copy()
                _, best_r = solve_lp(best_c)
            step = max(0.001, step * 0.96)
        else:
            step = min(0.05, step * 1.04)
            
    return best_c, best_r, best_sum

def run_packing():
    rng = np.random.default_rng(42)
    best_x = None
    best_sum = -np.inf
    
    inits = []

    ]
    
    # Generate diverse initial configurations
    for p in patterns:
        for s in range(5):
            rot = (s % 5 - 2) * 0.12

    for cycle in range(3):

    x = r + u * (1.0 - 2.0 * r)

    centers = np.column_stack([x, y])
    radii = np.maximum(r, 0.0)
    
    return centers, radii, float(best_sum)
