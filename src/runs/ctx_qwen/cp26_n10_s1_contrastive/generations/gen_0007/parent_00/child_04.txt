import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:,0,np.newaxis] - centers[np.newaxis,:,0]
    dy = centers[:,1,np.newaxis] - centers[np.newaxis,:,1]
    dists = np.sqrt(dx**2 + dy**2)
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i,j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0), -res.fun
    return np.full(n, 1e-5), 0.0

def to_params(centers, radii):
    """Converts physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj_func(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons_func(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def gen_hex(seed, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_e = 0.095
    y = r_e
    row = 0
    counts = [6,5,6,5,4]
    for cnt in counts:
        shift = (row % 2) * r_e
        x = r_e + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_e
        y += np.sqrt(3.0) * r_e
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def gen_grid(seed):
    """Generates a perturbed grid initialization."""
    rng = np.random.RandomState(seed)
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-0.03, 0.03, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    # Diverse initial configurations
    inits = []
    for s in range(15):
        inits.append(gen_hex(s, rot=np.random.uniform(-0.2, 0.2), scale=np.random.uniform(0.9, 1.1)))
    for s in range(5):
        inits.append(gen_grid(s))
        
    rng = np.random.default_rng(42)
    bounds_p = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': cons_func}
    
    # Phase 1 & 2: LP-driven Simulated Annealing on centers
    for init_c in inits:
        curr_c = init_c.copy()
        curr_r, curr_s = solve_lp(curr_c)
        
        temp = 0.06
        step = 0.05
        for it in range(7000):
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
            new_r, new_s = solve_lp(curr_c)
            delta = new_s - curr_s
            
            if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(delta / temp)):
                curr_s = new_s
                if new_s > best_sum:
                    best_sum = new_s
                    best_c = curr_c.copy()
                    best_r = new_r.copy()
            else:
                curr_c[idx] = old
                
            temp *= 0.9993
            step = max(0.002, step * 0.9995)
            
        # Phase 3: SLSQP Refinement on parameterized space
        if best_c is not None:
            r_shrink = np.maximum(best_r * 0.99, 1e-5)
            p0 = to_params(best_c, r_shrink)
            
            try:
                res = minimize(obj_func, p0, method='SLSQP', bounds=bounds_p,
                               constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13})
                if res.success and np.min(cons_func(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        r_opt = res.x[:N]
                        u_opt = res.x[N:2*N]
                        v_opt = res.x[2*N:3*N]
                        x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
                        y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
                        best_c = np.column_stack((x_opt, y_opt))
                        best_r = r_opt
                        
                # Perturbation escapes from SLSQP local minima
                for _ in range(25):
                    p_pert = res.x.copy()
                    p_pert[N:3*N] += rng.uniform(-0.015, 0.015, 2*N)
                    p_pert[:N] += rng.uniform(-0.001, 0.001, N)
                    p_pert[:N] = np.clip(p_pert[:N], 1e-6, 0.5)
                    p_pert[N:3*N] = np.clip(p_pert[N:3*N], 0.0, 1.0)
                    
                    try:
                        res2 = minimize(obj_func, p_pert, method='SLSQP', bounds=bounds_p,
                                        constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
                        if res2.success and np.min(cons_func(res2.x)) >= -1e-7:
                            s2 = -res2.fun
                            if s2 > best_sum:
                                best_sum = s2
                                r_opt = res2.x[:N]
                                u_opt = res2.x[N:2*N]
                                v_opt = res2.x[2*N:3*N]
                                x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
                                y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
                                best_c = np.column_stack((x_opt, y_opt))
                                best_r = r_opt
                    except Exception:
                        pass
            except Exception:
                pass
                
    # Phase 4: High-precision final polish
    if best_c is not None:
        r_shrink = np.maximum(best_r * 0.999, 1e-5)
        p0 = to_params(best_c, r_shrink)
        try:
            res_f = minimize(obj_func, p0, method='SLSQP', bounds=bounds_p,
                             constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
            if res_f.success and np.min(cons_func(res_f.x)) >= -1e-8:
                best_sum = -res_f.fun
                r_opt = res_f.x[:N]
                u_opt = res_f.x[N:2*N]
                v_opt = res_f.x[2*N:3*N]
                x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
                y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
                best_c = np.column_stack((x_opt, y_opt))
                best_r = r_opt
        except Exception:
            pass
            
    # Fallback valid configuration
    if best_c is None:
        best_c = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_r, best_sum = solve_lp(best_c)
        
    return best_c, best_r, float(best_sum)
