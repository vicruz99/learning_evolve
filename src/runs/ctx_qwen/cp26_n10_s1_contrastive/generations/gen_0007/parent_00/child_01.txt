N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective_func(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[:N])

def constraint_func(v):
    """
    Inequality constraints for SLSQP.
    Uses (r, u, w) parameterization which automatically satisfies boundary constraints.
    Returns dist^2 - (r_i + r_j)^2 >= 0.
    """
    r = v[:N]
    u = v[N:2*N]
    w = v[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def solve_lp(centers):
    """Given fixed centers, solves LP to find optimal radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A = []
    b = []
    # Boundary constraints
    for i in range(n):
        xi, yi = centers[i]
        for bound in [xi, 1.0-xi, yi, 1.0-yi]:
            row = np.zeros(n)
            row[i] = 1.0
            A.append(row)
            b.append(bound)
    # Pairwise constraints
    dx = centers[:, None, 0] - centers[None, :, 0]
    dy = centers[:, None, 1] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)

        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A.append(row)
            b.append(dists[i, j])
    A = np.array(A)
    b = np.array(b)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def pts_to_params(pts):
    """Converts physical centers to (r, u, w) optimization parameters."""
    r = np.zeros(N)
    for i in range(N):
        dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])

    r = np.maximum(r, 1e-4)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((pts[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((pts[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, w])

                row += 1

            pts += rng.uniform(-0.02, 0.02, pts.shape)
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
    return inits

def force_init(rng):

            f[:, d] -= 30.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.006 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts
