import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4*n + n*(n-1)//2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        A_ub[k, i] = 1.0; b_ub[k] = centers[i, 0]; k+=1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0-centers[i, 0]; k+=1
        A_ub[k, i] = 1.0; b_ub[k] = centers[i, 1]; k+=1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0-centers[i, 1]; k+=1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i, j in zip(I_IDX, J_IDX):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]
        k += 1
        
    bounds = [(0.0, None)]*n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 0.0

def obj_p(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[:N])

def cons_p(v):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = v[:N]
    u = v[N:2*N]
    w = v[2*N:3*N]
    
    x = r + u*(1.0-2.0*r)
    y = r + w*(1.0-2.0*r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_p(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    den = np.clip(1.0-2.0*r, 1e-6, 1.0)
    u = np.clip((centers[:,0]-r)/den, 0.0, 1.0)
    w = np.clip((centers[:,1]-r)/den, 0.0, 1.0)
    return np.concatenate([r, u, w])

def from_p(v):
    """Reconstruct physical centers and radii from parameters."""
    r = v[:N]
    u = v[N:2*N]
    w = v[2*N:3*N]
    x = r + u*(1.0-2.0*r)
    y = r + w*(1.0-2.0*r)
    return np.column_stack((x, y)), r

def gen_hex(rows, seed, rot, sc, jit):
    """Generates a hexagonal lattice initialization with specified parameters."""
    rng = np.random.RandomState(seed)
    pts = []
    re = 0.09
    y = re
    r_i = 0
    for cnt in rows:
        shift = (r_i%2)*re
        x = re + shift
        for _ in range(cnt):
            if len(pts)<N: pts.append([x,y])
            x += 2.0*re
        y += np.sqrt(3.0)*re
        r_i += 1
    pts = np.array(pts[:N])
    pts = (pts-0.5)*sc + 0.5
    if rot!=0:
        c,s = np.cos(rot), np.sin(rot)
        M = np.array([[c,-s],[s,c]])
        pts = (pts-0.5)@M.T + 0.5
    pts += rng.uniform(-jit, jit, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def run_packing():
    np.seterr(all='ignore')
    np.random.seed(42)
    
    bnds = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*(2*N)
    cons = {'type': 'ineq', 'fun': cons_p}
    
    best_v = None
    best_sum = -np.inf
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [4,5,5,5,7], [5,7,5,5,4],
        [8,6,6,6], [6,6,6,4,4], [7,6,5,6], [5,6,6,5,4],
        [6,5,5,5,5], [5,5,6,5,5], [4,5,6,6,5]
    ]
    
    # Phase 1: Broad search from diverse hexagonal patterns + SLSQP
    for pat in patterns:
        for s in range(8):
            rot = np.random.uniform(-0.25, 0.25)
            sc = np.random.uniform(0.88, 1.12)
            jit = np.random.uniform(0.01, 0.035)
            pts = gen_hex(pat, s, rot, sc, jit)
            r_lp, s_lp = solve_lp(pts)
            
            if s_lp > best_sum:
                best_sum = s_lp
                best_v = to_p(pts, r_lp*0.99)
                
            v0 = best_v.copy() if best_v is not None else to_p(pts, r_lp*0.99)
            try:
                res = minimize(obj_p, v0, method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-12})
                if res.success and np.min(cons_p(res.x)) >= -1e-7:
                    sv = -res.fun
                    if sv > best_sum:
                        best_sum = sv
                        best_v = res.x.copy()
            except: pass

    # Phase 2: LP-Driven Local Search (Move & Swap)
    if best_v is not None:
        curr_c, curr_r = from_p(best_v)

                curr_s = ns

                    best_v = to_p(curr_c, nr*0.995)

        # Polish LP result with SLSQP

                if sv > best_sum:

                    best_v = res.x.copy()
        except: pass

    # Phase 3: Iterative Perturbation + SLSQP to escape local basins
    if best_v is not None:

            try:
                res = minimize(obj_p, v_p, method='SLSQP', bounds=bnds, constraints=cons,

                    sv = -res.fun
                    if sv > best_sum:
                        best_sum = sv
                        best_v = res.x.copy()
            except: pass

    # Phase 4: High-precision final polish
    if best_v is not None:
        v0 = best_v.copy()
        v0[:N] *= 0.999  # Slight shrink for strict interior feasibility
        try:
            res = minimize(obj_p, v0, method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-14})
            if res.success and np.min(cons_p(res.x)) >= -1e-8:
                best_sum = -res.fun
                best_v = res.x.copy()
        except: pass

    # Fallback valid configuration
    if best_v is None:
        pts = np.array([[0.1+0.2*i, 0.1+0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        best_v = to_p(pts, np.full(N, 0.09))
        best_sum = np.sum(best_v[:N])

    centers, radii = from_p(best_v)
    return centers, radii, float(best_sum)
