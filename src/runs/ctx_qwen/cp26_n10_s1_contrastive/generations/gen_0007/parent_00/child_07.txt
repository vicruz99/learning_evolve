1. **LP-driven Center Optimization**: For fixed centers, finding optimal radii is a linear program. This objective function `-sum(solve_lp_radii(centers))` is highly non-smooth but perfectly suited for Simulated Annealing (SA) and direct search methods.
2. **Parameterized SLSQP Polish**: Once good centers are found, we switch to a constrained nonlinear optimizer (SLSQP) using a `(r, u, v)` parameterization that automatically satisfies boundary constraints (`r <= x <= 1-r`, `r <= y <= 1-r`), leaving only pairwise non-overlap constraints to be handled explicitly. This avoids constraint violations during optimization.
3. **Iterative Escaping**: We alternate between SA on centers (to change contact topology and escape local minima) and SLSQP on parameters (to precisely push circles to their tightest feasible configuration).
4. **Diverse Initializations**: We generate hexagonal lattices with varied row counts, rotations, scales, plus grids and random layouts to cover different structural basins.
5. **Strict Compliance**: All helper functions are top-level, no lambdas or closures are used, and numerical tolerances are carefully managed.

The pipeline:
- Phase 1: Generate ~30 diverse initial center configurations.
- Phase 2: Run SA on each initialization using LP-sum as the objective. Keep top candidates.
- Phase 3: Convert top candidates to `(r, u, v)` parameters and run SLSQP to polish.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def obj_params(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[:N])

def constr_params(v):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = v[:N]
    u = v[N:2*N]
    vt = v[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + vt * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    vt = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, vt])

def from_params(v):
    """Convert (r, u, v) parameters back to physical centers."""
    r = v[:N]
    u = v[N:2*N]
    vt = v[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + vt * (1.0 - 2.0 * r)
    return np.column_stack((x, y))

def gen_hex_init(rng, row_counts, rot, scale, jitter):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_sa_on_centers(centers, steps, temp_init, rng):
    """Simulated annealing on center positions optimizing LP sum of radii."""
    best_c = centers.copy()
    _, best_s = solve_lp(best_c)
    curr_c = best_c.copy()
    curr_s = best_s
    temp = temp_init
    step_size = 0.03
    
    for _ in range(steps):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.normal(0, step_size, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        
        _, new_s = solve_lp(curr_c)
        delta = new_s - curr_s
        
        if delta > 0 or (temp > 1e-7 and rng.random() < np.exp(delta / temp)):
            curr_s = new_s
            if new_s > best_s:
                best_s = new_s
                best_c = curr_c.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.998
        step_size = max(0.005, step_size * 0.998)
    return best_c, best_s

def get_score(item):
    return item[1]

def run_packing():
    rng = np.random.default_rng(42)
    bounds_opt = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constr_params}
    
    best_v = None
    best_sum = -np.inf
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [7,6,6,7], [8,6,6,6], [6,4,6,5,5], [4,5,6,5,6],
        [7,5,5,5,4], [4,5,5,5,7], [6,6,6,4,4], [8,8,5,5]
    ]
    
    inits = []
    for pat in patterns:
        for _ in range(4):
            rot = rng.uniform(-0.25, 0.25)
            scale = rng.uniform(0.9, 1.1)
            jit = rng.uniform(0.01, 0.03)
            inits.append(gen_hex_init(rng, pat, rot, scale, jit))
            
    for _ in range(10):
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng.uniform(-0.02, 0.02, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    for _ in range(10):
        inits.append(np.clip(rng.random((N, 2)), 0.05, 0.95))
        
    # Phase 1: SA on centers from each init, keep best
    top_centers = []
    for init_c in inits:
        c_sa, s_sa = run_sa_on_centers(init_c, steps=800, temp_init=0.015, rng=rng)
        top_centers.append((c_sa, s_sa))
        
    top_centers.sort(key=get_score, reverse=True)
    
    # Phase 2: SLSQP polish top candidates
    for c_cand, _ in top_centers[:10]:
        r_cand, _ = solve_lp(c_cand)
        r_cand = np.maximum(r_cand * 0.995, 1e-5)
        x0 = to_params(c_cand, r_cand)
        
        try:
            res = minimize(obj_params, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
            if res.success:
                if np.min(constr_params(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        c_fallback = top_centers[0][0]
        r_fallback, _ = solve_lp(c_fallback)
        best_v = to_params(c_fallback, r_fallback)
        best_sum = np.sum(r_fallback)
        
    # Phase 3: Iterative SA + SLSQP refinement from best found
    curr_v = best_v.copy()
    curr_sum = best_sum
    
    for _ in range(3):
        centers_curr = from_params(curr_v)
        centers_refined, _ = run_sa_on_centers(centers_curr, steps=600, temp_init=0.008, rng=rng)
        r_ref, _ = solve_lp(centers_refined)
        r_ref = np.maximum(r_ref * 0.995, 1e-5)
        x0 = to_params(centers_refined, r_ref)
        
        try:
            res = minimize(obj_params, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
            if res.success and np.min(constr_params(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > curr_sum:
                    curr_sum = s_val
                    curr_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 4: High precision final polish
    try:
        res_f = minimize(obj_params, curr_v, method='SLSQP', bounds=bounds_opt,
                         constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
        if res_f.success and np.min(constr_params(res_f.x)) >= -1e-8:
            curr_v = res_f.x
            curr_sum = -res_f.fun
    except Exception:
        pass
        
    # Reconstruct final centers and radii
    centers = from_params(curr_v)
    r_opt = curr_v[:N]
    
    return centers, r_opt, float(curr_sum)
