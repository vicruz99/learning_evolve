import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 0.0

def constraints_slsqp(v):
    """Inequality constraints for joint (x, y, r) optimization: g(v) >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise non-overlap constraints
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def objective_slsqp(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2::3])

def gen_hex(row_counts, rot=0.0, scale=1.0, jitter=0.0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_e = 0.095
    y = r_e
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_e
        x = r_e + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_e
        y += np.sqrt(3.0) * r_e
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = pts @ M.T
        
    if jitter > 0:
        pts += np.random.uniform(-jitter, jitter, pts.shape)
        
    return np.clip(pts, 0.02, 0.98)

def sa_and_ca(init_centers):
    """Runs Simulated Annealing on centers with LP radius evaluation, followed by Coordinate Ascent."""
    cur_c = init_centers.copy()
    cur_r, cur_sum = solve_lp(cur_c)
    best_c = cur_c.copy()
    best_r = cur_r.copy()
    best_sum = cur_sum
    
    T = 0.06
    n_iters = 1800
    
    # Simulated Annealing
    for i in range(n_iters):
        idx = np.random.randint(N)
        old = cur_c[idx].copy()
        step = np.random.uniform(-0.07, 0.07, 2)
        cur_c[idx] = np.clip(old + step, 0.01, 0.99)
        
        new_r, new_sum = solve_lp(cur_c)
        delta = new_sum - cur_sum
        
        if delta > 0 or (T > 1e-6 and np.random.rand() < np.exp(delta / T)):
            cur_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = cur_c.copy()
                best_r = new_r.copy()
        else:
            cur_c[idx] = old
            
        T *= 0.9992
        
    # Coordinate Ascent Refinement
    cur_c = best_c.copy()
    cur_r, cur_sum = solve_lp(cur_c)
    
    for _ in range(60):
        improved = False
        for i in range(N):
            for _ in range(2):
                dir_ = np.random.randn(2)
                dir_ /= np.linalg.norm(dir_)
                for s in [0.012, 0.004, 0.001]:
                    nc = cur_c.copy()
                    nc[i] = np.clip(nc[i] + s * dir_, 0.01, 0.99)
                    nr, ns = solve_lp(nc)
                    if ns > cur_sum + 1e-8:
                        cur_c = nc
                        cur_r = nr
                        cur_sum = ns
                        improved = True
        if cur_sum > best_sum:
            best_c = cur_c.copy()
            best_r = cur_r.copy()
            best_sum = cur_sum
        if not improved:
            break
            
    return best_c, best_r, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    # Diverse hexagonal patterns to explore different boundary alignments
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [7,6,6,7], [8,6,6,6], [6,4,6,5,5], [4,5,6,5,6],
        [7,5,5,5,4], [4,5,5,5,7]
    ]
    
    inits = []
    for pat in patterns:
        for s in range(4):
            rot = (s % 5 - 2) * 0.12
            sc = 0.92 + (s // 5) * 0.08
            jit = 0.015
            inits.append(gen_hex(pat, rot=rot, scale=sc, jitter=jit))
            
    # Add perturbed random layouts
    for s in range(12):
        np.random.seed(s + 100)
        pts = np.random.rand(N, 2) * 0.7 + 0.15
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    # Phase 1 & 2: SA + CA on diverse starts
    for init_pts in inits:
        c, r, s = sa_and_ca(init_pts)
        if s > best_sum:
            best_sum = s
            best_centers = c.copy()
            best_radii = r.copy()
            
    # Phase 3: High-precision SLSQP joint optimization
    if best_centers is not None:
        v0 = np.zeros(3 * N)
        v0[0::3] = best_centers[:, 0]
        v0[1::3] = best_centers[:, 1]
        # Shrink radii slightly to guarantee strict interior feasibility for SLSQP start
        v0[2::3] = np.maximum(best_radii * 0.985, 1e-5)
        
        bounds_slqp = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
        cons = {'type': 'ineq', 'fun': constraints_slsqp}
        
        # First polish
        try:
            res = minimize(objective_slsqp, v0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = constraints_slsqp(res.x)
                if np.min(c_val) >= -1e-7:
                    final_sum = -res.fun
                    if final_sum > best_sum:
                        best_sum = final_sum
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
        except Exception:
            pass
            
        # Second high-precision polish from the updated point
        if best_centers is not None:
            v0[0::3] = best_centers[:, 0]
            v0[1::3] = best_centers[:, 1]
            v0[2::3] = np.maximum(best_radii * 0.998, 1e-5)
            
            try:
                res2 = minimize(objective_slsqp, v0, method='SLSQP', bounds=bounds_slqp,
                                constraints=cons, options={'maxiter': 8000, 'ftol': 1e-15, 'disp': False})
                if res2.success:
                    c_val2 = constraints_slsqp(res2.x)
                    if np.min(c_val2) >= -1e-7:
                        final_sum2 = -res2.fun
                        if final_sum2 > best_sum:
                            best_sum = final_sum2
                            best_centers = np.column_stack((res2.x[0::3], res2.x[1::3]))
                            best_radii = res2.x[2::3]
            except Exception:
                pass

    # Fallback valid configuration (should rarely be reached)
    if best_centers is None:
        best_centers = gen_hex([6,5,6,5,4])
        best_radii, best_sum = solve_lp(best_centers)
        
    # Ensure non-negative radii
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_sum)
