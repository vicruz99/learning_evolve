import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
np.seterr(all='ignore')

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        x, y = centers[i]
        b_vals = [x, 1.0 - x, y, 1.0 - y]
        for b in b_vals:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 0.0

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[:N])

def constraints(v):
    """
    Computes inequality constraints g(v) >= 0.
    Uses (r, u, v) parameterization which automatically satisfies boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = v[:N]
    u = v[N:2*N]
    vt = v[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + vt * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_params_from_pts(pts):
    """Converts physical centers to (r, u, v) optimization parameters."""
    n = pts.shape[0]
    dw = np.minimum(np.minimum(pts[:, 0], 1.0 - pts[:, 0]), 
                    np.minimum(pts[:, 1], 1.0 - pts[:, 1]))
    dists_mat = np.sqrt(((pts[:, np.newaxis, :] - pts[np.newaxis, :, :])**2).sum(axis=2))
    np.fill_diagonal(dists_mat, np.inf)
    dp = np.min(dists_mat, axis=1)
    r = 0.85 * np.minimum(dw, dp / 2.0)
    r = np.maximum(r, 1e-4)
    
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((pts[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((pts[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def gen_hex(row_counts, rot, scale, jitter, rng):
    """Generates a hexagonal lattice initialization with specified parameters."""
    pts = []
    r_e = 0.09
    y = r_e
    row = 0
    for cnt in row_counts:
        x_s = r_e if row % 2 == 0 else 2.0 * r_e
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x_s, y])
            x_s += 2.0 * r_e
        y += np.sqrt(3.0) * r_e
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = pts @ M.T
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Force-directed layout that grows circles until tightly packed."""
    centers = rng.uniform(0.2, 0.8, (N, 2))
    radii = np.full(N, 0.02)
    dt = 0.01
    for step in range(1500):
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-5)
        rep = np.where(dists < radii[:, None] + radii[None, :] + 0.03,
                       (radii[:, None] + radii[None, :] + 0.03 - dists)**2 * 200.0,
                       1.0 / dists**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            margin = radii + 0.015
            forces[:, d] += np.maximum(0, margin - centers[:, d]) * 40.0
            forces[:, d] -= np.maximum(0, centers[:, d] - (1.0 - margin)) * 40.0
        centers += dt * forces
        centers = np.clip(centers, 0.0, 1.0)
        radii *= 1.0003
        dt *= 0.999
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        inits.append(make_params_from_pts(pts))
