import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(val)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, None, 0] - centers[None, :, 0]
    dy = centers[:, None, 1] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-6 * N

def obj_full(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constr_full(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c_list = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    c_list.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c_list)

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # Hexagonal patterns with varied row counts
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [7,6,6,7], [8,6,6,6], [6,4,6,5,5], [4,5,6,5,6],
        [7,5,5,5,4], [4,5,5,5,7], [6,6,6,4,4], [8,8,5,5],
        [6,6,7,5,2], [5,5,6,6,4]
    ]
    
    for pat in patterns:
        for _ in range(3):
            pts = []
            r_e = 0.095
            y = r_e
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_e
                x = r_e + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_e
                y += np.sqrt(3.0) * r_e
                row += 1
                
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.90, 1.10) + 0.5
            
            rot = rng.uniform(-0.25, 0.25)
            c, s = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            
            pts += rng.uniform(-0.02, 0.02, pts.shape)
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    # Perturbed grids
    for _ in range(5):
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng.uniform(-0.03, 0.03, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    # Force-directed layouts
    for s in range(5):
        rng_fd = np.random.RandomState(s + 42)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        for _ in range(200):
            f = np.zeros_like(pts)
            diff = pts[:, None, :] - pts[None, :, :]
            dist = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dist**2
            np.fill_diagonal(rep, 0.0)
            f += np.sum(rep[:, :, None] * diff / dist[:, :, None], axis=1)
            pts += 0.005 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_c = None
    best_r = None
    best_sum = -np.inf
    
    inits = generate_inits(rng)
    
    for init_pts in inits:
        c = init_pts.copy()
        r, curr_sum = solve_lp(c)
        
        # LP-Driven Simulated Annealing on centers
        temp = 0.025
        step = 0.04
        for it in range(1200):
            idx = rng.integers(N)
            old_c = c[idx].copy()
            
            # Random move or swap
            if rng.random() < 0.85:
                c[idx] += rng.normal(0, step, 2)
                c[idx] = np.clip(c[idx], 0.01, 0.99)
            else:
                jdx = rng.integers(N)
                if jdx != idx:
                    c[idx], c[jdx] = c[jdx].copy(), c[idx].copy()
                    
            r_new, new_s = solve_lp(c)
            delta = new_s - curr_sum
            
            # Accept if better or probabilistically if worse
            if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(min(0.0, delta / temp))):
                curr_sum = new_s
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_c = c.copy()
                    best_r = r_new.copy()
            else:
                c[idx] = old_c
                
            temp *= 0.998
            step = max(0.002, step * 0.997)
            
        # Check final state of this init
        r_final, s_final = solve_lp(c)
        if s_final > best_sum:
            best_sum = s_final
            best_c = c.copy()
            best_r = r_final.copy()
            
        # SLSQP Polish on best found so far
        if best_c is not None:
            v0 = np.zeros(3 * N)
            v0[0::3] = best_c[:, 0]
            v0[1::3] = best_c[:, 1]
            v0[2::3] = best_r * 0.995
            
            slqp_bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
            cons = {'type': 'ineq', 'fun': constr_full}

                res = minimize(obj_full, v0, method='SLSQP', bounds=slqp_bounds,
                               constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13})

                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_r = res.x[2::3]
            except Exception:
                pass
                
        # Perturbation & Restart SLSQP
        for _ in range(8):
            c_p = best_c + rng.normal(0, 0.012, (N, 2))
            c_p = np.clip(c_p, 0.02, 0.98)
            r_p, _ = solve_lp(c_p)
            
            v0 = np.zeros(3 * N)
            v0[0::3] = c_p[:, 0]
            v0[1::3] = c_p[:, 1]
            v0[2::3] = r_p * 0.995
            
            try:

                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_r = res.x[2::3]
            except Exception:
                pass
                
    # Final high-precision polish
    if best_c is not None:
        v0 = np.zeros(3 * N)
        v0[0::3] = best_c[:, 0]
        v0[1::3] = best_c[:, 1]
        v0[2::3] = best_r * 0.9999
        
        try:
            res = minimize(obj_full, v0, method='SLSQP', bounds=slqp_bounds,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
            if np.min(constr_full(res.x)) >= -1e-8:

        best_c = fallback
        best_r = np.full(N, 0.08)
        best_sum = np.sum(best_r)
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
