import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find optimal radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0

    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1

    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)

    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1

    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def objective_params(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[:N])

def constraints_params(v):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = v[:N]
    u = v[N:2*N]
    vt = v[2*N:3*N]
    
    # Map normalized u, v to actual coordinates within [r, 1-r]
    x = r + u * (1.0 - 2.0 * r)
    y = r + vt * (1.0 - 2.0 * r)
    
    # Pairwise squared distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    # Squared sum of radii
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_params(centers, radii):
    """Converts physical centers to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def gen_hex_init(seed, rot, scale):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    row_counts = [6, 5, 6, 5, 4]
    
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = pts @ M.T
        
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_sa_iter(seed, max_iter=3500):
    """LP-driven Simulated Annealing to find high-density topology."""
    rng = np.random.RandomState(seed)
    centers = gen_hex_init(seed, 0.0, 1.0)
    r_cur, s_cur = solve_lp(centers)
    best_c = centers.copy()
    best_r = r_cur.copy()
    best_s = s_cur

    temp = 0.06
    step = 0.045
    for it in range(max_iter):
        idx = rng.randint(N)
        old = centers[idx].copy()
        centers[idx] += rng.normal(0, step, 2)
        centers[idx] = np.clip(centers[idx], 0.02, 0.98)

        r_new, s_new = solve_lp(centers)
        delta = s_new - s_cur
        
        # Metropolis acceptance
        if delta > 0 or (temp > 1e-8 and rng.rand() < np.exp(max(-500.0, delta / temp))):
            s_cur = s_new
            if s_new > best_s:
                best_s = s_new
                best_c = centers.copy()
                best_r = r_new.copy()
        else:
            centers[idx] = old

        temp *= 0.9992
        step = max(0.002, step * 0.9994)

    return best_c, best_r, best_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_v = None
    best_sum = -np.inf
    bounds_opt = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N

    # Phase 1: SA-based topology search
    for s in range(12):

            # Prepare for SLSQP refinement
            v_sa = make_params(c_sa, r_sa)
            v_sa[:N] *= 0.995  # slight shrink for strict interior feasibility
            try:
                res = minimize(objective_params, v_sa, method='SLSQP',
                               bounds=bounds_opt, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(constraints_params(res.x)) >= -1e-7:
                    val = -res.fun

                        best_sum = val
                        best_v = res.x.copy()
            except Exception:

    # Phase 2: Direct SLSQP from diverse hex starts (catches symmetries SA might miss)

    centers = np.column_stack((x_out, y_out))

    return centers, radii, float(best_sum)
