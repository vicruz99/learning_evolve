import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find optimal radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Precompute boundary distances
    x, y = centers[:, 0], centers[:, 1]
    b_bounds = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Precompute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(b_bounds[i])
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback
    return np.full(n, 1e-6), 1e-5

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints(params):
    """
    Inequality constraints: pairwise non-overlap.
    Boundary constraints are automatically satisfied by parameterization.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_params(centers, radii):
    """Converts physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    r = np.clip(r, 1e-6, 0.499)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def gen_hex(row_counts, seed, rot, scale, jitter):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_e = 0.095
    y = r_e
    row = 0
    for cnt in row_counts:
        shift = 0.0 if row % 2 == 0 else r_e
        x = r_e + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_e
        y += np.sqrt(3.0) * r_e
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = pts @ M.T
        
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Force-directed layout to spread points organically."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        pts += 0.005 * forces
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def optimize_centers_lp(centers):
    """Coordinate ascent on centers maximizing LP sum of radii."""
    curr_centers = centers.copy()
    _, curr_sum = solve_radii_lp(curr_centers)
    
    for sweep in range(40):
        improved = False
        for i in np.random.permutation(N):
            best_loc = curr_centers[i].copy()
            best_val = curr_sum
            step = 0.025
            
            for _ in range(12):
                directions = [(step,0), (-step,0), (0,step), (0,-step),
                              (step,step), (step,-step), (-step,step), (-step,-step)]
                for dx, dy in directions:
                    cand = best_loc + np.array([dx, dy])
                    cand = np.clip(cand, 0.01, 0.99)
                    
                    temp = curr_centers.copy()
                    temp[i] = cand
                    _, val = solve_radii_lp(temp)
                    
                    if val > best_val + 1e-8:
                        best_val = val
                        best_loc = cand
                step *= 0.85
                
            if best_val > curr_sum + 1e-7:
                curr_centers[i] = best_loc
                curr_sum = best_val
                improved = True
        if not improved:
            break
    return curr_centers, curr_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds_opt = [(1e-6, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [7,6,6,7], [8,6,6,6], [6,4,6,5,5], [4,5,6,5,6],
        [7,5,5,5,4], [4,5,5,5,7], [6,6,6,4,4], [8,8,5,5],
        [9,5,5,5,2], [5,5,5,5,5,1]
    ]
    
    for pat in patterns:
        for s in range(5):
            rot = np.random.uniform(-0.4, 0.4)
            scale = np.random.uniform(0.85, 1.15)
            jit = np.random.uniform(0.01, 0.03)
            pts = gen_hex(pat, s*10 + 1, rot, scale, jit)
            r_est = np.full(N, 0.08)
            inits.append(make_params(pts, r_est))
            
    for s in range(10):
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += np.random.uniform(-0.015, 0.015, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(make_params(pts, np.full(N, 0.07)))
        
    for s in range(8):
        pts = force_init(s+200)
        inits.append(make_params(pts, np.full(N, 0.07)))
        
    # Phase 1: Broad SLSQP search
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: LP-driven center optimization
    if best_v is not None:
        r_tmp = best_v[:N]
        u_tmp = best_v[N:2*N]
        v_tmp = best_v[2*N:3*N]
        c_tmp = np.column_stack((r_tmp + u_tmp*(1.0-2.0*r_tmp), r_tmp + v_tmp*(1.0-2.0*r_tmp)))
        
        c_opt, sum_opt = optimize_centers_lp(c_tmp)
        if sum_opt > best_sum:
            best_sum = sum_opt
            r_opt, _ = solve_radii_lp(c_opt)
            best_v = make_params(c_opt, r_opt)
            
            # Re-optimize jointly after LP center refinement

                if res.success and np.min(constraints(res.x)) >= -1e-7:

                    if s_val > best_sum:
                        best_sum = s_val
                        best_v = res.x.copy()
            except Exception:
                pass

        r_tmp = best_v[:N]
        u_tmp = best_v[N:2*N]
        v_tmp = best_v[2*N:3*N]

        c_opt, sum_opt = optimize_centers_lp(c_tmp)

        # Slight shrink to guarantee strict interior feasibility for final pass

        try:

                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})

    # Fallback

    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    # Ensure strictly positive radii
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
