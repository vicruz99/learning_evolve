from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find radii maximizing sum(r_i) for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((4*n + n*(n-1)//2, n))
    b_ub = np.zeros(4*n + n*(n-1)//2)
    
    k = 0
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = centers[i]
        bounds = [x, 1.0-x, y, 1.0-y]
        for b in bounds:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def gen_hex(row_counts, rot, scale, jitter, rng):

    pts = []
    r_e = 0.095
    y = r_e
    row = 0
    for cnt in row_counts:

        x = r_e + shift

            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_e
        y += np.sqrt(3.0) * r_e
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:

        pts = pts @ M.T
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def gen_force(rng):

            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.9)
        pts += 0.005 * forces
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def obj_slqp(v):
    """Objective function for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[:N])

def cons_slqp(v):
    """Inequality constraints for SLSQP: pairwise non-overlap."""
    r = v[:N]
    u = v[N:2*N]
    vv = v[2*N:3*N]
    # Parameterization guarantees boundary constraints
    x = r + u * (1.0 - 2.0 * r)
    y = r + vv * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2
