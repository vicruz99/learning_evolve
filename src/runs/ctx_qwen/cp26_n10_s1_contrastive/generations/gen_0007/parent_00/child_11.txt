import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find optimal radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, None, 0] - centers[None, :, 0]
    dy = centers[:, None, 1] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-5

def make_params(centers, radii):
    """Converts physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def get_centers(params):
    """Reconstructs physical centers from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y))

def obj_slsqp(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slsqp(params):
    """Inequality constraints for SLSQP: pairwise non-overlap."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    rs2 = rs**2
    
    return dist2[I_IDX, J_IDX] - rs2[I_IDX, J_IDX]

def gen_hex(row_counts, rot, scale, jitter, seed):
    """Generates a hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    r_e = 0.095
    y = r_e
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_e
        x = r_e + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_e
        y += np.sqrt(3.0) * r_e
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = pts @ M.T
    pts += np.random.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [7,6,6,7], [8,6,6,6], [6,4,6,5,5], [4,5,6,5,6],
        [7,5,5,5,4], [4,5,5,5,7], [6,6,6,4,4], [8,8,5,5],
        [5,5,5,5,5,1], [6,6,5,5,4], [7,7,6,6], [8,8,7,3]
    ]
    
    # Generate diverse initial center configurations
    inits = []
    for pat in patterns:
        for s in range(8):
            rot = rng.uniform(-0.35, 0.35)
            sc = rng.uniform(0.88, 1.12)
            jit = rng.uniform(0.01, 0.04)
            inits.append(gen_hex(pat, rot, sc, jit, s))
            
    for s in range(20):
        rng_f = np.random.RandomState(s)
        pts = rng_f.rand(N, 2) * 0.8 + 0.1
        for step in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-5
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            pts += 0.005 * (0.99**step) * forces
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Find best initialization using LP
    for init_pts in inits:
        r, s = solve_lp_radii(init_pts)
        if s > best_sum:
            best_sum = s
            best_c = init_pts.copy()
            best_r = r.copy()
            
    # Simulated Annealing on centers using LP evaluation
    current_c = best_c.copy()
    current_r, curr_sum = solve_lp_radii(current_c)
    best_lp_sum = curr_sum
    
    temp = 0.05
    step = 0.035
    
    for it in range(8000):
        if rng.random() < 0.7:
            idx = rng.integers(N)
            old = current_c[idx].copy()
            current_c[idx] += rng.normal(0, step, 2)
            current_c[idx] = np.clip(current_c[idx], 0.005, 0.995)
            is_swap = False
        else:
            idx1, idx2 = rng.choice(N, 2, replace=False)
            current_c[idx1], current_c[idx2] = current_c[idx2].copy(), current_c[idx1].copy()
            is_swap = True
            
        new_r, new_sum = solve_lp_radii(current_c)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(delta / temp)):
            curr_sum = new_sum
            current_r = new_r
            if new_sum > best_lp_sum:
                best_lp_sum = new_sum
                best_c = current_c.copy()
                best_r = current_r.copy()
        else:
            if is_swap:
                current_c[idx1], current_c[idx2] = current_c[idx2].copy(), current_c[idx1].copy()
            else:
                current_c[idx] = old
                
        temp *= 0.9991
        step = max(0.0015, step * 0.9993)
        
    # Prepare for SLSQP refinement
    bounds_opt = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': cons_slsqp}
    
    params = make_params(best_c, best_r * 0.995)
    try:
        res = minimize(obj_slsqp, params, method='SLSQP', bounds=bounds_opt,
                       constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13})
        if res.success and np.min(cons_slsqp(res.x)) >= -1e-7:
            s_val = -res.fun
            if s_val > best_lp_sum:
                best_lp_sum = s_val
                best_c = get_centers(res.x)
                best_r = res.x[:N]
    except Exception:
        pass
        
    # Perturbation refinement to escape SLSQP local minima
    for k in range(30):
        x0 = make_params(best_c, best_r * 0.998)
        x0[N:3*N] += rng.normal(0, 0.015, 2*N)
        x0[:N] += rng.normal(0, 0.001, N)
        x0[:N] = np.clip(x0[:N], 1e-6, 0.5)
        x0[N:3*N] = np.clip(x0[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slsqp, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
            if res.success and np.min(cons_slsqp(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_lp_sum:
                    best_lp_sum = s_val
                    best_c = get_centers(res.x)
                    best_r = res.x[:N]
        except Exception:
            pass
            
    # High-precision final polish
    x0 = make_params(best_c, best_r * 0.9999)
    try:
        res = minimize(obj_slsqp, x0, method='SLSQP', bounds=bounds_opt,
                       constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
        if np.min(cons_slsqp(res.x)) >= -1e-8:
            best_c = get_centers(res.x)
            best_r = res.x[:N]
            best_lp_sum = np.sum(best_r)
    except Exception:
        pass
        
    # Fallback (should not be reached with valid inputs)
    if best_c is None:
        best_c = np.random.rand(N, 2)
        best_r, _ = solve_lp_radii(best_c)
        best_lp_sum = np.sum(best_r)
        
    return best_c, np.maximum(best_r, 0.0), float(best_lp_sum)
