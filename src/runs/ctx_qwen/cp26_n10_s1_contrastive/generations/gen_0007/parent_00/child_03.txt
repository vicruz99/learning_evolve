2. **LP Radii Solver**: For any fixed set of centers, the optimal radii that maximize `\sum r_i` subject to non-overlap and boundary constraints form a Linear Programming problem. I will implement a fast LP solver using `scipy.optimize.linprog` (HiGHS method).

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-5

def objective_slqp(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints_slqp(params):
    """
    Computes inequality constraints g(params) >= 0.
    Uses parameterization to automatically satisfy boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def centers_to_params(centers, radii):
    """Convert centers and radii to the (r, u, v) parameterization."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[:N] = r
    params[N:2*N] = u
    params[2*N:3*N] = v
    return params

def generate_hex_init(row_counts, seed, rot, scale, jitter):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
        row += 1
        
    pts = np.array(pts[:N])
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.uniform(-jitter, jitter, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def generate_force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    r_curr = np.full(N, 0.06)
    
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.01 * (0.99**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists_to_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_to_others[i] = np.inf
            d_pair = np.min(dists_to_others)
            r_curr[i] = 0.75 * min(d_wall, d_pair/2.0)
            
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_opt = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraints_slqp}
    
    best_vars = None
    best_sum = -np.inf
    
    # Generate diverse initial center configurations
    inits = []
    
    # 1. Hexagonal lattice patterns
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [5,5,6,5,5], [7,5,5,5,4], [4,5,5,5,7], [5,7,5,5,4],
        [8,6,5,4,3], [7,6,6,4,3], [6,6,6,5,3], [8,8,5,5]
    ]
    
    for pat in patterns:
        for _ in range(4):
            rot = rng.uniform(-0.35, 0.35)
            scale = rng.uniform(0.82, 1.18)
            jit = rng.uniform(0.01, 0.03)
            inits.append(generate_hex_init(pat, rng.integers(0, 10000), rot, scale, jit))
            
    # 2. Force-directed layouts
    for s in range(25):
        inits.append(generate_force_init(s))
        
    # 3. Perturbed grids
    for s in range(15):
        rng_g = np.random.RandomState(s)
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng_g.uniform(-0.025, 0.025, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # Phase 1: Broad search from diverse initializations using SLSQP
    for pts in inits:
        radii, _ = solve_radii_lp(pts)
        radii = np.maximum(radii * 0.985, 1e-5)
        x0 = centers_to_params(pts, radii)
        
        try:
            res = minimize(objective_slqp, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                cons_val = constraints_slqp(res.x)
                if np.min(cons_val) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: LP-Driven Simulated Annealing on centers to escape local minima
    if best_vars is not None:
        r_tmp = best_vars[:N]
        u_tmp = best_vars[N:2*N]
        v_tmp = best_vars[2*N:3*N]
        x_tmp = r_tmp + u_tmp * (1.0 - 2.0 * r_tmp)
        y_tmp = r_tmp + v_tmp * (1.0 - 2.0 * r_tmp)
        best_c = np.column_stack((x_tmp, y_tmp))
        best_r, best_sum = solve_radii_lp(best_c)
        
        current_c = best_c.copy()
        curr_sum = best_sum
        temp = 0.03
        step = 0.025
        
        for it in range(3000):
            idx = rng.integers(N)
            old = current_c[idx].copy()
            current_c[idx] += rng.normal(0, step, 2)
            current_c[idx] = np.clip(current_c[idx], 0.005, 0.995)
            
            r_new, sum_new = solve_radii_lp(current_c)
            delta = sum_new - curr_sum
            
            if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
                curr_sum = sum_new
                if sum_new > best_sum:
                    best_sum = sum_new
                    best_c = current_c.copy()
                    best_r = r_new.copy()
            else:
                current_c[idx] = old
                
            temp *= 0.9992
            step = max(0.001, step * 0.9995)
            
        # Convert best LP result back to params for SLSQP refinement
        best_vars = centers_to_params(best_c, np.maximum(best_r * 0.998, 1e-5))
        
    # Phase 3: Perturbation refinement to change contact topology
    if best_vars is not None:
        for k in range(60):
            x0 = best_vars.copy()
            x0[N:3*N] += rng.normal(0, 0.04, 2*N)
            x0[:N] += rng.normal(0, 0.0015, N)
            
            x0[:N] = np.clip(x0[:N], 1e-6, 0.49)
            x0[N:3*N] = np.clip(x0[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(objective_slqp, x0, method='SLSQP', bounds=bounds_opt,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                if res.success:
                    cons_val = constraints_slqp(res.x)
                    if np.min(cons_val) >= -1e-7:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_vars = res.x.copy()
            except Exception:
                continue
                
        # One final LP sweep on the optimized centers
        r_tmp = best_vars[:N]
        u_tmp = best_vars[N:2*N]
        v_tmp = best_vars[2*N:3*N]
        x_tmp = r_tmp + u_tmp * (1.0 - 2.0 * r_tmp)
        y_tmp = r_tmp + v_tmp * (1.0 - 2.0 * r_tmp)
        centers_lp = np.column_stack((x_tmp, y_tmp))
        
        r_lp, sum_lp = solve_radii_lp(centers_lp)
        if sum_lp > best_sum:
            best_sum = sum_lp
            best_vars = centers_to_params(centers_lp, np.maximum(r_lp * 0.999, 1e-5))
            
        # Phase 4: High-precision final polish
        try:
            res_f = minimize(objective_slqp, best_vars, method='SLSQP', bounds=bounds_opt,
                             constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if np.min(constraints_slqp(res_f.x)) >= -1e-8:
                best_vars = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Fallback configuration (should rarely be reached)
    if best_vars is None:
        fallback_pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        fallback_pts = np.vstack([fallback_pts, [0.5, 0.5]])
        r_f = np.full(N, 0.09)
        best_vars = centers_to_params(fallback_pts, r_f)
        best_sum = np.sum(r_f)
        
    # Reconstruct centers from optimized parameters
    r_opt = best_vars[:N]
    u_opt = best_vars[N:2*N]
    v_opt = best_vars[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    # Ensure radii are strictly positive
    r_final = np.maximum(r_opt, 0.0)
    
    return centers, r_final, float(best_sum)
