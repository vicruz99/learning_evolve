import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        bounds_list = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_list:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 0.0

def generate_hex(row_counts, seed, rot, scale, jitter):
    """Generates a hexagonal lattice initialization with specified parameters."""
    rng = np.random.RandomState(seed)
    pts = []
    r_e = 0.09
    y = r_e
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_e
        x = r_e + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_e
        y += np.sqrt(3.0) * r_e
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = pts @ M.T
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def generate_force(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    r_curr = np.full(N, 0.05)
    
    for step in range(600):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 30.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.01 * (0.995**step)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
        
        # Update radii estimates for repulsion strength
        dists_mat = np.sqrt(np.sum((pts[:, None, :] - pts[None, :, :])**2, axis=2))
        np.fill_diagonal(dists_mat, np.inf)
        d_p = np.min(dists_mat, axis=1)
        dw = np.minimum(np.minimum(pts[:,0], 1.0-pts[:,0]), np.minimum(pts[:,1], 1.0-pts[:,1]))
        r_curr = 0.8 * np.minimum(dw, d_p/2.0)
        
    return pts

def obj_slqp(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[:N])

def cons_slqp(v):
    """Inequality constraints for SLSQP using (r, u, v) parameterization."""
    r = v[:N]
    u = v[N:2*N]
    vt = v[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + vt * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [7,6,5,5,3], [6,7,5,5,3], [8,6,5,5,2], [7,7,5,5,2],
        [6,6,6,5,3], [8,8,5,5], [9,5,5,5,2], [6,4,6,5,5],
        [5,5,5,5,5,1], [4,5,6,5,6], [7,5,5,5,4], [4,5,5,5,7]
    ]
    
    inits = []
    for pat in patterns:
        for s in range(6):
            inits.append(generate_hex(pat, s, np.random.uniform(-0.25, 0.25), np.random.uniform(0.88, 1.12), 0.015))
            
    for s in range(25):
        inits.append(generate_force(s))
        
    for s in range(10):
        rng = np.random.RandomState(s)
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng.uniform(-0.025, 0.025, pts.shape)
        inits.append(np.clip(pts, 0.02, 0.98))

    # Evaluate all inits with LP to find a strong starting point
    for pts in inits:
        r, s_val = solve_radii_lp(pts)
        if s_val > best_sum:
            best_sum = s_val
            best_c = pts.copy()
            best_r = r.copy()
            
    # Simulated Annealing on centers
    curr_c = best_c.copy()
    curr_sum = best_sum
    temp = 0.025
    step = 0.035
    
    rng = np.random.RandomState(123)
    for it in range(10000):
        idx = rng.randint(N)
        old = curr_c[idx].copy()
        
        current_step = step * (3.0 if rng.rand() < 0.05 else 1.0)
        curr_c[idx] += rng.normal(0, current_step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
        
        new_r, new_sum = solve_radii_lp(curr_c)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-8 and rng.rand() < np.exp(min(0.0, delta / temp))):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = curr_c.copy()
                best_r = new_r.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9992
        step = max(0.0015, step * 0.9996)
        
    # SLSQP Polish with boundary-safe parameterization
    r_init = np.maximum(best_r * 0.995, 1e-5)
    denom = np.clip(1.0 - 2.0 * r_init, 1e-6, 1.0)
    u_init = np.clip((best_c[:, 0] - r_init) / denom, 0.0, 1.0)
    v_init = np.clip((best_c[:, 1] - r_init) / denom, 0.0, 1.0)
    v0 = np.concatenate([r_init, u_init, v_init])
    
    bounds_slqp = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': cons_slqp}
    
    # Try multiple perturbed starts for SLSQP to escape local minima
    for k in range(3):
        vp = v0.copy()
        if k > 0:
            vp[N:3*N] += rng.uniform(-0.015, 0.015, 2*N)
            vp[:N] += rng.uniform(-0.001, 0.001, N)
            vp[:N] = np.clip(vp[:N], 1e-6, 0.5)
            vp[N:3*N] = np.clip(vp[N:3*N], 0.0, 1.0)
            
        try:
            res = minimize(obj_slqp, vp, method='SLSQP', bounds=bounds_slqp, constraints=cons,
                           options={'maxiter': 6000, 'ftol': 1e-13})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    r_opt = res.x[:N]
                    u_opt = res.x[N:2*N]
                    v_opt = res.x[2*N:3*N]
                    best_c = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))
                    best_r = r_opt
                    v0 = res.x.copy()
        except Exception:
            pass
            
    # Final LP check on best centers to ensure radii are truly optimal for them
    r_lp, s_lp = solve_radii_lp(best_c)
    if s_lp > best_sum:
        best_sum = s_lp
        best_r = r_lp
        
    centers = best_c
    radii = np.maximum(best_r, 0.0)
    
    return centers, radii, float(best_sum)
