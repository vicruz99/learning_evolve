import numpy as np
from scipy.optimize import minimize, differential_evolution, linprog

N = 26

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[idx, i] = 1.0
            b_ub[idx] = b
            idx += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.hypot(dx, dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0), -res.fun
    return np.zeros(n), 0.0

def center_obj(x):
    """Objective for center optimization: negative sum of optimal radii."""
    centers = x.reshape(N, 2)
    _, s = solve_radii_lp(centers)
    return -s

def generate_hex_init(seed, pattern=None):
    """Generates a hexagonal lattice initialization with specified row counts."""
    np.random.seed(seed)
    if pattern is None:
        pattern = [5, 5, 5, 5, 6]  # Common efficient pattern for N=26
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, count in enumerate(pattern):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(count):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    pts += np.random.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Global search on centers using Differential Evolution
    bounds_c = [(0.05, 0.95)] * (2 * N)
    
    for seed in range(5):
        try:
            res_de = differential_evolution(
                center_obj, bounds_c, popsize=12, maxiter=250,
                mutation=(0.5, 1.0), recombination=0.8,
                seed=seed, tol=1e-6, polish=True
            )
            if -res_de.fun > best_sum:
                best_sum = -res_de.fun
                best_centers = res_de.x.reshape(N, 2)
                best_radii, _ = solve_radii_lp(best_centers)
        except Exception:
            pass
            
    # Supplement with structured hexagonal initializations
    patterns = [
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5],
        [5, 5, 6, 5, 5], [4, 5, 6, 5, 6], [6, 6, 5, 5, 4]
    ]
    for p in patterns:
        for s in range(3):
            c_init = generate_hex_init(s, p)
            _, s_val = solve_radii_lp(c_init)
            if s_val > best_sum:
                best_sum = s_val
                best_centers = c_init
                best_radii, _ = solve_radii_lp(best_centers)
                
    if best_centers is None:
        best_centers = generate_hex_init(0)
        best_radii, best_sum = solve_radii_lp(best_centers)
        
    # Phase 2: Joint refinement with SLSQP to enforce exact constraints
    def joint_obj(v):
        return -np.sum(v[2::3])
        
    def joint_constr(v):
        cx, cy, r = v[0::3], v[1::3], v[2::3]
        c = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        d2 = dx**2 + dy**2
        rs = r[:, None] + r[None, :]
        mask = np.triu(np.ones((N, N), dtype=bool), k=1)
        c.append(d2[mask] - rs[mask]**2)
        return np.concatenate(c)
        
    bounds_v = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': joint_constr}

    v0[0::3] = best_centers[:, 0]
    v0[1::3] = best_centers[:, 1]
    v0[2::3] = best_radii * 0.998  # Slight shrink for strict interior feasibility

                    if s > best_sum:

                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))

        except Exception:
            pass
            
    # Final safety clip
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
