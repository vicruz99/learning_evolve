import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(radii)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-4

def lp_objective(x_flat):
    """Objective for center optimization: negative of optimal radius sum."""
    centers = x_flat.reshape(N, 2)
    _, s = solve_radii_lp(centers)
    return -s

def make_hex_init(pattern, seed=0, scale=1.0, rot=0.0):
    """Generates a hexagonal lattice initialization with specified parameters."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.1
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def joint_constraints(v):
    """Inequality constraints: g(v) >= 0 for joint center/radius optimization."""
    cx, cy, r = v[0::3], v[1::3], v[2::3]
    c = []
    # Boundary containment
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def run_packing():
    np.random.seed(42)
    bounds_joint = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    
    best_sum = -np.inf
    best_v = None
    
    # Diverse hexagonal row patterns to capture different boundary alignments
    patterns = [
        [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 5, 6, 5, 5], [5, 5, 5, 6, 5], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [5, 6, 4, 6, 5], [4, 6, 5, 6, 5]
    ]
    
    # Phase 1: Structured initializations refined with SLSQP
    for pat in patterns:
        for s in range(4):
            rot = np.random.uniform(-0.15, 0.15)
            scale = np.random.uniform(0.9, 1.1)
            centers_init = make_hex_init(pat, seed=s, scale=scale, rot=rot)
            r_init, _ = solve_radii_lp(centers_init)
            
            v0 = np.zeros(3 * N)
            v0[0::3] = centers_init[:, 0]
            v0[1::3] = centers_init[:, 1]
            v0[2::3] = np.maximum(r_init * 0.995, 1e-6)
            
            try:
                res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                               constraints={'type': 'ineq', 'fun': joint_constraints},
                               options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                if res.success and np.min(joint_constraints(res.x)) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_v = res.x.copy()
            except Exception:
                pass
                
    # Phase 2: Differential Evolution on centers (global search)
    try:
        res_de = differential_evolution(lp_objective, bounds_centers, popsize=20, maxiter=200, 
                                        tol=1e-8, mutation=(0.5, 1.0), recombination=0.9, 
                                        seed=42, polish=False)
        curr_sum = -res_de.fun
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = res_de.x.reshape(N, 2)
            best_radii, _ = solve_radii_lp(best_centers)
            
            v0 = np.zeros(3 * N)
            v0[0::3] = best_centers[:, 0]
            v0[1::3] = best_centers[:, 1]
            v0[2::3] = np.maximum(best_radii * 0.995, 1e-6)
            
            try:
                res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                               constraints={'type': 'ineq', 'fun': joint_constraints},
                               options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(joint_constraints(res.x)) >= -1e-8:
                    best_sum = -res.fun
                    best_v = res.x.copy()
            except Exception:
                pass
    except Exception:
        pass
        
    # Phase 3: Local perturbation refinement to escape local minima
    if best_v is not None:
        rng = np.random.RandomState(123)
        for k in range(15):
            v_p = best_v.copy()
            # Perturb centers more aggressively, radii slightly
            v_p[0::3] += rng.randn(N) * 0.006
            v_p[1::3] += rng.randn(N) * 0.006
            v_p[0::3] = np.clip(v_p[0::3], 0.01, 0.99)
            v_p[1::3] = np.clip(v_p[1::3], 0.01, 0.99)
            
            try:
                res_p = minimize(obj_joint, v_p, method='SLSQP', bounds=bounds_joint,
                                 constraints={'type': 'ineq', 'fun': joint_constraints},
                                 options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
                if res_p.success and np.min(joint_constraints(res_p.x)) >= -1e-8:
                    s = -res_p.fun
                    if s > best_sum:
                        best_sum = s
                        best_v = res_p.x.copy()
            except Exception:
                pass
                
        # Phase 4: High-precision polish
        try:
            res_f = minimize(obj_joint, best_v, method='SLSQP', bounds=bounds_joint,
                             constraints={'type': 'ineq', 'fun': joint_constraints},
                             options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(joint_constraints(res_f.x)) >= -1e-9:
                best_v = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Fallback configuration (should rarely be reached)
    if best_v is None:
        centers_f = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        r_f, _ = solve_radii_lp(centers_f)
        best_v = np.zeros(3 * N)
        best_v[0::3] = centers_f[:, 0]
        best_v[1::3] = centers_f[:, 1]
        best_v[2::3] = np.maximum(r_f, 1e-6)
        best_sum = np.sum(best_v[2::3])
        
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3]
    return centers, radii, float(best_sum)
