N = 26

def solve_radii_lp(centers):

    n = centers.shape[0]
    c = -np.ones(n)
    m = n*4 + n*(n-1)//2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        b_list = [x, 1.0-x, y, 1.0-y]
        for b in b_list:
            A_ub[idx, i] = 1.0
            b_ub[idx] = b
            idx += 1

            A_ub[idx, j] = 1.0

            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 2.6e-5

def obj_centers(centers_flat):
    """Objective for center optimization: minimize negative LP sum of radii."""
    centers = centers_flat.reshape(N, 2)
    _, s = solve_radii_lp(centers)
    return -s

def joint_objective(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def joint_constraints(v):

    cx, cy, r = v[0::3], v[1::3], v[2::3]
    c = []
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    return np.concatenate(c)

def generate_hex_init(seed, rot, scale):

    pts = []
    r_est = 0.09
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ rot_mat.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def local_search_centers(centers, steps=800):
    """Hill-climbing local search on centers maximizing LP sum of radii."""
    n = centers.shape[0]
    best_c = centers.copy()
    _, best_s = solve_radii_lp(best_c)
    step = 0.015
    rng = np.random.RandomState(123)

            best_s = new_s

    return best_c
