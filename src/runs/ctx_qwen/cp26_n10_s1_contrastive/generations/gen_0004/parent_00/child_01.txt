import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def get_pairwise_dists(centers):
    """Compute pairwise Euclidean distances between centers."""
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.sqrt(np.sum(diff**2, axis=2))

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dists = get_pairwise_dists(centers)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 0.0

def objective_centers(x_flat):
    """Objective for center optimization: maximize LP sum of radii."""
    centers = x_flat.reshape(N, 2)
    # Keep centers strictly inside to avoid degenerate LP constraints
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, s = solve_lp(centers)
    return -s

def generate_hex(row_counts, seed, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_inits():
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [5,5,5,5,5,1], [6,5,5,5,5], [5,6,5,5,5], [4,6,6,6,4], 
        [6,6,5,5,4], [7,5,5,5,4], [5,5,5,5,6], [4,5,5,5,5,2]
    ]
    for p in patterns:
        for s in range(4):
            for r in np.linspace(-0.25, 0.25, 5):
                for sc in [0.9, 1.0, 1.1]:
                    inits.append(generate_hex(p, seed=s, rot=r, scale=sc).flatten())
                    
    # Grid-based + center
    for s in range(15):
        rng = np.random.RandomState(s)
        pts = np.array([[0.1 + i*0.2, 0.1 + j*0.2] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng.uniform(-0.04, 0.04, pts.shape)
        inits.append(np.clip(pts, 0.02, 0.98).flatten())
        
    # Pure random
    for s in range(20):
        rng = np.random.RandomState(s)
        pts = rng.rand(N, 2)
        inits.append(pts.flatten())
        
    return inits

def run_packing():
    np.random.seed(42)
    best_sum = -np.inf
    best_v = None

    bounds_full = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    inits = generate_inits()

    # Phase 1: Optimize centers using Powell (derivative-free, handles LP kinks well)
    for x0 in inits:
        try:
            res = minimize(objective_centers, x0, method='Powell',
                           options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
            if not res.success:
                continue
            centers_opt = res.x.reshape(N, 2)
            _, curr_sum = solve_lp(centers_opt)
            if curr_sum > best_sum:
                best_sum = curr_sum
                r_opt, _ = solve_lp(centers_opt)
                best_v = np.zeros(3 * N)
                best_v[0::3] = centers_opt[:, 0]
                best_v[1::3] = centers_opt[:, 1]
                best_v[2::3] = r_opt
        except Exception:
            continue

    # Fallback if all initializations fail
    if best_v is None:
        c0 = np.random.rand(N, 2)
        best_v = np.zeros(3 * N)
        best_v[0::3] = c0[:, 0]
        best_v[1::3] = c0[:, 1]
        best_v[2::3] = np.full(N, 0.01)
        best_sum = 0.26

    def joint_obj(v):
        return -np.sum(v[2::3])

    def joint_con(v):
        cx, cy, r = v[0::3], v[1::3], v[2::3]
        c = []
        c.append(cx - r)
        c.append(1.0 - cx - r)
        c.append(cy - r)
        c.append(1.0 - cy - r)
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        d2 = dx**2 + dy**2
        rs = r[:, None] + r[None, :]
        c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
        return np.concatenate(c)

    cons = {'type': 'ineq', 'fun': joint_con}

    # Phase 2: Joint SLSQP refinement
    try:
        res = minimize(joint_obj, best_v, method='SLSQP', bounds=bounds_full,
                       constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(joint_con(res.x)) >= -1e-8:
            best_v = res.x
            best_sum = -res.fun
    except Exception:
        pass

    # Phase 3: Perturbation & Refinement to escape local minima
    for _ in range(25):
        vp = best_v + np.random.randn(3 * N) * 0.0025
        vp[0::3] = np.clip(vp[0::3], 0.01, 0.99)
        vp[1::3] = np.clip(vp[1::3], 0.01, 0.99)
        vp[2::3] = np.clip(vp[2::3], 1e-6, 0.49)
        try:
            res_p = minimize(joint_obj, vp, method='SLSQP', bounds=bounds_full,
                             constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if res_p.success and np.min(joint_con(res_p.x)) >= -1e-8:
                s = -res_p.fun
                if s > best_sum:
                    best_sum = s
                    best_v = res_p.x
        except Exception:
            pass

    # Phase 4: High-precision polish
    try:
        res_f = minimize(joint_obj, best_v, method='SLSQP', bounds=bounds_full,
                         constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(joint_con(res_f.x)) >= -1e-9:
            best_v = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass

    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3]
    radii = np.maximum(radii, 0.0)
    
    # Final minor adjustment to guarantee strict validity against numerical noise
    for i in range(N):
        cx, cy = centers[i]
        r = radii[i]
        mx = min(cx, 1.0 - cx, cy, 1.0 - cy)
        if r > mx + 1e-12:
            radii[i] = mx - 1e-13
        for j in range(i + 1, N):
            d = np.sqrt((cx - centers[j, 0])**2 + (cy - centers[j, 1])**2)
            if r + radii[j] > d + 1e-12:
                overlap = (r + radii[j] - d) * 0.5
                radii[i] -= overlap
                radii[j] -= overlap
    radii = np.maximum(radii, 1e-7)
    
    return centers, radii, float(np.sum(radii))
