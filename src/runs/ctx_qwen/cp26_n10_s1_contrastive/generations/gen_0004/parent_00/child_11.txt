import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[0:N])

def constraint_func(vars_vec):
    """
    Computes inequality constraints g(vars_vec) >= 0.
    Uses parameterization to automatically satisfy boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = vars_vec[0:N]
    u = vars_vec[N:2*N]
    v = vars_vec[2*N:3*N]
    
    # Parameterization maps u, v in [0,1] to x, y such that circle is strictly inside [0,1]^2
    denom = 1.0 - 2.0 * r
    x = r + u * denom
    y = r + v * denom
    
    # Pairwise squared distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    # Squared sum of radii
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX]

def force_init(seed, steps=500):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    r_curr = np.full(N, 0.04)
    
    for step in range(steps):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        repulsion_mag = 1.0 / dists**2
        np.fill_diagonal(repulsion_mag, 0.0)
        forces = np.sum(repulsion_mag[:, :, None] * diff / dists[:, :, None], axis=1)
        
        # Wall repulsion to keep points inside
        for d in range(2):
            forces[:, d] += 10.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 10.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.008 * (0.995**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        # Adaptively update radii during simulation
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            d_pair = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            d_pair = max(d_pair, 1e-6)
            r_curr[i] = 0.85 * min(d_wall, d_pair/2.0)
            
    # Final feasible radii computation
    r_final = np.zeros(N)
    for i in range(N):
        d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        dists = np.linalg.norm(pts[i] - pts, axis=1)
        dists[i] = np.inf
        d_pair = np.min(dists)
        r_final[i] = 0.95 * min(d_wall, d_pair/2.0)
        
    denom = np.clip(1.0 - 2.0 * r_final, 1e-6, 1.0)
    u = np.clip((pts[:, 0] - r_final) / denom, 0.0, 1.0)
    v = np.clip((pts[:, 1] - r_final) / denom, 0.0, 1.0)
    
    vars0 = np.empty(3 * N)
    vars0[0:N] = r_final
    vars0[N:2*N] = u
    vars0[2*N:3*N] = v
    return vars0

def hex_init(seed, row_counts, rotation=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization with rotation and perturbation."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.09
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rotation != 0.0:
        c, s = np.cos(rotation), np.sin(rotation)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ rot_mat + 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    r_vals = np.zeros(N)
    for i in range(N):
        d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        dists = np.linalg.norm(pts[i] - pts, axis=1)
        dists[i] = np.inf
        d_pair = np.min(dists)
        r_vals[i] = 0.95 * min(d_wall, d_pair/2.0)
        
    denom = np.clip(1.0 - 2.0 * r_vals, 1e-6, 1.0)
    u = np.clip((pts[:, 0] - r_vals) / denom, 0.0, 1.0)
    v = np.clip((pts[:, 1] - r_vals) / denom, 0.0, 1.0)
    
    vars0 = np.empty(3 * N)
    vars0[0:N] = r_vals
    vars0[N:2*N] = u
    vars0[2*N:3*N] = v
    return vars0

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_sum = -1.0
    rng = np.random.RandomState(42)
    
    inits = []
    
    # 1. Force-based initializations (excellent at finding tight packings)
    for s in range(30):
        inits.append(force_init(seed=s))
        
    # 2. Hexagonal lattice variations with random rotations
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [5, 5, 5, 5, 6], [6, 4, 6, 6, 4], [7, 5, 5, 5, 4],
        [4, 5, 6, 5, 6], [5, 6, 4, 6, 5], [6, 6, 5, 5, 4],
        [7, 6, 5, 5, 3], [5, 5, 6, 6, 4], [4, 6, 6, 5, 5],
        [5, 4, 6, 6, 5], [6, 5, 5, 5, 5]
    ]
    for pat in patterns:
        for s in range(5):
            rot = rng.uniform(-0.25, 0.25)
            sc = rng.uniform(0.85, 1.15)
            inits.append(hex_init(seed=s, row_counts=pat, rotation=rot, scale=sc))
            
    # Phase 1: Broad search from diverse initializations
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                cons_val = constraint_func(res.x)
                if np.min(cons_val) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Local perturbation refinement to escape local minima
    if best_vars is not None:
        for k in range(40):
            x0 = best_vars.copy()
            # Perturb positions more aggressively to explore topology changes
            x0[N:2*N] += rng.randn(N) * 0.06
            x0[2*N:3*N] += rng.randn(N) * 0.06
            x0[0:N] += rng.randn(N) * 0.003
            
            x0[0:N] = np.clip(x0[0:N], 1e-6, 0.5)
            x0[N:3*N] = np.clip(x0[N:3*N], 0.0, 1.0)
            
            # Occasionally permute variables to break solver symmetry
            if k % 5 == 0:
                perm = rng.permutation(3 * N)
                x0 = x0[perm]
                
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                if res.success:
                    cons_val = constraint_func(res.x)
                    if np.min(cons_val) >= -1e-8:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_vars = res.x.copy()
            except Exception:
                continue
                
    # Phase 3: High-precision polish
    if best_vars is not None:
        try:
            res = minimize(objective, best_vars, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if np.min(constraint_func(res.x)) >= -1e-9:
                best_vars = res.x
                best_sum = -res.fun
        except Exception:
            pass
            
    # Fallback configuration (should not be reached given robust initialization)
    if best_vars is None:
        best_vars = force_init(seed=0)
        best_sum = -objective(best_vars)

    # Reconstruct centers from optimized parameters
    r_opt = best_vars[0:N]
    u_opt = best_vars[N:2*N]
    v_opt = best_vars[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    return centers, r_opt, float(best_sum)
