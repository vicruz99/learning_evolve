import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-4

def joint_objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def joint_constraints(v):
    """Inequality constraints: g(v) >= 0."""
    cx, cy, r = v[0::3], v[1::3], v[2::3]
    c = []
    # Boundary containment
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def generate_hex_init(row_counts, seed, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.09
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(row_counts[row % len(row_counts)] if len(row_counts) > row else 5):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_grid_init(seed, jitter=0.03):
    """Generates a perturbed grid initialization."""
    rng = np.random.RandomState(seed)
    pts = np.array([[0.08 + 0.2*i, 0.08 + 0.2*j] for i in range(5) for j in range(5)])
    # Add 26th point in a gap or center
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_corner_bias_init(seed):
    """Generates initialization biased towards corners and edges."""
    rng = np.random.RandomState(seed)
    pts = []
    # 4 corners
    for x in [0.12, 0.88]:
        for y in [0.12, 0.88]:
            pts.append([x + rng.uniform(-0.02, 0.02), y + rng.uniform(-0.02, 0.02)])
    # Fill remaining with hex pattern in center
    rem = N - len(pts)
    r_est = 0.08
    y = 0.5 - rem * 0.04
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = 0.5 - (row_counts_helper(row, rem) * r_est)
        for _ in range(row_counts_helper(row, rem)):
            if len(pts) >= N: break
            pts.append([x + rng.uniform(-0.01, 0.01), y + rng.uniform(-0.01, 0.01)])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    return np.clip(np.array(pts), 0.02, 0.98)

def row_counts_helper(row, rem):
    """Simple helper to distribute remaining points."""
    counts = [5, 5, 5, 5, 4]
    return counts[row % len(counts)]

def optimize_centers_lp(centers, rng, steps=300):
    """Local search on centers maximizing sum of radii via LP."""
    curr_centers = centers.copy()
    _, best_sum = solve_radii_lp(curr_centers)
    
    step_size = 0.04
    for step in range(steps):
        # Adaptive step size
        if step > 0 and step % 100 == 0:
            step_size *= 0.8
            
        # Pick a random circle to move
        idx = rng.randint(0, N)
        direction = rng.randn(2)
        direction /= np.linalg.norm(direction)
        
        new_centers = curr_centers.copy()
        new_centers[idx] += direction * step_size
        new_centers = np.clip(new_centers, 0.01, 0.99)
        
        _, new_sum = solve_radii_lp(new_centers)
        if new_sum > best_sum:
            curr_centers = new_centers
            best_sum = new_sum
            
    return curr_centers, best_sum
