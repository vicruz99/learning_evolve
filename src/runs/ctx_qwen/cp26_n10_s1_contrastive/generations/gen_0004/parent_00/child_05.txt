N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r when u,v in [0,1]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)

    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2

    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_params(c, r):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    d = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((c[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((c[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def get_safe_radii(centers):

    n = centers.shape[0]
