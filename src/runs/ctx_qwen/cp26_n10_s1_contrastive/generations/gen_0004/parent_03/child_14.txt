import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    c = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if d * 0.5 < mx:
                    mx = d * 0.5
        r[i] = max(1e-4, mx * 0.9)
    return r

def hex_init(pat, rot=0.0, sc=1.0, jitter=0.02, seed=0):
    """Generate a hexagonal lattice initialization with specified parameters."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.1
    y = r_est
    for r_idx, count in enumerate(pat):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(count):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * sc + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
        
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Generate a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    for _ in range(400):
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        f = np.sum((1.0/dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        
        wall = 15.0
        for d in range(2):
            f[:, d] += wall * (pts[:, d] < 0.12) * (0.12 - pts[:, d])
            f[:, d] -= wall * (pts[:, d] > 0.88) * (pts[:, d] - 0.88)
            
        pts += 0.004 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -np.inf
    best_x = None
    rng = np.random.RandomState(42)
    
    inits = []
    pats = [
        [6,5,5,5,5], [5,6,5,5,5], [5,5,6,5,5], [5,5,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,6,5,6,5], [7,5,5,5,4], [5,7,5,5,4],
        [6,6,5,5,4], [5,5,7,5,4], [4,5,6,5,6], [6,5,6,5,4], [5,5,5,7,4]
    ]
    for p in pats:
        for rot in [0.0, 0.08, -0.08, 0.15, -0.15]:
            for sc in [0.9, 1.0, 1.1]:
                inits.append(hex_init(p, rot=rot, sc=sc, jitter=0.02, seed=len(inits)))
                
    for s in range(15):
        inits.append(force_init(seed=s))
        
    for s in range(10):
        pts = np.sort(np.random.rand(N, 2), axis=0)

        inits.append(np.clip(pts, 0.05, 0.95))

    # Phase 1: Broad search from diverse initializations

        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r0
        try:

                           options={'maxiter': 2500, 'ftol': 1e-12, 'disp': False})
            c_val = constraints(res.x)
            if np.min(c_val) >= -1e-7:
                s_val = -res.fun

                    best_sum = s_val
                    best_x = res.x.copy()

            continue

                pass

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = np.maximum(best_x[2::3], 1e-6)
    return centers, radii, float(best_sum)
