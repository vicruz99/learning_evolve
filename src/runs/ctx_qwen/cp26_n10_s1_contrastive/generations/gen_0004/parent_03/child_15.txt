import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def constraints(vars_vec):
    """
    Inequality constraints: pairwise non-overlap.
    Boundary containment is implicitly satisfied by the (r, u, v) parameterization.
    Returns array of values that must be >= 0.
    """
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    # Implicit boundary mapping: x, y are guaranteed in [r, 1-r]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    # Pairwise squared distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    
    # Squared sum of radii
    rs = r[:, None] + r[None, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = 1.0 - 2.0 * r
    denom = np.clip(denom, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def safe_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i]-centers[j])**2))
                mx = min(mx, d * 0.5)
        r[i] = max(1e-4, mx * 0.8)
    return r

def hex_init(seed, pattern=None, rot=0.0, scale=1.0):
    """Generate hexagonal lattice initialization with rotation and scaling."""
    rng = np.random.RandomState(seed)
    if pattern is None:
        pattern = [6, 5, 6, 5, 4]
        
    pts = []
    r_est = 0.09
    y = r_est
    for r_idx, count in enumerate(pattern):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(count):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts -= 0.5
    pts *= scale
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = pts @ rot_mat.T
        
    pts += 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def force_init(seed):
    """Force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    
    for _ in range(200):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        
        # Repulsion force
        f += np.sum((1.0/dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        
        # Wall repulsion
        for d in range(2):
            f[:, d] += 5.0 * (pts[:, d] < 0.1) * (0.1 - pts[:, d])
            f[:, d] -= 5.0 * (pts[:, d] > 0.9) * (pts[:, d] - 0.9)
            
        pts += 0.003 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    bounds = [(1e-5, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -np.inf
    best_params = None
    rng = np.random.RandomState(42)
    
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5],
        [7,5,5,5,4], [5,7,5,5,4], [5,5,7,5,4], [5,5,5,7,4], [8,5,5,5,3],
        [6,6,5,5,4], [5,5,6,6,4], [7,6,5,4,4], [6,7,6,4,3], [5,8,5,4,4]
    ]
    
    # Generate diverse hexagonal initializations
    for p in patterns:
        for s in range(6):
            rot = rng.uniform(-0.2, 0.2)
            sc = rng.uniform(0.88, 1.12)
            pts = hex_init(s, pattern=p, rot=rot, scale=sc)
            r0 = safe_radii(pts)
            inits.append(make_params(pts, r0))
            
    # Force-directed initializations
    for s in range(15):
        pts = force_init(s)
        r0 = safe_radii(pts)
        inits.append(make_params(pts, r0))
        
    # Random initializations
    for s in range(10):
        rng2 = np.random.RandomState(s)
        pts = rng2.rand(N, 2)
        pts = np.clip(pts, 0.1, 0.9)
        r0 = safe_radii(pts)
        inits.append(make_params(pts, r0))
        
    rng.shuffle(inits)
    
    # Phase 1: Broad search from diverse initializations
    for x0 in inits:

                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            pass

            pass

    # Fallback configuration if optimization fails completely
    if best_params is None:
        fallback_centers = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        r_f = safe_radii(fallback_centers)
        best_params = make_params(fallback_centers, r_f)
        best_sum = np.sum(r_f)

    # Reconstruct physical coordinates
    r_opt = best_params[:N]
    u_opt = best_params[N:2*N]
    v_opt = best_params[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
