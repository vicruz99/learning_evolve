`x = r + u * (1 - 2r)`
`y = r + v * (1 - 2r)`
where `r ∈ [0, 0.5]` and `u, v ∈ [0, 1]`. This guarantees `r ≤ x, y ≤ 1-r` without explicit constraints, leaving only the pairwise non-overlap constraints for SLSQP to handle. This formulation is known to be numerically stable and converges faster.

Initialization is critical for escaping local minima. I will generate a diverse set of starting configurations:
1. Hexagonal lattices with systematic variations in rotation, scale, and jitter.
2. Force-directed layouts that simulate Coulomb repulsion and wall forces to organically spread points.
3. Perturbed grids and purely random points.

The best result will then undergo aggressive local perturbation refinement (shuffling positions, slightly shrinking radii to maintain feasibility, and re-optimizing) to escape shallow local minima. import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(params):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(params[:N])

def constraints(params):
    """
    Inequality constraints: pairwise non-overlap.
    Boundary constraints are handled automatically by parameterization.
    Returns array of values that must be >= 0.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization ensures r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    d = 1.0 - 2.0 * r
    d = np.clip(d, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def safe_radii(centers):
    """Compute strictly feasible initial radii based on boundaries and neighbors."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i]-centers[j])**2))
                mx = min(mx, d * 0.5)
        r[i] = max(1e-4, mx * 0.9)
    return r

def generate_inits():
    """Generates a diverse set of initial configurations."""
    inits = []
    rng = np.random.RandomState(42)
    
    # 1. Hexagonal lattices with various rotations and scales
    for rot in np.linspace(-0.25, 0.25, 11):
        for sc in np.linspace(0.80, 1.20, 7):
            pts = []
            r_est = 0.1
            y = r_est
            row = 0
            while len(pts) < N:
                shift = (row % 2) * r_est
                x = r_est + shift
                while x <= 1.0 - r_est and len(pts) < N:
                    pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * sc + 0.5
            if abs(rot) > 1e-5:
                c, s = np.cos(rot), np.sin(rot)
                R = np.array([[c, -s], [s, c]])
                pts = pts @ R.T
                pts = np.clip(pts, 0.02, 0.98)
            else:
                pts = np.clip(pts, 0.02, 0.98)
            pts += rng.uniform(-0.015, 0.015, pts.shape)
            pts = np.clip(pts, 0.02, 0.98)
            r0 = safe_radii(pts)
            inits.append(make_params(pts, r0))
            
    # 2. Force-directed layouts
    for seed in range(12):
        rng2 = np.random.RandomState(seed)
        pts = rng2.rand(N, 2) * 0.6 + 0.2
        for _ in range(250):
            forces = np.zeros_like(pts)
            diff = pts[:, None, :] - pts[None, :, :]
            dist = np.sqrt(np.sum(diff**2, axis=2))
            dist = np.maximum(dist, 1e-4)
            rep = 1.0 / (dist**2)
            np.fill_diagonal(rep, 0.0)
            forces += np.sum(rep[:, :, None] * diff / dist[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 30.0 * np.maximum(0, 0.1 - pts[:, d])
                forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - 0.9)
            pts += 0.0025 * forces
            pts = np.clip(pts, 0.05, 0.95)
        r0 = safe_radii(pts)
        inits.append(make_params(pts, r0))
        
    # 3. Random perturbed grids & random points
    for seed in range(12):
        rng3 = np.random.RandomState(seed)
        if seed < 8:
            pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
            pts = np.vstack([pts, [0.5, 0.5]])
            pts += rng3.uniform(-0.05, 0.05, (N, 2))
        else:
            pts = rng3.rand(N, 2)
        pts = np.clip(pts, 0.05, 0.95)
        r0 = safe_radii(pts)
        inits.append(make_params(pts, r0))
        
    return inits

def run_packing():
    np.random.seed(42)
    bounds = [(1e-5, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    
    inits = generate_inits()
    np.random.shuffle(inits)
    
    best_sum = -np.inf
    best_params = None
    
    # Phase 1: Broad search from diverse initializations
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 2500, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Local perturbation refinement to escape local minima
    if best_params is not None:
        for k in range(60):
            x_p = best_params.copy()
            # Perturb positions significantly, shrink radii slightly to maintain feasibility
            x_p[N:3*N] += np.random.uniform(-0.015, 0.015, 2*N)
            x_p[:N] *= np.random.uniform(0.96, 1.0, N)
            x_p = np.clip(x_p, [1e-5] + [0.0]*2*N, [0.5] + [1.0]*2*N)
            
            try:
                res = minimize(objective, x_p, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(constraints(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
            except Exception:
                continue
                
        # Phase 3: High-precision polish
        try:
            res_f = minimize(objective, best_params, method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(constraints(res_f.x)) >= -1e-8:
                best_params = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Fallback configuration (should rarely be needed)
    if best_params is None:
        centers_f = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                     np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        centers_f[25] = [0.5, 0.5]
        r_f = safe_radii(centers_f)
        best_params = make_params(centers_f, r_f)
        best_sum = np.sum(r_f)
        
    # Reconstruct centers from optimized parameters
    r_opt = best_params[:N]
    u_opt = best_params[N:2*N]
    v_opt = best_params[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack([x_opt, y_opt])
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
