   $x_i = r_i + u_i(1 - 2r_i)$, $y_i = r_i + v_i(1 - 2r_i)$ where $u_i, v_i \in [0, 1]$.
   This automatically guarantees $r_i \le x_i, y_i \le 1-r_i$, eliminating boundary constraints from the solver's burden. We only need to enforce pairwise non-overlap: $d_{ij}^2 \ge (r_i + r_j)^2$. This drastically reduces constraint complexity and improves SLSQP convergence.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints(params):
    """Pairwise non-overlap constraints using parameterized coordinates."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees boundary containment
    term = 1.0 - 2.0 * r
    x = r + u * term
    y = r + v * term
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def get_params(centers, radii):
    """Convert physical centers/radii to optimization parameters (r, u, v)."""
    r = np.clip(radii, 1e-6, 0.4999)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def compute_radii_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    for i in range(n):
        for j in range(i+1, n):
            d = dists[i, j]
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds_r = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def hex_init(row_counts, rotation=0.0, scale=1.0, jitter=0.02, seed=0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, count in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(count):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rotation != 0.0:
        c, s = np.cos(rotation), np.sin(rotation)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ rot_mat.T + 0.5
        
    pts += rng.uniform(-jitter, jitter, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def force_init(seed, steps=400):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    r_curr = np.full(N, 0.04)
    
    for step in range(steps):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 10.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 10.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.006 * (0.996**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        for i in range(N):
            dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            r_curr[i] = 0.85 * min(dw, dp/2.0)
            
    return pts, r_curr

def run_packing():
    bounds = [(1e-6, 0.4999), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_params = None
    best_sum = -np.inf
    
    inits = []
    patterns = [
        [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 5, 6, 5, 5], [5, 5, 5, 6, 5], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [5, 6, 4, 6, 5], [4, 6, 5, 6, 5],
        [7, 6, 5, 4, 4], [5, 7, 6, 5, 3], [8, 6, 5, 4, 3], [6, 7, 6, 5, 2],
        [6, 6, 6, 5, 3], [5, 5, 7, 5, 4], [7, 5, 7, 4, 3], [6, 6, 5, 6, 3],
        [5, 5, 5, 5, 5, 1], [6, 5, 5, 5, 4, 1]
    ]
    
    # Generate hexagonal initializations
    for pat in patterns:
        for rot in [0.0, 0.05, -0.05, 0.1, -0.1]:
            for sc in [0.9, 1.0, 1.1]:
                centers = hex_init(pat, rotation=rot, scale=sc, jitter=0.025, seed=len(inits))
                r, _ = compute_radii_lp(centers)
                inits.append(get_params(centers, r))
                
    # Generate force-directed initializations
    for s in range(12):
        centers, r = force_init(s)
        inits.append(get_params(centers, r))
        
    rng = np.random.RandomState(42)
    rng.shuffle(inits)
    
    # Phase 1: Broad search from diverse initializations
    for x0 in inits[:70]:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-8:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            continue
            
    if best_params is not None:
        # Phase 2: Local perturbation refinement to escape local minima
        for k in range(45):
            x_p = best_params.copy()
            # Perturb positions more aggressively than radii
            x_p[N:3*N] += np.random.randn(2*N) * 0.012
            x_p[:N] += np.random.randn(N) * 0.002
            x_p[:N] = np.clip(x_p[:N], 1e-5, 0.4999)
            x_p[N:3*N] = np.clip(x_p[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(objective, x_p, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(constraints(res.x)) >= -1e-8:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
            except Exception:
                pass
                
        # Phase 3: High-precision polish
        try:
            res_f = minimize(objective, best_params, method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 10000, 'ftol': 1e-15, 'disp': False})
            if res_f.success and np.min(constraints(res_f.x)) >= -1e-9:
                best_params = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass

    # Fallback configuration if optimization fails completely
    if best_params is None:
        centers = np.array([[0.1+i*0.2, 0.1+j*0.2] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        r, _ = compute_radii_lp(centers)
        best_params = get_params(centers, r)
        best_sum = np.sum(r)
        
    # Reconstruct physical coordinates
    r_opt = best_params[:N]
    u_opt = best_params[N:2*N]
    v_opt = best_params[2*N:3*N]
    term = 1.0 - 2.0 * r_opt
    x_opt = r_opt + u_opt * term
    y_opt = r_opt + v_opt * term
    centers_out = np.column_stack((x_opt, y_opt))
    radii_out = r_opt
    return centers_out, radii_out, float(best_sum)
