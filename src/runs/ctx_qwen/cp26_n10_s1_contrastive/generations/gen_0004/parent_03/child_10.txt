import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(params):
    """Objective: maximize sum of radii <=> minimize negative sum."""
    return -np.sum(params[:N])

def constraint_func(params):
    """
    Computes inequality constraints g(params) >= 0.
    Boundary constraints are automatically satisfied by the parameterization.
    Only pairwise non-overlap constraints are enforced.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization maps u, v in [0,1] to x, y such that circle is strictly inside [0,1]^2
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    # Pairwise squared distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    
    # Squared sum of radii
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def get_params_from_centers_r(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = 1.0 - 2.0 * r
    denom = np.clip(denom, 1e-8, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def hex_init(seed, pattern, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization with specified parameters."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(pattern):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts = pts - pts.mean(axis=0) + 0.5
        
    pts += rng.uniform(-0.025, 0.025, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    # Compute safe radii
    r_vals = np.full(N, 1e-6)
    for i in range(N):
        d_b = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        d_p = np.min(np.linalg.norm(pts[i] - pts, axis=1))
        d_p = max(d_p, 1e-6)
        r_vals[i] = 0.9 * min(d_b, d_p / 2.0)
        
    return get_params_from_centers_r(pts, r_vals)

def force_init(seed):
    """Force-directed layout initialization to spread points evenly."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        f += np.sum((1.0/dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        
        wall = 50.0
        for d in range(2):
            f[:, d] += wall * (pts[:, d] < 0.1) * (0.1 - pts[:, d])
            f[:, d] -= wall * (pts[:, d] > 0.9) * (pts[:, d] - 0.9)
            
        pts += 0.005 * f
        pts = np.clip(pts, 0.02, 0.98)
        
    r_vals = np.full(N, 1e-6)
    for i in range(N):
        d_b = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        d_p = np.min(np.linalg.norm(pts[i] - pts, axis=1))
        d_p = max(d_p, 1e-6)
        r_vals[i] = 0.85 * min(d_b, d_p / 2.0)
        
    return get_params_from_centers_r(pts, r_vals)

def run_packing():
    bounds = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -np.inf
    best_params = None
    rng = np.random.RandomState(42)
    
    inits = []
    
    # Diverse hexagonal patterns (sum to 26)
    patterns = [
        [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 5, 6, 5, 5], [5, 5, 5, 6, 5], [5, 5, 5, 5, 6],

    ]
    
    for pat in patterns:

            d_p = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            d_p = max(d_p, 1e-6)
            r_vals[i] = 0.8 * min(d_b, d_p / 2.0)
        inits.append(get_params_from_centers_r(pts, r_vals))
        
    rng.shuffle(inits)
    
    # Phase 1: Broad search from diverse initializations
    for x0 in inits[:100]:
        try:

            if res.success:
                if np.min(constraint_func(res.x)) >= -1e-7:
                    s = -res.fun

        except Exception:
            pass

                    if s > best_sum:

            except Exception:

        # Phase 3: High-precision polish

            pass

    # Fallback configuration if optimization fails completely
    if best_params is None:
        pts = np.array([[0.1+i*0.2, 0.1+j*0.2] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        r_vals = np.full(N, 0.09)

    v = best_params[2*N:3*N]

    y = r + v * (1.0 - 2.0 * r)
    centers = np.column_stack((x, y))
    radii = np.maximum(r, 0.0)
    return centers, radii, float(best_sum)
