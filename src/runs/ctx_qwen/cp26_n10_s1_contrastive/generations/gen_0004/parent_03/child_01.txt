import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(vars_vec):
    """Objective: maximize sum of radii <=> minimize negative sum."""
    return -np.sum(vars_vec[0::3])

def constraint_func(vars_vec):
    """
    Inequality constraints: pairwise non-overlap.
    Boundary constraints are handled automatically by parameterization.
    Returns array of values that must be >= 0.
    """
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    # Parameterization ensures r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def get_params_from_centers_radii(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def get_radii_for_centers(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        dists = np.linalg.norm(centers[i] - centers, axis=1)
        dists[i] = np.inf
        d_pair = np.min(dists)
        r[i] = 0.95 * min(d_wall, d_pair / 2.0)
    return np.maximum(r, 1e-5)

def hex_init(pattern, seed=0, rot=0.0, scale=1.0, jitter=0.02):
    """Generates a hexagonal lattice initialization with specified parameters."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.1
    y = r_est
    for r_idx, count in enumerate(pattern):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(count):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        mat = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ mat.T + 0.5
        
    pts += rng.uniform(-jitter, jitter, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def force_init(seed=0):
    """Force-directed layout initialization to spread points evenly."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    for _ in range(400):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        f += np.sum((1.0/dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        
        wall = 30.0
        f[:, 0] += wall * np.maximum(0, 0.1 - pts[:, 0])
        f[:, 0] -= wall * np.maximum(0, pts[:, 0] - 0.9)
        f[:, 1] += wall * np.maximum(0, 0.1 - pts[:, 1])
        f[:, 1] -= wall * np.maximum(0, pts[:, 1] - 0.9)
        
        pts += 0.005 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -np.inf
    best_vars = None
    
    # Diverse hexagonal row count patterns
    patterns = [
        [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 5, 6, 5, 5], [5, 5, 5, 6, 5], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [5, 6, 4, 6, 5], [4, 6, 5, 6, 5],
        [7, 6, 5, 4, 4], [5, 7, 6, 5, 3], [8, 6, 5, 4, 3], [6, 7, 6, 5, 2],
        [6, 6, 6, 5, 3], [5, 5, 7, 5, 4], [7, 5, 7, 4, 3], [7, 7, 6, 3, 3],
        [6, 6, 5, 5, 4], [5, 6, 6, 5, 4], [4, 5, 6, 6, 5], [5, 5, 5, 6, 5]
    ]
    
    configs = []
    for pat in patterns:
        for rot in np.linspace(-0.2, 0.2, 9):
            for sc in [0.9, 0.95, 1.0, 1.05, 1.1]:
                c = hex_init(pat, seed=len(configs), rot=rot, scale=sc, jitter=0.025)
                r = get_radii_for_centers(c)
                configs.append(get_params_from_centers_radii(c, r))
                
    for s in range(25):
        c = force_init(seed=s)
        r = get_radii_for_centers(c)
        configs.append(get_params_from_centers_radii(c, r))
        
    # Shuffle to diversify search order
    rng = np.random.RandomState(42)
    idx = np.arange(len(configs))
    rng.shuffle(idx)
    configs = [configs[i] for i in idx]
    
    # Phase 1: Broad search from diverse initializations
    for x0 in configs[:90]:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 2500, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = constraint_func(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Local perturbation refinement to escape local minima & symmetry
    if best_vars is not None:
        for k in range(60):
            x_p = best_vars.copy()
            # Perturb positions more aggressively to explore topology changes
            x_p[1::3] += np.random.randn(N) * 0.012
            x_p[2::3] += np.random.randn(N) * 0.012
            x_p[0::3] += np.random.randn(N) * 0.003
            
            x_p[0::3] = np.clip(x_p[0::3], 1e-6, 0.5)
            x_p[1::3] = np.clip(x_p[1::3], 0.0, 1.0)
            x_p[2::3] = np.clip(x_p[2::3], 0.0, 1.0)
            
            try:
                res = minimize(objective, x_p, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(constraint_func(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_vars = res.x.copy()
            except Exception:
                pass
                
        # Phase 3: High-precision polish
        try:
            res_f = minimize(objective, best_vars, method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(constraint_func(res_f.x)) >= -1e-8:
                best_vars = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Fallback configuration (should not be reached)
    if best_vars is None:
        c_f = np.array([[0.1+i*0.2, 0.1+j*0.2] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        r_f = get_radii_for_centers(c_f)
        best_vars = get_params_from_centers_radii(c_f, r_f)
        best_sum = np.sum(r_f)
        
    # Reconstruct centers from optimized parameters
    r_opt = best_vars[0::3]
    u_opt = best_vars[1::3]
    v_opt = best_vars[2::3]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
