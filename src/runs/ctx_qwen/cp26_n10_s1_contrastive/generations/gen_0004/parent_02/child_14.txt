import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii under non-overlap and boundary constraints."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 2.6e-4

def obj_params(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[:N])

def cons_params(v):
    """Inequality constraints for (r, u, w) parameterization: pairwise non-overlap."""
    r = v[:N]
    u = v[N:2*N]
    w = v[2*N:3*N]
    
    scale = np.clip(1.0 - 2.0 * r, 1e-9, 1.0)
    x = r + u * scale
    y = r + w * scale
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    rs2 = rs**2
    
    return dist2[I_IDX, J_IDX] - rs2[I_IDX, J_IDX]

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    scale = np.clip(1.0 - 2.0 * r, 1e-9, 1.0)
    u = np.clip((centers[:, 0] - r) / scale, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / scale, 0.0, 1.0)
    return np.concatenate([r, u, w])

def make_hex_init(row_counts, scale, rot, seed):
    """Generate a hexagonal lattice initialization with specified row counts."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, count in enumerate(row_counts):
        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_v = None
    best_sum = -np.inf
    bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': cons_params}
    
    inits = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5], 
        [5, 5, 5, 5, 6], [6, 4, 6, 5, 5], [5, 6, 4, 6, 5],
        [7, 5, 5, 5, 4], [4, 5, 5, 5, 7], [6, 6, 5, 5, 4]
    ]
    
    # Diverse hexagonal initializations with rotations and scales
    for pat in patterns:
        for s in range(3):
            for rot in np.linspace(-0.12, 0.12, 4):
                inits.append(make_hex_init(pat, scale=0.95 + 0.05 * (s % 3), rot=rot, seed=s * 100 + int(rot * 100)))

    # Add random sorted layouts
    rng_rand = np.random.RandomState(99)
    for _ in range(10):
        pts = rng_rand.rand(N, 2)
        pts = pts[pts[:, 1].argsort()]
        pts += rng_rand.uniform(-0.02, 0.02, pts.shape)
        pts = np.clip(pts, 0.03, 0.97)
        inits.append(pts)

    # Phase 1: Broad SLSQP search
    for pts in inits:
        r0, _ = solve_lp_radii(pts)
        r0 = np.maximum(r0 * 0.99, 1e-5)
        v0 = make_params(pts, r0)
        
        try:
            res = minimize(obj_params, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 2500, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = cons_params(res.x)
                if np.min(c_val) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_v = res.x.copy()
        except Exception:
            continue

    # Fallback if optimization fails
    if best_v is None:
        pts_f = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                 np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        r_f, _ = solve_lp_radii(pts_f)
        best_v = make_params(pts_f, r_f)
        best_sum = np.sum(r_f)

    # Phase 2: Center-based Hill Climbing with LP evaluation
    # This escapes local minima where joint (x,y,r) optimization stalls
    r_best = best_v[:N]
    u_best = best_v[N:2*N]
    w_best = best_v[2*N:3*N]
    scale = np.clip(1.0 - 2.0 * r_best, 1e-9, 1.0)
    c_best = np.column_stack([r_best + u_best * scale, r_best + w_best * scale])
    
    current_centers = c_best.copy()
    _, current_sum = solve_lp_radii(current_centers)
    
    rng_hc = np.random.RandomState(42)
    step = 0.025
    
    for _ in range(3000):
        idx = rng_hc.integers(N)
        direction = rng_hc.randn(2)
        direction /= np.linalg.norm(direction)
        
        new_centers = current_centers.copy()
        new_centers[idx] += step * direction
        new_centers[idx] = np.clip(new_centers[idx], 0.015, 0.985)
        
        _, new_sum = solve_lp_radii(new_centers)
        if new_sum > current_sum + 1e-8:
            current_centers = new_centers
            _, current_sum = solve_lp_radii(current_centers)
            if rng_hc.random() < 0.05:
                step *= 0.95
                
    if current_sum > best_sum:
        best_sum = current_sum
        best_radii, _ = solve_lp_radii(current_centers)
        best_v = make_params(current_centers, best_radii)

    # Phase 3: High-precision SLSQP polish
    v_polish = best_v.copy()
    v_polish[:N] *= 0.998
    v_polish[N:3*N] += rng_hc.uniform(-0.001, 0.001, 2 * N)
    v_polish[N:3*N] = np.clip(v_polish[N:3*N], 0.0, 1.0)
    
    try:
        res_final = minimize(obj_params, v_polish, method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
        if res_final.success:
            c_val = cons_params(res_final.x)
            if np.min(c_val) >= -1e-8:
                s_val = -res_final.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_v = res_final.x.copy()
    except Exception:
        pass
        
    # Reconstruct final centers and radii
    r_final = best_v[:N]
    u_final = best_v[N:2*N]
    w_final = best_v[2*N:3*N]
    scale_final = np.clip(1.0 - 2.0 * r_final, 1e-9, 1.0)
    centers_final = np.column_stack([r_final + u_final * scale_final, r_final + w_final * scale_final])
    radii_final = np.maximum(r_final, 0.0)
    
    return centers_final, radii_final, float(best_sum)
