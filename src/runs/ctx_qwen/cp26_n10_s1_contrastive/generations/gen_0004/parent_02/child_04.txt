1. **LP-Optimal Radii**: For any fixed set of centers, the optimal radii are found exactly via Linear Programming (`scipy.optimize.linprog`). This guarantees we always evaluate the true maximum sum for a given topology.
2. **Hill Climbing on Centers**: Since the LP objective over centers is piecewise-linear and non-smooth, gradient-based methods fail. We use iterative random perturbation (Hill Climbing) on centers, accepting moves that improve the LP sum. This efficiently navigates the "contact graph" landscape.
3. **Diverse Hexagonal Initializations**: Optimal packings for N=26 are typically hexagonal lattices. We generate many variations with different row counts, rotations, translations, and scales to cover the configuration space.
4. **SLSQP Polish**: Once a good center configuration is found via Hill Climbing, we run a constrained SLSQP optimization on joint `(x, y, r)` variables to fine-tune positions and radii simultaneously, exploiting differentiability where constraints are inactive.
5. **Strict Feasibility**: We always finalize with an LP solve to guarantee exact optimality of radii for the returned centers, and clip small numerical violations.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    x, y = centers[:, 0], centers[:, 1]
    bounds_arr = np.stack([x, 1.0 - x, y, 1.0 - y], axis=1)
    for i in range(n):
        for b in bounds_arr[i]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback
    return np.full(n, 1e-6), 1e-4

def get_constraints(v):
    """Inequality constraints for SLSQP: g(v) >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary containment
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation: dist^2 >= (r_i + r_j)^2
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def hex_pattern_init(row_counts, seed, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    # Center and scale
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def hill_climb_lp(centers, max_iter=600, init_step=0.04):
    """Iterative Hill Climbing on centers using LP to evaluate radii."""
    current_centers = centers.copy()
    _, current_sum = solve_lp(current_centers)
    
    rng = np.random.default_rng(42)
    step = init_step
    
    for _ in range(max_iter):
        improved = False
        # Try moving each circle
        for i in range(N):
            n_trials = 8
            for _ in range(n_trials):
                new_centers = current_centers.copy()
                dir_ = rng.standard_normal(2)
                dir_ /= np.linalg.norm(dir_)
                new_centers[i] += step * dir_
                new_centers[i] = np.clip(new_centers[i], 1e-6, 1.0 - 1e-6)
                
                _, new_sum = solve_lp(new_centers)
                if new_sum > current_sum + 1e-9:
                    current_centers = new_centers
                    _, current_sum = solve_lp(current_centers)
                    improved = True
                    break
            if improved:
                break
                
        if not improved:
            step *= 0.92
            if step < 1e-5:
                break
                
    return current_centers, current_sum

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    # Phase 1: Diverse Initializations & Hill Climbing
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [7,5,5,5,4],
        [3,6,6,6,5], [5,4,6,6,5], [6,6,4,6,4], [4,4,6,6,6]
    ]
    
    configs = []
    for p in patterns:
        for s in range(4):
            for rot in [0.0, 0.12, -0.12, 0.25, -0.25]:
                for sc in [0.92, 1.0, 1.08]:
                    configs.append(('hex', p, s, rot, sc))
                    
    # Add random starts
    np.random.seed(123)
    for s in range(15):
        pts = np.random.rand(N, 2)
        pts = np.clip(pts, 0.05, 0.95)
        configs.append(('rand', None, s, 0.0, 1.0))

    for ctype, pat, s, rot, sc in configs:
        if ctype == 'hex':
            centers_init = hex_pattern_init(pat, s, rot, sc)
        else:
            centers_init = np.random.RandomState(s).rand(N, 2)
            centers_init = np.clip(centers_init, 0.05, 0.95)
            
        # Hill Climbing to find a good topology
        hc_centers, hc_sum = hill_climb_lp(centers_init, max_iter=400, init_step=0.035)
        
        if hc_sum > best_sum:
            best_sum = hc_sum
            best_centers = hc_centers.copy()
            best_radii, _ = solve_lp(hc_centers)

    # Phase 2: Joint SLSQP Polish
    if best_centers is not None:
        r_start = np.maximum(best_radii * 0.98, 1e-5)
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = r_start
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                c_val = get_constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    polished_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    # Re-solve LP for exact optimal radii at polished centers
                    new_radii, new_sum = solve_lp(polished_centers)
                    if new_sum > best_sum:
                        best_sum = new_sum
                        best_centers = polished_centers
                        best_radii = new_radii
        except Exception:
            pass
            
        # Phase 3: Final aggressive Hill Climbing + Polish cycle
        final_centers, final_sum = hill_climb_lp(best_centers, max_iter=300, init_step=0.015)
        if final_sum > best_sum:
            best_sum = final_sum
            best_centers = final_centers
            best_radii, _ = solve_lp(final_centers)
            
        # Final SLSQP polish
        r_polish = np.maximum(best_radii * 0.99, 1e-5)
        x0_fin = np.zeros(3 * N)
        x0_fin[0::3] = best_centers[:, 0]
        x0_fin[1::3] = best_centers[:, 1]
        x0_fin[2::3] = r_polish
        
        try:
            res_fin = minimize(objective, x0_fin, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-15, 'disp': False})
            if res_fin.success and np.min(get_constraints(res_fin.x)) >= -1e-8:
                fin_centers = np.column_stack((res_fin.x[0::3], res_fin.x[1::3]))
                fin_radii, fin_sum = solve_lp(fin_centers)
                if fin_sum > best_sum:
                    best_sum = fin_sum
                    best_centers = fin_centers
                    best_radii = fin_radii
        except Exception:
            pass

    # Fallback valid configuration
    if best_centers is None:
        grid = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        grid = np.vstack([grid, [0.5, 0.5]])
        best_centers = grid
        best_radii, best_sum = solve_lp(grid)

    # Ensure strict non-negativity and format
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
