def objective(v):

    return -np.sum(v[2::3])

def constraints(v):

    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)

    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]

    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def solve_lp_radii(centers):

    A_ub = []
    b_ub = []

            A_ub.append(row)
            b_ub.append(b)
