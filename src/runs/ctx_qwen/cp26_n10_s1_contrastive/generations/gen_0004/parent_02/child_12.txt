1. **Global Search**: Generate diverse initial center configurations using rotated and scaled hexagonal lattices, which are known to be near-optimal for circle packing.
2. **Joint Refinement**: Use SLSQP to jointly optimize centers and radii from these starts to quickly reach a high-quality local optimum.
3. **Hill Climbing on Centers**: This is the critical improvement step. We fix the radii to their LP-optimal values and perform direct search (random perturbations) on the 2D centers. This effectively smooths the optimization landscape and escapes SLSQP stagnation.
4. **Final Polish**: Run SLSQP one last time on the hill-climbed configuration to tighten constraints and squeeze out remaining numerical slack.

This hybrid approach leverages the exactness of LP for radii evaluation and the robustness of direct search for topology changes, consistently pushing packings past traditional joint-optimization limits.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    x, y = centers[:, 0], centers[:, 1]
    bounds_vals = np.column_stack([x, 1.0 - x, y, 1.0 - y])
    for i in range(n):
        for b in bounds_vals[i]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 2.6e-4

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """Compute all inequality constraints: boundary containment and separation."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation constraints: dist_sq >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def get_hex_init(row_counts, seed, scale=1.0, rot=0.0):
    """Generate a hexagonal lattice initialization with specified row counts."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.098
    y = r_est
    for r_idx, count in enumerate(row_counts):
        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        R = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = pts @ R.T
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Diverse hexagonal patterns to capture different boundary alignments
    hex_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [5, 5, 5, 5, 6], [6, 4, 6, 5, 5], [5, 6, 4, 6, 5],
        [4, 5, 6, 5, 6], [7, 5, 5, 5, 4], [4, 5, 5, 5, 7]
    ]
    
    inits = []
    for pat in hex_patterns:
        for s in range(3):
            for rot in np.linspace(-0.12, 0.12, 3):
                for sc in [0.96, 1.0, 1.04]:
                    inits.append(get_hex_init(pat, seed=s, scale=sc, rot=rot))
    
    # Phase 1: Broad search from diverse initializations using joint SLSQP
    for pts in inits:
        r_init, _ = solve_lp_radii(pts)
        r_init = np.maximum(r_init * 0.99, 1e-6)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r_init
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
        except Exception:
            continue

    # Phase 2: Hill climbing on centers using exact LP radii evaluation
    # This decouples the problem and efficiently escapes local minima
    if best_centers is not None:
        rng = np.random.default_rng(42)
        current_centers = best_centers.copy()
        _, current_sum = solve_lp_radii(current_centers)
        
        step = 0.02
        while step > 0.0005:
            improved = False
            for _ in range(800):
                idx = rng.integers(N)
                direction = rng.standard_normal(2)
                direction /= np.linalg.norm(direction)
                
                new_c = current_centers[idx] + step * direction
                new_c = np.clip(new_c, 0.005, 0.995)
                
                temp_c = current_centers.copy()
                temp_c[idx] = new_c
                _, new_sum = solve_lp_radii(temp_c)
                
                if new_sum > current_sum + 1e-8:
                    current_centers = temp_c
                    current_sum = new_sum
                    improved = True
                    
            if not improved:
                step *= 0.75
                
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = current_centers
            best_radii, _ = solve_lp_radii(best_centers)

    # Phase 3: High-precision SLSQP polish on the improved configuration
    if best_centers is not None:
        r_polish = np.maximum(best_radii * 0.999, 1e-6)
        x0_final = np.zeros(3 * N)
        x0_final[0::3] = best_centers[:, 0]
        x0_final[1::3] = best_centers[:, 1]
        x0_final[2::3] = r_polish
        
        try:
            res_final = minimize(objective, x0_final, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
            if res_final.success and np.min(constraints(res_final.x)) >= -1e-8:
                final_sum = -res_final.fun
                if final_sum > best_sum:
                    best_sum = final_sum
                    best_centers = np.column_stack((res_final.x[0::3], res_final.x[1::3]))
                    best_radii = res_final.x[2::3]
        except Exception:
            pass

    # Fallback valid configuration
    if best_centers is None:
        best_centers = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                        np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        best_radii, best_sum = solve_lp_radii(best_centers)

    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
