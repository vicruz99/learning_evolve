import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def get_xy(r, u, v):
    """Maps parameters to physical centers. Automatically satisfies boundary constraints."""
    return r + u * (1.0 - 2.0 * r), r + v * (1.0 - 2.0 * r)

def objective(p):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(p[:N])

def constraints(p):
    """Compute pairwise separation constraints: dist_sq >= (r_i + r_j)^2"""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x, y = get_xy(r, u, v)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def centers_to_params(centers, r_val):
    """Converts physical centers and a base radius to optimization parameters (r, u, v)."""
    denom = 1.0 - 2.0 * r_val
    denom = np.clip(denom, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r_val) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r_val) / denom, 0.0, 1.0)
    
    p = np.empty(3 * N)
    p[:N] = r_val
    p[N:2*N] = u
    p[2*N:3*N] = v
    return p

def generate_hex_pattern(row_counts, scale, rot, seed):
    """Generates a hexagonal lattice configuration with scaling, rotation, and jitter."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, count in enumerate(row_counts):
        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    
    # Normalize to unit square
    pts -= pts.min(axis=0)
    span = pts.max(axis=0) - pts.min(axis=0) + 1e-9
    pts /= span
    pts *= 0.92 * scale
    pts += (1.0 - 0.92 * scale) * 0.5
    
    # Rotate
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c, -s], [s, c]])
        pts = pts @ rot_mat.T
        
    # Jitter and clip
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(1e-6, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -np.inf
    best_p = None
    
    inits = []
    
    # Phase 1: Generate diverse initial configurations
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], 
        [5,5,5,5,6], [6,4,6,5,5], [5,6,4,6,5],
        [4,5,6,5,6], [7,5,5,5,4], [4,5,5,5,7],
        [3,6,6,6,5], [5,6,6,6,3], [4,5,6,6,5]
    ]
    for pat in patterns:
        for s in range(8):
            rng = np.random.RandomState(s)
            sc = 0.9 + rng.uniform(0, 0.35)
            rot = rng.uniform(-0.3, 0.3)
            pts = generate_hex_pattern(pat, sc, rot, s)
            r_init = np.full(N, 0.045)
            inits.append(centers_to_params(pts, r_init))
            
    # Add sorted random layouts for variety
    for s in range(30):
        rng = np.random.RandomState(s)
        pts = rng.rand(N, 2)
        pts = pts[pts[:, 1].argsort()]
        for i in range(N):
            pts[i, 0] = 0.05 + pts[i, 0] * 0.9 + rng.uniform(-0.025, 0.025)
        pts = np.clip(pts, 0.02, 0.98)
        r_init = np.full(N, 0.04)
        inits.append(centers_to_params(pts, r_init))
        
    # Run optimization from all initializations
    for p0 in inits:
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-6:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Local perturbation refinement
    if best_p is not None:
        for k in range(60):
            rng = np.random.RandomState(k)
            p_pert = best_p.copy()
            p_pert += rng.normal(0, 0.0025, 3*N)
            p_pert[:N] = np.clip(p_pert[:N], 1e-5, 0.48)
            p_pert[N:2*N] = np.clip(p_pert[N:2*N], 0.0, 1.0)
            p_pert[2*N:3*N] = np.clip(p_pert[2*N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(objective, p_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if res.success:
                    c_val = constraints(res.x)
                    if np.min(c_val) >= -1e-6:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_p = res.x.copy()
            except Exception:
                continue
                
        # Phase 3: High-precision polish
        try:
            res = minimize(objective, best_p, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-7:
                best_p = res.x
                best_sum = -res.fun
        except Exception:
            pass

    # Fallback configuration (should rarely be reached)
    if best_p is None:
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_p = centers_to_params(pts, np.full(N, 0.05))
        best_sum = np.sum(best_p[:N])
        
    # Reconstruct physical centers and radii
    r_opt = best_p[:N]
    u_opt = best_p[N:2*N]
    v_opt = best_p[2*N:3*N]
    x_opt, y_opt = get_xy(r_opt, u_opt, v_opt)
    centers = np.column_stack((x_opt, y_opt))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
