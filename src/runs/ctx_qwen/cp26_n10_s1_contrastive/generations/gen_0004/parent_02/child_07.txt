import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def get_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to non-overlap and boundary constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b_val in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b_val)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0), -res.fun
    return np.full(n, 1e-6), 0.0

def obj_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def cons_joint(v):
    """Inequality constraints: g(v) >= 0 for boundaries and pairwise separation."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def gen_hex(row_counts, seed, rot=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization with specified row counts, rotation, and scale."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        R = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = pts @ R.T
        pts = np.clip(pts, 0.02, 0.98)
    else:
        pts += rng.uniform(-0.02, 0.02, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': cons_joint}
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    inits = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [7, 5, 5, 5, 4], [4, 5, 5, 5, 7], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [5, 6, 4, 6, 5], [4, 5, 6, 5, 6]
    ]
    
    np.random.seed(42)
    for pat in patterns:
        for s in range(3):
            rot = np.random.uniform(-0.2, 0.2)
            scale = np.random.uniform(0.88, 1.12)
            inits.append(gen_hex(pat, seed=s, rot=rot, scale=scale))
            
    # Phase 1: Broad search from diverse initializations using SLSQP
    for pts in inits:
        r_init, _ = get_lp_radii(pts)
        r_init = np.maximum(r_init * 0.95, 1e-5)
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r_init
        
        try:
            res = minimize(obj_joint, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_val = cons_joint(res.x)
                if np.min(c_val) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
        except Exception:
            continue
            
    # Phase 2: Hill Climbing on centers using exact LP radii evaluation
    if best_centers is not None:
        centers_hc = best_centers.copy()
        radii_hc, sum_hc = get_lp_radii(centers_hc)
        current_sum = sum_hc
        
        rng = np.random.RandomState(123)
        step_size = 0.05
        for step in range(2500):
            idx = rng.integers(N)
            angle = rng.uniform(0.0, 2.0 * np.pi)
            direction = np.array([np.cos(angle), np.sin(angle)])
            
            new_centers = centers_hc.copy()
            new_centers[idx] = np.clip(new_centers[idx] + direction * step_size, 0.01, 0.99)
            
            r_new, s_new = get_lp_radii(new_centers)
            if s_new > current_sum + 1e-9:
                centers_hc = new_centers
                radii_hc = r_new
                current_sum = s_new
            if step % 200 == 0 and step > 0:
                step_size *= 0.85
                
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_hc
            best_radii = radii_hc

    # Phase 3: High-precision SLSQP polish on the refined configuration
    if best_centers is not None:
        r_polish = np.maximum(best_radii * 0.99, 1e-6)
        x0_final = np.zeros(3 * N)
        x0_final[0::3] = best_centers[:, 0]
        x0_final[1::3] = best_centers[:, 1]
        x0_final[2::3] = r_polish
        
        try:
            res_final = minimize(obj_joint, x0_final, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if res_final.success and np.min(cons_joint(res_final.x)) >= -1e-9:
                final_centers = np.column_stack((res_final.x[0::3], res_final.x[1::3]))
                best_radii, best_sum = get_lp_radii(final_centers)
                best_centers = final_centers
        except Exception:
            pass

    # Fallback valid configuration
    if best_centers is None:
        best_centers = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                        np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        best_radii = np.full(N, 0.01)
        best_sum = np.sum(best_radii)
        
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
