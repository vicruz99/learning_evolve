import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 2.6e-4

def joint_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def joint_constr(v):
    """Inequality constraints for joint optimization."""
    xc = v[0::3]
    yc = v[1::3]
    rc = v[2::3]
    c = []
    # Boundary containment
    c.append(xc - rc)
    c.append(1.0 - xc - rc)
    c.append(yc - rc)
    c.append(1.0 - yc - rc)
    # Pairwise separation
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    d2 = dx**2 + dy**2
    rs = rc[:, None] + rc[None, :]
    c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def generate_hex_configs():
    """Generates diverse hexagonal lattice configurations with rotations."""
    configs = []
    base_rows = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [5, 5, 5, 5, 6], [6, 4, 6, 5, 5], [7, 5, 5, 5, 4],
        [5, 7, 5, 5, 4], [4, 5, 6, 5, 6], [5, 6, 4, 6, 5]
    ]
    rng = np.random.RandomState(42)
    for rows in base_rows:
        pts = []
        r_est = 0.095
        y = r_est
        for r_idx, count in enumerate(rows):
            x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
            for _ in range(count):
                if len(pts) < N:
                    pts.append([x_start, y])
                x_start += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
        pts = np.array(pts[:N])
        
        # Rotate and scale variations to explore boundary alignments
        for rot in np.linspace(-0.2, 0.2, 7):
            c_cos, s_sin = np.cos(rot), np.sin(rot)
            R = np.array([[c_cos, -s_sin], [s_sin, c_cos]])
            pts_rot = pts @ R.T
            pts_rot -= 0.5
            pts_rot *= 0.85  # Scale down to leave room for optimizer expansion
            pts_rot += 0.5
            pts_rot += rng.uniform(-0.015, 0.015, pts_rot.shape)
            pts_rot = np.clip(pts_rot, 0.02, 0.98)
            configs.append(pts_rot)
    return configs

def run_packing():
    np.random.seed(123)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    configs = generate_hex_configs()
    cons = {'type': 'ineq', 'fun': joint_constr}
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    
    # Phase 1: Broad search with LP initialization + SLSQP refinement
    for cfg in configs:
        r_init, s_init = solve_radii_lp(cfg)
        if s_init > best_sum:
            best_sum = s_init
            best_centers = cfg.copy()
            best_radii = r_init.copy()
            
        x0 = np.zeros(3 * N)
        x0[0::3] = cfg[:, 0]
        x0[1::3] = cfg[:, 1]
        x0[2::3] = np.maximum(r_init * 0.99, 1e-5)
        
        try:
            res = minimize(joint_obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                if np.min(joint_constr(res.x)) >= -1e-7:
                    s_new = np.sum(res.x[2::3])
                    if s_new > best_sum:
                        best_sum = s_new
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
        except Exception:
            pass

    if best_centers is None:
        best_centers = np.random.rand(N, 2) * 0.8 + 0.1
        best_radii, best_sum = solve_radii_lp(best_centers)

    # Phase 2: Stochastic Hill Climbing on Centers
    # Uses LP objective to navigate the non-smooth landscape efficiently
    curr_centers = best_centers.copy()
    curr_sum = best_sum
    step_sizes = [0.03, 0.015, 0.007, 0.003, 0.001]
    
    for step in step_sizes:
        improved = True
        while improved:
            improved = False
            for _ in range(N * 30):
                idx = np.random.randint(N)
                direction = np.random.randn(2)
                direction /= np.linalg.norm(direction)
                
                new_centers = curr_centers.copy()
                new_centers[idx] += step * direction
                new_centers[idx] = np.clip(new_centers[idx], 0.01, 0.99)
                
                _, new_sum = solve_radii_lp(new_centers)
                if new_sum > curr_sum + 1e-7:
                    curr_centers = new_centers
                    curr_sum = new_sum
                    improved = True
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_radii, _ = solve_radii_lp(curr_centers)
                        best_centers = curr_centers.copy()
                    break
                    
    # Phase 3: Perturbation & Final SLSQP Polish
    for _ in range(15):
        x_pert = np.concatenate([best_centers.flatten(), best_radii])
        x_pert += np.random.normal(0, 0.005, x_pert.shape)
        x_pert[0::3] = np.clip(x_pert[0::3], 0.01, 0.99)
        x_pert[1::3] = np.clip(x_pert[1::3], 0.01, 0.99)
        x_pert[2::3] = np.clip(x_pert[2::3], 1e-6, 0.49)
        
        try:
            res = minimize(joint_obj, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(joint_constr(res.x)) >= -1e-7:
                s_new = np.sum(res.x[2::3])
                if s_new > best_sum:
                    best_sum = s_new
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3]
        except Exception:
            pass

    # Final exact LP solve to guarantee feasibility and extract maximum radii
    best_radii, best_sum = solve_radii_lp(best_centers)
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
