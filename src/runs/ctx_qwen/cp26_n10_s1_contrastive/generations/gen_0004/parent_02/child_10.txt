import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 2.6e-4

def center_obj(x_flat):
    """Objective for center optimization: minimize negative sum of radii from LP."""
    centers = x_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    _, s = solve_radii_lp(centers)
    return -s

def joint_constraints(v):
    """Inequality constraints: g(v) >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def joint_objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def make_hex_init(row_counts, seed, scale=1.0, rot=0.0):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, count in enumerate(row_counts):
        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force_init(seed):
    """Spread points using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    
    for step in range(300):
        forces = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-4)
        
        rep_mag = 1.0 / (dists**2)
        rep_mag = np.minimum(rep_mag, 2.0)
        np.fill_diagonal(rep_mag, 0.0)
        
        forces += np.sum(rep_mag[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 10.0 * (pts[:, d] < 0.1) * (0.1 - pts[:, d])
            forces[:, d] -= 10.0 * (pts[:, d] > 0.9) * (pts[:, d] - 0.9)
            
        step_size = 0.005 * (0.99**step)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)
        
    return pts

def run_packing():
    np.random.seed(42)
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    bounds_joint = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_joint = {'type': 'ineq', 'fun': joint_constraints}
    
    # Phase 1: Diverse initializations & Center Optimization via Nelder-Mead
    hex_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [5, 5, 5, 5, 6], [6, 4, 6, 5, 5], [5, 6, 4, 6, 5]
    ]
    
    init_configs = []
    for pat in hex_patterns:
        for s in range(3):
            init_configs.append(make_hex_init(pat, seed=s, scale=0.95 + 0.05*(s%2), rot=0.05*(s-1)))
            
    for s in range(6):
        init_configs.append(make_force_init(seed=s))
        
    for pts in init_configs:
        try:
            res_nm = minimize(center_obj, pts.flatten(), method='Nelder-Mead',
                              options={'maxiter': 2000, 'xatol': 1e-6, 'fatol': 1e-8})
            curr_centers = np.clip(res_nm.x.reshape(N, 2), 0.01, 0.99)
            r_lp, s_lp = solve_radii_lp(curr_centers)
            
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = curr_centers
                best_radii = r_lp
        except Exception:
            continue

    # Phase 2: Joint SLSQP refinement from best LP results
    if best_centers is not None:
        # Prepare joint variables, shrink radii slightly for strict interior feasibility
        v0 = np.zeros(3 * N)
        v0[0::3] = best_centers[:, 0]
        v0[1::3] = best_centers[:, 1]
        v0[2::3] = np.maximum(best_radii * 0.99, 1e-5)
        
        try:
            res_slsqp = minimize(joint_objective, v0, method='SLSQP', bounds=bounds_joint,
                                 constraints=cons_joint, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res_slsqp.success and np.min(joint_constraints(res_slsqp.x)) >= -1e-7:
                curr_sum = -res_slsqp.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = np.column_stack((res_slsqp.x[0::3], res_slsqp.x[1::3]))
                    best_radii = res_slsqp.x[2::3]
        except Exception:
            pass

    # Phase 3: Perturbation & Restart Loop to escape local minima
    if best_centers is not None:
        rng = np.random.default_rng(42)
        for _ in range(20):
            v_p = np.zeros(3 * N)
            v_p[0::3] = best_centers[:, 0] + rng.normal(0, 0.004, N)
            v_p[1::3] = best_centers[:, 1] + rng.normal(0, 0.004, N)
            v_p[2::3] = best_radii + rng.normal(0, 0.001, N)
            
            v_p[0::3] = np.clip(v_p[0::3], 0.01, 0.99)
            v_p[1::3] = np.clip(v_p[1::3], 0.01, 0.99)
            v_p[2::3] = np.clip(v_p[2::3], 1e-5, 0.49)
            
            try:
                res_p = minimize(joint_objective, v_p, method='SLSQP', bounds=bounds_joint,
                                 constraints=cons_joint, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if res_p.success and np.min(joint_constraints(res_p.x)) >= -1e-7:
                    s_p = -res_p.fun
                    if s_p > best_sum:
                        best_sum = s_p
                        best_centers = np.column_stack((res_p.x[0::3], res_p.x[1::3]))
                        best_radii = res_p.x[2::3]
            except Exception:
                continue
                
        # Phase 4: Final high-precision polish
        v_final = np.zeros(3 * N)
        v_final[0::3] = best_centers[:, 0]
        v_final[1::3] = best_centers[:, 1]
        v_final[2::3] = best_radii
        
        try:
            res_f = minimize(joint_objective, v_final, method='SLSQP', bounds=bounds_joint,
                             constraints=cons_joint, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(joint_constraints(res_f.x)) >= -1e-8:
                best_centers = np.column_stack((res_f.x[0::3], res_f.x[1::3]))
                best_radii = res_f.x[2::3]
                best_sum = -res_f.fun
        except Exception:
            pass

    # Fallback (should not be reached)
    if best_centers is None:
        grid = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        grid = np.vstack([grid, [0.5, 0.5]])
        best_centers = grid
        best_radii, best_sum = solve_radii_lp(grid)

    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
