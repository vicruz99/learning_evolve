import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

N_CIRCLES = 26

def get_lp_objective(centers):
    """Solves LP to find radii maximizing sum for fixed centers. Returns (neg_sum, radii)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    xs = centers[:, 0]
    ys = centers[:, 1]
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        b = min(xs[i], 1.0 - xs[i], ys[i], 1.0 - ys[i])
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(b)
        
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def de_objective_flat(x_flat):
    """Objective for Differential Evolution: minimize negative LP sum of radii."""
    centers = x_flat.reshape(N_CIRCLES, 2)
    neg_sum, _ = get_lp_objective(centers)
    return neg_sum

def nm_objective_flat(x_flat):
    """Objective for Nelder-Mead: minimize negative LP sum of radii."""
    centers = x_flat.reshape(N_CIRCLES, 2)
    neg_sum, _ = get_lp_objective(centers)
    return neg_sum

def constraints_func(v):
    """Inequality constraints for SLSQP: boundary containment and pairwise separation."""
    n = len(v) // 3
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise: dist^2 - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    i_idx, j_idx = np.triu_indices(n, k=1)
    c.append(dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    return np.concatenate(c)

def objective_func(v):
    """SLSQP objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def run_packing():
