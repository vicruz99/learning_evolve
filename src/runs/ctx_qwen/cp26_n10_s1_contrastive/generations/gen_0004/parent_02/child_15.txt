To improve the packing sum of radii beyond 2.633, I will employ a two-stage optimization strategy that decouples center placement from radius assignment. This avoids the rugged joint (x,y,r) landscape where standard gradient-based solvers often stall.

1. **LP-based Center Optimization**: For any fixed set of centers, finding the optimal radii that maximize the sum is a convex Linear Program. I will use `scipy.optimize.linprog` (with the highly efficient `highs` solver) to compute the exact maximum sum of radii for given centers. This serves as a smooth(er), globally informative objective function for center optimization.
2. **Global & Local Center Search**: 
   - Use `differential_evolution` to explore the 52-dimensional center space globally, maximizing the LP sum.
   - Refine the best DE result with `Powell` (derivative-free, handles non-smooth LP objective well).
   - Supplement with structured initializations (hexagonal lattices with varying row counts, force-directed layouts) refined via `Nelder-Mead`.
3. **Joint SLSQP Polish**: Once high-quality centers are found, re-introduce radii as variables and run a joint constrained optimization with SLSQP. The squared-distance constraints and boundary containment are enforced directly. This phase exploits local gradients to squeeze out the final margin.
4. **Robustness**: All helper functions are strictly top-level to avoid closure overhead and comply with constraints. Numerical safeguards (clipping, small tolerances, feasibility checks) ensure the final output strictly passes validation.

This approach systematically escapes local minima by leveraging the convex structure of the radius subproblem, while still utilizing gradient-based refinement for precision.

import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def get_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        bounds = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds:
            A_ub[idx, i] = 1.0
            b_ub[idx] = b
            idx += 1
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds_r = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs', options={'presolve': True})
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def lp_objective(x_flat):
    """Objective for center optimization: negative LP sum of radii."""
    centers = np.clip(x_flat.reshape(N, 2), 0.001, 0.999)
    s, _ = get_lp_radii(centers)
    return -s

def hex_init(seed, pattern):
    """Generates a hexagonal lattice initialization with specified row counts."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.098
    y = r_est
    for r_idx, count in enumerate(pattern):
        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Force-directed layout initialization to spread points evenly."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    for _ in range(200):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        f += np.sum((1.0 / dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        for d in range(2):
            f[:, d] += 10.0 * (pts[:, d] < 0.1) * (0.1 - pts[:, d])
            f[:, d] -= 10.0 * (pts[:, d] > 0.9) * (pts[:, d] - 0.9)
        pts += 0.005 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def joint_objective(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def joint_constraints(v):
    """Computes inequality constraints g(v) >= 0 for joint optimization."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = []
    # Boundary containment
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    # Pairwise separation
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def run_packing():
    best_centers = None
    best_sum = -np.inf
    
    # Phase 1: Global search on centers using Differential Evolution
    bounds_de = [(0.01, 0.99)] * (2 * N)
    try:
        res_de = differential_evolution(lp_objective, bounds_de, popsize=15, maxiter=40, 
                                        seed=42, workers=1, polish=False, tol=1e-7)
        c_de = np.clip(res_de.x.reshape(N, 2), 0.001, 0.999)
        s_de, _ = get_lp_radii(c_de)
        if s_de > best_sum:
            best_sum = s_de
            best_centers = c_de
    except Exception:
        pass
        
    # Phase 2: Refine DE result with Powell method
    if best_centers is not None:
        try:
            res_pw = minimize(lp_objective, best_centers.flatten(), method='Powell', 
                              options={'maxiter': 1500, 'xtol': 1e-7, 'ftol': 1e-7})
            c_pw = np.clip(res_pw.x.reshape(N, 2), 0.001, 0.999)
            s_pw, _ = get_lp_radii(c_pw)
            if s_pw > best_sum:
                best_sum = s_pw
                best_centers = c_pw
        except Exception:
            pass
            
    # Phase 3: Explore structured initializations
    hex_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [5, 5, 5, 5, 6], [6, 4, 6, 5, 5], [5, 6, 4, 6, 5]
    ]
    candidates = []
    for pat in hex_patterns:
        for s in range(3):
            candidates.append(hex_init(s, pat))
    for s in range(5):
        candidates.append(force_init(s))
        
    for init_c in candidates:
        try:
            res_nm = minimize(lp_objective, init_c.flatten(), method='Nelder-Mead',
                              options={'maxiter': 1500, 'xatol': 1e-7, 'fatol': 1e-7})
            c_cand = np.clip(res_nm.x.reshape(N, 2), 0.001, 0.999)
            s_cand, _ = get_lp_radii(c_cand)
            if s_cand > best_sum:
                best_sum = s_cand
                best_centers = c_cand
        except Exception:
            continue
            
    # Fallback configuration
    if best_centers is None:
        best_centers = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                        np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        
    # Phase 4: Joint SLSQP refinement on (centers, radii)
    r_init, _ = get_lp_radii(best_centers)
    r_init = np.maximum(r_init * 0.99, 1e-6)
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = r_init
    
    bounds_full = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': joint_constraints}
    
    try:
        res_sl = minimize(joint_objective, x0, method='SLSQP', bounds=bounds_full,
                          constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
        if res_sl.success:
            c_val = joint_constraints(res_sl.x)
            if np.min(c_val) >= -1e-7:
                best_centers = np.column_stack((res_sl.x[0::3], res_sl.x[1::3]))
                best_radii = res_sl.x[2::3]
                best_sum = np.sum(best_radii)
        else:
            best_radii = r_init
    except Exception:
        best_radii = r_init
        
    # Phase 5: High-precision polish
    try:
        x0_polish = np.zeros(3 * N)
        x0_polish[0::3] = best_centers[:, 0]
        x0_polish[1::3] = best_centers[:, 1]
        x0_polish[2::3] = np.maximum(best_radii * 0.999, 1e-6)
        
        res_pol = minimize(joint_objective, x0_polish, method='SLSQP', bounds=bounds_full,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14})
        if res_pol.success and np.min(joint_constraints(res_pol.x)) >= -1e-8:
            best_centers = np.column_stack((res_pol.x[0::3], res_pol.x[1::3]))
            best_radii = res_pol.x[2::3]
            best_sum = np.sum(best_radii)
    except Exception:
        pass
        
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
