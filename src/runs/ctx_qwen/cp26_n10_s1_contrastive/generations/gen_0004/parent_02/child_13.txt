import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[:N])

def cons(v):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = v[:N]
    u = v[N:2*N]
    w = v[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_pv(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    d = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, w])

def safe_r(centers):
    """Compute strictly feasible initial radii."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        dw = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        ds = np.linalg.norm(centers[i] - centers, axis=1)
        ds[i] = np.inf
        r[i] = 0.82 * min(dw, 0.5 * np.min(ds))
    return np.maximum(r, 1e-4)

def hex_init(seed, rot=0.0, scale=1.0):
    """Generate hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    re = 0.098
    y = re
    row = 0
    while len(pts) < N:
        sh = re if row % 2 == 0 else 2.0 * re
        x = re + sh
        while x <= 1.0 - re and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * re
        y += np.sqrt(3.0) * re
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.012, 0.012, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():

    best_v = None
    best_sum = -np.inf

        pts = rng.rand(N, 2)
        for _ in range(150):
            diff = pts[:, None, :] - pts[None, :, :]
            dist = np.sqrt(np.sum(diff**2, axis=2))
            dist = np.maximum(dist, 1e-3)
            f = np.sum(diff / dist**3, axis=1)
            pts += 0.004 * f
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    # Phase 1: Initial optimization from diverse starts
    for pts in inits:
        r0 = safe_r(pts)
        v0 = make_pv(pts, r0)
        try:

            if res.success:

                if np.min(cv) >= -1e-6:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Iterative perturbation & restart to escape local minima
    if best_v is not None:

    r_opt = np.maximum(r_opt, 0.0)
    best_sum = float(np.sum(r_opt))
    
    return centers, r_opt, best_sum
