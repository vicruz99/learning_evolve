from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((n*4 + n*(n-1)//2, n))
    b_ub = np.zeros(n*4 + n*(n-1)//2)
    
    row_idx = 0
    for i in range(n):
        x, y = centers[i]
        vals = [x, 1.0 - x, y, 1.0 - y]
        for v in vals:
            A_ub[row_idx, i] = 1.0
            b_ub[row_idx] = v
            row_idx += 1
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            b_ub[row_idx] = dists[i, j]
            row_idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-5

def lp_objective(centers_flat):
    """Objective for center optimization: minimize negative LP sum of radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    _, sum_r = solve_lp_radii(centers)
    return -sum_r

def get_hex_init(row_counts, seed):
    """Generate a hexagonal lattice initialization with specified row counts."""
    pts = []
    rng = np.random.RandomState(seed)
    r_est = 0.098
    y = r_est

        x_start = r_est if r_idx % 2 == 0 else 2.0 * r_est
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def joint_obj(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def joint_cons(v):
    """Inequality constraints: boundary containment and separation."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise separation
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)
