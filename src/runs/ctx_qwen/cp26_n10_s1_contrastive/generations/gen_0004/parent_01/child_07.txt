import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def obj_centers(centers_flat):
    """Black-box objective for center optimization: minimize negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-3, 1.0 - 1e-3)
    _, neg_sum = solve_lp_radii(centers)
    return neg_sum

def joint_obj(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def joint_cons(v):
    """Inequality constraints for joint optimization: boundary and non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def generate_inits():
    """Generates diverse initial center configurations."""
    inits = []
    rng = np.random.RandomState(42)
    
    # 1. Hexagonal lattice variations
    for rot in rng.uniform(-0.3, 0.3, size=6):
        pts = []
        r_est = 0.09
        y = r_est
        row = 0
        while len(pts) < N:
            x_off = (row % 2) * r_est
            x = r_est + x_off
            while x <= 1.0 - r_est and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
        centers = np.array(pts[:N])
        if rot != 0.0:
            c, s = np.cos(rot), np.sin(rot)
            rot_mat = np.array([[c, -s], [s, c]])
            centers = centers @ rot_mat.T
            centers = np.clip(centers, 0.02, 0.98)
        inits.append(centers.flatten())
        
    # 2. Perturbed Grid
    for _ in range(8):
        grid = []
        for i in range(5):
            for j in range(5):
                grid.append([0.1 + 0.2*i, 0.1 + 0.2*j])
        grid.append([0.5, 0.5])
        centers = np.array(grid) + rng.uniform(-0.03, 0.03, (N, 2))
        centers = np.clip(centers, 0.02, 0.98)
        inits.append(centers.flatten())
        
    # 3. Force-directed layouts
    for s in range(8):
        rng_local = np.random.RandomState(s)
        pts = rng_local.rand(N, 2) * 0.8 + 0.1
        for _ in range(250):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            dists = np.maximum(dists, 1e-4)
            forces = np.sum((1.0/dists**2)[:, :, None] * diff / dists[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 8.0 * np.maximum(0, 0.05 - pts[:, d])
                forces[:, d] -= 8.0 * np.maximum(0, pts[:, d] - 0.95)
            pts += 0.004 * forces
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts.flatten())
        
    # 4. Pure random
    for _ in range(6):
        inits.append(rng.rand(N, 2).flatten())
        
    return inits

def run_packing():

    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).

    best_centers = None
    best_radii = None
    rng = np.random.RandomState(42)

    # Phase 1: Two-level optimization (Centers via Nelder-Mead, Radii via LP)
    for x0 in inits:

            centers_opt = res.x.reshape(N, 2)
            centers_opt = np.clip(centers_opt, 1e-3, 1.0 - 1e-3)
            radii_opt, curr_sum = solve_lp_radii(centers_opt)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
        except Exception:
            continue
            
    # Phase 2: Local perturbations on best centers to escape local minima

                centers_opt = res.x.reshape(N, 2)

                    best_centers = centers_opt.copy()

    if best_centers is not None:

        v0[1::3] = best_centers[:, 1]

            if res.success:

            pass

                    res = minimize(joint_obj, v_pert, method='SLSQP', bounds=bounds_j,

                    if res.success:
                        cons_val = joint_cons(res.x)

                            curr_sum = -res.fun
                            if curr_sum > best_sum:
                                best_sum = curr_sum
                                best_centers = np.column_stack((res.x[0::3], res.x[1::3]))

                                v0 = res.x.copy()

    # Fallback valid configuration

        best_centers = np.zeros((N, 2))

    return best_centers, best_radii, float(best_sum)
