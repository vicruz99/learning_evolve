from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """
    Computes inequality constraints g(v) >= 0.
    Uses unsquared distances for better gradient behavior near contacts.
    """
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(np.maximum(dx**2 + dy**2, 1e-12))
    r_sum = r[:, None] + r[None, :]
    
    c.append(dist[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX])
    return np.concatenate(c)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find exact optimal radii."""
    n = centers.shape[0]
    c_lp = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    return np.full(n, 0.01)

        pts = np.clip(pts, 1e-6, 1.0 - 1e-6)

            dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])

            r[i] = 0.95 * min(dw, dp/2.0)
            
    return np.column_stack([pts, r]).flatten()

def hex_init(rng, rot):
    """Generates a hexagonal lattice initialization with rotation and perturbation."""
    pts = []
    r_est = 0.09
    y = r_est
    row = 0
    while len(pts) < N:
        x_off = (row % 2) * r_est
        x = r_est + x_off
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    centers = np.array(pts[:N])
    centers += rng.uniform(-0.02, 0.02, (N, 2))
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c, -s], [s, c]])
        centers = centers @ rot_mat.T
        centers = np.clip(centers, 0.02, 0.98)
        
    r = np.full(N, 0.05)
    for i in range(N):
        dw = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])

        r[i] = 0.9 * min(dw, dp/2.0)
        
    return np.column_stack([centers, r]).flatten()
