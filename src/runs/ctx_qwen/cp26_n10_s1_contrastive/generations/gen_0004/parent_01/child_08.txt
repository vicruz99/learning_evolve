import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def constraint_func(vars_vec):
    """
    Computes inequality constraints g(vars_vec) >= 0.
    Uses parameterization to automatically satisfy boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    
    # Parameterization maps u, v in [0,1] to x, y such that circle is strictly inside [0,1]^2
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    # Pairwise squared distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    # Squared sum of radii
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX]

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    d = 1.0 - 2.0 * r
    d = np.clip(d, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    
    out = np.empty(3 * N)
    out[0::3] = r
    out[1::3] = u
    out[2::3] = v
    return out

def from_params(params):
    """Convert (r, u, v) parameters back to physical centers and radii."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def generate_hex(row_counts, rot=0.0, scale=1.0, jitter=0.01, seed=0):
    """Generates a hexagonal lattice initialization with rotation and perturbation."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    pts += rng.uniform(-jitter, jitter, (N, 2))
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def force_layout(seed, steps=200):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    r = np.full(N, 0.05)
    
    for step in range(steps):
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-6)
        rep = 1.0 / dist**2
        np.fill_diagonal(rep, 0.0)
        force = np.sum(rep[:, :, None] * diff / dist[:, :, None], axis=1)
        
        # Wall repulsion to keep points inside
        force += 10.0 * np.maximum(0, r - pts)
        force -= 10.0 * np.maximum(0, pts - (1.0 - r))
            
        step_sz = 0.005 * (0.99**step)
        pts += step_sz * force
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        # Adaptively update radii during simulation
        for i in range(N):
            dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            dp = max(dp, 1e-6)
            r[i] = 0.9 * min(dw, dp/2.0)
            
    return pts, r

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    bounds = [(1e-6, 0.5), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_params = None
    best_sum = -1.0
    rng = np.random.RandomState(42)
    
    inits = []
    
    # Hexagonal lattice variations
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5],
        [5, 5, 5, 5, 6], [6, 4, 6, 5, 5], [5, 6, 4, 6, 5],
        [4, 5, 6, 5, 6], [6, 6, 5, 5, 4]
    ]
    
    for pat in patterns:
        for s in range(3):
            rot = rng.uniform(-0.2, 0.2)
            scale = rng.uniform(0.9, 1.1)
            centers = generate_hex(pat, rot=rot, scale=scale, jitter=0.01, seed=s)
            r_safe = np.zeros(N)
            for i in range(N):
                dw = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
                dp = np.min(np.linalg.norm(centers[i] - centers, axis=1))
                dp = max(dp, 1e-6)
                r_safe[i] = 0.95 * min(dw, dp/2.0)
            inits.append(to_params(centers, r_safe))
            
    # Force-directed layouts

        centers, r_safe = force_layout(s)
        inits.append(to_params(centers, r_safe))

        r_safe = np.zeros(N)

            dp = max(dp, 1e-6)
            r_safe[i] = 0.8 * min(dw, dp/2.0)
        inits.append(to_params(centers, r_safe))
        
    # Phase 1: Broad search from diverse initializations
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                if np.min(constraint_func(res.x)) >= -1e-7:
                    curr_sum = -res.fun

                        best_sum = curr_sum
                        best_params = res.x.copy()
        except Exception:
            continue

                x0[3*i], x0[3*j] = x0[3*j], x0[3*i]
                x0[3*i+1], x0[3*j+1] = x0[3*j+1], x0[3*i+1]
                x0[3*i+2], x0[3*j+2] = x0[3*j+2], x0[3*i+2]
                
            try:

                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_params = res.x.copy()

                continue
                
        # Phase 3: High-precision polish
        try:
            res = minimize(objective, best_params, method='SLSQP', bounds=bounds,

            if np.min(constraint_func(res.x)) >= -1e-8:

            pass

    if best_params is None:
        centers_f = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        r_f = np.full(N, 0.09)
        best_params = to_params(centers_f, r_f)
        best_sum = np.sum(r_f)

    # Reconstruct centers from optimized parameters
    centers_opt, r_opt = from_params(best_params)
    return centers_opt, r_opt, float(best_sum)
