import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def get_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def obj_centers(centers_flat):
    """Objective for center optimization: minimize negative sum of optimal radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 0.0001, 0.9999)
    _, s = get_lp_radii(centers)
    return -s

def hex_centers(seed, pattern, scale=1.0, rot=0.0):
    """Generate hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts -= 0.5
    pts *= scale
    pts += 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.05, 0.95)

def force_centers(seed):
    """Generate force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    for _ in range(200):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        f += np.sum((1.0/dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        pts += 0.005 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def constr_slscp(x):
    """Inequality constraints for SLSQP polish."""
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    c = []
    c.append(xc - r)
    c.append(1.0 - xc - r)
    c.append(yc - r)
    c.append(1.0 - yc - r)
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)
    
def obj_slscp(x):
    """Objective for SLSQP polish."""
    return -np.sum(x[2::3])

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], 
        [5,5,5,5,6], [6,4,6,5,5], [5,6,4,6,5]
    ]
    
    inits = []
    for p in patterns:
        for s in range(2):
            inits.append(hex_centers(s, p, scale=0.95 + 0.1*(s%2), rot=0.05*(s-1)))
    for s in range(3):
        inits.append(force_centers(s))
    for s in range(3):
        rng = np.random.RandomState(s)
        inits.append(rng.rand(N, 2) * 0.9 + 0.05)
        
    best_centers_init = None
    best_sum_init = -np.inf
    
    # Phase 1: Optimize centers using LP-derived objective
    for c0 in inits:
        try:
            res = minimize(obj_centers, c0.flatten(), method='Powell', 
                           options={'maxiter': 1500, 'xtol': 1e-7, 'ftol': 1e-9, 'gtol': 1e-8})
            c_opt = np.clip(res.x.reshape(N, 2), 0.0001, 0.9999)
            radii, s_val = get_lp_radii(c_opt)
            if s_val > best_sum_init:
                best_sum_init = s_val
                best_centers_init = c_opt
                best_radii_init = radii
        except Exception:
            pass
            
    if best_centers_init is not None:
        best_centers = best_centers_init
        best_radii = best_radii_init
        best_sum = best_sum_init
    else:
        best_centers = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_radii = np.full(N, 0.09)
        best_sum = np.sum(best_radii)
        
    # Phase 2: SLSQP High-Precision Polish on full variables
    bounds_full = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constr_slscp}
    
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = best_radii * 0.995
    
    try:
        res_p = minimize(obj_slscp, x0, method='SLSQP', bounds=bounds_full,
                         constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
        if res_p.success:
            c_val = constr_slscp(res_p.x)
            if np.min(c_val) >= -1e-8:
                s_val = -res_p.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_centers = np.column_stack((res_p.x[0::3], res_p.x[1::3]))
                    best_radii = res_p.x[2::3]
    except Exception:
        pass
        
    # Phase 3: Perturbation Refinement to escape minor local minima
    rng = np.random.RandomState(42)
    for _ in range(15):
        x0_p = np.zeros(3 * N)
        x0_p[0::3] = best_centers[:, 0] + rng.uniform(-0.001, 0.001, N)
        x0_p[1::3] = best_centers[:, 1] + rng.uniform(-0.001, 0.001, N)
        x0_p[2::3] = best_radii * 0.99
        x0_p = np.clip(x0_p, 0.001, 0.999)
        
        try:
            res_p2 = minimize(obj_slscp, x0_p, method='SLSQP', bounds=bounds_full,
                              constraints=cons, options={'maxiter': 2000, 'ftol': 1e-13})
            if res_p2.success:
                if np.min(constr_slscp(res_p2.x)) >= -1e-8:
                    s_val = -res_p2.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_centers = np.column_stack((res_p2.x[0::3], res_p2.x[1::3]))
                        best_radii = res_p2.x[2::3]
        except Exception:
            pass
            
    return best_centers, best_radii, float(best_sum)
