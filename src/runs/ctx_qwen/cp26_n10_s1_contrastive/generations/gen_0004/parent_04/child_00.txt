import numpy as np
from scipy.optimize import linprog, minimize

N = 26
PAIRS = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    x = centers[:, 0]
    y = centers[:, 1]
    bounds_dist = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    A_ub.append(np.eye(n))
    b_ub.append(bounds_dist)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    n_pairs = len(PAIRS[0])
    A_ub_pair = np.zeros((n_pairs, n))
    A_ub_pair[np.arange(n_pairs), PAIRS[0]] = 1.0
    A_ub_pair[np.arange(n_pairs), PAIRS[1]] = 1.0
    b_ub_pair = dists[PAIRS]
    
    A_ub.append(A_ub_pair)
    b_ub.append(b_ub_pair)
    
    A_ub = np.vstack(A_ub)
    b_ub = np.concatenate(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def obj_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def cons_joint(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    x, y, r = v[0::3], v[1::3], v[2::3]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(dist2[PAIRS] - rs[PAIRS]**2)
    return np.concatenate(c)

def generate_inits():
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5], [5,5,5,6,5]]
    
    for p in patterns:
        pts = []
        r_est = 0.095
        y = r_est
        row = 0
        for cnt in p:
            x_start = r_est if row % 2 == 0 else 2.0 * r_est
            for _ in range(cnt):
                pts.append([x_start, y])
                x_start += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
        pts = np.array(pts[:N])
        inits.append(pts)
        
        for angle in [0.05, -0.05, 0.1, -0.1, 0.2]:
            c, s = np.cos(angle), np.sin(angle)
            R = np.array([[c, -s], [s, c]])
            p_rot = (pts - 0.5) @ R.T + 0.5
            inits.append(p_rot)
            
    for seed in range(5):
        rng = np.random.RandomState(seed)
        pts = rng.rand(N, 2)
        for _ in range(200):
            f = np.zeros_like(pts)
            diff = pts[:, None, :] - pts[None, :, :]
            dist = np.sqrt(np.sum(diff**2, axis=2)) + 1e-5
            f += np.sum((1.0/dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
            for d in range(2):
                f[:, d] += 10.0 * (pts[:, d] < 0.05) * (0.05 - pts[:, d])
                f[:, d] -= 10.0 * (pts[:, d] > 0.95) * (pts[:, d] - 0.95)
            pts += 0.01 * f
            pts = np.clip(pts, 0.01, 0.99)
        inits.append(pts)
        
    return inits

def run_packing():
    inits = generate_inits()
    
    init_sums = []
    for pts in inits:
        _, s = solve_radii_lp(pts)
        init_sums.append((s, pts))
    init_sums.sort(key=lambda x: x[0], reverse=True)
    
    best_global_sum = 0.0
    best_global_centers = None
    best_global_radii = None
    
    # Phase 1: Simulated Annealing on centers (top 5 inits)
    for s, init_centers in init_sums[:5]:
        centers = init_centers.copy()
        radii, curr_sum = solve_radii_lp(centers)
        if curr_sum > best_global_sum:
            best_global_sum = curr_sum
            best_global_centers = centers.copy()
            best_global_radii = radii.copy()
            
        rng = np.random.RandomState(int(curr_sum * 10000) % (2**31))
        step = 0.025
        temp = 0.005
        for it in range(3000):
            idx = rng.randint(N)
            old_pos = centers[idx].copy()
            centers[idx] += rng.normal(0, step, 2)
            centers[idx] = np.clip(centers[idx], 1e-4, 1.0-1e-4)
            
            _, new_sum = solve_radii_lp(centers)
            
            delta = new_sum - curr_sum
            if delta > 0 or rng.rand() < np.exp(delta / max(temp, 1e-6)):
                curr_sum = new_sum
                if curr_sum > best_global_sum:
                    best_global_sum = curr_sum
                    best_global_centers = centers.copy()
                    _, best_global_radii = solve_radii_lp(centers)
            else:
                centers[idx] = old_pos
                
            temp *= 0.999
            step *= 0.9998
            
    # Phase 2: Joint SLSQP polish
    if best_global_centers is not None:
        centers = best_global_centers
        radii = best_global_radii
        
        v0 = np.zeros(3*N)
        v0[0::3] = centers[:, 0]
        v0[1::3] = centers[:, 1]
        v0[2::3] = radii * 0.99
        
        bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
        cons = {'type': 'ineq', 'fun': cons_joint}
        
        try:
            res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                c_val = cons_joint(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_global_sum:
                        best_global_sum = s_val
                        best_global_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_global_radii = res.x[2::3]
        except Exception:
            pass
            
        for _ in range(15):
            v0_p = np.concatenate([best_global_centers.flatten(), best_global_radii])
            v0_p += np.random.randn(3*N) * 0.003
            v0_p[0::3] = np.clip(v0_p[0::3], 0.01, 0.99)
            v0_p[1::3] = np.clip(v0_p[1::3], 0.01, 0.99)
            v0_p[2::3] = np.clip(v0_p[2::3], 1e-5, 0.49)
            
            try:
                res = minimize(obj_joint, v0_p, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 4000, 'ftol': 1e-13})
                if res.success and np.min(cons_joint(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_global_sum:
                        best_global_sum = s_val
                        best_global_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_global_radii = res.x[2::3]
            except Exception:
                pass
                
    if best_global_centers is None:
        best_global_centers = np.tile([[0.5, 0.5]], (N, 1))
        best_global_centers += np.random.rand(N, 2) * 0.1 - 0.05
        best_global_radii = np.full(N, 0.05)
        best_global_sum = np.sum(best_global_radii)
        
    return best_global_centers, best_global_radii, float(best_global_sum)
