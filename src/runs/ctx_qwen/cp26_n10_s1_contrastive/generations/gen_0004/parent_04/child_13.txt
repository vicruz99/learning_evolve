import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 0.0

def obj_centers(c_flat):
    """Objective for center optimization: minimize negative LP sum of radii."""
    centers = c_flat.reshape(N, 2)
    _, s = solve_radii_lp(centers)
    return -s

def hex_init(pattern, seed, scale=1.0, rot=0.0):
    """Generate hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.09
    y = r_est
    row = 0
    for cnt in pattern:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def force_init(seed):
    """Force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    for _ in range(400):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        f += np.sum((1.0/dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        
        wall = 15.0
        for d in range(2):
            f[:, d] += wall * (pts[:, d] < 0.1) * (0.1 - pts[:, d])
            f[:, d] -= wall * (pts[:, d] > 0.9) * (pts[:, d] - 0.9)
        pts += 0.008 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def safe_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    d = 1.0 - 2.0 * r
    d = np.clip(d, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def constr(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def run_packing():
    bounds = [(1e-5, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constr}
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    inits = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5]]
    for p in patterns:
        for s in range(5):
            inits.append(hex_init(p, s, scale=0.95 + 0.1*(s%3), rot=0.05*(s-2)))
    for s in range(10):
        inits.append(force_init(s))
    for s in range(10):
        rng = np.random.RandomState(s)
        pts = rng.rand(N, 2)
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # Phase 1: Optimize centers using LP objective with Powell
    for pts in inits:
        c0 = pts.flatten()
        try:
            res = minimize(obj_centers, c0, method='Powell', 
                           options={'maxiter': 1500, 'ftol': 1e-12, 'xtol': 1e-12})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    best_radii, _ = solve_radii_lp(best_centers)
    
    # Phase 2: SLSQP refinement on parameterized space
    p0 = safe_params(best_centers, best_radii * 0.995)
    try:

                       options={'maxiter': 5000, 'ftol': 1e-14})
        if res.success:
            if np.min(constr(res.x)) >= -1e-7:
                best_sum = -res.fun
                best_radii = res.x[:N]
                best_centers = np.column_stack((
                    res.x[:N] + res.x[N:2*N]*(1.0 - 2.0*res.x[:N]),
                    res.x[:N] + res.x[2*N:3*N]*(1.0 - 2.0*res.x[:N])
                ))
    except Exception:
        pass
        
    # Phase 3: Perturbation and hill-climbing to escape local minima
    rng = np.random.RandomState(42)
    for _ in range(25):
        c_pert = best_centers.copy()
        c_pert += rng.uniform(-0.025, 0.025, c_pert.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        
        try:
            res = minimize(obj_centers, c_pert.flatten(), method='Powell',
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
                best_radii, _ = solve_radii_lp(best_centers)
        except Exception:
            pass
            
        if -obj_centers(best_centers.flatten()) > best_sum - 1e-6:
            p_pert = safe_params(best_centers, best_radii * 0.995)
            try:
                res_s = minimize(obj, p_pert, method='SLSQP', bounds=bounds, constraints=cons,
                                 options={'maxiter': 3000, 'ftol': 1e-13})
                if res_s.success and np.min(constr(res_s.x)) >= -1e-7:
                    s_val = -res_s.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_radii = res_s.x[:N]
                        best_centers = np.column_stack((
                            res_s.x[:N] + res_s.x[N:2*N]*(1.0 - 2.0*res_s.x[:N]),
                            res_s.x[:N] + res_s.x[2*N:3*N]*(1.0 - 2.0*res_s.x[:N])
                        ))
            except Exception:
                pass

    # Phase 4: Final high-precision polish
    p_final = safe_params(best_centers, best_radii * 0.998)
    try:
        res_f = minimize(obj, p_final, method='SLSQP', bounds=bounds, constraints=cons,

                res_f.x[:N] + res_f.x[N:2*N]*(1.0 - 2.0*res_f.x[:N]),
                res_f.x[:N] + res_f.x[2*N:3*N]*(1.0 - 2.0*res_f.x[:N])
            ))
    except Exception:
        pass

    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
