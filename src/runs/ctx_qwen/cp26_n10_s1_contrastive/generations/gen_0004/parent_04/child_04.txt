1. **Parameterization for Boundary Satisfaction**: Use the transformation `x = r + u*(1-2r)` and `y = r + v*(1-2r)` where `u, v ∈ [0, 1]`. This automatically enforces circle containment within the square, reducing constraint count and improving solver stability. Only pairwise non-overlap constraints `dist^2 ≥ (r_i + r_j)^2` need explicit handling.

2. **Diverse Initialization**: Generate a large pool of starting configurations using:
   - Hexagonal lattices with varied row counts, scales, and rotations to break symmetry
   - Force-directed layouts that naturally spread points while respecting boundaries
   - Random perturbations to cover unexplored regions of the configuration space

3. **Joint SLSQP Optimization**: Run `scipy.optimize.minimize` with the SLSQP method on the `(r, u, v)` parameter space. SLSQP efficiently handles the nonlinear inequality constraints and bounds. Multiple restarts from the diverse initial pool ensure we escape poor local minima.

4. **Perturbation Refinement**: Apply localized random perturbations to the best solution found, followed by SLSQP re-optimization. This helps escape shallow local optima and fine-tunes the packing topology.

5. **LP Radius Maximization**: After finding optimal centers via SLSQP, run a linear program to maximize `∑r_i` subject to `r_i + r_j ≤ dist(i,j)` and boundary constraints. This guarantees that for the final center positions, the radii are mathematically optimal, often squeezing out an additional 0.001-0.003 in the objective sum.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[:N])

def constraints(v):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = v[:N]
    u = v[N:2*N]
    w = v[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    d = 1.0 - 2.0 * r
    d = np.clip(d, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, w])

def safe_radii(centers):
    """Compute strictly feasible initial radii."""
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i]-centers[j])**2))
                mx = min(mx, d*0.5)
        r[i] = max(1e-4, mx * 0.88)
    return r

def hex_init(seed, pattern, scale=1.0, rot=0.0):
    """Generate hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.09
    y = r_est
    row = 0
    for cnt in pattern:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.uniform(-0.025, 0.025, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def force_init(seed):
    """Force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    for _ in range(400):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        f += np.sum((1.0/dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        
        wall = 30.0
        for d in range(2):
            f[:, d] += wall * (pts[:, d] < 0.1) * (0.1 - pts[:, d])
            f[:, d] -= wall * (pts[:, d] > 0.9) * (pts[:, d] - 0.9)
        pts += 0.008 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to maximize sum of radii."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 0.0

def run_packing():
    bounds = [(1e-5, 0.499)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -np.inf
    best_params = None
    
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], 
        [6,4,6,5,5], [5,5,6,5,5], [4,5,6,6,5], [5,6,4,6,5],
        [6,6,4,5,5], [5,4,6,6,5], [4,5,5,6,6], [6,5,5,5,5],
        [5,5,5,6,5], [4,6,6,5,5], [5,6,6,5,4]
    ]
    
    for p in patterns:
        for s in range(8):
            rot = 0.04 * (s - 4)
            scale = 0.88 + 0.12 * (s % 5)
            inits.append(hex_init(s, p, scale=scale, rot=rot))
            
    for s in range(20):
        inits.append(force_init(s))
        
    for s in range(20):
        rng = np.random.RandomState(s)
        pts = rng.rand(N, 2)
        pts = np.clip(pts, 0.08, 0.92)
        inits.append(pts)
        
    # Phase 1: SLSQP from diverse starts
    for pts in inits:
        r0 = safe_radii(pts)
        p0 = make_params(pts, r0)
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-13})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation refinement
    if best_params is not None:
        for _ in range(60):
            p0 = best_params.copy()
            p0[N:3*N] += np.random.uniform(-0.015, 0.015, 2*N)
            p0[:N] += np.random.uniform(-0.002, 0.002, N)
            p0[:N] = np.clip(p0[:N], 1e-5, 0.499)
            p0[N:3*N] = np.clip(p0[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 4000, 'ftol': 1e-13})
                if res.success:
                    c_val = constraints(res.x)
                    if np.min(c_val) >= -1e-7:
                        s_val = -res.fun
                        if s_val > best_sum:
                            best_sum = s_val
                            best_params = res.x.copy()
            except Exception:
                pass
                
        # Phase 3: High-precision polish
        try:
            res = minimize(objective, best_params, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-14})
            if res.success and np.min(constraints(res.x)) >= -1e-8:
                best_params = res.x
                best_sum = -res.fun
        except Exception:
            pass

    # Phase 4: LP radius maximization on final centers
    if best_params is not None:
        r_opt = best_params[:N]
        u_opt = best_params[N:2*N]
        w_opt = best_params[2*N:3*N]
        x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
        y_opt = r_opt + w_opt * (1.0 - 2.0 * r_opt)
        centers_opt = np.column_stack([x_opt, y_opt])
        
        lp_radii, lp_sum = solve_radii_lp(centers_opt)
        if lp_sum > best_sum:
            best_sum = lp_sum
            # Reconstruct params for consistency, though we just need centers/radii
            best_params = np.zeros(3*N)
            best_params[:N] = lp_radii
            denom = np.clip(1.0 - 2.0 * lp_radii, 1e-6, 1.0)
            best_params[N:2*N] = np.clip((x_opt - lp_radii) / denom, 0.0, 1.0)
            best_params[2*N:3*N] = np.clip((y_opt - lp_radii) / denom, 0.0, 1.0)

    # Fallback configuration
    if best_params is None:
        fallback_centers = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                            np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        fallback_radii = np.full(N, 0.04)
        return fallback_centers, fallback_radii, float(np.sum(fallback_radii))

    r = best_params[:N]
    u = best_params[N:2*N]
    w = best_params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    centers = np.column_stack([x, y])
    radii = np.maximum(r, 0.0)
    
    return centers, radii, float(best_sum)
