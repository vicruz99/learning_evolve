import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = max(1e-9, x); k += 1
        A_ub[k, i] = 1.0; b_ub[k] = max(1e-9, 1.0 - x); k += 1
        A_ub[k, i] = 1.0; b_ub[k] = max(1e-9, y); k += 1
        A_ub[k, i] = 1.0; b_ub[k] = max(1e-9, 1.0 - y); k += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_ub[k, i] = 1.0; A_ub[k, j] = 1.0; b_ub[k] = max(1e-9, d); k += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    r_fallback = np.full(n, 1e-4)
    return r_fallback, 2.6e-3

def get_lp_sum(centers):
    return solve_radii_lp(centers)[1]

def hex_init(seed, pattern, scale=1.0, rot=0.0):
    """Generate hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.09
    y = r_est
    row = 0
    for cnt in pattern:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    for _ in range(400):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        f += np.sum((1.0/dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        wall = 15.0
        for d in range(2):
            f[:, d] += wall * (pts[:, d] < 0.1) * (0.1 - pts[:, d])
            f[:, d] -= wall * (pts[:, d] > 0.9) * (pts[:, d] - 0.9)
        pts += 0.004 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def optimize_centers_sa(centers, max_iter=22000, T0=0.06):
    """Simulated Annealing on centers using LP for objective evaluation."""
    best_centers = centers.copy()
    best_sum = get_lp_sum(best_centers)
    curr_centers = best_centers.copy()
    curr_sum = best_sum
    
    rng = np.random.RandomState(123)
    sigma = 0.045
    T = T0
    
    for i in range(max_iter):
        if rng.rand() < 0.65:
            idx = rng.randint(N)
            new_centers = curr_centers.copy()
            new_centers[idx] += rng.normal(0, sigma, 2)
            new_centers[idx] = np.clip(new_centers[idx], 0.01, 0.99)
        else:
            idx = rng.choice(N, 2, replace=False)
            new_centers = curr_centers.copy()
            new_centers[idx] += rng.normal(0, sigma, (2, 2))
            new_centers[idx] = np.clip(new_centers[idx], 0.01, 0.99)
            
        new_sum = get_lp_sum(new_centers)
        delta = new_sum - curr_sum
        
        if delta > 0 or (T > 1e-6 and rng.rand() < np.exp(delta / T)):
            curr_centers = new_centers
            curr_sum = new_sum
            if curr_sum > best_sum:
                best_centers = curr_centers.copy()
                best_sum = curr_sum
                
        if i % 600 == 0:
            sigma *= 0.955
            T *= 0.915
            
    return best_centers, best_sum

def obj_param(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[:N])

def cons_param(v):
    """Inequality constraints for SLSQP: pairwise non-overlap."""
    r = v[:N]
    u = v[N:2*N]
    v_coord = v[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v_coord * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    return d2[mask] - rs[mask]**2

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5],
        [7,5,5,5,4], [4,5,5,5,7], [6,6,5,5,4], [5,5,6,5,5], [6,5,5,6,4],
        [5,7,5,5,4], [4,5,7,5,5], [6,5,6,4,5], [5,6,4,6,5], [4,6,5,5,6]
    ]
    
    inits = []
    for p in patterns:
        for s in range(5):
            inits.append(hex_init(s, p, scale=0.95 + 0.1*(s%3), rot=0.04*(s-2)))
    for s in range(20):
        inits.append(force_init(s))
    for s in range(15):
        rng = np.random.RandomState(s)
        pts = rng.rand(N, 2)
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    candidates = []
    for pts in inits:
        r, s = solve_radii_lp(pts)
        candidates.append((pts, r, s))
        
    candidates.sort(key=lambda x: x[2], reverse=True)
    
    # Phase 1: Deep SA refinement on top candidates
    for i in range(min(7, len(candidates))):
        c0, _, _ = candidates[i]
        c_opt, s_opt = optimize_centers_sa(c0, max_iter=22000)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            best_radii, _ = solve_radii_lp(c_opt)
            
    # Phase 2: High-precision SLSQP polish with boundary-parameterization
    if best_centers is not None:
        r0 = best_radii.copy()
        denom = np.clip(1.0 - 2.0 * r0, 1e-6, 1.0)
        u0 = np.clip((best_centers[:, 0] - r0) / denom, 0.0, 1.0)
        v0_coord = np.clip((best_centers[:, 1] - r0) / denom, 0.0, 1.0)
        
        v_init = np.concatenate([r0, u0, v0_coord])
        
        bounds_param = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
        cons = {'type': 'ineq', 'fun': cons_param}
        
        try:
            res = minimize(obj_param, v_init, method='SLSQP', bounds=bounds_param, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-14})
            if res.success:
                c_val = cons_param(res.x)
                if np.min(c_val) >= -1e-7:
                    r_opt = res.x[:N]
                    u_opt = res.x[N:2*N]
                    v_opt = res.x[2*N:3*N]
                    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
                    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
                    best_centers = np.column_stack((x_opt, y_opt))
                    best_radii = r_opt
                    best_sum = np.sum(best_radii)
        except Exception:
            pass

    # Fallback (should rarely be reached)
    if best_centers is None:
        best_centers = np.random.rand(N, 2)
        best_radii = np.full(N, 0.05)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
