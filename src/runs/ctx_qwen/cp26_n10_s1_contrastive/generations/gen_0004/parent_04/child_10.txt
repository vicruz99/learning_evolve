import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-4

def center_objective(centers_flat):
    """Objective for center optimization: negative LP max sum of radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, s = solve_radii_lp(centers)
    return -s

def make_hex_centers(seed, pattern, scale=1.0, rot=0.0):
    """Generate hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.09
    y = r_est
    row = 0
    for cnt in pattern:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force_centers(seed):
    """Force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    for _ in range(600):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        f += np.sum((1.0 / dist**2)[:, :, None] * diff / dist[:, :, None], axis=1)
        pts += 0.004 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    d = 1.0 - 2.0 * r
    d = np.clip(d, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def slsqp_obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def slsqp_constr(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def run_packing():
    bounds_nm = [(0.02, 0.98)] * (2 * N)
    best_sum = -np.inf
    best_centers = None
    
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5]]
    inits = []
    for p in patterns:
        for s in range(8):
            inits.append(make_hex_centers(s, p, scale=0.85 + 0.2*(s%4), rot=0.15*(s-2)))
    for s in range(25):
        inits.append(make_force_centers(s))
    for s in range(20):
        rng = np.random.RandomState(s)
        inits.append(np.clip(rng.rand(N, 2), 0.02, 0.98))
        
    # Phase 1: Nelder-Mead on centers using LP objective
    for pts in inits:
        res_nm = minimize(center_objective, pts.flatten(), method='Nelder-Mead',
                          bounds=bounds_nm, options={'maxiter': 3000, 'xatol': 1e-6, 'fatol': 1e-8})
        if -res_nm.fun > best_sum:
            best_sum = -res_nm.fun
            best_centers = res_nm.x.reshape(N, 2)
            
    if best_centers is None:
        best_centers = make_hex_centers(0, [6,5,6,5,4])
        
    # Phase 2: Polish with SLSQP using parameterization
    radii_init, _ = solve_radii_lp(best_centers)
    params0 = make_params(best_centers, radii_init)
    
    bounds_sl = [(1e-5, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': slsqp_constr}
    
    res_sl = minimize(slsqp_obj, params0, method='SLSQP', bounds=bounds_sl,
                      constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
    
    best_params = params0
    if res_sl.success:
        c_val = slsqp_constr(res_sl.x)
        if np.min(c_val) >= -1e-7:
            if -res_sl.fun > best_sum:
                best_sum = -res_sl.fun
                best_params = res_sl.x.copy()
                
    # Phase 3: Perturbation refinement
    for _ in range(20):
        p0 = best_params.copy()
        p0[N:3*N] += np.random.uniform(-0.006, 0.006, 2*N)
        p0[:N] += np.random.uniform(-0.0008, 0.0008, N)
        p0[:N] = np.clip(p0[:N], 1e-5, 0.5)
        p0[N:3*N] = np.clip(p0[N:3*N], 0.0, 1.0)
        
        res_p = minimize(slsqp_obj, p0, method='SLSQP', bounds=bounds_sl,
                         constraints=cons, options={'maxiter': 4000, 'ftol': 1e-14})
        if res_p.success and np.min(slsqp_constr(res_p.x)) >= -1e-7:
            if -res_p.fun > best_sum:
                best_sum = -res_p.fun
                best_params = res_p.x.copy()
                
    r = best_params[:N]
    u = best_params[N:2*N]
    v = best_params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    centers = np.column_stack([x, y])
    radii = np.maximum(r, 0.0)
    
    return centers, radii, float(best_sum)
