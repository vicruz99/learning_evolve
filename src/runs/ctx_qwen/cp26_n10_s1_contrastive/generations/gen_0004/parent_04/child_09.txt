import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def obj_centers(v):
    """Objective for center optimization: negative sum of LP-optimized radii."""
    return -solve_radii_lp(v.reshape(N, 2))[1]

def make_hex_init(seed, counts, rot=0.0, scale=1.0):
    """Generate hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.09
    y = r_est
    row = 0
    for cnt in counts:
        x_start = r_est if row % 2 == 0 else 2.0 * r_est
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def joint_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def joint_constr(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def run_packing():
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Phase 1: Diverse initializations
    inits = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], 
                [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6]]
    for p in patterns:
        for s in range(5):
            inits.append(make_hex_init(s, p, rot=0.06*(s-2), scale=0.95 + 0.04*(s%3)))
            
    for s in range(15):
        rng = np.random.RandomState(s)
        pts = rng.rand(N, 2) * 0.9 + 0.05
        inits.append(pts)
        
    # Phase 2: Optimize centers using Nelder-Mead on LP radii sum
    for pts in inits:
        try:
            res = minimize(obj_centers, pts.flatten(), method='Nelder-Mead',

            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    if best_centers is None:
        best_centers = inits[0]
        
    best_radii, best_sum = solve_radii_lp(best_centers)
    
    # Phase 3: Joint SLSQP polishing
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': joint_constr}
    
    v0 = np.zeros(3 * N)
    v0[0::3] = best_centers[:, 0]
    v0[1::3] = best_centers[:, 1]
    v0[2::3] = best_radii * 0.99
    
    try:
        res = minimize(joint_obj, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(joint_constr(res.x)) >= -1e-7:
            best_centers = np.column_stack((res.x[0::3], res.x[1::3]))

            best_sum = -res.fun
    except Exception:
        pass

        r_pert, s_pert = solve_radii_lp(c_pert)

            best_centers = c_pert
            best_radii = r_pert

                pass

    return best_centers, best_radii, float(best_sum)
