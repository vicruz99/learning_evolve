import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[0::3])

def constraints(vars):
    """
    Constraint function: ensures pairwise non-overlap.
    Boundary constraints are inherently satisfied by the (u, v) parameterization.
    Returns array of constraint values >= 0.
    """
    r = vars[0::3]
    u = vars[1::3]
    v = vars[2::3]
    
    # Transform normalized coordinates to actual positions
    # This guarantees r <= x <= 1-r and r <= y <= 1-r automatically
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    # Pairwise squared distances for i < j
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    
    dist_sq = dx[I_IDX, J_IDX]**2 + dy[I_IDX, J_IDX]**2
    r_sum_sq = (r[I_IDX] + r[J_IDX])**2
    
    return dist_sq - r_sum_sq

def run_packing():
    n = N_CIRCLES
    best_sum = -np.inf
    best_vars = None
    
    # Bounds for (r, u, v)
    bounds = [(1e-4, 0.5), (0.0, 1.0), (0.0, 1.0)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice layout
    r0 = 0.08
    pts1 = []
    rows = [5, 6, 5, 6, 4]
    y = r0
    for row_idx, count in enumerate(rows):
        x_off = r0 if row_idx % 2 == 0 else 2 * r0
        for col in range(count):
            pts1.append([x_off + col * 2 * r0, y])
        y += r0 * np.sqrt(3)
    pts1 = np.array(pts1[:n])
    inits.append(pts1)
    
    # 2. 5x5 Grid + center
    pts2 = []
    for i in range(5):
        for j in range(5):
            pts2.append([0.1 + i*0.2, 0.1 + j*0.2])
    pts2.append([0.5, 0.5])
    pts2 = np.array(pts2[:n])
    inits.append(pts2)
    
    # 3-5. Random feasible configurations in (x,y) space
    np.random.seed(42)
    for _ in range(3):
        r_rand = np.random.uniform(0.05, 0.15, n)
        u_rand = np.random.uniform(0.0, 1.0, n)
        v_rand = np.random.uniform(0.0, 1.0, n)
        pts = np.column_stack((r_rand + u_rand*(1.0 - 2.0*r_rand), 
                               r_rand + v_rand*(1.0 - 2.0*r_rand)))
        inits.append(pts)
        
    # Run optimization with multiple restarts and perturbations
    for seed in range(15):
        np.random.seed(seed)
        for p in inits:
            # Perturb centers slightly to break symmetry
            p_jit = p + np.random.randn(n, 2) * 0.015
            p_jit = np.clip(p_jit, 0.05, 0.95)
            
            vars0 = np.zeros(3 * n)
            base_r = 0.07 + np.random.uniform(-0.01, 0.01)
            base_r = np.clip(base_r, 0.05, 0.12)
            
            for i in range(n):
                r = base_r
                # Ensure initial centers are strictly within [r, 1-r]
                x = np.clip(p_jit[i, 0], r, 1.0 - r)
                y = np.clip(p_jit[i, 1], r, 1.0 - r)
                denom = 1.0 - 2.0 * r
                u = (x - r) / denom
                v = (y - r) / denom
                vars0[3*i] = r
                vars0[3*i+1] = np.clip(u, 0.0, 1.0)
                vars0[3*i+2] = np.clip(v, 0.0, 1.0)
                
            try:
                res = minimize(
                    objective, 
                    vars0, 
                    method='SLSQP', 
                    bounds=bounds,
                    constraints=cons, 
                    options={'maxiter': 2500, 'ftol': 1e-13}
                )
                if res.success:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_vars = res.x.copy()
            except Exception:
                continue
                
    if best_vars is None:
        # Fallback to a valid small packing if optimization fails
        centers = np.column_stack((np.full(n, 0.5), np.full(n, 0.5)))
        radii = np.full(n, 0.01)
        return centers, radii, 0.26
        
    # Reconstruct centers from optimized parameters
    r = best_vars[0::3]
    u = best_vars[1::3]
    v = best_vars[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    centers = np.column_stack((x, y))
    radii = r
    
    # Safety clipping for numerical drift
    radii = np.maximum(radii, 1e-6)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    
    return centers, radii, float(np.sum(radii))
