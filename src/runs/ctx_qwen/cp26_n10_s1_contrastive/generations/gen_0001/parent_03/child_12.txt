import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """
    Computes inequality constraints for boundary and separation.
    Returns array of values that must be >= 0.
    """
    C = vars.reshape(N_CIRCLES, 3)
    x = C[:, 0]
    y = C[:, 1]
    r = C[:, 2]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c_boundary = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Separation constraints: dist_sq - (ri + rj)^2 >= 0
    i_idx, j_idx = np.triu_indices(N_CIRCLES, k=1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    r_sum = r[i_idx] + r[j_idx]
    c_sep = dx*dx + dy*dy - r_sum*r_sum
    
    return np.concatenate([c_boundary, c_sep])

def get_hex_init():
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_init = 0.04
    y = r_init
    row = 0
    while len(pts) < N_CIRCLES:
        x = r_init + (row % 2) * r_init
        while x <= 1.0 - r_init and len(pts) < N_CIRCLES:
            pts.append([x, y, r_init])
            x += 2.0 * r_init
        y += np.sqrt(3) * r_init
        row += 1
    return np.array(pts).flatten()

def get_grid_init():
    """Generates a 5x5 grid + center initialization."""
    pts = []
    r_init = 0.04
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + 0.2*i, 0.1 + 0.2*j, r_init])
    pts.append([0.5, 0.5, r_init])
    return np.array(pts).flatten()

def get_force_init():
    """Generates a force-directed spread initialization."""
    pts = np.random.rand(N_CIRCLES, 2) * 0.8 + 0.1
    for _ in range(300):
        forces = np.zeros_like(pts)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                diff = pts[i] - pts[j]
                dist = np.linalg.norm(diff)
                if dist < 1e-5:
                    dist = 1e-5
                    diff = np.random.rand(2) * 1e-5
                # Repulsion force ~ 1/d^2
                f = diff / (dist**3)
                forces[i] += f
                forces[j] -= f
            # Wall repulsion
            margin = 0.05
            if pts[i, 0] < margin: forces[i, 0] += (margin - pts[i, 0])
            if pts[i, 0] > 1.0 - margin: forces[i, 0] -= (pts[i, 0] - (1.0 - margin))
            if pts[i, 1] < margin: forces[i, 1] += (margin - pts[i, 1])
            if pts[i, 1] > 1.0 - margin: forces[i, 1] -= (pts[i, 1] - (1.0 - margin))
        pts += forces * 0.01
        pts = np.clip(pts, 0.05, 0.95)
    r_init = 0.04
    return np.column_stack([pts, r_init]).flatten()

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_vars = None
    
    # Try multiple diverse initial configurations
    inits = [get_hex_init(), get_grid_init(), get_force_init()]
    
    for init in inits:
        res = minimize(objective, init, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-12})
        val = -res.fun
        # Check feasibility (allow tiny numerical tolerance)
        cons_val = np.min(constraints(res.x))
        if val > best_sum and cons_val >= -1e-7:
            best_sum = val
            best_vars = res.x.copy()
            
    # Local refinement via perturbation to escape local minima
    if best_vars is not None:
        for it in range(20):
            scale = 0.006 / (1.0 + it * 0.4)
            perturbed = best_vars + np.random.randn(len(best_vars)) * scale
            # Clip to bounds
            for k in range(0, len(perturbed), 3):
                perturbed[k] = np.clip(perturbed[k], 0.0, 1.0)
                perturbed[k+1] = np.clip(perturbed[k+1], 0.0, 1.0)
                perturbed[k+2] = np.clip(perturbed[k+2], 1e-6, 0.5)
                
            res = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-12})
            val = -res.fun
            cons_val = np.min(constraints(res.x))
            if val > best_sum and cons_val >= -1e-7:
                best_sum = val
                best_vars = res.x.copy()
                
    # Fallback safety check
    if best_vars is None:
        best_vars = get_grid_init()
        
    centers = best_vars.reshape(N_CIRCLES, 3)[:, :2]
    radii = best_vars.reshape(N_CIRCLES, 3)[:, 2]
    return centers, radii, float(np.sum(radii))
