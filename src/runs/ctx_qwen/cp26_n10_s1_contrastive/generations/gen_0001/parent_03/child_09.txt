import numpy as np
from scipy.optimize import minimize

def compute_objective(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def compute_constraints(x):
    """Compute all inequality constraints: boundary and separation."""
    N = 26
    C = x.reshape(N, 3)
    x_c = C[:, 0]
    y_c = C[:, 1]
    r = C[:, 2]
    
    c_list = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c_list.append(x_c - r)
    c_list.append(1.0 - x_c - r)
    c_list.append(y_c - r)
    c_list.append(1.0 - y_c - r)
    
    # Separation constraints: dist_sq >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = x_c[i_idx] - x_c[j_idx]
    dy = y_c[i_idx] - y_c[j_idx]
    r_sum = r[i_idx] + r[j_idx]
    c_list.append(dx*dx + dy*dy - r_sum*r_sum)
    
    return np.concatenate(c_list)

def get_safe_radii(centers):
    """Compute strictly feasible initial radii for given centers."""
    n = centers.shape[0]
    r = np.full(n, 0.05)
    for i in range(n):
        # Distance to boundaries
        min_d = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        # Distance to other circles
        for j in range(n):
            if i == j: continue
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if d < min_d:
                min_d = d
        # Set radius slightly less than half the minimum clearance
        r[i] = min_d / 2.2
    return r

def run_optimization(centers_init):
    """Run SLSQP optimization from a given set of initial centers."""
    n = 26
    r_init = get_safe_radii(centers_init)
    
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = r_init[i]
        
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    res = minimize(
        compute_objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': compute_constraints},
        options={'maxiter': 2000, 'ftol': 1e-12}
    )
    
    if res.success:
        return res.x, -res.fun
    return None, 0.0
