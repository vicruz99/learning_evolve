import numpy as np
from scipy.optimize import minimize, linprog

def objective_joint(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_joint(x):
    """Compute all inequality constraints: boundary and separation."""
    n = 26
    x_c = x[0::3]
    y_c = x[1::3]
    r = x[2::3]
    
    cons = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    cons.append(x_c - r)
    cons.append(1.0 - x_c - r)
    cons.append(y_c - r)
    cons.append(1.0 - y_c - r)
    
    # Separation constraints: dist_sq >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = x_c[i_idx] - x_c[j_idx]
    dy = y_c[i_idx] - y_c[j_idx]
    r_sum = r[i_idx] + r[j_idx]
    cons.append(dx*dx + dy*dy - r_sum*r_sum)
    
    return np.concatenate(cons)

def solve_radii_lp(centers):
    """
    Given fixed centers, solve an LP to maximize the sum of radii.
    Constraints ensure non-overlap and boundary containment.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Maximize sum(r) <=> Minimize -sum(r)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= dist_to_boundary
    for i in range(n):
        x, y = centers[i]
        bounds_i = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_i:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds_r = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_vars = [(0, 1), (0, 1), (0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints_joint}

    inits = []

    hex_pts = []
    r_est = 0.105
    y = r_est
    row = 0

        x = r_est + x_off

            x += 2 * r_est

        row += 1
    hex_pts = np.array(hex_pts[:n])
    inits.append(hex_pts)
    
    # 2. Perturbed hexagonal grids
    for seed in range(8):
        np.random.seed(seed)
        pts = hex_pts + np.random.uniform(-0.03, 0.03, size=(n, 2))
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    # 3. Structured grid (5x5 + center)
    grid_pts = []
    for i in range(5):
        for j in range(5):
            grid_pts.append([0.1 + i*0.2, 0.1 + j*0.2])
    grid_pts.append([0.5, 0.5])
    inits.append(np.array(grid_pts))
    
    # 4. Random placements
    for seed in range(5):
        np.random.seed(50 + seed)
        inits.append(np.random.rand(n, 2))

    # Optimization loop
    for init_centers in inits:
        r_safe = 0.01

        best_centers = inits[0]
        _, best_radii = solve_radii_lp(best_centers)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
