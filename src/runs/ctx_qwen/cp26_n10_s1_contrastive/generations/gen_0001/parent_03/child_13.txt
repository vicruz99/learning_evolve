import numpy as np
from scipy.optimize import minimize

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[2::3])

def constraints(vars_vec):
    """Compute all inequality constraints: boundary and separation."""
    n = 26
    X = vars_vec.reshape(n, 3)
    cx = X[:, 0]
    cy = X[:, 1]
    r = X[:, 2]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c_boundary = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Pairwise separation constraints: dist^2 >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    r_sum = r[i_idx] + r[j_idx]
    c_pair = dx*dx + dy*dy - r_sum*r_sum
    
    return np.concatenate([c_boundary, c_pair])

def run_packing():
    n = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_vars = None
    
    # Multiple restarts with diverse initial configurations
    for seed in range(20):
        rng = np.random.default_rng(seed)
        
        # Alternate between structured hex-grid and random spread
        if seed % 2 == 0:
            pts = []
            y = 0.1
            row = 0
            while len(pts) < n:
                shift = (row % 2) * 0.1
                x = 0.1 + shift
                while x < 0.9:
                    if len(pts) < n:
                        pts.append([x, y])
                    x += 0.2
                y += 0.175
                row += 1
            pts = np.array(pts[:n])
        else:
            pts = rng.uniform(0.15, 0.85, (n, 2))
            
        # Add jitter to break symmetry and ensure strict feasibility
        pts += rng.uniform(-0.025, 0.025, (n, 2))
        pts = np.clip(pts, 0.05, 0.95)
        
        # Flatten initial state: [x0, y0, r0, x1, y1, r1, ...]
        x0 = np.zeros(3 * n)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = 0.03  # Small initial radius to guarantee strict feasibility
        
        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons, 
                options={'maxiter': 2500, 'ftol': 1e-12}
            )
            
            # Verify constraint satisfaction with a tolerance
            if np.min(constraints(res.x)) >= -1e-6:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_vars = res.x.copy()
        except Exception:
            continue
            
    # Final polishing step from the best configuration found
    if best_vars is not None:
        try:
            res_final = minimize(
                objective, 
                best_vars, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons, 
                options={'maxiter': 3000, 'ftol': 1e-14}
            )
            if np.min(constraints(res_final.x)) >= -1e-6:
                best_vars = res_final.x
        except Exception:
            pass
            
        centers = best_vars.reshape(n, 3)[:, :2]
        radii = best_vars.reshape(n, 3)[:, 2]
    else:
        # Fallback to a valid small packing
        centers = np.tile([[0.5, 0.5]], (n, 1))
        radii = np.full(n, 0.01)
        
    return centers, radii, float(np.sum(radii))
