import numpy as np
from scipy.optimize import minimize

N = 26

def objective_fn(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_fn(x):
    """Compute all inequality constraints: boundary and separation."""
    X = x[0::3]
    Y = x[1::3]
    R = x[2::3]
    
    c = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c.append(X - R)
    c.append(1.0 - X - R)
    c.append(Y - R)
    c.append(1.0 - Y - R)
    
    # Separation constraints: dist_sq >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = X[i_idx] - X[j_idx]
    dy = Y[i_idx] - Y[j_idx]
    r_sum = R[i_idx] + R[j_idx]
    c.append(dx**2 + dy**2 - r_sum**2)
    
    return np.concatenate(c)

def compute_feasible_radii(pts):
    """Compute a strictly feasible radius for each point based on distances."""
    r = np.full(N, 1.0)
    for i in range(N):
        d_min = min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])
        for j in range(N):
            if i != j:
                d = np.sqrt(np.sum((pts[i] - pts[j])**2))
                if d < d_min:
                    d_min = d
        # Leave a small margin to ensure strict feasibility for SLSQP
        r[i] = d_min / 2.5
    return r

def get_hex_init(r0, shift_type):
    """Generate hexagonal lattice initialization."""
    counts = [6, 5, 6, 5, 4] if shift_type == 0 else [5, 6, 5, 6, 4]
    pts = []
    y = r0
    for r_idx, count in enumerate(counts):
        x = r0 if r_idx % 2 == 0 else 2 * r0
        for _ in range(count):
            if len(pts) < N:
                pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
    return np.array(pts[:N])

def get_grid_init():
    """Generate grid-based initialization."""
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + 0.2*i, 0.1 + 0.2*j])
    pts.append([0.5, 0.5])
    return np.array(pts[:N])
