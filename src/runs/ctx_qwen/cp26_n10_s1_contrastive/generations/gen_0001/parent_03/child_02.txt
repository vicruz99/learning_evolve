import numpy as np
from scipy.optimize import minimize

N = 26

def objective_func(v):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)."""
    return -np.sum(v[2::3])

def constraint_func(v):
    """
    Computes inequality constraints g(v) >= 0.
    Includes boundary containment and pairwise separation.
    """
    C = v.reshape(N, 3)
    xc = C[:, 0]
    yc = C[:, 1]
    r = C[:, 2]
    
    c = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c.append(xc - r)
    c.append(1.0 - xc - r)
    c.append(yc - r)
    c.append(1.0 - yc - r)
    
    # Pairwise separation: dist_sq - (r_i + r_j)^2 >= 0
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = xc[i_idx] - xc[j_idx]
    dy = yc[i_idx] - yc[j_idx]
    rs = r[i_idx] + r[j_idx]
    c.append(dx*dx + dy*dy - rs*rs)
    
    return np.concatenate(c)

def generate_hex_init(r, n):
    """Generates a hexagonal lattice initialization clipped to valid bounds."""
    pts = []
    y = r
    row = 0
    while len(pts) < n:
        x_off = (row % 2) * r
        x = r + x_off
        while x <= 1.0 - r + 1e-9 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3.0) * r
        row += 1
    return np.clip(np.array(pts[:n]), 0.05, 0.95)

def generate_repulsion_init(n, seed):
    """Generates a force-directed initialization that spreads points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.2, 0.8, (n, 2))
    for _ in range(150):
        forces = np.zeros_like(pts)
        for i in range(n):
            for j in range(i + 1, n):
                d = pts[i] - pts[j]
                dist = np.linalg.norm(d)
                if dist < 0.25 and dist > 1e-5:
                    f = d / dist * (0.25 - dist)
                    forces[i] += f
                    forces[j] -= f
            # Wall repulsion to keep points away from edges initially
            for k in range(2):
                if pts[i, k] < 0.2: forces[i, k] += 1.0
                elif pts[i, k] > 0.8: forces[i, k] -= 1.0
        pts += forces * 0.15
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """

    inits = []

    for r_hex in [0.09, 0.10, 0.105, 0.11]:
        pts = generate_hex_init(r_hex, N)
        for i in range(N):
            inits.append(np.concatenate([pts[i], [0.03]]))
            
    # 2. Force-repulsion layouts to discover non-lattice optimal basins
    for seed in range(8):
        pts = generate_repulsion_init(N, seed)
        for i in range(N):
            inits.append(np.concatenate([pts[i], [0.02]]))
            
    # 3. Perturbed 5x5 grid + center (breaks strict symmetry)
    pts_grid = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])

        p = pts_grid + np.random.uniform(-0.04, 0.04, (N, 2))
        p = np.clip(p, 0.05, 0.95)
        for i in range(N):
            inits.append(np.concatenate([p[i], [0.04]]))

    for i0 in inits:
        try:
            res = minimize(
                objective_func,
                i0,
                method='SLSQP',

                    best_vars = res.x.copy()
