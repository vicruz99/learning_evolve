import numpy as np
from scipy.optimize import minimize

def objective(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all inequality constraints: boundary containment and circle separation."""
    N = 26
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    
    c = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(xc - r)
    c.append(1.0 - xc - r)
    c.append(yc - r)
    c.append(1.0 - yc - r)
    
    # Pairwise separation constraints: dist^2 >= (r_i + r_j)^2
    i, j = np.triu_indices(N, 1)
    c.append((xc[i]-xc[j])**2 + (yc[i]-yc[j])**2 - (r[i]+r[j])**2)
    
    return np.concatenate(c)

def get_hex_grid():
    """Generates a hexagonal grid of points inside [0,1]^2."""
    pts = []
    y = 0.08
    row = 0
    while len(pts) < 36:
        x = 0.08 + (row % 2) * 0.04
        while x <= 0.92:
            pts.append([x, y])
            x += 0.08
        y += 0.08 * np.sqrt(3)
        row += 1
    return np.array(pts[:26])

def get_square_grid():
    """Generates a perturbed square grid of points inside [0,1]^2."""
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + 0.18*i, 0.1 + 0.18*j])
    pts.append([0.5, 0.5])
    return np.array(pts)

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    N = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.001, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -np.inf
    best_x = None
    
    # Try multiple base layouts
    inits = [get_hex_grid(), get_square_grid()]
    
    for base_pts in inits:
        # Compute a safe initial radius for strict feasibility
        d_min = 1.0
        for k in range(N):
            d_min = min(d_min, base_pts[k,0], 1-base_pts[k,0], base_pts[k,1], 1-base_pts[k,1])
            for m in range(k+1, N):
                d = np.sqrt(np.sum((base_pts[k]-base_pts[m])**2))
                if d < d_min: 
                    d_min = d
        r0 = d_min / 5.0
        
        # Flatten initial state
        x0_base = np.zeros(3*N)
        for k in range(N):
            x0_base[3*k] = base_pts[k,0]
            x0_base[3*k+1] = base_pts[k,1]
            x0_base[3*k+2] = r0
            
        # Multiple restarts with controlled perturbations
        for seed in range(10):
            np.random.seed(seed + len(inits)*100)
            x_init = x0_base.copy()
            x_init[:2*N] += np.random.randn(2*N) * 0.015
            x_init[:2*N] = np.clip(x_init[:2*N], 0.02, 0.98)
            
            try:
                res = minimize(objective, x_init, method='SLSQP', bounds=bounds, 
                               constraints=cons, options={'maxiter': 2000, 'ftol': 1e-10})
                # Accept only if constraints are satisfied within tolerance
                if np.min(constraints(res.x)) >= -1e-6:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x
            except Exception:
                pass
                
    # Local refinement phase to escape shallow local minima
    if best_x is not None:
        for _ in range(15):
            x_curr = best_x.copy()
            x_curr[:2*N] += np.random.randn(2*N) * 0.008

                pass
