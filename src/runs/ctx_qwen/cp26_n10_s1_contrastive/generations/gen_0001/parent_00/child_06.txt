import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars_vec):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(vars_vec[2::3])

def constraint_func(vars_vec):
    """
    Inequality constraints: g(vars) >= 0
    Includes boundary containment and pairwise non-overlap.
    """
    n = len(vars_vec) // 3
    x = vars_vec[0::3]
    y = vars_vec[1::3]
    r = vars_vec[2::3]
    
    c = []
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise non-overlap constraints: dist_sq >= (r_i + r_j)^2
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    c.append(dist_sq[i_idx, j_idx] - r_sum_sq[i_idx, j_idx])
    
    return np.concatenate(c)

def compute_safe_radii(pts):
    """Compute strictly feasible radii for a given set of centers."""
    n = pts.shape[0]
    r = np.full(n, 0.5)
    for i in range(n):
        # Distance to walls
        r[i] = min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])
        # Distance to other centers
        for j in range(n):
            if i != j:
                dist = np.hypot(pts[i, 0] - pts[j, 0], pts[i, 1] - pts[j, 1])
                r[i] = min(r[i], dist)
    return r * 0.85  # Safety margin to ensure strict feasibility

def generate_inits():
    """Generate multiple diverse and feasible initial configurations."""
    inits = []
    
    # --- Hexagonal Lattice Inits ---
    pts_hex = []
    r_est = 0.105
    dy = r_est * np.sqrt(3)
    dx = 2.0 * r_est
    y = r_est
    row = 0
    while len(pts_hex) < N_CIRCLES:
        x = r_est + (row % 2) * (dx / 2.0)
        while x <= 1.0 - r_est and len(pts_hex) < N_CIRCLES:
            pts_hex.append([x, y])
            x += dx
        y += dy
        row += 1
    pts_hex = np.array(pts_hex[:N_CIRCLES])
    
    for seed in range(8):
        np.random.seed(seed)
        p = pts_hex.copy() + np.random.uniform(-0.015, 0.015, pts_hex.shape)
        p = np.clip(p, 0.05, 0.95)
        r_init = compute_safe_radii(p)
        
        v = np.zeros(3 * N_CIRCLES)
        for i in range(N_CIRCLES):
            v[3*i] = p[i, 0]
            v[3*i+1] = p[i, 1]
            v[3*i+2] = r_init[i]
        inits.append(v)
        
    # --- Grid-based Inits ---
    pts_grid = np.zeros((N_CIRCLES, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < N_CIRCLES:
                pts_grid[idx] = [0.1 + i*0.2, 0.1 + j*0.2]
                idx += 1
    if idx < N_CIRCLES:
        pts_grid[idx] = [0.5, 0.5]
        idx += 1
        
    for seed in range(8):
        np.random.seed(seed + 5000)
        p = pts_grid + np.random.uniform(-0.02, 0.02, pts_grid.shape)
        p = np.clip(p, 0.05, 0.95)
        r_init = compute_safe_radii(p)
        
        v = np.zeros(3 * N_CIRCLES)
        for i in range(N_CIRCLES):
            v[3*i] = p[i, 0]
            v[3*i+1] = p[i, 1]
            v[3*i+2] = r_init[i]
        inits.append(v)
        
    return inits

def run_packing():
    best_sum = -np.inf
    best_vars = None
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES
    
    inits = generate_inits()
    
    for x0 in inits:
        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': constraint_func},
                options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
            )
            if res.success:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x
        except Exception:
            continue
            
    # Fallback if optimization fails entirely
    if best_vars is None:
        best_vars = np.zeros(3 * N_CIRCLES)
        best_vars[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 5)
        best_vars[1::3] = np.repeat(np.linspace(0.1, 0.9, 5), 5)
        best_vars[2::3] = 0.09
        best_sum = 0.09 * 25
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    # Safety clamp to guarantee boundary constraints within numerical tolerance
    radii = np.maximum(radii, 1e-6)
    centers = np.clip(centers, radii, 1.0 - radii)
    
    return centers, radii, float(best_sum)
