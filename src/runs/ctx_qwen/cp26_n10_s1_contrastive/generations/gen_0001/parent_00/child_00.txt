import numpy as np
from scipy.optimize import minimize

# Global constants to avoid closures and ensure top-level definition
N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """
    Returns inequality constraints >= 0.
    Enforces boundary containment and pairwise non-overlap.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    cons = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    cons.append(x - r)
    cons.append(1.0 - x - r)
    cons.append(y - r)
    cons.append(1.0 - y - r)
    
    # Pairwise non-overlap: dist_sq >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    r_sum_sq = r_sum**2
    
    # Use precomputed indices for efficiency
    cons.append(dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX])
    
    return np.concatenate(cons)

def get_hex_init(r_scale=0.09):
    """Generates a hexagonal lattice initialization for N circles."""
    rows = [6, 5, 6, 5, 4]
    pts = []
    y = r_scale
    for idx, count in enumerate(rows):
        shift = r_scale if idx % 2 == 1 else 0.0
        x = r_scale + shift
        for _ in range(count):
            pts.append((x, y))
            x += 2.0 * r_scale
        y += np.sqrt(3.0) * r_scale
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -np.inf
    best_vars = None
    
    # 1. Generate diverse initial configurations
    inits = []
    # Hexagonal packs at different scales
    for scale in [0.08, 0.09, 0.10, 0.11]:
        inits.append(get_hex_init(scale))
    # Grid pack
    grid_pts = []
    for i in range(5):
        for j in range(5):
            grid_pts.append((0.1 + i * 0.2, 0.1 + j * 0.2))
    grid_pts.append((0.5, 0.5))
    inits.append(np.array(grid_pts[:N]))
    # Random packs
    for _ in range(4):
        inits.append(np.random.rand(N, 2) * 0.8 + 0.1)
        
    # 2. Run optimization with controlled perturbations
    for c_init in inits:
        for noise_scale in [0.0, 0.005, 0.01, 0.02]:
            if noise_scale > 0:
                noise = np.random.randn(N, 2) * noise_scale
                c_pert = np.clip(c_init + noise, 0.01, 0.99)
            else:
                c_pert = c_init
                
            # Safe initial radius to guarantee feasibility
            r_init = np.full(N, 0.06)
            x0 = np.zeros(3 * N)
            x0[0::3] = c_pert[:, 0]
            x0[1::3] = c_pert[:, 1]
            x0[2::3] = r_init
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-11})
                if res.success:
                    # Verify constraints are satisfied within tolerance
                    cons_val = constraint_func(res.x)
                    if np.min(cons_val) >= -1e-6:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_vars = res.x.copy()
            except Exception:
                pass
                
    # 3. Iterative polishing to escape local minima and tighten packing
    if best_vars is not None:
        for _ in range(5):
            x_pert = best_vars + np.random.randn(3 * N) * 0.003
            x_pert[0::3] = np.clip(x_pert[0::3], 0.0, 1.0)
            x_pert[1::3] = np.clip(x_pert[1::3], 0.0, 1.0)
            x_pert[2::3] = np.clip(x_pert[2::3], 0.0, 0.5)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,

                    cons_val = constraint_func(res.x)
                    if np.min(cons_val) >= -1e-6 and -res.fun > best_sum:
                        best_sum = -res.fun
                        best_vars = res.x.copy()
            except Exception:
                pass
                
    # Fallback if optimization completely fails (highly unlikely)
    if best_vars is None:
        centers = get_hex_init(0.09)
        radii = np.full(N, 0.05)
        return centers, radii, float(np.sum(radii))
