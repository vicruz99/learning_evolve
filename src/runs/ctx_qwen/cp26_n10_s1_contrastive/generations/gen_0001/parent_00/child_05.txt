To maximize the sum of radii for 26 circles in a unit square, we decouple the problem into two parts: center placement and radius determination. For any fixed set of centers, finding the optimal radii is a linear programming (LP) problem: maximize $\sum r_i$ subject to boundary constraints ($r_i \leq \min(x_i, 1-x_i, y_i, 1-y_i)$) and non-overlap constraints ($r_i + r_j \leq \text{dist}_{ij}$). This LP is solved extremely fast using `scipy.optimize.linprog` with the HiGHS solver.

The optimization strategy proceeds in three phases:
1. **Global Search**: Use `differential_evolution` on the 52-dimensional space of center coordinates. The objective function is the negative LP-optimal sum of radii. DE is robust to the non-smooth landscape created by the LP objective and explores diverse configurations effectively.
2. **Cyclic Coordinate Descent**: Refine the DE result by fixing 25 circles at a time and optimizing the 2D position of the remaining circle using `Nelder-Mead`. This efficiently escapes local minima and tightens packings.
3. **Full-Space Refinement**: Apply `Powell`'s method over all center coordinates simultaneously to squeeze out remaining gains, using the already-optimized configuration as a warm start.

This approach separates geometric constraint satisfaction (handled exactly by LP) from topological optimization (handled by derivative-free global/local search), yielding highly dense and valid packings. All helper functions are defined at the top level, and no lambda functions are used, adhering strictly to the rules.

import numpy as np
from scipy.optimize import differential_evolution, linprog, minimize

N_CIRCLES = 26

def compute_sum_radii(centers):
    """Solves the LP to find the maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    c = -np.ones(n)
    x_c = centers[:, 0]
    y_c = centers[:, 1]
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    b_boundary = np.concatenate([x_c, 1.0 - x_c, y_c, 1.0 - y_c])
    A_boundary = np.eye(n)
    A_boundary = np.vstack([A_boundary, A_boundary, A_boundary, A_boundary])
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    i_idx, j_idx = np.triu_indices(n, k=1)
    n_pairs = len(i_idx)
    A_pairs = np.zeros((n_pairs, n))
    A_pairs[np.arange(n_pairs), i_idx] = 1.0
    A_pairs[np.arange(n_pairs), j_idx] = 1.0
    b_pairs = dists[i_idx, j_idx]
    
    A_ub = np.vstack([A_boundary, A_pairs])
    b_ub = np.concatenate([b_boundary, b_pairs])
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun
    except Exception:
        pass
    return 0.0

def solve_radii(centers):
    """Solves the LP and returns both the sum of radii and the radii array."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    c = -np.ones(n)
    x_c = centers[:, 0]
    y_c = centers[:, 1]
    
    b_boundary = np.concatenate([x_c, 1.0 - x_c, y_c, 1.0 - y_c])
    A_boundary = np.eye(n)
    A_boundary = np.vstack([A_boundary, A_boundary, A_boundary, A_boundary])
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    n_pairs = len(i_idx)
    A_pairs = np.zeros((n_pairs, n))
    A_pairs[np.arange(n_pairs), i_idx] = 1.0
    A_pairs[np.arange(n_pairs), j_idx] = 1.0
    b_pairs = dists[i_idx, j_idx]
    
    A_ub = np.vstack([A_boundary, A_pairs])
    b_ub = np.concatenate([b_boundary, b_pairs])
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def objective_de(vars):
    """Objective for differential evolution: minimize negative sum of radii."""
    centers = vars.reshape(-1, 2)
    return -compute_sum_radii(centers)

def obj_cyclic(pos, idx, current_centers):
    """Objective for optimizing a single circle's position cyclically."""
    temp_centers = current_centers.copy()
    temp_centers[idx] = np.clip(pos, 0.0, 1.0)
    return -compute_sum_radii(temp_centers)

def run_packing():
    n = N_CIRCLES
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # Phase 1: Global exploration with Differential Evolution
    best_res = differential_evolution(
        objective_de, bounds, popsize=25, maxiter=150,
        mutation=(0.5, 1.0), recombination=0.9, seed=42, polish=False
    )
    centers = best_res.x.reshape(n, 2)
    current_sum = compute_sum_radii(centers)
    
    # Phase 2: Cyclic Coordinate Descent to tighten local packing
    for _ in range(3):
        improved = False
        for i in range(n):
            x0 = centers[i]
            try:
                res = minimize(obj_cyclic, x0, args=(i, centers), method='Nelder-Mead', 
                               options={'xatol': 1e-5, 'fatol': 1e-7, 'maxiter': 150})
                new_pos = np.clip(res.x, 0.0, 1.0)
                temp_c = centers.copy()
                temp_c[i] = new_pos
                new_sum = compute_sum_radii(temp_c)
                if new_sum > current_sum + 1e-9:
                    centers[i] = new_pos
                    current_sum = new_sum
                    improved = True
            except Exception:
                pass
        if not improved:
            break
            
    # Phase 3: Full-space refinement with Powell's method
    try:
        res_final = minimize(objective_de, centers.flatten(), method='Powell', 
                             options={'maxiter': 40, 'xtol': 1e-6})
        centers = np.clip(res_final.x.reshape(n, 2), 0.0, 1.0)
        current_sum = compute_sum_radii(centers)
    except Exception:
        pass
        
    # Extract final valid radii corresponding to the optimized centers
    final_sum, final_radii = solve_radii(centers)
    
    return centers, final_radii, float(final_sum)
