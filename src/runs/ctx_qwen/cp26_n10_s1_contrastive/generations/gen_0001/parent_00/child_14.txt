import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
TRI_IDX = np.triu_indices(N_CIRCLES, k=1)

def objective(vars):
    """Objective: maximize sum of radii (minimize negative sum)."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Constraints: boundary containment and non-overlap."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    c = np.empty(4 * N_CIRCLES + (N_CIRCLES * (N_CIRCLES - 1)) // 2)
    idx = 0
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c[idx:idx+N_CIRCLES] = x - r
    idx += N_CIRCLES
    c[idx:idx+N_CIRCLES] = 1.0 - x - r
    idx += N_CIRCLES
    c[idx:idx+N_CIRCLES] = y - r
    idx += N_CIRCLES
    c[idx:idx+N_CIRCLES] = 1.0 - y - r
    idx += N_CIRCLES
    
    # Pairwise non-overlap: dist_sq >= (ri + rj)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    r_sum_sq = r_sum**2
    
    c[idx:] = dist_sq[TRI_IDX] - r_sum_sq[TRI_IDX]
    return c

def run_packing():
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -np.inf
    best_vars = None
    
    # Multiple restarts with hexagonal-inspired initialization
    for seed in range(40):
        np.random.seed(seed)
        
        # Hexagonal lattice parameters
        r0 = 0.065
        dy = r0 * np.sqrt(3.0)
        dx = 2.0 * r0
        
        centers = []
        # Row pattern: 6, 5, 6, 5, 4 circles closely matches optimal density
        row_counts = [6, 5, 6, 5, 4]
        y_cur = r0
        for i, cnt in enumerate(row_counts):
            x_start = r0 if i % 2 == 0 else r0 + dx / 2.0
            for j in range(cnt):
                centers.append([x_start + j * dx, y_cur])
            y_cur += dy
        centers = np.array(centers)
        
        # Add controlled perturbation to escape symmetries and local minima
        centers += np.random.uniform(-0.012, 0.012, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        
        # Slight radius variation to help optimizer break ties
        radii = np.full(N_CIRCLES, r0) + np.random.uniform(0, 0.003, N_CIRCLES)
        
        # Flatten to optimization vector: [x0, y0, r0, x1, y1, r1, ...]
        x0 = np.zeros(3 * N_CIRCLES)
        for i in range(N_CIRCLES):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-11})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_vars = res.x.copy()
        except Exception:
            continue
            
    # High-precision refinement on best configuration found
    if best_vars is not None:
        try:
            res_final = minimize(objective, best_vars, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
            if -res_final.fun > best_sum:
                best_vars = res_final.x
                best_sum = -res_final.fun
        except Exception:
            pass
            
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
