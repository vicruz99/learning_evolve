import numpy as np
from scipy.optimize import minimize

N = 26

def compute_safe_radius(centers):
    """Compute a radius r such that all circles with radius r at given centers are strictly valid."""
    n = centers.shape[0]
    r = 0.5
    for i in range(n):
        # Distance to walls
        d_wall = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
        if d_wall < r:
            r = d_wall
        # Distance to other centers
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            if d / 2.0 < r:
                r = d / 2.0
    return r * 0.95  # Slight margin to ensure strict feasibility

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Returns inequality constraints >= 0 for walls and pairwise non-overlap."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Wall constraints
    c_wall = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints: dist_sq >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    r_sum_sq = (r[:, None] + r[None, :])**2
    
    i, j = np.triu_indices(N, k=1)
    c_pair = dist_sq[i, j] - r_sum_sq[i, j]
    
    return np.concatenate([c_wall, c_pair])

def generate_hex_init(n, r_base):
    """Generate a hexagonal lattice of points."""
    pts = []
    y = r_base
    row = 0
    while len(pts) < n:
        x = r_base
        if row % 2 == 1:
            x += r_base  # Horizontal shift for hex packing
        while x <= 1.0 - r_base and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r_base
        y += np.sqrt(3) * r_base
        row += 1
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -np.inf
    best_vars = None
    
    bounds = [(0, 1), (0, 1), (0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Base configurations to perturb
    bases = []
    for scale in [0.07, 0.08, 0.085, 0.09]:
        bases.append(generate_hex_init(N, scale))
        
    # 5x5 grid + center
    grid = np.zeros((N, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            grid[idx] = [0.1 + i * 0.2, 0.1 + j * 0.2]
            idx += 1
    grid[-1] = [0.5, 0.5]
    bases.append(grid)
    
    # Multiple restarts
    for seed in range(15):
        np.random.seed(seed)
        
        for base in bases:
            pts = base.copy() + np.random.uniform(-0.015, 0.015, base.shape)
            pts = np.clip(pts, 0.01, 0.99)
            r_safe = compute_safe_radius(pts)
            r_init = np.full(N, r_safe)
            
            x0 = np.zeros(3 * N)
            x0[0::3] = pts[:, 0]
            x0[1::3] = pts[:, 1]
            x0[2::3] = r_init
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if not np.isnan(res.fun):
                    c_vals = constraint_func(res.x)
                    if np.min(c_vals) >= -1e-7:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_vars = res.x.copy()
            except Exception:
                continue
                
        # Random initialization
        pts_r = np.random.uniform(0.05, 0.95, (N, 2))
        r_safe_r = compute_safe_radius(pts_r)
        x0_r = np.zeros(3 * N)
        x0_r[0::3] = pts_r[:, 0]
        x0_r[1::3] = pts_r[:, 1]
        x0_r[2::3] = np.full(N, r_safe_r)
        
        try:
            res = minimize(objective, x0_r, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun):
                c_vals = constraint_func(res.x)
                if np.min(c_vals) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            continue

    # Polishing phase on the best configuration found
    if best_vars is not None:
        try:
            res_final = minimize(objective, best_vars, method='SLSQP', bounds=bounds,

            if not np.isnan(res_final.fun):

                if np.min(c_vals) >= -1e-7:
                    best_vars = res_final.x
                    best_sum = -res_final.fun
        except Exception:
            pass

    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
