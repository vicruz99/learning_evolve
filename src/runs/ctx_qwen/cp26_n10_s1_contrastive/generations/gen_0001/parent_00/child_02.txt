import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
TRI_INDICES = np.triu_indices(N_CIRCLES, k=1)

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """
    Computes all inequality constraints: boundary containment and non-overlap.
    Returns array of constraint values >= 0.
    """
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Wall constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c_wall = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints: dist_sq >= (ri + rj)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    # Extract upper triangular part to avoid duplicates and self-comparison
    c_pair = dist_sq[TRI_INDICES] - r_sum[TRI_INDICES]**2
    
    return np.concatenate([c_wall, c_pair])

def get_feasible_radii(centers):
    """Computes strictly feasible initial radii for given center positions."""
    n = centers.shape[0]
    r = np.full(n, 0.5)
    
    # Constrain by boundaries
    r = np.minimum(r, np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    r = np.minimum(r, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Constrain by pairwise distances
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    # Scale down slightly to ensure strict feasibility for optimizer
    return np.minimum(r, min_dists * 0.45)

def generate_hex_config(scale=1.0):
    """Generates a hexagonal lattice configuration scaled by `scale`."""
    n = N_CIRCLES
    r_est = 0.09 * scale
    dx = 2.0 * r_est
    dy = np.sqrt(3.0) * r_est
    pts = []
    y = r_est
    row = 0
    while y <= 1.0 - r_est:
        x = r_est + (row % 2) * (dx / 2.0)
        while x <= 1.0 - r_est:
            if len(pts) < n:
                pts.append([x, y])
            x += dx
        y += dy
        row += 1
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = N_CIRCLES
    best_sum = -np.inf
    best_vars = None
    
    # Variable bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    configs = []
    
    # 1. Hexagonal grids with varying densities
    for s in np.linspace(0.8, 1.3, 6):
        configs.append(generate_hex_config(s))
        
    # 2. Random + Force-directed layouts to escape grid symmetries
    for seed in range(10):
        np.random.seed(seed + 42)
        pts = np.random.rand(n, 2) * 0.8 + 0.1
        # Simple repulsion to spread points evenly
        for _ in range(60):
            forces = np.zeros_like(pts)
            for i in range(n):
                for j in range(i+1, n):
                    diff = pts[i] - pts[j]
                    d = np.linalg.norm(diff)
                    if d > 1e-4:
                        f = diff / (d**2) * 0.02
                        forces[i] += f
                        forces[j] -= f
            pts += forces * 0.15
            pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts)
        
    # Phase 1: Coarse multi-start optimization
    for init_centers in configs:
        init_radii = get_feasible_radii(init_centers)
        vars0 = np.zeros(3 * n)
        vars0[0::3] = init_centers[:, 0]
        vars0[1::3] = init_centers[:, 1]
        vars0[2::3] = init_radii
        
        # Add small perturbation to break exact symmetries
        vars0 += np.random.randn(3 * n) * 0.001
        vars0 = np.clip(vars0, [0.01, 0.01, 0.01]*n, [0.99, 0.99, 0.5]*n)
        
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraint_func},
                           options={'maxiter': 1500, 'ftol': 1e-13})
            
            curr_sum = -res.fun
            c_val = constraint_func(res.x)
            # Accept if constraints are satisfied (within tolerance) and better
            if np.min(c_val) >= -1e-5 and curr_sum > best_sum:
                best_sum = curr_sum
                best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: High-precision refinement on the best configuration found
    if best_vars is not None:
        try:
            res_final = minimize(objective, best_vars, method='SLSQP', bounds=bounds,

            c_val = constraint_func(res_final.x)
            if np.min(c_val) >= -1e-5:
                best_vars = res_final.x
                best_sum = -res_final.fun
        except Exception:
            pass
            
        centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
        radii = best_vars[2::3]
        radii = np.maximum(radii, 0.0)
        return centers, radii, float(best_sum)
        
    # Fallback to a valid grid if all optimizations fail
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)
    return centers, radii, float(np.sum(radii))
