import numpy as np
from scipy.optimize import minimize

N = 26
MASK = np.triu(np.ones((N, N), dtype=bool), k=1)

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """
    Inequality constraints: g(vars) >= 0
    - Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    - Separation: dist_sq(i,j) >= (r_i + r_j)^2 for all i < j
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    r_sum_sq = r_sum**2
    
    c_pairwise = dist_sq[MASK] - r_sum_sq[MASK]
    
    return np.concatenate([c1, c2, c3, c4, c_pairwise])

def run_packing():
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-5, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -np.inf
    best_vars = None
    
    inits = []
    
    # 1. Hexagonal lattice variations (exploits dense packing geometry)
    for scale in [0.08, 0.09, 0.10]:
        r_init = scale
        pts = []
        y = r_init
        row = 0
        while len(pts) < N:
            x = r_init + (0.5 if row % 2 == 1 else 0.0) * 2 * r_init
            while x <= 1.0 - r_init and len(pts) < N:
                pts.append([x, y, r_init])
                x += 2.0 * r_init
            y += np.sqrt(3) * r_init
            row += 1
        inits.append(np.array(pts[:N]).flatten())
        
    # 2. Random starts (explores asymmetric configurations)
    for seed in range(10):
        np.random.seed(seed + 42)
        xs = np.random.uniform(0.15, 0.85, N)
        ys = np.random.uniform(0.15, 0.85, N)
        rs = np.full(N, 0.04)
        inits.append(np.column_stack((xs, ys, rs)).flatten())
        
    # 3. Force-directed starts (pre-spreads points to reduce initial overlap)
    for seed in range(5):
        np.random.seed(seed + 100)
        centers_fd = np.random.rand(N, 2) * 0.7 + 0.15
        for _ in range(300):
            forces = np.zeros_like(centers_fd)
            for i in range(N):
                for j in range(i+1, N):
                    diff = centers_fd[i] - centers_fd[j]
                    dist = np.linalg.norm(diff)
                    if dist < 1e-6: continue
                    f = diff / (dist**2 + 0.01)
                    forces[i] += f
                    forces[j] -= f
            centers_fd += forces * 0.005
            centers_fd = np.clip(centers_fd, 0.05, 0.95)
        rs_fd = np.full(N, 0.04)
        inits.append(np.column_stack((centers_fd, rs_fd)).flatten())
        
    # 4. Grid starts with jitter
    for seed in range(5):
        np.random.seed(seed + 200)
        pts = np.random.uniform(0.1, 0.9, (N, 3))
        pts[:, 2] = 0.05
        inits.append(pts.flatten())
        
    # Phase 1: Global exploration via multi-start SLSQP
    for idx, x0 in enumerate(inits):
        # Perturb to break symmetry and ensure diversity
        x0_pert = x0 + np.random.uniform(-0.015, 0.015, len(x0))
        x0_pert = np.clip(x0_pert, 0.01, 0.99)
        x0_pert[2::3] = np.clip(x0_pert[2::3], 1e-5, 0.49)
        
        try:
            res = minimize(objective, x0_pert, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-10})
            if res.success:
                s = -res.fun
                # Verify feasibility (allow tiny numerical tolerance)
                min_c = np.min(constraint_func(res.x))
                if min_c >= -1e-6:
                    if s > best_sum:
                        best_sum = s
                        best_vars = res.x
        except Exception:
            continue
            
    # Phase 2: Local refinement on the best configuration found
    if best_vars is not None:
        for _ in range(10):
            x_ref = best_vars + np.random.uniform(-0.005, 0.005, len(best_vars))
            x_ref = np.clip(x_ref, 0.01, 0.99)
            x_ref[2::3] = np.clip(x_ref[2::3], 1e-5, 0.49)
            try:
                res = minimize(objective, x_ref, method='SLSQP', bounds=bounds,

                    min_c = np.min(constraint_func(res.x))

                        best_sum = s
                        best_vars = res.x
            except Exception:
                continue

    # Fallback to a safe valid configuration if optimization fails
    if best_vars is None:
        x0_safe = np.zeros(3*N)
        for i in range(N):
            x0_safe[3*i] = 0.25 + (i % 5) * 0.15

                       constraints=cons, options={'maxiter': 2000, 'ftol': 1e-10})
        best_vars = res.x
        best_sum = -res.fun

    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
