import numpy as np
from scipy.optimize import minimize

N = 26

def objective_func(x):
    """Objective: minimize negative sum of radii."""
    r = x[2::3]
    return -np.sum(r)

def constraint_func(x):
    """
    Vectorized constraint function for SLSQP.
    Returns array of values that must be >= 0.
    Includes boundary and pairwise non-overlap constraints.
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist_sq >= (r_i + r_j)^2
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    i, j = np.triu_indices(N, k=1)
    c.append(dist_sq[i, j] - r_sum_sq[i, j])
    
    return np.concatenate(c)

def generate_force_init(n, seed):
    """Generate initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    pts = np.random.rand(n, 2)
    for _ in range(600):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists_sq = np.sum(diff**2, axis=2)
        dists_sq[np.eye(n, dtype=bool)] = np.inf
        dists = np.sqrt(dists_sq)
        forces = np.sum(diff / (dists_sq + 1e-6)[:, :, np.newaxis], axis=1)
        
        for i in range(n):
            if pts[i, 0] < 0.05: forces[i, 0] += 5.0
            if pts[i, 0] > 0.95: forces[i, 0] -= 5.0
            if pts[i, 1] < 0.05: forces[i, 1] += 5.0
            if pts[i, 1] > 0.95: forces[i, 1] -= 5.0
            
        pts += 0.002 * forces
        pts = np.clip(pts, 0.05, 0.95)
        
    diff = pts[:, np.newaxis, :] - pts[np.newaxis, :]
    dists_sq = np.sum(diff**2, axis=2)
    dists_sq[np.eye(n, dtype=bool)] = np.inf
    min_d = np.sqrt(np.min(dists_sq, axis=1))
    r_init = np.min(min_d / 2.0, 0.08)
    
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3*i] = pts[i, 0]
        x0[3*i+1] = pts[i, 1]
        x0[3*i+2] = r_init[i]
    return x0

def generate_hex_init(n, seed):
    """Generate initial configuration using perturbed hexagonal lattice."""
    np.random.seed(seed + 100)
    r_est = 0.09
    pts = []
    y = r_est
    row = 0
    while len(pts) < n:
        x = r_est if row % 2 == 0 else 2 * r_est
        while x <= 1.0 - r_est and len(pts) < n:
            pts.append([x, y])
            x += 2 * r_est
        y += np.sqrt(3) * r_est
        row += 1
    pts = np.array(pts[:n])
    pts += np.random.uniform(-0.03, 0.03, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3*i] = pts[i, 0]
        x0[3*i+1] = pts[i, 1]
        x0[3*i+2] = 0.07
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-4, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -np.inf
    best_x = None
    
    # Run multiple restarts with diverse initializations
    for seed in range(20):
        if seed < 10:
            x0 = generate_force_init(n, seed)
        else:
            x0 = generate_hex_init(n, seed)
            
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-11})
            
            if res.success:
                curr_sum = -res.fun
                con_vals = constraint_func(res.x)
                # Accept if constraints are satisfied (within tolerance) and better than best
                if np.min(con_vals) >= -1e-6 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            continue
            
    if best_x is None:
        best_x = generate_force_init(n, 42)
        
    centers = best_x.reshape(n, 3)[:, :2]
    radii = best_x.reshape(n, 3)[:, 2]
    
    # Strict post-optimization correction to guarantee validity within 1e-10
    for _ in range(10):
        changed = False
        # Boundary enforcement
        for i in range(n):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-10:
                radii[i] = max_r
                changed = True
                
        # Pairwise overlap enforcement
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                max_r_sum = dist
                if radii[i] + radii[j] > max_r_sum + 1e-10:
                    avg = max_r_sum / 2.0
                    radii[i] = min(radii[i], avg)
                    radii[j] = min(radii[j], avg)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
