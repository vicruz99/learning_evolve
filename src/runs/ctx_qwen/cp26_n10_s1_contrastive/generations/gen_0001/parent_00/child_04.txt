import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def _objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def _constraints(x):
    """Compute inequality constraints >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    # Pre-allocate constraint array
    c = np.empty(4 * N + N * (N - 1) // 2)
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c[0::4] = cx - cr
    c[1::4] = 1.0 - cx - cr
    c[2::4] = cy - cr
    c[3::4] = 1.0 - cy - cr
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    dist_sq = dx * dx + dy * dy
    r_sum_sq = (cr[I_IDX] + cr[J_IDX]) ** 2
    c[4 * N:] = dist_sq - r_sum_sq
    
    return c

def _generate_hex_init(row_counts, r_init, scale=1.0, shift_x=0.0, shift_y=0.0):
    """Generate initial centers from a hexagonal pattern."""
    pts = []
    y = r_init
    for i, count in enumerate(row_counts):
        shift = (r_init if i % 2 == 1 else 0.0) + shift_x
        x = r_init + shift
        for _ in range(count):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_init
        y += np.sqrt(3.0) * r_init
        
    centers = np.array(pts[:N])
    centers = centers * scale + np.array([shift_x, shift_y])
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -np.inf
    best_vars = None
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-5, 0.5)] * N
    cons = {'type': 'ineq', 'fun': _constraints}
    
    # Diverse initial configurations
    inits = []
    inits.append(_generate_hex_init([6, 5, 6, 5, 4], 0.095))
    inits.append(_generate_hex_init([5, 6, 5, 6, 4], 0.095))
    inits.append(_generate_hex_init([6, 6, 5, 5, 4], 0.092))
    inits.append(_generate_hex_init([4, 5, 6, 5, 6], 0.094))
    
    # Randomized grid
    np.random.seed(10)
    inits.append(np.random.uniform(0.1, 0.9, (N, 2)))

    pts_c = []
    pts_c.append([0.5, 0.5])

        r = 0.35
        pts_c.append([0.5 + r*np.cos(angle), 0.5 + r*np.sin(angle)])
    inits.append(np.clip(pts_c, 0.05, 0.95))

    for idx, center_init in enumerate(inits):
        for trial in range(10):
            np.random.seed(idx * 100 + trial)
            
            # Perturb centers
            pert_centers = center_init + np.random.uniform(-0.02, 0.02, (N, 2))
            pert_centers = np.clip(pert_centers, 0.05, 0.95)
            
            # Initial radii
            r_init = np.full(N, 0.09)
            
            # Flatten
            x0 = np.zeros(3 * N)
            for i in range(N):
                x0[3*i] = pert_centers[i, 0]
                x0[3*i+1] = pert_centers[i, 1]
                x0[3*i+2] = r_init[i]
                
            try:
                res = minimize(_objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 2500, 'ftol': 1e-12})

                    current_sum = -res.fun
                    c_vals = _constraints(res.x)

                            best_sum = current_sum

            except Exception:
                continue

    # Fallback

        idx = 0

        best_vars[2::3] = radii
        best_sum = np.sum(radii)
            
    return centers, radii, float(best_sum)
