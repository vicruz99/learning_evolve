

<strategy>
The current SLSQP-only approach struggles with the highly non-convex constraint landscape of circle packing, often converging to local optima around 2.59. To break through the 2.63 barrier, we will use a hybrid optimization strategy:

1. **LP-Guided Center Optimization**: Fix centers and solve a Linear Program (LP) to find the optimal radii that maximize $\sum r_i$. This LP exactly handles all boundary and pairwise non-overlap constraints efficiently.
2. **Derivative-Free Global Search**: Use `scipy.optimize.minimize` with the `Powell` method to optimize the circle centers. The objective is the negative sum of radii returned by the LP. Powell's method is robust for non-smooth objectives and works well with bounds. Multiple restarts from structured (hexagonal, grid) and random initial configurations ensure exploration of the global landscape.
3. **Joint SLSQP Refinement**: Once the LP-guided search finds high-quality centers and radii, we pass them as an initial guess to a joint SLSQP optimization over all variables $(x_i, y_i, r_i)$. SLSQP excels at polishing solutions near active constraints, squeezing out additional precision and handling the smooth constraint boundaries optimally.

This decoupling separates the combinatorial placement difficulty (handled by Powell+LP) from the continuous constraint satisfaction (handled by SLSQP), consistently yielding superior packings for this class of problems. All helper functions are defined at the module level with no closures or lambdas, adhering strictly to the rules.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_max_radii_sum(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
        
    # Safe fallback
    r_fallback = np.ones(n) * 1e-6
    return np.sum(r_fallback), r_fallback

def neg_sum_radii(centers_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    s, _ = get_max_radii_sum(centers)
    return -s

def joint_objective(vars):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def joint_constraint(vars):
    """Constraints for joint optimization: boundary and non-overlap >= 0."""
    n = N
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    c = []
    # Boundary: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.extend(x - r)
    c.extend(1.0 - x - r)
    c.extend(y - r)
    c.extend(1.0 - y - r)
    
    # Pairwise: dist^2 >= (r_i + r_j)^2
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.extend((dist_sq - r_sum_sq)[mask])
    
    return np.array(c)

def run_packing():
    best_sum = -1.0
    best_centers = None
    
    inits = []
    
    # 1. Hexagonal lattice
    r_est = 0.095
    y_curr = r_est
    row = 0
    pts = []
    while len(pts) < N:
        x_start = r_est if row % 2 == 0 else 2 * r_est
        x = x_start
        while x < 1.0 - r_est + 1e-6 and len(pts) < N:
            pts.append([x, y_curr])
            x += 2 * r_est
        y_curr += np.sqrt(3) * r_est
        row += 1
    inits.append(np.array(pts[:N]).flatten())
    
    # 2. Structured grid + center
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + i*0.2, 0.1 + j*0.2])
    pts.append([0.5, 0.5])
    inits.append(np.array(pts).flatten())
    
    # 3. Random configurations
    np.random.seed(123)
    for _ in range(5):
        pts = np.random.rand(N, 2)
        inits.append(pts.flatten())
        
    bounds_c = [(0.0, 1.0)] * (2 * N)
    
    # Phase 1: Optimize centers using Powell + LP objective
    for init in inits:
        init_pert = init + np.random.randn(len(init)) * 0.002
        init_pert = np.clip(init_pert, 0.01, 0.99)
        
        res = minimize(neg_sum_radii, init_pert, method='Powell', 
                       bounds=bounds_c, 
                       options={'maxiter': 250, 'ftol': 1e-10, 'xtol': 1e-10})
        
        curr_sum = -res.fun
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = res.x.reshape(N, 2)
            
    if best_centers is None:
        best_centers = np.random.rand(N, 2)
        
    # Get optimal radii for the best centers found
    _, best_radii = get_max_radii_sum(best_centers)
    
    # Phase 2: Joint refinement with SLSQP
    vars_init = np.zeros(3 * N)
    for i in range(N):
        vars_init[3*i] = best_centers[i, 0]
        vars_init[3*i+1] = best_centers[i, 1]
        vars_init[3*i+2] = best_radii[i]
        
    bounds_joint = []
    for _ in range(N):
        bounds_joint.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
    cons = {'type': 'ineq', 'fun': joint_constraint}
    
    res_final = minimize(joint_objective, vars_init, method='SLSQP', 
                         bounds=bounds_joint, constraints=cons,
                         options={'maxiter': 4000, 'ftol': 1e-12})
                         
    final_vars = res_final.x
    centers_opt = np.column_stack((final_vars[0::3], final_vars[1::3]))
    radii_opt = final_vars[2::3]
    radii_opt = np.maximum(radii_opt, 0.0)
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
