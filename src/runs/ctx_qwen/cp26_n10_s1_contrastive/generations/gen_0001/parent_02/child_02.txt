import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def compute_max_radii_lp(centers):
    """
    Given fixed centers, solves an LP to find radii that maximize sum(r_i)
    subject to boundary and non-overlap constraints.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Maximize sum(r) <=> Minimize -sum(r)
    
    A_ub = []
    b_ub = []
    
    # Wall constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        row_i = [0.0] * n
        row_i[i] = 1.0
        A_ub.extend([row_i, row_i, row_i, row_i])
        b_ub.extend([x, 1.0 - x, y, 1.0 - y])
        
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = [0.0] * n
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return np.zeros(n)

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """
    Returns a 1D array of constraint values >= 0.
    Enforces boundary containment and pairwise non-overlap.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c_wall = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap: dist_sq - (r_i + r_j)^2 >= 0
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    mask = np.triu(np.ones((N_CIRCLES, N_CIRCLES), dtype=bool), k=1)
    c_pair = dist_sq[mask] - r_sum[mask]**2
    
    return np.concatenate([c_wall, c_pair])

def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Strategy 1: LP-initialized random starts
    for seed in range(12):
        np.random.seed(seed)
        c_init = np.random.rand(N_CIRCLES, 2) * 0.8 + 0.1
        r_init = compute_max_radii_lp(c_init)
        
        vars_init = np.empty(3 * N_CIRCLES)
        vars_init[0::3] = c_init[:, 0]
        vars_init[1::3] = c_init[:, 1]
        vars_init[2::3] = r_init
        
        try:
            res = minimize(objective, vars_init, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 1500, 'ftol': 1e-10})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                best_radii = res.x[2::3]
        except Exception:
            pass

    # Strategy 2: Hexagonal lattice starts
    for seed in range(6):
        np.random.seed(100 + seed)
        r0 = 0.075 + np.random.rand() * 0.015
        rows = [6, 5, 6, 5, 4]
        c_hex = []
        y_curr = r0
        for k, cnt in enumerate(rows):
            x_start = r0 if k % 2 == 0 else 2 * r0
            for m in range(cnt):
                c_hex.append([x_start + m * 2 * r0, y_curr])
            y_curr += np.sqrt(3) * r0
        c_hex = np.array(c_hex)
        
        vars_init = np.empty(3 * N_CIRCLES)
        vars_init[0::3] = c_hex[:, 0]
        vars_init[1::3] = c_hex[:, 1]
        vars_init[2::3] = r0
        
        # Add controlled noise to break symmetry
        vars_init += np.random.randn(3 * N_CIRCLES) * 0.005
        for i in range(0, 3 * N_CIRCLES, 3):
            vars_init[i] = np.clip(vars_init[i], 0.01, 0.99)
            vars_init[i+1] = np.clip(vars_init[i+1], 0.01, 0.99)
            vars_init[i+2] = np.clip(vars_init[i+2], 0.01, 0.4)

                best_sum = -res.fun
                best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                best_radii = res.x[2::3]
        except Exception:
            pass
            
    # Strategy 3: Perturbation refinement on best found
    if best_centers is not None:
        for _ in range(8):
            c_pert = best_centers + np.random.randn(N_CIRCLES, 2) * 0.008
            c_pert = np.clip(c_pert, 0.01, 0.99)
            vars_pert = np.empty(3 * N_CIRCLES)
            vars_pert[0::3] = c_pert[:, 0]
            vars_pert[1::3] = c_pert[:, 1]

            try:

                    best_sum = -res.fun
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3]
            except Exception:
                pass
                
    # Fallback to safe valid configuration if optimization fails
    if best_centers is None:
        best_centers = np.array([[0.1*(i+0.5), 0.1*(j+0.5)] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_radii = np.full(N_CIRCLES, 0.09)
        best_sum = np.sum(best_radii)
        
    # Ensure non-negative radii
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_sum)
