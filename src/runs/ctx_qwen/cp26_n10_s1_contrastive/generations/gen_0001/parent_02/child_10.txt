import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """
    Returns a 1D array of constraint values.
    All constraints are formulated as g(vars) >= 0.
    Includes boundary constraints and pairwise non-overlap constraints.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    
    # Non-overlap constraints: dist_sq - (r_i + r_j)^2 >= 0
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    r_sum_sq = (r[:, np.newaxis] + r[np.newaxis, :])**2
    
    # Extract upper triangular indices for i < j to avoid duplicates and self-comparison
    i, j = np.triu_indices(N, k=1)
    c_pair = dist_sq[i, j] - r_sum_sq[i, j]
    
    return np.concatenate([c1, c2, c3, c4, c_pair])

def run_packing():
    best_sum = -np.inf
    best_vars = None
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    np.random.seed(42)
    
    inits = []
    
    # 1. Hexagonal lattice initialization
    r_init = 0.08
    y = r_init
    row = 0
    pts = []
    while len(pts) < N:
        x_off = (row % 2) * r_init
        x = r_init + x_off
        while x <= 1.0 - r_init + 1e-6 and len(pts) < N:
            pts.append([x, y, r_init])
            x += 2.0 * r_init
        y += np.sqrt(3.0) * r_init
        row += 1
    inits.append(np.array(pts).flatten())
    
    # 2. Square grid + 1 initialization
    pts2 = []
    for i in range(5):
        for j in range(5):
            pts2.append([0.1 + 0.2*i, 0.1 + 0.2*j, 0.08])
    pts2.append([0.5, 0.5, 0.02])
    inits.append(np.array(pts2).flatten())
    
    # 3. Randomized perturbations to escape local minima
    for _ in range(40):
        base = inits[0].copy()
        base += np.random.randn(len(base)) * 0.01
        # Ensure positions stay within bounds and start with small radii for guaranteed feasibility
        for i in range(0, len(base), 3):
            base[i] = np.clip(base[i], 0.05, 0.95)
            base[i+1] = np.clip(base[i+1], 0.05, 0.95)
            base[i+2] = 0.01  
        inits.append(base)
        
    # Multi-start optimization
    for init_vars in inits:
        try:
            res = minimize(objective, init_vars, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-10})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_vars = res.x.copy()
        except Exception:
            continue
            
    # High-precision refinement on the best configuration
    if best_vars is not None:
        res_final = minimize(objective, best_vars, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
        best_vars = res_final.x
        
    centers = best_vars[:2*N].reshape(N, 2)
    radii = best_vars[2*N:]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
