1. **Simulated Annealing (SA) on Centers + Linear Programming (LP) for Radii**: 
   Instead of optimizing centers and radii jointly, we decouple them. At each SA step, we randomly perturb one circle's center. Then, we solve a Linear Program to find the optimal radii for that fixed configuration. The LP maximizes $\sum r_i$ subject to $r_i + r_j \le d_{ij}$ and boundary constraints. This is extremely fast and guarantees the largest possible radii for any given layout.

2. **Multiple Restarts**: We run the SA+LP process from multiple perturbed hexagonal starts to ensure global exploration.

3. **High-Precision Refinement**: After SA finds a high-quality layout, we pass the best centers and radii to SLSQP for a joint refinement. This fine-tunes both positions and sizes simultaneously to numerical precision, squeezing out the last bits of radius sum.

4. **Constraint Handling**: All helper functions are defined at the top level with explicit arguments, avoiding closures and lambdas as required. The LP constraint matrix is precomputed to maximize speed during the SA loop.

import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26

def solve_radii_lp(centers, A_ub):
    """
    Solves a Linear Program to maximize sum of radii given fixed centers.
    Constraints: r_i >= 0, r_i <= boundary_dist, r_i + r_j <= dist_ij
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    b_ub = np.empty(4 * n + n * (n - 1) // 2)
    
    idx = 0
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        b_ub[idx] = x; idx += 1
        b_ub[idx] = 1.0 - x; idx += 1
        b_ub[idx] = y; idx += 1
        b_ub[idx] = 1.0 - y; idx += 1
        
    # Pairwise non-overlap constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            b_ub[idx] = dists[i, j]
            idx += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def compute_constraints(vars):
    """
    Computes inequality constraints >= 0 for SLSQP.
    Includes boundary and pairwise non-overlap constraints.
    """
    n = N_CIRCLES
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    c_boundary = np.column_stack([
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
    ]).flatten()
    
    dx = centers[:, np.newaxis, 0] - centers[np.newaxis, :, 0]
    dy = centers[:, np.newaxis, 1] - centers[np.newaxis, :, 1]
    dist_sq = dx**2 + dy**2
    rad_sum_sq = (radii[:, np.newaxis] + radii[np.newaxis, :])**2
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    c_overlap = dist_sq[i_idx, j_idx] - rad_sum_sq[i_idx, j_idx]
    
    return np.concatenate([c_boundary, c_overlap])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    
    # Precompute sparse-ish constraint matrix for LP to speed up SA loop
    A_ub = np.zeros((4 * n + n * (n - 1) // 2, n))
    idx = 0
    for i in range(n):
        A_ub[idx, i] = 1.0; idx += 1
        A_ub[idx, i] = 1.0; idx += 1
        A_ub[idx, i] = 1.0; idx += 1
        A_ub[idx, i] = 1.0; idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1

    best_centers = None
    best_sum = 0.0
    
    np.random.seed(42)
    
    # Stage 1: Simulated Annealing on centers with LP radius assignment
    for restart in range(4):
        # Hexagonal initialization with controlled density
        centers = np.zeros((n, 2))
        r_init = 0.045
        y_curr = r_init
        row = 0
        idx_c = 0
        while idx_c < n:
            x_off = (row % 2) * r_init
            x_curr = r_init + x_off
            while x_curr < 1.0 - r_init and idx_c < n:
                centers[idx_c] = [x_curr, y_curr]
                idx_c += 1
                x_curr += 2.0 * r_init
            y_curr += np.sqrt(3.0) * r_init
            row += 1
            
        # Add initial perturbation
        centers += np.random.randn(*centers.shape) * 0.025
        centers = np.clip(centers, 0.01, 0.99)
        
        radii, sum_r = solve_radii_lp(centers, A_ub)
        current_centers = centers.copy()
        current_sum = sum_r
        
        temp = 0.045
        for step in range(10000):
            i = np.random.randint(n)
            # Step size slowly decays to fine-tune near optimum
            move_size = 0.035 * (0.9997 ** step)
            new_c = current_centers[i].copy()
            new_c += np.random.randn(2) * move_size
            new_c = np.clip(new_c, 0.005, 0.995)
            
            current_centers[i] = new_c
            new_radii, new_sum = solve_radii_lp(current_centers, A_ub)
            
            delta = new_sum - current_sum
            # Metropolis criterion
            if delta > 0 or np.random.rand() < np.exp(delta / temp):
                current_sum = new_sum
                if delta < 0:
                    temp *= 0.9996
            else:
                current_centers[i] = new_c  # Revert move
                
            temp *= 0.99995
            
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = current_centers.copy()

    # Stage 2: High-precision joint refinement with SLSQP
    if best_centers is not None:
        _, best_radii = solve_radii_lp(best_centers, A_ub)
        
        # Flatten to [x0, y0, r0, x1, y1, r1, ...]
        x0 = np.concatenate([best_centers.flatten(), best_radii])
        
        bounds = []
        for _ in range(n):
            bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
            
        cons = {'type': 'ineq', 'fun': compute_constraints}
        
        # SLSQP objective: minimize negative sum of radii
        def objective(vars):
            return -np.sum(vars[2::3])
            
        try:
            res = minimize(objective, x0, method='SLSQP', 
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'iprint': -1})
            
            if res.success:
                centers_opt = res.x[:2*n].reshape(n, 2)
                radii_opt = res.x[2*n:]
                # Safety clamp for numerical drift
                radii_opt = np.maximum(radii_opt, 0.0)
                return centers_opt, radii_opt, float(np.sum(radii_opt))
        except Exception:
            pass
            
    # Fallback to best SA result
    _, radii_fallback = solve_radii_lp(best_centers, A_ub)
    return best_centers, radii_fallback, float(best_sum)
