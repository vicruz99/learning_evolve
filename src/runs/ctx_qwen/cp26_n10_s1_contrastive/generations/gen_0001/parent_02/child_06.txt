N = 26

def compute_objective(vars):

    return -np.sum(vars[2::3])

def compute_constraints(vars):
    """
    Returns a 1D array of inequality constraints >= 0.
    Enforces boundary containment and pairwise non-overlap.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    c = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.extend(x - r)
    c.extend(1.0 - x - r)
    c.extend(y - r)
    c.extend(1.0 - y - r)
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]

    r_sum = r[:, None] + r[None, :]

    c.extend(dist_sq[i, j] - r_sum[i, j]**2)
    
    return np.array(c)
