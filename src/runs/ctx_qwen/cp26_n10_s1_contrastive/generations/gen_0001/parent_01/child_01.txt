import numpy as np
from scipy.optimize import minimize

def objective(vars):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(vars[0::3])

def constraint_func(vars):
    """
    Constraint function: ensures non-overlap between all circle pairs.
    Returns array of constraint values >= 0.
    """
    radii = vars[0::3]
    u = vars[1::3]
    v = vars[2::3]
    n = 26
    
    # Transform normalized coordinates to actual positions
    # This parameterization automatically satisfies boundary constraints:
    # x in [r, 1-r] and y in [r, 1-r]
    x = radii + u * (1.0 - 2.0 * radii)
    y = radii + v * (1.0 - 2.0 * radii)
    
    # Compute pairwise squared distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    
    # Compute squared minimum allowed distances
    r_sum = radii[:, None] + radii[None, :]
    min_dist_sq = r_sum**2
    
    # Extract upper triangular part (i < j) to avoid duplicates and self
    i, j = np.triu_indices(n, k=1)
    return dist_sq[i, j] - min_dist_sq[i, j]

def force_layout_init(n, seed, steps=800):
    """Generates a well-packed initial configuration using repulsion forces."""
    np.random.seed(seed)
    
    # Start with a perturbed hexagonal grid for better structure
    pts = []
    y = 0.1
    row = 0
    while len(pts) < n:
        x = 0.1
        shift = (row % 2) * 0.1
        while x <= 0.9 and len(pts) < n:
            pts.append([x + shift, y])
            x += 0.2
        y += 0.1732 # approx sqrt(3)/2 * 0.2
        row += 1
        
    centers = np.array(pts[:n])
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    target_r = 0.095
    radii = np.full(n, target_r)
    
    for t in range(steps):
        diff = centers[:, None, :] - centers[None, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        dist = np.sqrt(np.maximum(dist_sq, 1e-12))
        np.fill_diagonal(dist, 1e-12)
        
        r_sum = radii[:, None] + radii[None, :]
        overlap = np.maximum(0, r_sum - dist)
        # Force magnitude proportional to overlap, softened by distance
        force_mag = overlap / (dist + 1e-8)
        
        # Sum forces along each axis
        forces = np.sum(force_mag[:, :, None] * diff, axis=1)
        
        # Vectorized wall repulsion
        wall_f = np.zeros_like(centers)
        wall_f[:, 0] += np.maximum(0, radii - centers[:, 0]) * 30.0
        wall_f[:, 0] -= np.maximum(0, centers[:, 0] - (1.0 - radii)) * 30.0
        wall_f[:, 1] += np.maximum(0, radii - centers[:, 1]) * 30.0
        wall_f[:, 1] -= np.maximum(0, centers[:, 1] - (1.0 - radii)) * 30.0
        
        # Adaptive step size that decays over time
        step_size = 0.025 / (1.0 + t * 0.015)
        centers += (forces + wall_f) * step_size
        centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
        
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sol = None
    best_val = -np.inf
    
    # Bounds: r in [1e-4, 0.5], u in [0, 1], v in [0, 1]
    bounds = [(1e-4, 0.5), (0, 1), (0, 1)] * n
