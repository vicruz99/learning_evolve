import numpy as np
from scipy.optimize import minimize

def objective(vars):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """
    Returns inequality constraint values >= 0.
    Enforces boundary containment and non-overlap between all circle pairs.
    """
    n = 26
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c_boundary = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    rad_sum_sq = r_sum**2
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c_pair = dist_sq[mask] - rad_sum_sq[mask]
    
    return np.concatenate([c_boundary, c_pair])

def run_packing():
    n = 26
    best_vars = None
    best_val = np.inf
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Multiple restarts with perturbed hexagonal initialization
    for attempt in range(15):
        np.random.seed(attempt)
        
        # Hexagonal lattice base configuration
        rows = [5, 6, 5, 6, 4]
        pts = []
        y = 0.12
        for i, cnt in enumerate(rows):
            x = 0.12 + (i % 2) * 0.06
            for _ in range(cnt):
                pts.append([x, y])
                x += 0.12
            y += 0.105
        pts = np.array(pts[:n])
        
        # Add controlled perturbation to break symmetry
        pts += np.random.uniform(-0.03, 0.03, (n, 2))
        pts = np.clip(pts, 0.05, 0.95)
        
        # Compute strictly feasible initial radius
        d_min = 1.0
        for i in range(n):
            d_min = min(d_min, pts[i, 0], 1.0 - pts[i, 0], 
                             pts[i, 1], 1.0 - pts[i, 1])
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((pts[i] - pts[j]) ** 2))
                if d < d_min:
                    d_min = d
        r0 = d_min * 0.85  # Start safely inside feasible region
        
        vars0 = np.empty(3 * n)
        vars0[0::3] = pts[:, 0]
        vars0[1::3] = pts[:, 1]
        vars0[2::3] = r0
        
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
            if res.fun < best_val:
                best_val = res.fun
                best_vars = res.x.copy()
        except Exception:
            continue
            
    # High-precision refinement on the best configuration found
    if best_vars is not None:
        try:
            res_final = minimize(objective, best_vars, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
            best_vars = res_final.x
        except Exception:
            pass

    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(-best_val)
