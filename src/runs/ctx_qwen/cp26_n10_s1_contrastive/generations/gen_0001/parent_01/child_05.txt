from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[2::3])

def constraint_func(vars_vec):
    """
    Computes all inequality constraints (must be >= 0).
    Includes boundary constraints and pairwise non-overlap constraints.
    Fully vectorized for performance.
    """
    n = N_CIRCLES
    x = vars_vec[0::3]
    y = vars_vec[1::3]
    r = vars_vec[2::3]
    
    # Pre-allocate constraint array
    n_boundary = 4 * n
    n_pairwise = n * (n - 1) // 2
    c = np.empty(n_boundary + n_pairwise)
    
    # Boundary constraints
    c[0:n] = x - r
    c[n:2*n] = 1.0 - x - r
    c[2*n:3*n] = y - r
    c[3*n:4*n] = 1.0 - y - r
    
    # Pairwise non-overlap constraints: dist_sq >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    r_sum_sq = r_sum**2
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    c[4*n:] = dist_sq[i_idx, j_idx] - r_sum_sq[i_idx, j_idx]
    
    return c
