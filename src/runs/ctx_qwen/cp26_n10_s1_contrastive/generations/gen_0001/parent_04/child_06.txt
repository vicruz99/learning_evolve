from scipy.optimize import minimize

# Precompute indices for pairwise constraints to avoid repeated allocation
I_IDX, J_IDX = np.triu_indices(26, k=1)

def objective(vars_vec):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(vars_vec[2::3])

def constraint_func(vars_vec):
    """
    Returns array of constraint values that must be >= 0.
    Includes boundary constraints and pairwise non-overlap constraints.
    """
    x = vars_vec[0::3]
    y = vars_vec[1::3]
    r = vars_vec[2::3]
    
    # Preallocate constraint array: 4*26 boundary + 325 pairwise = 429
    c = np.empty(429)
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c[:26] = x - r
    c[26:52] = 1.0 - x - r
    c[52:78] = y - r
    c[78:104] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    r_sum = r[I_IDX] + r[J_IDX]
    c[104:] = dx**2 + dy**2 - r_sum**2
    
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0.0, 1.0)] * 52 + [(0.0, 0.5)] * 26
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_sum = -np.inf

    inits = []

    r0 = 0.08

            x += 2.0 * r0

        row += 1
    pts = np.array(pts[:n])
