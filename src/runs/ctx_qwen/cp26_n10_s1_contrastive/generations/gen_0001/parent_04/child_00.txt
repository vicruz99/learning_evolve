import numpy as np
from scipy.optimize import minimize

def objective(x):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(x[2::3])

def constraint_func(x):
    """
    Computes all inequality constraints.
    Returns an array where all elements must be >= 0.
    Constraints:
    1. Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    2. Pairwise: dist^2 >= (r_i + r_j)^2
    """
    N = 26
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints (4 per circle)
    c_boundary = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Pairwise non-overlap constraints (vectorized)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    c_pairwise = dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx]**2
    
    return np.concatenate([c_boundary, c_pairwise])

def get_init_5x5(seed):
    """Initializes a 5x5 grid plus one center circle."""
    np.random.seed(seed)
    grid_step = 0.2
    x_g = np.arange(0.1, 1.0, grid_step)
    y_g = np.arange(0.1, 1.0, grid_step)
    
    xs = []
    ys = []
    for x in x_g:
        for y in y_g:
            xs.append(x)
            ys.append(y)
            
    xs.append(0.5)
    ys.append(0.5)
    
    xs = np.array(xs) + np.random.uniform(-0.02, 0.02, 26)
    ys = np.array(ys) + np.random.uniform(-0.02, 0.02, 26)
    xs = np.clip(xs, 0.05, 0.95)
    ys = np.clip(ys, 0.05, 0.95)
    rs = np.full(26, 0.09)
    
    return np.concatenate([xs, ys, rs])

def get_init_hex(seed):
    """Initializes a hexagonal packing."""
    np.random.seed(seed)
    r_init = 0.075
    xs = []
    ys = []
    y = r_init
    row = 0
    while len(xs) < 26:
        x = r_init
        if row % 2 == 1:
            x += r_init
        while x <= 1.0 - r_init and len(xs) < 26:
            xs.append(x)
            ys.append(y)
            x += 2.0 * r_init
        y += np.sqrt(3) * r_init
        row += 1
        
    xs = np.array(xs) + np.random.uniform(-0.01, 0.01, 26)
    ys = np.array(ys) + np.random.uniform(-0.01, 0.01, 26)
    xs = np.clip(xs, 0.01, 0.99)
    ys = np.clip(ys, 0.01, 0.99)
    rs = np.full(26, r_init)
    
    return np.concatenate([xs, ys, rs])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    best_x = None
    best_val = -np.inf
    
    # Variable bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = [{'type': 'ineq', 'fun': constraint_func}]
    
    # Try multiple structured initializations
    init_funcs = [get_init_5x5, get_init_hex]
    
    for func in init_funcs:
        for seed in range(6):
            x0 = func(seed)
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                              constraints=cons,
                              options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if res.success:
                    val = -res.fun
                    # Verify feasibility with a small tolerance
                    if np.min(constraint_func(res.x)) >= -1e-6:
                        if val > best_val:
                            best_val = val
                            best_x = res.x.copy()
            except Exception:
                pass
                
    # Local perturbation search from the best solution found to escape local minima
    if best_x is not None:
        for pert_seed in range(8):
            np.random.seed(pert_seed + 1000)
            x_pert = best_x + np.random.uniform(-0.003, 0.003, 3*N)
            x_pert = np.clip(x_pert, 0.001, 0.999)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                              constraints=cons,
                              options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                if res.success:
                    val = -res.fun
                    if np.min(constraint_func(res.x)) >= -1e-6:
                        if val > best_val:
                            best_val = val
                            best_x = res.x.copy()
            except Exception:
                pass
                
    # Fallback if optimization fails entirely
    if best_x is None:
        best_x = get_init_5x5(0)
        best_val = np.sum(best_x[2::3])
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Ensure non-negative radii
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(best_val)
