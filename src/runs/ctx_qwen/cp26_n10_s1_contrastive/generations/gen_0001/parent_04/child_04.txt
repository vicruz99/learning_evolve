import numpy as np
from scipy.optimize import minimize

N = 26
MASK = np.triu(np.ones((N, N), dtype=bool), k=1)

def objective(vars):
    """Objective function: minimize negative sum of radii (maximize sum)."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """
    Vectorized inequality constraints:
    1. Boundary: circle inside [0,1]x[0,1]
    2. Non-overlap: distance >= sum of radii
    Returns array of constraint values (must be >= 0).
    """
    centers = vars[:2*N].reshape(N, 2)
    radii = vars[2*N:]
    
    # Boundary constraints
    c1 = centers[:, 0] - radii
    c2 = 1.0 - centers[:, 0] - radii
    c3 = centers[:, 1] - radii
    c4 = 1.0 - centers[:, 1] - radii
    
    # Pairwise non-overlap constraints (vectorized)
    dx = centers[:, 0, np.newaxis] - centers[:, 0]
    dy = centers[:, 1, np.newaxis] - centers[:, 1]
    dist_sq = dx**2 + dy**2
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    c_pair = dist_sq[MASK] - r_sum[MASK]**2
    
    return np.concatenate([c1, c2, c3, c4, c_pair])

def make_vars(pts, radii):
    """Helper to flatten positions and radii into optimization vector."""
    v = np.empty(3*N)
    v[:2*N] = pts.flatten()
    v[2*N:] = radii
    return v

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_sum = -np.inf

        y = r0
        row = 0

            x = r0 + shift
            while x <= 1.0 - r0 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
        pts = np.array(pts[:N])
        pts = np.clip(pts, r0 + 1e-5, 1.0 - r0 - 1e-5)

        pts = np.random.uniform(0.1, 0.9, (N, 2))

            forces = np.zeros_like(pts)

                for j in range(i+1, N):

                    d = np.linalg.norm(diff)

                    forces[i] += diff / (d**2)
