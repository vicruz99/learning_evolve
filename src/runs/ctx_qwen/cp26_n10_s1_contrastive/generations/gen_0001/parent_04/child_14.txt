from scipy.optimize import minimize

def objective_26(vars_vec):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)."""
    return -np.sum(vars_vec[2::3])

def constraint_bounds_26(vars_vec):
    """Boundary constraints: circles must stay inside [0,1]x[0,1]."""
    n = 26
    c = np.empty(4 * n)
    for i in range(n):
        idx = 3 * i
        x, y, r = vars_vec[idx], vars_vec[idx+1], vars_vec[idx+2]
        c[4*i] = x - r
        c[4*i+1] = 1.0 - x - r
        c[4*i+2] = y - r
        c[4*i+3] = 1.0 - y - r
    return c

def constraint_overlap_26(vars_vec):
    """Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2."""
    n = 26
    m = n * (n - 1) // 2
    c = np.empty(m)
    k = 0
    for i in range(n):

            c[k] = (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
            k += 1
    return c
