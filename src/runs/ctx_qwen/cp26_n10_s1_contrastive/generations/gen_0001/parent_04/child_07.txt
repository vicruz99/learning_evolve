import numpy as np
from scipy.optimize import minimize

def obj_26(v):
    """Objective: maximize sum of radii <=> minimize negative sum."""
    return -np.sum(v[2::3])

def cons_26(v):
    """
    Constraint function: returns array of values that must be >= 0.
    Includes 4 boundary constraints per circle and pairwise separation constraints.
    Fully vectorized for N=26.
    """
    n = 26
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    
    # Preallocate constraint array
    n_bound = 4 * n
    n_pair = n * (n - 1) // 2
    c = np.empty(n_bound + n_pair)
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c[0::4] = cx - r
    c[1::4] = 1.0 - cx - r
    c[2::4] = cy - r
    c[3::4] = 1.0 - cy - r
    
    # Pairwise separation: dist^2 >= (r_i + r_j)^2
    pts = np.stack((cx, cy), axis=1)
    diff = pts[:, None, :] - pts[None, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    r_sum = r[:, None] + r[None, :]
    r_sum_sq = r_sum**2
    
    # Extract upper triangular part (i < j)
    idx = np.triu_indices(n, k=1)
    c[n_bound:] = dist_sq[idx] - r_sum_sq[idx]
    
    return c

def generate_hex_init(n, perturb_scale):
    """Generate a hexagonal lattice initialization with random perturbation."""
    pts = []
    r_init = 0.06
    dx = 2.0 * r_init * 1.1
    dy = np.sqrt(3) * r_init * 1.1
    y = r_init
    row = 0
    while len(pts) < n:
        x = r_init
        if row % 2 == 1:
            x += dx / 2.0
        while x <= 1.0 - r_init and len(pts) < n:
            pts.append([x, y])
            x += dx
        y += dy
        row += 1
        
    centers = np.array(pts[:n])
    centers += np.random.uniform(-perturb_scale, perturb_scale, centers.shape)
    return np.clip(centers, 0.05, 0.95)

def generate_force_init(n, steps=300):
    """Generate a force-directed layout initialization for even spread."""
    pts = np.random.uniform(0.15, 0.85, (n, 2))
    for _ in range(steps):
        forces = np.zeros_like(pts)
        
        # Pairwise repulsion
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.linalg.norm(diff, axis=2)
        dist = np.clip(dist, 1e-4, None)
        rep = diff / (dist[:, :, None] * dist)
        np.fill_diagonal(rep, 0)
        forces += np.sum(rep, axis=1)
        
        # Wall repulsion
        for d in range(2):
            wall_dist_low = pts[:, d] - 0.05
            wall_dist_high = 0.95 - pts[:, d]
            forces[:, d] += np.where(wall_dist_low < 0.0, 10.0 * wall_dist_low, 0.0)
            forces[:, d] += np.where(wall_dist_high < 0.0, -10.0 * wall_dist_high, 0.0)
            
        pts += forces * 0.005
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def safe_radii(centers, n):
    """Compute strictly feasible initial radii based on local geometry."""
    pts = centers
    r_init = np.full(n, 0.01)
    for i in range(n):
        d_to_wall = min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])
        d_to_others = np.linalg.norm(pts - pts[i], axis=1)
        d_to_others[i] = np.inf
        d_min = min(d_to_wall, np.min(d_to_others) / 2.0)
        r_init[i] = d_min * 0.65  # Leave room for optimizer to expand
    return r_init

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
