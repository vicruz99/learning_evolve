import numpy as np
from scipy.optimize import minimize

_N = 26
_TRIU_IDX = np.triu_indices(_N, k=1)

def objective_26(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_26(vars):
    """
    Vectorized constraint function.
    Returns array of values that must be >= 0.
    Constraints: 
    1. Boundary: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    2. Pairwise: dist^2 - (r_i + r_j)^2 >= 0
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    n = _N
    total_c = 4 * n + n * (n - 1) // 2
    c = np.empty(total_c)
    
    # Boundary constraints
    c[0:n] = x - r
    c[n:2*n] = 1.0 - x - r
    c[2*n:3*n] = y - r
    c[3*n:4*n] = 1.0 - y - r
    
    # Pairwise separation constraints
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    r_sum_sq = r_sum**2
    
    c[4*n:] = dist_sq[_TRIU_IDX] - r_sum_sq[_TRIU_IDX]
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = _N
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-5, 0.5)] * n
    
    best_sum = -1.0
    best_vars = None
    
    # Precompute diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice approximation
    r_h = 0.075
    pts = []
    y = r_h
    row = 0
    while len(pts) < n:
        x = r_h + (0.5 if row % 2 == 1 else 0.0) * 2 * r_h
        while x <= 1 - r_h and len(pts) < n:
            pts.append([x, y])
            x += 2 * r_h
        y += np.sqrt(3) * r_h
        row += 1
    inits.append(np.array(pts[:n]))
    
    # 2. 5x5 Grid + Center
    grid = []
    for i in range(5):
        for j in range(5):
            grid.append([0.1 + 0.2 * i, 0.1 + 0.2 * j])
    grid.append([0.5, 0.5])
    inits.append(np.array(grid))
    
    # 3. Random uniform starts
    for _ in range(5):
        inits.append(np.random.rand(n, 2))
        
    n_restarts = 30
    
    for seed in range(n_restarts):
        np.random.seed(seed)
        base = inits[seed % len(inits)].copy()
        
        # Perturb to break symmetry and explore basins
        base += np.random.normal(0, 0.025, (n, 2))
        base = np.clip(base, 0.05, 0.95)
        
        # Compute strictly feasible initial radius
        r_safe = 0.05
        for i in range(n):
            r_safe = min(r_safe, base[i, 0], 1.0 - base[i, 0], 
                              base[i, 1], 1.0 - base[i, 1])
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((base[i] - base[j]) ** 2))
                r_safe = min(r_safe, d / 2.0)
        r_safe *= 0.9 # Safety margin
        
        x0 = np.zeros(3 * n)
        x0[0::3] = base[:, 0]
        x0[1::3] = base[:, 1]
        x0[2::3] = r_safe
        
        try:
            res = minimize(objective_26, x0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraint_26},
                           options={'maxiter': 800, 'ftol': 1e-11})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                # Accept if constraints are satisfied within tolerance
                if np.min(constraint_26(res.x)) >= -1e-6:
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            pass
            
    # Fallback if optimization fails completely
    if best_vars is None:
        best_vars = np.zeros(3 * n)
        best_vars[0::3] = np.linspace(0.1, 0.9, n)
        best_vars[1::3] = 0.5
        best_vars[2::3] = 0.01

        pass
