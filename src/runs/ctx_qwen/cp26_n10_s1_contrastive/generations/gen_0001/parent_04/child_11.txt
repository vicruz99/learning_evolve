2. **Radius Maximization (LP)**: For each optimized center configuration, fix the positions and solve a Linear Program to find the theoretically maximum possible radii. This step guarantees that for any valid layout found by the NLP, we extract the absolute maximum sum of radii possible. The LP solver (`scipy.optimize.linprog` with HiGHS) is extremely fast and numerically robust for this size.

This alternating/sequential approach decouples the non-convex geometry problem (placing centers) from the convex sizing problem (choosing radii), significantly improving both stability and final score. Multiple restarts ensure we explore the configuration space thoroughly.

import numpy as np
from scipy.optimize import minimize, linprog

def objective(vars_vec):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_vec[2::3])

def constraints(vars_vec):
    """
    Returns inequality constraint values >= 0.
    Includes boundary constraints and squared distance pairwise constraints.
    """
    n = 26
    x = vars_vec[0::3]
    y = vars_vec[1::3]
    r = vars_vec[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    # Extract upper triangular part to avoid duplicates and self-comparisons
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c = np.concatenate([c, dist_sq[mask] - r_sum[mask]**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Run multiple restarts with perturbed hexagonal initializations
    for seed in range(30):
        np.random.seed(seed)
        
        # 1. Initialization: Hexagonal grid
        pts = []
        r_est = 0.09
        y = r_est
        row = 0
        while len(pts) < n:
            x = r_est + (0.5 if row % 2 else 0) * 2 * r_est
            while x < 1.0 - r_est and len(pts) < n:
                pts.append([x, y])
                x += 2 * r_est
            y += np.sqrt(3) * r_est
            row += 1
            
        centers_init = np.array(pts[:n])
        radii_init = np.full(n, r_est)
        
        # Add controlled jitter to break symmetry and explore space
        centers_init += np.random.uniform(-0.02, 0.02, (n, 2))
        centers_init = np.clip(centers_init, 0.05, 0.95)
        
        # Flatten to optimization vector: [x0, y0, r0, x1, y1, r1, ...]
        x0 = np.zeros(3*n)
        for i in range(n):
            x0[3*i] = centers_init[i, 0]
            x0[3*i+1] = centers_init[i, 1]
            x0[3*i+2] = radii_init[i]
            
        # 2. NLP Phase: Optimize positions and radii simultaneously
        try:
            res = minimize(objective, x0, method='SLSQP', 
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-9})
            
            c_temp = np.column_stack((res.x[0::3], res.x[1::3]))
            
            # 3. LP Refinement Phase: Maximize radii for fixed optimal centers
            # Given centers, maximize sum(r_i) s.t. r_i >= 0, r_i <= boundary, r_i + r_j <= dist_ij
            c_obj = -np.ones(n)
            A_ub = []
            b_ub = []
            
            # Boundary constraints for LP: r_i <= min(x, 1-x, y, 1-y)
            for i in range(n):
                xb = min(c_temp[i, 0], 1.0 - c_temp[i, 0])
                yb = min(c_temp[i, 1], 1.0 - c_temp[i, 1])
                b_val = max(0.0, min(xb, yb))
                row = np.zeros(n)
                row[i] = 1.0
                A_ub.append(row)
                b_ub.append(b_val)
                
            # Pairwise constraints for LP: r_i + r_j <= dist_ij
            # Vectorized distance computation
            dists = np.sqrt(((c_temp[:, None, :] - c_temp[None, :, :])**2).sum(axis=2))
            for i in range(n):
                for j in range(i+1, n):
                    row = np.zeros(n)
                    row[i] = 1.0
                    row[j] = 1.0
                    A_ub.append(row)
                    b_ub.append(dists[i, j])
                    
            A_ub = np.array(A_ub)
            b_ub = np.array(b_ub)
            
            # Solve LP using HiGHS (fast and robust)
            lp_res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0.0, None), method='highs')
            
            if lp_res.success:
                r_lp = lp_res.x
                curr_sum = np.sum(r_lp)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = c_temp.copy()
                    best_radii = r_lp.copy()
        except Exception:
            continue
            
    # Fallback in case all optimizations fail (highly unlikely)
    if best_centers is None:
        best_centers = np.random.rand(n, 2) * 0.8 + 0.1
        best_radii = np.full(n, 0.01)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
