import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((4 * n + n * (n - 1) // 2, n))
    b_ub = np.zeros(4 * n + n * (n - 1) // 2)
    k = 0
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        bounds_vals = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_vals:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def to_params(centers, radii):
    """Convert centers and radii to the (r, u, v) parameterization."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    centers = np.column_stack((x, y))
    return centers, r

def objective(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints(params):
    """Computes inequality constraints g(params) >= 0 for pairwise non-overlap."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def gen_hex(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) >= N: break
            if x > 1.0 - r_est: break
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        if len(pts) >= N: break
        
    pts = np.array(pts[:N])
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    r = np.full(N, 0.08)
    return pts, r

def gen_force(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.1, 0.9, (N, 2))
    r = np.full(N, 0.08)
    
    for step in range(300):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-5)
        rep = 1.0 / (dists**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        
        for d in range(2):
            forces[:, d] += 10.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 10.0 * np.maximum(0, pts[:, d] - 0.95)
            
        step_size = 0.005 * (0.995**step)
        pts += step_size * forces
        pts = np.clip(pts, 0.05, 0.95)
        
    for i in range(N):
        dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
        r[i] = 0.85 * min(dw, dp/2.0)
        
    return pts, r

def run_packing():
    rng = np.random.default_rng(42)
    bounds_opt = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_p = None
    best_s = -np.inf
    
    inits = []
    
    # Diverse hexagonal patterns
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [7,5,5,5,4],
        [4,5,5,5,7], [5,7,5,5,4], [4,5,5,7,5], [7,6,6,7],
        [8,6,6,6], [9,5,6,6], [10,8,8], [4,4,6,6,6]
    ]
    
    for pat in patterns:
        for _ in range(3):
            rot = rng.uniform(-0.35, 0.35)
            scale = rng.uniform(0.85, 1.15)
            pts, r = gen_hex(rng, pat, rot, scale)
            inits.append(to_params(pts, r))
            
    # Force-directed layouts
    for seed in range(15):
        pts, r = gen_force(rng, seed)
        inits.append(to_params(pts, r))
        
    # Phase 1: Broad SLSQP search
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-7:
                    s = -res.fun
                    if s > best_s:
                        best_s = s
                        best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is not None:
        # Phase 2: Perturbation refinement to escape local minima
        for _ in range(50):
            xp = best_p.copy()
            xp[:N] += rng.normal(0, 0.002, N)
            xp[N:2*N] += rng.normal(0, 0.04, N)
            xp[2*N:3*N] += rng.normal(0, 0.04, N)
            xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
            xp[N:] = np.clip(xp[N:], 0.0, 1.0)
            
            try:
                res = minimize(objective, xp, method='SLSQP', bounds=bounds_opt,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
                if res.success:
                    c_val = constraints(res.x)
                    if np.min(c_val) >= -1e-7:
                        s = -res.fun
                        if s > best_s:
                            best_s = s
                            best_p = res.x.copy()
            except Exception:
                continue
                
        # Phase 3: LP-driven coordinate ascent on centers
        c_best, r_best = from_params(best_p)
        rad_lp, lp_sum = solve_lp(c_best)
        if lp_sum > best_s:
            best_s = lp_sum
            best_p = to_params(c_best, rad_lp)
            
        current_c = c_best.copy()
        current_s = best_s
        step = 0.025
        for _ in range(400):
            idx = rng.integers(N)
            direction = rng.standard_normal(2)
            direction /= np.linalg.norm(direction)
            nc = current_c.copy()
            nc[idx] = np.clip(nc[idx] + step * direction, 0.02, 0.98)
            _, ns = solve_lp(nc)
            
            if ns > current_s + 1e-7:
                current_c = nc
                current_s = ns
                step = np.maximum(0.001, step * 0.92)
            else:
                step = np.minimum(0.04, step * 1.03)
                
        if current_s > best_s:
            best_s = current_s
            c_best = current_c
            rad_lp, _ = solve_lp(c_best)
            best_p = to_params(c_best, rad_lp)
            
        # Phase 4: High-precision final polish
        final_p = best_p.copy()
        final_p[:N] *= 0.99995 # Slight shrink for strict interior feasibility
        try:
            res_f = minimize(objective, final_p, method='SLSQP', bounds=bounds_opt,
                             constraints=cons, options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(constraints(res_f.x)) >= -1e-8:
                best_p = res_f.x
                best_s = -res_f.fun
        except Exception:
            pass
            
    # Fallback configuration (should rarely be reached)
    if best_p is None:
        pts = np.array([[0.1+0.2*i, 0.1+0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        r = np.full(N, 0.09)
        best_p = to_params(pts, r)
        best_s = np.sum(r)
        
    centers, radii = from_params(best_p)
    return centers, radii, float(best_s)
