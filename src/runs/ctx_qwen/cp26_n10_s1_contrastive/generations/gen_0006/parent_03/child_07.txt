To maximize the sum of radii for 26 circles in a unit square, I will employ a hybrid optimization strategy that combines:
1. **Diverse Initializations**: Generate multiple rotated and scaled hexagonal lattice patterns, which are theoretically optimal for dense circle packings. Variations in row counts, rotation angles, and jitter ensure exploration of different boundary alignments and contact topologies.
2. **Parameterized SLSQP Optimization**: Use the `(r, u, v)` parameterization where `x = r + u*(1-2r)` and `y = r + v*(1-2r)`. This automatically satisfies boundary constraints, reducing the problem to only pairwise non-overlap constraints and significantly improving solver stability.
3. **LP-Driven Coordinate Ascent**: After an initial SLSQP run, fix the centers and solve a Linear Program to find the mathematically maximal radii. Then, perform a coordinate ascent on centers: randomly move one circle, re-solve the LP, and accept the move if the sum of radii increases. This smooths over non-convexities in the joint optimization landscape.
4. **Perturbation & Escape**: Inject controlled noise into the best configuration, re-run SLSQP, and repeat the coordinate ascent to escape local minima.
5. **High-Precision Polish**: Final SLSQP run with tight tolerances, followed by a final LP evaluation to extract the absolute maximal radii for the converged centers.

This approach separates the non-smooth radius optimization (handled exactly by LP) from the geometric center optimization (handled smoothly by SLSQP and coordinate ascent), reliably pushing the sum of radii beyond previous bests.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(params):
    """Minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def solve_lp(centers):
    """Given fixed centers, solves LP to find optimal feasible radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i,j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)]*n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Map (r, u, v) parameters back to physical centers and radii."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def gen_hex_pattern(rng, pattern, rot, scale, jitter):
    """Generate a hexagonal lattice initialization with specified parameters."""
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(pattern):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    pts += rng.uniform(-jitter, jitter, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    
    # Estimate initial radii safely
    dw = np.minimum(np.minimum(pts[:,0], 1.0-pts[:,0]), np.minimum(pts[:,1], 1.0-pts[:,1]))
    dp = np.linalg.norm(pts[:, np.newaxis, :] - pts[np.newaxis, :, :], axis=2)
    np.fill_diagonal(dp, np.inf)
    d_pair = np.min(dp, axis=1)
    rad = 0.85 * np.minimum(dw, d_pair/2.0)
    return pts, rad

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_p = None
    best_s = -np.inf
    best_c = None
    best_r = None
    
    # Diverse hexagonal patterns covering various boundary alignments
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5],
        [5,5,5,5,6], [6,4,6,5,5], [5,6,4,6,5],
        [4,5,6,5,6], [6,6,5,5,4], [5,5,6,5,5],
        [6,5,5,5,5], [5,5,5,6,5], [7,5,5,5,4],
        [4,5,5,5,7], [5,7,5,5,4], [4,5,5,7,5],
        [6,6,6,4,4], [5,5,5,5,5,1], [7,6,6,5,2],
        [5,5,6,6,4], [6,5,6,4,5]
    ]
    
    # Generate diverse starts
    starts = []
    for pat in patterns:

            rot = rng.uniform(-0.4, 0.4)
            scale = rng.uniform(0.85, 1.15)
            jit = rng.uniform(0.005, 0.025)
            pts, rads = gen_hex_pattern(rng, pat, rot, scale, jit)
            starts.append(to_params(pts, rads))
            
    # Phase 1: Broad SLSQP search from diverse initializations
    for x0 in starts:
        try:

                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_s:
                    best_s = s
                    best_p = res.x.copy()
                    best_c, best_r = from_params(best_p)
        except Exception:
            continue

    # Phase 2: LP-based Coordinate Ascent on Centers

            current_s = ns
            best_s = current_s

        else:
            step = np.minimum(0.05, step * 1.05)
            
    best_p = to_params(best_c, best_r)
    
    # Phase 3: Perturbation + SLSQP to escape local minima
    for _ in range(25):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.001, N)
        xp[N:] += rng.normal(0, 0.015, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:] = np.clip(xp[N:], 0.0, 1.0)
        try:
            res = minimize(objective, xp, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_s:
                    best_s = s
                    best_p = res.x.copy()
                    best_c, best_r = from_params(best_p)
                    
                    # Follow up with coordinate ascent from new best
                    current_c = best_c.copy()
                    current_s = best_s
                    step = 0.02
                    for _ in range(300):
                        idx = rng.integers(N)
                        direction = rng.standard_normal(2)
                        direction /= np.linalg.norm(direction)
                        nc = current_c.copy()
                        nc[idx] = np.clip(nc[idx] + step * direction, 0.01, 0.99)
                        nr, ns = solve_lp(nc)
                        if ns > current_s + 1e-6:
                            current_c = nc
                            current_s = ns
                            best_s = current_s
                            best_c = current_c.copy()

        pass
        
    # Final LP extraction to guarantee maximal radii for converged centers
    r_lp, s_lp = solve_lp(best_c)
    if s_lp > best_s - 1e-7:
        best_r = r_lp
        best_s = s_lp
        
    return best_c, np.maximum(best_r, 0.0), float(best_s)
