This avoids the non-convexity of joint optimization early on and allows efficient exploration of the center configuration space.

SA moves one circle at a time by a random step, clips to boundaries, and evaluates the maximum possible sum of radii via an LP. This separates the combinatorial geometry of placement from the continuous sizing problem.
2. **Coordinate Ascent Refinement**: After SA, we perform a local hill-climbing phase where we greedily move circles in random directions, accepting only moves that strictly increase the LP-derived sum of radii. Step sizes decay to fine-tune positions.
3. **Rotation Search**: Boundary alignment significantly impacts packing density in a square. We rotate the best configuration across a range of angles (±15°), re-evaluating the LP sum to find optimal boundary interactions.
4. **SLSQP Joint Polish**: Finally, we map the best centers/radii to a boundary-safe parameterization `(r, u, v)` and run SLSQP to jointly optimize positions and sizes. This handles remaining slack and pushes the solution to the mathematical boundary of feasibility.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((4*n + n*(n-1)//2, n))
    b_ub = np.zeros(4*n + n*(n-1)//2)
    k = 0
    
    xs = centers[:, 0]
    ys = centers[:, 1]
    for i in range(n):
        A_ub[k, i] = 1.0; b_ub[k] = xs[i]; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - xs[i]; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = ys[i]; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - ys[i]; k += 1
        
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 0.0

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj_slqp(p):
    return -np.sum(p[:N])

def cons_slqp(p):
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx*dx + dy*dy
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def run_packing():
    rng = np.random.default_rng(42)
    best_centers = None
    best_radii = None
    best_sum = 0.0

    n_runs = 12
    sa_iters = 1500
    temp_init = 0.025
    step_init = 0.05

    # Phase 1: Simulated Annealing on centers with LP evaluation
    for run_idx in range(n_runs):
        pts = []
        r_est = 0.09
        y = r_est
        row = 0
        while len(pts) < N:
            shift = (row % 2) * r_est
            x = r_est + shift
            while x <= 1.0 - r_est and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
        c = np.array(pts[:N])
        c += rng.uniform(-0.04, 0.04, (N, 2))
        c = np.clip(c, 0.05, 0.95)

        r_curr, curr_sum = solve_lp(c)
        best_c_run = c.copy()
        best_r_run = r_curr.copy()
        best_sum_run = curr_sum

        temp = temp_init
        step = step_init
        
        for _ in range(sa_iters):
            idx = rng.integers(N)
            nc = c.copy()
            nc[idx] += rng.uniform(-step, step, 2)
            nc[idx] = np.clip(nc[idx], 1e-5, 1.0-1e-5)
            
            nr, ns = solve_lp(nc)
            delta = ns - curr_sum
            
            if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
                c = nc
                r_curr = nr
                curr_sum = ns
                if ns > best_sum_run:
                    best_sum_run = ns
                    best_c_run = c.copy()
                    best_r_run = r_curr.copy()
            temp *= 0.9992
            step *= 0.9996
            
        if best_sum_run > best_sum:
            best_sum = best_sum_run
            best_centers = best_c_run
            best_radii = best_r_run

    # Phase 2: Coordinate Ascent Refinement
    c = best_centers.copy()
    r, s = solve_lp(c)
    curr_s = s
    step = 0.02
    for _ in range(200):
        idx = rng.integers(N)
        best_nc = None
        best_ns = curr_s
        for _ in range(8):
            d = rng.uniform(-step, step, 2)
            nc = c.copy()
            nc[idx] = np.clip(nc[idx] + d, 1e-5, 1.0-1e-5)
            _, ns = solve_lp(nc)
            if ns > best_ns + 1e-7:
                best_ns = ns
                best_nc = nc
        if best_nc is not None:
            c = best_nc
            r, curr_s = solve_lp(c)
            step *= 0.85
        else:
            step = max(1e-4, step * 0.5)
            
    if curr_s > best_sum:
        best_sum = curr_s
        best_centers = c
        best_radii = r

    # Phase 3: Rotation Search for optimal boundary alignment
    for rot_deg in np.linspace(-15, 15, 31):
        rot_rad = np.radians(rot_deg)
        rot_mat = np.array([[np.cos(rot_rad), -np.sin(rot_rad)], [np.sin(rot_rad), np.cos(rot_rad)]])
        c_rot = (best_centers - 0.5) @ rot_mat + 0.5
        c_rot = np.clip(c_rot, 1e-5, 1.0-1e-5)
        _, s_rot = solve_lp(c_rot)
        if s_rot > best_sum:
            best_sum = s_rot
            _, best_radii = solve_lp(c_rot)
            best_centers = c_rot

    # Phase 4: SLSQP Joint Polish
    p0 = to_params(best_centers, best_radii * 0.995)
    bounds_slqp = [(1e-5, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons = {'type': 'ineq', 'fun': cons_slqp}
    
    try:
        res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons,
                       options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
            r_opt = res.x[:N]
            u_opt = res.x[N:2*N]
            v_opt = res.x[2*N:3*N]
            x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
            y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
            final_c = np.column_stack((x_opt, y_opt))
            final_r = r_opt
            final_s = np.sum(final_r)
            if final_s > best_sum:
                best_centers = final_c
                best_radii = final_r
                best_sum = final_s
    except Exception:
        pass

    return best_centers, np.maximum(best_radii, 0.0), float(best_sum)
