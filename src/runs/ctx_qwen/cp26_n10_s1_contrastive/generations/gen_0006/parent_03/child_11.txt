

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we need a robust global search combined with precise local refinement. The problem is highly non-convex due to pairwise distance constraints. Previous approaches using SLSQP directly on `(x, y, r)` or parameterized `(r, u, v)` coordinates reach ~2.636 but can get trapped in local minima.

Strategy improvements:
1. **LP-Driven Evaluation**: For any fixed set of centers, the optimal radii maximizing the sum can be found exactly via Linear Programming. This gives a precise, non-smooth objective landscape that is ideal for stochastic search.
2. **Simulated Annealing on Centers**: We run simulated annealing that moves circle centers randomly and accepts moves based on the LP-computed sum of radii. This topology exploration efficiently navigates the rugged landscape.
3. **Diverse Initializations**: Generate starts from hexagonal lattices (varied rotations/scales) and force-directed repulsion layouts to cover different packing topologies.
4. **SLSQP Polishing**: Once a high-quality configuration is found via SA, we polish it using SLSQP on `(x, y, r)` variables with explicit inequality constraints. This handles the continuous gradient information to squeeze out the final precision.
5. **Efficiency & Safety**: Precompute triangular indices, use `highs` solver for speed, adaptive step/temperature cooling, and multiple independent runs to ensure we escape local optima. Fallback configurations guarantee validity.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def get_max_sum_radii(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constr = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constr, n))
    b_ub = np.zeros(num_constr)
    k = 0
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for val in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = val
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def constraints_slsqp(v):
    """Inequality constraints for SLSQP: boundary containment and non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def objective_slsqp(v):
    return -np.sum(v[2::3])

def to_slsqp_params(centers, radii):
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = radii
    return v

def hex_init(rng, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_e = 0.095
    y = r_e
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_e
        x = r_e + shift
        while x <= 1.0 - r_e and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_e
        y += np.sqrt(3.0) * r_e
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.012, 0.012, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Spreads points using force-directed repulsion."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(250):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-5)
        rep = 1.0 / (dists**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 10.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 10.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.004 * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    inits = []
    for _ in range(12):
        inits.append(hex_init(rng, rng.uniform(-0.4, 0.4), rng.uniform(0.85, 1.15)))
    for _ in range(5):
        inits.append(force_init(rng))
        
    bounds_slsqp = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_slsqp = {'type': 'ineq', 'fun': constraints_slsqp}
    
    for init_c in inits:
        centers = init_c.copy()
        curr_radii, curr_sum = get_max_sum_radii(centers)
        local_best_sum = curr_sum
        local_best_c = centers.copy()
        local_best_r = curr_radii.copy()
        
        temp = 0.025
        step = 0.025
        for it in range(1500):
            idx = rng.integers(N)
            nc = centers.copy()
            nc[idx] += rng.normal(0, step, 2)
            nc[idx] = np.clip(nc[idx], 1e-4, 1.0 - 1e-4)
            
            nr, ns = get_max_sum_radii(nc)
            delta = ns - curr_sum
            
            if delta > 0 or rng.random() < np.exp(delta / max(temp, 1e-6)):
                centers = nc
                curr_sum = ns
                curr_radii = nr
                if curr_sum > local_best_sum:
                    local_best_sum = curr_sum
                    local_best_c = centers.copy()
                    local_best_r = curr_radii.copy()
            
            temp *= 0.9992
            if it % 250 == 0:
                step *= 0.94
                step = max(0.0015, step)
                
        if local_best_sum > best_sum:
            best_sum = local_best_sum
            best_centers = local_best_c
            best_radii = local_best_r
            
        # SLSQP polish
        v0 = to_slsqp_params(best_centers, best_radii * 0.995)
        v0 = np.clip(v0, [0.01, 0.01, 1e-6] * N, [0.99, 0.99, 0.49] * N)
        try:
            res = minimize(objective_slsqp, v0, method='SLSQP', bounds=bounds_slsqp,
                           constraints=cons_slsqp, options={'maxiter': 2500, 'ftol': 1e-13})
            if res.success and np.min(constraints_slsqp(res.x)) >= -1e-8:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3]
        except Exception:
            pass
            
    # Final high-precision polish
    if best_centers is not None:
        # Slight random perturbation to help escape tight SLSQP basins
        best_centers_p = best_centers + rng.normal(0, 1e-4, (N, 2))
        best_centers_p = np.clip(best_centers_p, 0.01, 0.99)
        v0 = to_slsqp_params(best_centers_p, best_radii * 0.999)
        v0 = np.clip(v0, [0.01, 0.01, 1e-6] * N, [0.99, 0.99, 0.49] * N)
        try:
            res_f = minimize(objective_slsqp, v0, method='SLSQP', bounds=bounds_slsqp,
                             constraints=cons_slsqp, options={'maxiter': 4000, 'ftol': 1e-14})
            if res_f.success and np.min(constraints_slsqp(res_f.x)) >= -1e-9:
                best_centers = np.column_stack((res_f.x[0::3], res_f.x[1::3]))
                best_radii = res_f.x[2::3]
                best_sum = -res_f.fun
        except Exception:
            pass
            
    if best_centers is None:
        best_centers = hex_init(rng, 0.0, 1.0)
        best_radii, best_sum = get_max_sum_radii(best_centers)
        
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(best_sum)
```
