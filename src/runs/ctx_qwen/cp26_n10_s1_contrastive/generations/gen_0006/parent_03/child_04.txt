import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find optimal feasible radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-5

def obj_centers(c_flat):
    """Objective for center optimization: minimize negative sum of LP-optimal radii."""
    centers = np.clip(c_flat.reshape(N, 2), 1e-5, 1.0 - 1e-5)
    _, s = solve_radii_lp(centers)
    return -s

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def params_to_cv(p):
    """Reconstruct centers and radii from (r, u, v) parameters."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraint_func(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    c, r = params_to_cv(p)
    dx = c[:, 0, np.newaxis] - c[np.newaxis, :, 0]
    dy = c[:, 1, np.newaxis] - c[np.newaxis, :, 1]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def objective_p(p):
    """Minimize negative sum of radii."""
    return -np.sum(p[:N])

def gen_hex_centers(rng, rot, scale):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_e = 0.095
    y = r_e
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_e
        x = r_e + shift
        while x <= 1.0 - r_e and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_e
        y += np.sqrt(3.0) * r_e
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.05, 0.95)

def gen_force_centers(rng):
    """Spread points using force-directed layout."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(200):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-5)
        rep = 1.0 / (dists**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        for d in range(2):
            forces[:, d] += 5.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 5.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.005 * forces
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    bounds_p = [(1e-5, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_p = {'type': 'ineq', 'fun': constraint_func}
    
    best_p = None
    best_s = -np.inf
    best_c = None
    best_r = None
    
    # Generate diverse initial center configurations
    starts_c = []
    for _ in range(20):
        starts_c.append(gen_hex_centers(rng, rng.uniform(-0.3, 0.3), rng.uniform(0.85, 1.15)))
    for _ in range(15):
        starts_c.append(gen_force_centers(rng))
    for _ in range(10):
        starts_c.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # Phase 1: Optimize centers via Nelder-Mead + exact LP radii evaluation
    for sc in starts_c:
        try:
            res_c = minimize(obj_centers, sc.flatten(), method='Nelder-Mead',
                             options={'maxiter': 2500, 'xatol': 1e-7, 'fatol': 1e-8})
            c_opt = np.clip(res_c.x.reshape(N, 2), 1e-5, 1.0 - 1e-5)
            r_opt, s_opt = solve_radii_lp(c_opt)
            if s_opt > best_s:
                best_s = s_opt
                best_c = c_opt
                best_r = r_opt
        except Exception:
            continue
            
    if best_c is not None:
        # Phase 2: Joint SLSQP refinement on (r, u, v) parameters
        p0 = to_params(best_c, best_r)
        try:
            res_p = minimize(objective_p, p0, method='SLSQP', bounds=bounds_p,
                             constraints=cons_p, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res_p.success and np.min(constraint_func(res_p.x)) >= -1e-7:
                s_p = -res_p.fun
                if s_p > best_s:
                    best_s = s_p
                    best_p = res_p.x.copy()
        except Exception:
            pass
            
        # Phase 3: Perturbation + SLSQP to escape local minima
        if best_p is not None:
            for _ in range(40):
                p_pert = best_p + rng.normal(0, 0.005, 3 * N)
                p_pert[:N] = np.clip(p_pert[:N], 1e-5, 0.49)
                p_pert[N:] = np.clip(p_pert[N:], 0.0, 1.0)
                try:
                    res_pp = minimize(objective_p, p_pert, method='SLSQP', bounds=bounds_p,
                                      constraints=cons_p, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                    if res_pp.success and np.min(constraint_func(res_pp.x)) >= -1e-7:
                        s_pp = -res_pp.fun
                        if s_pp > best_s:
                            best_s = s_pp
                            best_p = res_pp.x.copy()
                except Exception:
                    continue
                    
        # Phase 4: Coordinate ascent on centers with LP re-evaluation
        if best_p is not None:
            c_curr, r_curr = params_to_cv(best_p)
            s_curr = best_s
            step = 0.02
            for _ in range(500):
                idx = rng.integers(N)
                direction = rng.standard_normal(2)
                direction /= np.linalg.norm(direction)
                nc = c_curr.copy()
                nc[idx] = np.clip(nc[idx] + step * direction, 0.01, 0.99)
                _, ns = solve_radii_lp(nc)
                if ns > s_curr + 1e-6:
                    c_curr = nc
                    s_curr = ns
                    step = max(0.001, step * 0.9)
                else:
                    step = min(0.05, step * 1.02)
            if s_curr > best_s:
                best_s = s_curr
                r_lp, _ = solve_radii_lp(c_curr)
                best_p = to_params(c_curr, r_lp)
                
        # Phase 5: High-precision final polish
        if best_p is not None:
            p_final = best_p.copy()
            p_final[:N] *= 0.99995  # Slight shrink for strict interior feasibility
            try:
                res_f = minimize(objective_p, p_final, method='SLSQP', bounds=bounds_p,
                                 constraints=cons_p, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if res_f.success and np.min(constraint_func(res_f.x)) >= -1e-8:
                    best_p = res_f.x
                    best_s = -res_f.fun
            except Exception:
                pass
                
    # Fallback configuration (should rarely be reached)
    if best_p is None:
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        rad = np.full(N, 0.09)
        best_p = to_params(pts, rad)
        best_s = np.sum(rad)
        
    # Reconstruct final centers and radii
    r_opt = best_p[:N]
    u_opt = best_p[N:2*N]
    v_opt = best_p[2*N:3*N]
    centers = np.column_stack((r_opt + u_opt * (1.0 - 2.0 * r_opt), r_opt + v_opt * (1.0 - 2.0 * r_opt)))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_s)
