import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find optimal feasible radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def objective_slsqp(vars):
    """Objective function for SLSQP: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints_slsqp(vars):
    """Inequality constraints for SLSQP: boundaries and pairwise non-overlap."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    c = []
    # Boundary containment: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx * dx + dy * dy
    rs = r[:, None] + r[None, :]
    c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX] ** 2)
    return np.concatenate(c)

def make_config(seed, mode):
    """Generates an initial center configuration based on mode."""
    rng = np.random.RandomState(seed)
    if mode == 0:  # Hexagonal lattice
        pts = []
        re = 0.095
        y = re
        row = 0
        while len(pts) < N:
            shift = (row % 2) * re
            x = re + shift
            while x <= 1.0 - re and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * re
            y += np.sqrt(3.0) * re
            row += 1
        pts = np.array(pts[:N])
        rot = rng.uniform(-0.3, 0.3)
        sc = rng.uniform(0.85, 1.15)
        pts = (pts - 0.5) * sc + 0.5
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        pts += rng.uniform(-0.02, 0.02, pts.shape)
        return np.clip(pts, 0.02, 0.98)
    elif mode == 1:  # Force-directed layout
        pts = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            dists = np.maximum(dists, 1e-5)
            rep = 1.0 / (dists ** 2)
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 10.0 * np.maximum(0, 0.05 - pts[:, d])
                forces[:, d] -= 10.0 * np.maximum(0, pts[:, d] - 0.95)
            pts += 0.005 * forces
            pts = np.clip(pts, 0.02, 0.98)
        return pts
    else:  # Perturbed grid
        pts = np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng.uniform(-0.03, 0.03, pts.shape)
        return np.clip(pts, 0.02, 0.98)

def local_search(centers, rng):
    """Hybrid Simulated Annealing + Coordinate Ascent on centers, optimizing radii via LP."""
    best_c = centers.copy()
    rad, curr_s = solve_lp(best_c)
    best_s = curr_s
    
    temp = 0.04
    step = 0.05
    
    # Phase 1: Simulated Annealing
    for it in range(1500):
        idx = rng.integers(N)
        old_c = best_c[idx].copy()
        direction = rng.standard_normal(2)
        direction /= np.linalg.norm(direction)
        new_c = best_c[idx] + step * direction
        new_c = np.clip(new_c, 0.02, 0.98)
        best_c[idx] = new_c
        
        _, new_s = solve_lp(best_c)
        delta = new_s - curr_s
        
        if delta > 0 or (temp > 1e-6 and rng.rand() < np.exp(delta / temp)):
            curr_s = new_s
            if curr_s > best_s:
                best_s = curr_s
        else:
            best_c[idx] = old_c
            
        temp *= 0.997
        step = max(0.002, step * 0.9985)
        
    # Phase 2: Deterministic Coordinate Ascent refinement
    step_ca = 0.02
    for _ in range(500):
        improved = False
        for i in range(N):
            old_c = best_c[i].copy()
            best_s_curr = best_s
            # Try axis-aligned moves
            dirs = np.array([[1, 0], [-1, 0], [0, 1], [0, -1]])
            for d in range(4):
                nc = best_c[i] + step_ca * dirs[d]
                nc = np.clip(nc, 0.02, 0.98)
                best_c[i] = nc
                _, ns = solve_lp(best_c)
                if ns > best_s_curr:
                    best_s_curr = ns
                else:
                    best_c[i] = old_c
            # Try random perturbation
            rp = best_c[i] + rng.normal(0, step_ca * 0.5, 2)
            rp = np.clip(rp, 0.02, 0.98)
            best_c[i] = rp
            _, ns = solve_lp(best_c)
            if ns > best_s_curr:
                best_s_curr = ns
            else:
                best_c[i] = old_c
                
            if best_s_curr > best_s:
                best_s = best_s_curr
                improved = True
        if not improved:
            step_ca *= 0.9
            if step_ca < 1e-4:
                break
                
    _, final_rad = solve_lp(best_c)
    return best_c, final_rad, best_s

def run_packing():
    rng = np.random.default_rng(42)
    best_c = None
    best_r = None
    best_s = -np.inf
    
    # Phase 1: Diverse starts with hybrid local search

                best_s = s_loc
                best_c = c_loc.copy()
                best_r = r_loc.copy()
                
    # Phase 2: High-precision SLSQP polish on the best configuration
    if best_c is not None:
        v0 = np.zeros(3 * N)
        v0[0::3] = best_c[:, 0]
        v0[1::3] = best_c[:, 1]
        v0[2::3] = best_r * 0.995  # Slight shrink for strict interior feasibility
        
        bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(1e-6, 0.5)] * N
        cons = {'type': 'ineq', 'fun': constraints_slsqp}
        
        try:
            res = minimize(objective_slsqp, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints_slsqp(res.x)) >= -1e-8:
                best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                best_r = res.x[2::3]
                best_s = -res.fun
        except Exception:
            pass
            
    # Safe fallback configuration
    if best_c is None:
        pts = np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        best_c = pts
        best_r, best_s = solve_lp(pts)
        
    best_r = np.maximum(best_r, 0.0)
    return best_c, best_r, float(best_s)
