import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def eval_centers(centers):
    """Given fixed centers, solves LP to find optimal feasible radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = np.zeros((4*n + n*(n-1)//2, n))
    b_ub = np.zeros(4*n + n*(n-1)//2)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k+=1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0-x; k+=1
        A_ub[k, i] = 1.0; b_ub[k] = y; k+=1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0-y; k+=1
        
    # Pairwise constraints: r_i + r_j <= dist_ij
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx*dx + dy*dy)
    np.fill_diagonal(dists, np.inf)
    
    ii, jj = np.triu_indices(n, k=1)
    A_ub[k:, ii] = 1.0
    A_ub[k:, jj] = 1.0
    b_ub[k:] = dists[ii, jj]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def neg_sum_radii(centers_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 0.005, 0.995)
    s, _ = eval_centers(centers)
    return -s

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, w])

def from_params(p):
    """Reconstruct centers and radii from (r, u, w) parameters."""
    r = p[:N]
    u = p[N:2*N]
    w = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraints(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    w = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx*dx + dy*dy
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def objective(p):
    """Minimize negative sum of radii."""
    return -np.sum(p[:N])

def gen_hex(rng, row_counts, rot, scale, jitter):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_e = 0.09
    y = r_e
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_e
        x = r_e + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_e
        y += np.sqrt(3.0) * r_e
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def gen_force(rng):
    """Spread points using force-directed layout."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(150):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-5
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 10.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 10.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.005 * forces
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5], 
                [7,6,6,7], [8,6,6,6], [5,7,5,7], [6,6,6,6,2], [7,5,6,6,2]]
    starts = []
    for pat in patterns:
        for _ in range(4):
            starts.append(gen_hex(rng, pat, rng.uniform(-0.2, 0.2), rng.uniform(0.85, 1.15), 0.015))
    for _ in range(15):
        starts.append(gen_force(rng))
        
    # Phase 1: Optimize centers with Nelder-Mead (handles non-smooth LP objective)
    for init_c in starts:
        try:
            res = minimize(neg_sum_radii, init_c.flatten(), method='Nelder-Mead',
                           options={'maxiter': 800, 'xatol': 1e-6, 'fatol': 1e-7})
            c_opt = res.x.reshape(N, 2)
            s, r = eval_centers(c_opt)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = r.copy()
        except Exception:
            continue
            
    # Phase 2: SLSQP joint optimization on (r, u, w)
    if best_centers is not None:
        bounds = [(1e-5, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
        cons = {'type': 'ineq', 'fun': constraints}
        
        p0 = to_params(best_centers, best_radii)
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-13})
            if res.success and np.min(constraints(res.x)) >= -1e-7:
                c_pol, r_pol = from_params(res.x)
                s_pol = np.sum(r_pol)
                if s_pol > best_sum:
                    best_sum = s_pol
                    best_centers = c_pol
                    best_radii = r_pol
        except Exception:
            pass
            
        # Perturbation search on centers to escape local minima
        for _ in range(25):
            p_pert = best_centers.copy()
            p_pert += rng.normal(0, 0.012, p_pert.shape)
            p_pert = np.clip(p_pert, 0.02, 0.98)
            try:
                res = minimize(neg_sum_radii, p_pert.flatten(), method='Nelder-Mead',
                               options={'maxiter': 500, 'xatol': 1e-6, 'fatol': 1e-7})
                c_tmp = res.x.reshape(N, 2)
                s_tmp, r_tmp = eval_centers(c_tmp)
                if s_tmp > best_sum:
                    best_sum = s_tmp
                    best_centers = c_tmp
                    best_radii = r_tmp
            except Exception:
                continue
                
        # Final high-precision SLSQP polish
        p0_f = to_params(best_centers, best_radii)
        try:
            res_f = minimize(objective, p0_f, method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 8000, 'ftol': 1e-14})
            if res_f.success and np.min(constraints(res_f.x)) >= -1e-7:
                c_f, r_f = from_params(res_f.x)
                s_f = np.sum(r_f)
                if s_f > best_sum:
                    best_sum = s_f
                    best_centers = c_f
                    best_radii = r_f
        except Exception:
            pass
            
    # Fallback configuration
    if best_centers is None:
        grid = np.array([[0.1+0.2*i, 0.1+0.2*j] for i in range(5) for j in range(5)])
        grid = np.vstack([grid, [0.5, 0.5]])
        s, r = eval_centers(grid)
        best_centers, best_radii, best_sum = grid, r, s
        
    # Ensure strict non-negativity
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_sum)
