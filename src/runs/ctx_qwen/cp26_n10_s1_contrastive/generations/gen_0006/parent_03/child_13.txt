from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback to tiny radii if LP fails
    return np.full(n, 1e-5), 1e-5 * n

def make_grid_init(rng, seed):
    """Generates a perturbed grid initialization."""
    rng_g = np.random.RandomState(seed)
    pts = np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng_g.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def obj_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[:N])

def cons_slqp(v):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = v[:N]
    u = v[N:2*N]
    w = v[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(c, r):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((c[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((c[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, w])

def from_params(v):
    """Reconstruct physical centers and radii from optimization parameters."""
    r = v[:N]
    u = v[N:2*N]
    w = v[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r
