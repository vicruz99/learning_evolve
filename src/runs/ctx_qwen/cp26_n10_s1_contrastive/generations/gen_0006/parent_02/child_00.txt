import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def obj_func(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[0::3])

def con_func(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[0::3]
    u = p[1::3]
    v = p[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers. Returns (radii, sum_radii)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = np.zeros((4*n + n*(n-1)//2, n))
    b_ub = np.zeros(4*n + n*(n-1)//2)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i+1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)]*n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def params_to_c(p):
    """Convert (r, u, v) parameters to physical centers."""
    r = p[0::3]
    u = p[1::3]
    v = p[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y))

def c_to_params(c, r):
    """Convert physical centers and radii to (r, u, v) parameters."""
    r_c = r.copy()
    denom = np.clip(1.0 - 2.0 * r_c, 1e-6, 1.0)
    u = np.clip((c[:, 0] - r_c) / denom, 0.0, 1.0)
    v = np.clip((c[:, 1] - r_c) / denom, 0.0, 1.0)
    p = np.empty(3*N)
    p[0::3] = r_c
    p[1::3] = u
    p[2::3] = v
    return p

def gen_hex(rng, rot, scale):
    """Generates a hexagonal lattice initialization with specified parameters."""
    pts = []
    r_e = 0.095
    y = r_e
    row = 0
    rows_counts = [6, 5, 6, 5, 4]
    for cnt in rows_counts:
        shift = (row % 2) * r_e
        x = r_e + shift
        for _ in range(cnt):
            if len(pts) >= N: break
            pts.append([x, y])
            x += 2.0 * r_e
        y += np.sqrt(3.0) * r_e
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        R = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ R.T + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def gen_force(rng, seed):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.005 * (0.99**step) * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    best_p = None
    best_s = -np.inf
    bounds_slqp = [(1e-5, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons_slqp = {'type': 'ineq', 'fun': con_func}

    # Phase 1: Diverse SLSQP starts
    starts = []
    for i in range(40):
        starts.append(gen_hex(rng, rng.uniform(-0.2, 0.2), rng.uniform(0.85, 1.15)))
    for i in range(20):
        starts.append(gen_force(rng, i))

    for c_init in starts:
        r_init, _ = solve_lp(c_init)
        r_init = np.maximum(r_init * 0.995, 1e-5)
        p0 = c_to_params(c_init, r_init)
        try:
            res = minimize(obj_func, p0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(con_func(res.x)) >= -1e-8:
                s = -res.fun
                if s > best_s:
                    best_s = s
                    best_p = res.x.copy()
        except Exception:
            pass

    # Phase 2: LP-based Local Search on Centers (Simulated Annealing)
    if best_p is not None:
        c_best = params_to_c(best_p)
        r_best, s_lp = solve_lp(c_best)
        if s_lp > best_s:
            best_s = s_lp
            best_p = c_to_params(c_best, r_best)
            c_best = params_to_c(best_p)

        rng_lp = np.random.RandomState(123)
        step_size = 0.025
        temp = 0.01
        for _ in range(8000):
            idx = rng_lp.integers(N)
            old_c = c_best[idx].copy()
            c_best[idx] += rng_lp.normal(0, step_size, 2)
            c_best[idx] = np.clip(c_best[idx], 0.005, 0.995)

            r_new, s_new = solve_lp(c_best)
            delta = s_new - s_lp
            if delta > 0 or (temp > 1e-5 and rng_lp.rand() < np.exp(delta / temp)):
                s_lp = s_new
                r_best = r_new
                if s_lp > best_s:
                    best_s = s_lp
                    best_p = c_to_params(c_best, r_best)
            else:
                c_best[idx] = old_c
            temp *= 0.999
            step_size = max(0.001, step_size * 0.9995)

        c_best = params_to_c(best_p)

    # Phase 3: Perturbation & SLSQP Polish to escape local minima
    if best_p is not None:
        for k in range(30):
            p_pert = best_p + rng.normal(0, 0.003, 3*N)
            p_pert[0::3] = np.clip(p_pert[0::3], 1e-5, 0.49)
            p_pert[1::3] = np.clip(p_pert[1::3], 0.0, 1.0)
            p_pert[2::3] = np.clip(p_pert[2::3], 0.0, 1.0)
            try:
                res = minimize(obj_func, p_pert, method='SLSQP', bounds=bounds_slqp,
                               constraints=cons_slqp, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(con_func(res.x)) >= -1e-8:
                    s = -res.fun
                    if s > best_s:
                        best_s = s
                        best_p = res.x.copy()
            except Exception:
                pass

    # Phase 4: Final High-Precision Polish
    if best_p is not None:
        p_polish = best_p.copy()
        p_polish[0::3] *= 0.9999
        try:
            res_f = minimize(obj_func, p_polish, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(con_func(res_f.x)) >= -1e-9:
                best_p = res_f.x
                best_s = -res_f.fun
        except Exception:
            pass

    # Fallback configuration (should not be reached)
    if best_p is None:
        c_fallback = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                      np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        r_fallback, _ = solve_lp(c_fallback)
        best_p = c_to_params(c_fallback, r_fallback)
        best_s = np.sum(r_fallback)

    c_out = params_to_c(best_p)
    r_out = best_p[0::3]
    return c_out, r_out, float(best_s)
