import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers. Returns (radii, sum_radii)."""
    n = centers.shape[0]
    num_pair = n * (n - 1) // 2
    num_bound = 4 * n
    A_ub = np.zeros((num_bound + num_pair, n))
    b_ub = np.zeros(num_bound + num_pair)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 2.6e-4

def constraints_slqp(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    i_idx, j_idx = np.triu_indices(N, k=1)
    c.append(dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    return np.concatenate(c)

def obj_slqp(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def generate_hex(seed, rows):
    """Generates a hexagonal lattice initialization with specified row counts."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(rows):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * rng.uniform(0.90, 1.10) + 0.5
    if rng.rand() > 0.5:
        rot = rng.uniform(-0.2, 0.2)
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.normal(0, 0.015, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def run_sa(initial_centers, n_iter, rng):
    """Simulated Annealing on center positions using LP for radius evaluation."""
    centers = initial_centers.copy()
    radii, best_sum = solve_lp(centers)
    best_centers = centers.copy()
    best_radii = radii.copy()
    curr_sum = best_sum
    temp = 0.04
    step = 0.025
    
    for _ in range(n_iter):
        op = rng.choice(['move', 'swap'])
        if op == 'move':
            idx = rng.randint(N)
            old = centers[idx].copy()
            centers[idx] += rng.normal(0, step, 2)
            centers[idx] = np.clip(centers[idx], 0.005, 0.995)
        else:
            i, j = rng.choice(N, 2, replace=False)
            centers[i], centers[j] = centers[j].copy(), centers[i].copy()
            
        r_new, sum_new = solve_lp(centers)
        delta = sum_new - curr_sum
        
        if delta > 0 or (rng.rand() < np.exp(delta / temp)):
            curr_sum = sum_new
            if sum_new > best_sum:
                best_sum = sum_new
                best_centers = centers.copy()
                best_radii = r_new.copy()
        else:
            if op == 'move':
                centers[idx] = old
                
        temp *= 0.997
        step = max(0.0005, step * 0.9985)
        
    return best_centers, best_radii, best_sum
