We solve this LP efficiently using `scipy.optimize.linprog` with the `highs` solver. We then use Simulated Annealing to perturb circle centers, accepting moves that increase the LP-optimal sum of radii. This effectively navigates the non-differentiable, piecewise-linear landscape of circle packings.

2. **Diverse Initializations**: We generate starting configurations using:

3. **SLSQP Joint Polish**: After SA finds a strong configuration, we refine it using `scipy.optimize.minimize` with the SLSQP method on joint `(x, y, r)` variables. import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers. Returns (radii, sum_radii)."""
    n = centers.shape[0]
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    num_cons = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_cons, n))
    b_ub = np.zeros(num_cons)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_vals = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_vals:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 2.6e-4

def make_hex_centers(rng, row_counts, rot, scale, jitter):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.1
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = pts @ rot_mat.T
        pts = pts - pts.mean(axis=0) + 0.5
        
    pts += rng.normal(0, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force_centers(rng, steps=300):
    """Generates organic packing using force-directed layout."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    
    for s in range(steps):
        forces = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        rep_mag = 1.0 / (dists**2)
        np.fill_diagonal(rep_mag, 0.0)
        forces += np.sum(rep_mag[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
            
        pts += 0.005 * forces * (1.0 - s / float(steps))
        pts = np.clip(pts, 0.01, 0.99)
        
    return pts

def run_sa(centers, rng, n_iter=4000, temp0=0.015, step0=0.025):
    """Simulated Annealing on centers optimizing LP sum."""
    curr_centers = centers.copy()
    _, curr_score = solve_lp(curr_centers)
    best_centers = curr_centers.copy()
    best_radii = np.zeros(N)
    best_score = curr_score
    
    temp = temp0
    step = step0
    
    for _ in range(n_iter):
        idx = rng.integers(N)
        old = curr_centers[idx].copy()
        curr_centers[idx] += rng.normal(0, step, 2)
        curr_centers[idx] = np.clip(curr_centers[idx], 0.01, 0.99)
        
        r_new, score_new = solve_lp(curr_centers)
        delta = score_new - curr_score
        
        if delta > 0 or (temp > 1e-9 and rng.rand() < np.exp(delta / temp)):
            curr_score = score_new
            if score_new > best_score:
                best_score = score_new
                best_centers = curr_centers.copy()
                best_radii = r_new.copy()
        else:
            curr_centers[idx] = old
            
        temp *= 0.9990
        step = max(0.0005, step * 0.9992)
        
    return best_centers, best_radii, best_score

def objective_slqp(v):
    return -np.sum(v[2::3])

def constraints_slqp(v):
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    c.append(dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    
    return np.concatenate(c)

def run_packing():
    rng = np.random.default_rng(42)
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 6, 6, 4],
        [7, 5, 6, 5, 3], [6, 6, 6, 4, 4], [5, 5, 6, 5, 5],
        [6, 4, 6, 5, 5], [4, 5, 6, 5, 6]
    ]
    
    # Phase 1: Hexagonal starts + SA
    for pat in row_patterns:
        for _ in range(3):
            rot = rng.uniform(-0.2, 0.2)
            scale = rng.uniform(0.90, 1.10)
            jit = rng.uniform(0.005, 0.02)
            init_c = make_hex_centers(rng, pat, rot, scale, jit)
            c_opt, r_opt, s_opt = run_sa(init_c, rng, n_iter=3000, temp0=0.012, step0=0.02)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt
                best_r = r_opt
                
    # Phase 2: Force-directed starts + SA
    for _ in range(4):
        init_c = make_force_centers(rng, steps=250)
        c_opt, r_opt, s_opt = run_sa(init_c, rng, n_iter=3000, temp0=0.01, step0=0.015)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt
            best_r = r_opt
            
    # Phase 3: SLSQP Polish
    if best_c is not None:
        v0 = np.zeros(3 * N)
        v0[0::3] = best_c[:, 0]
        v0[1::3] = best_c[:, 1]
        v0[2::3] = np.maximum(best_r * 0.998, 1e-6)
        
        bounds_slqp = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
        cons_slqp = {'type': 'ineq', 'fun': constraints_slqp}
        
        try:
            res = minimize(objective_slqp, v0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                if np.min(constraints_slqp(res.x)) >= -1e-8:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_r = res.x[2::3]
        except Exception:
            pass
            
        # Additional perturbation & SLSQP to escape shallow local minima
        for _ in range(10):
            v_p = v0.copy()
            v_p += rng.normal(0, 0.003, 3 * N)
            v_p[0::3] = np.clip(v_p[0::3], 0.01, 0.99)
            v_p[1::3] = np.clip(v_p[1::3], 0.01, 0.99)
            v_p[2::3] = np.clip(v_p[2::3], 1e-6, 0.5)
            
            try:
                res_p = minimize(objective_slqp, v_p, method='SLSQP', bounds=bounds_slqp,
                                 constraints=cons_slqp, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                if res_p.success and np.min(constraints_slqp(res_p.x)) >= -1e-8:
                    s_p = -res_p.fun
                    if s_p > best_sum:
                        best_sum = s_p
                        best_c = np.column_stack((res_p.x[0::3], res_p.x[1::3]))
                        best_r = res_p.x[2::3]
            except Exception:
                pass

    if best_c is None:
        best_c = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                  np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        best_r = np.full(N, 0.04)
        best_sum = np.sum(best_r)
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
