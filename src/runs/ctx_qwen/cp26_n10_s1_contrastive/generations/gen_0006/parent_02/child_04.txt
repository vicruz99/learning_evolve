import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers. Returns (radii, sum_radii)."""
    n = centers.shape[0]
    num_rows = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_rows, n))
    b_ub = np.zeros(num_rows)
    k = 0
    
    # Precompute pairwise distances efficiently
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        x, y = centers[i]
        vals = [x, 1.0 - x, y, 1.0 - y]
        for v in vals:
            A_ub[k, i] = 1.0
            b_ub[k] = v
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x, -res.fun
    # Fallback to tiny radii if LP fails (should rarely happen for valid centers)
    return np.full(n, 1e-4), 2.6e-3

def objective_centers(c_flat):
    """Objective for center optimization: negative sum of optimal radii."""
    c = c_flat.reshape(N, 2)
    c = np.clip(c, 0.01, 0.99)
    _, s = solve_lp(c)
    return -s

def objective_slqp(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def constraint_slqp(v):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2
