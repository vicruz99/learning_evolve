import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers. Returns (radii, sum_radii)."""
    n = centers.shape[0]
    num_pair = n * (n - 1) // 2
    num_ineq = 4 * n + num_pair
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        bounds_vals = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_vals:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    A_ub[k:, I_IDX] = 1.0
    A_ub[k:, J_IDX] = 1.0
    b_ub[k:] = dists[I_IDX, J_IDX]
    
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4 * n

def objective_slqp(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def constraints_slqp(v):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_hex_centers(seed, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization with specified parameters."""
    rng = np.random.RandomState(seed)
    rows_counts = [6, 5, 6, 5, 4]
    pts = []
    r_est = 0.1
    y = r_est
    for r_idx, cnt in enumerate(rows_counts):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        rot_mat = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = pts @ rot_mat.T
        pts = pts - pts.mean(axis=0) + 0.5
        
    pts += rng.randn(N, 2) * 0.015
    return np.clip(pts, 0.02, 0.98)

def generate_force_centers(seed):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    for _ in range(200):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / (dists**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
            
        pts += 0.005 * forces
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def run_sa(centers, rng):
    """Simulated Annealing on centers using LP for evaluation."""
    r_curr, sum_curr = solve_lp(centers)
    best_c, best_r, best_sum = centers.copy(), r_curr.copy(), sum_curr
    
    temp = 0.01
    step = 0.025
    
    for i in range(12000):
        idx = rng.randint(N)
        old_pos = centers[idx].copy()
        centers[idx] += rng.normal(0, step, 2)
        centers[idx] = np.clip(centers[idx], 0.005, 0.995)
            
        r_new, sum_new = solve_lp(centers)
        delta = sum_new - sum_curr
        
        if delta > 0 or (temp > 1e-6 and rng.rand() < np.exp(delta / temp)):
            sum_curr = sum_new
            if sum_new > best_sum:
                best_c, best_r, best_sum = centers.copy(), r_new.copy(), sum_new
        else:
            centers[idx] = old_pos
                
        if i % 2500 == 0:
            temp *= 0.4
            step *= 0.65
            
    return best_c, best_r, best_sum

    cons_slqp = {'type': 'ineq', 'fun': constraints_slqp}
    
    best_v = None
    best_sum = -np.inf

    inits = []
