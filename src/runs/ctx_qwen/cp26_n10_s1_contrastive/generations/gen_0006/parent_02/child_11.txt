import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective_slsqp(v):
    return -np.sum(v[0::3])

def constraints_slsqp(v):
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx*dx + dy*dy
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def solve_lp(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)]*n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 1e-5

def to_params(centers, radii):
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, w])

def from_params(v):
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def gen_hex(seed, rot, scale):
    rng = np.random.RandomState(seed)
    counts = [6, 5, 6, 5, 4]
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.05, 0.95)

def gen_force(seed, steps=300):
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    r = np.full(N, 0.06)
    for _ in range(steps):
        forces = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces += np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.005 * forces
        pts = np.clip(pts, 0.01, 0.99)
        np.fill_diagonal(dists, np.inf)
        d_pair = np.min(dists, axis=1)
        d_wall = np.minimum(np.minimum(pts[:, 0], 1.0-pts[:, 0]), np.minimum(pts[:, 1], 1.0-pts[:, 1]))
        r = 0.9 * np.minimum(d_wall, d_pair / 2.0)
    return pts, r

def run_packing():
    np.random.seed(42)
    bounds_slsqp = [(1e-6, 0.49)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons_slsqp = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_v = None
    best_sum = -np.inf

    inits = []

        r_init = np.maximum(r_init * 0.99, 1e-5)
        inits.append(to_params(pts, r_init))
        
    for s in range(20):
        pts, r_init = gen_force(s)
        inits.append(to_params(pts, r_init))
        
    for x0 in inits:
        try:

                           constraints=cons_slsqp, options={'maxiter': 5000, 'ftol': 1e-13})
            if res.success:
                if np.min(constraints_slsqp(res.x)) >= -1e-8:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_v = res.x.copy()
        except:
            pass
            
    # Phase 2: LP-based Simulated Annealing
    if best_v is not None:
        centers, radii = from_params(best_v)
        r_lp, curr_sum = solve_lp(centers)
        best_lp_sum = curr_sum
        best_lp_c = centers.copy()
        best_lp_r = r_lp.copy()

        temp = 0.015
        step = 0.02
        
        for it in range(8000):

            old_centers = centers.copy()

                idx = rng.randint(N)
                centers[idx] += rng.normal(0, step, 2)
                centers[idx] = np.clip(centers[idx], 0.005, 0.995)
            elif op == 'swap':
                i, j = rng.choice(N, 2, replace=False)
                centers[i], centers[j] = centers[j].copy(), centers[i].copy()
            else:

            delta = sum_new - curr_sum
            
            if delta > 0 or (rng.rand() < np.exp(delta / max(temp, 1e-6))):
                curr_sum = sum_new
                if sum_new > best_lp_sum:
                    best_lp_sum = sum_new
                    best_lp_c = centers.copy()
                    best_lp_r = r_new.copy()
            else:
                centers = old_centers
                
            temp *= 0.9995
            step = max(0.002, step * 0.999)
            
        best_v = to_params(best_lp_c, best_lp_r)
        best_sum = best_lp_sum
        
    # Phase 3: Coordinate Ascent
    if best_v is not None:
        centers, radii = from_params(best_v)
        r_lp, curr_sum = solve_lp(centers)
        for cycle in range(5):
            improved = True
            while improved:
                improved = False
                for i in range(N):
                    for d in range(2):
                        for sign in [1, -1]:
                            step = 0.01 * (0.9**cycle)
                            old = centers[i, d]
                            centers[i, d] += sign * step
                            centers[i, d] = np.clip(centers[i, d], 0.01, 0.99)
                            r_new, sum_new = solve_lp(centers)
                            if sum_new > curr_sum + 1e-9:
                                curr_sum = sum_new
                                improved = True
                            else:
                                centers[i, d] = old
                            
    # Phase 4: Final SLSQP Polish
    if best_v is not None:
        v0 = to_params(centers, r_lp)
        try:
            res_f = minimize(objective_slsqp, v0, method='SLSQP', bounds=bounds_slsqp,
                             constraints=cons_slsqp, options={'maxiter': 4000, 'ftol': 1e-14})
            if res_f.success and np.min(constraints_slsqp(res_f.x)) >= -1e-9:
                s_f = -res_f.fun
                if s_f > best_sum:
                    best_v = res_f.x
                    best_sum = s_f
        except:
            pass
            
    # Fallback
    if best_v is None:
        best_v = to_params(np.random.rand(N, 2), np.full(N, 0.05))
        best_sum = np.sum(best_v[0::3])
        
    centers_final, radii_final = from_params(best_v)
    return centers_final, np.maximum(radii_final, 0.0), float(best_sum)
