import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers. Returns (radii, sum_radii)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    n_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        limits = [x, 1.0 - x, y, 1.0 - y]
        for lim in limits:
            A_ub[k, i] = 1.0
            b_ub[k] = lim
            k += 1
            
    # Precompute pairwise distances efficiently
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def get_params_from_centers_radii(centers, radii):
    """Convert physical centers and radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def get_centers_radii_from_params(params):
    """Convert (r, u, v) parameters back to physical centers and radii."""
    r = params[0::3].copy()
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraints_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def objective_func(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def make_hex_init(rng, row_counts, rot, scale, jitter):
    """Generates a hexagonal lattice initialization with specified parameters."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        mat = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ mat.T + 0.5
        
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force_init(rng):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.005 * forces
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraints_func}
    
    best_sum = -np.inf
    best_params = None
    
    # Phase 1: Diverse global search from multiple topologies
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5],
        [5,5,5,5,6], [6,4,6,5,5], [5,6,4,6,5],
        [4,5,6,5,6], [6,6,5,5,4], [5,5,6,5,5]
    ]
    
    inits = []
    for pat in patterns:
        for _ in range(4):
            rot = rng.uniform(-0.25, 0.25)
            scale = rng.uniform(0.85, 1.15)
            jitt = rng.uniform(0.005, 0.03)
            inits.append(make_hex_init(rng, pat, rot, scale, jitt))
            
    for _ in range(20):
        inits.append(make_force_init(rng))
        
    for pts in inits:
        r_init, _ = solve_radii_lp(pts)
        r_init = np.maximum(r_init * 0.995, 1e-5)
        p0 = get_params_from_centers_radii(pts, r_init)
        
        try:
            res = minimize(objective_func, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                if np.min(constraints_func(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Perturbation refinement to escape local minima
    if best_params is not None:
        for k in range(40):
            p0 = best_params.copy()
            p0[1::3] += rng.normal(0, 0.015, N) # Perturb u more aggressively
            p0[2::3] += rng.normal(0, 0.015, N) # Perturb v more aggressively
            p0[0::3] += rng.normal(0, 0.0005, N)
            
            p0[0::3] = np.clip(p0[0::3], 1e-6, 0.49)
            p0[1::3] = np.clip(p0[1::3], 0.0, 1.0)
            p0[2::3] = np.clip(p0[2::3], 0.0, 1.0)
            
            try:
                res = minimize(objective_func, p0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                if res.success:
                    if np.min(constraints_func(res.x)) >= -1e-7:
                        s_val = -res.fun
                        if s_val > best_sum:
                            best_sum = s_val
                            best_params = res.x.copy()
            except Exception:
                continue
                
        # Phase 3: LP-based Coordinate Ascent on Centers with Simulated Annealing
        centers, radii = get_centers_radii_from_params(best_params)
        centers, lp_r = solve_radii_lp(centers)
        curr_sum = np.sum(lp_r)
        
        step = 0.025
        temp = 0.03
        for _ in range(2500):
            idx = rng.integers(N)
            old_pos = centers[idx].copy()
            centers[idx] += rng.normal(0, step, 2)
            centers[idx] = np.clip(centers[idx], 0.01, 0.99)
            
            new_r, new_sum = solve_radii_lp(centers)
            delta = new_sum - curr_sum
            
            # Accept if improvement, or with SA probability
            if delta > 0 or (temp > 1e-6 and rng.rand() < np.exp(delta / temp)):
                curr_sum = new_sum
                radii = new_r
                temp *= 0.999
                step = max(0.001, step * 0.995)
            else:
                centers[idx] = old_pos
                step = min(0.04, step * 1.01)
                
        best_sum = curr_sum
        best_params = get_params_from_centers_radii(centers, radii)
        
        # Phase 4: High-precision final polish
        p0 = best_params.copy()
        p0[0::3] *= 0.9999 # Slight shrink for strict interior feasibility
        try:
            res_f = minimize(objective_func, p0, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(constraints_func(res_f.x)) >= -1e-8:
                s_f = -res_f.fun
                if s_f > best_sum:
                    best_sum = s_f
                    best_params = res_f.x
        except Exception:
            pass
            
    # Fallback configuration (should rarely be reached)
    if best_params is None:
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        r_f = np.full(N, 0.05)
        best_params = get_params_from_centers_radii(pts, r_f)
        best_sum = np.sum(r_f)
        
    centers, radii = get_centers_radii_from_params(best_params)
    return centers, np.maximum(radii, 0.0), float(best_sum)
