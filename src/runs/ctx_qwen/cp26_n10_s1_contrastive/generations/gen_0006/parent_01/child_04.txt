import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_fast(centers):
    """Given fixed centers, solves LP to find optimal radii maximizing sum(r_i)."""
    n = centers.shape[0]
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    num_cons = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_cons, n))
    b_ub = np.zeros(num_cons)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0; A_ub[k, j] = 1.0; b_ub[k] = dists[i, j]; k += 1
            
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 1e-5

def slqp_objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def slqp_constraints(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    c.append(dist2[i_idx, j_idx] - rs[i_idx, j_idx]**2)
    return np.concatenate(c)

def make_hex_init(rng, seed):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = r_est if row % 2 == 1 else 0.0
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    rng_local = np.random.RandomState(seed)
    pts += rng_local.uniform(-0.02, 0.02, pts.shape)
    rot = rng_local.uniform(-0.2, 0.2)
    c_val, s_val = np.cos(rot), np.sin(rot)
    R = np.array([[c_val, -s_val], [s_val, c_val]])
    pts = (pts - 0.5) @ R.T + 0.5
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_force_init(rng, seed):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    rng_local = np.random.RandomState(seed)
    pts = rng_local.rand(N, 2)
    for _ in range(300):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        rep = 1.0 / (dists**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.008 * forces
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def run_sa(centers_init, rng, n_iters=35000):
    """Runs Simulated Annealing on centers, using LP to evaluate objective."""
    centers = centers_init.copy()
    radii, curr_sum = solve_lp_fast(centers)
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = curr_sum
    
    step = 0.06
    temp = 0.012
    
    for _ in range(n_iters):
        idx = rng.integers(0, N)
        old_c = centers[idx].copy()
        delta = rng.normal(0, step, 2)
        new_c = np.clip(old_c + delta, 0.001, 0.999)
        
        centers[idx] = new_c
        new_radii, new_sum = solve_lp_fast(centers)
        
        if new_sum > best_sum:
            best_centers = centers.copy()
            best_radii = new_radii.copy()
            best_sum = new_sum
            
        delta_e = new_sum - curr_sum
        if delta_e > 0 or rng.random() < np.exp(delta_e / temp):
            curr_sum = new_sum
            step *= 0.9997
            step = max(0.0008, step)
        else:
            centers[idx] = old_c
            step = min(0.09, step * 1.0003)
            
        temp *= 0.99995
        temp = max(1e-6, temp)
        
    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_slqp = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_slqp = {'type': 'ineq', 'fun': slqp_constraints}
    
    best_overall_sum = -np.inf
    best_overall_c = None
    best_overall_r = None
    
    inits = []
    for s in range(8):
        inits.append(make_hex_init(rng, s))
    for s in range(8):
        inits.append(make_force_init(rng, s))
        
    for init_c in inits:
        # Phase 1: SA on centers guided by LP
        sa_c, sa_r, sa_sum = run_sa(init_c, rng, n_iters=35000)
        if sa_sum > best_overall_sum:
            best_overall_sum = sa_sum
            best_overall_c = sa_c
            best_overall_r = sa_r
            
        # Phase 2: SLSQP polish from SA result
        v0 = np.zeros(3 * N)
        v0[0::3] = sa_c[:, 0]
        v0[1::3] = sa_c[:, 1]
        v0[2::3] = np.maximum(sa_r * 0.995, 1e-6)
        
        try:
            res = minimize(slqp_objective, v0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-13})
            if res.success:
                c_val = slqp_constraints(res.x)
                if np.min(c_val) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_overall_sum:
                        best_overall_sum = curr_sum
                        best_overall_c = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_overall_r = res.x[2::3]
        except Exception:
            pass
            
        # Phase 2b: Perturbed SLSQP starts to escape shallow local minima
        for _ in range(4):
            v_pert = v0.copy()
            v_pert[0::3] += rng.normal(0, 0.004, N)
            v_pert[1::3] += rng.normal(0, 0.004, N)
            v_pert[0::3] = np.clip(v_pert[0::3], 0.01, 0.99)
            v_pert[1::3] = np.clip(v_pert[1::3], 0.01, 0.99)
            v_pert[2::3] *= 0.995
            
            try:
                res_p = minimize(slqp_objective, v_pert, method='SLSQP', bounds=bounds_slqp,
                                 constraints=cons_slqp, options={'maxiter': 3000, 'ftol': 1e-13})
                if res_p.success:
                    if np.min(slqp_constraints(res_p.x)) >= -1e-8:
                        s_p = -res_p.fun
                        if s_p > best_overall_sum:
                            best_overall_sum = s_p
                            best_overall_c = np.column_stack((res_p.x[0::3], res_p.x[1::3]))
                            best_overall_r = res_p.x[2::3]
            except Exception:
                pass
                
    # Fallback (should rarely be reached)
    if best_overall_c is None:
        best_overall_c = make_force_init(rng, 0)
        best_overall_r, best_overall_sum = solve_lp_fast(best_overall_c)
        
    # Phase 3: High-precision final polish
    v_final = np.zeros(3 * N)
    v_final[0::3] = best_overall_c[:, 0]
    v_final[1::3] = best_overall_c[:, 1]
    v_final[2::3] = np.maximum(best_overall_r * 0.999, 1e-6)

        res_f = minimize(slqp_objective, v_final, method='SLSQP', bounds=bounds_slqp,

        if res_f.success and np.min(slqp_constraints(res_f.x)) >= -1e-9:
            best_overall_c = np.column_stack((res_f.x[0::3], res_f.x[1::3]))
            best_overall_r = res_f.x[2::3]
            best_overall_sum = -res_f.fun
    except Exception:
        pass
        
    return best_overall_c, np.maximum(best_overall_r, 0.0), float(best_overall_sum)
