import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def compute_optimal_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Preallocate constraint matrices
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((4 * n + n_pairs, n))
    b_ub = np.zeros(4 * n + n_pairs)
    
    idx = 0
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        bounds_list = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_list:
            A_ub[idx, i] = 1.0
            b_ub[idx] = b
            idx += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 2.6e-4

def obj_centers(x_flat):
    """Objective for optimizing centers: minimize negative sum of optimal radii."""
    centers = x_flat.reshape(N, 2)
    centers = np.clip(centers, 0.0, 1.0)
    _, s = compute_optimal_radii(centers)
    return -s

def make_hex_init(seed, rot, scale):
    """Generates a hexagonal lattice initialization with rotation and scaling."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def make_force_init(seed):
    """Generates a spread-out initialization using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    for _ in range(300):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        pts += 0.005 * forces
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def make_params(c, r):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((c[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((c[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def slqp_obj(x):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[:N])

def slqp_constr(x):
    """Inequality constraints for SLSQP: pairwise non-overlap."""
    r = x[:N]
    u = x[N:2*N]
    v = x[2*N:3*N]
    cx = r + u * (1.0 - 2.0 * r)
    cy = r + v * (1.0 - 2.0 * r)
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_centers = None
    best_sum = -np.inf
    best_radii = None
    
    # Generate diverse initial configurations
    inits = []
    for s in range(15):
        rot = np.random.uniform(-0.3, 0.3)
        sc = np.random.uniform(0.85, 1.15)
        inits.append(make_hex_init(s, rot, sc))
    for s in range(10):
        inits.append(make_force_init(s))
        
    # Phase 1: Nelder-Mead optimization on centers
    for init_c in inits:
        x0 = init_c.flatten()
        try:
            res = minimize(obj_centers, x0, method='Nelder-Mead',
                           options={'maxiter': 2000, 'xatol': 1e-7, 'fatol': 1e-9})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Phase 2: Simulated Annealing local search on centers
    if best_centers is not None:
        curr_c = best_centers.copy()
        curr_s = best_sum
        temp = 0.04
        step = 0.025
        
        for it in range(3500):
            idx = np.random.randint(N)
            old_c = curr_c[idx].copy()
            curr_c[idx] += np.random.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
            _, new_s = compute_optimal_radii(curr_c)
            delta = new_s - curr_s
            
            if delta > 0 or (temp > 1e-6 and np.random.rand() < np.exp(delta / temp)):
                curr_s = new_s
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_centers = curr_c.copy()
            else:
                curr_c[idx] = old_c
                
            temp *= 0.999
            step = max(0.002, step * 0.998)
            
    # Phase 3: High-precision SLSQP polish
    if best_centers is not None:
        radii, _ = compute_optimal_radii(best_centers)
        radii = np.maximum(radii * 0.995, 1e-5)
        x0_slqp = make_params(best_centers, radii)
        bounds_slqp = [(1e-5, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
        
        try:
            res_slqp = minimize(slqp_obj, x0_slqp, method='SLSQP', bounds=bounds_slqp,
                                constraints={'type': 'ineq', 'fun': slqp_constr},
                                options={'maxiter': 8000, 'ftol': 1e-13})
            if res_slqp.success:
                cv = slqp_constr(res_slqp.x)
                if np.min(cv) >= -1e-7:
                    new_sum = -res_slqp.fun
                    if new_sum > best_sum:
                        best_sum = new_sum
                        r_opt = res_slqp.x[:N]
                        u_opt = res_slqp.x[N:2*N]
                        v_opt = res_slqp.x[2*N:3*N]
                        cx = r_opt + u_opt*(1.0-2.0*r_opt)
                        cy = r_opt + v_opt*(1.0-2.0*r_opt)
                        best_centers = np.column_stack((cx, cy))
                        best_radii = r_opt
        except Exception:
            pass
            
    # Fallback safety
    if best_centers is None:
        best_centers = make_hex_init(0, 0.0, 1.0)
    if best_radii is None:
        best_radii, best_sum = compute_optimal_radii(best_centers)
        
    return best_centers, best_radii, float(best_sum)
