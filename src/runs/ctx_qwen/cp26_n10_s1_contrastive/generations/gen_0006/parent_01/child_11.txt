import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 2.6e-4

def obj_slsqp(p):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons_slsqp(p):
    """Inequality constraints for SLSQP: pairwise non-overlap only."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Generate diverse hexagonal lattice initializations
    inits = []
    for s in range(30):
        pts = []
        r_est = 0.095
        y = r_est
        row = 0
        while len(pts) < N:
            shift = (row % 2) * r_est
            x = r_est + shift
            while x <= 1.0 - r_est and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
        pts = np.array(pts[:N])
        pts -= 0.5
        scale = np.random.uniform(0.85, 1.15)
        pts *= scale
        pts += 0.5
        rot = np.random.uniform(-0.3, 0.3)
        c, s_mat = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s_mat], [s_mat, c]])
        pts -= pts.mean(axis=0)
        pts += 0.5
        pts += np.random.uniform(-0.02, 0.02, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    # Phase 1: LP-driven Simulated Annealing on centers
    for init_centers in inits:
        centers = init_centers.copy()
        radii, curr_sum = solve_radii_lp(centers)
        
        temp = 0.04
        step = 0.035
        
        for it in range(1200):
            idx = np.random.randint(N)
            old = centers[idx].copy()
            centers[idx] += np.random.normal(0, step, 2)
            centers[idx] = np.clip(centers[idx], 0.005, 0.995)
            
            new_radii, new_sum = solve_radii_lp(centers)
            delta = new_sum - curr_sum
            
            if delta > 0 or (temp > 1e-9 and np.random.rand() < np.exp(delta / temp)):
                curr_sum = new_sum
                radii = new_radii
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()
            else:
                centers[idx] = old
                
            temp *= 0.9994
            step = max(0.0008, step * 0.9994)
            
    # Phase 2: High-precision SLSQP polish with boundary-safe parameterization
    if best_centers is not None:
        r_init = best_radii * 0.995
        denom = np.clip(1.0 - 2.0 * r_init, 1e-6, 1.0)
        u_init = np.clip((best_centers[:, 0] - r_init) / denom, 0.0, 1.0)
        v_init = np.clip((best_centers[:, 1] - r_init) / denom, 0.0, 1.0)
        
        params = np.zeros(3 * N)
        params[:N] = r_init
        params[N:2*N] = u_init
        params[2*N:] = v_init
        
        bounds_slsqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * (2*N)
        
        try:
            res = minimize(obj_slsqp, params, method='SLSQP', bounds=bounds_slsqp,
                           constraints={'type': 'ineq', 'fun': cons_slsqp},
                           options={'maxiter': 10000, 'ftol': 1e-14})
            if res.success and np.min(cons_slsqp(res.x)) >= -1e-7:
                r_opt = res.x[:N]
                u_opt = res.x[N:2*N]
                v_opt = res.x[2*N:]
                x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
                y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
                centers_final = np.column_stack((x_opt, y_opt))
                sum_final = np.sum(r_opt)
                if sum_final > best_sum:
                    best_sum = sum_final
                    best_centers = centers_final
                    best_radii = r_opt
        except Exception:
            pass
            
    # Fallback (should rarely be reached)
    if best_centers is None:
        best_centers = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_radii, best_sum = solve_radii_lp(best_centers)
        
    return best_centers, np.maximum(best_radii, 0.0), float(best_sum)
