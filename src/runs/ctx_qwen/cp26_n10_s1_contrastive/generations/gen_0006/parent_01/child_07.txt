import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Precompute pairwise distances
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    num_ineq = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    idx = 0
    
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def constraint_func(params):
    """Computes inequality constraints g(params) >= 0 for SLSQP."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def objective(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[:N])

def make_hex_init(rng, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.09
    y = r_est
    row = 0
    while len(pts) < N:
        x_off = r_est if row % 2 == 0 else 2.0 * r_est
        x = x_off
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force_init(rng):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(300):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        rep_mag = 1.0 / (dists**2)
        np.fill_diagonal(rep_mag, 0.0)
        forces = np.sum(rep_mag[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        
        for d in range(2):
            forces[:, d] += 15.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 15.0 * np.maximum(0, pts[:, d] - 0.95)
            
        pts += 0.004 * forces
        pts = np.clip(pts, 0.01, 0.99)
        
    return pts

def local_search_centers(centers, rng, steps=600, step_init=0.025):
    """Hill-climbing local search on centers maximizing LP sum of radii."""
    n = centers.shape[0]
    curr_centers = centers.copy()
    curr_sum, _ = solve_lp(curr_centers)
    best_sum = curr_sum
    best_centers = curr_centers.copy()
    step = step_init
    
    for t in range(steps):
        i = rng.integers(n)
        old = curr_centers[i].copy()
        curr_centers[i] += rng.normal(0, step, 2)
        curr_centers[i] = np.clip(curr_centers[i], 1e-4, 0.999)
        
        new_sum, _ = solve_lp(curr_centers)
        if new_sum > curr_sum:
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = curr_centers.copy()
        else:
            curr_centers[i] = old
            
        step *= 0.9985
        
    return best_centers, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds = [(1e-5, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_centers_lp = None
    best_sum_lp = -np.inf
    
    # Phase 1: Generate diverse initializations
    inits = []
    for _ in range(25):
        rot = rng.uniform(-0.35, 0.35)
        scale = rng.uniform(0.80, 1.20)
        inits.append(make_hex_init(rng, rot, scale))

    for _ in range(10):

        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng.uniform(-0.02, 0.02, pts.shape)
        inits.append(np.clip(pts, 0.02, 0.98))
        
    # Phase 2: LP-Guided Local Search on each initialization
    for init_c in inits:
        c_best, s_best = local_search_centers(init_c, rng, steps=500, step_init=0.03)
        if s_best > best_sum_lp:
            best_sum_lp = s_best
            best_centers_lp = c_best.copy()
            
    # Phase 3: Joint SLSQP Refinement
    r_lp, _ = solve_lp(best_centers_lp)
    denom = np.clip(1.0 - 2.0 * r_lp, 1e-6, 1.0)
    u = np.clip((best_centers_lp[:, 0] - r_lp) / denom, 0.0, 1.0)
    v = np.clip((best_centers_lp[:, 1] - r_lp) / denom, 0.0, 1.0)
    x0 = np.concatenate([r_lp, u, v])
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13})
        if res.success and np.min(constraint_func(res.x)) >= -1e-7:
            r_opt = res.x[:N]
            u_opt = res.x[N:2*N]
            v_opt = res.x[2*N:3*N]
            x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
            y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
            best_centers_lp = np.column_stack((x_opt, y_opt))
    except Exception:
        pass
        
    # Phase 4: Fine-grained Local Search + Final Polish
    c_fine, _ = local_search_centers(best_centers_lp, rng, steps=800, step_init=0.008)
    
    r_lp_fine, _ = solve_lp(c_fine)
    denom_f = np.clip(1.0 - 2.0 * r_lp_fine, 1e-6, 1.0)
    u_f = np.clip((c_fine[:, 0] - r_lp_fine) / denom_f, 0.0, 1.0)
    v_f = np.clip((c_fine[:, 1] - r_lp_fine) / denom_f, 0.0, 1.0)
    x0_f = np.concatenate([r_lp_fine, u_f, v_f])
    
    try:
        res_f = minimize(objective, x0_f, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
        if res_f.success and np.min(constraint_func(res_f.x)) >= -1e-8:
            r_final = res_f.x[:N]
            u_final = res_f.x[N:2*N]
            v_final = res_f.x[2*N:3*N]
            x_final = r_final + u_final * (1.0 - 2.0 * r_final)
            y_final = r_final + v_final * (1.0 - 2.0 * r_final)

    # Final LP projection to guarantee strict feasibility
    radii_final, sum_final = solve_lp(final_centers)
    
    return final_centers, radii_final, float(sum_final)
