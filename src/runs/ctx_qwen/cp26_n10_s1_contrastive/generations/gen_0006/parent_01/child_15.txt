import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find optimal radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Precompute pairwise distances efficiently
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Constraints matrix and vector
    # 4 boundary constraints per circle + N*(N-1)/2 pairwise constraints
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-4

def slsqp_objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[:N])

def slsqp_constraints(v):
    """Inequality constraints: pairwise non-overlap using (r, u, w) parameterization."""
    r = v[:N]
    u = v[N:2*N]
    w = v[2*N:3*N]
    
    # Map to physical coordinates: automatically satisfies x in [r, 1-r]
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Constraint: dist_sq >= (r_i + r_j)^2
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_params(centers, radii):
    """Convert physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, w])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_slq = [(1e-5, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slq = {'type': 'ineq', 'fun': slsqp_constraints}
    
    best_v = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Rotated & Scaled Hexagonal Lattices
    for s in range(35):
        pts = []
        r_est = 0.095
        y = r_est
        row = 0
        while len(pts) < N:
            shift = (row % 2) * r_est
            x = r_est + shift
            while x <= 1.0 - r_est and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
        pts = np.array(pts[:N])
        
        # Center, scale, rotate, jitter
        pts -= 0.5
        sc = rng.uniform(0.82, 1.18)
        pts *= sc
        pts += 0.5
        
        rot = rng.uniform(-0.35, 0.35)
        c_val, s_val = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c_val, -s_val], [s_val, c_val]])
        pts -= pts.mean(axis=0)
        pts += 0.5
        pts += rng.uniform(-0.012, 0.012, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        
        radii, _ = solve_radii_lp(pts)
        inits.append(make_params(pts, radii))
        
    # 2. Force-directed organic layouts
    for s in range(25):
        pts = rng.uniform(0.05, 0.95, (N, 2))

            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            dists = np.maximum(dists, 1e-6)
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
            pts += 0.0025 * forces
            pts = np.clip(pts, 0.02, 0.98)
        radii, _ = solve_radii_lp(pts)
        inits.append(make_params(pts, radii))
        
    # Phase 1: Broad SLSQP search from diverse starts
    for x0 in inits:
        try:
            res = minimize(slsqp_objective, x0, method='SLSQP', bounds=bounds_slq,
                           constraints=cons_slq, options={'maxiter': 5000, 'ftol': 1e-12})
            if res.success:
                if np.min(slsqp_constraints(res.x)) >= -1e-7:
                    s_val = np.sum(res.x[:N])
                    if s_val > best_sum:
                        best_sum = s_val
                        best_v = res.x.copy()
        except Exception:
            continue
            
    if best_v is None:
        best_v = inits[0]
        
    # Extract centers for SA phase
    r_cur = best_v[:N]
    u_cur = best_v[N:2*N]
    w_cur = best_v[2*N:3*N]

                                  r_cur + w_cur * (1.0 - 2.0 * r_cur)))
    radii_sa, sum_sa = solve_radii_lp(centers_sa)
    
    # Phase 2: Simulated Annealing on centers using LP objective
    best_c = centers_sa.copy()
    best_r, best_s = radii_sa.copy(), sum_sa
    temp = 0.045
    step = 0.028
    
    for _ in range(9000):
        op = rng.choice(['move', 'swap'])
        old_c = best_c.copy()
        
        if op == 'move':
            idx = rng.integers(N)
            best_c[idx] += rng.normal(0, step, 2)
            best_c[idx] = np.clip(best_c[idx], 0.005, 0.995)
        else:
            i, j = rng.choice(N, 2, replace=False)
            best_c[[i, j]] = best_c[[j, i]]
            
        _, new_s = solve_radii_lp(best_c)
        delta = new_s - best_s
        
        if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(min(delta / temp, 100.0))):
            best_s = new_s
        else:
            best_c = old_c
            
        temp *= 0.9992
        step = max(0.002, step * 0.9995)
        
    best_r, _ = solve_radii_lp(best_c)
    best_v = make_params(best_c, best_r * 0.995)  # Slight shrink for strict interior feasibility
    
    # Phase 3: SLSQP refinement on SA result
    try:
        res = minimize(slsqp_objective, best_v, method='SLSQP', bounds=bounds_slq,
                       constraints=cons_slq, options={'maxiter': 6000, 'ftol': 1e-12})
        if res.success and np.min(slsqp_constraints(res.x)) >= -1e-7:
            if -res.fun > best_s:
                best_v = res.x
                best_s = -res.fun
    except Exception:
        pass
        
    # Phase 4: High-precision final polish
    try:
        res = minimize(slsqp_objective, best_v, method='SLSQP', bounds=bounds_slq,
                       constraints=cons_slq, options={'maxiter': 10000, 'ftol': 1e-14})
        if res.success and np.min(slsqp_constraints(res.x)) >= -1e-8:
            best_v = res.x
    except Exception:
        pass
        
    # Reconstruct physical centers and radii
    r_opt = best_v[:N]
    u_opt = best_v[N:2*N]
    w_opt = best_v[2*N:3*N]
    centers = np.column_stack((r_opt + u_opt * (1.0 - 2.0 * r_opt), 
                               r_opt + w_opt * (1.0 - 2.0 * r_opt)))
    radii = np.maximum(r_opt, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
