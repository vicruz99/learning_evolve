import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-5

def hex_centers(scale=1.0, rot=0.0, jitter=0.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.default_rng()
    pts = []
    r_est = 0.1
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R + 0.5
        
    if jitter > 0:
        pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def objective_slsqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints_slsqp(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Phase 1: Simulated Annealing on centers with LP oracle
    for trial in range(20):
        if trial == 0:
            centers = hex_centers(scale=1.0, rot=0.0, jitter=0.015)
        else:
            centers = hex_centers(
                scale=0.85 + 0.3 * np.random.rand(),
                rot=np.random.uniform(-0.35, 0.35),
                jitter=0.025
            )
            
        radii, curr_sum = solve_lp(centers)
        best_trial_sum = curr_sum
        best_trial_c = centers.copy()
        best_trial_r = radii.copy()
        
        temp = 0.06
        step = 0.045
        
        for _ in range(3500):
            idx = np.random.randint(N)
            old_pos = centers[idx].copy()
            direction = np.random.randn(2)
            direction /= np.linalg.norm(direction)
            centers[idx] = old_pos + step * direction
            centers[idx] = np.clip(centers[idx], 0.005, 0.995)
            
            new_radii, new_sum = solve_lp(centers)
            delta = new_sum - curr_sum
            
            # Accept if better, or with probability exp(delta/temp)
            if delta > 0 or (temp > 1e-8 and np.random.rand() < np.exp(min(delta / temp, 700.0))):
                curr_sum = new_sum
                if curr_sum > best_trial_sum:
                    best_trial_sum = curr_sum
                    best_trial_c = centers.copy()
                    best_trial_r = new_radii.copy()
            else:
                centers[idx] = old_pos
                
            temp *= 0.9983
            step = max(0.0015, step * 0.998)
            
        if best_trial_sum > best_sum:
            best_sum = best_trial_sum
            best_centers = best_trial_c.copy()
            best_radii = best_trial_r.copy()
            
    # Phase 2: SLSQP joint refinement
    if best_centers is not None:
        v0 = np.zeros(3 * N)
        v0[0::3] = best_centers[:, 0]
        v0[1::3] = best_centers[:, 1]
        v0[2::3] = np.maximum(best_radii * 0.992, 1e-6)

                        best_sum = curr_s
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
        except Exception:
            pass
            
        # Perturbation restarts around best SLSQP result

            vp = v0.copy()
            vp[0::3] += np.random.uniform(-0.004, 0.004, N)
            vp[1::3] += np.random.uniform(-0.004, 0.004, N)
            vp[2::3] += np.random.uniform(-0.0005, 0.0005, N)
            vp[0::3] = np.clip(vp[0::3], 0.01, 0.99)
            vp[1::3] = np.clip(vp[1::3], 0.01, 0.99)
            vp[2::3] = np.clip(vp[2::3], 1e-6, 0.49)

                        s_p = -res_p.fun
                        if s_p > best_sum:
                            best_sum = s_p
                            best_centers = np.column_stack((res_p.x[0::3], res_p.x[1::3]))
                            best_radii = res_p.x[2::3]
            except Exception:
                pass
                
        # Final high-precision polish
        v0_final = np.zeros(3 * N)
        v0_final[0::3] = best_centers[:, 0]
        v0_final[1::3] = best_centers[:, 1]
        v0_final[2::3] = np.maximum(best_radii * 0.999, 1e-6)
        try:
            res_f = minimize(objective_slsqp, v0_final, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons_slqp, options={'maxiter': 12000, 'ftol': 1e-14})
            if res_f.success and np.min(constraints_slsqp(res_f.x)) >= -1e-8:
                best_sum = -res_f.fun
                best_centers = np.column_stack((res_f.x[0::3], res_f.x[1::3]))
                best_radii = res_f.x[2::3]
        except Exception:
            pass

    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = hex_centers()
        best_radii, best_sum = solve_lp(best_centers)
        
    return best_centers, np.maximum(best_radii, 0.0), float(best_sum)
