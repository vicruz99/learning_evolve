

<strategy>
The strategy focuses on a hybrid optimization pipeline combining joint constrained optimization with LP-driven local search:
1. **Initialization**: Generate diverse starting configurations using rotated/scaled hexagonal lattices and force-directed layouts. This covers both structured and organic packing topologies.
2. **Joint Optimization (SLSQP)**: Use a specialized `(r, u, v)` parameterization that automatically satisfies boundary constraints (`r <= x <= 1-r`), allowing SLSQP to focus purely on pairwise non-overlap constraints. This phase finds a locally optimal configuration jointly optimizing centers and radii.
3. **LP-Driven Simulated Annealing**: The joint optimization can get stuck in local minima. To escape, we fix the centers and solve a Linear Program (LP) to find the optimal radii for those positions. We then perform simulated annealing on the *centers only*, using the LP-optimized sum of radii as the objective. This separates the difficult joint landscape into a smoother center-search landscape. Moves include single-circle translations and global rotations.
4. **Final Polish**: Convert the best LP-optimized configuration back to SLSQP parameters and run a high-precision final optimization step to squeeze out remaining gains.
This approach leverages the speed and global perspective of LP for radii, while using SA to navigate the non-convex center space effectively.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-6 * n

def objective(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraint_func(params):
    """
    Computes inequality constraints g(params) >= 0.
    Uses (r, u, v) parameterization which automatically satisfies boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_hex_params(rng, rot, scale):
    """Generates a hexagonal lattice initialization with rotation and scaling."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        x_off = r_est if row % 2 == 0 else 2.0 * r_est
        x = x_off
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    
    diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    d_pair = np.min(dists, axis=1)
    d_wall = np.minimum(np.minimum(pts[:, 0], 1.0 - pts[:, 0]), np.minimum(pts[:, 1], 1.0 - pts[:, 1]))
    r = 0.9 * np.minimum(d_wall, d_pair / 2.0)
    
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((pts[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((pts[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def make_force_params(rng, steps=300):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for s in range(steps):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        rep = 1.0 / (dists**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        
        for d in range(2):
            forces[:, d] += 5.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 5.0 * np.maximum(0, pts[:, d] - 0.95)
            
        step_sz = 0.008 * (1.0 - s / float(steps))
        pts += step_sz * forces
        pts = np.clip(pts, 0.02, 0.98)
        
    diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    d_pair = np.min(dists, axis=1)
    d_wall = np.minimum(np.minimum(pts[:, 0], 1.0 - pts[:, 0]), np.minimum(pts[:, 1], 1.0 - pts[:, 1]))
    r = 0.9 * np.minimum(d_wall, d_pair / 2.0)
    
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((pts[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((pts[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds = [(1e-5, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_sum = -np.inf
    
    inits = []
    # Generate diverse hexagonal initializations
    for s in range(30):
        rot = rng.uniform(-0.3, 0.3)
        scale = rng.uniform(0.85, 1.15)
        inits.append(make_hex_params(rng, rot, scale))
    # Generate force-directed initializations
    for s in range(20):
        inits.append(make_force_params(rng))
        
    # Phase 1: Broad SLSQP search
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
            if res.success:
                c_val = constraint_func(res.x)
                if np.min(c_val) >= -1e-7:
                    s_val = np.sum(res.x[:N])
                    if s_val > best_sum:
                        best_sum = s_val
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    best_centers = None
    best_radii = None
    
    if best_vars is not None:
        # Reconstruct centers for LP-based refinement
        r_opt = best_vars[:N]
        u_opt = best_vars[N:2*N]
        v_opt = best_vars[2*N:3*N]
        x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
        y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
        current_centers = np.column_stack((x_opt, y_opt))
        
        current_radii, current_sum = solve_lp(current_centers)
        best_lp_centers = current_centers.copy()
        best_lp_radii = current_radii.copy()
        best_lp_sum = current_sum
        
        # Phase 2: LP-driven Simulated Annealing on centers
        temp = 0.02
        step_size = 0.025
        
        for it in range(4000):
            op = rng.choice(['move', 'rotate'])
            if op == 'move':
                idx = rng.integers(N)
                old_pos = current_centers[idx].copy()
                direction = rng.standard_normal(2)
                direction /= np.linalg.norm(direction)
                current_centers[idx] += direction * step_size
                current_centers[idx] = np.clip(current_centers[idx], 0.01, 0.99)
            else:
                angle = rng.uniform(-0.05, 0.05)
                c_val, s_val = np.cos(angle), np.sin(angle)
                R = np.array([[c_val, -s_val], [s_val, c_val]])
                current_centers = (current_centers - 0.5) @ R.T + 0.5
                current_centers = np.clip(current_centers, 0.01, 0.99)
                
            new_radii, new_sum = solve_lp(current_centers)
            delta = new_sum - current_sum
            
            if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(delta / temp)):
                current_sum = new_sum
                if new_sum > best_lp_sum:
                    best_lp_sum = new_sum
                    best_lp_centers = current_centers.copy()
                    best_lp_radii = new_radii.copy()
            else:
                if op == 'move':
                    current_centers[idx] = old_pos
                    
            temp *= 0.9992
            step_size = max(0.001, step_size * 0.999)
            
        best_centers = best_lp_centers
        best_radii = best_lp_radii
        best_sum = best_lp_sum
        
        # Phase 3: High-precision SLSQP polish
        r_opt = best_radii.copy()
        denom = np.clip(1.0 - 2.0 * r_opt, 1e-6, 1.0)
        u_opt = np.clip((best_centers[:, 0] - r_opt) / denom, 0.0, 1.0)
        v_opt = np.clip((best_centers[:, 1] - r_opt) / denom, 0.0, 1.0)
        final_vars = np.concatenate([r_opt * 0.98, u_opt, v_opt])
        
        try:
            res_f = minimize(objective, final_vars, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13})
            if res_f.success and np.min(constraint_func(res_f.x)) >= -1e-7:
                r_opt = res_f.x[:N]
                u_opt = res_f.x[N:2*N]
                v_opt = res_f.x[2*N:3*N]
                x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
                y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
                best_centers = np.column_stack((x_opt, y_opt))
                best_radii = r_opt
                best_sum = np.sum(r_opt)
        except Exception:
            pass

    # Fallback configuration
    if best_centers is None:
        best_centers = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                        np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        best_radii = np.full(N, 0.04)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
