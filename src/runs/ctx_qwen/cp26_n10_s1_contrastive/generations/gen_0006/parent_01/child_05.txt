import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    
    # Precompute boundary bounds
    b_ub = np.concatenate([
        centers[:, 0], 
        1.0 - centers[:, 0], 
        centers[:, 1], 
        1.0 - centers[:, 1]
    ])
    
    # Precompute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = np.concatenate([b_ub, dists[I_IDX, J_IDX]])
    
    # Build constraint matrix A_ub @ r <= b_ub
    num_cons = len(b_ub)
    A_ub = np.zeros((num_cons, n))
    
    # Boundary constraints: r_i <= bound
    for i in range(n):
        A_ub[i, i] = 1.0
        A_ub[n + i, i] = 1.0
        A_ub[2 * n + i, i] = 1.0
        A_ub[3 * n + i, i] = 1.0
        
    # Pairwise constraints: r_i + r_j <= dist_ij
    idx = 4 * n
    for k in range(len(I_IDX)):
        i, j = I_IDX[k], J_IDX[k]
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        idx += 1
        
    bounds = [(0.0, None)] * n
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 1e-5

def lp_obj(c_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = c_flat.reshape(N, 2)
    _, s = solve_lp(centers)
    return -s

def get_hex_centers(row_counts, scale=1.0, rot=0.0):
    """Generates a hexagonal lattice configuration."""
    pts = []
    r = 0.09
    y = r
    row = 0
    for cnt in row_counts:
        x = r if row % 2 == 0 else 2.0 * r
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3.0) * r
        row += 1
    pts = np.array(pts[:N])
    pts -= 0.5
    pts *= scale
    pts += 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts -= pts.mean(axis=0)
        pts += 0.5
    return np.clip(pts, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_s = -np.inf

        sc = np.random.uniform(0.85, 1.15)
        starts.append(get_hex_centers([6, 5, 6, 5, 4], sc, rot))
    for s in range(10):
        starts.append(get_hex_centers([5, 6, 5, 6, 4], np.random.uniform(0.85, 1.15), np.random.uniform(-0.3, 0.3)))
    # Random scattered starts
    for s in range(10):
        rng = np.random.RandomState(s + 100)
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # Phase 2: Simulated Annealing on Centers with LP evaluation
    for c0 in starts:
        curr_c = c0.copy()
        _, curr_val = solve_lp(curr_c)
        
        temp = 0.025
        step = 0.025
        for it in range(1500):
            idx = np.random.randint(N)
            move = np.random.uniform(-step, step, 2)
            new_c = curr_c.copy()
            new_c[idx] = np.clip(new_c[idx] + move, 0.02, 0.98)
            
            _, new_val = solve_lp(new_c)
            delta = new_val - curr_val
            
            if delta > 0 or (temp > 1e-6 and np.random.rand() < np.exp(delta / temp)):
                curr_c = new_c
                curr_val = new_val
                if curr_val > best_s:
                    best_s = curr_val
                    best_c = curr_c.copy()
                    
            temp *= 0.9992
            step = max(0.0015, step * 0.999)
            
    # Phase 3: Nelder-Mead refinement on centers

        for _ in range(12):
            c_pert = best_c.copy()
            c_pert += np.random.uniform(-0.008, 0.008, c_pert.shape)

            res = minimize(lp_obj, c_pert.flatten(), method='Nelder-Mead',

            if res.fun < -best_s:
                best_s = -res.fun
                best_c = res.x.reshape(N, 2)

    r_lp, _ = solve_lp(best_c)

        return -np.sum(v[2::3])
        
    def slqp_cons(v):

        y = v[1::3]

        c.append(y - r)
        c.append(1.0 - y - r)

        dy = y[:, np.newaxis] - y[np.newaxis, :]
        d2 = dx**2 + dy**2
        rs = r[:, np.newaxis] + r[np.newaxis, :]
        c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
        return np.concatenate(c)
        
    cons_slqp = {'type': 'ineq', 'fun': slqp_cons}
    
    best_vars = None
    best_final_sum = best_s
    
    # Generate SLSQP starts from best LP result + perturbations
    slqp_starts = [x0.copy()]
    for _ in range(20):
        x_p = x0.copy()
        x_p[0::3] += np.random.uniform(-0.015, 0.015, N)
        x_p[1::3] += np.random.uniform(-0.015, 0.015, N)
        x_p[2::3] += np.random.uniform(-0.001, 0.001, N)
        x_p[:2*N] = np.clip(x_p[:2*N], 0.01, 0.99)
        x_p[2::3] = np.clip(x_p[2::3], 1e-5, 0.49)
        slqp_starts.append(x_p)
        
    for x_init in slqp_starts:
        try:
            res = minimize(slqp_obj, x_init, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 6000, 'ftol': 1e-13})
            if res.success:
                if np.min(slqp_cons(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_final_sum:
                        best_final_sum = s_val
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # Fallback to LP result if SLSQP fails to improve
    if best_vars is None:
        best_vars = x0
        best_final_sum = best_s
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = np.maximum(best_vars[2::3], 0.0)
    
    return centers, radii, float(best_final_sum)
