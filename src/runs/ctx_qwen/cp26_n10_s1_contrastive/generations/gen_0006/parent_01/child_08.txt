import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub_rows = []
    b_ub = []
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        b_ub.extend([x, 1.0 - x, y, 1.0 - y])
        for _ in range(4):
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_rows.append(row)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def obj_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def cons_joint(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def gen_hex_init(rng):
    """Generates a hexagonal lattice initialization with random rotation/scale."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = r_est if row % 2 == 1 else 0.0
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    pts -= 0.5
    scale = rng.uniform(0.85, 1.15)
    pts *= scale
    pts += 0.5
    
    rot = rng.uniform(-0.3, 0.3)
    c, s = np.cos(rot), np.sin(rot)
    R = np.array([[c, -s], [s, c]])
    pts = pts @ R.T
    
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def gen_force_init(rng):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(200):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        rep = 1.0 / (dists**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        
        for d in range(2):
            forces[:, d] += 5.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 5.0 * np.maximum(0, pts[:, d] - 0.95)
            
        pts += 0.004 * forces
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def hill_climb_sa(centers_init, rng, max_iters=2000):
    """Simulated Annealing on centers to maximize LP sum of radii."""
    centers = centers_init.copy()
    _, current_sum = solve_lp(centers)
    best_sum = current_sum
    best_c = centers.copy()
    
    temp = 0.015
    step = 0.02
    
    for _ in range(max_iters):
        idx = rng.integers(N)
        old = centers[idx].copy()
        centers[idx] = np.clip(old + rng.normal(0, step, 2), 0.005, 0.995)
        
        _, new_sum = solve_lp(centers)
        delta = new_sum - current_sum
        
        if delta > 0 or (rng.rand() < np.exp(delta / max(temp, 1e-6))):
            current_sum = new_sum
            if current_sum > best_sum:
                best_sum = current_sum
                best_c = centers.copy()
        else:
            centers[idx] = old
            
        if step > 0.0008:
            step *= 0.998
        temp *= 0.9995
            
    return best_c, best_sum

def slsqp_refine(centers, radii):
    """Joint SLSQP refinement of centers and radii."""
    v0 = np.zeros(3 * N)
    v0[0::3] = centers[:, 0]
    v0[1::3] = centers[:, 1]
    v0[2::3] = np.maximum(radii * 0.995, 1e-5)
    
    bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': cons_joint}
    
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(cons_joint(res.x)) >= -1e-7:
            c = np.column_stack((res.x[0::3], res.x[1::3]))
            r = res.x[2::3]
            return c, r, np.sum(r)
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def run_packing():
    rng = np.random.default_rng(42)
    
    best_c = None
    best_r = None
    best_s = -np.inf
    
    candidates = []
    
    # Phase 0: Generate diverse initial configurations
    for _ in range(5):
        candidates.append(gen_hex_init(rng))
    for _ in range(3):
        candidates.append(gen_force_init(rng))
        
    # Phase 1: SA on centers to maximize LP sum
    for c_init in candidates:
        c_sa, s_sa = hill_climb_sa(c_init, rng, max_iters=2000)
        _, r_sa = solve_lp(c_sa)
        if s_sa > best_s:
            best_s = s_sa
            best_c = c_sa
            best_r = r_sa
            
    # Phase 2: SLSQP refinement from best LP centers
    if best_c is not None:
        c_ref, r_ref, s_ref = slsqp_refine(best_c, best_r)
        if s_ref > best_s:
            best_s = s_ref
            best_c = c_ref
            best_r = r_ref
            
        # Perturb best centers and refine to escape local minima
        for _ in range(4):
            c_pert = best_c.copy()
            c_pert += rng.normal(0, 0.015, c_pert.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            c_ref2, r_ref2, s_ref2 = slsqp_refine(c_pert, best_r)
            if s_ref2 > best_s:
                best_s = s_ref2
                best_c = c_ref2
                best_r = r_ref2
                
    # Phase 3: High-precision final polish
    if best_c is not None:
        v0 = np.zeros(3 * N)
        v0[0::3] = best_c[:, 0]
        v0[1::3] = best_c[:, 1]
        v0[2::3] = np.maximum(best_r * 0.999, 1e-5)
        
        bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(1e-6, 0.5)] * N
        cons = {'type': 'ineq', 'fun': cons_joint}
        
        try:
            res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if res.success and np.min(cons_joint(res.x)) >= -1e-8:
                best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                best_r = res.x[2::3]
                best_s = np.sum(best_r)
        except Exception:
            pass
            
    # Fallback (should rarely be reached)
    if best_c is None:
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        best_c = pts
        best_r = np.full(N, 0.04)
        best_s = np.sum(best_r)
        
    return best_c, np.maximum(best_r, 0.0), float(best_s)
