import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 1e-5 * n

def objective_slqp(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints_slqp(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def make_hex_init(rng, rot, scale):
    """Generates a hexagonal lattice initialization with rotation and scaling."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = pts @ M.T
        pts = pts - pts.mean(axis=0) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_force_init(rng, steps=300):
    """Generates a strictly feasible initial configuration using force-directed layout."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    r = np.full(N, 0.05)
    for s in range(steps):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        rep = 1.0 / (dists**2)
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
        step_size = 0.005 * (1.0 - s / float(steps))
        pts += step_size * forces
        pts = np.clip(pts, 0.01, 0.99)
        np.fill_diagonal(dists, np.inf)
        d_pair = np.min(dists, axis=1)
        d_wall = np.minimum(np.minimum(pts[:, 0], 1.0 - pts[:, 0]), np.minimum(pts[:, 1], 1.0 - pts[:, 1]))
        r = 0.9 * np.minimum(d_wall, d_pair / 2.0)
    return pts, r

def make_params(centers, radii):
    """Map physical centers/radii to SLSQP optimization variables."""
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = radii
    return v

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_slqp = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(1e-6, 0.5)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints_slqp}
    
    best_vars = None
    best_sum = -np.inf
    
    # Phase 1: Diverse SLSQP starts
    starts = []
    for _ in range(25):
        rot = rng.uniform(-0.3, 0.3)
        sc = rng.uniform(0.85, 1.15)
        starts.append(make_hex_init(rng, rot, sc))
    for _ in range(15):
        pts, r = make_force_init(rng)
        starts.append(pts)
        
    for pts in starts:
        r_init, _ = solve_radii_lp(pts)
        r_init = np.maximum(r_init * 0.995, 1e-6)
        v0 = make_params(pts, r_init)
        v0[0::3] = np.clip(v0[0::3], 0.001, 0.999)
        v0[1::3] = np.clip(v0[1::3], 0.001, 0.999)
        
        try:
            res = minimize(objective_slqp, v0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                if np.min(constraints_slqp(res.x)) >= -1e-8:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    if best_vars is None:
        pts = make_hex_init(rng, 0.0, 1.0)
        r_f, _ = solve_radii_lp(pts)
        best_vars = make_params(pts, r_f * 0.995)
        best_sum = np.sum(best_vars[2::3])
        
    # Phase 2: LP-based Simulated Annealing on centers to escape topological local minima
    centers_best = np.column_stack((best_vars[0::3], best_vars[1::3]))
    
    sa_rng = np.random.default_rng(123)
    curr_c = centers_best.copy()
    _, curr_sum = solve_radii_lp(curr_c)
    best_c_sa = curr_c.copy()
    best_sum_sa = curr_sum
    best_r_sa, _ = solve_radii_lp(best_c_sa)
    
    temp = 0.02
    step_size = 0.025
    
    for it in range(2000):
        idx = sa_rng.integers(N)
        old = curr_c[idx].copy()
        curr_c[idx] += sa_rng.normal(0, step_size, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
        
        r_new, sum_new = solve_radii_lp(curr_c)
        delta = sum_new - curr_sum
        
        if delta > 0 or (temp > 1e-6 and sa_rng.random() < np.exp(delta / temp)):
            curr_sum = sum_new
            if sum_new > best_sum_sa:
                best_sum_sa = sum_new
                best_c_sa = curr_c.copy()
                best_r_sa = r_new.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9993
        step_size = max(0.001, step_size * 0.9995)
        
    if best_sum_sa > best_sum:
        best_sum = best_sum_sa
        centers_best = best_c_sa
        
    # Phase 3: High-precision SLSQP polish
    v0_polish = make_params(centers_best, best_r_sa * 0.9999)
    v0_polish[0::3] = np.clip(v0_polish[0::3], 0.001, 0.999)
    v0_polish[1::3] = np.clip(v0_polish[1::3], 0.001, 0.999)
    
    try:
        res_f = minimize(objective_slqp, v0_polish, method='SLSQP', bounds=bounds_slqp,
                         constraints=cons_slqp, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(constraints_slqp(res_f.x)) >= -1e-9:
            if -res_f.fun > best_sum:
                best_vars = res_f.x
                best_sum = -res_f.fun
    except Exception:
        pass
        
    centers_out = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii_out = best_vars[2::3]
    
    # Final validation & safety shrink to guarantee strict feasibility
    diff = centers_out[:, np.newaxis, :] - centers_out[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = radii_out[:, np.newaxis] + radii_out[np.newaxis, :]
    violation = np.maximum(0, r_sum - dists)
    max_viol = np.max(np.triu(violation, k=1))
    
    if max_viol > 1e-10:
        scale = 1.0 - max_viol / (np.sum(radii_out) + 1e-9)
        radii_out *= scale
        
    for i in range(N):
        x, y = centers_out[i]
        r = radii_out[i]
        max_r = min(x, 1.0-x, y, 1.0-y)
        if r > max_r + 1e-10:
            radii_out[i] = max_r
            
    return centers_out, np.maximum(radii_out, 0.0), float(np.sum(radii_out))
