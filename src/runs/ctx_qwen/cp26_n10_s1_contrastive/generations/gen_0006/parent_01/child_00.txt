import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def params_to_centers(params):
    """Converts (r, u, v) parameters back to physical centers."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y))

def centers_to_params(centers, radii):
    """Converts physical centers and radii to (r, u, v) parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def constraint_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def lp_obj_nm(x_flat):
    """Objective for Nelder-Mead: minimize negative LP sum of radii for fixed centers."""
    centers = x_flat.reshape(N, 2)
    _, s = solve_lp(centers)
    return -s

def make_hex_init(row_counts, rot, scale, jitter, rng):
    """Generates a hexagonal lattice initialization with specified parameters."""
    pts = []
    r_est = 0.09
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-jitter, jitter, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    
    diff = pts[:, None, :] - pts[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    d_pair = np.min(dists, axis=1)
    d_wall = np.minimum(np.minimum(pts[:, 0], 1.0 - pts[:, 0]), np.minimum(pts[:, 1], 1.0 - pts[:, 1]))
    r = 0.9 * np.minimum(d_wall, d_pair / 2.0)
    return centers_to_params(pts, r)

def local_search_lp(centers, n_iter=1500):
    """Iteratively perturbs centers to maximize the LP sum of radii."""
    rng = np.random.default_rng()
    curr_c = centers.copy()
    _, curr_s = solve_lp(curr_c)
    step = 0.025
    
    for _ in range(n_iter):
        idx = rng.integers(N)
        direction = rng.uniform(-1, 1, 2)
        direction /= np.linalg.norm(direction)
        new_c = curr_c.copy()
        new_c[idx] += step * direction
        new_c[idx] = np.clip(new_c[idx], 0.01, 0.99)
        
        _, new_s = solve_lp(new_c)
        if new_s > curr_s + 1e-9:
            curr_c = new_c
            curr_s = new_s
            step = min(0.05, step * 1.02)
        else:
            step = max(0.001, step * 0.96)
    return curr_c, curr_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_params = [(1e-5, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_sum = -np.inf
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [7,5,5,5,4], [4,5,5,5,7], [6,6,6,4,4], [7,6,6,7],
        [8,6,6,6], [6,4,6,5,5], [5,7,5,5,4], [4,5,5,7,5],
        [6,5,5,6,4], [5,5,6,5,5]
    ]
    
    inits = []
    # Hexagonal lattice variations
    for pat in patterns:
        for _ in range(3):
            rot = rng.uniform(-0.35, 0.35)
            scale = rng.uniform(0.85, 1.15)
            jit = rng.uniform(0.01, 0.035)
            inits.append(make_hex_init(pat, rot, scale, jit, rng))
            
    # Random feasible initializations
    for _ in range(25):
        pts = rng.uniform(0.1, 0.9, (N, 2))
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        d_pair = np.min(dists, axis=1)
        d_wall = np.minimum(np.minimum(pts[:, 0], 1.0 - pts[:, 0]), np.minimum(pts[:, 1], 1.0 - pts[:, 1]))
        r = 0.85 * np.minimum(d_wall, d_pair / 2.0)
        inits.append(centers_to_params(pts, r))
        
    # Phase 1: Broad SLSQP search
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_params,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
            if res.success and np.min(constraint_func(res.x)) >= -1e-7:
                curr_sum = np.sum(res.x[:N])
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x.copy()
        except Exception:
            continue
            
    if best_vars is None:
        best_vars = inits[0]
        
    # Phase 2: Hybrid Refinement Loop (LP Local Search + Nelder-Mead + SLSQP)
    best_c = params_to_centers(best_vars)
    for _ in range(4):
        # LP-based center optimization
        best_c, _ = local_search_lp(best_c, n_iter=1200)
        
        # Nelder-Mead on center coordinates to navigate non-smooth LP landscape
        c_flat = best_c.flatten()
        try:
            res_nm = minimize(lp_obj_nm, c_flat, method='Nelder-Mead',
                              options={'maxiter': 6000, 'xatol': 1e-7, 'fatol': 1e-9})
            best_c = res_nm.x.reshape(N, 2)
        except Exception:
            pass
            
        _, curr_lp_sum = solve_lp(best_c)
        if curr_lp_sum > best_sum:
            best_sum = curr_lp_sum
            
        # Refine with SLSQP from the improved LP centers
        _, radii_lp = solve_lp(best_c)
        x0_slqp = centers_to_params(best_c, radii_lp * 0.995)
        try:
            res_slqp = minimize(objective, x0_slqp, method='SLSQP', bounds=bounds_params,
                                constraints=cons, options={'maxiter': 6000, 'ftol': 1e-12})
            if res_slqp.success and np.min(constraint_func(res_slqp.x)) >= -1e-7:
                curr_sum = np.sum(res_slqp.x[:N])
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res_slqp.x.copy()
                    best_c = params_to_centers(best_vars)
        except Exception:
            pass
            
    # Phase 3: High-precision final polish
    if best_vars is not None:
        x0_polish = best_vars.copy()
        x0_polish[:N] *= 0.9999 # Ensure strict interior feasibility for final step
        try:
            res_polish = minimize(objective, x0_polish, method='SLSQP', bounds=bounds_params,
                                  constraints=cons, options={'maxiter': 12000, 'ftol': 1e-14})
            if res_polish.success and np.min(constraint_func(res_polish.x)) >= -1e-8:
                best_vars = res_polish.x
        except Exception:
            pass
            
    # Reconstruct final output
    r_opt = best_vars[:N]
    u_opt = best_vars[N:2*N]
    v_opt = best_vars[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    return centers, np.maximum(r_opt, 0.0), float(np.sum(r_opt))
