To maximize the sum of radii for 26 circles in a unit square, we employ a two-phase optimization strategy that combines global search with precise local refinement:

1. **Simulated Annealing on Centers + LP for Radii**: 
   For a fixed set of circle centers, finding the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints is a Linear Programming (LP) problem. This LP can be solved extremely fast. We use this property to define an objective landscape over center positions: $f(C) = \max \sum r_i$ given $C$. We explore this landscape using Simulated Annealing (SA), which allows probabilistic acceptance of worse moves to escape local minima. We include both single-circle moves and coordinated 2-circle moves to efficiently navigate the configuration space. Multiple diverse initializations (hexagonal lattices and force-directed layouts) ensure broad exploration.

2. **Parameterized SLSQP Polish**:
   Once SA finds a high-quality configuration, we convert it into a specialized parameterization $(r, u, v)$ where $x_i = r_i + u_i(1-2r_i)$ and $y_i = r_i + v_i(1-2r_i)$ with $u_i, v_i \in [0,1]$. This parameterization inherently satisfies all boundary constraints ($r_i \le x_i \le 1-r_i$, etc.), reducing the optimization to only pairwise non-overlap constraints. We then run `SLSQP` to fine-tune all variables simultaneously, squeezing out the last fraction of a radius sum.

This combination leverages the global reach of stochastic search, the exactness of LP for radii assignment, and the gradient-based precision of SLSQP for final convergence.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

# Precompute constant parts of LP constraint matrix for efficiency
A_ub_const = np.zeros((4*N + N*(N-1)//2, N))
c_obj = -np.ones(N)
bounds_lp = [(0.0, None)] * N

# Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
for i in range(N):
    A_ub_const[4*i, i] = 1.0
    A_ub_const[4*i+1, i] = 1.0
    A_ub_const[4*i+2, i] = 1.0
    A_ub_const[4*i+3, i] = 1.0

# Pairwise constraints: r_i + r_j <= dist(i, j)
for idx, (i, j) in enumerate(zip(I_IDX, J_IDX)):
    A_ub_const[4*N + idx, i] = 1.0
    A_ub_const[4*N + idx, j] = 1.0

def eval_sum(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    b_ub = np.empty(4*n + n*(n-1)//2)
    
    # Fill boundary RHS
    for i in range(n):
        x, y = centers[i]
        b_ub[4*i] = x
        b_ub[4*i+1] = 1.0 - x
        b_ub[4*i+2] = y
        b_ub[4*i+3] = 1.0 - y
        
    # Fill pairwise distance RHS
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    b_ub[4*n:] = dists[I_IDX, J_IDX]
    
    res = linprog(c_obj, A_ub=A_ub_const, b_ub=b_ub, bounds=bounds_lp, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 0.0

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice variations
    for s in range(10):
        pts = []
        r_est = 0.095 + rng.uniform(-0.01, 0.01)
        y = r_est
        row = 0
        while len(pts) < N:
            shift = (row % 2) * r_est
            x = r_est + shift
            while x <= 1.0 - r_est and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
        pts = np.array(pts[:N])
        
        rot = rng.uniform(-0.2, 0.2)
        sc = rng.uniform(0.95, 1.05)
        pts = (pts - 0.5) * sc + 0.5
        
        if rot != 0.0:
            c, s_val = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[c, -s_val], [s_val, c]])
            pts = pts - pts.mean(axis=0) + 0.5
            
        pts += rng.uniform(-0.02, 0.02, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    # 2. Force-directed layouts
    for s in range(10):
        pts = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            
            for d in range(2):
                forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
                forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
                
            pts += 0.005 * forces
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    return inits

def obj_slqp(x):
    """SLSQP objective: minimize negative sum of radii."""
    return -np.sum(x[0::3])

def cons_slqp(x):
    """SLSQP constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r_s = x[0::3]
    u_s = x[1::3]
    v_s = x[2::3]
    
    x_c = r_s + u_s * (1.0 - 2.0 * r_s)
    y_c = r_s + v_s * (1.0 - 2.0 * r_s)
    
    dx = x_c[:, None] - x_c[None, :]
    dy = y_c[:, None] - y_c[None, :]
    dist2 = dx**2 + dy**2
    
    rs = r_s[:, None] + r_s[None, :]
    return dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def run_packing():
    rng = np.random.default_rng(42)
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    inits = generate_inits(rng)
    
    # Phase 1: Simulated Annealing on centers with LP radius evaluation
    for init_c in inits:
        centers = init_c.copy()
        radii, curr_sum = eval_sum(centers)
        
        temp = 0.005
        step = 0.04
        min_step = 1e-5
        
        for it in range(8000):
            # 20% chance to move a pair of circles together (preserves local clustering)
            if rng.random() < 0.2:
                idx1, idx2 = rng.choice(N, 2, replace=False)
                old = [centers[idx1].copy(), centers[idx2].copy()]
                direction = rng.normal(0, 1, 2)
                direction /= np.linalg.norm(direction) + 1e-12
                centers[idx1] += step * direction
                centers[idx2] += step * direction
                centers[idx1] = np.clip(centers[idx1], 0.005, 0.995)
                centers[idx2] = np.clip(centers[idx2], 0.005, 0.995)
                modified = [idx1, idx2]
            else:
                idx = rng.integers(N)
                old = [centers[idx].copy()]
                direction = rng.normal(0, 1, 2)
                direction /= np.linalg.norm(direction) + 1e-12
                centers[idx] += step * direction
                centers[idx] = np.clip(centers[idx], 0.005, 0.995)
                modified = [idx]
                
            r_new, sum_new = eval_sum(centers)
            delta = sum_new - curr_sum
            
            # Metropolis criterion with overflow protection
            if delta > 0 or (temp > 1e-7 and rng.random() < np.exp(min(500.0, delta / temp))):
                curr_sum = sum_new
                radii = r_new
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()
                temp *= 0.9995
                step = max(min_step, step * 0.999)
            else:
                for idx, pt in zip(modified, old):
                    centers[idx] = pt
                temp *= 0.9995
                step = max(min_step, step * 0.999)
                
    # Phase 2: High-precision SLSQP polish using parameterized variables
    if best_centers is not None:
        r = best_radii.copy()
        denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
        u = np.clip((best_centers[:, 0] - r) / denom, 0.0, 1.0)
        v = np.clip((best_centers[:, 1] - r) / denom, 0.0, 1.0)
        
        x0 = np.empty(3 * N)
        x0[0::3] = r
        x0[1::3] = u
        x0[2::3] = v
        
        bounds_slqp = [(1e-6, 0.5), (0.0, 1.0), (0.0, 1.0)] * N
        
        try:
            res = minimize(obj_slqp, x0, method='SLSQP', bounds=bounds_slqp,
                           constraints={'type': 'ineq', 'fun': cons_slqp},
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    r_opt = res.x[0::3]
                    u_opt = res.x[1::3]
                    v_opt = res.x[2::3]
                    best_centers = np.column_stack((r_opt + u_opt*(1-2*r_opt), r_opt + v_opt*(1-2*r_opt)))
                    best_radii = r_opt
        except Exception:
            pass
            
    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = np.random.rand(N, 2)
        best_radii = np.full(N, 0.01)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
