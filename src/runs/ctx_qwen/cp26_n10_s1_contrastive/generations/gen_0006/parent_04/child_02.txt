import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    xs = centers[:, 0]
    ys = centers[:, 1]
    
    for i in range(n):
        A_ub[4*i, i] = 1.0; b_ub[4*i] = xs[i]
        A_ub[4*i+1, i] = 1.0; b_ub[4*i+1] = 1.0 - xs[i]
        A_ub[4*i+2, i] = 1.0; b_ub[4*i+2] = ys[i]
        A_ub[4*i+3, i] = 1.0; b_ub[4*i+3] = 1.0 - ys[i]
        
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    k = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0; A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 0.0

def objective_slsqp(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[0::3])

def constraints_slsqp(v):
    """Inequality constraints: dist^2 >= (r_i + r_j)^2."""
    r = v[0::3]
    u = v[1::3]
    t = v[2::3]
    
    # Parameterization guarantees boundary containment
    x = r + u * (1.0 - 2.0 * r)
    y = r + t * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    return dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX]

def make_params(centers, radii):
    """Map physical centers/radii to (r, u, t) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    t = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = t
    return params

def hex_start(rng, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_start(seed):
    """Generates a tightly packed initial configuration using force-directed layout."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.rand(N, 2)
    r_curr = np.full(N, 0.05)
    for step in range(500):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
        step_size = 0.006 * (0.995**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        for i in range(N):
            dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            dp = max(dp, 1e-6)
            r_curr[i] = 0.85 * min(dw, dp / 2.0)
    return pts, r_curr

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_vars = None
    best_sum = -np.inf
    
    inits = []
    # 1. Hexagonal lattice variations
    for s in range(30):
        rot = rng.uniform(-0.4, 0.4)
        sc = rng.uniform(0.80, 1.20)
        pts = hex_start(rng, rot, sc)
        r_init, _ = solve_radii_lp(pts)
        r_init = np.maximum(r_init * 0.99, 1e-5)
        inits.append(make_params(pts, r_init))
        
    # 2. Force-directed layouts
    for s in range(20):
        pts, r_init = force_start(s)
        r_init = np.maximum(r_init * 0.99, 1e-5)
        inits.append(make_params(pts, r_init))
        
    # 3. Perturbed grid layouts
    for s in range(10):
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng.uniform(-0.03, 0.03, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        r_init, _ = solve_radii_lp(pts)
        r_init = np.maximum(r_init * 0.99, 1e-5)
        inits.append(make_params(pts, r_init))
        
    # Phase 1: Broad SLSQP search from diverse initializations
    for x0 in inits:
        try:
            res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                if np.min(constraints_slsqp(res.x)) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: LP-Guided Coordinate Ascent on Centers
    # This efficiently searches for better topologies by optimizing radii exactly via LP
    if best_vars is not None:
        r_opt = best_vars[0::3]
        u_opt = best_vars[1::3]
        t_opt = best_vars[2::3]
        r_safe = np.clip(r_opt, 1e-6, 0.4999)
        centers = np.column_stack((r_safe + u_opt*(1-2*r_safe), r_safe + t_opt*(1-2*r_safe)))
        radii = r_opt
        
        current_sum = best_sum
        step = 0.02
        for it in range(4000):
            idx = rng.integers(N)
            direction = rng.standard_normal(2)
            direction /= np.linalg.norm(direction)
            
            new_centers = centers.copy()
            new_centers[idx] = np.clip(new_centers[idx] + step * direction, 0.01, 0.99)
            
            _, new_sum = solve_radii_lp(new_centers)
            
            if new_sum > current_sum + 1e-7:
                centers = new_centers
                current_sum = new_sum
                radii, _ = solve_radii_lp(centers)
                step = max(0.001, step * 0.95)
            else:
                step = min(0.05, step * 1.02)
                
        best_vars = make_params(centers, radii * 0.999)
        best_sum = current_sum
        
    # Phase 3: Perturbation & Re-optimization (Basin Hopping)
    if best_vars is not None:
        for k in range(30):
            x0 = best_vars.copy()
            x0 += rng.normal(0, 0.005, 3*N)
            x0[0::3] = np.clip(x0[0::3], 1e-6, 0.49)
            x0[1::3] = np.clip(x0[1::3], 0.0, 1.0)
            x0[2::3] = np.clip(x0[2::3], 0.0, 1.0)
            
            try:
                res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if res.success and np.min(constraints_slsqp(res.x)) >= -1e-7:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_vars = res.x.copy()
            except Exception:
                continue
                
        # Phase 4: High-precision Final Polish
        try:
            res = minimize(objective_slsqp, best_vars, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res.success and np.min(constraints_slsqp(res.x)) >= -1e-8:
                best_vars = res.x
                best_sum = -res.fun
        except Exception:
            pass

    # Fallback configuration
    if best_vars is None:
        pts = hex_start(rng)
        r_f, _ = solve_radii_lp(pts)
        best_vars = make_params(pts, r_f)
        best_sum = np.sum(r_f)
        
    # Reconstruct final centers and radii
    r_opt = best_vars[0::3]
    u_opt = best_vars[1::3]
    t_opt = best_vars[2::3]
    r_safe = np.clip(r_opt, 1e-6, 0.4999)
    x_opt = r_safe + u_opt * (1.0 - 2.0 * r_safe)
    y_opt = r_safe + t_opt * (1.0 - 2.0 * r_safe)
    centers = np.column_stack((x_opt, y_opt))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
