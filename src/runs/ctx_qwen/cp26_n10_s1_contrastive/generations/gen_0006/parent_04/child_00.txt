import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    m = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    k = 0
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 0.0

def hex_start(seed, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization with rotation, scaling, and jitter."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.1
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def force_start(seed):
    """Generates a tightly packed initial configuration using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    r_curr = np.full(N, 0.05)
    
    for step in range(400):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep_mag = 1.0 / dists**2
        np.fill_diagonal(rep_mag, 0.0)
        forces = np.sum(rep_mag[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 15.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 15.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.006 * (0.997**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        for i in range(N):
            d_wall = min(pts[i,0], 1.0 - pts[i,0], pts[i,1], 1.0 - pts[i,1])
            d_pair = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            d_pair = max(d_pair, 1e-6)
            r_curr[i] = 0.85 * min(d_wall, d_pair / 2.0)
            
    return pts

def objective_slqp(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[0::3])

def constraints_slqp(v):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundaries handled by parameterization."""
    r = v[0::3]
    u = v[1::3]
    w = v[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + w * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, w) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    w = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, w])

def run_packing():

    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations

    # Phase 1: LP-based Simulated Annealing on Centers
    for start_c in starts:

        best_lp_sum = curr_sum
        best_lp_c = centers.copy()
        best_lp_r = radii.copy()
        
        temp = 0.025
        step = 0.025
        for it in range(2500):
            op = rng.choice(['move', 'swap'])
            if op == 'move':
                idx = rng.integers(N)
                old = centers[idx].copy()
                centers[idx] += rng.normal(0, step, 2)
                centers[idx] = np.clip(centers[idx], 0.005, 0.995)
            else:
                i, j = rng.choice(N, 2, replace=False)
                centers[i], centers[j] = centers[j].copy(), centers[i].copy()
                
            r_new, sum_new = solve_radii_lp(centers)
            delta = sum_new - curr_sum
            
            if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(delta / temp)):
                curr_sum = sum_new
                if sum_new > best_lp_sum:
                    best_lp_sum = sum_new
                    best_lp_c = centers.copy()
                    best_lp_r = r_new.copy()
            else:
                if op == 'move':
                    centers[idx] = old
                    
            temp *= 0.9993
            step = max(0.001, step * 0.9993)
            
        if best_lp_sum > best_sum:
            best_sum = best_lp_sum
            best_c = best_lp_c
            best_r = best_lp_r
            
    # Phase 2: SLSQP Refinement with Parameterization
    if best_c is not None:
        v0 = to_params(best_c, best_r * 0.995)
        bounds = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
        cons = {'type': 'ineq', 'fun': constraints_slqp}
        
        # Multiple refinement passes to ensure convergence
        for _ in range(5):
            try:
                res = minimize(objective_slqp, v0, method='SLSQP', bounds=bounds, 
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(constraints_slqp(res.x)) >= -1e-8:

                pass

        r_opt = v0[0::3]
        u_opt = v0[1::3]
        w_opt = v0[2::3]
        x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
        y_opt = r_opt + w_opt * (1.0 - 2.0 * r_opt)
        best_c = np.column_stack((x_opt, y_opt))

            v_pert[0::3] = np.clip(v_pert[0::3], 1e-6, 0.5)
            v_pert[1::3] = np.clip(v_pert[1::3], 0.0, 1.0)
            v_pert[2::3] = np.clip(v_pert[2::3], 0.0, 1.0)
            
            try:
                res_p = minimize(objective_slqp, v_pert, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                if res_p.success and np.min(constraints_slqp(res_p.x)) >= -1e-8:

                    if s_p > best_sum:

        r_opt = v0[0::3]
        u_opt = v0[1::3]
        w_opt = v0[2::3]

        best_c = hex_start(0)
        best_r, best_sum = solve_radii_lp(best_c)

    return best_c, best_r, float(best_sum)
