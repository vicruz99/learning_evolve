import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-4

def make_vars_from_pts(pts, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((pts[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((pts[:, 1] - r) / denom, 0.0, 1.0)
    vars0 = np.empty(3 * N)
    vars0[0::3] = r
    vars0[1::3] = u
    vars0[2::3] = v
    return vars0

def objective(vars_vec):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_vec[0::3])

def constraint_func(vars_vec):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundaries handled by parameterization."""
    r = vars_vec[0::3]
    u = vars_vec[1::3]
    v = vars_vec[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    return dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX]

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice variations
    for s in range(8):
        pts = []
        r_est = 0.095
        y = r_est
        row = 0
        while len(pts) < N:
            shift = (row % 2) * r_est
            x = r_est + shift
            while x <= 1.0 - r_est and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
        pts = np.array(pts[:N])
        pts -= 0.5
        pts *= rng.uniform(0.90, 1.10)
        pts += 0.5
        rot = rng.uniform(-0.25, 0.25)
        c, s_val = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s_val], [s_val, c]])
        pts += rng.uniform(-0.015, 0.015, (N, 2))
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    # 2. Force-directed layouts
    for s in range(6):
        pts = rng.uniform(0.1, 0.9, (N, 2))
        r_curr = np.full(N, 0.05)
        for step in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            dists = np.maximum(dists, 1e-8)
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 5.0 * np.maximum(0, r_curr - pts[:, d])
                forces[:, d] -= 5.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            pts += 0.005 * forces * (1.0 - step / 300.0)
            pts = np.clip(pts, 0.01, 0.99)
            for i in range(N):
                dw = min(pts[i,0], 1.0 - pts[i,0], pts[i,1], 1.0 - pts[i,1])
                dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
                r_curr[i] = 0.8 * min(dw, dp / 2.0)
        inits.append(pts)
        
    # 3. Perturbed grid
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-0.02, 0.02, (N, 2))
    pts = np.clip(pts, 0.05, 0.95)
    inits.append(pts)
    
    return inits

def local_search_lp(centers, rng, max_iters=20000):
    """Simulated Annealing on centers with LP radii optimization."""
    n = centers.shape[0]
    radii, curr_sum = solve_radii_lp(centers)
    best_c = centers.copy()
    best_r = radii.copy()
    best_sum = curr_sum
    
    step = 0.035
    temp = 0.06
    
    for it in range(max_iters):
        if rng.random() < 0.80:
            op = 'move'
            idx = rng.integers(n)
            old = centers[idx].copy()
            centers[idx] += rng.normal(0, step, 2)
            centers[idx] = np.clip(centers[idx], 0.002, 0.998)
        else:
            op = 'swap'
            i, j = rng.choice(n, 2, replace=False)
            centers[i], centers[j] = centers[j].copy(), centers[i].copy()
            
        _, new_sum = solve_radii_lp(centers)
        delta = new_sum - curr_sum
        
        accept = delta > 0 or (temp > 1e-6 and rng.rand() < np.exp(delta / temp))
        if accept:
            curr_sum = new_sum
            if curr_sum > best_sum:
                best_c = centers.copy()
                _, best_r = solve_radii_lp(best_c)
                best_sum = curr_sum
        else:
            if op == 'move':
                centers[idx] = old
            else:
                centers[i], centers[j] = centers[j].copy(), centers[i].copy()
                
        temp *= 0.9993
        step = max(0.0015, step * 0.9995)
        
    return best_c, best_r, best_sum

def run_packing():
    rng = np.random.default_rng(42)
    best_c, best_r, best_sum = None, None, -np.inf
    
    inits = generate_inits(rng)
    
    # Phase 1 & 2: LP-driven Simulated Annealing on diverse starts
    for pts in inits:
        c, r, s = local_search_lp(pts, rng, max_iters=25000)
        if s > best_sum:
            best_c, best_r, best_sum = c, r, s
            
    # Phase 3: SLSQP High-Precision Polish
    bounds = [(1e-6, 0.5), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Shrink radii slightly to ensure strict interior feasibility for SLSQP
    r_polish = best_r * 0.995
    v0 = make_vars_from_pts(best_c, r_polish)
    
    try:
        res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(constraint_func(res.x)) >= -1e-7:
            s_polish = -res.fun
            if s_polish > best_sum:
                best_sum = s_polish
                best_vars = res.x
            else:
                best_vars = make_vars_from_pts(best_c, best_r)
        else:
            best_vars = make_vars_from_pts(best_c, best_r)
    except Exception:
        best_vars = make_vars_from_pts(best_c, best_r)
        
    # Reconstruct final centers and radii
    r_opt = best_vars[0::3]
    u_opt = best_vars[1::3]
    v_opt = best_vars[2::3]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    return centers, r_opt, float(best_sum)
