import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-5

def objective_slsqp(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints_slsqp(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def hex_init(rng, rot, scale):
    """Generates a hexagonal lattice initialization with rotation, scaling, and jitter."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generates a tightly packed initial configuration using force-directed layout."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    r_curr = np.full(N, 0.06)
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 20.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.008 * (0.995**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists_to_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_to_others[i] = np.inf
            d_pair = np.min(dists_to_others)
            r_curr[i] = 0.8 * min(d_wall, d_pair/2.0)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_vars = None
    best_sum = -np.inf
    
    # Generate diverse initial center configurations
    starts = []
    for s in range(10):
        rot = rng.uniform(-0.3, 0.3)
        sc = rng.uniform(0.85, 1.15)
        starts.append(hex_init(rng, rot, sc))
        
    for s in range(5):
        starts.append(force_init(rng))
        
    for s in range(5):
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng.uniform(-0.03, 0.03, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        starts.append(pts)
        
    # Phase 1: SLSQP from LP-initialized starts
    # LP gives exact optimal radii for fixed centers, providing a high-quality starting point
    for pts in starts:
        r_init, _ = solve_lp_radii(pts)
        # Slightly shrink radii to ensure strict interior feasibility for SLSQP start
        r_init = np.maximum(r_init * 0.99, 1e-5)
        
        v0 = np.zeros(3 * N)
        v0[0::3] = pts[:, 0]
        v0[1::3] = pts[:, 1]
        v0[2::3] = r_init
        
        try:
            res = minimize(objective_slsqp, v0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints_slsqp(res.x)) >= -1e-8:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation refinement to escape local minima
    if best_vars is not None:
        for k in range(30):
            v_p = best_vars.copy()
            v_p += rng.normal(0, 0.003, 3*N)
            v_p[0::3] = np.clip(v_p[0::3], 0.01, 0.99)
            v_p[1::3] = np.clip(v_p[1::3], 0.01, 0.99)
            v_p[2::3] = np.clip(v_p[2::3], 1e-5, 0.49)
            
            try:
                res_p = minimize(objective_slsqp, v_p, method='SLSQP', bounds=bounds_slqp,
                                 constraints=cons_slqp, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                if res_p.success and np.min(constraints_slsqp(res_p.x)) >= -1e-8:
                    s_p = -res_p.fun
                    if s_p > best_sum:
                        best_sum = s_p
                        best_vars = res_p.x.copy()
            except Exception:
                continue
                
        # Phase 3: High-precision polish
        try:
            res_f = minimize(objective_slsqp, best_vars, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons_slqp, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(constraints_slsqp(res_f.x)) >= -1e-9:
                best_vars = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Fallback configuration (should rarely be reached)
    if best_vars is None:
        pts = hex_init(rng, 0.0, 1.0)
        r_f, _ = solve_lp_radii(pts)
        best_vars = np.zeros(3*N)
        best_vars[0::3] = pts[:, 0]
        best_vars[1::3] = pts[:, 1]
        best_vars[2::3] = np.maximum(r_f, 1e-5)
        best_sum = np.sum(best_vars[2::3])
        
    # Reconstruct centers and radii
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    return centers, radii, float(best_sum)
