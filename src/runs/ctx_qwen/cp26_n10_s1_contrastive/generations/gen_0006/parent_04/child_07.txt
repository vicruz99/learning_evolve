import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    
    return np.full(n, 1e-5), 2.6e-4

def objective_ruv(p):
    """Objective for (r, u, v) parameterization."""
    return -np.sum(p[0::3])

def constraint_ruv(p):
    """Pairwise non-overlap constraint for (r, u, v) parameterization."""
    r = p[0::3]
    u = p[1::3]
    v = p[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def objective_xyz(v):
    """Objective for (x, y, r) parameterization."""
    return -np.sum(v[2::3])

def constraint_xyz(v):
    """Constraints for (x, y, r) parameterization."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def params_to_centers(r, u, v):
    """Convert (r, u, v) parameters to physical centers."""
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y))

def centers_to_params(centers, radii):
    """Convert physical centers and radii to (r, u, v) parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def make_hex_init(seed, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5
    
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force_init(seed, steps=700):
    """Generates tightly packed initial configuration using force-directed layout."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2)
    r_curr = np.full(N, 0.05)
    
    for step in range(steps):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        dists = np.maximum(dists, 1e-8)
        rep_mag = 1.0 / dists**2
        np.fill_diagonal(rep_mag, 0.0)
        forces = np.sum(rep_mag[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        
        for d in range(2):
            forces[:, d] += 25.0 * np.maximum(0, 0.05 - pts[:, d])
            forces[:, d] -= 25.0 * np.maximum(0, pts[:, d] - 0.95)
        
        step_size = 0.008 * (0.997**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        for i in range(N):
            d_wall = min(pts[i,0], 1.0 - pts[i,0], pts[i,1], 1.0 - pts[i,1])
            dists_to_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_to_others[i] = np.inf
            d_pair = np.min(dists_to_others)
            r_curr[i] = 0.9 * min(d_wall, d_pair / 2.0)
    
    return pts, r_curr

def make_grid_init(seed):
    """Generates a perturbed grid initialization."""
    rng = np.random.RandomState(seed)
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-0.03, 0.03, (N, 2))
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def make_corner_init(seed):
    """Generates an initialization focused on corners and edges."""
    rng = np.random.RandomState(seed)
    pts = []
    
    # 4 corners
    r_corner = 0.12
    pts.append([r_corner, r_corner])
    pts.append([1.0 - r_corner, r_corner])
    pts.append([r_corner, 1.0 - r_corner])
    pts.append([1.0 - r_corner, 1.0 - r_corner])
    
    # Edge centers
    r_edge = 0.09
    pts.append([0.5, r_edge])
    pts.append([0.5, 1.0 - r_edge])
    pts.append([r_edge, 0.5])
    pts.append([1.0 - r_edge, 0.5])
    
    # Fill rest with hex-like pattern
    r_fill = 0.085
    y = r_fill * 3
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_fill
        x = r_fill + shift
        while x <= 1.0 - r_fill and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_fill
        y += np.sqrt(3.0) * r_fill
        row += 1
    
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    rng = np.random.RandomState(42)
    
    bounds_ruv = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    bounds_xyz = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_ruv = {'type': 'ineq', 'fun': constraint_ruv}
    cons_xyz = {'type': 'ineq', 'fun': constraint_xyz}
    
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    
    # Hexagonal lattice variations
    for s in range(35):
        rot = rng.uniform(-0.35, 0.35)
        sc = rng.uniform(0.80, 1.20)
        inits.append(('hex', make_hex_init(s, rot, sc)))
    
    # Force-directed layouts
    for s in range(25):
        pts, r = make_force_init(s)
        inits.append(('force', pts))
    
    # Perturbed grids
    for s in range(15):
        inits.append(('grid', make_grid_init(s)))
    
    # Corner-focused layouts
    for s in range(10):
        inits.append(('corner', make_corner_init(s)))
    
    # Phase 2: SLSQP optimization from each configuration
    for cfg_type, pts in inits:
        r_init, _ = solve_lp(pts)
        r_init = np.maximum(r_init * 0.99, 1e-6)
        
        # Try (r, u, v) parameterization
        v0_ruv = centers_to_params(pts, r_init)
        try:
            res = minimize(objective_ruv, v0_ruv, method='SLSQP', bounds=bounds_ruv,
                           constraints=cons_ruv, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                cons_val = constraint_ruv(res.x)
                if np.min(cons_val) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        r_opt = res.x[0::3]
                        u_opt = res.x[1::3]
                        v_opt = res.x[2::3]
                        best_c = params_to_centers(r_opt, u_opt, v_opt)
                        best_r = r_opt
        except Exception:
            pass
        
        # Try (x, y, r) parameterization
        v0_xyz = np.zeros(3 * N)
        v0_xyz[0::3] = pts[:, 0]
        v0_xyz[1::3] = pts[:, 1]
        v0_xyz[2::3] = r_init
        try:
            res = minimize(objective_xyz, v0_xyz, method='SLSQP', bounds=bounds_xyz,
                           constraints=cons_xyz, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                cons_val = constraint_xyz(res.x)
                if np.min(cons_val) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_r = res.x[2::3]
        except Exception:
            pass
    
    # Phase 3: LP-based simulated annealing on centers
    if best_c is not None:
        centers = best_c.copy()
        _, curr_sum = solve_lp(centers)
        best_lp_c = centers.copy()
        best_lp_sum = curr_sum
        
        temp = 0.025
        step = 0.022
        
        for it in range(12000):
            op = rng.choice(['move', 'swap'])
            if op == 'move':
                idx = rng.randint(N)
                old = centers[idx].copy()
                centers[idx] += rng.normal(0, step, 2)
                centers[idx] = np.clip(centers[idx], 0.005, 0.995)
            else:
                i, j = rng.choice(N, 2, replace=False)
                centers[i], centers[j] = centers[j].copy(), centers[i].copy()
            
            r_new, sum_new = solve_lp(centers)
            delta = sum_new - curr_sum
            
            if delta > 0 or (temp > 1e-8 and rng.rand() < np.exp(delta / temp)):
                curr_sum = sum_new
                if sum_new > best_lp_sum:
                    best_lp_sum = sum_new
                    best_lp_c = centers.copy()
            else:
                if op == 'move':
                    centers[idx] = old
            
            temp *= 0.9993
            step = max(0.001, step * 0.9993)
        
        if best_lp_sum > best_sum:
            best_sum = best_lp_sum
            best_c = best_lp_c
            _, best_r = solve_lp(best_c)
    
    # Phase 4: Rotation-based escape and re-optimization
    if best_c is not None:
        for k in range(20):
            rot = rng.uniform(-0.15, 0.15)
            c_val, s_val = np.cos(rot), np.sin(rot)
            mat = np.array([[c_val, -s_val], [s_val, c_val]])
            c_rot = (best_c - 0.5) @ mat.T + 0.5
            c_rot = np.clip(c_rot, 0.01, 0.99)
            
            r_rot, _ = solve_lp(c_rot)
            r_rot = np.maximum(r_rot * 0.995, 1e-6)
            
            # Try with (x, y, r) parameterization
            v0 = np.zeros(3 * N)
            v0[0::3] = c_rot[:, 0]
            v0[1::3] = c_rot[:, 1]
            v0[2::3] = r_rot
            
            try:
                res = minimize(objective_xyz, v0, method='SLSQP', bounds=bounds_xyz,
                               constraints=cons_xyz, options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                if res.success:
                    if np.min(constraint_xyz(res.x)) >= -1e-7:
                        s_val = -res.fun
                        if s_val > best_sum:
                            best_sum = s_val
                            best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                            best_r = res.x[2::3]
            except Exception:
                pass
    
    # Phase 5: Perturbation refinement
    if best_c is not None:
        r_tmp = best_r.copy()
        u_tmp = np.clip((best_c[:, 0] - r_tmp) / np.clip(1.0 - 2.0 * r_tmp, 1e-6, 1.0), 0.0, 1.0)
        v_tmp = np.clip((best_c[:, 1] - r_tmp) / np.clip(1.0 - 2.0 * r_tmp, 1e-6, 1.0), 0.0, 1.0)
        best_vars_ruv = np.empty(3 * N)
        best_vars_ruv[0::3] = r_tmp
        best_vars_ruv[1::3] = u_tmp
        best_vars_ruv[2::3] = v_tmp

            x0 = best_vars_ruv.copy()

            x0[1::3] += rng.normal(0, 0.05, N)
            x0[2::3] += rng.normal(0, 0.05, N)
            
            x0[0::3] = np.clip(x0[0::3], 1e-6, 0.49)
            x0[1::3] = np.clip(x0[1::3], 0.0, 1.0)
            x0[2::3] = np.clip(x0[2::3], 0.0, 1.0)
            
            try:

                            best_sum = curr_sum
                            best_vars_ruv = res.x.copy()
            except Exception:
                continue
    
    # Phase 6: High-precision final polish
    if best_c is not None:

        r_tmp, _ = solve_lp(best_c)
        v0 = np.zeros(3 * N)
        v0[0::3] = best_c[:, 0]
        v0[1::3] = best_c[:, 1]

                        best_r = res.x[2::3]

            pass

                            u_opt = res.x[1::3]

                            best_r = r_opt
            except Exception:
                pass
    
    # Fallback
    if best_c is None:
        pts = make_hex_init(0)
        r_f, _ = solve_lp(pts)
        best_c = pts
        best_r = r_f
        best_sum = np.sum(r_f)
    
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
