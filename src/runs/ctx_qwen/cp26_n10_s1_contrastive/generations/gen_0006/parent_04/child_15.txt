from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    n_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_vals = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_vals:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 2.6e-4

def objective(params):
    """Minimize negative sum of radii."""
    return -np.sum(params[0::3])

def constraint_func(params):
    """Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def make_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    p = np.empty(3 * N)
    p[0::3] = r
    p[1::3] = u
    p[2::3] = v
    return p

def hex_pattern(rows_cfg, r_est, rng):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r_est
    for r_idx, cnt in enumerate(rows_cfg):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        
    pts = np.array(pts[:N])
    pts -= 0.5
    scale = rng.uniform(0.85, 1.15)
    pts *= scale
    pts += 0.5
    
    rot = rng.uniform(-0.25, 0.25)
    c_val, s_val = np.cos(rot), np.sin(rot)
    pts = pts @ np.array([[c_val, -s_val], [s_val, c_val]])
    pts -= pts.mean(axis=0)
    pts += 0.5
    
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):

    pts = rng.uniform(0.1, 0.9, (N, 2))

            forces[:, d] -= 15.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.008 * (0.997**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        for i in range(N):
            dw = min(pts[i,0], 1.0 - pts[i,0], pts[i,1], 1.0 - pts[i,1])

            dp = max(dp, 1e-6)
            r_curr[i] = 0.85 * min(dw, dp / 2.0)
            
    return pts
