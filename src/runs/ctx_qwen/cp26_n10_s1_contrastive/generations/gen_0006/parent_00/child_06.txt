import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    
    # Precompute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Build LP constraints: r_i + r_j <= dist_ij and r_i <= boundary
    n_cons = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_cons, n))
    b_ub = np.zeros(n_cons)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_i = np.array([x, 1.0 - x, y, 1.0 - y])
        for b in bounds_i:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-6), 1e-5

def objective(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def constraint_func(params):
    """
    Computes inequality constraints g(params) >= 0.
    Uses parameterization to automatically satisfy boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def centers_to_params(centers, radii):
    """Convert centers and radii to the (r, u, v) parameterization."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def generate_hex_init(rng, pattern=None):
    """Generates a hexagonal lattice initialization."""
    if pattern is None:
        pattern = [6, 5, 6, 5, 4]
        
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(pattern):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
        
    pts = np.array(pts[:N])
    pts -= 0.5
    scale = rng.uniform(0.85, 1.15)
    pts *= scale
    pts += 0.5
    
    rot = rng.uniform(-0.25, 0.25)
    c, s = np.cos(rot), np.sin(rot)
    pts = pts @ np.array([[c, -s], [s, c]])
    pts -= pts.mean(axis=0)
    pts += 0.5
    
    pts += rng.normal(0, 0.012, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def generate_force_init(rng):
    """Generates a force-directed initial configuration."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    r_curr = np.full(N, 0.06)
    
    for step in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 40.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 40.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.01 * (0.985**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists_o = np.linalg.norm(pts[i] - pts, axis=1)
            dists_o[i] = np.inf
            d_pair = np.min(dists_o)
            r_curr[i] = 0.75 * min(d_wall, d_pair/2.0)
            
    return pts

def lp_sa(centers_init, n_iter, rng):
    """LP-based Simulated Annealing to optimize centers."""
    centers = centers_init.copy()
    radii, curr_sum = solve_radii_lp(centers)
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = curr_sum
    
    temp = 0.015
    step = 0.025
    
    for i in range(n_iter):
        mt = rng.integers(4)
        old_c = centers.copy()
        
        if mt == 0:
            idx = rng.integers(N)
            centers[idx] += rng.normal(0, step, 2)
        elif mt == 1:
            idxs = rng.choice(N, 2, replace=False)
            centers[idxs] += rng.normal(0, step, (2, 2))
        elif mt == 2:
            ang = rng.uniform(-0.015, 0.015)
            c, s = np.cos(ang), np.sin(ang)
            centers = (centers - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        else:
            fac = rng.uniform(0.985, 1.015)
            centers = (centers - 0.5) * fac + 0.5
            
        centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
        
        radii_new, sum_new = solve_radii_lp(centers)
        
        delta = sum_new - curr_sum
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_sum = sum_new
            if sum_new > best_sum:
                best_sum = sum_new
                best_centers = centers.copy()
                best_radii = radii_new.copy()
        else:
            centers = old_c
            
        temp *= 0.9996
        step = max(0.0005, step * 0.9997)
        
    return best_centers, best_radii, best_sum

def greedy_coordinate_ascent(centers_init, n_iter, rng):
    """Deterministic greedy ascent to tighten packing."""
    centers = centers_init.copy()
    radii, curr_sum = solve_radii_lp(centers)
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = curr_sum
    step = 0.02
    
    for _ in range(n_iter):
        improved = False
        for i in range(N):
            old_c = centers.copy()
            for d in range(2):
                for sign in [1.0, -1.0]:
                    centers_copy = centers.copy()
                    centers_copy[i, d] += sign * step
                    centers_copy = np.clip(centers_copy, 1e-5, 1.0 - 1e-5)
                    _, s_new = solve_radii_lp(centers_copy)
                    if s_new > curr_sum:
                        centers = centers_copy
                        curr_sum = s_new
                        improved = True
            if not improved:
                centers = old_c
                
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            best_radii, _ = solve_radii_lp(centers)
        step = max(0.0002, step * 0.995)
        
    return best_centers, best_radii, best_sum

def run_packing():

            best_centers = c

        r_start = np.maximum(best_radii * 0.998, 1e-5)
        x0 = centers_to_params(best_centers, r_start)
        
        try:

                    v_opt = res.x[2::3]

        except Exception:
