import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to maximize sum of radii."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-5

def sa_local_search(centers, rng, steps=2500):
    """Simulated Annealing on centers optimizing LP-based sum of radii."""
    n = centers.shape[0]
    radii, curr_sum = solve_lp(centers)
    best_c = centers.copy()
    best_r = radii.copy()
    best_s = curr_sum
    
    temp = 0.012
    step = 0.02
    
    for it in range(steps):
        idx = rng.integers(n)
        old_pos = centers[idx].copy()
        
        centers[idx] += rng.normal(0, step, 2)
        centers[idx] = np.clip(centers[idx], 0.005, 0.995)
        
        r_new, s_new = solve_lp(centers)
        delta = s_new - best_s
        
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
            best_c = centers.copy()
            best_r = r_new.copy()
            best_s = s_new
        else:
            centers[idx] = old_pos
            
        temp *= 0.9992
        step = max(0.0008, step * 0.9995)
        
    return best_c, best_r, best_s

def make_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5],
        [5,5,5,5,6], [6,4,6,5,5], [5,6,4,6,5],
        [4,5,6,5,6], [6,6,5,5,4], [5,5,6,5,5]
    ]
    
    for pat in patterns:
        for _ in range(4):
            pts = []
            r_est = 0.095
            y = r_est
            row = 0
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
                
            pts = np.array(pts[:N])
            pts -= 0.5
            scale = rng.uniform(0.85, 1.15)
            pts *= scale
            pts += 0.5
            
            rot = rng.uniform(-0.3, 0.3)
            c, s = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[c, -s], [s, c]])
            pts -= pts.mean(axis=0)
            pts += 0.5
            
            pts += rng.normal(0, 0.012, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    # 2. Force-directed layouts
    for seed in range(15):
        rfd = np.random.RandomState(seed)
        pts = rfd.rand(N, 2)
        rc = np.full(N, 0.06)
        
        for step in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            
            for d in range(2):
                forces[:, d] += 25.0 * np.maximum(0, rc - pts[:, d])
                forces[:, d] -= 25.0 * np.maximum(0, pts[:, d] - (1.0 - rc))
                
            step_size = 0.006 * (1.0 - step / 300.0)
            pts += step_size * forces
            pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
            
            for i in range(N):
                dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
                dp = max(dp, 1e-6)
                rc[i] = 0.8 * min(dw, dp/2.0)
                
        inits.append(pts)
        
    # 3. Perturbed grids
    for seed in range(10):
        rng_g = np.random.RandomState(seed)
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng_g.uniform(-0.03, 0.03, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    return inits

def obj_slqp(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def con_slqp(v):
    """Constraints for SLSQP: boundaries and pairwise non-overlap."""
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary containment
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def run_packing():
    rng = np.random.default_rng(42)
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    inits = make_inits(rng)

            best_centers = c_best
            best_radii = r_best
            
    # Phase 2: Intensive refinement from best found
    if best_centers is not None:
        # Run multiple passes of SA with cooling schedule reset slightly to escape micro-local minima
        for _ in range(2):
            c_best, r_best, s_best = sa_local_search(best_centers, rng, steps=3000)
            if s_best > best_sum:
                best_sum = s_best
                best_centers = c_best
                best_radii = r_best
                
        # Phase 3: High-precision SLSQP polish
        v0 = np.zeros(3 * N)
        v0[0::3] = best_centers[:, 0]
        v0[1::3] = best_centers[:, 1]
        v0[2::3] = np.maximum(best_radii * 0.9995, 1e-6)
        
        bounds_slqp = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

        try:

            if res.success and np.min(con_slqp(res.x)) >= -1e-9:
                s_polished = -res.fun
                if s_polished > best_sum:
                    best_sum = s_polished
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))

    if best_centers is None:

        r_f = np.full(N, 0.09)
        best_centers = centers_f
        best_radii = r_f
        best_sum = np.sum(r_f)
        
    return best_centers, best_radii, float(best_sum)
