import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-5

def objective_slsqp(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def constraint_func_slsqp(params):
    """
    Computes inequality constraints g(params) >= 0.
    Uses parameterization to automatically satisfy boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def centers_to_params(centers, radii):
    """Convert centers and radii to the (r, u, v) parameterization."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def generate_hex_configs(rng):
    """Generates diverse hexagonal lattice initializations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5],
        [5,5,5,5,6], [6,4,6,5,5], [5,6,4,6,5],
        [4,5,6,5,6], [6,6,5,5,4], [5,5,6,5,5],
        [6,5,5,5,5], [5,5,5,6,5], [7,5,5,5,4],
        [4,5,5,5,7], [5,7,5,5,4], [4,5,5,7,5],
        [5,5,4,6,6], [4,6,6,5,5], [5,6,6,5,4]
    ]
    
    for pat in patterns:
        for _ in range(6):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
                
            pts = np.array(pts[:N])
            pts -= 0.5
            scale = rng.uniform(0.80, 1.20)
            pts *= scale
            pts += 0.5
            
            rot = rng.uniform(-0.35, 0.35)
            c_val, s_val = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[c_val, -s_val], [s_val, c_val]])
            pts -= pts.mean(axis=0)
            pts += 0.5
            
            pts += rng.normal(0, 0.015, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    # Force-directed layouts
    for seed in range(25):
        rng_fd = np.random.RandomState(seed)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        r_curr = np.full(N, 0.06)
        
        for step in range(500):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            
            for d in range(2):
                forces[:, d] += 30.0 * np.maximum(0, r_curr - pts[:, d])
                forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
                
            step_size = 0.008 * (0.990**step)
            pts += step_size * forces
            pts = np.clip(pts, 0.001, 0.999)
            
            for i in range(N):
                d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                dists_to_others = np.linalg.norm(pts[i] - pts, axis=1)
                dists_to_others[i] = np.inf
                d_pair = np.min(dists_to_others)
                r_curr[i] = 0.8 * min(d_wall, d_pair/2.0)
                
        inits.append(pts)
        
    return inits

def local_search_centers(centers, rng, max_iters=2500):
    """Simulated annealing on centers optimizing LP-computed sum of radii."""
    n = centers.shape[0]
    best_c = centers.copy()
    _, best_sum = solve_radii_lp(best_c)
    curr_c = best_c.copy()
    curr_sum = best_sum
    
    temp = 0.045
    step = 0.035
    
    for _ in range(max_iters):
        op = rng.integers(2)
        if op == 0:  # move
            idx = rng.integers(n)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
        else:  # swap
            i, j = rng.choice(n, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
            
        _, new_sum = solve_radii_lp(curr_c)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(delta / temp)):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = curr_c.copy()
        else:
            if op == 0:
                curr_c[idx] = old
                
        temp *= 0.9991
        step *= 0.9994
        
    return best_c, best_sum

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slsqp = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons_slsqp = {'type': 'ineq', 'fun': constraint_func_slsqp}
    
    best_vars = None

            best_c_sa = c_opt

        try:
