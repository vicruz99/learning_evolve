import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to find radii maximizing sum(r_i) for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 1e-5

def obj_func(p):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(p[0::3])

def cons_func(p):
    """Constraints for SLSQP: pairwise non-overlap (boundaries handled by parameterization)."""
    r = p[0::3]
    u = p[1::3]
    v = p[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def run_packing():
    rng = np.random.default_rng(42)
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    inits = []
    
    # 1. Hexagonal lattice variations
    for _ in range(12):
        pts = []
        r_est = 0.095
        y = r_est
        row = 0
        while len(pts) < N:
            shift = (row % 2) * r_est
            x = r_est + shift
            while x <= 1.0 - r_est and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3.0)
            row += 1
        pts = np.array(pts[:N])
        pts = (pts - 0.5) * rng.uniform(0.80, 1.20) + 0.5
        rot = rng.uniform(-0.25, 0.25)
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        pts += rng.normal(0, 0.015, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # 2. Force-directed layouts
    for seed in range(15):
        rng_fd = np.random.RandomState(seed)
        pts = rng_fd.uniform(0.1, 0.9, (N, 2))
        r_c = np.full(N, 0.06)
        for _ in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
                forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
            pts += 0.005 * forces
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    # 3. Perturbed grids
    for seed in range(10):
        rng_g = np.random.RandomState(seed)
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng_g.uniform(-0.04, 0.04, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # Phase 1: Simulated Annealing on centers with LP evaluation
    for pts in inits:
        centers = pts.copy()
        radii, curr_sum = solve_lp(centers)
        local_best_sum = curr_sum
        local_best_c = centers.copy()
        local_best_r = radii.copy()
        
        temp = 0.025
        step_size = 0.025
        
        for _ in range(5000):
            idx = rng.integers(N)
            old_pos = centers[idx].copy()
            
            centers[idx] += rng.normal(0, step_size, 2)
            centers[idx] = np.clip(centers[idx], 0.002, 0.998)
            
            radii_new, sum_new = solve_lp(centers)
            delta = sum_new - curr_sum
            
            if delta > 0 or rng.random() < np.exp(min(0, delta / max(temp, 1e-9))):
                curr_sum = sum_new
                if sum_new > local_best_sum:
                    local_best_sum = sum_new
                    local_best_c = centers.copy()
                    local_best_r = radii_new.copy()
            else:
                centers[idx] = old_pos
                
            temp *= 0.9995
            step_size = max(0.001, step_size * 0.9996)
            
        if local_best_sum > best_sum:
            best_sum = local_best_sum
            best_c = local_best_c
            best_r = local_best_r
            
    # Phase 2: SLSQP Polish on parameterized space
    if best_c is not None:
        r_opt = best_r.copy()
        denom = np.clip(1.0 - 2.0 * r_opt, 1e-6, 1.0)
        u_opt = np.clip((best_c[:, 0] - r_opt) / denom, 0.0, 1.0)
        v_opt = np.clip((best_c[:, 1] - r_opt) / denom, 0.0, 1.0)
        
        params = np.empty(3 * N)
        params[0::3] = r_opt * 0.995
        params[1::3] = u_opt
        params[2::3] = v_opt
        
        bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
        
        starts = [params.copy()]
        for _ in range(5):
            p_pert = params.copy()
            p_pert[1::3] += rng.normal(0, 0.03, N)
            p_pert[2::3] += rng.normal(0, 0.03, N)
            p_pert[0::3] += rng.normal(0, 0.001, N)
            p_pert[0::3] = np.clip(p_pert[0::3], 1e-6, 0.49)
            p_pert[1::3] = np.clip(p_pert[1::3], 0.0, 1.0)
            p_pert[2::3] = np.clip(p_pert[2::3], 0.0, 1.0)
            starts.append(p_pert)
            
        for p0 in starts:
            try:
                res = minimize(obj_func, p0, method='SLSQP', bounds=bounds,
                               constraints={'type': 'ineq', 'fun': cons_func},
                               options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if res.success and np.min(cons_func(res.x)) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        p_opt = res.x
                        r_f = p_opt[0::3]
                        u_f = p_opt[1::3]
                        v_f = p_opt[2::3]
                        x_f = r_f + u_f * (1.0 - 2.0 * r_f)
                        y_f = r_f + v_f * (1.0 - 2.0 * r_f)
                        best_c = np.column_stack((x_f, y_f))
                        best_r = r_f
            except Exception:
                continue
                
    # Fallback
    if best_c is None:
        best_c = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_r = np.full(N, 0.09)
        best_sum = np.sum(best_r)
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
