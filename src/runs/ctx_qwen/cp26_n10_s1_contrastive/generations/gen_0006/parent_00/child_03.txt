2. **SLSQP Joint Refinement**: Optimize centers and radii simultaneously using a parameterized formulation `(r, u, v)` that intrinsically satisfies boundary constraints. This allows smooth gradient-based descent while respecting geometry.

5. **High-Precision Final Polish**: A final SLSQP run with tight tolerances (`ftol=1e-14`) ensures numerical constraints are strictly met and extracts every last micro-unit of radius.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 1e-5

def objective(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def constraint_func(params):
    """
    Computes inequality constraints g(params) >= 0.
    Uses parameterization to automatically satisfy boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def centers_to_params(centers, radii):
    """Convert centers and radii to the (r, u, v) parameterization."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def params_to_centers(params):
    """Convert (r, u, v) parameters back to physical centers."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y))

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns with varied structures
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [7,5,5,5,4],
        [4,5,5,5,7], [5,7,5,5,4], [4,5,5,7,5], [6,5,6,4,5],
        [5,6,6,4,5], [4,6,6,5,5], [6,6,4,5,5], [5,5,7,5,4]
    ]
    
    for pat in patterns:
        for _ in range(4):
            pts = []
            r_est = 0.095
            y = r_est
            row = 0
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
                row += 1
                
            pts = np.array(pts[:N])
            pts -= 0.5
            scale = rng.uniform(0.85, 1.15)
            pts *= scale
            pts += 0.5
            
            rot = rng.uniform(-0.2, 0.2)
            c, s = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[c, -s], [s, c]])
            pts -= pts.mean(axis=0)
            pts += 0.5
            
            pts += rng.normal(0, 0.01, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    # 2. Force-directed layouts for organic packings
    for seed in range(15):
        rng_fd = np.random.RandomState(seed + 100)
        pts = rng_fd.uniform(0.1, 0.9, (N, 2))
        for _ in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            
            for d in range(2):
                forces[:, d] += 10.0 * np.maximum(0, 0.05 - pts[:, d])
                forces[:, d] -= 10.0 * np.maximum(0, pts[:, d] - 0.95)
                
            pts += 0.005 * forces
            pts = np.clip(pts, 0.01, 0.99)
        inits.append(pts)
        
    # 3. Perturbed grids
    for seed in range(10):
        rng_g = np.random.RandomState(seed + 200)
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng_g.uniform(-0.03, 0.03, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    return inits

def local_search_lp(centers, n_iter, rng):
    """Hill climbing on centers using LP objective evaluation."""
    best_c = centers.copy()
    _, best_s = solve_lp(best_c)
    step = 0.05
    
    for _ in range(n_iter):
        idx = rng.integers(N)
        direction = rng.standard_normal(2)
        direction /= np.linalg.norm(direction)
        
        new_c = best_c.copy()
        new_c[idx] = np.clip(best_c[idx] + step * direction, 1e-5, 1.0 - 1e-5)
        
        _, new_s = solve_lp(new_c)
        if new_s > best_s + 1e-9:
            best_c = new_c
            best_s = new_s
            step = max(0.0005, step * 0.97)
        else:
            step = min(0.1, step * 1.03)
            
    _, best_r = solve_lp(best_c)
    return best_c, best_r, best_s

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_sum = -np.inf
    
    inits = generate_inits(rng)
    
    # Phase 1: Broad search with SLSQP from diverse inits
    for pts in inits:
        r_init, _ = solve_lp(pts)
        r_init = np.maximum(r_init * 0.97, 1e-5)
        x0 = centers_to_params(pts, r_init)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                cv = constraint_func(res.x)
                if np.min(cv) >= -1e-6:
                    cs = -res.fun
                    if cs > best_sum:
                        best_sum = cs
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2 & 3: LP-driven refinement + Perturbation cycles
    if best_vars is not None:
        c_best = params_to_centers(best_vars)
        r_best = best_vars[0::3]
        
        # LP Local Search on centers
        for _ in range(3):
            c_best, r_best, s_best = local_search_lp(c_best, 2500, rng)
            if s_best > best_sum:
                best_sum = s_best
                x0 = centers_to_params(c_best, r_best * 0.98)
                try:
                    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
                    if res.success:
                        cv = constraint_func(res.x)
                        if np.min(cv) >= -1e-6:
                            cs = -res.fun
                            if cs > best_sum:
                                best_sum = cs
                                best_vars = res.x.copy()
                                c_best = params_to_centers(best_vars)
                except Exception:
                    pass
                    
        # Aggressive perturbation to escape local minima
        for k in range(40):
            x0 = best_vars.copy()
            x0[1::3] += rng.normal(0, 0.05, N)
            x0[2::3] += rng.normal(0, 0.05, N)
            x0[0::3] += rng.normal(0, 0.002, N)
            x0[0::3] = np.clip(x0[0::3], 1e-6, 0.49)
            x0[1::3] = np.clip(x0[1::3], 0.0, 1.0)
            x0[2::3] = np.clip(x0[2::3], 0.0, 1.0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
                if res.success:
                    cv = constraint_func(res.x)
                    if np.min(cv) >= -1e-6:
                        cs = -res.fun
                        if cs > best_sum:
                            best_sum = cs
                            best_vars = res.x.copy()
            except Exception:
                continue
                
        # Phase 4: High-precision final polish
        try:
            res = minimize(objective, best_vars, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if np.min(constraint_func(res.x)) >= -1e-7:
                best_vars = res.x
                best_sum = -res.fun
        except Exception:
            pass
            
    # Fallback (should rarely be reached)
    if best_vars is None:
        centers_f = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        r_f = np.full(N, 0.09)
        best_vars = centers_to_params(centers_f, r_f)
        best_sum = np.sum(r_f)
        
    # Reconstruct final centers from optimized parameters
    r_opt = best_vars[0::3]
    u_opt = best_vars[1::3]
    v_opt = best_vars[2::3]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    return centers, r_opt, float(best_sum)
