import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    n_rows = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_rows, n))
    b_ub = np.zeros(n_rows)
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    idx = np.arange(n)
    A_ub[idx, idx] = 1.0
    b_ub[idx] = centers[:, 0]
    idx = n + np.arange(n)
    A_ub[idx, idx - n] = 1.0
    b_ub[idx] = 1.0 - centers[:, 0]
    idx = 2 * n + np.arange(n)
    A_ub[idx, idx - 2 * n] = 1.0
    b_ub[idx] = centers[:, 1]
    idx = 3 * n + np.arange(n)
    A_ub[idx, idx - 3 * n] = 1.0
    b_ub[idx] = 1.0 - centers[:, 1]
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    k = 4 * n + np.arange(len(I_IDX))
    A_ub[k, I_IDX] = 1.0
    A_ub[k, J_IDX] = 1.0
    b_ub[k] = dists[I_IDX, J_IDX]
    
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def objective_slsqp(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints_slsqp(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    cx, cy, r = v[0::3], v[1::3], v[2::3]
    c = [cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r]
    
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    starts = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], 
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [4,5,6,5,6], 
        [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5]
    ]
    
    for pat in patterns:
        for _ in range(2):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
                
            pts = np.array(pts[:N])
            pts -= 0.5
            pts *= rng.uniform(0.9, 1.1)
            pts += 0.5
            
            rot = rng.uniform(-0.2, 0.2)
            c, s = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[c, -s], [s, c]])
            pts -= pts.mean(axis=0)
            pts += 0.5
            
            pts += rng.normal(0, 0.01, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            starts.append(pts)
            
    for seed in range(10):
        rng_f = np.random.RandomState(seed)
        pts = rng_f.rand(N, 2)
        r_c = np.full(N, 0.06)
        
        for step in range(300):
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
            
            for d in range(2):
                forces[:, d] += 30.0 * np.maximum(0, r_c - pts[:, d])
                forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - (1.0 - r_c))
                
            pts += 0.005 * forces * (0.99**step)
            pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
            
            for i in range(N):
                dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
                dp = max(dp, 1e-6)
                r_c[i] = 0.7 * min(dw, dp/2.0)
                
        starts.append(pts)
        
    return starts

def run_packing():
    rng = np.random.default_rng(42)
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    starts = generate_inits(rng)
    
    # Phase 1: Simulated Annealing on Centers with LP for Radii
    for init_c in starts:
        centers = init_c.copy()
        curr_r, curr_sum = solve_lp(centers)
        best_c_run = centers.copy()
        best_r_run = curr_r.copy()
        best_s_run = curr_sum
        
        temp = 0.008
        step = 0.025
        
        for it in range(6000):
            idx = rng.integers(N)
            old = centers[idx].copy()
            centers[idx] += rng.normal(0, step, 2)
            centers[idx] = np.clip(centers[idx], 1e-4, 0.999)
            
            r_new, s_new = solve_lp(centers)
            delta = s_new - curr_sum
            
            if delta > 0 or (temp > 1e-7 and rng.rand() < np.exp(delta / temp)):
                curr_sum = s_new
                if s_new > best_s_run:
                    best_s_run = s_new
                    best_c_run = centers.copy()
                    best_r_run = r_new.copy()
            else:
                centers[idx] = old
                
            temp *= 0.9998
            step = max(0.002, step * 0.9992)
            
        if best_s_run > best_sum:
            best_sum = best_s_run
            best_centers = best_c_run.copy()
            best_radii = best_r_run.copy()
            
    # Phase 2: High-Precision SLSQP Polish
    if best_centers is not None:
        bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
        cons = {'type': 'ineq', 'fun': constraints_slsqp}
        
        v_base = np.zeros(3 * N)
        v_base[0::3] = best_centers[:, 0]
        v_base[1::3] = best_centers[:, 1]
        v_base[2::3] = best_radii * 0.99
        
        # Multi-restart polish to escape shallow local minima
        for _ in range(12):
            vp = v_base.copy()
            vp[0::3] += rng.normal(0, 0.004, N)
            vp[1::3] += rng.normal(0, 0.004, N)
            vp[2::3] += rng.normal(0, 0.0004, N)
            
            vp[0::3] = np.clip(vp[0::3], 1e-4, 0.999)
            vp[1::3] = np.clip(vp[1::3], 1e-4, 0.999)
            vp[2::3] = np.maximum(vp[2::3], 1e-6)
            
            try:
                res = minimize(objective_slsqp, vp, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if res.success and np.min(constraints_slsqp(res.x)) >= -1e-8:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
            except Exception:
                pass
                
    # Fallback (should rarely be reached)
    if best_centers is None:
        best_centers = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_radii, best_sum = solve_lp(best_centers)
        
    return best_centers, np.maximum(best_radii, 0.0), float(best_sum)
