import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    k = 0
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def coordinate_ascent(centers, rng, max_iter=2000, init_step=0.04):
    """Optimizes centers by moving one circle at a time, evaluating LP sum of radii."""
    curr_c = centers.copy()
    _, sum_curr = solve_lp(curr_c)
    best_c = curr_c.copy()
    best_sum = sum_curr
    step = init_step
    temp = 0.06
    
    for it in range(max_iter):
        idx = rng.integers(N)
        direction = rng.normal(0, 1, 2)
        direction /= np.linalg.norm(direction)
        
        new_c = curr_c.copy()
        new_c[idx] = np.clip(new_c[idx] + step * direction, 0.005, 0.995)
        
        _, new_sum = solve_lp(new_c)
        delta = new_sum - sum_curr
        
        # Simulated Annealing acceptance
        if delta > 0 or (temp > 1e-6 and rng.random() < np.exp(delta / temp)):
            curr_c = new_c
            sum_curr = new_sum
            if sum_curr > best_sum:
                best_sum = sum_curr
                best_c = curr_c.copy()
                
        step = max(0.0003, step * 0.9992)
        temp *= 0.9988
    return best_c, best_sum

def slsqp_constraints(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    cx, cy, r = v[0::3], v[1::3], v[2::3]
    c = []
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dist2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def slsqp_objective(v):
    return -np.sum(v[2::3])

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5]]
    for pat in patterns:
        for _ in range(3):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            pts = np.array(pts[:N])
            pts -= 0.5
            scale = rng.uniform(0.85, 1.15)
            pts *= scale
            pts += 0.5
            rot = rng.uniform(-0.2, 0.2)
            c, s = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[c, -s], [s, c]])
            pts -= pts.mean(axis=0)
            pts += 0.5
            pts += rng.normal(0, 0.01, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    for _ in range(10):
        pts = rng.uniform(0.1, 0.9, (N, 2))
        r_curr = np.full(N, 0.06)
        for step in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 20.0 * np.maximum(0, r_curr - pts[:, d])
                forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            pts += 0.005 * (0.99**step) * forces
            pts = np.clip(pts, 0.01, 0.99)
            for i in range(N):
                dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
                r_curr[i] = 0.8 * min(dw, dp/2.0)
        inits.append(pts)
    return inits

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': slsqp_constraints}
    
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    inits = generate_inits(rng)
    
    # Phase 1 & 2: Broad search -> LP+SA -> SLSQP Polish
    for pts in inits:
        curr_c, curr_sum = coordinate_ascent(pts, rng, max_iter=1500, init_step=0.04)
        
        r_init, _ = solve_lp(curr_c)
        v0 = np.zeros(3 * N)
        v0[0::3] = curr_c[:, 0]
        v0[1::3] = curr_c[:, 1]
        v0[2::3] = np.maximum(r_init * 0.99, 1e-5)
        
        try:
            res = minimize(slsqp_objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                if np.min(slsqp_constraints(res.x)) >= -1e-7:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_r = res.x[2::3]
        except Exception:
            pass
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_c = curr_c
            best_r, _ = solve_lp(curr_c)
            
    # Phase 3: Micro-refinement & High-Precision Final Polish
    if best_c is not None:
        fine_c, _ = coordinate_ascent(best_c, rng, max_iter=2000, init_step=0.008)
        r_fine, _ = solve_lp(fine_c)
        v0 = np.zeros(3 * N)
        v0[0::3] = fine_c[:, 0]
        v0[1::3] = fine_c[:, 1]
        v0[2::3] = np.maximum(r_fine * 0.995, 1e-5)
        
        try:
            res = minimize(slsqp_objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-15, 'disp': False})
            if res.success and np.min(slsqp_constraints(res.x)) >= -1e-7:
                best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                best_r = res.x[2::3]
                best_sum = -res.fun
        except Exception:
            pass
            
    if best_c is None:
        best_c = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
        best_r = np.full(N, 0.09)
        best_sum = np.sum(best_r)
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
