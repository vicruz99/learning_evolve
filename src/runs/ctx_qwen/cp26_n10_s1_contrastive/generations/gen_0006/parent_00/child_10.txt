2. **Boundary Interactions**: Optimal packings heavily utilize corners and edges, requiring precise coordinate adjustments.

**Strategy:**
1. **LP-Based Simulated Annealing (SA)**: Instead of optimizing radii and centers jointly from the start, we decouple the problem. We perform SA directly on the *centers* of the circles. For each candidate center configuration, we solve a fast Linear Program (LP) to compute the maximum feasible radii and their sum. This LP approach guarantees strict feasibility and provides a smooth-ish landscape for SA to navigate. The SA phase excels at changing the contact topology (swapping neighbors, shifting clusters) without getting trapped by gradient barriers.

4. **Multi-Restart Polish**: Run multiple SLSQP refinements with slight perturbations around the best SA result to ensure we haven't missed a nearby gradient direction.

This hybrid approach leverages the global exploration of SA with LP feasibility checks, and the precise local convergence of constrained nonlinear programming.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    # Precompute pairwise distances efficiently
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Constraint matrices: 4*N boundary + N*(N-1)/2 pairwise
    num_cons = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_cons, n))
    b_ub = np.zeros(num_cons)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_val = np.array([x, 1.0 - x, y, 1.0 - y])
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def constraint_func(params):
    """
    Computes inequality constraints g(params) >= 0.
    Uses parameterization to automatically satisfy boundary constraints.
    Only pairwise non-overlap constraints are enforced.
    """
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def centers_to_params(centers, radii):
    """Convert centers and radii to the (r, u, v) parameterization."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def objective(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def run_sa(init_centers, rng):
    """Simulated Annealing on centers, evaluating objective via LP."""
    centers = init_centers.copy()
    radii, curr_sum = solve_lp(centers)
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = curr_sum
    
    temp = 0.025
    step = 0.028
    
    for _ in range(5000):
        idx = rng.integers(N)
        old_pos = centers[idx].copy()
        centers[idx] += rng.normal(0, step, 2)
        centers[idx] = np.clip(centers[idx], 0.002, 0.998)
        
        radii_new, new_sum = solve_lp(centers)
        delta = new_sum - curr_sum
        
        # Acceptance criterion
        if delta > 0 or (rng.random() < np.exp(delta / max(temp, 1e-6))):
            curr_sum = new_sum
            radii = radii_new
            # centers is already updated
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
        else:
            centers[idx] = old_pos
            
        temp *= 0.9993
        step = max(0.001, step * 0.9995)
        
    return best_centers, best_radii, best_sum

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns with varied row counts
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5],
        [5,5,5,5,6], [6,4,6,5,5], [5,6,4,6,5],
        [4,5,6,5,6], [6,6,5,5,4], [5,5,6,5,5],
        [6,5,5,5,5], [5,5,5,6,5], [7,5,5,5,4]
    ]
    
    for pat in patterns:
        pts = []
        r_est = 0.095
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = (r_idx % 2) * r_est
            x = r_est + shift
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3.0)
        pts = np.array(pts[:N])
        pts -= 0.5
        scale = rng.uniform(0.85, 1.15)
        pts *= scale
        pts += 0.5
        rot = rng.uniform(-0.2, 0.2)
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
        pts -= pts.mean(axis=0)
        pts += 0.5
        pts += rng.normal(0, 0.01, (N, 2))
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    # 2. Force-directed layouts to find organic packings
    for seed in range(10):
        rng_fd = np.random.RandomState(seed)
        pts = rng_fd.rand(N, 2)
        r_curr = np.full(N, 0.06)
        for step in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 20.0 * np.maximum(0, r_curr - pts[:, d])
                forces[:, d] -= 20.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            pts += 0.005 * forces * (0.99**step)
            pts = np.clip(pts, 0.01, 0.99)
            for i in range(N):
                dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                dp = np.min(np.linalg.norm(pts[i] - pts, axis=1))
                r_curr[i] = 0.8 * min(dw, dp/2.0)
        inits.append(pts)
        
    return inits

def run_packing():
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    # Generate diverse starting topologies
    inits = generate_inits(rng)
    
    # Phase 1: LP-based Simulated Annealing for global topology search
    for pts in inits:
        c, r, s = run_sa(pts, rng)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r
            
    # Phase 2: SLSQP refinement with boundary-parameterized variables
    if best_centers is not None:
        # Multiple restarts around best SA solution to catch gradient directions
        for k in range(15):
            c_pert = best_centers.copy()
            c_pert += rng.normal(0, 0.004, (N, 2))
            c_pert = np.clip(c_pert, 0.01, 0.99)
            r_pert, _ = solve_lp(c_pert)
            r_pert = np.maximum(r_pert * 0.995, 1e-5)
            
            x0 = centers_to_params(c_pert, r_pert)
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
                if res.success:
                    cons_val = constraint_func(res.x)
                    if np.min(cons_val) >= -1e-7:
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            r_opt = res.x[0::3]
                            u_opt = res.x[1::3]
                            v_opt = res.x[2::3]
                            best_centers = np.column_stack((r_opt + u_opt*(1-2*r_opt), 
                                                            r_opt + v_opt*(1-2*r_opt)))
                            best_radii = r_opt
            except Exception:
                pass
                
        # Phase 3: High-precision final polish
        x0 = centers_to_params(best_centers, best_radii * 0.999)
        try:
            res_f = minimize(objective, x0, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(constraint_func(res_f.x)) >= -1e-7:
                best_centers = np.column_stack((res_f.x[0::3] + res_f.x[1::3]*(1-2*res_f.x[0::3]), 
                                                res_f.x[0::3] + res_f.x[2::3]*(1-2*res_f.x[0::3])))
                best_radii = res_f.x[0::3]
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Fallback configuration
    if best_centers is None:
        best_centers = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_radii, best_sum = solve_lp(best_centers)
        
    return best_centers, best_radii, float(best_sum)
