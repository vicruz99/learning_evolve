import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def get_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_constr = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constr, n))
    b_ub = np.zeros(n_constr)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for p in range(len(I_IDX)):
        i, j = I_IDX[p], J_IDX[p]
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]
        k += 1
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def de_objective(centers_flat):
    """Objective for Differential Evolution: minimize negative LP sum of radii."""
    centers = centers_flat.reshape(N, 2)
    # Clip strictly inside to avoid degenerate LP bounds
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    s, _ = get_lp_radii(centers)
    return -s

def local_search(centers, n_iter=5000):
    """Greedy hill-climbing on centers with adaptive step size."""
    n = centers.shape[0]
    curr_sum, _ = get_lp_radii(centers)
    best_sum, best_c = curr_sum, centers.copy()
    step = 0.05
    
    for _ in range(n_iter):
        idx = np.random.randint(n)
        pert = np.random.randn(2) * step
        new_c = centers.copy()
        new_c[idx] = np.clip(new_c[idx] + pert, 1e-4, 1.0 - 1e-4)
        
        new_sum, _ = get_lp_radii(new_c)
        
        if new_sum > curr_sum:
            centers = new_c
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = new_c.copy()
            step *= 0.995
            step = max(step, 0.0005)
        else:
            step *= 1.01
            step = min(step, 0.1)
            
    return best_c, best_sum

def run_packing():
    np.random.seed(42)
    
    # Phase 1: Differential Evolution for global center configuration search
    bounds = [(0.0, 1.0)] * (2 * N)
    try:
        de_res = differential_evolution(de_objective, bounds, popsize=20, maxiter=80,
                                        mutation=(0.5, 1.0), recombination=0.9,
                                        seed=42, polishing=False)
        c_init = de_res.x.reshape(N, 2)
    except Exception:
        c_init = np.random.rand(N, 2) * 0.8 + 0.1
        
    # Phase 2: Intensive local search refinement
    c_best, s_best = local_search(c_init, n_iter=5000)
    
    # Phase 3: Perturbation restarts to escape local minima
    for _ in range(5):
        c_pert = c_best + np.random.randn(N, 2) * 0.04
        c_pert = np.clip(c_pert, 0.05, 0.95)
        c_new, s_new = local_search(c_pert, n_iter=2000)
        if s_new > s_best:
            c_best = c_new
            s_best = s_new
            
    _, r_best = get_lp_radii(c_best)
    
    # Phase 4: High-precision SLSQP polish on full (x, y, r) variables
    v0 = np.zeros(3 * N)
    v0[0::3] = c_best[:, 0]
    v0[1::3] = c_best[:, 1]
    # Shrink radii slightly to guarantee strict interior feasibility for the optimizer start
    v0[2::3] = r_best * 0.995
    
    bounds_full = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    
    def obj(v):
        return -np.sum(v[2::3])
        
    def constr(v):
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
        
        dx = x[:, None] - x[None, :]
        dy = y[:, None] - y[None, :]
        d2 = dx**2 + dy**2
        rs = r[:, None] + r[None, :]
        
        c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
        return np.concatenate(c)
        
    cons = {'type': 'ineq', 'fun': constr}
    
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds_full,
                       constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res.success and np.min(constr(res.x)) >= -1e-9:
            v0 = res.x
    except Exception:
        pass
        
    centers_out = np.column_stack((v0[0::3], v0[1::3]))
    radii_out = np.maximum(v0[2::3], 0.0)
    
    return centers_out, radii_out, float(np.sum(radii_out))
