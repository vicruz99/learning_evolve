import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def to_params_ruv(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params_ruv(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp_ruv(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp_ruv(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def obj_slqp_xyzr(params):
    """Objective for direct (x,y,r) optimization."""
    return -np.sum(params[2*N:])

def cons_slqp_xyzr(params):
    """Constraints for direct (x,y,r) optimization."""
    x = params[:N]
    y = params[N:2*N]
    r = params[2*N:]
    
    constraints = []
    # Boundary constraints
    constraints.extend(x - r)
    constraints.extend(1.0 - x - r)
    constraints.extend(y - r)
    constraints.extend(1.0 - y - r)
    
    # Pairwise non-overlap
    x_mat = x[:, np.newaxis] - x[np.newaxis, :]
    y_mat = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = x_mat**2 + y_mat**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    constraints.extend(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.array(constraints)

def generate_hex_centers(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_centers(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def generate_staggered_grid(rng, seed):
    """Generates staggered grid initializations."""
    rng_g = np.random.RandomState(seed)
    rows = rng.choice([5, 6, 7])
    cols_per_row = []
    remaining = N
    for r in range(rows):
        c = remaining // (rows - r)
        if c < 2:
            c = 2
        cols_per_row.append(c)
        remaining -= c
    if remaining > 0:
        cols_per_row[-1] += remaining
    
    pts = []
    y = 0.1
    row_h = 0.8 / max(rows - 1, 1)
    row_idx = 0
    for cnt in cols_per_row:
        row_width = 0.8 * (cnt - 1) / max(cnt - 1, 1) if cnt > 1 else 0.8
        x = 0.1 + (0.5 - cnt // 2) * row_width if cnt > 1 else 0.5
        shift = (row_idx % 2) * row_width * 0.5
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x + shift, y + rng_g.uniform(-0.02, 0.02)])
            x += row_width
        y += row_h
        row_idx += 1
    
    pts = np.array(pts[:N])
    return np.clip(pts, 0.02, 0.98)

def run_sa_on_centers(initial_centers, rng, iters=8000):
    """Run Simulated Annealing on center positions, guided by LP objective."""
    curr_c = initial_centers.copy()
    r_curr, curr_s = solve_lp_radii(curr_c)
    best_c = curr_c.copy()
    best_r = r_curr.copy()
    best_s = curr_s
    
    temp = 0.05
    step = 0.04
    
    for it in range(iters):
        move_type = rng.integers(0, 4)
        old = None
        
        if move_type == 0:
            # Small perturbation
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        elif move_type == 1:
            # Large jump
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        elif move_type == 2:
            # Pair swap/interchange
            i = rng.integers(N)
            j = rng.integers(N)
            while j == i:
                j = rng.integers(N)
            old = curr_c[[i, j]].copy()
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        else:
            # Multi-point perturbation (2-3 circles)
            n_move = rng.integers(2, 4)
            idxs = rng.choice(N, n_move, replace=False)
            old = curr_c[idxs].copy()
            curr_c[idxs] += rng.normal(0, step * 0.5, (n_move, 2))
            curr_c[idxs] = np.clip(curr_c[idxs], 0.01, 0.99)
        
        r_new, s_new = solve_lp_radii(curr_c)
        delta = s_new - curr_s
        
        accept = delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp))
        
        if accept:
            curr_s = s_new
            curr_c[idxs if move_type == 3 else [curr_c.shape[0]]]  # keep new
            if s_new > best_s:
                best_s = s_new
                best_c = curr_c.copy()
                best_r = r_new.copy()
        else:
            if move_type == 0:
                curr_c[idx] = old
            elif move_type == 1:
                curr_c[idx] = old
            elif move_type == 2:
                curr_c[[i, j]] = old
            else:
                curr_c[idxs] = old
        
        # Adaptive cooling and step size
        temp = max(1e-6, temp * 0.9995)
        if it % 1000 == 0:
            step = max(0.001, step * 0.9)
        
        # Occasional temperature boost
        if it % 2000 == 0 and temp < 0.01:
            temp = 0.03
    
    return best_c, best_r, best_s

def refine_with_slsqp_ruv(centers, radii, rng, iterations=40):
    """Refine using SLSQP with (r,u,v) parameterization."""
    bounds = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': cons_slqp_ruv}
    
    best_p = to_params_ruv(centers, np.clip(radii * 0.995, 1e-6, 0.49))
    best_s = np.sum(radii)
    
    for k in range(iterations):
        xp = best_p.copy()
        # Scale perturbation with iteration
        scale = 0.02 * (0.95 ** k)
        xp[:N] += rng.normal(0, scale * 0.1, N)
        xp[N:3*N] += rng.normal(0, scale, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp_ruv, xp, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp_ruv(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_s:
                    best_s = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
    
    # Final high-precision polish
    try:
        res_f = minimize(obj_slqp_ruv, best_p, method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_slqp_ruv(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_s = -res_f.fun
    except Exception:
        pass
    
    centers_out, radii_out = from_params_ruv(best_p)
    return centers_out, radii_out, best_s

def refine_with_slsqp_xyzr(centers, radii, rng, iterations=30):
    """Refine using direct (x,y,r) SLSQP."""
    x0 = np.concatenate([centers[:, 0], centers[:, 1], radii * 0.995])
    bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(1e-6, 0.49)] * N
    cons = {'type': 'ineq', 'fun': cons_slqp_xyzr}
    
    best_x = x0.copy()
    best_s = np.sum(radii)
    
    for k in range(iterations):
        xp = best_x.copy()
        scale = 0.015 * (0.96 ** k)
        xp[:2*N] += rng.normal(0, scale, 2*N)
        xp[2*N:] += rng.normal(0, scale * 0.1, N)
        xp[:2*N] = np.clip(xp[:2*N], 0.01, 0.99)
        xp[2*N:] = np.clip(xp[2*N:], 1e-6, 0.49)
        
        try:
            res = minimize(obj_slqp_xyzr, xp, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp_xyzr(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_s:
                    best_s = s_val
                    best_x = res.x.copy()
        except Exception:
            continue
    
    # Final polish
    try:
        res_f = minimize(obj_slqp_xyzr, best_x, method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_slqp_xyzr(res_f.x)) >= -1e-8:
            best_x = res_f.x
            best_s = -res_f.fun
    except Exception:
        pass
    
    centers_out = np.column_stack([best_x[:N], best_x[N:2*N]])
    radii_out = best_x[2*N:]
    return centers_out, radii_out, best_s

def run_packing():

    rng = np.random.default_rng(12345)

        inits.append(generate_force_centers(rng, s))

    # Evaluate all inits with LP
    init_scores = []
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)
        init_scores.append((s0, c0, r0))
    init_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Phase 2: SA on top initial configurations
    top_inits = init_scores[:15]
    for s0, c0, r0 in top_inits:
        sa_c, sa_r, sa_s = run_sa_on_centers(c0, rng, iters=6000)
        if sa_s > best_sum:
            best_sum = sa_s
            best_centers = sa_c.copy()
            best_radii = sa_r.copy()

    if best_centers is not None:
        r1_c, r1_r, r1_s = refine_with_slsqp_ruv(best_centers, best_radii, rng)

            best_sum = r1_s
            best_centers = r1_c.copy()
            best_radii = r1_r.copy()
    
    # Phase 4: SA + SLSQP cycle to escape local minima
    for cycle in range(3):

        if sa_s > best_sum:

            best_centers = sa_c.copy()
            best_radii = sa_r.copy()
        
        # Refine

            if r1_s > best_sum:
                best_sum = r1_s
                best_centers = r1_c.copy()
                best_radii = r1_r.copy()
    
    # Phase 8: Final LP polish to ensure radii are optimal for final centers
    if best_centers is not None:

        best_sum = final_s
    
    # Safety check and return
    if best_centers is None:
        # Fallback to top init
        best_centers = init_scores[0][1]
        best_radii = init_scores[0][2]
        best_sum = init_scores[0][0]
    
    return best_centers, np.maximum(best_radii, 0.0), float(best_sum)
