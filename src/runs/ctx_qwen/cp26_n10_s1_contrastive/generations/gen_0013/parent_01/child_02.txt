import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-4

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization with specified parameters."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def boundary_heavy_init(rng):
    """Generates an initialization heavily biased towards corners and edges."""
    pts = []
    # Corners
    for cx in [0.15, 0.85]:
        for cy in [0.15, 0.85]:
            pts.append([cx + rng.uniform(-0.05, 0.05), cy + rng.uniform(-0.05, 0.05)])
            
    # Edges
    for _ in range(4):
        pts.append([rng.uniform(0.05, 0.95), 0.1 + rng.uniform(-0.05, 0.05)])
    for _ in range(4):
        pts.append([rng.uniform(0.05, 0.95), 0.9 + rng.uniform(-0.05, 0.05)])
    for _ in range(4):
        pts.append([0.1 + rng.uniform(-0.05, 0.05), rng.uniform(0.05, 0.95)])
    for _ in range(4):
        pts.append([0.9 + rng.uniform(-0.05, 0.05), rng.uniform(0.05, 0.95)])
        
    # Fill remaining randomly in center
    while len(pts) < N:
        pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
        
    return np.array(pts[:N])

def local_search_ascent(centers, rng):
    """Performs coordinate ascent on centers to improve LP objective."""
    curr_c = centers.copy()
    _, curr_s = solve_lp_radii(curr_c)
    step = 0.02
    improved = True
    
    while improved:
        improved = False
        # Shuffle order to avoid bias
        order = rng.permutation(N)
        for idx in order:
            old_pos = curr_c[idx].copy()
            best_pos = old_pos
            best_s = curr_s
            
            # Probe 8 directions
            for dx in np.linspace(-step, step, 5):
                for dy in np.linspace(-step, step, 5):
                    if dx == 0 and dy == 0:
                        continue
                    curr_c[idx, 0] = np.clip(old_pos[0] + dx, 0.01, 0.99)
                    curr_c[idx, 1] = np.clip(old_pos[1] + dy, 0.01, 0.99)
                    
                    _, new_s = solve_lp_radii(curr_c)
                    if new_s > best_s + 1e-7:
                        best_s = new_s
                        best_pos = curr_c[idx].copy()
                        
            if best_s > curr_s + 1e-7:
                curr_c[idx] = best_pos
                curr_s = best_s
                improved = True
                step *= 0.95 # Decay step size slightly upon improvement
            else:
                curr_c[idx] = old_pos
                curr_c[idx, 0] = np.clip(curr_c[idx, 0], 0.01, 0.99)
                curr_c[idx, 1] = np.clip(curr_c[idx, 1], 0.01, 0.99)
                
    return curr_c, curr_s

def sa_optimize(initial_centers, rng):
    """Simulated Annealing optimization on centers."""
    curr_c = initial_centers.copy()
    _, curr_s = solve_lp_radii(curr_c)
    best_c = curr_c.copy()
    best_s = curr_s
    
    temp = 0.05
    step = 0.04
    no_improve_count = 0
    
    for it in range(25000):
        move_type = rng.integers(0, 4)
        if move_type == 0:
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        elif move_type == 1:
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] = rng.uniform(0.1, 0.9, 2)
        elif move_type == 2:
            # Shake 2 circles
            idxs = rng.choice(N, 2, replace=False)
            old = curr_c[idxs].copy()
            curr_c[idxs] += rng.normal(0, step * 1.5, (2, 2))
            curr_c[idxs] = np.clip(curr_c[idxs], 0.01, 0.99)
        else:
            i = rng.integers(N)
            j = rng.integers(N)
            while j == i:
                j = rng.integers(N)
            old = curr_c[[i, j]].copy()
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
            
        _, new_s = solve_lp_radii(curr_c)
        delta = new_s - curr_s
        
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
            curr_s = new_s
            if new_s > best_s:
                best_s = new_s
                best_c = curr_c.copy()
                step = min(0.05, step * 1.02)
                temp = min(0.1, temp * 1.01) # Rejuvenate temp slightly
                no_improve_count = 0
            else:
                step *= 0.998
                no_improve_count += 1
        else:
            if move_type < 3:
                if move_type < 2:
                    curr_c[idx] = old
                else:
                    curr_c[idxs] = old
            else:
                curr_c[[i, j]] = old
            step = max(0.005, step * 0.999)
            no_improve_count += 1
            
        temp *= 0.9996
        
        # Periodic shake if stuck
        if no_improve_count > 2000:
            idxs = rng.choice(N, 5, replace=False)
            curr_c[idxs] = rng.uniform(0.2, 0.8, (5, 2))
            no_improve_count = 0
            step = 0.03
            temp = 0.03
            
    return best_c, best_s

def run_packing():
