5. **Final LP Polish**: Always re-solve the LP on the final centers to guarantee optimal radii for the exact geometry, ensuring strict feasibility and maximum sum.

This hybrid approach balances global exploration (SA), local refinement (Nelder-Mead coordinate search, SLSQP), and exact radius optimization (LP), which is proven to yield high-quality packings for moderate N.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-5

def phys_to_params(c, r):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    d = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((c[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((c[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def params_to_phys(p):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons_slqp(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    c, r = params_to_phys(p)
    dx = c[:, 0, np.newaxis] - c[np.newaxis, :, 0]
    dy = c[:, 1, np.newaxis] - c[np.newaxis, :, 1]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, rows, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in rows:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(250):
        f = np.zeros_like(pts)
        d = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dist = np.sqrt(np.sum(d**2, axis=2)) + 1e-8
        f += np.sum(d / dist[:, :, np.newaxis]**2, axis=1)
        
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.08 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.92)
            
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    rng = np.random.default_rng(42)
    bounds_s = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_s = {'type': 'ineq', 'fun': cons_slqp}
    
    best_c, best_r, best_sum = None, None, -np.inf
    
    # Generate diverse initial configurations
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5], 
        [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4], [7,5,5,5,4], [5,7,5,5,4],
        [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3]
    ]
    for p in pats:
        for _ in range(3):
            inits.append(hex_init(rng, p, rng.uniform(-0.3, 0.3), rng.uniform(0.9, 1.1)))
    for _ in range(12):
        inits.append(force_init(rng))
        
    # Evaluate inits with LP
    init_scores = []
    for c in inits:
        r, s = solve_lp(c)
        init_scores.append((s, c, r))
    init_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Take the best few inits and run SA iteratively
    start_configs = init_scores[:5]
    
    for s0, c0, r0 in start_configs:
        curr_c, curr_sum = c0.copy(), s0
        
        # SA Phase
        temp, step = 0.05, 0.04
        for it in range(10000):
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            
            if rng.random() < 0.04:
                curr_c[idx] = rng.uniform(0.05, 0.95, 2)
            else:
                curr_c[idx] += rng.normal(0, step, 2)
                curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
                
            nr, ns = solve_lp(curr_c)
            delta = ns - curr_sum
            
            if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
                curr_sum = ns
                if ns > best_sum:
                    best_sum, best_c, best_r = ns, curr_c.copy(), nr.copy()
                    step = min(0.05, step * 1.02)
                else:
                    step = max(0.001, step * 0.995)
            else:
                curr_c[idx] = old
                
            temp *= 0.9993
            step = max(0.001, step * 0.9994)
            
        # Update best if SA finished at a higher point
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_c = curr_c.copy()
            best_r, _ = solve_lp(curr_c)

    # SLSQP Refinement on best SA result
    if best_c is not None:
        p0 = phys_to_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_s, constraints=cons_s,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_c, best_r = params_to_phys(res.x)
        except Exception:
            pass
            
        # Perturbation & SLSQP to escape local minima
        for k in range(25):
            xp = phys_to_params(best_c, best_r * 0.995)
            xp[1::3] += rng.normal(0, 0.025, N)
            xp[2::3] += rng.normal(0, 0.025, N)
            xp[1::3] = np.clip(xp[1::3], 0.0, 1.0)
            xp[2::3] = np.clip(xp[2::3], 0.0, 1.0)
            
            try:
                res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_s, constraints=cons_s,
                               options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
                if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_c, best_r = params_to_phys(res.x)
            except Exception:
                continue
                
        # Coordinate-wise local search on centers (Nelder-Mead)
        # Optimizes each circle's position while others are fixed
        curr_c_ls = best_c.copy()
        curr_r_ls, curr_sum_ls = solve_lp(curr_c_ls)
        
        for _ in range(3):
            for i in range(N):
                others = curr_c_ls.copy()
                
                def obj_2d(pos):
                    temp_c = others.copy()
                    temp_c[i] = pos
                    temp_c[i] = np.clip(temp_c[i], 1e-4, 1.0 - 1e-4)
                    _, val = solve_lp(temp_c)
                    return -val # Minimize negative sum
                
                try:
                    res_2d = minimize(obj_2d, curr_c_ls[i], method='Nelder-Mead',
                                      options={'maxiter': 80, 'xatol': 1e-5, 'fatol': 1e-8})
                    if res_2d.success:
                        curr_c_ls[i] = np.clip(res_2d.x, 1e-4, 1.0 - 1e-4)
                except Exception:
                    pass
                    
            r_ls, s_ls = solve_lp(curr_c_ls)
            if s_ls > best_sum:
                best_sum = s_ls
                best_c = curr_c_ls.copy()
                best_r = r_ls.copy()
                
        # Final high-precision polish with SLSQP
        try:
            p_final = phys_to_params(best_c, np.clip(best_r * 0.999, 1e-6, 0.49))
            res_f = minimize(obj_slqp, p_final, method='SLSQP', bounds=bounds_s,
                             constraints=cons_s, options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-8:
                best_c, best_r = params_to_phys(res_f.x)
                best_sum = -res_f.fun
        except Exception:
            pass

    # Final exact LP solve to guarantee optimal radii for precise centers
    final_r, final_sum = solve_lp(best_c)
    return best_c, np.maximum(final_r, 0.0), float(final_sum)
