import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-6

def obj_slqp(params):
    return -np.sum(params[0::3])

def cons_slqp(params):
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def to_params(centers, radii):
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def hex_init(rng, rows, rot, scale):
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in rows:
        shift = 0.0 if row % 2 == 0 else r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng, seed):
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.1, 0.9, (N, 2))
    r_curr = np.full(N, 0.05)
    for step in range(400):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        for d in range(2):
            forces[:, d] += 30.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
        step_size = 0.008 * (0.99**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_others[i] = np.inf
            d_pair = np.min(dists_others)
            r_curr[i] = 0.75 * min(d_wall, d_pair/2.0)
    return pts

def run_sa(centers, rng, iters=8000):
    curr_c = centers.copy()
    curr_r, curr_sum = solve_lp_radii(curr_c)
    best_c = curr_c.copy()
    best_r = curr_r.copy()
    best_sum = curr_sum
    
    temp = 0.05
    step = 0.04
    
    for it in range(iters):
        move_type = rng.integers(0, 4)
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        if move_type == 0:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        elif move_type == 1:
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        elif move_type == 2:
            jdx = rng.integers(N)
            curr_c[idx], curr_c[jdx] = curr_c[jdx].copy(), curr_c[idx].copy()
            old = None
        else:
            curr_c[idx] = rng.uniform(0.1, 0.9, 2)
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_sum
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_sum = ns
            curr_r = nr.copy()
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
                step = min(0.06, step * 1.01)
            else:
                step = max(0.001, step * 0.995)
        else:
            if old is not None:
                curr_c[idx] = old
                
        temp *= 0.9992
        step = max(0.001, step * 0.9993)
        
    return best_c, best_r, best_sum

def obj_i_wrapper(x, base_vars, i):
    tv = base_vars.copy()
    tv[N + i] = x[0]
    tv[2*N + i] = x[1]
    return obj_slqp(tv)

def cons_i_wrapper(x, base_vars, i):
    tv = base_vars.copy()
    tv[N + i] = x[0]
    tv[2*N + i] = x[1]
    return cons_slqp(tv)

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_vars = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [5,5,6,5,5], [7,5,5,5,4], [4,5,5,5,7], [5,7,5,5,4],
        [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3], [5,5,5,5,5,1],
        [6,5,5,5,5], [5,5,5,6,5], [4,6,6,5,5], [5,6,6,4,5],
        [5,5,5,5,5,1], [6,6,6,5,3], [5,5,6,6,4]
    ]
    
    for pat in patterns:
        for _ in range(4):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.4, 0.4), scale=rng.uniform(0.85, 1.15)))
            
    for s in range(25):
        inits.append(force_init(rng, s))
        
    init_scores = []
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)
        init_scores.append((s0, c0, r0))
    init_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Run SA on top 3 initializations to diversify search and avoid single-basin trapping
    for s0, c0, r0 in init_scores[:3]:
        curr_c, curr_r, curr_sum = run_sa(c0, rng, iters=10000)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_c = curr_c.copy()
            best_r = curr_r.copy()
            
    if best_c is None:
        best_c = init_scores[0][1].copy()
        best_r = init_scores[0][2].copy()
        best_sum = init_scores[0][0]
        
    # SLSQP Refinement
    r_safe = np.clip(best_r * 0.995, 1e-6, 0.49)
    p0 = to_params(best_c, r_safe)
    try:
        res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp,
                       constraints=cons_dict, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_vars = res.x.copy()
                best_c, best_r = from_params(best_vars)
    except Exception:
        pass
        
    if best_vars is None:
        best_vars = to_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
        
    # Perturbation & SLSQP to escape local minima
    for k in range(40):
        xp = best_vars.copy()
        xp[0::3] += rng.normal(0, 0.002, N)
        xp[1::3] += rng.normal(0, 0.025, N)
        xp[2::3] += rng.normal(0, 0.025, N)
        xp[0::3] = np.clip(xp[0::3], 1e-6, 0.49)
        xp[1::3] = np.clip(xp[1::3], 0.0, 1.0)
        xp[2::3] = np.clip(xp[2::3], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_dict, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x.copy()
                    best_c, best_r = from_params(best_vars)
        except Exception:
            continue
            
    # Coordinate-wise SLSQP to finely tune individual circle placements
    curr_vars = best_vars.copy()
    improved = True
    while improved:
        improved = False
        for i in range(N):
            x0_i = np.array([curr_vars[N + i], curr_vars[2*N + i]])
            try:
                res_i = minimize(obj_i_wrapper, x0_i, args=(curr_vars, i), method='SLSQP', 
                                 bounds=[(0.0, 1.0), (0.0, 1.0)],
                                 constraints={'type': 'ineq', 'fun': cons_i_wrapper, 'args': (curr_vars, i)},
                                 options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if res_i.success and np.min(cons_i_wrapper(res_i.x, curr_vars, i)) >= -1e-7:
                    if res_i.fun < obj_slqp(curr_vars) - 1e-10:
                        curr_vars[N + i] = res_i.x[0]
                        curr_vars[2*N + i] = res_i.x[1]
                        improved = True
            except Exception:
                pass
    best_vars = curr_vars
    best_c, best_r = from_params(best_vars)
    best_sum = -obj_slqp(best_vars)
        
    # Final exact LP polish to guarantee radii match centers perfectly
    if best_c is not None:
        final_r, final_sum = solve_lp_radii(best_c)
        best_c, best_r, best_sum = best_c, final_r, final_sum
    else:
        best_c = np.random.rand(N, 2) * 0.8 + 0.1
        best_r = np.full(N, 0.04)
        best_sum = np.sum(best_r)
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
