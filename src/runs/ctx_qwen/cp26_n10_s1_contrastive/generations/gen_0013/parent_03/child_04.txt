To improve the sum of radii for packing 26 circles in a unit square, I will employ a robust multi-phase optimization pipeline that has proven effective for high-density packings:

1. **Diverse Initialization**: Generate a large pool of initial configurations using perturbed hexagonal lattices (with varied row counts, rotations, and scales) and force-directed layouts. Hexagonal packings are theoretically near-optimal for dense circle arrangements.
2. **LP Radii Estimation**: For each initial center configuration, solve a Linear Program (LP) to find the maximum possible radii that satisfy all non-overlap and boundary constraints. This quickly evaluates the quality of each topology.
3. **Gradient-Based Refinement (SLSQP)**: Take the top-scoring initializations and refine them using SLSQP. A clever `(r, u, v)` parameterization is used where `x = r + u*(1-2r)` and `y = r + v*(1-2r)`. This automatically enforces boundary constraints `r <= x <= 1-r` and `r <= y <= 1-r`, leaving only pairwise non-overlap constraints for the optimizer.
4. **Simulated Annealing on Centers**: From the best SLSQP result, run Simulated Annealing directly on the center coordinates, evaluating the objective via the fast LP. This allows escaping local minima by accepting worse moves probabilistically and exploring the topology space more broadly.
5. **Perturbation & High-Precision Polish**: Perform multiple SLSQP restarts with randomized perturbations to escape remaining local basins, followed by a high-precision final optimization pass. A final exact LP solve guarantees radii are maximally optimal for the converged centers.

This strategy balances global exploration (SA, diverse inits) with local exploitation (SLSQP, LP polish) and strictly adheres to the constraint of no closures or lambdas.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves the LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def params_to_centers_r(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def centers_r_to_params(centers, radii):
    """Convert centers and radii to the (r, u, v) parameterization."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def cons(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        mat = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ mat.T + 0.5
        
    pts += rng.uniform(-0.012, 0.012, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_f = np.random.RandomState(seed)
    pts = rng_f.uniform(0.1, 0.9, (N, 2))
    r = np.full(N, 0.05)
    for step in range(400):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / d**2
        np.fill_diagonal(rep, 0.0)
        f = np.sum(rep[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, r - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - (1.0 - r))
            
        step_size = 0.006 * (0.995**step)
        pts += step_size * f
        pts = np.clip(pts, 0.01, 0.99)
        
        for i in range(N):
            dw = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            do = np.min(np.linalg.norm(pts[i] - pts, axis=1))
            do = max(do, 1e-4)
            r[i] = 0.85 * min(dw, do/2.0)
    return pts

def run_sa(centers, rng, iters=8000):
    """Simulated Annealing on centers with LP evaluation."""
    curr_c = centers.copy()
    curr_r, curr_s = solve_lp_radii(curr_c)
    best_c = curr_c.copy()
    best_r = curr_r.copy()
    best_s = curr_s
    
    temp = 0.05
    step = 0.04
    
    for it in range(iters):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        if rng.random() < 0.06:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            curr_r = nr.copy()
            if ns > best_s:
                best_s = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
                step = min(0.05, step * 1.015)
            else:
                step = max(0.001, step * 0.996)
        else:
            curr_c[idx] = old
            
        temp *= 0.9992
    return best_c, best_r, best_s

def run_packing():

    ]
    
    for pat in patterns:

    scores = []
