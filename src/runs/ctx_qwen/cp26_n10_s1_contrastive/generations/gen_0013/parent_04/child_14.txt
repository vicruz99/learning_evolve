

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we need a highly efficient optimization pipeline that balances global exploration with precise local refinement. The current approach uses Simulated Annealing (SA) on centers combined with SLSQP, but SA's random moves can be slow to escape tight local minima. 

Key improvements:
1. **LP-Dual Driven Gradient Ascent**: When circles are tightly packed, the LP that optimizes radii provides dual variables (marginals) that exactly indicate how much the objective would improve if constraints were relaxed. We compute forces on centers proportional to these marginals and move circles in directions that relieve tight constraints. This is essentially exact gradient ascent on the LP objective, vastly outperforming random SA.
2. **Boundary-Safe Parameterization**: We maintain the `(r, u, v)` parameterization where `x = r + u*(1-2r)` and `y = r + v*(1-2r)`. This automatically satisfies boundary constraints `r <= x <= 1-r` and `r <= y <= 1-r`, reducing SLSQP to only handling pairwise non-overlap constraints.
3. **Multi-Phase Optimization**: 
   - Generate diverse initial configurations (hexagonal lattices, force-directed layouts, perturbed grids).
   - Run SLSQP to find strong local optima.
   - Apply LP-Dual gradient ascent to push circles apart efficiently.
   - Run SLSQP again to refine.
   - Use perturbation restarts and global rotations to escape remaining basins.
   - Final high-precision polish.
4. **Robust LP Solver**: Use `scipy.optimize.linprog` with the `highs` method for fast, stable radius optimization and reliable marginal extraction.

This strategy leverages mathematical structure (LP duality) to guide moves, ensuring we efficiently approach the theoretical packing limit.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def build_lp_data(centers):
    """Constructs LP constraint matrix and bounds for fixed centers."""
    n = centers.shape[0]
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
    return A_ub, b_ub

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    A_ub, b_ub = build_lp_data(centers)
    bounds = [(0.0, None)] * N
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(N, 1e-5), 1e-4

def dual_force_refine(centers, max_iter=300, alpha=0.008):
    """Gradient ascent on centers driven by LP dual variables (marginals)."""
    radii = np.full(N, 1e-5)
    sum_r = 0.0
    for _ in range(max_iter):
        A_ub, b_ub = build_lp_data(centers)
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*N, method='highs')
        if not res.success:
            break
            
        radii = res.x
        sum_r = -res.fun
        marg = np.maximum(getattr(res.ineqlin, 'marginals', np.zeros(len(b_ub))), 0.0)
        
        forces = np.zeros((N, 2))
        k = 0
        # Boundary constraint forces
        for i in range(N):
            forces[i, 0] += marg[k]; k += 1  # r <= x
            forces[i, 0] -= marg[k]; k += 1  # r <= 1-x
            forces[i, 1] += marg[k]; k += 1  # r <= y
            forces[i, 1] -= marg[k]; k += 1  # r <= 1-y
            
        # Pairwise constraint forces
        for i in range(N):
            for j in range(i + 1, N):
                m = marg[k]
                if m > 1e-6:
                    diff = centers[i] - centers[j]
                    d = np.linalg.norm(diff)
                    if d > 1e-9:
                        f = m * diff / d
                        forces[i] += f
                        forces[j] -= f
                k += 1
                
        centers += alpha * forces
        centers = np.clip(centers, 0.005, 0.995)
        alpha *= 0.995
        
    return centers, radii, sum_r

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_func(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_hex_centers(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_centers(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.15, 0.85, (N, 2))
    for _ in range(400):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 30.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 30.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.002 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': cons_func}
    
    best_p = None
    best_sum = -np.inf
    
    # Phase 1: Diverse Initializations
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], 
        [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
        [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3], [5,5,5,5,5,1], [4,5,6,5,6]
    ]
    for pat in pats:
        for _ in range(3):
            inits.append(generate_hex_centers(rng, pat, rot=rng.uniform(-0.35, 0.35), scale=rng.uniform(0.85, 1.15)))
    for s in range(12):
        inits.append(generate_force_centers(rng, s))
        
    init_scores = []
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)
        init_scores.append((s0, c0, r0))
    init_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Phase 2: SLSQP Refinement on Top Inits
    for s0, c0, r0 in init_scores[:12]:
        p0 = to_params(c0, np.clip(r0 * 0.985, 1e-6, 0.49))
        try:
            res = minimize(obj_func, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_func(res.x)) >= -1e-8:
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        best_p = to_params(init_scores[0][1], np.clip(init_scores[0][2] * 0.99, 1e-6, 0.49))
        best_sum = -obj_func(best_p)
        
    curr_c, curr_r = from_params(best_p)
    
    # Phase 3: LP-Dual Gradient Ascent (Force Refinement)
    curr_c, curr_r, curr_s = dual_force_refine(curr_c, max_iter=300, alpha=0.006)
    if curr_s > best_sum:
        best_sum = curr_s
        best_p = to_params(curr_c, np.clip(curr_r * 0.995, 1e-6, 0.49))
        
    # Phase 4: SA on Centers with LP Evaluation
    temp = 0.04
    step = 0.03
    for it in range(5000):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.normal(0, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_p = to_params(curr_c, np.clip(nr * 0.995, 1e-6, 0.49))
                step = min(0.035, step * 1.02)
            else:
                step = max(0.001, step * 0.995)
        else:
            curr_c[idx] = old
        temp *= 0.9995
        
    # Phase 5: Perturbation & SLSQP Restarts
    for k in range(25):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.002, N)
        xp[N:3*N] += rng.normal(0, 0.02, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_func, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_func(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 6: Final Dual-Force + SLSQP Polish
    curr_c, curr_r, _ = from_params(best_p)
    curr_c, curr_r, curr_s = dual_force_refine(curr_c, max_iter=200, alpha=0.003)
    if curr_s > best_sum:
        best_sum = curr_s
        best_p = to_params(curr_c, np.clip(curr_r * 0.998, 1e-6, 0.49))
        
    try:
        res_f = minimize(obj_func, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                         options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_func(res_f.x)) >= -1e-9:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_p)
    radii_final, sum_final = solve_lp_radii(centers)
    
    return centers, np.maximum(radii_final, 0.0), float(sum_final)
