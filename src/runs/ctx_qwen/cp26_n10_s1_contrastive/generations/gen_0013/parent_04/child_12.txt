import numpy as np
from scipy.optimize import minimize, linprog
import operator

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = N * (N - 1) // 2
NUM_CONS = 4 * N + NUM_PAIRS

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((NUM_CONS, n))
    b_ub = np.zeros(NUM_CONS)
    k = 0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        bounds_val = np.array([x, 1.0 - x, y, 1.0 - y])
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-4

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_func(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_hex_centers(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_centers(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 30.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 30.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.002 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def generate_grid_centers(rng, seed):
    """Generates a perturbed grid initialization."""
    rng_g = np.random.RandomState(seed)
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng_g.uniform(-0.025, 0.025, (N, 2))
    return np.clip(pts, 0.05, 0.95)

def run_packing():

    cons_slqp = {'type': 'ineq', 'fun': cons_func}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], 
        [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
        [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3], [5,5,5,5,5,1], [4,5,6,5,6],
        [5,4,6,5,6], [6,5,4,6,5], [4,7,5,5,5], [5,4,7,5,5], [5,5,4,7,5]
    ]
    for pat in pats:
        for _ in range(3):
            inits.append(generate_hex_centers(rng, pat, rot=rng.uniform(-0.35, 0.35), scale=rng.uniform(0.85, 1.15)))
    for s in range(15):
        inits.append(generate_force_centers(rng, s))
    for s in range(10):
        inits.append(generate_grid_centers(rng, s))
        
    # Phase 2: LP-Driven Simulated Annealing on Centers
    # Run multiple independent SA runs to find robust local optima
    num_sa_runs = 12
    for run_idx in range(num_sa_runs):
        # Pick a random init or start from best found so far
        if run_idx == 0:
            curr_c = inits[0].copy()
        else:
            # Mix previous best with random perturbation
            curr_c = best_c.copy() if best_c is not None else inits[rng.integers(len(inits))].copy()
            curr_c += rng.normal(0, 0.05, curr_c.shape)

            best_r = curr_r.copy()
            
        temp = 0.045
        step = 0.035
        
        for it in range(2000):
            move_type = rng.integers(0, 4)
            idx = rng.integers(N)
            old_pos = curr_c[idx].copy()
            
            if move_type == 0:
                curr_c[idx] += rng.normal(0, step, 2)

            elif move_type == 2:
                # Swap
                i, j = rng.choice(N, 2, replace=False)
                curr_c[[i, j]] = curr_c[[j, i]].copy()
                old_pos = None # Swap is its own inverse, handle carefully
            else:

                curr_c[idx] += rng.normal(0, step * 1.5, 2)

            nr, ns = solve_lp_radii(curr_c)

                    best_c = curr_c.copy()
                    best_r = nr.copy()
                    step = min(0.04, step * 1.05)

            if res.success and np.min(cons_func(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            pass
            
        # Phase 4: Perturbation & SLSQP restarts to escape local minima
        if best_p is not None:

                try:
                    res = minimize(obj_func, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,

                    if res.success and np.min(cons_func(res.x)) >= -1e-8:
                        s_val = -res.fun
                        if s_val > best_sum:
                            best_sum = s_val
                            best_p = res.x.copy()
                except Exception:
                    continue
                    
        # Phase 5: Global rotation escapes
        if best_c is not None:
            for _ in range(10):
                rot = rng.uniform(-0.12, 0.12)

    radii_final, sum_final = solve_lp_radii(centers)
