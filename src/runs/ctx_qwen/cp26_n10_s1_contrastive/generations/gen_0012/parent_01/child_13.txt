The objective function $V(\mathbf{c}) = \max \sum r_i$ over centers $\mathbf{c}$ is concave and piecewise linear. By sensitivity analysis (dual variables/shadow prices), we can compute the exact gradient $\nabla_{\mathbf{c}} V(\mathbf{c})$ from the LP solution without finite differences.

**Strategy:**
1. **Gradient Ascent via LP Sensitivity:** Instead of random Simulated Annealing, we perform directed gradient ascent on circle centers. At each step, we solve the LP for radii, extract dual variables, and compute the gradient of the sum of radii with respect to centers. We move centers along this gradient to increase the objective.
2. **Diverse Initializations:** We start from multiple structured configurations (hexagonal lattices with varied row counts and rotations, plus force-directed layouts) to sample different topological basins.
3. **Adaptive Step & Noise:** We use normalized gradient steps with exponential decay. Periodic small Gaussian noise helps escape flat regions or local plateaus in the piecewise-linear landscape.
4. **SLSQP Polish:** After gradient ascent finds a promising configuration, we switch to a constrained nonlinear optimizer (SLSQP) with a boundary-safe parameterization $(r, u, v)$ to finely tune positions and radii, handling active constraints rigorously.
5. **Final LP Validation:** We re-solve the LP on the final centers to guarantee radii are exactly optimal and constraints are satisfied within numerical tolerance.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_and_grad(centers):
    """Solves LP for radii and computes gradient of sum(radii) w.r.t centers using dual variables."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    
    cons_types = []
    cons_idx = []
    k = 0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        for b, ct in [(x, 'x_min'), (1.0 - x, 'x_max'), (y, 'y_min'), (1.0 - y, 'y_max')]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            cons_types.append(ct)
            cons_idx.append((i,))
            k += 1
            
    # Pairwise distance constraints
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(np.maximum(dx**2 + dy**2, 1e-15))
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            cons_types.append('pair')
            cons_idx.append((i, j))
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return np.full(n, 1e-8), 0.0, np.zeros_like(centers)
        
    radii = np.maximum(res.x, 1e-10)
    grad = np.zeros_like(centers)
    
    try:
        duals = res.ineqlin.marginals
    except AttributeError:
        duals = np.zeros(num_c)
        
    for idx, dual in enumerate(duals):
        # Marginal from minimization of -sum(r) is <= 0. Shadow price is -dual.
        val = max(0.0, -dual)
        if val < 1e-9:
            continue
            
        ct = cons_types[idx]
        if ct == 'pair':
            i, j = cons_idx[idx]
            d = dists[i, j]
            if d > 1e-8:
                f = val / d
                grad[i, 0] += f * dx[i, j]
                grad[i, 1] += f * dy[i, j]
                grad[j, 0] -= f * dx[i, j]
                grad[j, 1] -= f * dy[i, j]
        elif ct == 'x_min':
            i, = cons_idx[idx]
            grad[i, 0] += val
        elif ct == 'x_max':
            i, = cons_idx[idx]
            grad[i, 0] -= val
        elif ct == 'y_min':
            i, = cons_idx[idx]
            grad[i, 1] += val
        elif ct == 'y_max':
            i, = cons_idx[idx]
            grad[i, 1] -= val
            
    return radii, -res.fun, grad

def to_params(centers, radii):
    """Map physical centers/radii to boundary-safe (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    best_centers = None
    best_sum = -np.inf
    best_radii = None
    
    # Generate diverse initial configurations
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], 
            [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
            [5,5,4,6,6], [4,6,6,5,5], [6,5,5,6,4], [5,4,6,6,5], [4,5,6,6,5]]
    for pat in pats:
        for _ in range(3):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.2, 0.2), scale=rng.uniform(0.9, 1.1)))
            
    # Force-directed inits
    for _ in range(10):
        pts = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(100):
            f = np.zeros_like(pts)
            diff = pts[:, None, :] - pts[None, :, :]
            d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-4
            f += np.sum((1.0 / d**2)[:, :, None] * diff / d[:, :, None], axis=1)
            pts += 0.005 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # Phase 1: Gradient Ascent on Centers
    for init_c in inits:
        curr_c = init_c.copy()
        radii, curr_sum, grad = solve_lp_and_grad(curr_c)
        step = 0.015
        
        for it in range(2500):
            radii, val, grad = solve_lp_and_grad(curr_c)
            g_norm = np.linalg.norm(grad)
            if g_norm < 1e-8:
                break
                
            curr_c += step * grad / g_norm
            curr_c = np.clip(curr_c, 0.005, 0.995)
            step *= 0.998
            
            # Periodic tracking
            if (it + 1) % 50 == 0:
                if val > best_sum:
                    best_sum = val
                    best_centers = curr_c.copy()
                    best_radii = radii.copy()
                    
            # Occasional noise to escape plateaus
            if it % 400 == 0 and it > 0:
                curr_c += rng.normal(0, 0.005, curr_c.shape)
                curr_c = np.clip(curr_c, 0.005, 0.995)
                
        # Local perturbation search around converged point
        for _ in range(5):
            pert = curr_c + rng.normal(0, 0.01, curr_c.shape)
            pert = np.clip(pert, 0.01, 0.99)
            radii, val, grad = solve_lp_and_grad(pert)
            if val > best_sum:
                curr_c = pert.copy()
                step = 0.008
                for it in range(1500):
                    radii, val, grad = solve_lp_and_grad(curr_c)
                    g_norm = np.linalg.norm(grad)
                    if g_norm < 1e-8: break
                    curr_c += step * grad / g_norm
                    curr_c = np.clip(curr_c, 0.005, 0.995)
                    step *= 0.998
                    if (it + 1) % 50 == 0:
                        if val > best_sum:
                            best_sum = val
                            best_centers = curr_c.copy()
                            best_radii = radii.copy()

    # Phase 2: SLSQP Polish with boundary-safe parameterization
    if best_centers is not None:
        r_init = np.clip(best_radii * 0.995, 1e-6, 0.49)
        p0 = to_params(best_centers, r_init)
        bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
        cons_dict = {'type': 'ineq', 'fun': cons_slqp}
        
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_centers, best_radii = from_params(res.x)
        except Exception:
            pass
            
        # Multiple restarts to escape SLSQP local minima
        for _ in range(15):
            p0 = to_params(best_centers, best_radii * 0.99)
            p0[N:3*N] += rng.normal(0, 0.02, 2*N)
            p0[:N] += rng.normal(0, 0.001, N)
            p0[:N] = np.clip(p0[:N], 1e-6, 0.49)
            p0[N:3*N] = np.clip(p0[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                               options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_centers, best_radii = from_params(res.x)
            except Exception:
                continue

    # Fallback (should rarely trigger)
    if best_centers is None:
        best_centers = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        best_radii = np.full(N, 0.04)
        best_sum = np.sum(best_radii)
        
    # Final exact LP solve to guarantee radii are optimal for precise centers
    final_r, final_sum, _ = solve_lp_and_grad(best_centers)
    return best_centers, np.maximum(final_r, 0.0), float(final_sum)
