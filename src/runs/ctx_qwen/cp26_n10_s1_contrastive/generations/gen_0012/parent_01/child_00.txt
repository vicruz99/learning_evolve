Key improvements in this strategy:
1. **Diverse & Rich Initialization**: Generate a larger pool of initial configurations including varied hexagonal row counts, rotated/scaled lattices, force-directed layouts with adaptive radius expansion, perturbed grids, and random starts. This increases the probability of hitting promising basins of attraction.
2. **LP-Preprocessed SLSQP**: For each initialization, first solve an LP to find the optimal radii for those fixed centers. This gives a feasible, tight starting point for SLSQP, drastically improving convergence speed and success rate.
3. **Aggressive Perturbation & Topology Escapes**: Instead of a slow simulated annealing walk, we use targeted perturbation restarts. 4. **Strict Constraint Handling**: The `(r, u, v)` parameterization inherently satisfies boundary constraints. We maintain this but tighten bounds and tolerances. import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 1e-10), -res.fun
    return np.full(n, 1e-8), 1e-6

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng, seed):
    """Generates a force-directed layout initialization with adaptive radius growth."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.1, 0.9, (N, 2))
    r_curr = np.full(N, 0.02)
    for step in range(600):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        rep = 1.0 / d**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        
        for dim in range(2):
            forces[:, dim] += 50.0 * np.maximum(0, r_curr - pts[:, dim])
            forces[:, dim] -= 50.0 * np.maximum(0, pts[:, dim] - (1.0 - r_curr))
            
        step_size = 0.006 * (0.9985**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        r_curr = np.maximum(r_curr * 1.0003, 0.02)
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_others[i] = np.inf
            d_pair = np.min(dists_others)
            r_curr[i] = 0.85 * min(d_wall, d_pair/2.0)
    return pts

def grid_init(rng, seed):
    """Generates a perturbed grid initialization."""
    rng_g = np.random.RandomState(seed)
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng_g.uniform(-0.025, 0.025, (N, 2))
    return np.clip(pts, 0.05, 0.95)

def get_score_tuple(t):
    """Helper to avoid lambda in sorting."""
    return t[0]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6],
        [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
        [5,5,4,6,6], [4,6,6,5,5], [6,5,5,6,4], [5,4,6,6,5], [4,5,6,6,5],
        [7,6,4,5,4], [6,7,5,4,4], [5,6,7,4,4], [8,5,5,5,3], [5,8,5,5,3],
        [6,5,7,5,3], [5,5,5,5,5,1], [7,7,5,5,2], [5,5,7,5,4]
    ]
    for pat in pats:
        for _ in range(3):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.35, 0.35), scale=rng.uniform(0.85, 1.15)))
            
    for s in range(25):
        inits.append(force_init(rng, s))
        
    for s in range(12):
        inits.append(grid_init(rng, s))
        
    for _ in range(15):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    init_scores = []
    for c in inits:
        r, s_val = solve_lp_radii(c)
        init_scores.append((s_val, c, r))
        
    init_scores.sort(key=get_score_tuple, reverse=True)
    
    # Phase 1: SLSQP refinement on diverse starts
    for s0, c0, r0 in init_scores:
        r_clip = np.clip(r0 * 0.995, 1e-6, 0.49)
        p0 = to_params(c0, r_clip)
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    if best_p is None:
        c_f, r_f = from_params(to_params(init_scores[0][1], init_scores[0][2] * 0.99))
        best_p = to_params(c_f, r_f * 0.99)
        best_sum = -obj_slqp(best_p)
        
    best_c, best_r = from_params(best_p)
    best_c = best_c.copy()
    best_r = np.maximum(best_r, 1e-8)
    
    # Phase 2: Aggressive Perturbation & SLSQP restarts
    for k in range(80):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.0015, N)
        xp[N:3*N] += rng.normal(0, 0.035, 2*N)
        
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
                    best_c, best_r = from_params(best_p)
        except Exception:
            continue
            
    # Phase 3: Topology change escapes
    for _ in range(35):
        c_pert = best_c.copy()
        n_move = rng.integers(3, 7)
        idxs = rng.choice(N, n_move, replace=False)
        c_pert[idxs] = rng.uniform(0.15, 0.85, (n_move, 2))
        r_pert, _ = solve_lp_radii(c_pert)
        r_pert = np.clip(r_pert * 0.995, 1e-6, 0.49)
        p_pert = to_params(c_pert, r_pert)
        try:
            res = minimize(obj_slqp, p_pert, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
                    best_c, best_r = from_params(best_p)
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-8:
            best_p = res_f.x
    except Exception:
        pass
        
    centers, radii = from_params(best_p)
    # Final LP to guarantee optimal radii for the precise final centers
    radii_final, sum_final = solve_lp_radii(centers)
    
    # Safety shrink to strictly pass validation tolerance
    radii_final = np.maximum(radii_final - 1e-9, 0.0)
    sum_final = np.sum(radii_final)
    
    return centers, radii_final, float(sum_final)
