import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-10), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-8), 1e-6

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6],
            [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
            [5,5,4,6,6], [4,6,6,5,5], [6,5,5,6,4], [5,4,6,6,5], [4,5,6,6,5],
            [7,6,4,5,4], [6,7,5,4,4], [5,6,7,4,4], [8,5,5,5,3], [5,5,5,5,5,1]]
    for pat in pats:
        for _ in range(4):
            pts = []
            r_est = 0.095
            y = r_est
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.85, 1.15) + 0.5
            rot = rng.uniform(-0.3, 0.3)
            c, s = np.cos(rot), np.sin(rot)
            M = np.array([[c, -s], [s, c]])
            pts = (pts - 0.5) @ M.T + 0.5
            pts += rng.uniform(-0.02, 0.02, pts.shape)
            inits.append(np.clip(pts, 0.02, 0.98))
            
    for _ in range(20):
        pts = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(400):
            f = np.zeros_like(pts)
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            d = np.sqrt(np.sum(diff**2, axis=2))
            d = np.maximum(d, 1e-4)
            f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
            for dim in range(2):
                f[:, dim] += 30.0 * np.maximum(0, 0.05 - pts[:, dim])
                f[:, dim] -= 30.0 * np.maximum(0, pts[:, dim] - 0.95)
            pts += 0.002 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    for _ in range(10):
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng.uniform(-0.025, 0.025, (N, 2))
        inits.append(np.clip(pts, 0.05, 0.95))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    inits = generate_inits(rng)
    init_scores = []
    for c in inits:
        r, s_val = solve_lp_radii(c)
        init_scores.append((s_val, c, r))
    init_scores.sort(reverse=True)
    
    # Phase 1: SLSQP on top inits
    for s0, c0, r0 in init_scores[:10]:
        r_clip = np.clip(r0 * 0.995, 1e-6, 0.49)
        p0 = to_params(c0, r_clip)
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Fallback initialization if Phase 1 fails completely
    if best_p is None:
        c_f = init_scores[0][1]
        r_f = init_scores[0][2]
        best_p = to_params(c_f, np.clip(r_f * 0.99, 1e-6, 0.49))
        best_sum = -obj_slqp(best_p)
        
    curr_c, curr_r = from_params(best_p)
    curr_c = curr_c.copy()
    curr_r = curr_r.copy()
    curr_s = best_sum
    best_c = curr_c.copy()
    best_r = curr_r.copy()
    
    temp = 0.05
    step = 0.04
    
    # Phase 2: SA on centers with periodic SLSQP checks to escape local minima
    for it in range(15000):
        move_type = rng.integers(0, 4)
        if move_type == 0:
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        elif move_type == 1:
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] = rng.uniform(0.1, 0.9, 2)
        elif move_type == 2:
            idx_pair = rng.permutation(N)[:2]
            i, j = idx_pair[0], idx_pair[1]
            old = curr_c[[i, j]].copy()
            curr_c[[i, j]] = curr_c[[j, i]]
        else:
            rot = rng.uniform(-0.05, 0.05)
            c_val, s_val = np.cos(rot), np.sin(rot)
            mat = np.array([[c_val, -s_val], [s_val, c_val]])
            old = curr_c.copy()
            curr_c = (curr_c - 0.5) @ mat.T + 0.5
            curr_c = np.clip(curr_c, 0.01, 0.99)
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
                step = min(0.05, step * 1.05)
            else:
                step = max(0.001, step * 0.998)
        else:
            if move_type < 3:
                if move_type < 2:
                    curr_c[idx] = old
                else:
                    curr_c[[i, j]] = old
            else:
                curr_c = old
            step = max(0.001, step * 0.98)
            
        temp *= 0.9995
        step = max(0.001, step * 0.9993)
        
        # Periodic SLSQP check to escape local minima using gradient information
        if it > 0 and it % 2000 == 0 and best_c is not None:
            r_clip = np.clip(best_r * 0.995, 1e-6, 0.49)
            p_check = to_params(best_c, r_clip)
            try:
                res_check = minimize(obj_slqp, p_check, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                                     options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if res_check.success and np.min(cons_slqp(res_check.x)) >= -1e-7:
                    s_check = -res_check.fun
                    if s_check > best_sum:
                        best_sum = s_check
                        best_p = res_check.x.copy()
                        best_c, best_r = from_params(best_p)
                        curr_c = best_c.copy()
                        curr_r = best_r.copy()
                        curr_s = best_sum
                        step = 0.02
            except Exception:
                pass
            
    if best_p is None and best_c is not None:
        best_p = to_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
        
    # Phase 3: Perturbation SLSQP restarts
    for k in range(40):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.0015, N)
        xp[N:3*N] += rng.normal(0, 0.02, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 4: Final high-precision polish
    try:
        res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-8:
            best_p = res_f.x
    except Exception:
        pass
        
    centers, radii = from_params(best_p)
    # Final exact LP solve to guarantee optimal radii for the precise final centers
    radii_final, sum_final = solve_lp_radii(centers)
    
    return centers, np.maximum(radii_final, 0.0), float(sum_final)
