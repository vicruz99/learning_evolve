import numpy as np
from scipy.optimize import minimize, linprog
from operator import itemgetter

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to maximize sum(r_i) for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def to_params(centers, radii):
    r = np.clip(radii, 1e-6, 0.49)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    return -np.sum(params[:N])

def cons_slqp(params):
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_inits(rng):
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [6,6,5,5,4],
        [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,5,6,5],
        [8,5,5,4,4], [5,5,5,5,5,1], [4,5,6,5,6]
    ]
    for pat in pats:
        for _ in range(3):
            pts = []
            r_est = 0.095
            y = r_est
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N: pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.85, 1.15) + 0.5
            rot = rng.uniform(-0.25, 0.25)
            c, s = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pts += rng.uniform(-0.012, 0.012, pts.shape)
            inits.append(np.clip(pts, 0.02, 0.98))

    # Force-directed layouts
    for s in range(15):
        rng_fd = np.random.RandomState(s)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        for _ in range(250):

            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-4
            f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
            pts += 0.0025 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # Perturbed grids & corner biased
    for s in range(15):
        rng_g = np.random.RandomState(s)

        pts += rng_g.uniform(-0.025, 0.025, (N, 2))
        inits.append(np.clip(pts, 0.05, 0.95))
        
    # Corner/Edge biased starts

        rng_c = np.random.RandomState(s)
        pts = np.zeros((N, 2))

        pts[22:] = rng_c.uniform(0.75, 0.95, 2)[:, None] + rng_c.uniform(0.3, 0.7, (4, 1))
        inits.append(pts)

    curr_c, curr_r = from_params(best_p)

    best_sa_r = curr_r.copy()
    best_sa_s = curr_s
    
    # Phase 2: LP-driven SA on centers with multi-move strategy
    temp = 0.045
    step = 0.035

        move_type = rng.integers(0, 4)
        old_config = None
        
        if move_type == 0:  # Shift one

            old_config = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        elif move_type == 1:  # Swap two

            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        elif move_type == 2:  # Random reposition
