import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to maximize sum(r_i) for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def obj_de(x_flat):
    """Objective for Differential Evolution: minimize negative sum of radii."""
    centers = x_flat.reshape(N, 2)
    _, s = solve_lp_radii(centers)
    return -s

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = np.clip(radii, 1e-6, 0.49)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints for SLSQP: pairwise non-overlap."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_hex_inits(rng, pats):
    """Generates hexagonal lattice initial configurations."""
    inits = []
    for pat in pats:
        for _ in range(2):
            pts = []
            r_est = 0.095
            y = r_est
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.9, 1.1) + 0.5
            rot = rng.uniform(-0.2, 0.2)
            c, s = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pts += rng.uniform(-0.01, 0.01, pts.shape)
            inits.append(np.clip(pts, 0.02, 0.98))
    return inits

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    
    # --- Phase 1: Differential Evolution on Centers ---
    de_bounds = [(0.0, 1.0)] * (2 * N)
    try:
        res_de = differential_evolution(
            obj_de, de_bounds, 
            popsize=20, maxiter=60, 
            mutation=(0.5, 1.0), recombination=0.7, 
            seed=42, polish=False
        )
        c_de = res_de.x.reshape(N, 2)
        r_de, s_de = solve_lp_radii(c_de)
        if s_de > best_sum:
            best_sum = s_de
            best_c = c_de
    except Exception:
        pass
        
    # --- Phase 2: Hexagonal Initializations + SLSQP ---
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [6,6,5,5,4]
    ]
    hex_inits = generate_hex_inits(rng, pats)
    for c0 in hex_inits:
        r0, s0 = solve_lp_radii(c0)
        if s0 > best_sum:
            best_sum = s0
            best_c = c0
            
    # SLSQP refinement on best found centers so far
    if best_c is not None:
        p0 = to_params(best_c, np.clip(best_c is not None and solve_lp_radii(best_c)[0] * 0.99, 1e-6, 0.49))
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
                    best_c, _ = from_params(best_p)
        except Exception:
            pass

    if best_c is None:
        best_c = np.random.rand(N, 2)
        best_c = np.clip(best_c, 0.02, 0.98)
        _, best_sum = solve_lp_radii(best_c)
        
    # --- Phase 3: Perturbation & SLSQP Escape Loop ---
    curr_p = to_params(best_c, np.clip(solve_lp_radii(best_c)[0] * 0.995, 1e-6, 0.49))

        xp = curr_p.copy()
        # Perturb relative positions more aggressively to change contact topology
        xp[N:3*N] += rng.normal(0, 0.015 + 0.01*k, 2*N)
        xp[:N] += rng.normal(0, 0.001, N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
                    curr_p = best_p.copy()
        except Exception:
            continue
            
    # --- Phase 4: High-Precision Final Polish ---
    if best_p is not None:
        try:
            res_f = minimize(obj_slqp, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                             options={'maxiter': 15000, 'ftol': 1e-15, 'disp': False})
            if res_f.success and np.min(cons_slqp(res_f.x)) >= -1e-9:
                best_p = res_f.x
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Reconstruct final centers
    if best_p is not None:
        best_c, _ = from_params(best_p)
        
    # Final exact LP solve to guarantee optimal radii for the precise final centers
    radii_final, sum_final = solve_lp_radii(best_c)
    
    return best_c, np.maximum(radii_final, 0.0), float(sum_final)
