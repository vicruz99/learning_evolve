import numpy as np
from scipy.optimize import minimize, linprog
import operator

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to maximize sum(r_i) for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def to_params(centers, radii):
    r = np.clip(radii, 1e-6, 0.49)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    return -np.sum(params[:N])

def cons_slqp(params):
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot, scale):
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N: pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    c_val, s_val = np.cos(rot), np.sin(rot)
    mat = np.array([[c_val, -s_val], [s_val, c_val]])
    pts = (pts - 0.5) @ mat.T + 0.5
    pts += rng.uniform(-0.012, 0.012, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng, seed):
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.rand(N, 2) * 0.8 + 0.1
    for _ in range(250):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-4
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def corner_init(rng):
    pts = []
    margin = 0.12
    # Corners
    pts.extend([[margin, margin], [1-margin, margin], [margin, 1-margin], [1-margin, 1-margin]])
    # Edge midpoints
    pts.extend([[0.5, margin], [0.5, 1-margin], [margin, 0.5], [1-margin, 0.5]])
    # Fill remaining with perturbed grid
    grid = np.array([[0.15 + 0.18*i, 0.15 + 0.18*j] for i in range(4) for j in range(4)])
    pts.extend(list(grid))
    pts = np.array(pts[:N]) + rng.uniform(-0.02, 0.02, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def run_optimization(cfg, rng):
    centers = cfg.copy()
    radii, curr_sum = solve_lp_radii(centers)
    best_c = centers.copy()
    best_r = radii.copy()
    best_s = curr_sum
    
    temp = 0.045
    step = 0.035
    
    for it in range(10000):
        move = rng.integers(0, 4)
        old = None
        if move == 0:
            idx = rng.integers(N)
            old = centers[idx].copy()
            centers[idx] += rng.normal(0, step, 2)
            centers[idx] = np.clip(centers[idx], 0.01, 0.99)
        elif move == 1:
            i, j = rng.choice(N, 2, replace=False)
            old = centers[[i, j]].copy()
            centers[i], centers[j] = centers[j].copy(), centers[i].copy()
        elif move == 2:
            idx = rng.integers(N)
            old = centers[idx].copy()
            centers[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            idx = rng.integers(N)
            old = centers[idx].copy()
            centers[idx] += rng.normal(0, step * 2.0, 2)
            centers[idx] = np.clip(centers[idx], 0.01, 0.99)
            
        nr, ns = solve_lp_radii(centers)
        delta = ns - curr_sum
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_sum = ns
            if ns > best_s:
                best_s = ns
                best_c = centers.copy()
                best_r = nr.copy()
        else:
            if old is not None:
                if move in (0, 2, 3):
                    centers[idx] = old
                else:
                    i, j = np.where(old.flatten() == centers[i].flatten())[0] # fallback, not ideal but safe
                    # Actually simpler: just revert indices
                    if move == 1:
                        centers[[i, j]] = old
            if move in (0, 2, 3):
                centers[idx] = old
            if move == 1:
                centers[[i, j]] = old
                
        temp *= 0.9994
        step = max(0.001, step * 0.9995)
        
    p0 = to_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    try:
        res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                       options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
            s_val = -res.fun
            if s_val > best_s:
                best_s = s_val
                best_c, best_r = from_params(res.x)
    except Exception:
        pass
        
    return best_c, best_r, best_s
