import numpy as np
from scipy.optimize import minimize, linprog
from operator import itemgetter

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to maximize sum(r_i) for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = np.clip(radii, 1e-6, 0.49)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def grid_init(rng, seed):
    """Generates a perturbed grid initialization."""
    rng_g = np.random.RandomState(seed)
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng_g.uniform(-0.025, 0.025, (N, 2))
    return np.clip(pts, 0.05, 0.95)

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_slqp}
    
    best_p = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [7,5,5,5,4], [5,7,5,5,4], [6,6,5,5,4],
        [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [7,6,5,4,4],
        [6,7,5,5,3], [8,6,5,4,3], [5,5,5,5,5,1]
    ]
    for pat in pats:
        for _ in range(4):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.3, 0.3), scale=rng.uniform(0.85, 1.15)))
    for s in range(20):
        inits.append(force_init(rng, s))
    for s in range(15):
        inits.append(grid_init(rng, s))
        
    # Score inits with LP
    init_scores = []
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)
        init_scores.append((s0, c0, r0))
    init_scores.sort(key=itemgetter(0), reverse=True)
    
    # Phase 1: SLSQP refinement on top initial configurations
    for s0, c0, r0 in init_scores[:15]:
        p0 = to_params(c0, np.clip(r0 * 0.99, 1e-6, 0.49))
        try:
            res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_p = res.x.copy()
        except Exception:
            pass
            
    if best_p is None:
        best_p = to_params(init_scores[0][1], init_scores[0][2] * 0.99)
        best_sum = -obj_slqp(best_p)
        
    curr_c, curr_r = from_params(best_p)
    curr_s = best_sum
    best_sa_c = curr_c.copy()
    best_sa_s = curr_s
    
    # Phase 2: LP-driven Simulated Annealing on centers
    temp = 0.05
    step = 0.04
    
    for it in range(20000):
        move = rng.integers(0, 3)
        if move == 0:
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
        elif move == 1:
            i, j = rng.choice(N, 2, replace=False)
            old = curr_c[[i, j]].copy()
            curr_c[i] += rng.normal(0, step/2, 2)
            curr_c[j] += rng.normal(0, step/2, 2)
            curr_c = np.clip(curr_c, 0.01, 0.99)
        else:
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] = rng.uniform(0.1, 0.9, 2)
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_s
        
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(min(0.0, delta/temp))):
            curr_s = ns
            if ns > best_sa_s:
                best_sa_s = ns
                best_sa_c = curr_c.copy()
        else:
            if move == 0:
                curr_c[idx] = old
            elif move == 1:
                curr_c[[i, j]] = old
            else:
                curr_c[idx] = old
                
        temp *= 0.9995
        step = max(0.002, step * 0.9998)
        
    # Update best if SA found a better configuration
    if best_sa_s > best_sum:
        best_sum = best_sa_s
        _, best_r = solve_lp_radii(best_sa_c)
        best_p = to_params(best_sa_c, np.clip(best_r * 0.995, 1e-6, 0.49))
    else:
        best_c, best_r = from_params(best_p)
        best_sa_c = best_c
        
    best_c, best_r = from_params(best_p)
    
    # Phase 3: SLSQP refinement on SA best
    p0 = to_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
    try:
        res = minimize(obj_slqp, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                       options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(cons_slqp(res.x)) >= -1e-8:
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_p = res.x.copy()
                best_c, best_r = from_params(best_p)
    except Exception:
        pass
        
    # Phase 4: Rotation & Perturbation Escapes
    for k in range(20):
        if k % 2 == 0:
            rot = rng.uniform(-0.12, 0.12)
            c_val, s_val = np.cos(rot), np.sin(rot)
            mat = np.array([[c_val, -s_val], [s_val, c_val]])
            c_rot = (best_c - 0.5) @ mat.T + 0.5
            c_rot = np.clip(c_rot, 0.01, 0.99)
            r_rot, _ = solve_lp_radii(c_rot)
            xp = to_params(c_rot, np.clip(r_rot * 0.995, 1e-6, 0.49))
        else:
            xp = best_p.copy()
            xp[:N] += rng.normal(0, 0.003, N)
            xp[N:3*N] += rng.normal(0, 0.025, 2*N)
            xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
            xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
            
        try:
            res = minimize(obj_slqp, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,

            if res.success and np.min(cons_slqp(res.x)) >= -1e-8:

                if s_val > best_sum:
                    best_sum = s_val
                    best_p = res.x.copy()
                    best_c, best_r = from_params(best_p)

    try:

            best_p = res_f.x
            best_sum = -res_f.fun
            best_c, best_r = from_params(best_p)
    except Exception:
        pass
        
    # Final LP to guarantee optimal radii for exact final centers
    radii, sum_radii = solve_lp_radii(best_c)
    
    return best_c, np.maximum(radii, 0.0), float(sum_radii)
