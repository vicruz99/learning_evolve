import numpy as np
from scipy.optimize import minimize, linprog
import operator

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_list = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_list:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_func(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def generate_hex_centers(rng, row_counts, rot, scale, shear=0.0):
    """Generates a hexagonal lattice initialization with shear transformation."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    
    if abs(shear) > 1e-6:
        shear_mat = np.array([[1, shear], [0, 1]])
        pts = (pts - 0.5) @ shear_mat.T + 0.5
    
    pts = (pts - 0.5) * scale + 0.5
    
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_corner_heavy(rng, seed):
    """Places circles preferentially in corners and edges where they can grow larger."""
    rng_c = np.random.RandomState(seed)
    pts = []
    margin = 0.12 + rng_c.uniform(-0.02, 0.02)
    
    # 4 corner circles (can be largest)
    pts.append([margin, margin])
    pts.append([1-margin, margin])
    pts.append([margin, 1-margin])
    pts.append([1-margin, 1-margin])
    
    # 12 edge circles (second-largest)
    n_edge = 12
    for _ in range(n_edge):
        side = rng_c.integers(4)
        if side == 0:
            pts.append([rng_c.uniform(0.15, 0.85), margin])
        elif side == 1:
            pts.append([rng_c.uniform(0.15, 0.85), 1-margin])
        elif side == 2:
            pts.append([margin, rng_c.uniform(0.15, 0.85)])
        else:
            pts.append([1-margin, rng_c.uniform(0.15, 0.85)])
    
    # Remaining in bulk (smallest)
    n_bulk = N - len(pts)
    for _ in range(n_bulk):
        pts.append([rng_c.uniform(0.2, 0.8), rng_c.uniform(0.2, 0.8)])
        
    pts = np.array(pts[:N])
    pts += rng_c.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_centers(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.uniform(0.1, 0.9, (N, 2))
    for _ in range(400):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-6
        f = np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 25.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 25.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.002 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts
