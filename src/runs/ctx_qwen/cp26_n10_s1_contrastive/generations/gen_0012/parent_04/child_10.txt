import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
_BOUNDS = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N

np.seterr(all='ignore')

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def get_params(centers, radii):
    """Map physical centers/radii to SLSQP parameter vector (x, y, r)."""
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = radii
    return v

def obj_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def cons_slqp(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def hex_init(rng, rows, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095 * scale
    y = r_est
    row = 0
    for cnt in rows:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
        
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    rng = np.random.default_rng(42)
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    pats = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6],
        [6,6,5,5,4], [5,5,6,5,5], [6,5,5,5,5], [5,5,5,6,5], [6,4,6,5,5],
        [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3], [5,5,5,5,5,1], [4,5,6,5,6],
        [5,5,5,4,7], [4,4,6,6,6], [6,5,5,6,4]
    ]
    
    for pat in pats:
        for _ in range(4):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.25, 0.25), scale=rng.uniform(0.9, 1.1)))
            
    # Force-directed organic layouts
    for s in range(15):
        rng_fd = np.random.RandomState(s)
        pts = rng_fd.uniform(0.1, 0.9, (N, 2))

            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            d = np.sqrt(np.sum(diff**2, axis=2))
            d = np.maximum(d, 1e-4)
            f = np.sum((1.0/d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
            for dim in range(2):
                f[:, dim] += 15.0 * np.maximum(0, 0.1 - pts[:, dim])
                f[:, dim] -= 15.0 * np.maximum(0, pts[:, dim] - 0.9)
            pts += 0.003 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # Evaluate all inits with LP to find the strongest starting topology
    for c0 in inits:

        if s0 > best_sum:
            best_sum = s0
            best_c = c0.copy()
            best_r = r0.copy()
            
    curr_c = best_c.copy()
    curr_r, curr_s = solve_lp_radii(curr_c)

            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            i, j = rng.choice(N, 2, replace=False)

        delta = ns - curr_s

            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
                step = min(0.05, step * 1.02)
            else:
                step = max(0.002, step * 0.996)
        else:
            if move_type < 2:
                curr_c[idx] = old
            else:
                curr_c[[i, j]] = old
            step = max(0.002, step * 0.996)
            
        temp *= 0.9994
        
        # Periodic SLSQP polish during SA to tighten contact graphs
        if it % 500 == 0 and best_c is not None:
            v0 = get_params(best_c, np.maximum(best_r * 0.995, 1e-6))
            try:
                res = minimize(obj_slqp, v0, method='SLSQP', bounds=_BOUNDS, constraints=cons_dict,
                               options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                if res.success and np.min(cons_slqp(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_r = res.x[2::3]
                        curr_c = best_c.copy()
                        curr_s = best_sum
            except Exception:
                pass
                
    # Phase 3: Post-SA Perturbation & SLSQP Restarts
    if best_c is not None:

            v0 = get_params(best_c, np.maximum(best_r * 0.995, 1e-6))
            v0[0::3] += rng.uniform(-0.008, 0.008, N)
            v0[1::3] += rng.uniform(-0.008, 0.008, N)
            v0[0::3] = np.clip(v0[0::3], 0.01, 0.99)
            v0[1::3] = np.clip(v0[1::3], 0.01, 0.99)
            
            try:
                res = minimize(obj_slqp, v0, method='SLSQP', bounds=_BOUNDS, constraints=cons_dict,
                               options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})

        c_rot = np.clip(c_rot, 0.01, 0.99)
        r_rot, _ = solve_lp_radii(c_rot)
        v0 = get_params(c_rot, np.maximum(r_rot * 0.995, 1e-6))
        try:
            res_r = minimize(obj_slqp, v0, method='SLSQP', bounds=_BOUNDS, constraints=cons_dict,
