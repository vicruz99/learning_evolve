import numpy as np
from scipy.optimize import linprog, minimize

N = 26
_MASK = np.triu(np.ones((N, N), dtype=bool), k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        bounds_list = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_list:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def obj_slqp(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[:N])

def cons_slqp(v):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = v[:N]
    u = v[N:2*N]
    v2 = v[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v2 * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[_MASK] - rs[_MASK]**2

def run_packing():
    np.random.seed(42)
    best_c = None
    best_r = None
    best_sum = -np.inf

    def hex_init(rows, rot, scale):

        r_est = 0.095
        y = r_est
        row = 0
        for cnt in rows:
            shift = (row % 2) * r_est
            x = r_est + shift
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3.0) * r_est
            row += 1
        pts = np.array(pts[:N])
        pts = (pts - 0.5) * scale + 0.5
        if abs(rot) > 1e-6:
            c, s = np.cos(rot), np.sin(rot)
            M = np.array([[c, -s], [s, c]])
            pts = (pts - 0.5) @ M.T + 0.5
        pts += np.random.uniform(-0.012, 0.012, pts.shape)
        return np.clip(pts, 0.02, 0.98)

    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], [6,6,5,5,4], [5,5,6,5,5]]
    for p in pats:
        for s in range(4):
            inits.append(hex_init(p, rot=np.random.uniform(-0.25, 0.25), scale=np.random.uniform(0.88, 1.12)))
            
    # Initialization 2: Force-directed layouts
    for s in range(15):
        rng_fd = np.random.RandomState(s)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1

            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]

    u_init = np.clip((best_c[:, 0] - r_init) / denom, 0.0, 1.0)

        p_test[0:N] = np.clip(p_test[0:N], 1e-6, 0.49)
        p_test[N:3*N] = np.clip(p_test[N:3*N], 0.0, 1.0)

                    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
                    best_c = np.column_stack((x_opt, y_opt))
                    best_r = r_opt
        except Exception:
            continue
            
    # Phase 4: Final LP polish to guarantee consistency and strict feasibility
    final_r, final_sum = solve_lp_radii(best_c)
    return best_c, final_r, float(final_sum)
