    n = centers.shape[0]
    c_obj = -np.ones(n)

    b_ub = np.zeros(num_c)

    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1

    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]

            k += 1

            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-4

def to_params(centers, radii):

    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):

    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def obj_param(params):

    return -np.sum(params[:N])

def cons_param(params):

    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]

    y = r + v * (1.0 - 2.0 * r)

    dy = y[:, np.newaxis] - y[np.newaxis, :]

    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def obj_direct(v):
    """Objective for direct (x,y,r) optimization."""
    return -np.sum(v[2::3])

def cons_direct(v):
    """Explicit inequality constraints for direct optimization."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)

    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
