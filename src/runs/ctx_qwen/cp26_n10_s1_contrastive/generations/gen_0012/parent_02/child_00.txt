import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-4

def obj_slqp(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons_slqp(params):
    """Inequality constraints using (r, u, v) parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def obj_direct(params):
    """Objective for direct (x,y,r) optimization."""
    return -np.sum(params[2::3])

def cons_direct(params):
    """Constraints for direct (x,y,r) optimization."""
    x = params[0::3]
    y = params[1::3]
    r = params[2::3]
    
    constraints = []
    constraints.append(x - r)
    constraints.append(1.0 - x - r)
    constraints.append(y - r)
    constraints.append(1.0 - y - r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    constraints.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    
    return np.concatenate(constraints)

def to_ruv_params(centers, radii):
    """Convert centers/radii to (r, u, v) parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_ruv_params(params):
    """Convert (r, u, v) parameters to centers/radii."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def to_direct_params(centers, radii):
    """Convert centers/radii to direct (x, y, r) parameters."""
    params = np.empty(3 * N)
    params[0::3] = centers[:, 0]
    params[1::3] = centers[:, 1]
    params[2::3] = radii
    return params

def from_direct_params(params):
    """Convert direct (x, y, r) parameters to centers/radii."""
    x = params[0::3]
    y = params[1::3]
    r = params[2::3]
    return np.column_stack([x, y]), r

def hex_init(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if abs(rot) > 1e-6:
        c_val, s_val = np.cos(rot), np.sin(rot)
        M = np.array([[c_val, -s_val], [s_val, c_val]])
        pts = (pts - 0.5) @ M.T + 0.5
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    r_curr = np.full(N, 0.05)
    
    for step in range(400):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)
        
        for d in range(2):
            forces[:, d] += 30.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        step_size = 0.008 * (0.992**step)
        pts += step_size * forces
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
        
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_others[i] = np.inf
            d_pair = np.min(dists_others)
            r_curr[i] = 0.75 * min(d_wall, d_pair/2.0)
    return pts

def grid_init(rng):
    """Generates a perturbed grid initialization."""
    pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    pts = np.vstack([pts, [0.5, 0.5]])
    pts += rng.uniform(-0.03, 0.03, (N, 2))
    return np.clip(pts, 0.05, 0.95)

def corner_init(rng):
    """Generates a corner-focused initialization."""
    pts = []
    # Place circles near corners
    corners = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
    for c in corners:
        pts.append(c)
    
    # Place circles along edges
    for t in np.linspace(0.15, 0.85, 6):
        pts.append([t, 0.12])
        pts.append([t, 0.88])
        pts.append([0.12, t])
        pts.append([0.88, t])
    
    # Fill remaining with center-focused
    while len(pts) < N:
        pts.append([0.5 + rng.uniform(-0.3, 0.3), 0.5 + rng.uniform(-0.3, 0.3)])
    
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.02, 0.02, (N, 2))
    return np.clip(pts, 0.02, 0.98)
