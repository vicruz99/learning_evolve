

<strategy>
1. **LP-based Radius Optimization**: For any fixed set of centers, finding the maximum sum of radii is a Linear Programming problem. This is exact and avoids the non-convexity of joint optimization.
2. **Simulated Annealing on Centers**: We use SA to explore the configuration space of centers. The objective function is the LP-optimized sum of radii. SA handles discrete changes in the contact graph (topology) well. We include both displacement moves and swap moves to escape local minima.
3. **Diverse Initializations**: Generate multiple hexagonal lattice patterns with varying rotations, scales, and force-directed layouts to seed the SA process.
4. **Gradient-Based Refinement**: After SA finds a high-quality configuration, we use SLSQP with a boundary-safe parameterization $(r, u, v)$ to polish the solution, maximizing the sum of radii simultaneously for centers and radii.
5. **Perturbation Restarts**: Run SLSQP multiple times with small random perturbations to escape any remaining local optima.
6. **High-Precision Polish**: Final SLSQP run with tight tolerance to squeeze out the last bits of precision.
</strategy>
```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def get_params(centers, radii):
    """Convert physical (centers, radii) to optimization parameters (r, u, v)."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def get_centers_radii(params):
    """Convert optimization parameters (r, u, v) back to physical (centers, radii)."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def hex_init(seed, row_counts, rot=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Generates a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        rep = 1.0 / (d**2)
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, None] * diff / d[:, :, None], axis=1)
        for dim in range(2):
            f[:, dim] += 30.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 30.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.006 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    np.random.seed(42)
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    # Generate initial configurations
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5],
            [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7], [6,6,5,5,4], [5,5,6,5,5]]
    for i, p in enumerate(pats):
        for s in range(5):
            inits.append(hex_init(i*5+s, p, rot=np.random.uniform(-0.3, 0.3), scale=np.random.uniform(0.85, 1.15)))
    for s in range(10):
        inits.append(force_init(s))
        
    # Phase 1: Evaluate inits with LP
    for pts in inits:
        r, s_val = solve_lp_radii(pts)
        if s_val > best_sum:
            best_sum = s_val
            best_c = pts.copy()
            best_r = r.copy()
            
    # Phase 2: Simulated Annealing on centers
    curr_c = best_c.copy()
    curr_sum = best_sum
    temp = 0.04
    step = 0.03
    
    for it in range(10000):
        # Refresh step size occasionally to escape deep basins
        if it > 0 and it % 2500 == 0:
            step = 0.035
            
        do_swap = np.random.rand() < 0.15
        if do_swap:
            i, j = np.random.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        else:
            idx = np.random.randint(N)
            old_pos = curr_c[idx].copy()
            curr_c[idx] += np.random.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_sum
        
        if delta > 0 or (temp > 1e-8 and np.random.rand() < np.exp(min(0.0, delta / temp))):
            curr_sum = ns
            if ns > best_sum:
                best_sum = ns
                best_c = curr_c.copy()
                best_r = nr.copy()
        else:
            if do_swap:
                curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
            else:
                curr_c[idx] = old_pos
                
        temp *= 0.9992
        step = max(0.0015, step * 0.9994)
        
    # Phase 3: SLSQP refinement
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * (2 * N)
    cons_slqp = {'type': 'ineq', 'fun': cons}
    best_params = None
    
    x0 = get_params(best_c, np.maximum(best_r * 0.995, 1e-6))
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds_slqp,
                       constraints=cons_slqp, options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
        if res.success and np.min(cons(res.x)) >= -1e-7:
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_params = res.x
    except Exception:
        pass
        
    # Phase 4: Perturbation & SLSQP restarts
    if best_params is not None:
        for k in range(20):
            xp = best_params.copy()
            xp[:N] += np.random.uniform(-0.001, 0.001, N)
            xp[N:3*N] += np.random.uniform(-0.01, 0.01, 2 * N)
            xp[:N] = np.clip(xp[:N], 1e-6, 0.5)
            xp[N:] = np.clip(xp[N:], 0.0, 1.0)
            
            try:
                res = minimize(obj, xp, method='SLSQP', bounds=bounds_slqp,
                               constraints=cons_slqp, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if res.success and np.min(cons(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x
            except Exception:
                pass
                
        # Final high-precision polish
        try:
            res_f = minimize(obj, best_params, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons_slqp, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(cons(res_f.x)) >= -1e-8:
                best_sum = -res_f.fun
                best_params = res_f.x
        except Exception:
            pass
            
        if best_params is not None:
            best_c, best_r = get_centers_radii(best_params)
    else:
        # Fallback to SA best if SLSQP failed completely
        best_c = best_c.copy()
        best_r = best_r.copy()
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
```
