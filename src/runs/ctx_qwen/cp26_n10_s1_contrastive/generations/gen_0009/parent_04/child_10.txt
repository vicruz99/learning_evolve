import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraint_func(params):
    """Inequality constraints: pairwise non-overlap. Boundary handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[I_IDX] + r[J_IDX]
    return dist_sq - r_sum**2

def objective_func(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def generate_hex_init(rng, pattern, rot, scale):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_sa(centers, rng, num_steps=8000, init_temp=0.04):
    """Simulated Annealing on centers maximizing LP-optimal sum of radii."""
    n = centers.shape[0]
    r, curr_sum = solve_lp(centers)
    best_c = centers.copy()
    best_r = r.copy()
    best_sum = curr_sum
    
    temp = init_temp
    step = 0.035
    
    for it in range(num_steps):
        if rng.random() < 0.12:
            i, j = rng.choice(n, 2, replace=False)
            centers[i], centers[j] = centers[j].copy(), centers[i].copy()
            is_swap = True
        else:
            idx = rng.integers(n)
            old = centers[idx].copy()
            step_cur = step * (4.0 if rng.random() < 0.03 else 1.0)
            centers[idx] += rng.normal(0, step_cur, 2)
            centers[idx] = np.clip(centers[idx], 0.008, 0.992)
            is_swap = False
            
        new_r, new_sum = solve_lp(centers)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = centers.copy()
                best_r = new_r.copy()
        else:
            if is_swap:
                centers[i], centers[j] = centers[j].copy(), centers[i].copy()
            else:
                centers[idx] = old
                
        temp *= 0.9992
        step = max(0.0015, step * 0.9994)
        
    return best_c, best_r, best_sum

def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * (2 * N)
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], 
        [5,5,5,5,6], [6,4,6,5,5], [5,7,5,5,4], [4,5,5,5,7],
        [6,6,5,5,4], [5,5,6,5,5], [7,6,5,4,4], [8,6,5,4,3]
    ]
    
    # Phase 1: SA from diverse hex inits
    for p in patterns:
        for _ in range(3):
            rot = rng.uniform(-0.3, 0.3)
            sc = rng.uniform(0.9, 1.1)
            init_c = generate_hex_init(rng, p, rot, sc)
            c, r, s = run_sa(init_c, rng)
            if s > best_sum:
                best_sum = s
                best_c = c
                best_r = r
                
    # Phase 2: SLSQP polish
    if best_c is not None:
        p0 = get_params(best_c, best_r * 0.995)
        try:
            res = minimize(objective_func, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_c, best_r = from_params(res.x)
        except Exception:
            pass
            
    # Phase 3: Perturbation + SLSQP restarts to escape local minima
    if best_c is not None:
        for _ in range(25):
            p0 = get_params(best_c, best_r * 0.998)
            p0[:N] += rng.normal(0, 0.0015, N)
            p0[N:3*N] += rng.normal(0, 0.012, 2*N)
            p0[:N] = np.clip(p0[:N], 1e-6, 0.5)
            p0[N:] = np.clip(p0[N:], 0.0, 1.0)
            
            try:
                res = minimize(objective_func, p0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_c, best_r = from_params(res.x)
            except Exception:
                pass
                
    # Phase 4: Final high-precision polish
    if best_c is not None:
        p0 = get_params(best_c, best_r * 0.999)
        try:
            res = minimize(objective_func, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-9:
                best_c, best_r = from_params(res.x)
                best_sum = np.sum(best_r)
        except Exception:
            pass
            
    # Fallback configuration (should rarely be reached)
    if best_c is None:
        best_c = generate_hex_init(rng, [6,5,6,5,4], 0.0, 1.0)
        best_r, best_sum = solve_lp(best_c)
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
