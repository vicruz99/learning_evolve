import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        vals = (x, 1.0 - x, y, 1.0 - y)
        for v in vals:
            A_ub[k, i] = 1.0
            b_ub[k] = v
            k += 1
            
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def get_centers_radii(params):
    """Reconstruct physical centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraint_func(params):
    """Inequality constraints: pairwise non-overlap. Boundary handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    dist_sq = dx**2 + dy**2
    r_sum = r[I_IDX] + r[J_IDX]
    return dist_sq - r_sum**2

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def hex_init(seed, pattern, rot, scale, jitter):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ R.T + 0.5
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(seed):
    """Generate a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dist**2
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, None] * diff / dist[:, :, None], axis=1)
        for d in range(2):
            f[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            f[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
        pts += 0.005 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def corner_init(seed):
    """Generate a corner/edge-biased initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    pts.extend([[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]])
    pts.extend([[0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5]])
    pts.append([0.5, 0.5])
    pts.extend([[0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75]])
    while len(pts) < N:
        pts.append(rng.uniform(0.2, 0.8, 2))
    pts = np.array(pts[:N])
    pts += rng.uniform(-0.03, 0.03, pts.shape)
    return np.clip(pts, 0.02, 0.98)

    best_r = None

            sc = rng.uniform(0.85, 1.15)

            inits.append(hex_init(s, p, rot, sc, jitt))

        inits.append(corner_init(s))
        
    # Phase 1: LP on inits to find a strong baseline
    for pts in inits:

        if s_val > best_sum:
            best_sum = s_val
            best_c = pts.copy()
            best_r = r.copy()

    if best_c is not None:
        curr_c = best_c.copy()
        curr_sum = best_sum

                    step = max(0.0015, step * 0.998)

                continue
                
            idx = rng.integers(N)

            cur_step = step * (4.0 if rng.random() < 0.02 else 1.0)

        best_r, best_sum = solve_lp_radii(best_c)
        
    # Fallback configuration
