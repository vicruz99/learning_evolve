import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds_r = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraint_func(params):
    """
    Inequality constraints: g(params) >= 0.
    Parameterization automatically satisfies boundary constraints.
    Only pairwise non-overlap is enforced.
    """
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Map parameters back to physical coordinates
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    dist_sq = dx * dx + dy * dy
    
    r_sum = r[I_IDX] + r[J_IDX]
    return dist_sq - r_sum * r_sum

def to_params(centers, radii):
    """Convert physical (centers, radii) to optimization parameters (r, u, v)."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Convert optimization parameters (r, u, v) back to physical (centers, radii)."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def generate_hex_init(rng, pattern, rot, scale):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(pattern):
        shift = (r_idx % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def generate_force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        rep = 1.0 / d**2
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.05 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.95)
        pts += 0.005 * f
        pts = np.clip(pts, 0.02, 0.98)
    return pts

    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).

    cons_opt = {'type': 'ineq', 'fun': constraint_func}
    
    best_params = None
    best_sum = -np.inf
    
    # Generate initial configurations
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [5,5,6,5,5], [7,5,5,5,4], [4,5,5,5,7], [5,7,5,5,4],
        [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3], [5,5,5,5,5,1]
    ]
    
    for p in patterns:
        for _ in range(3):
            rot = rng.uniform(-0.25, 0.25)
            sc = rng.uniform(0.90, 1.10)
            inits.append(generate_hex_init(rng, p, rot, sc))
            
    for _ in range(10):
        inits.append(generate_force_init(rng))
        
    # Phase 1: SLSQP on diverse starts
    for pts in inits:
        r_init, _ = solve_lp_radii(pts)
        r_init = np.clip(r_init * 0.985, 1e-6, 0.49)
        x0 = to_params(pts, r_init)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons_opt, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                if np.min(constraint_func(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            continue
            
    if best_params is None:
        # Fallback
        pts = inits[0]
        r_init, _ = solve_lp_radii(pts)

    # Phase 2: LP-driven Simulated Annealing on centers
    centers, radii = from_params(best_params)
    curr_sum = best_sum
    temp = 0.04
    step = 0.035

        old_c = centers.copy()

            centers[idx] += rng.normal(0, step, 2)
            centers[idx] = np.clip(centers[idx], 0.005, 0.995)
        else:
            i, j = rng.choice(N, 2, replace=False)
            centers[i], centers[j] = centers[j].copy(), centers[i].copy()
            
        r_new, s_new = solve_lp_radii(centers)

            if s_new > best_sum:
                best_sum = s_new
                best_params = to_params(centers, np.clip(radii * 0.995, 1e-6, 0.49))
        else:
            centers = old_c
            
        temp *= 0.9993
        step = max(0.0015, step * 0.9995)
        
    # Phase 3: Refine SA best with SLSQP
    try:
        res = minimize(objective, best_params, method='SLSQP', bounds=bounds_opt,
                       constraints=cons_opt, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(constraint_func(res.x)) >= -1e-8:
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_params = res.x.copy()
    except Exception:
        pass
        
    # Phase 4: Perturbation & SLSQP restarts to escape local minima
    for k in range(25):
        x0 = best_params.copy()
        x0[:N] += rng.uniform(-0.002, 0.002, N)
        x0[N:3*N] += rng.uniform(-0.02, 0.02, 2 * N)

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons_opt, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            continue
            
    # Phase 5: High-precision final polish
    try:
        res = minimize(objective, best_params, method='SLSQP', bounds=bounds_opt,
                       constraints=cons_opt, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        if res.success and np.min(constraint_func(res.x)) >= -1e-9:
            best_params = res.x
            best_sum = -res.fun
    except Exception:
        pass
        
    centers, radii = from_params(best_params)
    return centers, np.maximum(radii, 0.0), float(best_sum)
