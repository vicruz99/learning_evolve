

Looking at the past solutions, the best achieves ~2.636. Key patterns that work:
1. **LP-based radius optimization** (fix centers, LP for radii) is highly effective
2. **Simulated Annealing on centers** with LP evaluation escapes local minima well
3. **Diverse hex lattice patterns** provide good starting points
4. **Multi-phase refinement** (SA → SLSQP → perturb → polish) is needed

My strategy:
- Add many more hex patterns and force-directed inits
- Run longer SA with both move and swap operations, plus occasional large jumps
- Add rotation-based exploration (rotate all circles to find better wall contacts)
- Multiple SLSQP refinement passes with increasing precision
- Final LP polish to ensure radii are optimal for final centers

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)


def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    k = 0

    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1

    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)

    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1

    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4


def get_params(centers, radii):
    """Convert physical (centers, radii) to optimization parameters (r, u, v)."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])


def get_physical(params):
    """Convert optimization parameters (r, u, v) back to physical (centers, radii)."""
    r = params[:N]
    u = params[N:2 * N]
    v = params[2 * N:3 * N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r


def constraint_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2 * N]
    v = params[2 * N:3 * N]

    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)

    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2

    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2


def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])


def hex_init(rng, pattern, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1

    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5

    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5

    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)


def force_init(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.rand(N, 2) * 0.8 + 0.1
    r_curr = np.full(N, 0.05)

    for step in range(600):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, np.newaxis] * diff / dists[:, :, np.newaxis], axis=1)

        for d in range(2):
            forces[:, d] += 30.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))

        step_size = 0.006 * (0.994**step)
        pts += step_size * forces
        pts = np.clip(pts, 0.02, 0.98)

        for i in range(N):
            d_wall = min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])
            dists_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_others[i] = np.inf
            d_pair = np.min(dists_others)
            r_curr[i] = 0.8 * min(d_wall, d_pair / 2.0)

    return pts


def rotate_centers(centers, angle):
    """Rotate all centers around (0.5, 0.5)."""
    c, s = np.cos(angle), np.sin(angle)
    M = np.array([[c, -s], [s, c]])
    shifted = centers - 0.5
    rotated = shifted @ M.T + 0.5
    return np.clip(rotated, 0.01, 0.99)


def run_packing():
    """
    Solves the circle packing problem for N=26 in a unit square.
    Returns (centers, radii, sum_radii).
    """
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}

    best_params = None
    best_sum = -np.inf
    best_centers = None
    best_radii = None

    # Generate diverse initial configurations
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [4, 6, 5, 6, 5], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [5, 6, 4, 6, 5], [4, 5, 6, 5, 6], [6, 6, 5, 5, 4],
        [7, 5, 5, 5, 4], [4, 5, 5, 5, 7], [5, 7, 5, 5, 4], [6, 5, 5, 5, 5],
        [5, 5, 5, 6, 5], [7, 6, 5, 4, 4], [6, 7, 5, 5, 3], [8, 6, 5, 4, 3],
        [5, 5, 5, 5, 5, 1], [6, 5, 5, 6, 3], [5, 6, 5, 5, 5], [6, 6, 6, 5, 3],
        [5, 5, 7, 5, 4], [6, 5, 7, 4, 4], [7, 5, 6, 5, 3], [4, 7, 6, 5, 4],
        [5, 4, 7, 5, 5], [6, 4, 7, 6, 3], [4, 6, 7, 5, 4], [5, 5, 6, 6, 4],
        [6, 5, 6, 4, 5], [5, 6, 6, 5, 4], [4, 6, 6, 6, 4], [6, 6, 4, 6, 4],
        [5, 4, 6, 6, 5], [4, 5, 6, 6, 5], [5, 6, 4, 5, 6], [6, 4, 5, 6, 5],
        [5, 5, 4, 6, 6], [4, 6, 5, 5, 6], [6, 5, 4, 6, 5], [5, 6, 5, 4, 6],
        [7, 4, 5, 6, 4], [4, 7, 5, 6, 4], [5, 7, 4, 5, 5], [4, 5, 7, 5, 5]
    ]

    inits = []
    for pat in patterns:
        for _ in range(3):
            rot = rng.uniform(-0.45, 0.45)
            sc = rng.uniform(0.80, 1.20)
            inits.append(hex_init(rng, pat, rot, sc))

    for seed in range(30):
        inits.append(force_init(rng, seed))

    # Phase 1: SLSQP from diverse starts
    for pts in inits:
        r_init, _ = solve_lp_radii(pts)
        r_init = np.clip(r_init * 0.99, 1e-6, 0.49)
        x0 = get_params(pts, r_init)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                if np.min(constraint_func(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            continue

    if best_params is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)
        best_params = get_params(c_f, r_f)
        best_sum = np.sum(r_f)
        best_centers, best_radii = get_physical(best_params)
    else:
        best_centers, best_radii = get_physical(best_params)

    # Phase 2: Long LP-driven Simulated Annealing with swaps
    curr_c = best_centers.copy()
    curr_r, curr_sum = solve_lp_radii(curr_c)
    temp = 0.05
    step = 0.04

    for it in range(12000):
        op = rng.choice(['move', 'move', 'move', 'move', 'swap'])

        if op == 'swap':
            i, j = rng.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        else:
            idx = rng.integers(N)
            old = curr_c[idx].copy()

            if it % 2500 == 0 or rng.random() < 0.015:
                curr_c[idx] = rng.uniform(0.05, 0.95, 2)
            else:
                curr_c[idx] += rng.normal(0, step, 2)
                curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)

        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_sum

        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
            curr_sum = ns
            if ns > best_sum:
                best_sum = ns
                best_centers = curr_c.copy()
                best_radii = nr.copy()
        else:
            if op != 'swap':
                curr_c[idx] = old

        temp *= 0.9991
        step = max(0.0015, step * 0.9994)

    best_radii, best_sum = solve_lp_radii(best_centers)

    # Phase 3: Rotation exploration
    for rot_deg in range(0, 360, 5):
        angle = np.radians(rot_deg)
        c_rot = rotate_centers(best_centers, angle)
        r_rot, s_rot = solve_lp_radii(c_rot)
        if s_rot > best_sum:
            best_sum = s_rot
            best_centers = c_rot.copy()
            best_radii = r_rot.copy()

    # Phase 4: SLSQP refinement after SA
    best_params = get_params(best_centers, best_radii * 0.995)
    try:
        res = minimize(objective, best_params, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(constraint_func(res.x)) >= -1e-7:
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_params = res.x.copy()
    except Exception:
        pass

    # Phase 5: Perturbation escapes
    for k in range(40):
        x0 = best_params.copy()
        x0[:N] += rng.normal(0, 0.003, N)
        x0[N:3 * N] += rng.normal(0, 0.02, 2 * N)
        x0[:N] = np.clip(x0[:N], 1e-6, 0.49)
        x0[N:3 * N] = np.clip(x0[N:3 * N], 0.0, 1.0)

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            continue

    # Phase 6: Multi-point perturbation (move 2-3 circles at once)
    best_centers, best_radii = get_physical(best_params)
    for k in range(30):
        c_pert = best_centers.copy()
        num_perturb = rng.integers(2, 4)
        idxs = rng.choice(N, num_perturb, replace=False)
        for idx in idxs:
            c_pert[idx] += rng.normal(0, 0.015, 2)
            c_pert[idx] = np.clip(c_pert[idx], 0.02, 0.98)
        r_pert, s_pert = solve_lp_radii(c_pert)
        if s_pert > best_sum:
            best_sum = s_pert
            best_centers = c_pert.copy()
            best_radii = r_pert.copy()
            best_params = get_params(best_centers, best_radii * 0.998)

            # Follow up with SLSQP
            try:
                res = minimize(objective, best_params, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if res.success and np.min(constraint_func(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
                        best_centers, best_radii = get_physical(best_params)
            except Exception:
                pass

    # Phase 7: Second SA round with finer steps
    curr_c = best_centers.copy()
    curr_r, curr_sum = solve_lp_radii(curr_c)
    temp = 0.015
    step = 0.02

    for it in range(8000):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        curr_c[idx] += rng.normal(0, step, 2)
        curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)

        nr, ns = solve_lp_radii(curr_c)
        delta = ns - curr_sum

        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
            curr_sum = ns
            if ns > best_sum:
                best_sum = ns
                best_centers = curr_c.copy()
                best_radii = nr.copy()
        else:
            curr_c[idx] = old

        temp *= 0.9993
        step = max(0.001, step * 0.9995)

    best_radii, best_sum = solve_lp_radii(best_centers)
    best_params = get_params(best_centers, best_radii * 0.998)

    # Phase 8: Final high-precision SLSQP polish
    try:
        res = minimize(objective, best_params, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res.success and np.min(constraint_func(res.x)) >= -1e-8:
            best_params = res.x
            best_sum = -res.fun
    except Exception:
        pass

    centers, radii = get_physical(best_params)
    # Final LP to ensure radii are truly optimal for these centers
    radii, best_sum = solve_lp_radii(centers)

    return centers, np.maximum(radii, 0.0), float(best_sum)
```
