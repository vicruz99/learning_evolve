1. **Parameterization `(r, u, v)`**: We parameterize circle centers as `x = r + u*(1-2r)` and `y = r + v*(1-2r)` with `u, v ∈ [0, 1]`. 2. **Exact LP Radius Evaluation**: For any fixed set of centers, the optimal radii problem is a Linear Program. We solve this exactly using `scipy.optimize.linprog` with the 'highs' method. This provides a precise objective value and gradient-free evaluation for local search.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def get_centers_radii(params):
    """Reconstruct centers and radii from (r, u, v) parameters."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraint_func(params):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundary handled by parameterization."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def hex_init(pattern, rot, scale, rng):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    jitter = rng.uniform(-0.02, 0.02, pts.shape)
    pts += jitter
    return np.clip(pts, 0.02, 0.98)

def run_packing():
    rng = np.random.default_rng(42)
    
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraint_func}
    
    best_params = None
    best_sum = -np.inf
    
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5],
        [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7], [6,6,5,5,4], [5,5,6,5,5],
        [6,5,5,5,5], [5,5,5,6,5], [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3]
    ]
    for p in patterns:
        for _ in range(4):
            rot = rng.uniform(-0.3, 0.3)
            sc = rng.uniform(0.85, 1.15)
            inits.append(hex_init(p, rot, sc, rng))
            
    # Force-directed layouts
    for s in range(10):
        rng_fd = np.random.RandomState(s + 500)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        for _ in range(300):
            f = np.zeros_like(pts)
            diff = pts[:, None, :] - pts[None, :, :]
            dist = np.sqrt(np.sum(diff**2, axis=2))
            dist = np.maximum(dist, 1e-4)
            mag = 1.0 / (dist**2)
            np.fill_diagonal(mag, 0.0)
            f += np.sum(mag[:, :, None] * diff / dist[:, :, None], axis=1)
            for d in range(2):
                f[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
                f[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
            pts += 0.006 * f
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)

    # Phase 1: Broad SLSQP search from diverse initializations
    for pts in inits:
        r0, _ = solve_lp_radii(pts)
        r0 = np.clip(r0 * 0.99, 1e-6, 0.49)
        p0 = get_params(pts, r0)
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                if np.min(constraint_func(res.x)) >= -1e-8:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
        except Exception:
            pass

    if best_params is None:
        best_params = get_params(inits[0], solve_lp_radii(inits[0])[0])
        best_sum = -objective(best_params)

    # Phase 2: LP-driven Simulated Annealing on centers
    centers, radii = get_centers_radii(best_params)
    curr_sum = best_sum
    curr_c = centers.copy()
    temp = 0.03
    step = 0.035
    
    for it in range(6000):
        is_swap = rng.random() < 0.05
        if is_swap:
            i, j = rng.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        else:
            idx_m = rng.integers(N)
            old_c = curr_c[idx_m].copy()
            dir_v = rng.standard_normal(2)
            dir_v /= np.linalg.norm(dir_v)
            step_cur = step * (4.0 if rng.random() < 0.05 else 1.0)
            curr_c[idx_m] = np.clip(old_c + dir_v * step_cur, 0.01, 0.99)
            
        new_r, new_sum = solve_lp_radii(curr_c)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_params = get_params(curr_c, new_r)
        else:
            if not is_swap:
                curr_c[idx_m] = old_c
                
        temp *= 0.9993
        step = max(0.0015, step * 0.9995)
        
    if curr_sum > best_sum:
        best_sum = curr_sum
        best_params = get_params(curr_c, solve_lp_radii(curr_c)[0])

    # Phase 3: Perturbation & SLSQP restarts to escape local minima
    for k in range(30):
        xp = best_params.copy()
        xp[:N] += rng.uniform(-0.0015, 0.0015, N)
        xp[N:3*N] += rng.uniform(-0.02, 0.02, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(objective, xp, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            pass

    # Phase 4: High-precision final polish
    try:
        res_f = minimize(objective, best_params, method='SLSQP', bounds=bounds_slqp,
                         constraints=cons_slqp, options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(constraint_func(res_f.x)) >= -1e-9:
            best_params = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass

    # Reconstruct and ensure radii are perfectly optimal for the exact final centers
    centers_opt, radii_opt = get_centers_radii(best_params)
    radii_opt, final_sum = solve_lp_radii(centers_opt)
    
    return centers_opt, np.maximum(radii_opt, 0.0), float(final_sum)
