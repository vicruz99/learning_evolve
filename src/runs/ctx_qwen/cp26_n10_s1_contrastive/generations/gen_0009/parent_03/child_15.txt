import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in [x, 1.0-x, y, 1.0-y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def objective_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints_slqp(v):
    """Inequality constraints: boundary containment and pairwise non-overlap."""
    cx, cy, r = v[0::3], v[1::3], v[2::3]
    c = []
    # Boundary containment
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(v):
    """Reconstruct physical centers and radii from parameters."""
    r = v[:N]
    u = v[N:2*N]
    v_param = v[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v_param * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns with variations
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [7,5,5,5,4]]
    for pat in patterns:
        for _ in range(6):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N: pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
            pts = np.array(pts[:N])
            pts -= 0.5
            sc = rng.uniform(0.85, 1.15)
            pts *= sc
            pts += 0.5
            rot = rng.uniform(-0.25, 0.25)
            c, s = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pts += rng.uniform(-0.015, 0.015, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    # 2. Force-directed layouts
    for seed in range(12):
        rng_fd = np.random.RandomState(seed)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        for _ in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dist = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dist**2
            np.fill_diagonal(rep, 0.0)
            f = np.sum(rep[:, :, None] * diff / dist[:, :, None], axis=1)
            pts += 0.005 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    return inits

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    inits = generate_inits(rng)
    
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    # Evaluate initial configurations
    for pts in inits:
        r, s = solve_lp_radii(pts)
        if s > best_sum:
            best_sum = s
            best_c = pts.copy()
            best_r = r.copy()
            
    if best_c is None:
        best_c = rng.rand(N, 2) * 0.8 + 0.1
        best_r, best_sum = solve_lp_radii(best_c)
        
    # Phase 1: Simulated Annealing on centers
    curr_c = best_c.copy()
    curr_sum = best_sum
    temp = 0.045
    step = 0.038
    
    for it in range(10000):
        op = rng.choice(['move', 'swap', 'jump'])
        old_c = curr_c.copy()
        
        if op == 'move':
            idx = rng.integers(N)
            d = rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx] + d, 0.01, 0.99)
        elif op == 'swap':
            i, j = rng.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        else:
            idx = rng.integers(N)
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
            
        new_r, new_sum = solve_lp_radii(curr_c)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = curr_c.copy()
                best_r = new_r.copy()
        else:
            curr_c = old_c
            
        temp *= 0.9996
        step = max(0.002, step * 0.9995)
        
    # Phase 2: SLSQP refinement with boundary-safe parameterization
    bounds_slqp = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints_slqp}
    
    best_v = to_params(best_c, best_r * 0.995)
    
    for k in range(25):
        v_p = best_v.copy()
        v_p[N:3*N] += rng.uniform(-0.025, 0.025, 2*N)
        v_p[:N] += rng.uniform(-0.002, 0.002, N)
        v_p[:N] = np.clip(v_p[:N], 1e-6, 0.49)
        v_p[N:3*N] = np.clip(v_p[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(objective_slqp, v_p, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                cv = constraints_slqp(res.x)
                if np.min(cv) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_v = res.x.copy()
        except Exception:
            pass
            
    best_c, best_r = from_params(best_v)
    best_r, best_sum = solve_lp_radii(best_c)
    
    # Phase 3: High-precision final polish
    try:
        res_f = minimize(objective_slqp, best_v, method='SLSQP', bounds=bounds_slqp,
                         constraints=cons_slqp, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(constraints_slqp(res_f.x)) >= -1e-8:
            best_v = res_f.x
            best_c, best_r = from_params(best_v)
            best_r, best_sum = solve_lp_radii(best_c)
    except Exception:
        pass
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
