import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
ROWS, COLS = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    # Max radius allowed by boundaries
    max_r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, r) for r in max_r]
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    A_ub = np.zeros((len(ROWS), n))
    A_ub[np.arange(len(ROWS)), ROWS] = 1.0
    A_ub[np.arange(len(ROWS)), COLS] = 1.0
    b_ub = dists[ROWS, COLS]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback
    return np.full(n, 1e-6), 0.0

def objective_slqp(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[:N])

def constraints_slqp(v):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = v[:N]
    u = v[N:2*N]
    vp = v[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + vp * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, vp) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    vp = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, vp])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    vp = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + vp * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def generate_hex_init(rng, scale=1.0, rot=0.0, jitter=0.02):
    """Generates a hexagonal lattice initialization."""
    r_est = 0.095
    pts = []
    y = r_est
    row = 0
    while len(pts) < N:
        shift = (row % 2) * r_est
        x = r_est + shift
        while x <= 1.0 - r_est and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.03, 0.97)

def generate_force_init(rng):
    """Generates a force-directed layout initialization."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        dist = np.maximum(dist, 1e-4)
        rep = 1.0 / (dist**2 + 0.01)
        np.fill_diagonal(rep, 0.0)
        f += np.sum(rep[:, :, np.newaxis] * diff / dist[:, :, np.newaxis], axis=1)
        
        for d in range(2):
            f[:, d] += 20.0 * np.maximum(0, 0.05 - pts[:, d])
            f[:, d] -= 20.0 * np.maximum(0, pts[:, d] - 0.95)
            
        pts += 0.005 * f
        pts = np.clip(pts, 0.04, 0.96)
    return pts

def run_packing():
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraints_slqp}
    
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal patterns with variations
    for s in range(20):
        sc = rng.uniform(0.90, 1.10)
        rot = rng.uniform(-0.3, 0.3)
        jit = rng.uniform(0.01, 0.025)
        inits.append(generate_hex_init(rng, scale=sc, rot=rot, jitter=jit))
        
    # 2. Force-directed layouts
    for s in range(15):
        rng_fd = np.random.RandomState(s)
        inits.append(generate_force_init(rng_fd))
        
    # 3. Corner/Edge heavy patterns
    for s in range(10):
        pts = np.array([
            [0.05, 0.05], [0.95, 0.05], [0.05, 0.95], [0.95, 0.95],
            [0.5, 0.05], [0.5, 0.95], [0.05, 0.5], [0.95, 0.5],
            [0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75],
            [0.5, 0.5]
        ])
        # Fill remaining with perturbed grid
        grid_pts = np.array([[0.2 + 0.1*i, 0.2 + 0.1*j] for i in range(4) for j in range(4)])
        pts = np.vstack([pts, grid_pts[:N - len(pts)]])
        pts += rng.uniform(-0.02, 0.02, pts.shape)
        pts = np.clip(pts, 0.03, 0.97)
        inits.append(pts)
        
    # Phase 1: Initial SLSQP sweep to find strong basins
    for pts in inits:
        r0, _ = solve_lp_radii(pts)
        r0 = np.clip(r0 * 0.99, 1e-6, 0.49)
        v0 = to_params(pts, r0)
        
        try:
            res = minimize(objective_slqp, v0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                cv = constraints_slqp(res.x)
                if np.min(cv) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_c, best_r = from_params(res.x)
        except Exception:
            continue
            
    # Fallback if Phase 1 failed completely
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum = solve_lp_radii(best_c)
        
    # Phase 2: LP-driven Simulated Annealing on centers
    curr_c = best_c.copy()
    curr_r = best_r.copy()
    curr_sum = best_sum
    
    temp = 0.06
    step = 0.045
    
    for it in range(6000):
        op = rng.choice(['move', 'swap', 'jump'])
        old_c = curr_c.copy()
        
        if op == 'move':
            idx = rng.integers(N)
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.008, 0.992)
        elif op == 'swap':
            i, j = rng.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        else:
            idx = rng.integers(N)
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
            
        r_new, s_new = solve_lp_radii(curr_c)
        delta = s_new - curr_sum
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_sum = s_new
            curr_r = r_new
            if s_new > best_sum:
                best_sum = s_new
                best_c = curr_c.copy()
                best_r = r_new.copy()
        else:
            curr_c = old_c
            
        temp *= 0.9990
        step = max(0.0015, step * 0.9993)
        
    # Phase 3: SLSQP refinement of SA best + Perturbation restarts
    # Refine SA result
    r_init = np.clip(best_r * 0.995, 1e-6, 0.49)
    v0 = to_params(best_c, r_init)
    
    try:
        res = minimize(objective_slqp, v0, method='SLSQP', bounds=bounds_slqp,
                       constraints=cons_slqp, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(constraints_slqp(res.x)) >= -1e-7:
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_c, best_r = from_params(res.x)
    except Exception:
        pass
        
    # Perturbation & SLSQP to escape local minima
    for k in range(25):
        v_p = to_params(best_c, best_r * 0.995)
        v_p[N:3*N] += rng.normal(0, 0.015, 2*N)
        v_p[:N] += rng.normal(0, 0.002, N)
        v_p[:N] = np.clip(v_p[:N], 1e-6, 0.49)
        v_p[N:3*N] = np.clip(v_p[N:3*N], 0.0, 1.0)
        
        try:
            res_p = minimize(objective_slqp, v_p, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res_p.success:
                cv = constraints_slqp(res_p.x)
                if np.min(cv) >= -1e-7:
                    curr_sum = -res_p.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_c, best_r = from_params(res_p.x)
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    v_final = to_params(best_c, best_r * 0.999)
    try:
        res_f = minimize(objective_slqp, v_final, method='SLSQP', bounds=bounds_slqp,
                         constraints=cons_slqp, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(constraints_slqp(res_f.x)) >= -1e-8:
            best_c, best_r = from_params(res_f.x)
            best_sum = -res_f.fun
    except Exception:
        pass
        
    # Final LP consistency check (optional but safe)
    best_r, best_sum_lp = solve_lp_radii(best_c)
    if best_sum_lp > best_sum:
        best_sum = best_sum_lp
        best_r = best_r
        
    # Ensure non-negative radii
    best_r = np.maximum(best_r, 0.0)
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
