import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.fun < -1e-6:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def objective_slqp(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints_slqp(v):
    """Inequality constraints: g(v) >= 0."""
    cx, cy, r = v[0::3], v[1::3], v[2::3]
    c = []
    # Boundary containment
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    c.append(dist2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
    return np.concatenate(c)

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns with rotation/scale
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [5,5,5,5,6], [6,6,5,5,4]]
    for pat in patterns:
        for s in range(3):
            rng_h = np.random.RandomState(s)
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.9, 1.1) + 0.5
            rot = rng.uniform(-0.25, 0.25)
            c_r, s_r = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c_r, -s_r], [s_r, c_r]]) + 0.5
            pts += rng.uniform(-0.015, 0.015, pts.shape)
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)

    # 2. Force-directed layouts
    for s in range(5):
        rng_f = np.random.RandomState(s)
        pts = rng_f.rand(N, 2)
        for _ in range(200):
            diff = pts[:, None, :] - pts[None, :, :]
            dist = np.sqrt(np.sum(diff**2, axis=2))
            dist = np.maximum(dist, 1e-4)
            mag = 1.0 / (dist**2 + 0.01)
            f = np.sum(mag[:, :, None] * diff / dist[:, :, None], axis=1)
            pts += 0.004 * f
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # 3. Random layouts
    for s in range(3):
        rng_r = np.random.RandomState(s + 50)
        inits.append(rng_r.rand(N, 2) * 0.8 + 0.1)
        
    return inits

def run_packing():
    rng = np.random.default_rng(42)
    inits = generate_inits(rng)
    
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    # Phase 1: Simulated Annealing on centers
    for init_pts in inits:
        centers = init_pts.copy()
        r, curr_sum = solve_lp(centers)
        best_s = curr_sum
        best_c_s = centers.copy()
        best_r_s = r.copy()
        
        temp = 0.04
        step = 0.035
        
        for it in range(1500):
            if rng.random() < 0.1:
                i, j = rng.choice(N, 2, replace=False)
                centers[i], centers[j] = centers[j].copy(), centers[i].copy()
                swap = True
            else:
                idx = rng.integers(N)
                old = centers[idx].copy()
                d = rng.standard_normal(2)
                d /= np.linalg.norm(d) + 1e-9
                jump = 4.0 if rng.random() < 0.04 else 1.0
                centers[idx] = np.clip(old + d * step * jump, 0.005, 0.995)
                swap = False
                
            r_new, sum_new = solve_lp(centers)
            delta = sum_new - curr_sum
            
            if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
                curr_sum = sum_new
                if sum_new > best_s:
                    best_s = sum_new
                    best_c_s = centers.copy()
                    best_r_s = r_new.copy()
            else:
                if not swap:
                    centers[idx] = old
                    
            temp *= 0.9994
            step = max(0.0012, step * 0.9995)
            
        if best_s > best_sum:
            best_sum = best_s
            best_c = best_c_s
            best_r = best_r_s
            
    # Phase 2: SLSQP Polishing
    if best_c is not None:
        v0 = np.zeros(3 * N)
        v0[0::3] = best_c[:, 0]
        v0[1::3] = best_c[:, 1]
        v0[2::3] = np.clip(best_r * 0.99, 1e-6, 0.49)
        
        bounds_slqp = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
        cons = {'type': 'ineq', 'fun': constraints_slqp}
        
        # Perturbed restarts to escape local minima
        for _ in range(3):
            v0_p = v0.copy()
            v0_p[0::3] += rng.normal(0, 0.002, N)
            v0_p[1::3] += rng.normal(0, 0.002, N)
            v0_p[2::3] += rng.normal(0, 0.001, N)
            v0_p[0::3] = np.clip(v0_p[0::3], 0.01, 0.99)
            v0_p[1::3] = np.clip(v0_p[1::3], 0.01, 0.99)
            v0_p[2::3] = np.clip(v0_p[2::3], 1e-6, 0.49)
            
            try:
                res = minimize(objective_slqp, v0_p, method='SLSQP', bounds=bounds_slqp,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                if res.success:
                    cv = constraints_slqp(res.x)
                    if np.min(cv) >= -1e-7:
                        s_val = -res.fun
                        if s_val > best_sum:
                            best_sum = s_val
                            best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                            best_r = res.x[2::3]
            except Exception:
                pass
                
        # Final high-precision polish
        v0_f = np.zeros(3 * N)
        v0_f[0::3] = best_c[:, 0]
        v0_f[1::3] = best_c[:, 1]
        v0_f[2::3] = np.clip(best_r * 0.998, 1e-6, 0.49)
        try:
            res_f = minimize(objective_slqp, v0_f, method='SLSQP', bounds=bounds_slqp,
                             constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res_f.success and np.min(constraints_slqp(res_f.x)) >= -1e-8:
                best_c = np.column_stack((res_f.x[0::3], res_f.x[1::3]))
                best_r = res_f.x[2::3]
                best_sum = -res_f.fun
        except Exception:
            pass
            
    # Fallback (should rarely be reached)
    if best_c is None:
        best_c = np.random.rand(N, 2)
        best_r = np.full(N, 0.01)
        best_sum = 0.26
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
