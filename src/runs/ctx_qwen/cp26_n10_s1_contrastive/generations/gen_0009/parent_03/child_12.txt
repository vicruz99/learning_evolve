import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    k = 0
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.ones(n) * 1e-6, 0.0

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: dist^2 >= (r_i + r_j)^2. Boundary handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = np.clip(radii, 1e-6, 0.49)
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def gen_hex(rng, pattern, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if abs(rot) > 1e-6:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def gen_force(rng, seed):
    """Generates a force-directed layout initialization."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.rand(N, 2) * 0.8 + 0.1
    for _ in range(300):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        mag = 1.0 / (d**2)
        np.fill_diagonal(mag, 0.0)
        f += np.sum(mag[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.005 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def gen_seq(rng):
    """Generates a sequential greedy packing initialization."""
    pts = []
    for _ in range(N):
        samples = rng.uniform(0.05, 0.95, (1500, 2))
        d_bound = np.minimum(np.minimum(samples[:, 0], 1.0 - samples[:, 0]),
                             np.minimum(samples[:, 1], 1.0 - samples[:, 1]))
        d_pts = np.full_like(d_bound, 10.0)
        for p in pts:
            d_pts = np.minimum(d_pts, np.linalg.norm(samples - p, axis=1))
        min_d = np.minimum(d_bound, d_pts)
        pts.append(samples[np.argmax(min_d)])
    return np.array(pts)

def local_search(c0, rng):
    """LP-driven Simulated Annealing on centers."""
    c = c0.copy()
    r, s = solve_lp(c)
    temp = 0.05
    step = 0.03
    best_c, best_s = c.copy(), s
    
    for it in range(2000):
        idx = rng.integers(N)
        old = c[idx].copy()
        c[idx] = np.clip(c[idx] + rng.normal(0, step, 2), 0.01, 0.99)
        nr, ns = solve_lp(c)
        delta = ns - s
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            s = ns
            if ns > best_s:
                best_s = ns
                best_c = c.copy()
        else:
            c[idx] = old
            
        temp *= 0.999
        step = max(0.002, step * 0.9995)
    return best_c, best_s

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': cons}
    
    inits = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [6,4,6,5,5],
                [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7], [6,6,5,5,4], [5,5,6,5,5]]
    for p in patterns:
        for _ in range(3):
            rot = rng.uniform(-0.2, 0.2)
            sc = rng.uniform(0.9, 1.1)
            inits.append(gen_hex(rng, p, rot, sc))
    for s in range(5):
        inits.append(gen_force(rng, s))
    for _ in range(3):
        inits.append(gen_seq(rng))
        
    best_c, best_r, best_sum = None, None, -np.inf
    
    # Phase 1: LP-driven local search on diverse starts
    for c0 in inits:
        c_opt, s_opt = local_search(c0, rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt
            best_r, _ = solve_lp(best_c)
            
    # Phase 2: SLSQP joint refinement on (r, u, v)
    p0 = to_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
    try:
        res = minimize(obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success:
            cv = cons(res.x)
            if np.min(cv) >= -1e-7:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                best_p = res.x.copy()
            else:
                best_p = p0
        else:
            best_p = p0
    except Exception:
        best_p = p0
        
    # Phase 3: Perturbation restarts to escape local minima
    for k in range(15):
        xp = best_p.copy()
        xp += rng.normal(0, 0.003, 3*N)
        xp[0::3] = np.clip(xp[0::3], 1e-6, 0.49)
        xp[1::3] = np.clip(xp[1::3], 0.0, 1.0)
        xp[2::3] = np.clip(xp[2::3], 0.0, 1.0)
        
        try:
            res_p = minimize(obj, xp, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                             options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res_p.success:
                cv = cons(res_p.x)
                if np.min(cv) >= -1e-7:
                    s_val = -res_p.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_p = res_p.x.copy()
        except Exception:
            pass
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                         options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res_f.success:
            cv = cons(res_f.x)
            if np.min(cv) >= -1e-8:
                best_p = res_f.x
                best_sum = -res_f.fun
    except Exception:
        pass
        
    # Reconstruct centers
    r_opt = best_p[:N]
    u_opt = best_p[N:2*N]
    v_opt = best_p[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    
    # Final LP projection to guarantee maximal radii for exact centers
    radii_lp, sum_lp = solve_lp(centers)
    if sum_lp > best_sum:
        radii = radii_lp
        best_sum = sum_lp
    else:
        radii = np.maximum(r_opt, 0.0)
        
    return centers, radii, float(best_sum)
