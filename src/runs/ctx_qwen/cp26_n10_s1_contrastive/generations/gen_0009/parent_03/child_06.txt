import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    n_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 0.0

def slsqp_refine(centers, radii):
    """Refines centers and radii simultaneously using SLSQP."""
    v0 = np.zeros(3 * N)
    v0[0::3] = centers[:, 0]
    v0[1::3] = centers[:, 1]
    # Shrink slightly to ensure strict interior feasibility
    v0[2::3] = np.clip(radii * 0.998, 1e-7, 0.49)
    
    def objective(v):
        return -np.sum(v[2::3])
        
    def constraints(v):
        cx, cy, r = v[0::3], v[1::3], v[2::3]
        c = []
        # Boundary containment
        c.append(cx - r)
        c.append(1.0 - cx - r)
        c.append(cy - r)
        c.append(1.0 - cy - r)
        
        # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        d2 = dx**2 + dy**2
        rs = r[:, None] + r[None, :]
        c.append(d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2)
        return np.concatenate(c)
        
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    try:
        res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
        if res.success:
            cv = constraints(res.x)
            if np.min(cv) >= -1e-7:
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = res.x[2::3]
                return c_opt, r_opt, -res.fun
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def run_packing():

    best_sum = -np.inf
