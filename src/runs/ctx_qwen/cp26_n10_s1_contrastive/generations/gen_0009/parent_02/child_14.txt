import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    dists = np.maximum(dists, 1e-9)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def constraints(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def make_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    d = 1.0 - 2.0 * r
    d = np.clip(d, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def gen_hex(pattern, rot, scale, jitter, rng):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts -= 0.5
    pts *= scale
    pts += 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        R = np.array([[c, -s], [s, c]])
        pts = pts @ R.T
        pts -= pts.mean(axis=0)
        pts += 0.5
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def gen_force(rng):
    """Spread points using force-directed repulsion within unit square."""
    pts = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(200):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-5
        forces = np.sum(diff / dists[:, :, None] / dists[:, :, None], axis=1)
        pts += 0.015 * forces
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_sa(centers, rng, n_iters=1500):
    """LP-driven Simulated Annealing on centers."""
    best_c = centers.copy()
    best_r, best_sum = solve_lp_radii(best_c)
    curr_c = best_c.copy()
    curr_sum = best_sum
    temp = 0.04
    step = 0.035
    
    for it in range(n_iters):
        if rng.random() < 0.2:
            i, j = rng.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
            is_swap = True
        else:
            idx = rng.integers(N)
            old = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.015, 0.985)
            is_swap = False
            
        _, new_sum = solve_lp_radii(curr_c)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(delta / temp)):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = curr_c.copy()
                best_r, _ = solve_lp_radii(best_c)
        else:
            if not is_swap:
                curr_c[idx] = old
                
        temp *= 0.9992
        step = max(0.0015, step * 0.9994)
        
    return best_c, best_r, best_sum

def run_packing():
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons_slqp = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -np.inf
    best_params = None
    
    # Diverse hexagonal patterns covering various boundary alignments
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4], [5,5,6,5,5],
        [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3], [5,5,5,6,5], [6,5,5,5,5]
    ]
    
    inits = []
    for pat in patterns:
        for _ in range(3):
            rot = rng.uniform(-0.4, 0.4)
            sc = rng.uniform(0.85, 1.15)
            jit = rng.uniform(0.01, 0.04)
            inits.append(gen_hex(pat, rot, sc, jit, rng))
            
    for _ in range(10):
        inits.append(gen_force(rng))
        
    # Phase 1: SA on diverse inits to find strong basins
    cand_sums = []
    cand_c = []
    cand_r = []
    
    for pts in inits:
        c, r, s = run_sa(pts, rng, n_iters=1500)
        cand_c.append(c)
        cand_r.append(r)
        cand_sums.append(s)
        
    # Pick top 5 candidates for continuous refinement
    top_idx = np.argsort(cand_sums)[-5:][::-1]
    
    for idx in top_idx:
        curr_c = cand_c[idx]
        curr_r = cand_r[idx]
        
        # SLSQP refinement on (r, u, v)
        r_init = np.clip(curr_r * 0.995, 1e-6, 0.49)
        p0 = make_params(curr_c, r_init)
        
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-8:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation escapes on best to break symmetry/contact locks
    if best_params is not None:
        for _ in range(40):
            p0 = best_params.copy()
            p0[:N] += rng.uniform(-0.002, 0.002, N)
            p0[N:3*N] += rng.uniform(-0.02, 0.02, 2*N)
            p0[:N] = np.clip(p0[:N], 1e-6, 0.49)
            p0[N:3*N] = np.clip(p0[N:3*N], 0.0, 1.0)
            
            try:
                res = minimize(objective, p0, method='SLSQP', bounds=bounds_slqp,
                               constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(constraints(res.x)) >= -1e-8:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_params = res.x.copy()
            except Exception:
                pass
                
        # Phase 3: High precision polish
        try:
            res = minimize(objective, best_params, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res.success and np.min(constraints(res.x)) >= -1e-9:
                best_params = res.x
                best_sum = -res.fun
        except Exception:
            pass

    # Fallback configuration
    if best_params is None:
        fallback_c = gen_hex([6,5,6,5,4], 0.0, 1.0, 0.0, rng)
        fallback_r, _ = solve_lp_radii(fallback_c)
        best_params = make_params(fallback_c, fallback_r)
        best_sum = np.sum(best_params[:N])

    # Reconstruct physical centers and radii
    r = best_params[:N]
    u = best_params[N:2*N]
    v = best_params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    centers = np.column_stack([x, y])
    radii = np.maximum(r, 0.0)
    
    return centers, radii, float(best_sum)
