To maximize the sum of radii for 26 circles in a unit square, we employ a hybrid optimization strategy combining Linear Programming (LP) for exact radius computation and Sequential Least Squares Programming (SLSQP) for joint center-radius refinement, bridged by Simulated Annealing (SA) to escape local minima.

1. **Parameterization**: We use an `(r, u, v)` parameterization where `x = r + u(1-2r)` and `y = r + v(1-2r)` with `u, v ∈ [0,1]`. This automatically satisfies boundary constraints (`r ≤ x ≤ 1-r`), allowing SLSQP to focus solely on pairwise non-overlap constraints, significantly improving stability.
2. **Diverse Initialization**: We generate numerous starting configurations from hexagonal lattices (with varying row counts summing to 26, rotations, and scales) and force-directed layouts. This covers the vast majority of promising topological arrangements.
3. **LP-Driven Simulated Annealing**: For a fixed center configuration, the optimal radii problem is a convex LP. We use `scipy.optimize.linprog` to compute exact optimal radii efficiently. SA perturbs centers and accepts moves based on the change in the LP objective. This directly optimizes the true objective without parameterization artifacts and navigates the piecewise-linear objective landscape effectively.
4. **SLSQP Refinement**: Promising center configurations from SA are converted to `(r, u, v)` parameters and polished with SLSQP. This gradient-based method tightly packs circles by simultaneously adjusting positions and sizes.
5. **Multi-Stage Polishing**: Multiple rounds of perturbation + SLSQP restarts followed by a high-precision final pass ensure convergence to a deep local minimum. A final LP re-solves radii for the exact final centers to guarantee strict validity.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    x, y = centers[:, 0], centers[:, 1]
    for i in range(n):
        A_ub[k, i] = 1.0; b_ub[k] = x[i]; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x[i]; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y[i]; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y[i]; k += 1
        
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-5

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def get_centers_radii(params):
    """Reconstruct centers and radii from (r, u, v) parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraint_func(params):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def gen_hex(pattern, rot, scale, jitter, seed):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts -= 0.5
    pts *= scale
    pts += 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = pts @ M.T
        pts -= pts.mean(axis=0)
        pts += 0.5
    pts += rng.uniform(-jitter, jitter, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def gen_force(seed):
    """Spread points using force-directed repulsion."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    for _ in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-5
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        pts += 0.005 * forces
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    np.seterr(all='ignore')
    np.random.seed(42)
    bounds_slqp = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons_slqp = {'type': 'ineq', 'fun': constraint_func}
    
    best_params = None
    best_sum = -np.inf
    
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [5,5,6,5,5], [7,5,5,5,4], [4,5,5,5,7], [5,7,5,5,4],
        [6,7,5,5,3], [7,6,5,4,4], [8,6,5,4,3], [5,5,5,5,5,1],
        [6,6,6,4,4], [5,5,7,5,4], [7,7,5,5,2], [6,5,7,5,3]
    ]
    
    inits = []
    for pat in patterns:
        for _ in range(8):
            inits.append(gen_hex(pat, np.random.uniform(-0.4, 0.4), np.random.uniform(0.85, 1.15), np.random.uniform(0.01, 0.03), np.random.randint(10000)))
    for s in range(20):
        inits.append(gen_force(s))
        
    # Phase 1: Broad SLSQP search from diverse initializations
    for pts in inits:
        r0, _ = solve_lp_radii(pts)
        r0 = np.clip(r0 * 0.98, 1e-6, 0.49)
        p0 = get_params(pts, r0)
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            pass
            
    if best_params is None:
        best_c = inits[0]
        best_r, best_sum = solve_lp_radii(best_c)
        best_params = get_params(best_c, best_r * 0.98)
        
    best_c, best_r = get_centers_radii(best_params)
    
    # Phase 2: LP-driven Simulated Annealing on centers
    curr_c = best_c.copy()
    curr_r, curr_sum = solve_lp_radii(curr_c)
    
    rng = np.random.default_rng(42)
    temp = 0.03
    step = 0.025
    
    for it in range(8000):
        if rng.random() < 0.15:
            i, j = rng.choice(N, 2, replace=False)
            curr_c[i], curr_c[j] = curr_c[j].copy(), curr_c[i].copy()
        else:
            idx = rng.integers(N)
            old_pos = curr_c[idx].copy()
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
            
        new_r, new_sum = solve_lp_radii(curr_c)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
            curr_sum = new_sum
            curr_r = new_r
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = curr_c.copy()
                best_r = curr_r.copy()
                best_params = get_params(best_c, best_r * 0.995)
        else:
            if not (rng.random() < 0.15):
                curr_c[idx] = old_pos
                
        temp *= 0.9993
        step = max(0.001, step * 0.9994)
        
    # Phase 3: SLSQP refinement on SA best
    r_init = np.clip(best_r * 0.99, 1e-6, 0.49)
    p0 = get_params(best_c, r_init)
    try:
        res = minimize(objective, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                       options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
        if res.success and np.min(constraint_func(res.x)) >= -1e-7:
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_params = res.x.copy()
    except Exception:
        pass
        
    # Phase 4: Perturbation & SLSQP restarts to escape local minima
    for k in range(25):
        p0 = best_params.copy()
        p0[:N] += rng.uniform(-0.002, 0.002, N)
        p0[N:3*N] += rng.uniform(-0.02, 0.02, 2*N)
        p0[:N] = np.clip(p0[:N], 1e-6, 0.49)
        p0[N:3*N] = np.clip(p0[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                           options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_params = res.x.copy()
        except Exception:
            pass
            
    # Phase 5: High-precision final polish
    try:
        res = minimize(objective, best_params, method='SLSQP', bounds=bounds_slqp, constraints=cons_slqp,
                       options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res.success and np.min(constraint_func(res.x)) >= -1e-8:
            best_params = res.x
            best_sum = -res.fun
    except Exception:
        pass
        
    centers, radii = get_centers_radii(best_params)
    # Final LP ensures radii exactly match the optimized centers
    radii, best_sum = solve_lp_radii(centers)
    
    return centers, np.maximum(radii, 0.0), float(best_sum)
