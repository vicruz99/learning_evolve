import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    
    k = 0
    for i in range(n):
        x, y = np.clip(centers[i], 1e-6, 1.0 - 1e-6)
        for b in [x, 1.0-x, y, 1.0-y]:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i+1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-5)

def obj(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[:N])

def cons(params):
    """Inequality constraints: pairwise non-overlap. Boundary handled by parameterization."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    d = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / d, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / d, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack([x, y]), r

def gen_hex(seed, pattern, rot, scale):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in pattern:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
    pts += rng.uniform(-0.02, 0.02, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def gen_force(seed):
    """Generate a force-directed layout initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.rand(N, 2) * 0.8 + 0.1
    for _ in range(300):
        diff = pts[:, None, :] - pts[None, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / d**2
        np.fill_diagonal(rep, 0.0)
        f = np.sum(rep[:, :, None] * diff / d[:, :, None], axis=1)
        pts += 0.005 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    np.random.seed(42)
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [7,5,5,5,4], [5,7,5,5,4], [4,5,5,5,7], [6,4,6,5,5],
        [5,6,4,6,5], [6,6,5,5,4], [7,6,6,7], [8,6,6,6],
        [5,5,5,5,5,1], [6,5,5,5,5], [4,4,6,6,6]
    ]
    idx = 0
    for p in patterns:
        for _ in range(4):
            rot = np.random.uniform(-0.35, 0.35)
            sc = np.random.uniform(0.85, 1.15)
            inits.append(gen_hex(idx, p, rot, sc))
            idx += 1
    for s in range(15):
        inits.append(gen_force(s))
        
    bounds_slqp = [(1e-6, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N
    cons_dict = {'type': 'ineq', 'fun': cons}
    dirs = [np.array([-1, 0]), np.array([1, 0]), np.array([0, -1]), np.array([0, 1])]
    
    # Phase 1: LP-Driven Simulated Annealing on Centers
    for init_c in inits:
        r0 = solve_lp(init_c)
        curr_c = init_c.copy()
        curr_sum = np.sum(r0)
        best_local_c = curr_c.copy()
        best_local_sum = curr_sum
        
        temp = 0.025
        step = 0.035
        
        for it in range(3000):
            op = np.random.choice(['move', 'swap', 'rotate', 'boundary'])
            old_c = curr_c.copy()
            
            if op == 'move':
                i = np.random.randint(N)
                curr_c[i] += np.random.normal(0, step, 2)
                curr_c[i] = np.clip(curr_c[i], 0.005, 0.995)
            elif op == 'swap':
                i, j = np.random.choice(N, 2, replace=False)
                curr_c[[i, j]] = curr_c[[j, i]]
            elif op == 'rotate':
                k = np.random.randint(2, 6)
                subset = np.random.choice(N, k, replace=False)
                centroid = curr_c[subset].mean(axis=0)
                angle = np.random.uniform(-step, step)
                c_a, s_a = np.cos(angle), np.sin(angle)
                R = np.array([[c_a, -s_a], [s_a, c_a]])

            else: # boundary hug

                if new_sum > best_local_sum:
                    best_local_sum = new_sum
                    best_local_c = curr_c.copy()
            else:
                curr_c = old_c
                
            temp *= 0.9992
            step = max(0.0008, step * 0.9994)

            best_sum = best_local_sum
            best_c = best_local_c.copy()
            best_r = solve_lp(best_c)
            
    # Phase 2: SLSQP Refinement with boundary-safe parameterization
    if best_c is not None:
        best_p = to_params(best_c, best_r * 0.995)
        
        for trial in range(25):
            p0 = best_p.copy()

                p0[:N] += np.random.uniform(-0.0015, 0.0015, N)
                p0[N:3*N] += np.random.uniform(-0.015, 0.015, 2*N)
                p0[:N] = np.clip(p0[:N], 1e-6, 0.5)
                p0[N:3*N] = np.clip(p0[N:3*N], 0.0, 1.0)
                
            try:
                res = minimize(obj, p0, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,
                               options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(cons(res.x)) >= -1e-7:
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_p = res.x.copy()
            except Exception:
                pass
                
        # Phase 3: High-precision final polish
        try:
            res = minimize(obj, best_p, method='SLSQP', bounds=bounds_slqp, constraints=cons_dict,

                best_sum = -res.fun
        except Exception:
            pass
            
        final_c, final_r = from_params(best_p)
        final_r = solve_lp(final_c)
        final_sum = np.sum(final_r)
        return final_c, final_r, float(final_sum)

    # Fallback configuration (should rarely be reached)
    fallback_centers = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N], 
                                        np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]])
    fallback_radii = np.full(N, 0.04)
    return fallback_centers, fallback_radii, float(np.sum(fallback_radii))
