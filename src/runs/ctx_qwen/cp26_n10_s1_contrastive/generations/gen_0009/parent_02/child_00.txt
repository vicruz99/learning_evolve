import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers. Returns (radii, sum_radii)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Precompute pairwise distances efficiently
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        bounds_val = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_val:
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-6), -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def get_centers_radii(params):
    """Reconstruct centers and radii from (r, u, v) parameters."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraint_func(params):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundary handled by parameterization."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def gen_hex(rng, row_counts, rot, scale):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = (pts - 0.5) @ M.T + 0.5
        
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def gen_force(rng, seed):
    """Spread points using force-directed repulsion."""
    rng_fd = np.random.RandomState(seed)
    pts = rng_fd.rand(N, 2) * 0.8 + 0.1
    r_curr = np.full(N, 0.05)
    
    for _ in range(200):
        diff = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
        rep = 1.0 / dists**2
        np.fill_diagonal(rep, 0.0)
        forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
        
        for d in range(2):
            forces[:, d] += 25.0 * np.maximum(0, r_curr - pts[:, d])
            forces[:, d] -= 25.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            
        pts += 0.006 * forces
        pts = np.clip(pts, 0.02, 0.98)
        
        for i in range(N):
            d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            dists_others = np.linalg.norm(pts[i] - pts, axis=1)
            dists_others[i] = np.inf
            d_pair = np.min(dists_others)
            r_curr[i] = 0.8 * min(d_wall, d_pair/2.0)
            
    return pts

def gen_seq(rng):
    """Sequential best-circle heuristic."""
    pts = []
    for _ in range(N):
        samples = rng.uniform(0.02, 0.98, (2000, 2))
        d_bound = np.minimum(np.minimum(samples[:, 0], 1.0 - samples[:, 0]), 
                             np.minimum(samples[:, 1], 1.0 - samples[:, 1]))
        if len(pts) == 0:
            d_pts = np.full(2000, 10.0)
        else:
            pts_arr = np.array(pts)
            d_pts = np.min(np.linalg.norm(samples[:, None, :] - pts_arr[None, :, :], axis=2), axis=1)
        min_d = np.minimum(d_bound, d_pts)
        pts.append(samples[np.argmax(min_d)])
    return np.array(pts)

def run_packing():

    rng = np.random.default_rng(42)
    
    bounds_slqp = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [7,5,5,5,4], [5,7,5,5,4], [6,4,6,5,5], [6,6,5,5,4],
        [4,5,5,5,7], [5,5,6,5,5], [6,5,5,5,5], [7,6,5,4,4]
    ]
    
    for pat in patterns:
        for _ in range(3):
            rot = rng.uniform(-0.35, 0.35)
            sc = rng.uniform(0.88, 1.12)
            inits.append(gen_hex(rng, pat, rot, sc))
            
    for s in range(10):
        inits.append(gen_force(rng, s))
        
    for _ in range(5):
        inits.append(gen_seq(rng))
        
    # Phase 1: Broad SLSQP search from diverse starts
    for pts in inits:
        r_init, _ = solve_lp(pts)
        r_init = np.clip(r_init * 0.99, 1e-6, 0.49)
        vars0 = get_params(pts, r_init)
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                if np.min(constraint_func(res.x)) >= -1e-8:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # Fallback
    if best_vars is None:
        pts_f = inits[0]
        r_f, _ = solve_lp(pts_f)
        best_vars = get_params(pts_f, r_f * 0.99)
        best_sum = -objective(best_vars)
        
    best_c, best_r = get_centers_radii(best_vars)
    
    # Phase 2: LP-driven Simulated Annealing on centers
    curr_c = best_c.copy()
    curr_r, curr_sum = solve_lp(curr_c)
    temp = 0.04
    step = 0.035
    
    for it in range(4000):
        idx = rng.integers(N)
        old_pos = curr_c[idx].copy()
        
        # Periodic shock to escape deep local minima
        if it % 1200 == 0:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)

                best_vars = res.x.copy()

        pass

        x0[0::3] = np.clip(x0[0::3], 1e-6, 0.49)

        try:

    best_c, best_r = get_centers_radii(best_vars)

    best_r, best_sum = solve_lp(best_c)
    
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
