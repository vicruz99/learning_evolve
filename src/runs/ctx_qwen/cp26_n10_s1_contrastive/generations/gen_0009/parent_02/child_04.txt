import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii maximizing sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Number of constraints: 4 boundary per circle + pairwise distances
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Precompute pairwise distances
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds_r = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 1e-4

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def from_params(params):
    """Reconstruct physical centers and radii from parameters."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def slqp_constraints(params):
    """Inequality constraints for SLSQP: pairwise non-overlap."""
    r = params[:N]
    u = params[N:2*N]
    v = params[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def slqp_objective(params):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(params[:N])

def generate_initials(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Greedy sequential placement
    for _ in range(3):
        pts = []
        for _ in range(N):
            samples = rng.uniform(0.05, 0.95, (2000, 2))
            d_bound = np.minimum(np.minimum(samples[:, 0], 1.0 - samples[:, 0]), 
                                 np.minimum(samples[:, 1], 1.0 - samples[:, 1]))
            d_pts = np.full_like(d_bound, 10.0)
            for p in pts:
                d_pts = np.minimum(d_pts, np.linalg.norm(samples - p, axis=1))
            pts.append(samples[np.argmin(np.minimum(d_bound, d_pts))])
        inits.append(np.array(pts))
        
    # 2. Hexagonal lattices
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], 
        [5,5,5,5,6], [6,4,6,5,5], [5,7,5,7], [6,6,6,6,2]
    ]
    for pat in patterns:
        for _ in range(4):
            pts = []
            r_est = 0.095
            y = r_est
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3.0) * r_est
                row += 1
            pts = np.array(pts[:N])
            pts -= 0.5
            scale = rng.uniform(0.85, 1.15)
            pts *= scale
            pts += 0.5
            rot = rng.uniform(-0.35, 0.35)
            c, s = np.cos(rot), np.sin(rot)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pts += rng.uniform(-0.015, 0.015, pts.shape)
            inits.append(np.clip(pts, 0.02, 0.98))
            
    # 3. Force-directed layouts
    for seed in range(10):
        rng_fd = np.random.RandomState(seed)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        for _ in range(300):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-5
            forces = np.sum(diff / dists[:, :, None] / dists[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 40.0 * np.maximum(0, 0.05 - pts[:, d])
                forces[:, d] -= 40.0 * np.maximum(0, pts[:, d] - 0.95)
            pts += 0.015 * forces
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    return inits

def simulated_annealing(centers, rng, max_iter=8000):
    """SA on centers optimizing sum of radii via LP."""
    curr_c = centers.copy()
    curr_r, curr_sum = solve_lp_radii(curr_c)
    best_c = curr_c.copy()
    best_r = curr_r.copy()
    best_sum = curr_sum
    
    temp = 0.04
    step = 0.035
    
    for it in range(max_iter):
        idx = rng.integers(N)
        old = curr_c[idx].copy()
        
        # Occasional large jump
        if rng.random() < 0.03:
            curr_c[idx] = rng.uniform(0.05, 0.95, 2)
        else:
            curr_c[idx] += rng.normal(0, step, 2)
            curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
            
        new_r, new_sum = solve_lp_radii(curr_c)
        delta = new_sum - curr_sum
        
        if delta > 0 or (temp > 1e-7 and rng.random() < np.exp(delta / temp)):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = curr_c.copy()
                best_r = new_r.copy()
        else:
            curr_c[idx] = old
            
        temp *= 0.9992
        step = max(0.0015, step * 0.9994)
        
    return best_c, best_r, best_sum

def run_packing():

    bounds = [(1e-5, 0.5)]*N + [(0.0, 1.0)]*N + [(0.0, 1.0)]*N

    best_r = None
    best_sum = -np.inf

            best_c = c
            best_r = r
            
    # Phase 2: SLSQP refinement on best SA result
    if best_c is not None:
        r_init = np.maximum(best_r * 0.995, 1e-5)
        p0 = get_params(best_c, r_init)
        
        try:
            res = minimize(slqp_objective, p0, method='SLSQP', bounds=bounds,

            if res.success and np.min(slqp_constraints(res.x)) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_c, best_r = from_params(res.x)
        except Exception:
            pass
            
    # Phase 3: Perturbation & SLSQP restarts to escape local minima
    if best_c is not None:
        for k in range(30):
            p0 = get_params(best_c, best_r * 0.995)
            p0[:N] += rng.normal(0, 0.002, N)
            p0[N:3*N] += rng.normal(0, 0.015, 2*N)
            p0[:N] = np.clip(p0[:N], 1e-5, 0.5)
            p0[N:3*N] = np.clip(p0[N:3*N], 0.0, 1.0)
            
            try:

    # Phase 4: High-precision final polish

        except Exception:
            pass

    # Fallback

        best_c = np.column_stack([np.linspace(0.1, 0.9, 6).repeat(5)[:N],

        best_sum = np.sum(best_r)
        
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
