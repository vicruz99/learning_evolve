import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(val)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-5), 1e-4

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj_slqp(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons_slqp(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    # Recover centers from parameters
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    # Squared Euclidean distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    # Squared sum of radii
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def gen_hex(rows, rot, scale, seed):
    """Generate a hexagonal lattice initialization with specified parameters."""
    rng = np.random.RandomState(seed)
    pts = []
    r0 = 0.095
    y = r0
    row = 0
    for cnt in rows:
        shift = (row % 2) * r0
        x = r0 + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N])
    
    # Center and scale
    pts -= 0.5
    pts *= scale
    pts += 0.5
    
    # Rotate
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        M = np.array([[c, -s], [s, c]])
        pts = pts @ M.T
        pts -= pts.mean(axis=0)
        pts += 0.5
        
    # Jitter
    pts += rng.uniform(-0.025, 0.025, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_sa(centers0, rng):
    """Simulated Annealing on centers guided by LP radius optimization."""
    centers = centers0.copy()
    _, curr_sum = solve_lp(centers)
    best_c = centers.copy()
    best_sum = curr_sum
    
    temp = 0.06
    step = 0.045
    
    for it in range(10000):
        old_c = centers.copy()
        
        action = rng.random()
        if action < 0.05:
            i = rng.integers(N)
            centers[i] = rng.uniform(0.05, 0.95, 2)
        elif action < 0.15:
            i, j = rng.choice(N, 2, replace=False)
            centers[i], centers[j] = centers[j].copy(), centers[i].copy()
        else:
            i = rng.integers(N)
            centers[i] += rng.normal(0, step, 2)
            centers[i] = np.clip(centers[i], 0.02, 0.98)
            
        _, sum_new = solve_lp(centers)
        delta = sum_new - curr_sum
        
        if delta > 0 or (temp > 1e-9 and rng.random() < np.exp(min(0.0, delta / temp))):
            curr_sum = sum_new
            if sum_new > best_sum:
                best_sum = sum_new
                best_c = centers.copy()
        else:
            centers = old_c
            
        temp *= 0.9994
        step = max(0.002, step * 0.9995)
        
    return best_c, best_sum
