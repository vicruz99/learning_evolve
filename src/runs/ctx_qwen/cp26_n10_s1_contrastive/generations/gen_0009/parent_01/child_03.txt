

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we will employ a hybrid optimization strategy combining:
1. **Linear Programming (LP) for Radii**: Given fixed centers, the optimal radii that maximize $\sum r_i$ under non-overlap and boundary constraints form a linear program. This is solved efficiently using `scipy.optimize.linprog`.
2. **(r, u, v) Parameterization for SLSQP**: We parameterize each circle's position as $x = r + u(1-2r)$, $y = r + v(1-2r)$ where $u,v \in [0,1]$. This automatically satisfies boundary constraints ($r \le x \le 1-r$, $r \le y \le 1-r$), allowing SLSQP to focus purely on pairwise non-overlap constraints $d_{ij}^2 \ge (r_i + r_j)^2$.
3. **Diverse Initializations**: We generate many starting configurations using hexagonal lattices (with various row counts, scales, and rotations), force-directed layouts, and perturbed grids to cover different topological arrangements.
4. **Simulated Annealing (SA) on Centers**: We run an intensive SA loop that perturbs circle centers and re-optimizes radii via LP. This explores the configuration space and escapes local minima that gradient-based methods might get stuck in.
5. **Iterative SLSQP Refinement & Perturbation**: After SA finds a strong local optimum, we refine it with high-precision SLSQP. We then repeatedly perturb the parameters and restart SLSQP to climb out of remaining local basins, followed by a final high-precision polish.

This pipeline leverages the speed of LP for radii evaluation, the robustness of SA for topology search, and the precision of SLSQP for contact tightening, consistently pushing the sum of radii towards the theoretical optimum.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def to_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def from_params(params):
    """Reconstruct centers and radii from (r, u, v) parameters."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def slsqp_constraints(p):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundary handled by parameterization."""
    r = p[0::3]
    u = p[1::3]
    v = p[2::3]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def slsqp_obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[0::3])

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns with varied structures
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [5,5,6,5,5], [7,5,5,5,4], [4,5,5,5,7], [5,7,5,5,4],
        [7,6,5,4,4], [6,7,5,5,3], [8,6,5,4,3], [5,5,5,5,5,1]
    ]
    for pat in patterns:
        for _ in range(3):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            pts = np.array(pts[:N])
            pts -= 0.5
            scale = rng.uniform(0.85, 1.15)
            pts *= scale
            pts += 0.5
            rot = rng.uniform(-0.3, 0.3)
            c, s = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[c, -s], [s, c]])
            pts += rng.uniform(-0.015, 0.015, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
            
    # 2. Force-directed layouts to find organic packings
    for seed in range(20):
        rng_fd = np.random.RandomState(seed)
        pts = rng_fd.rand(N, 2) * 0.8 + 0.1
        r_curr = np.full(N, 0.06)
        for step in range(400):
            diff = pts[:, None, :] - pts[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8
            rep = 1.0 / dists**2
            np.fill_diagonal(rep, 0.0)
            forces = np.sum(rep[:, :, None] * diff / dists[:, :, None], axis=1)
            for d in range(2):
                forces[:, d] += 30.0 * np.maximum(0, r_curr - pts[:, d])
                forces[:, d] -= 30.0 * np.maximum(0, pts[:, d] - (1.0 - r_curr))
            step_size = 0.008 * (0.993**step)
            pts += step_size * forces
            pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
            for i in range(N):
                d_wall = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
                dists_others = np.linalg.norm(pts[i] - pts, axis=1)
                dists_others[i] = np.inf
                d_pair = np.min(dists_others)
                r_curr[i] = 0.80 * min(d_wall, d_pair/2.0)
        inits.append(pts)
        
    # 3. Perturbed grids
    for seed in range(10):
        rng_g = np.random.RandomState(seed)
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [0.5, 0.5]])
        pts += rng_g.uniform(-0.04, 0.04, (N, 2))
        pts = np.clip(pts, 0.04, 0.96)
        inits.append(pts)
        
    return inits

def run_packing():
    rng = np.random.default_rng(123)
    bounds = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': slsqp_constraints}
    
    best_vars = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    inits = generate_inits(rng)
    
    # Phase 1: Broad SLSQP search from diverse initializations
    for pts in inits:
        r_lp, _ = solve_lp_radii(pts)
        r_lp = np.clip(r_lp * 0.985, 1e-6, 0.49)
        v0 = to_params(pts, r_lp)
        try:
            res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(slsqp_constraints(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_vars = res.x.copy()
        except Exception:
            continue
            
    if best_vars is not None:
        best_c, best_r = from_params(best_vars)
        best_r, best_sum = solve_lp_radii(best_c)
        
        # Phase 2: Intensive Simulated Annealing on centers
        curr_c = best_c.copy()
        curr_r, curr_sum = solve_lp_radii(curr_c)
        temp = 0.03
        step = 0.035
        
        for it in range(12000):
            idx = rng.integers(N)
            old_pos = curr_c[idx].copy()
            
            # Occasional random jump to escape deep local minima
            if rng.random() < 0.02:
                curr_c[idx] = rng.uniform(0.02, 0.98, 2)
            else:
                curr_c[idx] += rng.normal(0, step, 2)
                curr_c[idx] = np.clip(curr_c[idx], 0.005, 0.995)
                
            nr, ns = solve_lp_radii(curr_c)
            delta = ns - curr_sum
            
            # Accept if improvement or probabilistically
            if delta > 0 or (temp > 1e-8 and rng.random() < np.exp(delta / temp)):
                curr_sum = ns
                if ns > best_sum:
                    best_sum = ns
                    best_c = curr_c.copy()
                    best_r = nr.copy()
            else:
                curr_c[idx] = old_pos
                
            temp *= 0.9993
            step = max(0.0015, step * 0.9995)
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_c = curr_c.copy()
            best_r, best_sum = solve_lp_radii(best_c)
            
        # Phase 3: SLSQP refinement on SA best
        r_init = np.clip(best_r * 0.995, 1e-6, 0.49)
        v0 = to_params(best_c, r_init)
        try:
            res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if res.success and np.min(slsqp_constraints(res.x)) >= -1e-8:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_vars = res.x.copy()
        except Exception:
            pass
            
        # Phase 4: Perturbation & SLSQP restarts to escape remaining local minima
        for k in range(30):
            r_p = best_r * 0.99
            v0 = to_params(best_c, r_p)
            v0[1::3] += rng.uniform(-0.03, 0.03, N)
            v0[2::3] += rng.uniform(-0.03, 0.03, N)
            v0[0::3] += rng.uniform(-0.002, 0.002, N)
            v0[0::3] = np.clip(v0[0::3], 1e-6, 0.49)
            v0[1::3] = np.clip(v0[1::3], 0.0, 1.0)
            v0[2::3] = np.clip(v0[2::3], 0.0, 1.0)
            
            try:
                res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
                if res.success and np.min(slsqp_constraints(res.x)) >= -1e-7:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_vars = res.x.copy()
            except Exception:
                continue
                
        if best_vars is not None:
            best_c, best_r = from_params(best_vars)
            best_r, best_sum = solve_lp_radii(best_c)
            
        # Phase 5: High-precision final polish
        r_init = np.clip(best_r * 0.999, 1e-6, 0.49)
        v0 = to_params(best_c, r_init)
        try:
            res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-15, 'disp': False})
            if res.success and np.min(slsqp_constraints(res.x)) >= -1e-9:
                best_vars = res.x
                best_sum = -res.fun
        except Exception:
            pass
            
    # Fallback configuration (should rarely be reached)
    if best_vars is None:
        pts = inits[0]
        r, _ = solve_lp_radii(pts)
        best_vars = to_params(pts, r)
        best_sum = np.sum(best_vars[0::3])
        
    best_c, best_r = from_params(best_vars)
    # Final LP to ensure radii are strictly optimal for the exact final centers
    best_r, best_sum = solve_lp_radii(best_c)
    
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
