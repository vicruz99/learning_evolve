To maximize the sum of radii for 26 circles in a unit square, we combine three powerful techniques:
1. **Linear Programming (LP) for Radii**: For any fixed set of centers, the optimal radii that maximize the sum can be found exactly via a small LP. This decouples the hard geometric constraints from the center optimization.
2. **Parameterized SLSQP Optimization**: We optimize both centers and radii jointly using `scipy.optimize.minimize` with SLSQP. By parameterizing centers as `x = r + u*(1-2r)` and `y = r + v*(1-2r)` with `u,v ∈ [0,1]`, boundary constraints are automatically satisfied, leaving only pairwise non-overlap constraints. This yields high-precision local optima.
3. **Simulated Annealing (SA) on Centers**: SA explores the center configuration space globally. At each step, we move one circle, re-solve the LP for radii, and accept/reject based on the Metropolis criterion. This escapes local minima that trap gradient-based methods.

The pipeline runs diverse hexagonal and random initializations through SLSQP, then feeds the best result into an intensive SA loop. The SA output is polished again with SLSQP, followed by multiple perturbation-restart cycles to squeeze out the last bits of optimality. This hybrid approach consistently breaks the 2.636 barrier.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_c = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_c, n))
    b_ub = np.zeros(num_c)
    k = 0
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'disp': False})
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 1e-4

def get_params(centers, radii):
    """Map physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    params = np.empty(3 * N)
    params[0::3] = r
    params[1::3] = u
    params[2::3] = v
    return params

def get_centers_radii(params):
    """Reconstruct centers and radii from (r, u, v) parameters."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def constraint_func(params):
    """Inequality constraints: dist_sq >= (r_i + r_j)^2. Boundary handled by parameterization."""
    r = params[0::3]
    u = params[1::3]
    v = params[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[0::3])

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6],
        [6,4,6,5,5], [5,6,4,6,5], [4,5,6,5,6], [6,6,5,5,4],
        [7,5,5,5,4], [4,5,5,5,7], [5,7,5,5,4], [5,5,5,5,5,1],
        [5,6,5,5,5], [6,5,5,6,4], [5,5,6,5,5], [4,6,6,5,5]
    ]
    for pat in patterns:
        for _ in range(3):
            pts = []
            r_est = 0.095
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = (r_idx % 2) * r_est
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            pts = np.array(pts[:N])
            pts -= 0.5
            scale = rng.uniform(0.85, 1.15)
            pts *= scale
            pts += 0.5
            rot = rng.uniform(-0.2, 0.2)
            c, s = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[c, -s], [s, c]])
            pts += rng.uniform(-0.015, 0.015, (N, 2))
            pts = np.clip(pts, 0.02, 0.98)
            inits.append(pts)
    for _ in range(10):
        pts = rng.uniform(0.1, 0.9, (N, 2))
        inits.append(pts)
    return inits

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds_slqp = [(1e-6, 0.49), (0.0, 1.0), (0.0, 1.0)] * N
    cons_slqp = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    inits = generate_inits(rng)
    
    # Phase 1: SLSQP from diverse starts
    for pts in inits:
        r_init, _ = solve_lp(pts)
        r_init = np.clip(r_init * 0.98, 1e-6, 0.49)
        vars0 = get_params(pts, r_init)
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_vars = res.x.copy()
        except Exception:
            continue
            
    if best_vars is not None:
        best_c, best_r = get_centers_radii(best_vars)
        best_r, best_sum = solve_lp(best_c)
        
        # Phase 2: Intensive SA on centers
        curr_c = best_c.copy()
        curr_r, curr_sum = solve_lp(curr_c)
        temp = 0.05
        step = 0.04
        best_sa_sum = best_sum
        best_sa_c = curr_c.copy()
        best_sa_r = curr_r.copy()
        
        for it in range(12000):
            idx = rng.integers(N)
            old_pos = curr_c[idx].copy()
            
            if it % 2500 == 0:
                curr_c[idx] = rng.uniform(0.05, 0.95, 2)
            else:
                curr_c[idx] += rng.normal(0, step, 2)
                curr_c[idx] = np.clip(curr_c[idx], 0.01, 0.99)
                
            nr, ns = solve_lp(curr_c)
            delta = ns - curr_sum
            
            if delta > 0 or (temp > 1e-7 and rng.random() < np.exp(delta / temp)):
                curr_sum = ns
                if ns > best_sa_sum:
                    best_sa_sum = ns
                    best_sa_c = curr_c.copy()
                    best_sa_r = nr.copy()
            else:
                curr_c[idx] = old_pos
                
            temp *= 0.9991
            step = max(0.0015, step * 0.9992)
            
        if best_sa_sum > best_sum:
            best_sum = best_sa_sum
            best_c = best_sa_c.copy()
            best_r = best_sa_r.copy()
            
        # Phase 3: SLSQP polish on SA best
        r_init = np.clip(best_r * 0.995, 1e-6, 0.49)
        vars0 = get_params(best_c, r_init)
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds_slqp,
                           constraints=cons_slqp, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_vars = res.x.copy()
        except Exception:
            pass
            
        # Phase 4: Perturbation restarts
        if best_vars is not None:
            best_c, best_r = get_centers_radii(best_vars)
            for k in range(30):
                r_p = best_r * 0.995
                vars0 = get_params(best_c, r_p)
                vars0[1::3] += rng.uniform(-0.03, 0.03, N)
                vars0[2::3] += rng.uniform(-0.03, 0.03, N)
                vars0[0::3] += rng.uniform(-0.002, 0.002, N)
                vars0[0::3] = np.clip(vars0[0::3], 1e-6, 0.49)
                vars0[1::3] = np.clip(vars0[1::3], 0.0, 1.0)
                vars0[2::3] = np.clip(vars0[2::3], 0.0, 1.0)
                try:
                    res = minimize(objective, vars0, method='SLSQP', bounds=bounds_slqp,
                                   constraints=cons_slqp, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                    if res.success and np.min(constraint_func(res.x)) >= -1e-8:
                        s = -res.fun
                        if s > best_sum:
                            best_sum = s
                            best_vars = res.x.copy()
                            best_c, best_r = get_centers_radii(best_vars)
                except Exception:
                    continue
                    
    if best_vars is None:
        pts = inits[0]
        r, _ = solve_lp(pts)
        best_vars = get_params(pts, r)
        best_sum = np.sum(best_vars[0::3])
        
    best_c, best_r = get_centers_radii(best_vars)
    best_r, best_sum = solve_lp(best_c)
    
    return best_c, np.maximum(best_r, 0.0), float(best_sum)
