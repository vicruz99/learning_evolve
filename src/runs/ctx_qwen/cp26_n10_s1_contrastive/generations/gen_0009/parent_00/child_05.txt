import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-5), 0.0

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def alt_opt(centers, iters=150):
    """Alternating optimization: fix centers -> solve LP radii -> move centers -> repeat"""
    curr_c = centers.copy()
    curr_r, curr_sum = solve_lp(curr_c)
    for t in range(iters):
        f = np.zeros_like(curr_c)
        diff = curr_c[:, np.newaxis, :] - curr_c[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2)) + 1e-9
        d[d > 0.4] = 1e-9
        rs = curr_r[:, np.newaxis] + curr_r[np.newaxis, :]
        np.fill_diagonal(d, 1e-9)
        w = (rs**2) / (d**3)
        np.fill_diagonal(w, 0.0)
        f += np.sum(w[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        
        for dim in range(2):
            f[:, dim] += 30.0 * np.maximum(0, curr_r - curr_c[:, dim])
            f[:, dim] -= 30.0 * np.maximum(0, curr_c[:, dim] - (1.0 - curr_r))
            
        step = 0.004 * (0.992**t)
        curr_c += step * f
        curr_c = np.clip(curr_c, 1e-4, 1.0 - 1e-4)
        curr_r, curr_sum = solve_lp(curr_c)
    return curr_c, curr_r, curr_sum

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6], [7,5,5,5,4]]
    for pat in pats:

            r_e = 0.095
            y = r_e
            row = 0
            for cnt in pat:
                shift = (row % 2) * r_e
                x = r_e + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_e
                y += np.sqrt(3.0) * r_e
                row += 1
            pts = np.array(pts[:N])
            pts = (pts - 0.5) * rng.uniform(0.95, 1.05) + 0.5
            pts += rng.uniform(-0.02, 0.02, pts.shape)
            inits.append(np.clip(pts, 0.02, 0.98))
            
    for _ in range(10):
        c = rng.uniform(0.05, 0.95, (N, 2))
        c, _, _ = alt_opt(c, iters=80)
        inits.append(c)
    return inits

    centers = np.column_stack((x_opt, y_opt))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
