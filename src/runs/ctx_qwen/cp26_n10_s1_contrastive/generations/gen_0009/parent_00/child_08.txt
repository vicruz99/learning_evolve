1. **Hybrid Optimization Strategy**: Combine LP-driven Simulated Annealing (SA) on circle centers with SLSQP optimization on a boundary-safe `(r, u, v)` parameterization. SA is excellent for escaping local minima in the center configuration space, while SLSQP excels at precise local refinement of both centers and radii simultaneously.
2. **LP for Radii**: For any fixed set of centers, the optimal radii maximizing $\sum r_i$ subject to non-overlap and boundary constraints form a Linear Program. We solve this efficiently using `scipy.optimize.linprog` with the 'highs' method. This acts as the objective function for the SA phase.
3. **Enhanced Simulated Annealing**: 
   - Run SA for 20,000 iterations to thoroughly explore center configurations.
   - Use adaptive step sizes and a cooling schedule.
   - Incorporate "random jump" moves (repositioning a random circle completely) to break rigid contact networks that trap standard local searches.
   - Acceptance based on Metropolis criterion to allow uphill moves early on.
4. **Diverse Initializations**: Generate initial configurations using hexagonal lattices (with varied row counts, rotations, and scales), force-directed layouts, and sequential Voronoi-like packings. This ensures we start SA from multiple promising basins.
5. **SLSQP Polishing**: 
   - Map the best center configuration from SA to the `(r, u, v)` parameterization, which automatically satisfies boundary constraints ($r \le x \le 1-r, r \le y \le 1-r$).
   - Run SLSQP to optimize $-\sum r$ subject to $dist^2 \ge (r_i + r_j)^2$.
   - Perform multiple restarts/perturbations around the best found point to escape local SLSQP minima.
   - Final high-precision polish with tight tolerances.
6. **Robustness**: Include fallbacks, numerical stability checks, and careful handling of constraints to ensure the returned packing is strictly valid per the validation function.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solves LP to find radii that maximize sum(r_i)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    k = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = b
            k += 1
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-5), 0.0

def get_params(centers, radii):
    """Convert physical centers/radii to (r, u, v) optimization parameters."""
    r = radii.copy()
    denom = np.clip(1.0 - 2.0 * r, 1e-6, 1.0)
    u = np.clip((centers[:, 0] - r) / denom, 0.0, 1.0)
    v = np.clip((centers[:, 1] - r) / denom, 0.0, 1.0)
    return np.concatenate([r, u, v])

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[:N])

def cons(p):
    """Inequality constraints: pairwise non-overlap. Boundaries handled by parameterization."""
    r = p[:N]
    u = p[N:2*N]
    v = p[2*N:3*N]
    
    # Parameterization guarantees r <= x <= 1-r and r <= y <= 1-r
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Constraint: dist^2 >= (r_i + r_j)^2
    return d2[I_IDX, J_IDX] - rs[I_IDX, J_IDX]**2

def seq_init(rng):
    """Generates a Voronoi-like sequential packing initialization."""
    pts = []
    for _ in range(N):
        samples = rng.uniform(0.1, 0.9, (1500, 2))
        d_bound = np.minimum(np.minimum(samples[:, 0], 1.0 - samples[:, 0]), 
                             np.minimum(samples[:, 1], 1.0 - samples[:, 1]))
        d_pts = np.full_like(d_bound, 10.0)
        for p in pts:
            d_pts = np.minimum(d_pts, np.linalg.norm(samples - p, axis=1))
        min_d = np.minimum(d_bound, d_pts)
        pts.append(samples[np.argmax(min_d)])
    return np.array(pts)

def hex_init(rng, row_counts, rot=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization."""
    pts = []
    r_est = 0.095
    y = r_est
    row = 0
    for cnt in row_counts:
        shift = (row % 2) * r_est
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts = (pts - 0.5) * scale + 0.5
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += rng.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def force_init(rng):
    """Generate a force-directed layout initialization."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(250):
        f = np.zeros_like(pts)
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        d = np.sqrt(np.sum(diff**2, axis=2))
        d = np.maximum(d, 1e-4)
        f += np.sum((1.0 / d**2)[:, :, np.newaxis] * diff / d[:, :, np.newaxis], axis=1)
        for dim in range(2):
            f[:, dim] += 20.0 * np.maximum(0, 0.1 - pts[:, dim])
            f[:, dim] -= 20.0 * np.maximum(0, pts[:, dim] - 0.9)
        pts += 0.0025 * f
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing():
    np.seterr(all='ignore')
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 0.5)] * N + [(0.0, 1.0)] * N + [(0.0, 1.0)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    best_p = None
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    inits = []
    for _ in range(5):
        inits.append(seq_init(rng))
    pats = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [7,5,5,5,4], [5,5,5,5,6], [5,5,6,5,5]]
    for pat in pats:
        for _ in range(4):
            inits.append(hex_init(rng, pat, rot=rng.uniform(-0.25, 0.25), scale=rng.uniform(0.9, 1.1)))
    for _ in range(5):
        inits.append(force_init(rng))
        
    # Phase 1: Broad SLSQP search from diverse starts to find a strong baseline
    for c in inits:
        r, _ = solve_lp_radii(c)
        r = np.clip(r * 0.985, 1e-6, 0.49)
        p0 = get_params(c, r)
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization failed completely
    if best_p is None:
        c_f = inits[0]
        r_f, _ = solve_lp_radii(c_f)
        best_p = get_params(c_f, r_f * 0.98)
        best_sum = -obj(best_p)
        
    # Extract current best for SA
    r_b = best_p[:N]
    u_b = best_p[N:2*N]
    v_b = best_p[2*N:3*N]
    c_best = np.column_stack((r_b + u_b * (1 - 2 * r_b), r_b + v_b * (1 - 2 * v_b)))
    
    # Phase 2: LP-Driven Simulated Annealing on Centers
    curr_c = c_best.copy()
    curr_s = best_sum
    step = 0.035
    temp = 0.025
    
    for it in range(20000):
        move_type = rng.integers(0, 5)
        if move_type < 4:
            idx = rng.integers(N)
            d = rng.standard_normal(2)
            nc = curr_c.copy()
            nc[idx] = np.clip(nc[idx] + step * d, 0.02, 0.98)
        else:
            idx = rng.integers(N)
            nc = curr_c.copy()
            nc[idx] = rng.uniform(0.05, 0.95, 2)
            
        _, ns = solve_lp_radii(nc)
        delta = ns - curr_s
        
        if delta > 0 or (rng.random() < np.exp(delta / max(temp, 1e-4))):
            curr_c = nc
            curr_s = ns
            if ns > best_sum:
                best_sum = ns
                best_c = nc.copy()
                best_r, _ = solve_lp_radii(best_c)
            step = max(0.001, step * 0.9995)
            temp = max(0.0001, temp * 0.9995)
        else:
            step = min(0.05, step * 1.001)
            
    # Update best_p if SA found better
    if best_c is not None:
        best_p = get_params(best_c, np.clip(best_r * 0.995, 1e-6, 0.49))
        
    # Phase 3: Perturbation & SLSQP to escape local minima
    for k in range(25):
        xp = best_p.copy()
        xp[:N] += rng.normal(0, 0.0025, N)
        xp[N:3*N] += rng.normal(0, 0.018, 2*N)
        xp[:N] = np.clip(xp[:N], 1e-6, 0.49)
        xp[N:3*N] = np.clip(xp[N:3*N], 0.0, 1.0)
        
        try:
            res = minimize(obj, xp, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
            if res.success and np.min(cons(res.x)) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 4: High-precision final polish
    try:
        res_f = minimize(obj, best_p, method='SLSQP', bounds=bounds, constraints=cons_dict,
                         options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
        if res_f.success and np.min(cons(res_f.x)) >= -1e-8:
            best_p = res_f.x
            best_sum = -res_f.fun
    except Exception:
        pass
        
    # Reconstruct physical centers and radii
    r_opt = best_p[:N]
    u_opt = best_p[N:2*N]
    v_opt = best_p[2*N:3*N]
    x_opt = r_opt + u_opt * (1.0 - 2.0 * r_opt)
    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
    centers = np.column_stack((x_opt, y_opt))
    radii = np.maximum(r_opt, 0.0)
    
    return centers, radii, float(best_sum)
