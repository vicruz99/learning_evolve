import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute sparse LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS, N))
A_LP[np.arange(NUM_PAIRS), I] = 1.0
A_LP[np.arange(NUM_PAIRS), J] = 1.0

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = []
    for i in range(N):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.zeros(N), 0.0

def objective_centers(x_flat):
    """Objective for center-only optimization: minimize negative LP sum of radii."""
    c = x_flat.reshape(N, 2)
    c = np.clip(c, 0.001, 0.999)
    _, s = solve_lp_radii(c)
    return -s

def objective_joint(x):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def constraints_joint(x):
    """Inequality constraints for joint optimization (must be >= 0)."""
    cx = x[0:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c[4*N:] = np.hypot(dx, dy) - r[I] - r[J]
    return c

def make_valid(centers, radii):
    """Deterministically resolve overlaps and boundary violations."""
    radii = radii.copy()
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], 
                 centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(0.0, radii[i])
        
    # Iteratively resolve overlaps
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(centers[i, 0] - centers[j, 0], 
                               centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc * 0.5
                    radii[j] -= exc * 0.5
                    changed = True
        if not changed:
            break
    return centers, np.maximum(radii, 0.0)

def generate_initializations(rng):
    """Generate diverse starting configurations for centers."""
    configs = []
    
    # 1. Hexagonal lattices with varying spacing
    for seed in range(15):
        r_rng = np.random.RandomState(seed)
        c = np.zeros((N, 2))
        idx = 0
        sp = 0.15 + r_rng.uniform(0.0, 0.06)
        y = sp / 2
        row = 0
        while idx < N and y < 1.0 - sp / 2:
            x = sp / 2 + (row % 2) * sp / 2
            while x < 1.0 - sp / 2 and idx < N:
                c[idx] = [x, y]
                idx += 1
                x += sp
            y += sp * math.sqrt(3) / 2
            row += 1
        while idx < N:
            c[idx] = r_rng.uniform(0.1, 0.9, 2)
            idx += 1
        c += r_rng.normal(0, 0.005, c.shape)
        configs.append(np.clip(c, 0.02, 0.98))
        
    # 2. Square grids
    for seed in range(10):
        r_rng = np.random.RandomState(seed + 100)
        c = np.zeros((N, 2))
        idx = 0
        step = 0.16 + r_rng.uniform(-0.02, 0.02)
        y = step / 2
        while y < 1.0 - step / 2 and idx < N:
            x = step / 2
            while x < 1.0 - step / 2 and idx < N:
                c[idx] = [x, y]
                idx += 1
                x += step
            y += step
        while idx < N:
            c[idx] = r_rng.uniform(0.1, 0.9, 2)
            idx += 1
        configs.append(np.clip(c, 0.02, 0.98))
        
    # 3. Random uniform
    for seed in range(20):
        r_rng = np.random.RandomState(seed + 200)
        configs.append(r_rng.uniform(0.1, 0.9, (N, 2)))
        
    # 4. Corner-biased (exploits boundary clearance for larger circles)
    for seed in range(10):
        r_rng = np.random.RandomState(seed + 300)
        c = r_rng.uniform(0.15, 0.85, (N, 2))
        corners = [[0.07, 0.07], [0.93, 0.07], [0.07, 0.93], [0.93, 0.93]]
        for i in range(8):
            c[i] = corners[i % 4] + r_rng.uniform(-0.03, 0.03, 2)
        configs.append(np.clip(c, 0.02, 0.98))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds_joint = [(0.0, 1.0)] * (2 * N) + [(1e-7, 0.5)] * N
    cons_joint = {'type': 'ineq', 'fun': constraints_joint}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)
    inits = generate_initializations(rng)
    
    # Phase 1: Powell optimization on centers using LP for radii evaluation
    for c0 in inits:

            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt = solve_lp_radii(c_opt)

                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()

            continue

    if best_centers is not None:

        x0 = np.zeros(3 * N)

        x0[N:2*N] = best_centers[:, 1]
        x0[2*N:] = r0
        
        try:
            res = minimize(objective_joint, x0, method='SLSQP', bounds=bounds_joint,
                           constraints=cons_joint,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})

            r_opt, s_opt = solve_lp_radii(c_opt)

                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Phase 3: Simulated Annealing / Basin Hopping on centers
    if best_centers is not None:
        curr_c = best_centers.copy()
        curr_r = best_radii.copy()
        curr_s = best_sum
        
        for step in range(150):
            # Decaying noise schedule
            noise_scale = 0.008 * math.exp(-step / 60.0)
            c_pert = curr_c + rng.normal(0, noise_scale, (N, 2))

            r_pert, s_pert = solve_lp_radii(c_pert)

                curr_r = r_pert
                curr_s = s_pert

                    r_pol = np.maximum(r_pol, 1e-5)
                    x_pol = np.zeros(3 * N)
                    x_pol[0:N] = best_centers[:, 0]

                        c_pol = np.column_stack((res_pol.x[0:N], res_pol.x[N:2*N]))
                        r_pol, s_pol = solve_lp_radii(c_pol)
                        if s_pol > best_sum:
                            best_sum = s_pol

                    except Exception:
                        pass

        best_centers = inits[0]
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    # Phase 4: Strict post-processing to guarantee validator compliance
    best_centers, best_radii = make_valid(best_centers.copy(), best_radii.copy())
    
    return best_centers, best_radii, float(np.sum(best_radii))
