import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute LP constraint matrix structure for speed
A_ub_lp = np.zeros((NUM_PAIRS, N))
A_ub_lp[np.arange(NUM_PAIRS), I] = 1.0
A_ub_lp[np.arange(NUM_PAIRS), J] = 1.0

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds_r = []
    for c in centers:
        ub = min(c[0], 1.0 - c[0], c[1], 1.0 - c[1])
        bounds_r.append((0.0, max(1e-9, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub_lp, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints (must be >= 0):
    - Pairwise distance >= sum of radii
    - Boundary clearance: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dists = np.hypot(dx, dy)
    
    c = np.empty(4 * N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    c[4*N:] = dists - (r[I] + r[J])
    return c

def generate_configs(rng):
    """Generate diverse initial center configurations."""
    configs = []
    
    # 1. Hexagonal lattices with varying spacings
    for s in np.linspace(0.14, 0.22, 12):
        c = np.zeros((N, 2))
        idx = 0
        row = 0
        y = s / 2
        while idx < N and y < 1.0 - s / 2:
            x_start = s / 2 + (row % 2) * s / 2
            col = 0
            while x_start + col * s < 1.0 - s / 2 and idx < N:
                c[idx] = [x_start + col * s, y]
                idx += 1
                col += 1
            y += s * np.sqrt(3) / 2
            row += 1
        while idx < N:
            c[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
        c += rng.normal(0, 0.005, c.shape)
        configs.append(np.clip(c, 0.02, 0.98))

    # 2. Square grids
    for step in np.linspace(0.16, 0.24, 10):
        c = np.zeros((N, 2))
        idx = 0
        y = step / 2
        while y < 1.0 - step / 2 and idx < N:
            x = step / 2
            while x < 1.0 - step / 2 and idx < N:
                c[idx] = [x, y]
                idx += 1
                x += step
            y += step
        while idx < N:
            c[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
        c += rng.normal(0, 0.005, c.shape)
        configs.append(np.clip(c, 0.02, 0.98))

    # 3. Corner/Edge biased starts to exploit boundary clearance
    for _ in range(20):
        c = rng.uniform(0.15, 0.85, (N, 2))
        corners = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        for i in range(8):
            c[i] = corners[i % 4] + rng.uniform(-0.05, 0.05, 2)
        c += rng.normal(0, 0.01, c.shape)
        configs.append(np.clip(c, 0.02, 0.98))

    # 4. Random uniform
    for _ in range(25):
        configs.append(np.clip(rng.uniform(0.1, 0.9, (N, 2)) + rng.normal(0, 0.01, (N, 2)), 0.02, 0.98))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    rng = np.random.RandomState(42)
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = generate_configs(rng)
    
    # Phase 1: Broad multi-start SLSQP
    for c0 in configs:
        r0, _ = solve_lp_radii(c0)
        r0 = np.maximum(r0 * 0.90, 1e-4)
        x0 = np.zeros(3 * N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r0
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            cx = res.x[0::3]
            cy = res.x[1::3]
            curr_c = np.column_stack((cx, cy))
            r_opt, s_opt = solve_lp_radii(curr_c)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = curr_c.copy()
                best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Simulated Annealing Basin Hopping
    if best_centers is not None:
        curr_c = best_centers.copy()
        curr_r = best_radii.copy()
        curr_s = best_sum
        
        for step in range(180):
            # Adaptive noise schedule
            scale = 0.008 * np.exp(-step / 70.0)
            
            # Random perturbation
            c_pert = curr_c + rng.normal(0, scale, (N, 2))
            c_pert = np.clip(c_pert, 0.01, 0.99)
            
            # Periodic swap to break symmetry and escape local traps
            if step % 12 == 0 and N > 2:
                idx1, idx2 = rng.choice(N, 2, replace=False)
                c_pert[[idx1, idx2]] = c_pert[[idx2, idx1]]
                
            r_pert, _ = solve_lp_radii(c_pert)
            if np.sum(r_pert) > 0.1:
                r_pert = np.maximum(r_pert * 0.92, 1e-4)
                x0 = np.zeros(3 * N)
                x0[0::3] = c_pert[:, 0]
                x0[1::3] = c_pert[:, 1]
                x0[2::3] = r_pert
                
                try:
                    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                                   options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
                    cx = res.x[0::3]
                    cy = res.x[1::3]
                    c_opt = np.column_stack((cx, cy))
                    r_opt, s_opt = solve_lp_radii(c_opt)
                    
                    if s_opt > curr_s:
                        curr_c, curr_r, curr_s = c_opt, r_opt, s_opt
                        if s_opt > best_sum:
                            best_sum = s_opt
                            best_centers = curr_c.copy()
                            best_radii = curr_r.copy()
                    elif rng.random() < np.exp((s_opt - curr_s) / max(0.001, scale)):
                        curr_c, curr_r, curr_s = c_opt, r_opt, s_opt
                except Exception:
                    pass
                    
    # Phase 3: Coordinate Descent Polish (move one circle at a time)
    if best_centers is not None:
        c_polish = best_centers.copy()
        r_polish = best_radii.copy()
        s_polish = best_sum
        
        for _ in range(40):
            improved = False
            for i in range(N):
                best_local_sum = s_polish
                best_local_c = c_polish[i].copy()
                
                for _ in range(10):
                    cand = c_polish.copy()
                    cand[i] = c_polish[i] + rng.normal(0, 0.0025, 2)
                    cand[i] = np.clip(cand[i], 0.01, 0.99)
                    r_cand, s_cand = solve_lp_radii(cand)
                    if s_cand > best_local_sum + 1e-7:
                        best_local_sum = s_cand
                        best_local_c = cand[i].copy()
                
                c_polish[i] = best_local_c
                r_polish, s_polish = solve_lp_radii(c_polish)
                if s_polish > best_sum:
                    best_sum = s_polish
                    best_centers = c_polish.copy()
                    best_radii = r_polish.copy()
                    improved = True
            if not improved:
                break
                
    # Fallback safety net
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    # Phase 4: Strict post-processing to guarantee validator compliance
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(radii[i], 0.0)
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-10:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc/2.0
                    radii[j] -= exc/2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
