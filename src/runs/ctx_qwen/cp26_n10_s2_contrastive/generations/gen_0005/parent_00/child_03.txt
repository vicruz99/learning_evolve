N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute LP constraint structure
A_ub_lp = np.zeros((NUM_PAIRS, N))
A_ub_lp[np.arange(NUM_PAIRS), I] = 1.0
A_ub_lp[np.arange(NUM_PAIRS), J] = 1.0


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints(x):
    """Inequality constraints (must be >= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]

    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dists = np.hypot(dx, dy)
    c_overlap = dists - (r[I] + r[J])

    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])


def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)

    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)

    bounds_r = []
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds_r.append((0.0, max(1e-9, mx)))

    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub_lp, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success:
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(n), 0.0


def force_relax(centers, radii, steps=80, strength=0.08):
    """Force-directed relaxation to push circles apart."""
    c = centers.copy()
    r = radii.copy()
    n = c.shape[0]

    for step in range(steps):
        forces = np.zeros_like(c)
        current_strength = strength * (1.0 - step / (steps + 1))

        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = np.hypot(dx, dy)
                if d < r[i] + r[j] and d > 1e-10:
                    overlap = r[i] + r[j] - d
                    f = overlap * current_strength / max(d, 1e-10)
                    fx = dx * f
                    fy = dy * f
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy

        # Boundary repulsion
        for i in range(n):
            if c[i, 0] - r[i] < 0:
                forces[i, 0] += (r[i] - c[i, 0]) * current_strength * 2
            if c[i, 0] + r[i] > 1:
                forces[i, 0] -= (c[i, 0] + r[i] - 1) * current_strength * 2
            if c[i, 1] - r[i] < 0:
                forces[i, 1] += (r[i] - c[i, 1]) * current_strength * 2
            if c[i, 1] + r[i] > 1:
                forces[i, 1] -= (c[i, 1] + r[i] - 1) * current_strength * 2

        c += forces * 0.5
        c = np.clip(c, 1e-6, 1.0 - 1e-6)

    return c


def generate_hex_init(spacing, rng, corner_bias=0.0):

            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2

def generate_row_pattern(rows, rng):

        idx += 1

def coordinate_descent(centers, radii, rng, steps=30):
    """Optimize one circle position at a time."""
    c = centers.copy()
    r = radii.copy()
    n = c.shape[0]

    for _ in range(steps):
        improved = False
        for i in range(n):
            best_sum = np.sum(r)
            best_pos = c[i].copy()

            # Try multiple perturbations for circle i
            for _ in range(12):
                scale = rng.uniform(0.001, 0.008)
                cand = c.copy()
                cand[i] = c[i] + rng.normal(0, scale, 2)
                cand[i] = np.clip(cand[i], 0.01, 0.99)

                r_cand, s_cand = solve_lp_radii(cand)
                if s_cand > best_sum + 1e-8:
                    best_sum = s_cand
                    best_pos = cand[i].copy()

            c[i] = best_pos
            if best_sum > np.sum(r) + 1e-8:
                r, _ = solve_lp_radii(c)
                improved = True

        if not improved:
            break

    return c, r, np.sum(r)


def pairwise_perturbation(centers, radii, rng, n_tries=50):
    """Perturb pairs of circles to find improvements."""
    c = centers.copy()
    r = radii.copy()
    best_sum = np.sum(r)

    for _ in range(n_tries):
        i, j = rng.choice(N, 2, replace=False)
        scale = rng.uniform(0.002, 0.012)
        cand = c.copy()
        cand[i] = c[i] + rng.normal(0, scale, 2)
        cand[j] = c[j] + rng.normal(0, scale, 2)

        r_cand, s_cand = solve_lp_radii(cand)
        if s_cand > best_sum:
            c = cand
            r = r_cand
            best_sum = s_cand
