import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute sparse-ish structure for LP pairwise constraints
A_ub_lp = np.zeros((NUM_PAIRS, N))
A_ub_lp[np.arange(NUM_PAIRS), I] = 1.0
A_ub_lp[np.arange(NUM_PAIRS), J] = 1.0

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds_r = []
    for i in range(n):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds_r.append((0.0, max(1e-9, mx)))
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_lp, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    # Fallback to interior-point if highs fails
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_lp, b_ub=b_ub, bounds=bounds_r, method='interior-point')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def get_score(centers):
    """Returns the maximum sum of radii for a given center configuration."""
    centers = np.clip(centers, 0.005, 0.995)
    _, s = solve_lp_radii(centers)
    return s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.RandomState(42)
    configs = []
    
    # Phase 0: Generate diverse initial configurations
    # 1. Hexagonal lattices with varying spacing
    for s in np.linspace(0.16, 0.20, 6):
        c = np.zeros((N, 2))
        idx = 0
        y = s/2
        row = 0
        while idx < N and y < 1.0 - s/2:
            x = s/2 + (row % 2) * s/2
            while x < 1.0 - s/2 and idx < N:
                c[idx] = [x, y]
                idx += 1
                x += s
            y += s * np.sqrt(3) / 2
            row += 1
        while idx < N:
            c[idx] = rng.uniform(0.2, 0.8, 2)
            idx += 1
        configs.append(c + rng.normal(0, 0.005, c.shape))
        
    # 2. Square grids
    for s in np.linspace(0.17, 0.21, 5):
        c = np.zeros((N, 2))
        idx = 0
        y = s/2
        while y < 1.0 - s/2 and idx < N:
            x = s/2
            while x < 1.0 - s/2 and idx < N:
                c[idx] = [x, y]
                idx += 1
                x += s
            y += s
        while idx < N:
            c[idx] = rng.uniform(0.2, 0.8, 2)
            idx += 1
        configs.append(c + rng.normal(0, 0.005, c.shape))
        
    # 3. Random with boundary/corner bias
    for _ in range(25):
        c = rng.uniform(0.1, 0.9, (N, 2))
        c[:4] = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        configs.append(c + rng.normal(0, 0.02, c.shape))
        
    configs = np.clip(configs, 0.02, 0.98)
    
    bounds_c = [(0.01, 0.99)] * (2 * N)

        curr_r = best_radii.copy()

                        best_local_c = cand[i].copy()

                r_opt, s_opt = solve_lp_radii(curr_c)
                curr_r = r_opt
                if s_opt > curr_s:
                    curr_s = s_opt
                    improved = True
                    
            if not improved:
                break
                
        if curr_s > best_sum:
            best_sum = curr_s
            best_centers = curr_c.copy()
            best_radii = curr_r.copy()

    # Fallback safety net
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    # Phase 4: Strict post-processing to guarantee validator compliance
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(radii[i], 0.0)
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-10:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc/2.0
                    radii[j] -= exc/2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
