import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((N_PAIRS, n))
    A_ub[np.arange(N_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(N_PAIRS), J_IDX] = 1.0
    
    dx = centers[I_IDX, 0] - centers[J_IDX, 0]
    dy = centers[I_IDX, 1] - centers[J_IDX, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds_r = []
    for i in range(n):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds_r.append((0.0, max(1e-9, mx)))
        
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success:
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(n), 0.0

def resolve_overlaps(centers, radii, steps=200):
    """Deterministically push overlapping circles and boundary-violating circles apart."""
    c = centers.copy()
    r = radii.copy()
    for _ in range(steps):
        max_ov = 0.0
        # Pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                dx = c[i,0] - c[j,0]
                dy = c[i,1] - c[j,1]
                d = np.hypot(dx, dy)
                if d < r[i] + r[j] and d > 1e-9:
                    ov = r[i] + r[j] - d
                    if ov > max_ov:
                        max_ov = ov
                        inv_d = 0.5 / d
                        c[i,0] += dx * inv_d * ov
                        c[i,1] += dy * inv_d * ov
                        c[j,0] -= dx * inv_d * ov
                        c[j,1] -= dy * inv_d * ov
        
        # Boundary overlaps
        for i in range(N):
            if c[i,0] - r[i] < 0:
                c[i,0] = r[i] + 1e-7
            if c[i,0] + r[i] > 1.0:
                c[i,0] = 1.0 - r[i] - 1e-7
            if c[i,1] - r[i] < 0:
                c[i,1] = r[i] + 1e-7
            if c[i,1] + r[i] > 1.0:
                c[i,1] = 1.0 - r[i] - 1e-7
                
        if max_ov < 1e-9:
            break
    return np.clip(c, 1e-6, 1.0 - 1e-6)

def scale_and_relax(centers, radii, scale=1.001, max_iter=40):
    """Iteratively scale up radii and resolve overlaps to find denser packings."""
    c = centers.copy()
    r = radii.copy()
    prev_sum = 0.0
    for _ in range(max_iter):
        r = r * scale
        c = resolve_overlaps(c, r)
        r_new, s_new = solve_lp_radii(c)
        if s_new <= prev_sum:
            break
        prev_sum = s_new
        r = r_new
    return c, r

def objective_joint(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_joint(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap (>= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    dists = np.hypot(dx, dy)
    c_overlap = dists - (r[I_IDX] + r[J_IDX])
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])

def generate_init(seed, style='hex'):
    """Generate structured or random initial center configurations."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    
    if style == 'hex':
        s = 0.15 + rng.uniform(0, 0.06)
        idx = 0
        row = 0
        y = s/2 + rng.uniform(0, 0.02)
        while idx < N and y < 1.0 - s/2:
            x_start = s/2 + (row % 2) * s/2 + rng.uniform(0, 0.02)
            col = 0
            while x_start + col*s < 1.0 - s/2 and idx < N:
                centers[idx] = [x_start + col*s, y]
                idx += 1
                col += 1
            y += s * np.sqrt(3) / 2
            row += 1
    elif style == 'grid':
        step = 0.18 + rng.uniform(-0.02, 0.02)
        idx = 0
        y = step/2
        while y < 1.0 - step/2 and idx < N:
            x = step/2
            while x < 1.0 - step/2 and idx < N:
                centers[idx] = [x, y]
                x += step
                idx += 1
            y += step
    else:
        centers = rng.uniform(0.1, 0.9, (N, 2))
        
    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': constraints_joint}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)
    candidates = []
    
    # Phase 1: Generate diverse initial configurations
    for seed in range(40):
        style = 'hex' if seed < 25 else 'grid' if seed < 35 else 'random'
        candidates.append(generate_init(seed, style))
        
    # Corner/edge biased starts
    for seed in range(15):
        c = rng.uniform(0.05, 0.95, (N, 2))
        corners = [[0.07, 0.07], [0.93, 0.07], [0.07, 0.93], [0.93, 0.93]]
        for i in range(8):
            c[i] = corners[i%4] + rng.uniform(-0.03, 0.03, 2)
        candidates.append(np.clip(c, 0.02, 0.98))

    # Phase 2: Scale-and-Relax heuristic
    for c_init in candidates:
        r_init, _ = solve_lp_radii(c_init)
        r_init = np.maximum(r_init, 1e-4)
        
        # Run scale-and-relax
        c_rel, r_rel = scale_and_relax(c_init, r_init, scale=1.0015, max_iter=60)
        
        if np.sum(r_rel) > best_sum:
            best_sum = np.sum(r_rel)
            best_centers = c_rel.copy()
            best_radii = r_rel.copy()

    # Phase 3: Joint SLSQP optimization on best candidates
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = np.maximum(best_radii * 0.95, 1e-5)
        
        try:
            res = minimize(objective_joint, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons_opt, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            cx = res.x[0::3]
            cy = res.x[1::3]
            c_opt = np.column_stack((cx, cy))
            r_opt, s_opt = solve_lp_radii(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            pass

        # Perturbation + SLSQP refinement
        for _ in range(20):
            scale = 0.004 * rng.uniform(0.5, 2.0)
            c_pert = best_centers + rng.normal(0, scale, (N, 2))

            x0_p = np.zeros(3 * N)
            x0_p[0::3] = c_pert[:, 0]

                c_opt_p = np.column_stack((cx_p, cy_p))

                    best_centers = c_opt_p.copy()

            except Exception:
                pass

    # Fallback safety net
    if best_centers is None:

        best_radii, best_sum = solve_lp_radii(best_centers)
        
    # Phase 4: Strict post-processing to guarantee validity

    radii = best_radii.copy()

        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(radii[i], 0.0)

    for _ in range(150):
        changed = False

                    radii[i] -= exc/2.0
                    radii[j] -= exc/2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
