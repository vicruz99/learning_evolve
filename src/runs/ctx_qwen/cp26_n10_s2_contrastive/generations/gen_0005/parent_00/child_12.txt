import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute sparse structure for LP pairwise constraints
A_ub_lp = np.zeros((NUM_PAIRS, N))
A_ub_lp[np.arange(NUM_PAIRS), I] = 1.0
A_ub_lp[np.arange(NUM_PAIRS), J] = 1.0

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    c_obj = -np.ones(N)
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds_r = []
    for i in range(N):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(1e-9, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub_lp, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.zeros(N)

def objective_joint(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_joint(x):
    """Inequality constraints (must be >= 0): boundary clearance and pairwise non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dists = np.hypot(dx, dy)
    c_overlap = dists - (r[I] + r[J])
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])

cons_joint = {'type': 'ineq', 'fun': constraints_joint}
bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N

def polish_slsqp(centers):
    """Run SLSQP to jointly optimize centers and radii from a starting point."""
    r_init = solve_lp_radii(centers) * 0.95
    r_init = np.maximum(r_init, 1e-5)
    x0 = np.zeros(3*N)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = r_init
    
    try:
        res = minimize(objective_joint, x0, method='SLSQP', bounds=bounds_opt,
                       constraints=cons_joint, options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
        c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
        r_opt = solve_lp_radii(c_opt)
        return c_opt, r_opt
    except Exception:
        return centers, r_init

def generate_starts(rng):
    """Generate diverse initial center configurations."""
    starts = []
    # 1. Hexagonal lattices with varying spacings and offsets
    for seed in range(60):
        c = np.zeros((N, 2))
        idx = 0
        s = 0.14 + rng.uniform(0, 0.08)
        margin = 0.02 + rng.uniform(0, 0.04)
        y = margin
        row = 0
        while idx < N and y < 1.0 - margin:
            x = margin + (row % 2) * s / 2.0
            while x < 1.0 - margin and idx < N:
                c[idx] = [x, y]
                idx += 1
                x += s
            y += s * np.sqrt(3) / 2.0
            row += 1
        while idx < N:
            c[idx] = rng.uniform(margin, 1.0-margin, 2)
            idx += 1
        c += rng.normal(0, 0.005, c.shape)
        starts.append(np.clip(c, 0.01, 0.99))
        
    # 2. Square grids
    for seed in range(20):
        c = np.zeros((N, 2))
        idx = 0
        step = 0.15 + rng.uniform(0, 0.08)
        margin = 0.02 + rng.uniform(0, 0.03)
        y = margin
        while y < 1.0 - margin and idx < N:
            x = margin
            while x < 1.0 - margin and idx < N:
                c[idx] = [x, y]
                idx += 1
                x += step
            y += step
        while idx < N:
            c[idx] = rng.uniform(margin, 1.0-margin, 2)
            idx += 1
        c += rng.normal(0, 0.004, c.shape)
        starts.append(np.clip(c, 0.01, 0.99))
        
    # 3. Random uniform placements
    for _ in range(30):
        starts.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_c = None
    best_r = None
    
    starts = generate_starts(rng)
    
    # Phase 1: Initial polish of diverse starts
    for c0 in starts:
        c_opt, r_opt = polish_slsqp(c0)
        s = np.sum(r_opt)
        if s > best_sum:
            best_sum = s
            best_c = c_opt.copy()
            best_r = r_opt.copy()
            
    # Phase 2: Simulated Annealing Basin Hopping
    if best_c is not None:
        curr_c = best_c.copy()
        curr_r = best_r.copy()
        curr_s = best_sum
        temp = 0.015
        
        for step in range(400):
            scale = 0.008 * (0.96 ** (step // 15))
            c_pert = curr_c + rng.normal(0, scale, (N, 2))
            c_pert = np.clip(c_pert, 0.01, 0.99)
            
            r_pert = solve_lp_radii(c_pert)
            s_pert = np.sum(r_pert)
            
            # Simulated annealing acceptance criterion
            delta = s_pert - curr_s
            if delta > 0 or rng.random() < min(1.0, np.exp(delta / max(temp, 1e-4))):
                curr_c = c_pert.copy()
                curr_r = r_pert.copy()
                curr_s = s_pert
                
                # Polish accepted jumps with SLSQP
                c_pol, r_pol = polish_slsqp(curr_c)
                s_pol = np.sum(r_pol)
                if s_pol > best_sum:
                    best_sum = s_pol
                    best_c = c_pol.copy()
                    best_r = r_pol.copy()
                    curr_c = c_pol.copy()
                    curr_r = r_pol.copy()
                    curr_s = s_pol
            temp *= 0.995

    # Phase 3: Coordinate Descent on Centers (move one circle at a time)
    if best_c is not None:
        curr_c = best_c.copy()
        for _ in range(50):
            improved = False
            for i in range(N):
                best_local_s = np.sum(solve_lp_radii(curr_c))
                best_local_pos = curr_c[i].copy()
                
                # Try multiple local perturbations for circle i
                for _ in range(15):
                    cand_c = curr_c.copy()
                    cand_c[i] = curr_c[i] + rng.normal(0, 0.003, 2)
                    cand_c[i] = np.clip(cand_c[i], 0.01, 0.99)
                    s_cand = np.sum(solve_lp_radii(cand_c))
                    if s_cand > best_local_s + 1e-7:
                        best_local_s = s_cand
                        best_local_pos = cand_c[i].copy()
                curr_c[i] = best_local_pos
                if best_local_s > best_sum:
                    best_sum = best_local_s
                    best_c = curr_c.copy()
                    best_r = solve_lp_radii(curr_c)
                    improved = True
            if not improved:
                break
                
    # Phase 4: Final SLSQP global polish
    if best_c is not None:
        c_final, r_final = polish_slsqp(best_c)
        s_final = np.sum(r_final)
        if s_final > best_sum:
            best_c = c_final
            best_r = r_final
            best_sum = s_final
            
    # Fallback safety net
    if best_c is None:
        best_c = starts[0]
        best_r = solve_lp_radii(best_c)
        best_sum = np.sum(best_r)
        
    # Phase 5: Strict post-processing to guarantee validator compliance
    radii = best_r.copy()
    centers = best_c.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(radii[i], 0.0)
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-10:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc/2.0
                    radii[j] -= exc/2.0
                    changed = True
        if not changed: break
        
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
