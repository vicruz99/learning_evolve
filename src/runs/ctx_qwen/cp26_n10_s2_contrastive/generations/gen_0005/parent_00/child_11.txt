import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

# Precompute constant structure for LP pairwise constraints
A_LP = np.zeros((NUM_PAIRS, N))
A_LP[np.arange(NUM_PAIRS), I_IDX] = 1.0
A_LP[np.arange(NUM_PAIRS), J_IDX] = 1.0

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    dx = centers[I_IDX, 0] - centers[J_IDX, 0]
    dy = centers[I_IDX, 1] - centers[J_IDX, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = []
    for i in range(n):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds.append((0.0, max(1e-9, mx)))
        
    try:
        res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    try:
        res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='interior-point')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def obj_func(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def cons_func(x):
    """Inequality constraints (must be >= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    dist = np.hypot(dx, dy)
    c_ov = dist - (r[I_IDX] + r[J_IDX])
    c_b = np.concatenate([cx-r, 1.0-cx-r, cy-r, 1.0-cy-r])
    return np.concatenate([c_ov, c_b])

def generate_hex_init(pattern, rng):
    """Generate structured hexagonal initial center configurations."""
    centers = np.zeros((N, 2))
    idx = 0
    y = 0.05
    dy = 0.90 / (len(pattern) - 0.5)
    for r_idx, cnt in enumerate(pattern):
        shift = 0.04 if r_idx % 2 == 1 else 0.0
        x = 0.05 + shift
        dx = 0.85 / (cnt - 0.5) if cnt > 1 else 0.0
        for _ in range(cnt):
            if idx < N:
                centers[idx] = [x + rng.uniform(-0.02, 0.02), y + rng.uniform(-0.02, 0.02)]
                idx += 1
                x += dx
        y += dy * np.sqrt(3) / 2
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    return np.clip(centers, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    rng = np.random.RandomState(42)
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    
    # Diverse row patterns covering different aspect ratios and symmetries
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [7,6,5,4,4], [4,5,6,5,6], 
        [8,6,5,4,3], [5,5,5,5,6], [6,6,4,5,5], [7,7,5,4,3],
        [6,6,6,6,2], [9,6,5,4,2], [4,4,4,4,4,4,2], [5,4,5,4,5,3],
        [6,4,6,4,6], [7,5,6,4,4], [5,5,4,5,5,2]
    ]
    
    inits = []
    for pat in patterns:
        inits.append(generate_hex_init(pat, rng))
    for _ in range(30):
        inits.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Phase 1: SLSQP polish on initial configurations
    for c0 in inits:
        r0, _ = solve_lp_radii(c0)
        r0 = np.maximum(r0, 1e-4)
        x0 = np.zeros(3*N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r0 * 0.90
        
        try:
            res = minimize(obj_func, x0, method='SLSQP', bounds=bounds_opt, 
                           constraints={'type': 'ineq', 'fun': cons_func},
                           options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            cx = res.x[0::3]
            cy = res.x[1::3]
            cc = np.column_stack((cx, cy))
            rr, ss = solve_lp_radii(cc)
            if ss > best_sum:
                best_sum = ss
                best_c = cc.copy()
                best_r = rr.copy()
        except Exception:
            pass

    # Phase 2: Simulated Annealing on centers using LP evaluation
    if best_c is not None:
        cur_c = best_c.copy()
        cur_r, cur_s = solve_lp_radii(cur_c)
        best_c, best_r, best_s = cur_c.copy(), cur_r.copy(), cur_s
        
        temp = 0.012
        decay = 0.9985
        
        for step in range(4500):
            # Perturb 1-3 random circles to explore configuration space
            num_pert = rng.choice([1, 2, 3])
            idxs = rng.choice(N, num_pert, replace=False)
            new_c = cur_c.copy()
            new_c[idxs] += rng.normal(0, temp, (num_pert, 2))
            new_c = np.clip(new_c, 0.01, 0.99)
            
            r_new, s_new = solve_lp_radii(new_c)
            delta = s_new - best_s
            
            # Metropolis acceptance criterion
            if delta > 0 or rng.random() < np.exp(delta / max(temp, 1e-9)):
                cur_c = new_c
                cur_r, cur_s = r_new, s_new
                if s_new > best_s:
                    best_c, best_r, best_s = cur_c.copy(), cur_r.copy(), s_new
                    # Occasionally polish with SLSQP after significant improvement
                    if rng.random() < 0.25:
                        x0 = np.zeros(3*N)
                        x0[0::3] = best_c[:, 0]
                        x0[1::3] = best_c[:, 1]
                        x0[2::3] = best_r * 0.95
                        try:
                            res = minimize(obj_func, x0, method='SLSQP', bounds=bounds_opt, 
                                           constraints={'type': 'ineq', 'fun': cons_func},
                                           options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                            cx = res.x[0::3]
                            cy = res.x[1::3]
                            cc = np.column_stack((cx, cy))
                            rr, ss = solve_lp_radii(cc)
                            if ss > best_s:
                                best_c, best_r, best_s = cc.copy(), rr.copy(), ss
                                cur_c, cur_r, cur_s = best_c.copy(), best_r.copy(), best_s
                        except Exception:
                            pass
            temp *= decay
            
    # Phase 3: Fine coordinate descent polishing
    if best_c is not None:
        for _ in range(15):
            improved = False
            for i in range(N):
                local_best_s = best_s
                local_best_c = best_c[i].copy()
                for _ in range(15):
                    cand = best_c.copy()
                    cand[i] += rng.normal(0, 0.0008, 2)
                    cand[i] = np.clip(cand[i], 0.01, 0.99)
                    _, s = solve_lp_radii(cand)
                    if s > local_best_s + 1e-7:
                        local_best_s = s
                        local_best_c = cand[i].copy()
                best_c[i] = local_best_c
                _, new_s = solve_lp_radii(best_c)
                if new_s > best_s + 1e-7:
                    best_s = new_s
                    best_r, _ = solve_lp_radii(best_c)
                    improved = True
            if not improved:
                break
                
    # Fallback safety net
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum = solve_lp_radii(best_c)
        
    # Phase 4: Strict post-processing to guarantee validator compliance
    centers = best_c.copy()
    radii = best_r.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(radii[i], 0.0)
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-10:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc/2.0
                    radii[j] -= exc/2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
