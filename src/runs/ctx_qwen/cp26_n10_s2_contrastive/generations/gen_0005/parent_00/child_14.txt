import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute LP constraint matrix structure for speed
A_ub_lp = np.zeros((NUM_PAIRS, N))
A_ub_lp[np.arange(NUM_PAIRS), I] = 1.0
A_ub_lp[np.arange(NUM_PAIRS), J] = 1.0

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds_r = []
    for i in range(n):
        mx = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
        bounds_r.append((0.0, max(1e-9, mx)))
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_lp, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 1e-6)

def lp_objective(centers_flat):
    """Objective for center optimization: negative sum of optimal radii."""
    centers = centers_flat.reshape(N, 2)
    radii = solve_lp_radii(centers)
    return -np.sum(radii)

def joint_objective(x):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def joint_constraints(x):
    """Inequality constraints for joint optimization (must be >= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dists = np.hypot(dx, dy)
    c_overlap = dists - (r[I] + r[J])
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])

def generate_hex_init(seed, spacing=0.18, margin=0.04, noise_scale=0.005):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    row = 0
    y = margin
    while idx < N and y < 1.0 - margin:
        x = margin + (row % 2) * spacing / 2.0
        while x < 1.0 - margin and idx < N:
            centers[idx] = [x, y]
            idx += 1
            x += spacing
        y += spacing * np.sqrt(3) / 2.0
        row += 1
    while idx < N:
        centers[idx] = rng.uniform(0.2, 0.8, 2)
        idx += 1
    centers += rng.normal(0, noise_scale, centers.shape)
    return np.clip(centers, 0.02, 0.98)

                           bounds=[(0.02, 0.98)] * (2 * N),

    return centers, radii, float(np.sum(radii))
