

Looking at this problem, I need to improve a circle packing from 2.634 to 2.636. The key insight is that for fixed centers, the optimal radii can be found via LP. So the problem reduces to finding better center configurations. I'll use a combination of structured initializations, aggressive basin hopping, and coordinate descent refinement.

<strategy>
1. **Diverse structured initializations**: Generate hexagonal lattices, square grids, and specific row patterns (like [6,5,6,5,4]) that are known to be efficient for packing. Add corner-biased starts since circles near corners can grow larger.

2. **LP-based radius optimization**: For any center configuration, use HiGHS LP solver to find the optimal radii. This gives a fast evaluation of any center arrangement.

3. **Joint SLSQP optimization**: After LP evaluation, jointly optimize centers and radii with SLSQP to refine.

4. **Aggressive basin hopping**: Systematically perturb the best configuration with decaying noise scales, accepting improvements immediately and polishing with SLSQP.

5. **Coordinate descent**: Optimize one circle at a time by trying multiple random perturbations for each circle, which can escape local minima that joint optimization gets stuck in.

6. **Strict post-processing**: Ensure all constraints are satisfied within numerical tolerance by clamping boundary violations and iteratively resolving overlaps.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute LP structure
A_lp = np.zeros((NUM_PAIRS, N))
A_lp[np.arange(NUM_PAIRS), I] = 1.0
A_lp[np.arange(NUM_PAIRS), J] = 1.0

def objective(x):
    return -np.sum(x[2::3])

def constraints(x):
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dists = np.hypot(dx, dy)
    c_overlap = dists - (r[I] + r[J])
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds_r = []
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds_r.append((0.0, max(1e-9, mx)))
        
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(-np.ones(N), A_ub=A_lp, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(N), 0.0

def polish(x0, maxiter=10000):
    """Run SLSQP optimization from x0."""
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                      constraints={'type': 'ineq', 'fun': constraints},
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        return res.x
    except Exception:
        return x0

def generate_hex_lattice(seed, spacing=0.16, margin=0.04):
    """Generate hexagonal lattice with given spacing."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    row = 0
    y = margin
    
    while idx < N and y < 1.0 - margin:
        x_start = margin + (row % 2) * spacing / 2
        col = 0
        while x_start + col * spacing < 1.0 - margin and idx < N:
            centers[idx, 0] = x_start + col * spacing
            centers[idx, 1] = y
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2
        row += 1
        
    while idx < N:
        centers[idx] = rng.uniform(margin, 1.0 - margin, 2)
        idx += 1
        
    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.02, 0.98)

def generate_row_pattern(row_counts, base_shift=0.085, base_x=0.06, spacing=0.17):
    """Generate pattern with specified row counts."""
    centers = np.zeros((N, 2))
    idx = 0
    y = 0.06
    dy = 0.88 / (len(row_counts) - 0.5)
    
    for r_idx, cnt in enumerate(row_counts):
        shift = 0.0 if r_idx % 2 == 0 else base_shift
        x = base_x + shift
        for _ in range(cnt):
            if idx < N:
                centers[idx] = [x, y]
                idx += 1
            x += spacing
        y += dy
        
    while idx < N:
        centers[idx] = [0.5, 0.5]
        idx += 1
        
    return np.clip(centers, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = []
    rng = np.random.RandomState(42)
    
    # 1. Hexagonal lattices with varying parameters
    for seed in range(50):
        rng_local = np.random.RandomState(seed)
        sp = 0.13 + rng_local.uniform(0, 0.09)
        margin = 0.02 + rng_local.uniform(0, 0.04)
        configs.append(generate_hex_lattice(seed, sp, margin))
    
    # 2. Row patterns - many variations
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [7, 6, 5, 4, 4],
        [4, 5, 6, 5, 6], [8, 6, 5, 4, 3], [6, 6, 6, 6, 2],
        [6, 7, 6, 4, 3], [5, 7, 7, 4, 3], [7, 7, 6, 3, 3],
        [5, 5, 5, 5, 6], [6, 6, 4, 5, 5], [4, 6, 6, 6, 4],
        [5, 6, 6, 5, 4], [6, 5, 7, 4, 4], [7, 5, 6, 5, 3],
        [6, 6, 5, 5, 4], [5, 5, 6, 6, 4], [4, 7, 6, 5, 4],
        [6, 5, 5, 6, 4], [5, 6, 5, 5, 5], [7, 5, 5, 5, 4],
        [5, 5, 5, 7, 4], [6, 4, 6, 6, 4], [4, 6, 5, 6, 5],
    ]
    for pat in patterns:
        configs.append(generate_row_pattern(pat))
        # Variations with different shifts
        configs.append(generate_row_pattern(pat, base_shift=0.07, spacing=0.16))
        configs.append(generate_row_pattern(pat, base_shift=0.09, spacing=0.18))
    
    # 3. Corner-biased starts
    corners = [[0.07, 0.07], [0.93, 0.07], [0.07, 0.93], [0.93, 0.93]]
    for seed in range(20):
        rng_local = np.random.RandomState(seed + 500)
        c = rng_local.uniform(0.1, 0.9, (N, 2))
        for i in range(8):
            c[i] = np.array(corners[i % 4]) + rng_local.uniform(-0.05, 0.05, 2)
        configs.append(np.clip(c, 0.02, 0.98))
    
    # 4. Random starts
    for _ in range(30):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
    
    # Evaluate all configs with LP and SLSQP polishing
    for c0 in configs:
        r0, s0 = solve_lp_radii(c0)
        if s0 > best_sum * 0.9:  # Only polish promising starts
            x0 = np.zeros(3 * N)
            x0[0::3] = c0[:, 0]
            x0[1::3] = c0[:, 1]
            x0[2::3] = np.maximum(r0 * 0.92, 1e-5)
            
            x_opt = polish(x0, maxiter=8000)
            c_opt = np.column_stack((x_opt[0::3], x_opt[1::3]))
            
            r_opt, s_opt = solve_lp_radii(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
    
    # Basin hopping with decaying noise
    if best_centers is not None:
        rng = np.random.RandomState(123)
        current_c = best_centers.copy()
        current_r = best_radii.copy()
        
        for step in range(400):
            scale = 0.012 * np.exp(-step / 80)
            c_pert = current_c + rng.normal(0, scale, (N, 2))
            c_pert = np.clip(c_pert, 0.015, 0.985)
            
            r_pert, s_pert = solve_lp_radii(c_pert)
            if s_pert > best_sum:
                best_sum = s_pert
                best_centers = c_pert.copy()
                best_radii = r_pert.copy()
                
                # Polish improvement
                x0 = np.zeros(3 * N)
                x0[0::3] = best_centers[:, 0]
                x0[1::3] = best_centers[:, 1]
                x0[2::3] = np.maximum(best_radii * 0.93, 1e-5)
                
                x_opt = polish(x0, maxiter=6000)
                c_opt = np.column_stack((x_opt[0::3], x_opt[1::3]))
                r_opt, s_opt = solve_lp_radii(c_opt)
                
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
                
                current_c = best_centers
                current_r = best_radii
            elif s_pert > np.sum(current_r):
                current_c = c_pert
                current_r = r_pert
    
    # Coordinate descent: optimize one circle at a time
    if best_centers is not None:
        rng_local = np.random.RandomState(999)
        for iteration in range(8):
            improved_any = False
            for i in range(N):
                best_local_s = np.sum(best_radii)
                best_local_c = best_centers[i].copy()
                
                # Try multiple random perturbations
                for trial in range(40):
                    c_try = best_centers.copy()
                    c_try[i] = best_centers[i] + rng_local.normal(0, 0.004, 2)
                    c_try[i] = np.clip(c_try[i], 0.01, 0.99)
                    
                    r_try, s_try = solve_lp_radii(c_try)
                    if s_try > best_local_s + 1e-8:
                        best_local_s = s_try
                        best_local_c = c_try[i].copy()
                
                if np.linalg.norm(best_local_c - best_centers[i]) > 1e-8:
                    best_centers[i] = best_local_c
                    improved_any = True
            
            r_new, s_new = solve_lp_radii(best_centers)
            best_radii = r_new
            if s_new > best_sum:
                best_sum = s_new
            
            if not improved_any and iteration > 2:
                break
    
    # Targeted small perturbations on promising circles
    if best_centers is not None:
        rng_small = np.random.RandomState(777)
        for _ in range(200):
            idx = rng_small.randint(N)
            c_try = best_centers.copy()
            c_try[idx] = best_centers[idx] + rng_small.normal(0, 0.0015, 2)
            c_try[idx] = np.clip(c_try[idx], 0.01, 0.99)
            
            r_try, s_try = solve_lp_radii(c_try)
            if s_try > best_sum + 1e-9:
                best_sum = s_try
                best_centers = c_try.copy()
                best_radii = r_try.copy()
    
    # Fallback
    if best_centers is None:
        best_centers = generate_hex_lattice(0)
        best_radii, best_sum = solve_lp_radii(best_centers)
    
    # Strict post-processing
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(radii[i], 0.0)
    
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc / 2.0
                    radii[j] -= exc / 2.0
                    changed = True
        if not changed:
            break
    
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
