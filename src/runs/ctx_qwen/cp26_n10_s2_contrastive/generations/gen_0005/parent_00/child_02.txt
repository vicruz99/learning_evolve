import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute LP constraint matrix structure
A_ub_lp = np.zeros((NUM_PAIRS, N))
A_ub_lp[np.arange(NUM_PAIRS), I] = 1.0
A_ub_lp[np.arange(NUM_PAIRS), J] = 1.0


def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints(x):
    """Inequality constraints (must be >= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dists = np.hypot(dx, dy)
    c_overlap = dists - (r[I] + r[J])
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])


def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds_r = []
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds_r.append((0.0, max(1e-9, mx)))
    
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(-np.ones(n), A_ub=A_ub_lp, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success:
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(n), 0.0


def make_valid(centers, radii):
    """Strictly enforce all constraints."""
    c = centers.copy()
    r = radii.copy()
    
    for i in range(N):
        mx = min(c[i, 0], 1.0 - c[i, 0], c[i, 1], 1.0 - c[i, 1])
        r[i] = min(r[i], mx - 1e-9)
        r[i] = max(r[i], 0.0)
    
    for _ in range(200):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
                if d < r[i] + r[j] - 1e-11:
                    exc = r[i] + r[j] - d
                    r[i] -= exc / 2.0
                    r[j] -= exc / 2.0
                    changed = True
        if not changed:
            break
    r = np.maximum(r, 0.0)
    return c, r


def generate_hex_lattice(seed, spacing=None):
    """Generate hexagonal lattice with random perturbation."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    
    if spacing is None:
        spacing = 0.13 + rng.uniform(0, 0.10)
    
    idx = 0
    row = 0
    y = spacing / 2 + rng.uniform(0, spacing / 4)
    while idx < N and y < 1.0 - spacing / 2:
        x_start = spacing / 2 + (row % 2) * spacing / 2 + rng.uniform(0, spacing / 4)
        col = 0
        while x_start + col * spacing < 1.0 - spacing / 2 and idx < N:
            centers[idx] = [x_start + col * spacing, y]
            idx += 1
            col += 1
        y += spacing * math.sqrt(3) / 2
        row += 1
    
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    
    centers += rng.normal(0, 0.003, centers.shape)
    return np.clip(centers, 0.02, 0.98)


def generate_row_pattern(seed, pattern=None):

    rng = np.random.RandomState(seed)
    
    if pattern is None:
        patterns = [
            [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [7, 6, 5, 4, 4],

            [7, 5, 5, 5, 4], [6, 6, 6, 5, 3], [5, 6, 6, 5, 4],
            [7, 7, 6, 6], [8, 7, 6, 5], [9, 6, 5, 6],

        ]

    idx = 0
    y = 0.04

                idx += 1
            x += dx_val

        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    
    return np.clip(centers, 0.02, 0.98)


def optimize_from_start(c0, maxiter=8000):
    """Run SLSQP optimization from a given start."""
    r0, _ = solve_lp_radii(c0)
    r0 = np.maximum(r0, 1e-4)
    
    x0 = np.zeros(3 * N)
    x0[0::3] = c0[:, 0]
    x0[1::3] = c0[:, 1]
    x0[2::3] = r0 * 0.93
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        
        cx = res.x[0::3]
        cy = res.x[1::3]
        c_opt = np.column_stack((cx, cy))
        r_opt, s_opt = solve_lp_radii(c_opt)
        return c_opt, r_opt, s_opt
    except Exception:
        pass
    
    r_fallback, s_fallback = solve_lp_radii(c0)
    return c0, r_fallback, s_fallback


def force_relaxation(centers, radii, steps=100, rng=None):
    """Apply repulsive forces to relax configuration."""
    if rng is None:
        rng = np.random.RandomState(999)
    c = centers.copy()
    r = radii.copy()
    n = c.shape[0]
    
    for step in range(steps):
        force = np.zeros_like(c)
        alpha = max(0.01, 0.1 * (1.0 - step / steps))
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = math.hypot(dx, dy)
                if d < r[i] + r[j] + 0.02 and d > 1e-12:
                    overlap = max(0, r[i] + r[j] - d)
                    f = overlap * alpha / max(d, 0.001)
                    fx = dx / d * f
                    fy = dy / d * f
                    force[i, 0] += fx
                    force[i, 1] += fy
                    force[j, 0] -= fx
                    force[j, 1] -= fy
        
        # Boundary repulsion
        for i in range(n):
            if c[i, 0] < r[i] + 0.02:
                force[i, 0] += (r[i] + 0.02 - c[i, 0]) * alpha * 5
            if c[i, 0] > 1.0 - r[i] - 0.02:
                force[i, 0] -= (c[i, 0] - (1.0 - r[i] - 0.02)) * alpha * 5
            if c[i, 1] < r[i] + 0.02:
                force[i, 1] += (r[i] + 0.02 - c[i, 1]) * alpha * 5
            if c[i, 1] > 1.0 - r[i] - 0.02:
                force[i, 1] -= (c[i, 1] - (1.0 - r[i] - 0.02)) * alpha * 5
        
        c += force * 0.3
        c = np.clip(c, 0.02, 0.98)
        
        # Add small random noise to escape minima
        c += rng.normal(0, 0.001, c.shape)
        c = np.clip(c, 0.02, 0.98)
    
    return c
