import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution
import itertools

def solve_lp_radii(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii.
    Constraints: r_i + r_j <= dist(i,j) for all pairs, and 0 <= r_i <= min boundary distance.
    """
    n = len(centers)
    if n == 0:
        return np.array([]), 0.0
    
    c_obj = -np.ones(n)
    
    # Build pairwise constraints: r_i + r_j <= dist(i, j)
    n_pairs = n * (n - 1) // 2
    if n_pairs > 0:
        A_ub = np.zeros((n_pairs, n))
        b_ub = np.zeros(n_pairs)
        
        # Efficient construction using meshgrid-like approach
        indices = np.triu_indices(n, k=1)
        i_idx, j_idx = indices
        
        # Compute distances
        dx = centers[i_idx, 0] - centers[j_idx, 0]
        dy = centers[i_idx, 1] - centers[j_idx, 1]
        dists = np.sqrt(dx * dx + dy * dy)
        
        for k, (i, j) in enumerate(zip(i_idx, j_idx)):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[k]
    else:
        A_ub = np.zeros((0, n))
        b_ub = np.array([])
    
    # Variable bounds: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, max_r)))
    
    # Try methods in order of preference
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method,
                         options={'presolve': True})
            if res.success:
                radii = np.maximum(res.x, 0.0)
                return radii, np.sum(radii)
        except Exception:
            continue
    
    # Fallback: compute safe radii
    safe_radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        safe_radii[i] = min(x, 1.0 - x, y, 1.0 - y) * 0.01
    return safe_radii, np.sum(safe_radii)


def objective_for_centers(x_flat):
    """Objective function for optimizing centers: returns negative sum of radii."""
    n = 26
    centers = x_flat.reshape((n, 2))
    # Clip to valid range
    centers = np.clip(centers, 0.0, 1.0)
    _, sum_r = solve_lp_radii(centers)
    return -sum_r


def generate_hexagonal_centers(row_counts, spacing, margin, height=None):
    """Generate hexagonal lattice points with given row counts and spacing."""
    n = 26
    points = []
    
    if height is None:
        n_rows = len(row_counts)
        height = 1.0 - 2 * margin
    
    dy = height / max(n_rows - 1, 1) if n_rows > 1 else 0.5
    
    y = margin
    for r_idx, cnt in enumerate(row_counts):
        # Alternate horizontal offset for hexagonal pattern
        shift = spacing / 2.0 if r_idx % 2 == 1 else 0.0
        x = margin + shift
        
        for _ in range(cnt):
            if len(points) < n and x + margin <= 1.0:
                points.append([x, y])
            x += spacing
        
        y += dy
    
    # Fill remaining points if needed
    while len(points) < n:
        points.append([0.5, 0.5])
    
    return np.array(points[:n])
