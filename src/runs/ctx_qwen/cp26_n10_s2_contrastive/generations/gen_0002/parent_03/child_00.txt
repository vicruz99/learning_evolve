import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
# Precompute indices for pairwise constraints to enable vectorization
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
N_PAIRS = len(PAIR_I)

def objective(x):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Allocate constraint array
    c = np.empty(4 * N + N_PAIRS)
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    # Pairwise non-overlap constraints: dist(i,j) >= r_i + r_j
    dx = cx[PAIR_I] - cx[PAIR_J]
    dy = cy[PAIR_I] - cy[PAIR_J]
    dists = np.hypot(dx, dy)
    c[4*N:] = dists - r[PAIR_I] - r[PAIR_J]
    
    return c

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    A_ub = np.zeros((N_PAIRS, n))
    A_ub[np.arange(N_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(N_PAIRS), PAIR_J] = 1.0
    
    dx = centers[PAIR_I, 0] - centers[PAIR_J, 0]
    dy = centers[PAIR_I, 1] - centers[PAIR_J, 1]
    b_ub = np.hypot(dx, dy)
    
    # Boundary constraints handled via variable bounds: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        mx, my = centers[i]
        ub = min(mx, 1.0 - mx, my, 1.0 - my)
        bounds.append((0.0, max(0.0, ub)))
        
    # Solve LP using HiGHS (fast/reliable) or fallback
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return res.x
        except Exception:
            continue
            
    # Fallback safe radii
    return np.full(n, 0.01)

def hex_init(seed, margin=0.04, spacing=0.18):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.default_rng(seed)
    pts = []
    row = 0
    y = margin
    while len(pts) < N:
        x_off = margin + (row % 2) * spacing / 2.0
        x = x_off
        while x < 1.0 - margin and len(pts) < N:
            pts.append([x, y])
            x += spacing
        y += spacing * math.sqrt(3) / 2.0
        row += 1
    pts = np.array(pts[:N])
    pts += rng.normal(0, 0.012, pts.shape)
    return np.clip(pts, 0.05, 0.95)
