import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_radii_lp(centers):
    """
    Given fixed centers, solve LP to find radii that maximize sum(radii)
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Compute max possible radius for each circle based on boundaries
    max_r = np.min(np.array([
        centers[:, 0], 
        1.0 - centers[:, 0], 
        centers[:, 1], 
        1.0 - centers[:, 1]
    ]), axis=0)
    max_r = np.maximum(max_r, 0.0)
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds = [(0.0, max_r[i]) for i in range(n)]
    
    # Try HiGHS (fastest), fallback to interior-point
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return res.x, -res.fun
        except Exception:
            continue
            
    # Fallback safe radii
    return np.full(n, 0.01), 0.0

def objective(centers_flat):
    """Objective function for center optimization: minimize negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    # Clip to safe interior to prevent boundary degeneracies during search
    centers = np.clip(centers, 0.01, 0.99)
    _, s = solve_radii_lp(centers)
    return -s

        row = 0

                centers[idx] = [x_start + col * spacing, y]
                idx += 1
                col += 1
            row += 1
        while idx < n:
            centers[idx] = rng.uniform(0.2, 0.8, 2)
            idx += 1
        centers += rng.normal(0, 0.008, centers.shape)

                x += step
