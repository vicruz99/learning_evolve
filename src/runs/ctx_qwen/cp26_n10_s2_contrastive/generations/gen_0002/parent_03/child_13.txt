import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def solve_radii_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii 
    subject to r_i >= 0, boundary constraints, and non-overlap.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, mx)))
        
    # Try HiGHS first (fast), fallback to interior-point
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success:
                radii = np.maximum(res.x, 0.0)
                return radii, np.sum(radii)
        except Exception:
            continue
            
    # Fallback if LP fails
    return np.full(n, 0.01), 0.26

def obj_func(x):
    """Objective: minimize negative sum of optimal radii for given centers."""
    centers = np.clip(x.reshape(N, 2), 0.0, 1.0)
    _, val = solve_radii_lp(centers)
    return -val

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0.0
    best_c = None
    best_r = None
    
    rng = np.random.default_rng(123)
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Perturbed Hexagonal Lattices
    for _ in range(15):
        c = np.zeros((N, 2))
        idx = 0
        y = 0.04 + rng.uniform(0, 0.02)
        row = 0
        dy = 0.15 + rng.uniform(-0.02, 0.02)
        while idx < N and y <= 0.96:
            x = 0.04 + (0.09 if row % 2 == 1 else 0.0) + rng.uniform(-0.01, 0.01)
            dx = 0.17 + rng.uniform(-0.02, 0.02)
            while idx < N and x <= 0.96:
                c[idx] = [x, y]
                idx += 1
                x += dx
            y += dy
            row += 1
        while idx < N:
            c[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
        inits.append(c)
        
    # 2. Uniform Random Configurations
    for _ in range(15):
        inits.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    # 3. Regular Grid Configurations
    for _ in range(10):
        c = np.zeros((N, 2))
        idx = 0
        step = 0.16 + rng.uniform(0, 0.05)
        y = step
        while idx < N and y <= 0.95:
            x = step
            while idx < N and x <= 0.95:
                c[idx] = [x, y]
                idx += 1
                x += step
            y += step
        while idx < N:
            c[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
        inits.append(c)

    # Multi-start local optimization
    for c0 in inits:
        c0_flat = c0.flatten()
        try:
            res = minimize(obj_func, c0_flat, method='Powell',
                           options={'maxiter': 800, 'ftol': 1e-13, 'xtol': 1e-13, 'disp': False})
            c_opt = np.clip(res.x.reshape(N, 2), 0.0, 1.0)
            r_opt, s_opt = solve_radii_lp(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            pass
            
        # Decaying perturbation search to fine-tune and escape shallow minima
        if best_c is not None:
            for trial in range(40):
                step = 0.008 * (1.0 - trial / 40.0)
                pert = best_c + rng.normal(0, step, best_c.shape)
                pert = np.clip(pert, 0.01, 0.99)
                r_p, s_p = solve_radii_lp(pert)
                if s_p > best_sum:
                    best_sum = s_p
                    best_c = pert.copy()
                    best_r = r_p.copy()

    # Fallback safety net
    if best_c is None:
        best_c = rng.uniform(0.1, 0.9, (N, 2))
        best_r, best_sum = solve_radii_lp(best_c)
        
    # Final validation and strict numerical repair
    c_final = best_c.copy()
    r_final = best_r.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        x, y = c_final[i]
        r_final[i] = min(r_final[i], x, 1.0 - x, y, 1.0 - y)
        r_final[i] = max(0.0, r_final[i])
        
    # Iteratively resolve microscopic overlaps from numerical drift
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(c_final[i, 0] - c_final[j, 0], c_final[i, 1] - c_final[j, 1])
                if d < r_final[i] + r_final[j] - 1e-12:
                    exc = r_final[i] + r_final[j] - d
                    r_final[i] -= exc * 0.5
                    r_final[j] -= exc * 0.5
                    r_final[i] = max(0.0, r_final[i])
                    r_final[j] = max(0.0, r_final[j])
                    changed = True
        if not changed:
            break
            
    return c_final, r_final, float(np.sum(r_final))
