import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution
import warnings

# Suppress solver warnings for cleaner execution
warnings.filterwarnings('ignore')

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """
    Given fixed centers, solve LP to find radii that maximize sum(radii)
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    dx = centers[TRIU_I, 0] - centers[TRIU_J, 0]
    dy = centers[TRIU_I, 1] - centers[TRIU_J, 1]
    b_ub = np.hypot(dx, dy)
    
    # Construct sparse-ish A_ub efficiently
    n_pairs = len(TRIU_I)
    A_ub = np.zeros((n_pairs, n))
    A_ub[np.arange(n_pairs), TRIU_I] = 1.0
    A_ub[np.arange(n_pairs), TRIU_J] = 1.0
    
    # Variable bounds: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for k in range(n):
        mx = min(centers[k, 0], 1.0 - centers[k, 0], centers[k, 1], 1.0 - centers[k, 1])
        bounds.append((0.0, max(0.0, mx)))
        
    # Solve LP
    for method in ['highs', 'interior-point', 'revised simplex']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return res.x
        except Exception:
            continue
            
    # Fallback: return tiny radii if LP fails (e.g., duplicate centers)
    return np.full(n, 1e-6)

def de_objective(c_flat):
    """Objective for Differential Evolution: maximize sum of radii via LP."""
    centers = c_flat.reshape(N, 2)
    radii = solve_radii_lp(centers)
    return -np.sum(radii)

def objective_joint(x):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(x[2::3])

def constraints_joint(x):
    """Inequality constraints g(x) >= 0 for SLSQP."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    n_cons = 4 * N + N * (N - 1) // 2
    c = np.empty(n_cons)
    
    # Boundary constraints
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    # Pairwise distance constraints
    dx = cx[TRIU_I] - cx[TRIU_J]
    dy = cy[TRIU_I] - cy[TRIU_J]
    dists = np.hypot(dx, dy)
    r_sum = r[TRIU_I] + r[TRIU_J]
    c[4*N:] = dists - r_sum
    
    return c

def generate_hex_init(density=1.0, seed=0):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    s = 0.14 * density
    pts = []
    y = 0.05
    row = 0
    while len(pts) < N and y < 0.95:
        shift = s / 2.0 if row % 2 == 1 else 0.0
        x = 0.05 + shift
        while x < 0.95 and len(pts) < N:
            pts.append([x + rng.uniform(-0.01, 0.01), y + rng.uniform(-0.01, 0.01)])
            x += s
        y += s * np.sqrt(3) / 2.0
        row += 1
        
    while len(pts) < N:
        pts.append([0.5 + rng.uniform(-0.1, 0.1), 0.5 + rng.uniform(-0.1, 0.1)])
        
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    """
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': constraints_joint}
    
    # 1. Global search for centers using Differential Evolution
    print("Running global center search...")
    bounds_de = [(0.05, 0.95)] * (2 * N)
    try:
        res_de = differential_evolution(
            de_objective,
            bounds_de,
            seed=42,
            maxiter=25,
            popsize=10,
            tol=1e-7,
            polish=False,
            updating='deferred'
        )
        de_c = res_de.x.reshape(N, 2)
        de_r = solve_radii_lp(de_c)
        de_s = np.sum(de_r)
        
        if de_s > best_sum:
            best_sum = de_s
            best_centers = de_c.copy()
            best_radii = de_r.copy()
    except Exception:
        pass
        
    # 2. Prepare diverse starting points for joint SLSQP optimization
    starts = []
    
    # From DE result
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii
        starts.append(x0)
        
        # Perturbed versions
        for _ in range(4):
            p = np.random.randn(3 * N) * 0.004
            x0_p = np.clip(x0 + p, [b[0] for b in bounds_opt], [b[1] for b in bounds_opt])
            starts.append(x0_p)
            
    # Hexagonal patterns with varying densities
    for d in [0.85, 0.95, 1.05, 1.15]:
        c_hex = generate_hex_init(density=d, seed=int(d*100))
        r_hex = solve_radii_lp(c_hex)
        x0_hex = np.zeros(3 * N)
        x0_hex[0::3] = c_hex[:, 0]
        x0_hex[1::3] = c_hex[:, 1]
        x0_hex[2::3] = r_hex
        starts.append(x0_hex)
        
    # Random initializations
    for seed in range(6):
        rng = np.random.RandomState(seed + 100)
        c_rand = rng.uniform(0.1, 0.9, (N, 2))
        r_rand = solve_radii_lp(c_rand)
        x0_rand = np.zeros(3 * N)
        x0_rand[0::3] = c_rand[:, 0]
        x0_rand[1::3] = c_rand[:, 1]
        x0_rand[2::3] = r_rand
        starts.append(x0_rand)
        
    # Run SLSQP from all starts
    print("Running joint optimization...")
    for i, x0 in enumerate(starts):
        try:
            res = minimize(
                objective_joint, x0,
                method='SLSQP',
                bounds=bounds_opt,
                constraints=cons_dict,
                options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False}
            )
            if res.success:
                curr_c = np.column_stack((res.x[0::3], res.x[1::3]))
                # LP polish guarantees optimal radii for these centers
                curr_r = solve_radii_lp(curr_c)
                curr_s = np.sum(curr_r)
                
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_centers = curr_c.copy()
                    best_radii = curr_r.copy()
        except Exception:
            pass
            
    # 3. Intensive local refinement around the best solution
    if best_centers is not None:
        print("Refining best solution...")
        for trial in range(12):
            rng = np.random.RandomState(trial * 31 + 7)
            c_p = best_centers + rng.randn(N, 2) * 0.002
            c_p = np.clip(c_p, 0.01, 0.99)
            r_p = solve_radii_lp(c_p) * 0.995
            
            x0_p = np.zeros(3 * N)
            x0_p[0::3] = c_p[:, 0]
            x0_p[1::3] = c_p[:, 1]
            x0_p[2::3] = r_p
            
            try:
                res_p = minimize(
                    objective_joint, x0_p, method='SLSQP', bounds=bounds_opt,
                    constraints=cons_dict,
                    options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False}
                )
                if res_p.success:
                    cp = np.column_stack((res_p.x[0::3], res_p.x[1::3]))
                    rp = solve_radii_lp(cp)
                    sp = np.sum(rp)
                    if sp > best_sum:
                        best_sum = sp
                        best_centers = cp.copy()
                        best_radii = rp.copy()
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = np.random.rand(N, 2) * 0.6 + 0.2
        best_radii = np.full(N, 0.02)
        best_sum = np.sum(best_radii)
        
    # 4. Final strict validation and numerical safety adjustment
    c_final = best_centers.copy()
    r_final = best_radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        x, y = c_final[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        r_final[i] = min(r_final[i], max_r - 1e-9)
        r_final[i] = max(0.0, r_final[i])
        
    # Iteratively resolve microscopic overlaps from numerical drift
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(c_final[i,0] - c_final[j,0], c_final[i,1] - c_final[j,1])
                if d < r_final[i] + r_final[j] - 1e-12:
                    exc = r_final[i] + r_final[j] - d
                    r_final[i] -= exc * 0.5
                    r_final[j] -= exc * 0.5
                    r_final[i] = max(0.0, r_final[i])
                    r_final[j] = max(0.0, r_final[j])
                    changed = True
        if not changed:
            break
            
    return c_final, r_final, float(np.sum(r_final))
