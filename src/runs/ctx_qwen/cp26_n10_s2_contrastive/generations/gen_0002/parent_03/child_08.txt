import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[2*N:])

def constraints(params):
    """Returns array of inequality constraints g(params) >= 0."""
    cx = params[:N]
    cy = params[N:2*N]
    r = params[2*N:]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b = np.empty(4*N)
    b[:N] = cx - r
    b[N:2*N] = 1.0 - cx - r
    b[2*N:3*N] = cy - r
    b[3*N:4*N] = 1.0 - cy - r
    
    # Pairwise non-overlap constraints: dist(i,j) - r[i] - r[j] >= 0
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dists = np.sqrt(dx*dx + dy*dy)
    r_sum = r[:, None] + r[None, :]
    
    overlaps = dists[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]
    return np.concatenate([b, overlaps])

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    i_idx, j_idx = np.triu_indices(n, k=1)
    n_pairs = len(i_idx)
    A_ub = np.zeros((n_pairs, n))
    A_ub[np.arange(n_pairs), i_idx] = 1.0
    A_ub[np.arange(n_pairs), j_idx] = 1.0
    
    # Compute pairwise distances efficiently
    dists = np.sqrt(((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2).sum(axis=2))
    b_ub = dists[I_IDX, J_IDX]
    
    # Variable bounds: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, mx)))
        
    # Solve LP
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
        
    return np.full(n, 0.01)

def generate_init(spacing, rotation, seed):
    """Generate a hexagonal lattice initialization with rotation and jitter."""
    rng = np.random.RandomState(seed)
    pts = []
    y = spacing * 0.5
    row = 0
    while len(pts) < N + 10:
        x_start = spacing * (0.5 if row % 2 else 0.0)
        col = 0
        while x_start + col * spacing <= 1.0 + spacing:
            x = x_start + col * spacing
            pts.append([x, y])
            col += 1
        y += spacing * np.sqrt(3) / 2.0
        row += 1
        
    pts = np.array(pts)
    if rotation != 0.0:
        cos_r, sin_r = np.cos(rotation), np.sin(rotation)
        center = np.mean(pts, axis=0)
        pts = (pts - center) @ [[cos_r, -sin_r], [sin_r, cos_r]] + center
        
    # Filter points inside [0,1]x[0,1]
    mask = (pts[:, 0] >= 0) & (pts[:, 0] <= 1) & (pts[:, 1] >= 0) & (pts[:, 1] <= 1)
    valid = pts[mask]
    
    if len(valid) < N:
        needed = N - len(valid)
        valid = np.vstack([valid, rng.uniform(0.1, 0.9, (needed, 2))])
        
    valid += rng.normal(0, 0.008, valid.shape)
    valid = np.clip(valid, 0.02, 0.98)
    return valid[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds_opt = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_c = None
    best_r = None

                    break

                try:
                    res = minimize(
                        objective, x0,
                        method='SLSQP',
                        bounds=bounds_opt,
                        constraints=cons_dict,

                    )

                    pass

                    r_final[j] = max(0.0, r_final[j])

        if not changed:
            break
            
    return c_final, r_final, float(np.sum(r_final))
