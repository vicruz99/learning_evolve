import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist
import math

N = 26

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """
    Inequality constraints g(x) >= 0:
    Boundary: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    Overlap: dist(i,j) - r_i - r_j >= 0
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints (4N)
    c_bound = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Overlap constraints (N*(N-1)/2)
    pts = np.column_stack((cx, cy))
    dists = pdist(pts)
    r_sums = r[:, None] + r[None, :]
    mask = np.triu_indices(N, k=1)
    c_overlap = dists - r_sums[mask]
    
    return np.concatenate([c_bound, c_overlap])

def get_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    dists = pdist(centers)
    
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = dists.copy()
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
            
    bounds = []
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], 
                 centers[i, 1], 1.0 - centers[i, 1])
        bounds.append((0.0, max(0.0, mx)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 0.01), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Strategy 1: Diverse lattice initializations
    configs = []
    # Hexagonal patterns summing to 26
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [7,6,5,4,4], [8,6,5,3,4], [5,5,5,5,6]]
    for pat in patterns:
        pts = []
        y = 0.06
        dy = 0.155
        for r_idx, cnt in enumerate(pat):
            shift = 0.0 if r_idx % 2 == 0 else 0.085
            x = 0.06 + shift
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 0.17
            y += dy
        while len(pts) < N:
            pts.append([0.5, 0.5])
        configs.append(np.array(pts[:N]))
        
    # Random initializations
    rng = np.random.default_rng(42)
    for _ in range(10):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))

    trial = 0
    max_trials = 40
    
    for base in configs:
        if trial >= max_trials:
            break
            
        for scale in [0.0, 0.015, 0.04]:
            if trial >= max_trials:
                break
                
            rng_trial = np.random.default_rng(trial * 31 + 7)
            c_init = base.copy()
            c_init += rng_trial.normal(0, scale, c_init.shape)
            c_init = np.clip(c_init, 0.03, 0.97)
            
            # Initial radii: conservative to ensure feasibility
            r_init = np.full(N, 0.035)
            x0 = np.zeros(3 * N)
            for i in range(N):
                x0[3*i] = c_init[i, 0]
                x0[3*i+1] = c_init[i, 1]
                x0[3*i+2] = r_init[i]
                
            try:
                res = minimize(
                    objective, x0,
                    method='SLSQP',
                    bounds=bounds_opt,
                    constraints=cons_opt,
                    options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False}
                )
                
                curr_c = np.column_stack((res.x[0::3], res.x[1::3]))
                # LP polishing for exact radii given centers
                curr_r, curr_s = get_lp_radii(curr_c)
                
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_centers = curr_c.copy()
                    best_radii = curr_r.copy()
            except Exception:
                pass
            trial += 1

    # Strategy 2: Local perturbation refinement around best found
    if best_centers is not None:
        rng_ref = np.random.default_rng(123)
        for step in range(30):
            # Adaptive perturbation magnitude
            noise = 0.005 * (1.0 - step / 40.0) 
            c_pert = best_centers + rng_ref.normal(0, noise, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            
            r_p, s_p = get_lp_radii(c_pert)

                        objective, x0_p,
                        method='SLSQP',

                    pass
    
    # Fallback safety net
    if best_centers is None:
        best_centers = np.random.rand(N, 2) * 0.6 + 0.2
        best_radii = np.full(N, 0.02)
        best_sum = np.sum(best_radii)

    # Final strict validation and numerical correction
    c_final = best_centers.copy()
    r_final = best_radii.copy()
    
    for i in range(N):
        x, y = c_final[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        r_final[i] = min(r_final[i], max_r - 1e-9)
        r_final[i] = max(0.0, r_final[i])
        
    for _ in range(20):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(c_final[i,0]-c_final[j,0], c_final[i,1]-c_final[j,1])
                if d < r_final[i] + r_final[j] - 1e-11:
                    exc = r_final[i] + r_final[j] - d
                    r_final[i] -= exc * 0.5
                    r_final[j] -= exc * 0.5
                    r_final[i] = max(0.0, r_final[i])
                    r_final[j] = max(0.0, r_final[j])
                    changed = True
        if not changed:
            break
            
    return c_final, r_final, float(np.sum(r_final))
