import numpy as np
from scipy.optimize import minimize
import math

N = 26
TRIU_IDX = np.triu_indices(N, k=1)

def objective(x):
    """Negative sum of radii to be minimized."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints: boundary and non-overlap (squared)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    n_cons = 4 * N + N * (N - 1) // 2
    c = np.empty(n_cons)
    
    # Boundary constraints
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    # Overlap constraints (squared distances for smooth gradients)
    i_idx, j_idx = TRIU_IDX
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    dist_sq = dx**2 + dy**2
    r_sum = r[i_idx] + r[j_idx]
    overlap = dist_sq - r_sum**2
    
    c[4*N:] = overlap
    return c

def center_dist(p):
    """Distance metric from center of square for sorting lattice points."""
    return (p[0] - 0.5)**2 + (p[1] - 0.5)**2

def enforce_validity(centers, radii):
    """Project an infeasible configuration to the feasible region."""
    margin = 1e-9
    n = centers.shape[0]
    
    # Enforce boundary constraints strictly
    for i in range(n):
        max_r = min(centers[i,0], 1.0 - centers[i,0], 
                    centers[i,1], 1.0 - centers[i,1]) - margin
        if radii[i] > max_r:
            radii[i] = max(0.0, max_r)
            
    # Iteratively resolve overlaps
    for _ in range(150):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = math.hypot(centers[i,0] - centers[j,0], 
                               centers[i,1] - centers[j,1])
                if d < radii[i] + radii[j] - 1e-12:
                    excess = radii[i] + radii[j] - d + 1e-9
                    radii[i] -= excess / 2.0
                    radii[j] -= excess / 2.0
                    changed = True
        if not changed:
            break
            
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

                    points.append((xr, yr))
                    
        points.sort(key=center_dist)
        points = points[:N]
        for k, (px, py) in enumerate(points):

        cy = np.clip(cy, 0.02, 0.98)
