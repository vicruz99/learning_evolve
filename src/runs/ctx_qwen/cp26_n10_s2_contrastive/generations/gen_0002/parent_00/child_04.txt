PAIRS = np.array([(i, j) for i in range(N) for j in range(i+1, N)])

        ub = min(x, 1.0-x, y, 1.0-y)
        bounds.append((0.0, max(0.0, ub)))

        res = linprog(c_obj, A_ub=A_ub_lp, b_ub=b_ub, bounds=bounds, method='highs')

        pass
    return np.zeros(n), 0.0
