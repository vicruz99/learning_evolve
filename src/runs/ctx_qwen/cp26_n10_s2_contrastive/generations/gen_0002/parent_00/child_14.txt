import numpy as np
from scipy.optimize import linprog, minimize
import math

N = 26
I_IND, J_IND = np.triu_indices(N, k=1)
N_PAIRS = len(I_IND)

# Precompute the structure of pairwise constraint matrix for LP
A_UUB = np.zeros((N_PAIRS, N))
for k in range(N_PAIRS):
    A_UUB[k, I_IND[k]] = 1.0
    A_UUB[k, J_IND[k]] = 1.0

def get_radii_lp(centers):
    """Solve LP to find radii maximizing sum(radii) for fixed centers."""
    # Compute pairwise distances
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    b_ub = dists[I_IND, J_IND]
    
    # Boundary bounds for each radius
    bounds_r = []
    for i in range(N):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(-np.ones(N), A_ub=A_UUB, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(N, 1e-5), 0.0

def obj_centers(centers_flat):
    """Objective for optimizing centers: minimize negative sum of optimal radii."""
    centers = centers_flat.reshape((N, 2))
    centers = np.clip(centers, 0.001, 0.999)
    _, s = get_radii_lp(centers)
    return -s

def obj_joint(x):
    """Objective for joint optimization."""
    return -np.sum(x[2*N:])

def constr_joint(x):
    """Constraints for joint optimization."""
    c = x[:2*N].reshape(N, 2)
    r = x[2*N:]
    
    vals = np.empty(4*N + N_PAIRS)
    vals[:N] = c[:, 0] - r
    vals[N:2*N] = 1.0 - c[:, 0] - r
    vals[2*N:3*N] = c[:, 1] - r
    vals[3*N:4*N] = 1.0 - c[:, 1] - r
    
    dx = c[:, 0][:, None] - c[:, 0][None, :]
    dy = c[:, 1][:, None] - c[:, 1][None, :]
    dists = np.sqrt(dx*dx + dy*dy)
    
    vals[4*N:] = dists[I_IND, J_IND] - (r[I_IND] + r[J_IND])
    return vals

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_centers = None
    best_radii = None
    
    bounds_c = [(0.001, 0.999)] * (2 * N)

    inits = []

                c[idx] = [x + rng.uniform(-0.025, 0.025), y + rng.uniform(-0.025, 0.025)]
                idx += 1
                x += 0.19
            y += 0.165
            row += 1
        inits.append(c)
        
    # Grid variations
    for seed in range(6):
        rng = np.random.RandomState(seed)
        c = np.zeros((N, 2))
        idx = 0
        step = 0.195
        y = step
        while y < 1.0 and idx < N:
            x = step
            while x < 1.0 and idx < N:
                c[idx] = [x + rng.uniform(-0.02, 0.02), y + rng.uniform(-0.02, 0.02)]
                idx += 1
                x += step
            y += step
        inits.append(c)
        
    # Random inits
    for seed in range(4):
        rng = np.random.RandomState(seed)
        c = rng.uniform(0.1, 0.9, (N, 2))
        inits.append(c)

    # Phase 1: Optimize centers using LP-based objective
    for init_c in inits:
        x0 = init_c.flatten()
        try:
            res = minimize(obj_centers, x0, method='L-BFGS-B', bounds=bounds_c,
                           options={'maxiter': 400, 'ftol': 1e-12, 'disp': False})
            c_opt = np.clip(res.x.reshape((N, 2)), 0.001, 0.999)
            r_opt, s_opt = get_radii_lp(c_opt)

                best_sum = s_opt
                best_centers = c_opt.copy()

                    pass

                    best_radii[i] -= excess / 2.0
                    best_radii[j] -= excess / 2.0
                    changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
