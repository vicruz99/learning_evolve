import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
PAIR_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_IDX[0])

def objective(x):
    """Negative sum of radii to minimize."""
    return -np.sum(x[2*N:])

def constraints(x):
    """Inequality constraints: g(x) >= 0."""
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    # Boundary constraints
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    # Pairwise non-overlap constraints
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dists = np.hypot(dx, dy)
    r_sum = r[:, None] + r[None, :]
    c[4*N:] = dists[PAIR_IDX] - r_sum[PAIR_IDX]
    return c

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to find radii maximizing sum(radii)."""
    try:
        c_obj = -np.ones(N)
        A_ub = np.zeros((NUM_PAIRS, N))
        b_ub = np.zeros(NUM_PAIRS)
        
        i_idx, j_idx = PAIR_IDX
        A_ub[np.arange(NUM_PAIRS), i_idx] = 1.0
        A_ub[np.arange(NUM_PAIRS), j_idx] = 1.0
        
        dx = centers[:, 0][:, None] - centers[:, 0][None, :]
        dy = centers[:, 1][:, None] - centers[:, 1][None, :]
        dists = np.hypot(dx, dy)
        b_ub = dists[PAIR_IDX]
        
        bounds_r = []
        for i in range(N):
            x, y = centers[i]
            mx = min(x, 1.0 - x, y, 1.0 - y)
            bounds_r.append((0.0, max(0.0, mx)))
            
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return np.full(N, 0.02)

def make_valid(c, r):
    """Post-process to strictly satisfy boundary and overlap constraints."""
    # Enforce boundaries
    for i in range(N):
        mx = min(c[i, 0], 1.0 - c[i, 0], c[i, 1], 1.0 - c[i, 1])
        if r[i] > mx:
            r[i] = max(0.0, mx - 1e-9)
            
    # Iteratively resolve overlaps
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
                if d < r[i] + r[j] - 1e-10:
                    excess = r[i] + r[j] - d + 1e-10
                    r[i] -= excess * 0.5
                    r[j] -= excess * 0.5
                    changed = True
        if not changed:
            break
    return c, r

def hex_grid_init(seed, density=1.0):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    dx = 0.18 * density
    dy = dx * math.sqrt(3) / 2.0
    
    row = 0
    y = 0.06
    while len(pts) < N and y < 0.95:
        x = 0.06 + (row % 2) * dx / 2.0
        while x < 0.95 and len(pts) < N:
            pts.append([x + rng.uniform(-0.02, 0.02), y + rng.uniform(-0.02, 0.02)])
            x += dx
        y += dy
        row += 1
        
    pts = np.array(pts[:N])
    return np.clip(pts, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Phase 1: Diverse LP-initialized starts
    starts = []
    for seed in range(60):
        density = 0.92 + (seed % 10) * 0.008
        c = hex_grid_init(seed, density=density)
        r = solve_radii_lp(c) * 0.96  # Scale down slightly for strict feasibility
        starts.append((c, r))
        
    for c_init, r_init in starts:
        x0 = np.concatenate([c_init.ravel(), r_init])
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons_dict,
                           options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_c = res.x[:2*N].reshape(N, 2)
                    best_r = res.x[2*N:].copy()
                    
                    # Phase 2: Local refinement from high-quality state
                    for _ in range(2):
                        cp = best_c + np.random.randn(N, 2) * 0.0025
                        cp = np.clip(cp, 0.01, 0.99)
                        rp = solve_radii_lp(cp) * 0.97
                        xp = np.concatenate([cp.ravel(), rp])
                        
                        res2 = minimize(objective, xp, method='SLSQP', bounds=bounds,
                                        constraints=cons_dict,
                                        options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                        if res2.success and -res2.fun > best_sum:
                            best_sum = -res2.fun
                            best_c = res2.x[:2*N].reshape(N, 2)
                            best_r = res2.x[2*N:].copy()
        except Exception:
            continue

    # Fallback & Strict Validation
    if best_c is None:
        best_c = hex_grid_init(42)
        best_r = solve_radii_lp(best_c)
        best_sum = np.sum(best_r)
    else:
        best_c, best_r = make_valid(best_c, best_r)
        best_sum = np.sum(best_r)
        
    return best_c, best_r, float(best_sum)
