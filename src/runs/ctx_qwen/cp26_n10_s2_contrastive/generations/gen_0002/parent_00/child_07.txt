import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii under constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(NUM_PAIRS), J_IDX] = 1.0
    
    dx = centers[:, 0][I_IDX] - centers[:, 0][J_IDX]
    dy = centers[:, 1][I_IDX] - centers[:, 1][J_IDX]
    dists = np.hypot(dx, dy)
    b_ub = dists
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
        return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0

def objective(x):
    return -np.sum(x[2*N:])

def constraints(x):
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    c = []
    # Boundary constraints
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Squared distance pairwise constraints (smoother gradients)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    c.append(dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def make_init(seed, spacing=0.18, margin=0.04):
    """Generate a perturbed hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = margin
    row = 0
    while idx < N and y < 1.0 - margin:
        x = margin + (row % 2) * spacing / 2.0
        while x < 1.0 - margin and idx < N:
            centers[idx, 0] = x + rng.uniform(-0.02, 0.02)
            centers[idx, 1] = y + rng.uniform(-0.02, 0.02)
            idx += 1
            x += spacing
        y += spacing * np.sqrt(3) / 2.0
        row += 1
    while idx < N:
        centers[idx, 0] = rng.uniform(0.1, 0.9)
        centers[idx, 1] = rng.uniform(0.1, 0.9)
        idx += 1
    centers = np.clip(centers, 0.01, 0.99)
    
    # Solve LP for feasible initial radii, scale down slightly for SLSQP safety
    r, _ = solve_lp_radii(centers)
    r = np.maximum(r * 0.95, 0.001)
    return np.concatenate([centers[:, 0], centers[:, 1], r])

def run_packing():
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: SLSQP optimization from diverse lattice starts
    for seed in range(50):
        spacing = 0.16 + (seed % 12) * 0.004
        margin = 0.03 + (seed % 6) * 0.01
        x0 = make_init(seed, spacing=spacing, margin=margin)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = np.column_stack((res.x[:N], res.x[N:2*N]))
                best_radii = res.x[2*N:].copy()
                
                # Local refinement to escape shallow traps
                for _ in range(3):
                    x_pert = res.x + np.random.randn(len(res.x)) * 0.002
                    x_pert = np.clip(x_pert, [b[0] for b in bounds], [b[1] for b in bounds])
                    res2 = minimize(objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                                    options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
                    if -res2.fun > best_sum:
                        best_sum = -res2.fun
                        best_centers = np.column_stack((res2.x[:N], res2.x[N:2*N]))
                        best_radii = res2.x[2*N:].copy()
        except Exception:
            continue
            
    # Phase 2: LP polishing on best centers found
    if best_centers is not None:
        r_lp, s_lp = solve_lp_radii(best_centers)
        if s_lp > best_sum:
            best_radii = r_lp
            best_sum = s_lp
            
        # Phase 3: Perturbative search combined with LP/SLSQP polishing
        for _ in range(30):
            rng = np.random.RandomState()
            c_pert = best_centers + rng.randn(N, 2) * 0.004
            c_pert = np.clip(c_pert, 0.01, 0.99)
            r_p, s_p = solve_lp_radii(c_pert)
            
            if s_p > best_sum:
                best_sum = s_p
                best_centers = c_pert
                best_radii = r_p
                
                x0_p = np.concatenate([c_pert.ravel(), r_p * 0.99])
                try:
                    res_p = minimize(objective, x0_p, method='SLSQP', bounds=bounds, constraints=cons,
                                     options={'maxiter': 5000, 'ftol': 1e-14})
                    if -res_p.fun > best_sum:
                        best_sum = -res_p.fun
                        best_centers = np.column_stack((res_p.x[:N], res_p.x[N:2*N]))

                        # Final LP polish for exactness
                        r_lp2, s_lp2 = solve_lp_radii(best_centers)
                        if s_lp2 > best_sum:
                            best_sum = s_lp2
                            best_radii = r_lp2
                except Exception:
                    pass
                    
    # Fallback initialization (highly unlikely to trigger)
    if best_centers is None:
        centers = np.random.rand(N, 2) * 0.8 + 0.1
        radii, best_sum = solve_lp_radii(centers)
        best_centers = centers
        best_radii = radii
        
    # Phase 4: Strict numerical validity enforcement
    margin_tol = 1e-10
    for i in range(N):
        x, y = best_centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y) - margin_tol
        if best_radii[i] > max_r:
            best_radii[i] = max_r
            
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                if d < best_radii[i] + best_radii[j] - margin_tol:
                    excess = best_radii[i] + best_radii[j] - d
                    best_radii[i] -= excess/2.0
                    best_radii[j] -= excess/2.0
                    changed = True
        if not changed:
            break
            
    best_sum = np.sum(best_radii)
    return best_centers, best_radii, float(best_sum)
