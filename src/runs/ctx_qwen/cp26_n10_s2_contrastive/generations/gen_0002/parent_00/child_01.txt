import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(x):
    """Negative sum of radii to be minimized."""
    return -np.sum(x[2 * N :])

def constraints(x):
    """Inequality constraints >= 0: boundary and squared pairwise distances."""
    cx = x[:N]
    cy = x[N : 2 * N]
    r = x[2 * N :]
    
    # Boundary constraints
    b = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Squared pairwise distance constraints (avoids sqrt singularities)
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    r_sum = r[I_IDX] + r[J_IDX]
    o = dx * dx + dy * dy - r_sum * r_sum
    
    return np.concatenate([b, o])

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds_r = []
    for i in range(n):
        ub = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds_r.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def project_to_feasible(centers, radii):
    """Strictly enforce constraints within numerical tolerance."""
    c = centers.copy()
    r = radii.copy()
    margin = 1e-10
    
    # Boundary projection
    for i in range(N):
        max_r = min(c[i, 0], 1.0 - c[i, 0], c[i, 1], 1.0 - c[i, 1]) - margin
        if r[i] > max_r:
            r[i] = max(0.0, max_r)
            
    # Overlap projection
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
                if d < r[i] + r[j] - margin:
                    excess = r[i] + r[j] - d + margin
                    r[i] -= excess / 2.0
                    r[j] -= excess / 2.0
                    changed = True
        if not changed:
            break
            
    r = np.maximum(r, 0.0)
    return c, r, np.sum(r)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_c = None
    best_r = None

            res = minimize(
                objective, x0, method='SLSQP', bounds=bounds,
                constraints=cons, options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False}
            )

                c = res.x[:2 * N].reshape(N, 2)
                r = res.x[2 * N :]
                c, r, s = project_to_feasible(c, r)
                return c, r, s
        except Exception:
            pass
        return None, None, 0.0

    inits = []

    return best_c, best_r, float(best_sum)
