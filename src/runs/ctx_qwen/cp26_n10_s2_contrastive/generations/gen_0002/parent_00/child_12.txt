from scipy.optimize import minimize
import math

N = 26
PAIR_MASK = np.triu(np.ones((N, N), dtype=bool), k=1)

def objective(x):

    return -np.sum(x[2 * N:])

def constraints(x):

    cy = x[N:2*N]
    r = x[2*N:]

    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r

    dy = cy[:, None] - cy[None, :]
