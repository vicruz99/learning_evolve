import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def compute_lp_radii(centers):
    """Solve LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    A_ub = np.zeros((n*(n-1)//2, n))
    A_ub[np.arange(n*(n-1)//2), I_IDX] = 1.0
    A_ub[np.arange(n*(n-1)//2), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, mx)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.zeros(n)

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[2*N:])

def constraints(params):
    """Inequality constraints: boundary and non-overlap."""
    c = params[:2*N].reshape(N, 2)
    r = params[2*N:]
    
    cons = []
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap constraints
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    cons.append(dists[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX])
    
    return np.concatenate(cons)

def run_packing():

    cons_dict = {'type': 'ineq', 'fun': constraints}

    best_c = None
    best_r = None

        row = 0

            while x < 0.96 and idx < N:
                c_hex[idx] = [x, y]
                x += sp
                idx += 1
            y += sp * np.sqrt(3) / 2.0
            row += 1
        c_hex += rng.randn(N, 2) * 0.008
        c_hex = np.clip(c_hex, 0.01, 0.99)
        inits.append(c_hex)
        
        # Grid lattice pattern
        c_grid = np.zeros((N, 2))
        idx = 0
        step = 0.18 + rng.uniform(-0.02, 0.02)
        y = 0.04
        while y < 0.96 and idx < N:
            x = 0.04
            while x < 0.96 and idx < N:
                c_grid[idx] = [x, y]
                x += step
                idx += 1
            y += step
        c_grid += rng.randn(N, 2) * 0.008
        c_grid = np.clip(c_grid, 0.01, 0.99)
        inits.append(c_grid)

    # Optimization loop
    for c0 in inits:
        r0 = compute_lp_radii(c0) * 0.98
        x0 = np.concatenate([c0.flatten(), r0])
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons_dict,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})

            ro = res.x[2*N:]
            ro = np.maximum(ro, 0.0)

        for i in range(N):

        for _ in range(150):

    if best_c is None:
        best_c = np.random.uniform(0.1, 0.9, (N, 2))
        best_r = compute_lp_radii(best_c)
        best_sum = np.sum(best_r)
        
    return best_c, best_r, float(best_sum)
