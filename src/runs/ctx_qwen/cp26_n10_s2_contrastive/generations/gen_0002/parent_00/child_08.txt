import numpy as np
from scipy.optimize import minimize, linprog
import math

N_CIRCLES = 26

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Preallocate constraint matrix for pairwise distances
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b_ub[idx] = math.hypot(dx, dy)
            idx += 1
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective_lp(centers_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = np.clip(centers_flat.reshape(N_CIRCLES, 2), 0.0, 1.0)
    _, sum_r = solve_radii_lp(centers)
    return -sum_r

def obj_full(params):
    """Objective for full joint optimization."""
    return -np.sum(params[2 * N_CIRCLES:])

def cons_full(params):
    """Constraints for full joint optimization."""
    c = params[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    r = params[2 * N_CIRCLES:]
    
    con = []
    # Boundary constraints
    con.extend(c[:, 0] - r)
    con.extend(1.0 - c[:, 0] - r)
    con.extend(c[:, 1] - r)
    con.extend(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            d = math.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
            con.append(d - r[i] - r[j])
    return np.array(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial center configurations
    inits = []
    
    # 1. Hexagonal lattices with varying spacing
    for seed in range(12):
        rng = np.random.RandomState(seed)
        c = np.zeros((N_CIRCLES, 2))
        idx = 0
        row = 0
        y = 0.05
        spacing = 0.175 + (seed % 6) * 0.005
        while idx < N_CIRCLES and y < 0.95:
            x = 0.05 + (row % 2) * spacing / 2.0
            while x < 0.95 and idx < N_CIRCLES:
                c[idx] = [x + rng.uniform(-0.012, 0.012), y + rng.uniform(-0.012, 0.012)]
                idx += 1
                x += spacing
            y += spacing * math.sqrt(3) / 2.0
            row += 1
        inits.append(c.flatten())
        
    # 2. Dense random clusters
    for seed in range(8):
        rng = np.random.RandomState(seed + 200)
        c = rng.uniform(0.1, 0.9, (N_CIRCLES, 2))
        inits.append(c.flatten())
        
    # 3. Staggered grid
    for seed in range(5):
        rng = np.random.RandomState(seed + 400)
        c = np.zeros((N_CIRCLES, 2))
        idx = 0
        y = 0.1
        while idx < N_CIRCLES and y < 0.95:
            x = 0.1
            shift = 0.05 if int(y * 10) % 2 == 1 else 0.0
            while x < 0.95 and idx < N_CIRCLES:
                c[idx] = [x + shift + rng.uniform(-0.01, 0.01), y + rng.uniform(-0.01, 0.01)]
                idx += 1
                x += 0.18
            y += 0.18
        inits.append(c.flatten())

    # Optimize centers using Nelder-Mead on LP objective
    for x0 in inits:
        try:
            res = minimize(objective_lp, x0, method='Nelder-Mead',
                           options={'maxiter': 10000, 'xatol': 1e-6, 'fatol': 1e-8})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = np.clip(res.x.reshape(N_CIRCLES, 2), 0.0, 1.0)
                _, best_radii = solve_radii_lp(best_centers)
                
                # Local perturbation search around best found so far
                for _ in range(4):
                    pert = best_centers + np.random.randn(N_CIRCLES, 2) * 0.004
                    pert = np.clip(pert, 0.02, 0.98)
                    res2 = minimize(objective_lp, pert.flatten(), method='Nelder-Mead',
                                    options={'maxiter': 6000, 'xatol': 1e-6, 'fatol': 1e-8})
                    s2 = -res2.fun
                    if s2 > best_sum:
                        best_sum = s2
                        best_centers = np.clip(res2.x.reshape(N_CIRCLES, 2), 0.0, 1.0)
                        _, best_radii = solve_radii_lp(best_centers)
        except Exception:
            continue

    # Fallback if optimization completely failed
    if best_centers is None:
        best_centers = np.random.rand(N_CIRCLES, 2) * 0.6 + 0.2
        best_radii, best_sum = solve_radii_lp(best_centers)

    # Final joint refinement with SLSQP to tighten constraints and squeeze margins
    x0_full = np.concatenate([best_centers.flatten(), best_radii])
    try:
        res_full = minimize(obj_full, x0_full, method='SLSQP',
                            bounds=[(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES,
                            constraints={'type': 'ineq', 'fun': cons_full},
                            options={'maxiter': 12000, 'ftol': 1e-14})
        if -res_full.fun > best_sum:
            best_sum = -res_full.fun
            best_centers = res_full.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
            best_radii = res_full.x[2 * N_CIRCLES:]
    except Exception:
        pass

    # Strict post-processing to guarantee validity
    for i in range(N_CIRCLES):
        max_r = min(best_centers[i, 0], 1.0 - best_centers[i, 0],
                    best_centers[i, 1], 1.0 - best_centers[i, 1])
        best_radii[i] = min(best_radii[i], max(0.0, max_r - 1e-9))

    # Iterative overlap resolution
    for _ in range(150):
        changed = False
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                d = math.hypot(best_centers[i, 0] - best_centers[j, 0],
                               best_centers[i, 1] - best_centers[j, 1])
                if d < best_radii[i] + best_radii[j] - 1e-12:
                    excess = best_radii[i] + best_radii[j] - d + 1e-10
                    best_radii[i] -= excess / 2.0
                    best_radii[j] -= excess / 2.0
                    changed = True
        if not changed:
            break

    best_sum = np.sum(best_radii)
    return best_centers, best_radii, float(best_sum)
