import numpy as np
from scipy.optimize import minimize
import math

N = 26
TRIU_IDX = np.triu_indices(N, k=1)

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[2 * N:])

def constraints(params):
    """
    Inequality constraints g(params) >= 0:
    - Boundary: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    - Non-overlap: dist_sq >= (r_i + r_j)^2
    """
    cx = params[:N]
    cy = params[N:2*N]
    r = params[2*N:]
    
    # Boundary constraints
    c_bound = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Pairwise squared distance constraints (vectorized)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    c_pair = dist_sq[TRIU_IDX] - r_sum[TRIU_IDX]**2
    return np.concatenate([c_bound, c_pair])

def hex_init(spacing, margin=0.05, seed=0):
    """Generate a hexagonal lattice initialization with controlled perturbation."""
    rng = np.random.RandomState(seed)
    centers = []
    y = margin
    row = 0
    while len(centers) < N:
        x_start = margin + (row % 2) * spacing / 2.0
        x = x_start
        while x < 1.0 - margin:
            centers.append([x, y])
            x += spacing
        y += spacing * math.sqrt(3) / 2.0
        row += 1
        
    centers = np.array(centers[:N])
    centers += rng.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    
    # Start with small feasible radii to guarantee initial feasibility for SLSQP
    radii = np.full(N, 0.01)
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    def optimize_from(start_c, start_r, maxiter=20000):
        x0 = np.concatenate([start_c.flatten(), start_r])
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons_dict,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
            if res.success:
                c_opt = res.x[:2*N].reshape(N, 2)
                r_opt = res.x[2*N:]
                r_opt = np.maximum(r_opt, 0.0)
                return c_opt, r_opt, np.sum(r_opt)
        except Exception:
            pass
        return None, None, 0.0

    # Phase 1: Broad search from diverse hexagonal lattices
    spacings = np.linspace(0.14, 0.21, 12)
    for s in spacings:
        for seed in range(4):
            c, r = hex_init(s, margin=0.05, seed=seed)
            opt_c, opt_r, s_val = optimize_from(c, r, maxiter=15000)
            if opt_c is not None and s_val > best_sum:
                best_sum = s_val
                best_centers = opt_c.copy()
                best_radii = opt_r.copy()

    # Phase 2: Local basin hopping to escape shallow local minima
    if best_centers is not None:
        for iteration in range(25):
            rng = np.random.RandomState(iteration * 13 + 7)
            noise_scale = 0.005 * (1.0 - iteration / 30.0)
            pert_c = best_centers + rng.normal(0, noise_scale, best_centers.shape)
            pert_c = np.clip(pert_c, 0.02, 0.98)
            pert_r = best_radii + rng.normal(0, noise_scale * 0.5, N)
            pert_r = np.clip(pert_r, 0.005, 0.4)
            
            opt_c, opt_r, s_val = optimize_from(pert_c, pert_r, maxiter=15000)
            if opt_c is not None and s_val > best_sum:
                best_sum = s_val
                best_centers = opt_c.copy()
                best_radii = opt_r.copy()

    # Phase 3: High-precision refinement from the best found configuration
    if best_centers is not None:
        opt_c, opt_r, s_val = optimize_from(best_centers, best_radii, maxiter=30000)
        if opt_c is not None and s_val > best_sum:
            best_sum = s_val
            best_centers = opt_c.copy()
            best_radii = opt_r.copy()

    # Final strict validity check and numerical adjustment
    if best_centers is not None:
        margin = 1e-9
        # Enforce boundary constraints strictly
        for i in range(N):
            mx = min(best_centers[i, 0], 1.0 - best_centers[i, 0], 
                     best_centers[i, 1], 1.0 - best_centers[i, 1])
            if best_radii[i] > mx:
                best_radii[i] = max(0.0, mx - margin)
                
        # Iteratively resolve any remaining overlaps to strict tolerance
        for _ in range(150):
            changed = False
            for i in range(N):
                for j in range(i + 1, N):
                    dx = best_centers[i, 0] - best_centers[j, 0]
                    dy = best_centers[i, 1] - best_centers[j, 1]
                    d = math.hypot(dx, dy)
                    if d < best_radii[i] + best_radii[j] - 1e-11:
                        excess = best_radii[i] + best_radii[j] - d + 1e-10
                        best_radii[i] -= excess / 2.0
                        best_radii[j] -= excess / 2.0
                        changed = True
            if not changed:
                break
        best_radii = np.maximum(best_radii, 0.0)
        best_sum = np.sum(best_radii)
    else:
        # Fallback (should not be reached)
        best_centers = np.random.rand(N, 2) * 0.6 + 0.2
        best_radii = np.full(N, 0.02)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
