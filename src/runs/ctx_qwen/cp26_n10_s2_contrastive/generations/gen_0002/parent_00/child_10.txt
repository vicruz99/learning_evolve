import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def solve_lp_radii(centers):
    """Solve LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((n * (n - 1) // 2, n))
    b_ub = np.zeros(n * (n - 1) // 2)
    bounds_r = []
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, ub)))
        
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.hypot(dx, dy)
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = d
            k += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.02)

def objective(x):
    """Negative sum of radii."""
    return -np.sum(x[2 * N_CIRCLES:])

def constraints(x):
    """Inequality constraints: boundary and squared pairwise non-overlap."""
    cx = x[:N_CIRCLES]
    cy = x[N_CIRCLES:2 * N_CIRCLES]
    r = x[2 * N_CIRCLES:]
    
    c = []
    # Boundary constraints
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Squared distance constraints for smoothness
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    c.append(dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2)
    
    return np.concatenate(c)

def generate_init_configs(rng):
    """Generate diverse initial center configurations."""
    configs = []
    n = N_CIRCLES
    
    # 1. Random placements
    for _ in range(25):
        c = rng.uniform(0.15, 0.85, (n, 2))
        configs.append(c)

        c = np.zeros((n, 2))
        idx = 0
        row = 0
        y = 0.08 + row * 0.185 * np.sqrt(3) / 2
        while idx < n and y < 0.95:
            x_start = 0.08 + (row % 2) * 0.0925
            col = 0
            while x_start + col * 0.185 < 0.95 and idx < n:
                c[idx] = [x_start + col * 0.185, y]
                idx += 1
                col += 1
            y += 0.185 * np.sqrt(3) / 2
            row += 1
        c += rng.uniform(-0.025, 0.025, c.shape)
        c = np.clip(c, 0.05, 0.95)
        configs.append(c)
        
    # 3. Grid-like variants
    for seed in range(10):
        c = np.zeros((n, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                if idx < n:
                    c[idx] = [0.12 + j * 0.16, 0.12 + i * 0.16]
                    idx += 1
        while idx < n:
            c[idx] = rng.uniform(0.2, 0.8, 2)
            idx += 1
        c += rng.uniform(-0.02, 0.02, c.shape)
        c = np.clip(c, 0.05, 0.95)
        configs.append(c)
        
    return configs

def run_packing():

    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    cons_dict = {'type': 'ineq', 'fun': constraints}
