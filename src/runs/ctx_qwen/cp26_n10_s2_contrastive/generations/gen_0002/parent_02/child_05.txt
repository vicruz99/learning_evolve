import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)

def obj_func(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[2*N:])

def constr_func(params):
    """Inequality constraints: boundary and pairwise non-overlap."""
    cx = params[:N]
    cy = params[N:2*N]
    r = params[2*N:]
    
    cons = []
    # Boundary constraints
    cons.append(cx - r)
    cons.append(1.0 - cx - r)
    cons.append(cy - r)
    cons.append(1.0 - cy - r)
    
    # Pairwise non-overlap constraints
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dists = np.sqrt(dx*dx + dy*dy + 1e-14)
    r_sum = r[:, None] + r[None, :]
    cons.append(dists[PAIRS_I, PAIRS_J] - r_sum[PAIRS_I, PAIRS_J])
    
    return np.concatenate(cons)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to find radii maximizing sum(radii)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b_ub[idx] = np.sqrt(dx*dx + dy*dy)
            idx += 1
            
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0-x, y, 1.0-y)
        bounds.append((0.0, max(0.0, ub)))
        
    # Try fast modern solver first, fallback if unavailable
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0
