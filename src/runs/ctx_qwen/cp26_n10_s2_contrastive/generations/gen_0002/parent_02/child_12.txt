The landscape is highly non-convex, and joint optimization of centers and radii often gets stuck in configurations where circles are symmetrically packed but suboptimally sized.

Key improvements in this approach:
1. **Hybrid Optimization (SLSQP + LP)**: We use SLSQP to find good center configurations, but for each promising set of centers, we solve a Linear Program (LP) to compute the exact maximum sum of radii. LP is extremely fast and exact for fixed centers, allowing us to evaluate the true potential of any layout without gradient coupling issues.
2. **Diverse Initialization & Force-Directed Pre-spread**: We generate initial layouts from hexagonal lattices, square grids, and random seeds. Before optimization, we apply a quick force-directed repulsion step to spread circles apart, ensuring the optimizer starts from a valid, well-separated state rather than an overlapping one.
3. **Multi-Stage Refinement**: After finding the best global configuration, we perform targeted local perturbations and re-run optimization. We also use a "radius inflation" schedule conceptually by letting SLSQP naturally push radii against constraints.
4. **Robust Constraint Handling**: We use `np.hypot` for pairwise distances to avoid singularities in gradient computation. Constraints are vectorized for speed.
5. **Strict Post-Processing**: A final pass ensures all circles strictly satisfy boundary and non-overlap constraints within the `1e-12` tolerance required by the validator, with a tiny safety margin to prevent numerical false failures.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_PAIRS, J_PAIRS = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_PAIRS)

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2 * N:])

def constraints(x):
    """Inequality constraints g(x) >= 0."""
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    c = np.empty(4 * N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[I_PAIRS] - cx[J_PAIRS]
    dy = cy[I_PAIRS] - cy[J_PAIRS]
    c[4*N:] = np.hypot(dx, dy) - (r[I_PAIRS] + r[J_PAIRS])
    return c

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    dx = centers[I_PAIRS, 0] - centers[J_PAIRS, 0]
    dy = centers[I_PAIRS, 1] - centers[J_PAIRS, 1]
    dists = np.hypot(dx, dy)
    
    A_ub_pairs = np.zeros((NUM_PAIRS, n))
    A_ub_pairs[np.arange(NUM_PAIRS), I_PAIRS] = 1.0
    A_ub_pairs[np.arange(NUM_PAIRS), J_PAIRS] = 1.0
    b_ub_pairs = dists
    
    # Boundary constraints handled via variable bounds for efficiency
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub_pairs, b_ub=b_ub_pairs, 
                      bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def force_spread(centers, steps=50, strength=0.05):
    """Quick repulsion step to separate overlapping circles."""
    cx = centers[:, 0].copy()
    cy = centers[:, 1].copy()
    n = len(cx)
    for _ in range(steps):
        fx = np.zeros(n)
        fy = np.zeros(n)
        for i in range(n):
            for j in range(i + 1, n):
                dx = cx[i] - cx[j]
                dy = cy[i] - cy[j]
                d = max(np.hypot(dx, dy), 1e-6)
                f = strength / (d**2)
                fx[i] += f * dx / d
                fy[i] += f * dy / d
                fx[j] -= f * dx / d
                fy[j] -= f * dy / d
        cx += fx
        cy += fy
        cx = np.clip(cx, 0.05, 0.95)
        cy = np.clip(cy, 0.05, 0.95)
    return np.column_stack((cx, cy))

def generate_initialization(seed, method='hex'):
    """Generate diverse initial layouts."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    
    if method == 'hex':
        r0 = 0.09 + rng.uniform(0, 0.03)
        dy = r0 * np.sqrt(3)
        y = r0 + 0.02
        row = 0
        while idx < N:
            shift = r0 if row % 2 == 1 else 0
            x = r0 + 0.02 + shift
            while x <= 1.0 - r0 - 0.02 and idx < N:
                centers[idx] = [x, y]
                idx += 1
                x += 2.0 * r0
            y += dy
            row += 1
    elif method == 'grid':
        step = 0.18 + rng.uniform(0, 0.04)
        y = step
        while y <= 1.0 - step and idx < N:
            x = step
            while x <= 1.0 - step and idx < N:
                centers[idx] = [x, y]
                idx += 1
                x += step
            y += step
    else:
        # Random
        centers[:, 0] = rng.uniform(0.15, 0.85, N)
        centers[:, 1] = rng.uniform(0.15, 0.85, N)
        idx = N
        
    # Pad or trim
    while idx < N:
        centers[idx] = [0.5, 0.5]
        idx += 1
        
    centers += rng.normal(0, 0.01, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    return force_spread(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Stage 1: Broad search with diverse initializations
    methods = ['hex', 'hex', 'hex', 'grid', 'grid', 'rand']
    for i in range(60):
        method = methods[i % len(methods)]
        seed = i * 7 + 3
        init_centers = generate_initialization(seed, method)
        
        # Initial radii guess
        r_init = np.full(N, 0.06)
        x0 = np.concatenate([init_centers.ravel(), r_init])
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            
            # Extract centers and solve exact LP for radii
            curr_centers = res.x[:2*N].reshape(N, 2)
            lp_r, lp_sum = solve_lp_radii(curr_centers)
            
            if lp_sum > best_sum:
                best_sum = lp_sum
                best_centers = curr_centers.copy()
                best_radii = lp_r.copy()
        except Exception:
            continue
            
    # Stage 2: Local refinement around best solution
    if best_centers is not None:
        rng = np.random.RandomState(42)
        for trial in range(20):
            # Perturb centers slightly
            pert_centers = best_centers + rng.normal(0, 0.004, best_centers.shape)
            pert_centers = np.clip(pert_centers, 0.02, 0.98)
            
            r_pert = np.full(N, 0.07)
            x0 = np.concatenate([pert_centers.ravel(), r_pert])
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons_dict,
                               options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
                
                curr_centers = res.x[:2*N].reshape(N, 2)
                lp_r, lp_sum = solve_lp_radii(curr_centers)
                
                if lp_sum > best_sum:
                    best_sum = lp_sum
                    best_centers = curr_centers.copy()
                    best_radii = lp_r.copy()
            except Exception:
                continue
                
    # Fallback safety
    if best_centers is None:
        best_centers = generate_initialization(0, 'hex')
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    # Stage 3: Strict post-processing to guarantee validity
    # Enforce boundaries strictly
    for i in range(N):
        mx = min(best_centers[i, 0], 1.0 - best_centers[i, 0])
        my = min(best_centers[i, 1], 1.0 - best_centers[i, 1])
        max_r = min(mx, my)
        if best_radii[i] > max_r:
            best_radii[i] = max(0.0, max_r - 1e-9)
            
    # Resolve any residual overlaps via iterative shrinking
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0],
                             best_centers[i, 1] - best_centers[j, 1])
                if d < best_radii[i] + best_radii[j] - 1e-10:
                    excess = best_radii[i] + best_radii[j] - d
                    best_radii[i] -= excess * 0.5
                    best_radii[j] -= excess * 0.5
                    changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
