import numpy as np
from scipy.optimize import minimize

N = 26

# Precompute pair indices for constraints to avoid repeated allocation
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def constraints(x):
    """Inequality constraints: g(x) >= 0"""
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    # Total constraints: 4*N boundary + N*(N-1)//2 pairwise
    c = np.empty(4*N + len(I_IDX))
    k = 0
    
    # Boundary constraints
    c[k:k+N] = cx - r; k += N
    c[k:k+N] = 1.0 - cx - r; k += N
    c[k:k+N] = cy - r; k += N
    c[k:k+N] = 1.0 - cy - r; k += N
    
    # Pairwise non-overlap constraints
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    d = np.hypot(dx, dy)
    r_sum = r[I_IDX] + r[J_IDX]
    c[k:] = d - r_sum
    
    return c

def hex_init(spacing, margin, rng):
    """Generate a perturbed hexagonal lattice initialization."""
    pts = []
    y = margin
    row = 0
    while len(pts) < N:
        x = margin + (row % 2) * spacing / 2.0
        while x < 1.0 - margin and len(pts) < N:
            pts.append([x + rng.uniform(-0.015, 0.015), y + rng.uniform(-0.015, 0.015)])
            x += spacing
        y += spacing * np.sqrt(3) / 2.0
        row += 1
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_x = None
    
    # Stage 1: Generate diverse initializations
    trials = []
    
    # Hexagonal lattice variations
    for i in range(60):
        rng = np.random.RandomState(i)
        spacing = 0.16 + rng.uniform(0.0, 0.06)
        margin = 0.03 + rng.uniform(0.0, 0.03)
        c = hex_init(spacing, margin, rng)
        r = np.full(N, 0.05)
        trials.append(np.concatenate([c.ravel(), r]))
        
    # Random inner-region placements
    for i in range(20):
        rng = np.random.RandomState(i + 200)
        c = rng.uniform(0.08, 0.92, (N, 2))
        r = np.full(N, 0.03)  # Smaller radii guarantee initial feasibility
        trials.append(np.concatenate([c.ravel(), r]))
        
    # Stage 2: Global search via SLSQP
    for x0 in trials:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_x = res.x.copy()
        except Exception:
            continue
            
    # Stage 3: Local basin refinement to escape shallow minima
    if best_x is not None:
        for _ in range(30):
            rng = np.random.RandomState()
            x_pert = best_x + rng.randn(len(best_x)) * 0.0015
            x_pert = np.clip(x_pert, [b[0] for b in bounds], [b[1] for b in bounds])
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
            except Exception:
                pass
                
    # Fallback safety net
    if best_x is None:
        c_fb = hex_init(0.18, 0.05, np.random.RandomState(42))
        r_fb = np.full(N, 0.05)
        best_x = np.concatenate([c_fb.ravel(), r_fb])
        
    # Extract results
    cx = best_x[:N].copy()
    cy = best_x[N:2*N].copy()
    r = best_x[2*N:].copy()
    
    # Stage 4: Strict feasibility enforcement
    # Iteratively clamp boundary and overlap violations to guarantee validation passes
    for _ in range(200):
        changed = False
        for i in range(N):
            mx = min(cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
            if r[i] > mx + 1e-10:
                r[i] = max(0.0, mx - 1e-10)
                changed = True
                
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                if d < r[i] + r[j] - 1e-10:
                    diff = r[i] + r[j] - d
                    r[i] -= diff / 2.0
                    r[j] -= diff / 2.0
                    changed = True
        if not changed:
            break
            
    r = np.maximum(r, 0.0)
    centers = np.column_stack((cx, cy))
    
    return centers, r, float(np.sum(r))
