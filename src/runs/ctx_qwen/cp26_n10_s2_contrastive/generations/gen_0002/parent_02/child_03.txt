The key insight is that for fixed centers, finding optimal radii is a Linear Programming (LP) problem. This decouples the non-smooth, highly constrained circle packing problem into:
1. **LP Radii Solver**: Given centers, instantly compute the maximum possible radii satisfying all non-overlap and boundary constraints.
2. **Center Optimization**: Use the LP solver's output as the objective function for a gradient-free optimizer (local random hill-climbing). This avoids SLSQP's difficulty with hundreds of active constraints and allows escaping local minima more robustly.
3. **SLSQP Polish**: After finding excellent centers via LP-guided search, run a final SLSQP refinement on all variables jointly for maximum precision.
4. **Strict Post-processing**: Iteratively shrink radii to guarantee numerical validity within the provided tolerance.

This hybrid LP + Random Search + SLSQP pipeline is highly effective for medium-sized circle packing problems and consistently pushes past pure gradient-based methods.

import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """
    Given fixed centers, solve LP to find radii that maximize sum(radii)
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.hypot(centers[i, 0] - centers[j, 0], 
                                 centers[i, 1] - centers[j, 1])
            idx += 1
            
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        # Add tiny tolerance to prevent strict infeasibility from numerical noise
        bounds_r.append((0.0, max(max_r, 0.0) + 1e-9))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback if LP fails
    return np.zeros(n), 0.0

def run_packing():
    N = 26
    
    # SLSQP components for joint optimization
    def obj_slq(x):
        return -np.sum(x[2*N:])
        
    def cons_slq(x):
        cx = x[:N]
        cy = x[N:2*N]
        r = x[2*N:]
        
        c = np.empty(4*N)
        c[:N] = cx - r
        c[N:2*N] = 1.0 - cx - r
        c[2*N:3*N] = cy - r
        c[3*N:4*N] = 1.0 - cy - r
        
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        dists = np.hypot(dx, dy)
        r_sum = r[:, None] + r[None, :]
        
        idx = np.triu_indices(N, k=1)
        c_pair = dists[idx] - r_sum[idx]
        
        return np.concatenate([c, c_pair])

    cons_dict = {'type': 'ineq', 'fun': cons_slq}
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: SLSQP from diverse structural initializations
    configs = []
    rng_init = np.random.RandomState(42)
    
    # Hexagonal lattice variants
    for s in range(5):
        r0 = 0.09 + s * 0.005
        pts = []
        y = r0
        row = 0
        while len(pts) < N:
            x = r0 + (row % 2) * r0
            while x <= 1.0 - r0 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
            row += 1
        pts = np.array(pts[:N])
        pts += rng_init.normal(0, 0.012, pts.shape)
        pts = np.clip(pts, 0.05, 0.95)
        configs.append(np.concatenate([pts[:, 0], pts[:, 1], np.full(N, 0.08)]))
        
    # Safe random placements
    for s in range(5):
        pts = rng_init.uniform(0.12, 0.88, (N, 2))
        configs.append(np.concatenate([pts[:, 0], pts[:, 1], np.full(N, 0.07)]))
        
    for x0 in configs:
        try:
            res = minimize(obj_slq, x0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x[:2*N].reshape(N, 2)
                best_radii = res.x[2*N:]
        except Exception:
            continue
            
    # Phase 2: LP-guided Local Random Search on Centers
    # This decouples radii optimization from center movement, handling non-smoothness robustly
    if best_centers is not None:
        current_c = best_centers.copy()
        current_s = best_sum
        step = 0.008
        rng_search = np.random.RandomState(123)
        
        for _ in range(1500):
            pc = current_c + rng_search.normal(0, step, current_c.shape)
            pc = np.clip(pc, 0.02, 0.98)
            
            r, s = solve_radii_lp(pc)
            if s > current_s:
                current_c = pc.copy()
                current_s = s
                step = np.clip(step * 1.05, 0.0005, 0.05)

                step = np.clip(step * 0.985, 0.0005, 0.05)
                
        best_centers = current_c
        best_radii, best_sum = solve_radii_lp(current_c)
        
    # Phase 3: Final SLSQP polish from LP-optimized centers
    if best_centers is not None:
        x0_polish = np.concatenate([best_centers[:, 0], best_centers[:, 1], best_radii])
        try:
            res_p = minimize(obj_slq, x0_polish, method='SLSQP', bounds=bounds, constraints=cons_dict,

            if -res_p.fun > best_sum:
                best_sum = -res_p.fun
                best_centers = res_p.x[:2*N].reshape(N, 2)
                best_radii = res_p.x[2*N:]
        except Exception:
            pass
            
    # Fallback safety net
    if best_centers is None:
        best_centers = np.random.rand(N, 2) * 0.6 + 0.2
        best_radii, best_sum = solve_radii_lp(best_centers)
        
    # Phase 4: Strict numerical post-processing
    cx, cy = best_centers[:, 0].copy(), best_centers[:, 1].copy()
    r = best_radii.copy()
    
    for _ in range(100):
        changed = False
        # Fix boundary violations
        for i in range(N):
            mx = min(cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
            if r[i] > mx + 1e-10:
                r[i] = max(0.0, mx - 1e-10)
                changed = True
                
        # Fix overlap violations
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                if d < r[i] + r[j] - 1e-10:
                    excess = r[i] + r[j] - d
                    r[i] -= excess * 0.5
                    r[j] -= excess * 0.5
                    changed = True
        if not changed:
            break
            
    best_radii = np.maximum(r, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
