import numpy as np
from scipy.optimize import minimize, linprog

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.sqrt(dx*dx + dy*dy)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # 1. Generate diverse initial center configurations
    inits = []
    
    # Hexagonal lattice variations
    for s in np.linspace(0.085, 0.115, 12):
        pts = []
        y = s
        row = 0
        while len(pts) < n:
            shift = s if row % 2 == 1 else 0
            x = s + shift
            while x <= 1.0 - s and len(pts) < n:
                pts.append([x, y])
                x += 2.0 * s
            y += s * np.sqrt(3)
            row += 1
        inits.append(np.array(pts[:n]))
        
    # Grid variations
    for sp in np.linspace(0.18, 0.22, 10):
        pts = []
        y = sp
        while y <= 1.0 - sp and len(pts) < n:
            x = sp
            while x <= 1.0 - sp and len(pts) < n:
                pts.append([x, y])
                x += sp
            y += sp
        inits.append(np.array(pts[:n]))
        
    # Random placements
    for seed in range(15):
        rng = np.random.RandomState(seed)
        pts = rng.uniform(0.1, 0.9, (n, 2))
        inits.append(pts)
        
    # Perturbations of structural inits
    for init in inits[:15]:
        for _ in range(3):
            rng = np.random.RandomState()
            p = init + rng.normal(0, 0.02, init.shape)
            inits.append(np.clip(p, 0.05, 0.95))
            
    # 2. Joint optimization function definitions
    def obj_joint(x):
        return -np.sum(x[2*n:])
        
    def constr_joint(x):
        cx = x[:n]
        cy = x[n:2*n]
        r = x[2*n:]
        
        c = np.empty(4*n + n*(n-1)//2)
        c[:n] = cx - r
        c[n:2*n] = 1.0 - cx - r
        c[2*n:3*n] = cy - r
        c[3*n:4*n] = 1.0 - cy - r
        
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        dist_sq = dx**2 + dy**2
        r_sum = r[:, None] + r[None, :]
        c[4*n:] = dist_sq[IDX_I, IDX_J] - r_sum[IDX_I, IDX_J]**2
        return c
        
    bounds_joint = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constr_joint}
    
    # 3. Optimize each initialization
    for init_c in inits:
        # Get LP optimal radii for these centers to start SLSQP on the feasible boundary
        r_lp, _ = solve_lp_radii(init_c)
        
        x0 = np.concatenate([init_c.flatten(), r_lp])
        
        try:
            res = minimize(obj_joint, x0, method='SLSQP', bounds=bounds_joint, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            
            if res.success:
                cx = res.x[:n]
                cy = res.x[n:2*n]
                r = res.x[2*n:]
                s_val = np.sum(r)
                
                if s_val > best_sum:
                    best_sum = s_val
                    best_centers = np.column_stack((cx, cy))
                    best_radii = r.copy()
        except Exception:
            continue
            
    # 4. Local refinement around the best found solution
    if best_centers is not None:
        rng = np.random.RandomState(42)
        for _ in range(20):
            c_p = best_centers + rng.normal(0, 0.004, best_centers.shape)
            c_p = np.clip(c_p, 0.01, 0.99)
            r_p, _ = solve_lp_radii(c_p)
            x0_p = np.concatenate([c_p.flatten(), r_p])
            
            try:
                res_p = minimize(obj_joint, x0_p, method='SLSQP', bounds=bounds_joint, constraints=cons,
                                 options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                if res_p.success:
                    cx_p = res_p.x[:n]
                    cy_p = res_p.x[n:2*n]
                    r_p_opt = res_p.x[2*n:]
                    s_p = np.sum(r_p_opt)
                    if s_p > best_sum:
                        best_sum = s_p
                        best_centers = np.column_stack((cx_p, cy_p))
                        best_radii = r_p_opt.copy()
            except Exception:
                continue
                
    # 5. Final strict validation and numerical cleanup
    if best_centers is None:
        best_centers = np.tile([0.5, 0.5], (n, 1))
        best_radii = np.full(n, 0.01)
        best_sum = np.sum(best_radii)
        
    # Iteratively enforce constraints strictly
    for _ in range(50):
        changed = False
        for i in range(n):
            mx = min(best_centers[i,0], 1.0-best_centers[i,0], 
                     best_centers[i,1], 1.0-best_centers[i,1])
            if best_radii[i] > mx + 1e-9:
                best_radii[i] = max(0.0, mx - 1e-9)
                changed = True
                
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], 
                             best_centers[i,1]-best_centers[j,1])
                if d < best_radii[i] + best_radii[j] - 1e-9:
                    exc = best_radii[i] + best_radii[j] - d
                    best_radii[i] -= exc / 2.0
                    best_radii[j] -= exc / 2.0
                    changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
