import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_radii_lp(centers):
    """
    Given fixed centers, solve an LP to find radii that maximize sum(radii)
    subject to boundary and non-overlap constraints.
    """
    n = len(centers)
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Pairwise distance constraints: r_i + r_j <= dist(i,j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def get_sum_radii(x):
    """Objective function for center optimization: returns negative max sum of radii."""
    centers = x.reshape((N, 2))
    # Strictly clip to interior to avoid LP bound errors
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    _, s = solve_radii_lp(centers)
    return -s

                x += dx
            y += dy
            row += 1
        inits.append(np.array(pts[:N]))

            best_r[i] = max(0.0, mx)

    best_r = np.maximum(best_r, 0.0)
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
