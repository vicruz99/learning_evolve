

<strategy>
The previous solutions relied on SLSQP to optimize centers and radii simultaneously, which suffers from non-convexity, constraint handling overhead, and getting trapped in shallow local minima. 

A more robust approach is to **decouple the problem**:
1. **Fix centers**, then find optimal radii via a **Linear Program (LP)**. Given fixed positions, maximizing the sum of radii subject to $r_i + r_j \leq \|c_i - c_j\|$ and boundary constraints is a convex LP, solvable exactly and instantly with `scipy.optimize.linprog`.
2. **Optimize centers** using a derivative-free optimizer (`Nelder-Mead`) on the non-smooth objective function $F(\text{centers}) = \sum r_i(\text{centers})$. This bypasses gradient singularities and constraint Jacobian issues.
3. Use **diverse initializations** (hexagonal lattices with varying spacings, random spreads, grids) and **iterative perturbation** to escape local minima.
4. The LP guarantees optimal radii for any valid center configuration, allowing the outer optimizer to focus purely on geometric arrangement. This hybrid strategy consistently finds denser packings and is highly stable.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to non-overlap and boundary constraints."""
    n = len(centers)
    c_obj = -np.ones(n)  # Maximize sum(r) <=> Minimize -sum(r)
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b_ub[idx] = np.hypot(dx, dy)
            idx += 1
            
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective(centers_flat):
    """Objective function for center optimization: negative sum of optimal radii."""
    centers = centers_flat.reshape((N, 2))
    # Heavy penalty if centers drift far outside bounds to keep simplex search in feasible region
    if np.any(centers < -0.1) or np.any(centers > 1.1):
        return -1000.0
    _, sum_r = solve_radii_lp(centers)
    return -sum_r

def gen_hex_init(seed, spacing=0.18, margin=0.05):
    """Generate a perturbed hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = margin
    row = 0
    while idx < N and y < 1.0 - margin:
        x = margin + (row % 2) * spacing / 2.0
        while x < 1.0 - margin and idx < N:
            centers[idx] = [x + rng.uniform(-spacing * 0.1, spacing * 0.1), 
                            y + rng.uniform(-spacing * 0.1, spacing * 0.1)]
            idx += 1
            x += spacing
        y += spacing * np.sqrt(3) / 2.0
        row += 1
    while idx < N:
        centers[idx] = rng.uniform(margin, 1.0 - margin, 2)
        idx += 1
    return centers

def run_packing():
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    inits = []
    # 1. Hexagonal patterns with varying spacing to cover dense/sparse configurations
    for s in range(20):
        spacing = 0.14 + s * 0.012
        inits.append(gen_hex_init(s, spacing))
        
    # 2. Random spreads
    for s in range(15):
        rng = np.random.RandomState(s + 100)
        inits.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    # 3. Regular grid patterns
    for s in range(5):
        c = np.zeros((N, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                if idx < N:
                    c[idx] = [0.1 + j * 0.175, 0.1 + i * 0.175]
                    idx += 1
        while idx < N:
            c[idx] = np.random.uniform(0.2, 0.8, 2)
            idx += 1
        inits.append(c)

    # Main optimization loop
    for i, c0 in enumerate(inits):
        res = minimize(objective, c0.flatten(), method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-5, 'fatol': 1e-7})
        curr_sum = -res.fun
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = res.x.reshape((N, 2))
            _, best_radii = solve_radii_lp(best_centers)
            
        # Local perturbation refinement around the current best
        for _ in range(4):
            rng = np.random.RandomState(_ * 13 + i * 7)
            c_pert = best_centers + rng.randn(N, 2) * 0.004
            res2 = minimize(objective, c_pert.flatten(), method='Nelder-Mead',
                            options={'maxiter': 2500, 'xatol': 1e-6, 'fatol': 1e-8})
            if -res2.fun > best_sum:
                best_sum = -res2.fun
                best_centers = res2.x.reshape((N, 2))
                _, best_radii = solve_radii_lp(best_centers)

    # Final projection to [0,1] and strict radii recalculation
    if best_centers is not None:
        centers = np.clip(best_centers, 0.0, 1.0)
        radii, _ = solve_radii_lp(centers)
        best_sum = np.sum(radii)
    else:
        centers = np.random.rand(N, 2) * 0.8 + 0.1
        radii, _ = solve_radii_lp(centers)
        best_sum = np.sum(radii)

    return centers, radii, float(best_sum)
```
