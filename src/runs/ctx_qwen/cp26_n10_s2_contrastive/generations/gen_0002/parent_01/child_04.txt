import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
BOUNDS = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[2 * N:])

def constraints(params):
    """Inequality constraints: boundary and squared non-overlap (must be >= 0)."""
    cx = params[:N]
    cy = params[N:2 * N]
    r = params[2 * N:]
    
    # Boundary constraints
    c = np.empty(4 * N)
    c[0:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    # Squared distance constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist_sq = dx*dx + dy*dy
    r_sum = r[:, None] + r[None, :]
    
    c_overlap = dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2
    return np.concatenate([c, c_overlap])

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, max_r)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success and np.all(res.x >= -1e-12):
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback radii if LP fails
    safe_r = np.array([min(c[0], 1.0-c[0], c[1], 1.0-c[1]) * 0.9 for c in centers])
    safe_r = np.maximum(safe_r, 0.0)
    return safe_r, np.sum(safe_r)

def generate_hex_initialization(seed, row_counts, scale=0.16, shift_x=0.0, shift_y=0.0):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = 0.05 + shift_y
    row = 0
    
    while idx < N and row < len(row_counts):
        count = row_counts[row]
        x_start = 0.05 + (row % 2) * scale / 2.0 + shift_x
        for k in range(count):
            if idx < N:
                x = x_start + k * scale
                centers[idx, 0] = np.clip(x + rng.uniform(-0.005, 0.005), 0.02, 0.98)
                centers[idx, 1] = np.clip(y + rng.uniform(-0.005, 0.005), 0.02, 0.98)
                idx += 1
        y += scale * math.sqrt(3) / 2.0
        row += 1
        
    # Fill remaining if needed
    while idx < N:
        centers[idx, 0] = rng.uniform(0.2, 0.8)
        centers[idx, 1] = rng.uniform(0.2, 0.8)
        idx += 1
        
    return centers

def fix_violations(centers, radii):
    """Strictly enforce boundary and overlap constraints."""
    n = centers.shape[0]
    # Clamp to boundaries
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        if radii[i] > mx:
            radii[i] = max(0.0, mx - 1e-9)
            
    # Resolve overlaps
    for _ in range(150):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                s = radii[i] + radii[j]
                if d < s - 1e-10:
                    diff = s - d
                    radii[i] -= diff * 0.5
                    radii[j] -= diff * 0.5
                    changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    return centers, radii

def run_packing():
    best_sum = 0.0
    best_centers = None
    best_radii = None
    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    # Diverse row configurations summing to 26
    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [7, 5, 5, 5, 4], 
        [5, 5, 5, 5, 6], [8, 6, 5, 4, 3], [4, 5, 6, 5, 6]
    ]
    
    # Phase 1: Broad search with varied hexagonal starts
    for pattern in row_patterns:
        for seed in range(5):
            for scale in [0.155, 0.165, 0.175]:
                c_init = generate_hex_initialization(seed * 100 + len(pattern), pattern, scale)
                r_init = np.full(N, 0.03)
                x0 = np.concatenate([c_init.ravel(), r_init])
                
                try:
                    res = minimize(objective, x0, method='SLSQP', bounds=BOUNDS, constraints=cons_dict,
                                   options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                    if res.success:
                        c_opt = res.x[:2*N].reshape(N, 2)
                        r_opt = res.x[2*N:]
                        
                        # LP Polish for exact radii
                        r_lp, s_lp = solve_radii_lp(c_opt)
                        if s_lp > best_sum:
                            best_sum = s_lp
                            best_centers = c_opt.copy()
                            best_radii = r_lp.copy()
                except Exception:
                    continue

    # Phase 2: Local refinement and basin hopping around best
    if best_centers is not None:
        c_curr = best_centers.copy()
        r_curr = best_radii.copy()
        
        for iteration in range(25):
            # Perturb centers
            rng = np.random.RandomState(iteration * 31 + 7)
            c_pert = c_curr + rng.randn(N, 2) * (0.002 + 0.001 * iteration)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            r_pert = np.full(N, 0.03)
            x0_pert = np.concatenate([c_pert.ravel(), r_pert])
            
            try:
                res_p = minimize(objective, x0_pert, method='SLSQP', bounds=BOUNDS, constraints=cons_dict,
                                 options={'maxiter': 5000, 'ftol': 1e-13})
                if res_p.success:
                    c_opt_p = res_p.x[:2*N].reshape(N, 2)
                    r_lp_p, s_lp_p = solve_radii_lp(c_opt_p)
                    if s_lp_p > best_sum:
                        best_sum = s_lp_p
                        best_centers = c_opt_p.copy()
                        best_radii = r_lp_p.copy()
                        c_curr = c_opt_p.copy()
                        r_curr = r_lp_p.copy()
            except Exception:
                continue
                
    # Fallback if optimization completely failed
    if best_centers is None:
        best_centers = generate_hex_initialization(42, [6, 5, 6, 5, 4], 0.16)
        best_radii, best_sum = solve_radii_lp(best_centers)
        
    # Final strict validation and numerical adjustment
    best_centers, best_radii = fix_violations(best_centers, best_radii)
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(final_sum)
