import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
N_PAIRS = len(I)
BOUNDS = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

# Precompute LP constraint matrix structure (constant for all LP calls)
A_UB_LP = np.zeros((N_PAIRS, N))
for k, (i, j) in enumerate(zip(I, J)):
    A_UB_LP[k, i] = 1.0
    A_UB_LP[k, j] = 1.0

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2 * N:])

def constraints(x):
    """Inequality constraints g(x) >= 0: boundary and non-overlap."""
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    c = np.empty(4 * N + N_PAIRS)
    # Boundary constraints
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    # Pairwise non-overlap constraints
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dists = np.sqrt(dx * dx + dy * dy)
    c[4*N:] = dists[I, J] - (r[I] + r[J])
    return c

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    # Compute pairwise distances efficiently
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.sqrt(dx * dx + dy * dy)
    b_ub = dists[I, J]
    
    # Boundary constraints for radii
    bounds_r = []
    for i in range(N):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, ub)))
        
    try:
        # Use HiGHS solver for speed and reliability
        res = linprog(-np.ones(N), A_ub=A_UB_LP, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            r_opt = np.maximum(res.x, 0.0)
            return r_opt, -res.fun
    except Exception:
        pass
        
    # Fallback if LP fails (should be rare)
    r_fallback = np.zeros(N)
    for i in range(N):
        x, y = centers[i]
        r_fallback[i] = min(x, 1.0 - x, y, 1.0 - y) * 0.1
    return r_fallback, 0.0

def make_hex_init(seed, s_min=0.14, s_max=0.18):
    """Generate a perturbed hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    s = s_min + rng.uniform(0, s_max - s_min)
    cx = np.zeros(N)
    cy = np.zeros(N)
    idx = 0
    row = 0
    y = 0.05 + row * s * np.sqrt(3) / 2
    while idx < N:
        x_start = 0.05 + (row % 2) * s / 2
        col = 0
        while x_start + col * s <= 0.95 and idx < N:
            cx[idx] = x_start + col * s
            cy[idx] = y
            idx += 1
            col += 1
        y += s * np.sqrt(3) / 2
        row += 1
        
    # Add small perturbation and clip
    cx += rng.normal(0, 0.005, N)
    cy += rng.normal(0, 0.005, N)
    cx = np.clip(cx, 0.02, 0.98)
    cy = np.clip(cy, 0.02, 0.98)
    
    centers = np.column_stack((cx, cy))
    # Initialize radii optimally for these centers
    r, s_sum = solve_lp_radii(centers)
    return np.concatenate([cx, cy, r])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0.0
    best_x = None
    
    # Phase 1: Broad search with diverse hexagonal seeds
    for seed in range(60):
        x0 = make_hex_init(seed)
        try:
            res = minimize(
                objective, x0, method='SLSQP', bounds=BOUNDS, constraints=constraints,
                options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False}
            )
            curr_c = np.column_stack((res.x[:N], res.x[N:2*N]))
            # Polish radii with LP for exact optimality given centers
            curr_r, curr_s = solve_lp_radii(curr_c)
            if curr_s > best_sum:
                best_sum = curr_s
                best_x = np.concatenate([curr_c.ravel(), curr_r])
        except Exception:
            continue
            
    # Phase 2: Intensive local refinement around the best solution
    if best_x is not None:
        best_c = best_x[:2*N].reshape(N, 2)
        best_r = best_x[2*N:]
        
        for _ in range(40):
            rng = np.random.RandomState()
            c_p = best_c + rng.normal(0, 0.002, (N, 2))
            c_p = np.clip(c_p, 0.01, 0.99)
            r_p, _ = solve_lp_radii(c_p)
            x0_p = np.concatenate([c_p.ravel(), r_p * 0.99])
            
            try:
                res = minimize(
                    objective, x0_p, method='SLSQP', bounds=BOUNDS, constraints=constraints,
                    options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False}
                )
                curr_c = np.column_stack((res.x[:N], res.x[N:2*N]))
                curr_r, curr_s = solve_lp_radii(curr_c)
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_x = np.concatenate([curr_c.ravel(), curr_r])
                    best_c = curr_c
                    best_r = curr_r
            except Exception:
                continue

    # Fallback initialization
    if best_x is None:
        best_x = make_hex_init(0)
        
    best_c = best_x[:2*N].reshape(N, 2)
    best_r = best_x[2*N:]
    
    # Final strict validation fix to guarantee passing the checker
    for i in range(N):
        x, y = best_c[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        best_r[i] = min(best_r[i], max(0.0, mx - 1e-9))
        
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                if d < best_r[i] + best_r[j] - 1e-9:
                    diff = best_r[i] + best_r[j] - d + 1e-9
                    best_r[i] -= diff * 0.5
                    best_r[j] -= diff * 0.5
                    changed = True
        if not changed:
            break
            
    best_r = np.maximum(best_r, 0.0)
    best_sum = np.sum(best_r)
    
    return best_c, best_r, float(best_sum)
