Strategy:
1. **LP Radii Solver**: For any given set of centers, compute the optimal radii that maximize the sum while satisfying non-overlap (`r_i + r_j <= dist_ij`) and boundary constraints. This is a standard LP.
2. **Global Center Search**: Use `scipy.optimize.differential_evolution` to search the 52-dimensional space of circle centers. DE is highly effective for non-smooth, multi-modal objectives and will naturally push circles apart to allow larger radii.
3. **Local Refinement**: After DE finds a promising configuration, apply `Nelder-Mead` local optimization on the centers to fine-tune positions and squeeze out extra radius sum.
4. **Validation & Correction**: Apply strict boundary enforcement and iterative overlap resolution to guarantee the solution passes the validator's `1e-12` tolerance checks.
5. **Fallback**: Provide a deterministic hexagonal lattice fallback to ensure robustness if global optimization fails or times out.

import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize
import warnings
warnings.filterwarnings('ignore')

N_CIRCLES = 26

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise distances
    dx = centers[:, 0, np.newaxis] - centers[:, 0]
    dy = centers[:, 1, np.newaxis] - centers[:, 1]
    dists = np.hypot(dx, dy)
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints as variable bounds: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        ub = max(0.0, max_r - 1e-9)  # Slight shrinkage for numerical safety
        bounds_r.append((0.0, ub))
        
    # Try HiGHS first (fast), fallback to interior-point
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success and np.all(res.x >= -1e-7):
                return res.x, -res.fun
        except Exception:
            continue
            
    return np.zeros(n), 0.0

def objective_centers(centers_flat):
    """Objective for optimizing centers: maximize sum of radii (minimize negative sum)."""
    centers = centers_flat.reshape(N_CIRCLES, 2)
    # Keep centers strictly inside to allow positive radii
    centers = np.clip(centers, 0.005, 0.995)
    _, s = solve_lp_radii(centers)
    return -s

def run_packing():
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds_de = [(0.01, 0.99)] * (2 * N_CIRCLES)
    
    # Phase 1: Global search on centers using Differential Evolution
    # DE handles the non-smooth landscape created by LP constraints well.
    for seed in [10, 20, 30, 40]:
        try:
            res = differential_evolution(
                objective_centers, 
                bounds_de, 
                seed=seed, 
                popsize=10, 
                maxiter=60, 
                tol=1e-7, 
                atol=1e-7, 
                polish=False,
                workers=1
            )
            c_de = np.clip(res.x.reshape(N_CIRCLES, 2), 0.005, 0.995)
            r_de, s_de = solve_lp_radii(c_de)
            if s_de > best_sum:
                best_sum = s_de
                best_centers = c_de.copy()
                best_radii = r_de.copy()
        except Exception:
            pass
            
    # Phase 2: Local refinement using Nelder-Mead to escape local plateaus
    if best_centers is not None:
        try:
            res_nm = minimize(
                objective_centers, 
                best_centers.ravel(), 
                method='Nelder-Mead', 
                options={'maxiter': 10000, 'xatol': 1e-7, 'fatol': 1e-8}
            )
            c_nm = np.clip(res_nm.x.reshape(N_CIRCLES, 2), 0.005, 0.995)
            r_nm, s_nm = solve_lp_radii(c_nm)
            if s_nm > best_sum:
                best_sum = s_nm
                best_centers = c_nm.copy()
                best_radii = r_nm.copy()
        except Exception:
            pass

    # Phase 3: Fallback to deterministic hexagonal lattice if global search fails
    if best_centers is None:
        best_centers = np.zeros((N_CIRCLES, 2))
        idx = 0
        row = 0
        while idx < N_CIRCLES:
            y = 0.08 + row * 0.155
            x_start = 0.08 + (row % 2) * 0.085
            col = 0
            while x_start + col * 0.17 <= 0.92 and idx < N_CIRCLES:
                best_centers[idx] = [x_start + col * 0.17, y]
                idx += 1
                col += 1
            row += 1
        while idx < N_CIRCLES:
            best_centers[idx] = [0.5, 0.5]
            idx += 1
        best_radii, best_sum = solve_lp_radii(best_centers)

    # Phase 4: Strict validation and numerical correction
    c_final = best_centers.copy()
    r_final = best_radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N_CIRCLES):
        x, y = c_final[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if r_final[i] > max_r:
            r_final[i] = max(0.0, max_r - 1e-9)
            
    # Iteratively resolve any microscopic overlaps from numerical drift
    for _ in range(200):
        changed = False
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                d = np.hypot(c_final[i, 0] - c_final[j, 0], c_final[i, 1] - c_final[j, 1])
                s = r_final[i] + r_final[j]
                if d < s - 1e-12:
                    excess = s - d + 1e-10
                    r_final[i] -= excess * 0.5
                    r_final[j] -= excess * 0.5
                    changed = True
        if not changed:
            break
            
    r_final = np.maximum(r_final, 0.0)
    final_sum = np.sum(r_final)
    
    return c_final, r_final, float(final_sum)
