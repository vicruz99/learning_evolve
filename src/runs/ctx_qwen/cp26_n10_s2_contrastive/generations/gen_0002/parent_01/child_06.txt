import numpy as np
from scipy.optimize import minimize, linprog

N = 26
N_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure (constant for all center configurations)
A_LP = np.zeros((N_PAIRS, N))
PAIR_I = np.empty(N_PAIRS, dtype=int)
PAIR_J = np.empty(N_PAIRS, dtype=int)
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_I[idx] = i
        PAIR_J[idx] = j
        idx += 1

def solve_radii_lp(centers):
    """Given centers, solve LP to maximize sum of radii subject to non-overlap and boundaries."""
    # Compute all pairwise distances efficiently
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt((diff ** 2).sum(axis=2))
    b_ub = dists[PAIR_I, PAIR_J]
    
    # Boundary bounds: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for x, y in centers:
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, mx)))
        
    # Try HiGHS first (fast), fallback to interior-point
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method=method)
            if res.success:
                return res.x, -res.fun
        except Exception:
            continue
            
    return np.zeros(N), 0.0

def obj_centers_only(x):
    """Objective for optimizing centers: negative sum of optimal radii."""
    centers = np.clip(x.reshape(N, 2), 1e-7, 1.0 - 1e-7)
    _, s = solve_radii_lp(centers)
    return -s

def joint_obj(x):
    return -np.sum(x[2 * N:])

def joint_constr(x):
    cx = x[:N]
    cy = x[N : 2 * N]
    r = x[2 * N :]
    c = []
    # Boundary constraints
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    # Pairwise non-overlap
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dists = np.hypot(dx, dy)
    r_sum = r[:, None] + r[None, :]
    triu_idx = np.triu_indices(N, k=1)
    c.extend(dists[triu_idx] - r_sum[triu_idx])
    return np.array(c)

def generate_inits():
    inits = []
    rng = np.random.default_rng(42)
    
    # Hexagonal grids with varying spacing and offsets
    for seed in range(12):
        c = np.zeros((N, 2))
        idx = 0
        row = 0
        spacing = 0.145 + seed * 0.006
        dy = spacing * np.sqrt(3) / 2.0
        margin = 0.035
        y = margin + dy / 2.0
        while idx < N and y < 1.0 - margin:
            x_off = margin + (row % 2) * spacing / 2.0
            col = 0
            while x_off + col * spacing <= 1.0 - margin and idx < N:
                c[idx] = [x_off + col * spacing, y]
                idx += 1
                col += 1
            y += dy
            row += 1
        while idx < N:
            c[idx] = rng.uniform(0.15, 0.85, 2)
            idx += 1
        inits.append(c.flatten())
        
    # Random inits to cover non-lattice basins
    for _ in range(5):
        c = rng.uniform(0.15, 0.85, (N, 2))
        inits.append(c.flatten())
        
    return inits

def run_packing():
    best_sum = 0.0
    best_c = None
    best_r = None
    
    inits = generate_inits()
    
    # Phase 1: Optimize centers using derivative-free method
    # Nelder-Mead handles the non-smooth LP landscape robustly
    for p0 in inits:
        try:
            res = minimize(obj_centers_only, p0, method='Nelder-Mead', 
                           options={'maxiter': 4000, 'xatol': 1e-6, 'fatol': 1e-8})
            c_opt = np.clip(res.x.reshape(N, 2), 1e-6, 1.0 - 1e-6)
            r_opt, s_opt = solve_radii_lp(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            continue
            
    # Phase 2: Joint refinement with SLSQP to polish both centers and radii
    if best_c is not None:
        bounds_joint = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        cons_joint = {'type': 'ineq', 'fun': joint_constr}
        
        x0 = np.concatenate([best_c.flatten(), best_r])
        rng = np.random.default_rng(123)
        
        for _ in range(20):
            x_pert = x0 + rng.normal(0, 0.003, len(x0))
            x_pert = np.clip(x_pert, [b[0] for b in bounds_joint], [b[1] for b in bounds_joint])
            try:
                res2 = minimize(joint_obj, x_pert, method='SLSQP', bounds=bounds_joint, 
                                constraints=cons_joint, options={'maxiter': 6000, 'ftol': 1e-13})
                if res2.success:
                    c2 = res2.x[:2 * N].reshape(N, 2)
                    r2 = res2.x[2 * N :]
                    
                    # Strict boundary enforcement
                    for i in range(N):
                        mx = min(c2[i, 0], 1.0 - c2[i, 0], c2[i, 1], 1.0 - c2[i, 1])
                        if r2[i] > mx:
                            r2[i] = max(0.0, mx)
                            
                    # Strict overlap enforcement via iterative shrinking
                    for _ in range(100):
                        changed = False
                        for i in range(N):
                            for j in range(i + 1, N):
                                d = np.hypot(c2[i, 0] - c2[j, 0], c2[i, 1] - c2[j, 1])
                                s = r2[i] + r2[j]
                                if d < s - 1e-10:
                                    exc = s - d
                                    r2[i] -= exc * 0.5
                                    r2[j] -= exc * 0.5
                                    changed = True
                        if not changed:
                            break
                            
                    r2 = np.maximum(r2, 0.0)
                    s2 = np.sum(r2)
                    if s2 > best_sum:
                        best_sum = s2
                        best_c = c2.copy()
                        best_r = r2.copy()
            except Exception:
                continue

    # Fallback safety net
    if best_c is None:
        best_c = np.random.rand(N, 2) * 0.6 + 0.2
        best_r = np.full(N, 0.02)
        best_sum = np.sum(best_r)
        
    return best_c, best_r, float(best_sum)
