import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds_r = []
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds_r.append((0.0, max(0.0, mx)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success and np.all(res.x >= -1e-9):
            return res.x, -res.fun
    except Exception:
        pass
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='interior-point')
        if res.success and np.all(res.x >= -1e-9):
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def objective_func(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_func(x):
    """Inequality constraints: boundary and pairwise non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = []
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dists = np.hypot(dx, dy)
    c.append(dists[I_IDX, J_IDX] - (r[I_IDX] + r[J_IDX]))
    
    return np.concatenate(c)

def fix_violations(centers, radii):
    """Post-process to strictly satisfy boundary and overlap constraints."""
    n = centers.shape[0]
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        if radii[i] > mx:
            radii[i] = max(0.0, mx - 1e-9)
            
    for _ in range(100):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                s = radii[i] + radii[j]
                if d < s - 1e-9:
                    diff = s - d + 1e-9
                    radii[i] -= diff * 0.5
                    radii[j] -= diff * 0.5
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii

def generate_initializations():
    """Create diverse starting points for optimization."""
    inits = []
    
    # Hexagonal lattices with varying spacing
    for s in range(15):
        rng = np.random.RandomState(s)
        c = np.zeros((N, 2))
        idx = 0
        row = 0
        spacing = 0.17 + s * 0.003
        while idx < N:
            y = 0.05 + row * spacing * math.sqrt(3) / 2.0
            if y > 0.95: break
            x_start = 0.05 + (row % 2) * spacing / 2.0
            col = 0
            while x_start + col * spacing <= 0.95 and idx < N:
                c[idx] = [x_start + col * spacing, y]
                idx += 1
                col += 1
            row += 1
        c += rng.randn(N, 2) * 0.008
        c = np.clip(c, 0.05, 0.95)
        while len(c) < N:
            c = np.vstack([c, rng.uniform(0.1, 0.9, 2)])
        inits.append(c[:N])

    # Random initializations
    for s in range(10):
        rng = np.random.RandomState(s + 100)
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))

    # Boundary/Corner heavy initializations
    for s in range(5):
        rng = np.random.RandomState(s + 200)
        c = np.zeros((N, 2))
        c[0] = [0.1, 0.1]; c[1] = [0.9, 0.1]; c[2] = [0.1, 0.9]; c[3] = [0.9, 0.9]
        for i in range(4, N):
            u = rng.rand()
            if u < 0.25: c[i] = [rng.uniform(0.05, 0.95), rng.uniform(0.05, 0.15)]
            elif u < 0.5: c[i] = [rng.uniform(0.05, 0.95), rng.uniform(0.85, 0.95)]
            elif u < 0.75: c[i] = [rng.uniform(0.05, 0.15), rng.uniform(0.05, 0.95)]
            else: c[i] = [rng.uniform(0.85, 0.95), rng.uniform(0.05, 0.95)]
        c += rng.randn(N, 2) * 0.015
        c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    return inits

def run_packing():
    best_sum = 0.0
    best_c = None
    best_r = None
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': constraints_func}
    
    inits = generate_initializations()
    
    # Stage 1: Multiple restarts with LP-polished starts
    for base in inits:
        r_lp, s_lp = solve_lp_radii(base)
        x0 = np.zeros(3 * N)
        x0[0::3] = base[:, 0]
        x0[1::3] = base[:, 1]
        x0[2::3] = np.maximum(r_lp * 0.99, 0.001)
        
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons_dict, options={'maxiter': 10000, 'ftol': 1e-13})
            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
            r_opt, s_opt = solve_lp_radii(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            continue
            
    # Stage 2: Local perturbation refinement
    if best_c is not None:
        for trial in range(20):
            rng = np.random.RandomState(trial * 17 + 3)
            c_p = best_c + rng.randn(N, 2) * 0.004
            c_p = np.clip(c_p, 0.01, 0.99)
            r_p, _ = solve_lp_radii(c_p)
            x0 = np.zeros(3 * N)
            x0[0::3] = c_p[:, 0]
            x0[1::3] = c_p[:, 1]
            x0[2::3] = np.maximum(r_p * 0.99, 0.001)
            try:
                res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_opt,
                               constraints=cons_dict, options={'maxiter': 6000, 'ftol': 1e-13})
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt, s_opt = solve_lp_radii(c_opt)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
            except Exception:
                continue
                
    # Fallback safety net
    if best_c is None:
        best_c = np.random.rand(N, 2) * 0.6 + 0.2
        best_r = np.full(N, 0.02)
        
    # Strict validity enforcement
    best_c, best_r = fix_violations(best_c, best_r)
    return best_c, best_r, float(np.sum(best_r))
