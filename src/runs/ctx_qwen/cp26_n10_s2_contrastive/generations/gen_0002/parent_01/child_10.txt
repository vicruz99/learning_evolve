import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

# Bounds: x,y in [0,1], r in [1e-7, 0.5]
BOUNDS = [(0.0, 1.0)] * (2 * N) + [(1e-7, 0.5)] * N

def objective(params):
    """Negative sum of radii to be minimized."""
    return -np.sum(params[2 * N:])

def constraints(params):
    """Boundary and non-overlap constraints (must be >= 0)."""
    c = params[:2 * N].reshape(N, 2)
    r = params[2 * N:]
    
    # Boundary constraints
    b = np.empty(4 * N)
    b[0:N] = c[:, 0] - r
    b[N:2*N] = 1.0 - c[:, 0] - r
    b[2*N:3*N] = c[:, 1] - r
    b[3*N:4*N] = 1.0 - c[:, 1] - r
    
    # Pairwise distance constraints
    diff = c[:, None, :] - c[None, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    overlaps = dists[I_IDX, J_IDX] - (r[I_IDX] + r[J_IDX])
    
    return np.concatenate([b, overlaps])

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    A_ub = np.zeros((N_PAIRS, n))
    b_ub = np.zeros(N_PAIRS)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = math.hypot(dx, dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    # Variable bounds: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, max_r)))
        
    # Try HiGHS first, fallback to interior-point
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return res.x, -res.fun
        except Exception:
            continue
            
    return np.zeros(n), 0.0

def fix_violations(centers, radii):
    """Post-process to strictly satisfy boundary and overlap constraints."""
    n = centers.shape[0]
    # Fix boundaries
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        if radii[i] > mx:
            radii[i] = max(0.0, mx - 1e-9)
            
    # Fix overlaps iteratively
    for _ in range(100):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                s = radii[i] + radii[j]
                if d < s - 1e-10:
                    diff = s - d
                    radii[i] -= diff * 0.5
                    radii[j] -= diff * 0.5
                    changed = True
        if not changed:
            break
    return centers, radii

def generate_initializations():
    """Create diverse starting points for optimization."""
    inits = []
    
    # 1. Hexagonal lattices with varying spacing and offsets
    for seed in range(15):
        rng = np.random.RandomState(seed)
        c = np.zeros((N, 2))
        r = np.full(N, 0.085)
        idx = 0
        row = 0
        spacing = 0.16 + rng.uniform(-0.02, 0.02)
        margin = 0.05
        while idx < N:
            y = margin + row * spacing * math.sqrt(3) / 2
            if y > 0.95: break
            x_off = margin + (row % 2) * spacing / 2
            col = 0
            while x_off + col * spacing <= 0.95 and idx < N:
                c[idx] = [x_off + col * spacing, y]
                idx += 1
                col += 1
            row += 1
        if idx < N:
            while idx < N:
                c[idx] = rng.uniform(0.1, 0.9, 2)
                idx += 1
        c += rng.randn(N, 2) * 0.005
        c = np.clip(c, 0.02, 0.98)
        inits.append(np.concatenate([c.ravel(), r]))
        
    # 2. Random uniform placements
    for seed in range(20):
        rng = np.random.RandomState(seed + 100)
        c = rng.uniform(0.1, 0.9, (N, 2))
        r = np.full(N, 0.08)
        inits.append(np.concatenate([c.ravel(), r]))
        
    # 3. Grid patterns
    for seed in range(10):
        rng = np.random.RandomState(seed + 200)
        c = np.zeros((N, 2))
        r = np.full(N, 0.085)
        idx = 0
        step = 0.19
        for i in range(7):
            for j in range(7):
                if idx < N:
                    c[idx] = [0.05 + j * step, 0.05 + i * step]
                    idx += 1
        c += rng.randn(N, 2) * 0.01
        c = np.clip(c, 0.05, 0.95)
        inits.append(np.concatenate([c.ravel(), r]))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0.0
    best_c = None
    best_r = None
    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    inits = generate_initializations()
    
    # Stage 1: Optimization from diverse starts
    for p0 in inits:
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=BOUNDS, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                c_opt = res.x[:2 * N].reshape(N, 2)
                r_lp, s_lp = solve_radii_lp(c_opt)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_c = c_opt.copy()
                    best_r = r_lp.copy()
        except Exception:
            continue

    # Stage 2: Local refinement around best
    if best_c is not None:
        for trial in range(30):
            rng = np.random.RandomState(trial * 31 + 7)
            c_pert = best_c + rng.randn(N, 2) * 0.004
            c_pert = np.clip(c_pert, 0.01, 0.99)
            r_pert = best_r * 0.98
            p_pert = np.concatenate([c_pert.ravel(), r_pert])
            
            try:
                res_p = minimize(objective, p_pert, method='SLSQP', bounds=BOUNDS, constraints=cons_dict,
                                 options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                if res_p.success:
                    c_opt = res_p.x[:2 * N].reshape(N, 2)
                    r_lp, s_lp = solve_radii_lp(c_opt)
                    if s_lp > best_sum:
                        best_sum = s_lp
                        best_c = c_opt.copy()
                        best_r = r_lp.copy()
            except Exception:
                continue

    # Stage 3: Basin hopping / iterative refinement
    if best_c is not None:
        current_c = best_c.copy()
        current_r = best_r.copy()
        current_sum = best_sum
        
        for step in range(50):
            rng = np.random.RandomState(step * 13 + 1)
            noise_scale = 0.003 * (0.85 ** (step // 10))
            c_hop = current_c + rng.randn(N, 2) * noise_scale
            c_hop = np.clip(c_hop, 0.02, 0.98)
            r_hop = current_r * 0.95
            p_hop = np.concatenate([c_hop.ravel(), r_hop])
            
            try:
                res_hop = minimize(objective, p_hop, method='SLSQP', bounds=BOUNDS, constraints=cons_dict,
                                   options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
                if res_hop.success:
                    c_opt = res_hop.x[:2 * N].reshape(N, 2)
                    r_lp, s_lp = solve_radii_lp(c_opt)
                    if s_lp > current_sum:
                        current_sum = s_lp
                        current_c = c_opt.copy()
                        current_r = r_lp.copy()
                        best_sum = current_sum
                        best_c = current_c.copy()
                        best_r = current_r.copy()
            except Exception:
                continue

    # Fallback safety net
    if best_c is None:
        best_c = np.random.uniform(0.2, 0.8, (N, 2))
        best_r = np.full(N, 0.02)
        best_sum = np.sum(best_r)
        
    # Final validation and strict fixing
    best_c, best_r = fix_violations(best_c, best_r)
    best_sum = float(np.sum(best_r))
    
    return best_c, best_r, best_sum
