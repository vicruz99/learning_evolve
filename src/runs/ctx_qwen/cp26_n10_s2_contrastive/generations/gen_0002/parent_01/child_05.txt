from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)
BOUNDS = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

def objective_func(x):
    """Negative sum of radii to be minimized."""
    return -np.sum(x[2 * N:])

def constraints_func(x):
    """Vectorized boundary and non-overlap constraints (must be >= 0)."""
    c = x[:2 * N].reshape(N, 2)
    r = x[2 * N:]
    
    cons = np.empty(4 * N + N_PAIRS)
    cons[:N] = c[:, 0] - r
    cons[N:2 * N] = 1.0 - c[:, 0] - r
    cons[2 * N:3 * N] = c[:, 1] - r
    cons[3 * N:4 * N] = 1.0 - c[:, 1] - r
    
    dx = c[:, 0][:, None] - c[:, 0][None, :]
    dy = c[:, 1][:, None] - c[:, 1][None, :]
    dists = np.hypot(dx, dy)
    r_sum = r[:, None] + r[None, :]
    
    cons[4 * N:] = dists[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]
    return cons

def get_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    A_ub = np.zeros((N_PAIRS, n))
    b_ub = np.zeros(N_PAIRS)
    for k, (i, j) in enumerate(zip(I_IDX, J_IDX)):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

                x += spacing

    pts = np.clip(pts, 0.05, 0.95)

    return np.concatenate([pts.ravel(), r_init])
