import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

def solve_radii_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii subject to
    pairwise non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    A_ub = np.zeros((N_PAIRS, n))
    b_ub = np.zeros(N_PAIRS)
    idx = 0
    for i, j in zip(I_IDX, J_IDX):
        d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        b_ub[idx] = d
        idx += 1
        
    # Boundary constraints handled via variable bounds: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, mx)))
        
    # Try HiGHS first (fast and robust), fallback to interior-point
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success:
                return res.x, -res.fun
        except Exception:
            continue
            
    return np.zeros(n), 0.0

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[2 * N:])

def constraints(params):
    """
    Inequality constraints g(params) >= 0:
    - Boundary: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    - Non-overlap (squared): dist^2 - (r_i + r_j)^2 >= 0
    """
    c = params[:2 * N].reshape(N, 2)
    r = params[2 * N:]
    
    # Boundary constraints
    b = np.empty(4 * N)
    b[0:N] = c[:, 0] - r
    b[N:2*N] = 1.0 - c[:, 0] - r
    b[2*N:3*N] = c[:, 1] - r
    b[3*N:4*N] = 1.0 - c[:, 1] - r
    
    # Pairwise squared distance constraints
    dx = c[:, 0][:, None] - c[:, 0][None, :]
    dy = c[:, 1][:, None] - c[:, 1][None, :]
    dist_sq = dx**2 + dy**2
    r_sum_sq = (r[:, None] + r[None, :])**2
    
    c_pair = dist_sq[I_IDX, J_IDX] - r_sum_sq[I_IDX, J_IDX]
    
    return np.concatenate([b, c_pair])
