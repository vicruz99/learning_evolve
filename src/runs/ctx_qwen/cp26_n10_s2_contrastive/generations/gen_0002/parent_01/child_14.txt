import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIRS = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIRS)
IDX_I = [p[0] for p in PAIRS]
IDX_J = [p[1] for p in PAIRS]

def fix_violations(centers, radii):
    """Post-process to strictly satisfy boundary and overlap constraints."""
    n = centers.shape[0]
    # Enforce boundaries strictly
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        if radii[i] > mx:
            radii[i] = max(0.0, mx - 1e-9)

    for _ in range(100):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                s = radii[i] + radii[j]
                if d < s - 1e-10:
                    diff = s - d + 1e-9
                    radii[i] -= diff * 0.5
                    radii[j] -= diff * 0.5
                    changed = True
        if not changed:
            break
    return centers, radii

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((N_PAIRS, n))
    b_ub = np.zeros(N_PAIRS)
    
    # Vectorized distance calculation
    dx = centers[:, np.newaxis, 0] - centers[np.newaxis, :, 0]
    dy = centers[:, np.newaxis, 1] - centers[np.newaxis, :, 1]
    dists = np.hypot(dx, dy)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, max_r)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def lp_objective(x):
    """Objective for center optimization: negative sum of optimal radii."""
    centers = x.reshape((N, 2))
    _, s = solve_lp_radii(centers)
    return -s

def repulse_and_optimize(centers):

    c = centers.copy()
    r = np.zeros(N)

            for j in range(i + 1, N):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = np.hypot(dx, dy)
                if d < r[i] + r[j] + 1e-7:
                    push = (r[i] + r[j] - d) * 0.5
                    if d > 1e-9:
                        fx = dx / d * push
                        fy = dy / d * push
                        force[i] += [fx, fy]
                        force[j] -= [fx, fy]
                    max_overlap = max(max_overlap, push)
        # Boundary repulsion
        for i in range(N):
            if c[i, 0] < r[i]: force[i, 0] += (r[i] - c[i, 0]) * 2.0

        c += force * 0.05
        c = np.clip(c, 0.0, 1.0)
        if max_overlap < 1e-5:
            break
    return c, r, np.sum(r)
