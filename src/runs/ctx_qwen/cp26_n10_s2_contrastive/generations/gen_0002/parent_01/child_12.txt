import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIRS = [(i, j) for i in range(N) for j in range(i + 1, N)]
NUM_PAIRS = len(PAIRS)

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Boundary constraints as variable bounds: 0 <= r_i <= min(dist to edges)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        if ub < 0.0:
            ub = 0.0
        bounds.append((0.0, ub))
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    A_ub = np.zeros((NUM_PAIRS, n))
    b_ub = np.zeros(NUM_PAIRS)
    
    idx = 0
    for i, j in PAIRS:
        d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        b_ub[idx] = d
        idx += 1
        
    # Try HiGHS first (fast), fallback to interior-point
    res = None
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-7):
                return res.x, -res.fun
        except Exception:
            continue
            
    # Fallback
    return np.zeros(n), 0.0

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2 * N:])

def constraints(x):
    """Boundary and non-overlap constraints (must be >= 0)."""
    c = x[:2 * N].reshape(N, 2)
    r = x[2 * N:]
    
    vals = np.empty(4 * N + NUM_PAIRS)
    vals[:N] = c[:, 0] - r
    vals[N:2 * N] = 1.0 - c[:, 0] - r
    vals[2 * N:3 * N] = c[:, 1] - r
    vals[3 * N:4 * N] = 1.0 - c[:, 1] - r
    
    dx = c[:, 0][:, None] - c[:, 0][None, :]
    dy = c[:, 1][:, None] - c[:, 1][None, :]
    dists = np.hypot(dx, dy)
    r_sum = r[:, None] + r[None, :]
    
    triu = np.triu_indices(N, k=1)
    vals[4 * N:] = dists[triu] - r_sum[triu]
    return vals

def run_packing():
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # 1. Generate diverse initial configurations
    inits = []

        row = 0
        while len(pts) < N:
            x = r0 + 0.01 + (row % 2) * r0
            while x < 1.0 - r0 - 0.01 and len(pts) < N:
                pts.append([x + rng.uniform(-0.01, 0.01), y + rng.uniform(-0.01, 0.01)])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
            row += 1
        pts = np.clip(pts, 0.05, 0.95)
        while len(pts) < N:
            pts.append(rng.uniform(0.2, 0.8, 2))
        inits.append(np.array(pts[:N]))
        
    # Corner-focused + inner grid
    for seed in range(8):
        rng = np.random.RandomState(seed)
        pts = []
        pts.extend([[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]])
        for i in range(1, 5):

                    pts.append([0.2 + i * 0.15, 0.2 + j * 0.15])
        pts = np.array(pts[:N]) + rng.normal(0, 0.015, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # Pure random
    for seed in range(5):
        rng = np.random.RandomState(seed)
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))

            r_final, s_final = solve_radii_lp(c_final)
            if s_final > best_sum:
                best_sum = s_final
                best_c = c_final
                best_r = r_final
        except Exception:
            pass

    # 5. Strict post-processing to guarantee validity
    centers = best_c.copy()
    radii = best_r.copy()
    
    # Enforce boundary constraints
    for i in range(N):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if radii[i] > max_r + 1e-9:
            radii[i] = max(0.0, max_r - 1e-9)

    for _ in range(50):
        changed = False
        for i in range(N):

                s = radii[i] + radii[j]
                if d < s - 1e-10:
                    excess = s - d + 1e-9
                    radii[i] -= excess * 0.5
                    radii[j] -= excess * 0.5
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    best_sum = np.sum(radii)
    
    return centers, radii, float(best_sum)
