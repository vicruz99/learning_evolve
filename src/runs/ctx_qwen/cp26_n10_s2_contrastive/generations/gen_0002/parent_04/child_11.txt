import numpy as np
from scipy.optimize import minimize
import math

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

def objective(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def constraints(x):
    """
    Inequality constraints:
    - Boundary: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    - Non-overlap: dist(i,j) >= r[i] + r[j]
    Returns array of constraint values (must be >= 0)
    """
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dists = np.hypot(dx, dy)
    c[4*N:] = dists - r[I] - r[J]
    return c

def make_hex_init(seed, density=1.0, noise=0.01):
    """Generate a hexagonal lattice initialization with controlled perturbation."""
    rng = np.random.RandomState(seed)
    s = 0.175 * density
    pts = []
    row = 0
    while len(pts) < N:
        y = 0.04 + row * s * np.sqrt(3)/2
        if y > 0.96: break
        x_start = 0.04 + (row % 2) * s/2
        col = 0
        while x_start + col * s <= 0.96 and len(pts) < N:
            pts.append([x_start + col * s, y])
            col += 1
        row += 1
    pts = np.array(pts[:N])
    if len(pts) < N:
        pts = np.vstack([pts, rng.uniform(0.1, 0.9, (N-len(pts), 2))])
    pts += rng.normal(0, noise, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    r = np.full(N, 0.045 * density)
    return np.concatenate([pts.flatten(), r])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_x = None
    rng = np.random.RandomState(42)
    
    # Stage 1: Broad search with diverse lattice configurations
    for seed in range(90):
        # Vary density and noise systematically
        d = 0.70 + (seed % 25) * 0.025
        n_val = 0.005 + (seed % 12) * 0.004
        x0 = make_hex_init(seed, density=d, noise=n_val)
        
        try:
            res = minimize(
                objective, x0, method='SLSQP', bounds=bounds, 
                constraints=cons, 
                options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False}
            )
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_x = res.x.copy()
        except Exception:
            continue
            
    # Stage 2: Local refinement around the best solution
    if best_x is not None:
        for _ in range(40):
            x_pert = best_x + rng.normal(0, 0.0015, len(best_x))
            x_pert = np.clip(x_pert, [b[0] for b in bounds], [b[1] for b in bounds])
            try:
                res = minimize(
                    objective, x_pert, method='SLSQP', bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False}
                )
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
            except Exception:
                pass
                
    # Fallback initialization if optimization failed entirely
    if best_x is None:
        x0 = make_hex_init(0, 1.0, 0.01)
        best_x = x0
        
    # Extract results
    cx = best_x[:N].copy()
    cy = best_x[N:2*N].copy()
    r = best_x[2*N:].copy()
    
    # Post-processing: enforce strict boundary constraints
    for i in range(N):
        mx = min(cx[i], 1.0-cx[i], cy[i], 1.0-cy[i])
        if r[i] > mx:
            r[i] = max(0.0, mx - 1e-9)
            
    # Iteratively resolve numerical overlaps
    for _ in range(200):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = math.hypot(cx[i]-cx[j], cy[i]-cy[j])
                if d < r[i] + r[j] - 1e-10:
                    exc = r[i] + r[j] - d
                    r[i] -= exc / 2.0
                    r[j] -= exc / 2.0
                    changed = True
        if not changed:
            break
            
    r = np.maximum(r, 0.0)
    centers = np.column_stack([cx, cy])
    return centers, r, float(np.sum(r))
