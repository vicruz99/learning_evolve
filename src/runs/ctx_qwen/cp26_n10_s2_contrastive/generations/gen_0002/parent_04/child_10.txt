import numpy as np
from scipy.optimize import minimize

N = 26
I, J = np.triu_indices(N, k=1)

def objective(x):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2::3])

def constraints(x):
    """
    Inequality constraints: g(x) >= 0
    - Boundary: circles inside [0,1]^2
    - Non-overlap: squared distance >= squared sum of radii
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]

    n_cons = 4 * N + len(I)
    c = np.empty(n_cons)

    c[k:k+N] = 1.0 - cx - r; k += N
    c[k:k+N] = cy - r; k += N
    c[k:k+N] = 1.0 - cy - r; k += N
    
    # Pairwise non-overlap constraints (squared formulation for stability)
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c[k:] = dx**2 + dy**2 - (r[I] + r[J])**2
    
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

                col += 1
