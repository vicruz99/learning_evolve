2. **LP Radius Refinement**: After SLSQP finds a promising configuration of centers, we fix those centers and solve a Linear Program (LP) to find the exact optimal radii. import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def constraints(x):
    """
    Inequality constraints g(x) >= 0:
    - Boundary: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    - Non-overlap: squared_dist >= (r_i + r_j)^2
    """
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    # Preallocate constraint array
    n_cons = 4 * N + N * (N - 1) // 2
    c = np.empty(n_cons)
    
    # Boundary constraints
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    # Pairwise squared distance constraints
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    dist_sq = dx**2 + dy**2
    r_sum_sq = (r[I_IDX] + r[J_IDX])**2
    c[4*N:] = dist_sq - r_sum_sq
    
    return c

def solve_radii_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum(radii)
    subject to r_i + r_j <= dist(i,j) and boundary constraints.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Compute all pairwise distances
    dists = np.sqrt(((centers[:, None, :] - centers[None, :, :])**2).sum(axis=2))
    
    # Pairwise constraints matrix: r_i + r_j <= d_ij
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    # Bounds for each radius: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds_r = []
    for i in range(n):
        mx, my = centers[i]
        ub = min(mx, 1.0 - mx, my, 1.0 - my)
        bounds_r.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def create_hex_init(seed, density=0.8, pert=0.012):
    """Generate a hexagonal lattice initialization with controlled perturbation."""
    rng = np.random.RandomState(seed)
    s = 0.18 + 0.06 * density  # lattice spacing
    points = []
    row = 0
    while len(points) < N:
        y = 0.04 + row * s * np.sqrt(3)/2
        if y > 0.96: break
        x_start = 0.04 + (row % 2) * s/2
        col = 0
        while x_start + col * s <= 0.96 and len(points) < N:
            points.append([x_start + col * s, y])
            col += 1
        row += 1
        
    points = np.array(points[:N])
    # Fill remaining if hex pattern yields fewer than N
    if len(points) < N:
        while len(points) < N:
            points = np.vstack([points, rng.uniform(0.15, 0.85, 2)])
            
    # Add perturbation
    points += rng.randn(*points.shape) * pert
    points = np.clip(points, 0.02, 0.98)
    
    # Initial radii small enough to guarantee feasibility
    r_init = np.full(N, 0.04 * density)
    return np.concatenate([points.flatten(), r_init])

def run_packing():
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Broad search with diverse hexagonal initializations
    seeds = list(range(60))
    densities = [0.7, 0.8, 0.9, 1.0, 1.1, 1.2]
    
    for seed in seeds:
        for d in densities:
            x0 = create_hex_init(seed, density=d, pert=0.010)
            
            try:
                res = minimize(
                    objective,
                    x0,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False}
                )
                
                curr_centers = res.x[:2*N].reshape(N, 2)
                # Refine radii with LP for exact optimal radii given these centers
                opt_r, opt_s = solve_radii_lp(curr_centers)
                
                if opt_s > best_sum:
                    best_sum = opt_s
                    best_centers = curr_centers.copy()
                    best_radii = opt_r.copy()
            except Exception:
                continue
                
    # Phase 2: Local refinement & perturbation search
    if best_centers is not None:
        rng = np.random.RandomState(42)
        for _ in range(20):
            # Perturb centers
            c_pert = best_centers + rng.randn(N, 2) * 0.004
            c_pert = np.clip(c_pert, 0.01, 0.99)
            
            # Solve LP for perturbed centers
            r_pert, s_pert = solve_radii_lp(c_pert)
            
            if s_pert > best_sum:
                # If LP found better radii, run SLSQP to further optimize centers
                x0_pert = np.concatenate([c_pert.flatten(), r_pert * 0.99])
                try:
                    res2 = minimize(
                        objective, x0_pert, method='SLSQP', bounds=bounds, constraints=cons,
                        options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False}
                    )
                    c_opt = res2.x[:2*N].reshape(N, 2)
                    r_opt, s_opt = solve_radii_lp(c_opt)
                    
                    if s_opt > best_sum:
                        best_sum = s_opt
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
                except Exception:
                    continue
            else:
                # Just accept the perturbation if it improved via LP alone
                best_centers = c_pert.copy()
                best_radii = r_pert.copy()
                best_sum = s_pert

    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = create_hex_init(0, 0.9)[:2*N].reshape(N, 2)
        best_radii = np.full(N, 0.05)
        best_sum = np.sum(best_radii)
        
    # Post-processing: guarantee strict validity against numerical errors
    # Enforce boundary constraints
    for i in range(N):
        mx, my = best_centers[i]
        max_r = min(mx, 1.0 - mx, my, 1.0 - my)
        if best_radii[i] > max_r:
            best_radii[i] = max(0.0, max_r - 1e-9)
            
    # Iteratively resolve any microscopic overlaps
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0],
                             best_centers[i, 1] - best_centers[j, 1])
                if d < best_radii[i] + best_radii[j] - 1e-11:
                    excess = best_radii[i] + best_radii[j] - d
                    best_radii[i] -= excess * 0.5
                    best_radii[j] -= excess * 0.5
                    changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(final_sum)
