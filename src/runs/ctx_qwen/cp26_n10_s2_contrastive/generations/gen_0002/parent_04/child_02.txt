import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution
import math

N = 26
I_PAIRS, J_PAIRS = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_PAIRS)

# Precompute sparse-like structure for pairwise constraints to speed up LP
A_Ub_pairs = np.zeros((NUM_PAIRS, N))
A_Ub_pairs[np.arange(NUM_PAIRS), I_PAIRS] = 1.0
A_Ub_pairs[np.arange(NUM_PAIRS), J_PAIRS] = 1.0

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    dists = np.sqrt(((centers[:, None] - centers[None, :])**2).sum(axis=2))
    b_ub = dists[I_PAIRS, J_PAIRS]
    
    bounds_r = []
    for i in range(N):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, ub - 1e-9)))
        
    try:
        res = linprog(-np.ones(N), A_ub=A_Ub_pairs, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x
        else:
            return np.full(N, 1e-5)
    except Exception:
        return np.full(N, 1e-5)

def obj_de(centers_flat):
    """Objective for Differential Evolution: minimize negative sum of optimal radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    radii = solve_radii_lp(centers)
    return -np.sum(radii)

def hex_init(seed, spacing=0.19):
    """Generates a perturbed hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    row = 0
    y = 0.05
    while len(pts) < N:
        x = 0.05 + (row % 2) * spacing / 2.0
        while x < 0.95 and len(pts) < N:
            pts.append([x + rng.uniform(-0.01, 0.01), y + rng.uniform(-0.01, 0.01)])
            x += spacing
        y += spacing * math.sqrt(3) / 2.0
        row += 1
    return np.array(pts[:N])

def run_packing():

    best_sum = 0.0
    best_centers = None
    best_radii = None

    try:

                                        seed=42, polish=True, workers=1, tol=1e-8)

        curr_radii = solve_radii_lp(curr_centers)
        curr_sum = np.sum(curr_radii)
        best_sum = curr_sum
        best_centers = curr_centers
        best_radii = curr_radii
    except Exception:
        pass
        
    # Strategy 2: Joint SLSQP optimization from structured seeds
    def obj_joint(x):
        return -np.sum(x[2*N:])
        
    def cons_joint(x):
        cx = x[:N]; cy = x[N:2*N]; r = x[2*N:]
        c = np.empty(4*N + NUM_PAIRS)
        c[:N] = cx - r
        c[N:2*N] = 1.0 - cx - r
        c[2*N:3*N] = cy - r
        c[3*N:4*N] = 1.0 - cy - r
        
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        dists = np.hypot(dx, dy)
        c[4*N:] = dists[I_PAIRS, J_PAIRS] - r[I_PAIRS] - r[J_PAIRS]
        return c
        
    bounds_j = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
    cons_dict = {'type': 'ineq', 'fun': cons_joint}

        r0 = solve_radii_lp(c0) * 0.90
        x0 = np.concatenate([c0.ravel(), r0])

            res = minimize(obj_joint, x0, method='SLSQP', bounds=bounds_j,

    return best_centers, best_radii, float(np.sum(best_radii))
