import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
N_PAIRS = len(I)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def constraints(x):
    """Inequality constraints: boundary and non-overlap (squared form for smoothness)."""
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    c = np.empty(4*N + N_PAIRS)
    # Boundary constraints
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dist_sq = dx*dx + dy*dy
    r_sum = r[I] + r[J]
    c[4*N:] = dist_sq - r_sum*r_sum
    return c

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Vectorized pairwise distances
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.hypot(dx, dy)
    
    # Build LP constraints: r_i + r_j <= dist(i, j)
    idx = np.triu_indices(n, k=1)
    A_ub = np.zeros((n*(n-1)//2, n))
    A_ub[np.arange(n*(n-1)//2), idx[0]] = 1.0
    A_ub[np.arange(n*(n-1)//2), idx[1]] = 1.0
    b_ub = dists[idx]
    
    # Boundary constraints as variable bounds: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0-x, y, 1.0-y)
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def make_hex_init(seed, pattern_type=0):
    """Generate a hexagonal lattice initialization with controlled perturbation."""
    rng = np.random.RandomState(seed)
    cx = np.zeros(N)
    cy = np.zeros(N)
    
    if pattern_type == 0:
        rows = [5, 6, 5, 6, 4]
    elif pattern_type == 1:
        rows = [6, 5, 6, 5, 4]
    elif pattern_type == 2:
        rows = [5, 5, 6, 5, 5]
    else:
        rows = [6, 6, 6, 6, 2]
        
    y_step = 0.85 / len(rows)
    idx = 0
    for r_idx, ncols in enumerate(rows):
        y = 0.1 + r_idx * y_step
        x_step = 0.8 / ncols
        x_start = 0.1 if r_idx % 2 == 0 else 0.1 + x_step/2
        for c_idx in range(ncols):
            if idx < N:
                cx[idx] = x_start + c_idx * x_step + rng.uniform(-0.02, 0.02)
                cy[idx] = y + rng.uniform(-0.02, 0.02)
                idx += 1
    while idx < N:
        cx[idx] = rng.uniform(0.2, 0.8)
        cy[idx] = rng.uniform(0.2, 0.8)
        idx += 1
        
    cx = np.clip(cx, 0.05, 0.95)
    cy = np.clip(cy, 0.05, 0.95)
    r = np.full(N, 0.05)
    return np.concatenate([cx, cy, r])

def run_packing():
    """Optimizes circle packing in a unit square to maximize sum of radii."""
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    
    # Phase 1: SLSQP from diverse hexagonal starts
    for seed in range(30):
        for p_type in range(3):
            x0 = make_hex_init(seed, p_type)
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
                cx = res.x[:N]
                cy = res.x[N:2*N]
                centers_slsqp = np.column_stack((cx, cy))
                
                # Polish radii exactly using LP
                r_lp, s_lp = solve_lp_radii(centers_slsqp)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = centers_slsqp.copy()
            except Exception:
                continue
                
    # Phase 2: Hill climbing on centers using LP as exact evaluator
    if best_centers is not None:
        rng = np.random.RandomState(42)
        centers_curr = best_centers.copy()
        sum_curr = best_sum
        step = 0.02
        
        for _ in range(2500):
            # Perturb centers
            centers_pert = centers_curr + rng.randn(N, 2) * step
            centers_pert = np.clip(centers_pert, 0.05, 0.95)
            
            s_pert, _ = solve_lp_radii(centers_pert)
            if s_pert > sum_curr:
                sum_curr = s_pert
                centers_curr = centers_pert.copy()
                best_sum = sum_curr
                best_centers = centers_curr.copy()
                step *= 0.98
            else:
                step = min(step * 1.005, 0.05)
                
    # Fallback safety net
    if best_centers is None:
        x0 = make_hex_init(0, 0)
        best_centers = np.column_stack((x0[:N], x0[N:2*N]))
        
    # Final LP polish on the best found centers
    best_radii, _ = solve_lp_radii(best_centers)
    
    # Strict validity enforcement
    cx = best_centers[:, 0]
    cy = best_centers[:, 1]
    r = best_radii.copy()
    
    for i in range(N):
        max_r = min(cx[i], 1.0-cx[i], cy[i], 1.0-cy[i])
        if r[i] > max_r:
            r[i] = max_r
            
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(cx[i]-cx[j], cy[i]-cy[j])
                if d < r[i] + r[j] - 1e-10:
                    excess = r[i] + r[j] - d
                    r[i] -= excess/2
                    r[j] -= excess/2
                    changed = True
        if not changed: break
        
    r = np.maximum(r, 0.0)
    return best_centers, r, float(np.sum(r))
