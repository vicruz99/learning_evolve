import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)
BOUNDS = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def constraints(x):
    """Inequality constraints: boundary and non-overlap (squared for smooth gradients)."""
    cx, cy, r = x[:N], x[N:2*N], x[2*N:]
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c[4*N:] = dx**2 + dy**2 - (r[I] + r[J])**2
    return c

def hex_init(seed, spacing=0.18, margin=0.05):
    """Generate a perturbed hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    cx, cy = np.zeros(N), np.zeros(N)
    idx = 0
    row = 0
    y = margin
    while idx < N and y < 1.0 - margin:
        x_start = margin + (row % 2) * spacing / 2.0
        col = 0
        while x_start + col * spacing < 1.0 - margin and idx < N:
            cx[idx] = x_start + col * spacing
            cy[idx] = y
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2.0
        row += 1
    while idx < N:
        cx[idx] = rng.uniform(0.1, 0.9)
        cy[idx] = rng.uniform(0.1, 0.9)
        idx += 1
        
    cx += rng.randn(N) * 0.01
    cy += rng.randn(N) * 0.01
    cx = np.clip(cx, 0.05, 0.95)
    cy = np.clip(cy, 0.05, 0.95)
    r = np.full(N, 0.085)
    return np.concatenate([cx, cy, r])

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii under non-overlap/boundary constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    
    dists = np.sqrt(((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2).sum(axis=2))
    
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds_r = []
    for i in range(n):
        mx, my = centers[i]
        ub = min(mx, 1.0-mx, my, 1.0-my)
        bounds_r.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def run_packing():
    cons = {'type': 'ineq', 'fun': constraints}
    best_x = None
    best_sum = -1.0
    
    # Phase 1: Systematic hexagonal starts
    for seed in range(60):
        s = 0.16 + (seed % 5) * 0.015
        m = 0.04 + (seed // 5) * 0.005
        x0 = hex_init(seed, spacing=s, margin=m)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=BOUNDS, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Random starts
    for seed in range(20):
        rng = np.random.RandomState(seed)
        x0 = np.zeros(3*N)
        x0[:N] = rng.uniform(0.15, 0.85, N)
        x0[N:2*N] = rng.uniform(0.15, 0.85, N)
        x0[2*N:] = 0.08
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=BOUNDS, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_x = res.x.copy()
        except Exception:
            pass

    # Phase 3: Local perturbation of best found solution
    if best_x is not None:
        rng = np.random.RandomState(0)
        for _ in range(15):
            x_pert = best_x + rng.randn(3*N) * 0.005
            x_pert = np.clip(x_pert, [b[0] for b in BOUNDS], [b[1] for b in BOUNDS])
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=BOUNDS, constraints=cons,
                               options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
            except Exception:
                pass
                
    # Fallback
    if best_x is None:
        best_x = hex_init(0)
        
    cx = best_x[:N].copy()
    cy = best_x[N:2*N].copy()
    r = best_x[2*N:].copy()
    
    centers = np.column_stack([cx, cy])
    
    # LP refinement: optimizes radii for the fixed optimal centers
    opt_r, opt_sum = solve_radii_lp(centers)
    if opt_sum > np.sum(r):
        r = opt_r
        
    # Strict boundary enforcement
    for i in range(N):
        max_r = min(cx[i], 1.0-cx[i], cy[i], 1.0-cy[i])
        if r[i] > max_r:
            r[i] = max_r
            
    # Iterative overlap resolution
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(cx[i]-cx[j], cy[i]-cy[j])
                if d < r[i] + r[j] - 1e-10:
                    excess = r[i] + r[j] - d
                    r[i] -= excess / 2
                    r[j] -= excess / 2
                    changed = True
        if not changed:
            break
            
    r = np.maximum(r, 0.0)
    return centers, r, float(np.sum(r))
