import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to find radii maximizing sum(r) subject to packing constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            idx += 1
            
    # Boundary constraints handled via variable bounds: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, mx)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        # Fallback to interior-point if highs unavailable
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def constraints_full(x):
    """Inequality constraints g(x) >= 0 for SLSQP."""
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    num_constraints = 4 * N + N * (N - 1) // 2
    c = np.empty(num_constraints)
    
    # Boundary constraints
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    # Pairwise non-overlap
    k = 4 * N
    for i in range(N):
        for j in range(i + 1, N):
            c[k] = math.hypot(cx[i] - cx[j], cy[i] - cy[j]) - r[i] - r[j]
            k += 1
    return c

def objective_full(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def generate_initial_centers():
    """Generate diverse structural starting points for circle centers."""
    starts = []
    rng = np.random.RandomState(42)
    
    # 1. Dense Hexagonal Lattice
    pts = []
    y = 0.07
    row = 0
    while len(pts) < N:
        x = 0.07 + (row % 2) * 0.09
        while x <= 0.93 and len(pts) < N:
            pts.append([x, y])
            x += 0.18
        y += 0.156
        row += 1
    starts.append(np.array(pts[:N]))
    
    # 2. 5x5 Grid + 1 center
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + j * 0.2, 0.1 + i * 0.2])
    pts.append([0.5, 0.5])
    starts.append(np.array(pts[:N]))
    
    # 3. Corner-Biased Layout
    pts = []
    for cx in [0.08, 0.92]:
        for cy in [0.08, 0.92]:
            pts.append([cx, cy])
    for i in range(18):
        pts.append([0.15 + (i // 6) * 0.27, 0.15 + (i % 6) * 0.15])
    starts.append(np.array(pts[:N]))
    
    # 4. Random Perturbation of Hex
    starts.append(starts[0] + rng.randn(N, 2) * 0.025)
    
    # 5. Pure Random (safe region)
    starts.append(rng.uniform(0.12, 0.88, (N, 2)))
    
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0.0
    best_c = None
    best_r = None
    
    bounds_full = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_full}
    
    init_centers_list = generate_initial_centers()
    
    # Phase 1: Optimize from structural starts
    for idx, c_init in enumerate(init_centers_list):
        # Get tight initial radii via LP
        r_lp, _ = solve_radii_lp(c_init)
        x0 = np.concatenate([c_init.ravel(), r_lp])
        
        try:
            res = minimize(objective_full, x0, method='SLSQP', bounds=bounds_full,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            c_opt = res.x[:2*N].reshape(N, 2)
            r_opt = res.x[2*N:]
            s_val = np.sum(r_opt)
            
            if s_val > best_sum:
                best_sum = s_val
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            pass
            
        # Try perturbations of this layout
        rng = np.random.RandomState(idx + 100)
        for _ in range(2):
            c_p = c_init + rng.randn(N, 2) * 0.02
            c_p = np.clip(c_p, 0.02, 0.98)
            r_lp_p, _ = solve_radii_lp(c_p)
            x0_p = np.concatenate([c_p.ravel(), r_lp_p])
            try:
                res = minimize(objective_full, x0_p, method='SLSQP', bounds=bounds_full,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
                c_opt = res.x[:2*N].reshape(N, 2)
                r_opt = res.x[2*N:]
                s_val = np.sum(r_opt)
                if s_val > best_sum:
                    best_sum = s_val
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
            except Exception:
                pass

    # Phase 2: Local refinement around the best found solution
    if best_c is not None:
        rng = np.random.RandomState(99)
        for _ in range(6):
            c_ref = best_c + rng.randn(N, 2) * 0.004
            c_ref = np.clip(c_ref, 0.01, 0.99)
            r_ref, _ = solve_radii_lp(c_ref)
            x0_ref = np.concatenate([c_ref.ravel(), r_ref])
            try:
                res = minimize(objective_full, x0_ref, method='SLSQP', bounds=bounds_full,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                c_opt = res.x[:2*N].reshape(N, 2)
                r_opt = res.x[2*N:]
                s_val = np.sum(r_opt)
                if s_val > best_sum:
                    best_sum = s_val
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
            except Exception:
                pass

    # Phase 3: Strict post-processing to guarantee validity
    centers = best_c.copy() if best_c is not None else np.zeros((N, 2))
    radii = best_r.copy() if best_r is not None else np.zeros(N)
    
    # Enforce hard boundary limits
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(radii[i], 0.0)
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc * 0.5
                    radii[j] -= exc * 0.5
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
