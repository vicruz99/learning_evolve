

<strategy>
The previous approaches optimized centers and radii simultaneously using SLSQP, which struggles with the highly non-convex and non-smooth landscape of circle packing constraints. A more robust strategy is to decouple the problem:
1. **Linear Programming for Radii**: For any fixed set of centers, the optimal radii that maximize the sum are found exactly by a Linear Program (LP). The constraints are linear: $r_i \ge 0$, $r_i \le \text{dist\_to\_boundary}$, and $r_i + r_j \le \text{dist}(C_i, C_j)$. This guarantees the best possible radii for any center configuration.
2. **Derivative-Free Optimization for Centers**: The objective becomes maximizing the LP's optimal value over the space of centers. This objective is continuous but piecewise linear/non-smooth. We use multiple restarts of `Nelder-Mead` from diverse initial configurations (hexagonal lattices, random grids) followed by a `Powell` refinement. This avoids gradient issues and efficiently explores the configuration space.
3. **Strict Post-Processing**: After optimization, we clamp radii to strictly satisfy boundary conditions and iteratively resolve any floating-point overlaps, ensuring the final packing passes the validation function flawlessly.

This hybrid LP + derivative-free optimization approach is mathematically sound for this problem class and consistently pushes past local minima that trap simultaneous optimizers.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_radii_lp(centers):
    """
    Given fixed centers, solve an LP to find radii that maximize sum(radii)
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    A_ub = np.zeros((n * (n - 1) // 2, n))
    b_ub = np.zeros(n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b_ub[idx] = np.hypot(dx, dy)
            idx += 1
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, mx)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective_centers(centers_flat):
    """
    Objective for optimizing centers: minimize negative sum of optimal radii.
    Clips centers strictly inside [0,1]^2 to prevent LP infeasibility.
    """
    centers = centers_flat.reshape((N, 2))
    centers = np.clip(centers, 0.001, 0.999)
    _, sum_r = compute_radii_lp(centers)
    return -sum_r

def run_packing():
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    best_sum = 0.0
    best_centers = None
    
    # Strategy 1: Hexagonal lattice initializations (structurally near-optimal)
    for seed in range(30):
        rng = np.random.RandomState(seed)
        centers = np.zeros((N, 2))
        idx = 0
        
        # Approximate spacing for dense packing
        r_init = 0.105
        dx = 2.0 * r_init
        dy = np.sqrt(3) * r_init
        y = r_init + 0.02
        row = 0
        while idx < N and y + r_init < 0.98:
            x_start = r_init + 0.02 + (row % 2) * (dx / 2.0)
            col = 0
            while x_start + col * dx + r_init < 0.98 and idx < N:
                centers[idx] = [x_start + col * dx, y]
                idx += 1
                col += 1
            y += dy
            row += 1
            
        while idx < N:
            centers[idx] = rng.uniform(0.15, 0.85, 2)
            idx += 1
            
        centers += rng.randn(N, 2) * 0.015
        centers = np.clip(centers, 0.05, 0.95)
        
        res = minimize(objective_centers, centers.flatten(), 
                       method='Nelder-Mead',
                       options={'maxiter': 4000, 'xatol': 1e-6, 'fatol': 1e-8})
        
        curr_sum = -res.fun
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = res.x.reshape((N, 2))
            
    # Strategy 2: Random initializations to escape structural local minima
    for seed in range(20):
        rng = np.random.RandomState(seed + 500)
        centers = rng.uniform(0.1, 0.9, (N, 2))
        
        res = minimize(objective_centers, centers.flatten(),
                       method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-6, 'fatol': 1e-8})
                       
        curr_sum = -res.fun
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = res.x.reshape((N, 2))
            
    # Strategy 3: Powell refinement on the best configuration found
    if best_centers is not None:
        res2 = minimize(objective_centers, best_centers.flatten(),
                        method='Powell',
                        options={'maxiter': 5000, 'xtol': 1e-7, 'ftol': 1e-10})
        curr_sum2 = -res2.fun
        if curr_sum2 > best_sum:
            best_sum = curr_sum2
            best_centers = res2.x.reshape((N, 2))
            
    # Final radii computation via LP
    if best_centers is None:
        best_centers = np.random.uniform(0.1, 0.9, (N, 2))
        
    radii, final_sum = compute_radii_lp(best_centers)
    
    # Strict post-processing to guarantee validation passes
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(best_centers[i,0], 1.0-best_centers[i,0], 
                 best_centers[i,1], 1.0-best_centers[i,1])
        if radii[i] > mx:
            radii[i] = max(0.0, mx - 1e-9)
            
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], 
                             best_centers[i,1]-best_centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    excess = radii[i] + radii[j] - d
                    radii[i] -= excess / 2.0
                    radii[j] -= excess / 2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = np.sum(radii)
    
    return best_centers, radii, float(final_sum)
```
