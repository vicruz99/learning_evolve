import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to find radii maximizing sum(radii)."""
    n = centers.shape[0]
    # Pairwise distances
    dists = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(axis=2))
    b_ub = dists[I_IDX, J_IDX]
    
    # Constraint matrix: r_i + r_j <= d_ij
    A_ub = np.zeros((len(I_IDX), n))
    A_ub[np.arange(len(I_IDX)), I_IDX] = 1.0
    A_ub[np.arange(len(J_IDX)), J_IDX] = 1.0
    
    # Boundary constraints: 0 <= r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    bounds = [(0.0, max(0.0, u)) for u in ub]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except ValueError:
        try:
            res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
            if res.success:
                return res.x, -res.fun
        except:
            pass
    return np.zeros(n), 0.0

def obj_centers(c_flat):
    """Objective for center optimization: negative of max sum of radii."""
    c = c_flat.reshape(N, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    _, s = solve_radii_lp(c)
    return -s

def make_init(seed, mode='hex'):
    """Generate diverse initial center configurations."""
    rng = np.random.RandomState(seed)
    c = np.zeros((N, 2))
    if mode == 'hex':
        s = 0.20 + rng.uniform(-0.05, 0.05)
        pts = []
        row = 0
        while len(pts) < N:
            y = 0.05 + row * s * np.sqrt(3)/2
            if y > 0.95: break
            x_start = 0.05 + (row % 2) * s/2
            col = 0
            while x_start + col * s <= 0.95 and len(pts) < N:
                pts.append([x_start + col * s, y])
                col += 1
            row += 1
        c[:len(pts)] = pts
        c[len(pts):] = rng.uniform(0.1, 0.9, (N-len(pts), 2))
    elif mode == 'grid':
        idx = 0
        for i in range(6):
            for j in range(6):
                if idx < N:
                    c[idx] = [0.1 + j*0.15, 0.1 + i*0.15]
                    idx += 1
        c[idx:] = rng.uniform(0.2, 0.8, (N-idx, 2))
    else:
        c = rng.uniform(0.1, 0.9, (N, 2))
        
    c += rng.randn(N, 2) * 0.02
    c = np.clip(c, 0.02, 0.98)
    return c.ravel()

def obj_joint(x):
    """Objective for joint optimization: negative sum of radii."""
    return -np.sum(x[2*N:])

def constr_joint(x):
    """Inequality constraints for joint optimization."""
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    # Boundary constraints
    c_b = np.concatenate([cx-r, 1-cx-r, cy-r, 1-cy-r])
    # Pairwise non-overlap constraints
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    return np.concatenate([c_b, dists[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]])

def run_packing():
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Phase 1: Optimize centers using derivative-free methods on LP objective
    modes = ['hex']*8 + ['grid']*4 + ['rand']*4
    for i, mode in enumerate(modes):
        x0 = make_init(seed=i*13+7, mode=mode)
        try:
            res = minimize(obj_centers, x0, method='Nelder-Mead', 
                           options={'maxiter': 2500, 'xatol': 1e-5, 'fatol': 1e-6})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_c = res.x.reshape(N, 2)
                best_r, _ = solve_radii_lp(best_c)
        except:
            pass
            
    # Phase 2: Joint refinement with SLSQP to push into tight contacts
    if best_c is not None:
        x0_joint = np.concatenate([best_c.ravel(), best_r])
        bounds = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
        cons = {'type': 'ineq', 'fun': constr_joint}
        
        try:
            res = minimize(obj_joint, x0_joint, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_c = res.x[:2*N].reshape(N, 2)
                best_r = res.x[2*N:]
        except:
            pass
            
        # Perturbation search to escape local minima around the best joint solution
        rng = np.random.RandomState(42)
        for _ in range(8):
            x_pert = np.concatenate([best_c.ravel() + rng.randn(2*N)*0.002, 
                                     best_r + rng.randn(N)*0.0005])
            x_pert = np.clip(x_pert, [b[0] for b in bounds], [b[1] for b in bounds])
            try:
                res = minimize(obj_joint, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-14})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_c = res.x[:2*N].reshape(N, 2)
                    best_r = res.x[2*N:]
            except:
                pass

    # Fallback if optimization fails completely
    if best_c is None:
        best_c = make_init(0, 'hex').reshape(N, 2)
        best_r, best_sum = solve_radii_lp(best_c)
        
    # Post-processing to guarantee strict validity against numerical errors
    for i in range(N):
        mx = min(best_c[i,0], 1-best_c[i,0], best_c[i,1], 1-best_c[i,1])
        best_r[i] = min(best_r[i], max(0, mx - 1e-9))
        
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                if d < best_r[i] + best_r[j] - 1e-10:
                    exc = best_r[i] + best_r[j] - d
                    best_r[i] -= exc/2
                    best_r[j] -= exc/2
                    changed = True
        if not changed: break
        
    best_r = np.maximum(best_r, 0.0)
    return best_c, best_r, float(np.sum(best_r))
