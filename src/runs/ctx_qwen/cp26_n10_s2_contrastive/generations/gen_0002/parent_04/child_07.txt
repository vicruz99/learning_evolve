import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I, J = np.triu_indices(N, k=1)

def objective(x):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2::3])

def constraints(x):
    """
    Inequality constraints:
    - Pairwise distance >= sum of radii
    - Circle boundaries within [0,1]x[0,1]
    Returns array of constraint values (must be >= 0)
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c_bound = np.concatenate([
        cx - r, 
        1.0 - cx - r, 
        cy - r, 
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist(i,j) >= r[i] + r[j]
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dists = np.hypot(dx, dy)
    c_overlap = dists - r[I] - r[J]
    
    return np.concatenate([c_bound, c_overlap])

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to find radii that maximize sum(radii)"""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            idx += 1
            
    bounds_r = []
    for c in centers:
        mx = min(c[0], 1.0-c[0], c[1], 1.0-c[1])
        bounds_r.append((0.0, max(0.0, mx)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success and not np.isnan(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback radii
    r = np.array([b[1] for b in bounds_r])
    return r, np.sum(r)

def make_hex_init(seed):
    """Generate a hexagonal lattice initialization with controlled perturbation."""
    rng = np.random.RandomState(seed)
    spacing = 0.16 + 0.06 * (seed % 10) / 10.0
    pts = []
    for r in range(10):
        y = 0.05 + r * spacing * np.sqrt(3)/2
        if y > 0.95: break
        offset = spacing/2 if r % 2 == 1 else 0.0
        for c in range(10):
            x = 0.05 + offset + c * spacing
            if x > 0.95: break
            pts.append([x, y])
            
    pts = np.array(pts[:N])
    if len(pts) < N:
        while len(pts) < N:
            pts = np.vstack([pts, rng.uniform(0.1, 0.9, 2)])
            
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    
    r_init = np.full(N, 0.05 + 0.03 * (seed % 5) / 5.0)
    return np.column_stack([pts, r_init]).flatten()

def make_random_init(seed):
    """Generate a random initialization in the safe inner region."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    r_init = np.full(N, 0.06)
    return np.column_stack([pts, r_init]).flatten()

def run_packing():
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_x = None
    
    # Stage 1: Broad search with mixed initializations
    for seed in range(50):
        if seed % 3 == 0:
            x0 = make_random_init(seed)
        else:
            x0 = make_hex_init(seed)
            
        try:
            res = minimize(
                objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False}
            )
            
            cx = res.x[0::3]
            cy = res.x[1::3]
            centers = np.column_stack([cx, cy])
            
            # LP refinement extracts maximum radii for these centers
            r_lp, sum_lp = solve_radii_lp(centers)
            
            if sum_lp > best_sum:
                best_sum = sum_lp
                best_x = np.concatenate([centers.flatten(), r_lp])
        except Exception:
            continue
            
    # Stage 2: Local refinement around the best solution found
    if best_x is not None:
        rng = np.random.RandomState(42)
        for _ in range(20):
            x_pert = best_x + rng.randn(len(best_x)) * 0.003
            x_pert = np.clip(x_pert, [b[0] for b in bounds], [b[1] for b in bounds])
            try:
                res = minimize(
                    objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 10000, 'ftol': 1e-13}
                )
                cx = res.x[0::3]
                cy = res.x[1::3]
                centers = np.column_stack([cx, cy])
                r_lp, sum_lp = solve_radii_lp(centers)
                
                if sum_lp > best_sum:
                    best_sum = sum_lp
                    best_x = np.concatenate([centers.flatten(), r_lp])
            except Exception:
                continue
                
    # Fallback initialization
    if best_x is None:
        best_x = make_hex_init(0)
        
    # Extract and post-process to guarantee strict validity
    cx = best_x[0::3].copy()
    cy = best_x[1::3].copy()
    r = best_x[2::3].copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        max_r = min(cx[i], 1-cx[i], cy[i], 1-cy[i])
        if r[i] > max_r:
            r[i] = max_r
            
    # Iteratively resolve overlaps to tolerance
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                dist = math.hypot(cx[i]-cx[j], cy[i]-cy[j])
                if dist < r[i] + r[j] - 1e-10:
                    excess = r[i] + r[j] - dist
                    r[i] -= excess / 2
                    r[j] -= excess / 2
                    changed = True
        if not changed:
            break
            
    r = np.maximum(r, 0.0)
    centers = np.column_stack([cx, cy])
    return centers, r, float(np.sum(r))
