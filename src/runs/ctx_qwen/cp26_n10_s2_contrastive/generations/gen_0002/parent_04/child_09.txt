1. **Center Placement Optimization**: We treat the center coordinates as continuous variables. Since the maximum sum of radii for fixed centers is a linear programming problem, the objective function becomes a non-smooth but concave piecewise-linear function of the centers.
2. **Radius Calculation**: For any given set of centers, we solve a small LP to find the optimal radii that satisfy `r_i + r_j <= dist(i,j)` and boundary constraints `r_i <= min(x_i, 1-x_i, y_i, 1-y_i)`. This guarantees that every evaluated configuration is feasible and optimally packed for those centers.

import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def solve_radii_lp(centers):
    """
    Given fixed centers, solve an LP to maximize sum of radii
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        xi, yi = centers[i]
        for j in range(i + 1, n):
            xj, yj = centers[j]
            d = math.hypot(xi - xj, yi - yj)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    # Boundary constraints: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        # Clamp coordinates to [0,1] to avoid negative max_r
        xc = max(0.0, min(1.0, x))
        yc = max(0.0, min(1.0, y))
        max_r = min(xc, 1.0 - xc, yc, 1.0 - yc)
        bounds.append((0.0, max(0.0, max_r)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
        
    # Fallback
    return 0.0, np.zeros(n)

def objective_func(centers_flat):
    """Objective: minimize negative sum of radii."""
    centers = centers_flat.reshape((N, 2))
    s, r = solve_radii_lp(centers)
    return -s

def generate_initializations():
    """Create diverse starting points for center optimization."""
    inits = []
    rng = np.random.RandomState(42)
    
    # 1. Hexagonal lattice
    pts = []
    spacing = 0.185
    y = spacing
    row = 0
    while len(pts) < N:
        x = spacing + (row % 2) * spacing / 2
        while x <= 1.0 - spacing and len(pts) < N:
            pts.append([x, y])
            x += spacing
        y += spacing * np.sqrt(3) / 2
        row += 1
    inits.append(np.array(pts[:N]))
    
    # 2. Perturbed hexagonal lattices
    for _ in range(3):
        c = np.array(pts[:N]) + rng.randn(N, 2) * 0.025
        inits.append(np.clip(c, 0.05, 0.95))
        
    # 3. Square grid
    pts_grid = []
    for i in range(6):
        for j in range(6):
            if len(pts_grid) < N:
                pts_grid.append([0.1 + j * 0.16, 0.1 + i * 0.16])
    inits.append(np.array(pts_grid[:N]))
    
    # 4. Corner/Edge focused
    c_corner = []
    corners = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
    for cx, cy in corners:
        for _ in range(4):
            c_corner.append([cx + rng.uniform(-0.06, 0.06), cy + rng.uniform(-0.06, 0.06)])
    while len(c_corner) < N:
        c_corner.append(rng.uniform(0.25, 0.75, 2))
    inits.append(np.array(c_corner[:N]))
    
    # 5. Pure random
    for _ in range(4):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return [c.flatten() for c in inits]

def run_packing():
    best_sum = 0.0
    best_c = None
    best_r = None
    
    inits = generate_initializations()
    rng = np.random.RandomState(77)
    
    # Primary optimization from diverse starts
    for x0 in inits:
        try:
            # Nelder-Mead is robust for the non-smooth LP value landscape
            res = minimize(objective_func, x0, method='Nelder-Mead',
                           options={'maxiter': 4000, 'xatol': 1e-5, 'fatol': 1e-5})
            c_opt = np.clip(res.x.reshape((N, 2)), 0.0, 1.0)
            s, r = solve_radii_lp(c_opt)
            
            if s > best_sum:
                best_sum = s
                best_c = c_opt.copy()
                best_r = r.copy()
                
                # Refine with Powell's method
                res2 = minimize(objective_func, res.x, method='Powell',
                                options={'maxiter': 2000, 'ftol': 1e-6})
                c_opt2 = np.clip(res2.x.reshape((N, 2)), 0.0, 1.0)
                s2, r2 = solve_radii_lp(c_opt2)
                if s2 > best_sum:
                    best_sum = s2
                    best_c = c_opt2.copy()
                    best_r = r2.copy()
        except Exception:
            continue
            
    # Fallback safety
    if best_c is None:
        best_c = inits[0].reshape((N, 2))
        _, best_r = solve_radii_lp(best_c)
        best_sum = np.sum(best_r)
        
    # Local hill-climbing to escape shallow local minima
    for _ in range(200):
        i = rng.randint(N)
        old_pos = best_c[i].copy()
        for _ in range(10):
            new_pos = old_pos + rng.randn(2) * 0.015
            new_pos = np.clip(new_pos, 0.02, 0.98)
            test_c = best_c.copy()
            test_c[i] = new_pos
            s, r = solve_radii_lp(test_c)
            if s > best_sum:
                best_sum = s
                best_c = test_c.copy()
                best_r = r.copy()
                break
                
    # Strict post-processing to guarantee validity against numerical drift
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                if d < best_r[i] + best_r[j] - 1e-10:
                    excess = best_r[i] + best_r[j] - d
                    best_r[i] -= excess / 2.0
                    best_r[j] -= excess / 2.0
                    changed = True
        if not changed:
            break
            
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(best_c[i, 0], 1.0 - best_c[i, 0], best_c[i, 1], 1.0 - best_c[i, 1])
        if best_r[i] > mx:
            best_r[i] = mx
            
    best_r = np.maximum(best_r, 0.0)
    best_sum = np.sum(best_r)
    
    return best_c, best_r, float(best_sum)
