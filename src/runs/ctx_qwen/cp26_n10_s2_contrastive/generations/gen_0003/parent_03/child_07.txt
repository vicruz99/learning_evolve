import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NP = len(I)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((NP, n))
    A_ub[np.arange(NP), I] = 1.0
    A_ub[np.arange(NP), J] = 1.0
    
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub - 1e-9)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.01)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints: pairwise non-overlap and boundary clearance."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dists = np.hypot(dx, dy)
    c = dists - (r[I] + r[J])
    c = np.concatenate([c, cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    best_sum = 0.0
    best_c = None
    best_r = None
    
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    inits = []
    rng = np.random.RandomState(42)
    
    # 1. Hexagonal lattices with varying spacing
    for sp in np.linspace(0.14, 0.22, 12):
        pts = []
        y = sp / 2
        row = 0
        while len(pts) < N and y < 1.0 - sp / 2:
            x = sp / 2 + (row % 2) * sp / 2
            while x < 1.0 - sp / 2 and len(pts) < N:
                pts.append([x, y])
                x += sp
            y += sp * np.sqrt(3) / 2
            row += 1
        pts = np.array(pts[:N])
        pts += rng.normal(0, 0.004, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        r_init = solve_lp_radii(pts)
        inits.append(np.concatenate([pts.ravel(), r_init]))
        
    # 2. Square grids
    for sp in np.linspace(0.16, 0.24, 10):
        pts = []
        y = sp / 2
        while len(pts) < N and y < 1.0 - sp / 2:
            x = sp / 2
            while x < 1.0 - sp / 2 and len(pts) < N:
                pts.append([x, y])
                x += sp
            y += sp
        pts = np.array(pts[:N])
        pts += rng.normal(0, 0.004, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        r_init = solve_lp_radii(pts)
        inits.append(np.concatenate([pts.ravel(), r_init]))
        
    # 3. Random placements
    for _ in range(25):
        pts = rng.uniform(0.1, 0.9, (N, 2))
        r_init = solve_lp_radii(pts)
        inits.append(np.concatenate([pts.ravel(), r_init]))
        
    # 4. Boundary/Corner focused
    for _ in range(15):
        pts = np.zeros((N, 2))
        corners = [[0.05, 0.05], [0.95, 0.05], [0.05, 0.95], [0.95, 0.95]]
        for i in range(4):
            pts[i] = corners[i]
        for i in range(4, N):
            pts[i] = rng.uniform(0.15, 0.85, 2)
        pts += rng.normal(0, 0.025, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        r_init = solve_lp_radii(pts)
        inits.append(np.concatenate([pts.ravel(), r_init]))

    for x0 in inits:
        # Scale radii down slightly to ensure strict initial feasibility

            if np.all(constraints(res.x) >= -1e-5):
                cx = res.x[0::3]
                cy = res.x[1::3]
                c_opt = np.column_stack((cx, cy))
                
                # Exact LP refinement for radii
                r_opt = solve_lp_radii(c_opt)
                s_opt = np.sum(r_opt)

                    best_sum = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()

                    for _ in range(6):
                        c_pert = best_c + rng.normal(0, 0.002, best_c.shape)

                        res2 = minimize(objective, x_pert, method='SLSQP', bounds=bounds_opt,

                            s2 = np.sum(r2)
                            if s2 > best_sum:
                                best_sum = s2
                                best_c = c2.copy()
                                best_r = r2.copy()
        except Exception:
            continue

    # Fallback safety net
    if best_c is None:
        best_c = inits[0][:2 * N].reshape(N, 2)
        best_r = solve_lp_radii(best_c)
        best_sum = np.sum(best_r)
        
    # Strict post-processing to guarantee validity
    c_final = best_c.copy()
    r_final = best_r.copy()
    
    for i in range(N):
        mx = min(c_final[i, 0], 1.0 - c_final[i, 0], c_final[i, 1], 1.0 - c_final[i, 1])
        r_final[i] = min(r_final[i], mx - 1e-9)
        r_final[i] = max(r_final[i], 0.0)
        
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(c_final[i, 0] - c_final[j, 0], c_final[i, 1] - c_final[j, 1])
                if d < r_final[i] + r_final[j] - 1e-10:
                    exc = r_final[i] + r_final[j] - d
                    r_final[i] -= exc / 2.0
                    r_final[j] -= exc / 2.0
                    changed = True
        if not changed:
            break
            
    return c_final, r_final, float(np.sum(r_final))
