from scipy.optimize import linprog, minimize

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIRS_I)

# Precompute constant LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS, N))
A_LP[np.arange(NUM_PAIRS), PAIRS_I] = 1.0
A_LP[np.arange(NUM_PAIRS), PAIRS_J] = 1.0
c_LP = -np.ones(N)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.hypot(dx, dy)
    b_ub = dists[PAIRS_I, PAIRS_J]
    
    bounds_r = []
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds_r.append((0.0, max(0.0, mx)))
        
    try:
        res = linprog(c_LP, A_ub=A_LP, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return -1.0, np.zeros(N)

def obj_centers_powell(x):
    """Objective for Powell: maximize LP sum of radii (minimize negative)."""
    centers = x.reshape(N, 2)
    s, _ = solve_lp_radii(centers)
    return -s

    dy = cy[PAIRS_I] - cy[PAIRS_J]

    c_bound = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    return np.concatenate([c_dist, c_bound])
