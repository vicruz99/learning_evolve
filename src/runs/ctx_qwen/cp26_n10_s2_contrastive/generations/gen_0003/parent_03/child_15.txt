N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

    A_ub[np.arange(len(I_IDX)), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]
    
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, mx)))

            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)

                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(n), 0.0

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):

    r = x[2::3]

    dy = cy[I_IDX] - cy[J_IDX]

    c_bound = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    return np.concatenate([c_overlap, c_bound])

            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2
        row += 1

        idx += 1
    
    centers += rng.normal(0, 0.006, centers.shape)
