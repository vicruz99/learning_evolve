import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

# Precompute constant A matrix for LP: r_i + r_j <= d_ij
A_LP = np.zeros((N_PAIRS, N))
A_LP[np.arange(N_PAIRS), I_IDX] = 1.0
A_LP[np.arange(N_PAIRS), J_IDX] = 1.0

def get_lp_data(centers):
    """Solves LP to find max sum of radii for fixed centers. Returns radii, sum, and constraint duals."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    dx = centers[I_IDX, 0] - centers[J_IDX, 0]
    dy = centers[I_IDX, 1] - centers[J_IDX, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, mx - 1e-9)))
        
    try:
        res = linprog(c_obj, A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return np.zeros(n), 0.0, np.zeros(N_PAIRS)
        
        radii = np.maximum(res.x, 0.0)
        duals = np.zeros(N_PAIRS)
        if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            duals = np.maximum(res.ineqlin.marginals, 0.0)
        return radii, -res.fun, duals
    except Exception:
        return np.zeros(n), 0.0, np.zeros(N_PAIRS)

def optimize_centers_dual_guided(centers, steps=50, init_step=0.006):

    radii, best_sum, _ = get_lp_data(best_c)

            break

        forces[mask_b, 1] += (margin - best_c[mask_b, 1]) * bf

            best_c = new_c
            best_sum = new_sum

def s_obj(x):
    """SLSQP objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def s_constr(x):
    """SLSQP inequality constraints: g(x) >= 0"""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    d = np.hypot(dx, dy)
    c = d - (r[I_IDX] + r[J_IDX])
    b = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    return np.concatenate([c, b])
