import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints(x):
    """Inequality constraints using squared distances for numerical stability."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]

    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c_overlap = dx**2 + dy**2 - (r[I] + r[J])**2

    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])

    return np.concatenate([c_overlap, c_bound])


def solve_radii_lp(centers):
    """Given fixed centers, solve LP to find radii maximizing sum."""
    n = centers.shape[0]
    c_obj = -np.ones(n)

    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])

    A_ub = np.zeros((n * (n - 1) // 2, n))
    b_ub = np.zeros(n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1

    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1 - x, y, 1 - y)
        bounds_r.append((0.0, max(0.0, ub)))

    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success:
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue

    return np.zeros(n), 0.0


def make_hex_init(spacing, seed=0):
    """Create hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    row = 0
    y = spacing / 2 + rng.uniform(-0.02, 0.02)
    while idx < N and y < 1.0 - spacing / 2:
        x_start = spacing / 2 + (row % 2) * spacing / 2 + rng.uniform(-0.01, 0.01)
        col = 0
        while x_start + col * spacing < 1.0 - spacing / 2 and idx < N:
            centers[idx, 0] = x_start + col * spacing
            centers[idx, 1] = y
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2
        row += 1
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    centers += rng.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    return centers


def make_pattern_init(pat, spacing, seed=0):
    """Create pattern-based initialization with specified row counts."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    dy = spacing * np.sqrt(3) / 2
    y = spacing / 2
    for row_idx, cnt in enumerate(pat):
        x_start = spacing / 2 + (row_idx % 2) * spacing / 2
        for col in range(cnt):
            if idx < N:
                centers[idx, 0] = x_start + col * spacing
                centers[idx, 1] = y
                idx += 1
        y += dy
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    centers += rng.normal(0, 0.004, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    return centers


def rotate_centers(centers, angle, pivot=None):
    """Rotate centers around a pivot point."""
    if pivot is None:
        pivot = np.array([0.5, 0.5])
    cos_a = np.cos(angle)
    sin_a = np.sin(angle)
    dx = centers[:, 0] - pivot[0]
    dy = centers[:, 1] - pivot[1]
    new_x = pivot[0] + dx * cos_a - dy * sin_a
    new_y = pivot[1] + dx * sin_a + dy * cos_a
    return np.column_stack([new_x, new_y])


def run_optimization(x0, bounds, cons, maxiter=8000):
    """Run SLSQP optimization and return LP-refined result."""
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        curr_centers = np.column_stack((res.x[0::3], res.x[1::3]))
        r_opt, s_opt = solve_radii_lp(curr_centers)
        return curr_centers, r_opt, s_opt, res.success
    except Exception:
        return None, None, 0.0, False


def run_packing():
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}

    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Phase 1: Hexagonal lattice search - broad coverage
    for sp in np.linspace(0.13, 0.22, 30):
        for seed in range(5):
            c0 = make_hex_init(sp, seed)
            r_lp, _ = solve_radii_lp(c0)
            r_start = np.maximum(r_lp, 0.02) * 0.95

            x0 = np.zeros(3 * N)
            x0[0::3] = c0[:, 0]
            x0[1::3] = c0[:, 1]
            x0[2::3] = r_start

            curr_c, curr_r, s, success = run_optimization(x0, bounds, cons, maxiter=6000)
            if curr_c is not None and s > best_sum:
                best_sum = s
                best_centers = curr_c.copy()
                best_radii = curr_r.copy()

    # Phase 2: Pattern-based initializations
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [7, 6, 5, 4, 4],
        [4, 5, 6, 5, 6], [8, 6, 5, 4, 3], [6, 6, 6, 6, 2], [5, 5, 5, 5, 6]
    ]
    for pat in patterns:
        for sp in np.linspace(0.15, 0.20, 8):
            for seed in range(3):
                c0 = make_pattern_init(pat, sp, seed)
                r_lp, _ = solve_radii_lp(c0)
                r_start = np.maximum(r_lp, 0.02) * 0.95

                x0 = np.zeros(3 * N)
                x0[0::3] = c0[:, 0]
                x0[1::3] = c0[:, 1]
                x0[2::3] = r_start

                curr_c, curr_r, s, success = run_optimization(x0, bounds, cons, maxiter=5000)
                if curr_c is not None and s > best_sum:
                    best_sum = s
                    best_centers = curr_c.copy()
                    best_radii = curr_r.copy()

    # Phase 3: Random initializations
    for seed in range(25):
        rng = np.random.RandomState(seed)
        c0 = rng.uniform(0.1, 0.9, (N, 2))
        r_lp, _ = solve_radii_lp(c0)
        r_start = np.maximum(r_lp, 0.02) * 0.95

        x0 = np.zeros(3 * N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r_start

        curr_c, curr_r, s, success = run_optimization(x0, bounds, cons, maxiter=6000)
        if curr_c is not None and s > best_sum:
            best_sum = s
            best_centers = curr_c.copy()
            best_radii = curr_r.copy()

    # Phase 4: Rotation perturbation of best solution
    if best_centers is not None:
        for trial in range(20):
            rng = np.random.RandomState(trial + 5000)
            angle = rng.uniform(-0.15, 0.15)
            c_rot = rotate_centers(best_centers, angle)
            c_rot = np.clip(c_rot, 0.02, 0.98)
            r_rot, _ = solve_radii_lp(c_rot)
            r_start = np.maximum(r_rot, 0.02) * 0.95

            x0 = np.zeros(3 * N)
            x0[0::3] = c_rot[:, 0]
            x0[1::3] = c_rot[:, 1]
            x0[2::3] = r_start

            curr_c, curr_r, s, success = run_optimization(x0, bounds, cons, maxiter=5000)
            if curr_c is not None and s > best_sum:
                best_sum = s
                best_centers = curr_c.copy()
                best_radii = curr_r.copy()

    # Phase 5: Multi-scale local perturbation refinement
    if best_centers is not None:
        # Coarse perturbations
        for trial in range(30):
            rng = np.random.RandomState(trial)
            c_pert = best_centers + rng.normal(0, 0.005, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            r_pert, _ = solve_radii_lp(c_pert)
            r_start = np.maximum(r_pert, 0.02) * 0.98

            x0 = np.zeros(3 * N)
            x0[0::3] = c_pert[:, 0]
            x0[1::3] = c_pert[:, 1]
            x0[2::3] = r_start

            curr_c, curr_r, s, success = run_optimization(x0, bounds, cons, maxiter=4000)
            if curr_c is not None and s > best_sum:
                best_sum = s
                best_centers = curr_c.copy()
                best_radii = curr_r.copy()

        # Fine perturbations
        for trial in range(40):
            rng = np.random.RandomState(trial + 2000)
            c_pert = best_centers + rng.normal(0, 0.002, best_centers.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            r_pert, s_pert = solve_radii_lp(c_pert)

            if s_pert > best_sum:
                best_sum = s_pert
                best_centers = c_pert.copy()
                best_radii = r_pert.copy()

        # Very fine perturbations (LP only, no SLSQP)
        for trial in range(50):
            rng = np.random.RandomState(trial + 3000)
            c_pert = best_centers + rng.normal(0, 0.0008, best_centers.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            r_pert, s_pert = solve_radii_lp(c_pert)

            if s_pert > best_sum:
                best_sum = s_pert
                best_centers = c_pert.copy()
                best_radii = r_pert.copy()

    # Fallback
    if best_centers is None:
        best_centers = make_hex_init(0.17)
        best_radii, best_sum = solve_radii_lp(best_centers)

    # Strict post-processing to guarantee validity
    radii = best_radii.copy()
    centers = best_centers.copy()

    for i in range(N):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        if radii[i] > max_r:
            radii[i] = max(0.0, max_r - 1e-9)

    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0],
                             centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    overlap = radii[i] + radii[j] - d
                    radii[i] -= overlap / 2.0
                    radii[j] -= overlap / 2.0
                    changed = True
        if not changed:
            break

    radii = np.maximum(radii, 0.0)
    best_sum = float(np.sum(radii))

    return centers, radii, best_sum
