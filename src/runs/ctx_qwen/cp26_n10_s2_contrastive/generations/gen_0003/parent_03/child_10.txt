import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_UP, J_UP = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    
    k = 0
    for i in range(n):
        for j in range(i+1, n):
            d = math.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = d
            k += 1
            
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0-x, y, 1.0-y)
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= -1e-9):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    dx = cx[I_UP] - cx[J_UP]
    dy = cy[I_UP] - cy[J_UP]
    dists = np.hypot(dx, dy)
    c_overlap = dists - (r[I_UP] + r[J_UP])
    c_bound = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    return np.concatenate([c_overlap, c_bound])

def make_hex_init(seed, spacing=0.15, noise=0.01):
    """Create a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = spacing/2
    row = 0
    while idx < N and y < 1.0 - spacing/2:
        x_start = spacing/2 + (row % 2) * spacing/2
        col = 0
        while x_start + col*spacing < 1.0 - spacing/2 and idx < N:
            centers[idx, 0] = x_start + col*spacing
            centers[idx, 1] = y
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2
        row += 1
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    centers += rng.normal(0, noise, centers.shape)
    return np.clip(centers, 0.05, 0.95)

def make_grid_init(seed, step=0.16):
    """Create a square grid initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = step/2
    while idx < N and y < 1.0 - step/2:
        x = step/2
        while x < 1.0 - step/2 and idx < N:
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
            x += step
        y += step
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    centers += rng.normal(0, 0.01, centers.shape)
    return np.clip(centers, 0.05, 0.95)

def make_random_init(seed):
    """Create a random uniform initialization."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.15, 0.85, (N, 2))
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    best_sum = 0.0
    best_centers = None
    best_radii = None
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Diverse initializations
    inits = []
    for s in range(25):
        inits.append(make_hex_init(s, spacing=0.13 + (s%6)*0.01, noise=0.008))
    for s in range(15):
        inits.append(make_grid_init(s, step=0.14 + (s%5)*0.012))
    for s in range(20):
        inits.append(make_random_init(s))
        
    for c0 in inits:
        r0 = np.full(N, 0.04)
        x0 = np.zeros(3*N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r0
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons_dict, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                cx = res.x[0::3]
                cy = res.x[1::3]
                c_curr = np.column_stack((cx, cy))
                r_lp, s_lp = solve_lp_radii(c_curr)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c_curr.copy()
                    best_radii = r_lp.copy()
        except Exception:
            continue

    # Phase 2: Cooling perturbation refinement around best solution
    if best_centers is not None:
        rng = np.random.RandomState(42)
        for step in range(120):
            noise_scale = 0.012 * (0.96 ** (step // 8))
            c_pert = best_centers + rng.normal(0, noise_scale, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            
            # LP provides a feasible, optimized radius baseline for the perturbed centers
            r_pert, _ = solve_lp_radii(c_pert)
            r_pert = np.maximum(r_pert, 1e-5)
            
            x_pert = np.zeros(3*N)
            x_pert[0::3] = c_pert[:, 0]
            x_pert[1::3] = c_pert[:, 1]
            x_pert[2::3] = r_pert
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds_opt,
                               constraints=cons_dict, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                if res.success:
                    cx = res.x[0::3]
                    cy = res.x[1::3]
                    c_opt = np.column_stack((cx, cy))
                    r_lp, s_lp = solve_lp_radii(c_opt)
                    if s_lp > best_sum:
                        best_sum = s_lp
                        best_centers = c_opt.copy()
                        best_radii = r_lp.copy()
            except Exception:
                continue

    # Fallback safety net
    if best_centers is None:
        best_centers = make_hex_init(0)
        best_radii, best_sum = solve_lp_radii(best_centers)

    # Strict post-processing to guarantee numerical validity
    radii = best_radii.copy()
    for i in range(N):
        x, y = best_centers[i]
        max_r = min(x, 1.0-x, y, 1.0-y)
        if radii[i] > max_r - 1e-9:
            radii[i] = max(0.0, max_r - 1e-9)
            
    # Iteratively resolve any remaining microscopic overlaps
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = math.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                if d < radii[i] + radii[j] - 1e-10:
                    overlap = radii[i] + radii[j] - d
                    radii[i] -= overlap/2.0
                    radii[j] -= overlap/2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return best_centers, radii, float(np.sum(radii))
