1. Using **LP refinement** after SLSQP to optimally solve for radii given centers
2. Using **squared distance constraints** for better numerical behavior
3. **Diverse initializations** (hexagonal with varying parameters, grid patterns, random)
4. **Block coordinate descent** alternating between center optimization and radius optimization
5. **Local perturbation refinement** around the best solution

<strategy>
1. Implement LP-based radius optimization for fixed centers (from Solution 1)
2. Use squared distance constraints in SLSQP for numerical stability
3. Generate diverse initializations: hexagonal lattices with varying spacings/offsets, grid patterns, corner-focused patterns, and random
4. Run SLSQP optimization with many restarts (70+ seeds)
5. After each SLSQP run, refine radii using LP
6. Implement local perturbation around best solution
7. Strict post-processing to guarantee validity

import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)


def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    A_ub = np.zeros((N_PAIRS, n))
    A_ub[np.arange(N_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(N_PAIRS), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]
    
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, max_r)))
    
    for method in ['highs', 'interior-point', 'revised simplex']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return res.x, -res.fun
        except Exception:
            continue
    return np.zeros(n), 0.0


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints(x):
    """Squared distance and boundary constraints (must be >= 0)."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c_overlap = dx**2 + dy**2 - (r[I_IDX] + r[J_IDX])**2
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    return np.concatenate([c_overlap, c_bound])


def make_hex_init(seed, spacing, margin, offset_x=0.0, offset_y=0.0):
    """Generate hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    row = 0
    y = margin + offset_y + row * spacing * np.sqrt(3) / 2
    
    while idx < N and y < 1.0 - margin:
        x_start = margin + offset_x + (row % 2) * spacing / 2
        col = 0
        while x_start + col * spacing < 1.0 - margin and idx < N:
            centers[idx, 0] = x_start + col * spacing + rng.uniform(-0.005, 0.005)
            centers[idx, 1] = y + rng.uniform(-0.005, 0.005)
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2
        row += 1
    
    while idx < N:
        centers[idx, 0] = rng.uniform(0.1, 0.9)
        centers[idx, 1] = rng.uniform(0.1, 0.9)
        idx += 1
    
    return np.clip(centers, 0.02, 0.98)


def make_grid_init(seed, spacing, margin):
    """Generate grid initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = margin
    
    while idx < N and y < 1.0 - margin:
        x = margin
        while x < 1.0 - margin and idx < N:
            centers[idx, 0] = x + rng.uniform(-0.005, 0.005)
            centers[idx, 1] = y + rng.uniform(-0.005, 0.005)
            idx += 1
            x += spacing
        y += spacing
    
    while idx < N:
        centers[idx, 0] = rng.uniform(0.1, 0.9)
        centers[idx, 1] = rng.uniform(0.1, 0.9)
        idx += 1
    
    return np.clip(centers, 0.02, 0.98)


def make_corner_init(seed):
    """Generate initialization with circles near corners and edges."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    
    # Place 4 circles near corners
    for cx, cy in [(0.06, 0.06), (0.94, 0.06), (0.06, 0.94), (0.94, 0.94)]:
        centers[idx, 0] = cx + rng.uniform(-0.02, 0.02)
        centers[idx, 1] = cy + rng.uniform(-0.02, 0.02)
        idx += 1
    
    # Fill rest with hexagonal pattern in center
    spacing = 0.16
    y = 0.25
    row = 0
    while idx < N and y < 0.85:
        x = 0.25 + (row % 2) * spacing / 2
        while x < 0.85 and idx < N:
            centers[idx, 0] = x + rng.uniform(-0.01, 0.01)
            centers[idx, 1] = y + rng.uniform(-0.01, 0.01)
            idx += 1
            x += spacing
        y += spacing * np.sqrt(3) / 2
        row += 1
    
    while idx < N:
        centers[idx, 0] = rng.uniform(0.15, 0.85)
        centers[idx, 1] = rng.uniform(0.15, 0.85)
        idx += 1
    
    return np.clip(centers, 0.02, 0.98)


def fix_violations(centers, radii):
    """Post-process to strictly satisfy boundary and overlap constraints."""
    n = centers.shape[0]
    radii = radii.copy()
    
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        if radii[i] > mx - 1e-9:
            radii[i] = max(0.0, mx - 1e-9)
    
    for _ in range(200):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                s = radii[i] + radii[j]
                if d < s - 1e-10:
                    diff = s - d
                    radii[i] -= diff * 0.5
                    radii[j] -= diff * 0.5
                    changed = True
        if not changed:
            break
    
    return centers, np.maximum(radii, 0.0)


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initializations
    inits = []
    
    # Hexagonal lattices with varying spacing
    for seed in range(25):
        spacing = 0.14 + seed * 0.004
        margin = 0.04 + (seed % 5) * 0.01
        c = make_hex_init(seed, spacing, margin)
        inits.append(c)
    
    # Hexagonal with row/column offsets
    for seed in range(20):
        spacing = 0.15 + seed * 0.003
        margin = 0.05
        ox = (seed % 5) * spacing / 4
        oy = (seed // 5) * spacing * np.sqrt(3) / 4
        c = make_hex_init(seed + 100, spacing, margin, offset_x=ox, offset_y=oy)
        inits.append(c)
    
    # Grid patterns
    for seed in range(15):
        spacing = 0.17 + seed * 0.005
        c = make_grid_init(seed + 200, spacing, 0.05)
        inits.append(c)
    
    # Corner-focused initializations
    for seed in range(10):
        c = make_corner_init(seed + 300)
        inits.append(c)
    
    # Random initializations
    for seed in range(15):
        rng = np.random.RandomState(seed + 400)
        c = rng.uniform(0.1, 0.9, (N, 2))
        inits.append(c)
    
    # Phase 1: Broad multi-start search
    for trial, c0 in enumerate(inits):
        r0 = np.full(N, 0.05)
        x0 = np.zeros(3 * N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r0
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            
            cx = res.x[0::3]
            cy = res.x[1::3]
            curr_centers = np.column_stack((cx, cy))
            
            r_opt, s_opt = solve_radii_lp(curr_centers)
            
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = curr_centers.copy()
                best_radii = r_opt.copy()
        except Exception:
            continue
    
    # Phase 2: Local perturbation refinement around best solution
    if best_centers is not None:
        rng = np.random.RandomState(99)
        for _ in range(40):
            # Perturb centers
            c_pert = best_centers + rng.normal(0, 0.003, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            r_pert, _ = solve_radii_lp(c_pert)
            
            x_pert = np.zeros(3 * N)
            x_pert[0::3] = c_pert[:, 0]
            x_pert[1::3] = c_pert[:, 1]
            x_pert[2::3] = np.maximum(r_pert, 1e-5)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                cx = res.x[0::3]
                cy = res.x[1::3]
                curr_centers = np.column_stack((cx, cy))
                r_opt, s_opt = solve_radii_lp(curr_centers)
                
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = curr_centers.copy()
                    best_radii = r_opt.copy()
            except Exception:
                continue
        
        # Medium perturbations to escape local minima
        for _ in range(30):
            noise_scale = rng.uniform(0.008, 0.025)
            c_pert = best_centers + rng.normal(0, noise_scale, best_centers.shape)
            c_pert = np.clip(c_pert, 0.05, 0.95)
            r_pert, _ = solve_radii_lp(c_pert)
            
            x_pert = np.zeros(3 * N)
            x_pert[0::3] = c_pert[:, 0]
            x_pert[1::3] = c_pert[:, 1]
            x_pert[2::3] = np.maximum(r_pert, 1e-5)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
                cx = res.x[0::3]
                cy = res.x[1::3]
                curr_centers = np.column_stack((cx, cy))
                r_opt, s_opt = solve_radii_lp(curr_centers)
                
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = curr_centers.copy()
                    best_radii = r_opt.copy()
            except Exception:
                continue
    
    # Fallback safety net
    if best_centers is None:
        best_centers = make_hex_init(0, 0.16, 0.05)
        best_radii, best_sum = solve_radii_lp(best_centers)
    
    # Phase 3: Strict post-processing
    best_centers, best_radii = fix_violations(best_centers, best_radii)
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
