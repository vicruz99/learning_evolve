import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, 1)
NUM_PAIRS = len(I_IDX)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((NUM_PAIRS, n))
    b_ub = np.zeros(NUM_PAIRS)
    
    # Compute pairwise distances efficiently
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    for k, (i, j) in enumerate(zip(I_IDX, J_IDX)):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]
        
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, mx)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.02)

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap (squared)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4 * N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c[4*N:] = dx**2 + dy**2 - (r[I_IDX] + r[J_IDX])**2
    return c

def fix_violations(centers, radii):
    """Strictly enforce boundary and non-overlap constraints."""
    n = centers.shape[0]
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        if radii[i] > mx:
            radii[i] = max(0.0, mx - 1e-9)
            
    for _ in range(100):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc * 0.5
                    radii[j] -= exc * 0.5
                    changed = True
        if not changed:
            break
    return centers, radii

def run_packing():
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.49)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_x = None
    
    # Phase 1: Diverse initializations with LP pre-solving
    inits = []
    for seed in range(45):
        rng = np.random.RandomState(seed)
        if seed < 25:
            # Hexagonal lattice with varying spacing
            c = np.zeros((N, 2))
            idx = 0
            sp = 0.15 + rng.uniform(0, 0.07)
            y = sp / 2
            row = 0
            while idx < N and y < 1.0 - sp / 2:
                x_start = sp / 2 + (row % 2) * sp / 2
                col = 0
                while x_start + col * sp < 1.0 - sp / 2 and idx < N:
                    c[idx] = [x_start + col * sp, y]
                    idx += 1
                    col += 1
                y += sp * np.sqrt(3) / 2
                row += 1
            while idx < N:
                c[idx] = rng.uniform(0.1, 0.9, 2)
                idx += 1
            c += rng.normal(0, 0.008, c.shape)
            c = np.clip(c, 0.02, 0.98)
        else:
            # Random placements
            c = rng.uniform(0.1, 0.9, (N, 2))
            
        r0 = solve_lp_radii(c) * 0.99
        inits.append(np.concatenate([c.ravel(), r0]))
        
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 7000, 'ftol': 1e-13, 'disp': False})
            if res.success or -res.fun > best_sum:
                cx = res.x[0::3]
                cy = res.x[1::3]
                c_opt = np.column_stack((cx, cy))
                r_opt = solve_lp_radii(c_opt)
                s_opt = np.sum(r_opt)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_x = np.concatenate([c_opt.ravel(), r_opt])
        except Exception:
            pass
            
    # Phase 2: Local perturbation refinement to escape local minima
    if best_x is not None:

            xp[:2*N] = np.clip(xp[:2*N], 0.02, 0.98)
            xp[2*N:] = solve_lp_radii(xp[:2*N].reshape(N, 2)) * 0.99
            
            try:
                res = minimize(objective, xp, method='SLSQP', bounds=bounds, constraints=cons,

                cy = res.x[1::3]
                c_opt = np.column_stack((cx, cy))

                s_opt = np.sum(r_opt)

                pass

        best_x = inits[0]
        
    cx = best_x[0::3]
    cy = best_x[1::3]
    r = best_x[2*N:]
    centers = np.column_stack((cx, cy))
    radii = r.copy()
    
    # Strict post-processing to guarantee validity
    centers, radii = fix_violations(centers, radii)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
