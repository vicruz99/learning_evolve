import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIRS_I)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Squared distance and boundary constraints (must be >= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[PAIRS_I] - cx[PAIRS_J]
    dy = cy[PAIRS_I] - cy[PAIRS_J]
    r_sum = r[PAIRS_I] + r[PAIRS_J]
    c_overlap = dx**2 + dy**2 - r_sum**2
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    return np.concatenate([c_overlap, c_bound])

def solve_radii_lp(centers):
    """Given fixed centers, find radii that maximize sum via LP."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((NUM_PAIRS, n))
    b_ub = np.zeros(NUM_PAIRS)
    
    dists = np.sqrt(((centers[:, None, :] - centers[None, :, :])**2).sum(axis=2))
    
    A_ub[np.arange(NUM_PAIRS), PAIRS_I] = 1.0
    A_ub[np.arange(NUM_PAIRS), PAIRS_J] = 1.0
    b_ub[:] = dists[PAIRS_I, PAIRS_J]
    
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1-x, y, 1-y)
        bounds_r.append((0.0, max(0.0, ub)))
        
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return res.x, -res.fun
        except Exception:
            pass
        
    return np.zeros(n), 0.0

def make_hex_init(seed, spacing_range=(0.13, 0.17)):
    """Generate hexagonal lattice initial center configuration."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    s = spacing_range[0] + rng.uniform(0, spacing_range[1] - spacing_range[0])
    idx = 0
    row = 0
    y = s * 0.6 + rng.uniform(0, 0.03)
    
    while idx < N and y < 1.0 - s * 0.6:
        x_start = s * 0.5 + (row % 2) * s * 0.5 + rng.uniform(-0.01, 0.01)
        col = 0
        while x_start + col * s < 1.0 - s * 0.5 and idx < N:
            centers[idx, 0] = x_start + col * s
            centers[idx, 1] = y
            idx += 1
            col += 1
        y += s * np.sqrt(3) / 2
        row += 1
    
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
        
    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.02, 0.98)

def make_corner_init(seed):
    """Generate initialization that places circles near corners and edges."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    
    # Place circles in 4 corners
    margin = 0.12 + rng.uniform(0, 0.05)
    corners = [[margin, margin], [1-margin, margin], [margin, 1-margin], [1-margin, 1-margin]]
    for c in corners:
        if idx < N:
            centers[idx] = [c[0] + rng.uniform(-0.02, 0.02), c[1] + rng.uniform(-0.02, 0.02)]
            idx += 1
    
    # Place along edges
    edge_count = 8
    for i in range(edge_count):
        if idx >= N:
            break
        if i < 2:
            centers[idx] = [0.3 + i*0.1 + rng.uniform(-0.02, 0.02), margin + rng.uniform(-0.01, 0.01)]
        elif i < 4:
            centers[idx] = [0.3 + (i-2)*0.1 + rng.uniform(-0.02, 0.02), 1-margin + rng.uniform(-0.01, 0.01)]
        elif i < 6:
            centers[idx] = [margin + rng.uniform(-0.01, 0.01), 0.3 + (i-4)*0.1 + rng.uniform(-0.02, 0.02)]
        else:
            centers[idx] = [1-margin + rng.uniform(-0.01, 0.01), 0.3 + (i-6)*0.1 + rng.uniform(-0.02, 0.02)]
        idx += 1
    
    # Fill rest with hex-like pattern
    remaining = N - idx
    if remaining > 0:
        s = 0.16
        row = 0
        y = 0.25
        while idx < N:
            x = 0.25 + (row % 2) * s * 0.5
            while x < 0.75 and idx < N:
                centers[idx] = [x + rng.uniform(-0.02, 0.02), y + rng.uniform(-0.02, 0.02)]
                idx += 1
                x += s
            y += s * np.sqrt(3) / 2
            row += 1
    
    return np.clip(centers, 0.02, 0.98)

def make_structured_init(pattern):
    """Generate structured initialization based on row counts pattern."""
    centers = np.zeros((N, 2))
    idx = 0
    y = 0.08
    dy = 0.155
    
    for r_idx, cnt in enumerate(pattern):
        shift = 0.04 if r_idx % 2 == 0 else 0.04 + 0.08
        x = shift
        for _ in range(cnt):
            if idx < N:
                centers[idx] = [x, y]
                x += 0.165
                idx += 1
        y += dy
    
    while idx < N:
        centers[idx] = [0.5, 0.5]
        idx += 1
        
    return centers

def fix_violations(centers, radii):
    """Post-process to strictly satisfy constraints."""
    radii = radii.copy()
    
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        if radii[i] > mx:
            radii[i] = max(0.0, mx - 1e-9)
            
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                s = radii[i] + radii[j]
                if d < s - 1e-10:
                    diff = s - d
                    radii[i] -= diff * 0.5
                    radii[j] -= diff * 0.5
                    changed = True
        if not changed:
            break
            
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    ]
    
    initializations = []

    for pat in patterns:

        initializations.append(c)

    for seed in range(40):

        c = make_corner_init(seed)

    # Random initializations
    for seed in range(20):

        initializations.append(c)
    
    trial = 0
    max_trials = 100
    
    for base_centers in initializations:
        if trial >= max_trials:
            break
