import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4 * N + NUM_PAIRS)
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c[0:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    # Pairwise distance constraints: dist(i,j) >= r_i + r_j
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c[4*N:] = np.sqrt(dx*dx + dy*dy) - r[I] - r[J]
    return c

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to find radii that maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((NUM_PAIRS, n))
    b_ub = np.zeros(NUM_PAIRS)
    
    # Vectorized distance computation
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(diff[:,:,0]**2 + diff[:,:,1]**2)
    
    for k in range(NUM_PAIRS):
        i, j = I[k], J[k]
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]
        
    bounds_r = []
    for i in range(n):
        ub = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds_r.append((0.0, max(0.0, ub)))
        
    # HiGHS is typically fastest and most reliable for this LP size
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    # Fallback
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='interior-point')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def generate_init(seed, style='hex'):
    """Generate structured or random initial center configurations."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    
    if style == 'hex':
        s = 0.14 + rng.uniform(0, 0.06)
        idx = 0
        row = 0
        y = s/2 + rng.uniform(0, 0.05)
        while idx < N and y < 1.0 - s/2:
            x_start = s/2 + (row % 2) * s/2 + rng.uniform(0, 0.02)
            col = 0
            while x_start + col*s < 1.0 - s/2 and idx < N:
                centers[idx] = [x_start + col*s, y]
                idx += 1
                col += 1
            y += s * math.sqrt(3) / 2
            row += 1
    elif style == 'grid':
        idx = 0
        step = 0.16 + rng.uniform(0, 0.04)
        y = step/2
        while y < 1.0 - step/2 and idx < N:
            x = step/2
            while x < 1.0 - step/2 and idx < N:
                centers[idx] = [x, y]
                idx += 1
                x += step
            y += step
    else:
        centers = rng.uniform(0.1, 0.9, (N, 2))
        
    # Fill any remaining slots randomly
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
        
    # Add controlled noise to break symmetry and avoid exact overlaps
    centers += rng.normal(0, 0.008, centers.shape)
    return np.clip(centers, 0.05, 0.95)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng_main = np.random.RandomState(42)
    
    # Phase 1: Broad multi-start search with diverse topologies
    for seed in range(55):
        style = 'hex' if seed < 35 else ('grid' if seed < 45 else 'random')
        c0 = generate_init(seed, style)
        r0, _ = solve_lp_radii(c0)
        
        # Scale down radii slightly to ensure strict interior feasibility for SLSQP
        x0 = np.zeros(3*N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = np.maximum(r0 * 0.98, 1e-4)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-15, 'disp': False})
            
            cx = res.x[0::3]
            cy = res.x[1::3]
            cur_centers = np.column_stack((cx, cy))
            
            # Phase 2: Exact LP refinement for radii given optimized centers
            r_opt, s_opt = solve_lp_radii(cur_centers)
            
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = cur_centers.copy()
                best_radii = r_opt.copy()
                
                # Phase 3: Aggressive local perturbation search around new best
                for _ in range(15):
                    pert_scale = rng_main.uniform(0.002, 0.008)
                    c_pert = best_centers + rng_main.normal(0, pert_scale, best_centers.shape)
                    c_pert = np.clip(c_pert, 0.02, 0.98)
                    r_pert, s_pert = solve_lp_radii(c_pert)
                    
                    if s_pert > best_sum:
                        best_sum = s_pert
                        best_centers = c_pert.copy()
                        best_radii = r_pert.copy()
        except Exception:
            continue

    # Fallback safety net
    if best_centers is None:
        best_centers = generate_init(0)
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    # Phase 4: Strict post-processing to guarantee validity within numerical tolerance
    radii = best_radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        x, y = best_centers[i]
        max_r = min(x, 1.0-x, y, 1.0-y)
        if radii[i] > max_r - 1e-9:
            radii[i] = max(0.0, max_r - 1e-9)
            
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                dx = best_centers[i,0] - best_centers[j,0]
                dy = best_centers[i,1] - best_centers[j,1]
                d = math.hypot(dx, dy)
                if d < radii[i] + radii[j] - 1e-9:
                    overlap = radii[i] + radii[j] - d
                    radii[i] -= overlap / 2.0
                    radii[j] -= overlap / 2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return best_centers, radii, final_sum
