

<strategy>
1. **Robust Initialization**: Generate diverse initial configurations using explicit hexagonal lattice patterns with varying row counts and spacing, plus random perturbations. Ensure strict feasibility for the optimizer by clipping centers to `[0.07, 0.93]` and starting with small radii (`0.025`).
2. **Joint Optimization**: Use `SLSQP` to simultaneously optimize centers and radii. The constraint formulation uses squared distances to avoid sqrt singularities and improve gradient behavior.
3. **LP Refinement**: After each SLSQP run, fix the optimized centers and solve a Linear Program to find the absolute maximum sum of radii for those positions. This decouples the radii optimization, guaranteeing optimality for the given centers.
4. **Basin Hopping / Perturbation**: From the best found configuration, run a decay-perturbation schedule. Randomly jitter centers, solve LP for new radii, and re-run SLSQP to escape shallow local minima.
5. **Strict Post-Processing**: Apply boundary and overlap corrections with tight tolerances to guarantee the final output passes the validation function, even if numerical drift occurred during optimization.
6. **Vectorization & Efficiency**: Precompute the LP constraint matrix structure and use vectorized numpy operations in constraints to keep runtime low while allowing many restarts and refinement steps.
</strategy>
```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

# Precompute LP constraint matrix structure (constant over iterations)
A_LIP = np.zeros((N_PAIRS, N))
for k in range(N_PAIRS):
    A_LIP[k, I_IDX[k]] = 1.0
    A_LIP[k, J_IDX[k]] = 1.0

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    # Compute pairwise distances
    dx = centers[I_IDX, 0] - centers[J_IDX, 0]
    dy = centers[I_IDX, 1] - centers[J_IDX, 1]
    b_ub = np.sqrt(dx**2 + dy**2)
    
    # Boundary constraints for each circle: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds_r = []
    for i in range(N):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, ub - 1e-9)))
        
    try:
        res = linprog(-np.ones(N), A_ub=A_LIP, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            r = np.maximum(res.x, 0.0)
            # Enforce strict boundary compliance
            for i in range(N):
                x, y = centers[i]
                max_r = min(x, 1.0 - x, y, 1.0 - y)
                if r[i] > max_r:
                    r[i] = max(0.0, max_r - 1e-9)
            return r, np.sum(r)
    except Exception:
        pass
        
    # Fallback if LP fails
    return np.full(N, 0.01), 0.0

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints: g(x) >= 0"""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    c1 = cx - r
    c2 = 1.0 - cx - r
    c3 = cy - r
    c4 = 1.0 - cy - r
    
    # Pairwise non-overlap (squared distance avoids singularities)
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c5 = dx**2 + dy**2 - (r[I_IDX] + r[J_IDX])**2
    
    return np.concatenate([c1, c2, c3, c4, c5])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.RandomState(42)
    inits = []
    
    # 1. Structured hexagonal lattice patterns
    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [7, 6, 5, 4, 4], [4, 5, 6, 6, 5], [5, 5, 5, 5, 6],
        [8, 6, 5, 4, 3], [6, 7, 5, 5, 3]
    ]
    
    for pat in row_patterns:
        for r_est in np.linspace(0.08, 0.12, 6):
            centers = np.zeros((N, 2))
            y = r_est + 0.01
            idx = 0
            for r_idx, cnt in enumerate(pat):
                if cnt > 0 and idx < N:
                    spacing = 2.0 * r_est
                    shift = spacing / 2.0 if r_idx % 2 == 1 else 0.0
                    x_start = shift + r_est
                    for c in range(cnt):
                        if idx < N:
                            centers[idx] = [x_start + c * spacing, y]
                            idx += 1
                y += r_est * np.sqrt(3)
            while idx < N:
                centers[idx] = rng.uniform(0.1, 0.9, 2)
                idx += 1
                
            centers += rng.normal(0, 0.006, centers.shape)
            centers = np.clip(centers, 0.07, 0.93)
            inits.append(centers)
            
    # 2. Random feasible starts
    for _ in range(20):
        c = rng.uniform(0.15, 0.85, (N, 2))
        c += rng.normal(0, 0.025, c.shape)
        inits.append(np.clip(c, 0.07, 0.93))
        
    # Phase 1: Multi-start joint optimization
    for c0 in inits:
        r0 = np.full(N, 0.025) # Strictly feasible small radii
        x0 = np.zeros(3 * N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r0
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                cx = res.x[0::3]
                cy = res.x[1::3]
                curr_c = np.column_stack((cx, cy))
                
                # LP refinement for exact radii at optimized centers
                r_opt, s_opt = solve_lp_radii(curr_c)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = curr_c
                    best_radii = r_opt
        except Exception:
            continue

    # Phase 2: Basin hopping / Local perturbation refinement
    if best_centers is not None:
        for trial in range(50):
            # Decay perturbation magnitude over time
            noise_scale = 0.005 * (0.85 ** (trial // 10))
            c_pert = best_centers + rng.normal(0, noise_scale, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            
            r_pert, _ = solve_lp_radii(c_pert)
            
            x_pert = np.zeros(3 * N)
            x_pert[0::3] = c_pert[:, 0]
            x_pert[1::3] = c_pert[:, 1]
            x_pert[2::3] = np.maximum(r_pert, 1e-6)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
                if res.success:
                    cx = res.x[0::3]
                    cy = res.x[1::3]
                    curr_c = np.column_stack((cx, cy))
                    r_opt, s_opt = solve_lp_radii(curr_c)
                    
                    if s_opt > best_sum:
                        best_sum = s_opt
                        best_centers = curr_c
                        best_radii = r_opt
            except Exception:
                continue

    # Fallback safety net
    if best_centers is None:
        best_centers = inits[0]
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    # Phase 3: Strict post-processing for numerical validity
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if radii[i] > max_r:
            radii[i] = max(0.0, max_r - 1e-9)
            
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(300):
        changed = False
        for k in range(N_PAIRS):
            i, j = I_IDX[k], J_IDX[k]
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            s = radii[i] + radii[j]
            if d < s - 1e-11:
                diff = s - d
                radii[i] -= diff * 0.5
                radii[j] -= diff * 0.5
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
```
