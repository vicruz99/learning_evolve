import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I, J = np.triu_indices(N, k=1)
N_PAIRS = len(I)

def constraint_fun(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap."""
    c = x[:2*N].reshape(N, 2)
    r = x[2*N:]
    
    # Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    bc = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    # Pairwise: dist >= r_i + r_j
    dx = c[I, 0] - c[J, 0]
    dy = c[I, 1] - c[J, 1]
    dist = np.hypot(dx, dy)
    pc = dist - (r[I] + r[J])
    
    return np.concatenate([bc, pc])

def solve_lp_radii(centers):
    """Given fixed centers, find radii that maximize sum via LP."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((N_PAIRS, n))
    A_ub[np.arange(N_PAIRS), I] = 1.0
    A_ub[np.arange(N_PAIRS), J] = 1.0
    
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0-x, y, 1.0-y)
        bounds_r.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='interior-point')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def relax_centers(centers, radii, margin=0.02, iters=80):
    """Fast pairwise repulsion to improve feasibility before optimization."""
    c = centers.copy()
    r = radii.copy()
    for _ in range(iters):
        moved = False
        for i in range(N):
            for j in range(i+1, N):
                dx = c[i,0]-c[j,0]
                dy = c[i,1]-c[j,1]
                d = math.hypot(dx, dy)
                if d < r[i] + r[j] + margin and d > 1e-8:
                    nx, ny = dx/d, dy/d
                    shift = (r[i] + r[j] + margin - d) * 0.5
                    c[i,0] -= nx*shift
                    c[i,1] -= ny*shift
                    c[j,0] += nx*shift
                    c[j,1] += ny*shift
                    moved = True
        # Boundary repulsion
        for i in range(N):
            if c[i,0] < r[i] + margin: c[i,0] = r[i] + margin
            if c[i,0] > 1.0 - r[i] - margin: c[i,0] = 1.0 - r[i] - margin
            if c[i,1] < r[i] + margin: c[i,1] = r[i] + margin
            if c[i,1] > 1.0 - r[i] - margin: c[i,1] = 1.0 - r[i] - margin
        if not moved:
            break
    return np.clip(c, 0.01, 0.99)

def objective_func(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
