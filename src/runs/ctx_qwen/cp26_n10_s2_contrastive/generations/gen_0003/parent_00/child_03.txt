import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
I, J = np.triu_indices(N, k=1)
N_PAIRS = len(I)

def solve_radii_lp(centers):
    """Given fixed centers, find radii that maximize sum via LP."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    A_ub = np.zeros((N_PAIRS, n))
    A_ub[np.arange(N_PAIRS), I] = 1.0
    A_ub[np.arange(N_PAIRS), J] = 1.0
    
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='interior-point')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def de_objective(centers_flat):
    """Objective for DE: minimize negative sum of radii for given centers."""
    centers = centers_flat.reshape(N, 2)
    _, s = solve_radii_lp(centers)
    return -s

def slsqp_objective(x):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def slsqp_constraints(x):
    """Constraints for SLSQP: squared distances and boundaries."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c_overlap = dx**2 + dy**2 - (r[I] + r[J])**2
    
    c_bound = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    return np.concatenate([c_overlap, c_bound])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = 0.0

    best_radii = None

    # DE explores the non-convex space of center coordinates effectively.

    for seed in [42, 123, 456]:

                de_objective, bounds_centers, seed=seed, maxiter=200, popsize=20,
                mutation=(0.5, 1.0), recombination=0.9, tol=1e-7, polish=False
            )
            curr_centers = res_de.x.reshape(N, 2)
            curr_radii, curr_sum = solve_radii_lp(curr_centers)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = curr_centers
                best_radii = curr_radii
        except Exception:
            continue
            
    # Phase 2: SLSQP refinement from DE results
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii
        
        for trial in range(15):
            rng = np.random.RandomState(trial * 13 + seed)
            x_pert = x0.copy()
            x_pert[:2 * N] += rng.normal(0, 0.0015, 2 * N)
            x_pert[:2 * N] = np.clip(x_pert[:2 * N], 0.01, 0.99)

                    slsqp_objective, x_pert, method='SLSQP', bounds=bounds_slsqp,

        mx = min(x, 1.0 - x, y, 1.0 - y)

            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
