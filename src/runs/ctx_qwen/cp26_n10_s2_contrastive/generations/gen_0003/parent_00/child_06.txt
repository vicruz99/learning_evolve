import numpy as np
from scipy.optimize import linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

# Precompute constant parts of the LP constraint matrix
A_LP = np.zeros((N_PAIRS, N))
A_LP[np.arange(N_PAIRS), I_IDX] = 1.0
A_LP[np.arange(N_PAIRS), J_IDX] = 1.0
C_OBJ_LP = -np.ones(N)

def compute_lp_bounds(centers):
    bounds = []
    for i in range(N):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
    return bounds

def solve_radii_lp(centers):
    dx = centers[I_IDX, 0] - centers[J_IDX, 0]
    dy = centers[I_IDX, 1] - centers[J_IDX, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds_r = compute_lp_bounds(centers)
    
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(C_OBJ_LP, A_ub=A_LP, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success:
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(N), 0.0

def make_hex_init(seed):
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    s = 0.15 + rng.uniform(0, 0.05)
    y = s / 2.0
    row = 0
    while idx < N and y < 1.0 - s / 2.0:
        x_start = s / 2.0 + (row % 2) * s / 2.0
        col = 0
        while x_start + col * s < 1.0 - s / 2.0 and idx < N:
            centers[idx, 0] = x_start + col * s
            centers[idx, 1] = y
            idx += 1
            col += 1
        y += s * np.sqrt(3) / 2.0
        row += 1
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    return centers + rng.normal(0, 0.005, centers.shape)

def make_grid_init(seed):
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    step = 0.18 + rng.uniform(-0.02, 0.02)
    for i in range(8):
        for j in range(8):
            if idx < N:
                centers[idx, 0] = step + j * step + rng.uniform(0, step / 2)
                centers[idx, 1] = step + i * step + rng.uniform(0, step / 2)
                idx += 1
    return np.clip(centers, 0.05, 0.95)

def make_random_init(seed):
    rng = np.random.RandomState(seed)
    return rng.uniform(0.15, 0.85, (N, 2))

def repair_packing(centers, radii):
    radii = radii.copy()
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        if radii[i] > mx:
            radii[i] = max(0.0, mx - 1e-9)
            
    for _ in range(200):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc * 0.5
                    radii[j] -= exc * 0.5
                    changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    return radii

def run_packing():
    np.random.seed(42)
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    inits = []
    for s in range(25):
        inits.append(make_hex_init(s))
    for s in range(15):
        inits.append(make_grid_init(s))
    for s in range(20):
        inits.append(make_random_init(s))
        
    for init_c in inits:
        c_opt, s_opt = solve_radii_lp(init_c)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = init_c.copy()
            best_radii = c_opt.copy()
            
        curr_c = init_c.copy()
        curr_s = s_opt
        sigma = 0.015
        for step in range(1500):
            temp = 0.005 * (1.0 - step / 1500.0) + 1e-5
            trial_c = curr_c + np.random.randn(N, 2) * sigma
            trial_c = np.clip(trial_c, 0.02, 0.98)
            trial_r, trial_s = solve_radii_lp(trial_c)
            
            delta = trial_s - curr_s
            if delta > 0 or np.random.rand() < np.exp(delta / temp):
                curr_c = trial_c
                curr_s = trial_s
                
            sigma *= 0.9995
            
            if curr_s > best_sum:
                best_sum = curr_s
                best_centers = curr_c.copy()
                best_radii = trial_r.copy()
                
    if best_centers is None:
        best_centers = make_hex_init(0)
        best_radii, best_sum = solve_radii_lp(best_centers)
        
    best_radii = repair_packing(best_centers, best_radii)
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(final_sum)
