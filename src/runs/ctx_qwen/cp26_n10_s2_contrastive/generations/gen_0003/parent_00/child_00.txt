import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

# Precompute constant LP constraint matrix for speed
A_LP = np.zeros((N_PAIRS, N))
A_LP[np.arange(N_PAIRS), I_IDX] = 1.0
A_LP[np.arange(N_PAIRS), J_IDX] = 1.0

def compute_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to non-overlap and boundary constraints."""
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    b_ub = dists[I_IDX, J_IDX]
    
    bounds = []
    for i in range(N):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub - 1e-9)))
        
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
        
    # Fallback solver if highs fails
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='interior-point')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
        
    return 0.0, np.zeros(N)

def obj_centers(x):
    """Objective for center optimization: maximize LP sum of radii."""
    c = x.reshape(N, 2)
    c = np.clip(c, 0.02, 0.98)
    s, _ = compute_lp(c)
    return -max(s, 0.0)

def objective_joint(x):
    """Joint objective: minimize negative sum of radii."""
    return -np.sum(x[2 * N:])

def constraints_joint(x):
    """Squared distance and boundary constraints for joint optimization."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c_overlap = dx**2 + dy**2 - (r[I_IDX] + r[J_IDX])**2
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    return np.concatenate([c_overlap, c_bound])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0.0
    best_c = None
    best_r = None
    
    inits = []
    
    # 1. Generate diverse hexagonal lattice initializations
    for seed in range(50):
        rng = np.random.RandomState(seed)
        c = np.zeros((N, 2))
        idx = 0
        s = 0.12 + rng.uniform(0, 0.08)
        y = s/2 + rng.uniform(0, 0.05)
        row = 0
        while idx < N and y < 1.0 - s/2:
            x_start = s/2 + (row % 2) * s/2 + rng.uniform(0, 0.02)
            col = 0
            while x_start + col * s < 1.0 - s/2 and idx < N:
                c[idx] = [x_start + col * s, y]
                idx += 1
                col += 1
            y += s * math.sqrt(3) / 2
            row += 1
        while idx < N:
            c[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
        inits.append(c)
        
    # 2. Add random initializations
    for seed in range(20):
        rng = np.random.RandomState(2000 + seed)
        c = rng.uniform(0.1, 0.9, (N, 2))
        inits.append(c)
        
    # Phase 1: Derivative-free optimization of centers
    for c0 in inits:
        x0 = c0.flatten()
        try:
            res = minimize(obj_centers, x0, method='Nelder-Mead', 
                           options={'maxiter': 800, 'xatol': 1e-4, 'fatol': 1e-6})
            c_opt = res.x.reshape(N, 2)
            s_opt, r_opt = compute_lp(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            continue
            
    # Phase 2: Joint refinement with SLSQP
    if best_c is not None:
        x_joint = np.concatenate([best_c.flatten(), best_r])
        bounds_joint = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
        cons_joint = {'type': 'ineq', 'fun': constraints_joint}
        
        try:
            res = minimize(objective_joint, x_joint, method='SLSQP', bounds=bounds_joint,
                           constraints=cons_joint, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                cx = res.x[0::3]
                cy = res.x[1::3]
                curr_c = np.column_stack((cx, cy))
                curr_s, curr_r = compute_lp(curr_c)
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_c = curr_c
                    best_r = curr_r

                rng = np.random.RandomState(trial * 31 + 7)

                            best_r = r_pp
                except Exception:
                    pass
        except Exception:
            pass
            
    # Fallback safety net
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum = compute_lp(best_c)
        
    # Phase 3: Strict post-processing to guarantee validity
    radii = best_r.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        x, y = best_c[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if radii[i] > max_r - 1e-9:
            radii[i] = max(0.0, max_r - 1e-9)
            
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    overlap = radii[i] + radii[j] - d
                    radii[i] -= overlap / 2.0
                    radii[j] -= overlap / 2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return best_c, radii, float(np.sum(radii))
