import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to non-overlap and boundaries."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    
    # Pairwise distances
    dx = centers[:, None, 0] - centers[None, :, 0]
    dy = centers[:, None, 1] - centers[None, :, 1]
    dists = np.hypot(dx, dy)
    
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1-x, y, 1-y)
        bounds_r.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='interior-point')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Boundary and pairwise non-overlap constraints (must be >= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    # Overlap constraints: distance >= r_i + r_j
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dists = np.hypot(dx, dy)
    c_overlap = dists - (r[I] + r[J])
    
    return np.concatenate([c_overlap, c_bound])

def get_init_hex(spacing, seed):
    """Generate a perturbed hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = spacing / 2
    row = 0
    while idx < N and y < 1.0 - spacing/2:
        x = spacing / 2 + (row % 2) * spacing / 2
        while x < 1.0 - spacing/2 and idx < N:
            centers[idx, 0] = x + rng.uniform(-0.01, 0.01)
            centers[idx, 1] = y + rng.uniform(-0.01, 0.01)
            idx += 1
            x += spacing
        y += spacing * np.sqrt(3) / 2
        row += 1
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    return np.clip(centers, 0.02, 0.98)

def run_packing():
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Diverse multi-start search
    inits = []
    for s in np.linspace(0.13, 0.24, 14):
        for seed in range(8):
            inits.append(get_init_hex(s, seed))
    for seed in range(30):
        rng = np.random.RandomState(seed)
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    for c0 in inits:
        r0 = np.full(N, 0.04)
        x0 = np.zeros(3*N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r0
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 18000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                cx = res.x[0::3]
                cy = res.x[1::3]
                cc = np.column_stack((cx, cy))
                r_lp, s_lp = solve_radii_lp(cc)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = cc
                    best_radii = r_lp
        except Exception:
            continue
            
    # Phase 2: Local refinement with adaptive cooling
    if best_centers is not None:
        rng = np.random.RandomState(42)
        for step in range(180):
            noise = 0.009 * (0.955 ** step)
            c_pert = best_centers + rng.normal(0, noise, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            
            r_pert, _ = solve_radii_lp(c_pert)
            x_pert = np.zeros(3*N)
            x_pert[0::3] = c_pert[:, 0]
            x_pert[1::3] = c_pert[:, 1]
            x_pert[2::3] = np.maximum(r_pert, 1e-6)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if res.success:
                    cc = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_opt, s_opt = solve_radii_lp(cc)
                    if s_opt > best_sum:
                        best_sum = s_opt
                        best_centers = cc
                        best_radii = r_opt
            except Exception:
                continue
                
        # Phase 3: Swap perturbations to escape topological local minima
        for _ in range(70):
            idx1, idx2 = rng.choice(N, 2, replace=False)
            c_swap = best_centers.copy()
            c_swap[[idx1, idx2]] = c_swap[[idx2, idx1]]
            r_swap, _ = solve_radii_lp(c_swap)
            x_swap = np.zeros(3*N)
            x_swap[0::3] = c_swap[:, 0]
            x_swap[1::3] = c_swap[:, 1]
            x_swap[2::3] = np.maximum(r_swap, 1e-6)
            
            try:
                res = minimize(objective, x_swap, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
                if res.success:
                    cc = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_opt, s_opt = solve_radii_lp(cc)
                    if s_opt > best_sum:
                        best_sum = s_opt
                        best_centers = cc
                        best_radii = r_opt
            except Exception:
                continue

    # Fallback safety net
    if best_centers is None:
        best_centers = get_init_hex(0.18, 0)
        best_radii, best_sum = solve_radii_lp(best_centers)
        
    # Phase 4: Strict post-processing to guarantee validity
    radii = best_radii.copy()
    for i in range(N):
        x, y = best_centers[i]
        mx = min(x, 1-x, y, 1-y)
        if radii[i] > mx - 1e-9:
            radii[i] = max(0.0, mx - 1e-9)
            
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], 
                             best_centers[i,1]-best_centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc/2
                    radii[j] -= exc/2
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return best_centers, radii, float(np.sum(radii))
