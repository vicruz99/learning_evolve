from scipy.optimize import minimize, linprog

N = 26

def get_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    triu_i, triu_j = np.triu_indices(n, k=1)
    m = len(triu_i)
    
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), triu_i] = 1.0
    A_ub[np.arange(m), triu_j] = 1.0
    b_ub = dists[triu_i, triu_j]
    
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            radii = np.maximum(res.x, 0.0)
            return radii, np.sum(radii)
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def objective_sim(params):
    """Negative sum of radii for simultaneous optimization."""
    return -np.sum(params[2 * N:])

def constraints_sim(params):
    """Smooth squared-distance and boundary constraints."""
    cx = params[0::3]
    cy = params[1::3]
    r = params[2::3]
    
    triu_i, triu_j = np.triu_indices(N, k=1)
    dx = cx[triu_i] - cx[triu_j]
    dy = cy[triu_i] - cy[triu_j]
    c_overlap = dx**2 + dy**2 - (r[triu_i] + r[triu_j])**2
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    return np.concatenate([c_overlap, c_bound])
