import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
N_PAIRS = len(I)

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((N_PAIRS, n))
    b_ub = np.zeros(N_PAIRS)
    
    # Compute pairwise distances
    dists = np.sqrt(((centers[:, None, :] - centers[None, :, :])**2).sum(axis=2))
    A_ub[np.arange(N_PAIRS), I] = 1.0
    A_ub[np.arange(N_PAIRS), J] = 1.0
    b_ub[:] = dists[I, J]
    
    # Boundary constraints: 0 <= r_i <= distance to nearest wall
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, ub - 1e-9)))
        
    # Solve LP
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='interior-point')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective_full(x):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def constraints_full(x):
    """Constraints for SLSQP: boundary and squared non-overlap."""
    c = x[:2*N].reshape(N, 2)
    r = x[2*N:]
    
    # Boundary constraints
    b = np.empty(4*N)
    b[:N] = c[:, 0] - r
    b[N:2*N] = 1.0 - c[:, 0] - r
    b[2*N:3*N] = c[:, 1] - r
    b[3*N:4*N] = 1.0 - c[:, 1] - r
    
    # Squared distance constraints (avoids sqrt gradient issues)
    diff = c[I] - c[J]
    d_sq = (diff**2).sum(axis=1)
    r_sum = r[I] + r[J]
    o = d_sq - r_sum**2
    
    return np.concatenate([b, o])

def make_initialization(seed, style='hex'):
    """Generate structured or random initial center configurations."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    
    if style == 'hex':
        s = 0.13 + rng.uniform(0, 0.06)
        idx = 0
        row = 0
        y = s / 2
        while idx < N and y < 1.0 - s / 2:
            x_start = s / 2 + (row % 2) * s / 2
            col = 0
            while x_start + col * s < 1.0 - s / 2 and idx < N:
                centers[idx] = [x_start + col * s, y]
                idx += 1
                col += 1
            y += s * np.sqrt(3) / 2
            row += 1
        while idx < N:
            centers[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
    elif style == 'grid':
        step = 0.16 + rng.uniform(0, 0.04)
        idx = 0
        y = step
        while y < 1.0 and idx < N:
            x = step
            while x < 1.0 and idx < N:
                centers[idx] = [x, y]
                idx += 1
                x += step
            y += step
        while idx < N:
            centers[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
    else:
        centers = rng.uniform(0.1, 0.9, (N, 2))
        
    # Add small noise to break symmetry
    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_full}

            r_lp, s_lp = solve_radii_lp(cx)

                best_radii = r_lp.copy()

                best_radii = r_lp.copy()

            pass

        best_radii, best_sum = solve_radii_lp(best_centers)

    radii = best_radii.copy()

    for i in range(N):
        x, y = best_centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        if radii[i] > mx - 1e-9:
            radii[i] = max(0.0, mx - 1e-9)

        changed = False
        for i in range(N):
            for j in range(i + 1, N):

                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc / 2.0
                    radii[j] -= exc / 2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return best_centers, radii, float(np.sum(radii))
