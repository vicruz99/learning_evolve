import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints(x):
    """Inequality constraints: boundary and non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]

    c = np.empty(4 * N + N_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r

    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    dist = np.sqrt(np.maximum(dx**2 + dy**2, 1e-20))
    c[4*N:] = dist - r[I_IDX] - r[J_IDX]
    return c


def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    dists = pdist(centers)

    A_ub = np.zeros((N_PAIRS, n))
    A_ub[np.arange(N_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(N_PAIRS), J_IDX] = 1.0
    b_ub = dists

    bounds_r = []
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds_r.append((0.0, max(0.0, mx)))

    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success:
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(n), 0.0


def make_hex_lattice(spacing, margin, rng):
    """Create a hexagonal lattice arrangement of circle centers."""
    c = []
    y = margin
    row = 0
    while len(c) < N:
        xs = margin + (row % 2) * spacing / 2
        while xs < 1.0 - margin and len(c) < N:
            c.append([xs, y])
            xs += spacing
        y += spacing * np.sqrt(3) / 2
        row += 1
    while len(c) < N:
        c.append([0.5 + rng.uniform(-0.3, 0.3), 0.5 + rng.uniform(-0.3, 0.3)])
    c = np.array(c[:N])
    c += rng.normal(0, 0.004, c.shape)
    return np.clip(c, 0.01, 0.99)


def run_local_refinement(centers, radii, noise_scale, max_iter, rng):
    """Run local refinement from a given configuration."""
    best_c = centers.copy()
    best_r = radii.copy()
    best_s = np.sum(radii)

    for seed_offset in range(3):
        c_pert = best_c + rng.normal(0, noise_scale, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        r_pert, _ = solve_lp_radii(c_pert)

        x_pert = np.zeros(3 * N)
        x_pert[0::3] = c_pert[:, 0]
        x_pert[1::3] = c_pert[:, 1]
        x_pert[2::3] = np.maximum(r_pert * 0.97, 1e-5)

        try:
            cons_opt = {'type': 'ineq', 'fun': constraints}
            bounds_opt = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
            res = minimize(objective, x_pert, method='SLSQP', bounds=bounds_opt,
                          constraints=cons_opt,
                          options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False})

            curr_c = np.column_stack((res.x[0::3], res.x[1::3]))
            curr_r, curr_s = solve_lp_radii(curr_c)

            if curr_s > best_s:
                best_s = curr_s
                best_c = curr_c.copy()
                best_r = curr_r.copy()
        except Exception:
            pass

    return best_c, best_r, best_s
