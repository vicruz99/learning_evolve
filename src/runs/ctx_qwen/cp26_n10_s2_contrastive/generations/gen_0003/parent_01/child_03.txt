import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIRS_I)

# Precompute the constant part of the LP constraint matrix: r_i + r_j <= dist_ij
A_LP = np.zeros((NUM_PAIRS, N))
for k in range(NUM_PAIRS):
    A_LP[k, PAIRS_I[k]] = 1.0
    A_LP[k, PAIRS_J[k]] = 1.0

def get_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Compute pairwise distances for RHS of LP constraints
    dx = centers[PAIRS_I, 0] - centers[PAIRS_J, 0]
    dy = centers[PAIRS_I, 1] - centers[PAIRS_J, 1]
    b_ub = np.sqrt(dx**2 + dy**2)
    
    # Boundary constraints: 0 <= r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(centers, 1.0 - centers).min(axis=1)
    bounds = [(0.0, max(0.0, u)) for u in ub]
    
    try:
        res = linprog(c_obj, A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= -1e-7):
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective_joint(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_joint(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap (squared)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = []
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    dx = cx[PAIRS_I] - cx[PAIRS_J]
    dy = cy[PAIRS_I] - cy[PAIRS_J]
    rs = r[PAIRS_I] + r[PAIRS_J]
    c.append(dx**2 + dy**2 - rs**2)
    
    return np.concatenate(c)

                idx += 1

    rng_sa = np.random.default_rng(42)
    current_c = inits[0].copy()
    _, current_sum = get_lp_radii(current_c)
    
    step = 0.025
    temperature = 0.015
    
    # Simulated Annealing on centers
    for step_idx in range(800):
