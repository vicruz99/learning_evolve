import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist

N = 26

def get_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_mat = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    I, J = np.triu_indices(n, k=1)
    n_pairs = len(I)
    A_ub = np.zeros((n_pairs, n))
    A_ub[np.arange(n_pairs), I] = 1.0
    A_ub[np.arange(n_pairs), J] = 1.0
    b_ub = dists_mat[I, J]
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
    
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success:
                return np.maximum(res.x, 0.0)
        except Exception:
            continue
    return np.zeros(n)


def obj_func(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constr_func(x):
    """Inequality constraints: boundary and non-overlap (using hypot for better gradients)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    I, J = np.triu_indices(N, k=1)
    n_pairs = len(I)
    
    c = np.empty(4 * N + n_pairs)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dists = np.hypot(dx, dy)
    c[4*N:] = dists - r[I] - r[J]
    return c


def make_hex_init(spacing, seed=0, noise_scale=0.005):
    """Generate hexagonal lattice pattern."""
    rng = np.random.RandomState(seed)
    centers = []
    y = spacing / 2.0
    row = 0
    while len(centers) < N:
        x = spacing / 2.0 + (row % 2) * spacing / 2.0
        while x < 1.0 - spacing / 2.0 and len(centers) < N:
            centers.append([x + rng.normal(0, noise_scale),
                          y + rng.normal(0, noise_scale)])
            x += spacing
        y += spacing * np.sqrt(3) / 2.0
        row += 1
    c = np.array(centers[:N])
    return np.clip(c, 0.015, 0.985)


def make_row_pattern(row_counts, seed=0, noise_scale=0.005):
    """Generate circles with specified row counts, staggered like hex lattice."""
    rng = np.random.RandomState(seed)
    centers = []
    if sum(row_counts) < N:
        return np.zeros((N, 2))
    
    n_rows = len(row_counts)
    y_spacing = 0.85 / n_rows
    y = 0.05
    
    for row_idx, cnt in enumerate(row_counts):
        if len(centers) >= N:
            break
        shift = 0.0 if row_idx % 2 == 0 else y_spacing * 0.55
        x_spacing = 0.9 / (cnt + 1)
        x = 0.05 + shift
        
        for col in range(cnt):
            if len(centers) >= N:
                break
            cx = x + col * x_spacing + rng.normal(0, noise_scale)
            cy = y + rng.normal(0, noise_scale)
            centers.append([cx, cy])
        y += y_spacing
    
    while len(centers) < N:
        centers.append([0.5 + rng.normal(0, 0.05), 0.5 + rng.normal(0, 0.05)])
    
    c = np.array(centers[:N])
    return np.clip(c, 0.015, 0.985)


def force_relax(centers, radii, n_iter=200, target_extra=0.03):
    """Physics-based repulsion to spread circles apart."""
    c = centers.copy()
    r = radii.copy()
    
    for step in range(n_iter):
        F = np.zeros_like(c)
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = np.hypot(dx, dy)
                target = r[i] + r[j] + target_extra
                if d < target and d > 1e-10:
                    f = (target - d) / d
                    F[i, 0] += f * dx
                    F[i, 1] += f * dy
                    F[j, 0] -= f * dx
                    F[j, 1] -= f * dy
            
            for w in [0, 1]:
                if c[i, w] < r[i] + 1e-6:
                    F[i, w] += 20.0 * (r[i] + 1e-6 - c[i, w])
                if c[i, w] > 1.0 - r[i] - 1e-6:
                    F[i, w] -= 20.0 * (c[i, w] - (1.0 - r[i] - 1e-6))
        
        dt = 0.03 * max(0.01, 1.0 - step / n_iter)
        c += dt * F
        for i in range(N):
            c[i, 0] = max(r[i] + 1e-6, min(1.0 - r[i] - 1e-6, c[i, 0]))
            c[i, 1] = max(r[i] + 1e-6, min(1.0 - r[i] - 1e-6, c[i, 1]))
    
    return c
