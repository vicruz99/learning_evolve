N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIRS_I)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints: boundary and non-overlap (squared distance)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[PAIRS_I] - cx[PAIRS_J]
    dy = cy[PAIRS_I] - cy[PAIRS_J]
    r_sum = r[PAIRS_I] + r[PAIRS_J]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    return c

def get_lp_radii(centers):

    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]

    A_ub[np.arange(NUM_PAIRS), PAIRS_J] = 1.0
    b_ub = dists[PAIRS_I, PAIRS_J]

        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, mx)))

            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)

def make_hex_config(spacing, margin, seed, y_offset=0.0, x_phase=0.0):
