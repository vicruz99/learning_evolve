from scipy.optimize import minimize, linprog

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIRS_I)
BOUNDS = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_fn(x):
    """Inequality constraints using squared distances for smoothness."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4 * N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[PAIRS_I] - cx[PAIRS_J]
    dy = cy[PAIRS_I] - cy[PAIRS_J]
    r_sum = r[PAIRS_I] + r[PAIRS_J]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    return c

CONS = {'type': 'ineq', 'fun': constraints_fn}

def get_lp_radii(centers):

    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), PAIRS_I] = 1.0
    A_ub[np.arange(NUM_PAIRS), PAIRS_J] = 1.0
    b_ub = dists[PAIRS_I, PAIRS_J]
    
    bounds = []

        mx, my = centers[i]
        ub = min(mx, 1.0 - mx, my, 1.0 - my)
        bounds.append((0.0, max(0.0, ub)))
    
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)

        except Exception:
            continue
    return np.full(n, 0.01)

def make_hex_config(sx, margin, rng=None):
    """Create hexagonal lattice configuration."""
    centers = np.zeros((N, 2))
    idx = 0
    y = margin
    row = 0
    sy = sx * np.sqrt(3) / 2
    while y < 1.0 - margin and idx < N:
        x = margin + (row % 2) * sx / 2
        while x < 1.0 - margin and idx < N:
            centers[idx] = [x, y]
            idx += 1
            x += sx
        y += sy
        row += 1
    if rng is not None:
        while idx < N:
            centers[idx] = rng.uniform(margin, 1.0 - margin, 2)
            idx += 1
    return centers

def make_staggered_config(sx, sy, margin, rng=None):
    """Create staggered grid configuration."""
    centers = np.zeros((N, 2))
    idx = 0
    y = margin
    row = 0
    while y < 1.0 - margin and idx < N:
        x_off = margin + (row % 2) * sx / 2
        x = x_off
        while x < 1.0 - margin and idx < N:
            centers[idx] = [x, y]
            idx += 1
            x += sx
        y += sy
        row += 1
    if rng is not None:
        while idx < N:
            centers[idx] = rng.uniform(margin, 1.0 - margin, 2)
            idx += 1
    return centers

def run_opt(x0, maxiter=5000):
    """Run SLSQP optimization."""
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=BOUNDS,
                      constraints=CONS, options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if res.success:
            return res.x
    except Exception:
        pass
    return x0
