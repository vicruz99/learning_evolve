import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIRS_I)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints: boundary and non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[PAIRS_I] - cx[PAIRS_J]
    dy = cy[PAIRS_I] - cy[PAIRS_J]
    r_sum = r[PAIRS_I] + r[PAIRS_J]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    return c

def get_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((NUM_PAIRS, n))
    b_ub = np.zeros(NUM_PAIRS)
    
    for k in range(NUM_PAIRS):
        i, j = PAIRS_I[k], PAIRS_J[k]
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        d = math.sqrt(dx * dx + dy * dy)
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = d
        
    bounds = []
    for i in range(n):
        mx = centers[i, 0]
        my = centers[i, 1]
        ub = min(mx, 1.0 - mx, my, 1.0 - my)
        bounds.append((0.0, max(0.0, ub)))
        
    res = None
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method,
                         options={'presolve': True})
            if res.success and np.all(res.x >= -1e-10):
                return np.maximum(res.x, 0.0)
        except Exception:
            continue
    return np.full(n, 0.01)

def make_hex_init(seed, spacing=None):
    """Hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    if spacing is None:
        spacing = 0.165 + rng.uniform(-0.02, 0.03)
    
    pts = []
    y = spacing / 2
    row = 0
    while len(pts) < N:
        x = spacing / 2 + (row % 2) * spacing / 2
        while x < 1.0 - spacing / 2 and len(pts) < N:
            pts.append([x, y])
            x += spacing
        y += spacing * math.sqrt(3) / 2
        row += 1
    
    pts = np.array(pts[:N])
    pts += rng.normal(0, 0.01, pts.shape)
    pts = np.clip(pts, 0.03, 0.97)
    return pts

def make_pattern_init(pat, noise=0.01, seed=0):
    """Create initialization from a pattern of row counts."""
    rng = np.random.RandomState(seed)
    spacing = 0.17
    pts = []
    y = 0.05
    dy = spacing * math.sqrt(3) / 2
    
    for r_idx, cnt in enumerate(pat):
        shift = 0.0 if r_idx % 2 == 0 else spacing / 2
        x = 0.05 + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += spacing
        y += dy
    
    while len(pts) < N:
        pts.append([0.5, 0.5])
    pts = np.array(pts[:N])
    pts += rng.normal(0, noise, pts.shape)
    pts = np.clip(pts, 0.03, 0.97)
    return pts

def make_random_init(seed):
    """Random initialization."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.1, 0.9, (N, 2))
    return pts

def make_sequential_init(seed):
    """Sequential packing: place circles where most space is available."""
    rng = np.random.RandomState(seed)
    pts = []
    
    # Place first circle
    x = rng.uniform(0.2, 0.8)
    y = rng.uniform(0.2, 0.8)
    pts.append([x, y])
    
    for _ in range(N - 1):
        best_pos = None
        best_score = 0
        
        for _ in range(500):
            x = rng.uniform(0.05, 0.95)
            y = rng.uniform(0.05, 0.95)
            
            # Score: min distance to existing circles and boundaries
            min_dist_to_circles = 1.0
            for px, py in pts:
                d = math.hypot(x - px, y - py)
                min_dist_to_circles = min(min_dist_to_circles, d)
            
            min_dist_to_boundary = min(x, 1.0 - x, y, 1.0 - y)
            score = min(min_dist_to_circles, min_dist_to_boundary)
            
            if score > best_score:
                best_score = score
                best_pos = [x, y]
        
        if best_pos is not None:
            pts.append(best_pos)
        else:
            pts.append(rng.uniform(0.15, 0.85, 2))
    
    return np.array(pts)

def make_corner_init(seed):
    """Place larger circles near corners and edges."""
    rng = np.random.RandomState(seed)
    pts = np.zeros((N, 2))
    
    # Four corners with small offset
    pts[0] = [0.12, 0.12]
    pts[1] = [0.88, 0.12]
    pts[2] = [0.12, 0.88]
    pts[3] = [0.88, 0.88]
    
    # Along edges
    edge_pts = [
        [0.35, 0.12], [0.65, 0.12],
        [0.35, 0.88], [0.65, 0.88],
        [0.12, 0.35], [0.12, 0.65],
        [0.88, 0.35], [0.88, 0.65],
    ]
    for i, ep in enumerate(edge_pts):
        pts[i + 4] = ep
    
    # Fill remaining with grid-like pattern in center
    remaining = N - 12
    gx = np.linspace(0.3, 0.7, remaining)
    gy = np.linspace(0.3, 0.7, remaining)
    idx = 12
    for i in range(remaining):
        pts[i + idx if idx < N else idx] = [gx[i % remaining], gy[i // remaining if remaining > 1 else 0]]
    
    # Fill any remaining
    for i in range(4, N):
        if pts[i, 0] == 0 and pts[i, 1] == 0:
            pts[i] = rng.uniform(0.25, 0.75, 2)
    
    pts += rng.normal(0, 0.02, pts.shape)
    pts = np.clip(pts, 0.04, 0.96)
    return pts

def make_spiral_init(seed):
    """Spiral initialization from center."""
    rng = np.random.RandomState(seed)
    pts = np.zeros((N, 2))
    pts[0] = [0.5, 0.5]
    
    for i in range(1, N):
        angle = i * 2.399963 * math.pi  # golden angle
        r = 0.12 * math.sqrt(i)
        pts[i, 0] = 0.5 + r * math.cos(angle)
        pts[i, 1] = 0.5 + r * math.sin(angle)
    
    pts += rng.normal(0, 0.015, pts.shape)
    pts = np.clip(pts, 0.04, 0.96)
    return pts

def make_grid_init(seed):
    """Grid-based initialization."""
    rng = np.random.RandomState(seed)
    pts = []
    step = 0.19
    for i in range(7):
        for j in range(7):
            if len(pts) < N:
                pts.append([0.05 + i * step, 0.05 + j * step])
    
    pts = np.array(pts[:N])
    pts += rng.normal(0, 0.015, pts.shape)
    pts = np.clip(pts, 0.03, 0.97)
    return pts

def optimize_single(c_init, trial_seed, max_iters=8000):
    """Run SLSQP + LP alternating optimization on a single initialization."""
    rng = np.random.RandomState(trial_seed)
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': constraints}
    
    # Get initial radii via LP
    r_init = get_lp_radii(c_init)
    r_init = np.maximum(r_init, 1e-5)
    
    x0 = np.zeros(3 * N)
    x0[0::3] = c_init[:, 0]
    x0[1::3] = c_init[:, 1]
    x0[2::3] = r_init * 0.95  # Slightly shrink for feasibility
    
    curr_x = x0.copy()
    
    # Alternating SLSQP + LP for multiple rounds
    for round_num in range(4):
        # SLSQP step
        try:
            maxiter = max_iters // (round_num + 1)
            res = minimize(objective, curr_x, method='SLSQP', bounds=bounds_opt,
                          constraints=cons_opt, 
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
            if res.success or res.nit > 0:
                curr_x = res.x.copy()
        except Exception:
            pass
        
        # LP step to update radii
        curr_c = np.column_stack((curr_x[0::3], curr_x[1::3]))
        curr_r = get_lp_radii(curr_c)
        curr_r = np.maximum(curr_r, 1e-6)
        curr_x[2::3] = curr_r
    
    # Final SLSQP refinement
    try:
        res = minimize(objective, curr_x, method='SLSQP', bounds=bounds_opt,
                      constraints=cons_opt,
                      options={'maxiter': max_iters, 'ftol': 1e-15, 'disp': False})
        if res.success:
            curr_x = res.x.copy()
    except Exception:
        pass
    
    # Final LP for radii
    curr_c = np.column_stack((curr_x[0::3], curr_x[1::3]))
    curr_r = get_lp_radii(curr_c)
    curr_s = np.sum(curr_r)
    
    return curr_c, curr_r, curr_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """

    """
    best_sum = -1.0

    ]
    for pat in patterns:

            inits.append(make_pattern_init(pat, noise=ns, seed=np.random.randint(0, 100)))

    for seed in range(25):
        inits.append(make_random_init(seed))

    for seed in range(10):
        inits.append(make_spiral_init(seed))
    
    # Grid
    for seed in range(10):
        inits.append(make_grid_init(seed))
    
    # Main optimization loop with perturbations
    trial = 0
    max_trials = 250
    
    for base in inits:
        if trial >= max_trials:
            break
        
        # Try with different noise levels
        noise_levels = [0.0, 0.005, 0.015, 0.03, 0.05]

            rng = np.random.RandomState(trial * 31 + 7)
            c_init = base.copy()
            c_init += rng.normal(0, nl, base.shape)

                best_radii = curr_r.copy()

                for pr in range(8):
                    noise_scale = 0.008 * (0.85 ** pr)
                    c_pert = best_centers + rng.normal(0, noise_scale, N)
                    c_pert = np.clip(c_pert, 0.02, 0.98)
                    
                    rc, rr, rs = optimize_single(c_pert, trial * 100 + pr, max_iters=4000)

    c_final = best_centers.copy()
    r_final = best_radii.copy()
    
    # Enforce boundary constraints
    for i in range(N):

        ub = min(mx, 1.0 - mx, my, 1.0 - my)

                r_final[j] -= exc * 0.5
                r_final[i] = max(0.0, r_final[i])
                r_final[j] = max(0.0, r_final[j])
                changed = True
        if not changed:
            break
    
    return c_final, r_final, float(np.sum(r_final))
