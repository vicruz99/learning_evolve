import numpy as np
from scipy.optimize import minimize, linprog, basinhopping
import math

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIRS_I)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), PAIRS_I] = 1.0
    A_ub[np.arange(NUM_PAIRS), PAIRS_J] = 1.0
    
    dx = centers[PAIRS_I, 0] - centers[PAIRS_J, 0]
    dy = centers[PAIRS_I, 1] - centers[PAIRS_J, 1]
    b_ub = np.sqrt(dx**2 + dy**2)
    
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, mx)))
        
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return np.sum(res.x), np.maximum(res.x, 0.0)
        except Exception:
            continue
    return 0.0, np.zeros(n)

def objective_joint(x):

    return -np.sum(x[2::3])

def constraints_joint(x):

    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]

    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[PAIRS_I] - cx[PAIRS_J]
    dy = cy[PAIRS_I] - cy[PAIRS_J]
    dist = np.hypot(dx, dy)
    c[4*N:] = dist - (r[PAIRS_I] + r[PAIRS_J])
    return c

def lp_obj_for_bh(x_flat):
    """Objective for basin hopping: negative of LP-optimal sum of radii."""
    c = x_flat.reshape(N, 2)
    c = np.clip(c, 0.01, 0.99)
    s, _ = solve_lp_radii(c)
    return -s
