import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIRS_I)

# Precompute LP constraint matrix structure (constant for all configurations)
A_LP = np.zeros((NUM_PAIRS, N))
A_LP[np.arange(NUM_PAIRS), PAIRS_I] = 1.0
A_LP[np.arange(NUM_PAIRS), PAIRS_J] = 1.0
C_OBJ_LP = -np.ones(N)

def solve_lp_radii(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii.
    Constraints: r_i + r_j <= dist(i,j) and 0 <= r_i <= dist_to_wall.
    """
    dx = centers[PAIRS_I, 0] - centers[PAIRS_J, 0]
    dy = centers[PAIRS_I, 1] - centers[PAIRS_J, 1]
    dists = np.hypot(dx, dy)
    
    bounds = []
    for i in range(N):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, mx)))
        
    try:
        res = linprog(C_OBJ_LP, A_ub=A_LP, b_ub=dists, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(N, 0.02)

def coordinate_descent(init_centers, iterations=2000):
    """
    Optimizes centers by randomly perturbing them and accepting moves
    that increase the optimal sum of radii (computed via LP).
    """
    c = init_centers.copy()
    r = solve_lp_radii(c)
    best_sum = np.sum(r)
    
    for t in range(iterations):
        decay = 1.0 - t / iterations
        step = 0.12 * decay + 0.005
        
        i = np.random.randint(N)
        new_c = c.copy()
        
        # Local perturbation with occasional global jumps to escape minima
        if np.random.rand() < 0.03:
            new_c[i] = np.random.uniform(0, 1, 2)
        else:
            new_c[i] = np.clip(c[i] + np.random.uniform(-step, step, 2), 0, 1)
            
        new_r = solve_lp_radii(new_c)
        new_sum = np.sum(new_r)
        
        if new_sum > best_sum:
            c = new_c
            r = new_r
            best_sum = new_sum
            
    return c, r, best_sum

def slsqp_objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def slsqp_constraints(x):
    """Inequality constraints: g(x) >= 0"""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = []
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    dx = cx[PAIRS_I] - cx[PAIRS_J]
    dy = cy[PAIRS_I] - cy[PAIRS_J]
    d = np.hypot(dx, dy)
    c.append(d - r[PAIRS_I] - r[PAIRS_J])
    
    return np.concatenate(c)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_c = None
    best_r = None
    best_sum = -1.0
    
    rng = np.random.default_rng(42)
    inits = []
    
    # 1. Hexagonal lattice patterns with varying spacings
    for spacing in [0.14, 0.16, 0.18, 0.20]:
        pts = []
        y = spacing / 2
        row = 0
        while len(pts) < N:
            x_start = spacing / 2 + (row % 2) * spacing / 2
            col = 0
            while x_start + col * spacing < 1.0 - spacing / 2 and len(pts) < N:
                pts.append([x_start + col * spacing, y])
                col += 1
            y += spacing * np.sqrt(3) / 2
            row += 1
        while len(pts) < N:
            pts.append([0.5, 0.5])
        inits.append(np.array(pts[:N]) + rng.normal(0, 0.01, (N, 2)))
        
    # 2. Corner-focused & edge-aligned patterns
    corner_pts = []
    margin = 0.08
    for x in [margin, 1-margin]:
        for y in [margin, 1-margin]:
            corner_pts.append([x, y])
    # Fill rest with grid
    y = 0.15
    while len(corner_pts) < N and y < 0.95:
        x = 0.15
        while x < 0.95 and len(corner_pts) < N:
            corner_pts.append([x, y])
            x += 0.18
        y += 0.18
    inits.append(np.array(corner_pts[:N]) + rng.normal(0, 0.015, (N, 2)))
    
    # 3. Random diverse placements
    for _ in range(5):
        inits.append(np.clip(rng.uniform(0.05, 0.95, (N, 2)) + rng.normal(0, 0.02, (N, 2)), 0, 1))
        
    # Phase 1 & 2: Coordinate Descent from diverse starts
    for init_c in inits:
        c_opt, r_opt, s_opt = coordinate_descent(init_c, iterations=1800)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            best_r = r_opt.copy()
            
    # Phase 3: SLSQP Polishing
    if best_c is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_c[:, 0]
        x0[1::3] = best_c[:, 1]
        x0[2::3] = best_r * 0.995  # Slightly shrink to ensure strict feasibility
        
        bounds_opt = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
        cons_opt = {'type': 'ineq', 'fun': slsqp_constraints}
        
        try:
            res = minimize(slsqp_objective, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons_opt, options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = solve_lp_radii(c_opt)
                s_opt = np.sum(r_opt)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            pass
            
    # Fallback safety
    if best_c is None:
        best_c = inits[0]
        best_r = solve_lp_radii(best_c)
        best_sum = np.sum(best_r)
        
    # Phase 4: Strict Post-Processing & Validation Repair
    c_final = best_c.copy()
    r_final = best_r.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(c_final[i, 0], 1.0 - c_final[i, 0], c_final[i, 1], 1.0 - c_final[i, 1])
        if r_final[i] > mx - 1e-9:
            r_final[i] = max(0.0, mx - 1e-9)
            
    # Iteratively resolve numerical overlaps
    for _ in range(50):
        changed = False
        for k in range(NUM_PAIRS):
            i, j = PAIRS_I[k], PAIRS_J[k]
            d = np.hypot(c_final[i, 0] - c_final[j, 0], c_final[i, 1] - c_final[j, 1])
            if d < r_final[i] + r_final[j] - 1e-11:
                excess = r_final[i] + r_final[j] - d
                r_final[i] -= excess * 0.5
                r_final[j] -= excess * 0.5
                changed = True
        if not changed:
            break
            
    r_final = np.maximum(r_final, 0.0)
    final_sum = float(np.sum(r_final))
    
    return c_final, r_final, final_sum
