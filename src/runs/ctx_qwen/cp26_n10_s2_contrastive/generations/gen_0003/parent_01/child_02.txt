import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

def get_lp_radii(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((N_PAIRS, n))
    A_ub[np.arange(N_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(N_PAIRS), J_IDX] = 1.0
    
    diff = centers[I_IDX] - centers[J_IDX]
    b_ub = np.hypot(diff[:, 0], diff[:, 1])
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    # Fallback solver
    res2 = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
    if res2.success:
        return np.maximum(res2.x, 0.0)
    return np.full(n, 0.01)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints: boundary and non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4 * N + N_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    diff = np.column_stack((cx, cy))
    d = np.hypot(diff[I_IDX, 0] - diff[J_IDX, 0], diff[I_IDX, 1] - diff[J_IDX, 1])
    c[4*N:] = d - r[I_IDX] - r[J_IDX]
    return c

def generate_hex_init(seed, spacing=0.17, offset=0.0):
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = 0.05 + rng.uniform(0, 0.02)
    row = 0
    while idx < N and y < 0.95:
        x = 0.05 + offset + (row % 2) * spacing / 2.0 + rng.uniform(0, 0.01)
        while x < 0.95 and idx < N:
            centers[idx] = [x, y]
            x += spacing
            idx += 1
        y += spacing * np.sqrt(3) / 2.0
        row += 1
    while idx < N:
        centers[idx] = rng.uniform(0.2, 0.8, 2)
        idx += 1
    return centers

def generate_force_directed_init(seed):
    rng = np.random.default_rng(seed)
    c = rng.uniform(0.1, 0.9, (N, 2))
    r = np.full(N, 0.06)
    for _ in range(300):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = np.hypot(dx, dy)
                if d < 1e-4: d = 1e-4
                target_d = r[i] + r[j] + 0.015
                if d < target_d:
                    repulse = (target_d - d) / d
                    forces[i, 0] += dx * repulse
                    forces[i, 1] += dy * repulse
                    forces[j, 0] -= dx * repulse
                    forces[j, 1] -= dy * repulse
        for i in range(N):
            if c[i, 0] < r[i] + 0.02: forces[i, 0] += 0.5
            elif c[i, 0] > 1.0 - r[i] - 0.02: forces[i, 0] -= 0.5
            if c[i, 1] < r[i] + 0.02: forces[i, 1] += 0.5
            elif c[i, 1] > 1.0 - r[i] - 0.02: forces[i, 1] -= 0.5
        c += forces * 0.005
        c = np.clip(c, 0.02, 0.98)
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    inits = []
    # Hexagonal variations
    for s in np.linspace(0.16, 0.20, 6):
        for off in [0.0, 0.05, 0.10]:
            inits.append(generate_hex_init(seed=len(inits), spacing=s, offset=off))
    # Grid variations
    for step in np.linspace(0.18, 0.22, 4):
        c = []
        y = 0.05
        while y < 0.95 and len(c) < N:
            x = 0.05
            while x < 0.95 and len(c) < N:
                c.append([x, y])
                x += step
            y += step
        inits.append(np.array(c[:N]))
    # Force-directed
    for _ in range(10):
        inits.append(generate_force_directed_init(seed=_))
    # Random
    for _ in range(10):
        rng = np.random.default_rng(_ * 7 + 1)
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    trial = 0
    max_trials = 60
    
    for base in inits:
        if trial >= max_trials:
            break
        for nl in [0.0, 0.015, 0.04]:
            if trial >= max_trials:
                break
                
            rng = np.random.default_rng(trial * 31 + 7)
            c_init = base + rng.normal(0, nl, base.shape)
            c_init = np.clip(c_init, 0.02, 0.98)
            
            x0 = np.zeros(3 * N)
            x0[0::3] = c_init[:, 0]
            x0[1::3] = c_init[:, 1]
            
            # Initial radii from LP
            r_init = get_lp_radii(c_init) * 0.95
            x0[2::3] = np.maximum(r_init, 1e-5)
            
            curr_x = x0.copy()
            
            # SLSQP -> LP refinement loop
            for _ in range(2):
                try:
                    res = minimize(objective, curr_x, method='SLSQP', bounds=bounds_opt,
                                   constraints=cons_opt, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                    if res.success:
                        curr_x = res.x
                except Exception:
                    pass
                    
                curr_c = np.column_stack((curr_x[0::3], curr_x[1::3]))
                curr_r = get_lp_radii(curr_c)
                curr_x[2::3] = np.maximum(curr_r, 1e-5)
                
            # Final polish
            try:
                res = minimize(objective, curr_x, method='SLSQP', bounds=bounds_opt,
                               constraints=cons_opt, options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
                if res.success:
                    curr_c = np.column_stack((res.x[0::3], res.x[1::3]))
                    curr_r = get_lp_radii(curr_c)
                    curr_s = np.sum(curr_r)
                    if curr_s > best_sum:
                        best_sum = curr_s
                        best_centers = curr_c.copy()
                        best_radii = curr_r.copy()
            except Exception:
                pass
                
            trial += 1

    # Local refinement on best
    if best_centers is not None:
        for step in range(30):
            rng = np.random.default_rng(step * 13 + 3)
            sigma = 0.008 * (0.85 ** (step // 10))
            c_pert = best_centers + rng.normal(0, sigma, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            
            x0_p = np.zeros(3 * N)
            x0_p[0::3] = c_pert[:, 0]
            x0_p[1::3] = c_pert[:, 1]
            r_pert = get_lp_radii(c_pert) * 0.98
            x0_p[2::3] = np.maximum(r_pert, 1e-5)
            
            try:
                res_p = minimize(objective, x0_p, method='SLSQP', bounds=bounds_opt,
                                 constraints=cons_opt, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                if res_p.success:
                    c_opt = np.column_stack((res_p.x[0::3], res_p.x[1::3]))
                    r_opt = get_lp_radii(c_opt)
                    s_opt = np.sum(r_opt)
                    if s_opt > best_sum:
                        best_sum = s_opt
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
            except Exception:
                pass
                
    # Fallback
    if best_centers is None:
        best_centers = generate_hex_init(0)
        best_radii = get_lp_radii(best_centers)
        best_sum = np.sum(best_radii)
        
    # Strict post-processing
    c_final = best_centers.copy()
    r_final = best_radii.copy()
    
    for i in range(N):
        mx, my = c_final[i]
        max_r = min(mx, 1.0 - mx, my, 1.0 - my)
        r_final[i] = min(r_final[i], max_r - 1e-9)
        r_final[i] = max(r_final[i], 0.0)
        
    for _ in range(50):
        changed = False
        for k in range(N_PAIRS):
            i, j = I_IDX[k], J_IDX[k]
            d = np.hypot(c_final[i, 0] - c_final[j, 0], c_final[i, 1] - c_final[j, 1])
            if d < r_final[i] + r_final[j] - 1e-12:
                exc = r_final[i] + r_final[j] - d
                r_final[i] -= exc * 0.5
                r_final[j] -= exc * 0.5
                r_final[i] = max(r_final[i], 0.0)
                r_final[j] = max(r_final[j], 0.0)
                changed = True
        if not changed:
            break
            
    return c_final, r_final, float(np.sum(r_final))
