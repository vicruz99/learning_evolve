import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise distance matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    # LP constraints: r_i + r_j <= dist_ij
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(NUM_PAIRS), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]
    
    # Boundary constraints: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, mx)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def de_objective(centers_flat):
    """Objective for Differential Evolution: maximize LP-optimal sum of radii."""
    c = centers_flat.reshape(N, 2)
    r, s = solve_radii_lp(c)
    if np.isnan(s) or s < 0:
        return 1e6  # Penalty for invalid configurations
    return -s

def joint_objective(x):
    """SLSQP objective: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def joint_constraints(x):
    """SLSQP constraints: boundary clearance and squared non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    r_sum = r[I_IDX] + r[J_IDX]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    return c

def generate_structured_inits():

    inits = []
    
    # 1. Hexagonal lattices with varying spacing
    for sp in np.linspace(0.14, 0.20, 8):

        idx = 0
        y = sp/2
        row = 0
        while idx < N and y < 1.0 - sp/2:
            x = sp/2 + (row % 2) * sp/2
            while x < 1.0 - sp/2 and idx < N:
                centers[idx] = [x, y]
                x += sp
                idx += 1
            y += sp * np.sqrt(3) / 2
            row += 1
        while idx < N:
            centers[idx] = [0.5, 0.5]
            idx += 1
        inits.append(np.clip(centers, 0.05, 0.95))
        
    # 2. Grid patterns
    for sp in np.linspace(0.18, 0.24, 6):
        centers = np.zeros((N, 2))
        idx = 0
        y = sp
        while y < 1.0 and idx < N:
            x = sp
            while x < 1.0 and idx < N:
                centers[idx] = [x, y]
                x += sp
                idx += 1
            y += sp
        while idx < N:
            centers[idx] = [0.5, 0.5]
            idx += 1
        inits.append(np.clip(centers, 0.05, 0.95))
        
    # 3. Corner-heavy + hex fill
    centers = np.zeros((N, 2))
    centers[0] = [0.05, 0.05]
    centers[1] = [0.95, 0.05]
    centers[2] = [0.05, 0.95]
    centers[3] = [0.95, 0.95]
    idx = 4

        inits.append(c)

        x0[1::3] = c0[:, 1]
        x0[2::3] = np.maximum(r0, 1e-5)

        pass

    if best_c is not None:

        for i in range(N):
            mx = min(best_c[i,0], 1.0 - best_c[i,0], best_c[i,1], 1.0 - best_c[i,1])

        best_c = np.random.uniform(0.2, 0.8, (N, 2))
        best_r = np.full(N, 0.02)
        best_sum = np.sum(best_r)
        
    return best_c, best_r, float(best_sum)
