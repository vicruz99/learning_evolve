from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

# Precompute constant constraint matrix structure for LP: r_i + r_j <= d_ij
A_UUB = np.zeros((NUM_PAIRS, N))
A_UUB[np.arange(NUM_PAIRS), I_IDX] = 1.0
A_UUB[np.arange(NUM_PAIRS), J_IDX] = 1.0

def get_lp_radii(centers):
    """Solve LP to find optimal radii for fixed centers."""
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[I_IDX, J_IDX]
    
    bounds = []
    for i in range(N):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(-np.ones(N), A_ub=A_UUB, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(N), 0.0

def joint_objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2 * N:])

def joint_constraints(x):

    c = x[:2 * N].reshape(N, 2)
    r = x[2 * N:]
    
    cons = []
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
