N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)


def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(NUM_PAIRS), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]
    
    # Boundary bounds for each radius
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, mx)))
    
    # Try multiple LP solvers
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success and np.all(res.x >= -1e-10):
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    
    return np.zeros(n), 0.0


def objective(params):
    """Minimize negative sum of radii."""
    return -np.sum(params[2 * N:])


def constraints_sq(params):
    """Squared distance constraints for better numerical behavior."""

    cons = []

    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    # Squared non-overlap: dist^2 >= (r_i + r_j)^2
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist_sq = diff[:, :, 0]**2 + diff[:, :, 1]**2

    cons.append(dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2)
    
    return np.concatenate(cons)


def constraints_lin(params):
    """Linear distance constraints (dist >= r_i + r_j)."""
    c = params[:2 * N].reshape(N, 2)
    r = params[2 * N:]
    
    cons = []
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)

    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    cons.append(dists[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX])
    
    return np.concatenate(cons)


def generate_hex_init(spacing, margin, shift_x=0.0, shift_y=0.0, rng=None):
    """Generate hexagonal lattice initialization."""
    if rng is None:
        rng = np.random.RandomState(0)
    
    centers = np.zeros((N, 2))
    idx = 0
    row = 0
    y = margin + shift_y + row * spacing * np.sqrt(3) / 2.0
    
    while idx < N and y < 1.0 - margin:
        x_start = margin + shift_x + (row % 2) * spacing / 2.0
        col = 0
        while x_start + col * spacing < 1.0 - margin and idx < N:
            centers[idx, 0] = x_start + col * spacing
            centers[idx, 1] = y
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2.0
        row += 1
    
    # Fill remaining with random
    while idx < N:
        centers[idx] = rng.uniform(margin, 1.0 - margin, 2)
        idx += 1
    
    return centers


def generate_grid_init(step, margin, rng=None):
    """Generate grid initialization."""
    if rng is None:
        rng = np.random.RandomState(0)
    
    centers = np.zeros((N, 2))
    idx = 0
    y = margin
    while y < 1.0 - margin and idx < N:
        x = margin
        while x < 1.0 - margin and idx < N:
            centers[idx] = [x, y]
            x += step
            idx += 1
        y += step
    
    while idx < N:
        centers[idx] = rng.uniform(margin, 1.0 - margin, 2)
        idx += 1
    
    return centers


def make_feasible_centers_and_radii(centers):
    """Make centers strictly feasible and compute initial radii."""
    n = centers.shape[0]
    r = np.zeros(n)
    
    # Scale down radii based on distances to boundaries and neighbors
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        r[i] = mx * 0.9
    
    # Further reduce based on pairwise distances
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            if d < r[i] + r[j] + 0.01:
                scale = d / (r[i] + r[j] + 0.01)
                r[i] *= scale
                r[j] *= scale
    
    return centers, np.maximum(r, 1e-6)
