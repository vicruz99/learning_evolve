

Looking at this problem, I need to pack 26 circles in a unit square to maximize the sum of radii. The current best is ~2.626, and I'm targeting 2.636.

<strategy>
1. **More diverse initializations**: Hexagonal lattice (optimal in 2D), grid, corner-biased patterns with varied spacing/offsets
2. **Multi-phase optimization**: SLSQP for joint center-radius -> LP for radii-only -> SLSQP again -> LP final
3. **Intensive local search**: Multi-scale perturbation around best found, with cooling schedule from large to small steps
4. **Better LP formulation**: HiGHS solver fallback to interior-point, proper bound handling
5. **Numerical robustness**: Strict post-processing to handle floating-point violations
6. **Constraint handling**: Use distance directly (not squared) for better conditioning with SLSQP

Key insight: The LP step (fixed centers, optimize radii) is fast and gives the true optimal radii for any center configuration. The bottleneck is finding good centers. By alternating SLSQP (which moves centers and radii) with LP (which optimally recalculates radii), we can escape shallow local minima.

The local search uses a cooling schedule - starting with larger perturbations to explore broadly, then finer perturbations to polish the solution.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)


def compute_lp_radii(centers):
    """Solve LP to find optimal radii for fixed centers, maximizing sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)

    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])

    A_ub = np.zeros((N_PAIRS, n))
    A_ub[np.arange(N_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(N_PAIRS), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]

    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, mx)))

    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            pass
    return np.zeros(n), 0.0


def objective(params):
    """Minimize negative sum of radii."""
    return -np.sum(params[2 * N:])


def constraints(params):
    """Boundary and non-overlap constraints (must be >= 0)."""
    c = params[:2 * N].reshape(N, 2)
    r = params[2 * N:]

    cons = [
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r,
    ]

    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]

    cons.append(dists[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX])

    return np.concatenate(cons)


def generate_hex_centers(seed):
    """Generate hexagonal lattice centers with random spacing/offset."""
    rng = np.random.RandomState(seed)
    c = np.zeros((N, 2))
    idx = 0
    y = 0.03 + rng.uniform(0, 0.02)
    row = 0
    sp = 0.15 + rng.uniform(0, 0.07)
    margin = 0.02 + rng.uniform(0, 0.03)
    while idx < N and y < 1.0 - margin:
        x = margin + (row % 2) * sp / 2.0
        while x < 1.0 - margin and idx < N:
            c[idx] = [x + rng.uniform(-0.005, 0.005), y + rng.uniform(-0.005, 0.005)]
            x += sp
            idx += 1
        y += sp * np.sqrt(3) / 2.0
        row += 1
    while idx < N:
        c[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    return np.clip(c, 0.01, 0.99)


def generate_grid_centers(seed):
    """Generate grid centers with random spacing."""
    rng = np.random.RandomState(seed)
    c = np.zeros((N, 2))
    idx = 0
    step = 0.16 + rng.uniform(0, 0.07)
    margin = 0.03 + rng.uniform(0, 0.03)
    y = margin
    while y < 1.0 - margin and idx < N:
        x = margin
        while x < 1.0 - margin and idx < N:
            c[idx] = [x + rng.uniform(-0.005, 0.005), y + rng.uniform(-0.005, 0.005)]
            x += step
            idx += 1
        y += step
    while idx < N:
        c[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    return np.clip(c, 0.01, 0.99)


def generate_random_centers(seed):
    """Generate random centers."""
    rng = np.random.RandomState(seed)
    return rng.uniform(0.1, 0.9, (N, 2))


def generate_corner_bias_centers(seed):
    """Generate centers biased toward corners and edges for tighter packing."""
    rng = np.random.RandomState(seed)
    c = np.zeros((N, 2))
    idx = 0

    # Place 4 circles near corners
    corners = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
    for corner in corners:
        if idx < N:
            c[idx] = [corner[0] + rng.uniform(-0.02, 0.02),
                      corner[1] + rng.uniform(-0.02, 0.02)]
            idx += 1

    # Place circles along edges
    for edge in range(4):
        remaining = N - idx
        count = min(4, remaining)
        for k in range(count):
            if idx < N:
                t = 0.3 + k * 0.2 + rng.uniform(-0.03, 0.03)
                if edge == 0:  # bottom
                    c[idx] = [t, 0.08 + rng.uniform(-0.01, 0.01)]
                elif edge == 1:  # top
                    c[idx] = [t, 0.92 + rng.uniform(-0.01, 0.01)]
                elif edge == 2:  # left
                    c[idx] = [0.08 + rng.uniform(-0.01, 0.01), t]
                else:  # right
                    c[idx] = [0.92 + rng.uniform(-0.01, 0.01), t]
                idx += 1

    # Fill remaining randomly in interior
    while idx < N:
        c[idx] = rng.uniform(0.2, 0.8, 2)
        idx += 1

    return np.clip(c, 0.01, 0.99)


def generate_dense_hex_centers(seed):
    """Generate denser hexagonal lattice (more rows, tighter spacing)."""
    rng = np.random.RandomState(seed)
    c = np.zeros((N, 2))
    idx = 0
    y = 0.025
    row = 0
    sp = 0.13 + rng.uniform(0, 0.05)
    while idx < N and y < 0.975:
        x = 0.025 + (row % 2) * sp / 2.0
        while x < 0.975 and idx < N:
            c[idx] = [x, y]
            x += sp
            idx += 1
        y += sp * np.sqrt(3) / 2.0
        row += 1
    while idx < N:
        c[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    c += rng.randn(N, 2) * 0.003
    return np.clip(c, 0.01, 0.99)


def run_slsqp_opt(centers, radii, maxiter=12000, ftol=1e-13):
    """Run SLSQP optimization."""
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': constraints}

    x0 = np.concatenate([centers.flatten(), radii])

    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons_dict,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        if res.success:
            co = res.x[:2 * N].reshape(N, 2)
            ro = np.maximum(res.x[2 * N:], 0.0)
            return co, ro
    except Exception:
        pass
    return None, None


def run_packing():
    best_sum = 0.0
    best_c = None
    best_r = None

    # Phase 1: Broad multi-start search
    for seed in range(120):
        if seed < 50:
            c0 = generate_hex_centers(seed)
        elif seed < 70:
            c0 = generate_dense_hex_centers(seed)
        elif seed < 85:
            c0 = generate_grid_centers(seed)
        elif seed < 100:
            c0 = generate_corner_bias_centers(seed)
        else:
            c0 = generate_random_centers(seed)

        r0, _ = compute_lp_radii(c0)
        r0 = r0 * 0.97

        # SLSQP optimization
        co, ro = run_slsqp_opt(c0, r0, maxiter=12000)

        if co is not None:
            # LP refinement
            ro, s = compute_lp_radii(co)
            if s > best_sum:
                best_sum = s
                best_c = co.copy()
                best_r = ro.copy()

                # SLSQP again starting from LP radii
                co2, ro2 = run_slsqp_opt(co, ro * 0.99, maxiter=8000)
                if co2 is not None:
                    ro2, s2 = compute_lp_radii(co2)
                    if s2 > best_sum:
                        best_sum = s2
                        best_c = co2.copy()
                        best_r = ro2.copy()

    # Phase 2: Intensive local search with cooling schedule
    if best_c is not None:
        pert_scales = [
            0.010, 0.008, 0.006, 0.005, 0.004, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001
        ]
        for ps_idx, pert_scale in enumerate(pert_scales):
            for trial in range(40):
                rng = np.random.RandomState(trial * 31 + ps_idx * 1000 + 7)
                cp = best_c + rng.randn(N, 2) * pert_scale
                cp = np.clip(cp, 0.01, 0.99)

                rp, _ = compute_lp_radii(cp)
                rp = rp * 0.99

                co, ro = run_slsqp_opt(cp, rp, maxiter=8000, ftol=1e-13)

                if co is not None:
                    ro, s = compute_lp_radii(co)
                    if s > best_sum:
                        best_sum = s
                        best_c = co.copy()
                        best_r = ro.copy()

                        # Second pass refinement
                        co2, ro2 = run_slsqp_opt(co, ro * 0.995, maxiter=6000, ftol=1e-14)
                        if co2 is not None:
                            ro2, s2 = compute_lp_radii(co2)
                            if s2 > best_sum:
                                best_sum = s2
                                best_c = co2.copy()
                                best_r = ro2.copy()

    # Phase 3: Ultra-fine local search around absolute best
    if best_c is not None:
        for trial in range(60):
            rng = np.random.RandomState(trial * 43 + 2024)
            # Very small random perturbations
            perturb = rng.choice([0, 1, 2], size=2 * N, p=[0.7, 0.15, 0.15])
            pert_magnitude = rng.uniform(1e-4, 5e-4)
            cp = best_c.copy()
            for i in range(N):
                if perturb[2 * i] == 1:
                    cp[i, 0] += rng.uniform(-1, 1) * pert_magnitude
                if perturb[2 * i + 1] == 1:
                    cp[i, 1] += rng.uniform(-1, 1) * pert_magnitude
            cp = np.clip(cp, 0.01, 0.99)

            rp, _ = compute_lp_radii(cp)
            rp = rp * 0.999

            co, ro = run_slsqp_opt(cp, rp, maxiter=4000, ftol=1e-14)

            if co is not None:
                ro, s = compute_lp_radii(co)
                if s > best_sum:
                    best_sum = s
                    best_c = co.copy()
                    best_r = ro.copy()

    # Phase 4: Strict post-processing to guarantee validity
    if best_c is not None:
        radii = best_r.copy()

        # Enforce boundary constraints strictly
        for i in range(N):
            x, y = best_c[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if radii[i] > max_r - 1e-9:
                radii[i] = max(0.0, max_r - 1e-9)

        # Iteratively resolve any remaining numerical overlaps
        for _ in range(300):
            changed = False
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(best_c[i, 0] - best_c[j, 0],
                                 best_c[i, 1] - best_c[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        exc = radii[i] + radii[j] - d
                        radii[i] -= exc / 2.0
                        radii[j] -= exc / 2.0
                        changed = True
            if not changed:
                break

        radii = np.maximum(radii, 0.0)
        best_r = radii
        best_sum = float(np.sum(radii))

    # Fallback (should not be reached)
    if best_c is None:
        best_c = np.random.uniform(0.2, 0.8, (N, 2))
        best_r, best_sum = compute_lp_radii(best_c)

    return best_c, best_r, float(best_sum)
