import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    A_ub = np.zeros((N_PAIRS, n))
    A_ub[np.arange(N_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(N_PAIRS), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]
    
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, ub)))
        
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success:
                return np.maximum(res.x, 0.0)
        except Exception:
            continue
    return np.zeros(n)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2 * N:])

def constraints(x):
    """Squared distance constraints for better numerical behavior."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4 * N + N_PAIRS)
    
    # Boundary constraints
    c[0:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    # Squared distance constraints: (xi-xj)^2 + (yi-yj)^2 >= (ri+rj)^2
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    r_sum = r[I_IDX] + r[J_IDX]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    
    return c

def make_hex_centers(seed, spacing, margin, row_pattern=None):
    """Generate hexagonal lattice centers with optional row pattern."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))

                centers_out.append([x, y])
                x += spacing
            y += spacing * np.sqrt(3) / 2.0
            row += 1
        centers[:len(centers_out)] = centers_out[:N]
        if len(centers_out) < N:

                centers[i] = rng.uniform(0.1, 0.9, 2)
    else:

        y = margin

                    x_start = max(margin, x_start)

def make_grid_centers(seed, step, margin):
    """Generate grid lattice centers."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = margin
    while y < 1.0 - margin and idx < N:
        x = margin
        while x < 1.0 - margin and idx < N:
            centers[idx] = [x, y]
            x += step
            idx += 1
        y += step
    while idx < N:
        centers[idx] = rng.uniform(0.15, 0.85, 2)
        idx += 1
    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.02, 0.98)

def make_random_centers(seed):
    """Generate random centers."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.1, 0.9, (N, 2))
    return centers

def optimize_from_start(c0, maxiter=10000):
    """Run SLSQP optimization followed by LP radii refinement."""
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    r0 = solve_lp_radii(c0)
    x0 = np.zeros(3 * N)
    x0[0::3] = c0[:, 0]
    x0[1::3] = c0[:, 1]
    x0[2::3] = np.maximum(r0 * 0.95, 1e-6)
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        
        co = res.x[0::3]
        yo = res.x[1::3]
        centers = np.column_stack((co, yo))
        
        # LP radii refinement
        r_opt = solve_lp_radii(centers)
        s_opt = np.sum(r_opt)
        
        return centers, r_opt, s_opt
    except Exception:
        return None, None, 0.0
