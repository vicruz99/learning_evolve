import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(IDX_I)

def get_lp_radii(centers):
    """Solve LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Vectorized distance calculation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:,:,0], diff[:,:,1])
    b_ub = dists[IDX_I, IDX_J]
    
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), IDX_I] = 1.0
    A_ub[np.arange(NUM_PAIRS), IDX_J] = 1.0
    
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.zeros(n)

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def constraints(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap."""
    cx = x[0:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[0:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[IDX_I] - cx[IDX_J]
    dy = cy[IDX_I] - cy[IDX_J]
    dist = np.hypot(dx, dy)
    c[4*N:] = dist - r[IDX_I] - r[IDX_J]
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0)]*(2*N) + [(1e-8, 0.5)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_c = None
    best_r = None
    rng = np.random.RandomState(42)

    inits = []

    for s in np.linspace(0.155, 0.215, 7):

            idx = 0
            y = margin + s/2
            row = 0
            while idx < N and y < 1.0 - margin - s/2:
                x_start = margin + s/2 + (row % 2) * s/2
                col = 0
                while idx < N and x_start + col * s < 1.0 - margin - s/2:
                    c[idx] = [x_start + col * s, y]
                    idx += 1
                    col += 1
                y += s * math.sqrt(3) / 2.0
                row += 1
            while idx < N:
                c[idx] = rng.uniform(margin, 1.0-margin, 2)
                idx += 1
            inits.append(c)
            
    # 2. Repulsion-initialized random configurations
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        # Simple iterative repulsion to spread points
        for _ in range(60):
            for i in range(N):
                dx = c[i] - c
                d = np.hypot(dx[:,0], dx[:,1])
                d[i] = 1.0
                # Soft repulsion force
                force = dx / (d**2 + 0.005)
                c[i] += 0.003 * force[i]
                c[i] = np.clip(c[i], 0.05, 0.95)
        inits.append(c)
        
    # 3. Explicit corner/edge packing seeds
    corners = np.array([[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]])
    edges = np.array([
        [0.5, 0.08], [0.3, 0.08], [0.7, 0.08],
        [0.5, 0.92], [0.3, 0.92], [0.7, 0.92],
        [0.08, 0.5], [0.92, 0.5]
    ])
    interior = rng.uniform(0.35, 0.65, (N-12, 2))
    c_corner = np.vstack([corners, edges, interior])
    c_corner += rng.normal(0, 0.012, c_corner.shape)
    c_corner = np.clip(c_corner, 0.05, 0.95)
    inits.append(c_corner)

    for trial, c0 in enumerate(inits):

                co = res.x[:2*N].reshape(N, 2)
                ro = np.maximum(res.x[2*N:], 0.0)
                s = np.sum(ro)

                    best_sum = s
                    best_c = co.copy()
                    best_r = ro.copy()

            mx = min(best_c[i,0], 1.0-best_c[i,0], best_c[i,1], 1.0-best_c[i,1])
            best_r[i] = min(best_r[i], mx - 1e-9)
            best_r[i] = max(best_r[i], 0.0)
            
        # Iteratively resolve any remaining numerical overlaps
        for _ in range(150):
            changed = False
            for i in range(N):
                for j in range(i+1, N):
                    d = math.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                    if d < best_r[i] + best_r[j] - 1e-10:
                        exc = best_r[i] + best_r[j] - d
                        best_r[i] -= exc / 2.0
                        best_r[j] -= exc / 2.0
                        changed = True
            if not changed:
                break
                
        # Final LP pass to guarantee maximal radii for the converged centers
        final_r = get_lp_radii(best_c)
        best_r = final_r
        best_sum = float(np.sum(best_r))
        
    # Fallback safety net
    if best_c is None:
        best_c = np.random.uniform(0.15, 0.85, (N, 2))
        best_r = get_lp_radii(best_c)
        best_sum = float(np.sum(best_r))
        
    return best_c, best_r, best_sum
