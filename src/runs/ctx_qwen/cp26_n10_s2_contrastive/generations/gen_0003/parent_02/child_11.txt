N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

def solve_radii_lp(centers):

    return np.zeros(n), 0.0

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):

    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    
    if spacing is None:
        spacing = 0.14 + rng.uniform(0, 0.08)
    
    idx = 0
    row = 0
    y = spacing / 2 + rng.uniform(0, 0.04)

        x_start = spacing / 2 + (row % 2) * spacing / 2 + rng.uniform(0, 0.02)
        col = 0

            col += 1
        y += spacing * np.sqrt(3) / 2
        row += 1

        idx += 1

        idx += 1

def optimize_from(x0, maxiter=15000, ftol=1e-14):
    """Run SLSQP optimization from given starting point."""
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})

            cx = res.x[0::3]
            cy = res.x[1::3]
            r = res.x[2::3]
            centers = np.column_stack((cx, cy))
            r_lp, s_lp = solve_radii_lp(centers)
            return centers, r_lp, s_lp
    except Exception:
        pass
    
    return None, None, 0.0
