from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

def compute_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    dx = centers[I_IDX, 0] - centers[J_IDX, 0]
    dy = centers[I_IDX, 1] - centers[J_IDX, 1]
    dists = np.hypot(dx, dy)

    A_ub[np.arange(NUM_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(NUM_PAIRS), J_IDX] = 1.0
    b_ub = dists

    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, mx)))
        
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success:
                return np.maximum(res.x, 0.0)
        except Exception:
            continue
    return np.zeros(n)

def objective(params):
    """Minimize negative sum of radii."""
    return -np.sum(params[2*N:])

def constraints(params):

    cx = params[:N]
    cy = params[N:2*N]
    r = params[2*N:]

    c[:N] = cx - r

    dy = cy[I_IDX] - cy[J_IDX]
    c[4*N:] = np.hypot(dx, dy) - r[I_IDX] - r[J_IDX]
    return c

def make_hex_init(seed, spacing=None):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    if spacing is None:
        spacing = 0.16 + rng.uniform(-0.02, 0.02)

    idx = 0
    y = spacing / 2.0
    row = 0
    while idx < N and y < 1.0 - spacing/2.0:

        col = 0
        while x_start + col * spacing < 1.0 - spacing/2.0 and idx < N:
            centers[idx, 0] = x_start + col * spacing
            centers[idx, 1] = y
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2.0
        row += 1
        
    while idx < N:
        centers[idx] = rng.uniform(0.2, 0.8, 2)
        idx += 1
        
    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.01, 0.99)

def make_grid_init(seed):
    """Generate a square grid initialization."""
    rng = np.random.RandomState(seed)
    step = 0.18 + rng.uniform(-0.03, 0.03)
    centers = np.zeros((N, 2))
    idx = 0
    y = step / 2.0
    while y < 1.0 - step/2.0 and idx < N:
        x = step / 2.0

            centers[idx] = [x, y]
            x += step
            idx += 1
        y += step

        centers[idx] = rng.uniform(0.2, 0.8, 2)
        idx += 1
    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.01, 0.99)
