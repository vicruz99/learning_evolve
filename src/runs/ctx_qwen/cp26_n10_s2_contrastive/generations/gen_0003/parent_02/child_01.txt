The key insight is that while LP optimization for radii is efficient, the center placement is what limits the result. I need more aggressive exploration of center configurations.

2. Use a smarter alternating optimization: after LP radii refinement, perturb centers strategically (not just randomly) and re-optimize
3. Use larger perturbation scales to escape local minima, combined with finer refinement passes
4. Run significantly more iterations with better convergence criteria
5. Use squared distance constraints in SLSQP for better numerical behavior
6. Try corner-exploiting patterns where larger circles sit near boundaries

import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

def compute_lp_radii(centers):
    """Solve LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    A_ub = np.zeros((n * (n - 1) // 2, n))
    A_ub[np.arange(n * (n - 1) // 2), I_IDX] = 1.0
    A_ub[np.arange(n * (n - 1) // 2), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]
    
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, mx)))
        
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success:
                return np.maximum(res.x, 0.0)
        except Exception:
            pass
    return np.zeros(n)

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[2 * N:])

def constraints_sqr(params):
    """Inequality constraints using squared distances for better numerical behavior."""
    c = params[:2 * N].reshape(N, 2)
    r = params[2 * N:]
    
    cons = []
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap using squared distances
    dx = c[:, np.newaxis, 0] - c[np.newaxis, :, 0]
    dy = c[:, np.newaxis, 1] - c[np.newaxis, :, 1]
    dist_sq = dx**2 + dy**2
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    cons.append(dist_sq[I_IDX, J_IDX] - r_sum[I_IDX, J_IDX]**2)
    
    return np.concatenate(cons)

def hex_init(spacing, margin, offset_x, offset_y, noise_scale, rng):
    """Generate hexagonal lattice initialization."""
    c = np.zeros((N, 2))
    idx = 0
    y = margin + offset_y
    row = 0
    while idx < N and y < 1.0 - margin:
        x = margin + offset_x + (row % 2) * spacing / 2.0
        while x < 1.0 - margin and idx < N:
            c[idx] = [x + rng.uniform(-noise_scale, noise_scale),
                      y + rng.uniform(-noise_scale, noise_scale)]
            x += spacing
            idx += 1
        y += spacing * math.sqrt(3) / 2.0
        row += 1
    while idx < N:
        c[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    return np.clip(c, 0.01, 0.99)

def grid_init(spacing, margin, noise_scale, rng):
    """Generate grid initialization."""
    c = np.zeros((N, 2))
    idx = 0
    y = margin
    while y < 1.0 - margin and idx < N:
        x = margin
        while x < 1.0 - margin and idx < N:
            c[idx] = [x + rng.uniform(-noise_scale, noise_scale),
                      y + rng.uniform(-noise_scale, noise_scale)]
            x += spacing
            idx += 1
        y += spacing
    while idx < N:
        c[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    return np.clip(c, 0.01, 0.99)

def corner_init(rng):
    """Generate corner-focused initialization."""
    c = np.zeros((N, 2))
    # Place some circles near corners and edges
    corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    edge_mids = [[0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5]]
    center_pts = [[0.5, 0.5]]
    
    fixed_pts = corners + edge_mids + center_pts
    c[:len(fixed_pts)] = fixed_pts
    
    # Fill rest with hexagonal-like pattern
    c = c.copy()
    idx = len(fixed_pts)
    sp = 0.18
    y = 0.2
    row = 0
    while idx < N:
        x = 0.2 + (row % 2) * sp / 2.0
        while x < 0.8 and idx < N:
            c[idx] = [x + rng.uniform(-0.01, 0.01), y + rng.uniform(-0.01, 0.01)]
            x += sp
            idx += 1
        y += sp * math.sqrt(3) / 2.0
        row += 1
    while idx < N:
        c[idx] = rng.uniform(0.15, 0.85, 2)
        idx += 1
    return np.clip(c, 0.02, 0.98)

def repel_centers(centers, radii, steps=50, strength=0.001):
    """Iteratively push overlapping circles apart."""
    c = centers.copy()
    n = c.shape[0]
    for _ in range(steps):
        grad = np.zeros((n, 2))
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = math.hypot(dx, dy)
                if d < radii[i] + radii[j] + 0.01 and d > 1e-10:
                    force = strength / (d + 1e-10)
                    grad[i, 0] += dx * force
                    grad[i, 1] += dy * force
                    grad[j, 0] -= dx * force
                    grad[j, 1] -= dy * force
            # Boundary repulsion
            x, y = c[i]
            r = radii[i]
            if x - r < 0.02:
                grad[i, 0] += strength * 10
            if x + r > 0.98:
                grad[i, 0] -= strength * 10
            if y - r < 0.02:
                grad[i, 1] += strength * 10
            if y + r > 0.98:
                grad[i, 1] -= strength * 10
        c += grad
        c = np.clip(c, 0.01, 0.99)
    return c

def optimize_centers_for_radii(centers, radii, rng, maxiter=2000):
    """Optimize center positions while keeping radii roughly fixed."""
    n = centers.shape[0]
    
    def center_obj(params):
        c = params.reshape(n, 2)
        penalty = 0.0
        for i in range(n):
            x, y = c[i]
            r = radii[i]
            if x - r < 0:
                penalty += 100 * (r - x)**2
            if x + r > 1:
                penalty += 100 * (x + r - 1)**2
            if y - r < 0:
                penalty += 100 * (r - y)**2
            if y + r > 1:
                penalty += 100 * (y + r - 1)**2
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = math.hypot(dx, dy)
                min_d = radii[i] + radii[j]
                if d < min_d:
                    penalty += 100 * (min_d - d)**2
        return penalty
    
    def center_cons(params):
        c = params.reshape(n, 2)
        cons = []
        for i in range(n):
            x, y = c[i]
            cons.append(x)
            cons.append(1.0 - x)
            cons.append(y)
            cons.append(1.0 - y)
        return np.concatenate(cons)
    
    try:
        res = minimize(center_obj, centers.flatten(), method='SLSQP',
                       bounds=[(0.01, 0.99)] * (2 * n),
                       constraints={'type': 'ineq', 'fun': center_cons},
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if res.success:
            return res.x.reshape(n, 2)
    except Exception:
        pass
    return centers
