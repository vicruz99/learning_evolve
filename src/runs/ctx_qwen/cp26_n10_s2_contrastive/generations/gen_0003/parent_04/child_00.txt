import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints(x):
    """Squared distance and boundary constraints (must be >= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]

    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c_overlap = dx**2 + dy**2 - (r[I_IDX] + r[J_IDX])**2

    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])

    return np.concatenate([c_overlap, c_bound])


def solve_radii_lp(centers):
    """Given fixed centers, find radii that maximize sum via LP."""
    n = centers.shape[0]
    c_obj = -np.ones(n)

    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt((diff**2).sum(axis=2))

    A_ub = np.zeros((NUM_PAIRS, n))
    b_ub = np.zeros(NUM_PAIRS)

    for k in range(NUM_PAIRS):
        i, j = I_IDX[k], J_IDX[k]
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]

    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds_r.append((0.0, max(0.0, ub)))

    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success and np.all(res.x >= -1e-10):
                return res.x, -res.fun
        except Exception:
            pass

    return np.zeros(n), 0.0


def make_init_centers_hex(seed, spacing=None):
    """Generate hexagonal lattice initial centers with varying parameters."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))

    if spacing is None:
        spacing = 0.13 + rng.uniform(0, 0.08)

    patterns = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [7, 6, 5, 4, 4],
        [6, 6, 5, 5, 4],
        [5, 5, 6, 6, 4],
        [7, 5, 5, 5, 4],
        [4, 6, 6, 6, 4],
        [6, 4, 6, 6, 4],
        [5, 7, 5, 5, 4],
        [8, 5, 6, 4, 3],
    ]
    pattern = patterns[seed % len(patterns)]

    idx = 0
    row = 0
    margin = 0.02 + rng.uniform(0, 0.03)
    y = margin + spacing * np.sqrt(3) / 2 * 0.5

    while idx < N and y < 1.0 - margin:
        x_start = margin + (row % 2) * spacing / 2
        count = pattern[row] if row < len(pattern) else 5
        col = 0

        while col < count and idx < N:
            x = x_start + col * spacing
            if x < 1.0 - margin:
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
            col += 1

        y += spacing * np.sqrt(3) / 2
        row += 1

    # Fill remaining
    while idx < N:
        centers[idx, 0] = rng.uniform(margin + 0.02, 1.0 - margin - 0.02)
        centers[idx, 1] = rng.uniform(margin + 0.02, 1.0 - margin - 0.02)
        idx += 1

    centers += rng.normal(0, 0.006, centers.shape)
    return np.clip(centers, 0.03, 0.97)


def make_init_centers_grid(seed):
    """Generate grid-like initial centers."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))

    # Try different grid dimensions
    for cols, rows in [(6, 5), (5, 6), (7, 4), (4, 7)]:
        if cols * rows >= N:
            break

    x_step = 1.0 / (cols + 1)
    y_step = 1.0 / (rows + 1)

    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= N:
                break
            x = (c + 1) * x_step
            y = (r + 1) * y_step
            if r % 2 == 1:
                x += x_step / 2
            centers[idx, 0] = x + rng.uniform(-0.02, 0.02)
            centers[idx, 1] = y + rng.uniform(-0.02, 0.02)
            idx += 1

    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1

    centers += rng.normal(0, 0.004, centers.shape)
    return np.clip(centers, 0.04, 0.96)


def make_init_centers_random(seed):
    """Generate random initial centers with some spacing heuristic."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.1, 0.9, (N, 2))
    return centers


def run_optimization(c0, r0=None, maxiter=10000):
    """Run SLSQP optimization and return result."""
    if r0 is None:
        r0 = np.full(N, 0.04)

    x0 = np.zeros(3 * N)
    x0[0::3] = c0[:, 0]
    x0[1::3] = c0[:, 1]
    x0[2::3] = r0

    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}

    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter,
                                                  'ftol': 1e-15, 'disp': False})
        cx = res.x[0::3]
        cy = res.x[1::3]
        r = res.x[2::3]
        return np.column_stack((cx, cy)), np.maximum(r, 1e-7)
    except Exception:
        return c0, r0
