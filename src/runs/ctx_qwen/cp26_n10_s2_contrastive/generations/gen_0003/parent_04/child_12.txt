import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
M_PAIRS = len(TRIU_I)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    A_ub = np.zeros((M_PAIRS, n))
    b_ub = np.zeros(M_PAIRS)
    
    A_ub[np.arange(M_PAIRS), TRIU_I] = 1.0
    A_ub[np.arange(M_PAIRS), TRIU_J] = 1.0
    b_ub[:] = dists[TRIU_I, TRIU_J]
    
    # Boundary constraints: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0), -res.fun
    return np.zeros(n), 0.0

def generate_initials():
    """Generate diverse structured initial configurations."""
    initials = []
    # Hexagonal lattices with varying spacing
    for seed in range(10):
        rng = np.random.RandomState(seed)
        c = np.zeros((N, 2))
        idx = 0
        s = 0.18 + rng.uniform(-0.02, 0.02)
        y = s / 2
        row = 0
        while idx < N and y < 1.0 - s / 2:
            x = s / 2 + (row % 2) * s / 2
            while x < 1.0 - s / 2 and idx < N:
                c[idx] = [x + rng.uniform(-0.01, 0.01), y + rng.uniform(-0.01, 0.01)]
                idx += 1
                x += s
            y += s * np.sqrt(3) / 2
            row += 1
        while idx < N:
            c[idx] = rng.uniform(0.2, 0.8, 2)
            idx += 1
        initials.append(c)
    return initials

def jiggle_optimize(centers, rng, epochs=30, step_init=0.01):
    """Randomized coordinate descent to maximize sum of radii."""
    n = centers.shape[0]
    best_c = centers.copy()
    _, best_s = solve_lp_radii(best_c)
    step = step_init
    
    for _ in range(epochs):
        improved = False
        order = rng.permutation(n)
        for i in order:
            candidates = rng.randn(8, 2) * step
            for cand in candidates:
                trial_c = best_c.copy()
                trial_c[i] += cand
                trial_c[i] = np.clip(trial_c[i], 0.01, 0.99)
                _, s = solve_lp_radii(trial_c)
                if s > best_s:
                    best_s = s
                    best_c = trial_c
                    improved = True
                    break
        if not improved:
            step *= 0.88
        if step < 5e-5:
            break
    return best_c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    initials = generate_initials()
    
    # Phase 1: Jiggle optimization from diverse starts
    for seed in range(12):
        rng = np.random.RandomState(seed * 17 + 5)
        idx = seed % len(initials)
        init_c = initials[idx].copy()
        init_c += rng.uniform(-0.01, 0.01, init_c.shape)
        init_c = np.clip(init_c, 0.02, 0.98)
        
        c_opt = jiggle_optimize(init_c, rng, epochs=35, step_init=0.012)
        r_opt, s_opt = solve_lp_radii(c_opt)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    # Phase 2: Refine best configuration with finer steps
    if best_centers is not None:
        rng = np.random.RandomState(42)
        c_opt = jiggle_optimize(best_centers, rng, epochs=50, step_init=0.005)
        r_opt, s_opt = solve_lp_radii(c_opt)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    # Phase 3: Nelder-Mead polish to exploit smooth regions of the LP objective
    if best_centers is not None:
        def nm_obj(c_flat):
            c = np.clip(c_flat.reshape(N, 2), 0.01, 0.99)
            _, s = solve_lp_radii(c)
            return -s
            
        try:
            res = minimize(nm_obj, best_centers.flatten(), method='Nelder-Mead',
                           options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10})
            c_polish = np.clip(res.x.reshape(N, 2), 0.01, 0.99)
            r_polish, s_polish = solve_lp_radii(c_polish)
            if s_polish > best_sum:
                best_sum = s_polish
                best_centers = c_polish
                best_radii = r_polish
        except Exception:
            pass
            
    # Strict post-processing to guarantee validity
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        if radii[i] > mx:
            radii[i] = max(0.0, mx - 1e-9)
            
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc / 2.0
                    radii[j] -= exc / 2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
