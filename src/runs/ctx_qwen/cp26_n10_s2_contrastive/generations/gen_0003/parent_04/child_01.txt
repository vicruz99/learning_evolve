import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x, n, i_idx, j_idx):
    """Vectorized inequality constraints: boundary clearance and non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4 * n + n * (n - 1) // 2)
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    c[4*n:] = dx**2 + dy**2 - (r[i_idx] + r[j_idx])**2
    return c

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    A_ub = np.zeros((NUM_PAIRS + n, n))
    b_ub = np.zeros(NUM_PAIRS + n)
    
    # r_i + r_j <= dist_ij
    A_ub[:NUM_PAIRS, I_IDX] = 1.0
    A_ub[:NUM_PAIRS, J_IDX] = 1.0
    b_ub[:NUM_PAIRS] = dists[I_IDX, J_IDX]
    
    # r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        ub = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[NUM_PAIRS + i, i] = 1.0
        b_ub[NUM_PAIRS + i] = max(0.0, ub)
        
    bounds_r = [(0.0, 0.5)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success and np.all(res.x >= -1e-9):
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 0.01), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    cons = {'type': 'ineq', 'fun': constraints, 'args': (n, I_IDX, J_IDX)}
    
    best_sum = 0.0
    best_c = None
    best_r = None

            while x < 0.92 and idx < n:
                c[idx] = [x, y]
                x += sp
                idx += 1
            y += sp * np.sqrt(3) / 2.0
            row += 1
        while idx < n:
            c[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
        inits.append(c)
        
    # 2. Random feasible placements
    for seed in range(15):
        rng = np.random.RandomState(seed + 100)
        c = rng.uniform(0.1, 0.9, (n, 2))
        inits.append(c)
        
    # 3. Grid patterns
    for seed in range(10):
        rng = np.random.RandomState(seed + 200)
        c = np.zeros((n, 2))
        idx = 0
        step = 0.18 + rng.uniform(-0.02, 0.02)
        y = 0.05
        while y < 0.95 and idx < n:
            x = 0.05
            while x < 0.95 and idx < n:
                c[idx] = [x, y]
                x += step
                idx += 1
            y += step
        while idx < n:
            c[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
        inits.append(c)

    # Phase 1: Broad search
    for c0 in inits:
        r0 = np.full(n, 0.03)
        x0 = np.zeros(3 * n)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r0
        
        try:

    # Fallback
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum = solve_lp_radii(best_c)
        
    # Strict post-processing to guarantee validity
    # Enforce boundaries
    for i in range(n):
        mx = min(best_c[i, 0], 1.0 - best_c[i, 0], best_c[i, 1], 1.0 - best_c[i, 1])
        if best_r[i] > mx:
            best_r[i] = max(0.0, mx - 1e-9)

    for _ in range(200):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])

                    exc = best_r[i] + best_r[j] - d
                    best_r[i] -= exc / 2.0
                    best_r[j] -= exc / 2.0
                    changed = True
        if not changed:
            break
            
    best_r = np.maximum(best_r, 0.0)
    return best_c, best_r, float(np.sum(best_r))
