import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

def solve_lp_radii(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    d_pair = dists[I_IDX, J_IDX]
    
    d_bound = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    
    A = np.zeros((N_PAIRS, n))
    A[np.arange(N_PAIRS), I_IDX] = 1.0
    A[np.arange(N_PAIRS), J_IDX] = 1.0
    
    bounds = [(0.0, max(0.0, d)) for d in d_bound]
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=d_pair, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun, res
    except Exception:
        pass
    return np.zeros(n), 0.0, None

def compute_center_gradient(centers, res):
    """Compute gradient of LP objective w.r.t centers using dual variables."""
    if res is None:
        return np.zeros_like(centers)
    
    n = centers.shape[0]
    grad = np.zeros((n, 2))
    
    # Pair constraints duals
    try:
        lambdas = np.abs(res.ineqlin.marginals)
    except (AttributeError, TypeError):
        lambdas = np.zeros(N_PAIRS)

    dists = np.sqrt(np.sum(diff**2, axis=2))
    d_pair = dists[I_IDX, J_IDX]
    
    for k in range(N_PAIRS):
        lam = lambdas[k]
        if lam > 1e-8:
            i, j = I_IDX[k], J_IDX[k]
            d = d_pair[k]
            if d > 1e-9:
                vec = centers[i] - centers[j]
                grad[i] += lam * vec / d
                grad[j] -= lam * vec / d
                
    # Boundary constraints duals
    try:
        ub_duals = np.abs(res.upper.marginals)
    except (AttributeError, TypeError):
        ub_duals = np.zeros(n)
        
    for i in range(n):
        if ub_duals[i] > 1e-8:
            x, y = centers[i]
            vals = np.array([x, 1.0-x, y, 1.0-y])
            active = np.argmin(vals)
            if active == 0: grad[i] += ub_duals[i] * np.array([1.0, 0.0])

            elif active == 2: grad[i] += ub_duals[i] * np.array([0.0, 1.0])
            else: grad[i] += ub_duals[i] * np.array([0.0, -1.0])
            
    return grad

def local_search_gradient(centers, n_steps=400):
    """Improve centers via gradient ascent on LP objective."""
    best_centers = centers.copy()
    best_sum = 0.0
    step_size = 3e-4
    
    for _ in range(n_steps):
        r, s, res = solve_lp_radii(centers)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            
        grad = compute_center_gradient(centers, res)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-9:
            centers += (step_size / g_norm) * grad
            
        centers = np.clip(centers, 0.001, 0.999)
        step_size *= 0.998
        
    return best_centers, best_sum

def obj_func(x):
    return -np.sum(x[2*N:])

def constr_func(x):
    c = x[:2*N].reshape(N, 2)
    r = x[2*N:]
    
    con = np.concatenate([
        c[:, 0] - r, 1.0 - c[:, 0] - r,
        c[:, 1] - r, 1.0 - c[:, 1] - r
    ])
    
    diff = c[:, None, :] - c[None, :, :]

    con = np.concatenate([con, dists[I_IDX, J_IDX] - r[I_IDX] - r[J_IDX]])
    return con

def init_hex(seed):

    row_counts = [6, 5, 6, 5, 4]

            centers[idx] = [x, y]
            idx += 1
            
    centers += rng.uniform(-0.03, 0.03, centers.shape)
    return np.clip(centers, 0.05, 0.95)

    for seed in range(5):
