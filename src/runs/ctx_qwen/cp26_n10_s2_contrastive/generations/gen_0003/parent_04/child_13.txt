import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    A_ub = np.zeros((N_PAIRS, n))
    b_ub = np.zeros(N_PAIRS)
    k = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0-x, y, 1.0-y)
        bounds_r.append((0.0, max(1e-9, ub)))
        
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
            
    return np.zeros(n), 0.0

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap (squared)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4*N + N_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    dist_sq = dx**2 + dy**2
    r_sum = r[I_IDX] + r[J_IDX]
    c[4*N:] = dist_sq - r_sum**2
    return c

def make_init(seed, style='hex'):
    """Generate structured or random initial center configurations."""
    rng = np.random.RandomState(seed)
    c = np.zeros((N, 2))
    
    if style == 'hex':
        sp = 0.14 + rng.uniform(0, 0.06)
        idx = 0
        row = 0
        y = sp/2 + rng.uniform(0, 0.04)
        while idx < N and y < 1.0 - sp/2:
            x = sp/2 + (row % 2) * sp/2
            while x < 1.0 - sp/2 and idx < N:
                c[idx] = [x, y]
                idx += 1
                x += sp
            y += sp * np.sqrt(3) / 2
            row += 1
        while idx < N:
            c[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
    elif style == 'corner':
        corners = np.array([[0.06, 0.06], [0.94, 0.06], [0.06, 0.94], [0.94, 0.94]])
        c[:4] = corners + rng.uniform(-0.02, 0.02, (4, 2))
        # Edge positions
        edges = []
        for i in range(1, 4):
            edges.extend([[i*0.25, 0.06], [i*0.25, 0.94], [0.06, i*0.25], [0.94, i*0.25]])
        c[4:12] = np.array(edges) + rng.uniform(-0.02, 0.02, (8, 2))
        c[12:] = rng.uniform(0.2, 0.8, (14, 2))
    else:
        c = rng.uniform(0.15, 0.85, (N, 2))
        
    c += rng.normal(0, 0.004, c.shape)
    return np.clip(c, 0.05, 0.95)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Uses a hybrid SLSQP + LP strategy with extensive multi-start and refinement.
    """
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Phase 1: Broad multi-start search
    for seed in range(80):
        style = 'hex' if seed < 50 else ('corner' if seed < 65 else 'rand')
        c0 = make_init(seed, style)
        r_lp, _ = solve_lp_radii(c0)
        
        x0 = np.zeros(3*N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = np.maximum(r_lp * 0.95, 1e-6)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            
            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
            r_opt, s_opt = solve_lp_radii(c_opt)
            
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            continue
            
    # Phase 2: Local perturbation refinement around the best solution
    if best_c is not None:
        rng = np.random.RandomState(42)
        for trial in range(60):
            scale = 0.004 * (0.88 ** (trial // 20))
            c_pert = best_c + rng.normal(0, scale, best_c.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            r_pert = best_r * 0.97
            
            x_pert = np.zeros(3*N)
            x_pert[0::3] = c_pert[:, 0]
            x_pert[1::3] = c_pert[:, 1]
            x_pert[2::3] = r_pert
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt, s_opt = solve_lp_radii(c_opt)
                
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
            except Exception:
                continue

    # Fallback safety net
    if best_c is None:
        best_c = make_init(0)
        best_radii, best_sum = solve_lp_radii(best_c)
        best_r = best_radii
        
    # Phase 3: Strict post-processing to guarantee validity
    radii = best_r.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        x, y = best_c[i]
        max_r = min(x, 1.0-x, y, 1.0-y)
        if radii[i] > max_r - 1e-11:
            radii[i] = max(0.0, max_r - 1e-11)
            
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                if d < radii[i] + radii[j] - 1e-11:
                    overlap = radii[i] + radii[j] - d
                    radii[i] -= overlap/2.0
                    radii[j] -= overlap/2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return best_c, radii, final_sum
