import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((N_PAIRS, n))
    b_ub = np.zeros(N_PAIRS)
    
    # Vectorized pairwise distances
    diff = centers[I_IDX, :] - centers[J_IDX, :]
    dists = np.hypot(diff[:, 0], diff[:, 1])
    
    A_ub[np.arange(N_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(N_PAIRS), J_IDX] = 1.0
    b_ub[:] = dists
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    # Try high-performance solver first, fallback to interior-point
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-9):
                radii = np.maximum(res.x, 0.0)
                return radii, np.sum(radii)
        except Exception:
            continue
    return np.zeros(n), 0.0

def obj_func(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def constraint_func(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap (g(x) >= 0)."""
    c = x[:2*N].reshape(N, 2)
    r = x[2*N:]
    
    bc = np.empty(4*N)
    bc[0:N] = c[:, 0] - r
    bc[N:2*N] = 1.0 - c[:, 0] - r
    bc[2*N:3*N] = c[:, 1] - r
    bc[3*N:4*N] = 1.0 - c[:, 1] - r
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    overlaps = dists[I_IDX, J_IDX] - r[I_IDX] - r[J_IDX]
    
    return np.concatenate([bc, overlaps])

def fix_violations(centers, radii):
    """Strictly enforce boundary and non-overlap constraints to guarantee validity."""
    # Fix boundary violations
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        if radii[i] > mx + 1e-12:
            radii[i] = max(0.0, mx - 1e-10)

    for _ in range(100):
        changed = False

            i, j = I_IDX[k], J_IDX[k]
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])

                exc = radii[i] + radii[j] - d
                radii[i] -= exc * 0.5
                radii[j] -= exc * 0.5
                radii[i] = max(0.0, radii[i])
                radii[j] = max(0.0, radii[j])
                changed = True
        if not changed:
            break
    return centers, radii
