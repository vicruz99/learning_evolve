I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):

    cx = x[0::3]
    cy = x[1::3]

    dy = cy[I_IDX] - cy[J_IDX]
    r_sum = r[I_IDX] + r[J_IDX]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    return c

    A_ub[np.arange(N_PAIRS), J_IDX] = 1.0

        idx += 1

    return centers
