import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

# Precompute constraint matrix structure for LP
A_ub_lp = np.zeros((NUM_PAIRS, N))
A_ub_lp[np.arange(NUM_PAIRS), I_IDX] = 1.0
A_ub_lp[np.arange(NUM_PAIRS), J_IDX] = 1.0

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.hypot(dx, dy)
    b_ub = dists[I_IDX, J_IDX]
    
    bounds = []
    for i in range(N):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(1e-9, mx)))
        
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_lp, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= -1e-9):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(N, 0.01), 0.0

def local_search(c0, steps, init_step, rng):
    """Stochastic hill climbing on centers with LP radius evaluation."""
    c = c0.copy()
    r, s = solve_lp_radii(c)
    step = init_step
    best_c_loc = c.copy()
    best_r_loc = r.copy()
    best_s_loc = s
    
    for _ in range(steps):
        # Occasional large jumps to escape shallow basins
        if rng.random() < 0.05:
            step = rng.uniform(0.01, 0.05)
        else:
            step *= 0.97
            
        c_new = c + rng.normal(0, step, c.shape)
        c_new = np.clip(c_new, 0.01, 0.99)
        
        r_new, s_new = solve_lp_radii(c_new)
        
        if s_new > best_s_loc + 1e-7:
            c, r, best_s_loc = c_new, r_new, s_new
            best_c_loc = c.copy()
            best_r_loc = r.copy()
            step = min(step * 1.15, 0.06)
        else:
            step *= 0.96
            
    return best_c_loc, best_r_loc, best_s_loc

def obj_func(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def cons_func(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    c_arr = np.column_stack((cx, cy))
    diff = c_arr[:, None, :] - c_arr[None, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    return np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r,
        dists[I_IDX, J_IDX] - r[I_IDX] - r[J_IDX]
    ])

def run_packing():
    rng = np.random.default_rng(42)
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    starts = []
    
    # 1. Hexagonal lattices with varying spacing
    for sp in np.linspace(0.155, 0.205, 8):
        c = np.zeros((N, 2))
        idx = 0
        y = sp/2
        row = 0
        while idx < N and y < 1.0 - sp/2:
            x = sp/2 + (row % 2) * sp/2
            while idx < N and x < 1.0 - sp/2:
                c[idx] = [x, y]
                x += sp
                idx += 1
            y += sp * np.sqrt(3) / 2
            row += 1
        while idx < N:
            c[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
        starts.append(c)
        
    # 2. Square grids
    for sp in np.linspace(0.17, 0.22, 6):
        c = np.zeros((N, 2))
        idx = 0
        y = sp/2
        while idx < N and y < 1.0 - sp/2:
            x = sp/2
            while idx < N and x < 1.0 - sp/2:
                c[idx] = [x, y]
                x += sp
                idx += 1
            y += sp
        while idx < N:
            c[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
        starts.append(c)
            
    # 3. Random feasible placements
    for _ in range(15):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # Phase 1: Basin hopping on centers with LP radius resolution
    for c_start in starts:
        c_noisy = c_start + rng.normal(0, 0.004, c_start.shape)
        c_noisy = np.clip(c_noisy, 0.02, 0.98)
        c_opt, r_opt, s_opt = local_search(c_noisy, steps=800, init_step=0.012, rng=rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            best_r = r_opt.copy()
            
    # Phase 2: SLSQP joint polishing
    if best_c is not None:
        # Shrink radii slightly to ensure strict initial feasibility for SLSQP
        x0 = np.concatenate([best_c.flatten(), best_r * 0.99])
        bounds = [(0,1), (0,1), (0, 0.5)] * N
        try:
            res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, 
                           constraints={'type': 'ineq', 'fun': cons_func},
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                co = res.x[:2*N].reshape(N, 2)
                ro, so = solve_lp_radii(co)
                if so > best_sum:
                    best_c = co
                    best_r = ro
                    best_sum = so
        except Exception:
            pass

    # Phase 3: Strict post-processing to guarantee validator compliance
    c_final = best_c.copy()
    r_final = best_r.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(c_final[i,0], 1.0-c_final[i,0], c_final[i,1], 1.0-c_final[i,1])
        r_final[i] = min(r_final[i], max(0.0, mx - 1e-10))
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(60):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(c_final[i,0]-c_final[j,0], c_final[i,1]-c_final[j,1])
                if d < r_final[i] + r_final[j] - 1e-11:
                    exc = r_final[i] + r_final[j] - d
                    r_final[i] -= exc/2
                    r_final[j] -= exc/2
                    changed = True
        if not changed:
            break
            
    r_final = np.maximum(r_final, 0.0)
    return c_final, r_final, float(np.sum(r_final))
