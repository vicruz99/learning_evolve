N = 26

def solve_lp_radii(centers):

    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])

    A_ub[np.arange(NUM_PAIRS), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]

        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(1e-9, mx)))

        except Exception:
            pass
    return np.full(n, 0.01), 0.0

def resolve_overlaps(centers, radii, iterations=60):
    """Deterministically push overlapping circles apart."""
    n = centers.shape[0]
    for _ in range(iterations):
        max_overlap = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)

                    overlap = r_sum - d

                    uy = dy / d

                    centers[i, 1] += uy * shift
                    centers[j, 0] -= ux * shift
                    centers[j, 1] -= uy * shift
                    max_overlap = max(max_overlap, overlap)
        centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
        if max_overlap < 1e-8:
            break
    return centers
