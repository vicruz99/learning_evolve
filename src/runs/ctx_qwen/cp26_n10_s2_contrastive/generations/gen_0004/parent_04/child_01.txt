import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

# Precompute constant structure for pairwise LP constraints
A_ub_pairs = np.zeros((NUM_PAIRS, N))
A_ub_pairs[np.arange(NUM_PAIRS), I_IDX] = 1.0
A_ub_pairs[np.arange(NUM_PAIRS), J_IDX] = 1.0

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    b_ub = dists[I_IDX, J_IDX]
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_pairs, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            r = np.maximum(res.x, 0.0)
            return r, np.sum(r)
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def constraints(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap (must be >= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c[4*N:] = np.hypot(dx, dy) - (r[I_IDX] + r[J_IDX])
    return c

def local_hill_climb(centers, rng):
    """Refine centers using LP oracle to maximize sum of radii."""
    curr_c = centers.copy()
    curr_r, curr_s = solve_lp_radii(curr_c)
    step = 0.008

        improved = False
        
        # Global perturbation
        pc = curr_c + rng.normal(0, step, curr_c.shape)
        pc = np.clip(pc, 0.005, 0.995)
        pr, ps = solve_lp_radii(pc)
        if ps > curr_s:
            curr_c, curr_r, curr_s = pc, pr, ps
            step *= 1.15
            improved = True
            
        # Coordinate-wise perturbation
        if not improved:
            for i in range(N):
                for _ in range(2):
                    pc2 = curr_c.copy()
                    pc2[i] += rng.normal(0, step, 2)

                    if ps2 > curr_s:
                        curr_c, curr_r, curr_s = pc2, pr2, ps2
                        improved = True
                        break
                if improved:
                    break
                    
        if not improved:
            step *= 0.85
        step = max(step, 1e-5)
        
    return curr_c, curr_r, curr_s
