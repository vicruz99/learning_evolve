import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
PAIRS_I = []
PAIRS_J = []
for i in range(N):
    for j in range(i + 1, N):
        PAIRS_I.append(i)
        PAIRS_J.append(j)
N_PAIRS = len(PAIRS_I)

# Precompute constant structure for LP pairwise constraints
A_ub_const = np.zeros((N_PAIRS, N))
A_ub_const[np.arange(N_PAIRS), PAIRS_I] = 1.0
A_ub_const[np.arange(N_PAIRS), PAIRS_J] = 1.0

def solve_lp_fast(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    c_obj = -np.ones(N)
    dx = centers[PAIRS_I, 0] - centers[PAIRS_J, 0]
    dy = centers[PAIRS_I, 1] - centers[PAIRS_J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = []
    for i in range(N):
        ub = min(centers[i, 0], 1.0 - centers[i, 0], 
                 centers[i, 1], 1.0 - centers[i, 1])
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub_const, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.zeros(N), 0.0

def objective_joint(x):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2 * N:])

def constraints_joint(x):
    """Inequality constraints for SLSQP (must be >= 0)."""
    c = x[:2 * N].reshape(N, 2)
    r = x[2 * N:]
    
    b1 = c[:, 0] - r
    b2 = 1.0 - c[:, 0] - r
    b3 = c[:, 1] - r
    b4 = 1.0 - c[:, 1] - r
    
    dx = c[PAIRS_I, 0] - c[PAIRS_J, 0]
    dy = c[PAIRS_I, 1] - c[PAIRS_J, 1]
    dists = np.hypot(dx, dy)
    r_sum = r[PAIRS_I] + r[PAIRS_J]
    overlaps = dists - r_sum
    
    return np.concatenate([b1, b2, b3, b4, overlaps])

def slsqp_polish(centers, maxiter=5000):
    """Local refinement using SLSQP on joint center+radius space."""
    r0, _ = solve_lp_fast(centers)
    r0 = np.maximum(r0 * 0.99, 1e-5)
    x0 = np.concatenate([centers.flatten(), r0])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_joint}
    
    try:
        res = minimize(objective_joint, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if res.success:
            co = res.x[:2 * N].reshape(N, 2)
            ro, so = solve_lp_fast(co)
            return co, ro, so
    except Exception:
        pass
    return centers, r0, np.sum(r0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_c = None
    best_r = None
    best_obj = 0.0

        row = 0

                x += s
                idx += 1
            y += s * math.sqrt(3) / 2
            row += 1

            c[idx] = rng.uniform(0.2, 0.8, 2)
            idx += 1
        c += rng.normal(0, 0.003, c.shape)
        c = np.clip(c, 0.02, 0.98)
        configs.append(c)
        
    for _ in range(15):
        configs.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # Phase 2: SLSQP polishing from multiple starts

    best_r = np.maximum(best_r, 0.0)
    best_obj = float(np.sum(best_r))
    
    return best_c, best_r, best_obj
