import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
M = len(I)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((M, n))
    A_ub[np.arange(M), I] = 1.0
    A_ub[np.arange(M), J] = 1.0
    
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = []
    for k in range(n):
        x, y = centers[k]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.01)

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap (>= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4 * N + M)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c[4*N:] = np.hypot(dx, dy) - (r[I] + r[J])
    return c

def generate_hex_init(spacing, row_offset, col_offset, seed):
    """Generate a hexagonal lattice initialization with controlled noise."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = 0.05 + row_offset
    row = 0
    while idx < N and y < 0.95:
        x_start = 0.05 + col_offset + (row % 2) * spacing / 2.0
        col = 0
        while x_start + col * spacing < 0.95 and idx < N:
            centers[idx] = [x_start + col * spacing, y]
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2.0
        row += 1
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    return centers + rng.normal(0, 0.005, centers.shape)
