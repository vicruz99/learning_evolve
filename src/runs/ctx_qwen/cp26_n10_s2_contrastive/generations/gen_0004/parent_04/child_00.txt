import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    # LP constraint matrix: r_i + r_j <= dist(i,j)
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(NUM_PAIRS), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]
    
    # Boundary constraints: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = [(0.0, max(1e-9, min(c[0], 1.0-c[0], c[1], 1.0-c[1]))) for c in centers]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = np.maximum(res.x, 0.0)
            return radii, np.sum(radii)
    except Exception:
        pass
    return np.full(n, 0.01), 0.26

def objective(x):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints for SLSQP: boundary clearance and non-overlap."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dists = np.hypot(dx, dy)
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    c[4*N:] = dists[I_IDX, J_IDX] - (r[I_IDX] + r[J_IDX])
    return c

def lp_center_search(centers, radii, n_iter=15000, seed=42):
    """Hill climbing on center coordinates evaluated exactly via LP."""
    rng = np.random.default_rng(seed)
    best_c = centers.copy()
    best_r = radii.copy()
    best_s = np.sum(radii)
    curr_c = best_c.copy()
    scale = 0.012
    
    for _ in range(n_iter):
        k = rng.integers(1, 4)
        idx = rng.choice(N, k, replace=False)
        c_pert = curr_c.copy()
        c_pert[idx] += rng.normal(0, scale, (k, 2))
        c_pert = np.clip(c_pert, 1e-5, 1-1e-5)
        
        r_pert, s_pert = solve_lp_radii(c_pert)
        if s_pert > best_s:
            best_c = c_pert.copy()
            best_r = r_pert.copy()
            best_s = s_pert
            curr_c = c_pert.copy()
            scale *= 1.03
        else:
            scale *= 0.985
            if rng.random() < 0.02:
                curr_c = best_c + rng.normal(0, scale*1.5, best_c.shape)
                curr_c = np.clip(curr_c, 1e-5, 1-1e-5)
    return best_c, best_r, best_s

def run_packing():
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    rng = np.random.default_rng(123)
    inits = []
    
    # Generate diverse hexagonal lattice initializations
    for seed in range(40):
        r_gen = np.random.RandomState(seed * 13 + 7)
        c = np.zeros((N, 2))
        idx = 0
        y = 0.06
        row = 0
        sp = 0.15 + r_gen.uniform(-0.02, 0.04)
        while idx < N and y < 0.94:
            x = 0.06 + (row % 2) * sp / 2.0
            while x < 0.94 and idx < N:
                c[idx] = [x, y]
                x += sp
                idx += 1
            y += sp * np.sqrt(3) / 2.0
            row += 1
        while idx < N:
            c[idx] = r_gen.uniform(0.1, 0.9, 2)
            idx += 1
        c += r_gen.normal(0, 0.005, c.shape)
        c = np.clip(c, 0.02, 0.98)
        inits.append(c)

    # Phase 1: SLSQP from diverse starts
    for c0 in inits:

        x0 = np.zeros(3*N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]

            co = res.x[:2*N].reshape(N, 2)

            if so > best_sum:
                best_sum = so
                best_c = co.copy()
                best_r = ro.copy()
        except Exception:
            pass

    if best_c is not None:

    if best_c is None:
        best_c = inits[0]
        best_r, best_sum = solve_lp_radii(best_c)
        
    # Strict post-processing to guarantee validator compliance
    c_final = best_c.copy()
    r_final = best_r.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(c_final[i,0], 1.0-c_final[i,0], c_final[i,1], 1.0-c_final[i,1])
        r_final[i] = min(r_final[i], mx - 1e-9)

                    exc = r_final[i] + r_final[j] - d
                    r_final[i] -= exc/2.0
                    r_final[j] -= exc/2.0
                    changed = True
        if not changed: break
    r_final = np.maximum(r_final, 0.0)
    
    return c_final, r_final, float(np.sum(r_final))
