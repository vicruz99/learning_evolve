

<strategy>
The strategy focuses on a hybrid optimization approach combining Linear Programming (LP) for exact radius computation given fixed centers, Sequential Least Squares Programming (SLSQP) for joint center-radius optimization, and an iterative "grow-and-resolve" phase using L-BFGS-B to steadily inflate the packing density.

1. **Initialization**: Generate diverse starting configurations using hexagonal lattices, square grids, and random placements. This ensures coverage of different structural basins.
2. **Broad Search**: Run joint SLSQP optimization from each initialization to quickly converge to local optima. Refine radii using LP after each SLSQP run.
3. **Iterative Inflation (Grow & Resolve)**: For the top candidates, iteratively scale all radii by a small factor (1.002) and use L-BFGS-B to minimize a smooth penalty function of constraint violations. This pushes the configuration towards tighter packings that SLSQP might miss. After inflation, LP refines the exact maximal radii.
4. **Polishing & Perturbation**: Apply further SLSQP polishing on the inflated configurations. Use small random perturbations to escape shallow local minima, followed by LP/SLSQP refinement.
5. **Strict Post-processing**: Enforce boundary constraints strictly and iteratively resolve any remaining numerical overlaps to guarantee validator compliance.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I] = 1.0
    A_ub[np.arange(NUM_PAIRS), J] = 1.0
    
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = []
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds.append((0.0, max(0.0, mx)))
        
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(n), 0.0

def obj_slsqp(x):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2::3])

def cons_slsqp(x):
    """Inequality constraints for SLSQP"""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c_dist = np.hypot(dx, dy) - (r[I] + r[J])
    c_b = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    return np.concatenate([c_dist, c_b])

def compute_penalty(c_flat, radii):
    """Smooth penalty function for constraint violations"""
    c = c_flat.reshape(N, 2)
    p = 0.0
    x, y = c[:, 0], c[:, 1]
    r = radii
    p += np.sum(np.maximum(0.0, r - x)**2)
    p += np.sum(np.maximum(0.0, r - (1.0 - x))**2)
    p += np.sum(np.maximum(0.0, r - y)**2)
    p += np.sum(np.maximum(0.0, r - (1.0 - y))**2)
    
    dx = c[I, 0] - c[J, 0]
    dy = c[I, 1] - c[J, 1]
    dist = np.hypot(dx, dy)
    ov = r[I] + r[J] - dist
    p += np.sum(np.maximum(0.0, ov)**2)
    return p

def run_grow_resolve(centers, radii, steps=20, scale=1.002):
    """Iteratively scale radii and resolve center positions to allow denser packing"""
    for _ in range(steps):
        radii_new = radii * scale
        res = minimize(compute_penalty, centers.flatten(), args=(radii_new,), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)]*(2*N), options={'maxiter': 300, 'ftol': 1e-10})
        centers = res.x.reshape(N, 2)
        if compute_penalty(centers.flatten(), radii_new) < 1e-5:
            radii = radii_new
        else:
            break
    return centers, radii

def generate_inits(rng, n_configs=40):
    """Generate diverse initial center configurations"""
    inits = []
    # Hexagonal lattices
    for s in np.linspace(0.15, 0.22, 12):
        c = np.zeros((N, 2))
        idx = 0
        row = 0
        y = s/2
        while idx < N and y < 1.0 - s/2:
            x = s/2 + (row % 2) * s/2
            while x < 1.0 - s/2 and idx < N:
                c[idx] = [x, y]
                x += s
                idx += 1
            y += s * np.sqrt(3)/2
            row += 1
        while idx < N:
            c[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
        inits.append(c)
        
    # Square grids
    for s in np.linspace(0.17, 0.23, 8):
        c = np.zeros((N, 2))
        idx = 0
        y = s/2
        while y < 1.0 - s/2 and idx < N:
            x = s/2
            while x < 1.0 - s/2 and idx < N:
                c[idx] = [x, y]
                x += s
                idx += 1
            y += s
        while idx < N:
            c[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
        inits.append(c)
        
    # Random placements
    for _ in range(max(0, n_configs - len(inits))):
        inits.append(rng.uniform(0.05, 0.95, (N, 2)))
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    """
    rng = np.random.RandomState(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    candidates = []
    
    inits = generate_inits(rng, 45)
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    
    # Phase 1: Broad multi-start SLSQP search
    for c0 in inits:
        r0, _ = solve_lp_radii(c0)
        if np.sum(r0) < 0.1: continue
        
        x0 = np.zeros(3*N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r0 * 0.95
        
        try:
            res = minimize(obj_slsqp, x0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': cons_slsqp},
                           options={'maxiter': 7000, 'ftol': 1e-13})
            c_opt = res.x[:2*N].reshape(N, 2)
            r_opt, s_opt = solve_lp_radii(c_opt)
            candidates.append((s_opt, c_opt, r_opt))
        except Exception:
            pass

    candidates.sort(reverse=True)
    
    # Phase 2: Deep refinement on top candidates using iterative inflation
    for _, c_base, r_base in candidates[:8]:
        # Grow & Resolve to push past local minima
        c_grow, r_grow = run_grow_resolve(c_base, r_base, steps=20, scale=1.002)
        r_grow_lp, s_grow = solve_lp_radii(c_grow)
        
        # Polish with joint SLSQP
        x0 = np.zeros(3*N)
        x0[0::3] = c_grow[:, 0]
        x0[1::3] = c_grow[:, 1]
        x0[2::3] = r_grow_lp * 0.95
        try:
            res = minimize(obj_slsqp, x0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': cons_slsqp},
                           options={'maxiter': 9000, 'ftol': 1e-13})
            c_polish = res.x[:2*N].reshape(N, 2)
            r_polish, s_polish = solve_lp_radii(c_polish)
            if s_polish > best_sum:
                best_sum = s_polish
                best_centers = c_polish.copy()
                best_radii = r_polish.copy()
        except Exception:
            pass
            
        # Perturbation jumps to escape basins
        for _ in range(4):
            cp = best_centers + rng.normal(0, 0.0025, best_centers.shape)
            cp = np.clip(cp, 0.02, 0.98)
            rp, _ = solve_lp_radii(cp)
            xp = np.zeros(3*N)
            xp[0::3] = cp[:, 0]
            xp[1::3] = cp[:, 1]
            xp[2::3] = rp * 0.95
            try:
                res2 = minimize(obj_slsqp, xp, method='SLSQP', bounds=bounds,
                                constraints={'type': 'ineq', 'fun': cons_slsqp},
                                options={'maxiter': 6000, 'ftol': 1e-13})
                c2 = res2.x[:2*N].reshape(N, 2)
                r2, s2 = solve_lp_radii(c2)
                if s2 > best_sum:
                    best_sum = s2
                    best_centers = c2.copy()
                    best_radii = r2.copy()
            except Exception:
                pass

    # Fallback safety net
    if best_centers is None:
        best_centers = inits[0]
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    # Strict post-processing to guarantee validator compliance
    radii = best_radii.copy()
    centers = best_centers.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(0.0, radii[i])
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-10:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc/2.0
                    radii[j] -= exc/2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
