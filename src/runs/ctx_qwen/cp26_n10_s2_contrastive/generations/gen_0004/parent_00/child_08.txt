import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    A_ub = np.zeros((len(I_IDX), n))
    A_ub[np.arange(len(I_IDX)), I_IDX] = 1.0
    A_ub[np.arange(len(I_IDX)), J_IDX] = 1.0
    
    dx = centers[I_IDX, 0] - centers[J_IDX, 0]
    dy = centers[I_IDX, 1] - centers[J_IDX, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = []
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds.append((0.0, max(0.0, mx)))
        
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(n), 0.0

def push_apart(centers, radii, steps):
    """Expand radii and push centers apart to find better configurations."""
    c = centers.copy()
    r = radii.copy()
    for _ in range(steps):
        r *= 1.0015
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = math.hypot(dx, dy)
                if d < r[i] + r[j] and d > 1e-12:
                    overlap = r[i] + r[j] - d
                    shift = overlap * 0.5
                    ux, uy = dx / d, dy / d
                    c[i, 0] += ux * shift
                    c[i, 1] += uy * shift
                    c[j, 0] -= ux * shift
                    c[j, 1] -= uy * shift
                    changed = True
        for i in range(N):
            for dim in range(2):
                if c[i, dim] < r[i]:
                    c[i, dim] = r[i]
                if c[i, dim] > 1.0 - r[i]:
                    c[i, dim] = 1.0 - r[i]
        if not changed:
            break
    return c, r

def objective(x):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints for SLSQP"""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c_dist = np.hypot(dx, dy) - (r[I_IDX] + r[J_IDX])
    
    c_b = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    return np.concatenate([c_dist, c_b])

def make_init(seed, style):
    """Generate structured or random initial center configurations."""
    rng = np.random.RandomState(seed)
    c = np.zeros((N, 2))
    if style == 'hex':
        s = 0.15 + rng.uniform(-0.02, 0.04)
        idx = 0
        row = 0
        y = 0.05 + rng.uniform(0, 0.05)
        while idx < N and y < 0.95:
            x = 0.05 + (row % 2) * s / 2
            while x < 0.95 and idx < N:
                c[idx] = [x, y]
                idx += 1
                x += s
            y += s * np.sqrt(3) / 2
            row += 1
    elif style == 'grid':
        step = 0.18 + rng.uniform(-0.02, 0.02)
        idx = 0
        y = step / 2
        while y < 1.0 - step / 2 and idx < N:
            x = step / 2
            while x < 1.0 - step / 2 and idx < N:
                c[idx] = [x, y]
                idx += 1
                x += step
            y += step
    else:
        c = rng.uniform(0.1, 0.9, (N, 2))
        
    if seed % 5 == 0:
        c[0] = [0.05, 0.05]
        c[1] = [0.95, 0.05]
        c[2] = [0.05, 0.95]
        c[3] = [0.95, 0.95]
        
    c += rng.normal(0, 0.005, c.shape)
    return np.clip(c, 0.02, 0.98)

def obj_single(p, centers, idx):
    """Objective for single circle position optimization"""
    centers[idx] = p
    _, s = solve_lp_radii(centers)
    return -s

def optimize_single(centers, radii, idx):
    """Optimize position of a single circle while others are fixed"""
    res = minimize(obj_single, centers[idx], args=(centers, idx),
                   method='Nelder-Mead', options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 250})
    centers[idx] = res.x
    r_new, s_new = solve_lp_radii(centers)
    return centers, r_new, s_new

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    """
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    rng = np.random.RandomState(123)
    
    # Phase 1: Diverse initializations & Expansion
    inits = []

        inits.append(make_init(s, 'hex'))

        inits.append(make_init(s, 'grid'))
    for s in range(15):
        inits.append(make_init(s, 'rand'))
        
    for c0 in inits:

        r0 = np.maximum(r0, 0.01)
        c_exp, r_exp = push_apart(c0, r0, steps=120)
        r_lp, s_lp = solve_lp_radii(c_exp)
        
        # Phase 2: SLSQP joint refinement
        x0 = np.zeros(3 * N)
        x0[0::3] = c_exp[:, 0]
        x0[1::3] = c_exp[:, 1]

            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))

            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            pass
            
        # Phase 3: Single-circle coordinate descent
        if best_centers is not None:
            curr_c = best_centers.copy()
            curr_r = best_radii.copy()
            curr_s = best_sum

                for i in range(N):
                    curr_c, curr_r, curr_s = optimize_single(curr_c, curr_r, i)
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_centers = curr_c.copy()
                    best_radii = curr_r.copy()

        best_centers = make_init(0, 'hex')
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    # Strict post-processing to guarantee validator compliance
    radii = best_radii.copy()
    centers = best_centers.copy()
    
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])

                    radii[i] -= overlap / 2.0
                    radii[j] -= overlap / 2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
