import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Use hypot for pairwise (stable gradient away from zero)
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c_overlap = np.hypot(dx, dy) - (r[I_IDX] + r[J_IDX])
    
    # Boundary constraints
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    
    dx = centers[I_IDX, 0] - centers[J_IDX, 0]
    dy = centers[I_IDX, 1] - centers[J_IDX, 1]
    b_ub = np.hypot(dx, dy)
    
    A_ub = np.zeros((N_PAIRS, n))
    A_ub[np.arange(N_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(N_PAIRS), J_IDX] = 1.0
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(1e-12, ub)))
    
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(n), 0.0

def rotate_points(points, angle):
    """Rotate points around center of unit square."""
    c = np.array([0.5, 0.5])
    r = points - c
    cos_a = np.cos(angle)
    sin_a = np.sin(angle)
    rotated = np.empty_like(r)
    rotated[:, 0] = r[:, 0] * cos_a - r[:, 1] * sin_a
    rotated[:, 1] = r[:, 0] * sin_a + r[:, 1] * cos_a
    return rotated + c

def generate_hex_centers(spacing, angle=0.0, seed=42):
    """Generate hexagonal lattice centers with optional rotation."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    row = 0
    y = spacing / 2
    while idx < N and y < 1.0 - spacing / 2:
        x_start = spacing / 2 + (row % 2) * spacing / 2
        col = 0
        while x_start + col * spacing < 1.0 - spacing / 2 and idx < N:
            centers[idx] = [x_start + col * spacing, y]
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2
        row += 1
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    
    if abs(angle) > 1e-6:
        centers = rotate_points(centers, angle)
    
    centers += rng.normal(0, 0.001, centers.shape)
    return np.clip(centers, 0.015, 0.985)

def generate_row_pattern(row_counts, spacing=0.17, seed=42):
    """Generate circles arranged in rows with given counts."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = 0.05
    dy = 1.0 / (len(row_counts) + 1)
    
    for r_idx, cnt in enumerate(row_counts):
        shift = 0.0 if r_idx % 2 == 0 else spacing / 2.0
        x_start = 0.05 + shift
        for _ in range(cnt):
            if idx < N:
                centers[idx] = [x_start + rng.uniform(-0.005, 0.005), 
                                y + rng.uniform(-0.005, 0.005)]
                x_start += spacing
                idx += 1
        y += dy
    
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    
    return np.clip(centers, 0.015, 0.985)

def push_centers_apart(centers, radii, iterations=50):
    """Push overlapping circles apart by moving centers."""
    centers = centers.copy()
    radii = radii.copy()
    n = centers.shape[0]
    
    for _ in range(iterations):
        max_shift = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)
                r_sum = radii[i] + radii[j]
                
                if d < r_sum and d > 1e-14:
                    overlap = r_sum - d
                    shift = overlap / 2.0 * 0.5
                    ux = dx / d
                    uy = dy / d
                    
                    centers[i, 0] += ux * shift
                    centers[i, 1] += uy * shift
                    centers[j, 0] -= ux * shift
                    centers[j, 1] -= uy * shift
                    max_shift = max(max_shift, shift)
        
        centers = np.clip(centers, 1e-8, 1.0 - 1e-8)
        if max_shift < 1e-12:
            break
    
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

            if res.success or -res.fun > best_sum:
