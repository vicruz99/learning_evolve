import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to non-overlap and boundary constraints."""
    n = centers.shape[0]
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(NUM_PAIRS), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]
    
    bounds = []
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds.append((0.0, max(0.0, mx - 1e-9)))
        
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return res.x, -res.fun
        except Exception:
            continue
    return np.zeros(n), 0.0

def objective(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints: pairwise distance >= sum of radii, and circles inside unit square."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c_dist = np.hypot(dx, dy) - (r[I_IDX] + r[J_IDX])
    
    c_b = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    return np.concatenate([c_dist, c_b])

def generate_initializations(num_trials=60):
    """Generate diverse initial center configurations."""
    inits = []
    rng = np.random.RandomState(42)
    
    # Hexagonal lattices with varying spacing
    for seed in range(25):
        s = 0.13 + rng.uniform(0, 0.07)
        centers = np.zeros((N, 2))
        idx = 0
        row = 0
        y = s/2 + rng.uniform(-0.01, 0.01)
        while idx < N and y < 1.0 - s/2:
            x_start = s/2 + (row % 2) * s/2 + rng.uniform(-0.01, 0.01)
            col = 0
            while x_start + col*s < 1.0 - s/2 and idx < N:
                centers[idx] = [x_start + col*s, y]
                idx += 1
                col += 1
            y += s * np.sqrt(3) / 2
            row += 1
        while idx < N:
            centers[idx] = rng.uniform(0.05, 0.95, 2)
            idx += 1
        inits.append(centers)
        
    # Grid lattices
    for seed in range(15):
        step = 0.18 + rng.uniform(-0.03, 0.03)
        centers = np.zeros((N, 2))
        idx = 0
        y = step/2
        while y < 1.0 - step/2 and idx < N:
            x = step/2
            while x < 1.0 - step/2 and idx < N:
                centers[idx] = [x, y]
                x += step
                idx += 1
            y += step
        while idx < N:
            centers[idx] = rng.uniform(0.05, 0.95, 2)
            idx += 1
        inits.append(centers)
        
    # Random uniform placements
    for seed in range(15):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # Corner/Edge biased placements
    for seed in range(10):
        c = rng.uniform(0.05, 0.95, (N, 2))
        corners = [[0.05,0.05], [0.95,0.05], [0.05,0.95], [0.95,0.95]]
        for i in range(4):
            c[i] = corners[i] + rng.uniform(-0.02, 0.02, 2)
        inits.append(c)
        
    return inits[:num_trials]

def push_apart(centers, radii, max_iter=50, push_scale=0.005):
    """Iteratively push overlapping circles apart to resolve constraints."""
    n = centers.shape[0]
    for _ in range(max_iter):
        max_viol = 0.0
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                d = math.hypot(dx, dy)
                r_sum = radii[i] + radii[j]
                if d < r_sum and d > 1e-8:
                    overlap = r_sum - d
                    shift = overlap * push_scale
                    ux = dx / d
                    uy = dy / d
                    centers[i,0] += ux * shift
                    centers[i,1] += uy * shift
                    centers[j,0] -= ux * shift
                    centers[j,1] -= uy * shift
                    max_viol = max(max_viol, overlap)
        centers = np.clip(centers, 0.01, 0.99)
        if max_viol < 1e-5:
            break
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    cons_opt = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.RandomState(123)
    inits = generate_initializations(60)
    
    # Phase 1: Broad multi-start search with feasibility enforcement
    for c0 in inits:
        r0, s0 = solve_lp_radii(c0)

            r0 = np.full(N, 0.03)
            
        # Ensure strict feasibility before SLSQP
        c_feas = push_apart(c0.copy(), r0.copy())
        x0 = np.zeros(3*N)
        x0[0::3] = c_feas[:, 0]
        x0[1::3] = c_feas[:, 1]
        x0[2::3] = np.maximum(r0 * 0.98, 1e-5)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt, constraints=cons_opt,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if res.success or -res.fun > best_sum:
                curr_c = np.column_stack((res.x[0::3], res.x[1::3]))
                curr_r, curr_s = solve_lp_radii(curr_c)
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_centers = curr_c.copy()
                    best_radii = curr_r.copy()
        except Exception:
            continue
            
    # Phase 2: Adaptive basin hopping with decaying noise
    if best_centers is not None:
        for scale in [0.02, 0.01, 0.005, 0.002, 0.001]:
            for _ in range(25):
                c_pert = best_centers + rng.normal(0, scale, best_centers.shape)
                c_pert = np.clip(c_pert, 0.02, 0.98)
                r_pert, s_pert = solve_lp_radii(c_pert)
                if s_pert <= 0: continue
                
                x_pert = np.zeros(3*N)
                x_pert[0::3] = c_pert[:, 0]
                x_pert[1::3] = c_pert[:, 1]
                x_pert[2::3] = r_pert * 0.99
                
                try:
                    res = minimize(objective, x_pert, method='SLSQP', bounds=bounds_opt, constraints=cons_opt,
                                   options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_opt, s_opt = solve_lp_radii(c_opt)
                    if s_opt > best_sum:
                        best_sum = s_opt
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
                except Exception:
                    continue
                    
    # Phase 3: Sequential radius expansion & center adjustment
    if best_centers is not None:
        curr_c = best_centers.copy()
        curr_r = best_radii.copy()
        
        for step in range(40):
            alpha = 0.006 * (0.94 ** step)
            new_r = curr_r * (1 + alpha)
            new_c = push_apart(curr_c.copy(), new_r, max_iter=30, push_scale=0.01)
            
            x0 = np.zeros(3*N)
            x0[0::3] = new_c[:, 0]
            x0[1::3] = new_c[:, 1]
            x0[2::3] = new_r * 0.995
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt, constraints=cons_opt,

                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt, s_opt = solve_lp_radii(c_opt)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
                    curr_c = c_opt.copy()
                    curr_r = r_opt.copy()
            except Exception:
                break

    # Fallback safety net
    if best_centers is None:
        best_centers = inits[0]
        best_radii, best_sum = solve_lp_radii(best_centers)
        
    # Strict post-processing to guarantee numerical validity
    radii = best_radii.copy()
    centers = best_centers.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(0.0, radii[i])
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = math.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-10:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc * 0.5
                    radii[j] -= exc * 0.5
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
