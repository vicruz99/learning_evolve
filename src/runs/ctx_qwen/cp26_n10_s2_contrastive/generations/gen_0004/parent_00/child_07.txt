

<strategy>
1. **Hybrid Optimization Pipeline**: Combine force-directed relaxation, SLSQP joint optimization, and Linear Programming (LP) radius refinement. Force relaxation handles large-scale rearrangements and escapes shallow local minima where gradient methods stall. SLSQP provides precise gradient-based tuning. LP guarantees maximal radii for any fixed center configuration.
2. **Diverse Initialization**: Generate hexagonal lattices, square grids, and random configurations with varying densities to cover different packing symmetries.
3. **Decaying Perturbation Schedule**: After each successful optimization, apply random perturbations to centers with a decaying scale. This basin-hopping strategy allows the solver to jump out of local optima while gradually focusing on fine-tuning.
4. **Strict Feasibility Enforcement**: Ensure all initial points for SLSQP are strictly feasible by scaling down LP radii slightly. Post-process the final result to rigorously satisfy boundary and overlap constraints within numerical tolerances.
5. **Efficient LP Solver**: Precompute the constraint matrix structure for the LP solver and use the `highs` method for fast, robust radius maximization.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(I_IDX)

# Precompute LP constraint matrix structure (constant across calls)
A_LP = np.zeros((N_PAIRS, N))
A_LP[np.arange(N_PAIRS), I_IDX] = 1.0
A_LP[np.arange(N_PAIRS), J_IDX] = 1.0

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    b_ub = dists[I_IDX, J_IDX]
    
    # Boundary upper bounds for each circle
    bounds = []
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds.append((0.0, max(0.0, mx)))
        
    # Try HiGHS first (fastest), fallback to interior-point
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(N), 0.0

def objective(x):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(x[2::3])

def constraints(x):
    """
    Inequality constraints g(x) >= 0:
    - Pairwise non-overlap: dist(i,j) >= r_i + r_j
    - Boundary clearance: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c_dist = np.hypot(dx, dy) - (r[I_IDX] + r[J_IDX])
    
    c_b = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    return np.concatenate([c_dist, c_b])

def relax_centers(centers, radii, steps=150, dt=0.004):
    """Force-directed relaxation to push overlapping circles apart and away from boundaries."""
    c = centers.copy()
    r = radii.copy()
    for _ in range(steps):
        forces = np.zeros_like(c)
        active = False
        
        for i in range(N):
            # Boundary repulsion
            for b in range(2):
                if c[i, b] - r[i] < 0:
                    forces[i, b] += -(c[i, b] - r[i])
                if c[i, b] + r[i] > 1.0:
                    forces[i, b] -= (c[i, b] + r[i] - 1.0)
                    
            # Pairwise repulsion
            for j in range(i + 1, N):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = math.hypot(dx, dy)
                if d < r[i] + r[j] and d > 1e-9:
                    overlap = r[i] + r[j] - d
                    # Soft repulsion force scaling
                    f = overlap / (d * d + 1e-6)
                    fx = dx / d * f
                    fy = dy / d * f
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
                    active = True
                    
        if not active:
            break
            
        c += forces * dt
        c = np.clip(c, 1e-7, 1.0 - 1e-7)
    return c

def generate_inits(rng):
    """Generate diverse initial center configurations."""
    inits = []
    
    # Hexagonal lattices with varying spacing
    for sp in np.linspace(0.14, 0.22, 8):
        c = np.zeros((N, 2))
        idx = 0
        row = 0
        y = sp / 2
        while idx < N and y < 1.0 - sp / 2:
            x = sp / 2 + (row % 2) * sp / 2
            while x < 1.0 - sp / 2 and idx < N:
                c[idx] = [x, y]
                x += sp
                idx += 1
            y += sp * math.sqrt(3) / 2
            row += 1
        while idx < N:
            c[idx] = rng.uniform(0.2, 0.8, 2)
            idx += 1
        inits.append(c)
        
    # Square grids
    for sp in np.linspace(0.15, 0.22, 6):
        c = np.zeros((N, 2))
        idx = 0
        y = sp / 2
        while y < 1.0 - sp / 2 and idx < N:
            x = sp / 2
            while x < 1.0 - sp / 2 and idx < N:
                c[idx] = [x, y]
                x += sp
                idx += 1
            y += sp
        while idx < N:
            c[idx] = rng.uniform(0.2, 0.8, 2)
            idx += 1
        inits.append(c)
        
    # Random uniform placements
    for _ in range(20):
        c = rng.uniform(0.1, 0.9, (N, 2))
        inits.append(c)
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Uses a hybrid pipeline: Initialization -> Force Relaxation -> SLSQP -> LP Refinement -> Basin Hopping.
    """
    rng = np.random.RandomState(42)
    best_sum = 0.0
    best_c = None
    best_r = None
    
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': constraints}
    
    inits = generate_inits(rng)
    
    # Phase 1: Broad multi-start search
    for c0 in inits:
        r0, _ = solve_lp_radii(c0)
        r0 = np.maximum(r0 * 0.97, 1e-5)  # Ensure strict feasibility for SLSQP
        
        # Force relaxation to clean up overlaps before gradient optimization
        c0 = relax_centers(c0, r0, steps=120, dt=0.005)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r0
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt, constraints=cons_opt,
                           options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                co = np.column_stack((res.x[0::3], res.x[1::3]))
                ro, so = solve_lp_radii(co)
                if so > best_sum:
                    best_sum = so
                    best_c = co.copy()
                    best_r = ro.copy()
        except Exception:
            continue
            
    # Phase 2: Basin hopping / perturbation refinement
    if best_c is not None:
        for step in range(70):
            # Adaptive decaying noise schedule
            scale = 0.018 * (0.94 ** (step // 12))
            cp = best_c + rng.normal(0, scale, best_c.shape)
            cp = np.clip(cp, 0.02, 0.98)
            
            rp, _ = solve_lp_radii(cp)
            rp = np.maximum(rp * 0.98, 1e-5)
            
            # Quick relaxation to make perturbed config feasible
            cp = relax_centers(cp, rp, steps=80, dt=0.004)
            
            x0 = np.zeros(3 * N)
            x0[0::3] = cp[:, 0]
            x0[1::3] = cp[:, 1]
            x0[2::3] = rp
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt, constraints=cons_opt,
                               options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if res.success:
                    co = np.column_stack((res.x[0::3], res.x[1::3]))
                    ro, so = solve_lp_radii(co)
                    if so > best_sum:
                        best_sum = so
                        best_c = co.copy()
                        best_r = ro.copy()
            except Exception:
                continue
                
    # Fallback safety net
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum = solve_lp_radii(best_c)
        
    # Phase 3: Strict post-processing to guarantee validator compliance
    radii = best_r.copy()
    centers = best_c.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(0.0, radii[i])
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc / 2.0
                    radii[j] -= exc / 2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
```
