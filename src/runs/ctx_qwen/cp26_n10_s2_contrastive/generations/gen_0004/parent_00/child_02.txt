import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

# Precompute constant structure for LP pairwise constraints
A_LP = np.zeros((NUM_PAIRS, N))
A_LP[np.arange(NUM_PAIRS), I_IDX] = 1.0
A_LP[np.arange(NUM_PAIRS), J_IDX] = 1.0

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    dx = centers[I_IDX, 0] - centers[J_IDX, 0]
    dy = centers[I_IDX, 1] - centers[J_IDX, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = []
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds.append((0.0, max(0.0, mx)))
        
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(N), 0.0

def de_objective(centers_flat):
    """Objective for Differential Evolution: negative sum of maximal radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    try:
        _, s = solve_lp_radii(centers)
        return -s
    except Exception:
        return 100.0  # Large penalty for invalid states

def obj_joint(x):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2::3])

def cons_joint(x):
    """
    Inequality constraints for joint optimization:
    - Pairwise distance >= sum of radii
    - Circle boundaries within [0,1]x[0,1]
    Returns array of constraint values (must be >= 0)
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c_dist = np.hypot(dx, dy) - (r[I_IDX] + r[J_IDX])
    
    c_b = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    return np.concatenate([c_dist, c_b])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    # Phase 1: Differential Evolution for global center search
    bounds_de = [(0.02, 0.98)] * (2 * N)
    res_de = differential_evolution(de_objective, bounds_de, seed=42, maxiter=200, popsize=35,
                                    mutation=(0.5, 1.5), recombination=0.8, polishing=False,
                                    workers=1, disp=False)
    
    best_centers = res_de.x.reshape(N, 2)
    best_radii, best_sum = solve_lp_radii(best_centers)
    
    # Phase 2: Joint optimization with SLSQP from DE result
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    
    x0 = np.zeros(3*N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = np.maximum(best_radii * 0.95, 1e-4)
    
    res_slsqp = minimize(obj_joint, x0, method='SLSQP', bounds=bounds_opt,
                         constraints={'type': 'ineq', 'fun': cons_joint},
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                         
    curr_c = np.column_stack((res_slsqp.x[0::3], res_slsqp.x[1::3]))
    curr_r, curr_s = solve_lp_radii(curr_c)
    if curr_s > best_sum:
        best_sum = curr_s
        best_centers = curr_c
        best_radii = curr_r
        
    # Phase 3: Adaptive basin hopping & LP refinement
    rng = np.random.RandomState(123)
    for step in range(90):
        noise = 0.012 * np.exp(-step / 30.0)
        c_pert = best_centers + rng.normal(0, noise, best_centers.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        
        r_pert, s_pert = solve_lp_radii(c_pert)
        if s_pert > best_sum:
            best_sum = s_pert
            best_centers = c_pert.copy()
            best_radii = r_pert.copy()
            
            # Refine with SLSQP
            x0 = np.zeros(3*N)
            x0[0::3] = best_centers[:, 0]
            x0[1::3] = best_centers[:, 1]
            x0[2::3] = np.maximum(best_radii * 0.96, 1e-4)
            try:
                res2 = minimize(obj_joint, x0, method='SLSQP', bounds=bounds_opt,
                                constraints={'type': 'ineq', 'fun': cons_joint},
                                options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
                c2 = np.column_stack((res2.x[0::3], res2.x[1::3]))
                r2, s2 = solve_lp_radii(c2)
                if s2 > best_sum:
                    best_sum = s2
                    best_centers = c2
                    best_radii = r2
            except Exception:
                pass
                
    # Phase 4: Strict post-processing to guarantee validator compliance
    radii = best_radii.copy()
    centers = best_centers.copy()

    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(0.0, radii[i])

    for _ in range(100):
        changed = False

                if d < radii[i] + radii[j] - 1e-11:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc/2.0
                    radii[j] -= exc/2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
