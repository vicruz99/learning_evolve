import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]

    A_ub[np.arange(NUM_PAIRS), J_IDX] = 1.0

    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds.append((0.0, max(0.0, mx)))

            res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)

                return r, np.sum(r)
        except Exception:
            continue
    return np.zeros(n), 0.0

def objective(x):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2::3])

def constraints(x):

    - Pairwise distance >= sum of radii
    - Circle boundaries within [0,1]x[0,1]
    Returns array of constraint values (must be >= 0)
    """

    cy = x[1::3]
    r = x[2::3]

    dy = cy[I_IDX] - cy[J_IDX]

    c_b = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    return np.concatenate([c_dist, c_b])

def refine_centers(centers, maxiter=12000):
    """Run SLSQP + LP refinement on a center configuration."""
    r0, _ = solve_lp_radii(centers)
    r0 = np.maximum(r0, 1e-6)
    
    x0 = np.zeros(3 * N)
    x0[0::3] = centers[:, 0]

    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': constraints}

        res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt, constraints=cons_opt,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
        return c_opt
    except Exception:
        return centers

def push_tight_centers(centers, radii, step=0.002):
    """Identify tight constraints and push centers apart slightly."""
    c_new = centers.copy()
    diff = c_new[:, np.newaxis, :] - c_new[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    tight_mask = dists[I_IDX, J_IDX] < r_sum[I_IDX, J_IDX] + 1e-6
    tight_i = I_IDX[tight_mask]
    tight_j = J_IDX[tight_mask]
    
    for i, j in zip(tight_i, tight_j):
        dx = c_new[i, 0] - c_new[j, 0]
        dy = c_new[i, 1] - c_new[j, 1]
        d = math.hypot(dx, dy)
        if d < 1e-9:
            continue
        shift = step * (1.0 - d / (radii[i] + radii[j]))
        c_new[i, 0] += (dx / d) * shift
        c_new[i, 1] += (dy / d) * shift
        c_new[j, 0] -= (dx / d) * shift
        c_new[j, 1] -= (dy / d) * shift
        
    return np.clip(c_new, 0.01, 0.99)
