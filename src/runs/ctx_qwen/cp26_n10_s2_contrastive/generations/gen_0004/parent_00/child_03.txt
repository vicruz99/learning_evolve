To improve upon the current sum of radii (~2.634), I will implement a dual-gradient ascent strategy on the circle centers. The problem of maximizing $\sum r_i$ subject to $r_i + r_j \le |c_i - c_j|$ and boundary constraints can be viewed as optimizing the centers $c_i$ to maximize the optimal value of a linear program. 

Key improvements:
1. **Dual Gradient Ascent**: For a fixed set of centers, solving the LP gives optimal radii and dual variables (Lagrange multipliers) for the pairwise distance constraints. The envelope theorem states that the gradient of the LP's optimal value with respect to centers is $\nabla_{c_i} \sum r = \sum_j \lambda_{ij} \frac{c_i - c_j}{|c_i - c_j|}$. Moving centers along this gradient directly increases the sum of radii.
2. **Adaptive Step Sizing**: The gradient ascent uses a backtracking/forward-adapting step size to ensure steady progress without overshooting.
3. **Diverse Initializations**: Generate multiple hexagonal, grid, and random patterns. Run gradient ascent on each to find high-quality basins.
4. **Hybrid Polishing**: After gradient ascent finds a strong configuration, apply SLSQP for joint center-radius refinement, followed by local random perturbations to escape shallow local minima.
5. **Strict Post-processing**: Guarantee validator compliance by clipping boundaries and iteratively resolving any numerical overlaps by symmetrically shrinking radii.

This approach avoids the non-smoothness issues of purely constraint-based SLSQP on non-convex problems and directly optimizes the objective landscape, typically yielding tighter packings.

import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

def solve_lp(centers):
    """Given centers, solve LP to maximize sum of radii. Returns radii, sum, and duals."""
    n = centers.shape[0]
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(NUM_PAIRS), J_IDX] = 1.0
    
    dx = centers[I_IDX, 0] - centers[J_IDX, 0]
    dy = centers[I_IDX, 1] - centers[J_IDX, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = []
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        bounds.append((0.0, max(0.0, mx)))
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            r = np.maximum(res.x, 0.0)
            try:
                duals = res.ineqlin.marginals
                if duals is None:
                    duals = np.zeros(NUM_PAIRS)
            except Exception:
                duals = np.zeros(NUM_PAIRS)
            return r, -res.fun, duals
    except Exception:
        pass
    return np.zeros(n), 0.0, np.zeros(NUM_PAIRS)

def gradient_ascent(centers, max_iter=400):
    """Optimize centers by ascending the gradient of the LP optimal value."""
    best_c = centers.copy()
    best_r, best_s, best_duals = solve_lp(best_c)
    if best_s <= 0:
        return best_c, best_r, best_s
        
    step = 0.008
    stagnation = 0
    for _ in range(max_iter):
        grad = np.zeros((N, 2))
        for k, lam in enumerate(best_duals):
            if lam > 1e-9:
                i, j = I_IDX[k], J_IDX[k]
                dx = best_c[i, 0] - best_c[j, 0]
                dy = best_c[i, 1] - best_c[j, 1]
                d = math.hypot(dx, dy)
                if d > 1e-12:
                    ux, uy = dx / d, dy / d
                    grad[i] += lam * np.array([ux, uy])
                    grad[j] += lam * np.array([-ux, -uy])
                    
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-10:
            break
            
        new_c = best_c + (grad / g_norm) * step
        new_c = np.clip(new_c, 1e-5, 1.0 - 1e-5)
        
        _, s_new, _ = solve_lp(new_c)
        if s_new > best_s - 1e-11:
            best_c = new_c
            best_s = s_new
            best_r, _, best_duals = solve_lp(best_c)
            step *= 1.15
            if step > 0.06:
                step = 0.06
            stagnation = 0
        else:
            step *= 0.75
            stagnation += 1
            if step < 1e-8 or stagnation > 50:
                break
    return best_c, best_r, best_s

def generate_starts():
    """Create diverse initial center configurations."""
    starts = []
    rng = np.random.RandomState(42)
    
    # Hexagonal lattices with varying densities
    for s in np.linspace(0.15, 0.225, 10):
        centers = []
        y = 0.04
        row = 0
        while len(centers) < N:
            x = 0.04 + (row % 2) * s / 2.0
            while x < 0.96 and len(centers) < N:
                centers.append([x + rng.uniform(-0.005, 0.005), 
                                y + rng.uniform(-0.005, 0.005)])
                x += s
            y += s * np.sqrt(3) / 2.0
            row += 1
        while len(centers) < N:
            centers.append([0.5, 0.5])
        starts.append(np.array(centers[:N]))
        
    # Square grids
    for s in np.linspace(0.17, 0.22, 6):
        centers = []
        y = 0.05
        while y < 0.95 and len(centers) < N:
            x = 0.05
            while x < 0.95 and len(centers) < N:
                centers.append([x, y])
                x += s
            y += s
        while len(centers) < N:
            centers.append([0.5, 0.5])
        starts.append(np.array(centers[:N]))
        
    # Random uniform placements
    for _ in range(15):
        c = rng.uniform(0.1, 0.9, (N, 2))
        starts.append(c)
        
    return starts

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Boundary and pairwise non-overlap constraints."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c_dist = np.hypot(dx, dy) - (r[I_IDX] + r[J_IDX])
    c_b = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    return np.concatenate([c_dist, c_b])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': constraints}
    
    best_c, best_r, best_s = None, None, 0.0
    starts = generate_starts()
    
    # Phase 1: Dual-gradient ascent from diverse starts
    for c0 in starts:
        c, r, s = gradient_ascent(c0)
        if s > best_s:
            best_c, best_r, best_s = c, r, s
            
    # Phase 2: SLSQP joint refinement on the best configuration
    if best_c is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_c[:, 0]
        x0[1::3] = best_c[:, 1]
        x0[2::3] = best_r * 0.96  # Slight shrink ensures strict feasibility
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt, constraints=cons_opt,
                           options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                c_polish = np.column_stack((res.x[0::3], res.x[1::3]))
                r_polish, s_polish, _ = solve_lp(c_polish)
                if s_polish > best_s:
                    best_c, best_r, best_s = c_polish, r_polish, s_polish
        except Exception:
            pass
            
    # Phase 3: Local basin hopping / perturbation refinement
    rng = np.random.RandomState(99)
    if best_c is not None:
        for _ in range(30):
            scale = rng.uniform(0.002, 0.012)
            c_pert = best_c + rng.normal(0, scale, best_c.shape)
            c_pert = np.clip(c_pert, 0.015, 0.985)
            c_g, r_g, s_g = gradient_ascent(c_pert, max_iter=200)
            if s_g > best_s:
                best_c, best_r, best_s = c_g, r_g, s_g
                
    # Fallback safety net
    if best_c is None:
        best_c = starts[0]
        best_r, best_s, _ = solve_lp(best_c)
        
    # Phase 4: Strict post-processing to guarantee validator compliance
    radii = best_r.copy()
    centers = best_c.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(0.0, radii[i])
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    overlap = radii[i] + radii[j] - d
                    radii[i] -= overlap / 2.0
                    radii[j] -= overlap / 2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
