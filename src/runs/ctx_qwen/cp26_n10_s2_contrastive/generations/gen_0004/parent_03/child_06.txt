import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

# Precompute constant LP constraint matrix for pairwise non-overlap
A_LP = np.zeros((NUM_PAIRS, N))
A_LP[np.arange(NUM_PAIRS), I_IDX] = 1.0
A_LP[np.arange(NUM_PAIRS), J_IDX] = 1.0

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    dx = centers[I_IDX, 0] - centers[J_IDX, 0]
    dy = centers[I_IDX, 1] - centers[J_IDX, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = []
    for i in range(N):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(N, 0.01)

def lp_obj(x):
    """Objective for center optimization: negative sum of LP-optimal radii."""
    c = np.clip(x.reshape(N, 2), 0.01, 0.99)
    return -np.sum(solve_radii_lp(c))

                        c[idx] = [xs + col * sp, y]
                        idx += 1
                        col += 1
                    y += sp * np.sqrt(3) / 2.0

            curr_r = solve_radii_lp(curr_c)
            curr_s = np.sum(curr_r)
            if curr_s > best_sum:

                best_c = curr_c.copy()
        except Exception:
            pass

            r_p = solve_radii_lp(c_p)
            s_p = np.sum(r_p)
