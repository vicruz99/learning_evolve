import numpy as np
from scipy.optimize import linprog

N = 26
M = N*(N-1)//2

# Precompute the constant pairwise constraint matrix for the LP
A_ub_global = np.zeros((M, N))
_k = 0
for _i in range(N):
    for _j in range(_i+1, N):
        A_ub_global[_k, _i] = 1.0
        A_ub_global[_k, _j] = 1.0
        _k += 1

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    b_ub = np.zeros(M)
    k = 0
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i,0] - centers[j,0]
            dy = centers[i,1] - centers[j,1]
            b_ub[k] = np.sqrt(dx*dx + dy*dy)
            k += 1
            
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0-x, y, 1.0-y)
        bounds.append((0.0, max(1e-9, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub_global, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.005)

def jiggle_optimize(centers, rng, max_iter=800, init_scale=0.02):
    """Iterative perturbation optimizer that maximizes sum of radii via LP."""
    best_centers = centers.copy()
    best_radii = solve_radii_lp(centers)
    best_sum = np.sum(best_radii)
    
    scale = init_scale
    no_improve = 0
    
    for step in range(max_iter):
        # Move a random subset of circles to allow fine local tuning
        move_mask = rng.random(N) < 0.5
        if not move_mask.any():
            move_mask[rng.choice(N)] = True
            
        noise = np.zeros((N, 2))
        noise[move_mask] = rng.standard_normal((move_mask.sum(), 2)) * scale
        
        new_centers = best_centers + noise
        new_centers = np.clip(new_centers, 1e-4, 1.0 - 1e-4)
        
        new_radii = solve_radii_lp(new_centers)
        new_sum = np.sum(new_radii)
        
        if new_sum > best_sum:
            best_centers = new_centers
            best_radii = new_radii
            best_sum = new_sum
            scale = min(scale * 1.01, 0.05)
            no_improve = 0
        else:
            scale *= 0.995
            no_improve += 1
            
        # Reheat if stuck
        if no_improve > 150:
            scale = max(scale, 0.005)
            no_improve = 0
            
        if scale < 1e-6:
            scale = 1e-5
            
    return best_centers, best_radii, best_sum
