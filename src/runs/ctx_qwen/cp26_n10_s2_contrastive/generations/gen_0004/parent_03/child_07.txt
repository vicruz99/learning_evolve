

Looking at the current solutions, I notice several opportunities for improvement:

1. **Constraint formulation**: The current code uses squared distances (`dx**2 + dy**2 - r_sum**2`), which gives poor gradients near contact points. Using `np.hypot` directly (like Solution 1) provides better numerical behavior.

2. **Search diversity**: Need more diverse initial patterns and better exploration of corner/edge configurations.

3. **Simulated annealing**: Can add a direct center-space SA loop that evaluates sum of radii via LP, enabling better global exploration than SLSQP alone.

4. **Finer perturbation schedules**: The current basin hopping needs more decaying perturbation scales for fine-tuning.

<strategy>
1. Replace squared distance constraints with np.hypot for better gradients
2. Add more diverse row patterns and hexagonal spacings to initial configurations
3. Implement simulated annealing on center positions (evaluating LP radii sum as objective)
4. Use multi-stage optimization: broad SLSQP search → SA exploration → fine local LP perturbation
5. Increase number of perturbation scales in fine local search
6. Better integration of LP refinement at every stage

import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)


def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)

    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])

    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(NUM_PAIRS), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]

    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, mx)))

    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.full(n, 0.01), 0.0


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints(x):
    """Inequality constraints using np.hypot for stable gradients."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]

    c = np.empty(4 * N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2 * N] = 1.0 - cx - r
    c[2 * N:3 * N] = cy - r
    c[3 * N:4 * N] = 1.0 - cy - r

    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c[4 * N:] = np.hypot(dx, dy) - (r[I_IDX] + r[J_IDX])
    return c


def generate_hex_init(spacing, seed, rng):
    """Generate hexagonal lattice initialization with varying parameters."""
    centers = np.zeros((N, 2))
    idx = 0
    row = 0
    y = spacing / 2 + rng.uniform(-0.02, 0.02)
    margin_x = rng.uniform(0.0, 0.02)

    while idx < N and y < 1.0 - spacing / 2:
        x_start = spacing / 2 + (row % 2) * spacing / 2 + margin_x
        col = 0
        while x_start + col * spacing < 1.0 - spacing / 2 and idx < N:
            centers[idx, 0] = x_start + col * spacing
            centers[idx, 1] = y
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2
        row += 1

    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1

    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.02, 0.98)


def generate_pattern_init(pat, spacing, rng):
    """Generate row pattern initialization with hexagonal staggering."""
    centers = []
    y = spacing * 0.52
    dy = spacing * np.sqrt(3) / 2 * 0.96
    margin_x = rng.uniform(-0.01, 0.01)

    for r_idx, cnt in enumerate(pat):
        shift = 0.0 if r_idx % 2 == 0 else spacing / 2.0
        x = spacing * 0.48 + shift + margin_x
        for _ in range(cnt):
            if len(centers) < N:
                centers.append([x, y])
            x += spacing
        y += dy

    while len(centers) < N:
        centers.append([0.5, 0.5])

    centers = np.array(centers[:N])
    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.02, 0.98)


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': constraints}

    best_sum = 0.0
    best_centers = None
    best_radii = None

    rng_main = np.random.default_rng(42)

    # Generate diverse initial configurations
    configs = []

    # Hexagonal lattices with various spacings and perturbations
    for seed in range(50):
        rng = np.random.default_rng(seed * 17 + 3)
        sp = 0.13 + rng.uniform(0, 0.08)
        configs.append(generate_hex_init(sp, seed, rng))

    # Row patterns with varying configurations
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [7, 6, 5, 4, 4],
        [4, 5, 6, 5, 6], [8, 6, 5, 4, 3], [6, 6, 6, 6, 2],
        [5, 5, 5, 5, 6], [7, 5, 6, 5, 3], [4, 6, 6, 6, 4],
        [5, 7, 5, 5, 4], [6, 4, 6, 5, 5], [3, 6, 7, 6, 4],
        [5, 5, 7, 4, 5], [6, 5, 5, 6, 4], [4, 4, 8, 5, 5],
        [7, 6, 5, 5, 3], [5, 4, 7, 5, 5], [6, 6, 5, 5, 4],
        [5, 6, 7, 4, 4], [4, 5, 5, 7, 5], [6, 7, 5, 4, 4],
        [7, 5, 5, 6, 3], [5, 5, 6, 6, 4], [6, 5, 4, 6, 5],
        [4, 7, 6, 5, 4], [5, 6, 4, 7, 4], [8, 5, 5, 5, 3],
    ]

    for pat in patterns:
        for sp in np.linspace(0.145, 0.215, 8):
            rng = np.random.default_rng(int(sp * 10000) + sum(pat) % 100)
            configs.append(generate_pattern_init(pat, sp, rng))

    # Random placements (filter for quality)
    for seed in range(30):
        rng = np.random.default_rng(seed * 23 + 11)
        c = rng.uniform(0.08, 0.92, (N, 2))
        r, s = solve_radii_lp(c)
        if s > 1.8:
            configs.append(c)

    # Phase 1: Multi-start SLSQP optimization
    trial = 0
    max_trials = 200

    for base in configs:
        for noise in [0.0, 0.005, 0.012, 0.02]:
            if trial >= max_trials:
                break

            rng_t = np.random.default_rng(trial * 31 + 7)
            c_init = base + rng_t.normal(0, noise, base.shape)
            c_init = np.clip(c_init, 0.02, 0.98)

            r_init, _ = solve_radii_lp(c_init)
            r_init = np.maximum(r_init * 0.95, 1e-5)

            x0 = np.zeros(3 * N)
            x0[0::3] = c_init[:, 0]
            x0[1::3] = c_init[:, 1]
            x0[2::3] = r_init

            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt,
                               constraints=cons_opt,
                               options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                if res.success:
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_opt, s_opt = solve_radii_lp(c_opt)
                    if s_opt > best_sum:
                        best_sum = s_opt
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
            except Exception:
                pass
            trial += 1

    # Phase 2: Simulated annealing on center positions
    if best_centers is not None:
        curr_c = best_centers.copy()
        curr_r, curr_s = solve_radii_lp(curr_c)

        T = 0.008
        T_min = 1e-8
        T_decay = 0.9995

        for step in range(3000):
            T *= T_decay
            if T < T_min:
                T = T_min

            # Perturb centers
            scale = max(T, 0.0005)
            c_new = curr_c + np.random.randn(N, 2) * scale
            c_new = np.clip(c_new, 0.02, 0.98)

            r_new, s_new = solve_radii_lp(c_new)

            # Acceptance criterion
            delta = s_new - curr_s
            if delta > 0 or np.random.rand() < np.exp(delta / max(T, T_min)):
                curr_c, curr_r, curr_s = c_new, r_new, s_new
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_centers = curr_c.copy()
                    best_radii = curr_r.copy()

                    # Polish with SLSQP after improvement
                    x0 = np.zeros(3 * N)
                    x0[0::3] = curr_c[:, 0]
                    x0[1::3] = curr_c[:, 1]
                    x0[2::3] = np.maximum(curr_r * 0.97, 1e-5)
                    try:
                        res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt,
                                       constraints=cons_opt,
                                       options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
                        if res.success:
                            c_ref = np.column_stack((res.x[0::3], res.x[1::3]))
                            r_ref, s_ref = solve_radii_lp(c_ref)
                            if s_ref > curr_s:
                                curr_c, curr_r, curr_s = c_ref, r_ref, s_ref
                                if curr_s > best_sum:
                                    best_sum = curr_s
                                    best_centers = curr_c.copy()
                                    best_radii = curr_r.copy()
                    except Exception:
                        pass

    # Phase 3: Basin hopping with decaying perturbation
    if best_centers is not None:
        curr_c = best_centers.copy()
        curr_r = best_radii.copy()
        curr_s = best_sum

        for step in range(150):
            scale = 0.015 * (0.3 + 0.7 * np.exp(-step / 75.0))
            c_pert = curr_c + np.random.randn(N, 2) * scale
            c_pert = np.clip(c_pert, 0.015, 0.985)

            r_pert, s_pert = solve_radii_lp(c_pert)

            if s_pert > curr_s:
                curr_c, curr_r, curr_s = c_pert, r_pert, s_pert

                # Local SLSQP polish
                x0 = np.zeros(3 * N)
                x0[0::3] = curr_c[:, 0]
                x0[1::3] = curr_c[:, 1]
                x0[2::3] = np.maximum(curr_r * 0.97, 1e-5)
                try:
                    res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt,
                                   constraints=cons_opt,
                                   options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
                    if res.success:
                        c_ref = np.column_stack((res.x[0::3], res.x[1::3]))
                        r_ref, s_ref = solve_radii_lp(c_ref)
                        if s_ref > curr_s:
                            curr_c, curr_r, curr_s = c_ref, r_ref, s_ref
                except Exception:
                    pass

                if curr_s > best_sum:
                    best_sum = curr_s
                    best_centers = curr_c.copy()
                    best_radii = curr_r.copy()

    # Phase 4: Very fine local search via LP perturbation
    if best_centers is not None:
        for scale in [0.003, 0.001, 0.0003, 0.0001]:
            for _ in range(60):

                c_p = np.clip(c_p, 0.02, 0.98)
                r_p, s_p = solve_radii_lp(c_p)

                    best_sum = s_p
                    best_centers = c_p.copy()
                    best_radii = r_p.copy()

    c_final = best_centers.copy()
    r_final = best_radii.copy()

    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(c_final[i, 0], 1.0 - c_final[i, 0],
                 c_final[i, 1], 1.0 - c_final[i, 1])
        r_final[i] = min(r_final[i], mx - 1e-9)
        r_final[i] = max(0.0, r_final[i])

    for _ in range(100):

                r_final[j] = max(0.0, r_final[j])
                changed = True
        if not changed:
