import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist
import math

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

# Precompute constant LP constraint matrix structure
A_ub_global = np.zeros((NUM_PAIRS, N))
A_ub_global[np.arange(NUM_PAIRS), I_IDX] = 1.0
A_ub_global[np.arange(NUM_PAIRS), J_IDX] = 1.0

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    dists = pdist(centers)
    bounds = []
    for x, y in centers:
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub - 1e-9)))
        
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_global, b_ub=dists, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(N, 1e-5)

def objective_sl(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_sl(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap (>= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    r_sum = r[I_IDX] + r[J_IDX]
    c[4*N:] = dx**2 + dy**2 - r_sum**2
    return c

def polish_slsqp(centers, radii):
    """Refine configuration using gradient-based constrained optimization."""
    x0 = np.zeros(3*N)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': constraints_sl}
    
    try:
        res = minimize(objective_sl, x0, method='SLSQP', bounds=bounds_opt,
                       constraints=cons_opt, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if res.success:
            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
            r_opt = solve_lp_radii(c_opt)
            return c_opt, r_opt, np.sum(r_opt)
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def sa_search(start_centers, steps=2000, T0=0.06):
    """Simulated annealing over center coordinates with LP evaluation."""
    centers = start_centers.copy()
    radii = solve_lp_radii(centers)
    best_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    T = T0
    decay = 0.994
    
    for step in range(steps):
        # Adaptive perturbation scale based on temperature
        scale = 0.012 * (T / T0 + 0.1) * (1.0 + 0.5 * np.random.rand())
        pert = centers + np.random.randn(N, 2) * scale
        pert = np.clip(pert, 0.01, 0.99)
        
        r_new = solve_lp_radii(pert)
        s_new = np.sum(r_new)
        
        delta = s_new - best_sum
        # Accept if better, or with SA probability
        if delta > 0 or np.random.rand() < math.exp(delta / max(T, 1e-7)):
            centers = pert.copy()
            radii = r_new.copy()
            if s_new > best_sum:
                best_sum = s_new
                best_centers = centers.copy()
                best_radii = radii.copy()
                
        T *= decay
        
    return best_centers, best_radii, best_sum

def repulsion_init(rng):
    """Generate a well-spread initial configuration via repulsion dynamics."""
    c = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(150):
        for i in range(N):
            for j in range(i + 1, N):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = math.hypot(dx, dy)
                if d < 0.14 and d > 1e-6:
                    shift = (0.14 - d) / 2.0
                    ux, uy = dx / d, dy / d
                    c[i, 0] += ux * shift
                    c[i, 1] += uy * shift
                    c[j, 0] -= ux * shift
                    c[j, 1] -= uy * shift
        c = np.clip(c, 0.02, 0.98)
    return c

def generate_inits(rng):
    """Generate diverse structured initial configurations."""
    inits = []
    # Hexagonal grids with varying spacing
    for sp in [0.175, 0.195, 0.215]:
        pts = []
        y = sp / 2
        row = 0
        while len(pts) < N:
            x = sp / 2 + (row % 2) * sp / 2
            while x < 1.0 - sp / 2 and len(pts) < N:
                pts.append([x, y])
                x += sp
            y += sp * math.sqrt(3) / 2
            row += 1
        pts = np.array(pts[:N]) + rng.normal(0, 0.006, (N, 2))
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # Random uniform
    for _ in range(4):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # Corner/edge biased
    corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    mid_edges = [[0.5, 0.08], [0.5, 0.92], [0.08, 0.5], [0.92, 0.5]]
    base = rng.uniform(0.2, 0.8, (N - 8, 2))
    pts = np.vstack([corners, mid_edges, base])
    rng.shuffle(pts)
    inits.append(pts)
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).

    best_radii = None

        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r
            
    # Phase 2: Gradient polishing of best found
    if best_centers is not None:
        c_pol, r_pol, s_pol = polish_slsqp(best_centers, best_radii)
        if s_pol > best_sum:
            best_sum = s_pol
            best_centers = c_pol
            best_radii = r_pol

            best_sum = s
            best_centers = c
            best_radii = r
            
        c_pol, r_pol, s_pol = polish_slsqp(c, r)

            best_sum = s_pol
            best_centers = c_pol
            best_radii = r_pol
            
    # Fallback safety net
    if best_centers is None:
        best_centers = rng.uniform(0.1, 0.9, (N, 2))
        best_radii = solve_lp_radii(best_centers)
        best_sum = np.sum(best_radii)
        
    # Phase 4: Strict post-processing to guarantee validator compliance
    c_final = best_centers.copy()
    r_final = best_radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(c_final[i, 0], 1.0 - c_final[i, 0], 
                 c_final[i, 1], 1.0 - c_final[i, 1])
        r_final[i] = min(r_final[i], mx - 1e-9)
        r_final[i] = max(0.0, r_final[i])
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(80):
        changed = False
        for k in range(NUM_PAIRS):
            i, j = I_IDX[k], J_IDX[k]
            d = math.hypot(c_final[i, 0] - c_final[j, 0], 
                           c_final[i, 1] - c_final[j, 1])
            if d < r_final[i] + r_final[j] - 1e-11:
                exc = r_final[i] + r_final[j] - d
                r_final[i] -= exc * 0.5
                r_final[j] -= exc * 0.5
                r_final[i] = max(0.0, r_final[i])
                r_final[j] = max(0.0, r_final[j])
                changed = True
        if not changed:
            break
            
    final_sum = float(np.sum(r_final))
    return c_final, r_final, final_sum
