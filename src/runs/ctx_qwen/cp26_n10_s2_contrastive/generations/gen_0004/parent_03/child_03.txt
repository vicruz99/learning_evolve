import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)


def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)

    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])

    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(NUM_PAIRS), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]

    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, mx)))

    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success and np.all(res.x >= -1e-9):
                return np.maximum(res.x, 0.0)
        except Exception:
            continue
    return np.full(n, 0.01)


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints(x):
    """Inequality constraints using hypot for stable gradients at contact."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]

    c = np.empty(4 * N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2 * N] = 1.0 - cx - r
    c[2 * N:3 * N] = cy - r
    c[3 * N:4 * N] = 1.0 - cy - r

    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c[4 * N:] = np.hypot(dx, dy) - (r[I_IDX] + r[J_IDX])
    return c


def generate_hex_init(spacing, seed, margin=0.02):
    """Generate hexagonal lattice initialization with noise."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = margin + spacing / 2
    row = 0
    while idx < N and y < 1.0 - margin:
        x_start = margin + spacing / 2 + (row % 2) * spacing / 2.0
        col = 0
        while x_start + col * spacing < 1.0 - margin and idx < N:
            centers[idx] = [x_start + col * spacing, y]
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2.0
        row += 1
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.02, 0.98)


def generate_grid_init(spacing, seed):
    """Generate square grid initialization."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = spacing / 2
    while y < 1.0 - spacing / 2 and idx < N:
        x = spacing / 2
        while x < 1.0 - spacing / 2 and idx < N:
            centers[idx] = [x, y]
            idx += 1
            x += spacing
        y += spacing
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.02, 0.98)


def generate_pattern_init(pat, spacing, seed):
    """Generate pattern-based initialization with specified row counts."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = spacing * 0.5
    dy = spacing * np.sqrt(3) / 2 * 0.95
    for r_idx, cnt in enumerate(pat):
        shift = 0.0 if r_idx % 2 == 0 else spacing / 2.0
        x = spacing * 0.5 + shift
        for _ in range(cnt):
            if idx < N:
                centers[idx] = [x, y]
                idx += 1
            x += spacing
        y += dy
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    centers += rng.normal(0, 0.004, centers.shape)
    return np.clip(centers, 0.02, 0.98)


def generate_corner_init(seed):
    """Generate initialization that places circles near corners for larger radii."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    idx = 0

    corners = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
    for c in corners:
        if idx < N:
            centers[idx] = [c[0] + rng.normal(0, 0.02), c[1] + rng.normal(0, 0.02)]
            idx += 1

    edge_mids = [[0.5, 0.12], [0.5, 0.88], [0.12, 0.5], [0.88, 0.5]]
    for c in edge_mids:
        if idx < N:
            centers[idx] = [c[0] + rng.normal(0, 0.02), c[1] + rng.normal(0, 0.02)]
            idx += 1

    remaining = N - idx
    if remaining > 0:
        s = 0.15 + rng.uniform(0, 0.05)
        y = 0.35
        row = 0
        while idx < N and y < 0.65:
            x = 0.35 + (row % 2) * s / 2
            while x < 0.65 and idx < N:
                centers[idx] = [x + rng.normal(0, 0.005), y + rng.normal(0, 0.005)]
                idx += 1
                x += s
            y += s * np.sqrt(3) / 2
            row += 1

    while idx < N:
        centers[idx] = rng.uniform(0.2, 0.8, 2)
        idx += 1

    return np.clip(centers, 0.03, 0.97)


def run_packing():

    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': constraints}

    best_sum = 0.0
    best_centers = None
    best_radii = None

        inits.append(rng.uniform(0.05, 0.95, (N, 2)))

    def optimize_from(c_init, maxiter=10000):
        """Run SLSQP optimization from initial centers."""
        nonlocal best_sum, best_centers, best_radii

        r_init = solve_radii_lp(c_init)
        r_init = np.maximum(r_init * 0.96, 1e-5)

        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons_opt,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})

            if res.success:
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = solve_radii_lp(c_opt)
                s_opt = np.sum(r_opt)

                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()

        except Exception:
            pass
        return False

    for init in inits:

        curr_r = best_radii.copy()

            c_pert = np.clip(c_pert, 0.02, 0.98)

            s_pert = np.sum(r_pert)

                # Full SLSQP polish after successful jump
                x0 = np.zeros(3 * N)
                x0[0::3] = curr_c[:, 0]
                x0[1::3] = curr_c[:, 1]

                try:

            scale = rng.uniform(0.001, 0.006)

                r_final[j] = max(0.0, r_final[j])
                changed = True
        if not changed:
            break

    return c_final, r_final, float(np.sum(r_final))
