import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I, J = np.triu_indices(N, k=1)
N_PAIRS = len(I)

# Precompute sparse-like constraint structure for LP
A_LP = np.zeros((N_PAIRS, N))
A_LP[np.arange(N_PAIRS), I] = 1.0
A_LP[np.arange(N_PAIRS), J] = 1.0

def solve_lp_fast(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    # Margin to each boundary
    margins = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    margins = np.maximum(margins, 0.0)
    
    bounds = [(0.0, m) for m in margins]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= -1e-9):
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.zeros(N), 0.0

def get_objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2*N:])

def get_constraints(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap."""
    cx = x[0:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    c = np.empty(4*N + N_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c[4*N:] = np.hypot(dx, dy) - (r[I] + r[J])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    inits = []
    rng = np.random.default_rng(42)
    
    # Generate diverse hexagonal lattice initializations
    for s in np.linspace(0.13, 0.21, 8):
        c = np.zeros((N, 2))
        idx = 0
        y = 0.05
        row = 0
        while idx < N and y < 0.95:
            x = 0.05 + (row % 2) * s / 2.0
            while x < 0.95 and idx < N:
                c[idx] = [x, y]
                idx += 1
                x += s
            y += s * np.sqrt(3) / 2.0
            row += 1
        while idx < N:
            c[idx] = [0.5, 0.5]
            idx += 1
        c += rng.normal(0, 0.005, c.shape)
        c = np.clip(c, 0.02, 0.98)
        inits.append(c)
        
    # Add random uniform placements
    for _ in range(10):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    best_sum = -1.0
    best_centers = None
    
    # Phase 1: LP-evaluated stochastic hill climbing with re-heating
    for c_init in inits:
        curr_c = c_init.copy()
        _, curr_s = solve_lp_fast(curr_c)
        scale = 0.03
        no_improve = 0
        
        for step in range(5000):
            # Re-heat if stuck to escape local minima
            if no_improve > 800:
                scale = min(0.05, scale * 1.5)
                no_improve = 0
                
            # Adaptive perturbation scale
            current_scale = scale * (0.5 + 0.5 * (1.0 - step/5000.0))
            n_pert = rng.integers(1, 7)
            idx_pert = rng.choice(N, n_pert, replace=False)
            
            new_c = curr_c.copy()
            new_c[idx_pert] += rng.normal(0, current_scale, (n_pert, 2))
            new_c = np.clip(new_c, 0.005, 0.995)
            
            _, new_s = solve_lp_fast(new_c)
            
            if new_s > curr_s + 1e-7:
                curr_c = new_c
                _, curr_s = solve_lp_fast(curr_c)
                no_improve = 0
                if curr_s > best_sum:
                    best_sum = curr_s
                    best_centers = curr_c.copy()
            else:
                no_improve += 1
                scale = max(0.001, scale * 0.995)
                
    # Fallback safety net
    if best_centers is None:
        best_centers = inits[0]
        best_radii, best_sum = solve_lp_fast(best_centers)
    else:
        best_radii, best_sum = solve_lp_fast(best_centers)
        
    # Phase 2: High-precision SLSQP joint refinement
    x0 = np.concatenate([best_centers.ravel(), best_radii])
    bounds_opt = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
    try:
        res = minimize(get_objective, x0, method='SLSQP', bounds=bounds_opt,
                       constraints={'type': 'ineq', 'fun': get_constraints},
                       options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res.success:
            c_opt = res.x[:2*N].reshape(N, 2)
            r_opt, s_opt = solve_lp_fast(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
    except Exception:
        pass
        
    # Phase 3: Strict post-processing to guarantee validator compliance
    c_final = best_centers.copy()
    r_final = best_radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(c_final[i,0], 1.0-c_final[i,0], c_final[i,1], 1.0-c_final[i,1])
        if r_final[i] > mx:
            r_final[i] = max(0.0, mx - 1e-9)
            
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(50):
        changed = False
        for k in range(N_PAIRS):
            i, j = I[k], J[k]
            d = np.hypot(c_final[i,0]-c_final[j,0], c_final[i,1]-c_final[j,1])
            if d < r_final[i] + r_final[j] - 1e-11:
                exc = r_final[i] + r_final[j] - d
                r_final[i] -= exc/2.0
                r_final[j] -= exc/2.0
                changed = True
        if not changed:
            break
            
    r_final = np.maximum(r_final, 0.0)
    return c_final, r_final, float(np.sum(r_final))
