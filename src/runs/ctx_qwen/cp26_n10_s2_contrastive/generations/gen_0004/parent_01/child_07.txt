import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_hypot(x):
    """Inequality constraints using hypot for better gradient at contact."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c_overlap = np.hypot(dx, dy) - (r[I] + r[J])
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])

def constraints_squared(x):
    """Inequality constraints using squared distances."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c_overlap = dx**2 + dy**2 - (r[I] + r[J])**2
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I] = 1.0
    A_ub[np.arange(NUM_PAIRS), J] = 1.0
    
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds_r = []
    for i in range(n):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        ub = max(1e-10, min(mx, 0.5))
        bounds_r.append((0.0, ub))
        
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success and res.fun is not None:
                return np.maximum(res.x, 0.0), float(-res.fun)
        except Exception:
            continue
    return np.zeros(n), 0.0

def make_hex_init(spacing, seed, margin=0.03, offset_x=0, offset_y=0):
    """Generate hexagonal lattice initialization with parameters."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    row = 0
    y = margin + spacing/2 + offset_y
    while idx < N and y < 1.0 - margin:
        x_start = margin + (row % 2) * spacing/2 + offset_x
        col = 0
        while x_start + col * spacing < 1.0 - margin and idx < N:
            centers[idx] = [x_start + col * spacing, y]
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2
        row += 1
    while idx < N:
        centers[idx] = rng.uniform(margin, 1.0-margin, 2)
        idx += 1
    centers += rng.normal(0, 0.002, centers.shape)
    return np.clip(centers, margin, 1.0-margin)

def make_structured_row_init(pat, seed, dy=0.17, dx=0.17):
    """Generate structured row pattern initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = 0.05
    for r_idx, cnt in enumerate(pat):
        shift = 0.0 if r_idx % 2 == 0 else dx / 2.0
        x = 0.05 + shift
        for _ in range(cnt):
            if idx < N:
                centers[idx] = [x + rng.uniform(-0.005, 0.005), y + rng.uniform(-0.005, 0.005)]
                idx += 1
            x += dx
        y += dy
    while idx < N:
        centers[idx] = rng.uniform(0.1, 0.9, 2)
        idx += 1
    return np.clip(centers, 0.03, 0.97)

def run_optimization(c0, cons_func, maxiter=20000):
    """Run SLSQP optimization from given initial point."""
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': cons_func}
    
    r0, _ = solve_lp_radii(c0)
    r0 = np.maximum(r0 * 0.95, 1e-5)
    
    x0 = np.zeros(3 * N)
    x0[0::3] = c0[:, 0]
    x0[1::3] = c0[:, 1]
    x0[2::3] = r0
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        
        cx = res.x[0::3]
        cy = res.x[1::3]
        curr_centers = np.column_stack((cx, cy))
        
        r_opt, s_opt = solve_lp_radii(curr_centers)
        return curr_centers, r_opt, s_opt
    except Exception:
        return None, None, 0.0

def iterative_refinement(c0, cons_func, n_iter=3):
    """Iterate SLSQP + LP refinement multiple times."""
    centers = c0
    for _ in range(n_iter):
        centers, radii, s = run_optimization(centers, cons_func, maxiter=15000)
        if centers is None:
            break
    if centers is not None:
        radii, s = solve_lp_radii(centers)
        return centers, radii, s
    return None, None, 0.0
