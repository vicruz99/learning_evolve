import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """
    Inequality constraints (must be >= 0):
    - Pairwise distance >= sum of radii (using hypot for non-vanishing gradient)
    - Boundary clearance: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c_overlap = np.hypot(dx, dy) - (r[I] + r[J])
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I] = 1.0
    A_ub[np.arange(NUM_PAIRS), J] = 1.0
    
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds_r = []
    for i in range(n):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        ub = max(1e-12, min(mx, 0.5))
        bounds_r.append((0.0, ub))
    
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success:
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(n), 0.0

def make_row_pattern(rows_pattern, y_spacing=0.165, margin=0.04, shift=0.085):
    """Generate a row-based pattern with specified number of circles per row."""
    centers = []
    y = margin
    for r_idx, count in enumerate(rows_pattern):
        if len(centers) >= N:
            break
        x_start = margin + (shift if r_idx % 2 == 1 else 0.0)
        for c_idx in range(count):
            if len(centers) >= N:
                break
            x = x_start + c_idx * (1.0 - 2.0 * margin) / max(count - 1, 1) if count > 1 else 0.5
            centers.append([x, y])
        y += y_spacing
    while len(centers) < N:
        centers.append([0.5 + (len(centers) - N) * 0.05, 0.5])
    return np.array(centers[:N])

def make_hex_init(seed, spacing=0.17, margin=0.04):
    """Generate hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    row = 0
    y = margin + rng.uniform(-0.01, 0.01)
    while idx < N and y < 1.0 - margin:
        x_start = margin + (row % 2) * spacing / 2.0 + rng.uniform(-0.005, 0.005)
        col = 0
        while x_start + col * spacing < 1.0 - margin and idx < N:
            centers[idx] = [x_start + col * spacing, y]
            idx += 1
            col += 1
        y += spacing * math.sqrt(3) / 2.0
        row += 1
    while idx < N:
        centers[idx] = rng.uniform(margin, 1.0 - margin, 2)
        idx += 1
    centers += rng.normal(0, 0.003, centers.shape)
    return np.clip(centers, 0.02, 0.98)

def make_corner_anchored(seed):
    """Place circles anchored near corners and edges for larger boundary radii."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    
    # Corner regions
    corners = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
    for cx, cy in corners:
        for _ in range(2):
            if idx < N:
                centers[idx] = [cx + rng.uniform(-0.05, 0.05), cy + rng.uniform(-0.05, 0.05)]
                idx += 1
    
    # Edge regions
    edge_pts = [
        [0.5, 0.08], [0.5, 0.92],
        [0.08, 0.5], [0.92, 0.5],
        [0.25, 0.08], [0.75, 0.08], [0.25, 0.92], [0.75, 0.92],
        [0.08, 0.25], [0.08, 0.75], [0.92, 0.25], [0.92, 0.75]
    ]
    for cx, cy in edge_pts:
        if idx < N:
            centers[idx] = [cx + rng.uniform(-0.03, 0.03), cy + rng.uniform(-0.03, 0.03)]
            idx += 1
    
    # Fill remaining
    while idx < N:
        centers[idx] = rng.uniform(0.15, 0.85, 2)
        idx += 1
    
    return np.clip(centers, 0.03, 0.97)

def coordinate_wise_optimize(centers, radii, n_iter=5):
    """Optimize one circle at a time while keeping others fixed."""
    n = centers.shape[0]
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(n_iter):
        for k in range(n):
            # Fix all other circles, optimize circle k
            x0 = np.array([centers[k, 0], centers[k, 1], radii[k]])
            
            def obj_k(x):
                return -x[2]
            
            def cons_k(x):
                cx, cy, r = x
                c = []
                # Boundary
                c.append(cx - r)
                c.append(1.0 - cx - r)
                c.append(cy - r)
                c.append(1.0 - cy - r)
                # Non-overlap with others
                for i in range(n):
                    if i != k:
                        dx = cx - centers[i, 0]
                        dy = cy - centers[i, 1]
                        d = math.hypot(dx, dy)
                        c.append(d - r - radii[i])
                return np.array(c)
            
            try:
                res = minimize(obj_k, x0, method='SLSQP', 
                              bounds=[(0.01, 0.99), (0.01, 0.99), (1e-6, 0.5)],
                              constraints={'type': 'ineq', 'fun': cons_k},
                              options={'maxiter': 2000, 'ftol': 1e-12})
                if res.success:
                    centers[k, 0] = res.x[0]
                    centers[k, 1] = res.x[1]
                    radii[k] = max(1e-6, res.x[2])
            except Exception:
                pass
    
    return centers, radii
