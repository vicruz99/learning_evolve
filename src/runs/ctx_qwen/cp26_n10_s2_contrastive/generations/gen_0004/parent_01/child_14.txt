1. **Diverse SLSQP Starts**: Run SLSQP from ~60 structured (hexagonal) and random initializations. After each SLSQP run, we extract the true maximal radii for those optimized centers using Linear Programming (LP). LP is exact and extremely fast for fixed centers.
2. **LP-Driven Local Search**: This is the main improvement. Instead of relying on SLSQP's gradient estimates which can be noisy near constraints, we treat the LP sum of radii as a black-box objective function of the centers. We perform an intensive coordinate-wise hill climbing: repeatedly pick a random circle, perturb its position, solve the LP, and accept the move if it increases the total sum. This allows circles to "slide" into micro-gaps and boundaries more naturally than gradient methods. The step size anneals slowly to fine-tune positions.
3. **Final Polishing & Validation**: Run one last SLSQP pass on the best configuration found by the local search to ensure mathematical optimality at the constraints boundaries. This hybrid approach leverages the speed of LP for radius extraction and the precision of SLSQP for constraint satisfaction, while the local search aggressively exploits the non-convex landscape.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute constant parts of the LP constraint matrix
A_ub = np.zeros((NUM_PAIRS, N))
A_ub[np.arange(NUM_PAIRS), I] = 1.0
A_ub[np.arange(NUM_PAIRS), J] = 1.0
c_obj_lp = -np.ones(N)

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    try:
        dx = centers[I, 0] - centers[J, 0]
        dy = centers[I, 1] - centers[J, 1]
        b_ub = np.sqrt(dx*dx + dy*dy)
        
        # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
        ub = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]), 
                        np.minimum(centers[:,1], 1.0-centers[:,1]))
        bounds = [(0.0, max(1e-9, u)) for u in ub]
        
        res = linprog(c_obj_lp, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.full(N, 0.01), 0.0

def objective_slsqp(x):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: boundary clearance and pairwise non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Squared distance constraints are smooth and differentiable everywhere
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c_overlap = dx**2 + dy**2 - (r[I] + r[J])**2
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])

def generate_init(seed, style='hex'):
    """Generate structured or random initial center configurations."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    if style == 'hex':
        s = 0.14 + rng.uniform(0, 0.06)
        idx = 0
        row = 0
        y = s/2 + rng.uniform(0, 0.05)
        while idx < N and y < 1.0 - s/2:
            x_start = s/2 + (row % 2) * s/2 + rng.uniform(0, 0.02)
            col = 0
            while x_start + col*s < 1.0 - s/2 and idx < N:
                centers[idx] = [x_start + col*s, y]
                idx += 1
                col += 1
            y += s * np.sqrt(3) / 2
            row += 1
        while idx < N:
            centers[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
    else:
        centers = rng.uniform(0.1, 0.9, (N, 2))
    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng_main = np.random.RandomState(42)
    
    # Phase 1: Broad multi-start search with SLSQP
    for seed in range(60):
        style = 'hex' if seed < 45 else 'random'
        c0 = generate_init(seed, style)
        r0, _ = solve_lp_radii(c0)
        r0 = np.maximum(r0, 1e-4)
        
        x0 = np.zeros(3*N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r0 * 0.92  # Slight shrinkage guarantees initial strict feasibility
        
        try:
            res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons_opt, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            cx = res.x[0::3]
            cy = res.x[1::3]
            curr_c = np.column_stack((cx, cy))
            curr_r, curr_s = solve_lp_radii(curr_c)
            
            if curr_s > best_sum:
                best_sum = curr_s
                best_centers = curr_c.copy()
                best_radii = curr_r.copy()
        except Exception:
            continue
            
    # Phase 2: Intensive local search on centers using LP objective
    # This phase is crucial for escaping SLSQP local minima by directly optimizing 
    # the exact maximal radius sum for perturbed center layouts.
    if best_centers is not None:
        c_curr = best_centers.copy()
        r_curr, s_curr = solve_lp_radii(c_curr)
        step = 0.025
        
        for _ in range(25000):
            i = rng_main.randint(N)
            best_move = None
            best_val = s_curr
            
            orig_pos = c_curr[i].copy()
            # Try multiple perturbations for circle i before accepting the best
            for _ in range(5):
                c_curr[i] += rng_main.uniform(-step, step, 2)
                c_curr[i] = np.clip(c_curr[i], 0.005, 0.995)
                _, s_new = solve_lp_radii(c_curr)
                if s_new > best_val:
                    best_val = s_new
                    best_move = c_curr[i].copy()
                c_curr[i] = orig_pos  # revert for next candidate
                
            if best_move is not None:
                c_curr[i] = best_move
                r_curr, s_curr = solve_lp_radii(c_curr)
                if s_curr > best_sum:
                    best_sum = s_curr
                    best_centers = c_curr.copy()
                    best_radii = r_curr.copy()
                    
            step = max(0.002, step * 0.999)
            
    # Phase 3: Final SLSQP polishing on the best configuration found
    if best_centers is not None:
        c_fin = best_centers.copy()
        r_fin, s_fin = solve_lp_radii(c_fin)
        r_fin = np.maximum(r_fin, 1e-5)
        x0 = np.zeros(3*N)
        x0[0::3] = c_fin[:, 0]
        x0[1::3] = c_fin[:, 1]
        x0[2::3] = r_fin * 0.98
        
        try:
            res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons_opt, options={'maxiter': 12000, 'ftol': 1e-15, 'disp': False})
            c_polished = np.column_stack((res.x[0::3], res.x[1::3]))
            r_polished, s_polished = solve_lp_radii(c_polished)
            if s_polished > best_sum:
                best_sum = s_polished
                best_centers = c_polished
                best_radii = r_polished
        except Exception:
            pass

    # Strict post-processing to guarantee validator compliance
    radii = best_radii.copy()
    centers = best_centers.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(radii[i], 0.0)
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc/2.0
                    radii[j] -= exc/2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
