import numpy as np
from scipy.optimize import minimize, linprog, NonlinearConstraint, Bounds, LinearConstraint

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_eq(x):
    """
    Inequality constraints (must be >= 0):
    - Pairwise distance >= sum of radii
    - Boundary clearance
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dists = np.hypot(dx, dy)
    c_overlap = dists - (r[I] + r[J])
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])

def constraints_sq(x):
    """
    Squared distance constraints (must be >= 0).
    Numerically stable but can be less informative for far circles.
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    c_overlap = dx**2 + dy**2 - (r[I] + r[J])**2
    
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I] = 1.0
    A_ub[np.arange(NUM_PAIRS), J] = 1.0
    
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds_r = []
    for i in range(n):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        ub = max(0.0, min(mx, 0.5))
        bounds_r.append((0.0, ub))
    
    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success:
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(n), 0.0

def make_hex_init(seed, spacing=None, margin=None):
    """Generate hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    row = 0
    
    if spacing is None:
        spacing = 0.14 + rng.uniform(0.01, 0.07)
    if margin is None:
        margin = spacing / 2.0
    
    y = margin + rng.uniform(0, spacing * 0.3)
    while idx < N and y < 1.0 - margin:
        x_start = margin + (row % 2) * spacing / 2.0
        col = 0
        while x_start + col * spacing < 1.0 - margin and idx < N:
            centers[idx] = [x_start + col * spacing, y]
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2
        row += 1
    
    while idx < N:
        centers[idx] = rng.uniform(margin, 1.0 - margin, 2)
        idx += 1
    
    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.02, 0.98)

def make_grid_init(seed, spacing=None):

def make_row_pattern_init(row_pattern, seed):

    centers = np.zeros((N, 2))

        idx += 1
    
    return np.clip(centers, 0.02, 0.98)

def make_valid(centers, radii):
    """Project configuration to strictly satisfy all constraints."""
    for i in range(N):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if radii[i] > max_r:
            radii[i] = max(0.0, max_r - 1e-9)
    
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if d < radii[i] + radii[j] - 1e-12:
                    excess = radii[i] + radii[j] - d
                    radii[i] -= excess / 2.0
                    radii[j] -= excess / 2.0
                    changed = True
        if not changed:
            break
    
    radii = np.maximum(radii, 0.0)
    return centers, radii

def optimize_from_start(c0, r0, method='SLSQP', maxiter=10000, ftol=1e-14):
    """Run optimization from a given starting point."""
    x0 = np.zeros(3 * N)
    x0[0::3] = c0[:, 0]
    x0[1::3] = c0[:, 1]
    x0[2::3] = np.maximum(r0, 1e-5)
    
    bounds = Bounds(
        lb=[0.0] * (2*N) + [1e-6] * N,
        ub=[1.0] * (2*N) + [0.5] * N
    )
    cons = NonlinearConstraint(constraints_eq, 0, np.inf)
    
    try:
        if method == 'trust-constr':
            res = minimize(objective, x0, method='trust-constr', bounds=bounds,
                          constraints=[cons],
                          options={'maxiter': maxiter, 'gtol': ftol, 'disp': False})
        else:
            cons_dict = {'type': 'ineq', 'fun': constraints_eq}
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                          constraints=cons_dict,
                          options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        
        if res.success or res.status >= 0:
            cx = res.x[0::3]
            cy = res.x[1::3]
            c_opt = np.column_stack((cx, cy))
            r_opt, s_opt = solve_lp_radii(c_opt)
            return c_opt, r_opt, s_opt
    except Exception:
        pass
    
    return None, None, 0.0
