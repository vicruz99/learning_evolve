2. **SLSQP Joint Optimization**: Use SLSQP to simultaneously optimize centers and radii. The constraint $d_{ij} \ge r_i + r_j$ is formulated using `np.hypot` for smooth, non-vanishing gradients at contact points.
3. **Force-Based Center Relaxation**: After SLSQP converges, run a repulsive force simulation on the centers. Circles are pushed apart proportionally to their overlap/tightness. 5. **Basin Hopping**: From the best solution found, perform controlled random perturbations followed by the full optimization pipeline to explore adjacent basins.
6. **Strict Post-Processing**: Ensure all constraints are satisfied within the validator's tolerance ($10^{-12}$) by clipping radii to boundary distances and iteratively reducing overlapping radii.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

def get_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(NUM_PAIRS), J_IDX] = 1.0
    
    dx = centers[I_IDX, 0] - centers[J_IDX, 0]
    dy = centers[I_IDX, 1] - centers[J_IDX, 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = []
    for i in range(n):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        bounds.append((0.0, max(1e-9, mx)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0), -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_func(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c[4*N:] = np.hypot(dx, dy) - (r[I_IDX] + r[J_IDX])
    return c

def relax_centers(centers, radii, steps=100):
    """Repulsive force simulation to push circles apart and escape local minima."""
    c = centers.copy()
    r = radii.copy()
    for step in range(steps):
        dt = 0.05 * (1.0 - step/steps)
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i+1, N):
                dx = c[i,0] - c[j,0]
                dy = c[i,1] - c[j,1]
                d = np.hypot(dx, dy)
                if d < 1e-12:
                    d = 1e-12
                    dx, dy = 0.0, 0.0
                rep = (r[i] + r[j] - d) / d
                fx = dx * rep
                fy = dy * rep
                forces[i] += [fx, fy]
                forces[j] -= [fx, fy]
        c += forces * dt
        c = np.clip(c, 1e-6, 1.0 - 1e-6)
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns centers (26, 2), radii (26,), and sum_radii.
    """
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-5, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': constraints_func}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng = np.random.RandomState(42)
    inits = []
    
    # 1. Hexagonal lattice initializations with varying density and alignment
    for seed in range(40):
        s = 0.13 + rng.uniform(0, 0.07)
        centers = np.zeros((N, 2))
        idx = 0
        row = 0
        y = s/2 + rng.uniform(0, 0.04)
        while idx < N and y < 1.0 - s/2:
            x_start = s/2 + (row % 2) * s/2 + rng.uniform(0, 0.02)
            col = 0
            while x_start + col*s < 1.0 - s/2 and idx < N:
                centers[idx] = [x_start + col*s, y]
                idx += 1
                col += 1
            y += s * np.sqrt(3) / 2
            row += 1
        while idx < N:
            centers[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
        centers += rng.normal(0, 0.008, centers.shape)
        inits.append(np.clip(centers, 0.02, 0.98))
        
    # 2. Random uniform placements
    for seed in range(30):
        centers = rng.uniform(0.15, 0.85, (N, 2))
        inits.append(centers)
        
    # 3. Corner/Edge biased placements
    for seed in range(20):
        centers = rng.uniform(0.05, 0.95, (N, 2))
        # Force some points closer to corners
        corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        for k, crn in enumerate(corners):
            centers[k] = crn + rng.normal(0, 0.02, 2)
        inits.append(np.clip(centers, 0.01, 0.99))
        
    # Main optimization loop
    for c0 in inits:
        r0, _ = get_lp_radii(c0)
        r0 = np.maximum(r0, 1e-4)
        x0 = np.zeros(3*N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r0 * 0.92
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt,
                           constraints=cons_opt, options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            cx = res.x[0::3]
            cy = res.x[1::3]
            c_opt = np.column_stack((cx, cy))
            r_opt, s_opt = get_lp_radii(c_opt)
            
            if s_opt > best_sum:

                best_centers = c_opt.copy()
                best_radii = r_opt.copy()

                curr_c, curr_r = c_opt, r_opt

                        best_centers = c_rel.copy()
                        best_radii = r_new.copy()
                        curr_c, curr_r = c_rel, r_new
                    else:
                        break
        except Exception:
            continue
            
    # Basin hopping from best solution
    if best_centers is not None:
        for step in range(50):

            c_pert = best_centers + rng.normal(0, noise_scale, (N, 2))
            c_pert = np.clip(c_pert, 0.02, 0.98)
            
            r0, _ = get_lp_radii(c_pert)
            r0 = np.maximum(r0, 1e-4)
            x0 = np.zeros(3*N)
            x0[0::3] = c_pert[:, 0]
            x0[1::3] = c_pert[:, 1]
            x0[2::3] = r0 * 0.92
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt,
                               constraints=cons_opt, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                cx = res.x[0::3]
                cy = res.x[1::3]
                c_opt = np.column_stack((cx, cy))
                r_opt, s_opt = get_lp_radii(c_opt)
                
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
                    
                    curr_c, curr_r = c_opt, r_opt
                    for _ in range(3):
                        c_rel = relax_centers(curr_c, curr_r, steps=60)
                        r_new, s_new = get_lp_radii(c_rel)
                        if s_new > best_sum:
                            best_sum = s_new
                            best_centers = c_rel.copy()
                            best_radii = r_new.copy()
                            curr_c, curr_r = c_rel, r_new
                        else:
                            break
            except Exception:
                continue
                
    # Fallback safety net
    if best_centers is None:
        best_centers = inits[0]
        best_radii, best_sum = get_lp_radii(best_centers)
        
    # Strict post-processing to guarantee validity
    radii = best_radii.copy()
    centers = best_centers.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        radii[i] = min(radii[i], mx - 1e-9)
        radii[i] = max(radii[i], 0.0)
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    exc = radii[i] + radii[j] - d
                    radii[i] -= exc/2.0
                    radii[j] -= exc/2.0
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
