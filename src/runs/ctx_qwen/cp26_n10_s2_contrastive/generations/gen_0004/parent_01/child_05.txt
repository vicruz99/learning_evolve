import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute LP constraint matrix
A_LP = np.zeros((NUM_PAIRS, N))
A_LP[np.arange(NUM_PAIRS), I] = 1.0
A_LP[np.arange(NUM_PAIRS), J] = 1.0


def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    c_obj = -np.ones(N)
    dx = centers[I, 0] - centers[J, 0]
    dy = centers[I, 1] - centers[J, 1]
    b_ub = np.hypot(dx, dy)

    bounds_r = []
    for i in range(N):
        mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        ub = max(0.0, min(mx, 0.5))
        bounds_r.append((0.0, ub))

    for method in ['highs', 'interior-point']:
        try:
            res = linprog(c_obj, A_ub=A_LP, b_ub=b_ub, bounds=bounds_r, method=method)
            if res.success:
                return np.maximum(res.x, 0.0), -res.fun
        except Exception:
            continue
    return np.zeros(N), 0.0


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints(x):
    """Inequality constraints using non-squared distance for better gradients."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]

    dx = cx[I] - cx[J]
    dy = cy[I] - cy[J]
    dists = np.hypot(dx, dy)
    c_overlap = dists - (r[I] + r[J])

    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    return np.concatenate([c_overlap, c_bound])


def make_hex_init(spacing, margin=0.03, noise=0.003, seed=42):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    idx = 0
    row = 0
    y = margin + spacing / 2
    while idx < N and y < 1.0 - margin:
        x_start = margin + (row % 2) * spacing / 2
        col = 0
        while x_start + col * spacing < 1.0 - margin and idx < N:
            centers[idx] = [x_start + col * spacing, y]
            idx += 1
            col += 1
        y += spacing * np.sqrt(3) / 2
        row += 1
    while idx < N:
        centers[idx] = rng.uniform(margin, 1 - margin, 2)
        idx += 1
    centers += rng.normal(0, noise, centers.shape)
    return np.clip(centers, 0.01, 0.99)

def run_slsqp_optimization(c0, r0, bounds_opt, cons_opt, maxiter=10000, ftol=1e-14):
    """Run SLSQP from given starting point and return optimized centers."""
    x0 = np.zeros(3 * N)
    x0[0::3] = c0[:, 0]
    x0[1::3] = c0[:, 1]
    x0[2::3] = r0

    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds_opt, constraints=cons_opt,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        cx = res.x[0::3]
        cy = res.x[1::3]
        return np.column_stack((cx, cy))
    except Exception:
        return c0
