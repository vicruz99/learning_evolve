This reduces the problem to optimizing only the 52 center coordinates over a piecewise-linear, concave-like objective landscape.
2. **Adaptive Hill Climbing (Basin Hopping)**: We run a stochastic search over centers, perturbing them and evaluating the LP objective. import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIRS_I)

# Precompute LP constraint matrix structure for speed
A_LP = np.zeros((NUM_PAIRS, N))
A_LP[np.arange(NUM_PAIRS), PAIRS_I] = 1.0
A_LP[np.arange(NUM_PAIRS), PAIRS_J] = 1.0

def get_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    dx = centers[PAIRS_I, 0] - centers[PAIRS_J, 0]
    dy = centers[PAIRS_I, 1] - centers[PAIRS_J, 1]
    b_ub = np.hypot(dx, dy)
    
    x_c, y_c = centers[:, 0], centers[:, 1]
    ub = np.minimum(np.minimum(x_c, 1.0 - x_c), np.minimum(y_c, 1.0 - y_c))
    bounds = [(0.0, max(1e-9, b)) for b in ub]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='interior-point')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(N, 0.02)

def joint_objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def joint_constraints(x):
    """Inequality constraints: boundary clearance and pairwise non-overlap (>= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[PAIRS_I] - cx[PAIRS_J]
    dy = cy[PAIRS_I] - cy[PAIRS_J]
    c[4*N:] = np.hypot(dx, dy) - (r[PAIRS_I] + r[PAIRS_J])
    return c

def generate_init(style, seed):
    """Generate initial center configurations."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    
    if style == 'hex':
        sp = 0.18 + rng.uniform(-0.02, 0.04)
        idx = 0
        row = 0
        y = 0.04
        while idx < N and y < 0.96:
            x_start = 0.04 + (row % 2) * sp / 2.0
            col = 0
            while x_start + col * sp < 0.96 and idx < N:
                centers[idx] = [x_start + col * sp, y]
                idx += 1
                col += 1
            y += sp * np.sqrt(3) / 2.0
            row += 1
        while idx < N:
            centers[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
    elif style == 'grid':
        sp = 0.19 + rng.uniform(-0.02, 0.02)
        idx = 0
        y = 0.04
        while y < 0.96 and idx < N:
            x = 0.04
            while x < 0.96 and idx < N:
                centers[idx] = [x, y]
                idx += 1
                x += sp
            y += sp
        while idx < N:
            centers[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
    else:
        centers = rng.uniform(0.08, 0.92, (N, 2))
        
    centers += rng.normal(0, 0.005, centers.shape)
    return np.clip(centers, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    """
    bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N
    cons_opt = {'type': 'ineq', 'fun': joint_constraints}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    rng_main = np.random.default_rng(42)
    
    # Phase 1: Generate diverse initializations
    inits = []
    styles = ['hex', 'grid', 'rand']
    for s_idx, style in enumerate(styles):
        for seed in range(8):
            inits.append(generate_init(style, seed=s_idx*100 + seed))
            
    # Phase 2: Basin Hopping / Hill Climbing on LP objective
    for base in inits:
        current_c = base.copy()
        current_r = get_lp_radii(current_c)
        current_s = np.sum(current_r)
        
        sigma = 0.015
        # Adaptive hill climbing
        for step in range(1200):
            sigma *= 0.9992
            pert = current_c + rng_main.normal(0, sigma, current_c.shape)
            pert = np.clip(pert, 0.01, 0.99)
            
            new_r = get_lp_radii(pert)
            new_s = np.sum(new_r)
            
            if new_s > current_s:
                current_c, current_r, current_s = pert, new_r, new_s
                sigma = min(0.02, sigma * 1.15) # Explode step size on success
                
            # Interleaved SLSQP polishing
            if step % 300 == 299 and step < 900:
                x0 = np.zeros(3*N)
                x0[0::3] = current_c[:, 0]
                x0[1::3] = current_c[:, 1]
                x0[2::3] = np.maximum(current_r * 0.97, 1e-5)
                
                try:
                    res = minimize(joint_objective, x0, method='SLSQP', bounds=bounds_opt,
                                   constraints=cons_opt,
                                   options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                    if res.success:
                        c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                        r_opt = get_lp_radii(c_opt)
                        s_opt = np.sum(r_opt)
                        if s_opt > current_s:
                            current_c, current_r, current_s = c_opt, r_opt, s_opt
                            sigma = 0.01 # Reset noise for new basin
                except Exception:
                    pass
                    
        if current_s > best_sum:
            best_sum = current_s
            best_centers = current_c.copy()
            best_radii = current_r.copy()
            
    # Fallback safety
    if best_centers is None:
        best_centers = inits[0]
        best_radii = get_lp_radii(best_centers)
        best_sum = np.sum(best_radii)
        
    # Phase 3: Final high-precision SLSQP polish
    x0_final = np.zeros(3*N)
    x0_final[0::3] = best_centers[:, 0]
    x0_final[1::3] = best_centers[:, 1]
    x0_final[2::3] = np.maximum(best_radii * 0.98, 1e-5)
    
    try:
        res_final = minimize(joint_objective, x0_final, method='SLSQP', bounds=bounds_opt,
                             constraints=cons_opt,
                             options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if res_final.success:
            c_final = np.column_stack((res_final.x[0::3], res_final.x[1::3]))
            r_final = get_lp_radii(c_final)
            s_final = np.sum(r_final)
            if s_final > best_sum:
                best_centers, best_radii, best_sum = c_final, r_final, s_final
    except Exception:
        pass

    # Phase 4: Strict post-processing to guarantee validator compliance
    c_out = best_centers.copy()
    r_out = best_radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx = min(c_out[i, 0], 1.0 - c_out[i, 0], c_out[i, 1], 1.0 - c_out[i, 1])
        r_out[i] = min(r_out[i], mx - 1e-9)
        r_out[i] = max(0.0, r_out[i])
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(60):
        changed = False
        for k in range(NUM_PAIRS):
            i, j = PAIRS_I[k], PAIRS_J[k]
            d = np.hypot(c_out[i,0]-c_out[j,0], c_out[i,1]-c_out[j,1])
            if d < r_out[i] + r_out[j] - 1e-11:
                exc = r_out[i] + r_out[j] - d
                r_out[i] -= exc * 0.5
                r_out[j] -= exc * 0.5
                r_out[i] = max(0.0, r_out[i])
                r_out[j] = max(0.0, r_out[j])
                changed = True
        if not changed:
            break
            
    return c_out, r_out, float(np.sum(r_out))
