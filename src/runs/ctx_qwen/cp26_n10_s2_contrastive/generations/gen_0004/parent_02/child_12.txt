def objective(x):

def constraints(x):

    cy = x[1::3]

def try_slsqp_optimize(c_init, r_init, maxiter=8000):
    """Run SLSQP optimization and return best result."""

    x0[1::3] = c_init[:, 1]

    try:

                       constraints=cons_opt,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})

            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
            r_opt = get_lp_radii(c_opt)
            return c_opt, r_opt
    except Exception:
        pass
    return None, None
