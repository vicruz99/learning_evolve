import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIRS_I)

# Precompute constant LP constraint matrix for pairwise distances
A_ub_lp = np.zeros((NUM_PAIRS, N))
A_ub_lp[np.arange(NUM_PAIRS), PAIRS_I] = 1.0
A_ub_lp[np.arange(NUM_PAIRS), PAIRS_J] = 1.0

def get_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Vectorized distance computation
    dx = centers[PAIRS_I, 0] - centers[PAIRS_J, 0]
    dy = centers[PAIRS_I, 1] - centers[PAIRS_J, 1]
    b_ub = np.hypot(dx, dy)
    
    # Boundary limits for each circle
    bounds = []
    for i in range(n):
        mx, my = centers[i]
        ub = min(mx, 1.0 - mx, my, 1.0 - my)
        bounds.append((0.0, max(1e-9, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub_lp, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.01)

def mc_lp_optimize(initial_centers, n_iter=6000, init_step=0.025):
    """Monte Carlo local search optimizing centers, using LP for radii evaluation."""
    rng = np.random.default_rng(np.random.randint(0, 100000))
    curr = initial_centers.copy()
    curr_r = get_lp_radii(curr)
    curr_s = np.sum(curr_r)
    
    best_c = curr.copy()
    best_r = curr_r.copy()
    best_s = curr_s
    step = init_step
    
    for i in range(n_iter):
        # Adaptive step size schedule
        step = init_step * (0.999 ** i) + 0.0005
        
        # Perturb a random subset of circles (1 to 5)
        k = rng.choice([1, 2, 3, 5], p=[0.3, 0.3, 0.2, 0.2])
        idx = rng.choice(N, k, replace=False)
        
        new_c = curr.copy()
        new_c[idx] += rng.normal(0, step, size=(k, 2))
        new_c = np.clip(new_c, 0.002, 0.998)
        
        new_r = get_lp_radii(new_c)
        new_s = np.sum(new_r)
        
        # Accept if improves
        if new_s > curr_s:
            curr = new_c
            curr_r = new_r
            curr_s = new_s
            if curr_s > best_s:
                best_s = curr_s
                best_c = curr.copy()
                best_r = curr_r.copy()
                
    return best_c, best_r, best_s

def objective_slsqp(x):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: boundary clearance and pairwise non-overlap."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4*N + NUM_PAIRS)
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    dx = cx[PAIRS_I] - cx[PAIRS_J]
    dy = cy[PAIRS_I] - cy[PAIRS_J]
    c[4*N:] = np.hypot(dx, dy) - (r[PAIRS_I] + r[PAIRS_J])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """

    inits = []
    rng_main = np.random.default_rng(42)
    
    # Hexagonal lattices with varying spacing and vertical shifts
    for sp in np.linspace(0.175, 0.225, 5):

            c_hex = np.zeros((N, 2))
            idx = 0
            y = 0.04 + shift_y
            row = 0
            while idx < N and y < 0.96:
                x_start = 0.04 + (row % 2) * sp / 2.0

                    idx += 1
                    col += 1
                y += sp * np.sqrt(3) / 2.0
                row += 1
            while idx < N:
                c_hex[idx] = rng_main.uniform(0.2, 0.8, 2)
                idx += 1
            c_hex += rng_main.normal(0, 0.004, c_hex.shape)
            c_hex = np.clip(c_hex, 0.02, 0.98)
            inits.append(c_hex)
            
    # Random uniform placements
    for _ in range(5):
        c_rand = rng_main.uniform(0.05, 0.95, (N, 2))
        inits.append(c_rand)

    best_radii = None

        pass

    c_res = best_centers.copy()

    for i in range(N):
        mx, my = c_res[i]
        ub = min(mx, 1.0 - mx, my, 1.0 - my)
        r_res[i] = min(r_res[i], ub - 1e-9)
        r_res[i] = max(0.0, r_res[i])

    for _ in range(50):
        changed = False
        for k in range(NUM_PAIRS):
            i, j = PAIRS_I[k], PAIRS_J[k]

                exc = r_res[i] + r_res[j] - d
                r_res[i] -= exc * 0.5
                r_res[j] -= exc * 0.5
                r_res[i] = max(0.0, r_res[i])
                r_res[j] = max(0.0, r_res[j])
                changed = True
        if not changed:
            break
            
    return c_res, r_res, float(np.sum(r_res))
