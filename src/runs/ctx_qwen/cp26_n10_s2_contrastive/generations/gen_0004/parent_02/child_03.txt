

<strategy>
To improve the packing sum of radii, we move away from standard SLSQP which often struggles with the non-smooth contact constraints and high dimensionality. Instead, we exploit the structure of the problem:
1. **LP Oracle for Radii**: For any fixed set of centers, the optimal radii that maximize the sum subject to non-overlap and boundary constraints can be found exactly via Linear Programming.
2. **Dual Variables as Forces**: The optimal value of an LP is a convex function of the right-hand side parameters. The dual variables (shadow prices) of the pairwise distance constraints $\lambda_{ij} \ge 0$ directly give the subgradient of the maximum sum of radii with respect to the distances $d_{ij}$. This translates to repulsive forces between circles: $\vec{F}_i = \sum_{j \ne i} \lambda_{ij} \frac{\vec{C}_i - \vec{C}_j}{d_{ij}}$.
3. **Gradient Ascent on Centers**: By solving the LP, extracting duals, and moving centers along these forces, we perform gradient ascent on the center coordinates. This efficiently pushes circles apart to open up space for larger radii, naturally converging to locally optimal packings where forces balance or boundaries are reached.
4. **Boundary Repulsion & Robustness**: We add soft repulsive forces from the square boundaries to prevent circles from getting stuck at walls where they cannot grow. We run multiple independent trajectories from diverse initializations (hexagonal lattices and random placements) and keep the best result. A strict post-processing step guarantees numerical validity for the validator.

This approach is mathematically grounded, avoids SLSQP's constraint handling overhead, and typically escapes shallow local minima more effectively than pure heuristic jitter.
</strategy>

import numpy as np
from scipy.optimize import linprog

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(I_IDX)

def get_duals(res):
    """Robustly extract inequality duals from linprog result across scipy versions."""
    try:
        return res.marginals.ineqlin.marginals
    except AttributeError:
        try:
            return res.ineqlin.marginals
        except AttributeError:
            return None

def solve_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii.
    Returns radii, repulsive forces (derived from duals), and sum_radii.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), I_IDX] = 1.0
    A_ub[np.arange(NUM_PAIRS), J_IDX] = 1.0
    b_ub = dists[I_IDX, J_IDX]
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        mx = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(1e-9, mx)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = np.maximum(res.x, 0.0)
            forces = None
            duals = get_duals(res)
            if duals is not None:
                forces = np.zeros((n, 2))
                for k in range(NUM_PAIRS):
                    if duals[k] > 1e-9:
                        i, j = I_IDX[k], J_IDX[k]
                        d = dists[i, j]
                        if d > 1e-12:
                            dir_vec = (centers[i] - centers[j]) / d
                            forces[i] += duals[k] * dir_vec
                            forces[j] -= duals[k] * dir_vec
            return radii, forces, -res.fun
        return np.zeros(n), None, 0.0
    except Exception:
        return np.zeros(n), None, 0.0

def generate_initial_configs(num_configs):
    """Generate diverse initial center configurations."""
    configs = []
    rng = np.random.default_rng(42)
    
    # Hexagonal lattices with varying densities and offsets
    for sp in np.linspace(0.17, 0.22, 6):
        for roff in [0.0, 0.04]:
            for coff in [0.0, 0.04]:
                centers = np.zeros((N, 2))
                idx = 0
                y = 0.05 + roff
                row = 0
                while idx < N and y < 0.95:
                    x_start = 0.05 + coff + (row % 2) * sp / 2.0
                    col = 0
                    while x_start + col * sp < 0.95 and idx < N:
                        centers[idx] = [x_start + col * sp, y]
                        idx += 1
                        col += 1
                    y += sp * np.sqrt(3) / 2.0
                    row += 1
                while idx < N:
                    centers[idx] = rng.uniform(0.1, 0.9, 2)
                    idx += 1
                configs.append(centers)
                
    # Random uniform placements
    for _ in range(num_configs):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return configs

def optimize_trajectory(init_centers, seed):
    """Gradient ascent on centers using LP dual forces."""
    rng = np.random.default_rng(seed)
    centers = init_centers.copy()
    best_score = -1.0
    best_centers = None
    best_radii = None
    
    step_size = 0.004
    margin = 0.03
    
    for step in range(2000):
        radii, forces, score = solve_lp(centers)
        
        if score > best_score:
            best_score = score
            best_centers = centers.copy()
            best_radii = radii.copy()
            
        if forces is not None:
            # Add soft boundary repulsion to keep circles away from walls
            repulsion = np.zeros_like(centers)
            repulsion[:, 0] += 15.0 * np.maximum(0.0, margin - centers[:, 0])
            repulsion[:, 0] -= 15.0 * np.maximum(0.0, margin - (1.0 - centers[:, 0]))
            repulsion[:, 1] += 15.0 * np.maximum(0.0, margin - centers[:, 1])
            repulsion[:, 1] -= 15.0 * np.maximum(0.0, margin - (1.0 - centers[:, 1]))
            
            total_force = forces + repulsion
            centers += step_size * total_force
            centers = np.clip(centers, margin + 1e-4, 1.0 - margin - 1e-4)
        else:
            # Fallback jitter if LP duals unavailable
            centers += rng.normal(0, 0.003, centers.shape)
            centers = np.clip(centers, 0.02, 0.98)
            
        # Gentle step decay to settle into basin
        if step % 500 == 0:
            step_size *= 0.95
            
    return best_centers, best_radii, best_score

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    configs = generate_initial_configs(20)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    for i, cfg in enumerate(configs):
        # Slight perturbation to break symmetries
        rng = np.random.default_rng(i * 17 + 3)
        cfg_pert = cfg + rng.normal(0, 0.004, cfg.shape)
        cfg_pert = np.clip(cfg_pert, 0.05, 0.95)
        
        c_opt, r_opt, s_opt = optimize_trajectory(cfg_pert, seed=i*13+7)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            best_radii = r_opt
            
    # Fallback safety net
    if best_centers is None:
        best_centers = configs[0]
        best_radii, _, best_sum = solve_lp(best_centers)
        
    # Strict post-processing to guarantee validator compliance
    c_final = best_centers.copy()
    r_final = best_radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx, my = c_final[i]
        ub = min(mx, 1.0 - mx, my, 1.0 - my)
        r_final[i] = min(r_final[i], ub - 1e-9)
        r_final[i] = max(0.0, r_final[i])
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(50):
        changed = False
        for k in range(NUM_PAIRS):
            i, j = I_IDX[k], J_IDX[k]
            d = np.hypot(c_final[i,0] - c_final[j,0], c_final[i,1] - c_final[j,1])
            if d < r_final[i] + r_final[j] - 1e-11:
                exc = r_final[i] + r_final[j] - d
                r_final[i] -= exc * 0.5
                r_final[j] -= exc * 0.5
                r_final[i] = max(0.0, r_final[i])
                r_final[j] = max(0.0, r_final[j])
                changed = True
        if not changed:
            break
            
    return c_final, r_final, float(np.sum(r_final))
