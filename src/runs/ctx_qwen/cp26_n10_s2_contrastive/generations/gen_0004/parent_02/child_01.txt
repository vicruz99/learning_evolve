1. **Optimize centers only**: For any fixed set of centers, the optimal radii are found exactly and rapidly via Linear Programming (LP). This reduces the 78-dimensional joint problem to a 52-dimensional problem over centers, where the objective is the negative sum of LP-optimal radii.
2. **Global search over centers**: The landscape of `-sum(LP_radii(centers))` is rugged but piecewise smooth. We then apply `basinhopping` to escape local optima.
3. **Joint refinement**: Finally, we polish the best configuration using SLSQP on joint (center, radius) variables to squeeze out boundary contact precision.
4. **Strict post-processing**: We enforce boundary clearance and resolve any numerical overlaps to guarantee validator compliance.

This approach leverages the speed of modern LP solvers (`highs`) to evaluate the objective thousands of times efficiently, allowing thorough exploration of the configuration space.

import numpy as np
from scipy.optimize import minimize, linprog, basinhopping

N = 26
PAIRS_I, PAIRS_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIRS_I)

def get_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Vectorized pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), PAIRS_I] = 1.0
    A_ub[np.arange(NUM_PAIRS), PAIRS_J] = 1.0
    b_ub = dists[PAIRS_I, PAIRS_J]
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, ub)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.01)

def objective_centers(x):
    """Objective for center optimization: minimize negative sum of LP-optimal radii."""
    centers = x.reshape(N, 2)
    radii = get_lp_radii(centers)
    return -np.sum(radii)

def accept_all(*args):
    """Accept any step during basin hopping for aggressive exploration."""
    return True

def generate_initial_configs():
    """Generate diverse structured and random initial configurations."""
    configs = []
    rng = np.random.default_rng(42)
    
    # 1. Hexagonal lattices with varying spacing
    for sp in np.linspace(0.18, 0.22, 5):
        c = np.zeros((N, 2))
        idx = 0
        y = 0.05
        row = 0
        while idx < N and y < 0.95:
            x_start = 0.05 + (row % 2) * sp / 2.0
            col = 0
            while x_start + col * sp < 0.95 and idx < N:
                c[idx] = [x_start + col * sp, y]
                idx += 1
                col += 1
            y += sp * np.sqrt(3) / 2.0
            row += 1
        while idx < N:
            c[idx] = rng.uniform(0.1, 0.9, 2)
            idx += 1
        configs.append(c + rng.normal(0, 0.005, c.shape))
        
    # 2. Structured row patterns: 6,5,6,5,4 (optimal for 26 circles)
    for sp in [0.175, 0.195, 0.215]:
        c = np.zeros((N, 2))
        idx = 0
        y = 0.06
        dy = sp * np.sqrt(3) / 2.0 * 0.95
        rows = [6, 5, 6, 5, 4]
        for r_idx, cnt in enumerate(rows):
            shift = 0.0 if r_idx % 2 == 0 else sp / 2.0
            x = 0.06 + shift
            for _ in range(cnt):
                if idx < N:
                    c[idx] = [x, y]
                    idx += 1
                x += sp
            y += dy
        while idx < N:
            c[idx] = rng.uniform(0.2, 0.8, 2)
            idx += 1
        configs.append(c + rng.normal(0, 0.008, c.shape))
        
    # 3. Random uniform placements
    for _ in range(12):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return [np.clip(cfg, 0.02, 0.98) for cfg in configs]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    bounds_coords = [(0.02, 0.98)] * (2 * N)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = generate_initial_configs()
    
    # Phase 1: Local optimization from diverse starts
    for c0 in configs:
        x0 = c0.flatten()
        try:
            res = minimize(objective_centers, x0, method='Powell', 
                           bounds=bounds_coords, 
                           options={'maxiter': 400, 'ftol': 1e-12, 'xtol': 1e-10})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
                best_radii = get_lp_radii(best_centers)
        except Exception:
            pass
            
    # Phase 2: Basin hopping to escape local minima
    if best_centers is not None:
        x0 = best_centers.flatten()
        minimizer_kwargs = {"method": "Powell", "bounds": bounds_coords, 
                            "options": {"maxiter": 250, "ftol": 1e-12}}
        try:
            bp = basinhopping(objective_centers, x0, minimizer_kwargs=minimizer_kwargs,
                              niter=40, stepsize=0.025, accept_test=accept_all)
            if -bp.fun > best_sum:
                best_sum = -bp.fun
                best_centers = bp.x.reshape(N, 2)
                best_radii = get_lp_radii(best_centers)
        except Exception:
            pass

    # Phase 3: Joint SLSQP refinement for boundary/contact precision
    if best_centers is not None:
        c = best_centers.copy()
        r = best_radii.copy()
        x0 = np.zeros(3 * N)
        x0[0::3] = c[:, 0]
        x0[1::3] = c[:, 1]
        x0[2::3] = np.maximum(r * 0.995, 1e-5)
        
        def constraints_slsqp(x):
            cx = x[0::3]
            cy = x[1::3]
            ri = x[2::3]
            c_val = np.empty(4 * N + NUM_PAIRS)
            c_val[:N] = cx - ri
            c_val[N:2*N] = 1.0 - cx - ri
            c_val[2*N:3*N] = cy - ri
            c_val[3*N:4*N] = 1.0 - cy - ri
            dx = cx[PAIRS_I] - cx[PAIRS_J]
            dy = cy[PAIRS_I] - cy[PAIRS_J]
            c_val[4*N:] = np.hypot(dx, dy) - (ri[PAIRS_I] + ri[PAIRS_J])
            return c_val
            
        def obj_slsqp(x):
            return -np.sum(x[2::3])
            
        try:
            res_sl = minimize(obj_slsqp, x0, method='SLSQP', 
                              bounds=[(0.0,1.0)]*(2*N) + [(1e-6, 0.5)]*N,
                              constraints={'type': 'ineq', 'fun': constraints_slsqp},
                              options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            if res_sl.success:
                c_sl = np.column_stack((res_sl.x[0::3], res_sl.x[1::3]))
                r_sl = get_lp_radii(c_sl)
                s_sl = np.sum(r_sl)
                if s_sl > best_sum:
                    best_sum = s_sl
                    best_centers = c_sl
                    best_radii = r_sl
        except Exception:
            pass

    # Fallback safety net
    if best_centers is None:
        best_centers = configs[0]
        best_radii = get_lp_radii(best_centers)
        best_sum = np.sum(best_radii)
        
    # Phase 4: Strict post-processing to guarantee validator compliance
    c_final = best_centers.copy()
    r_final = best_radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(N):
        mx, my = c_final[i]
        ub = min(mx, 1.0 - mx, my, 1.0 - my)
        r_final[i] = min(r_final[i], ub - 1e-9)
        r_final[i] = max(0.0, r_final[i])
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(50):
        changed = False
        for k in range(NUM_PAIRS):
            i, j = PAIRS_I[k], PAIRS_J[k]
            d = np.hypot(c_final[i,0] - c_final[j,0], c_final[i,1] - c_final[j,1])
            if d < r_final[i] + r_final[j] - 1e-11:
                exc = r_final[i] + r_final[j] - d
                r_final[i] -= exc * 0.5
                r_final[j] -= exc * 0.5
                r_final[i] = max(0.0, r_final[i])
                r_final[j] = max(0.0, r_final[j])
                changed = True
        if not changed:
            break
            
    return c_final, r_final, float(np.sum(r_final))
