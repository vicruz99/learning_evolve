from scipy.optimize import minimize

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0"""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    n = N_CIRCLES
    n_pairs = len(I_IDX)
    
    c = np.empty(n_pairs + 4 * n)
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c[:n] = cx - cr
    c[n:2*n] = 1.0 - cx - cr
    c[2*n:3*n] = cy - cr
    c[3*n:4*n] = 1.0 - cy - cr
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    c[4*n:] = np.hypot(dx, dy) - (cr[I_IDX] + cr[J_IDX])
    
    return c
