import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(x):
    # x is arranged as [x1, y1, r1, x2, y2, r2, ...]
    # Radii are at indices 2, 5, 8, ...
    return -np.sum(x[2::3])

def constraints(x):
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
    cons = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    dists = np.hypot(dx, dy)
    r_sums = r[I_IDX] + r[J_IDX]
    
    return np.concatenate([cons, dists - r_sums])

def make_init(seed, scale):
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    
    # Hexagonal pattern generation for 26 circles
    row_counts = [6, 5, 6, 5, 4]
    y = scale * 0.5
    dy = scale * np.sqrt(3)
    dx = 2 * scale
    
    idx = 0
    for row_idx, count in enumerate(row_counts):
        shift = (dx / 2) if row_idx % 2 == 1 else 0
        x_start = scale + shift
        for col in range(count):
            if idx < N:
                x = x_start + col * dx
                centers[idx] = [x + rng.uniform(-0.005, 0.005), 
                                y + rng.uniform(-0.005, 0.005)]
                idx += 1
        y += dy
        
    while idx < N:
        centers[idx] = [rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)]
        idx += 1
        
    r = np.full(N, scale)
    x0 = np.zeros(3*N)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = r
    return x0

def run_packing():

    best_val = -np.inf

    bounds = [(0, 1), (0, 1), (0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}

    for scale in [0.08, 0.09, 0.10, 0.11, 0.12]:
        for seed in range(10):
            x0 = make_init(seed, scale)
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,

                    best_val = val
                    best_x = res.x.copy()

                pass

                best_x = res2.x.copy()
        except Exception:

    cx = best_x[0::3]
    cy = best_x[1::3]
    r = best_x[2::3]
