import numpy as np
from scipy.optimize import minimize

def objective_func(x, n):
    """Minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def constraint_func(x, n, triu_idx):
    """Vectorized inequality constraints: g(x) >= 0"""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    
    num_cons = 4*n + len(triu_idx[0])
    c = np.empty(num_cons)
    
    # Boundary constraints
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    
    # Overlap constraints (vectorized)
    r_arr = r[:, np.newaxis] + r[np.newaxis, :]
    cx_arr = cx[:, np.newaxis]
    cy_arr = cy[:, np.newaxis]
    dists = np.sqrt((cx_arr - cx_arr.T)**2 + (cy_arr - cy_arr.T)**2)
    
    c[4*n:] = dists[triu_idx] - r_arr[triu_idx]
    return c

def make_init_config(n, seed):
    """Generate a feasible hexagonal initialization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    rows = [6, 5, 6, 5, 4]  # Sums to 26
    y_step = 0.18
    x_step = 0.17
    idx = 0
    
    for r_idx, cnt in enumerate(rows):
        y = 0.12 + r_idx * y_step
        x_start = 0.08 + (0.085 if r_idx % 2 == 1 else 0.0)
        for c in range(cnt):
            x = x_start + c * x_step
            if x <= 0.92:
                centers[idx] = [x + rng.uniform(-0.015, 0.015), 
                                y + rng.uniform(-0.015, 0.015)]
                idx += 1
                
    # Fill any remaining slots (should be exactly 0)
    while idx < n:
        centers[idx] = [0.5 + rng.uniform(-0.05, 0.05), 
                        0.5 + rng.uniform(-0.05, 0.05)]
        idx += 1
        
    centers = np.clip(centers, 0.06, 0.94)
    radii = np.full(n, 0.02)  # Small enough to guarantee initial feasibility
    return np.concatenate([centers.ravel(), radii])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Optimizes circle packing in a unit square to maximize sum of radii."""
    n = 26
    triu_idx = np.triu_indices(n, k=1)
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n, triu_idx)}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Multi-start optimization
    for trial in range(60):
        x0 = make_init_config(n, trial * 13 + 7)
        try:
            res = minimize(objective_func, x0, args=(n,), method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                s = np.sum(res.x[2*n:])
                if s > best_sum:
                    best_sum = s
                    best_centers = res.x[:2*n].reshape(n, 2).copy()
                    best_radii = res.x[2*n:].copy()
        except Exception:
            pass
            
    # Phase 2: High-precision local refinement
    if best_centers is not None:
        for _ in range(20):
            x0 = np.concatenate([best_centers.ravel(), best_radii])
            noise = np.random.randn(len(x0)) * 0.0005
            x0 = np.clip(x0 + noise, 0.0, 0.5)
            res = minimize(objective_func, x0, args=(n,), method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if res.success:
                s = np.sum(res.x[2*n:])
                if s > best_sum:
                    best_sum = s
                    best_centers = res.x[:2*n].reshape(n, 2).copy()
                    best_radii = res.x[2*n:].copy()
                    
    # Phase 3: Deterministic projection to guarantee strict validity
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(n):
        x, y = centers[i]
        max_r_boundary = min(x, 1.0 - x, y, 1.0 - y)
        radii[i] = min(radii[i], max_r_boundary - 1e-10)
        
    # Iteratively resolve any residual overlaps
    for _ in range(50):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    overlap = radii[i] + radii[j] - d
                    radii[i] -= overlap * 0.51
                    radii[j] -= overlap * 0.49
                    radii[i] = max(radii[i], 0.0)
                    radii[j] = max(radii[j], 0.0)
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
