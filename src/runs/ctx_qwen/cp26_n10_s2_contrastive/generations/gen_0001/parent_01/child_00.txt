import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to find optimal radii maximizing sum."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        mr = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(mr, 0.0)))
        
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                                 (centers[i, 1] - centers[j, 1])**2)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except:
        pass
    return np.full(n, 1e-5), 1e-4

def obj_func(vars_flat):
    r = vars_flat[2 * N:]
    return -np.sum(r)

def constraint_func(vars_flat):
    x = vars_flat[:N]
    y = vars_flat[N:2 * N]
    r = vars_flat[2 * N:]
    
    cons = []
    # Boundary constraints
    cons.extend(x - r)
    cons.extend(1.0 - x - r)
    cons.extend(y - r)
    cons.extend(1.0 - y - r)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(N):
        for j in range(i + 1, N):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist_sq = dx * dx + dy * dy
            r_sum = r[i] + r[j]
            cons.append(dist_sq - r_sum * r_sum)
            
    return np.array(cons)

def fix_violations(centers, radii):
    """Ensure strict feasibility by shrinking radii if necessary due to numerical drift."""
    n = len(radii)
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        mr = min(x, 1.0 - x, y, 1.0 - y)
        if radii[i] > mr:
            radii[i] = max(0.0, mr - 1e-10)
            
    # Overlap violations
    for _ in range(200):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                            (centers[i, 1] - centers[j, 1])**2)
                if d < radii[i] + radii[j] - 1e-10:
                    overlap = radii[i] + radii[j] - d
                    radii[i] -= overlap / 2 + 1e-9
                    radii[j] -= overlap / 2 + 1e-9
                    radii[i] = max(radii[i], 1e-9)
                    radii[j] = max(radii[j], 1e-9)
                    changed = True
        if not changed:
            break
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
