1. **Block Coordinate Descent with LP**: For any fixed set of centers, finding the optimal radii is a convex Linear Programming (LP) problem. We solve `max sum(r_i)` subject to `r_i + r_j <= dist(i,j)` and boundary constraints. This LP solver is exact and extremely fast for N=26.
2. **Center Optimization**: We use the LP result as a smooth-ish objective function `f(C) = LP(C)` and optimize the centers using the `Powell` method. This effectively navigates the configuration space by always keeping radii optimal for the current positions.
3. **Diverse Initialization**: We generate multiple starting configurations (random spread, hexagonal lattice, grid) and apply a quick force-directed relaxation to evenly distribute points and avoid initial overlaps.
4. **SLSQP Refinement**: After finding the best centers and radii via the LP approach, we run a final `SLSQP` optimization on the combined variables (centers + radii) to tighten constraints and polish the solution to high precision.
5. **Safety & Validation**: Final clamping and overlap resolution ensure strict compliance with the validator's tolerance (`1e-12`).

This approach decouples the combinatorial geometry from the radius expansion, consistently reaching higher sums than direct nonlinear optimization.

import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    """Solves the Linear Program to maximize sum of radii for fixed centers."""
    n = len(centers)
    c = -np.ones(n)  # Minimize -sum(r) <=> Maximize sum(r)
    
    # Pairwise constraints: r_i + r_j <= distance(i, j)
    A_ub = np.zeros((n * (n - 1) // 2, n))
    b_ub = np.zeros(n * (n - 1) // 2)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j])
            idx += 1
            
    # Boundary constraints handled via bounds for efficiency: 0 <= r_i <= min(dist_to_walls)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        # Ensure feasibility even if center is slightly out of bounds during optimization
        bounds.append((0.0, max(1e-12, max_r)))
        
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.status == 0:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback solver
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def objective_centers(x_flat):
    """Objective for optimizing centers: returns negative sum of optimal radii."""
    centers = x_flat.reshape(26, 2)
    # Strong penalty if centers go outside the unit square
    out_penalty = 100.0 * (np.sum(np.maximum(0, -centers)) + np.sum(np.maximum(0, centers - 1.0)))
    
    _, sum_r = solve_radii_lp(centers)
    return -sum_r + out_penalty

def force_directed_init(n, seed, steps=250):
    """Initializes and spreads points using simple repulsion forces."""
    np.random.seed(seed)
    pts = np.random.rand(n, 2) * 0.6 + 0.2
    
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(n):
            for j in range(i + 1, n):
                diff = pts[i] - pts[j]
                d = np.linalg.norm(diff)
                if d < 0.12 and d > 1e-6:
                    repel = diff / d * (0.12 - d) * 0.08
                    forces[i] += repel
                    forces[j] -= repel
        pts += forces
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Strategy 1: LP-based center optimization from multiple starts
    for seed in range(25):
        # Generate initial configuration
        if seed < 5:
            # Hexagonal-ish grid
            pts = np.zeros((n, 2))
            idx = 0
            row_counts = [6, 5, 6, 5, 4]
            y = 0.15
            for rc in row_counts:
                x = 0.15
                for _ in range(rc):
                    if idx < n:
                        pts[idx] = [x, y]
                        idx += 1
                    x += 0.15
                y += 0.17
        else:
            # Random spread with force-directed relaxation
            pts = force_directed_init(n, seed)
            
        # Optimize centers using Powell method on the LP objective
        res = minimize(objective_centers, pts.flatten(), method='Powell',
                       options={'xtol': 1e-8, 'ftol': 1e-10, 'maxiter': 400})
        
        opt_centers = res.x.reshape(n, 2)
        opt_centers = np.clip(opt_centers, 0.001, 0.999)
        
        # Solve for optimal radii given these centers
        radii, s = solve_radii_lp(opt_centers)
        if s > best_sum:
            best_sum = s
            best_centers = opt_centers.copy()
            best_radii = radii.copy()
            
    # Strategy 2: SLSQP refinement on combined variables
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii])
        
        def obj_combined(x):
            return -np.sum(x[2 * n:])
            
        def cons_combined(x):
            cx = x[:n]
            cy = x[n:2 * n]
            r = x[2 * n:]
            
            # Boundary constraints: x-r>=0, 1-x-r>=0, y-r>=0, 1-y-r>=0
            cons = np.concatenate([
                cx - r, 1.0 - cx - r,
                cy - r, 1.0 - cy - r
            ])
            
            # Non-overlap constraints: dist >= r_i + r_j
            for i in range(n):
                for j in range(i + 1, n):
                    dx = cx[i] - cx[j]
                    dy = cy[i] - cy[j]
                    dist = np.hypot(dx, dy)
                    cons = np.append(cons, dist - (r[i] + r[j]))
            return cons
            
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        res2 = minimize(obj_combined, x0, method='SLSQP', bounds=bounds,
                        constraints={'type': 'ineq', 'fun': cons_combined},
                        options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
        
        final_centers = res2.x[:2 * n].reshape(n, 2)
        final_radii = res2.x[2 * n:]
        final_sum = np.sum(final_radii)
        
        if final_sum > best_sum:
            best_sum = final_sum
            best_centers = final_centers
            best_radii = final_radii
            
    # Final safety adjustments to strictly satisfy validation tolerances
    best_radii = np.maximum(best_radii, 1e-12)
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    # Enforce boundary constraints strictly
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y) - 1e-11
        if r > max_r:
            best_radii[i] = max(1e-12, max_r)
            
    # Enforce non-overlap strictly
    for _ in range(10):
        overlap_found = False
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                r_sum = best_radii[i] + best_radii[j]
                if d < r_sum - 1e-12:
                    excess = r_sum - d
                    reduction = excess / 2.0 + 1e-12
                    best_radii[i] -= reduction
                    best_radii[j] -= reduction
                    best_radii[i] = max(best_radii[i], 1e-12)
                    best_radii[j] = max(best_radii[j], 1e-12)
                    overlap_found = True
        if not overlap_found:
            break
            
    return best_centers, best_radii, float(np.sum(best_radii))
