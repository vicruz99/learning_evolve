import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_I = None
PAIR_J = None

def setup_globals():
    global PAIR_I, PAIR_J
    PAIR_I, PAIR_J = np.triu_indices(N, k=1)

def objective(x):
    # Minimize negative sum of radii
    return -np.sum(x[2::3])

def constraint_func(x):
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Pre-allocate constraint array: 4 boundary per circle + 1 per pair
    cons = np.empty(N * 4 + len(PAIR_I))
    
    # Boundary constraints (formulated as >= 0)
    cons[0:N] = cx - r
    cons[N:2*N] = 1.0 - cx - r
    cons[2*N:3*N] = cy - r
    cons[3*N:4*N] = 1.0 - cy - r
    
    # Non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = cx[PAIR_I] - cx[PAIR_J]
    dy = cy[PAIR_I] - cy[PAIR_J]
    dist_sq = dx*dx + dy*dy
    r_sum = r[PAIR_I] + r[PAIR_J]
    cons[4*N:] = dist_sq - r_sum**2
    
    return cons

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    setup_globals()
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': constraint_func}
    
    # State holder to avoid nonlocal/closure issues
    state = {'best_sum': -1.0, 'best_x': None}
    
    def try_opt(x0):
        try:
            res = minimize(
                objective, x0, method='SLSQP', bounds=bounds,
                constraints=cons_dict, 
                options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False}
            )
            if res.success:
                curr = -res.fun
                if curr > state['best_sum']:
                    state['best_sum'] = curr
                    state['best_x'] = res.x.copy()
        except Exception:
            pass

    # Generate initial configurations
    configs = []
    
    # 1. Hexagonal lattice patterns tailored for N=26
    for seed in range(15):
        rng = np.random.RandomState(seed)
        cx, cy = np.zeros(N), np.zeros(N)
        row_counts_options = [[6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [4,6,5,6,5]]
        rows = row_counts_options[seed % len(row_counts_options)]
        
        y = 0.15
        idx = 0
        for ri, cnt in enumerate(rows):
            shift = 0.05 if ri % 2 == 1 else 0.0
            xs = np.linspace(0.1 + shift, 0.9 - shift, cnt)
            for k in range(cnt):
                cx[idx], cy[idx] = xs[k], y
                idx += 1
            y += 0.185
            
        cx += rng.uniform(-0.02, 0.02, N)
        cy += rng.uniform(-0.02, 0.02, N)
        configs.append(np.concatenate([cx, cy, np.full(N, 0.04)]))
        
    # 2. Random safe-zone initializations
    for seed in range(20):
        rng = np.random.RandomState(seed + 1000)
        cx = rng.uniform(0.15, 0.85, N)
        cy = rng.uniform(0.15, 0.85, N)
        configs.append(np.concatenate([cx, cy, np.full(N, 0.04)]))
        
    # Run optimization on all configs
    for x0 in configs:
        try_opt(x0)
        
    # 3. Local refinement around the best found solution
    if state['best_x'] is not None:
        for _ in range(30):
            x_pert = state['best_x'] + np.random.randn(3 * N) * 0.003
            try_opt(x_pert)
            
    # Extract best results
    best_x = state['best_x']
    if best_x is not None:
        centers = np.column_stack((best_x[0::3], best_x[1::3]))
        radii = best_x[2::3].copy()
    else:
        centers = np.tile([0.5, 0.5], (N, 1))
        radii = np.full(N, 0.01)
        
    # Post-processing: Guarantee strict validity for the checker
    # 1. Project to boundaries
    for i in range(N):
        x, y = centers[i]
        max_r_boundary = min(x, 1.0 - x, y, 1.0 - y)
        if radii[i] > max_r_boundary:
            radii[i] = max_r_boundary
            
    # 2. Iteratively shrink overlapping circles
    for _ in range(200):
        max_viol = 0.0
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                sum_r = radii[i] + radii[j]

                    excess = sum_r - dist
                    radii[i] -= excess * 0.5
                    radii[j] -= excess * 0.5
                    max_viol = max(max_viol, excess)
        if max_viol < 1e-12:
            break
            
    radii = np.maximum(radii, 1e-9)
    
    return centers, radii, float(np.sum(radii))
