import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26

# Precompute static constraint matrix for pairwise distance constraints: r_i + r_j <= dist_ij
A_STATIC = np.zeros((N_CIRCLES*(N_CIRCLES-1)//2, N_CIRCLES))
_idx = 0
for _i in range(N_CIRCLES):
    for _j in range(_i + 1, N_CIRCLES):
        A_STATIC[_idx, _i] = 1.0
        A_STATIC[_idx, _j] = 1.0
        _idx += 1

def solve_radii_lp(centers):
    """Solves for optimal radii given fixed centers using Linear Programming."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(A_STATIC.shape[0])
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, max_r)))
        
    res = linprog(-np.ones(n), A_ub=A_STATIC, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def objective(centers_flat):
    """Objective function for optimizing centers: minimizes negative sum of radii."""
    n = len(centers_flat) // 2
    centers = np.clip(centers_flat.reshape((n, 2)), 1e-6, 1.0 - 1e-6)
    _, sum_r = solve_radii_lp(centers)
    return -sum_r

def full_obj(x, n):
    """Objective for final SLSQP polishing."""
    return -np.sum(x[2*n:])

def full_cons(x, n):
    """Constraints for final SLSQP polishing."""
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    cons = []
    cons.extend(cx - r)
    cons.extend(1.0 - cx - r)
    cons.extend(cy - r)
    cons.extend(1.0 - cy - r)
    for i in range(n):
        for j in range(i+1, n):
            d2 = (cx[i]-cx[j])**2 + (cy[i]-cy[j])**2
            cons.append(d2 - (r[i]+r[j])**2)
    return np.array(cons)

def run_packing():

    best_sum = 0.0
    best_centers = None
    best_radii = None

            r_guess = 0.075 * scale

                x = x_start
                while x + r_guess <= 1.0 and len(centers) < n:
                    centers.append([x, y])
                    x += 2 * r_guess
                y += r_guess * np.sqrt(3)
                row += 1
            centers = np.array(centers[:n])
            centers += np.random.randn(*centers.shape) * 0.005
            inits.append(centers.flatten())
            
    # Add random uniform initializations
    for _ in range(5):
        c = np.random.uniform(0.15, 0.85, (n, 2))
        inits.append(c.flatten())
        
    # Stage 1: Optimize centers using Nelder-Mead with LP-based radius evaluation
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='Nelder-Mead', 
                           options={'maxiter': 1500, 'xatol': 1e-6, 'fatol': 1e-8})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape((n, 2))
        except Exception:
            continue
            
    # Fallback if optimization fails
    if best_centers is None:
        best_centers = np.array([[0.1 + j*0.2, 0.1 + i*0.2] for i in range(5) for j in range(5)] + [[0.5, 0.5]])

            break
            
    return centers, radii, float(np.sum(radii))
