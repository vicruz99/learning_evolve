from scipy.optimize import minimize, linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Helper: SLSQP Objective & Constraints
    # -----------------------------
    def objective(params):
        # params structure: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
        radii = params[2::3]
        return -np.sum(radii)  # Minimize negative sum => Maximize sum

        cy = params[1::3]
        r = params[2::3]

        c_list.append(1.0 - cx - r)
        c_list.append(cy - r)
        c_list.append(1.0 - cy - r)

        dx = cx[:, np.newaxis] - cx[np.newaxis, :]
        dy = cy[:, np.newaxis] - cy[np.newaxis, :]
        dists = np.sqrt(dx**2 + dy**2)

        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        d_upper = dists[mask]
        r_sum = (r[:, np.newaxis] + r[np.newaxis, :])[mask]
        c_list.append(d_upper - r_sum)
        
        return np.concatenate(c_list)

    # -----------------------------
    def maximize_radii_lp(centers):

        cy = centers[:, 1]

            np.minimum(cx, 1.0 - cx),
            np.minimum(cy, 1.0 - cy)
        )

    return centers, radii, final_sum
