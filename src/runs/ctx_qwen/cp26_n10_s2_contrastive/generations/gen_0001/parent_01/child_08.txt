import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26

def solve_radii_lp(centers):
    """Solves LP to find radii maximizing sum(r_i) for fixed centers."""
    n = len(centers)
    c = -np.ones(n)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Build A_ub and b_ub for r_i + r_j <= dists[i,j]
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Bounds for radii based on center positions
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, max_r)))
        
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-6), 26e-6

def objective_centers(x_flat):
    """Objective for center optimization: negative of optimal sum of radii."""
    n = N_CIRCLES
    centers = np.clip(x_flat.reshape(n, 2), 1e-5, 1.0 - 1e-5)
    _, sum_r = solve_radii_lp(centers)
    return -sum_r

def slsqp_objective(vars_flat):
    """SLSQP objective: maximize sum of radii."""
    n = N_CIRCLES
    r = vars_flat[2*n:]
    return -np.sum(r)

def slsqp_constraints(vars_flat):
    """SLSQP constraints: boundaries and non-overlaps (squared form)."""
    n = N_CIRCLES
    cx = vars_flat[:n]
    cy = vars_flat[n:2*n]
    r = vars_flat[2*n:]
    cons = []
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    cons.extend(cx - r)
    cons.extend(1.0 - cx - r)
    cons.extend(cy - r)
    cons.extend(1.0 - cy - r)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            d2 = dx*dx + dy*dy
            rs = r[i] + r[j]
            cons.append(d2 - rs*rs)
            
    return np.array(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    initial_configs = []
    
    # 1. Hexagonal lattice variations
    for seed in range(5):
        rng = np.random.RandomState(seed)
        centers = np.zeros((n, 2))
        r_base = 0.09 + rng.uniform(0, 0.02)
        dy = r_base * np.sqrt(3)
        dx = 2 * r_base
        idx = 0
        y = r_base + rng.uniform(0, 0.05)
        row_counts = [6, 5, 6, 5, 4]
        for r_idx, ncols in enumerate(row_counts):
            x_start = r_base + (r_idx % 2) * dx * 0.5 + rng.uniform(-0.02, 0.02)
            for col in range(ncols):
                if idx < n:
                    centers[idx] = [x_start + col * dx, y]
                    idx += 1
            y += dy + rng.uniform(-0.01, 0.01)
        while idx < n:
            centers[idx] = [rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)]
            idx += 1
        initial_configs.append(centers.flatten())
        
    # 2. Grid variations
    for seed in range(3):
        rng = np.random.RandomState(seed + 10)
        centers = np.zeros((n, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                if idx < n:
                    centers[idx] = [0.1 + j*0.2 + rng.uniform(-0.02, 0.02), 
                                    0.1 + i*0.2 + rng.uniform(-0.02, 0.02)]
                    idx += 1
        while idx < n:
            centers[idx] = [rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)]
            idx += 1
        initial_configs.append(centers.flatten())
        
    # 3. Random spread
    for seed in range(5):
        rng = np.random.RandomState(seed + 20)
        centers = rng.uniform(0.15, 0.85, (n, 2))
        initial_configs.append(centers.flatten())

    # Optimize centers using Powell's method on the LP envelope
    for config in initial_configs:
        try:
            res = minimize(objective_centers, config, method='Powell',
                           options={'maxiter': 2500, 'xtol': 1e-6, 'ftol': 1e-8})
            c_opt = np.clip(res.x.reshape(n, 2), 1e-5, 1.0 - 1e-5)
            r_opt, s_opt = solve_radii_lp(c_opt)
            
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            continue

    # Joint refinement using SLSQP
    if best_centers is not None:
        try:
            x0 = np.concatenate([best_centers.ravel(), best_radii])
            bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
            cons_dict = {'type': 'ineq', 'fun': slsqp_constraints}
            
            res2 = minimize(slsqp_objective, x0, method='SLSQP', bounds=bounds,

            if res2.success or -res2.fun > best_sum - 1e-4:
                c_ref = res2.x[:2*n].reshape(n, 2)
                r_ref = res2.x[2*n:]
                
                # Project radii to valid boundary limits
                r_ref = np.maximum(r_ref, 0.0)
                for i in range(n):
                    mr = min(c_ref[i,0], 1.0-c_ref[i,0], c_ref[i,1], 1.0-c_ref[i,1])
                    if r_ref[i] > mr:
                        r_ref[i] = mr
                        
                # Iterative overlap fixing to guarantee feasibility
                for _ in range(100):
                    max_ov = 0.0
                    for i in range(n):
                        for j in range(i+1, n):
                            d = np.sqrt((c_ref[i,0]-c_ref[j,0])**2 + (c_ref[i,1]-c_ref[j,1])**2)
                            if d < r_ref[i] + r_ref[j] - 1e-10:
                                ov = r_ref[i] + r_ref[j] - d + 1e-9
                                r_ref[i] -= ov/2.0
                                r_ref[j] -= ov/2.0
                                max_ov = max(max_ov, ov)
                    if max_ov < 1e-12:
                        break
                        
                r_ref = np.maximum(r_ref, 0.0)
                s_ref = np.sum(r_ref)
                
                if s_ref > best_sum:
                    best_sum = s_ref
                    best_centers = c_ref
                    best_radii = r_ref
        except Exception:
            pass

    # Fallback safety net
    if best_centers is None:
        best_centers = np.random.uniform(0.2, 0.8, (n, 2))
        best_radii = np.full(n, 0.02)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
