import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def objective(v):
    """Objective: maximize sum of radii (minimize negative sum)"""
    return -np.sum(v[2*N_CIRCLES:])

def constraints(v):
    """Vectorized inequality constraints: boundary + non-overlap"""
    x = v[:N_CIRCLES]
    y = v[N_CIRCLES:2*N_CIRCLES]
    r = v[2*N_CIRCLES:]
    
    c = np.empty(4*N_CIRCLES + len(I_IDX))
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c[:N_CIRCLES] = x - r
    c[N_CIRCLES:2*N_CIRCLES] = 1.0 - x - r
    c[2*N_CIRCLES:3*N_CIRCLES] = y - r
    c[3*N_CIRCLES:4*N_CIRCLES] = 1.0 - y - r
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    c[4*N_CIRCLES:] = np.hypot(dx, dy) - r[I_IDX] - r[J_IDX]
    
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * N_CIRCLES + [(0, 1)] * N_CIRCLES + [(0, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraints}

    best_sum = -1.0
    best_v = None

    inits = []
    # Hexagonal lattice initialization (6-5-6-5-4 pattern sums to 26)
    rows = [(6, 0), (5, 1), (6, 0), (5, 1), (4, 0)]
    pts = []
    y = 0.1
    for cnt, sh in rows:
        xs = [0.1 + (sh * 0.1) + k * 0.19 for k in range(cnt)]
        for x in xs:
            pts.append((x, y))
        y += 0.165
    pts_arr = np.array(pts[:N_CIRCLES])
    inits.append(np.concatenate([pts_arr[:, 0], pts_arr[:, 1], np.full(N_CIRCLES, 0.06)]))

    # Generate perturbations of the hexagonal lattice
    base = inits[0]
    for i in range(20):
        np.random.seed(i * 7 + 1)
        inits.append(np.clip(base + np.random.randn(3 * N_CIRCLES) * 0.01, 0.02, 0.98))

    # Random initializations
    for i in range(15):
        np.random.seed(i * 13 + 3)
        x = np.random.uniform(0.2, 0.8, N_CIRCLES)
        y = np.random.uniform(0.2, 0.8, N_CIRCLES)
        inits.append(np.concatenate([x, y, np.full(N_CIRCLES, 0.05)]))

    # Main optimization loop
    for init in inits:
        try:
            res = minimize(objective, init, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-12})
            if res.success and -res.fun > best_sum:
                best_sum = -res.fun
                best_v = res.x.copy()
        except Exception:
            continue

    # Refinement loop to escape local optima
    if best_v is not None:
        for _ in range(5):
            np.random.seed(42)
            pert = best_v + np.random.randn(3 * N_CIRCLES) * 0.005
            pert = np.clip(pert, 0.01, 0.99)
            try:
                res = minimize(objective, pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
                if res.success and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_v = res.x.copy()
            except Exception:
                pass

    # Fallback if optimization fails completely
    if best_v is None:
        x_coords = np.linspace(0.2, 0.8, N_CIRCLES)
        y_coords = np.linspace(0.2, 0.8, N_CIRCLES)
        best_v = np.concatenate([x_coords, y_coords, np.full(N_CIRCLES, 0.05)])

    centers = best_v[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = best_v[2 * N_CIRCLES:].copy()

    # Post-processing: strictly enforce boundaries
    for i in range(N_CIRCLES):
        x, y = centers[i]
        r = radii[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if r > max_r + 1e-12:
            radii[i] = max(max_r - 1e-9, 0.0)
    radii = np.maximum(radii, 0.0)

    # Post-processing: resolve numerical overlaps iteratively
    for _ in range(100):
        max_violation = 0.0
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if dist < req - 1e-12:
                    max_violation = max(max_violation, req - dist)
        if max_violation < 1e-14:
            break
        radii *= 0.9999
    radii = np.maximum(radii, 0.0)

    return centers, radii, float(np.sum(radii))
