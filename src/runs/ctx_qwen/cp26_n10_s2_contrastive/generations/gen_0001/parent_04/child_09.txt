import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

# Precompute constant structure for pairwise constraints to speed up LP evaluations
N_CIRCLES = 26
i_pairs, j_pairs = np.triu_indices(N_CIRCLES, k=1)
NUM_PAIRS = N_CIRCLES * (N_CIRCLES - 1) // 2
A_ub_pairs = np.zeros((NUM_PAIRS, N_CIRCLES))
A_ub_pairs[np.arange(NUM_PAIRS), i_pairs] = 1.0
A_ub_pairs[np.arange(NUM_PAIRS), j_pairs] = 1.0

def solve_radii(centers):
    """
    Given fixed centers, solve LP to find radii that maximize sum(radii)
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    # Ensure centers are within valid bounds for LP feasibility
    centers = np.clip(centers, 0.0, 1.0)
    
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # Ignore self-distances
    
    b_ub = dists[i_pairs, j_pairs]
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    x, y = centers[:, 0], centers[:, 1]
    max_r = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    bounds = [(0.0, r_max) for r_max in max_r]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)
    
    # Solve LP
    res = linprog(c, A_ub=A_ub_pairs, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    # Fallback for rare numerical failures
    return np.zeros(n), 0.0

def objective_func(centers_flat):
    """
    Objective function for optimizing centers.
    Returns negative sum of optimal radii (to be minimized).
    """
    n = len(centers_flat) // 2
    centers = centers_flat.reshape((n, 2))
    radii, s = solve_radii(centers)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = N_CIRCLES
    bounds = [(0.0, 1.0)] * (2 * n)
    
    best_val = -np.inf
    best_centers = None
    
    # Strategy 1: Global search using Differential Evolution
    try:
        res_de = differential_evolution(
            objective_func,
            bounds,
            seed=42,
            maxiter=250,
            popsize=15,
            tol=1e-7,
            atol=1e-7,
            mutation=(0.5, 1.5),
            recombination=0.9,
            polish=True,  # Runs local optimizer at the end
            disp=False
        )
        val = -res_de.fun
        if val > best_val:
            best_val = val
            best_centers = res_de.x.reshape((n, 2))
    except Exception as e:
        print(f"DE failed: {e}")

    # Strategy 2: Local refinement from best found so far
    if best_centers is not None:
        res_local = minimize(
            objective_func,
            best_centers.flatten(),
            method='Nelder-Mead',
            options={'maxiter': 5000, 'xatol': 1e-9, 'fatol': 1e-10}
        )
        if -res_local.fun > best_val:
            best_val = -res_local.fun
            best_centers = res_local.x.reshape((n, 2))
            
    # Strategy 3: Multiple random restarts with local search to ensure robustness
    for seed in range(12):
        np.random.seed(seed * 7 + 13)
        x0 = np.random.uniform(0.2, 0.8, (n, 2)).flatten()
        res = minimize(
            objective_func,
            x0,
            method='Nelder-Mead',
            options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-10}
        )
        if -res.fun > best_val:
            best_val = -res.fun
            best_centers = res.x.reshape((n, 2))
            
    # Fallback if optimization failed completely
    if best_centers is None:
        np.random.seed(0)
        best_centers = np.random.uniform(0.2, 0.8, (n, 2))
        
    # Strictly enforce boundary constraints on centers
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    # Compute final optimal radii
    radii, sum_r = solve_radii(best_centers)
    
    # Microscopic shrinkage to guarantee strict satisfaction of validator tolerances
    radii *= 0.999999999
    sum_r = np.sum(radii)
    
    return best_centers, radii, float(sum_r)
