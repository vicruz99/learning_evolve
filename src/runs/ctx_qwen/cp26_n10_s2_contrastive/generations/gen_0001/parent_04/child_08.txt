import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_func(x):
    """
    Vectorized inequality constraints:
    - Boundary: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    - Non-overlap: dist^2 - (r_i + r_j)^2 >= 0
    """
    n = N_CIRCLES
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(4*n + n*(n-1)//2)
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    
    idx = np.triu_indices(n, 1)
    dx = cx[idx[0]] - cx[idx[1]]
    dy = cy[idx[0]] - cy[idx[1]]
    d2 = dx**2 + dy**2
    rs = r[idx[0]] + r[idx[1]]
    c[4*n:] = d2 - rs**2
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    bounds = [(0.0, 1.0)]*(2*n) + [(0.0, 0.5)]*n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_x = None
    best_val = -np.inf
    
    # Generate diverse initial configurations using perturbed hexagonal lattices
    inits = []
    for k in range(25):
        np.random.seed(k)
        r0 = 0.025 + k * 0.002
        cx, cy = [], []
        y = r0 + 0.05
        row = 0
        while len(cx) < n:
            x = r0 + 0.05 + (row % 2) * r0 * 0.8
            while x < 1.0 - r0 and len(cx) < n:
                cx.append(x)
                cy.append(y)
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
            
        cx = np.array(cx[:n])
        cy = np.array(cy[:n])
        r = np.full(n, r0)
        
        # Add controlled noise to break symmetry and explore landscape
        cx += np.random.randn(n) * 0.01
        cy += np.random.randn(n) * 0.01
        cx = np.clip(cx, r0, 1.0 - r0)
        cy = np.clip(cy, r0, 1.0 - r0)
        
        inits.append(np.concatenate([cx, cy, r]))
        
    # Phase 1: Joint optimization of centers and radii
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-12})
            if -res.fun > best_val:
                best_val = -res.fun
                best_x = res.x.copy()
        except Exception:
            pass
            
    if best_x is None:
        return np.zeros((n, 2)), np.zeros(n), 0.0
        
    # Refine the best solution with tighter tolerances
    res2 = minimize(objective, best_x, method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 6000, 'ftol': 1e-14})
    best_x = res2.x
    
    cx = best_x[0::3]
    cy = best_x[1::3]
    
    # Phase 2: LP refinement for radii given fixed centers
    # This guarantees optimal radii for the found layout, often improving sum significantly
    d_mat = np.sqrt((cx[:, None] - cx[None, :])**2 + (cy[:, None] - cy[None, :])**2)
    idx = np.triu_indices(n, 1)
    d_pairs = np.maximum(d_mat[idx], 1e-12)  # Clamp to avoid infeasibility from coincident points
    
    c_lp = -np.ones(n)
    A_ub = np.zeros((len(idx[0]), n))
    for k in range(len(idx[0])):
        A_ub[k, idx[0][k]] = 1.0
        A_ub[k, idx[1][k]] = 1.0
    b_ub = d_pairs
    
    bnds = []
    for i in range(n):
        max_r = min(cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
        bnds.append((0.0, max(max_r, 0.0)))
        
    r_final = np.full(n, 0.01)  # Fallback
    try:
        res_lp = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, bounds=bnds, method='highs')
        if res_lp.success:
            r_final = res_lp.x
    except Exception:
        pass
        
    # Phase 3: Numerical safety fix to guarantee validation passes
    for _ in range(100):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((cx[i] - cx[j])**2 + (cy[i] - cy[j])**2)
                if dist < r_final[i] + r_final[j] - 1e-10:
                    red = (r_final[i] + r_final[j] - dist) / 2.0 + 1e-10
                    r_final[i] -= red
                    r_final[j] -= red
                    changed = True
        if not changed:
            break
            
    # Ensure non-negative radii
    r_final = np.maximum(r_final, 0.0)
    
    return np.column_stack((cx, cy)), r_final, float(np.sum(r_final))
