To maximize the sum of radii for 26 circles in a unit square, we employ a hybrid optimization strategy that decouples center and radius optimization in the first phase, then refines both simultaneously in the second phase.

1. **LP-based Center Optimization (Phase 1)**: For any fixed set of centers, finding the optimal radii that maximize their sum is a Linear Programming (LP) problem. The constraints are linear: $r_i + r_j \leq d_{ij}$ and boundary constraints $r_i \leq x_i, 1-x_i, y_i, 1-y_i$. We use `scipy.optimize.linprog` (method='highs') to solve this exactly and quickly. We then use a derivative-free optimizer (Nelder-Mead) to search the center space, maximizing the LP objective value. This avoids the non-smoothness and constraint handling difficulties of optimizing centers and radii jointly from scratch.

2. **Hexagonal Initialization**: We generate multiple initial configurations based on a perturbed hexagonal lattice (6-5-6-5-4 row structure), which is known to be near-optimal for dense circle packing. Random perturbations help escape symmetric local optima.

3. **Smooth Refinement (Phase 2)**: Once a high-quality configuration is found from Phase 1, we switch to `SLSQP` to optimize both centers and radii simultaneously. We use squared distance constraints $(x_i-x_j)^2 + (y_i-y_j)^2 \geq (r_i+r_j)^2$ for better numerical conditioning and avoid square-root singularities. Tight tolerances and post-processing ensure strict feasibility.

4. **Robustness**: We run multiple random seeds, clip centers to valid ranges during evaluation, and apply a final corrective pass to guarantee all constraints are satisfied within numerical tolerance before returning.

import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise distances
    dists = np.sqrt(((centers[:, None] - centers[None, :])**2).sum(axis=2))
    idx = np.triu_indices(n, k=1)
    
    n_constr = n * (n - 1) // 2 + 4 * n
    A_ub = np.zeros((n_constr, n))
    b_ub = np.zeros(n_constr)
    
    k = 0
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i, j in zip(idx[0], idx[1]):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]
        k += 1
        
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        vals = [x, 1.0 - x, y, 1.0 - y]
        for v in vals:
            A_ub[k, i] = 1.0
            b_ub[k] = v
            k += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def obj_centers(centers_flat):
    """Objective for center optimization: negative max sum of radii."""
    n = len(centers_flat) // 2
    centers = centers_flat.reshape((n, 2))
    # Clip to avoid negative bounds in LP
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    _, s = solve_radii_lp(centers)
    return -s

def obj_slscqp(x, n):
    """Objective for joint refinement: negative sum of radii."""
    return -np.sum(x[2::3])

def cons_slscqp(x, n):
    """Inequality constraints >= 0 for SLSQP."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.empty(n * (n - 1) // 2 + 4 * n)
    
    # Boundary: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c[:n] = cx - r
    c[n:2*n] = 1.0 - cx - r
    c[2*n:3*n] = cy - r
    c[3*n:4*n] = 1.0 - cy - r
    
    # Pairwise squared distance: dist^2 - (r_i + r_j)^2 >= 0
    idx = np.triu_indices(n, k=1)
    i_idx, j_idx = idx
    dist_sq = (cx[i_idx] - cx[j_idx])**2 + (cy[i_idx] - cy[j_idx])**2
    r_sum_sq = (r[i_idx] + r[j_idx])**2
    c[4*n:] = dist_sq - r_sum_sq
    
    return c

def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    def hex_init(seed):
        rng = np.random.RandomState(seed)
        centers = np.zeros((n, 2))
        rows = [6, 5, 6, 5, 4]
        y_base, y_step = 0.12, 0.18
        x_base = 0.12
        
        idx = 0
        for r_idx, cnt in enumerate(rows):
            y = y_base + r_idx * y_step
            shift = 0.08 if r_idx % 2 == 1 else 0.0
            xs = np.linspace(x_base + shift, 1.0 - x_base - shift, cnt)
            for k in range(cnt):
                centers[idx] = [xs[k], y]
                idx += 1
        centers += rng.normal(0, 0.015, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        return centers.flatten()

    # Phase 1: Optimize centers using LP for radii
    for seed in range(25):
        x0 = hex_init(seed)
        # Nelder-Mead handles non-smooth LP objective well
        res = minimize(obj_centers, x0, method='Nelder-Mead',
                       options={'maxiter': 2500, 'xatol': 1e-7, 'fatol': 1e-9})
        
        curr_c = res.x.reshape((n, 2))
        curr_r, curr_s = solve_radii_lp(curr_c)
        if curr_s > best_sum:
            best_sum = curr_s
            best_centers = curr_c.copy()
            best_radii = curr_r.copy()
            
    # Phase 2: Joint refinement with SLSQP
    if best_centers is not None:
        x0_ref = np.concatenate([best_centers.ravel(), best_radii])
        bnd_all = [(0, 1), (0, 1), (0, 0.5)] * n
        cons = {'type': 'ineq', 'fun': cons_slscqp, 'args': (n,)}
        
        res2 = minimize(obj_slscqp, x0_ref, args=(n,), method='SLSQP',
                        bounds=bnd_all, constraints=cons,
                        options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
        
        opt_c = res2.x[:2*n].reshape((n, 2))
        opt_r = res2.x[2*n:]
        opt_r = np.maximum(opt_r, 0.0)
        
        # Enforce boundary limits strictly
        for i in range(n):
            mr = min(opt_c[i, 0], 1.0 - opt_c[i, 0], opt_c[i, 1], 1.0 - opt_c[i, 1])
            opt_r[i] = min(opt_r[i], mr)
            
        # Iterative overlap resolution
        for _ in range(50):
            changed = False
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt(np.sum((opt_c[i] - opt_c[j])**2))
                    if d < opt_r[i] + opt_r[j] - 1e-9:
                        exc = opt_r[i] + opt_r[j] - d
                        opt_r[i] -= exc / 2.0
                        opt_r[j] -= exc / 2.0
                        changed = True
            if not changed:
                break
                
        opt_r = np.maximum(opt_r, 1e-9)
        
        new_s = np.sum(opt_r)
        if new_s > best_sum:
            best_centers, best_radii, best_sum = opt_c, opt_r, new_s
            
    return best_centers, best_radii, float(best_sum)
