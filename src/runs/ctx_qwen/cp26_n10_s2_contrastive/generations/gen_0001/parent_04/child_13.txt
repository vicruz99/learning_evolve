import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to find radii maximizing sum(radii)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Maximize sum(r) <=> Minimize -sum(r)
    
    # Boundary constraints: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(max_r, 0.0)))
        
    # Pairwise non-overlap constraints: r_i + r_j <= distance(i, j)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(d, 1e-9)  # Prevent degenerate 0-distance
            idx += 1
            
    # Solve using Highs (fast interior-point/dual-simplex hybrid)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def objective(x):
    """Objective for center optimization: negative sum of optimal radii."""
    n = 26
    centers = np.clip(x.reshape(n, 2), 1e-6, 1.0 - 1e-6)
    _, val = solve_radii_lp(centers)
    return -val

def make_hex(rows_cols, noise=0.0, seed=None):
    """Generate hexagonal-like initial placement."""
    n = 26
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    idx = 0
    for r_idx, (count, shift) in enumerate(rows_cols):
        y = (r_idx + 1.0) / (len(rows_cols) + 1.0)
        for c in range(count):
            x = (c + 1.0 + shift) / (count + 2.0)
            centers[idx] = [x, y]
            idx += 1
            if idx >= n: break
        if idx >= n: break
    if noise > 0:
        centers += rng.randn(n, 2) * noise
        centers = np.clip(centers, 0.05, 0.95)
    return centers.flatten()

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Diverse hexagonal lattice patterns optimized for n=26
    patterns = [
        [(6, 0.5), (5, 0.0), (6, 0.5), (5, 0.0), (4, 0.5)],
        [(5, 0.0), (6, 0.5), (5, 0.0), (6, 0.5), (4, 0.0)],
        [(4, 0.0), (6, 0.5), (6, 0.0), (6, 0.5), (4, 0.0)],
        [(5, 0.0), (5, 0.5), (5, 0.0), (5, 0.5), (6, 0.0)]
    ]
    
    # Create initial seeds
    seeds = []
    for pat in patterns:
        seeds.append(make_hex(pat, noise=0.005))
    for i in range(4):
        seeds.append(make_hex(patterns[0], noise=0.015 + i*0.005, seed=i+100))
        
    # Optimize centers using Powell's method (derivative-free, robust for non-smooth LP objectives)
    for i, x0 in enumerate(seeds):
        try:
            res = minimize(objective, x0, method='Powell', 
                           options={'maxiter': 1500, 'xatol': 1e-10, 'fatol': 1e-10})
            if -res.fun > best_sum:
                centers_opt = np.clip(res.x.reshape(n, 2), 1e-6, 1.0 - 1e-6)
                radii_opt, s = solve_radii_lp(centers_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
        except Exception:
            continue
            
    # Fallback if optimization fails
    if best_centers is None:
        centers = np.zeros((n, 2))
        for i in range(n):
            centers[i] = [0.1 + (i%5)*0.18, 0.1 + (i//5)*0.18]
        radii, best_sum = solve_radii_lp(centers)
        best_centers = centers
        best_radii = radii
        
    return best_centers, best_radii, float(best_sum)
