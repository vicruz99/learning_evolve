import numpy as np
from scipy.optimize import minimize
import math


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Uses multiple diverse initializations with SLSQP optimization and post-processing.
    """
    n = 26

    def objective(x):
        """Minimize negative sum of radii (maximize sum)."""
        return -np.sum(x[2::3])

    def constraints(x):
        """Boundary and non-overlap constraints (must be >= 0)."""
        cx = x[0::3]
        cy = x[1::3]
        r = x[2::3]

        c = []
        # Boundary: r <= x <= 1-r, r <= y <= 1-r
        c.extend(cx - r)
        c.extend(1.0 - cx - r)
        c.extend(cy - r)
        c.extend(1.0 - cy - r)

        # Non-overlap: dist >= r_i + r_j
        for i in range(n):
            for j in range(i + 1, n):
                dx = cx[i] - cx[j]
                dy = cy[i] - cy[j]
                dist = math.sqrt(max(dx * dx + dy * dy, 1e-30))
                c.append(dist - r[i] - r[j])

        return np.array(c)

    def make_hex_init(spacing, margin, seed):
        """Create hexagonal lattice initialization with noise."""
        np.random.seed(seed)
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.035

        idx = 0
        row = 0
        while idx < n:
            y = margin + row * spacing * math.sqrt(3) / 2
            x_start = margin + (row % 2) * spacing / 2

            col = 0
            while x_start + col * spacing <= 1 - margin and idx < n:
                centers[idx] = [x_start + col * spacing, y]
                idx += 1
                col += 1
            row += 1

        # Add noise
        centers += np.random.randn(n, 2) * 0.01
        centers = np.clip(centers, 0.02, 0.98)
        return centers, radii

    def make_random_init(seed):
        """Create random initialization."""
        np.random.seed(seed)
        centers = np.random.uniform(0.15, 0.85, (n, 2))
        radii = np.ones(n) * 0.04
        return centers, radii

    def make_grid_init(seed):
        """Create grid-like initialization."""
        np.random.seed(seed)
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.035

        # Try 5x5 grid + extra
        idx = 0
        for row in range(5):
            for col in range(5):
                if idx < n:
                    x = 0.08 + col * 0.18
                    y = 0.08 + row * 0.18
                    centers[idx] = [x, y]
                    idx += 1

        # Place remaining in gaps
        while idx < n:
            centers[idx] = [0.17 + (idx - 25) * 0.08, 0.5]
            idx += 1

        centers += np.random.randn(n, 2) * 0.008
        centers = np.clip(centers, 0.03, 0.97)
        return centers, radii

    def fix_violations(centers, radii):
        """Fix boundary and overlap violations."""
        radii = radii.copy()
        centers = centers.copy()

        # Fix boundary
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if max_r < r:
                radii[i] = max(0.0, max_r - 1e-10)

        # Iteratively fix overlaps
        for _ in range(200):
            max_violation = 0
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = math.sqrt(dx * dx + dy * dy)
                    min_dist = radii[i] + radii[j]
                    if dist < min_dist:
                        max_violation = max(max_violation, min_dist - dist)

            if max_violation < 1e-12:
                break

            # Scale radii down slightly
            radii *= (1.0 - 0.5 * max_violation)
            radii = np.maximum(radii, 0.0)

        return centers, radii

    best_sum = -1.0
    best_centers = None
    best_radii = None

    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-10, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}

    # Strategy 1: Hexagonal patterns with varied parameters
    for trial in range(18):
        spacing = 0.135 + (trial % 6) * 0.008
        margin = 0.06 + (trial // 6) * 0.01
        centers, radii = make_hex_init(spacing, margin, seed=trial * 7 + 13)

        x0 = np.concatenate([centers.ravel(), radii])

        try:
            res = minimize(
                objective, x0, method='SLSQP', bounds=bounds,
                constraints=cons,
                options={'maxiter': 8000, 'ftol': 1e-15, 'disp': False}
            )

            if res.success:
                opt_centers = res.x[:2 * n].reshape(n, 2)
                opt_radii = res.x[2 * n:]
                opt_centers, opt_radii = fix_violations(opt_centers, opt_radii)
                current_sum = np.sum(opt_radii)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = opt_centers.copy()
                    best_radii = opt_radii.copy()
        except Exception:
            continue

    # Strategy 2: Grid patterns
    for trial in range(8):
        centers, radii = make_grid_init(seed=trial * 31 + 7)

        x0 = np.concatenate([centers.ravel(), radii])

        try:
            res = minimize(
                objective, x0, method='SLSQP', bounds=bounds,
                constraints=cons,
                options={'maxiter': 8000, 'ftol': 1e-15, 'disp': False}
            )

            if res.success:
                opt_centers = res.x[:2 * n].reshape(n, 2)
                opt_radii = res.x[2 * n:]
                opt_centers, opt_radii = fix_violations(opt_centers, opt_radii)
                current_sum = np.sum(opt_radii)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = opt_centers.copy()
                    best_radii = opt_radii.copy()
        except Exception:
            continue

    # Strategy 3: Random initializations
    for trial in range(10):
        centers, radii = make_random_init(seed=trial * 41 + 17)

        x0 = np.concatenate([centers.ravel(), radii])

        try:
            res = minimize(
                objective, x0, method='SLSQP', bounds=bounds,
                constraints=cons,
                options={'maxiter': 6000, 'ftol': 1e-15, 'disp': False}
            )

            if res.success:
                opt_centers = res.x[:2 * n].reshape(n, 2)
                opt_radii = res.x[2 * n:]
                opt_centers, opt_radii = fix_violations(opt_centers, opt_radii)
                current_sum = np.sum(opt_radii)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = opt_centers.copy()
                    best_radii = opt_radii.copy()
        except Exception:
            continue

    # Strategy 4: Perturbations of best solution for refinement
    if best_centers is not None:
        for trial in range(15):
            np.random.seed(trial * 53 + 11)
            centers = best_centers.copy()
            radii = best_radii.copy()

            # Perturb positions and radii
            noise_pos = np.random.randn(n, 2) * 0.005
            centers += noise_pos
            centers = np.clip(centers, 0.02, 0.98)

            noise_rad = np.random.randn(n) * 0.002
            radii += noise_rad
            radii = np.clip(radii, 0.001, 0.5)

            x0 = np.concatenate([centers.ravel(), radii])

            try:
                res = minimize(
                    objective, x0, method='SLSQP', bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 10000, 'ftol': 1e-15, 'disp': False}
                )

                if res.success:
                    opt_centers = res.x[:2 * n].reshape(n, 2)
                    opt_radii = res.x[2 * n:]
                    opt_centers, opt_radii = fix_violations(opt_centers, opt_radii)
                    current_sum = np.sum(opt_radii)
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_centers = opt_centers.copy()
                        best_radii = opt_radii.copy()
            except Exception:
                continue

    # Fallback if nothing worked
    if best_centers is None:
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        idx = 0
        for row in range(5):
            for col in range(5):
                if idx < n:
                    centers[idx] = [0.1 + col * 0.2, 0.1 + row * 0.2]
                    radii[idx] = 0.09
                    idx += 1
        while idx < n:
            centers[idx] = [0.5, 0.5]
            radii[idx] = 1e-6
            idx += 1
        best_sum = np.sum(radii)
        best_centers = centers
        best_radii = radii

    return best_centers, best_radii, float(best_sum)
