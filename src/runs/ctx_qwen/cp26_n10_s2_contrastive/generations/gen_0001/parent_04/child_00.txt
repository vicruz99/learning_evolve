

<strategy>
The problem of packing circles to maximize the sum of radii can be decoupled into optimizing circle centers while solving a Linear Programming (LP) problem for radii. For fixed centers, the optimal radii that maximize their sum subject to non-overlap and boundary constraints are given by a simple LP. This transforms the problem into a derivative-free optimization over 52 variables (the centers), where the objective is the negative sum of radii from the LP. This approach avoids the non-smoothness and constraint handling difficulties of joint optimization.

Strategy:
1. Define a function `solve_packing_lp(centers)` that constructs the LP matrix and solves for optimal radii. Vectorized distance and constraint matrix generation ensures speed.
2. Define `lp_objective(centers_flat)` that calls the LP solver and returns the negative sum of radii. Centers are clipped to stay strictly inside the square to maintain LP feasibility.
3. Generate multiple initial center configurations: a hexagonal lattice, perturbed versions of it, and random points.
4. Use `Nelder-Mead` optimization on each initialization to find local maxima of the sum of radii.
5. Run a refinement phase with smaller perturbations around the best found solution to escape shallow local optima and polish the result.
6. Apply a tiny scaling factor to radii to satisfy the validator's strict `1e-12` tolerance against numerical boundary/overlap violations.
7. Return the best centers, radii, and sum.

import numpy as np
from scipy.optimize import linprog, minimize

def solve_packing_lp(centers):
    n = len(centers)
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.hypot(diffs[:,:,0], diffs[:,:,1])
    
    I, J = np.triu_indices(n, k=1)
    A_ub = np.zeros((len(I), n))
    A_ub[np.arange(len(I)), I] = 1.0
    A_ub[np.arange(len(I)), J] = 1.0
    b_ub = dists[I, J]
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, max_r)))
        
    try:
        res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
        else:
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0

def lp_objective(centers_flat):
    n = len(centers_flat) // 2
    centers = centers_flat.reshape(n, 2)
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    _, neg_sum = solve_packing_lp(centers)
    return neg_sum

def run_packing():
    np.random.seed(42)
    n = 26
    best_val = -1.0
    best_centers = None
    best_radii = None

    inits = []
    # Hexagonal pattern
    pts = []
    y = 0.08
    row = 0
    while len(pts) < n:
        x = 0.08 if row % 2 == 0 else 0.16
        while x <= 0.92 and len(pts) < n:
            pts.append([x, y])
            x += 0.16
        y += 0.14
        row += 1
    inits.append(np.array(pts[:n]))

    for _ in range(6):
        p = inits[0].copy()
        p += np.random.normal(0, 0.025, p.shape)
        p = np.clip(p, 0.05, 0.95)
        inits.append(p)

    inits.append(np.random.uniform(0.15, 0.85, (n, 2)))

    for c0 in inits:
        x0 = c0.flatten()
        res = minimize(lp_objective, x0, method='Nelder-Mead',
                       options={'maxiter': 5000, 'xatol': 1e-6, 'fatol': 1e-6})
        curr_centers = np.clip(res.x.reshape(n, 2), 0, 1)
        radii, neg_s = solve_packing_lp(curr_centers)
        curr_sum = -neg_s
        if curr_sum > best_val:
            best_val = curr_sum
            best_centers = curr_centers.copy()
            best_radii = radii.copy()

    # Refinement phase
    if best_centers is not None:
        for _ in range(10):
            c_ref = best_centers + np.random.normal(0, 0.003, best_centers.shape)
            c_ref = np.clip(c_ref, 0.02, 0.98)
            res = minimize(lp_objective, c_ref.flatten(), method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-7, 'fatol': 1e-7})
            curr_centers = np.clip(res.x.reshape(n, 2), 0, 1)
            radii, neg_s = solve_packing_lp(curr_centers)
            curr_sum = -neg_s
            if curr_sum > best_val:
                best_val = curr_sum
                best_centers = curr_centers.copy()
                best_radii = radii.copy()

    if best_radii is not None:
        best_radii *= (1.0 - 1e-9)

    return best_centers, best_radii, float(best_val)
