import numpy as np
from scipy.optimize import minimize

def objective(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_fun(x, n):
    """Constraint function: returns array of inequality constraints (must be >= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    c = []
    
    # Boundary constraints: circles inside [0,1]x[0,1]
    c.extend(cx - r)          # x >= r
    c.extend(1.0 - cx - r)    # x + r <= 1
    c.extend(cy - r)          # y >= r
    c.extend(1.0 - cy - r)    # y + r <= 1
    
    # Overlap constraints: distance between centers >= sum of radii
    # Vectorized computation for speed
    i, j = np.triu_indices(n, k=1)
    dx = cx[i] - cx[j]
    dy = cy[i] - cy[j]
    dists = np.hypot(dx, dy)
    c.extend(dists - (r[i] + r[j]))
    
    return np.array(c)

    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n

    cons = {'type': 'ineq', 'fun': constraint_fun, 'args': (n,)}
