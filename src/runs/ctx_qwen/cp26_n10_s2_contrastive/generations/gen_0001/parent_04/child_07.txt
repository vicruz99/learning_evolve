import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
TRIU_IDX = np.triu_indices(N_CIRCLES, k=1)

def objective(x):
    """Negative sum of radii (to be minimized)."""
    return -np.sum(x[2::3])

def constr(x):
    """
    Inequality constraints: g(x) >= 0
    - Boundary: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    - Non-overlap: dist^2 - (r_i + r_j)^2 >= 0
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    c = []
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    c.extend(d2[TRIU_IDX] - r_sum[TRIU_IDX]**2)
    return np.array(c)

def make_hex(r_start, shift_x=0.0, shift_y=0.0):
    """Generate a hexagonal lattice initialization."""
    centers = []
    y = r_start + shift_y
    row = 0
    while len(centers) < N_CIRCLES:
        x_start = r_start + shift_x + (r_start if row % 2 == 1 else 0)
        x = x_start
        while x <= 1.0 - r_start + 1e-9 and len(centers) < N_CIRCLES:
            centers.append([x, y, r_start])
            x += 2 * r_start
        y += r_start * np.sqrt(3)
        row += 1
    while len(centers) < N_CIRCLES:
        centers.append([0.5, 0.5, r_start])
    return np.array(centers[:N_CIRCLES]).flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constr}
    
    best_sum = -1.0
    best_x = None
    
    # 1. Generate diverse initial configurations
    inits = []
    # Structured hexagonal starts with varying parameters

        r = np.full(N_CIRCLES, 0.04)

        inits.append(np.concatenate([cx, cy, r]))

    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,

            curr_sum = -res.fun
        except Exception:
            continue
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_x = res.x.copy()
            
        # Perturbation search to escape local minima
        for _ in range(5):
            x_pert = best_x + np.random.randn(3*N_CIRCLES) * 0.001

                curr_sum2 = -res2.fun
