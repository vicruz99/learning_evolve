from scipy.optimize import minimize

N = 26

def objective(x):

def constraints(x):

    cy = x[1::3]
    r = x[2::3]
    
    c = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)

    dy = cy[:, None] - cy[None, :]
