import numpy as np
from scipy.optimize import minimize

N = 26

def objective(x):
    """Negative sum of radii (minimizing this maximizes the sum)."""
    return -np.sum(x[2*N:])

def constraints(x):
    """
    Inequality constraints g(x) >= 0:
    - Boundary: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    - Non-overlap: (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    """
    cx = x[:N]
    cy = x[N:2*N]
    r = x[2*N:]
    
    num_overlap = N * (N - 1) // 2
    c = np.empty(4 * N + num_overlap)
    
    # Boundary constraints
    c[:N] = cx - r
    c[N:2*N] = 1.0 - cx - r
    c[2*N:3*N] = cy - r
    c[3*N:4*N] = 1.0 - cy - r
    
    # Pairwise non-overlap constraints (squared distance)
    idx = 4 * N
    for i in range(N):
        for j in range(i + 1, N):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            dr = r[i] + r[j]
            c[idx] = dx * dx + dy * dy - dr * dr
            idx += 1
    return c

def hex_init(seed, spacing=0.16, margin=0.08):
    """Create a hexagonal lattice initialization with random perturbation."""
    np.random.seed(seed)
    centers = []
    y = margin
    row = 0
    while len(centers) < N:
        x = margin + (row % 2) * spacing / 2
        while x <= 1.0 - margin and len(centers) < N:
            centers.append([x, y])
            x += spacing
        y += spacing * np.sqrt(3) / 2
        row += 1
        
    centers = np.array(centers[:N])
    centers += np.random.randn(*centers.shape) * 0.015
    centers = np.clip(centers, 0.05, 0.95)
    r = np.full(N, 0.035)
    return np.concatenate([centers.ravel(), r])

def random_init(seed):
    """Create a random initialization within safe inner bounds."""
    np.random.seed(seed)
    cx = np.random.uniform(0.1, 0.9, N)
    cy = np.random.uniform(0.1, 0.9, N)
    r = np.full(N, 0.03)
    return np.concatenate([cx, cy, r])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Strategy 1: Multiple diverse restarts
    for s in range(40):
        if s % 3 == 0:
            x0 = hex_init(s, spacing=0.15 + 0.01*(s%4), margin=0.05 + 0.02*(s%4))
        elif s % 3 == 1:
            x0 = random_init(s)
        else:
            # Perturbed hex grid
            x0 = hex_init(s, spacing=0.14, margin=0.06)
            x0[:2*N] += np.random.randn(2*N) * 0.02
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-12})
            if res.success:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = res.x[:2*N].reshape(N, 2)
                    best_radii = res.x[2*N:].copy()
        except Exception:
            pass
            
    # Strategy 2: Local refinement of the best found solution
    if best_centers is not None:
        for s in range(15):
            np.random.seed(s + 500)
            x0 = best_centers.ravel().copy()
            r0 = best_radii.copy()
            
            x0 += np.random.randn(2*N) * 0.005
            r0 += np.random.randn(N) * 0.002
            r0 = np.clip(r0, 0.001, 0.5)
            x_init = np.concatenate([x0, r0])
            
            try:
                res = minimize(objective, x_init, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-13})
                if res.success:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = res.x[:2*N].reshape(N, 2)
                        best_radii = res.x[2*N:].copy()
            except Exception:
                pass
                
    # Final validation and numerical safety adjustment
    if best_centers is not None:
        cx, cy, r = best_centers[:,0], best_centers[:,1], best_radii
        max_viol = 0.0
        
        # Check boundary violations
        for i in range(N):
            mx = min(cx[i], 1.0-cx[i], cy[i], 1.0-cy[i])
            if r[i] > mx:
                max_viol = max(max_viol, r[i] - mx)
                
        # Check overlap violations
        for i in range(N):
            for j in range(i+1, N):
                d = np.sqrt((cx[i]-cx[j])**2 + (cy[i]-cy[j])**2)
                if r[i]+r[j] > d:
                    max_viol = max(max_viol, r[i]+r[j] - d)
                    
        # Scale down radii slightly if any numerical violation exists
        if max_viol > 1e-9:
            scale = max(0.99, 1.0 - max_viol / np.sum(r) * 1.1)
            best_radii *= scale
            best_sum = np.sum(best_radii)
            
    # Fallback if all optimizers fail (highly unlikely)
    if best_centers is None:
        np.random.seed(42)
        best_centers = np.random.rand(N, 2) * 0.8 + 0.1
        best_radii = np.full(N, 0.02)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
