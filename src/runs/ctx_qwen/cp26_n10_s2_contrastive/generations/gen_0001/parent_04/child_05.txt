import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers):
    """Solve LP to find radii maximizing sum(r) for fixed centers."""
    n = centers.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            dists[i, j] = d
            dists[j, i] = d
            
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = []
    for i in range(n):
        x, y = centers[i]
        r_max = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, r_max)))
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except ValueError:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def obj_centers(centers_flat):
    """Objective for center optimization: negative max sum of radii."""
    c = np.clip(centers_flat.reshape(-1, 2), 1e-7, 1.0 - 1e-7)
    _, s = solve_lp(c)
    return -s

def full_obj(x):
    """Objective for joint optimization: negative sum of radii."""
    return -np.sum(x[2::3])

def full_constraints(x):
    """Constraints for joint optimization."""
    n = 26
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    cons = []
    cons.append(cx - r)
    cons.append(1.0 - cx - r)
    cons.append(cy - r)
    cons.append(1.0 - cy - r)
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    d2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    
    idx = np.tril_indices(n, k=-1)
    cons.append(d2[idx] - rs[idx]**2)
    
    return np.concatenate(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_s = 0.0
    best_c = None
    best_r = None

        y += 0.105
        row += 1
    inits.append(np.array(hc[:n]).flatten())
    
    # Grid pattern
    gc = []
    for i in range(5):
        for j in range(5):
            gc.append([0.1 + 0.18 * i, 0.1 + 0.18 * j])
    gc.append([0.5, 0.5])
    inits.append(np.array(gc[:n]).flatten())
    
    # Random pattern
    np.random.seed(101)
    inits.append(np.random.uniform(0.15, 0.85, (n, 2)).flatten())
    
    for x0 in inits:

        curr_s = -res.fun
        if curr_s > best_s:
            best_s = curr_s

            best_r[i] = max(0.0, mr)
            
    best_r = np.maximum(best_r, 0.0)
    return best_c, best_r, float(np.sum(best_r))
