To maximize the sum of radii for 26 circles in a unit square, we decouple the problem into two stages that are iteratively refined:
1. **Linear Programming for Radii**: Given fixed centers, the optimal radii can be found exactly by solving a linear program. The constraints are linear: $r_i + r_j \leq d_{ij}$ (pairwise distances) and $0 \leq r_i \leq \min(x_i, 1-x_i, y_i, 1-y_i)$ (boundary constraints). This LP is solved very efficiently using `scipy.optimize.linprog` with the `highs` solver.
2. **Derivative-Free Optimization for Centers**: We optimize the center coordinates to maximize the result of the LP. Since the LP's optimal value as a function of centers is continuous but non-smooth (piecewise linear), we use `Nelder-Mead`, which handles non-smooth objectives robustly.
3. **SLSQP Joint Refinement**: After obtaining a strong configuration from the LP+Nelder-Mead pipeline, we perform a final joint refinement of centers and radii using `SLSQP` with smooth squared-distance constraints. This polishes the solution and ensures strict feasibility.
4. **Multiple Restarts & Initialization**: We seed the optimization with a hexagonal lattice, a uniform grid, and several random initializations to escape local optima. Each is processed through the full pipeline, and the best valid result is retained.
5. **Numerical Safety**: A final post-processing pass clamps any minor boundary/overlap violations to guarantee the output passes the strict validation function.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to find radii that maximize sum(radii)."""
    n = len(centers)
    c = -np.ones(n)
    
    # Constraints: r_i + r_j <= dist_ij
    idx = 0
    A_ub = np.zeros((n * (n - 1) // 2, n))
    b_ub = np.zeros(n * (n - 1) // 2)
    
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    # Bounds: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        if ub < 0.0:
            ub = 0.0
        bounds.append((0.0, ub))
        
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 1e-7), 1e-7 * n


def obj_centers(centers_flat):
    """Objective for optimizing centers: negative sum of optimal radii."""
    centers = centers_flat.reshape(-1, 2)
    _, s = solve_radii_lp(centers)
    return -s


def obj_sl(x):
    """Objective for SLSQP joint refinement."""
    return -np.sum(x[2::3])


def cons_sl(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    n = len(r)
    cons = []
    
    # Boundary constraints
    cons.extend(cx - r)
    cons.extend(1.0 - cx - r)
    cons.extend(cy - r)
    cons.extend(1.0 - cy - r)
    
    # Overlap constraints (squared distance >= sum of radii squared)
    for i in range(n):
        for j in range(i + 1, n):
            d2 = (cx[i] - cx[j]) ** 2 + (cy[i] - cy[j]) ** 2
            rs = r[i] + r[j]
            cons.append(d2 - rs ** 2)
            
    return np.array(cons)


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Optimizes the packing of 26 circles in a unit square to maximize the sum of radii."""
    n = N_CIRCLES
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds_cent = [(0.0, 1.0)] * (2 * n)
    bounds_sl = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons_dict = {'type': 'ineq', 'fun': cons_sl}
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice
    hex_c = []
    r_h = 0.095
    dy = r_h * np.sqrt(3)
    for row in range(10):
        y = r_h + row * dy
        if y + r_h > 1.0:
            break
        for col in range(10):
            x = r_h + col * 2 * r_h + (row % 2) * r_h
            if x + r_h <= 1.0:
                hex_c.append([x, y])
    while len(hex_c) < n:
        hex_c.append([0.5, 0.5])
    inits.append(np.array(hex_c[:n]).flatten())
    
    # 2. Grid pattern
    grid_c = []
    for i in range(6):
        for j in range(5):
            if len(grid_c) < n:
                grid_c.append([0.1 + j * 0.18, 0.1 + i * 0.18])
    while len(grid_c) < n:
        grid_c.append([0.5, 0.5])
    inits.append(np.array(grid_c[:n]).flatten())
    
    # 3. Random restarts with perturbations
    np.random.seed(42)
    for _ in range(6):
        c = np.random.rand(n, 2) * 0.6 + 0.2
        c += np.random.randn(n, 2) * 0.01
        c = np.clip(c, 0.01, 0.99)
        inits.append(c.flatten())
        
    # Optimization pipeline
    for x0 in inits:
        # Stage 1: Nelder-Mead on centers (handles non-smooth LP objective well)
        res_nm = minimize(obj_centers, x0, method='Nelder-Mead', bounds=bounds_cent,
                          options={'maxiter': 4000, 'fatol': 1e-9, 'xatol': 1e-9})
        
        curr_centers = res_nm.x.reshape(n, 2)
        curr_radii, curr_sum = solve_radii_lp(curr_centers)
        
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = curr_centers.copy()
            best_radii = curr_radii.copy()
            
        # Stage 2: SLSQP joint refinement for polishing
        full_x0 = np.zeros(3 * n)
        for i in range(n):
            full_x0[3 * i] = curr_centers[i, 0]
            full_x0[3 * i + 1] = curr_centers[i, 1]
            full_x0[3 * i + 2] = curr_radii[i]
            
        # Add tiny perturbation to help SLSQP escape flat regions
        full_x0 += np.random.randn(3 * n) * 1e-6
        full_x0 = np.clip(full_x0, 0.0, 1.0)
        full_x0[2::3] = np.clip(full_x0[2::3], 0.0, 0.5)
        
        try:
            res_sl = minimize(obj_sl, full_x0, method='SLSQP', bounds=bounds_sl,
                              constraints=cons_dict, options={'maxiter': 2000, 'ftol': 1e-12})
            
            if res_sl.success or -res_sl.fun > best_sum:
                s_sl = -res_sl.fun
                if s_sl > best_sum:
                    best_sum = s_sl
                    best_centers = res_sl.x[:2 * n].reshape(n, 2)
                    best_radii = res_sl.x[2 * n:]
        except Exception:
            pass
            
    # Final numerical safety pass to guarantee validation passes
    centers = best_centers
    radii = best_radii
    for _ in range(10):
        valid = True
        # Fix boundary violations
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-12:
                radii[i] = max(0.0, max_r - 1e-9)
                valid = False
                
        # Fix overlap violations
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if d < radii[i] + radii[j] - 1e-12:
                    excess = radii[i] + radii[j] - d
                    radii[i] -= excess / 2.0
                    radii[j] -= excess / 2.0
                    radii[i] = max(0.0, radii[i])
                    radii[j] = max(0.0, radii[j])
                    valid = False
                    
        if valid:
            break
            
    return centers, radii, float(np.sum(radii))
