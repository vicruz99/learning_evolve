N = 26

def objective(x):

def constraints(x):

    cy = x[1::3]
    r = x[2::3]
    
    n_cons = 4 * N + N * (N - 1) // 2
    c = np.empty(n_cons)
    k = 0
    
    # Boundary constraints: x >= r, x+r <= 1, y >= r, y+r <= 1
    c[k:k+N] = cx - r
    c[k+N:k+2*N] = 1.0 - cx - r
    c[k+2*N:k+3*N] = cy - r

    k += 4 * N
