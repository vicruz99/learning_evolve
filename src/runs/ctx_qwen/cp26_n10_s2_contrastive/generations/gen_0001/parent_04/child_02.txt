import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
I_UPPER, J_UPPER = np.triu_indices(N_CIRCLES, k=1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_constraints(x):
    """Evaluate boundary and non-overlap constraints (must be >= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c_bound = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: squared distance >= squared sum of radii
    dx = cx[I_UPPER] - cx[J_UPPER]
    dy = cy[I_UPPER] - cy[J_UPPER]
    r_sum = r[I_UPPER] + r[J_UPPER]
    c_overlap = dx**2 + dy**2 - r_sum**2
    
    return np.concatenate([c_bound, c_overlap])

def make_init(seed, strategy='hex'):
    """Generate feasible initial configuration."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N_CIRCLES, 2))
    
    if strategy == 'hex':
        scale = 0.14 + rng.uniform(0, 0.06)
        # Row counts summing to 26
        rows = [6, 5, 6, 5, 4]
        y = 0.1 + rng.uniform(0, 0.05)
        dy = scale * np.sqrt(3)
        idx = 0
        for r_idx, count in enumerate(rows):
            x = 0.1 + (0 if r_idx % 2 == 0 else scale/2) + rng.uniform(0, 0.05)
            for _ in range(count):
                if idx < N_CIRCLES:
                    centers[idx] = [x, y]
                    idx += 1
                x += scale
            y += dy
    else:
        centers = rng.uniform(0.15, 0.85, (N_CIRCLES, 2))
        
    # Add noise and ensure inside bounds
    centers += rng.normal(0, 0.01, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    # Start with small, feasible radii
    r0 = np.full(N_CIRCLES, 0.04 + rng.uniform(0, 0.02))
    
    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = r0
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    best_x = None
    best_val = -float('inf')
    
    # Phase 1: Broad search with diverse initializations
    for seed in range(60):
        strat = 'hex' if seed < 40 else 'rand'
        x0 = make_init(seed, strat)

            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,

            if -res.fun > best_val:
                # Verify constraint satisfaction with tolerance
                c_vals = get_constraints(res.x)
                if np.all(c_vals >= -1e-7):
                    best_val = -res.fun
                    best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Local refinement around the best solution
    if best_x is not None:
        for _ in range(8):
            x_pert = best_x + np.random.randn(3 * N_CIRCLES) * 0.0015
            x_pert[2::3] = np.clip(x_pert[2::3], 1e-6, 0.5)
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,

    radii = np.maximum(radii, 1e-9)
    centers = np.clip(centers, 1e-9, 1.0 - 1e-9)
    
    return centers, radii, float(np.sum(radii))
