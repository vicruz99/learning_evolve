import numpy as np
from scipy.optimize import minimize
import math

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Return inequality constraints (must be >= 0)."""
    c = []
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    # Using squared distances avoids square roots and improves smoothness
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            d2 = dx*dx + dy*dy
            rs = r[i] + r[j]
            c.append(d2 - rs*rs)
            
    return np.array(c)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]"""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def is_feasible(x, tol=1e-4):
    """Check if a configuration satisfies all constraints within tolerance."""
    return np.min(constraints(x)) >= -tol

def make_hex_initial(seed):
    """Generate hexagonal lattice initialization with perturbation."""
    rng = np.random.RandomState(seed)
    cx = np.zeros(N_CIRCLES)
    cy = np.zeros(N_CIRCLES)
    
    # 6-5-6-5-4 pattern sums to 26
    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    for row_idx, count in enumerate(row_counts):
        y = (row_idx + 1.0) / 6.0
        offset = 0.0
        if row_idx % 2 == 1:
            offset = 0.5 / count
        for col in range(count):
            x = (col + 0.5) / count + offset
            cx[idx] = x
            cy[idx] = y
            idx += 1
            
    # Add controlled noise
    cx += rng.uniform(-0.04, 0.04, N_CIRCLES)
    cy += rng.uniform(-0.04, 0.04, N_CIRCLES)
    cx = np.clip(cx, 0.05, 0.95)
    cy = np.clip(cy, 0.05, 0.95)
    
    r = np.full(N_CIRCLES, 0.02)  # Start small to guarantee initial feasibility
    return np.concatenate([cx, cy, r])

def make_random_initial(seed):
    """Generate random initialization."""
    rng = np.random.RandomState(seed)
    cx = rng.uniform(0.15, 0.85, N_CIRCLES)
    cy = rng.uniform(0.15, 0.85, N_CIRCLES)
    r = np.full(N_CIRCLES, 0.02)
    return np.concatenate([cx, cy, r])

def run_packing():
    best_x = None
    best_val = np.inf
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Hexagonal lattice restarts
    for seed in range(30):
        x0 = make_hex_initial(seed)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,

            if is_feasible(res.x, 1e-3) and res.fun < best_val:
                best_val = res.fun
                best_x = res.x.copy()
        except Exception:
            pass

    # Phase 2: Random restarts to explore different topologies
    for seed in range(15):
        x0 = make_random_initial(seed)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,

            if is_feasible(res.x, 1e-3) and res.fun < best_val:
                best_val = res.fun
                best_x = res.x.copy()
        except Exception:
            pass

    # Phase 3: Local refinement of the best solution found

        if not overlap_found:
            break
            
    radii = np.maximum(radii, 0.0)
    total_sum = np.sum(radii)
    
    return centers, radii, float(total_sum)
