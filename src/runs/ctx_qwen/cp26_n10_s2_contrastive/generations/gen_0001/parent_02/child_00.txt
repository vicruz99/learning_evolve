import numpy as np
from scipy.optimize import minimize, linprog
import math

def solve_radii_lp(centers):
    """
    Given fixed centers, solve an LP to find radii that maximize sum(radii)
    subject to non-overlap and boundary constraints.
    """
    n = len(centers)
    c_obj = -np.ones(n)
    
    # Boundary constraints become upper bounds on variables
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, max_r)))
        
    # Pairwise constraints: r_i + r_j <= dist_ij
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        xi, yi = centers[i]
        for j in range(i + 1, n):
            xj, yj = centers[j]
            dist = math.hypot(xi - xj, yi - yj)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Try high-performance HiGHS solver first, fallback to interior-point
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback for degenerate cases
    return np.full(n, 1e-8), n * 1e-8

def obj_centers(x):
    """Objective for center optimization: minimize negative sum of optimal radii."""
    n = len(x) // 2
    centers = x.reshape((n, 2))
    # Clip to valid interior region
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    _, s = solve_radii_lp(centers)
    return -s

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    starts.append(np.array(hc[:n]))

    for _ in range(8):

    for init_c in starts:

        c_opt = np.clip(res.x.reshape((n, 2)), 1e-6, 1.0 - 1e-6)
        r_opt, s_opt = solve_radii_lp(c_opt)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    # Phase 2: Joint refinement with SLSQP for high precision
    x0 = np.concatenate([best_centers.flatten(), best_radii])
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n

        return -np.sum(params[2*n:])

        cx = params[:2*n][0::2]

        c.extend(1.0 - cx - r)
        c.extend(cy - r)

            best_centers = c_j
            best_radii = r_j
