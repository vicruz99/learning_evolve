import numpy as np
from scipy.optimize import linprog, minimize


def compute_max_radii(centers):
    """
    Given fixed centers, solve LP to find radii that maximize sum(radii)
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    if n == 0:
        return np.array([]), 0.0
        
    # Objective: minimize -sum(r_i)  <=>  maximize sum(r_i)
    c_obj = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    # Boundary constraints handled via variable bounds:
    # 0 <= r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        r_max = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, r_max)))
        
    # Solve LP
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
        else:
            # Fallback if LP fails (rare)
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0


def packing_objective(centers_flat):
    """Objective function for outer optimizer: minimize negative sum of radii."""
    centers = centers_flat.reshape((26, 2))
    # Soft clip to keep centers strictly inside to avoid degenerate bounds
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    _, sum_r = compute_max_radii(centers)
    return -sum_r


def generate_hex_init(seed):
    """Generate a hexagonal lattice initialization with perturbations."""
    rng = np.random.RandomState(seed)
    centers = []
    
    # Pattern: 6, 5, 6, 5, 4 = 26 circles
    row_configs = [
        (6, 0.0), (5, 0.5), (6, 0.0), (5, 0.5), (4, 0.0)
    ]
    
    for row_idx, (count, offset) in enumerate(row_configs):
        y = (row_idx + 1.0) / 6.0
        for col in range(count):
            x = (col + 1.0) / (count + 1.0) + offset / (count + 1.0)
            centers.append([x, y])
            
    centers = np.array(centers)
    # Add random perturbation to break symmetry and explore variations
    centers += rng.randn(26, 2) * 0.06
    centers = np.clip(centers, 0.1, 0.9)
    return centers


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds_2d = [(0.0, 1.0)] * (2 * n)
    
    # Try multiple restarts from different hexagonal configurations
    for trial in range(12):
        init_centers = generate_hex_init(seed=trial * 7 + 13)
        x0 = init_centers.flatten()
        
        # Phase 1: Derivative-free global exploration
        res_nm = minimize(
            packing_objective,
            x0,
            method='Nelder-Mead',
            options={'maxiter': 2500, 'xatol': 1e-7, 'fatol': 1e-9}
        )
        
        opt_centers_nm = res_nm.x.reshape((n, 2))
        opt_radii_nm, s_nm = compute_max_radii(opt_centers_nm)
        
        # Phase 2: Gradient-based local refinement with bounds
        res_lbfgs = minimize(
            packing_objective,
            res_nm.x,
            method='L-BFGS-B',
            bounds=bounds_2d,
            options={'maxiter': 1500, 'ftol': 1e-12}
        )
        
        opt_centers_ref = res_lbfgs.x.reshape((n, 2))
        opt_radii_ref, s_ref = compute_max_radii(opt_centers_ref)
        
        # Pick the best from this trial
        if s_ref > s_nm:
            current_best_s = s_ref
            current_best_c = opt_centers_ref
            current_best_r = opt_radii_ref
        else:
            current_best_s = s_nm
            current_best_c = opt_centers_nm
            current_best_r = opt_radii_nm
            
        if current_best_s > best_sum:
            best_sum = current_best_s
            best_centers = current_best_c.copy()
            best_radii = current_best_r.copy()
            
    # Final validation and cleanup
    if best_centers is not None:
        # Ensure strict feasibility for the validator
        final_radii = best_radii.copy()
        for i in range(n):
            x, y = best_centers[i]
            r_max = min(x, 1.0 - x, y, 1.0 - y)
            final_radii[i] = min(final_radii[i], max(0.0, r_max - 1e-9))
            
        # Shrink radii slightly to guarantee non-overlap tolerance
        final_radii *= 0.99999
        
        return best_centers, final_radii, float(np.sum(final_radii))
    else:
        # Fallback (should not be reached)
        centers = np.random.rand(n, 2) * 0.6 + 0.2
        radii = np.full(n, 0.01)
        return centers, radii, float(np.sum(radii))
