import numpy as np
from scipy.optimize import minimize
import math


def objective(params, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[2::3])


def get_constraints(params, n):
    """
    Inequality constraints g(params) >= 0.
    Includes boundary constraints and squared pairwise non-overlap constraints.
    """
    cx = params[0::3]
    cy = params[1::3]
    r = params[2::3]

    cons = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons.extend(cx - r)
    cons.extend(1.0 - cx - r)
    cons.extend(cy - r)
    cons.extend(1.0 - cy - r)

    # Overlap constraints: dist_ij^2 >= (r_i + r_j)^2
    cx_mat = cx[:, None]
    cy_mat = cy[:, None]
    r_mat = r[:, None]

    dx = cx_mat - cx_mat.T
    dy = cy_mat - cy_mat.T
    dist_sq = dx**2 + dy**2
    r_sum = r_mat + r_mat.T

    # Extract upper triangle to avoid double counting and self-constraints
    i_idx, j_idx = np.triu_indices(n, k=1)
    cons.extend(dist_sq[i_idx, j_idx] - r_sum[i_idx, j_idx]**2)

    return np.array(cons)


def constraint_wrapper(params):
    """Wrapper to satisfy SLSQP constraint interface."""
    return get_constraints(params, 26)


def create_hex_config(n, spacing, seed=0):
    """Generate a centered hexagonal lattice configuration for n circles."""
    rng = np.random.RandomState(seed)
    row_counts = [6, 5, 6, 5, 4]  # Sums to 26
    num_rows = len(row_counts)
    
    y_step = spacing * math.sqrt(3) / 2
    total_h = (num_rows - 1) * y_step
    y_start = (1.0 - total_h) / 2.0

    centers = np.zeros((n, 2))
    idx = 0
    
    for row_idx, rc in enumerate(row_counts):
        y = y_start + row_idx * y_step
        x_step = spacing
        
        # Center x-coordinates for each row
        row_width = (rc - 1) * x_step
        x_start = (1.0 - row_width) / 2.0
        
        # Offset odd rows for hexagonal packing
        if row_idx % 2 == 1:
            x_start += x_step / 2.0

        for col in range(rc):
            if idx >= n:
                break
            x = x_start + col * x_step
            centers[idx] = [
                np.clip(x + rng.uniform(-0.002, 0.002), 0.05, 0.95),
                np.clip(y + rng.uniform(-0.002, 0.002), 0.05, 0.95)
            ]
            idx += 1

    # Initialize radii small enough to guarantee feasibility
    radii = np.full(n, spacing * 0.4)
    return centers, radii


def run_packing():
    """Main packing function. Returns (centers, radii, sum_radii)."""
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_wrapper}

    # Strategy 1: Systematic hexagonal lattice searches with varying densities
    for spacing in np.linspace(0.18, 0.24, 15):
        centers, radii = create_hex_config(n, spacing)
        x0 = np.concatenate([centers.ravel(), radii])
        
        res = minimize(
            objective, x0, args=(n,), method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False}
        )
        
        if res.success:
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_centers = res.x[:2 * n].reshape((n, 2)).copy()
                best_radii = res.x[2 * n:].copy()

    # Strategy 2: Perturbation-based refinement from the best found so far
    if best_centers is not None:
        for _ in range(15):
            x0 = np.concatenate([best_centers.ravel(), best_radii])
            x0 += np.random.randn(3 * n) * 0.005
            x0 = np.clip(x0, 0.02, 0.98)
            x0[2::3] = np.maximum(x0[2::3], 0.01)  # Keep radii positive
            
            res = minimize(
                objective, x0, args=(n,), method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False}
            )
            
            if res.success and -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x[:2 * n].reshape((n, 2)).copy()
                best_radii = res.x[2 * n:].copy()

    # Final validation and strict constraint enforcement
    if best_centers is not None:
        best_radii = np.maximum(best_radii, 0.0)
        
        # Enforce boundary constraints strictly
        for i in range(n):
            r_max = min(
                best_centers[i, 0], 1.0 - best_centers[i, 0],
                best_centers[i, 1], 1.0 - best_centers[i, 1]
            )
            best_radii[i] = min(best_radii[i], r_max)

        # Iteratively resolve any remaining pairwise overlaps
        for _ in range(200):
            changed = False
            for i in range(n):
                for j in range(i + 1, n):
                    dx = best_centers[i, 0] - best_centers[j, 0]
                    dy = best_centers[i, 1] - best_centers[j, 1]
                    d = math.sqrt(dx * dx + dy * dy)
                    if d < best_radii[i] + best_radii[j] - 1e-10:
                        excess = best_radii[i] + best_radii[j] - d
                        best_radii[i] -= excess / 2.0
                        best_radii[j] -= excess / 2.0
                        changed = True
            if not changed:
                break

    return best_centers, best_radii, float(best_sum)
