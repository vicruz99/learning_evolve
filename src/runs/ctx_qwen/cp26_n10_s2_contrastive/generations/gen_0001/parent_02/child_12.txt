The problem of maximizing the sum of radii for 26 circles in a unit square can be efficiently solved by decoupling the optimization into two stages:
1. **Radius Computation (Linear Programming)**: For any fixed set of centers, finding the optimal radii that maximize $\sum r_i$ while satisfying non-overlap ($r_i + r_j \leq \|c_i - c_j\|$) and boundary constraints ($r_i \leq \text{dist to wall}$) is a Linear Program. This LP is small (~325 constraints for $n=26$) and solves extremely fast.
2. **Center Optimization (Derivative-Free)**: We optimize the 52 center coordinates using `Nelder-Mead` on the negative LP objective value. The LP value function is concave but non-smooth (piecewise linear), making derivative-free methods like Nelder-Mead highly effective. We use multiple high-quality initializations (hexagonal lattice, grid, perturbed random) and two-pass refinement to escape local optima.

This approach avoids the numerical instability of passing hundreds of non-smooth constraints directly to SLSQP, and instead leverages exact constraint satisfaction at each evaluation step, leading to robust convergence near the global optimum.

import numpy as np
from scipy.optimize import linprog, minimize

def get_optimal_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to constraints."""
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Minimize negative sum => Maximize sum
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    x = centers[:, 0]
    y = centers[:, 1]
    ub = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    ub = np.maximum(ub, 0.0)
    bounds = [(0.0, u) for u in ub]
    
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    idx = np.triu_indices(n, k=1)
    diff = centers[idx[0]] - centers[idx[1]]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    
    A_ub = np.zeros((len(idx[0]), n))
    A_ub[idx[0], idx[1]] = 1.0
    A_ub[idx[1], idx[0]] = 1.0
    b_ub = dists
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def objective(centers_flat):
    """Objective for center optimization: negative max sum of radii."""
    n = 26
    centers = centers_flat.reshape(n, 2)
    # Keep centers strictly inside to prevent LP numerical issues
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, val = get_optimal_radii(centers)
    return -val

def generate_hex_initial(n):
    """Hexagonal lattice initialization."""
    centers = np.zeros((n, 2))
    r_guess = 0.085
    dy = r_guess * np.sqrt(3)
    dx = 2.0 * r_guess
    
    count = 0
    row = 0
    y = r_guess + 0.02
    while count < n:
        shift = 0.0 if row % 2 == 0 else dx / 2.0
        x = r_guess + 0.02 + shift
        while x <= 1.0 - r_guess - 0.02 and count < n:
            centers[count] = [x, y]
            count += 1
            x += dx
        y += dy
        row += 1
    return centers.flatten()

def generate_grid_initial(n):
    """Uniform grid initialization with extra point."""
    centers = np.zeros((n, 2))
    x_vals = np.linspace(0.1, 0.9, 5)
    y_vals = np.linspace(0.1, 0.9, 5)
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < n:
                centers[idx] = [x_vals[j], y_vals[i]]
                idx += 1
    if idx < n:
        centers[idx] = [0.25, 0.25]
    return centers.flatten()

def generate_random_initial(n):
    """Random initialization in safe inner region."""
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    return centers.flatten()

def run_packing():
    """Main packing function. Returns (centers, radii, sum_radii)."""
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    np.random.seed(42)
    
    initial_configs = [
        generate_hex_initial(n),
        generate_grid_initial(n),
        generate_random_initial(n),
        generate_random_initial(n) + np.random.normal(0, 0.1, 52),
    ]
    
    for x0 in initial_configs:
        x0 = np.clip(x0.reshape(n, 2), 0.05, 0.95).flatten()
        
        # First optimization pass
        res = minimize(objective, x0, method='Nelder-Mead', 
                       options={'maxiter': 4000, 'xatol': 1e-6, 'fatol': 1e-7})
        
        centers = np.clip(res.x.reshape(n, 2), 1e-4, 1.0 - 1e-4)
        radii, s = get_optimal_radii(centers)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
            
        # Refinement pass around the best found
        x0_ref = res.x + np.random.normal(0, 0.005, 52)
        res2 = minimize(objective, x0_ref, method='Nelder-Mead',
                        options={'maxiter': 3000, 'xatol': 1e-7, 'fatol': 1e-8})
        centers2 = np.clip(res2.x.reshape(n, 2), 1e-4, 1.0 - 1e-4)
        radii2, s2 = get_optimal_radii(centers2)
        if s2 > best_sum:
            best_sum = s2
            best_centers = centers2.copy()
            best_radii = radii2.copy()

    if best_centers is not None:
        # Recompute radii exactly on the final centers for consistency
        radii_final, _ = get_optimal_radii(best_centers)
        # Apply tiny shrinkage to guarantee strict validity against float tolerances
        radii_final = radii_final * 0.99995
        best_radii = radii_final
        best_sum = np.sum(radii_final)
        
    return best_centers, best_radii, float(best_sum)
