import numpy as np
from scipy.optimize import minimize
import math

N_CIRCLES = 26

def obj_func(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[2*N_CIRCLES:])

def cons_func(params):
    """Constraint function: boundary and non-overlap constraints (>= 0)."""
    n = N_CIRCLES
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    x, y = centers[:, 0], centers[:, 1]
    
    # Pre-allocate constraint array
    c = np.empty(n*4 + n*(n-1)//2)
    idx = 0
    
    # Boundary constraints
    c[idx:idx+n] = x - radii; idx += n
    c[idx:idx+n] = 1.0 - x - radii; idx += n
    c[idx:idx+n] = y - radii; idx += n
    c[idx:idx+n] = 1.0 - y - radii; idx += n
    
    # Overlap constraints: distance >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx*dx + dy*dy)
    
    i, j = np.tril_indices(n, -1)
    c[idx:] = dist[i, j] - (radii[i] + radii[j])
    return c

def make_init(seed):
    """Create an initial configuration based on seed."""
    n = N_CIRCLES
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    idx = 0
    pattern = seed % 4
    rows = [6, 5, 6, 5, 4]
    if pattern == 1: rows = [5, 6, 5, 6, 4]
    elif pattern == 2: rows = [6, 6, 5, 5, 4]
    elif pattern == 3: 
        centers = rng.uniform(0.2, 0.8, (n, 2))
        return np.concatenate([centers.ravel(), np.full(n, 0.01)])
        
    num_rows = len(rows)
    for r_idx, count in enumerate(rows):
        y = (r_idx + 0.5) / num_rows + rng.uniform(-0.02, 0.02)
        shift = (1.0 / count) * 0.5 if r_idx % 2 == 1 else 0.0
        for c_idx in range(count):
            x = (c_idx + 0.5) / count + shift + rng.uniform(-0.02, 0.02)
            if idx < n:
                centers[idx] = [np.clip(x, 0.05, 0.95), np.clip(y, 0.05, 0.95)]
                idx += 1
    while idx < n:
        centers[idx] = rng.uniform(0.2, 0.8, 2)
        idx += 1
    return np.concatenate([centers.ravel(), np.full(n, 0.01)])

def run_packing():
    n = N_CIRCLES
    best_centers = None
    best_radii = None
    best_sum = 0.0

    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    constraints = {'type': 'ineq', 'fun': cons_func}

    # Phase 1: Multiple restarts with structured initializations
    for seed in range(30):
        x0 = make_init(seed)
        x0[:2*n] = np.clip(x0[:2*n], 0.0, 1.0)
        x0[2*n:] = np.clip(x0[2*n:], 0.0, 0.5)
        
        res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, 
                       constraints=constraints, 
                       options={'maxiter': 10000, 'ftol': 1e-10, 'disp': False})
        
        if -res.fun > best_sum:
            best_sum = -res.fun
            best_centers = res.x[:2*n].reshape(n, 2).copy()
            best_radii = res.x[2*n:].copy()

    # Phase 2: Local refinement on the best found solution
    if best_centers is not None:
        for _ in range(5):
            x0 = np.concatenate([best_centers.ravel(), best_radii])
            x0 += np.random.randn(len(x0)) * 1e-4

            x0[2*n:] = np.clip(x0[2*n:], 0.0, 0.5)
            res = minimize(obj_func, x0, method='SLSQP', bounds=bounds,

            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x[:2*n].reshape(n, 2).copy()
                best_radii = res.x[2*n:].copy()

    # Fallback if optimization completely fails
    if best_centers is None:

        best_radii = np.full(n, 0.03)
        best_sum = np.sum(best_radii)

        r = best_radii[i]

    best_sum = np.sum(best_radii)
    return best_centers, best_radii, float(best_sum)
