import numpy as np
from scipy.optimize import minimize
import math

def compute_constraints(x, n):
    """Compute inequality constraints (must be >= 0)."""
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    # Boundary constraints
    cons = [
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
    ]
    
    # Overlap constraints using squared distances for smoothness
    diff = centers[:, None, :] - centers[None, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum = radii[:, None] + radii[None, :]
    
    # Only upper triangle to avoid duplicates
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cons.append(dist_sq[mask] - r_sum[mask]**2)
    
    return np.concatenate(cons)

def obj_func(x, n):
    """Objective: maximize sum of radii (minimize negative sum)."""
    return -np.sum(x[2*n:])

def make_valid(centers, radii, n):
    """Project an infeasible configuration to the feasible region by shrinking radii."""
    centers = np.clip(centers, 0.0, 1.0)
    for _ in range(300):
        changed = False
        # Boundary violations
        for i in range(n):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > max_r + 1e-12:
                radii[i] = max(0.0, max_r)
                changed = True
        # Overlap violations
        for i in range(n):
            for j in range(i+1, n):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    excess = radii[i] + radii[j] - d
                    radii[i] -= excess / 2.0
                    radii[j] -= excess / 2.0
                    changed = True
        if not changed:
            break
    return centers, radii

def init_packing(n, seed):
    """Generate a feasible initial configuration based on a perturbed hexagonal lattice."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    
    # 6-5-6-5-4 pattern sums to 26
    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    y_spacing = 1.0 / (len(row_counts) + 1)
    
    for r_idx, count in enumerate(row_counts):
        y = (r_idx + 1) * y_spacing
        x_spacing = 1.0 / (count + 1)
        shift = 0.5 * x_spacing if r_idx % 2 == 1 else 0.0
        for c_idx in range(count):
            x = (c_idx + 1) * x_spacing + shift
            centers[idx] = [x, y]
            idx += 1
            
    # Add controlled noise to break symmetry and escape trivial local minima
    centers += rng.uniform(-0.04, 0.04, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    # Start with small feasible radii
    radii = np.full(n, 0.025)
    return make_valid(centers, radii, n)

def run_single_opt(n, centers, radii, maxiter):
    """Run SLSQP optimization from a given starting point."""
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    cons = {'type': 'ineq', 'fun': compute_constraints, 'args': (n,)}
    
    res = minimize(
        obj_func, x0, args=(n,), method='SLSQP', bounds=bounds,
        constraints=cons, 
        options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False}
    )
    
    c_opt = res.x[:2*n].reshape(n, 2)
    r_opt = res.x[2*n:]
    
    # Ensure strict feasibility after optimization
    c_opt, r_opt = make_valid(c_opt, r_opt, n)
    return c_opt, r_opt, np.sum(r_opt)

def run_packing():
    """Main packing routine: multiple restarts + local refinement."""
    n = 26
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Strategy 1: Diverse hexagonal lattice restarts
    for i in range(12):
        c, r = init_packing(n, seed=i*17+3)
        c, r, s = run_single_opt(n, c, r, maxiter=12000)
        if s > best_sum:
            best_sum = s
            best_c = c.copy()
            best_r = r.copy()
            
    # Strategy 2: Perturb best solution to escape shallow local optima
    if best_c is not None:
        for i in range(8):
            rng = np.random.RandomState(i*23+5)
            c = best_c + rng.normal(0, 0.003, best_c.shape)
            c = np.clip(c, 0.02, 0.98)
            r = best_r + rng.normal(0, 0.001, 26)
            r = np.maximum(r, 0.001)
            c, r = make_valid(c, r, n)
            c, r, s = run_single_opt(n, c, r, maxiter=15000)
            if s > best_sum:
                best_sum = s
                best_c = c.copy()
                best_r = r.copy()
                
    # Fallback (should not be reached given the inits above)
    if best_c is None:
        best_c, best_r = init_packing(n, seed=0)
        best_sum = np.sum(best_r)
        
    return best_c, best_r, float(best_sum)
