To maximize the sum of radii for 26 circles in a unit square, we can decompose the problem into two stages:
1. **Radius Optimization for Fixed Centers**: Given any set of centers, finding the optimal radii that maximize the sum is a Linear Programming (LP) problem. The constraints are $r_i + r_j \leq d_{ij}$ (non-overlap) and $0 \leq r_i \leq \min(x_i, 1-x_i, y_i, 1-y_i)$ (boundary). This LP can be solved extremely fast using `scipy.optimize.linprog`.
2. **Center Optimization**: We optimize the centers to maximize the LP result. Since the LP solution is continuous but non-smooth (kinks occur when active constraints change), derivative-free methods like Nelder-Mead are highly effective. We run multiple restarts with varied initial configurations (hexagonal lattice and random perturbations) to escape local minima.

This approach decouples the difficult non-linear overlap constraints from the geometry optimizer, delegates them to an exact LP solver, and allows the center optimizer to efficiently explore the configuration space. We strictly avoid closures and lambdas as requested.

import numpy as np
from scipy.optimize import minimize, linprog

def lp_objective(x_flat, n, A_ub, rows, cols):
    """
    Computes the negative sum of optimal radii for given centers.
    Uses Linear Programming to exactly solve the radius expansion problem.
    """
    centers = np.clip(x_flat.reshape((n, 2)), 1e-4, 1.0 - 1e-4)
    x_c, y_c = centers[:, 0], centers[:, 1]
    
    # Maximum radius allowed by square boundaries
    ub = np.minimum(np.minimum(x_c, 1.0 - x_c), np.minimum(y_c, 1.0 - y_c))
    
    # Pairwise distances between centers
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[rows, cols]
    
    try:
        # Solve LP: min -sum(r) subject to r_i + r_j <= dist_ij and 0 <= r_i <= ub_i
        res = linprog(
            -np.ones(n), 
            A_ub=A_ub, 
            b_ub=b_ub, 
            bounds=[(0.0, max(u, 1e-9)) for u in ub], 
            method='highs'
        )
        if res.success:
            return -res.fun
    except Exception:
        pass
        
    # Fallback if LP fails (should be rare)
    return -np.sum(np.minimum(ub, 0.05))


def run_packing():
    n = 26
    
    # Precompute LP constraint matrix structure for pairwise non-overlap
    rows, cols = np.triu_indices(n, k=1)
    A_ub = np.zeros((len(rows), n))
    A_ub[np.arange(len(rows)), rows] = 1.0
    A_ub[np.arange(len(rows)), cols] = 1.0
    
    best_sum = 0.0
    best_centers = None
    
    # Multiple restarts to explore different regions of the configuration space
    n_restarts = 40
    
    for trial in range(n_restarts):
        np.random.seed(trial * 13 + 7)
        
        # Initialize centers: mix of structured hexagonal lattice and random placement
        if trial % 3 == 0:
            cx = np.zeros(n)
            cy = np.zeros(n)
            idx = 0
            y = 0.12
            row = 0
            while idx < n:
                shift = 0.06 if row % 2 == 1 else 0.0
                ncols = 5 if row % 2 == 0 else 4
                for col in range(ncols):
                    if idx < n:
                        cx[idx] = 0.1 + col * 0.16 + shift
                        cy[idx] = y
                        idx += 1
                y += 0.14
                row += 1
        else:
            cx = np.random.uniform(0.1, 0.9, n)
            cy = np.random.uniform(0.1, 0.9, n)
            
        x0 = np.concatenate([cx, cy])
        # Add controlled noise to break symmetry and avoid grid traps
        x0 += np.random.randn(len(x0)) * 0.035
        x0 = np.clip(x0, 0.05, 0.95)
        
        # Optimize centers using Nelder-Mead, which handles the non-smooth LP objective well
        res = minimize(
            lp_objective, 
            x0, 
            method='Nelder-Mead', 
            args=(n, A_ub, rows, cols),
            options={'maxiter': 2000, 'xatol': 1e-6, 'fatol': 1e-8}
        )
        
        curr_sum = -res.fun
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = np.clip(res.x.reshape((n, 2)), 1e-4, 1.0 - 1e-4)
            
    # Final exact LP solve to determine precise optimal radii for the best centers found
    if best_centers is not None:
        x_c, y_c = best_centers[:, 0], best_centers[:, 1]
        ub = np.minimum(np.minimum(x_c, 1.0 - x_c), np.minimum(y_c, 1.0 - y_c))
        diff = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        b_ub = dists[rows, cols]
        
        try:
            final_res = linprog(
                -np.ones(n), 
                A_ub=A_ub, 
                b_ub=b_ub, 
                bounds=[(0.0, max(u, 1e-9)) for u in ub], 
                method='highs'
            )
            if final_res.success:
                best_radii = final_res.x
                best_sum = -final_res.fun
            else:
                best_radii = np.minimum(ub, 0.05)
        except Exception:
            best_radii = np.minimum(ub, 0.05)
    else:
        # Ultimate fallback configuration
        best_centers = np.random.uniform(0.2, 0.8, (n, 2))
        best_radii = np.full(n, 0.02)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
