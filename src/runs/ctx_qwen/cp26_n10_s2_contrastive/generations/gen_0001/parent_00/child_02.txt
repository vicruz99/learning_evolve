import numpy as np
from scipy.optimize import minimize
import math

def objective(params, n):
    """Negative sum of radii (to be minimized)."""
    return -np.sum(params[2*n:])

def constraints(params, n):
    """
    Inequality constraints: g(params) >= 0
    - Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    - Non-overlap: dist_sq >= (r_i + r_j)^2
    """
    cx = params[:n]
    cy = params[n:2*n]
    r = params[2*n:]
    
    c = []
    # Boundary constraints (4 per circle)
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    # Overlap constraints (squared distance to avoid sqrt gradients)
    idx = np.triu_indices(n, k=1)
    dx = cx[idx[0]] - cx[idx[1]]
    dy = cy[idx[0]] - cy[idx[1]]
    dist_sq = dx*dx + dy*dy
    r_sum = r[idx[0]] + r[idx[1]]
    c.extend(dist_sq - r_sum*r_sum)
    
    return np.array(c)

def run_single_opt(x0, n, bounds, cons, max_iter=5000):
    """Run a single SLSQP optimization and return the result."""
    try:
        res = minimize(
            objective, x0, args=(n,),
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False}
        )
        return res
    except Exception:
        return None

def run_packing():

    cons = {'type': 'ineq', 'fun': constraints, 'args': (n,)}
    
    inits = []

    for spacing in [0.14, 0.15, 0.16, 0.17, 0.18]:

            col = 0
            while x_start + col*spacing + r_init <= 1.0 and len(centers) < n:
                centers.append([x_start + col*spacing, y])
                col += 1
            row += 1
        while len(centers) < n:
            centers.append([0.5, 0.5])
        cx, cy = zip(*centers[:n])
        inits.append(np.concatenate([list(cx), list(cy), np.full(n, r_init)]))

        cy = np.random.uniform(0.1, 0.9, n)

        cy[0:4] = [0.12, 0.12, 0.88, 0.88]
        r = np.full(n, 0.05)
        inits.append(np.concatenate([cx, cy, r]))

    # Run initial optimizations
    for x0 in inits:

    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
