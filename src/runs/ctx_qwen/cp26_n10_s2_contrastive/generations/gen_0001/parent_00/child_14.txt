def compute_feasible_radii(centers):
    """Compute maximum feasible radii for given centers to ensure a valid start."""
    n = len(centers)
    r = np.full(n, 0.5)
    # Iteratively shrink radii to satisfy all constraints
    for _ in range(200):
        changed = False
        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r[i] > max_r + 1e-12:
                r[i] = max(0.0, max_r)
                changed = True
        # Pairwise constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                if dist < r[i] + r[j] - 1e-12:
                    # Distribute the reduction proportionally
                    factor = dist / (r[i] + r[j] + 1e-12)
                    r[i] *= factor
                    r[j] *= factor
                    changed = True
        if not changed:
            break
    return np.maximum(r, 1e-6)

def generate_hex_init(rows_config, spacing, margin):
    """Generate centers for a hexagonal lattice pattern."""
    centers = []
    y = margin
    for idx, cnt in enumerate(rows_config):
        shift = spacing / 2.0 if idx % 2 == 1 else 0.0
        for k in range(cnt):
            x = margin + shift + k * spacing
            centers.append([x, y])
        y += spacing * math.sqrt(3) / 2.0
    return np.array(centers)

def objective_func(x, n):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(x[2 * n :])

def constraint_func(x, n):
    """Inequality constraints g(x) >= 0."""
    cx = x[:n]
    cy = x[n : 2 * n]
    r = x[2 * n :]
    c = []
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c.extend(cx - r)
    c.extend(1.0 - cx - r)
    c.extend(cy - r)
    c.extend(1.0 - cy - r)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2

        for j in range(i + 1, n):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            c.append(dx * dx + dy * dy - (r[i] + r[j]) ** 2)
            
    return np.array(c)

def repair_packing(centers, radii, n):
    """Ensure strict validity by minimally shrinking radii."""
    # Fix boundaries
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if radii[i] > max_r:
            radii[i] = max(0.0, max_r - 1e-9)
            
    # Fix overlaps iteratively

        max_overlap = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                overlap = (radii[i] + radii[j]) - dist
                if overlap > max_overlap:
                    max_overlap = overlap
                    
        if max_overlap < 1e-12:
            break
            
        # Scale down radii proportionally to remove overlap
        for i in range(n):
            radii[i] *= (1.0 - max_overlap / (radii[i] + 0.01) * 0.5)
            
    radii = np.maximum(radii, 1e-9)
    return centers, radii
