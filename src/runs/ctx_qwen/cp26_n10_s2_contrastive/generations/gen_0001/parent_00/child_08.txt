from scipy.optimize import minimize

N = 26

def obj(v):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2::3])

def cons(v):
    """Constraints: boundary and non-overlap."""
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # Using broadcasting for efficiency
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    rs = r[:, None] + r[None, :]
    o = dist2 - rs**2
    
    # Extract upper triangle to avoid duplicate constraints
    iu, ju = np.triu_indices(N, k=1)
    return np.concatenate([b, o[iu, ju]])

def get_hex_init(seed=0):

    rng = np.random.RandomState(seed)
    centers = []
    # Row counts summing to 26
    counts = [6, 5, 6, 5, 4]

    dy = 0.16 * np.sqrt(3)
    dx = 0.15

            x += dx
        y += dy
    centers = np.array(centers[:N])

def get_random_init(seed=0):

    cons_dict = {'type': 'ineq', 'fun': cons}
