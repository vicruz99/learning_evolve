import numpy as np
from scipy.optimize import minimize, linprog
from operator import itemgetter

def solve_radii_lp(centers):
    """
    Given fixed centers, solve LP to find radii that maximize sum(radii)
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    
    # Compute pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Build LP constraint matrix: r_i + r_j <= dist(i,j)
    i_idx, j_idx = np.triu_indices(n, k=1)
    A_ub = np.zeros((len(i_idx), n))
    A_ub[np.arange(len(i_idx)), i_idx] = 1
    A_ub[np.arange(len(i_idx)), j_idx] = 1
    b_ub = dists[i_idx, j_idx]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Boundary bounds: 0 <= r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    max_radii = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]),
                           np.minimum(centers[:, 1], 1 - centers[:, 1]))
    max_radii = np.maximum(max_radii, 0.0)
    bounds = list(zip(np.zeros(n), max_radii))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def objective(centers_flat):
    """Outer objective: negative sum of radii for given centers."""
    centers = centers_flat.reshape(-1, 2)
    centers = np.clip(centers, 0.0, 1.0)
    radii, s = solve_radii_lp(centers)
    return -s

def run_packing():
    """Main packing function. Returns (centers, radii, sum_radii)."""
    n = 26
    
    # Generate diverse initial configurations
    configs = []
    
    # 1. Hexagonal lattice pattern
    c_hex = np.zeros((n, 2))
    y, row, idx = 0.08, 0, 0
    while idx < n:
        x = 0.08 if row % 2 == 0 else 0.16
        col = 0
        while x <= 0.92 and idx < n:
            c_hex[idx] = [x, y]
            x += 0.17
            col += 1
            idx += 1
        y += 0.148
        row += 1
    configs.append(c_hex)
    
    # 2. Random initializations
    for seed in range(8):
        rng = np.random.RandomState(seed)
        configs.append(rng.uniform(0.1, 0.9, (n, 2)))
        
    candidates = []
    
    # Stage 1: Quick exploration with Nelder-Mead
    for c_init in configs:
        x0 = c_init.flatten()
        res = minimize(objective, x0, method='Nelder-Mead',
                       options={'maxiter': 8000, 'xatol': 1e-4, 'fatol': 1e-6})
        curr_c = np.clip(res.x.reshape(n, 2), 0, 1)
        curr_r, curr_s = solve_radii_lp(curr_c)
        candidates.append((curr_s, curr_c, curr_r))
        
    # Sort candidates by sum of radii (descending)
    candidates.sort(key=itemgetter(0), reverse=True)
    
    best_sum = candidates[0][0]
    best_centers = candidates[0][1].copy()
    best_radii = candidates[0][2].copy()
    
    # Stage 2: Refine top candidates with Powell
    for s, c, r in candidates[:5]:
        if s <= best_sum:
            break
            
        noise = np.random.randn(n, 2) * 0.005
        c_pert = np.clip(c + noise, 0.02, 0.98)
        
        res_ref = minimize(objective, c_pert.flatten(), method='Powell',
                           options={'maxiter': 15000, 'xtol': 1e-6, 'ftol': 1e-8})
        curr_c = np.clip(res_ref.x.reshape(n, 2), 0, 1)
        curr_r, curr_s = solve_radii_lp(curr_c)
        
        if curr_s > best_sum:
            best_sum = curr_s
            best_centers = curr_c.copy()
            best_radii = curr_r.copy()
            
    # Stage 3: High-precision final refinement
    for _ in range(3):
        noise = np.random.randn(n, 2) * 0.001
        c_pert = np.clip(best_centers + noise, 0.01, 0.99)
        res_final = minimize(objective, c_pert.flatten(), method='Powell',
                             options={'maxiter': 20000, 'xtol': 1e-7, 'ftol': 1e-9})
        curr_c = np.clip(res_final.x.reshape(n, 2), 0, 1)
        curr_r, curr_s = solve_radii_lp(curr_c)
        if curr_s > best_sum:
            best_sum = curr_s
            best_centers = curr_c.copy()
            best_radii = curr_r.copy()
            
    # Apply minimal shrinkage to guarantee strict validation compliance
    # against floating-point precision issues in the checker
    best_radii = best_radii * (1.0 - 1e-9)
    
    return best_centers, best_radii, float(np.sum(best_radii))
