import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    """
    Given fixed centers, solve LP to find radii that maximize sum(radii)
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Maximize sum(r) => Minimize -sum(r)
    
    # Vectorized distance computation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Upper triangle indices for pairwise constraints
    idx_i, idx_j = np.triu_indices(n, k=1)
    n_pairs = len(idx_i)
    
    A_ub = np.zeros((n_pairs, n))
    A_ub[np.arange(n_pairs), idx_i] = 1.0
    A_ub[np.arange(n_pairs), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    # Boundary constraints handled via variable bounds for efficiency
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if max_r < 0:
            max_r = 0.0
        bounds.append((0.0, max_r))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback in case of solver failure
    return np.full(n, 1e-6), 1e-6 * n

def objective_centers(centers_flat):
    """
    Objective function for optimizing centers.
    Returns negative sum of optimal radii.
    """
    n = len(centers_flat) // 2
    centers = centers_flat.reshape((n, 2))
    # Clip centers to strictly inside to prevent degenerate LP bounds
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, val = solve_radii_lp(centers)
    return -val

        initial_configs.append(c)
