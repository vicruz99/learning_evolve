import numpy as np
from scipy.optimize import linprog, minimize
import math


def solve_radii_lp(centers, n):
    """
    Given fixed centers, solve LP to find radii that maximize sum(radii)
    subject to non-overlap and boundary constraints.
    """
    c_obj = -np.ones(n)
    m = n * (n - 1) // 2 + 4 * n
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = math.hypot(dx, dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        b_ub[idx] = x; A_ub[idx, i] = 1.0; idx += 1
        b_ub[idx] = 1 - x; A_ub[idx, i] = 1.0; idx += 1
        b_ub[idx] = y; A_ub[idx, i] = 1.0; idx += 1
        b_ub[idx] = 1 - y; A_ub[idx, i] = 1.0; idx += 1
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    except Exception:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='interior-point')
        
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0


def repulsion_step(centers, radii, n, steps=100, damping=0.5):
    """
    Adjust centers to resolve overlaps and boundary violations using repulsive forces.
    """
    c = centers.copy()
    r = radii.copy()
    
    for step in range(steps):
        forces = np.zeros_like(c)
        max_viol = 0.0
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = math.hypot(dx, dy)
                if d < 1e-9:
                    dx, dy = 1e-4, 1e-4
                    d = 1.414e-4
                    
                overlap = r[i] + r[j] - d
                if overlap > 0:
                    max_viol = max(max_viol, overlap)
                    f = overlap / d
                    forces[i, 0] += dx * f
                    forces[i, 1] += dy * f
                    forces[j, 0] -= dx * f
                    forces[j, 1] -= dy * f
                    
        # Boundary repulsion
        for i in range(n):
            if c[i, 0] < r[i]: forces[i, 0] += (r[i] - c[i, 0])
            if c[i, 0] > 1 - r[i]: forces[i, 0] -= (c[i, 0] - (1 - r[i]))
            if c[i, 1] < r[i]: forces[i, 1] += (r[i] - c[i, 1])
            if c[i, 1] > 1 - r[i]: forces[i, 1] -= (c[i, 1] - (1 - r[i]))
            
        # Adaptive damping: start strong, settle gently
        current_damp = damping * (1.0 - step / steps) + 0.05
        c += forces * current_damp
        c = np.clip(c, 1e-7, 1 - 1e-7)
        
        if max_viol < 1e-8:
            break
    return c


def joint_optimize(centers, radii, n):
    """
    Final polish using SLSQP to simultaneously optimize centers and radii.
    """
    params = np.concatenate([centers.ravel(), radii])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    def obj(p):
        return -np.sum(p[2 * n:])
        
    def constr(p):
        c = p[:2 * n].reshape(n, 2)
        r = p[2 * n:]
        con = []
        for i in range(n):
            con.append(c[i, 0] - r[i])
            con.append(1.0 - c[i, 0] - r[i])
            con.append(c[i, 1] - r[i])
            con.append(1.0 - c[i, 1] - r[i])
        for i in range(n):
            for j in range(i + 1, n):
                d = math.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
                con.append(max(d, 1e-12) - r[i] - r[j])
        return np.array(con)
        
    try:
        res = minimize(obj, params, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constr},
                       options={'maxiter': 3000, 'ftol': 1e-14})
        return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
    except Exception:
        return centers, radii


def run_packing():
    """
    Main packing function. Returns (centers, radii, sum_radii).
    """
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Try multiple initial hexagonal grids with different densities/seeds
    for seed in range(8):
        np.random.seed(seed)
        
        # Generate hexagonal lattice
        centers = np.zeros((n, 2))
        r_start = 0.04 + seed * 0.005
        idx = 0
        row = 0
        while idx < n:
            y = r_start + row * r_start * math.sqrt(3)
            shift = (r_start * 0.5) if row % 2 == 1 else 0.0
            col = 0
            while True:
                x = r_start + shift + col * 2 * r_start
                if x + r_start > 1.0:
                    break
                centers[idx] = [x, y]
                idx += 1
                col += 1
            row += 1
            if y + r_start > 1.0 and idx < n:
                # Fill remaining circles in available space
                while idx < n:
                    centers[idx] = [0.5 + np.random.uniform(-0.2, 0.2), 
                                    0.5 + np.random.uniform(-0.2, 0.2)]
                    idx += 1
                break
                
        # Perturb initialization
        centers += np.random.randn(n, 2) * 0.015
        centers = np.clip(centers, 0.05, 0.95)
        
        # Alternating LP + Repulsion optimization
        for _ in range(60):
            radii, s = solve_radii_lp(centers, n)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()
                
            centers = repulsion_step(centers, radii, n, steps=40, damping=0.6)
            
    # Final joint optimization polish from the best found configuration
    if best_centers is not None:
        opt_centers, opt_radii = joint_optimize(best_centers, best_radii, n)
        final_sum = np.sum(opt_radii)
        
        if final_sum > best_sum:
            best_centers = opt_centers
            best_radii = opt_radii
            best_sum = final_sum

    # Final validation & numerical safety adjustment
    # Shrink radii slightly if they exactly touch boundaries to pass strict validation
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        margin = 1e-9
        max_r = min(x, 1 - x, y, 1 - y) - margin
        if r > max_r and max_r > 0:
            best_radii[i] = max_r
        elif max_r <= 0:
            best_centers[i] = [0.5, 0.5]
            best_radii[i] = 1e-12

    return best_centers, best_radii, float(best_sum)
