import numpy as np
from scipy.optimize import minimize, linprog
import math

def get_hex_initial(n, seed=0):
    """Generate a hexagonal lattice initialization."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.035)
    
    idx = 0
    row = 0
    spacing = 0.17
    margin = 0.08
    
    while idx < n:
        y = margin + row * spacing * math.sqrt(3) / 2
        shift = (row % 2) * spacing / 2
        x = margin + shift
        
        col = 0
        while x + spacing/2 <= 1.0 - margin and idx < n:
            centers[idx] = [x + col * spacing, y]
            idx += 1
            col += 1
        row += 1
        
    # Perturb slightly to break symmetry
    centers += rng.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    return centers, radii

def get_random_initial(n, seed=0):
    """Generate a valid random initialization."""
    rng = np.random.default_rng(seed)
    centers = rng.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, 0.03)
    return centers, radii

def compute_overlap_constraints_sq(centers, radii):
    """Compute squared overlap constraints: dist^2 - (r_i + r_j)^2"""
    n = len(radii)
    cons = np.zeros(n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            rs = radii[i] + radii[j]
            cons[idx] = dx * dx + dy * dy - rs * rs
            idx += 1
    return cons

def compute_boundary_constraints(centers, radii):
    """Compute boundary constraints: x-r, 1-x-r, y-r, 1-y-r"""
    n = len(radii)
    cons = np.zeros(4 * n)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        cons[4*i] = x - r
        cons[4*i + 1] = 1.0 - x - r
        cons[4*i + 2] = y - r
        cons[4*i + 3] = 1.0 - y - r
    return cons

def objective_func(params, n):
    """Negative sum of radii."""
    return -np.sum(params[2::3])

def constraints_func(params, n):
    """Combined constraints function for SLSQP."""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    return np.concatenate([
        compute_boundary_constraints(centers, radii),
        compute_overlap_constraints_sq(centers, radii)
    ])

def optimize_radii_lp(centers):
    """Given fixed centers, find optimal radii using Linear Programming."""
    n = len(centers)
    c_obj = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    # Boundary constraints as variable bounds: 0 <= r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        # Strictly inside to avoid numerical boundary issues
        max_r = max(0.0, max_r - 1e-10)
        bounds.append((0.0, max_r))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback to tiny radii if LP fails
    return np.full(n, 1e-6), 0.0

def run_single_optimization(centers, radii, max_iter=10000):
    """Run SLSQP optimization from given state."""
    n = len(radii)
    params = np.concatenate([centers.ravel(), radii])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    cons = {'type': 'ineq', 'fun': constraints_func, 'args': (n,)}
    
    try:
        res = minimize(
            objective_func,
            params,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False}
        )
        opt_centers = res.x[:2*n].reshape(n, 2)
        opt_radii = res.x[2*n:]
        return opt_centers, opt_radii
    except Exception:
        return centers, radii

def fix_and_validate(centers, radii):
    """Ensure strict validity and return cleaned arrays."""
    n = len(radii)
    centers = centers.copy()
    radii = radii.copy()
    
    # Enforce boundary constraints strictly
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        valid_r = min(x, 1.0 - x, y, 1.0 - y)
        if valid_r < r - 1e-12:
            radii[i] = max(0.0, valid_r)
            
    # Iterative overlap resolution if SLSQP left tiny violations
    for _ in range(50):
        max_viol = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    max_viol = max(max_viol, req - d)
        if max_viol < 1e-12:
            break
        # Proportionally reduce radii to satisfy constraints
        for i in range(n):
            radii[i] *= (1.0 - 0.01 * max_viol)
            
    return centers, radii
