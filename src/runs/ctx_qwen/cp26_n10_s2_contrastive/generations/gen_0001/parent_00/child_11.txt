import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

N = 26

# Precompute A_ub structure for LP: r_i + r_j <= d_ij
# This matrix is constant regardless of center positions
_PAIRS_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        _PAIRS_IDX.append((i, j))
        idx += 1

A_UB_STATIC = np.zeros((len(_PAIRS_IDX), N))
for k, (i, j) in enumerate(_PAIRS_IDX):
    A_UB_STATIC[k, i] = 1.0
    A_UB_STATIC[k, j] = 1.0


def solve_radii_lp(centers):
    """Given fixed centers, solve LP to find radii maximizing sum(radii)."""
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    max_r = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    max_r = np.maximum(max_r, 0.0)
    bounds = [(0.0, mr) for mr in max_r]

    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

    b_ub = np.array([dists[i, j] for i, j in _PAIRS_IDX])
    c_obj = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=A_UB_STATIC, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_UB_STATIC, b_ub=b_ub, bounds=bounds, method='interior-point')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-6), 1e-5


def objective_de(centers_flat):
    """Objective for Differential Evolution: negative max sum of radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 0.0, 1.0)
    _, sum_r = solve_radii_lp(centers)
    return -sum_r


def obj_joint(p):
    """Objective for joint optimization: negative sum of radii."""
    return -np.sum(p[2*N:])


def joint_constraints(p):
    """Inequality constraints for joint optimization (must be >= 0)."""
    cx = p[:N]
    cy = p[N:2*N]
    r = p[2*N:]
    cons = []
    for i in range(N):
        cons.append(cx[i] - r[i])
        cons.append(1.0 - cx[i] - r[i])
        cons.append(cy[i] - r[i])
        cons.append(1.0 - cy[i] - r[i])
    for i in range(N):
        for j in range(i + 1, N):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            d = np.sqrt(dx*dx + dy*dy)
            cons.append(d - r[i] - r[j])
    return np.array(cons)


def run_packing():
    best_centers = np.zeros((N, 2))
    best_radii = np.zeros(N)
    best_sum = 0.0

    bounds_de = [(0.0, 1.0)] * (2 * N)
    bounds_sl = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons_sl = {'type': 'ineq', 'fun': joint_constraints}

    # 1. Differential Evolution for global center placement
    try:
        res_de = differential_evolution(
            objective_de, bounds_de, maxiter=400, popsize=12,
            mutation=(0.5, 1.0), recombination=0.7, seed=42, polish=False
        )
        de_centers = np.clip(res_de.x.reshape(N, 2), 0.0, 1.0)
        de_radii, de_sum = solve_radii_lp(de_centers)
        if de_sum > best_sum:
            best_sum = de_sum
            best_centers = de_centers
            best_radii = de_radii
    except Exception:
        pass

    # 2. SLSQP refinement from DE result
    params = np.concatenate([best_centers.ravel(), best_radii])
    try:
        res_sl = minimize(
            obj_joint, params, method='SLSQP', bounds=bounds_sl,
            constraints=cons_sl, options={'maxiter': 8000, 'ftol': 1e-13}
        )
        if res_sl.success:
            r_sum = -res_sl.fun
            if r_sum > best_sum:
                best_sum = r_sum
                best_centers = res_sl.x[:2*N].reshape(N, 2)
                best_radii = res_sl.x[2*N:]
    except Exception:
        pass

    # 3. Structured hexagonal initializations + SLSQP
    for seed in range(6):
        np.random.seed(seed)
        centers_hex = []
        r_init = 0.085
        y = r_init + 0.02
        row = 0
        while len(centers_hex) < N:
            x_start = r_init + 0.02 + (row % 2) * r_init * 0.866
            col = 0
            while x_start + col * 2 * r_init <= 1.0 - r_init and len(centers_hex) < N:
                cx = x_start + col * 2 * r_init
                centers_hex.append([cx, y])
                col += 1
            y += r_init * 1.433
            row += 1
        centers_hex = np.array(centers_hex[:N])
        centers_hex += np.random.randn(N, 2) * 0.015
        centers_hex = np.clip(centers_hex, 0.05, 0.95)

        _, init_radii = solve_radii_lp(centers_hex)
        if np.sum(init_radii) < 1e-5:
            init_radii = np.full(N, 0.05)

        params_hex = np.concatenate([centers_hex.ravel(), init_radii])
        try:
            res_hex = minimize(
                obj_joint, params_hex, method='SLSQP', bounds=bounds_sl,
                constraints=cons_sl, options={'maxiter': 6000, 'ftol': 1e-13}
            )
            if res_hex.success:
                r_sum = -res_hex.fun
                if r_sum > best_sum:
                    best_sum = r_sum
                    best_centers = res_hex.x[:2*N].reshape(N, 2)
                    best_radii = res_hex.x[2*N:]
        except Exception:
            pass

    # Final validation and numerical fixing
    centers_fix = best_centers.copy()
    radii_fix = best_radii.copy()

    for i in range(N):
        x, y = centers_fix[i]
        r = radii_fix[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if max_r < r:
            radii_fix[i] = max(0.0, max_r - 1e-10)

    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers_fix[i, 0] - centers_fix[j, 0]
                dy = centers_fix[i, 1] - centers_fix[j, 1]
                d = np.sqrt(dx*dx + dy*dy)
                if d < radii_fix[i] + radii_fix[j] - 1e-10:
                    overlap = radii_fix[i] + radii_fix[j] - d
                    radii_fix[i] -= overlap / 2.0
                    radii_fix[j] -= overlap / 2.0
                    changed = True
        if not changed:
            break

    return centers_fix, radii_fix, float(np.sum(radii_fix))
