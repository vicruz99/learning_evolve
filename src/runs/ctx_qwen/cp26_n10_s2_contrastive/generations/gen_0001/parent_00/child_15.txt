import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers, n):
    """Given fixed centers, solve LP to find radii that maximize sum(radii)
    subject to non-overlap and boundary constraints."""
    A_ub = []
    b_ub = []
    
    # Precompute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.maximum(np.sum(diff ** 2, axis=2), 0.0))
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Boundary constraints as variable bounds: r_i <= min(x, 1-x, y, 1-y)
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if max_r < 0.0:
            max_r = 0.0
        bounds_r.append((0.0, max_r))
        
    try:
        # Maximize sum(r) <=> Minimize -sum(r)
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
        else:
            # Fallback if LP fails (rare, usually due to degenerate centers)
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0

def obj_func(params, n):
    """Objective: minimize negative sum of radii."""
    radii = params[2 * n :]
    return -np.sum(radii)

def cons_func(params, n):
    """Constraints: boundaries and pairwise non-overlap (squared form)."""
    centers = params[: 2 * n].reshape(n, 2)
    radii = params[2 * n :]
    c = []
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        c.append(x - r)
        c.append(1.0 - x - r)
        c.append(y - r)
        c.append(1.0 - y - r)
        
    # Overlap constraints: (x_i-x_j)^2 + (y_i-y_j)^2 - (r_i+r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            r_sum = radii[i] + radii[j]
            c.append(dx * dx + dy * dy - r_sum * r_sum)
            
    return np.array(c)

def run_packing():
    """Main packing function. Returns (centers, radii, sum_radii)."""
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Variable bounds
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Generate initial configurations
    configs = []
    
    # 1. Hexagonal lattice variations
    hex_row_counts = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [4, 6, 5, 6, 5],
        [6, 5, 5, 5, 5]
    ]
    
    for row_counts in hex_row_counts:
        pts = []
        r_init = 0.09
        y_step = r_init * np.sqrt(3)
        x_step = 2 * r_init
        y_curr = r_init + 0.015
        for row_idx, cnt in enumerate(row_counts):
            x_start = r_init + 0.015 + (row_idx % 2) * (x_step / 2)
            for k in range(cnt):
                x = x_start + k * x_step
                if x + r_init <= 1.0:
                    pts.append([x, y_curr])
            y_curr += y_step
        while len(pts) < n:
            pts.append([0.5, 0.5])
        configs.append(np.array(pts[:n]))
        
    # 2. Grid variations
    for spacing in [0.18, 0.195, 0.21]:
        pts = []
        margin = 0.05
        for i in range(6):
            for j in range(6):
                x = margin + i * spacing
                y = margin + j * spacing
                if x <= 1.0 - margin and y <= 1.0 - margin:
                    pts.append([x, y])
        configs.append(np.array(pts[:n]))
        
    # 3. Random perturbations of hex
    for seed in range(10):
        np.random.seed(seed * 7 + 3)
        base = configs[0].copy()
        noise = np.random.randn(n, 2) * 0.025
        perturbed = np.clip(base + noise, 0.03, 0.97)
        configs.append(perturbed)
        
    # Primary optimization loop
    for cfg in configs:
        # Initialize radii optimally for these centers
        init_radii, _ = solve_radii_lp(cfg, n)
        params0 = np.concatenate([cfg.ravel(), init_radii])
        
        try:
            res = minimize(
                obj_func, params0, args=(n,),
                method='SLSQP', bounds=bounds,
                constraints={'type': 'ineq', 'fun': cons_func, 'args': (n,)},
                options={'maxiter': 15000, 'ftol': 1e-12, 'disp': False}
            )
            
            # Always project to valid radii using LP to guarantee feasibility
            curr_centers = res.x[:2 * n].reshape(n, 2)
            safe_radii, safe_sum = solve_radii_lp(curr_centers, n)
            
            if safe_sum > best_sum:
                best_sum = safe_sum
                best_centers = curr_centers.copy()
                best_radii = safe_radii.copy()
        except Exception:
            continue
            
    # Refinement phase: perturb best solution to escape local minima
    if best_centers is not None:
        for trial in range(15):
            np.random.seed(trial * 13 + 7)
            centers_pert = best_centers + np.random.randn(n, 2) * 0.008
            centers_pert = np.clip(centers_pert, 0.02, 0.98)
            radii_pert = best_radii + np.random.randn(n) * 0.003
            radii_pert = np.clip(radii_pert, 0.001, 0.5)
            
            params_pert = np.concatenate([centers_pert.ravel(), radii_pert])
            try:
                res = minimize(
                    obj_func, params_pert, args=(n,),
                    method='SLSQP', bounds=bounds,
                    constraints={'type': 'ineq', 'fun': cons_func, 'args': (n,)},
                    options={'maxiter': 12000, 'ftol': 1e-12, 'disp': False}
                )
                curr_centers = res.x[:2 * n].reshape(n, 2)
                safe_radii, safe_sum = solve_radii_lp(curr_centers, n)
                if safe_sum > best_sum:
                    best_sum = safe_sum
                    best_centers = curr_centers.copy()
                    best_radii = safe_radii.copy()
            except Exception:
                continue
                
    # Fallback if everything fails (should not happen)
    if best_centers is None:
        best_centers = configs[0]
        best_radii, best_sum = solve_radii_lp(best_centers, n)
        
    return best_centers, best_radii, float(best_sum)
