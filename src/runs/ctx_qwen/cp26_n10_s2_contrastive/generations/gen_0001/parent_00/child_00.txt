import numpy as np
from scipy.optimize import minimize

def objective(params, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[2 * n:])

def constraints(params, n):
    """Inequality constraints: g(x) >= 0."""
    cx = params[:n]
    cy = params[n:2*n]
    r = params[2*n:]
    
    cons = []
    # Boundary constraints: circles must stay inside [0,1]x[0,1]
    cons.extend(cx - r)
    cons.extend(1.0 - cx - r)
    cons.extend(cy - r)
    cons.extend(1.0 - cy - r)
    
    # Non-overlap constraints using squared distances for smoothness
    for i in range(n):
        for j in range(i + 1, n):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            cons.append(dx * dx + dy * dy - (r[i] + r[j]) ** 2)
            
    return np.array(cons)

def make_hex(n, r, shift_x=0.0, shift_y=0.0):
    """Generate a hexagonal lattice initialization for n circles."""
    centers = []
    row_h = r * np.sqrt(3)
    col_w = 2.0 * r
    y = r + shift_y
    row = 0
    
    while len(centers) < n and row < 20:
        x = r + shift_x + (col_w / 2.0 if row % 2 == 1 else 0.0)
        while x <= 1.0 - r and len(centers) < n:
            centers.append([x, y])
            x += col_w
        y += row_h
        row += 1
        
    # Fallback to simple grid if hex lattice doesn't yield n points
    if len(centers) < n:
        step = 0.15
        for i in range(n - len(centers)):
            centers.append([0.1 + (i % 6) * step, 0.1 + (i // 6) * step])
            
    return np.array(centers[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles in a unit square."""
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds: x, y in [0, 1], r in [1e-5, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-5, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints, 'args': (n,)}
    
    # Generate diverse initial configurations
    trials = []
    for r in [0.05, 0.07, 0.09, 0.11, 0.13]:
        for sx in [0.0, 0.02, 0.05]:
            for sy in [0.0, 0.02, 0.05]:
                try:
                    c = make_hex(n, r, sx, sy)
                    trials.append((c, np.full(n, r)))
                except Exception:
                    pass
                    
    for _ in range(10):
        c = np.random.uniform(0.2, 0.8, (n, 2))
        trials.append((c, np.full(n, 0.04)))
        
    np.random.seed(42)
    
    # Phase 1: Broad search over configurations
    for _, (c_init, r_init) in enumerate(trials):
        x0 = np.concatenate([c_init.ravel(), r_init])
        # Add small perturbation to break symmetry
        x0[:2 * n] += np.random.randn(2 * n) * 0.005
        x0[:2 * n] = np.clip(x0[:2 * n], 1e-4, 1.0 - 1e-4)
        
        try:
            res = minimize(
                objective, x0, args=(n,), method='SLSQP', bounds=bounds,
                constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False}
            )
            cur_sum = -res.fun
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = res.x[:2 * n].reshape(n, 2)
                best_radii = res.x[2 * n:]
        except Exception:
            continue
            
    # Phase 2: Local refinement around the best found solution
    if best_centers is not None:
        for _ in range(15):
            cx = best_centers[:, 0] + np.random.randn(n) * 0.001
            cy = best_centers[:, 1] + np.random.randn(n) * 0.001
            r = best_radii + np.random.randn(n) * 0.0005
            r = np.clip(r, 1e-5, 0.5)
            cx = np.clip(cx, 1e-4, 1.0 - 1e-4)
            cy = np.clip(cy, 1e-4, 1.0 - 1e-4)
            
            x0 = np.concatenate([cx, cy, r])
            try:
                res = minimize(
                    objective, x0, args=(n,), method='SLSQP', bounds=bounds,

                )
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_centers = res.x[:2 * n].reshape(n, 2)
                    best_radii = res.x[2 * n:]
            except Exception:
                continue
                
    # Phase 3: Strict post-processing to guarantee validity
    margin = 1e-9
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        max_r_boundary = min(x, 1.0 - x, y, 1.0 - y) - margin
        if r > max_r_boundary:
            best_radii[i] = max(1e-9, max_r_boundary)

    for _ in range(100):
        overlap_found = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = best_centers[i, 0] - best_centers[j, 0]
                dy = best_centers[i, 1] - best_centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < best_radii[i] + best_radii[j] - 1e-12:
                    excess = best_radii[i] + best_radii[j] - dist
                    reduction = excess / 2.0 + 1e-9
                    best_radii[i] -= reduction
                    best_radii[j] -= reduction
                    best_radii[i] = max(best_radii[i], 1e-9)
                    best_radii[j] = max(best_radii[j], 1e-9)
                    overlap_found = True
        if not overlap_found:
            break
            
    best_radii = np.maximum(best_radii, 1e-9)
    return best_centers, best_radii, float(np.sum(best_radii))
