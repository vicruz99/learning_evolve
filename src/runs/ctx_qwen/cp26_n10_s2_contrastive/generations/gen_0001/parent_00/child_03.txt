import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to find radii that maximize sum(radii)."""
    n = centers.shape[0]
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Boundary limits for each circle
    r_max = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, max(0.0, r)) for r in r_max]
    
    # Pairwise constraints: r_i + r_j <= d_ij
    i_idx, j_idx = np.triu_indices(n, k=1)
    n_pairs = len(i_idx)
    A_ub = np.zeros((n_pairs, n))
    A_ub[np.arange(n_pairs), i_idx] = 1.0
    A_ub[np.arange(n_pairs), j_idx] = 1.0
    b_ub = dists[i_idx, j_idx]
    
    # Objective: max sum(r) => min -sum(r)
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    # Fallback for infeasible centers
    return np.full(n, 1e-7), n * 1e-7

def obj_centers(centers_flat):
    """Objective for center optimization: negative max sum of radii."""
    centers = centers_flat.reshape((26, 2))
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    _, s = solve_radii_lp(centers)
    return -s

def obj_full(params):
    """Objective for joint optimization: negative sum of radii."""
    return -np.sum(params[52:])

def cons_full(params):
    """Constraints for joint optimization: boundaries and non-overlap."""
    n = 26
    cx = params[0::3]
    cy = params[1::3]
    r = params[2::3]
    
    cons = []
    # Boundary: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    cons.extend(cx - r)
    cons.extend(1.0 - cx - r)
    cons.extend(cy - r)
    cons.extend(1.0 - cy - r)
    
    # Non-overlap: dist^2 >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    r_sum = r[i_idx] + r[j_idx]
    cons.extend((dx**2 + dy**2) - r_sum**2)
    
    return np.array(cons)
