import numpy as np
from scipy.optimize import minimize

def objective(params, n):
    """Negative sum of radii (minimizing this maximizes the sum)."""
    return -np.sum(params[2 * n :])

def constraints(params, n):
    """
    Returns all inequality constraint values g(x) >= 0.
    Includes boundary constraints and pairwise non-overlap constraints.
    """
    cx = params[0::3]
    cy = params[1::3]
    r = params[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    cons = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    # Vectorized distance matrix computation
    CX = cx[:, None]
    CY = cy[:, None]
    R = r[:, None]
    
    DX = CX - CX.T
    DY = CY - CY.T
    # Add small epsilon to avoid gradient singularity at dist=0
    dist = np.sqrt(DX**2 + DY**2 + 1e-15)
    r_sum = R + R.T
    
    # Extract upper triangle to avoid duplicates and self-comparisons
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cons_dist = dist[mask] - r_sum[mask]
    
    return np.concatenate([cons, cons_dist])

def get_hex_initial(n, seed):
    """Generate a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    r_init = 0.085
    cx, cy = [], []
    row = 0
    y = r_init
    
    while len(cx) < n:
        x_start = r_init + (row % 2) * r_init
        col = 0
        while x_start + col * 2 * r_init <= 1 - r_init and len(cx) < n:
            cx.append(x_start + col * 2 * r_init)
            cy.append(y)
            col += 1
        y += r_init * np.sqrt(3)
        row += 1
        
    cx = np.array(cx[:n])
    cy = np.array(cy[:n])
    
    # Apply random perturbation to break symmetry and avoid bad local minima
    cx += np.random.randn(n) * 0.01
    cy += np.random.randn(n) * 0.01
    cx = np.clip(cx, 0.02, 0.98)
    cy = np.clip(cy, 0.02, 0.98)
    r = np.full(n, r_init)
    return np.concatenate([cx, cy, r])

    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
