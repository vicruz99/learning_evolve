import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Precompute indices for pairwise constraints to save time
    idx = np.triu_indices(n, k=1)
    
    def obj(x):
        return -np.sum(x[2 * n:])
        
    def constr(x):
        cx = x[:n]
        cy = x[n:2*n]
        r = x[2*n:]
        
        # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
        c = []
        c.append(cx - r)
        c.append(1.0 - cx - r)
        c.append(cy - r)
        c.append(1.0 - cy - r)
        
        # Pairwise distance constraints: dist(i,j) >= r_i + r_j
        dx = cx[:, np.newaxis] - cx[np.newaxis, :]
        dy = cy[:, np.newaxis] - cy[np.newaxis, :]
        dists = np.hypot(dx, dy)
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        
        c.append(dists[idx] - r_sum[idx])
        
        return np.concatenate(c)
        
    cons = {'type': 'ineq', 'fun': constr}
    
    def try_optimize(x0):
        nonlocal best_sum, best_centers, best_radii
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if res.success or -res.fun > best_sum:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = np.column_stack((res.x[:n], res.x[n:2*n]))
                    best_radii = res.x[2*n:].copy()
        except Exception:
            pass
            
    # Strategy 1: Hexagonal lattice initializations with varying densities
    for trial in range(25):
        np.random.seed(trial)
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.02)
        
        r_init = 0.07 + trial * 0.008
        dy = r_init * np.sqrt(3)
        dx = 2.0 * r_init
        
        row = 0
        idx_pos = 0
        y = r_init + 0.01
        while idx_pos < n and y + r_init <= 0.99:
            x_start = r_init + 0.01 if row % 2 == 0 else r_init + r_init + 0.01
            col = 0
            while x_start + col * dx + r_init <= 0.99 and idx_pos < n:
                centers[idx_pos, 0] = x_start + col * dx
                centers[idx_pos, 1] = y
                idx_pos += 1
                col += 1
            y += dy
            row += 1
            
        while idx_pos < n:
            centers[idx_pos, 0] = np.random.uniform(0.1, 0.9)
            centers[idx_pos, 1] = np.random.uniform(0.1, 0.9)
            idx_pos += 1

        try_optimize(x0)

        step = 0.14
        idx_pos = 0
        for i in range(grid_size):

                if idx_pos < n:
                    centers[idx_pos, 0] = 0.05 + j * step
                    centers[idx_pos, 1] = 0.05 + i * step
                    idx_pos += 1

            centers[idx_pos] = [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]

        try_optimize(x0)

            x0 = np.concatenate([best_centers.ravel(), best_radii])

            r = best_radii[i]

        best_radii = np.full(n, 0.02)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
