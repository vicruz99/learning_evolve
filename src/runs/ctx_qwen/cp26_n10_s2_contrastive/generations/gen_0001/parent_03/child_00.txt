import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
# Precompute upper triangle indices for pairwise constraints
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def objective(x):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2::3])

def constraints(x):
    """
    Inequality constraints g(x) >= 0:
    - Boundary: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    - Overlap: dist(i,j) - r_i - r_j >= 0
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints (4 per circle)
    c_bound = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints (C(N,2) pairs)
    dx = cx[I_IDX] - cx[J_IDX]
    dy = cy[I_IDX] - cy[J_IDX]
    dist = np.hypot(dx, dy)
    c_overlap = dist - r[I_IDX] - r[J_IDX]
    
    return np.concatenate([c_bound, c_overlap])

def run_optimization(x0):
    """Run SLSQP optimization from a given initial state."""
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraints}
    
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        options={'maxiter': 15000, 'ftol': 1e-12, 'disp': False}
    )
    return res.x
