import numpy as np
from scipy.optimize import minimize
import math

N = 26

def objective(params):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(params[2 * N :])


def constraints(params):
    """
    Inequality constraints g(x) >= 0:
    - Circle boundaries within [0,1]x[0,1]
    - Pairwise non-overlap: distance >= r_i + r_j
    """
    centers = params[: 2 * N].reshape(N, 2)
    radii = params[2 * N :]
    
    # Boundary constraints
    c_bound = np.concatenate(
        [
            centers[:, 0] - radii,
            1.0 - centers[:, 0] - radii,
            centers[:, 1] - radii,
            1.0 - centers[:, 1] - radii,
        ]
    )
    
    # Pairwise distance constraints (vectorized)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Strictly lower triangle to avoid duplicates and self-comparison
    mask = np.tril(np.ones((N, N), dtype=bool), k=-1)
    c_dist = dists[mask] - r_sum[mask]
    
    return np.concatenate([c_bound, c_dist])


def get_initial_configurations():
    """Generate diverse initial configurations based on perturbed hex grids."""
    configs = []
    for seed in range(30):
        rng = np.random.default_rng(seed)
        
        # Vary grid spacing and margin to cover different packing densities
        spacing = 0.13 + (seed % 6) * 0.015
        margin = 0.04 + (seed % 4) * 0.02
        
        centers = []
        y = margin
        row = 0
        while y < 1.0 - margin and len(centers) < N + 10:
            x = margin + (row % 2) * spacing * 0.5
            while x < 1.0 - margin and len(centers) < N + 10:
                centers.append([x, y])
                x += spacing
            y += spacing * math.sqrt(3) / 2
            row += 1
            
        centers = np.array(centers[:N])
        # Add controlled noise to break symmetry and avoid degeneracy
        centers += rng.standard_normal((N, 2)) * 0.01
        centers = np.clip(centers, 0.02, 0.98)
        
        # Compute strictly feasible initial radii
        radii = np.zeros(N)
        for i in range(N):
            d_bound = min(centers[i, 0], 1.0 - centers[i, 0], 
                          centers[i, 1], 1.0 - centers[i, 1])
            d_min = np.min(np.sqrt(np.sum((centers[i] - centers) ** 2, axis=1)))
            # 0.85 factor ensures strict feasibility initially
            radii[i] = min(d_bound, d_min / 2.0) * 0.85
            
        configs.append(np.concatenate([centers.ravel(), radii]))
    return configs


def fix_violations(centers, radii):
    """Conservatively fix numerical drift while preserving solution quality."""
    n = len(radii)
    
    # Fix boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if r > max_r + 1e-9:
            radii[i] = max_r
            
    # Fix overlap violations iteratively
    for _ in range(100):
        max_violation = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                d = math.hypot(centers[i, 0] - centers[j, 0], 
                               centers[i, 1] - centers[j, 1])
                v = radii[i] + radii[j] - d
                if v > max_violation:
                    max_violation = v
                    
        if max_violation < 1e-11:
            break
            
        # Scale down radii just enough to resolve overlaps
        factor = 1.0 - max_violation / (2.0 * np.max(radii) + 1e-12)
        radii *= factor
        
    return centers, radii


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {"type": "ineq", "fun": constraints}
    
    best_sum = -1.0
    best_params = None
    
    configs = get_initial_configurations()
    
    for x0 in configs:
        # Tiny perturbation to avoid exact grid stagnation
        x0 += np.random.randn(len(x0)) * 1e-6
        
        res = minimize(
            objective, x0, method="SLSQP", bounds=bounds, constraints=cons,
            options={"maxiter": 10000, "ftol": 1e-14, "disp": False}
        )
        
        curr_sum = -res.fun
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_params = res.x.copy()

                x_ref = best_params + np.random.randn(len(best_params)) * 5e-5

                    best_params = res2.x.copy()

        centers = best_params[: 2 * N].reshape(N, 2)
        radii = best_params[2 * N :]
        
        centers, radii = fix_violations(centers, radii)
        return centers, radii, float(np.sum(radii))
    else:

        centers = np.random.rand(N, 2) * 0.6 + 0.2
        radii = np.full(N, 0.02)
        return centers, radii, float(np.sum(radii))
