from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2::3])

def constraints(x, idx):
    """
    Inequality constraints g(x) >= 0.
    - Pairwise distance >= sum of radii
    - Circle boundaries within [0,1]x[0,1]
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c_bound = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist(i,j) - (r_i + r_j) >= 0
    dx = cx[idx[0]] - cx[idx[1]]
    dy = cy[idx[0]] - cy[idx[1]]
    dist = np.hypot(dx, dy)
    c_olap = dist - (r[idx[0]] + r[idx[1]])
    
    return np.concatenate([c_bound, c_olap])

def make_feasible_radii(cx, cy):
    """Compute strictly feasible initial radii for given centers."""
    n = len(cx)
    r = np.full(n, 0.5)
    for i in range(n):
        # Max radius allowed by boundaries
        mx = min(cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
        r[i] = mx
        # Max radius allowed by other circles

            if i != j:
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                r[i] = min(r[i], d * 0.5)
    # Apply safety margin to ensure strict feasibility for optimizer
    return r * 0.85

    cy_base = np.array(cy_base[:N_CIRCLES])
