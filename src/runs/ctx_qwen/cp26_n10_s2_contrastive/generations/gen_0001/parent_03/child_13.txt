import numpy as np
from scipy.optimize import minimize
import math

N = 26

def objective(params):
    """Minimize negative sum of radii."""
    return -np.sum(params[2::3])


def make_constraints(params):
    """
    Constraints: boundary (circle inside square) and non-overlap (pairwise).
    Uses squared distance formulation for numerical stability.
    Returns array of values that must be >= 0.
    """
    cx = params[0::3]
    cy = params[1::3]
    r = params[2::3]

    # Boundary: x-r>=0, 1-x-r>=0, y-r>=0, 1-y-r>=0
    c_boundary = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r,
    ])

    # Overlap: dist^2 >= (r_i + r_j)^2  (equivalent to dist >= r_i + r_j for non-negative values)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist_sq = dx * dx + dy * dy
    r_sum_sq = (r[:, None] + r[None, :]) ** 2

    triu_idx = np.triu_indices(N, k=1)
    c_overlap = dist_sq[triu_idx] - r_sum_sq[triu_idx]

    return np.concatenate([c_boundary, c_overlap])


def hex_init(row_configs, seed=42):
    """
    Create hexagonal lattice initialization.
    row_configs: list of (count, offset_frac) tuples.

    """
    rng = np.random.RandomState(seed)
    n_rows = len(row_configs)
    y_spacing = 1.0 / (n_rows + 1)

    params = np.zeros(3 * N)
    idx = 0

    for row_idx, (count, offset_frac) in enumerate(row_configs):
        y = y_spacing * (row_idx + 1)

            if idx >= N:
                break
            x = x_spacing * (col + 1) + offset_frac * x_spacing * 0.5

            params[3 * idx] = np.clip(x, 0.03, 0.97)
            params[3 * idx + 1] = np.clip(y, 0.03, 0.97)

            idx += 1

        params[3 * idx + 1] = 0.5 + rng.uniform(-0.15, 0.15)
        params[3 * idx + 2] = 0.035

    return params

def fix_violations(params):
    """
    Post-process to ensure constraints are satisfied.
    Fix boundary by capping radii, fix overlaps by selective reduction.
    """
    cx = params[0::3].copy()
    cy = params[1::3].copy()
    r = params[2::3].copy()

    # Fix boundary violations
    for i in range(N):
        max_r = min(cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
        if r[i] > max_r - 1e-10:
            r[i] = max(0.0, max_r - 1e-10)

    # Iteratively fix overlap violations with selective reduction
    for _ in range(300):
        max_viol = 0.0
        worst_i, worst_j = 0, 1
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(cx[i] - cx[j], cy[i] - cy[j])
                overlap = r[i] + r[j] - d
                if overlap > max_viol:
                    max_viol = overlap
                    worst_i, worst_j = i, j

        if max_viol < 1e-12:
            break

        # Reduce the two offending circles
        reduction = max_viol * 0.6 + 1e-10
        r[worst_i] = max(0.0, r[worst_i] - reduction)
        r[worst_j] = max(0.0, r[worst_j] - reduction)

    new_params = np.zeros(3 * N)
    new_params[0::3] = cx
    new_params[1::3] = cy
    new_params[2::3] = r
    return new_params


def run_optimization(params, bounds, cons, max_iter=3000, ftol=1e-12):
    """Run a single SLSQP optimization."""
    try:
        res = minimize(
            objective,
            params,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': ftol, 'disp': False},
        )
        return res
    except Exception:
        return None
