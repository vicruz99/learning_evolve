import numpy as np
from scipy.optimize import minimize
import math

N = 26
I_UPPER, J_UPPER = np.triu_indices(N, k=1)

def objective(x):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2::3])

def constraints(x):
    """
    Inequality constraints:
    - Pairwise distance >= sum of radii
    - Circle boundaries within [0,1]x[0,1]
    Returns array of constraint values (must be >= 0)
    """

    cy = x[1::3]
    r = x[2::3]
    
    # Distance constraints between all pairs
    dx = cx[I_UPPER] - cx[J_UPPER]
    dy = cy[I_UPPER] - cy[J_UPPER]
    dists = np.hypot(dx, dy)
    c_dist = dists - (r[I_UPPER] + r[J_UPPER])
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c_bound = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    return np.concatenate([c_dist, c_bound])

def get_hex_centers(rows):
    """Generate centers for a hexagonal lattice with specified row counts."""
    centers = []
    r_init = 0.11
    dx = 2.0 * r_init
    dy = math.sqrt(3.0) * r_init
    y = r_init
    for row_idx, ncols in enumerate(rows):
        x_start = r_init + (dx / 2.0 if row_idx % 2 == 1 else 0.0)
        for c in range(ncols):
            centers.append([x_start + c * dx, y])
        y += dy
    return np.array(centers)

    min_dist = np.min(dists)
