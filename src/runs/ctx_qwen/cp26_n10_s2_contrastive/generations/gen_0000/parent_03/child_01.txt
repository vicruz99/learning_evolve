import numpy as np
from scipy.optimize import minimize
import math


def neg_sum_radii(params, n):
    """Objective: minimize negative sum of radii"""
    r = params[2::3]
    return -np.sum(r)


def constraint_x_min(params, i):
    """x_i - r_i >= 0"""
    return params[3*i] - params[3*i + 2]


def constraint_x_max(params, i):
    """1 - x_i - r_i >= 0"""
    return 1 - params[3*i] - params[3*i + 2]


def constraint_y_min(params, i):
    """y_i - r_i >= 0"""
    return params[3*i + 1] - params[3*i + 2]


def constraint_y_max(params, i):
    """1 - y_i - r_i >= 0"""
    return 1 - params[3*i + 1] - params[3*i + 2]


def constraint_overlap(params, i, j):
    """dist(i,j) - r_i - r_j >= 0"""
    dx = params[3*i] - params[3*j]
    dy = params[3*i + 1] - params[3*j + 1]
    dist_sq = dx*dx + dy*dy
    dist = math.sqrt(dist_sq + 1e-14)
    return dist - params[3*i + 2] - params[3*j + 2]


class ConstraintWrapper:
    """Wrapper to create callable constraint functions without lambdas"""
    def __init__(self, func, *args):
        self.func = func
        self.args = args
    
    def __call__(self, params):
        return self.func(params, *self.args)


def get_constraints(n):
    """Build all boundary and overlap constraints"""
    constraints = []

    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': ConstraintWrapper(constraint_x_min, i)})

        constraints.append({'type': 'ineq', 'fun': ConstraintWrapper(constraint_y_max, i)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': ConstraintWrapper(constraint_overlap, i, j)})
    
    return constraints


def get_bounds(n):

    return bounds

            x = x_start + col * 2 * r_init

                idx += 1

    for i in range(n):
        params[3*i] = centers[i, 0]
        params[3*i + 1] = centers[i, 1]
        params[3*i + 2] = r_init
    
    return params

def validate_and_fix(centers, radii, n):

                     centers[i, 1], 1 - centers[i, 1])
        if radii[i] > max_r:
            radii[i] = max(0.0, max_r)
    
    return centers, radii
