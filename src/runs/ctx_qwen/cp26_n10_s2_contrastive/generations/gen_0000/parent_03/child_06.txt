import numpy as np
from scipy.optimize import minimize

def compute_loss(params, n, mu):
    """
    Computes the objective function: negative sum of radii plus penalty for violations.
    """
    xs = params[0::3]
    ys = params[1::3]
    rs = params[2::3]
    
    # Primary objective: maximize sum of radii (minimize negative sum)
    loss = -np.sum(rs)
    
    # Penalty term accumulator
    pen = 0.0
    
    # Boundary constraints: circles must be inside [0,1]x[0,1]
    # r <= x, r <= 1-x, r <= y, r <= 1-y
    pen += np.sum(np.maximum(0.0, rs - xs)**2)
    pen += np.sum(np.maximum(0.0, rs - (1.0 - xs))**2)
    pen += np.sum(np.maximum(0.0, rs - ys)**2)
    pen += np.sum(np.maximum(0.0, rs - (1.0 - ys))**2)
    
    # Overlap constraints: distance(i,j) >= r_i + r_j
    # Compute pairwise distances using broadcasting
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    
    # Required distance sum
    r_sum = rs[:, None] + rs[None, :]
    
    # Amount of overlap (positive if overlapping)
    overlap = np.maximum(0.0, r_sum - dist)
    
    # Sum squared overlaps over the upper triangle to avoid double-counting and self-overlap
    pen += np.sum(np.triu(overlap**2, k=1))
    
    return loss + mu * pen
