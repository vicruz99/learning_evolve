import numpy as np
from scipy.optimize import minimize

def compute_overlap_penalty(centers, radii):
    """Compute squared overlap penalty for all circle pairs."""
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    min_dists = radii[:, None] + radii[None, :]
    np.fill_diagonal(min_dists, 0.0)
    overlaps = np.maximum(0.0, min_dists - dists)
    upper_tri = np.triu(overlaps, k=1)
    return np.sum(upper_tri**2)
