        x = params[0::3]
        y = params[1::3]

        # Equivalent to: r - x <= 0, x + r - 1 <= 0, r - y <= 0, y + r - 1 <= 0
        viol_x_min = np.maximum(0, r - x)
        viol_x_max = np.maximum(0, x + r - 1)
        viol_y_min = np.maximum(0, r - y)
        viol_y_max = np.maximum(0, y + r - 1)
        
        boundary_violations = np.sum(viol_x_min**2) + np.sum(viol_x_max**2) + \
                              np.sum(viol_y_min**2) + np.sum(viol_y_max**2)
        penalty += boundary_violations * penalty_weight

        # Overlap penalties
        # Distance between centers i and j must be >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # We compute violations: max(0, r_i + r_j - dist)^2

        # coords shape (n, 2)
        coords = np.column_stack((x, y))
        
        # Compute pairwise distances
        # diff shape (n, n, 2)
        diff = coords[:, np.newaxis, :] - coords[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
