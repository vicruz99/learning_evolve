import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # --- 1. Initialize Configuration ---
    # Generate a staggered (hexagonal) grid to provide a high-quality initial guess.
    # 5 rows with 5, 5, 5, 5, 6 circles = 26 circles.
    # We start with a radius of 0.08 to ensure a valid, non-overlapping start.
    centers = np.zeros((n, 2))
    r_start = 0.08
    
    # Vertical spacing for hexagonal packing
    h = r_start * np.sqrt(3)
    
    # Row configuration: 5, 5, 5, 5, 6
    row_counts = [5, 5, 5, 5, 6]
    idx = 0
    
    for i, count in enumerate(row_counts):
        # Calculate y coordinate for the row
        y = r_start + i * h
        
        # Calculate x coordinates (shift every other row by r_start for hex packing)
        x_start = r_start + (i % 2) * r_start
        
        # Calculate horizontal spacing to fit 'count' circles in width 1
        # Width required = 2*r + (count-1)*spacing. We want spacing approx 2*r.
        # To fit in [0,1]: 2*r + (count-1)*spacing = 1 => spacing = (1-2*r)/(count-1)
        spacing = (1 - 2 * r_start) / (count - 1)
        
        for j in range(count):
            x = x_start + j * spacing
            centers[idx] = [x, y]
            idx += 1
            
    # --- 2. Setup Optimization ---
    # Flatten centers and add radii
    # Initial radii: all 0.08
    radii_init = np.full(n, r_start)
    x0 = np.concatenate([centers.flatten(), radii_init])
    
    # Define bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])

    # --- 3. Define Constraints ---
    # Inequality constraints: fun(x) >= 0
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        xi, yi, ri = 3*i, 3*i + 1, 3*i + 2
        
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[xi] - v[ri]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[xi] - v[ri]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[yi] - v[ri]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[yi] - v[ri]})
        
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # Equivalent to: (x_i-x_j)^2 + (y_i-y_j)^2 - (r_i+r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = 3*i, 3*i + 1, 3*i + 2
            xj, yj, rj = 3*j, 3*j + 1, 3*j + 2
            
            def overlap_constraint(v, i=i, j=j):
                dist_sq = (v[xi] - v[xj])**2 + (v[yi] - v[yj])**2
                return dist_sq - (v[ri] + v[rj])**2
            
            constraints.append({'type': 'ineq', 'fun': overlap_constraint})

    # --- 4. Run Optimization ---
    # Objective: Minimize -sum(radii)
    def objective(v):
        return -np.sum(v[2::3]) # Radii are at indices 2, 5, 8...

    # Use trust-constr for robust handling of non-linear constraints
    # Note: We add a small regularization or rely on default Jacobian approximation
    # to handle the many constraints.
    result = minimize(objective, x0, method='trust-constr', bounds=bounds, 
                      constraints=constraints, options={'verbose': 0, 'maxiter': 200})
    
    # --- 5. Extract and Validate Results ---
    if result.success or result.status == 0:
        final_centers = result.x[:3*n].reshape((n, 2))
        final_radii = result.x[3*n:]
    else:
        # Fallback if optimization fails (return initial valid config)
        final_centers = centers
        final_radii = radii_init

    sum_radii = np.sum(final_radii)
    
    # Ensure final_radii are non-negative (clip for safety)
    final_radii = np.maximum(final_radii, 0)

    return final_centers, final_radii, float(sum_radii)
