2. **Sequential Inflation Method**: 
   - Start with small non-overlapping circles
   - Gradually increase all radii uniformly
   - After each increase, use gradient-based optimization to adjust positions to minimize overlaps
   - This avoids the optimizer getting stuck in bad local minima

3. **Final Optimization**: Once circles are grown as large as possible with the sequential method, use scipy.optimize.minimize with SLSQP for fine-tuning to maximize sum of radii with proper constraints.

4. **Multiple Restarts**: Try from different initial configurations (hexagonal, grid, perturbed) to escape local optima.

Key constraints to enforce:
- Each circle stays within [0,1]²
- No two circles overlap (distance between centers ≥ sum of radii)
- All radii non-negative

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26


def objective(vars):
    """Negative sum of radii (we minimize the negative to maximize)."""
    return -np.sum(vars[2::3])


def constraints(vars):
    """Inequality constraints: boundary and non-overlap."""
    n = N_CIRCLES
    c = []
    for i in range(n):
        x, y, r = vars[3 * i], vars[3 * i + 1], vars[3 * i + 2]
        c.append(x - r)
        c.append(1.0 - x - r)
        c.append(y - r)
        c.append(1.0 - y - r)
        c.append(r)

    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = vars[3 * i], vars[3 * i + 1], vars[3 * i + 2]
            xj, yj, rj = vars[3 * j], vars[3 * j + 1], vars[3 * j + 2]
            dist_sq = (xi - xj) ** 2 + (yi - yj) ** 2
            min_dist = ri + rj
            c.append(dist_sq - min_dist ** 2)
    return np.array(c)


def create_hex_config(n):
    """Create initial hexagonal lattice configuration."""
    centers = np.zeros((n, 2))
    idx = 0

    row_counts = [5, 5, 5, 5, 4, 2]

    h_spacing = 1.0 / 6.0
    v_spacing = np.sqrt(3) / 2 * h_spacing

    for row, count in enumerate(row_counts):
        y = v_spacing * row + 0.08
        x_offset = h_spacing / 2 if row % 2 == 1 else 0.0

        for col in range(count):
            if idx < n:
                x = x_offset + col * h_spacing + 0.06
                centers[idx] = [x, y]
                idx += 1

    return centers


def create_grid_config(n):
    """Create initial grid configuration."""
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(6):
        for col in range(5):
            if idx < n:
                centers[idx] = [0.12 + col * 0.17, 0.12 + row * 0.15]
                idx += 1
    return centers


def create_staggered_config(n):
    """Create staggered configuration with better spacing."""
    centers = np.zeros((n, 2))
    idx = 0

    row_configs = [
        (5, 0.0),
        (5, 0.075),
        (5, 0.15),
        (5, 0.225),
        (4, 0.30),
        (2, 0.375),
    ]

    for row_idx, (count, y_base) in enumerate(row_configs):
        y = y_base + row_idx * 0.13
        x_start = 0.08 + (0.04 if row_idx % 2 == 1 else 0.0)
        x_spacing = 0.16

        for col in range(count):
            if idx < n:
                centers[idx] = [x_start + col * x_spacing, y]
                idx += 1

    return centers


def relax_positions(centers, radii, n, iterations=200, dt=0.01):
    """Use force-directed relaxation to minimize overlaps."""
    centers = centers.copy()

    for _ in range(iterations):
        forces = np.zeros((n, 2))

        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2))
                if dist < 1e-10:
                    dist = 1e-10
                    diff = np.array([1e-10, 0.0])

                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    force_mag = 50.0 * overlap / dist
                    forces[i] += diff * force_mag
                    forces[j] -= diff * force_mag

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            margin = r * 0.5

            if x - r < margin:
                forces[i, 0] += 100.0 * (margin - (x - r))
            if x + r > 1.0 - margin:
                forces[i, 0] -= 100.0 * ((x + r) - (1.0 - margin))
            if y - r < margin:
                forces[i, 1] += 100.0 * (margin - (y - r))
            if y + r > 1.0 - margin:
                forces[i, 1] -= 100.0 * ((y + r) - (1.0 - margin))

        centers += forces * dt
        centers = np.clip(centers, 0.005, 0.995)

    return centers


def sequential_inflation(centers, radii, n):
    """Gradually inflate circles and relax positions."""
    centers = centers.copy()
    radii = radii.copy()

    for step in range(300):
        radii += 0.0008

        centers = relax_positions(centers, radii, n, iterations=30, dt=0.005)

        min_clearance = 1.0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                clearance = dist - radii[i] - radii[j]
                min_clearance = min(min_clearance, clearance)
            for coord in range(2):
                clearance = centers[i, coord] - radii[i]
                min_clearance = min(min_clearance, clearance)
                clearance = 1.0 - centers[i, coord] - radii[i]
                min_clearance = min(min_clearance, clearance)

        if min_clearance < -0.01:
            shrink = 1.0 + min_clearance / (np.mean(radii) + 0.001)
            radii *= max(0.95, shrink)

    return centers, radii


def pack_vars(centers, radii):
    """Pack centers and radii into optimization variable array."""
    n = centers.shape[0]
    vars = np.zeros(3 * n)
    for i in range(n):
        vars[3 * i] = centers[i, 0]
        vars[3 * i + 1] = centers[i, 1]
        vars[3 * i + 2] = radii[i]
    return vars


def unpack_result(vars, n):
    """Unpack optimization result into centers and radii."""
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]
    return centers, radii


def run_packing():
    """Main packing function."""
    n = N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraints}
    bounds = [(0.001, 0.999), (0.001, 0.999), (0.001, 0.499)] * n

    best_sum = 0.0
    best_centers = None
    best_radii = None

    configs = [
        create_hex_config(n),
        create_grid_config(n),
        create_staggered_config(n),
    ]

    for config_idx, centers_init in enumerate(configs):
        radii_init = np.ones(n) * 0.03

        centers_relaxed, radii_relaxed = sequential_inflation(
            centers_init, radii_init, n
        )

        current_sum = np.sum(radii_relaxed)

        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_relaxed.copy()
            best_radii = radii_relaxed.copy()

        vars0 = pack_vars(centers_relaxed, radii_relaxed)

        for restart in range(3):
            current_vars = vars0.copy()
            if restart > 0:
                noise = np.random.randn(3 * n) * 0.005
                current_vars = np.clip(current_vars + noise, 0.01, 0.99)
                current_vars[2::3] = np.clip(current_vars[2::3], 0.01, 0.4)

            try:
                result = minimize(
                    objective,
                    current_vars,
                    bounds=bounds,
                    constraints=cons,
                    method='SLSQP',
                    options={
                        'maxiter': 2000,
                        'ftol': 1e-14,
                        'disp': False,
                    },
                )

                if -result.fun > best_sum:
                    best_sum = -result.fun
                    best_centers, best_radii = unpack_result(result.x, n)
            except Exception:
                continue

        centers_init = best_centers.copy()
        radii_init = best_radii.copy()

    return best_centers, best_radii, best_sum
