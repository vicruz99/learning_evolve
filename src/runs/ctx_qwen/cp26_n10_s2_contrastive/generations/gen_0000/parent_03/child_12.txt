

Looking at this circle packing problem, I need to maximize the sum of radii of 26 circles in a unit square.

<strategy>
My approach combines several strategies:

1. **Initial Placement**: Use a hexagonal grid pattern (most space-efficient packing) to place 26 circles with reasonable initial radii.

2. **Iterative Expansion**: Greedily expand each circle's radius to its maximum possible (limited by boundaries and neighbors).

3. **Local Movement Optimization**: For each circle, try random perturbations to find positions where the circle can be larger.

4. **Multi-start Optimization**: Use scipy's SLSQP optimizer with multiple different starting points to escape local optima.

5. **Refinement Phase**: After optimization, apply additional expansion and movement passes.

Key insights:
- Hexagonal packing is optimal for equal circles, so it's a good starting point
- Circles can benefit from being near corners (fewer boundary constraints)
- The problem is non-convex with many local optima, requiring multi-start strategies
- The sum of radii target is ~2.636, suggesting average radius ~0.101 per circle

I'll implement a combination of analytical grid placement, greedy expansion, random perturbation search, and gradient-based optimization.

import numpy as np
from scipy.optimize import minimize
from itertools import combinations


def make_initial_grid(n, method="hex"):
    """Create initial grid placement for n circles."""
    centers = []
    
    if method == "hex":
        # Hexagonal grid with good spacing
        spacing = 0.14
        for row in range(8):
            for col in range(8):
                x = 0.12 + col * spacing + (row % 2) * spacing * 0.5
                y = 0.12 + row * spacing
                if x < 0.88 and y < 0.88:
                    centers.append([x, y])
                if len(centers) >= n:
                    break
            if len(centers) >= n:
                break
    
    if len(centers) < n:
        # Fallback: fill with random positions
        for _ in range(n - len(centers)):
            x = 0.2 + np.random.random() * 0.6
            y = 0.2 + np.random.random() * 0.6
            centers.append([x, y])
    
    return np.array(centers[:n])


def compute_max_radius(centers, radii, idx):
    """Compute the maximum radius for circle idx given current configuration."""
    x, y = centers[idx]
    r = radii[idx]
    
    max_r = min(x, 1 - x, y, 1 - y)
    for j in range(len(centers)):
        if j == idx:
            continue
        dx = x - centers[j][0]
        dy = y - centers[j][1]
        d = np.sqrt(dx * dx + dy * dy)
        max_r = min(max_r, d - radii[j])
    
    return max(0, max_r)


def expand_all_radii(centers, radii):
    """Expand each circle to its maximum possible radius."""
    n = len(centers)
    changed = False
    for i in range(n):
        new_r = compute_max_radius(centers, radii, i)
        if new_r > radii[i] + 1e-12:
            radii[i] = new_r
            changed = True
    return centers, radii, changed


def try_random_moves(centers, radii, n_moves=50, step=0.05):
    """Try random moves to improve circle positions and radii."""
    n = len(centers)
    centers = centers.copy()
    radii = radii.copy()
    improved = False
    
    for _ in range(n_moves):
        i = np.random.randint(n)
        x, y = centers[i]
        r = radii[i]
        
        for _ in range(30):
            dx = (np.random.random() - 0.5) * step
            dy = (np.random.random() - 0.5) * step
            nx = np.clip(x + dx, r, 1 - r)
            ny = np.clip(y + dy, r, 1 - r)
            
            # Check if this position allows larger radius
            new_max_r = min(nx, 1 - nx, ny, 1 - ny)
            for j in range(n):
                if j == i:
                    continue
                dx2 = nx - centers[j][0]
                dy2 = ny - centers[j][1]
                d = np.sqrt(dx2 * dx2 + dy2 * dy2)
                new_max_r = min(new_max_r, d - radii[j])
            
            if new_max_r > r + 1e-10:
                centers[i] = [nx, ny]
                radii[i] = new_max_r
                improved = True
                break
    
    return centers, radii, improved


def try_push_apart(centers, radii, n_pushes=100):
    """Push overlapping circles apart."""
    n = len(centers)
    centers = centers.copy()
    radii = radii.copy()
    
    for _ in range(n_pushes):
        # Find closest pair
        min_dist = float('inf')
        pair = (0, 1)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                d = np.sqrt(dx * dx + dy * dy)
                required = radii[i] + radii[j]
                if d < required and d < min_dist:
                    min_dist = d
                    pair = (i, j)
        
        i, j = pair
        dx = centers[i][0] - centers[j][0]
        dy = centers[i][1] - centers[j][1]
        d = np.sqrt(dx * dx + dy * dy)
        
        if d < radii[i] + radii[j]:
            # Push apart
            overlap = (radii[i] + radii[j] - d) / 2 + 0.001
            if d > 0:
                nx, ny = dx / d, dy / d
                centers[i][0] += nx * overlap
                centers[i][1] += ny * overlap
                centers[j][0] -= nx * overlap
                centers[j][1] -= ny * overlap
                # Clamp to bounds
                for k in [i, j]:
                    centers[k][0] = max(radii[k], min(1 - radii[k], centers[k][0]))
                    centers[k][1] = max(radii[k], min(1 - radii[k], centers[k][1]))


def objective_func(params, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[2::3])


def get_boundary_constraints(n):
    """Get boundary constraints for optimization."""
    constraints = []
    for i in range(n):
        # x >= r
        def c1(p, idx=i):
            return p[3 * idx] - p[3 * idx + 2]
        constraints.append({"type": "ineq", "fun": c1})
        
        # x <= 1 - r => x + r <= 1
        def c2(p, idx=i):
            return 1 - p[3 * idx] - p[3 * idx + 2]
        constraints.append({"type": "ineq", "fun": c2})
        
        # y >= r
        def c3(p, idx=i):
            return p[3 * idx + 1] - p[3 * idx + 2]
        constraints.append({"type": "ineq", "fun": c3})
        
        # y <= 1 - r => y + r <= 1
        def c4(p, idx=i):
            return 1 - p[3 * idx + 1] - p[3 * idx + 2]
        constraints.append({"type": "ineq", "fun": c4})
    
    return constraints


def get_overlap_constraints(n):
    """Get overlap constraints for optimization."""
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            def c_ov(p, i=i, j=j):
                xi, yi, ri = p[3 * i], p[3 * i + 1], p[3 * i + 2]
                xj, yj, rj = p[3 * j], p[3 * j + 1], p[3 * j + 2]
                dx = xi - xj
                dy = yi - yj
                return np.sqrt(dx * dx + dy * dy) - ri - rj
            constraints.append({"type": "ineq", "fun": c_ov})
    
    return constraints


def scipy_optimize(centers, radii, n, max_iter=300):
    """Run scipy SLSQP optimization."""
    params = np.zeros(3 * n)
    for i in range(n):
        params[3 * i] = centers[i][0]
        params[3 * i + 1] = centers[i][1]
        params[3 * i + 2] = radii[i]
    
    bounds = [(0.001, 0.999) for _ in range(2 * n)]
    bounds += [(0.001, 0.5) for _ in range(n)]
    
    constraints = get_boundary_constraints(n)
    constraints.extend(get_overlap_constraints(n))
    
    result = minimize(
        objective_func,
        params,
        args=(n,),
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-15}
    )
    
    new_centers = np.array([[result.x[3 * i], result.x[3 * i + 1]] for i in range(n)])
    new_radii = np.array([result.x[3 * i + 2] for i in range(n)])
    
    return new_centers, new_radii


def random_init(n):
    """Generate random initial configuration."""
    centers = []
    radii = []
    for _ in range(n):
        x = np.random.uniform(0.15, 0.85)
        y = np.random.uniform(0.15, 0.85)
        r = 0.05 + np.random.random() * 0.08
        centers.append([x, y])
        radii.append(r)
    return np.array(centers), np.array(radii)


def optimize_full(centers, radii, n=26, iterations=5000):
    """Full optimization: expand, move, optimize iteratively."""
    np.random.seed(12345)
    
    best_sum = -float('inf')
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for iteration in range(iterations):
        # Phase 1: Expand all radii
        centers, radii, exp_changed = expand_all_radii(centers, radii)
        
        # Phase 2: Try random moves
        centers, radii, moved = try_random_moves(centers, radii, n_moves=20, step=0.03)
        
        # Phase 3: Push overlapping circles apart
        try_push_apart(centers, radii, n_pushes=50)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
        
        # Phase 4: Occasionally run scipy optimizer from current position
        if iteration % 200 == 0 and iteration > 0:
            try:
                opt_centers, opt_radii = scipy_optimize(best_centers, best_radii, n, max_iter=100)
                opt_sum = np.sum(opt_radii)
                if opt_sum > best_sum + 1e-8:
                    best_sum = opt_sum
                    best_centers = opt_centers
                    best_radii = opt_radii
                    centers = opt_centers.copy()
                    radii = opt_radii.copy()
            except Exception:
                pass
    
    return best_centers, best_radii, best_sum
