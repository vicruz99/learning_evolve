import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)

    centers = np.zeros((n, 2))

            centers[idx] = [0.1 + c * 0.2, 0.1 + r * 0.2]
            idx += 1
    centers[25] = [0.5, 0.5]

    centers += np.random.uniform(-0.01, 0.01, size=centers.shape)
    
    # Clip to valid range for initial small radii
    centers = np.clip(centers, 0.05, 0.95)
    
    radii = np.full(n, 0.05) # Start with small radius

    max_steps = 5000
    growth_rate = 0.0001
    repulsion_strength = 100.0
    damping = 0.1

    for step in range(max_steps):
        # --- Growth Phase ---
        # Increase radii. Add slight noise to allow unequal sizes
        noise = np.random.uniform(0.95, 1.05, n)
        radii += growth_rate * noise
        
        # --- Repulsion Phase (Solver) ---
        # Run multiple sub-steps of relaxation per growth step
        for _ in range(20):

            # Circle-Circle Repulsion
            # Vectorized computation for performance
            diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # Shape (n, n, 2)
            dists = np.linalg.norm(diffs, axis=2) # Shape (n, n)
            dists = np.maximum(dists, 1e-9) # Avoid div by zero
            
            sums_radii = radii[:, np.newaxis] + radii[np.newaxis, :]
            overlaps = sums_radii - dists
            
            # Only compute forces for overlapping pairs
            # overlap > 0 means collision
            collision_mask = overlaps > 0
            # Set diagonal to 0 to ignore self-interaction
            np.fill_diagonal(collision_mask, False)
            
            # Force direction: push apart
            # Force magnitude proportional to overlap
            # Normalize direction vector
            safe_dists = np.where(collision_mask, dists, 1.0)
            directions = diffs / safe_dists[:, :, np.newaxis]
            
            # Apply force
            force_mag = overlaps * repulsion_strength
            # Add to force accumulator
            # We sum forces from all neighbors
            forces += np.sum(directions * force_mag[:, :, np.newaxis], axis=1)
            
            # Wall Repulsion (Boundary constraints)
            # If x - r < 0, push right. If x + r > 1, push left.
            # Left Wall
            left_overlap = radii - centers[:, 0]
            left_overlap = np.maximum(left_overlap, 0)
            forces[:, 0] += left_overlap * repulsion_strength
            
            # Right Wall
            right_overlap = centers[:, 0] + radii - 1
            right_overlap = np.maximum(right_overlap, 0)
            forces[:, 0] -= right_overlap * repulsion_strength
            
            # Bottom Wall
            bottom_overlap = radii - centers[:, 1]
            bottom_overlap = np.maximum(bottom_overlap, 0)
            forces[:, 1] += bottom_overlap * repulsion_strength
            
            # Top Wall
            top_overlap = centers[:, 1] + radii - 1
            top_overlap = np.maximum(top_overlap, 0)
            forces[:, 1] -= top_overlap * repulsion_strength
            
            # Update positions
            centers += damping * forces
            centers = np.clip(centers, 0.0, 1.0) # Hard clip
            
            # If overlaps are resolved, we can speed up or break? 
            # But we need to keep pushing to settle.
            
        # Slow down growth slightly as we approach max density
        if step > 2000:
            growth_rate *= 0.999
            damping *= 0.995
            
    # 4. Final Validation and Safety
    # Ensure no negative radii
    radii = np.maximum(radii, 0.0)
    
    # Hard clamp centers to valid range based on current radii
    # This ensures validity even if simulation drifted slightly
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
