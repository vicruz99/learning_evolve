import numpy as np
from scipy.optimize import minimize

def _objective(params, n):
    """
    Objective function to minimize: negative sum of radii plus penalty terms.
    """
    C = params[:2*n].reshape(n, 2)
    R = params[2*n:]
    
    # Boundary penalties: r <= x, r <= 1-x, r <= y, r <= 1-y
    b1 = np.maximum(0.0, R - C[:, 0])**2
    b2 = np.maximum(0.0, R - (1.0 - C[:, 0]))**2
    b3 = np.maximum(0.0, R - C[:, 1])**2
    b4 = np.maximum(0.0, R - (1.0 - C[:, 1]))**2
    bound_pen = np.sum(b1 + b2 + b3 + b4)
    
    # Pairwise non-overlap penalties: r_i + r_j <= ||c_i - c_j||
    diff = C[:, None, :] - C[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    
    overlap = np.maximum(0.0, R[:, None] + R[None, :] - dist)
    np.fill_diagonal(overlap, 0.0)
    overlap_pen = np.sum(overlap**2)
    
    # Penalty weight
    w = 5000.0
    
    # Maximize sum of radii -> minimize negative sum
    return -np.sum(R) + w * bound_pen + w * overlap_pen

def _get_initial_config(n, seed):
    """
    Generate an initial hexagonal-like configuration with random perturbation.
    """
    rng = np.random.RandomState(seed)
    pts = []
    row = 0
    # Hexagonal lattice spacing approx 0.14
    while len(pts) < n:
        col = 0
        while len(pts) < n:
            x = col * 0.14 + (0.07 if row % 2 == 1 else 0.0)
            y = row * 0.12
            if x < 1.0 and y < 1.0:
                pts.append([x, y])
            col += 1
        row += 1
    pts = pts[:n]
    
    # Add controlled noise
    C = np.array(pts) + rng.uniform(-0.02, 0.02, size=(n, 2))
    C = np.clip(C, 0.05, 0.95)
    R = np.full(n, 0.05)
    
    return np.concatenate([C.flatten(), R])

def run_packing():
    """
    Optimizes the positions and radii of 26 circles in a unit square
    to maximize the sum of radii while satisfying non-overlap and boundary constraints.
    """
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds for variables: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    # Multiple restarts to escape local minima
    for i in range(10):
        x0 = _get_initial_config(n, seed=i*123 + 42)
        
        try:
            res = minimize(_objective, x0, args=(n,), method='L-BFGS-B', bounds=bounds, 
                           options={'ftol': 1e-9, 'gtol': 1e-6, 'maxiter': 5000})
            
            curr_C = res.x[:2*n].reshape(n, 2)
            curr_R = res.x[2*n:]
            
            # Ensure radii are non-negative
            if np.any(curr_R < 0):
                continue
                
            # Boundary check
            if np.any(curr_R > curr_C[:, 0] + 1e-12) or np.any(curr_R > 1.0 - curr_C[:, 0] + 1e-12) or \
               np.any(curr_R > curr_C[:, 1] + 1e-12) or np.any(curr_R > 1.0 - curr_C[:, 1] + 1e-12):
                continue
                
            # Overlap check
            diff = curr_C[:, None, :] - curr_C[None, :, :]
            dist = np.sqrt(np.sum(diff**2, axis=2))
            overlaps = curr_R[:, None] + curr_R[None, :] - dist
            if np.any(np.triu(overlaps, 1) > 1e-6):
                continue
                
            s = np.sum(curr_R)
            if s > best_sum:
                best_sum = s
                best_centers = curr_C.copy()
                best_radii = curr_R.copy()
        except Exception:
            continue
            
    if best_centers is None:
        best_centers = np.random.rand(26, 2)
        best_radii = np.full(26, 0.01)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
