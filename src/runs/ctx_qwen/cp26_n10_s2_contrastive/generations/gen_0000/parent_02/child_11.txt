import numpy as np
from scipy.optimize import minimize

def smooth_max(t, eps=1e-6):
    """Smooth approximation of max(0, t) for differentiable optimization."""
    return (t + np.sqrt(t**2 + eps)) / 2.0

def objective(vars, n=26):
    """Objective function: maximize sum of radii with smooth penalty for violations."""
    centers = vars[:n*2].reshape(n, 2)
    radii = vars[n*2:]
    
    # Pairwise distances and radius sums
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap violations (upper triangle only)
    violations = r_sums - dists
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    violations = violations[mask]
    
    # Boundary violations
    b_viols = np.concatenate([
        radii - centers[:, 0],
        radii - centers[:, 1],
        radii - (1.0 - centers[:, 0]),
        radii - (1.0 - centers[:, 1])
    ])
    
    all_viols = np.concatenate([violations, b_viols])
    penalty = np.sum(smooth_max(all_viols)**2)
    
    # Maximize sum of radii <=> minimize negative sum + penalty
    return -np.sum(radii) + 5000.0 * penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize sum of radii."""
    n = 26
    best_obj = np.inf
    best_vars = None
    
    starts = []
    
    # Random initializations
    for seed in range(15):
        rng = np.random.default_rng(seed)
        centers = rng.uniform(0.15, 0.85, size=(n, 2))
        radii = np.full(n, 0.06)
        starts.append(np.concatenate([centers.flatten(), radii]))
        
    # Hexagonal grid initializations with jitter
    for seed in range(10):
        rng = np.random.default_rng(seed)
        xs = np.linspace(0.12, 0.88, 7)
        ys = np.linspace(0.12, 0.88, 4)
        centers = []
        for i, y in enumerate(ys):
            for j, x in enumerate(xs):
                if i % 2 == 1:
                    x += (xs[1] - xs[0]) / 2
                centers.append([x, y])
                if len(centers) == n:
                    break
            if len(centers) == n:
                break
        centers = np.array(centers) + rng.normal(0, 0.02, size=(n, 2))
        centers = np.clip(centers, 0.1, 0.9)
        radii = np.full(n, 0.07)
        starts.append(np.concatenate([centers.flatten(), radii]))
        
    bounds = [(1e-4, 1.0 - 1e-4)] * (n*2) + [(1e-4, 0.5)] * n
    
    for init_vars in starts:
        res = minimize(objective, init_vars, method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-14})
        if res.fun < best_obj:
            best_obj = res.fun
            best_vars = res.x.copy()
            
    centers = best_vars[:n*2].reshape(n, 2)
    radii = best_vars[n*2:]
    
    # Safety clipping
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    radii = np.clip(radii, 1e-5, None)
    
    # Compute maximum violation and shrink radii uniformly to guarantee feasibility
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    max_overlap = np.max(r_sums[mask] - dists[mask])
    
    b_max = np.max(np.concatenate([
        radii - centers[:, 0], radii - centers[:, 1],
        radii - (1.0 - centers[:, 0]), radii - (1.0 - centers[:, 1])
    ]))
    
    shrink = max(0.0, max_overlap, b_max) + 1e-6
    if shrink > 1e-7:
        radii -= shrink
        
    sum_r = float(np.sum(radii))
    return centers, radii, sum_r
