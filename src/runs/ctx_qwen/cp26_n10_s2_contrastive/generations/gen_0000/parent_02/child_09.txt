def resolve_overlaps_force(centers, radii, n, stiffness=50.0, steps=50, dt=0.01):
    """Resolve overlaps using force-based simulation"""
    for step in range(steps):
        forces = np.zeros((n, 2))
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist_sq = dx * dx + dy * dy
                dist = math.sqrt(dist_sq) if dist_sq > 1e-16 else 1e-8
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    force_mag = stiffness * overlap
                    fx = force_mag * dx / dist
                    fy = force_mag * dy / dist
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
            
            r = radii[i]
            if centers[i, 0] < r:
                forces[i, 0] += stiffness * 5 * (r - centers[i, 0])
            if centers[i, 0] > 1 - r:
                forces[i, 0] -= stiffness * 5 * (centers[i, 0] + r - 1)
            if centers[i, 1] < r:
                forces[i, 1] += stiffness * 5 * (r - centers[i, 1])
            if centers[i, 1] > 1 - r:
                forces[i, 1] -= stiffness * 5 * (centers[i, 1] + r - 1)
        
        centers += dt * forces
        
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r + 1e-10, min(1 - r - 1e-10, centers[i, 0]))
            centers[i, 1] = max(r + 1e-10, min(1 - r - 1e-10, centers[i, 1]))
    
    return centers
