import numpy as np
from scipy.optimize import minimize

def compute_objective(params, n, penalty_weight):
    """
    Computes the penalized objective value.
    Objective: maximize sum(radii) <=> minimize -sum(radii)
    Penalty adds squared violations for boundaries and overlaps.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    val = -np.sum(radii)
    penalty = 0.0
    
    # Boundary penalty: circles must stay inside [0,1]x[0,1]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        v = r - x
        if v > 0: penalty += v**2
        v = r + x - 1.0
        if v > 0: penalty += v**2
        v = r - y
        if v > 0: penalty += v**2
        v = r + y - 1.0
        if v > 0: penalty += v**2
        
    # Overlap penalty: distance between centers must be >= sum of radii
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i,0] - centers[j,0]
            dy = centers[i,1] - centers[j,1]
            dist = np.sqrt(dx*dx + dy*dy)
            v = radii[i] + radii[j] - dist
            if v > 0: penalty += v**2
            
    return val + penalty_weight * penalty

    # Optimize each configuration
    for cfg in configs:
        params = cfg.copy()
        # Annealing penalty weight to guide convergence
        for pw in [100.0, 500.0, 2000.0, 5000.0]:
            res = minimize(compute_objective, params, args=(n, pw), 
                           method='L-BFGS-B', bounds=bounds,

            params = res.x
            
        curr_r = params[2*n:]
        curr_sum = np.sum(curr_r)
        if curr_sum > best_sum:

            best_params = params
            
    centers = best_params[:2*n].reshape(n, 2)
    radii = best_params[2*n:]
    
    # Post-processing to guarantee strict validity
    radii = np.clip(radii, 0.0, 0.5)
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0-x, y, 1.0-y)
        if radii[i] > max_r:
            radii[i] = max_r

    for i in range(n):

                # Scale down proportionally to preserve relative sizes
                scale = dist / sum_r
                radii[i] *= scale
                radii[j] *= scale
                
    return centers, radii, np.sum(radii)
