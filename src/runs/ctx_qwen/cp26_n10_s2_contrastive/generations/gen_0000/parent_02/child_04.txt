def objective_function(centers_flat):
    """Wrapper for scipy minimize"""
    n = 26
    centers = centers_flat.reshape((n, 2))
    sum_radii, grad_sum = get_sum_radii_and_grad(centers)
    
    # We want to minimize -sum_radii
    obj_val = -sum_radii
    grad_obj = -grad_sum
    
    return obj_val, grad_obj
