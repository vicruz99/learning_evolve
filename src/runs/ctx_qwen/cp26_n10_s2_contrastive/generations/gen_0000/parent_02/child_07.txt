import numpy as np
from scipy.optimize import minimize
import math

def compute_constraints(centers, radii):
    n = centers.shape[0]
    cons = []
    # Boundary constraints: circle must stay inside [0,1]x[0,1]
    for i in range(n):
        cons.append(centers[i, 0] - radii[i])
        cons.append(1.0 - centers[i, 0] - radii[i])
        cons.append(centers[i, 1] - radii[i])
        cons.append(1.0 - centers[i, 1] - radii[i])
    # Overlap constraints: distance between centers must be >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.hypot(dx, dy)
            cons.append(dist - radii[i] - radii[j])
    return np.array(cons)

def objective_func(x):
    # Variables: first 52 are x,y coordinates, next 26 are radii
    return -np.sum(x[52:])

def constraint_func(x):
    n = 26
    c = x[:52].reshape(n, 2)
    r = x[52:]
    return compute_constraints(c, r)

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Bounds: x,y in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * 52 + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Run optimization from multiple starting configurations
    for trial in range(20):
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.02)
        
        # Hexagonal grid initialization
        idx = 0
        cols = int(np.ceil(np.sqrt(n)))
        rows = int(np.ceil(n / cols))
        
        y_spacing = 0.95 / (rows + 0.5)
        x_spacing = 0.95 / (cols + 0.5)
        
        for r in range(rows):
            y = 0.025 + r * y_spacing
            for c in range(cols):
                if idx >= n: break
                x = 0.025 + c * x_spacing
                if r % 2 == 1:
                    x += x_spacing / 2
                centers[idx] = [x, y]
                idx += 1
                
        # Add random perturbation to escape symmetry/local minima
        centers += np.random.uniform(-0.04, 0.04, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        
        x0 = np.concatenate([centers.flatten(), radii])
        
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-10, 'disp': False})
            
            if not np.isnan(res.fun) and -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x[:52].reshape(n, 2)
                best_radii = res.x[52:]
        except Exception:
            continue
            
    # Post-processing: repair any numerical boundary or overlap violations
    for _ in range(10):
        valid = True
        for i in range(n):
            x, y = best_centers[i]
            r = best_radii[i]
            if x < r: best_centers[i, 0] = r
            if x > 1-r: best_centers[i, 0] = 1-r
            if y < r: best_centers[i, 1] = r
            if y > 1-r: best_centers[i, 1] = 1-r
            
        for i in range(n):
            for j in range(i+1, n):
                dist = math.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                if dist < best_radii[i] + best_radii[j] - 1e-9:
                    overlap = best_radii[i] + best_radii[j] - dist
                    best_radii[i] -= overlap * 0.51
                    best_radii[j] -= overlap * 0.51
                    valid = False
        if valid:
            break
            
    # Ensure radii are non-negative
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, np.sum(best_radii)
