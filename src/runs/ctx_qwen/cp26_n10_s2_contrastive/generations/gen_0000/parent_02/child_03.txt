import numpy as np
from scipy.optimize import minimize, linprog


def objective(x, n):
    """Minimize negative sum of radii (equivalent to maximizing sum of radii)."""
    return -np.sum(x[2 * n:])


def boundary_constraint_x_min(x, i, n):
    """Constraint: x_i - r_i >= 0"""
    return x[2 * i] - x[2 * n + i]


def boundary_constraint_x_max(x, i, n):
    """Constraint: 1 - x_i - r_i >= 0"""
    return 1.0 - x[2 * i] - x[2 * n + i]


def boundary_constraint_y_min(x, i, n):
    """Constraint: y_i - r_i >= 0"""
    return x[2 * i + 1] - x[2 * n + i]


def boundary_constraint_y_max(x, i, n):
    """Constraint: 1 - y_i - r_i >= 0"""
    return 1.0 - x[2 * i + 1] - x[2 * n + i]


def overlap_constraint(x, i, j, n):
    """Constraint: dist(i,j) - r_i - r_j >= 0"""
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - x[2 * n + i] - x[2 * n + j]


def radius_nonneg(x, i, n):
    """Constraint: r_i >= 0"""
    return x[2 * n + i]


def create_hexagonal_init(n):
    """Create hexagonal grid initialization."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04

    idx = 0
    for row in range(8):
        for col in range(6):
            if idx >= n:
                break
            x = 0.08 + col * 0.16 + (row % 2) * 0.08
            y = 0.08 + row * 0.13
            if x <= 0.92 and y <= 0.92:
                centers[idx] = [x, y]
                idx += 1

    # Fill remaining if needed
    while idx < n:
        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.02
        idx += 1

    return centers, radii


def create_random_init(n, seed):
    """Create random initialization."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.15, 0.85, (n, 2))
    radii = rng.uniform(0.02, 0.06, n)
    return centers, radii


def optimize_radii_given_centers(centers):
    """Given fixed centers, find optimal radii via linear programming."""
    n = centers.shape[0]

    # Compute pairwise distances
    dist = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            if i < j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                dist[i, j] = d
                dist[j, i] = d

    # LP: maximize sum(r_i)
    # Equivalent to: minimize -sum(r_i)
    c = -np.ones(n)

    # Inequality constraints: A_ub @ r <= b_ub
    constraints_count = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((constraints_count, n))
    b_ub = np.zeros(constraints_count)

    row_idx = 0
    for i in range(n):
        # r_i <= x_i
        A_ub[row_idx, i] = 1
        b_ub[row_idx] = centers[i, 0]
        row_idx += 1

        # r_i <= 1 - x_i
        A_ub[row_idx, i] = 1
        b_ub[row_idx] = 1 - centers[i, 0]
        row_idx += 1

        # r_i <= y_i
        A_ub[row_idx, i] = 1
        b_ub[row_idx] = centers[i, 1]
        row_idx += 1

        # r_i <= 1 - y_i
        A_ub[row_idx, i] = 1
        b_ub[row_idx] = 1 - centers[i, 1]
        row_idx += 1

    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row_idx, i] = 1
            A_ub[row_idx, j] = 1
            b_ub[row_idx] = dist[i, j]
            row_idx += 1

    bounds = [(0, None)] * n

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success and np.all(result.x >= -1e-10):
        radii = np.maximum(result.x, 0)
        return radii, np.sum(radii)
    else:
        return np.ones(n) * 0.01, 0.26


def run_packing():
    """Main function to pack 26 circles in a unit square."""
    n = 26

    # Generate constraints for SLSQP
    cons = []

    for i in range(n):
        cons.append({'type': 'ineq', 'fun': boundary_constraint_x_min, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': boundary_constraint_x_max, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': boundary_constraint_y_min, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': boundary_constraint_y_max, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': radius_nonneg, 'args': (i, n)})

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': overlap_constraint, 'args': (i, j, n)})

    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Try multiple initializations
    for trial in range(8):
        if trial == 0:
            centers_init, radii_init = create_hexagonal_init(n)
        else:
            centers_init, radii_init = create_random_init(n, trial * 42 + 7)

        x0 = np.concatenate([centers_init.flatten(), radii_init])

        try:
            result = minimize(
                objective,
                x0,
                method='SLSQP',
                args=(n,),
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 10000, 'ftol': 1e-12, 'disp': False}
            )

            current_sum = -result.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = result.x[:2 * n].reshape((n, 2))
                best_radii = result.x[2 * n:]
        except Exception:
            continue

    # Fallback if no successful optimization
    if best_centers is None:
        best_centers, best_radii = create_hexagonal_init(n)
        best_sum = np.sum(best_radii)

    # Refinement phase: alternate between position optimization and radius maximization
    centers = best_centers.copy()
    radii = best_radii.copy()

    for iteration in range(20):
        # Step 1: Given current radii, optimize positions using penalty method
        def position_objective(x_pos):
            """Optimize positions to reduce constraint violations."""
            c = x_pos.reshape((n, 2))
            penalty = 0.0
            weight = 100.0

            # Boundary penalties
            for i in range(n):
                xi, yi = c[i]
                ri = radii[i]
                violation = max(0, ri - xi)
                penalty += weight * violation ** 2
                violation = max(0, xi + ri - 1.0)
                penalty += weight * violation ** 2
                violation = max(0, ri - yi)
                penalty += weight * violation ** 2
                violation = max(0, yi + ri - 1.0)
                penalty += weight * violation ** 2

            # Overlap penalties
            for i in range(n):
                for j in range(i + 1, n):
                    dx = c[i, 0] - c[j, 0]
                    dy = c[i, 1] - c[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    overlap = max(0, radii[i] + radii[j] - dist)
                    penalty += weight * overlap ** 2

            return penalty

        pos_bounds = [(0.01, 0.99)] * (2 * n)
        x_pos_0 = centers.flatten()

        result_pos = minimize(
            position_objective,
            x_pos_0,
            method='L-BFGS-B',
            bounds=pos_bounds,
            options={'maxiter': 500, 'ftol': 1e-15}
        )

        centers = result_pos.x.reshape((n, 2))

        # Step 2: Given positions, maximize radii via LP
        radii, sum_r = optimize_radii_given_centers(centers)

        if sum_r <= best_sum:
            radii = best_radii
            centers = best_centers
            break

        best_sum = sum_r
        best_centers = centers.copy()
        best_radii = radii.copy()

    # Final refinement with joint optimization
    x0_final = np.concatenate([best_centers.flatten(), best_radii])
    result_final = minimize(
        objective,
        x0_final,
        method='SLSQP',
        args=(n,),
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 10000, 'ftol': 1e-12, 'disp': False}
    )

    if -result_final.fun > best_sum:
        best_centers = result_final.x[:2 * n].reshape((n, 2))
        best_radii = result_final.x[2 * n:]
        best_sum = -result_final.fun

    # Clip any negative radii
    best_radii = np.maximum(best_radii, 0)

    # Ensure centers are within bounds
    best_centers = np.clip(best_centers, best_radii.reshape(-1, 1), 1 - best_radii.reshape(-1, 1))
    best_centers = np.clip(best_centers, 0, 1)

    final_sum = np.sum(best_radii)

    return best_centers, best_radii, final_sum
