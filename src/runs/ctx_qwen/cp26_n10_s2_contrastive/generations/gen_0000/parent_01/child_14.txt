import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(z):
    radii = z[2*N_CIRCLES:]
    return -np.sum(radii)

def get_constraints(z):
    n = N_CIRCLES
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    con_list = []
    con_list.append(centers[:, 0] - radii)
    con_list.append(1.0 - centers[:, 0] - radii)
    con_list.append(centers[:, 1] - radii)
    con_list.append(1.0 - centers[:, 1] - radii)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    con_list.append(dist_sq[mask] - r_sum[mask]**2)
            
    return np.concatenate(con_list)

def constraint_wrapper(z):
    return get_constraints(z)

def run_packing():
    n = N_CIRCLES
    best_z = None
    best_val = -np.inf
    
    pts = []
    dx = 0.2
    dy = dx * np.sqrt(3) / 2
    for i in range(12):
        for j in range(12):
            x = j * dx + (i % 2) * dx / 2
            y = i * dy
            if x <= 1.0 and y <= 1.0:
                pts.append([x, y])
    pts = np.array(pts[:n])
    
    init_radii = np.full(n, 0.07)
    bounds = [(0.0, 1.0)] * 2*n + [(0.0, 0.5)] * n
    
    for seed in range(15):
        np.random.seed(seed)
        z0 = np.concatenate([pts.reshape(-1), init_radii])
        perturbation = np.random.normal(0, 0.02, size=2*n)
        z0[:2*n] += perturbation
        z0[:2*n] = np.clip(z0[:2*n], 0.05, 0.95)
        
        res = minimize(objective, z0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constraint_wrapper},
                       options={'maxiter': 2000, 'ftol': 1e-10})
        
        if -res.fun > best_val:
            best_val = -res.fun
            best_z = res.x.copy()

    centers = best_z[:2*n].reshape(n, 2)
    radii = best_z[2*n:]
    return centers, radii, float(best_val)
