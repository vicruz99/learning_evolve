**Objective:** Maximize $\sum r_i$ (equivalently, minimize $-\sum r_i$).

**Constraints:**
1. **Boundary:** Each circle must lie within $[0,1]^2$: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$.
2. **Non-overlap:** For every pair $i < j$, the distance between centers must be at least the sum of radii: $\sqrt{(x_i-x_j)^2 + (y_i-y_j)^2} \ge r_i + r_j$.

import numpy as np
from scipy.optimize import minimize

def objective_func(v):
    # Maximize sum of radii => minimize negative sum
    # Radii are stored at indices 2, 5, 8, ... (every 3rd element starting from 2)
    return -np.sum(v[2::3])

def constraint_func(v):
    n = len(v) // 3
    res = []
    
    # Boundary constraints: circle i must be inside [0,1]x[0,1]
    for i in range(n):
        x, y, r = v[3*i], v[3*i+1], v[3*i+2]
        res.append(x - r)          # left boundary
        res.append(1.0 - x - r)    # right boundary
        res.append(y - r)          # bottom boundary
        res.append(1.0 - y - r)    # top boundary
        
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = v[3*i] - v[3*j]
            dy = v[3*i+1] - v[3*j+1]
            r_sum = v[3*i+2] + v[3*j+2]
            # Add small epsilon to avoid singularity at identical positions during optimization
            dist = np.sqrt(dx*dx + dy*dy + 1e-12)
            res.append(dist - r_sum)
            
    return np.array(res)

def generate_initial_state(n):
    v = np.zeros(3 * n)
    r_init = 0.095
    centers = []
    row = 0
    # Create a dense hexagonal lattice arrangement
    while len(centers) < n:
        cols = 6 if row % 2 == 0 else 5
        for c in range(cols):
            if len(centers) >= n:
                break
            x = (c + 0.5 + (0.5 if row % 2 != 0 else 0)) * 2 * r_init
            y = row * np.sqrt(3) * r_init
            centers.append((x, y))
        row += 1
        
    xs = [c[0] for c in centers]
    ys = [c[1] for c in centers]
    min_x, max_x = min(xs), max(xs)
    min_y, max_y = min(ys), max(ys)
    
    # Scale and center the configuration inside the unit square with padding
    pad = 0.02
    scale_x = (1.0 - 2.0 * pad) / (max_x - min_x)
    scale_y = (1.0 - 2.0 * pad) / (max_y - min_y)
    scale = min(scale_x, scale_y)
    
    for i, (cx, cy) in enumerate(centers):
        nx = pad + (cx - min_x) * scale
        ny = pad + (cy - min_y) * scale
        v[3*i] = nx
        v[3*i+1] = ny
        v[3*i+2] = r_init
    return v

def run_packing():
    n = 26
    v0 = generate_initial_state(n)
    
    # Bounds: x in [0,1], y in [0,1], r in [small_positive, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
        
    constraints = {'type': 'ineq', 'fun': constraint_func}
    
    # Run constrained non-linear optimization
    res = minimize(objective_func, v0, method='SLSQP', bounds=bounds, 
                   constraints=constraints, options={'maxiter': 8000, 'ftol': 1e-12})
    
    # Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = res.x[3*i : 3*i+2]
        radii[i] = res.x[3*i+2]
        
    # Post-optimization safety clip to strictly satisfy boundary conditions
    # (handles tiny floating-point drift)
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
        
    return centers, radii, np.sum(radii)
