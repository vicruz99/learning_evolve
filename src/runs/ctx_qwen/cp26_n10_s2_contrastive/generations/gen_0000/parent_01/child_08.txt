import numpy as np
from scipy.optimize import minimize

def compute_penalty(vars, n):
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    pen = 0.0
    for i in range(n):
        pen += max(0.0, r[i] - c[i,0])**2
        pen += max(0.0, r[i] - (1.0 - c[i,0]))**2
        pen += max(0.0, r[i] - c[i,1])**2
        pen += max(0.0, r[i] - (1.0 - c[i,1]))**2
    for i in range(n):
        for j in range(i+1, n):
            dx = c[i,0] - c[j,0]
            dy = c[i,1] - c[j,1]
            d = np.sqrt(dx*dx + dy*dy)
            pen += max(0.0, r[i] + r[j] - d)**2
    return pen

def objective(vars, n):
    r = vars[2*n:]
    pen = compute_penalty(vars, n)
    return -np.sum(r) + 1000.0 * pen

def get_initial_config(n, seed):
    np.random.seed(seed)
    r_init = 0.085
    centers = []
    for i in range(12):
        for j in range(12):
            x = j * 2 * r_init + (i % 2) * r_init + np.random.uniform(-0.03, 0.03)
            y = i * np.sqrt(3) * r_init + np.random.uniform(-0.03, 0.03)
            if -0.1 <= x <= 1.1 and -0.1 <= y <= 1.1:
                centers.append([x, y])
    
    centers = [c for c in centers if 0.0 <= c[0] <= 1.0 and 0.0 <= c[1] <= 1.0]
    while len(centers) < n:
        centers.append([np.random.rand(), np.random.rand()])
    centers = np.array(centers[:n])
    
    radii = np.zeros(n)
    for i in range(n):
        max_r = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(n):
            if i != j:
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                d = np.sqrt(dx*dx + dy*dy)
                if d < max_r * 2:
                    max_r = d / 2.0
        radii[i] = max(0.01, max_r)
        
    return np.concatenate([centers.flat, radii])

def run_packing():
    n = 26
    best_sum_r = -1.0
    best_vars = None
    
    for seed in range(20):
        x0 = get_initial_config(n, seed)
        bounds = [(0.0, 1.0)] * 2*n + [(0.0, 0.5)] * n
        
        try:
            res = minimize(objective, x0, args=(n,), method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 3000, 'ftol': 1e-9})
            
            pen = compute_penalty(res.x, n)
            if pen < 1e-3:
                current_sum = np.sum(res.x[2*n:])
                if current_sum > best_sum_r:
                    best_sum_r = current_sum
                    best_vars = res.x
        except Exception:
            pass
            
    if best_vars is not None:
        centers = best_vars[:2*n].reshape(n, 2)
        radii = best_vars[2*n:]
        
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
        for i in range(n):
            max_r = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            for j in range(i+1, n):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                d = np.sqrt(dx*dx + dy*dy)
                max_r = min(max_r, d / 2.0)
            radii[i] = max_r
            
        return centers, radii, np.sum(radii)
        
    # Fallback configuration
    fallback_centers = np.zeros((n, 2))
    fallback_radii = np.zeros(n)
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < n:
                fallback_centers[idx, 0] = j * 0.2 + 0.1
                fallback_centers[idx, 1] = i * 0.2 + 0.1
                fallback_radii[idx] = 0.05
                idx += 1
    if idx < n:
        fallback_centers[idx, 0] = 0.5
        fallback_centers[idx, 1] = 0.5
        fallback_radii[idx] = 0.04
    return fallback_centers, fallback_radii, np.sum(fallback_radii)
