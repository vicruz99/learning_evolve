import numpy as np
import scipy.optimize
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """

    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Helper function for constraints and objective
    def objective(x_flat):
        return -np.sum(x_flat[2::3])

    def constraints_gen(centers, radii):

        for i in range(n):
            # Boundary constraints
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i] - x[3*i+2]})       # x >= r
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[3*i] - x[3*i+2]}) # 1-x >= r
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+1] - x[3*i+2]})     # y >= r
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[3*i+1] - x[3*i+2]}) # 1-y >= r

        for i in range(n):
            for j in range(i + 1, n):
                def dist_constraint(x, i=i, j=j):
                    xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
                    xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
                    return np.sqrt((xi - xj)**2 + (yi - yj)**2) - ri - rj
                cons.append({'type': 'ineq', 'fun': dist_constraint})
        return cons

    # Initialization: Staggered Hexagonal Lattice
    def get_hex_init():
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.05)
        row_counts = [5, 6, 5, 6, 4]
        idx = 0
        for r_idx, count in enumerate(row_counts):
            y = 0.1 + r_idx * 0.2
            offset = 0.05 if r_idx % 2 == 1 else 0.0
            x_start = offset
            for c in range(count):
                if idx < n:
                    centers[idx, 0] = x_start + c * (2 * 0.05) + 0.05
                    centers[idx, 1] = y
                    idx += 1
        return centers, radii

    # Run optimization multiple times to find global optimum
    best_sum = 0.0
    for trial in range(10):

        if trial == 0:
            centers, radii = get_hex_init()

            centers, radii = get_hex_init()

            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]

                 [(0.0, 0.5) for _ in range(n)]

                cur_sum = -res.fun

    if best_centers is None:
        best_centers, best_radii = get_hex_init()
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
