import numpy as np
from scipy.optimize import minimize
import math

def get_hexagonal_initialization(n):

    Generates an initial configuration of n circles in a hexagonal pattern 
    inside the unit square.
    """
    centers = []
    radii = []
    
    # Estimate radius for initial packing to fit n circles roughly
    # Hexagonal density ~ 0.9069. Area per circle ~ 1/n.
    # pi * r^2 ~ 1/n / 0.9069 => r ~ sqrt(1/(pi * n * 0.9069))

    effective_h = 1 - 2 * margin

    # Horizontal spacing is diameter

    # Try r = 0.08 initially
    r = 0.08
    diam = 2 * r
    
    # Calculate rows needed for this r
    # height = 2r + (k-1)*sqrt(3)*r
    # k ~ 1 + (1 - 2r) / (sqrt(3)*r)

    # We need to place n circles.

    idx = 0
    y = r

        # Determine x start
        # Shift alternating rows
        if int((y - r) / (math.sqrt(3)*r)) % 2 == 1:
            x_start = r + diam # Shifted
        else:
            x_start = r
            
        x = x_start
        while idx < n and x + r <= 1:
            centers.append([x, y])
            radii.append(r)
            idx += 1
            x += diam
        y += math.sqrt(3) * r
        
    # If we didn't fit all (unlikely with small r), add remaining randomly
    while len(centers) < n:
        cx = np.random.uniform(r, 1-r)
        cy = np.random.uniform(r, 1-r)
        centers.append([cx, cy])
        radii.append(r)
        
    # If we fit more than n (shouldn't happen with loop), truncate
    centers = centers[:n]
    radii = radii[:n]
    
    return np.array(centers), np.array(radii)

def objective_function(vars, n):
    # vars contains [x1, y1, r1, x2, y2, r2, ...]
    # But it's easier to manage as separate arrays if possible, 
    # but optimizer passes 1D array.
    # Shape: 3 * n
    r = vars[2::3]
    return -np.sum(r) # Minimize negative sum

def constraint_boundary(vars, n):
    # x - r >= 0, 1 - x - r >= 0
    # y - r >= 0, 1 - y - r >= 0
    # 4 constraints per circle -> 4n constraints
    centers = vars.reshape(n, 3) # x, y, r
    # x >= r
    c1 = centers[:, 0] - centers[:, 2]
    # 1 - x >= r
    c2 = 1 - centers[:, 0] - centers[:, 2]
    # y >= r
    c3 = centers[:, 1] - centers[:, 2]
    # 1 - y >= r
    c4 = 1 - centers[:, 1] - centers[:, 2]
    return np.concatenate([c1, c2, c3, c4])

def constraint_overlap(vars, n):
    # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
    # dist^2 - (ri + rj)^2 >= 0
    centers = vars.reshape(n, 3) # x, y, r
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = centers[i, 2] + centers[j, 2]
            constraints.append(dist_sq - r_sum*r_sum)
    return np.array(constraints)

    # Constraints setup

    bounds = []
