For packing 26 circles in a unit square to maximize sum of radii, I'll use a multi-phase approach:

**Phase 1: LP-Based Radius Optimization**
Given fixed centers, finding optimal radii is a Linear Program:
- Maximize Σr_i
- Subject to: r_i ≤ distance to boundaries, r_i + r_j ≤ distance(i,j)

**Phase 2: Center Optimization**
Use Nelder-Mead local optimization on centers, where each function evaluation solves the LP for radii. This is non-convex but effective.

**Phase 3: Multiple Strategic Initializations**
1. Hexagonal grid patterns (6-5-6-5-4 rows) - mimicking optimal equal-circle packings
2. Random restarts to escape local optima
3. Perturbed versions of best solutions

**Phase 4: Fine-tuning**
- Use the constrained joint optimization (SLSQP) to refine the best LP-based solution
- Run multiple trials with different seeds

The key insight: hexagonal arrangements give the highest density for circle packings. By allowing varying radii and optimizing positions, we can beat the equal-circle packing bound.

Target: sum ≈ 2.636 (average radius ≈ 0.1014)

import numpy as np
from scipy.optimize import linprog, minimize
import math
import itertools


def solve_radius_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii.
    
    Constraints:
    - r_i <= min(x_i, 1-x_i, y_i, 1-y_i)  [boundary]
    - r_i + r_j <= ||c_i - c_j||           [non-overlap]
    - r_i >= 0
    """
    n = len(centers)
    c_obj = -np.ones(n)  # maximize sum(r_i) => minimize -sum
    
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        bound = min(x, 1 - x, y, 1 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(bound)
    
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dist)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x, -result.fun
    else:
        # Fallback
        r_fallback = 0.01
        return np.ones(n) * r_fallback, n * r_fallback


def neg_sum_radii(params, n=26):
    """Objective for center optimization: negative sum of optimal radii."""
    centers = params.reshape(n, 2)
    _, sum_r = solve_radius_lp(centers)
    return -sum_r


def create_hexagonal_config(row_specs):
    """
    Create centers from row specifications.
    row_specs: list of (y, x_start, x_step, count, offset_flag)
    """
    centers = []
    for spec in row_specs:
        y, x_start, x_step, count = spec[0], spec[1], spec[2], spec[3]
        offset = spec[4] if len(spec) > 4 else 0
        for col in range(count):
            x = x_start + offset + col * x_step
            centers.append([x, y])
    return np.array(centers)


def generate_hexagonal_init(n, num_rows, y_spacing, x_spacing, y_start=0.06, x_start=0.06):
    """Generate hexagonal grid initial configuration."""
    centers = []
    for row in range(num_rows):
        y = y_start + row * y_spacing
        offset = (x_spacing / 2) if row % 2 == 1 else 0
        x = x_start + offset
        while x <= 1 - x_start and len(centers) < n:
            if x >= 0 and x <= 1:
                centers.append([x, y])
            x += x_spacing
    # Fill remaining if needed
    while len(centers) < n:
        centers.append([0.5, 0.5])
    return np.array(centers[:n])


def optimize_from_init(centers_init, n=26, max_iter=15000, verbose=False):
    """Run Nelder-Mead optimization from an initial configuration."""
    try:
        res = minimize(neg_sum_radii, centers_init.flatten(), 
                      args=(n,),
                      method='Nelder-Mead',
                      options={
                          'maxiter': max_iter,
                          'xatol': 1e-10,
                          'fatol': 1e-10,
                          'adaptive': True
                      })
        centers_opt = res.x.reshape(n, 2)
        radii, sum_r = solve_radius_lp(centers_opt)
        return centers_opt, radii, sum_r, True
    except Exception as e:
        if verbose:
            print(f"Optimization failed: {e}")
        return centers_init, np.ones(n) * 0.01, 0.26, False

            best_centers = c_opt.copy()
            best_radii = r_opt.copy()

        for i in range(n):

        def joint_objective(p):
            radii = p[2::3]
            return -np.sum(radii)
        
        def joint_boundary_ineq(p):
            """Returns values that must be >= 0"""
            vals = []
            for i in range(n):
                x, y, r = p[3*i], p[3*i+1], p[3*i+2]
                vals.append(x - r)       # r <= x
                vals.append((1-x) - r)   # r <= 1-x
                vals.append(y - r)       # r <= y
                vals.append((1-y) - r)   # r <= 1-y
            return np.array(vals)
        
        def joint_separation_ineq(p):
            """Returns values that must be >= 0: dist^2 - (r_i+r_j)^2"""
            vals = []
            for i in range(n):
                for j in range(i + 1, n):
                    xi, yi, ri = p[3*i], p[3*i+1], p[3*i+2]

                    vals.append(dist_sq - min_dist_sq)
            return np.array(vals)

def validate_quick(centers, radii):
    """Quick validation without printing."""
    n = centers.shape[0]
    
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    
    for i in range(n):
        if radii[i] < -1e-12:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True
