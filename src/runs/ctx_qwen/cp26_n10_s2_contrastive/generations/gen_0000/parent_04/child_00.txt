import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    best_centers = None
    best_radii = None
    best_sum = -1.0

    # Collect diverse initializations
    initializations = []

    # Strategy 1: Hexagonal lattice with perturbation
    for seed in range(40):
        np.random.seed(seed)
        centers, radii = init_hexagonal(n)
        centers += np.random.randn(n, 2) * 0.015
        centers = np.clip(centers, 0.02, 0.98)
        initializations.append((centers, radii))

    # Strategy 2: Random sparse placement
    for seed in range(30):
        np.random.seed(seed)
        centers = np.random.rand(n, 2)
        radii = np.ones(n) * 0.015
        initializations.append((centers, radii))

    # Strategy 3: Grid-based with perturbation
    for seed in range(20):
        np.random.seed(seed)
        centers, radii = init_grid(n)
        centers += np.random.randn(n, 2) * 0.03
        centers = np.clip(centers, 0.02, 0.98)
        initializations.append((centers, radii))

    # Strategy 4: Corner-focused initialization
    for seed in range(15):
        np.random.seed(seed)
        centers, radii = init_corner_focused(n)
        centers += np.random.randn(n, 2) * 0.02
        centers = np.clip(centers, 0.02, 0.98)
        initializations.append((centers, radii))

    # Optimize each initialization
    for centers, radii in initializations:
        x0 = np.concatenate([centers.ravel(), radii])
        bounds = build_bounds(n)

        result = minimize(
            objective, x0, args=(n,), method="L-BFGS-B",
            bounds=bounds, options={"maxiter": 60000, "ftol": 1e-16, "gtol": 1e-12}
        )

        centers_opt = result.x[: 2 * n].reshape(n, 2)
        radii_opt = result.x[2 * n :]

        # Post-process for strict validity
        centers_opt, radii_opt = ensure_validity(centers_opt, radii_opt, n)

        s = np.sum(radii_opt)
        if s > best_sum:
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
            best_sum = s

    return best_centers, best_radii, best_sum


def build_bounds(n):
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))
    return bounds


def objective(x, n):
    centers = x[: 2 * n].reshape(n, 2)
    radii = x[2 * n :]

    obj = -np.sum(radii)
    penalty = 1e10

    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]

        # Boundary violations
        violation = max(0.0, ri - xi)
        obj += penalty * violation * violation
        violation = max(0.0, ri - (1.0 - xi))
        obj += penalty * violation * violation
        violation = max(0.0, ri - yi)
        obj += penalty * violation * violation
        violation = max(0.0, ri - (1.0 - yi))
        obj += penalty * violation * violation

    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(diff[0] * diff[0] + diff[1] * diff[1])
            overlap = radii[i] + radii[j] - dist
            if overlap > 0.0:
                obj += penalty * overlap * overlap

    return obj


def init_hexagonal(n):
    centers = []
    radii = []
    spacing = 0.16
    radius = 0.04

    row = 0
    while len(centers) < n:
        col = 0
        while len(centers) < n:
            x = 0.08 + col * spacing * 1.732 + (row % 2) * spacing * 0.866
            y = 0.08 + row * spacing * 1.5
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
                radii.append(radius)
            col += 1
        row += 1

    return np.array(centers[:n]), np.array(radii[:n])


def init_grid(n):
    grid = int(np.ceil(np.sqrt(n)))
    spacing = 1.0 / (grid + 1)
    centers = []
    radii = []
    for i in range(grid):
        for j in range(grid):
            if len(centers) < n:
                centers.append([(i + 1) * spacing, (j + 1) * spacing])
                radii.append(spacing * 0.28)
    return np.array(centers[:n]), np.array(radii[:n])


def init_corner_focused(n):
    centers = []
    radii = []

    # 4 corner circles
    corners = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
    for c in corners:
        centers.append(c)
        radii.append(0.08)

    # Edge circles
    for t in np.linspace(0.2, 0.8, 4):
        centers.append([t, 0.06])
        radii.append(0.05)
        centers.append([t, 0.94])
        radii.append(0.05)
        centers.append([0.06, t])
        radii.append(0.05)
        centers.append([0.94, t])
        radii.append(0.05)

    # Interior circles
    for r in range(3):
        for c in range(3):
            if len(centers) >= n:
                break
            centers.append([0.3 + c * 0.2, 0.3 + r * 0.2])
            radii.append(0.04)

    return np.array(centers[:n]), np.array(radii[:n])


def ensure_validity(centers, radii, n):
    """Post-process to guarantee all constraints are satisfied."""
    centers = centers.copy()
    radii = radii.copy()

    for _ in range(200):
        changed = False

        # Clamp to boundaries
        for i in range(n):
            max_r = min(
                centers[i, 0], 1.0 - centers[i, 0],
                centers[i, 1], 1.0 - centers[i, 1]
            )
            if radii[i] > max_r + 1e-12:
                radii[i] = max(max_r, 0.0)
                changed = True

        # Fix overlaps by reducing radii
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(diff[0] * diff[0] + diff[1] * diff[1])
                required = radii[i] + radii[j] - dist
                if required > 1e-12:
                    reduction = required / 2.0 + 1e-12
                    radii[i] = max(radii[i] - reduction, 0.0)
                    radii[j] = max(radii[j] - reduction, 0.0)
                    changed = True

        if not changed:
            break

    return centers, np.maximum(radii, 0.0)
