import numpy as np
from scipy.optimize import minimize
import math


N_CIRCLES = 26


def neg_sum_radii(x):
    """Objective: maximize sum of radii (negated for minimization)."""
    radii = x[2 * N_CIRCLES:]
    return -np.sum(radii)


def constr_x_lower(x, i):
    """Circle i must be inside left boundary: x_i >= r_i."""
    return x[2 * i] - x[2 * N_CIRCLES + i]


def constr_x_upper(x, i):
    """Circle i must be inside right boundary: x_i + r_i <= 1."""
    return 1 - x[2 * i] - x[2 * N_CIRCLES + i]


def constr_y_lower(x, i):
    """Circle i must be inside bottom boundary: y_i >= r_i."""
    return x[2 * i + 1] - x[2 * N_CIRCLES + i]


def constr_y_upper(x, i):
    """Circle i must be inside top boundary: y_i + r_i <= 1."""
    return 1 - x[2 * i + 1] - x[2 * N_CIRCLES + i]


def constr_nonoverlap(x, i, j):
    """Circles i and j must not overlap: dist(i,j) >= r_i + r_j."""
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist = math.sqrt(dx * dx + dy * dy)
    return dist - x[2 * N_CIRCLES + i] - x[2 * N_CIRCLES + j]


def make_hexagonal_init():
    """Create initial positions using hexagonal grid pattern."""
    centers = np.zeros((N_CIRCLES, 2))
    idx = 0

    r_est = 0.098
    sx = 2 * r_est
    sy = r_est * math.sqrt(3)

    row = 0
    while idx < N_CIRCLES:
        y = r_est + row * sy
        if y + r_est > 1.0:
            break
        offset = sx / 2 if row % 2 == 1 else 0
        col = 0
        while idx < N_CIRCLES:
            x = r_est + offset + col * sx
            if x + r_est <= 1.0:
                centers[idx] = [x, y]
                idx += 1
            else:
                break
            col += 1
        row += 1

    # Fill any remaining spots
    while idx < N_CIRCLES:
        centers[idx] = [0.5, 0.5 + 0.02 * (idx - N_CIRCLES + 1)]
        idx += 1

    return centers


def make_grid_init():

    idx = 0

    # 5x5 = 25 circles, plus 1 more

            idx += 1

    # 26th circle in center gap
    centers[idx] = [0.5, 0.5]

    return centers


def make_staggered_init():
    """Create initial positions with staggered rows."""
    centers = np.zeros((N_CIRCLES, 2))
    idx = 0

    r_est = 0.09
    # Row configuration: 5, 6, 5, 6, 4 = 26

    y_spacing = 1.0 / (len(row_counts) + 1)

        y = y_spacing * (row_idx + 1)
        x_spacing = 1.0 / (count + 1)
        for col in range(count):
            x = x_spacing * (col + 1)
            centers[idx] = [x, y]
            idx += 1

    return centers


def make_optimized_hexagonal_init():

        idx += 1

def build_constraints():
    """Build list of constraint dictionaries."""
    constraints = []

    # Boundary constraints for each circle
    for i in range(N_CIRCLES):
        constraints.append({'type': 'ineq', 'fun': constr_x_lower, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_x_upper, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_y_lower, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_y_upper, 'args': (i,)})

    # Non-overlap constraints for each pair
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            constraints.append({'type': 'ineq', 'fun': constr_nonoverlap, 'args': (i, j)})

    return constraints


def build_bounds():
    """Build bounds for optimization variables."""
    bounds = []
    for i in range(N_CIRCLES):
        # Center x and y in [0, 1]
        bounds.append((0, 1))
        bounds.append((0, 1))
    # Radii in [0, 0.5]
    for i in range(N_CIRCLES):
        bounds.append((0, 0.5))
    return bounds


def run_optimization(centers_init, r_init, constraints, bounds, maxiter=5000):
    """Run single optimization from given initialization."""
    radii_init = np.full(N_CIRCLES, r_init)
    x0 = np.concatenate([centers_init.flatten(), radii_init])

    result = minimize(
        neg_sum_radii,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'maxiter': maxiter,
            'ftol': 1e-15,
            'disp': False,
        }
    )

    centers_out = result.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii_out = result.x[2 * N_CIRCLES:]
    radii_out = np.maximum(radii_out, 0)  # Ensure non-negative

    return centers_out, radii_out, float(np.sum(radii_out)), result

            if centers[i, 1] + radii[i] > 1:

                    forces[j, 0] += fx
