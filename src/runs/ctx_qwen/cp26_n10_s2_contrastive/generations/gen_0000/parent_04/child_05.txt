import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Configuration for optimization
    n_restarts = 15
    penalty_weight = 1000.0
    
    for restart_idx in range(n_restarts):
        # 1. Initial Guess
        # Place centers randomly in [0.1, 0.9] and start with small valid radii
        np.random.seed(restart_idx * 42 + 123)
        centers = np.random.rand(n, 2) * 0.8 + 0.1
        radii = np.full(n, 0.02)
        
        # 2. Define Variables
        # Vector x = [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
        
        # 3. Define Bounds
        # x, y in [0, 1], r in [0, 0.5]
        bounds = []
        for i in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append(0.0, 0.5)) # r
        
        # 4. Objective Function with Penalty
        def objective(vars):
            # Extract centers and radii
            c_x = vars[0::3]
            c_y = vars[1::3]
            r = vars[2::3]
            
            # Primary objective: minimize negative sum of radii
            cost = -np.sum(r)
            
            # Penalty for pairwise overlaps
            # dist >= r_i + r_j  =>  (r_i + r_j - dist)^2 if violation
            # Using squared Euclidean distance for efficiency
            # dist_sq = (xi-xj)^2 + (yi-yj)^2
            # constraint: sqrt(dist_sq) >= ri + rj  <=> dist_sq >= (ri + rj)^2
            
            # Vectorized computation of distances and radii sums
            # Shape (n, n)
            # X matrix
            X = np.column_stack((c_x, c_y))
            # Compute pairwise squared distances
            # Using broadcasting: ||a - b||^2 = ||a||^2 + ||b||^2 - 2*a.b
            sq_norms = np.sum(X**2, axis=1)
            dist_sq = sq_norms[:, None] + sq_norms[None, :] - 2 * (X @ X.T)
            
            # Lower triangle indices to avoid duplication and self-distance
            i_indices, j_indices = np.tril_indices(n, -1)
            
            # Sum of radii for pairs
            r_sums = r[i_indices] + r[j_indices]
            r_sums_sq = r_sums**2
            
            # Violations: dist_sq < r_sums_sq
            violations = r_sums_sq - dist_sq[i_indices, j_indices]
            # Only positive violations count
            violations = np.maximum(0, violations)
            
            cost += penalty_weight * np.sum(violations**2)
            
            # Penalty for boundary constraints
            # x >= r  => r - x <= 0
            # x <= 1-r => x + r - 1 <= 0
            # same for y
            
            # Violations for x boundaries
            viol_x_left = np.maximum(0, r - c_x)
            viol_x_right = np.maximum(0, c_x + r - 1.0)
            
            # Violations for y boundaries
            viol_y_bottom = np.maximum(0, r - c_y)
            viol_y_top = np.maximum(0, c_y + r - 1.0)
            
            boundary_violations = viol_x_left + viol_x_right + viol_y_bottom + viol_y_top
            cost += penalty_weight * np.sum(boundary_violations**2)
            
            return cost

        # 5. Run Optimization
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            # 6. Extract Result
            opt_vars = res.x
            opt_centers = np.column_stack((opt_vars[0::3], opt_vars[1::3]))
            opt_radii = opt_vars[2::3]

            if np.any(opt_radii < 0) or np.any(np.isnan(opt_radii)) or np.any(np.isnan(opt_centers)):

            # Check boundaries

                r = opt_radii[i]
                if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
                    is_valid = False
                    break
            
            if not is_valid:
                continue

            # Check overlaps
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(np.sum((opt_centers[i] - opt_centers[j]) ** 2))
                    if dist < opt_radii[i] + opt_radii[j] - 1e-9:

                        break

                    break

                    best_radii = opt_radii.copy()

            continue

    # If no valid packing found (unlikely), return a fallback

                    idx += 1
        best_centers = fallback_centers
        best_radii = fallback_radii

    return best_centers, best_radii, best_sum
