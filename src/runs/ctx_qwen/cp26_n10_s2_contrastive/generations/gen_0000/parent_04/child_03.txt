The problem is to maximize the sum of radii of 26 circles packed in a unit square. This is a non-convex optimization problem with non-overlap and boundary constraints.

My strategy:

1. **Initialize with a hexagonal packing pattern** - Hexagonal packing is the densest arrangement for equal circles. I'll place 26 circles in a roughly hexagonal grid with a small initial radius to ensure feasibility.

2. **Use SLSQP optimization** with explicit constraints and analytical Jacobian:
   - Objective: maximize sum of radii (minimize negative sum)
   - Boundary constraints: each circle must stay within [0,1]²
   - Overlap constraints: distance between centers ≥ sum of radii

3. **Provide analytical Jacobian** for faster convergence - The constraint gradients are straightforward: ±1 for boundary constraints, directional derivatives for overlap constraints.

4. **Multi-start optimization** - Run from multiple initial configurations to avoid local optima, keeping the best result.

5. **Refinement passes** - After the main optimization, do additional passes to squeeze out more radius by re-running with the best solution as starting point.

The hexagonal initial arrangement uses ~6 rows with alternating 5 and 4 circles per row (offset by half-width for odd rows), which totals around 26 circles. This gives a good starting configuration that the optimizer can refine.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26


def objective(x):
    radii = x[2::3]
    return -np.sum(radii)


def constraints_func(x):
    n = N_CIRCLES
    n_constraints = 4 * n + n * (n - 1) // 2
    c = np.zeros(n_constraints)

    idx = 0
    for i in range(n):
        c[idx] = x[3 * i] - x[3 * i + 2]
        idx += 1
        c[idx] = 1.0 - x[3 * i] - x[3 * i + 2]
        idx += 1
        c[idx] = x[3 * i + 1] - x[3 * i + 2]
        idx += 1
        c[idx] = 1.0 - x[3 * i + 1] - x[3 * i + 2]
        idx += 1

    for i in range(n):
        for j in range(i + 1, n):
            dx = x[3 * i] - x[3 * j]
            dy = x[3 * i + 1] - x[3 * j + 1]
            dist = np.sqrt(dx * dx + dy * dy)
            c[idx] = dist - x[3 * i + 2] - x[3 * j + 2]
            idx += 1

    return c


def constraints_jac(x):
    n = N_CIRCLES
    n_constraints = 4 * n + n * (n - 1) // 2
    n_vars = 3 * n
    jac = np.zeros((n_constraints, n_vars))

    idx = 0
    for i in range(n):
        jac[idx, 3 * i] = 1.0
        jac[idx, 3 * i + 2] = -1.0
        idx += 1

        jac[idx, 3 * i] = -1.0
        jac[idx, 3 * i + 2] = -1.0
        idx += 1

        jac[idx, 3 * i + 1] = 1.0
        jac[idx, 3 * i + 2] = -1.0
        idx += 1

        jac[idx, 3 * i + 1] = -1.0
        jac[idx, 3 * i + 2] = -1.0
        idx += 1

    for i in range(n):
        for j in range(i + 1, n):
            dx = x[3 * i] - x[3 * j]
            dy = x[3 * i + 1] - x[3 * j + 1]
            dist = np.sqrt(dx * dx + dy * dy)

            if dist > 1e-10:
                jac[idx, 3 * i] = dx / dist
                jac[idx, 3 * i + 1] = dy / dist
                jac[idx, 3 * i + 2] = -1.0
                jac[idx, 3 * j] = -dx / dist
                jac[idx, 3 * j + 1] = -dy / dist
                jac[idx, 3 * j + 2] = -1.0

            idx += 1

    return jac


def create_hexagonal_init(r_init):
    n = N_CIRCLES
    x0 = np.zeros(3 * n)

    idx = 0
    row_height = r_init * np.sqrt(3)
    col_width = 2.0 * r_init

    row_configs = []
    # Try to fit 26 circles with alternating row lengths
    total = 0
    row = 0
    while total < n:
        if row % 2 == 0:
            ncols = min(5, n - total)
        else:
            ncols = min(4, n - total)
        row_configs.append(ncols)
        total += ncols
        row += 1

    row = 0
    for ncols in row_configs:
        offset_x = col_width / 2.0 if row % 2 == 1 else 0.0
        for col in range(ncols):
            x = r_init + offset_x + col * col_width
            y = r_init + row * row_height
            x0[3 * idx] = x
            x0[3 * idx + 1] = y
            x0[3 * idx + 2] = r_init
            idx += 1
        row += 1

    while idx < n:
        x0[3 * idx] = 0.5 + (idx % 3 - 1) * 0.15
        x0[3 * idx + 1] = 0.5 + (idx // 3 - 1) * 0.15
        x0[3 * idx + 2] = r_init
        idx += 1

    return x0


def create_corner_heavy_init():
    """Place larger circles near corners, smaller in middle"""
    n = N_CIRCLES
    x0 = np.zeros(3 * n)

    # 4 corner circles with larger radius
    corner_r = 0.22
    corners = [(corner_r, corner_r), (1 - corner_r, corner_r),
               (corner_r, 1 - corner_r), (1 - corner_r, 1 - corner_r)]
    for i, (cx, cy) in enumerate(corners):
        x0[3 * i] = cx
        x0[3 * i + 1] = cy
        x0[3 * i + 2] = corner_r

    # Edge circles
    edge_r = 0.12
    edge_configs = [
        (0.5, edge_r), (1 - edge_r, 0.5),
        (edge_r, 0.5), (0.5, 1 - edge_r)
    ]
    for i in range(4, 8):
        idx = i - 4
        x0[3 * i] = edge_configs[idx][0]
        x0[3 * i + 1] = edge_configs[idx][1]
        x0[3 * i + 2] = edge_r

    # Remaining circles in interior
    mid_r = 0.08
    for i in range(8, n):
        row = (i - 8) // 5
        col = (i - 8) % 5
        x0[3 * i] = 0.2 + col * 0.14
        x0[3 * i + 1] = 0.2 + row * 0.14
        x0[3 * i + 2] = mid_r

    return x0


def create_random_init():
    """Random feasible initialization"""
    n = N_CIRCLES
    x0 = np.zeros(3 * n)
    r = 0.06

    idx = 0
    row = 0
    col = 0
    spacing = 0.18

    while idx < n:
        x = r + col * spacing
        y = r + row * spacing
        if x <= 1 - r and y <= 1 - r:
            x0[3 * idx] = x
            x0[3 * idx + 1] = y
            x0[3 * idx + 2] = r
            idx += 1
        col += 1
        if col >= 6:
            col = 0
            row += 1

    return x0


def optimize_from_init(x0):
    n = N_CIRCLES
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    cons = {'type': 'ineq', 'fun': constraints_func, 'jac': constraints_jac}

    result = minimize(
        objective, x0, method='SLSQP', bounds=bounds,
        constraints=cons,
        options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
    )

    return result


def extract_result(x):
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = x[3 * i]
        centers[i, 1] = x[3 * i + 1]
        radii[i] = x[3 * i + 2]
    return centers, radii, np.sum(radii)


def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Try multiple initial configurations
    init_configs = [
        create_hexagonal_init(0.07),
        create_hexagonal_init(0.08),
        create_hexagonal_init(0.06),
        create_corner_heavy_init(),
        create_random_init(),
    ]

    for x0 in init_configs:
        result = optimize_from_init(x0)
        centers, radii, s = extract_result(result.x)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()

    # Refinement: use best solution as new starting point
    x0_refine = np.zeros(3 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x0_refine[3 * i] = best_centers[i, 0]
        x0_refine[3 * i + 1] = best_centers[i, 1]
        x0_refine[3 * i + 2] = best_radii[i]

    result2 = optimize_from_init(x0_refine)
    centers2, radii2, s2 = extract_result(result2.x)
    if s2 > best_sum:
        best_sum = s2
        best_centers = centers2
        best_radii = radii2

    # Final refinement
    x0_final = np.zeros(3 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x0_final[3 * i] = best_centers[i, 0]
        x0_final[3 * i + 1] = best_centers[i, 1]
        x0_final[3 * i + 2] = best_radii[i]

    result3 = optimize_from_init(x0_final)
    centers3, radii3, s3 = extract_result(result3.x)
    if s3 > best_sum:
        best_sum = s3
        best_centers = centers3
        best_radii = radii3

    return best_centers, best_radii, best_sum
