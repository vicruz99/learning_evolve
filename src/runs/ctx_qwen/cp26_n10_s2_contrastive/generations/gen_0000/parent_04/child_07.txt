import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    best_centers = None
    best_radii = None
    max_sum_radii = 0.0

    # Number of restarts to avoid local minima
    n_restarts = 20

    for attempt in range(n_restarts):
        # Initialize centers randomly
        # We try to keep centers away from boundaries initially to allow expansion
        np.random.seed(attempt)
        x = np.random.uniform(0.2, 0.8, n_circles)
        y = np.random.uniform(0.2, 0.8, n_circles)
        r = np.full(n_circles, 0.05) # Start with small radii

        # Flatten variables for optimizer: [x1, y1, r1, x2, y2, r2, ...]
        # But SLSQP handles bounds well, so we can structure variables as [x, y, r] arrays
        # However, scipy expects a 1D array.
        # Let's define variables as [x_0, ..., x_{n-1}, y_0, ..., y_{n-1}, r_0, ..., r_{n-1}]
        x0 = np.concatenate([x, y, r])

        # Bounds:
        # x, y in [0, 1]
        # r in [0, 0.5]
        bounds = [(0, 1)] * n_circles + [(0, 1)] * n_circles + [(0, 0.5)] * n_circles

        def objective(vars_flat):
            # vars_flat: [x, y, r] concatenated
            r_vals = vars_flat[2 * n_circles:]
            return -np.sum(r_vals) # Maximize sum of radii

        def constraints(vars_flat):
            constraints_list = []
            x_vals = vars_flat[:n_circles]
            y_vals = vars_flat[n_circles:2 * n_circles]
            r_vals = vars_flat[2 * n_circles:]

            # Boundary constraints
            # x - r >= 0  => r - x <= 0
            # x + r <= 1  => x + r - 1 <= 0
            # y - r >= 0  => r - y <= 0
            # y + r <= 1  => y + r - 1 <= 0
            
            # These are linear constraints, but we can just add them as inequality constraints >= 0
            # scipy constraints are usually func(x) >= 0
            
            # x - r >= 0
            constraints_list.extend(x_vals - r_vals)
            # 1 - (x + r) >= 0
            constraints_list.extend(1.0 - (x_vals + r_vals))
            # y - r >= 0
            constraints_list.extend(y_vals - r_vals)
            # 1 - (y + r) >= 0
            constraints_list.extend(1.0 - (y_vals + r_vals))

            # Non-overlap constraints
            # dist(i, j) >= r_i + r_j
            # dist^2 >= (r_i + r_j)^2
            # dist^2 - (r_i + r_j)^2 >= 0
            
            for i in range(n_circles):
                for j in range(i + 1, n_circles):
                    dx = x_vals[i] - x_vals[j]
                    dy = y_vals[i] - y_vals[j]
                    dist_sq = dx*dx + dy*dy
                    r_sum = r_vals[i] + r_vals[j]
                    constraints_list.append(dist_sq - r_sum**2)

            return np.array(constraints_list)

        # Define constraint structure for SLSQP
        # SLSQP expects a list of dictionaries or a single constraint dict with fun returning array
        constr = {
            'type': 'ineq',
            'fun': constraints
        }

        try:
            # Use SLSQP for constrained optimization
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constr,
                options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
            )

            if res.success:
                final_vars = res.x
                c_x = final_vars[:n_circles]
                c_y = final_vars[n_circles:2 * n_circles]
                c_r = final_vars[2 * n_circles:]
                
                current_sum = np.sum(c_r)
                if current_sum > max_sum_radii:
                    max_sum_radii = current_sum
                    best_centers = np.column_stack((c_x, c_y))
                    best_radii = c_r
        except Exception as e:
            # If optimization fails, skip this attempt
            continue

    # Fallback if optimization fails completely (should not happen with good seeds)
    if best_centers is None:
        # Simple grid fallback
        best_centers = np.zeros((n_circles, 2))
        best_radii = np.zeros(n_circles)
        idx = 0
        for i in range(5):
            for j in range(5):
                if idx < n_circles:
                    best_centers[idx] = [0.1 + j*0.2, 0.1 + i*0.2]
                    best_radii[idx] = 0.1
                    idx += 1
        # Fill remaining with small radii
        while idx < n_circles:
            best_centers[idx] = [0.5, 0.5]
            best_radii[idx] = 1e-6
            idx += 1
        max_sum_radii = np.sum(best_radii)

    # Final validation and adjustment for numerical stability
    # Ensure all circles are strictly inside and non-overlapping
    # We might need to shrink radii slightly if they are on the boundary
    centers = best_centers.copy()
    radii = best_radii.copy()

    # Check boundary constraints
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        # Shrink if too close to boundary
        margin = 1e-9
        max_r_boundary = min(x, 1-x, y, 1-y) - margin
        if r > max_r_boundary and max_r_boundary > 0:
            radii[i] = max_r_boundary
        elif max_r_boundary <= 0:
            # Circle is invalid position, move to center and set small radius
            centers[i] = [0.5, 0.5]
            radii[i] = 1e-9

    # Check overlap constraints
    # If overlap exists, reduce radii. 
    # A simple iterative reduction for overlapping pairs
    for _ in range(10): # Few passes
        overlap_found = False
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                sum_r = radii[i] + radii[j]
                if dist < sum_r - 1e-12:
                    # Reduce radii to satisfy constraint
                    # Scale down both proportionally or just reduce sum
                    excess = sum_r - dist
                    # Distribute excess reduction
                    reduction = excess / 2 + 1e-9
                    radii[i] -= reduction
                    radii[j] -= reduction
                    # Ensure non-negative
                    radii[i] = max(radii[i], 1e-9)
                    radii[j] = max(radii[j], 1e-9)
                    overlap_found = True
        if not overlap_found:
            break

    # Recalculate sum
    final_sum = np.sum(radii)

    return centers, radii, final_sum
