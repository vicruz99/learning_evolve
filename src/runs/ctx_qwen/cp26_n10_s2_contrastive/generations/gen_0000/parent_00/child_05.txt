3. **Iterative Scaling**: Start with a low `alpha` to allow the optimizer to explore configurations and increase radii, then gradually increase `alpha` to strictly enforce constraints. This prevents the optimizer from getting trapped by stiff penalty gradients early on.

import numpy as np
from scipy.optimize import minimize

def smooth_max(x, eps=1e-9):
    """Smooth approximation of max(0, x) for differentiable optimization."""
    return 0.5 * (x + np.sqrt(x**2 + eps))

def compute_loss(vars, n, alpha):
    """
    Objective function: -sum(radii) + alpha * (boundary_violations + overlap_violations)
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    loss = -np.sum(radii)

    # Boundary penalties
    loss += alpha * np.sum(smooth_max(radii - centers[:, 0])**2)
    loss += alpha * np.sum(smooth_max(centers[:, 0] + radii - 1.0)**2)
    loss += alpha * np.sum(smooth_max(radii - centers[:, 1])**2)
    loss += alpha * np.sum(smooth_max(centers[:, 1] + radii - 1.0)**2)

    # Overlap penalties (vectorized pairwise)
    diff0 = centers[:, 0][:, None] - centers[:, 0]
    diff1 = centers[:, 1][:, None] - centers[:, 1]
    dists = np.sqrt(diff0**2 + diff1**2)
    overlaps = radii[:, None] + radii[None, :] - dists
    np.fill_diagonal(overlaps, 0.0)
    upper = np.triu(overlaps, k=1)
    loss += alpha * np.sum(smooth_max(upper)**2)
    
    return loss

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Run multiple trials from different initial perturbations
    for trial in range(4):
        # Initialize centers in a perturbed grid
        centers = np.zeros((n, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                if idx < n:
                    centers[idx] = [0.1 + i * 0.2, 0.1 + j * 0.2]
                    idx += 1
        centers += np.random.uniform(-0.015, 0.015, size=centers.shape)
        centers = np.clip(centers, 0.05, 0.95)

        radii = np.ones(n) * 0.09
        x0 = np.concatenate([centers.flatten(), radii])

        current_x = x0
        # Iteratively increase penalty strength to enforce constraints
        alphas = [500, 2000, 6000, 15000]
        bounds = [(0, 1)] * (2 * n) + [(1e-5, 0.5)] * n

        for alpha in alphas:
            res = minimize(compute_loss, current_x, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-14})
            current_x = res.x

        c_opt = current_x[:2*n].reshape(n, 2)
        r_opt = current_x[2*n:]
        s = np.sum(r_opt)
        
        if s > best_sum:
            best_sum = s
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()

    # Ensure non-negative radii due to numerical precision
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, best_sum
