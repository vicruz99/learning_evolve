import numpy as np
from scipy.optimize import minimize


def objective_penalty(vars, n):
    """Objective with penalty for constraint violations."""
    centers_opt = vars[:n * 2].reshape(n, 2)
    radii_opt = vars[n * 2:]

    obj = -np.sum(radii_opt)
    penalty = 0

    for i in range(n):
        x, y = centers_opt[i]
        r = radii_opt[i]
        if x < r:
            penalty += (x - r) ** 2
        if x > 1 - r:
            penalty += (x - (1 - r)) ** 2
        if y < r:
            penalty += (y - r) ** 2
        if y > 1 - r:
            penalty += (y - (1 - r)) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers_opt[i, 0] - centers_opt[j, 0]
            dy = centers_opt[i, 1] - centers_opt[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            r_sum = radii_opt[i] + radii_opt[j]
            if dist < r_sum:
                penalty += (dist - r_sum) ** 2

    return obj + 1000.0 * penalty


def constraint_x_lower(vars, i):
    return vars[2 * i] - vars[26 * 2 + i]


def constraint_x_upper(vars, i):
    return 1 - vars[2 * i] - vars[26 * 2 + i]


def constraint_y_lower(vars, i):
    return vars[2 * i + 1] - vars[26 * 2 + i]


def constraint_y_upper(vars, i):
    return 1 - vars[2 * i + 1] - vars[26 * 2 + i]


def constraint_radius(vars, i):
    return vars[26 * 2 + i]


def constraint_non_overlap(vars, i, j):
    x_i = vars[2 * i]
    y_i = vars[2 * i + 1]
    x_j = vars[2 * j]
    y_j = vars[2 * j + 1]
    r_i = vars[26 * 2 + i]
    r_j = vars[26 * 2 + j]
    dist = np.sqrt((x_i - x_j) ** 2 + (y_i - y_j) ** 2)
    return dist - r_i - r_j


def make_hexagonal_init(n):
    """Create hexagonal packing initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08

    idx = 0
    row_counts = [6, 5, 6, 5, 4]
    spacing = 0.17

    for row_idx, count in enumerate(row_counts):
        for col in range(count):
            if idx < n:
                x = 0.12 + col * spacing
                if row_idx % 2 == 1:
                    x += spacing / 2
                y = 0.12 + row_idx * spacing * np.sqrt(3) / 2
                centers[idx] = [np.clip(x, 0.02, 0.98), np.clip(y, 0.02, 0.98)]
                idx += 1

    return centers, radii


def make_grid_init(n):
    """Create grid initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.07

    rows = 5
    cols = 6
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx < n:
                centers[idx] = [0.1 + j * 0.18, 0.1 + i * 0.2]
                idx += 1

    return centers, radii


def make_random_init(n):
    """Create random initial configuration."""
    centers = np.random.rand(n, 2) * 0.7 + 0.15
    radii = np.ones(n) * 0.05
    return centers, radii


def run_slsqp(centers, radii, n):
    """Run SLSQP optimization from given initial configuration."""
    x0 = np.concatenate([centers.flatten(), radii])

    def objective(vars):
        return -np.sum(vars[n * 2:])

    cons = []

    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: constraint_x_lower(v, i)})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: constraint_x_upper(v, i)})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: constraint_y_lower(v, i)})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: constraint_y_upper(v, i)})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: constraint_radius(v, i)})

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: constraint_non_overlap(v, i, j)
            })

    bounds = [(0.001, 0.999) for _ in range(n * 2)] + [(0, 0.5) for _ in range(n)]

    try:
        result = minimize(objective, x0, method='SLSQP', constraints=cons, bounds=bounds,
                         options={'maxiter': 2000, 'ftol': 1e-14})
        return result.x, result.fun
    except Exception:
        return x0, 0
