def calculate_penalty(centers, radii):
    """
    Calculates a penalty value for overlaps and boundary violations.
    Penalty is the sum of squared violations.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary violations
    # x must be in [r, 1-r], y in [r, 1-r]
    # Violation = max(0, r - x)^2 + max(0, x - (1-r))^2 + ...
    # We can vectorize this
    
    # Check x bounds
    # x < r -> violation (r - x)
    # x > 1-r -> violation (x - (1-r))
    
    # Lower bounds
    lower_viol_x = np.maximum(0, radii - centers[:, 0])
    upper_viol_x = np.maximum(0, centers[:, 0] - (1 - radii))
    
    lower_viol_y = np.maximum(0, radii - centers[:, 1])
    upper_viol_y = np.maximum(0, centers[:, 1] - (1 - radii))
    
    penalty += np.sum(lower_viol_x**2) + np.sum(upper_viol_x**2)
    penalty += np.sum(lower_viol_y**2) + np.sum(upper_viol_y**2)
    
    # Overlaps
    # dist_ij < r_i + r_j -> violation (r_i + r_j - dist_ij)
    # We compute pairwise distances.
    # Using broadcasting for efficiency
    
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    # dist shape (n, n)
    
    # To save memory/time for n=26, explicit loops or vectorized chunks are fine.
    # Vectorized approach:
    # c1 = centers[:, np.newaxis, :] # (n, 1, 2)
    # c2 = centers[np.newaxis, :, :] # (1, n, 2)
    # diff = c1 - c2
    # dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # However, for n=26, n^2 = 676, very small.
    
    c1 = centers[:, np.newaxis, :]
    c2 = centers[np.newaxis, :, :]
    diff = c1 - c2
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Radii sum matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Violations where dist < r_sum
    overlap = np.maximum(0, r_sum - dists)
    
    # Sum squared overlaps. Note: overlap matrix is symmetric and diagonal is 0.
    # We sum upper triangle or just sum all and divide by 2.
    penalty += np.sum(overlap**2) / 2.0 # Divide by 2 because i,j and j,i counted
    
    return penalty

def get_penalty_gradient(centers, radii):
    """
    Numerical gradient of penalty function with respect to centers.
    """
    n = centers.shape[0]
    grad = np.zeros_like(centers)
    h = 1e-5
    base_penalty = calculate_penalty(centers, radii)
    
    for i in range(n):
        for d in range(2):
            centers_plus = centers.copy()
            centers_plus[i, d] += h
            penalty_plus = calculate_penalty(centers_plus, radii)
            grad[i, d] = (penalty_plus - base_penalty) / h
            
    return grad

def optimize_positions(centers, radii, max_iters=500):
    """
    Optimizes centers to minimize penalty for fixed radii.
    """
    x0 = centers.flatten()
    
    def obj(x):
        c = x.reshape(-1, 2)
        return calculate_penalty(c, radii)
    
    # Use L-BFGS-B as it handles bounds well, but here we use penalty for bounds.
    # Unconstrained minimization of penalty is fine.
    result = opt.minimize(obj, x0, method='L-BFGS-B', options={'maxiter': max_iters, 'ftol': 1e-12})
    
    best_centers = result.x.reshape(-1, 2)
    return best_centers
