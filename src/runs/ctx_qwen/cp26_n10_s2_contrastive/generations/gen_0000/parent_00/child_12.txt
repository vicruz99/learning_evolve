import numpy as np
import scipy.optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # 1. Initialization: Hexagonal Grid
    # A hexagonal grid is denser than a square grid.
    # We attempt to fit 26 circles in a staggered pattern.
    # Rows with 6 circles, rows with 5 circles...
    # 6 + 5 + 6 + 5 + 4 = 26
    
    row_counts = [6, 5, 6, 5, 4]
    current_idx = 0
    vertical_spacing = 0.18  # Initial guess for row spacing
    
    for r_idx, count in enumerate(row_counts):
        # Horizontal spacing
        if count > 0:
            # Distribute centers evenly with margin
            # Margin r implies center in [r, 1-r]. 
            # For r ~ 0.1, range is [0.1, 0.9]. Span 0.8.
            # 5 intervals for 6 circles: 0.8 / 5 = 0.16
            margin = 0.12
            span = 1.0 - 2 * margin
            step = span / (count - 1) if count > 1 else 0
            
            # Stagger odd/even rows
            x_offset = (margin + step) / 2 if r_idx % 2 == 1 else margin
            
            # Vertical position
            y_pos = margin + (vertical_spacing * r_idx)
            
            for c_idx in range(count):
                x_pos = x_offset + c_idx * step
                centers[current_idx] = [x_pos, y_pos]
                radii[current_idx] = 0.09 # Initial small radius
                current_idx += 1

    # 2. Optimization Loop
    # We use a loop to iteratively expand radii and resolve collisions.
    # This "expansion" helps the optimizer escape local optima where 
    # radii are small but positions are stuck.
    
    for step in range(15):
        # Define bounds for variables: [x1, y1, r1, ..., x26, y26, r26]
        # x, y in [0, 1], r >= 0.
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        
        # Flatten variables for scipy
        x0 = np.hstack([centers.flatten(), radii])
        
        # Objective: Maximize sum of radii -> Minimize negative sum
        def objective(vars_flat):
            r = vars_flat[2*n:]
            return -np.sum(r)
        
        # Constraints
        constraints = []
        
        # Helper to extract variables
        def get_centers(vars_flat):
            return vars_flat[:2*n].reshape((n, 2))
        
        def get_radii(vars_flat):
            return vars_flat[2*n:]

        # 1. Boundary Constraints: r_i <= min(x, 1-x, y, 1-y)
        # r_i - x_i <= 0
        # r_i - (1 - x_i) <= 0
        # r_i - y_i <= 0
        # r_i - (1 - y_i) <= 0
        
        # We can define these as a single function returning an array of inequalities <= 0
        def boundary_constraints(vars_flat):
            C = get_centers(vars_flat)
            R = get_radii(vars_flat)
            res = []
            for i in range(n):
                x, y = C[i]
                r = R[i]
                # r <= x  => r - x <= 0
                res.append(r - x)
                # r <= 1-x => r - (1-x) <= 0
                res.append(r - (1.0 - x))
                # r <= y
                res.append(r - y)
                # r <= 1-y
                res.append(r - (1.0 - y))
            return np.array(res)

        constraints.append({'type': 'ineq', 'fun': lambda v: -boundary_constraints(v)})

        # 2. Non-overlap Constraints: r_i + r_j <= dist(C_i, C_j)
        # dist - (r_i + r_j) >= 0  =>  (r_i + r_j) - dist <= 0
        # We handle this carefully to avoid NaN gradients at dist=0
        
        def non_overlap_constraints(vars_flat):
            C = get_centers(vars_flat)
            R = get_radii(vars_flat)
            res = []
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.linalg.norm(C[i] - C[j])
                    # If distance is extremely small, force radius sum to be 0
                    # But practically, we want r_i + r_j <= d
                    # Inequality: r_i + r_j - d <= 0
                    res.append(R[i] + R[j] - d)
            return np.array(res)

        constraints.append({'type': 'ineq', 'fun': lambda v: -non_overlap_constraints(v)})

        # Run optimization
        # Using SLSQP. Method choice is critical.
        try:
            res = scipy.optimize.minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'ftol': 1e-8, 'maxiter': 200, 'disp': False}
            )
            
            if res.success or res.fun < objective(x0):
                # Update state
                x_opt = res.x
                centers = x_opt[:2*n].reshape((n, 2))
                radii = x_opt[2*n:]
                
                # Expansion Step: Slightly increase radii to push optimizer to find new space
                # This effectively searches for a "tighter" packing.
                expansion_factor = 1.005
                radii = np.clip(radii * expansion_factor, 0, 0.5)
                # Ensure feasibility roughly by clamping to boundaries
                # This might create slight overlaps, which the next iter will fix.
                # But to be safe, we can just let the solver handle it.
                # We reset x0 to include expanded radii.
                
        except Exception:
            pass

    # 3. Final Validation and Cleanup
    # Ensure strict non-negative radii and bounds
    radii = np.maximum(radii, 0.0)
    centers = np.clip(centers, radii.reshape(-1, 1), 1.0 - radii.reshape(-1, 1))
    
    # Re-calculate optimal radii for the fixed centers via a simple LP-like adjustment
    # or just a few more iterations to ensure sum is maximized.
    # Given the solver worked on r and C jointly, this should be fine.
    
    # Final check to ensure no overlaps (numerical safety)
    # If overlaps exist due to numerical error, reduce radii slightly.
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii[i] + radii[j] - 1e-9:
                # Overlap detected, reduce radii
                avg_r = (radii[i] + radii[j]) / 2.0
                radii[i] = dist / 2.0 * 0.999
                radii[j] = dist / 2.0 * 0.999

    # Check boundaries again
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        max_r = min(x, 1-x, y, 1-y)
        if r > max_r:
            radii[i] = max_r

    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
