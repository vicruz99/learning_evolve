import numpy as np
import math
from scipy.optimize import linprog


N_CIRCLES = 26


def build_constraint_matrix(n):
    """Build the fixed constraint matrix A_ub for the LP."""
    m_boundary = n
    m_overlap = n * (n - 1) // 2
    m_total = m_boundary + m_overlap

    A_ub = np.zeros((m_total, n))

    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        A_ub[i, i] = 1.0

    # Overlap constraints: r_i + r_j <= dist(i, j)
    idx = n
    overlap_pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            overlap_pairs.append((i, j))
            idx += 1

    return A_ub, overlap_pairs


def compute_radii_lp(centers, n, A_ub, overlap_pairs):
    """Compute maximum sum-of-radii for given centers using LP."""
    c = -np.ones(n)

    m_boundary = n
    m_overlap = len(overlap_pairs)
    b_ub = np.zeros(m_boundary + m_overlap)

    # Boundary constraints RHS
    for i in range(n):
        x, y = centers[i]
        b_ub[i] = min(x, 1.0 - x, y, 1.0 - y)

    # Overlap constraints RHS
    for idx, (i, j) in enumerate(overlap_pairs):
        dx = centers[i][0] - centers[j][0]
        dy = centers[i][1] - centers[j][1]
        dist = math.sqrt(dx * dx + dy * dy)
        b_ub[m_boundary + idx] = dist

    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'disp': False})

    if result.success:
        return result.x
    return np.zeros(n)


def initialize_hexagonal(n, variant):
    """Initialize centers in hexagonal patterns with different row arrangements."""
    patterns = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [4, 6, 5, 6, 5],
        [7, 5, 6, 5, 3],
        [6, 6, 6, 5, 3],
        [5, 5, 6, 5, 5],
        [6, 4, 6, 4, 6],
        [7, 7, 6, 6],
        [8, 6, 6, 6],
        [6, 7, 6, 7],
    ]

    pattern = patterns[variant % len(patterns)]
    # Ensure sum is exactly n
    current_sum = sum(pattern)
    while current_sum > n:
        pattern[-1] -= 1
        current_sum -= 1
    while sum(pattern) < n:
        pattern.append(1)

    centers = []
    # Estimate radius from available space
    max_in_row = max(pattern)
    r_est = 0.5 / max_in_row * 0.85

    y = 0.5
    y_step = r_est * math.sqrt(3)

    for row_idx, count in enumerate(pattern):
        if count <= 0:
            y += y_step
            continue
        x_start = 0.5 - (count - 1) * r_est
        for col in range(count):
            x = x_start + col * 2 * r_est
            if row_idx % 2 == 1:
                x += r_est
            centers.append([x, y])
        y += y_step

    result = np.array(centers[:n])
    return result


def initialize_staggered_grid(n, variant):
    """Initialize with staggered grid patterns."""
    centers = []
    
    if variant % 4 == 0:
        # 5x5 plus 1
        for i in range(5):
            for j in range(5):
                centers.append([0.1 + j * 0.2, 0.1 + i * 0.2])
        centers.append([0.5, 0.55])
    elif variant % 4 == 1:
        # 4x6 plus 2
        for i in range(4):
            for j in range(6):
                centers.append([0.07 + j * 0.157, 0.12 + i * 0.25])
        centers.append([0.5, 0.03])
        centers.append([0.5, 0.97])
    elif variant % 4 == 2:
        # 6x4 plus 2
        for i in range(6):
            for j in range(4):
                centers.append([0.1 + j * 0.233, 0.08 + i * 0.166])
        centers.append([0.05, 0.5])
        centers.append([0.95, 0.5])
    else:
        # Hexagonal-inspired with 5 rows
        pattern = [6, 5, 6, 5, 4]
        r_est = 0.09
        y = 0.5
        y_step = r_est * math.sqrt(3)
        for row_idx, count in enumerate(pattern):
            x_start = 0.5 - (count - 1) * r_est
            for col in range(count):
                x = x_start + col * 2 * r_est
                if row_idx % 2 == 1:
                    x += r_est
                centers.append([x, y])
            y += y_step

    return np.array(centers[:n])


def initialize_random(n, seed):
    """Random initialization with small jitter."""
    rng = np.random.RandomState(seed)
    centers = rng.rand(n, 2)
    centers = centers * 0.8 + 0.1  # Keep away from boundaries initially
    return centers


def optimize_packing(n, trial, A_ub, overlap_pairs):
    """Run one optimization trial using simulated annealing."""
    # Choose initialization strategy
    if trial < 30:
        centers = initialize_hexagonal(n, trial)
    elif trial < 50:
        centers = initialize_staggered_grid(n, trial)
    else:
        centers = initialize_random(n, trial)

    # Compute initial radii
    current_radii = compute_radii_lp(centers, n, A_ub, overlap_pairs)
    current_sum = np.sum(current_radii)

    best_centers = centers.copy()
    best_radii = current_radii.copy()
    best_sum = current_sum

    best_centers_sa = centers.copy()
    best_radii_sa = current_radii.copy()
    best_sum_sa = current_sum

    # Simulated annealing parameters
    temperature = 0.012
    cooling_rate = 0.9992
    n_iterations = 30000

    rng = np.random.RandomState(42 + trial * 7)

    for iteration in range(n_iterations):
        # Generate neighbor solution
        new_centers = current_centers if False else centers.copy()
        i = rng.randint(n)
        step_size = temperature * (1.0 + 0.5 * math.sin(iteration * 0.01))
        new_centers[i] += rng.randn(2) * step_size
        new_centers[i] = np.clip(new_centers[i], 0.002, 0.998)

        # Compute new radii and sum
        new_radii = compute_radii_lp(new_centers, n, A_ub, overlap_pairs)
        new_sum = np.sum(new_radii)

        # Acceptance criterion
        delta = new_sum - current_sum
        if delta > 0 or rng.random() < math.exp(delta / temperature):
            centers = new_centers
            current_radii = new_radii
            current_sum = new_sum

            if new_sum > best_sum_sa:
                best_sum_sa = new_sum
                best_centers_sa = centers.copy()
                best_radii_sa = new_radii.copy()

        # Cool down
        temperature *= cooling_rate
        if temperature < 1e-8:
            temperature = 1e-8

    # Return best found
    if best_sum_sa > best_sum:
        best_sum = best_sum_sa
        best_centers = best_centers_sa
        best_radii = best_radii_sa

    return best_centers, best_radii, best_sum


def local_optimization(centers, n, A_ub, overlap_pairs):
    """Local hill-climbing optimization."""
    current_radii = compute_radii_lp(centers, n, A_ub, overlap_pairs)
    current_sum = np.sum(current_radii)

    best_centers = centers.copy()
    best_radii = current_radii.copy()
    best_sum = current_sum

    step = 0.003
    rng = np.random.RandomState(999)

    for iteration in range(5000):
        improved = False
        for _ in range(n):
            i = rng.randint(n)
            for direction in range(4):
                new_centers = centers.copy()
                if direction == 0:
                    new_centers[i][0] += step
                elif direction == 1:
                    new_centers[i][0] -= step
                elif direction == 2:
                    new_centers[i][1] += step
                else:
                    new_centers[i][1] -= step

                new_centers[i] = np.clip(new_centers[i], 0.002, 0.998)

                new_radii = compute_radii_lp(new_centers, n, A_ub, overlap_pairs)
                new_sum = np.sum(new_radii)

                if new_sum > current_sum:
                    centers = new_centers
                    current_radii = new_radii
                    current_sum = new_sum
                    improved = True

                    if new_sum > best_sum:
                        best_sum = new_sum
                        best_centers = centers.copy()
                        best_radii = new_radii.copy()

        if not improved:
            step *= 0.5
            if step < 1e-6:
                break

    return best_centers, best_radii, best_sum


def run_packing():
    """Main function to pack 26 circles in unit square."""
    n = N_CIRCLES
    A_ub, overlap_pairs = build_constraint_matrix(n)

    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Run multiple optimization trials
    for trial in range(60):
        centers, radii, s = optimize_packing(n, trial, A_ub, overlap_pairs)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()

    # Local optimization on the best result
    best_centers, best_radii, best_sum = local_optimization(
        best_centers, n, A_ub, overlap_pairs
    )

    # Final validation pass - ensure no numerical issues
    best_centers = np.clip(best_centers, 0.001, 0.999)
    best_radii = np.maximum(best_radii, 0.0)

    # Recompute radii one final time for consistency
    final_radii = compute_radii_lp(best_centers, n, A_ub, overlap_pairs)
    final_sum = np.sum(final_radii)

    return best_centers, final_radii, final_sum
