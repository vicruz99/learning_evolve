import numpy as np
from scipy.optimize import linprog, minimize

def get_max_radii_sum(centers):
    """
    Solves an LP to find radii that maximize the sum of radii 
    for a fixed set of circle centers.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        sum_radii: float (maximum sum of radii)
        radii: np.array of shape (n)
    """
    n = centers.shape[0]
    if n == 0:
        return 0.0, np.array([])
    
    # Compute pairwise distances between centers
    # Using broadcasting for efficiency
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    # c vector for linprog (minimization)
    c = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # 1. r_i + r_j <= dist(i, j) for all i < j
    # 2. r_i <= x_i
    # 3. r_i <= 1 - x_i
    # 4. r_i <= y_i
    # 5. r_i <= 1 - y_i
    # 6. r_i >= 0 (handled by bounds)
    
    n_pairs = n * (n - 1) // 2
    n_bound_constraints = 4 * n
    n_constraints = n_pairs + n_bound_constraints
    
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    
    # Pairwise non-overlap constraints
    # Only upper triangle (i < j) is needed
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        A_ub[idx, i] = 1.0
        b_ub[idx] = x
        idx += 1
        
        # r_i <= 1 - x
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - x
        idx += 1
        
        # r_i <= y
        A_ub[idx, i] = 1.0
        b_ub[idx] = y
        idx += 1
        
        # r_i <= 1 - y
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - y
        idx += 1
        
    # Bounds for radii: r_i >= 0
    bounds = [(0, None)] * n
    
    # Solve LP using HiGHS method (fast and robust)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            return -res.fun, res.x
        else:
            # If infeasible (should not happen with valid centers), return 0
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective_function(centers_flat):
    """
    Objective function for the outer optimization loop.
    Returns negative sum of radii (to be minimized).
    """
    centers = centers_flat.reshape(-1, 2)
    
    # Clip centers to [0, 1] to ensure validity inside LP constraints
    # This acts as a barrier for the optimizer
    centers = np.clip(centers, 0.0, 1.0)
    
    sum_radii, _ = get_max_radii_sum(centers)
    
    # We want to maximize sum_radii, so we minimize -sum_radii
    return -sum_radii

def generate_hex_grid(n, spacing_scale=1.0):
    """
    Generates a hexagonal grid of points within [0, 1]x[0, 1].
    """
    # Base spacing estimation
    # Target radius approx 0.1, so spacing approx 0.2
    base_spacing = 0.2 * spacing_scale
    dx = base_spacing
    dy = dx * np.sqrt(3) / 2
    
    points = []
    y = 0
    row = 0
    while y <= 1.0:
        offset = 0.0 if row % 2 == 0 else dx / 2.0
        x = offset
        while x <= 1.0:
            points.append([x, y])
            x += dx
        y += dy
        row += 1
        
    # Select first n points
    if len(points) < n:
        # If grid is too sparse, just return random or fill up
        # But for our parameters, it should be dense enough
        pass
        
    return np.array(points[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Main function to run the circle packing optimization.
    """
    n = 26
    
    best_sum_radii = -1.0

    best_radii = None

    initial_configs = []
    
    # 1. Hexagonal Grid (Standard Density)
    initial_configs.append(generate_hex_grid(n, spacing_scale=1.0))
    
    # 2. Hexagonal Grid (Slightly Denser)
    initial_configs.append(generate_hex_grid(n, spacing_scale=0.9))
    
    # 3. Hexagonal Grid (Slightly Sparser)
    initial_configs.append(generate_hex_grid(n, spacing_scale=1.1))
    
    # 4. Perturbed Hexagonal Grid

        grid = generate_hex_grid(n, spacing_scale=1.0)
        noise = np.random.uniform(-0.05, 0.05, grid.shape)

        initial_configs.append(perturbed)

        initial_configs.append(np.random.uniform(0, 1, (n, 2)))

    # maxiter needs to be high enough to converge

    options = {
        'maxiter': 3000,
        'xatol': 1e-5,
        'fatol': 1e-5,
        'adaptive': True
    }
    
    for i, start_centers in enumerate(initial_configs):
        # Flatten centers for optimizer
        x0 = start_centers.flatten()

        try:
            # Run Nelder-Mead optimization
            res = minimize(objective_function, x0, method='Nelder-Mead', options=options)

        best_sum_radii = np.sum(best_radii)
