import numpy as np
import math


def compute_overlap_penalty(centers, radii, n):
    """Compute penalty for overlaps and boundary violations."""
    penalty = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += 1e6 * (x - r) ** 2
        if x + r > 1:
            penalty += 1e6 * (x + r - 1) ** 2
        if y - r < 0:
            penalty += 1e6 * (y - r) ** 2
        if y + r > 1:
            penalty += 1e6 * (y + r - 1) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += 1e6 * (min_dist - dist) ** 2

    return penalty


def resolve_overlaps(centers, radii, n, iterations=100):
    """Resolve overlaps by pushing circles apart."""
    for _ in range(iterations):
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff ** 2)
                dist = math.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
                min_dist = radii[i] + radii[j]

                if dist < min_dist:
                    push = (min_dist - dist) / (2 * dist)
                    centers[i] += diff * push
                    centers[j] -= diff * push

    # Enforce boundaries
    for i in range(n):
        centers[i, 0] = max(radii[i], min(1 - radii[i], centers[i, 0]))
        centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))


def expand_and_resolve(centers, radii, n, expansion_rate=1.0001, relax_iters=50):
    """Expand all circles slightly and resolve overlaps."""
    for i in range(n):
        radii[i] *= expansion_rate

    resolve_overlaps(centers, radii, n, relax_iters)

    # Enforce boundaries strictly

        centers[i, 0] = max(radii[i] + 1e-12, min(1 - radii[i] - 1e-12, centers[i, 0]))
        centers[i, 1] = max(radii[i] + 1e-12, min(1 - radii[i] - 1e-12, centers[i, 1]))
