import numpy as np

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses force-directed relaxation with adaptive radius scaling.
    """
    n = 26
    np.random.seed(42)
    
    # 1. Initialize centers in a hexagonal lattice pattern
    centers = np.zeros((n, 2))
    idx = 0
    y = 0.15
    x_start = 0.15
    spacing_x = 0.22
    spacing_y = 0.20
    
    while idx < n:
        x = x_start
        while x <= 0.95 and idx < n:
            centers[idx] = [x, y]
            idx += 1
            x += spacing_x
        y += spacing_y
        x_start += spacing_x / 2.0  # Shift for hexagonal staggering
        
    # 2. Force-directed optimization parameters
    r = 0.10
    lr = 0.005
    max_iter = 60000
    
    for step in range(max_iter):
        forces = np.zeros_like(centers)
        min_dist = 1.0
        min_bnd = 1.0
        
        # Compute pairwise interactions and boundary constraints
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                d = np.sqrt(np.sum(diff**2))
                
                if d < min_dist:
                    min_dist = d
                
                # Strong repulsion when circles overlap
                if d < 2 * r - 1e-9:
                    overlap = 2 * r - d
                    dir = diff / (d + 1e-10)
                    f = overlap * 9.0
                    forces[i] += dir * f
                    forces[j] -= dir * f
                    
            # Boundary repulsion and tracking
            for d_idx in range(2):
                if centers[i, d_idx] < r:
                    forces[i, d_idx] += (r - centers[i, d_idx]) * 9.0
                if centers[i, d_idx] > 1.0 - r:
                    forces[i, d_idx] -= (centers[i, d_idx] - (1.0 - r)) * 9.0
                    
                b = min(centers[i, d_idx], 1.0 - centers[i, d_idx])
                if b < min_bnd:
                    min_bnd = b
                    
        # Update positions
        centers += lr * forces
        centers = np.clip(centers, 1e-9, 1.0 - 1e-9)
        
        # Adaptively scale radius towards feasible limit
        r_limit = min(min_dist / 2.0, min_bnd)
        r = 0.997 * r_limit
        
        # Cooling schedule
        if step % 300 == 0:
            lr *= 0.996
            
    # 3. Final exact computation to ensure strict validity
    min_d = 1.0
    min_b = 1.0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if d < min_d:
                min_d = d
        for d_idx in range(2):
            b = min(centers[i, d_idx], 1.0 - centers[i, d_idx])
            if b < min_b:
                min_b = b
                
    # Subtract tolerance buffer to guarantee validation passes
    r_final = min(min_d / 2.0, min_b) - 1e-12
    radii = np.full(n, r_final)
    
    return centers, radii, float(np.sum(radii))
