import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_IND = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

def build_lp_matrices():
    """Pre-construct the constant structure of the LP constraint matrix."""
    A = np.zeros((NUM_PAIRS + 4 * N, N))
    pairs = []
    k = 0
    for i in range(N):
        for j in range(i + 1, N):
            A[k, i] = 1.0
            A[k, j] = 1.0
            pairs.append((i, j))
            k += 1
    for i in range(N):
        base = NUM_PAIRS + 4 * i
        A[base, i] = 1.0
        A[base + 1, i] = 1.0
        A[base + 2, i] = 1.0
        A[base + 3, i] = 1.0
    return A, pairs

A_LP, LP_PAIRS = build_lp_matrices()

def solve_lp_radii(centers):
    """Solves LP for maximal radii given fixed centers. Returns radii, sum, and dual multipliers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-20)
    
    b = np.zeros(A_LP.shape[0])
    idx = 0
    for i, j in LP_PAIRS:
        b[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    radii = np.zeros(N)
    duals = np.zeros_like(b)
    s_sum = 0.0
    
    if res.success:
        radii = res.x
        s_sum = np.sum(radii)
        try:
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                duals = np.asarray(res.ineqlin.marginals)
            elif hasattr(res, 'marginals'):
                duals = np.asarray(res.marginals)
        except Exception:
            pass
    return radii, s_sum, duals

def obj_grad_lbfgs(x_flat):
    """Objective and exact gradient for L-BFGS-B: minimizes negative sum of radii."""
    c = x_flat.reshape(N, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    r, s, duals = solve_lp_radii(c)
    
    grad = np.zeros_like(c)
    diffs = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-20)
    
    idx = 0
    for i, j in LP_PAIRS:
        mu = duals[idx]
        if mu > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
        idx += 1
        
    bound_start = NUM_PAIRS
    for i in range(N):
        grad[i, 0] += duals[bound_start + 4*i] - duals[bound_start + 4*i + 1]
        grad[i, 1] += duals[bound_start + 4*i + 2] - duals[bound_start + 4*i + 3]
        
    return -s, -grad.flatten()

def optimize_lbfgs(c0):
    """Runs L-BFGS-B optimization on centers using exact LP gradient."""
    bnds = [(1e-5, 1.0-1e-5)] * (2*N)
    try:
        res = minimize(obj_grad_lbfgs, c0.flatten(), jac=True, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': 5000, 'ftol': 1e-14, 'gtol': 1e-10})
        c_opt = res.x.reshape(N, 2)
        _, s_opt, _ = solve_lp_radii(c_opt)
        return c_opt, s_opt
    except Exception:
        return c0, 0.0

def coordinate_opt(c_init, rng):
    """Refines configuration by optimizing each circle's position independently."""
    c = c_init.copy()
    _, best_sum, _ = solve_lp_radii(c)
    improved = True
    cycles = 0
    while improved and cycles < 4:
        improved = False
        cycles += 1
        for i in range(N):
            def obj_single(p):
                c_tmp = c.copy()
                c_tmp[i] = p
                return -solve_lp_radii(c_tmp)[1]
            try:
                res = minimize(obj_single, c[i], method='Nelder-Mead',
                               options={'maxiter': 600, 'xatol': 1e-8, 'fatol': 1e-12})
                if -res.fun > best_sum + 1e-9:
                    c[i] = np.clip(res.x, 1e-5, 1.0-1e-5)
                    best_sum = -res.fun
                    improved = True
            except Exception:
                pass
    return c

def slsqp_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def slsqp_cons(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    i, j = TRIU_IND
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

SLSQP_BOUNDS = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

def slsqp_polish(c_init, r_init, maxiter=6000):
    """Joint SLSQP optimization of centers and radii."""
    v0 = np.concatenate([c_init.flatten(), r_init])
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=SLSQP_BOUNDS,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if np.min(slsqp_cons(res.x)) >= -1e-7:
            return res.x[:2*N].reshape(N, 2), res.x[2*N:], -res.fun
    except Exception:
        pass
    return c_init, r_init, 0.0

def generate_inits(rng, n_inits=50):
    """Generates diverse initial configurations."""
    inits = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
        [5, 4, 6, 6, 5], [6, 5, 5, 5, 5], [5, 5, 5, 6, 5],
        [5, 6, 4, 5, 6], [6, 4, 6, 5, 5], [5, 7, 5, 5, 4],
        [5, 5, 5, 5, 5, 1], [6, 5, 5, 5, 5]
    ]
    for pat in patterns:
        for r0 in [0.085, 0.090, 0.095, 0.100, 0.105, 0.110]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            inits.append(np.array(c[:N]))
            
    # Rotated hex grids
    base_c = inits[0].copy()
    for angle in [0.15, 0.3, -0.15, -0.3]:
        cx, cy = 0.5, 0.5
        c_rot = base_c - [cx, cy]
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        c_rot = np.column_stack([c_rot[:,0]*cos_a - c_rot[:,1]*sin_a, 
                                 c_rot[:,0]*sin_a + c_rot[:,1]*cos_a])
        c_rot += [cx, cy]
        inits.append(np.clip(c_rot, 0.05, 0.95))
        
    # Force-directed layouts
    for _ in range(12):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(1200):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c[i] - c[j]
                    d = np.linalg.norm(d_vec) + 1e-8
                    if d < 0.26:
                        f = (0.26 - d) / d * 0.012
                        forces[i] += d_vec * f
                        forces[j] -= d_vec * f
            c += forces
            c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    return inits[:n_inits]

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(200):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    rng = np.random.default_rng(42)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    inits = generate_inits(rng, 50)
    
    # Phase 1: L-BFGS-B from multiple starts
    for c_init in inits:
        c_opt, s_opt = optimize_lbfgs(c_init)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum, _ = solve_lp_radii(best_c)
    else:
        best_r, best_sum, _ = solve_lp_radii(best_c)
        
    # Phase 2: Coordinate optimization
    best_c = coordinate_opt(best_c, rng)
    _, best_sum, _ = solve_lp_radii(best_c)
    
    # Phase 3: Perturbation & Restart (Basin Hopping style)
    for step_idx in range(25):
        noise_scale = 0.018 * (0.91 ** step_idx)
        c_pert = best_c + rng.normal(0, noise_scale, best_c.shape)
        c_pert = np.clip(c_pert, 0.05, 0.95)
        
        c_opt, s_opt = optimize_lbfgs(c_pert)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            
        c_coord = coordinate_opt(c_opt, rng)
        _, s_coord, _ = solve_lp_radii(c_coord)
        if s_coord > best_sum:
            best_sum = s_coord
            best_c = c_coord.copy()
            
    # Phase 4: Joint SLSQP Polish
    _, best_r, _ = solve_lp_radii(best_c)
    c_sl, r_sl, s_sl = slsqp_polish(best_c, best_r, maxiter=8000)
    if s_sl > best_sum:
        best_sum = s_sl
        best_c = c_sl
        best_r = r_sl
        
    # Phase 5: Final LP refinement
    best_r, best_sum, _ = solve_lp_radii(best_c)
    
    # Phase 6: Strict Numerical Repair
    radii = repair(best_c.copy(), best_r.copy())
    return best_c, radii, float(np.sum(radii))
