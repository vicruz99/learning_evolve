import numpy as np
from scipy.optimize import linprog, minimize

N = 26
NUM_PAIRS = N * (N - 1) // 2

def build_lp_matrices():
    """Pre-construct the constant structure of the LP constraint matrix."""
    A = np.zeros((NUM_PAIRS + 4 * N, N))
    pairs = []
    k = 0
    for i in range(N):
        for j in range(i + 1, N):
            A[k, i] = 1.0
            A[k, j] = 1.0
            pairs.append((i, j))
            k += 1
    for i in range(N):
        base = NUM_PAIRS + 4 * i
        A[base, i] = 1.0
        A[base + 1, i] = 1.0
        A[base + 2, i] = 1.0
        A[base + 3, i] = 1.0
    return A, pairs

A_LP, LP_PAIRS = build_lp_matrices()

def solve_lp(centers):
    """Solves LP for maximal radii given fixed centers. Returns radii, sum, and dual multipliers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    b = np.zeros(A_LP.shape[0])
    idx = 0
    for i, j in LP_PAIRS:
        b[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(b)
        
    radii = res.x
    try:
        duals = np.asarray(res.ineqlin.marginals)
    except (AttributeError, ValueError, TypeError):
        try:
            duals = np.asarray(res.marginals)
        except:
            duals = np.zeros_like(b)
    return radii, np.sum(radii), duals

def compute_grad(centers, duals):
    """Computes exact gradient of sum of radii w.r.t centers using LP duals."""
    grad = np.zeros_like(centers)
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-15)
    
    idx = 0
    for i, j in LP_PAIRS:
        mu = duals[idx]
        if mu > 1e-8:
            d = dists[i, j]
            vec = (centers[i] - centers[j]) / d
            grad[i] += mu * vec
            grad[j] -= mu * vec
        idx += 1
        
    bound_start = NUM_PAIRS
    for i in range(N):
        grad[i, 0] += duals[bound_start + 4*i] - duals[bound_start + 4*i + 1]
        grad[i, 1] += duals[bound_start + 4*i + 2] - duals[bound_start + 4*i + 3]
    return grad

def gradient_ascent(centers0, max_iter=3500, step=0.006):
    """Runs gradient ascent on centers to maximize sum of radii."""
    c = centers0.copy()
    radii, s, duals = solve_lp(c)
    best_c, best_s = c.copy(), s
    
    for _ in range(max_iter):
        grad = compute_grad(c, duals)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        c_new = c + step * (grad / gn)
        c_new = np.clip(c_new, 1e-5, 1.0 - 1e-5)
        radii, s_new, duals = solve_lp(c_new)
        if s_new > s:
            s = s_new
            c = c_new
            if s > best_s:
                best_s = s
                best_c = c.copy()
            step = min(step * 1.03, 0.025)
        else:
            step *= 0.94
    return best_c, best_s

def coord_obj(params, centers, idx):
    """Objective for single-circle coordinate descent: maximize total radius sum."""
    c_tmp = centers.copy()
    c_tmp[idx] = np.clip(params, 1e-5, 1.0 - 1e-5)
    _, s, _ = solve_lp(c_tmp)
    return -s

def coord_descent(centers, rounds=3):
    """Optimizes each circle's position independently while fixing others."""
    c = centers.copy()
    _, best_s, _ = solve_lp(c)
    for _ in range(rounds):
        for i in range(N):
            try:
                res = minimize(coord_obj, c[i], args=(c, i), method='Nelder-Mead',
                               options={'maxiter': 300, 'xatol': 1e-8, 'fatol': 1e-10})
                if -res.fun > best_s + 1e-9:
                    best_s = -res.fun
                    c[i] = res.x
            except:
                pass
    return c, best_s

def kick_optimize(centers, n_kicks=20, rng=None):
    """Stochastic basin hopping: perturb subsets of circles and re-optimize."""
    c = centers.copy()
    _, best_s, _ = solve_lp(c)
    for _ in range(n_kicks):
        c_try = c.copy()
        n_kick = rng.integers(1, 4)
        idxs = rng.choice(N, n_kick, replace=False)
        for idx in idxs:
            c_try[idx] += rng.normal(0, 0.025, 2)
        c_try = np.clip(c_try, 0.05, 0.95)
        
        # Quick local search on kicked circles
        for _ in range(1):
            for idx in idxs:
                res = minimize(coord_obj, c_try[idx], args=(c_try, idx), method='Nelder-Mead',
                               options={'maxiter': 200, 'xatol': 1e-7, 'fatol': 1e-9})
                c_try[idx] = res.x
                
        _, s_try, _ = solve_lp(c_try)
        if s_try > best_s:
            best_s = s_try
            c = c_try
    return c, best_s

def generate_inits(rng):

        c[:4] = corners

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(200):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
