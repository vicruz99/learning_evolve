import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_IND = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

def build_lp_matrices():
    """Pre-construct the constant structure of the LP constraint matrix."""
    A = np.zeros((NUM_PAIRS + 4 * N, N))
    pairs = []
    k = 0
    for i in range(N):
        for j in range(i + 1, N):
            A[k, i] = 1.0
            A[k, j] = 1.0
            pairs.append((i, j))
            k += 1
    for i in range(N):
        base = NUM_PAIRS + 4 * i
        A[base, i] = 1.0
        A[base + 1, i] = 1.0
        A[base + 2, i] = 1.0
        A[base + 3, i] = 1.0
    return A, pairs

A_LP, LP_PAIRS = build_lp_matrices()

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii given fixed centers. Returns radii, sum, and exact gradient."""
    c = np.clip(centers, 1e-7, 1.0 - 1e-7)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diffs = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    b = np.zeros(A_LP.shape[0])
    idx = 0
    for i, j in LP_PAIRS:
        b[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b[idx] = c[i, 0]; idx += 1
        b[idx] = 1.0 - c[i, 0]; idx += 1
        b[idx] = c[i, 1]; idx += 1
        b[idx] = 1.0 - c[i, 1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(c)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    duals = np.zeros_like(b)
    # Safe extraction of dual marginals across SciPy versions
    if hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
        duals = np.asarray(res.marginals.ineqlin)
    elif hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
        duals = np.asarray(res.ineqlin.marginals)
        
    grad = np.zeros_like(c)
    idx = 0
    for i, j in LP_PAIRS:
        mu = duals[idx]
        if mu > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
        idx += 1
        
    bound_start = NUM_PAIRS
    for i in range(N):
        grad[i, 0] += duals[bound_start + 4*i] - duals[bound_start + 4*i + 1]
        grad[i, 1] += duals[bound_start + 4*i + 2] - duals[bound_start + 4*i + 3]
        
    return radii, s_sum, grad

def obj_grad_lbfgs(v):
    """Objective and gradient for L-BFGS-B: minimize negative sum of radii."""
    centers = v.reshape(N, 2)
    _, s, g = solve_lp_and_grad(centers)
    return -s, -g.flatten()

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 5, 5, 6], [6, 5, 5, 5, 5],
        [5, 7, 5, 5, 4], [5, 5, 5, 6, 5], [6, 5, 6, 4, 5]
    ]
    for pat in patterns:
        for r0 in [0.088, 0.095, 0.102, 0.108]:
            c = []
            y = r0

                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            c = np.array(c[:N])
            c += rng.normal(0, 0.004, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    # Force-directed layouts for high initial separation
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(600):
            forces = np.zeros_like(c)

                    d = np.linalg.norm(d_vec)

                        forces[i] += d_vec * f
                        forces[j] -= d_vec * f
            c += forces
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    return starts

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
