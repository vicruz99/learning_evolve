import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIRS = [(i, j) for i in range(N) for j in range(i + 1, N)]
NUM_PAIRS = len(PAIRS)
TRIU_IND = np.triu_indices(N, 1)

# Precompute constant structure of LP constraint matrix
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
for k, (i, j) in enumerate(PAIRS):
    A_LP[k, i] = 1.0
    A_LP[k, j] = 1.0
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii given fixed centers. Returns radii, sum_radii, and gradient."""
    # Clip centers slightly inside to ensure feasible upper bounds
    c = np.clip(centers, 1e-5, 1.0 - 1e-5)
    
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    # Vectorized distance computation
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    
    b_ub = np.zeros(NUM_PAIRS + 4 * N)
    for k, (i, j) in enumerate(PAIRS):
        b_ub[k] = dists[i, j]
    for i in range(N):
        base = NUM_PAIRS + 4 * i
        b_ub[base] = c[i, 0]
        b_ub[base + 1] = 1.0 - c[i, 0]
        b_ub[base + 2] = c[i, 1]
        b_ub[base + 3] = 1.0 - c[i, 1]
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub,
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(c)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    # Extract dual marginals safely across scipy versions
    duals = np.zeros_like(b_ub)
    if hasattr(res, 'marginals') and res.marginals is not None:
        duals = res.marginals.ineqlin
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = res.ineqlin.marginals
        
    # Compute gradient using dual variables
    grad = np.zeros_like(c)
    for k, (i, j) in enumerate(PAIRS):
        mu = duals[k]
        if mu > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                v = (c[i] - c[j]) / d
                grad[i] += mu * v
                grad[j] -= mu * v
                
    bound_start = NUM_PAIRS
    for i in range(N):
        base = bound_start + 4 * i
        grad[i, 0] += duals[base] - duals[base + 1]
        grad[i, 1] += duals[base + 2] - duals[base + 3]
        
    return radii, s_sum, grad

def obj_lp(v):
    """Objective and gradient for L-BFGS-B / Powell on centers."""
    c = v.reshape(N, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    _, s, g = solve_lp_and_grad(c)
    return -s, -g.flatten()

def generate_inits(rng):
    """Generates diverse initial configurations."""
    inits = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
        [5, 4, 6, 6, 5], [6, 5, 5, 5, 5], [5, 5, 5, 6, 5],
        [5, 6, 4, 5, 6], [6, 4, 6, 5, 5], [5, 7, 5, 5, 4]
    ]
    
    for pat in patterns:
        for r0 in [0.085, 0.090, 0.095, 0.100, 0.105, 0.110]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            inits.append(np.array(c[:N]))
            
    # 5x5 grid + 1 center
    gx = np.linspace(0.1, 0.9, 5)
    gy = np.linspace(0.1, 0.9, 5)
    cx, cy = np.meshgrid(gx, gy)
    c_grid = np.column_stack((cx.flatten(), cy.flatten()))
    c_plus = np.vstack([c_grid, [0.5, 0.5]])
    c_plus += rng.normal(0, 0.012, c_plus.shape)
    inits.append(np.clip(c_plus, 0.05, 0.95))
    
    # Force-directed random starts
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(600):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    d_vec = c[i] - c[j]
                    d = np.linalg.norm(d_vec)
                    if d < 0.20 and d > 1e-6:
                        f = (0.20 - d) / (d**2 + 1e-8) * 0.01
                        forces[i] += d_vec * f
                        forces[j] -= d_vec * f
            c += forces
            c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    return inits

def coordinate_refinement(centers, rng):
    """Refines each circle's position independently."""
    c_best = centers.copy()
    _, s_best, _ = solve_lp_and_grad(c_best)
    
    for _ in range(3):
        for i in range(N):
            def obj_2d(xy):
                temp = c_best.copy()
                temp[i] = np.clip(xy, 1e-5, 1.0 - 1e-5)
                _, s, _ = solve_lp_and_grad(temp)
                return -s
                
            try:
                res = minimize(obj_2d, c_best[i], method='Nelder-Mead',
                               options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-12})
                if -res.fun > s_best + 1e-9:
                    c_best[i] = np.clip(res.x, 1e-5, 1.0 - 1e-5)
                    _, s_best, _ = solve_lp_and_grad(c_best)
            except Exception:
                pass
    return c_best, s_best

def slsqp_obj(v):
    """Objective for joint SLSQP."""
    return -np.sum(v[2 * N:])

def slsqp_cons(v):
    """Constraints for joint SLSQP."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    dx = c[TRIU_IND[0], 0] - c[TRIU_IND[1], 0]
    dy = c[TRIU_IND[0], 1] - c[TRIU_IND[1], 1]
    dr = r[TRIU_IND[0]] + r[TRIU_IND[1]]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

SLSQP_BOUNDS = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

def slsqp_polish(c_init, r_init, maxiter=8000):
    """Joint SLSQP optimization."""
    v0 = np.concatenate([c_init.flatten(), r_init])
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=SLSQP_BOUNDS,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if np.min(slsqp_cons(res.x)) >= -1e-7:
            return res.x[:2*N].reshape(N, 2), res.x[2*N:], -res.fun
    except Exception:
        pass
    return c_init, r_init, 0.0

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    rng = np.random.default_rng(42)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    inits = generate_inits(rng)
    bounds_xy = [(0.001, 0.999)] * (2 * N)
    
    # Phase 1: Multi-start L-BFGS-B and Powell
    for c_init in inits:
        v0 = np.clip(c_init, 0.005, 0.995).flatten()
        
        # L-BFGS-B
        try:
            res = minimize(obj_lp, v0, jac=True, method='L-BFGS-B', bounds=bounds_xy,
                          options={'maxiter': 3000, 'ftol': 1e-14})
            c_opt = res.x.reshape(N, 2)
            _, s_opt, _ = solve_lp_and_grad(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
        except Exception:
            pass
            
        # Powell
        try:
            res_p = minimize(obj_lp, v0, method='Powell', bounds=bounds_xy,
                            options={'maxiter': 1500, 'ftol': 1e-12})
            c_opt_p = res_p.x.reshape(N, 2)
            _, s_opt_p, _ = solve_lp_and_grad(c_opt_p)
            if s_opt_p > best_sum:
                best_sum = s_opt_p
                best_c = c_opt_p.copy()
        except Exception:
            pass

    if best_c is None:
        best_c = inits[0]
        best_r, best_sum, _ = solve_lp_and_grad(best_c)
    else:
        best_r, _, _ = solve_lp_and_grad(best_c)

    # Phase 2: Coordinate-wise refinement
    best_c, best_sum = coordinate_refinement(best_c, rng)
    best_r, _, _ = solve_lp_and_grad(best_c)

    # Phase 3: Basin Hopping / Perturbation Search
    for step in range(30):
        noise_scale = 0.012 * (0.90 ** step)
        c_pert = best_c + rng.normal(0, noise_scale, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        
        v_pert = c_pert.flatten()
        try:
            res_bh = minimize(obj_lp, v_pert, jac=True, method='L-BFGS-B', bounds=bounds_xy,
                             options={'maxiter': 2000, 'ftol': 1e-13})
            c_bh = res_bh.x.reshape(N, 2)
            _, s_bh, _ = solve_lp_and_grad(c_bh)
            if s_bh > best_sum:
                best_sum = s_bh
                best_c = c_bh.copy()
                best_r, _, _ = solve_lp_and_grad(best_c)
        except Exception:
            pass

    # Phase 4: Final Joint Polish
    c_sl, r_sl, s_sl = slsqp_polish(best_c, best_r * 0.95, maxiter=8000)
    if s_sl > best_sum:
        best_sum = s_sl
        best_c = c_sl
        best_r = r_sl
        
    # Re-solve LP for exact radii on best centers
    best_r, s_lp, _ = solve_lp_and_grad(best_c)
    if s_lp > best_sum:
        best_sum = s_lp
        best_r = best_r
        
    # Phase 5: Strict Numerical Repair
    radii = repair(best_c.copy(), best_r.copy())
    final_sum = float(np.sum(radii))
    
    return best_c, radii, final_sum
