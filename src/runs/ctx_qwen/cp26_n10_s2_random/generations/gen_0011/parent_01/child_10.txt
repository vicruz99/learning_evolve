import numpy as np
from scipy.optimize import linprog, minimize

N = 26
NUM_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure: r_i + r_j <= dist(i, j)
A_LP = np.zeros((NUM_PAIRS, N))
PAIR_INDICES = []
_lp_row = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[_lp_row, i] = 1.0
        A_LP[_lp_row, j] = 1.0
        PAIR_INDICES.append((i, j))
        _lp_row += 1

# Index arrays for fast vectorized operations
IDX_I = np.array([p[0] for p in PAIR_INDICES])
IDX_J = np.array([p[1] for p in PAIR_INDICES])

def solve_lp_and_gradient(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns radii, sum_radii, and gradient w.r.t centers.
    """
    n = centers.shape[0]
    
    # Upper bounds from boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-7)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # RHS for LP constraints
    b_ub = dists[IDX_I, IDX_J]
    
    # Solve LP: max sum(r) s.t. A r <= b, 0 <= r <= ub
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub,
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(n), 0.0, np.zeros_like(centers)
        
    radii = res.x
    
    # Extract dual marginals safely
    duals_ineq = np.zeros(NUM_PAIRS)
    if hasattr(res, 'marginals') and res.marginals is not None:
        if hasattr(res.marginals, 'ineqlin'):
            duals_ineq = res.marginals.ineqlin
        elif hasattr(res.marginals, 'ineq'):
            duals_ineq = res.marginals.ineq
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals_ineq = res.ineqlin.marginals
        
    # Compute gradient from active pairwise constraints
    # grad[i] = sum_j duals[i,j] * (c_i - c_j) / d_ij
    active_mask = duals_ineq > 1e-9
    act_i = IDX_I[active_mask]
    act_j = IDX_J[active_mask]
    act_duals = duals_ineq[active_mask]
    act_dists = dists[act_i, act_j]
    
    safe_dists = np.maximum(act_dists, 1e-9)
    weights = act_duals / safe_dists
    
    grad = np.zeros_like(centers)
    diff_act = diff[act_i, act_j]
    
    # Vectorized accumulation using explicit loop for safety and clarity
    for k in range(len(act_i)):
        i = act_i[k]
        j = act_j[k]
        w = weights[k]
        vec = diff_act[k]
        grad[i] += w * vec
        grad[j] -= w * vec
        
    return radii, np.sum(radii), grad

def obj_and_grad_for_lbfgsb(v):
    """Objective and gradient wrapper for L-BFGS-B."""
    centers = v.reshape(N, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    _, s, g = solve_lp_and_gradient(centers)
    return -s, -g.flatten()

def generate_initial_configs(rng):

    configs = []
    
    # 1. Hexagonal lattices with varying densities
    patterns = [
        [5, 5, 5, 5, 6], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4],

    ]

                x = r0 + shift

                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            while len(c) < N:
                c.append(rng.uniform(0.1, 0.9, 2))
            c = np.array(c[:N])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.05, 0.95)
            configs.append(c)

        c = rng.uniform(0.1, 0.9, (N, 2))
        configs.append(c)

        c[:4] = corners
        c += rng.normal(0, 0.005, c.shape)
        c = np.clip(c, 0.02, 0.98)
        configs.append(c)
        
    return configs

def single_circle_obj(x, centers, idx):
    """Objective for optimizing a single circle's position."""
    temp = centers.copy()
    temp[idx] = np.clip(x, 1e-5, 1.0 - 1e-5)
    _, s, _ = solve_lp_and_gradient(temp)
    return -s

def coordinate_descent_refinement(centers, rng):
    """Optimize one circle at a time using Nelder-Mead."""
    c = centers.copy()
    _, best_s, _ = solve_lp_and_gradient(c)
    
    for iteration in range(3):
        for i in range(N):
            try:
                res = minimize(single_circle_obj, c[i], args=(c, i), method='Nelder-Mead',
                               options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-12})

                    c[i] = np.clip(res.x, 1e-5, 1.0 - 1e-5)
                    _, best_s, _ = solve_lp_and_gradient(c)
            except Exception:
                pass
    return c

def obj_joint(v):
    """Objective for joint SLSQP optimization."""
    return -np.sum(v[2 * N:])

def cons_joint(v):
    """Constraints for joint SLSQP optimization."""
    cc = v[:2 * N].reshape(N, 2)
    rr = v[2 * N:]
    c_list = [
        cc[:, 0] - rr,
        1.0 - cc[:, 0] - rr,
        cc[:, 1] - rr,
        1.0 - cc[:, 1] - rr
    ]
    d = np.sqrt(np.sum((cc[IDX_I] - cc[IDX_J])**2, axis=1))
    c_list.append(d - (rr[IDX_I] + rr[IDX_J]))
    return np.concatenate(c_list)

def polish_joint_slsqp(centers, radii, rng):
    """Joint SLSQP refinement for precise constraint satisfaction."""
    n = centers.shape[0]
    v0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
        if np.min(cons_joint(res.x)) >= -1e-8:
            return res.x[:2 * n].reshape(n, 2), res.x[2 * n:], np.sum(res.x[2 * n:])
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def repair_packing(centers, radii):
    """Deterministically shrinks radii to guarantee strict validation compliance."""
    radii = radii.copy()
    n = centers.shape[0]
    
    for _ in range(80):
        changed = False
        for i in range(n):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
                
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
