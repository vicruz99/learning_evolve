from scipy.optimize import linprog, minimize

N = 26
NUM_PAIRS = N * (N - 1) // 2

# Precompute constant LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
_lp_row = 0
for i in range(N):
    for j in range(i + 1, N):

def solve_lp(centers):
    c = np.clip(centers, 1e-6, 1.0 - 1e-6)
    ub = np.minimum(
        np.minimum(c[:, 0], 1.0 - c[:, 0]),
        np.minimum(c[:, 1], 1.0 - c[:, 1])
    )
    ub = np.maximum(ub, 1e-6)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    np.fill_diagonal(dists, 0.0)
    
    b_ub = np.zeros(A_LP.shape[0])
    k = 0
    for i, j in PAIR_IDX:
        b_ub[k] = dists[i, j]
        k += 1
    for i in range(N):
        b_ub[k] = c[i, 0]; k += 1
        b_ub[k] = 1.0 - c[i, 0]; k += 1
        b_ub[k] = c[i, 1]; k += 1
        b_ub[k] = 1.0 - c[i, 1]; k += 1
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub,
                  bounds=[(0.0, u) for u in ub], method='highs')

        return np.zeros(N), 0.0, np.zeros_like(c)

    s = np.sum(radii)

    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = res.ineqlin.marginals

        k += 1

def obj_grad(x):

    _, val, g = solve_lp(c)
    return -val, -g.flatten()

def repair(centers, radii):
    radii = radii.copy()
    for _ in range(80):
        changed = False
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d)/2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
