import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute constant LP constraint matrix structure for pairwise distances
# Constraint: r_i + r_j <= dist(i, j)
N_PAIRS = N * (N - 1) // 2
A_LP = np.zeros((N_PAIRS, N))
PAIR_INDICES = []
_lp_k = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[_lp_k, i] = 1.0
        A_LP[_lp_k, j] = 1.0
        PAIR_INDICES.append((i, j))
        _lp_k += 1

def solve_lp(centers):
    """Solves LP for optimal radii given fixed centers and computes gradient w.r.t centers."""
    c = np.clip(centers, 1e-7, 1.0 - 1e-7)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    b_ub = dists[np.triu_indices(N, k=1)]
    
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(c)
        
    radii = res.x
    
    # Extract dual marginals safely across scipy versions
    duals = np.zeros(N_PAIRS)
    if hasattr(res, 'marginals') and res.marginals is not None:
        duals = getattr(res.marginals, 'ineqlin', getattr(res.marginals, 'ineq', duals))
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = res.ineqlin.marginals
        
    grad = np.zeros_like(c)
    k = 0
    for i, j in PAIR_INDICES:
        lam = duals[k]
        if lam > 1e-10:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        k += 1
        
    return radii, np.sum(radii), grad

def obj_grad(x):
    """Objective and gradient wrapper for L-BFGS-B."""
    c = x.reshape(N, 2)
    _, s, g = solve_lp(c)
    return -s, -g.flatten()

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5]
    ]
    
    for pat in patterns:
        for r_est in [0.085, 0.095, 0.105, 0.112]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            c = np.array(c[:N])
            c += rng.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    # Force-directed random starts to ensure good initial spacing
    for _ in range(10):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(600):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if 0.0 < dist < 0.18:
                        push = (0.18 - dist) * 0.04 / (dist + 1e-6)
                        forces[i] += d_vec / dist * push
                        forces[j] -= d_vec / dist * push
            c += forces * 0.1
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    return starts

def obj_2d_wrapper(xy, idx, centers):
    """Objective for single-circle coordinate descent."""
    temp = centers.copy()
    temp[idx] = np.clip(xy, 0.001, 0.999)
    _, s, _ = solve_lp(temp)
    return -s

def coordinate_refinement(centers, rng):
    """Refines each circle's position independently using Nelder-Mead."""
    best_c = centers.copy()
    _, best_s, _ = solve_lp(best_c)
    
    for _ in range(3):
        for i in range(N):
            try:
                res = minimize(obj_2d_wrapper, best_c[i], args=(i, best_c), method='Nelder-Mead', 
                               options={'maxiter': 300, 'xatol': 1e-8, 'fatol': 1e-12})
                if -res.fun > best_s + 1e-9:
                    best_c[i] = np.clip(res.x, 0.001, 0.999)
                    _, best_s, _ = solve_lp(best_c)
            except Exception:
                pass
    return best_c

def obj_joint(v):
    """Objective for joint SLSQP optimization."""
    return -np.sum(v[2 * N:])

def cons_joint(v):
    """Constraints for joint SLSQP optimization."""
    cc = v[:2 * N].reshape(N, 2)
    rr = v[2 * N:]
    con = []
    con.append(cc[:, 0] - rr)
    con.append(1.0 - cc[:, 0] - rr)
    con.append(cc[:, 1] - rr)
    con.append(1.0 - cc[:, 1] - rr)
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(cc[idx[0]] - cc[idx[1]], axis=1)
    con.append(d - (rr[idx[0]] + rr[idx[1]]))
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_c = None
    best_sum = -1.0
    bounds_xy = [(0.005, 0.995)] * (2 * N)
    
    starts = generate_starts(rng)
    
    # Phase 1: L-BFGS-B from multiple starts
    candidates = []
    for c0 in starts:
        try:
            res = minimize(obj_grad, c0.flatten(), method='L-BFGS-B', 
                           bounds=bounds_xy, jac=True,
                           options={'maxiter': 2500, 'ftol': 1e-14, 'gtol': 1e-10})
            c_opt = res.x.reshape(N, 2)
            _, s_opt, _ = solve_lp(c_opt)
            candidates.append((c_opt, s_opt))
        except Exception:
            pass
            
    candidates.sort(key=lambda x: x[1], reverse=True)
    if candidates:
        best_c = candidates[0][0]
        best_sum = candidates[0][1]
    else:
        best_c = starts[0]
        _, best_sum, _ = solve_lp(best_c)
    
    # Phase 2: Coordinate Refinement on top candidates
    for c_opt, s_opt in candidates[:5]:
        c_ref = coordinate_refinement(c_opt, rng)
        _, s_ref, _ = solve_lp(c_ref)
        if s_ref > best_sum:
            best_sum = s_ref
            best_c = c_ref
            
    # Phase 3: Simulated Annealing to escape local optima
    c_curr = best_c.copy()
    _, s_curr, _ = solve_lp(c_curr)
    best_c_sa = c_curr.copy()
    best_sum_sa = s_curr
    T = 0.005
    
    for step in range(600):
        T *= 0.996
        c_new = c_curr + rng.normal(0, T, c_curr.shape)
        c_new = np.clip(c_new, 0.02, 0.98)
        
        _, s_new, _ = solve_lp(c_new)
        
        delta = s_new - s_curr
        if delta > 0 or rng.random() < np.exp(delta / max(T * 5.0, 1e-8)):
            c_curr = c_new
            s_curr = s_new
            if s_curr > best_sum_sa:
                best_sum_sa = s_curr
                best_c_sa = c_curr.copy()
                
        # Periodic gradient settlement during cooling
        if step % 120 == 0:
            try:
                res = minimize(obj_grad, c_curr.flatten(), method='L-BFGS-B',
                               bounds=bounds_xy, jac=True,
                               options={'maxiter': 500, 'ftol': 1e-13})
                c_curr = res.x.reshape(N, 2)
                _, s_curr, _ = solve_lp(c_curr)
            except Exception:
                pass
                
    if best_sum_sa > best_sum:
        best_sum = best_sum_sa
        best_c = best_c_sa
        
    # Phase 4: Joint SLSQP Polish for precision
    radii_lp, _, _ = solve_lp(best_c)
    v0 = np.concatenate([best_c.flatten(), radii_lp])
    
    bounds_j = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res_sl = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                          constraints={'type': 'ineq', 'fun': cons_joint},
                          options={'maxiter': 8000, 'ftol': 1e-14})
        if np.min(cons_joint(res_sl.x)) >= -1e-9:
            s_sl = np.sum(res_sl.x[2 * N:])
            if s_sl > best_sum:
                best_sum = s_sl
                best_c = res_sl.x[:2 * N].reshape(N, 2)
                radii_lp = res_sl.x[2 * N:]
    except Exception:
        pass
        
    # Phase 5: Strict Deterministic Repair to guarantee validation compliance
    radii = radii_lp.copy()
    for _ in range(50):
        changed = False
        # Boundary clamp
        for i in range(N):
            mr = min(best_c[i, 0], 1.0 - best_c[i, 0], 
                     best_c[i, 1], 1.0 - best_c[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        # Pairwise overlap resolution
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(best_c[i] - best_c[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return best_c, radii, float(np.sum(radii))
