import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute LP constraint matrix structure for pairwise distances
# Constraint: r_i + r_j <= dist(i, j)
A_LP = np.zeros((N * (N - 1) // 2, N))
PAIR_INDICES = []
_lp_row = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[_lp_row, i] = 1.0
        A_LP[_lp_row, j] = 1.0
        PAIR_INDICES.append((i, j))
        _lp_row += 1

def solve_lp_and_gradient(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns radii, sum_radii, and gradient w.r.t centers.
    """
    n = centers.shape[0]
    
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-7)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    b_ub = dists[np.triu_indices(n, k=1)]
    
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(n), 0.0, np.zeros_like(centers)
        
    radii = res.x
    
    duals = np.zeros(A_LP.shape[0])
    if hasattr(res, 'marginals') and res.marginals is not None:
        if hasattr(res.marginals, 'ineqlin'):
            duals = res.marginals.ineqlin
        elif hasattr(res.marginals, 'ineq'):
            duals = res.marginals.ineq
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = res.ineqlin.marginals
        
    grad = np.zeros_like(centers)
    idx = 0
    for i, j in PAIR_INDICES:
        lam = duals[idx]
        if lam > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    return radii, np.sum(radii), grad

def generate_inits(rng):
    """Generates diverse starting configurations."""
    configs = []
    patterns = [
        [5, 5, 5, 5, 6], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
        [4, 5, 6, 5, 6], [6, 5, 5, 5, 5], [5, 6, 4, 6, 5]
    ]
    
    for pat in patterns:
        for scale in [0.95, 1.0, 1.05]:
            c = []
            r0 = 0.095 * scale
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            while len(c) < N:
                c.append(rng.uniform(0.1, 0.9, 2))
            c = np.array(c[:N])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.02, 0.98)
            configs.append(c)
            
    for _ in range(15):
        c = rng.uniform(0.1, 0.9, (N, 2))
        configs.append(c)
        
    for _ in range(10):
        c = rng.uniform(0.05, 0.95, (N, 2))
        corners = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        c[:4] = corners
        c += rng.normal(0, 0.005, c.shape)
        c = np.clip(c, 0.02, 0.98)
        configs.append(c)
        
    return configs

def obj_grad_l_bfgs(x):
    """Objective and gradient for L-BFGS-B optimization."""
    c = np.clip(x.reshape(N, 2), 0.005, 0.995)
    _, s, g = solve_lp_and_gradient(c)
    return -s, -g.flatten()

def optimize_l_bfgs(c0):
    """Runs L-BFGS-B optimization on centers."""
    bounds = [(0.01, 0.99)] * (2 * N)
    res = minimize(obj_grad_l_bfgs, c0.flatten(), method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-10})
    return res.x.reshape(N, 2)

def obj_powell(x):
    """Objective for Powell optimization."""
    c = np.clip(x.reshape(N, 2), 0.005, 0.995)
    _, s, _ = solve_lp_and_gradient(c)
    return -s

def optimize_powell(c0):
    """Runs Powell method optimization on centers."""
    bounds = [(0.01, 0.99)] * (2 * N)
    res = minimize(obj_powell, c0.flatten(), method='Powell', bounds=bounds,
                   options={'maxiter': 1500, 'ftol': 1e-13})
    return res.x.reshape(N, 2)

def obj_local_cd(p, full_centers, idx):
    """Objective for coordinate descent single-circle optimization."""
    temp = full_centers.copy()
    temp[idx] = np.clip(p, 0.005, 0.995)
    _, s, _ = solve_lp_and_gradient(temp)
    return -s

def coordinate_descent(centers, radii, rng):
    """Refines each circle's position independently using Nelder-Mead."""
    c = centers.copy()
    r = radii.copy()
    s_curr = np.sum(r)
    improved = True
    iteration = 0
    
    while improved and iteration < 3:
        improved = False
        iteration += 1
        for i in range(N):
            try:
                res = minimize(obj_local_cd, c[i], args=(c, i), method='Nelder-Mead', 
                               options={'maxiter': 300, 'xatol': 1e-8, 'fatol': 1e-12})
                if -res.fun > s_curr + 1e-9:
                    c[i] = np.clip(res.x, 0.01, 0.99)
                    _, r, _ = solve_lp_and_gradient(c)
                    s_curr = np.sum(r)
                    improved = True
            except Exception:
                pass
    return c, r

def obj_slsqp(v):
    """Objective for joint SLSQP optimization."""
    return -np.sum(v[2 * N:])

def cons_slsqp(v):
    """Constraints for joint SLSQP optimization."""
    cc = v[:2 * N].reshape(N, 2)
    rr = v[2 * N:]
    c_list = [
        cc[:, 0] - rr,
        1.0 - cc[:, 0] - rr,
        cc[:, 1] - rr,
        1.0 - cc[:, 1] - rr
    ]
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(cc[idx[0]] - cc[idx[1]], axis=1)
    c_list.append(d - (rr[idx[0]] + rr[idx[1]]))
    return np.concatenate(c_list)

def polish_slsqp(centers, radii):
    """Joint SLSQP refinement for precise constraint satisfaction."""
    n = centers.shape[0]
    v0 = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    try:
        res = minimize(obj_slsqp, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons_slsqp},
                       options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        if np.min(cons_slsqp(res.x)) >= -1e-9:
            return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
    except Exception:
        pass
    return centers, radii

def repair_packing(centers, radii):
    """Deterministically shrinks radii to guarantee strict validation compliance."""
    radii = radii.copy()
    n = centers.shape[0]
    
    for _ in range(100):
        changed = False
        # Boundary clamp
        for i in range(n):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
                
        # Pairwise overlap resolution
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    return np.maximum(radii, 0.0)

def get_sort_key(item):
    """Helper to sort candidates by sum of radii."""
    return item[2]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
