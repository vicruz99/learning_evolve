import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute constant structure for LP constraints to speed up repeated calls
pair_idx = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(pair_idx)
A_ub_const = np.zeros((N_PAIRS + 4 * N, N))

for k, (i, j) in enumerate(pair_idx):
    A_ub_const[k, i] = 1.0
    A_ub_const[k, j] = 1.0
    
for i in range(N):
    base = N_PAIRS + 4 * i
    A_ub_const[base, i] = 1.0      # r_i <= x_i
    A_ub_const[base + 1, i] = 1.0  # r_i <= 1 - x_i
    A_ub_const[base + 2, i] = 1.0  # r_i <= y_i
    A_ub_const[base + 3, i] = 1.0  # r_i <= 1 - y_i

def solve_lp_and_grad(centers):
    """
    Solves LP for optimal radii given fixed centers.
    Returns radii, objective value (-sum_r), and gradient of objective w.r.t centers.
    """
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Build RHS vector for LP constraints
    b_ub = np.zeros(N_PAIRS + 4 * N)
    for k, (i, j) in enumerate(pair_idx):
        b_ub[k] = dists[i, j]
        
    for i in range(N):
        base = N_PAIRS + 4 * i
        b_ub[base] = centers[i, 0]
        b_ub[base + 1] = 1.0 - centers[i, 0]
        b_ub[base + 2] = centers[i, 1]
        b_ub[base + 3] = 1.0 - centers[i, 1]
        
    # Solve LP: minimize -sum(r) subject to A_ub @ r <= b_ub, r >= 0
    res = linprog(-np.ones(N), A_ub=A_ub_const, b_ub=b_ub, 
                  bounds=[(0.0, None)] * N, method='highs')
                  
    if res.success:
        radii = res.x
        duals = np.zeros_like(b_ub)
        if hasattr(res, 'marginals') and res.marginals is not None:
            duals = res.marginals.ineqlin
        elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
            duals = res.ineqlin.marginals
            
        # Compute gradient using dual variables
        grad = np.zeros_like(centers)
        for k, (i, j) in enumerate(pair_idx):
            lam = duals[k]
            if lam > 1e-8:
                d = dists[i, j]
                if d > 1e-9:
                    v = (centers[i] - centers[j]) / d
                    grad[i] += lam * v
                    grad[j] -= lam * v
                    
        for i in range(N):
            base = N_PAIRS + 4 * i
            grad[i, 0] += duals[base] - duals[base + 1]
            grad[i, 1] += duals[base + 2] - duals[base + 3]
            
        obj_val = -np.sum(radii)
        # We minimize obj_val, so gradient is -grad (since grad points to increase sum_r)
        return radii, obj_val, -grad.flatten()
        
    # Fallback for infeasible or failed LP (should not happen often)
    return np.zeros(N), 0.0, np.zeros(2 * N)

def obj_grad(centers_flat):
    """Wrapper for scipy minimize: returns (f, g)"""
    centers = centers_flat.reshape(N, 2)
    _, val, g = solve_lp_and_grad(centers)
    return val, g

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    
    # --- Phase 1: Diverse Initializations ---
    inits = []
    
    # Hexagonal lattice patterns
    patterns = [
        [5, 5, 5, 5, 6], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
        [4, 5, 6, 5, 6], [5, 4, 6, 6, 5], [6, 5, 5, 6, 4],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5]
    ]
    for pat in patterns:
        for r_est in [0.095, 0.100, 0.105, 0.110]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    c.append([x + rng.normal(0, 0.001), y + rng.normal(0, 0.001)])
                y += r_est * np.sqrt(3.0)
            inits.append(np.array(c[:N]))
            
    # Force-directed spreads
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(800):
            f = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    d_vec = c[i] - c[j]
                    d = np.linalg.norm(d_vec)
                    if d < 0.2 and d > 1e-6:
                        push = (0.2 - d) * 0.02
                        f[i] += d_vec / d * push
                        f[j] -= d_vec / d * push
            c += f
            c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    best_c = None
    best_val = 0.0
    bounds_centers = [(1e-4, 0.999)] * (2 * N)

    for c0 in inits:
        try:
            res = minimize(obj_grad, c0.flatten(), method='L-BFGS-B', jac=True,

            if res.fun < best_val:
                best_val = res.fun
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass

            val_trial = res.fun
            
            delta = val_trial - best_val
            # Accept if better, or probabilistically if worse
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-9)):
                best_val = val_trial
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
        T *= 0.985
        
    # --- Phase 4: Joint SLSQP Polish ---
    _, _, _ = solve_lp_and_grad(best_c)
    best_r, best_val, _ = solve_lp_and_grad(best_c)
    
    v0 = np.concatenate([best_c.flatten(), best_r])
    bounds_slsqp = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    def obj_sl(v):
        return -np.sum(v[2 * N:])
        
    def cons_sl(v):
        c = v[:2 * N].reshape(N, 2)
        r = v[2 * N:]
        con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
        idx = np.triu_indices(N, 1)
        dx = c[idx[0], 0] - c[idx[1], 0]
        dy = c[idx[0], 1] - c[idx[1], 1]
        dr = r[idx[0]] + r[idx[1]]
        con = np.concatenate([con, dx**2 + dy**2 - dr**2])
        return con
        
    for _ in range(4):
        v_pert = v0 + rng.normal(0, 0.0006, v0.shape)
        v_pert = np.clip(v_pert, 0.0, 1.0)
        v_pert[2 * N:] = np.clip(v_pert[2 * N:], 0.0, 0.45)
        try:
            res = minimize(obj_sl, v_pert, method='SLSQP', bounds=bounds_slsqp,
                          constraints={'type': 'ineq', 'fun': cons_sl},
                          options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            if np.min(cons_sl(res.x)) >= -1e-7:
                s = np.sum(res.x[2 * N:])
                if s > -best_val:
                    best_val = -s
                    best_c = res.x[:2 * N].reshape(N, 2).copy()
                    best_r = res.x[2 * N:].copy()
                    v0 = res.x.copy()
        except Exception:
            pass
            
    # Final LP verification
    final_r, final_val, _ = solve_lp_and_grad(best_c)
    if -final_val > -best_val:
        best_r = final_r
        best_val = final_val
        
    # --- Phase 5: Deterministic Repair ---
    centers = best_c.copy()
    radii = best_r.copy()
    
    for _ in range(150):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Fix boundary violations
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
