from scipy.optimize import minimize, linprog

N = 26
PAIRS = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIRS)

# Precompute constant part of A_ub matrix for LP
A_ub_const = np.zeros((N_PAIRS + 4 * N, N))
for k, (i, j) in enumerate(PAIRS):
    A_ub_const[k, i] = 1.0
    A_ub_const[k, j] = 1.0
for i in range(N):
    base = N_PAIRS + 4 * i
    A_ub_const[base, i] = 1.0
    A_ub_const[base + 1, i] = 1.0
    A_ub_const[base + 2, i] = 1.0
    A_ub_const[base + 3, i] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii and dual variables."""
    c = np.clip(centers, 1e-7, 1.0 - 1e-7)
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(N_PAIRS + 4 * N)
    for k, (i, j) in enumerate(PAIRS):
        b_ub[k] = dists[i, j]
    for i in range(N):
        base = N_PAIRS + 4 * i
        b_ub[base] = c[i, 0]
        b_ub[base + 1] = 1.0 - c[i, 0]
        b_ub[base + 2] = c[i, 1]
        b_ub[base + 3] = 1.0 - c[i, 1]
        
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    res = linprog(-np.ones(N), A_ub=A_ub_const, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return None, None
        
    radii = res.x
    duals = np.zeros_like(b_ub)
    if hasattr(res, 'marginals') and res.marginals is not None:
        duals = res.marginals.ineqlin
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = res.ineqlin.marginals
        
    return radii, duals

def compute_grad(centers, duals):
    """Computes exact gradient of sum of radii w.r.t centers using LP duals."""
    grad = np.zeros_like(centers)

                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1

        base = N_PAIRS + 4 * i
        grad[i, 0] += duals[base] - duals[base + 1]
        grad[i, 1] += duals[base + 2] - duals[base + 3]
        
    return grad

def lp_obj_grad(v):
    """Objective and exact gradient for L-BFGS-B optimization of centers."""
    centers = v.reshape(N, 2)
    radii, duals = solve_lp(centers)
    if radii is None:
        return 0.0, np.zeros_like(v)
    obj = -np.sum(radii)
    g = compute_grad(centers, duals)
    return obj, -g.flatten()

def generate_hex_init(pattern, r_est, rng):
    """Generates a hexagonal lattice initialization with specified row counts."""
    c = []
    y = r_est
    for r_idx, cnt in enumerate(pattern):
        shift = r_est if r_idx % 2 == 1 else 0.0
        x = r_est + shift
        for _ in range(cnt):
            if len(c) < N:
                c.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    while len(c) < N:
        c.append(rng.uniform(0.2, 0.8, 2))
    return np.array(c[:N])

def force_init(rng):
    """Generates an evenly spread initial configuration using repulsive forces."""
    c = rng.uniform(0.2, 0.8, (N, 2))
    for _ in range(2000):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                d_vec = c[i] - c[j]
                d = np.linalg.norm(d_vec)
                if d < 0.2 and d > 1e-6:
                    push = (0.2 - d) * 0.1
                    forces[i] += d_vec / d * push
                    forces[j] -= d_vec / d * push
        c += forces
        c = np.clip(c, 0.1, 0.9)
    return c

def obj_joint(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def cons_joint(v):
    """Constraints for SLSQP: boundaries and non-overlap using squared distances."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    i, j = np.triu_indices(N, 1)
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con = np.concatenate([con, dx**2 + dy**2 - dr**2])
    return con

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(200):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
