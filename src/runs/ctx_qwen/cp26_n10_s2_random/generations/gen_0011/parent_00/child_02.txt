import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_and_grad(centers):
    """Solves LP for max sum of radii given fixed centers and returns exact gradient."""
    n = centers.shape[0]
    # Keep centers strictly inside to ensure LP stability
    c = np.clip(centers, 1e-5, 1.0 - 1e-5)
    
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    m = num_pairs + num_bound
    
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = c[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - c[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = c[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - c[i, 1]; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if not res.success:
        return None, 0.0, None
        
    radii = res.x
    try:
        duals = res.marginals.ineqlin
    except AttributeError:
        try:
            duals = res.ineqlin.marginals
        except AttributeError:
            duals = np.zeros(m)
            
    grad = np.zeros((n, 2))
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]
            if lam > 1e-10:
                d = dists[i, j]
                if d > 1e-9:
                    vec = (c[i] - c[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
            idx += 1
            
    for i in range(n):
        mu_L = duals[num_pairs + 4 * i]
        mu_R = duals[num_pairs + 4 * i + 1]
        mu_B = duals[num_pairs + 4 * i + 2]
        mu_T = duals[num_pairs + 4 * i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, np.sum(radii), grad

def lp_obj(v):
    return -solve_lp_and_grad(v.reshape(N, 2))[1]

def lp_jac(v):
    _, _, g = solve_lp_and_grad(v.reshape(N, 2))
    return -g.flatten() if g is not None else np.zeros_like(v)

def generate_inits(rng):
    inits = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
        [4, 5, 6, 5, 6], [5, 4, 6, 6, 5], [6, 5, 5, 6, 4]
    ]
    
    # Hexagonal lattice starts
    for pat in patterns:
        for r_est in [0.092, 0.098, 0.105, 0.112]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            while len(c) < N:
                c.append(rng.uniform(0.15, 0.85, 2))
            inits.append(np.array(c[:N]))
            
    # Force-directed random starts
    for _ in range(12):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(600):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    d_vec = c[i] - c[j]
                    d = np.linalg.norm(d_vec)
                    if d < 0.22 and d > 1e-4:
                        push = (0.22 - d) * 0.08 / (d + 1e-4)
                        forces[i] += d_vec * push
                        forces[j] -= d_vec * push
            c += forces * 0.01
            c = np.clip(c, 0.08, 0.92)
        inits.append(c)
        
    # Corner/Edge biased starts
    for _ in range(8):
        c = rng.uniform(0.2, 0.8, (N, 2))
        c[:4] = [[0.13, 0.13], [0.87, 0.13], [0.13, 0.87], [0.87, 0.87]]
        c[4:8] = [[0.5, 0.13], [0.5, 0.87], [0.13, 0.5], [0.87, 0.5]]
        inits.append(c + rng.normal(0, 0.005, c.shape))
        
    return inits

def coordinate_refinement(centers, rng):
    """Optimizes each circle's position independently."""
    c = centers.copy()
    _, best_s, _ = solve_lp_and_grad(c)
    
    for _ in range(4):
        for i in range(N):
            def obj_2d(xy):
                temp = c.copy()
                temp[i] = np.clip(xy, 0.001, 0.999)
                return -solve_lp_and_grad(temp)[1]
                
            try:
                res = minimize(obj_2d, c[i], method='Nelder-Mead', 
                               options={'maxiter': 400, 'xatol': 1e-8, 'fatol': 1e-12})
                if -res.fun > best_s + 1e-9:
                    c[i] = np.clip(res.x, 0.001, 0.999)
                    _, best_s, _ = solve_lp_and_grad(c)
            except Exception:
                pass
    return c

def repair_packing(centers, radii):
    """Deterministic repair for strict validation compliance."""
    radii = radii.copy()
    for _ in range(80):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds = [(0.005, 0.995)] * (2 * N)
    
    best_c = None
    best_sum = -1.0
    
    inits = generate_inits(rng)
    
    # Phase 1: Multi-start L-BFGS-B with exact LP gradient
    for c0 in inits:

            res = minimize(lp_obj, c0.flatten(), jac=lp_jac, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 2500, 'ftol': 1e-14})
            _, s, _ = solve_lp_and_grad(res.x.reshape(N, 2))
            if s > best_sum:
                best_sum = s
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = inits[0]
        
    # Phase 2: Adaptive Basin Hopping
    T = 0.012
    for step in range(250):
        c_pert = best_c + rng.normal(0, T, best_c.shape)
        c_pert = np.clip(c_pert, 0.01, 0.99)
        
        try:
            res = minimize(lp_obj, c_pert.flatten(), jac=lp_jac, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 1200, 'ftol': 1e-14})
            _, s_pert, _ = solve_lp_and_grad(res.x.reshape(N, 2))
            
            # Accept if better or probabilistically
            if s_pert > best_sum or rng.random() < np.exp((s_pert - best_sum) / max(T * 8, 1e-8)):
                best_sum = s_pert
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
        T *= 0.996
        
    # Phase 3: Coordinate-wise refinement to escape shallow local minima
    best_c = coordinate_refinement(best_c, rng)
    _, best_sum, _ = solve_lp_and_grad(best_c)
    
    # Phase 4: Joint SLSQP Polish
    _, best_r, _ = solve_lp_and_grad(best_c)
    v0 = np.concatenate([best_c.flatten(), best_r])
    
    def obj_joint(v):
        return -np.sum(v[2 * N:])
        
    def cons_joint(v):
        c = v[:2 * N].reshape(N, 2)
        r = v[2 * N:]
        con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
        idx_i, idx_j = np.triu_indices(N, 1)
        dx = c[idx_i, 0] - c[idx_j, 0]
        dy = c[idx_i, 1] - c[idx_j, 1]
        dr = r[idx_i] + r[idx_j]
        con.append(dx**2 + dy**2 - dr**2)
        return np.concatenate(con)
        
    bounds_j = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res_sl = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                          constraints={'type': 'ineq', 'fun': cons_joint},
                          options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})

            s_sl = np.sum(res_sl.x[2 * N:])
            if s_sl > best_sum:
                best_sum = s_sl
                best_c = res_sl.x[:2 * N].reshape(N, 2)
                best_r = res_sl.x[2 * N:]
    except Exception:
        pass
        
    # Phase 5: Final LP verification & strict repair
    lp_r, _, _ = solve_lp_and_grad(best_c)
    if lp_r is not None and np.sum(lp_r) > best_sum:
        best_r = lp_r
        best_sum = np.sum(lp_r)
        
    radii = repair_packing(best_c.copy(), best_r.copy())
    return best_c, radii, float(np.sum(radii))
