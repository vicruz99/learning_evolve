import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIRS = [(i, j) for i in range(N) for j in range(i + 1, N)]
NUM_PAIRS = len(PAIRS)

def solve_lp(centers):

    centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)

    b_ub = np.zeros(NUM_PAIRS + 4 * n)
    
    k = 0
    for i, j in PAIRS:
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]
        k += 1
        
    for i in range(n):
        A_ub[k, i] = 1.0; b_ub[k] = centers[i, 0]; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - centers[i, 0]; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = centers[i, 1]; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - centers[i, 1]; k += 1

            duals = res.ineqlin.marginals
        return res.x, duals
    return None, None

def compute_grad(centers, duals):
    """Computes gradient of sum of radii w.r.t centers using LP duals."""
    n = centers.shape[0]
    grad = np.zeros_like(centers)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)

                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        k += 1
        
    bound_start = NUM_PAIRS
    for i in range(n):
        mu_x0 = duals[bound_start + 4 * i]
        mu_x1 = duals[bound_start + 4 * i + 1]
        mu_y0 = duals[bound_start + 4 * i + 2]
        mu_y1 = duals[bound_start + 4 * i + 3]
        grad[i, 0] += mu_x0 - mu_x1
        grad[i, 1] += mu_y0 - mu_y1
        
    return grad

def lp_obj_grad(x_flat):
    """Objective and exact gradient for L-BFGS-B optimization of centers."""
    centers = x_flat.reshape(N, 2)
    radii, duals = solve_lp(centers)
    if radii is None:
        return 0.0, np.zeros_like(x_flat)
    obj = -np.sum(radii)
    grad = compute_grad(centers, duals)
    return obj, -grad.flatten()

def solve_lp_and_sum(c):
    """Wrapper to get radii and sum from solve_lp."""
    r, _ = solve_lp(c)
    if r is None:
        return r, 0.0, None
    return r, np.sum(r), None

def sub_obj(xy, centers, idx):
    """Objective for coordinate-wise optimization."""
    tmp = centers.copy()
    tmp[idx] = np.clip(xy, 0.001, 0.999)
    _, s, _ = solve_lp_and_sum(tmp)
    return -s

def coord_opt(centers, rng):

    _, best_s, _ = solve_lp_and_sum(best_c)

            except Exception:

def hex_init(pattern, r_est, rng):
    """Generates a hexagonal lattice initialization with specified row counts."""
    c = []
    y = r_est
    for r_idx, cnt in enumerate(pattern):
        shift = r_est if r_idx % 2 == 1 else 0.0
        x = r_est + shift
        for _ in range(cnt):
            if len(c) < N:
                c.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    while len(c) < N:
        c.append(rng.uniform(0.2, 0.8, 2))
    return np.array(c[:N])

def force_init(rng):
    """Generates an evenly spread initial configuration using repulsive forces."""
    c = rng.uniform(0.2, 0.8, (N, 2))
    for _ in range(1500):
        f = np.zeros_like(c)
        for i in range(N):

                d_vec = c[i] - c[j]
                d = np.linalg.norm(d_vec)
                if d < 0.25 and d > 1e-6:
                    push = (0.25 - d) * 0.05
                    f[i] += d_vec / d * push
                    f[j] -= d_vec / d * push
        c += f
        c = np.clip(c, 0.1, 0.9)
    return c

def repair(centers, radii):

                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def obj_joint(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def cons_joint(v):
    """Constraints for SLSQP: boundaries and non-overlap using squared distances."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    i_idx, j_idx = np.triu_indices(N, 1)
    dx = c[i_idx, 0] - c[j_idx, 0]
    dy = c[i_idx, 1] - c[j_idx, 1]
    dr = r[i_idx] + r[j_idx]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)
