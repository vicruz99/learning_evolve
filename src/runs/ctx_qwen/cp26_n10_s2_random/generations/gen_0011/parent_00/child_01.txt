N = 26

def solve_lp_and_grad(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns radii, duals, gradient, and sum_radii.
    """
    c = np.clip(centers, 1e-6, 1.0 - 1e-6)
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    np.fill_diagonal(dists, 0.0)

    num_bound = 4 * N
    
    A_ub = np.zeros((num_pairs + num_bound, N))
    b_ub = np.zeros(num_pairs + num_bound)

    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(N):
        A_ub[idx, i] = 1.0; b_ub[idx] = c[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - c[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = c[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - c[i, 1]; idx += 1
        
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, 
                      bounds=[(0.0, 0.5)] * N, method='highs')
        if not res.success:
            return None, None, None, 0.0
            
        radii = res.x
        sum_r = np.sum(radii)

        try:
            duals = res.marginals.ineqlin
        except AttributeError:
            try:
                duals = res.ineqlin.marginals
            except AttributeError:
                pass
                
        grad = np.zeros_like(c)
        idx = 0
        for i in range(N):
            for j in range(i + 1, N):

            grad[i, 1] += mu_B - mu_T
            
        return radii, duals, grad, sum_r
        
    except Exception:
        return None, None, None, 0.0

def lp_obj_and_grad(centers_flat):
    """Objective and gradient for L-BFGS-B center optimization."""
    c = centers_flat.reshape(N, 2)
    r, _, g, s = solve_lp_and_grad(c)
    if r is None:
        return 0.0, np.zeros_like(centers_flat)
    return -s, -g.flatten()

def generate_hex_init(rng, pattern, r_est, noise_std=0.002):
    """Generates a hexagonal lattice initialization."""
    c = []
    y = r_est
    for r_idx, cnt in enumerate(pattern):
        shift = r_est if r_idx % 2 == 1 else 0.0
        x = r_est + shift
        for _ in range(cnt):
            if len(c) < N:
                c.append([x + rng.normal(0, noise_std), y + rng.normal(0, noise_std)])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    while len(c) < N:
        c.append(rng.uniform(0.2, 0.8, 2))
    return np.array(c[:N])

def force_directed_init(rng):
    """Generates an evenly spread initial configuration using repulsive forces."""
    c = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(1500):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                d_vec = c[i] - c[j]
                d = np.linalg.norm(d_vec)
                if d < 0.22 and d > 1e-6:
                    push = (0.22 - d) * 0.06
                    forces[i] += d_vec / d * push
                    forces[j] -= d_vec / d * push
        c += forces
        c = np.clip(c, 0.1, 0.9)
    return c

def obj_joint(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def cons_joint_sq(v):
    """Constraints for SLSQP: boundaries and non-overlap using squared distances."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    
    i, j = np.triu_indices(N, 1)
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(150):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
