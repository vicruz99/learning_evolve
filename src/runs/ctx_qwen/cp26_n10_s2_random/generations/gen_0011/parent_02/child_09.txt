import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute constant structure of the LP constraint matrix
NUM_PAIRS = N * (N - 1) // 2
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

TRIU_I, TRIU_J = np.triu_indices(N, 1)

def solve_lp_and_grad(centers):

                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))

        k += 1

    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:

def gradient_ascent(centers, rng, max_iter=1500, step_init=0.01):

    best_c = c.copy()

            break

            step *= 1.12

def sa_optimize(centers, rng, steps=2000, T0=0.012):

def obj_coord(centers, idx, xy):
    """Objective for coordinate-wise refinement."""
    temp = centers.copy()
    temp[idx] = np.clip(xy, 1e-5, 1.0 - 1e-5)
    _, s, _ = solve_lp_and_grad(temp)
    return -s

def coordinate_refine(centers, rng, passes=3):
    """Optimizes each circle's position independently."""
    c = centers.copy()
    for _ in range(passes):
        for i in range(N):
            try:
                res = minimize(obj_coord, c[i], args=(c, i), method='Nelder-Mead', 
                               options={'maxiter': 300, 'xatol': 1e-8, 'fatol': 1e-12})
                c[i] = res.x
            except Exception:
                pass
    return c

def slsqp_polish(c0, r0):
    """Joint SLSQP optimization of centers and radii."""
    v0 = np.concatenate([c0.flatten(), r0])
    
    def obj_sl(v):
        return -np.sum(v[2*N:])

    def cons_sl(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        con = []
        con.append(c[:, 0] - r)
        con.append(1.0 - c[:, 0] - r)
        con.append(c[:, 1] - r)
        con.append(1.0 - c[:, 1] - r)
        dx = c[TRIU_I, 0] - c[TRIU_J, 0]
        dy = c[TRIU_I, 1] - c[TRIU_J, 1]
        dr = r[TRIU_I] + r[TRIU_J]
        con.append(dx**2 + dy**2 - dr**2)
        return np.concatenate(con)
        
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(obj_sl, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons_sl},
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if np.min(cons_sl(res.x)) >= -1e-7:
            return res.x[:2*N].reshape(N, 2), res.x[2*N:], -res.fun
    except Exception:
        pass
    return c0, r0, np.sum(r0)

def generate_hex_init(pat, r0, rng):
    """Generates a hexagonal lattice initialization."""
    c = []
    y = r0
    for ri, cnt in enumerate(pat):
        sh = r0 if ri % 2 == 1 else 0.0
        x = r0 + sh
        for _ in range(cnt):
            if len(c) < N:
                c.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
    return np.clip(np.array(c[:N]), 0.05, 0.95)

def force_init(rng):
    """Generates a well-spaced configuration via repulsive forces."""
    c = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(600):

        for i in range(N):

                dv = c[i] - c[j]
                d = np.linalg.norm(dv)
                if d < 0.22 and d > 1e-4:
                    push = (0.22 - d) * 0.04 / (d + 1e-4)
                    f[i] += dv / d * push
                    f[j] -= dv / d * push
        c += f
        c = np.clip(c, 0.05, 0.95)
    return c

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
