N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):

                    np.minimum(c[:, 1], 1.0 - c[:, 1]))

    k = 0

        b[k] = dists[i, j]
        k += 1

        b[k] = c[i, 1]; k += 1
        b[k] = 1.0 - c[i, 1]; k += 1

        return np.zeros(N), 0.0, np.zeros_like(c)

        grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]

def obj_grad(v):
    """Objective and gradient for L-BFGS-B: minimizes negative sum of radii."""
    c = v.reshape(N, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    _, val, g = solve_lp_and_grad(c)
    return -val, -g.flatten()

def slsqp_polish(c0, r0):
    """Joint SLSQP optimization of centers and radii."""
    v0 = np.concatenate([c0.flatten(), r0])
    
    def obj(v):
        return -np.sum(v[2*N:])

    def cons(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        con = []
        con.append(c[:, 0] - r)
        con.append(1.0 - c[:, 0] - r)
        con.append(c[:, 1] - r)
        con.append(1.0 - c[:, 1] - r)
        dx = c[TRIU_I, 0] - c[TRIU_J, 0]
        dy = c[TRIU_I, 1] - c[TRIU_J, 1]
        dr = r[TRIU_I] + r[TRIU_J]
        con.append(dx**2 + dy**2 - dr**2)
        return np.concatenate(con)
        
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})

        if np.min(c_cons) >= -1e-7:
            return res.x[:2*N].reshape(N, 2), res.x[2*N:], -res.fun
    except Exception:
        pass
    return c0, r0, np.sum(r0)

def generate_hex_centers(pat, r_est, rng):
    """Generates centers from a hexagonal pattern."""
    c = []
    y = r_est
    for ri, cnt in enumerate(pat):
        sh = r_est if ri % 2 == 1 else 0.0
        x = r_est + sh
        for _ in range(cnt):
            if len(c) < N:
                c.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    return np.clip(np.array(c[:N]), 0.05, 0.95)
