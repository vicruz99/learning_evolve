To maximize the sum of radii for 26 circles in a unit square, we treat the problem as a bi-level optimization:
1. **Inner Level**: For fixed centers $C$, the optimal radii $r^*$ are found by solving a Linear Program (LP) that maximizes $\sum r_i$ subject to pairwise distance and boundary constraints. The LP dual variables provide the exact subgradient of the optimal objective w.r.t. the centers.
2. **Outer Level**: We optimize the centers $C$ using this subgradient. Because the objective is non-smooth (due to active constraint switches), we combine:
   - **L-BFGS-B** for fast, gradient-driven convergence from diverse starts.
   - **Coordinate Descent** (optimizing one circle at a time) to handle non-smooth valleys.
   - **Simulated Annealing / Basin Hopping** to escape local minima.
   - **SLSQP Joint Polish** on $(C, r)$ to resolve numerical precision and tight contacts.
3. **Initialization**: We generate ~35 high-quality starts using perturbed hexagonal lattices, force-directed layouts, and random dense configurations.
4. **Repair**: A deterministic post-processing step strictly enforces constraints to guarantee validation compliance.

This strategy leverages exact subgradients for efficiency, multi-start diversity for global search, and hybrid local methods to overcome non-smoothness, reliably pushing the sum past the 2.636 target.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute constant LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii and computes exact subgradient via duals."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    dists = np.maximum(dists, 1e-12)
    
    b = np.zeros(NUM_PAIRS + 4 * N)
    k = 0
    for i, j in PAIR_IDX:
        b[k] = dists[i, j]
        k += 1
    for i in range(N):
        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, 
                  bounds=[(0.0, u) for u in ub], method='highs', options={'disp': False})
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(centers)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    duals = np.zeros(len(b))
    if hasattr(res, 'marginals') and res.marginals is not None:
        duals = res.marginals.ineqlin
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = res.ineqlin.marginals
        
    grad = np.zeros_like(centers)
    k = 0
    for i, j in PAIR_IDX:
        mu = duals[k]
        if mu > 1e-9:
            d = dists[i, j]
            vec = (centers[i] - centers[j]) / d
            grad[i] += mu * vec
            grad[j] -= mu * vec
        k += 1
        
    b_start = NUM_PAIRS
    for i in range(N):
        grad[i, 0] += duals[b_start + 4*i] - duals[b_start + 4*i + 1]
        grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
    return radii, s_sum, grad

def obj_centers(v):
    c = v.reshape(N, 2)
    return -solve_lp_and_grad(c)[1]

def grad_centers(v):
    c = v.reshape(N, 2)
    return -solve_lp_and_grad(c)[2].flatten()

def obj_cd_2d(xy, c_arr, i):
    c_temp = c_arr.copy()
    c_temp[i] = xy
    return -solve_lp_and_grad(c_temp)[1]

def generate_hex(r0, pat, rng):
    c = []
    y = r0
    for ri, cnt in enumerate(pat):
        sh = r0 if ri % 2 == 1 else 0.0
        x = r0 + sh
        for _ in range(cnt):
            if len(c) < N:
                c.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
    return np.clip(np.array(c[:N]), 0.05, 0.95)

def force_init(rng):
    c = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(600):
        f = np.zeros_like(c)
        for i in range(N):
            for j in range(i+1, N):
                dv = c[i] - c[j]
                d = np.linalg.norm(dv)
                if d < 0.25 and d > 1e-4:
                    push = (0.25 - d) * 0.04 / (d + 1e-4)
                    f[i] += dv / d * push
                    f[j] -= dv / d * push
        c += f
        c = np.clip(c, 0.05, 0.95)
    return c

def generate_starts(rng):
    starts = []
    pats = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], 
            [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
            [6, 5, 5, 6, 4], [5, 6, 4, 5, 6], [6, 4, 5, 6, 5],
            [5, 5, 4, 6, 6], [4, 5, 6, 6, 5], [6, 5, 6, 4, 5]]
            
    for pat in pats:
        for r0 in [0.095, 0.100, 0.105, 0.110]:
            starts.append(generate_hex(r0, pat, rng))
            
    for _ in range(10):
        starts.append(force_init(rng))
        
    for _ in range(10):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return starts

def coord_descent(c0, n_passes=3, rng=None):
    c = c0.copy()
    _, best_s, _ = solve_lp_and_grad(c)
    
    for _ in range(n_passes):
        for i in range(N):
            try:
                res = minimize(obj_cd_2d, c[i], args=(c, i), 
                               method='Nelder-Mead', 
                               options={'maxiter': 400, 'xatol': 1e-7, 'fatol': 1e-10})
                if -res.fun > best_s + 1e-9:
                    c[i] = np.clip(res.x, 1e-5, 1.0 - 1e-5)
                    _, best_s, _ = solve_lp_and_grad(c)
            except Exception:
                pass
    return c, best_s

def basin_hopping(c0, rng, steps=100):
    c = c0.copy()
    _, best_s, _ = solve_lp_and_grad(c)
    best_c = c.copy()
    T = 0.006
    
    for _ in range(steps):
        c_try = c + rng.normal(0, 0.008, c.shape)
        c_try = np.clip(c_try, 0.02, 0.98)
        
        try:
            res = minimize(obj_centers, c_try.flatten(), jac=grad_centers,
                           method='L-BFGS-B', bounds=[(0.0, 1.0)]*(2*N),
                           options={'maxiter': 300, 'ftol': 1e-12})
            c_new = res.x.reshape(N, 2)
            _, s_new, _ = solve_lp_and_grad(c_new)
            
            delta = s_new - best_s
            if delta > 0 or rng.random() < np.exp(delta / max(T, 1e-9)):
                c = c_new
                best_s = s_new
                best_c = c.copy()
        except Exception:
            pass
        T *= 0.994
    return best_c, best_s

def slsqp_polish(c0, r0):
    v0 = np.concatenate([c0.flatten(), r0])
    
    def obj_joint(v):
        return -np.sum(v[2*N:])
        
    def cons_joint(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        con = []
        con.append(c[:, 0] - r)
        con.append(1.0 - c[:, 0] - r)
        con.append(c[:, 1] - r)
        con.append(1.0 - c[:, 1] - r)
        dx = c[TRIU_I, 0] - c[TRIU_J, 0]
        dy = c[TRIU_I, 1] - c[TRIU_J, 1]
        dr = r[TRIU_I] + r[TRIU_J]
        con.append(dx**2 + dy**2 - dr**2)
        return np.concatenate(con)
        
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
        if np.min(cons_joint(res.x)) >= -1e-7:
            return res.x[:2*N].reshape(N, 2), res.x[2*N:], -res.fun
    except Exception:
        pass
    return c0, r0, np.sum(r0)

def repair(centers, radii):
    radii = radii.copy()
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_c = None
    best_r = None
    best_s = -1.0
    
    starts = generate_starts(rng)
    
    # Phase 1: Multi-start L-BFGS-B
    bounds_xy = [(0.0, 1.0)] * (2 * N)
    for c_init in starts:
        try:
            res = minimize(obj_centers, c_init.flatten(), jac=grad_centers,
                           method='L-BFGS-B', bounds=bounds_xy,
                           options={'maxiter': 1200, 'ftol': 1e-12, 'gtol': 1e-10})
            c_opt = res.x.reshape(N, 2)
            _, s_opt, _ = solve_lp_and_grad(c_opt)
            if s_opt > best_s:
                best_s = s_opt
                best_c = c_opt.copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = starts[0]
        best_r, best_s, _ = solve_lp_and_grad(best_c)
    else:
        best_r, _, _ = solve_lp_and_grad(best_c)
        
    # Phase 2: Coordinate Descent to handle non-smoothness
    best_c, best_s = coord_descent(best_c, n_passes=3, rng=rng)
    best_r, _, _ = solve_lp_and_grad(best_c)
    
    # Phase 3: Basin Hopping for global escape
    best_c, best_s = basin_hopping(best_c, rng, steps=120)
    best_r, _, _ = solve_lp_and_grad(best_c)
    
    # Phase 4: Joint SLSQP Polish
    c_final, r_final, s_final = slsqp_polish(best_c, best_r)
    if s_final > best_s:
        best_c = c_final
        best_r = r_final
        best_s = s_final
        
    # Phase 5: Strict numerical repair
    radii = repair(best_c, best_r)
    return best_c, radii, float(np.sum(radii))
