import numpy as np
from scipy.optimize import minimize

N = 26

def compute_radii(centers):
    """Compute maximum valid radii for a fixed set of centers."""
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    
    # Distance to boundaries: min(x, 1-x, y, 1-y)
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    dists[np.arange(n), np.arange(n)] = np.inf  # Ignore self-distance
    
    # Radius limited by half the distance to the nearest neighbor
    min_dists = np.min(dists, axis=1)
    r_pair = 0.5 * min_dists
    
    return np.minimum(r_bound, r_pair)

def objective(centers_flat):
    """Objective function: minimize negative sum of radii."""
    centers = centers_flat.reshape(-1, 2)
    centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
    radii = compute_radii(centers)
    return -np.sum(radii)

def run_packing():
    np.random.seed(42)
    bounds = [(1e-7, 1.0 - 1e-7)] * (2 * N)
    
    best_val = np.inf
    best_centers = None
    
    inits = []
    
    # 1. Structured hexagonal patterns known to pack N=26 well
    row_configs = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 5, 5, 6]
    ]
    
    for cfg in row_configs:
        for r_init in [0.10, 0.105, 0.108, 0.11]:
            pts = []
            y = r_init
            row = 0
            dy = np.sqrt(3) * r_init
            for count in cfg:
                if len(pts) >= N: break
                x = r_init if row % 2 == 0 else 2 * r_init
                for _ in range(count):
                    if len(pts) >= N: break
                    pts.append([x, y])
                    x += 2 * r_init
                y += dy
                row += 1
            inits.append(np.array(pts[:N]).flatten())
            
    # 2. Perturbed hex configs to break symmetry
    for seed in range(15):
        rng = np.random.default_rng(seed)
        idx = rng.integers(len(inits))
        c = inits[idx].copy().reshape(-1, 2)
        c += rng.normal(0, 0.004, c.shape)
        c = np.clip(c, 0.05, 0.95)
        inits.append(c.flatten())
        
    # 3. Random starts for diverse basin exploration
    for seed in range(20):
        rng = np.random.default_rng(seed)
        c = rng.uniform(0.15, 0.85, (N, 2))
        inits.append(c.flatten())
        
    # Phase 1: Multi-start Powell optimization
    for x0 in inits:
        res = minimize(objective, x0, method='Powell', bounds=bounds,
                       options={'maxiter': 15000, 'ftol': 1e-14, 'xtol': 1e-14})
        if res.fun < best_val:
            best_val = res.fun
            best_centers = res.x.reshape(-1, 2)
            
        # Local perturbations around each promising start
        for _ in range(4):
            pert = best_centers.copy()
            pert += np.random.normal(0, 0.003, pert.shape)
            pert = np.clip(pert, 1e-7, 1.0 - 1e-7)
            try:
                res2 = minimize(objective, pert.flatten(), method='Powell', bounds=bounds,
                                options={'maxiter': 8000})
                if res2.fun < best_val:
                    best_val = res2.fun
                    best_centers = res2.x.reshape(-1, 2)
            except Exception:
                pass

    # Phase 2: Intensive basin hopping / cluster perturbation
    for _ in range(80):
        pert = best_centers.copy()
        num_move = np.random.randint(1, 5)
        idx = np.random.choice(N, num_move, replace=False)
        pert[idx] += np.random.normal(0, 0.008, (num_move, 2))
        pert = np.clip(pert, 1e-7, 1.0 - 1e-7)
        try:
            res = minimize(objective, pert.flatten(), method='Powell', bounds=bounds,
                           options={'maxiter': 6000})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(-1, 2)
        except Exception:
            pass

    # Compute final radii
    radii = compute_radii(best_centers)
    
    # Phase 3: Deterministic repair for strict validation compliance
    for _ in range(30):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            x, y = best_centers[i]
            r_bound = min(x, 1.0 - x, y, 1.0 - y)
            if radii[i] > r_bound + 1e-12:
                radii[i] = r_bound
                changed = True
                
        # Resolve overlaps by proportional shrinking
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return best_centers, radii, float(np.sum(radii))
