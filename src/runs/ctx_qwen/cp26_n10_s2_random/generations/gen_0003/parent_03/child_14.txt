import numpy as np
from scipy.optimize import minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)

def compute_constraints(vars_flat):
    """
    Computes boundary and non-overlap constraints.
    Returns a 1D array where each element must be >= 0.
    """
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    xi, xj = x[IDX_I], x[IDX_J]
    yi, yj = y[IDX_I], y[IDX_J]
    ri, rj = r[IDX_I], r[IDX_J]
    c.append((xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2)
    
    return np.concatenate(c)

def compute_objective(vars_flat):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars_flat[2::3])

def run_packing():

    # ==========================================

    # ==========================================
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.02)
    
    # Hexagonal lattice initialization
    idx = 0
    y = 0.12
    row = 0
    while idx < N:
        shift = (0.5 if row % 2 == 1 else 0.0) * 0.16
        x = 0.08 + shift
        while x <= 0.92 and idx < N:
            centers[idx] = [x, y]
            idx += 1
            x += 0.16
        y += 0.14
        row += 1
        
    # Break symmetry with small random noise
    centers += np.random.uniform(-0.005, 0.005, (N, 2))
    centers = np.clip(centers, 0.05, 0.95)
    
    velocities = np.zeros_like(centers)
    k_rep = 800.0      # Repulsion stiffness
    k_bnd = 2000.0     # Boundary stiffness
    dt = 0.004         # Time step
    damp = 0.88        # Velocity damping
    growth = 1.0002    # Radius growth factor when stable
    
    for step in range(40000):
        # Vectorized pairwise distance and overlap calculation
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        min_d = radii[:, None] + radii[None, :]
        overlap = np.maximum(0, min_d - dists)
        
        # Repulsive forces from overlaps
        dir_v = diff / (dists[:, :, None] + 1e-12)
        forces = np.sum(overlap[:, :, None] * dir_v * k_rep, axis=1)
        
        # Boundary repulsion forces
        for i in range(N):
            r = radii[i]
            if centers[i, 0] < r: forces[i, 0] += k_bnd * (r - centers[i, 0])
            if centers[i, 0] > 1 - r: forces[i, 0] -= k_bnd * (centers[i, 0] - (1 - r))
            if centers[i, 1] < r: forces[i, 1] += k_bnd * (r - centers[i, 1])
            if centers[i, 1] > 1 - r: forces[i, 1] -= k_bnd * (centers[i, 1] - (1 - r))
            
        # Verlet integration step
        velocities = damp * velocities + forces * dt
        centers += velocities * dt
        centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
        
        # Grow radii when packing is stable
        if np.max(overlap) < 1e-6:
            radii *= growth

    sim_r = np.zeros(N)
    for i in range(N):
        db = min(centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1])
        dn = np.min([np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1]) for j in range(N) if i!=j])
        sim_r[i] = min(db, dn/2.0)
    sim_r *= 0.995  # Shrink slightly to ensure strict feasibility for SLSQP
        
    # ==========================================
    # Phase 2: Constrained Refinement (SLSQP)
    # ==========================================
    x0 = np.zeros(3*N)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]

    # Phase 3: Deterministic Repair & Validation

        # Clamp to boundaries
        for i in range(N):

            if r > mr + 1e-9:
                final_radii[i] = mr
                changed = True

        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(final_centers[i,0]-final_centers[j,0], final_centers[i,1]-final_centers[j,1])

                    ov = (final_radii[i] + final_radii[j] - d)/2.0 + 1e-9
                    final_radii[i] -= ov
                    final_radii[j] -= ov
                    changed = True
                    
        if not changed:
            break
            
    final_radii = np.maximum(final_radii, 0.0)
    
    return final_centers, final_radii, float(np.sum(final_radii))
