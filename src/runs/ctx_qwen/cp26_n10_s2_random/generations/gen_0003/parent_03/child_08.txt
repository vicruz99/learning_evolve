import numpy as np
from scipy.optimize import minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)

def objective(vars_flat):
    """Minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def constraint_fun(vars_flat):
    """
    Returns a 1D array of constraint violations.
    SLSQP expects g(x) >= 0.
    """
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    c = []
    # Boundary constraints: x >= r, x + r <= 1, y >= r, y + r <= 1
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Non-overlap constraints: dist(i, j) >= r_i + r_j
    # Using direct Euclidean distance avoids zero-gradient issues at contact points
    dx = x[IDX_I] - x[IDX_J]
    dy = y[IDX_I] - y[IDX_J]
    dr = r[IDX_I] + r[IDX_J]
    dists = np.sqrt(dx**2 + dy**2)
    c.append(dists - dr)
    
    return np.concatenate(c)

def get_bounds():
    """Variable bounds for x, y in [0,1] and r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def compute_feasible_radii(centers):
    """Computes strictly feasible radii for a given set of centers."""
    r_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_pair = 0.5 * np.min(dists, axis=1)
    # Apply 5% safety margin to guarantee strict feasibility initially
    return np.minimum(r_bound, r_pair) * 0.95

def make_hex_init(r_est, noise=0.005, seed=42):
    """Generates a perturbed hexagonal lattice initialization."""
    rng = np.random.default_rng(seed)
    pts = []
    y = r_est
    row = 0
    while len(pts) < N:
        x_start = r_est if row % 2 == 0 else 2 * r_est
        x = x_start
        while x + r_est <= 1.0 and len(pts) < N:
            pts.append([x, y])
            x += 2 * r_est
        y += np.sqrt(3) * r_est
        row += 1
    pts = np.array(pts[:N])
    pts += rng.normal(0, noise, pts.shape)
    pts = np.clip(pts, 0.01, 0.99)
    rs = compute_feasible_radii(pts)
    v = np.zeros(3 * N)
    v[0::3] = pts[:, 0]
    v[1::3] = pts[:, 1]
    v[2::3] = rs
    return v

def run_packing():
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    best_val = -np.inf
    best_v = None
    
    # Phase 1: Generate diverse feasible initial configurations
    starts = []
    for r in [0.09, 0.095, 0.10, 0.105]:
        for seed in range(8):
            starts.append(make_hex_init(r, noise=0.008, seed=seed))
            
    for seed in range(15):
        rng = np.random.default_rng(seed)
        pts = rng.uniform(0.1, 0.9, (N, 2))
        rs = compute_feasible_radii(pts)
        v = np.zeros(3 * N)
        v[0::3] = pts[:, 0]
        v[1::3] = pts[:, 1]
        v[2::3] = rs
        starts.append(v)
        
    # Phase 2: Multi-start optimization
    for x0 in starts:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
            c_vals = constraint_fun(res.x)
            if np.all(c_vals >= -1e-7):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Iterative Basin-Hopping Refinement
    if best_v is not None:
        curr_v = best_v.copy()
        for step in range(50):
            pert = curr_v.copy()

            pert[0::3] += np.random.normal(0, 0.0015, N)
            pert[1::3] += np.random.normal(0, 0.0015, N)
            pert[0::3] = np.clip(pert[0::3], 0.01, 0.99)
            pert[1::3] = np.clip(pert[1::3], 0.01, 0.99)

            pert[2::3] *= 1.001
            
            try:

                c_vals = constraint_fun(res.x)
                if np.all(c_vals >= -1e-7):

            except Exception:
                pass
                
    centers = best_v[:2*N].reshape(N, 2)
    radii = best_v[2*N:].copy()

    for _ in range(30):
        changed = False

                radii[i] = max_r

                    radii[i] -= overlap
                    radii[j] -= overlap
                    changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
