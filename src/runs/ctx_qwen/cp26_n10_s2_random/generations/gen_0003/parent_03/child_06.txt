import numpy as np
from scipy.optimize import minimize

N = 26

def compute_radii(centers):
    n = centers.shape[0]
    # Distance to boundaries
    xb = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    yb = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    r_bound = np.minimum(xb, yb)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_pair = 0.5 * np.min(dists, axis=1)
    
    return np.minimum(r_bound, r_pair)

def objective(x):
    centers = x.reshape(-1, 2)
    # Keep centers strictly inside the square to ensure valid radii
    centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
    r = compute_radii(centers)
    return -np.sum(r)

def generate_starts():
    starts = []
    rng = np.random.default_rng(42)
    
    # 1. Hexagonal lattices with various densities and shifts
    for seed in range(15):
        r_est = 0.09 + rng.uniform(-0.01, 0.01)
        dy = r_est * np.sqrt(3)
        dx = 2 * r_est
        centers = []
        row = 0
        while len(centers) < N:
            y = r_est + row * dy
            if y + r_est > 1.0: break
            offset = (dx / 2.0) if row % 2 == 1 else 0.0
            x = r_est + offset
            while x + r_est <= 1.0 and len(centers) < N:
                centers.append([x, y])
                x += dx
            row += 1
        while len(centers) < N:
            centers.append(rng.uniform(0.2, 0.8, 2))
        centers = np.array(centers[:N])
        centers += rng.normal(0, 0.005, centers.shape)
        starts.append(np.clip(centers, 0.02, 0.98).flatten())
        
    # 2. Corner and edge focused configurations
    for seed in range(10):
        centers = rng.uniform(0.2, 0.8, (N, 2))
        corners = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
        centers[:4] = corners
        starts.append(centers.flatten())
        
    # 3. Dense random with rejection sampling
    for seed in range(10):
        centers = []
        att = 0
        while len(centers) < N and att < 5000:
            pt = rng.uniform(0.15, 0.85, 2)
            if all(np.hypot(pt[0]-c[0], pt[1]-c[1]) > 0.1 for c in centers):
                centers.append(pt)
            att += 1
        while len(centers) < N:
            centers.append(rng.uniform(0.2, 0.8, 2))
        starts.append(np.array(centers[:N]).flatten())
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_val = np.inf
    best_centers = None
    
    starts = generate_starts()
    
    # Phase 1: Initial multi-start optimization
    for s in starts:
        try:
            res = minimize(objective, s, method='Powell',
                           options={'maxiter': 3000, 'ftol': 1e-12, 'xtol': 1e-12})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(-1, 2)
        except Exception:
            pass
            
    # Phase 2: Basin hopping / Perturbation refinement to escape local minima
    for step in range(40):
        pert_size = 0.01 * (0.9 ** step)
        pert = best_centers + np.random.normal(0, pert_size, best_centers.shape)
        pert = np.clip(pert, 0.02, 0.98)
        try:
            res = minimize(objective, pert.flatten(), method='Powell',
                           options={'maxiter': 2000, 'ftol': 1e-12})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(-1, 2)
        except Exception:
            pass
            
    # Phase 3: Compute final radii and deterministic repair
    radii = compute_radii(best_centers)
    
    # Iterative shrink to guarantee strict non-overlap within tolerance
    for _ in range(5):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                req = radii[i] + radii[j]
                if d < req - 1e-9:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed: break
        
    # Clamp to boundaries strictly
    for i in range(N):
        x, y = best_centers[i]
        max_r = min(x, 1.0-x, y, 1.0-y)
        if radii[i] > max_r + 1e-9:
            radii[i] = max_r
            
    radii = np.maximum(radii, 0.0)
    return best_centers, radii, float(np.sum(radii))
