import numpy as np
from scipy.optimize import minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)

def compute_constraints(vars_flat):
    """
    Computes all boundary and non-overlap constraints.
    Returns a 1D array where each element must be >= 0.
    """
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    c = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    xi, xj = x[IDX_I], x[IDX_J]
    yi, yj = y[IDX_I], y[IDX_J]
    ri, rj = r[IDX_I], r[IDX_J]
    c.append((xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2)
    
    return np.concatenate(c)

def compute_objective(vars_flat):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars_flat[2::3])

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def make_init(centers, scale=0.9):
    """Assigns strictly feasible radii to centers based on distances to boundaries and neighbors."""
    centers = np.clip(centers, 0.001, 0.999)
    rs = np.zeros(N)
    for i in range(N):
        d_bound = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        d_neighbor = np.inf
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < d_neighbor:
                    d_neighbor = d
        rs[i] = min(d_bound, d_neighbor / 2.0) * scale
        
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = rs
    return v

def generate_hex(seed, scale=0.9):
    """Generates initial configuration from a perturbed hexagonal lattice."""
    rng = np.random.default_rng(seed)
    centers = []
    r0 = 0.105 + rng.uniform(-0.012, 0.012)
    y = r0
    row = 0
    while len(centers) < N:
        shift = (r0 if row % 2 == 1 else 0.0)
        x = r0 + shift
        while x + r0 <= 1.0 and len(centers) < N:
            centers.append([x + rng.uniform(-0.008, 0.008), y + rng.uniform(-0.008, 0.008)])
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
    while len(centers) < N:
        centers.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])
    return make_init(np.array(centers[:N]), scale)

def generate_corner_heavy(seed, scale=0.9):
    """Generates initial configuration with larger circles pushed into corners."""
    rng = np.random.default_rng(seed)
    centers = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
    r0 = 0.09
    y = r0 + 0.22
    row = 0
    while len(centers) < N:
        shift = (r0 if row % 2 == 1 else 0.0)
        x = r0 + shift + 0.22
        while x + r0 <= 0.78 and len(centers) < N:
            centers.append([x + rng.uniform(-0.005, 0.005), y + rng.uniform(-0.005, 0.005)])
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
    while len(centers) < N:
        centers.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])
    return make_init(np.array(centers[:N]), scale)

def generate_random_dense(seed, scale=0.9):
    """Generates initial configuration using rejection sampling for uniform density."""
    rng = np.random.default_rng(seed)
    centers = []
    attempts = 0
    while len(centers) < N and attempts < 20000:
        pt = rng.uniform(0.05, 0.95, 2)
        ok = True
        for c in centers:
            if np.hypot(pt[0]-c[0], pt[1]-c[1]) < 0.11:
                ok = False
                break
        if ok:
            centers.append(pt)
        attempts += 1
    while len(centers) < N:
        centers.append(rng.uniform(0.1, 0.9, 2))
    return make_init(np.array(centers[:N]), scale)

def generate_grid_plus_one(seed, scale=0.9):
    """Generates a 5x5 grid with an extra circle in the central gap."""
    rng = np.random.default_rng(seed)
    centers = []
    for i in range(5):
        for j in range(5):
            centers.append([0.1 + i*0.2 + rng.uniform(-0.005, 0.005), 
                            0.1 + j*0.2 + rng.uniform(-0.005, 0.005)])
    centers.append([0.4 + rng.uniform(-0.005, 0.005), 0.4 + rng.uniform(-0.005, 0.005)])
    return make_init(np.array(centers[:N]), scale)

def run_packing():
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_v = None
    best_val = -np.inf
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    for i in range(15):
        inits.append(generate_hex(i))
    for i in range(10):
        inits.append(generate_corner_heavy(i))
    for i in range(10):
        inits.append(generate_random_dense(i))
    for i in range(5):
        inits.append(generate_grid_plus_one(i))
        
    # Phase 2: Multi-start optimization
    for x0 in inits:
        try:
            res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13})
            if np.all(compute_constraints(res.x) >= -1e-7):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = inits[0]
        
    # Phase 3: Iterative Perturbation & Refinement (Basin Hopping)
    curr_v = best_v.copy()
    for step in range(50):
        pert_v = curr_v.copy()
        
        # Perturb centers slightly
        pert_v[0::3] += np.random.normal(0, 0.002, N)
        pert_v[1::3] += np.random.normal(0, 0.002, N)
        
        # Perturb radii to break degeneracy
        pert_v[2::3] *= np.random.uniform(0.995, 1.005, N)
        
        # Clip to valid bounds
        for i in range(N):
            pert_v[3*i] = np.clip(pert_v[3*i], 0.01, 0.99)
            pert_v[3*i+1] = np.clip(pert_v[3*i+1], 0.01, 0.99)
            pert_v[3*i+2] = np.clip(pert_v[3*i+2], 0.001, 0.49)
            
        try:
            res = minimize(compute_objective, pert_v, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
            if np.all(compute_constraints(res.x) >= -1e-7):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
                    curr_v = best_v.copy()
        except Exception:
            pass
            
    # Extract final centers and radii
    centers = best_v.reshape(N, 3)[:, :2]
    radii = best_v.reshape(N, 3)[:, 2].copy()
    
    # Phase 4: Deterministic Repair for Strict Validation Compliance
    for _ in range(30):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-9:
                radii[i] = max_r
                changed = True
                
        # Resolve overlaps by proportional shrinking
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    overlap = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= overlap
                    radii[j] -= overlap
                    changed = True
                    
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
