import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)
NUM_PAIRS = N * (N - 1) // 2


def compute_valid_radii(centers):
    """Compute maximum valid radii for fixed centers."""
    n = centers.shape[0]
    # Distance to boundary
    r_bound = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    # Distance to nearest neighbor
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-20)
    np.fill_diagonal(dists, np.inf)
    r_pair = 0.5 * np.min(dists, axis=1)
    return np.minimum(r_bound, r_pair)


def sum_radii_objective(centers_flat):
    """Objective for centers-only optimization: maximize sum of valid radii."""
    centers = centers_flat.reshape(N, 2)
    radii = compute_valid_radii(centers)
    return -np.sum(radii)


def compute_constraints_full(vars_flat):
    """Full constraints for (x, y, r) optimization."""
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Overlap constraints
    xi, xj = x[TRIU_I], x[TRIU_J]
    yi, yj = y[TRIU_I], y[TRIU_J]
    ri, rj = r[TRIU_I], r[TRIU_J]
    
    dist = np.sqrt((xi - xj)**2 + (yi - yj)**2 + 1e-20)
    c.append(dist - (ri + rj))
    
    return np.concatenate(c)


def compute_objective_full(vars_flat):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])


def get_bounds_full():
    """Bounds for (x, y, r) variables."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b


def get_bounds_centers():
    """Bounds for centers-only variables."""
    return [(0.001, 0.999)] * (2 * N)


def make_xy_r_from_centers(centers, scale=0.85):
    """Create (x, y, r) vector from centers with feasible initial radii."""
    radii = compute_valid_radii(centers) * scale
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = radii
    return v


def generate_hex_pattern(seed, pattern_idx=0):
    """Generate hexagonal lattice pattern."""
    rng = np.random.default_rng(seed)
    
    patterns = [
        [5, 5, 5, 5, 5, 1],
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4],
        [6, 6, 5, 5, 4],
        [5, 4, 5, 4, 5, 3],
        [5, 6, 4, 6, 5],
        [6, 4, 6, 4, 6],
        [5, 5, 5, 6, 5],
        [6, 5, 5, 5, 5],
        [5, 5, 4, 5, 7],
        [4, 5, 6, 5, 6],
    ]
    
    pat = patterns[pattern_idx % len(patterns)]
    num_rows = len(pat)
    max_per_row = max(pat)
    
    # Calculate spacing
    dx = 0.85 / max_per_row
    dy = 0.85 / max(num_rows - 1, 1)
    
    centers = []
    y = 0.075
    row_idx = 0
    for count in pat:
        shift = (0.5 if row_idx % 2 == 1 else 0.0) * dx
        x = 0.075 + shift
        for _ in range(count):
            centers.append([
                x + rng.uniform(-0.003, 0.003),
                y + rng.uniform(-0.003, 0.003)
            ])
            x += dx
        y += dy
        row_idx += 1
    
    centers = np.array(centers[:N])
    return np.clip(centers, 0.03, 0.97)


def generate_corner_anchored(seed):

    rng = np.random.default_rng(seed)
    centers = []

    # Fill remaining with hexagonal packing

    row_idx = 0

        shift = (0.065 if row_idx % 2 == 1 else 0.0)

        row_idx += 1
    
    centers = np.array(centers[:N])
    return np.clip(centers, 0.03, 0.97)


def generate_rejection_sampled(seed):
    """Generate well-distributed random centers via rejection sampling."""
    rng = np.random.default_rng(seed)
    centers = []
    min_dist = 0.09
    
    for _ in range(N * 500):
        if len(centers) >= N:
            break
        pt = rng.uniform(0.05, 0.95, 2)
        ok = True
        for c in centers:
            if np.hypot(pt[0] - c[0], pt[1] - c[1]) < min_dist:
                ok = False
                break
        if ok:
            centers.append(pt)
    
    while len(centers) < N:
        centers.append(rng.uniform(0.1, 0.9, 2))
    
    return np.array(centers[:N])


def generate_grid_perturbed(seed):
    """Generate perturbed grid pattern."""
    rng = np.random.default_rng(seed)
    centers = []
    
    # 5x5 grid perturbed
    x_vals = np.linspace(0.12, 0.88, 5)
    y_vals = np.linspace(0.12, 0.88, 5)
    
    for y in y_vals:
        for x in x_vals:
            if len(centers) < N:
                centers.append([
                    x + rng.uniform(-0.03, 0.03),
                    y + rng.uniform(-0.03, 0.03)
                ])
    
    # Add extra circle if needed
    while len(centers) < N:
        centers.append(rng.uniform(0.2, 0.8, 2))
    
    centers = np.array(centers[:N])
    return np.clip(centers, 0.03, 0.97)


def iterative_force_optimization(centers, num_steps=3000, grow_rate=1.0001):
    """
    Iteratively grow radii and resolve overlaps using force-based method.
    This is a physically motivated approach that often finds good packings.
    """
    centers = centers.copy()
    radii = compute_valid_radii(centers) * 0.5  # Start small
    
    for step in range(num_steps):
        # Grow radii slightly
        radii *= grow_rate
        
        # Clamp to boundary
        for i in range(N):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0], 
                       centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > max_r:
                radii[i] = max_r
        
        # Compute forces from overlaps and push centers apart
        forces = np.zeros_like(centers)
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    force_mag = overlap * 50.0
                    fx = dx / dist * force_mag
                    fy = dy / dist * force_mag
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
        
        # Boundary forces
        for i in range(N):
            if centers[i, 0] < radii[i]:
                forces[i, 0] += (radii[i] - centers[i, 0]) * 100.0
            if centers[i, 0] > 1.0 - radii[i]:
                forces[i, 0] -= (centers[i, 0] - (1.0 - radii[i])) * 100.0
            if centers[i, 1] < radii[i]:
                forces[i, 1] += (radii[i] - centers[i, 1]) * 100.0
            if centers[i, 1] > 1.0 - radii[i]:
                forces[i, 1] -= (centers[i, 1] - (1.0 - radii[i])) * 100.0
        
        # Apply forces with damping
        dt = 0.001 / (1.0 + step * 0.0002)
        centers += forces * dt
        centers = np.clip(centers, 0.001, 0.999)
        
        # Reclamp radii to boundaries after center movement
        for i in range(N):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                       centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > max_r:
                radii[i] = max_r
        
        # Reduce radius if overlaps persist after movement
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j]:
                    excess = (radii[i] + radii[j] - d) / 2.0
                    radii[i] -= excess * 0.9
                    radii[j] -= excess * 0.9
    
    radii = np.maximum(radii, 0.0)
    return centers, radii


def solve_lp_for_radii(centers):
    """Use LP to find optimal radii for fixed centers."""
    # Build constraint matrix A_ub * r <= b_ub
    # Constraints: r_i + r_j <= dist(i,j) for i < j
    #               r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    
    A_ub = np.zeros((NUM_PAIRS + 4 * N, N))
    b_ub = np.zeros(NUM_PAIRS + 4 * N)
    
    # Pairwise constraints
    diff_x = centers[:, 0][:, np.newaxis] - centers[:, 0][np.newaxis, :]
    diff_y = centers[:, 1][:, np.newaxis] - centers[:, 1][np.newaxis, :]
    dists = np.sqrt(diff_x**2 + diff_y**2)
    
    for k, (i, j) in enumerate(zip(TRIU_I, TRIU_J)):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]
    
    # Boundary constraints
    base = NUM_PAIRS
    for i in range(N):
        A_ub[base + 4*i, i] = 1.0
        b_ub[base + 4*i] = centers[i, 0]
        
        A_ub[base + 4*i + 1, i] = 1.0
        b_ub[base + 4*i + 1] = 1.0 - centers[i, 0]
        
        A_ub[base + 4*i + 2, i] = 1.0
        b_ub[base + 4*i + 2] = centers[i, 1]
        
        A_ub[base + 4*i + 3, i] = 1.0
        b_ub[base + 4*i + 3] = 1.0 - centers[i, 1]
    
    # Maximize sum of radii = minimize negative sum
    c_obj = -np.ones(N)
    bounds = [(0, None)] * N
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    
    return None, 0.0


def repair_solution(centers, radii):
    """Deterministically repair to ensure strict validity."""
    radii = radii.copy()
    
    for _ in range(50):
        changed = False
        
        # Boundary clamping
        for i in range(N):
            x, y = centers[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if radii[i] > max_r + 1e-10:
                radii[i] = max(0.0, max_r)
                changed = True
        
        # Overlap resolution
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                           centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    overlap = (radii[i] + radii[j] - d + 1e-10) / 2.0
                    radii[i] = max(0.0, radii[i] - overlap)
                    radii[j] = max(0.0, radii[j] - overlap)
                    changed = True
        
        if not changed:
            break
    
    return radii

        try:
