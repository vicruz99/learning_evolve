Key insights:

Solving this LP provides a mathematically tight, feasible starting point for the radii, which dramatically accelerates convergence and prevents the non-linear optimizer from getting stuck in loose local minima.
2. **Multi-Start Lattice Patterns**: I will generate diverse initial center configurations using perturbed hexagonal and quasi-hexagonal lattice patterns (e.g., row distributions like [6,5,6,5,4], [5,6,5,6,4], etc.). These patterns naturally approximate optimal packing densities.
3. **Basin Hopping Refinement**: After finding the best initial configuration, I will perform iterative perturbations on the centers, slightly shrink the radii to maintain feasibility, and re-run the optimizer. This allows escaping local optima while preserving the high density of the packing.
4. **Strict Feasibility Repair**: A deterministic post-processing step will clamp centers to valid boundary regions and proportionally shrink overlapping circles to guarantee strict compliance with the `validate_packing` tolerance (`1e-12`).

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)

def compute_constraints(vars_flat):
    """Computes boundary and non-overlap constraints as inequalities >= 0."""
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise non-overlap: dist >= r_i + r_j
    dx = x[IDX_I] - x[IDX_J]
    dy = y[IDX_I] - y[IDX_J]
    dist = np.sqrt(dx**2 + dy**2)
    c.append(dist - (r[IDX_I] + r[IDX_J]))
    
    return np.concatenate(c)

def compute_objective(vars_flat):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars_flat[2::3])

def get_bounds():
    """Variable bounds for x, y in [0,1] and r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """
    Solves a Linear Program to find the maximum feasible radii for fixed centers.
    This provides a tight, feasible starting point for the non-linear optimizer.
    """
    num_pairs = N * (N - 1) // 2
    A = np.zeros((num_pairs + 4*N, N))
    b = np.zeros(num_pairs + 4*N)
    
    idx = 0
    for i in range(N):
        for j in range(i+1, N):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            idx += 1
            
    for i in range(N):
        A[idx, i] = 1.0; b[idx] = centers[i,0]; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - centers[i,0]; idx += 1
        A[idx, i] = 1.0; b[idx] = centers[i,1]; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - centers[i,1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A, b_ub=b, bounds=[(0, None)]*N, method='highs')
    if res.success:
        return res.x
    return None

def make_init_centers(seed):
    """Generates diverse initial center configurations based on lattice patterns."""
    rng = np.random.default_rng(seed)
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], 
        [7,6,5,5,3], [4,6,6,6,4], [6,4,6,5,5]
    ]
    pat = patterns[seed % len(patterns)]
    
    centers = []
    y = 0.12
    dy = 0.145
    dx = 0.16
    row_idx = 0
    for count in pat:
        shift = (0.5 if row_idx % 2 == 1 else 0.0) * dx
        x = 0.12 + shift
        for _ in range(count):
            centers.append([x + rng.uniform(-0.015, 0.015), y + rng.uniform(-0.015, 0.015)])
            x += dx
        y += dy
        row_idx += 1
        
    centers = np.array(centers[:N])
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def run_optimization(x0):
    """Runs SLSQP optimization from a given starting point."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}
    try:
        res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12})
        # Check feasibility with numerical tolerance
        if np.all(compute_constraints(res.x) >= -1e-8):
            return res.x, -res.fun
    except Exception:
        pass
    return None, -np.inf

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_x = None
    best_val = -np.inf
    
    # Phase 1: Multi-start optimization with LP-initialized radii
    for seed in range(35):
        centers = make_init_centers(seed)
        radii = solve_lp_radii(centers)
        
        if radii is not None:
            v0 = np.zeros(3*N)
            v0[0::3] = centers[:,0]
            v0[1::3] = centers[:,1]
            v0[2::3] = radii
            
            x_opt, val = run_optimization(v0)
            if x_opt is not None and val > best_val:
                best_val = val
                best_x = x_opt
                
    # Phase 2: Basin Hopping / Iterative Perturbation Refinement
    # Pushes the packing tighter by exploring the neighborhood of the best found solution
    for step in range(50):
        curr = best_x.copy()
        
        # Perturb centers to escape local minima
        curr[0::3] += np.random.normal(0, 0.006, N)
        curr[1::3] += np.random.normal(0, 0.006, N)
        curr[0::3] = np.clip(curr[0::3], 0.02, 0.98)
        curr[1::3] = np.clip(curr[1::3], 0.02, 0.98)
        
        # Slightly reduce radii to guarantee feasibility for the next optimization step
        curr[2::3] *= 0.996
        
        x_opt, val = run_optimization(curr)
        if x_opt is not None and val > best_val:
            best_val = val
            best_x = x_opt
            
    # Extract final configuration
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x.reshape(N, 3)[:, 2].copy()
    
    # Phase 3: Deterministic Repair for Strict Validation Compliance
    for _ in range(25):
        changed = False
        
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i,0], centers[i,1], radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-9:
                radii[i] = max_r
                changed = True
                
        # Resolve overlaps by proportional shrinking
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    ov = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= ov
                    radii[j] -= ov
                    changed = True
                    
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
