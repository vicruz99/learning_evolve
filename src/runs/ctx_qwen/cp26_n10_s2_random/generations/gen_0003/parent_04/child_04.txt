import numpy as np
import scipy.optimize as opt

N = 26
i_idx, j_idx = np.triu_indices(N, k=1)

def objective(p):
    """Minimize negative sum of radii to maximize total radius."""
    return -np.sum(p[2::3])

def constraints(p):
    """
    Computes all boundary and non-overlap constraints.
    Returns a 1D array where each element must be >= 0.
    """
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    c = []
    # Boundary constraints
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Non-overlap constraints
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    # Add small epsilon to avoid singularity at dist=0, though feasible states avoid it
    dist = np.sqrt(dx**2 + dy**2 + 1e-16)
    c.append(dist - (r[i_idx] + r[j_idx]))
    
    return np.concatenate(c)

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def make_feasible_radii(centers, scale=0.95):
    """Assigns feasible radii to centers based on distances to boundaries and neighbors."""
    rs = np.zeros(N)
    for i in range(N):
        d_bound = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        d_near = np.inf
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < d_near:
                    d_near = d
        rs[i] = min(d_bound, d_near / 2.0) * scale
    return rs

def generate_inits():

    inits = []
    
    # 1. Perturbed Hexagonal Lattices
    for _ in range(8):
        centers = np.zeros((N, 2))
        idx = 0
        y = 0.08
        row = 0
        r0 = 0.09
        while idx < N:
            x_start = r0 if row % 2 == 0 else 2*r0
            x = x_start

                idx += 1
                x += 2*r0
            y += np.sqrt(3)*r0
            row += 1
        centers += np.random.normal(0, 0.005, centers.shape)

        p[1::3] = centers[:,1]

                centers.append([x, y])
                x += 2*0.09
            y += dy
            row_idx += 1
        while len(centers) < N:

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    inits = generate_inits()
    
    best_p = None
    best_val = -np.inf
    
    # Main optimization loop with iterative growth
    for p0 in inits:
        curr_p = p0.copy()
        
        # Phase 1: Baseline optimization

            if np.all(constraints(res.x) >= -1e-8):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_p = res.x.copy()
                curr_p = res.x.copy()
        except Exception:
            pass
            
        # Phase 2: Iterative Growth & Perturbation Refinement
        # Forces the solver to find tighter packings by expanding radii and resolving overlaps
        for step in range(25):

            curr_p[2::3] *= 1.003

                break
                
    # Fallback if optimization fails entirely
    if best_p is None:
        best_p = inits[0]
        
    # Extract final centers and radii
    centers = np.column_stack((best_p[0::3], best_p[1::3]))
    radii = best_p[2::3].copy()

    for _ in range(50):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]

                    radii[i] -= overlap
                    radii[j] -= overlap
                    changed = True
                    
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
