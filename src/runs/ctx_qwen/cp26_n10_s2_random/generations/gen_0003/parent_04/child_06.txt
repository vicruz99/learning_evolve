import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

N = 26

# Precompute the inequality constraint matrix A_ub for the LP
# Constraints: A_ub @ r <= b_ub
num_pair = N * (N - 1) // 2
A_ub = np.zeros((4 * N + num_pair, N))

# Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
for i in range(N):
    A_ub[i, i] = 1.0
    A_ub[N + i, i] = 1.0
    A_ub[2*N + i, i] = 1.0
    A_ub[3*N + i, i] = 1.0

# Pairwise non-overlap constraints: r_i + r_j <= dist(i,j)
pair_idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub[4*N + pair_idx, i] = 1.0
        A_ub[4*N + pair_idx, j] = 1.0
        pair_idx += 1

bounds_r = [(0, None)] * N
c_obj = -np.ones(N)

def compute_max_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    x = centers[:, 0]
    y = centers[:, 1]
    
    b_ub = np.empty(4 * N + num_pair)
    b_ub[:N] = x
    b_ub[N:2*N] = 1.0 - x
    b_ub[2*N:3*N] = y
    b_ub[3*N:4*N] = 1.0 - y
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    triu_mask = np.triu_indices(N, k=1)
    b_ub[4*N:] = dists[triu_mask]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return res.x, -res.fun
    # Fallback for infeasible numerical states
    return np.full(N, 1e-5), 26e-5

def obj_centers(centers_flat):
    """Objective for center optimization: negative max sum of radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 0.0, 1.0)
    _, sum_r = compute_max_radii(centers)
    return -sum_r

def generate_hex_init(seed):
    """Generates a perturbed hexagonal lattice initialization."""
    rng = np.random.default_rng(seed)
    c = []
    r = 0.095
    y = r
    row = 0
    while len(c) < N:
        x_start = r if row % 2 == 0 else 2 * r
        x = x_start
        while x + r <= 1.0 + 1e-9 and len(c) < N:
            c.append([x, y])
            x += 2 * r
        y += np.sqrt(3) * r
        row += 1
    c = np.array(c[:N]) + rng.normal(0, 0.015, (N, 2))
    return np.clip(c, 0.05, 0.95).flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    best_centers = None
    best_val = np.inf
    
    # Phase 1: Global search to escape poor initial basins
    bounds_c = [(0.0, 1.0)] * (2 * N)
    try:
        res_de = differential_evolution(
            obj_centers, bounds_c, seed=42, popsize=8, maxiter=35,
            tol=1e-7, atol=1e-7, mutation=(0.5, 1.5), recombination=0.9, polish=False
        )
        best_centers = res_de.x
        best_val = res_de.fun
    except Exception:
        pass
        
    # Phase 2: Local refinement of the global best
    if best_centers is not None:
        res_local = minimize(obj_centers, best_centers, method='Nelder-Mead',
                             options={'xatol': 1e-8, 'fatol': 1e-10, 'maxiter': 3000})
        if res_local.fun < best_val:
            best_centers = res_local.x
            best_val = res_local.fun
            
    # Phase 3: Deterministic local searches from high-density hexagonal patterns
    for seed in range(12):
        x0 = generate_hex_init(seed)
        try:
            res = minimize(obj_centers, x0, method='Nelder-Mead',
                           options={'xatol': 1e-8, 'fatol': 1e-10, 'maxiter': 2500})
            if res.fun < best_val:
                best_centers = res.x
                best_val = res.fun
        except Exception:
            pass
            
    # Extract optimal configuration
    centers = np.clip(best_centers.reshape(N, 2), 0.0, 1.0)
    radii, sum_r = compute_max_radii(centers)
    
    # Phase 4: Deterministic repair to guarantee strict validation compliance
    for _ in range(50):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    ov = radii[i] + radii[j] - dist
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        # Fix boundary violations
        for i in range(N):
            x, y = centers[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if radii[i] > max_r - 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
