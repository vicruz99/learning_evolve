import numpy as np
import scipy.optimize as opt

N = 26
# Precompute pairwise indices for overlap constraints
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective(p):
    """Minimize negative sum of radii to maximize total radius."""
    return -np.sum(p[2::3])

def constraint_func(p):
    """
    Computes non-overlap constraints using squared distances.
    Variables: p[0::3] = u, p[1::3] = v, p[2::3] = r
    The transformation x = r + u*(1-2r) automatically satisfies boundary constraints.
    """
    u = p[0::3]
    v = p[1::3]
    r = p[2::3]
    
    # Transform to physical coordinates
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    sum_r = r[I_IDX] + r[J_IDX]
    
    # Squared distance constraint: dist^2 >= (r_i + r_j)^2
    return dx**2 + dy**2 - sum_r**2

def get_centers_and_radii(p):
    """Convert optimization variables back to centers and radii."""
    u = p[0::3]
    v = p[1::3]
    r = p[2::3]
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    return np.column_stack((x, y)), r

def generate_hex_init(perturb_scale=0.01, r_start=0.09):
    """Generates initial configuration from a perturbed hexagonal lattice."""
    centers = np.zeros((N, 2))
    idx = 0
    y_pos = r_start
    row = 0
    
    while idx < N:
        x_start = r_start if row % 2 == 0 else 2 * r_start
        x = x_start
        while x + r_start <= 1.0 + 1e-9 and idx < N:
            centers[idx] = [x, y_pos]
            idx += 1
            x += 2 * r_start
        y_pos += np.sqrt(3) * r_start
        row += 1
        
    # Convert to u, v coordinates
    denom = 1.0 - 2.0 * r_start
    u = (centers[:, 0] - r_start) / denom
    v = (centers[:, 1] - r_start) / denom
    
    # Add perturbation
    u += np.random.normal(0, perturb_scale, N)
    v += np.random.normal(0, perturb_scale, N)
    u = np.clip(u, 0.01, 0.99)
    v = np.clip(v, 0.01, 0.99)
    
    p = np.zeros(N * 3)
    p[0::3] = u
    p[1::3] = v
    p[2::3] = r_start
    return p

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # Bounds: u, v in [0, 1], r in [1e-7, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_p = None
    best_sum = -np.inf
    
    # Phase 1: Diverse initializations
    inits = []
    for seed in range(30):
        np.random.seed(seed)
        r_init = 0.085 + (seed % 5) * 0.002
        perturb = 0.005 * ((seed % 6) + 1)
        inits.append(generate_hex_init(perturb_scale=perturb, r_start=r_init))
        
    for p0 in inits:
        try:
            res = opt.minimize(objective, p0, method='SLSQP', bounds=bounds, 
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_p = res.x.copy()
                
            # Phase 2: Iterative Growth & Perturbation
            curr = res.x.copy()
            for step in range(25):
                p_trial = curr.copy()
                # Perturb u, v
                p_trial[0::3] += np.random.normal(0, 0.0008, N)
                p_trial[1::3] += np.random.normal(0, 0.0008, N)
                p_trial[0::3] = np.clip(p_trial[0::3], 0.0, 1.0)
                p_trial[1::3] = np.clip(p_trial[1::3], 0.0, 1.0)
                # Grow radii to push density limits
                p_trial[2::3] = np.maximum(p_trial[2::3], curr[2::3] * 1.0006)
                
                res2 = opt.minimize(objective, p_trial, method='SLSQP', bounds=bounds,
                                    constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
                
                if -res2.fun > best_sum:
                    best_sum = -res2.fun
                    best_p = res2.x.copy()
                    curr = best_p.copy()
                else:
                    curr = res2.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization completely failed
    if best_p is None:
        best_p = generate_hex_init(perturb_scale=0.0)
        
    centers, radii = get_centers_and_radii(best_p)
    
    # Phase 3: Strict Repair for Validation Compliance
    for _ in range(50):
        changed = False
        # Fix overlaps
        for i in range(N):
            for j in range(i + 1, N):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if dist < radii[i] + radii[j] - 1e-11:
                    shrink = (radii[i] + radii[j] - dist) * 0.5001
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Fix boundaries (safety net)
        for i in range(N):
            x, y = centers[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if radii[i] > max_r - 1e-11:
                radii[i] = max_r
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
