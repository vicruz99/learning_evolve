import numpy as np
from scipy.optimize import differential_evolution, minimize

N = 26

def get_sum_radii(centers):
    """
    Computes the maximum possible sum of radii for a given set of centers.
    Each radius is constrained by the distance to the boundary and half the distance to the nearest neighbor.
    """
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Distance to the four boundaries
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise Euclidean distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    
    # Each circle's radius is limited by half the distance to its nearest neighbor
    r_pair = 0.5 * np.min(dists, axis=1)
    
    # The actual maximum radius for each circle
    radii = np.minimum(r_bound, r_pair)
    return np.sum(radii)

def objective(centers_flat):
    """Objective function for minimizers: return negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    # Clip to maintain a safe margin from exact boundaries during optimization
    centers = np.clip(centers, 0.005, 0.995)
    return -get_sum_radii(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # Bounds for center coordinates
    bounds = [(0.005, 0.995)] * (2 * N)
    
    # Phase 1: Global optimization using Differential Evolution
    # DE is robust for non-smooth, multimodal objective functions common in packing problems.
    res_de = differential_evolution(
        objective, 
        bounds, 
        seed=42, 
        popsize=25, 
        maxiter=250, 
        mutation=(0.5, 1.0), 
        recombination=0.9, 
        polish=False, 
        atol=1e-8,
        disp=False
    )
    
    best_centers = res_de.x.reshape(N, 2)
    
    # Phase 2: Local refinement using Powell's method (derivative-free)
    # Fine-tunes the configuration found by DE.
    res_local = minimize(
        objective, 
        best_centers.flatten(), 
        method='Powell', 
        options={'maxiter': 2000, 'ftol': 1e-12, 'xtol': 1e-12}
    )
    
    best_centers = res_local.x.reshape(N, 2)
    best_centers = np.clip(best_centers, 0.005, 0.995)
    
    # Phase 3: Compute final maximum radii deterministically
    x = best_centers[:, 0]
    y = best_centers[:, 1]
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    r_pair = 0.5 * np.min(dists, axis=1)
    
    radii = np.minimum(r_bound, r_pair)
    
    # Phase 4: Deterministic repair to guarantee strict validity against numerical tolerances
    for _ in range(30):
        changed = False
        # Resolve overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i,0] - best_centers[j,0], best_centers[i,1] - best_centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) * 0.51 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            mx = min(best_centers[i,0], 1.0 - best_centers[i,0], best_centers[i,1], 1.0 - best_centers[i,1])
            if radii[i] > mx - 1e-9:
                radii[i] = mx
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    total_sum = float(np.sum(radii))
    
    return best_centers, radii, total_sum
