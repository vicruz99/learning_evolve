import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def solve_radii_lp(centers):
    """Solve LP to find optimal radii for fixed centers. Returns radii, sum, and dual marginals."""
    c = -np.ones(N)
    num_pairs = N * (N - 1) // 2
    A_ub = np.zeros((4 * N + num_pairs, N))
    b_ub = np.zeros(4 * N + num_pairs)
    
    idx = 0
    for i in range(N):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            idx += 1
            
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * N, method='highs')
    if res.success:
        return res.x, -res.fun, res.ineqlin.marginals
    return None, -np.inf, None

def compute_gradient(centers, marginals):
    """Compute gradient of max radii sum w.r.t centers using LP dual variables."""
    grad = np.zeros_like(centers)
    if marginals is None or len(marginals) == 0:
        return grad
        
    idx = 0
    for i in range(N):
        mu_L = marginals[idx]; idx += 1
        mu_R = marginals[idx]; idx += 1
        mu_B = marginals[idx]; idx += 1
        mu_T = marginals[idx]; idx += 1
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    for i in range(N):
        for j in range(i + 1, N):
            lam = marginals[idx]; idx += 1
            if lam > 1e-9:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)
                if d > 1e-12:
                    fx = lam * dx / d
                    fy = lam * dy / d
                    grad[i, 0] += fx
                    grad[i, 1] += fy
                    grad[j, 0] -= fx
                    grad[j, 1] -= fy
    return grad

def run_gradient_ascent(centers0, max_iter=2000):
    """Iteratively optimize centers using LP-derived gradients and momentum."""
    centers = centers0.copy()
    best_sum = -1.0
    best_centers = centers.copy()
    best_radii = None
    vel = np.zeros_like(centers)
    
    lr = 0.03
    decay = 0.999
    momentum = 0.7
    
    for t in range(max_iter):
        radii, cur_sum, marginals = solve_radii_lp(centers)
        if radii is None:
            break
            
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
        grad = compute_gradient(centers, marginals)
        vel = momentum * vel + (1.0 - momentum) * grad
        centers += lr * vel
        centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
        
        lr *= decay
        if lr < 1e-7:
            lr = 1e-7
            
    return best_centers, best_radii, best_sum

def generate_hex_init(seed):
    """Generates an initial hexagonal packing configuration with controlled noise."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    idx = 0
    r = 0.095
    y = r
    row = 0
    while idx < N:
        x_start = r if row % 2 == 0 else 2 * r
        x = x_start
        while x + r <= 1.0 + 1e-9 and idx < N:
            centers[idx] = [x, y]
            idx += 1
            x += 2 * r
        y += np.sqrt(3) * r
        row += 1
    noise = np.random.normal(0, 0.01, centers.shape)
    centers += noise
    return np.clip(centers, 0.05, 0.95)

def obj_func_slscp(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[2::3])

def con_func_slscp(p):
    """Computes all boundary and non-overlap constraints for SLSQP."""
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    c = []
    c.extend(x - r)
    c.extend(1.0 - x - r)
    c.extend(y - r)
    c.extend(1.0 - y - r)
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.hypot(x[i] - x[j], y[i] - y[j])
            c.append(dist - r[i] - r[j])
    return np.array(c)

def polish_with_slscp(centers, radii):
    """Polish configuration using SLSQP on all variables."""
    p0 = np.zeros(N * 3)
    p0[0::3] = centers[:, 0]
    p0[1::3] = centers[:, 1]
    p0[2::3] = radii
    
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(obj_func_slscp, p0, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': con_func_slscp},
                       options={'maxiter': 3000, 'ftol': 1e-13})
        if res.success or -res.fun > np.sum(radii):
            return np.column_stack((res.x[0::3], res.x[1::3])), res.x[2::3], -res.fun
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Phase 1: LP Gradient Ascent from diverse hexagonal starts
    for seed in range(30):
        c0 = generate_hex_init(seed)
        c_opt, r_opt, s_opt = run_gradient_ascent(c0, max_iter=2000)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    # Phase 2: SLSQP Polishing on best and perturbed configurations
    configs_to_polish = [(best_centers, best_radii)]
    for _ in range(5):
        pert = best_centers + np.random.normal(0, 0.002, best_centers.shape)
        pert = np.clip(pert, 0.05, 0.95)
        configs_to_polish.append((pert, best_radii))
        
    for c_cand, r_cand in configs_to_polish:
        c_pol, r_pol, s_pol = polish_with_slscp(c_cand, r_cand)
        if s_pol > best_sum:
            best_sum = s_pol
            best_centers = c_pol.copy()
            best_radii = r_pol.copy()
            
    # Phase 3: Final deterministic repair for strict validation compliance
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                dist = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                                best_centers[i, 1] - best_centers[j, 1])
                if dist < best_radii[i] + best_radii[j] - 1e-10:
                    ov = best_radii[i] + best_radii[j] - dist
                    best_radii[i] -= ov * 0.5
                    best_radii[j] -= ov * 0.5
                    changed = True
        for i in range(N):
            x, y = best_centers[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if best_radii[i] > max_r - 1e-10:
                best_radii[i] = max_r
                changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
