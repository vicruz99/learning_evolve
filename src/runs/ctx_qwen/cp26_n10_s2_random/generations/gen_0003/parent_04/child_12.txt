import numpy as np
from scipy.optimize import minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)

def objective_func(p):
    """Minimize negative sum of radii to maximize total radius."""
    return -np.sum(p[2::3])

def constraint_func(p):
    """
    Computes all boundary and non-overlap constraints.
    Returns a 1D array where each element must be >= 0.
    Uses squared distances for smoother gradients.
    """
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    dist_sq = dx**2 + dy**2
    sum_r = r[I_IDX] + r[J_IDX]
    c = np.concatenate([c, dist_sq - sum_r**2])
    
    return c

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def generate_hex_init(perturb_scale=0.0):
    """Generates a feasible hexagonal lattice configuration with optional noise."""
    centers = np.zeros((N, 2))
    idx = 0
    # Parameters chosen to ensure initial feasibility (spacing >= 2*r_init)
    r0 = 0.10
    dy = r0 * np.sqrt(3)
    dx = 2.0 * r0
    
    y = r0
    row = 0
    while idx < N:
        offset = (dx / 2.0) if row % 2 == 1 else 0.0
        x = r0 + offset
        while x + r0 <= 1.0 + 1e-7 and idx < N:
            centers[idx] = [x, y]
            idx += 1
            x += dx
        y += dy
        row += 1
        
    if perturb_scale > 0:
        centers += np.random.normal(0, perturb_scale, centers.shape)
        centers = np.clip(centers, 0.02, 0.98)
        
    p = np.zeros(3 * N)
    p[0::3] = centers[:, 0]
    p[1::3] = centers[:, 1]
    p[2::3] = 0.09  # Initial radii slightly smaller than spacing/2 ensures feasibility
    return p

def run_packing():

    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_p = None
    best_sum = -np.inf

        p[0::3] = c[:, 0]
        p[1::3] = c[:, 1]
        p[2::3] = 0.08
        inits.append(p)

            res = minimize(
                objective_func, p0, method='SLSQP', bounds=bounds,
                constraints=cons,
                options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False}
            )
            s = -res.fun
            if s > best_sum:
                # Strict feasibility check
                if np.min(constraint_func(res.x)) >= -1e-7:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            pass

        best_p = inits[0]

        curr_p[0::3] = np.clip(curr_p[0::3], 0.01, 0.99)

    return centers, radii, final_sum
