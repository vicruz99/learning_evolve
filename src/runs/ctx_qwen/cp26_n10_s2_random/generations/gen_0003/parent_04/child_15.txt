import numpy as np
import scipy.optimize as opt

N = 26
i_idx, j_idx = np.triu_indices(N, k=1)

def objective(p):
    """Minimize negative sum of radii to maximize total radius."""
    return -np.sum(p[2::3])

def constraints(p):
    """
    Computes all boundary and non-overlap constraints.
    Returns a 1D array where each element must be >= 0.
    Uses squared distances for smoother gradient conditioning.
    """
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    c = []
    # Boundary constraints: circles must stay inside [0,1]x[0,1]
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Overlap constraints: dx^2 + dy^2 >= (r_i + r_j)^2
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    c.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(c)

def get_bounds():
    """Creates variable bounds: x,y in [0,1], r in [1e-7, 0.5]"""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def generate_initial_configurations():
    """Generates diverse starting points to escape local minima."""
    configs = []
    np.random.seed(42)
    
    # 1. Hexagonal lattice variants with controlled noise
    for seed in range(10):
        rng = np.random.default_rng(seed)
        r0 = 0.085
        centers = []
        y = r0
        row = 0
        while len(centers) < N:
            x_start = r0 if row % 2 == 0 else 2 * r0
            x = x_start
            while x + r0 <= 1.0 + 1e-9 and len(centers) < N:
                centers.append([x, y])
                x += 2 * r0
            y += np.sqrt(3) * r0
            row += 1
        c = np.array(centers[:N])
        c += rng.normal(0, 0.003, c.shape)
        c = np.clip(c, 0.02, 0.98)
        p = np.zeros(3 * N)
        p[0::3] = c[:, 0]
        p[1::3] = c[:, 1]
        p[2::3] = r0
        configs.append(p)
        
    # 2. Random dense packings
    for seed in range(5):
        rng = np.random.default_rng(seed)
        c = rng.uniform(0.1, 0.9, (N, 2))
        p = np.zeros(3 * N)
        p[0::3] = c[:, 0]
        p[1::3] = c[:, 1]
        p[2::3] = 0.06
        configs.append(p)
        
    # 3. Corner-focused configurations (boundaries often allow tighter packing)
    for seed in range(5):
        rng = np.random.default_rng(seed)
        c = rng.uniform(0.05, 0.95, (N, 2))
        # Pull a subset of circles toward corners to test boundary-limited growth
        corners = rng.choice(4, size=N//2)
        for i, corner in enumerate(corners):
            if corner == 0: c[i] = [0.15 + rng.uniform(0,0.1), 0.15 + rng.uniform(0,0.1)]
            elif corner == 1: c[i] = [0.85 + rng.uniform(0,0.1), 0.15 + rng.uniform(0,0.1)]
            elif corner == 2: c[i] = [0.15 + rng.uniform(0,0.1), 0.85 + rng.uniform(0,0.1)]
            else: c[i] = [0.85 + rng.uniform(0,0.1), 0.85 + rng.uniform(0,0.1)]
        p = np.zeros(3 * N)
        p[0::3] = c[:, 0]
        p[1::3] = c[:, 1]
        p[2::3] = 0.07
        configs.append(p)
        
    return configs

def repair_packing(centers, radii):
    """
    Strictly enforces constraints to guarantee validation passes.
    Shrinks overlaps proportionally to preserve larger circles and minimize sum loss.
    """
    for _ in range(200):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    overlap = req - d
                    total_r = radii[i] + radii[j]
                    if total_r > 1e-12:
                        fi = radii[i] / total_r
                        fj = radii[j] / total_r
                    else:
                        fi = fj = 0.5
                    radii[i] -= overlap * fi * 0.51
                    radii[j] -= overlap * fj * 0.49
                    changed = True
                    
        # Fix boundary violations
        for i in range(N):
            x, y = centers[i]
            mr = min(x, 1.0 - x, y, 1.0 - y)
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    configs = generate_initial_configurations()
    best_p = None
    best_sum = -np.inf
    
    # Phase 1: Multi-start optimization
    for p0 in configs:
        try:
            res = opt.minimize(objective, p0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
            s = -res.fun
            c_vals = constraints(res.x)
            if np.min(c_vals) >= -1e-6 and s > best_sum:
                best_sum = s
                best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Iterative Expansion (Inflate & Relax)
    # Pushes the solution against constraint boundaries to find denser packings
    if best_p is not None:
        for step in range(60):
            p = best_p.copy()
            # Gradually inflate radii
            inflate_factor = 1.0 + 0.003 * (1.0 - step / 60.0)
            p[2::3] *= inflate_factor
            
            # Small perturbation to help optimizer find new equilibrium
            p[0::3] += np.random.normal(0, 0.0008, N)
            p[1::3] += np.random.normal(0, 0.0008, N)
            p[0::3] = np.clip(p[0::3], 0.005, 0.995)
            p[1::3] = np.clip(p[1::3], 0.005, 0.995)
            
            try:
                res = opt.minimize(objective, p, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': 4000, 'ftol': 1e-14})
                s = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6 and s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
            except Exception:
                pass
                
    # Fallback
    if best_p is None:
        best_p = configs[0]
        
    centers = np.column_stack((best_p[0::3], best_p[1::3]))
    radii = best_p[2::3].copy()
    
    # Final strict repair for validation compliance
    centers, radii = repair_packing(centers, radii)
    
    return centers, radii, float(np.sum(radii))
