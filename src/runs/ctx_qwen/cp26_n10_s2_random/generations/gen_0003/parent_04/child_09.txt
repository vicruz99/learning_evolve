import numpy as np
from scipy.optimize import minimize
import math

N = 26
# Precompute pairwise indices for overlap constraints to speed up evaluation
II, JJ = np.triu_indices(N, k=1)

def objective(p):
    """Minimize negative sum of radii to maximize total radius."""
    return -np.sum(p[2::3])

def get_constraints(p):
    """
    Computes all boundary and non-overlap constraints.
    Returns a 1D array where each element must be >= 0.
    """
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Preallocate constraint array
    n_con = 4 * N + N * (N - 1) // 2
    c = np.empty(n_con)
    idx = 0
    
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    c[idx:idx+N] = x - r; idx += N
    c[idx:idx+N] = 1.0 - x - r; idx += N
    c[idx:idx+N] = y - r; idx += N
    c[idx:idx+N] = 1.0 - y - r; idx += N
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    # Vectorized using precomputed upper-triangle indices
    dx = x[II] - x[JJ]
    dy = y[II] - y[JJ]
    c[idx:] = np.hypot(dx, dy) - (r[II] + r[JJ])
    
    return c

def generate_hex_init(seed):
    """Generates initial configuration from a perturbed hexagonal lattice."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = 0.085
    row = 0
    r0 = 0.095
    
    while idx < N:
        x = r0 + (row % 2) * r0
        while x + r0 <= 1.0 + 1e-9 and idx < N:
            centers[idx] = [x, y]
            idx += 1
            x += 2 * r0
        y += math.sqrt(3) * r0
        row += 1
        
    # Add controlled noise to break symmetry
    noise_scale = 0.005 * ((seed % 5) + 1)
    centers += np.random.normal(0, noise_scale, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    p = np.zeros(N * 3)
    p[0::3] = centers[:, 0]
    p[1::3] = centers[:, 1]
    p[2::3] = 0.085
    return p

def generate_random_init(seed):
    """Generates initial configuration using rejection sampling for diversity."""
    np.random.seed(seed)
    centers = []
    attempts = 0
    min_dist = 0.12
    while len(centers) < N and attempts < 20000:
        pt = np.random.uniform(0.15, 0.85, 2)
        ok = True
        for c in centers:
            if np.hypot(pt[0]-c[0], pt[1]-c[1]) < min_dist:
                ok = False
                break
        if ok:
            centers.append(pt)
        attempts += 1
        
    while len(centers) < N:
        centers.append(np.random.uniform(0.1, 0.9, 2))
        
    centers = np.array(centers[:N])
    p = np.zeros(N * 3)
    p[0::3] = centers[:, 0]
    p[1::3] = centers[:, 1]
    p[2::3] = 0.06
    return p

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.001, 0.5)] * N
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    best_p = None
    best_sum = -np.inf
    
    # Phase 1: Diverse multi-start optimization
    # 25 hexagonal variants
    for seed in range(25):
        p0 = generate_hex_init(seed)
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-11, 'disp': False})
            curr_sum = -res.fun
            if np.min(get_constraints(res.x)) >= -1e-6 and curr_sum > best_sum:
                best_sum = curr_sum
                best_p = res.x.copy()
        except Exception:
            continue
            
    # 15 random variants
    for seed in range(15):
        p0 = generate_random_init(seed)
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds,

            curr_sum = -res.fun

                best_sum = curr_sum
                best_p = res.x.copy()

            continue

    if best_p is not None:
        for i in range(20):
            decay = 1.0 - i / 20.0
            p_trial = best_p + np.random.normal(0, 0.003 * decay, N * 3)

            p_trial[1::3] = np.clip(p_trial[1::3], 0.02, 0.98)

                continue

    if best_p is None:
        best_p = generate_hex_init(0)
        
    centers = np.column_stack((best_p[0::3], best_p[1::3]))
    radii = best_p[2::3].copy()

    # Resolves microscopic overlaps and boundary violations from numerical precision

                radii[i] = max_r
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 1e-7)
    return centers, radii, float(np.sum(radii))
