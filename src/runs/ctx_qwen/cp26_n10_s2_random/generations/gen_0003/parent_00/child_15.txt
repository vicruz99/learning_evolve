from scipy.optimize import minimize

N = 26

def compute_constraints(v):
    """
    Computes boundary and pairwise non-overlap constraints.
    Returns a 1D numpy array where each element must be >= 0 for feasibility.
    """
    centers = v[:2 * N].reshape(N, 2)
    radii = v[2 * N:]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con = np.concatenate([
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
    ])
    
    # Pairwise constraints: dist^2 >= (r_i + r_j)^2
    idx = np.triu_indices(N, k=1)
    dx = centers[idx[0], 0] - centers[idx[1], 0]
    dy = centers[idx[0], 1] - centers[idx[1], 1]
    dr = radii[idx[0]] + radii[idx[1]]
    con = np.concatenate([con, dx**2 + dy**2 - dr**2])
    
    return con

def objective_func(v):

    return -np.sum(v[2 * N:])

    y = r0

        offset = (r_idx % 2) * r0
        x = r0 + offset

            centers.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
