import numpy as np
from scipy.optimize import minimize

N = 26

def get_constraints(v):
    """
    Computes inequality constraints: g(v) >= 0.
    Returns concatenated array of boundary and non-overlap constraints.
    """
    centers = v[:2 * N].reshape(N, 2)
    radii = v[2 * N:]
    
    cons = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons.append(centers[:, 0] - radii)
    cons.append(1.0 - centers[:, 0] - radii)
    cons.append(centers[:, 1] - radii)
    cons.append(1.0 - centers[:, 1] - radii)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    c1 = centers[:, np.newaxis, :]
    c2 = centers[np.newaxis, :, :]
    dist_sq = np.sum((c1 - c2)**2, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Upper triangle indices to avoid duplicates and self-comparison
    idx = np.triu_indices(N, k=1)
    cons.append(dist_sq[idx] - r_sum[idx]**2)
    
    return np.concatenate(cons)

def obj_func(v):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2 * N:])

def compute_initial_radii(centers, scale=0.95):
    """Computes strictly feasible initial radii for given centers."""
    x, y = centers[:, 0], centers[:, 1]
    # Distance to boundaries
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    
    # Radius limited by half distance to nearest neighbor
    r_pair = 0.5 * np.min(dist, axis=1)
    
    return np.minimum(r_bound, r_pair) * scale

def generate_inits():
    """Generates a list of diverse, feasible initial parameter arrays."""
    inits = []
    
    # 1. Hexagonal lattices with varying densities and rotations
    for seed in range(5):
        rng = np.random.default_rng(seed)
        r0 = 0.095 + rng.uniform(-0.005, 0.005)
        pts = []
        y = r0
        row = 0
        while len(pts) < N:
            x = r0 + (row % 2) * r0
            while x + r0 <= 1.0 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        pts = np.array(pts[:N])
        pts += rng.normal(0, 0.002, pts.shape)
        pts = np.clip(pts, 0.05, 0.95)
        r_init = compute_initial_radii(pts)
        inits.append(np.concatenate([pts.flatten(), r_init]))

    for seed in range(5):

        pts[1] = [0.88, 0.12]
        pts[2] = [0.12, 0.88]
        pts[3] = [0.88, 0.88]
        # Edge midpoints

        pts[5] = [0.5, 0.88]
        pts[6] = [0.12, 0.5]
        pts[7] = [0.88, 0.5]
        # Center
        pts[8] = [0.5, 0.5]
        # Random fill for remaining

        pts += rng.normal(0, 0.005, pts.shape)
        pts = np.clip(pts, 0.06, 0.94)
        r_init = compute_initial_radii(pts)
        inits.append(np.concatenate([pts.flatten(), r_init]))
        
    # 3. Dense random starts

        pts = rng.uniform(0.1, 0.9, (N, 2))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    constraints = {'type': 'ineq', 'fun': get_constraints}
    
    best_v = None
    best_val = -np.inf

    for v0 in inits:

                obj_func, v0, method='SLSQP', bounds=bounds,

            # Check feasibility with tolerance

                best_val = val
                best_v = res.x.copy()

            continue

    if best_v is None:
        best_v = inits[0]

    for _ in range(50):
        valid = True
        
        # Enforce boundaries strictly

            max_r = min(x, 1.0 - x, y, 1.0 - y)

                radii[i] = max_r
                valid = False

            for j in range(i + 1, N):

                    radii[i] -= shrink

                    valid = False

            break
            
    # Ensure non-negative radii
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
