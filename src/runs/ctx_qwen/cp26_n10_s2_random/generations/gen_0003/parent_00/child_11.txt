To improve the packing sum from ~2.628 to >2.636, we need a more robust global optimization strategy combined with an exact radius extraction method. 

Key insights:
1. **Joint Optimization vs. LP Radii Extraction**: Optimizing centers and radii jointly with SLSQP is good for finding tight configurations, but SLSQP may not find the absolute maximum radii for a given set of centers due to non-convex constraint interactions. However, for *fixed* centers, finding the maximum sum of radii is a Linear Programming (LP) problem: maximize $\sum r_i$ subject to $r_i + r_j \le d_{ij}$ and $r_i \le \text{boundary}_i$. Solving this LP after each successful optimization run guarantees we extract the maximal possible radii for those centers, often pushing the sum higher.
2. **Diverse Initialization**: Hexagonal lattices with slight perturbations are known to yield high-density packings. We will generate many such starts with varying base radii (0.095 to 0.105) to cover different density regimes.
3. **Basin Hopping Refinement**: After finding the best configuration, we perform local perturbations with decaying step sizes, followed by SLSQP refinement and LP radius extraction. This escapes shallow local minima.
4. **Strict Validation Repair**: A final deterministic shrinking loop ensures the output strictly satisfies the validator's $10^{-12}$ tolerance.

The implementation uses `scipy.optimize.minimize` (SLSQP) for center optimization and `scipy.optimize.linprog` (HiGHS) for exact radius extraction. This hybrid approach is both efficient and highly effective for this class of problems.

import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def get_bounds():
    """Returns variable bounds for x, y in [0,1] and r in [0, 0.5]."""
    return [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

def constraints(v):
    """Computes all boundary and non-overlap constraints. Returns array >= 0 for feasibility."""
    centers = v[:2 * N].reshape(N, 2)
    radii = v[2 * N:]
    
    c = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c.append(centers[:, 0] - radii)
    c.append(1.0 - centers[:, 0] - radii)
    c.append(centers[:, 1] - radii)
    c.append(1.0 - centers[:, 1] - radii)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    idx = np.triu_indices(N, k=1)
    dx = centers[idx[0], 0] - centers[idx[1], 0]
    dy = centers[idx[0], 1] - centers[idx[1], 1]
    dr = radii[idx[0]] + radii[idx[1]]
    c.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(c)

def objective_func(v):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2 * N:])

def solve_lp_radii(centers):
    """
    For fixed centers, solves LP to find radii that maximize sum(r_i)
    subject to r_i + r_j <= dist(i,j) and r_i <= boundary_dist(i).
    """
    x, y = centers[:, 0], centers[:, 1]
    b_i = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    
    # LP formulation: min -1^T r  s.t. A_ub r <= b_ub, r >= 0
    c_lp = -np.ones(N)
    
    num_pairs = N * (N - 1) // 2
    num_bounds = N
    A_ub = np.zeros((num_pairs + num_bounds, N))
    b_ub = np.zeros(num_pairs + num_bounds)
    
    # Pairwise constraints: r_i + r_j <= d_ij
    idx = np.triu_indices(N, k=1)
    A_ub[:num_pairs, idx[0]] = 1.0
    A_ub[:num_pairs, idx[1]] = 1.0
    b_ub[:num_pairs] = dist[idx]
    
    # Boundary constraints: r_i <= b_i
    A_ub[num_pairs:, np.arange(N)] = 1.0
    b_ub[num_pairs:] = b_i
    
    # Solve LP
    res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * N, method='highs')
    if res.success:
        return res.x, -res.fun
    # Fallback to safe small radii if LP fails
    return np.full(N, 0.01), 0.01 * N

def init_config(seed, r_base=0.101):
    """Generates an initial configuration from a perturbed hexagonal lattice."""
    rng = np.random.default_rng(seed)
    pts = []
    y = r_base
    row = 0
    while len(pts) < N:
        x = r_base + (row % 2) * r_base
        while x <= 1.0 - r_base and len(pts) < N:
            pts.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
            x += 2.0 * r_base
        y += r_base * math.sqrt(3)
        row += 1
        
    pts = np.array(pts[:N])
    pts = np.clip(pts, 0.05, 0.95)
    
    # Compute strictly feasible initial radii via LP
    c_flat = pts.flatten()
    c_arr = c_flat.reshape(N, 2)
    r_est, _ = solve_lp_radii(c_arr)
    
    return np.concatenate([c_flat, r_est * 0.95])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    best_v = None
    best_sum = -np.inf
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Generate diverse initial configurations
    starts = []
    for s in range(30):
        starts.append(init_config(s, r_base=0.100))
        starts.append(init_config(s, r_base=0.105))
        starts.append(init_config(s, r_base=0.095))
        
    # Phase 2: Multi-start SLSQP optimization with LP radius extraction
    for v0 in starts:
        try:
            res = minimize(
                objective_func, v0, method='SLSQP', bounds=bounds,
                constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False}
            )
            c_vals = constraints(res.x)
            if np.min(c_vals) > -1e-7:
                centers_temp = res.x[:2 * N].reshape(N, 2)
                radii_temp, s_lp = solve_lp_radii(centers_temp)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_v = np.concatenate([centers_temp.flatten(), radii_temp])
        except Exception:
            continue
            
    # Phase 3: Basin hopping refinement to escape local minima
    if best_v is not None:
        rng = np.random.default_rng(42)
        for step in range(50):
            scale = 0.002 * (0.9 ** step)
            v_curr = best_v + rng.normal(0, scale, best_v.shape)
            v_curr[:2 * N] = np.clip(v_curr[:2 * N], 0.01, 0.99)
            
            try:
                res = minimize(
                    objective_func, v_curr, method='SLSQP', bounds=bounds,
                    constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
                )
                c_vals = constraints(res.x)
                if np.min(c_vals) > -1e-7:
                    centers_temp = res.x[:2 * N].reshape(N, 2)
                    radii_temp, s_lp = solve_lp_radii(centers_temp)
                    if s_lp > best_sum:
                        best_sum = s_lp
                        best_v = np.concatenate([centers_temp.flatten(), radii_temp])
            except Exception:
                continue
                
    # Phase 4: Extract and strictly validate
    centers = best_v[:2 * N].reshape(N, 2)
    radii = best_v[2 * N:].copy()
    
    # Final safety repair to guarantee strict feasibility within tolerance
    for _ in range(30):
        valid = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            if x - r < -1e-12 or x + r > 1.0 + 1e-12 or y - r < -1e-12 or y + r > 1.0 + 1e-12:
                valid = False
                break
        if not valid:
            radii *= 0.9995
            continue
            
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            break
        radii *= 0.9995
        
    return centers, radii, float(np.sum(radii))
