import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_constraints(v):
    """Computes boundary and non-overlap constraints. Returns array >= 0 for feasibility."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    idx = np.triu_indices(N, k=1)
    dx = c[idx[0], 0] - c[idx[1], 0]
    dy = c[idx[0], 1] - c[idx[1], 1]
    dr = r[idx[0]] + r[idx[1]]
    con.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(con)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def get_lp_radii(centers):
    """Solves LP to exactly maximize sum of radii for fixed centers."""
    num_pairs = N * (N - 1) // 2
    A = np.zeros((num_pairs + 4*N, N))
    b = np.zeros(num_pairs + 4*N)
    k = 0
    
    # Pairwise distance constraints: r_i + r_j <= d_ij
    for i in range(N):
        for j in range(i+1, N):
            A[k, i] = 1.0
            A[k, j] = 1.0
            b[k] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            k += 1
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(N):
        A[k, i] = 1.0; b[k] = centers[i, 0]; k += 1
        A[k, i] = 1.0; b[k] = 1.0 - centers[i, 0]; k += 1
        A[k, i] = 1.0; b[k] = centers[i, 1]; k += 1
        A[k, i] = 1.0; b[k] = 1.0 - centers[i, 1]; k += 1
        
    try:
        res = linprog(-np.ones(N), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, -np.inf

def make_init(seed, strategy):
    """Generates a feasible initial parameter array based on a strategy."""
    rng = np.random.default_rng(seed)
    c = np.zeros((N, 2))
    
    if strategy == 'hex':
        r0 = 0.092
        y = r0
        row = 0
        idx = 0
        while idx < N:
            xs = r0 if row % 2 == 0 else 2*r0
            x = xs
            while x + r0 <= 1.0 - 1e-9 and idx < N:
                c[idx] = [x, y]
                idx += 1
                x += 2*r0
            y += r0 * np.sqrt(3)
            row += 1
    elif strategy == 'grid':
        pts = []
        for i in range(6):
            for j in range(5):
                pts.append([0.08 + 0.16*i, 0.08 + 0.18*j])
        c = np.array(pts[:N])
    elif strategy == 'rand':
        c = rng.uniform(0.15, 0.85, (N, 2))
    elif strategy == 'corners':
        c[:4] = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
        c[4:] = rng.uniform(0.2, 0.8, (N-4, 2))
        
    # Add noise and clamp
    c += rng.normal(0, 0.005, c.shape)
    c = np.clip(c, 0.05, 0.95)
    
    # Compute strictly feasible initial radii
    r_bound = np.minimum(np.minimum(c[:, 0], 1.0-c[:, 0]), np.minimum(c[:, 1], 1.0-c[:, 1]))
    dists = np.linalg.norm(c[:, None, :] - c[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r_pair = 0.5 * np.min(dists, axis=1)
    r = np.minimum(r_bound, r_pair) * 0.9
    r = np.maximum(r, 1e-6)
    
    return np.concatenate([c.flatten(), r])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    best_v = None
    best_sum = -np.inf
    bounds = [(0.0, 1.0)]*(2*N) + [(1e-7, 0.5)]*N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # Phase 1: Multi-start global optimization
    rng_main = np.random.default_rng(42)
    seeds = rng_main.integers(0, 10000, 30)
    strats = ['hex', 'hex', 'grid', 'rand', 'corners']
    
    for s in seeds:
        strat = strats[s % len(strats)]
        for trial in range(3):
            v0 = make_init(s + trial, strat)
            try:
                res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                # Check feasibility with tolerance
                if np.min(compute_constraints(res.x)) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_v = res.x.copy()
            except Exception:
                pass
                
    # Fallback if optimization failed
    if best_v is None:
        best_v = make_init(42, 'hex')
        
    # Phase 2: Perturbation search (Basin Hopping style) to escape local minima
    rng_pert = np.random.default_rng(123)
    for step in range(60):
        scale = 0.004 * (0.85 ** step)
        v_curr = best_v + rng_pert.normal(0, scale, best_v.shape)
        v_curr = np.clip(v_curr, 0.0, 1.0)
        v_curr[2*N:] = np.clip(v_curr[2*N:], 1e-7, 0.5)
        
        try:
            res = minimize(objective, v_curr, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if np.min(compute_constraints(res.x)) >= -1e-7:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Exact radius maximization via Linear Programming for fixed centers
    centers = best_v[:2*N].reshape(N, 2)
    radii, final_sum = get_lp_radii(centers)
    
    if radii is None:
        # Fallback to optimized radii if LP fails (rare)
        radii = best_v[2*N:].copy()
        final_sum = float(best_sum)
        
    return centers, radii, float(final_sum)
