

<strategy>
1. **Global Exploration with Differential Evolution (DE)**: Since the objective function (max sum of radii) is non-convex and non-smooth with respect to centers, DE is excellent for finding a good basin of attraction without getting trapped in poor local minima. We optimize only the 52 center coordinates, computing radii deterministically as the minimum distance to boundaries or half the distance to the nearest neighbor.
2. **Local Refinement with SLSQP**: Once DE identifies promising center positions, we switch to SLSQP to jointly optimize centers and radii with explicit inequality constraints. This allows precise tuning of radii and positions to satisfy boundary and non-overlap constraints strictly while pushing the sum of radii higher.
3. **Multi-Start Perturbation**: We run SLSQP multiple times from slightly perturbed versions of the DE result to escape any remaining local optima and refine the solution further.
4. **Strict Validation Repair**: A final deterministic repair step shrinks radii minimally to guarantee all constraints pass the strict `1e-12` tolerance of the validator, ensuring a valid submission.
</strategy>
```python
import numpy as np
from scipy.optimize import minimize, differential_evolution

N = 26

def compute_max_radii(centers):
    """Computes the maximum valid radius for each circle given fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    # Distance to boundaries
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    
    # Radius limited by half the distance to nearest neighbor
    r_pair = 0.5 * np.min(dist, axis=1)
    
    return np.minimum(r_bound, r_pair)

def objective_centers(centers_flat):
    """Objective for DE: maximize sum of radii (minimize negative sum)."""
    centers = centers_flat.reshape(N, 2)
    radii = compute_max_radii(centers)
    return -np.sum(radii)

def compute_constraints(v):
    """Computes all boundary and non-overlap constraints. Returns array >= 0 for feasibility."""
    centers = v[:2 * N].reshape(N, 2)
    radii = v[2 * N:]
    
    con = []
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con.append(centers[:, 0] - radii)
    con.append(1.0 - centers[:, 0] - radii)
    con.append(centers[:, 1] - radii)
    con.append(1.0 - centers[:, 1] - radii)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    idx = np.triu_indices(N, k=1)
    dx = centers[idx[0], 0] - centers[idx[1], 0]
    dy = centers[idx[0], 1] - centers[idx[1], 1]
    dr = radii[idx[0]] + radii[idx[1]]
    con.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(con)

def objective_v(v):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(v[2 * N:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    
    # Phase 1: Global search using Differential Evolution on centers
    # DE handles the non-smooth nature of the radius computation well
    de_res = differential_evolution(
        objective_centers, bounds_centers,
        popsize=25, maxiter=500, tol=1e-8, mutation=(0.5, 1.5),
        recombination=0.9, seed=42, polish=False
    )
    
    best_centers = de_res.x.reshape(N, 2)
    best_radii = compute_max_radii(best_centers)
    best_sum = np.sum(best_radii)
    
    # Phase 2: Local refinement with SLSQP on (centers, radii)
    v0 = np.concatenate([best_centers.flatten(), best_radii])
    bounds_v = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    np.random.seed(42)
    for _ in range(15):
        # Perturb current best to explore nearby basins
        v_curr = v0.copy()
        v_curr += np.random.normal(0, 0.002, v_curr.shape)
        v_curr = np.clip(v_curr, 0.0, 1.0)
        v_curr[2 * N:] = np.maximum(v_curr[2 * N:], 1e-6)
        
        res = minimize(
            objective_v, v_curr, method='SLSQP', bounds=bounds_v,
            constraints={'type': 'ineq', 'fun': compute_constraints},
            options={'maxiter': 10000, 'ftol': 1e-12, 'disp': False}
        )
        
        curr_sum = -res.fun
        # Check feasibility with tolerance
        c_vals = compute_constraints(res.x)
        if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
            best_sum = curr_sum
            v0 = res.x.copy()
            
    # Extract results
    centers = v0[:2 * N].reshape(N, 2)
    radii = v0[2 * N:].copy()
    
    # Phase 3: Strict validation repair (handles floating point drift)
    for _ in range(30):
        valid = True
        
        # Check boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            if x - r < -1e-12 or x + r > 1.0 + 1e-12 or y - r < -1e-12 or y + r > 1.0 + 1e-12:
                valid = False
                break
        if not valid:
            radii *= 0.9999
            continue
            
        # Check overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            break
        radii *= 0.9999
        
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
```
