import numpy as np
from scipy.optimize import minimize

N = 26

def compute_max_radii(centers):
    """Computes the maximum valid radius for each circle given fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    # Distance to boundaries
    rb = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    
    # Radius limited by half the distance to nearest neighbor
    rp = 0.5 * np.min(dist, axis=1)
    
    return np.minimum(rb, rp)

def objective(v):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2 * N:])

def constraints(v):
    """Computes all boundary and non-overlap constraints. Returns array >= 0 for feasibility."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    
    con = []
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    idx = np.triu_indices(N, k=1)
    dx = c[idx[0], 0] - c[idx[1], 0]
    dy = c[idx[0], 1] - c[idx[1], 1]
    dr = r[idx[0]] + r[idx[1]]
    con.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(con)

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    return [(0.0, 1.0)] * (2 * N) + [(1e-8, 0.5)] * N

                idx += 1
                x += 2.0 * r0

            row += 1
        c = np.clip(c, 0.02, 0.98)

        r = compute_max_radii(c) * 0.96

    # Pattern 3: Corner-focused + random fill

        c = np.zeros((N, 2))
        c[:4] = [[0.06, 0.06], [0.94, 0.06], [0.06, 0.94], [0.94, 0.94]]
        c[4:] = rng.uniform(0.15, 0.85, (N - 4, 2))
        r = compute_max_radii(c) * 0.96
        inits.append(np.concatenate([c.flatten(), np.maximum(r, 1e-6)]))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}

    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
