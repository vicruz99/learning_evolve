N = 26

def compute_objective(v):

    return -np.sum(v.reshape(N, 3)[:, 2])

def compute_constraints(v):
    """Inequality constraints: boundary and non-overlap (must be >= 0)."""
    pts = v.reshape(N, 3)
    cx, cy, cr = pts[:, 0], pts[:, 1], pts[:, 2]
    
    # Boundary constraints
    c = [cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr]
    
    # Non-overlap constraints
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    sr = cr[:, None] + cr[None, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(dist[mask] - sr[mask])
    
    return np.concatenate(c)

def get_feasible_radii(centers):

    b = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),

    diff = centers[:, None, :] - centers[None, :, :]
    d = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(d, np.inf)
    p = 0.5 * np.min(d, axis=1)
    return 0.95 * np.minimum(b, p)
