import numpy as np
from scipy.optimize import minimize, linprog

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
NUM_PAIRS = N * (N - 1) // 2

def fast_sum_radii(centers):
    """
    Fast heuristic for sum of radii given fixed centers.
    r_i = min(distance_to_boundary, 0.5 * min_distance_to_other_circles)
    """
    b = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    np.fill_diagonal(d, np.inf)
    p = 0.5 * np.min(d, axis=1)
    r = np.minimum(b, p)
    return np.sum(r), r

def solve_lp_radii(centers):
    """
    Solves LP to find exact maximum radii for fixed centers.
    Maximize sum(r_i) subject to r_i + r_j <= dist(i,j) and boundary constraints.
    """
    A_ub = np.zeros((NUM_PAIRS + 4 * N, N))
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
    for i in range(N):
        for _ in range(4):
            A_ub[idx, i] = 1.0
            idx += 1
            
    b_ub = np.zeros(NUM_PAIRS + 4 * N)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            b_ub[idx] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            idx += 1
    for i in range(N):
        x, y = centers[i]
        b_ub[idx] = x; idx += 1
        b_ub[idx] = 1.0 - x; idx += 1
        b_ub[idx] = y; idx += 1
        b_ub[idx] = 1.0 - y; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return res.x if res.success else None

def get_slsqp_constraints(v):
    """Inequality constraints for SLSQP: must be >= 0"""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    cons = [
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ]
    
    xi, xj = x[IDX_I], x[IDX_J]
    yi, yj = y[IDX_I], y[IDX_J]
    ri, rj = r[IDX_I], r[IDX_J]
    dists = np.sqrt((xi - xj)**2 + (yi - yj)**2 + 1e-16)
    cons.append(dists - (ri + rj))
    
    return np.concatenate(cons)

def get_slsqp_obj(v):
    return -np.sum(v[2::3])

def run_packing():

        return [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': get_slsqp_constraints}
    
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattices
    for seed in range(8):
        np.random.seed(seed)
        c = np.zeros((N, 2))
        r_est = 0.095
        row_counts = [6, 5, 6, 5, 4]
        y = 0.08

        dx = 0.17
        idx = 0

                    x += dx
                    idx += 1
            y += dy
        c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    # 2. Perturbed grids

                    best_radii[i] = mx
                    changed = True
            if not changed:
                break
                
        best_radii = np.maximum(best_radii, 0.0)
        
    return best_centers, best_radii, float(best_sum)
