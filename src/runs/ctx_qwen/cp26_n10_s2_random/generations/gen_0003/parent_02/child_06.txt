import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_sum_radii(centers):
    """Computes maximum valid radii for fixed centers and returns their sum."""
    # Distance to boundaries: min(x, 1-x, y, 1-y)
    b = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(d, np.inf)
    
    # Radius limited by half the distance to nearest neighbor
    p = 0.5 * np.min(d, axis=1)
    r = np.minimum(b, p)
    return np.sum(r), r

def obj_centers(centers_flat):
    """Objective for center optimization: maximize sum of radii."""
    centers = centers_flat.reshape(N_CIRCLES, 2)
    centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
    s, _ = compute_sum_radii(centers)
    return -s

def neg_sum_radii(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def get_bounds_3n():
    """Variable bounds for (x, y, r) per circle."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def constraint_3n(v):
    """Inequality constraints for SLSQP (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    # Non-overlap constraints: dist(i,j) >= r_i + r_j
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    
    mask = np.triu_indices(N_CIRCLES, k=1)
    c.append(dist[mask] - (r[mask[0]] + r[mask[1]]))
    return np.concatenate(c)

def generate_inits():
    """Generates diverse initial center configurations."""
    inits = []
    # Hexagonal row patterns summing to 26
    patterns = [
        [5,5,5,5,5,1], [6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], 
        [4,6,6,6,4], [5,5,6,5,5], [6,5,5,5,5], [5,6,6,5,4],
        [7,5,5,5,4], [4,5,6,6,5], [5,5,5,6,5], [6,4,6,5,5]
    ]
    
    for seed in range(25):
        pat = patterns[seed % len(patterns)]
        centers = []
        r_est = 0.09
        dy = r_est * np.sqrt(3)
        y = r_est
        rng = np.random.default_rng(seed)
        for i, count in enumerate(pat):
            x_start = r_est if i % 2 == 0 else 2 * r_est
            x = x_start
            for _ in range(count):
                centers.append([x + rng.uniform(-0.008, 0.008), y + rng.uniform(-0.008, 0.008)])
                x += 2 * r_est
            y += dy
        while len(centers) < N_CIRCLES:
            centers.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])
        inits.append(np.clip(np.array(centers[:N_CIRCLES]).flatten(), 0.01, 0.99))
        
    # Uniform random starts
    for seed in range(15):
        rng = np.random.default_rng(seed)
        c = rng.uniform(0.1, 0.9, (N_CIRCLES, 2))
        inits.append(np.clip(c.flatten(), 0.01, 0.99))
        
    return inits

def run_packing():
    best_c = None
    best_r = None
    best_sum = -1.0
    
    bounds_3n = get_bounds_3n()
    inits = generate_inits()
    
    # Phase 1: Optimize centers using Powell (derivative-free, robust to non-smooth min-operators)
    candidates = []
    for x0 in inits:
        try:
            res = minimize(obj_centers, x0, method='Powell', 
                           options={'maxiter': 3000, 'ftol': 1e-13})
            s = -res.fun
            candidates.append((s, res.x.reshape(N_CIRCLES, 2)))
        except Exception:
            pass
            
    # Sort by sum of radii (tuple comparison uses first element)
    candidates.sort(reverse=True)
    
    # Phase 2: SLSQP joint refinement on top candidates
    for s, c_opt in candidates[:8]:
        r_opt, _ = compute_sum_radii(c_opt)
        v0 = np.zeros(3 * N_CIRCLES)
        v0[0::3] = c_opt[:, 0]
        v0[1::3] = c_opt[:, 1]
        v0[2::3] = r_opt * 0.995  # Slightly feasible start
        
        try:
            res = minimize(neg_sum_radii, v0, method='SLSQP',
                           bounds=bounds_3n,
                           constraints={'type': 'ineq', 'fun': constraint_3n},
                           options={'maxiter': 4000, 'ftol': 1e-13})
            
            curr_sum = -res.fun
            if curr_sum > best_sum:
                v_best = res.x
                c_best = np.column_stack((v_best[0::3], v_best[1::3]))
                r_best = v_best[2::3].copy()
                best_sum = curr_sum
                best_c = c_best
                best_r = r_best
        except Exception:
            pass
            
    # Fallback safety
    if best_c is None:
        rng = np.random.default_rng(0)
        best_c = rng.uniform(0.1, 0.9, (N_CIRCLES, 2))
        _, best_r = compute_sum_radii(best_c)
        best_sum = np.sum(best_r)
        
    # Phase 3: Strict repair to guarantee validation passes within 1e-12 tolerance
    for _ in range(20):
        changed = False
        # Boundary repair
        for i in range(N_CIRCLES):
            mx = min(best_c[i,0], 1.0-best_c[i,0], best_c[i,1], 1.0-best_c[i,1])
            if best_r[i] > mx + 1e-9:
                best_r[i] = mx
                changed = True
                
        # Overlap repair
        for i in range(N_CIRCLES):
            for j in range(i+1, N_CIRCLES):
                d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                if d < best_r[i]+best_r[j]-1e-9:
                    shrink = (best_r[i]+best_r[j]-d)/2.0 + 1e-9
                    best_r[i] -= shrink
                    best_r[j] -= shrink
                    changed = True
                    
        if not changed:
            break
            
    best_r = np.maximum(best_r, 0.0)
    final_sum = float(np.sum(best_r))
    
    return best_c, best_r, final_sum
