import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
TRI_MASK = np.triu(np.ones((N_CIRCLES, N_CIRCLES), dtype=bool), k=1)
IDX_I, IDX_J = np.where(TRI_MASK)

def get_max_radii(centers):
    """Compute the maximum feasible radii for a fixed set of centers."""
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    rn = 0.5 * np.min(dist, axis=1)
    return np.minimum(rb, rn)

def objective(vars):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars[2*N_CIRCLES:])

def constraint_fun(vars):
    """Computes all boundary and non-overlap constraints (must be >= 0)."""
    c = vars[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    r = vars[2*N_CIRCLES:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    diff = c[:, None, :] - c[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    cons.append(dist[TRI_MASK] - (r[IDX_I] + r[IDX_J]))
    
    return np.concatenate(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = -1.0
    best_c = None
    best_r = None
    
    bounds = [(0, 1)]*(2*N_CIRCLES) + [(0, 0.5)]*N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    # Generate diverse initial configurations
    starts = []
    
    # 1. Hexagonal lattices with various shifts
    for shift in np.linspace(0, 0.08, 12):
        centers = []
        r0 = 0.1
        y = r0
        row = 0
        while len(centers) < N_CIRCLES:
            x = r0 + shift + (row % 2) * r0
            while x <= 1 - r0 and len(centers) < N_CIRCLES:
                centers.append([x, y])
                x += 2 * r0
            y += r0 * np.sqrt(3)
            row += 1
        while len(centers) < N_CIRCLES:
            centers.append([np.random.rand(), np.random.rand()])
        starts.append(np.array(centers[:N_CIRCLES]))
        
    # 2. Random starts
    for _ in range(15):
        centers = np.random.rand(N_CIRCLES, 2) * 0.8 + 0.1
        starts.append(centers)
        
    # Optimization loop over diverse starts
    for cfg in starts:
        r_init = 0.95 * get_max_radii(cfg)
        x0 = np.concatenate([cfg.flatten(), r_init])
        
        try:
            # Initial optimization
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-13})
            
            c_opt = res.x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
            r_opt = res.x[2*N_CIRCLES:]
            
            # Verify feasibility
            if np.min(constraint_fun(res.x)) >= -1e-8:
                curr_c, curr_r = c_opt, r_opt
                
                # Expand & Resolve loop: pushes boundaries of local minima
                for _ in range(25):
                    curr_r *= 1.0006
                    x0_exp = np.concatenate([curr_c.flatten(), curr_r])
                    res2 = minimize(objective, x0_exp, method='SLSQP', bounds=bounds,
                                    constraints=cons, options={'maxiter': 1500, 'ftol': 1e-13})
                    
                    if np.min(constraint_fun(res2.x)) >= -1e-8:
                        curr_c = res2.x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
                        curr_r = res2.x[2*N_CIRCLES:]
                    else:
                        break
                        
                curr_sum = np.sum(curr_r)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_c = curr_c.copy()
                    best_r = curr_r.copy()
        except Exception:
            continue
            
    # Final deterministic repair to guarantee strict validation compliance
    if best_c is not None:
        centers = best_c
        radii = best_r
        
        for _ in range(50):
            changed = False
            # Resolve overlaps by proportional shrinking
            for i in range(N_CIRCLES):
                for j in range(i+1, N_CIRCLES):
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i] + radii[j] - 1e-12:
                        shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                        radii[i] -= shrink
                        radii[j] -= shrink
                        changed = True
            
            # Clamp to boundaries
            for i in range(N_CIRCLES):
                mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
                if radii[i] > mx + 1e-12:
                    radii[i] = mx
                    changed = True
                    
            if not changed:
                break
        radii = np.maximum(radii, 0.0)
        best_c = centers
        best_r = radii
        
    return best_c, best_r, float(np.sum(best_r))
