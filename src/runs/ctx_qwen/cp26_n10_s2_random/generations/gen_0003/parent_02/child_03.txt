from scipy.optimize import minimize

N_CIRCLES = 26
TRIU_IDX = np.triu_indices(N_CIRCLES, k=1)
i_idx, j_idx = TRIU_IDX

def obj(v):

    return -np.sum(v[2*N_CIRCLES:])

def cons(v):
    """Inequality constraints: must be >= 0 for validity."""
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    n_constr = 4*N_CIRCLES + N_CIRCLES*(N_CIRCLES-1)//2
    c = np.empty(n_constr)

    c[2*N_CIRCLES:3*N_CIRCLES] = cy - r
    c[3*N_CIRCLES:4*N_CIRCLES] = 1.0 - cy - r

    dy = cy[:, None] - cy[None, :]

    r_sum = r[:, None] + r[None, :]
    c[4*N_CIRCLES:] = d[TRIU_IDX] - r_sum[TRIU_IDX]
    return c

    cx = v_c[::2]
    cy = v_c[1::2]
