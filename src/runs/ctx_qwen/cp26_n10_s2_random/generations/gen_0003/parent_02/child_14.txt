import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_constraints(v):
    """
    Computes inequality constraints (must be >= 0):
    - Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    - Non-overlap: ||c_i - c_j|| >= r_i + r_j
    """
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Overlap constraints (vectorized)
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dist = np.sqrt(dx**2 + dy**2)
    sr = r[:, np.newaxis] + r[np.newaxis, :]
    
    mask = np.triu(np.ones((N_CIRCLES, N_CIRCLES), dtype=bool), k=1)
    c.append(dist[mask] - sr[mask])
    
    return np.concatenate(c)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def get_feasible_radii(centers):
    """Computes maximum feasible radii for fixed centers, scaled slightly down for safety."""
    x, y = centers[:, 0], centers[:, 1]
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    r_pair = 0.5 * np.min(dist, axis=1)
    
    return np.minimum(r_bound, r_pair) * 0.92

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_v = None
    best_sum = -1.0
    
    # ---------------------------------------------------------
    # Phase 1: Diverse Initial Configurations
    # ---------------------------------------------------------
    starts = []
    
    # 1. Hexagonal lattices with various densities
    for seed in range(5):
        rng_loc = np.random.default_rng(seed)
        centers = np.zeros((N_CIRCLES, 2))
        r0 = 0.095 + seed * 0.003
        y = r0
        row = 0
        cnt = 0
        while cnt < N_CIRCLES:
            offset = r0 if row % 2 == 1 else 0.0
            x = r0 + offset
            while x + r0 <= 1.0 and cnt < N_CIRCLES:
                centers[cnt] = [x + rng_loc.uniform(-0.005, 0.005), y + rng_loc.uniform(-0.005, 0.005)]
                x += 2 * r0
                cnt += 1
            y += np.sqrt(3) * r0
            row += 1
        while cnt < N_CIRCLES:
            centers[cnt] = rng_loc.uniform(0.1, 0.9, 2)
            cnt += 1
        starts.append(centers)
        
    # 2. Perturbed square grids
    for seed in range(5):
        rng_loc = np.random.default_rng(seed + 50)
        grid = np.array(np.meshgrid(np.linspace(0.1, 0.9, 5), np.linspace(0.1, 0.9, 5))).T.reshape(-1, 2)
        centers = grid[:N_CIRCLES] + rng_loc.uniform(-0.02, 0.02, (N_CIRCLES, 2))
        starts.append(np.clip(centers, 0.05, 0.95))
        
    # 3. Corner-anchored + random
    for seed in range(5):
        rng_loc = np.random.default_rng(seed + 100)
        centers = rng_loc.uniform(0.15, 0.85, (N_CIRCLES, 2))
        # Force 4 circles into corners
        corners = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
        centers[:4] = corners
        starts.append(centers)
        
    # 4. Pure random
    for seed in range(10):
        rng_loc = np.random.default_rng(seed + 200)
        starts.append(rng_loc.uniform(0.1, 0.9, (N_CIRCLES, 2)))
        
    # ---------------------------------------------------------
    # Phase 2: Multi-Start Optimization
    # ---------------------------------------------------------
    for centers_init in starts:
        r_init = get_feasible_radii(centers_init)
        v0 = np.zeros(3 * N_CIRCLES)
        v0[0::3] = centers_init[:, 0]
        v0[1::3] = centers_init[:, 1]
        v0[2::3] = r_init
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            
            # Verify feasibility with tolerance
            c_vals = compute_constraints(res.x)
            if np.all(c_vals >= -1e-8):
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_v = res.x.copy()
        except Exception:
            continue
            
    # ---------------------------------------------------------
    # Phase 3: Iterative Growth & Perturbation Refinement
    # ---------------------------------------------------------
    if best_v is not None:
        for step in range(25):
            curr = best_v.copy()
            
            # Alternate between growing radii and perturbing centers
            if step % 2 == 0:
                curr[2::3] *= 1.0015
            else:
                curr[0::3] += rng.normal(0, 0.001, N_CIRCLES)
                curr[1::3] += rng.normal(0, 0.001, N_CIRCLES)
                curr[0::3] = np.clip(curr[0::3], 0.01, 0.99)
                curr[1::3] = np.clip(curr[1::3], 0.01, 0.99)
                
            try:
                res = minimize(objective, curr, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-14})
                c_vals = compute_constraints(res.x)
                if np.all(c_vals >= -1e-8):
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_v = res.x.copy()
            except Exception:
                pass
                
    # ---------------------------------------------------------
    # Phase 4: Final Repair for Strict Validation Compliance
    # ---------------------------------------------------------
    centers = best_v[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = best_v[2*N_CIRCLES:].copy()
    
    # Deterministic shrink to guarantee validation passes within 1e-12 tolerance
    for _ in range(20):
        changed = False
        # Resolve overlaps
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    overlap = (req - d) / 2.0 + 1e-9
                    radii[i] -= overlap
                    radii[j] -= overlap
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N_CIRCLES):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx + 1e-12:
                radii[i] = mx
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
