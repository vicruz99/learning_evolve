import numpy as np
from scipy.optimize import minimize, linprog

N = 26
NUM_PAIRS = N * (N - 1) // 2
TRIU_I, TRIU_J = np.triu_indices(N, k=1)

def get_lp_arrays(centers):
    """Constructs A_ub and b_ub for the LP: max sum(r) s.t. r_i + r_j <= d_ij, r_i <= boundaries."""
    A_ub = np.zeros((NUM_PAIRS + 4 * N, N))
    A_ub[:NUM_PAIRS, TRIU_I] = 1.0
    A_ub[:NUM_PAIRS, TRIU_J] = 1.0
    A_ub[NUM_PAIRS:NUM_PAIRS + 4 * N, :] = np.eye(N)
    
    b_ub = np.zeros(NUM_PAIRS + 4 * N)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub[:NUM_PAIRS] = dists[TRIU_I, TRIU_J]
    
    x, y = centers[:, 0], centers[:, 1]
    b_ub[NUM_PAIRS:NUM_PAIRS + N] = x
    b_ub[NUM_PAIRS + N:NUM_PAIRS + 2 * N] = 1.0 - x
    b_ub[NUM_PAIRS + 2 * N:NUM_PAIRS + 3 * N] = y
    b_ub[NUM_PAIRS + 3 * N:NUM_PAIRS + 4 * N] = 1.0 - y
    
    return A_ub, b_ub

def solve_lp_and_gradient(centers):
    """Solves LP for max radii sum and computes the gradient w.r.t centers using dual multipliers."""
    A_ub, b_ub = get_lp_arrays(centers)
    c_obj = -np.ones(N)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    
    if not res.success:
        return np.full(N, 0.0), np.zeros((N, 2))
        
    radii = res.x
    marginals = np.zeros(NUM_PAIRS + 4 * N)
    if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
        marginals = res.ineqlin.marginals
    
    # Compute gradient of optimal value w.r.t centers
    grad = np.zeros((N, 2))
    
    # Pairwise constraint gradients
    lam = marginals[:NUM_PAIRS]
    for k in range(NUM_PAIRS):
        if lam[k] > 1e-9:
            i, j = TRIU_I[k], TRIU_J[k]
            di = centers[i] - centers[j]
            d = np.linalg.norm(di)
            if d > 1e-9:
                v = di / d * lam[k]
                grad[i] += v
                grad[j] -= v
                
    # Boundary constraint gradients
    idx = NUM_PAIRS
    for i in range(N):
        mu_L = marginals[idx]     # r_i <= x_i
        mu_R = marginals[idx + 1] # r_i <= 1 - x_i
        mu_B = marginals[idx + 2] # r_i <= y_i
        mu_T = marginals[idx + 3] # r_i <= 1 - y_i
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        idx += 4
        
    return radii, grad

def compute_slsqp_obj(v):
    return -np.sum(v[2 * N:])

def compute_slsqp_cons(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    out = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    
    d2 = (c[:, np.newaxis, :] - c[np.newaxis, :, :]) ** 2
    dists = np.sqrt(np.sum(d2, axis=2))
    np.fill_diagonal(dists, np.inf)
    mask = np.triu_indices(N, k=1)
    out.append(dists[mask] - (r[mask[0]] + r[mask[1]]))
    return np.concatenate(out)

def generate_hex_starts(rng, seed_id):
    """Generates diverse hexagonal lattice configurations."""
    configs = []
    patterns = [[6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 6, 6, 6, 4]]
    pats = patterns * 4  # Repeat to fill seed_id
    pat = pats[seed_id % len(pats)]
    
    density_variants = [0.09, 0.095, 0.10, 0.102]
    r0 = density_variants[(seed_id // 5) % len(density_variants)]
    
    y = r0
    dy = r0 * np.sqrt(3)
    dx = 2 * r0
    row = 0
    centers = []
    
    for count in pat:
        shift = (dx / 2.0) if row % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(count):
            centers.append([x, y])
            x += dx
        y += dy
        row += 1
        
    centers = np.array(centers[:N])
    # Perturb
    noise = rng.normal(0, 0.003, centers.shape)
    centers += noise
    centers = np.clip(centers, 0.01, 0.99)
    configs.append(centers)
    return configs

            continue

                    radii[i] -= ov
                    radii[j] -= ov
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
