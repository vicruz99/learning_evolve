import numpy as np
from scipy.optimize import minimize
import math

N = 26

def compute_constraints(vars_flat):
    """Computes all boundary and non-overlap constraints for the packing."""
    centers = vars_flat[:2*N].reshape(N, 2)
    radii = vars_flat[2*N:]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c_bound = np.concatenate([
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
    ])
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    pair_dists = dists[mask]
    pair_radii_sum = (radii[:, np.newaxis] + radii[np.newaxis, :])[mask]
    c_overlap = pair_dists - pair_radii_sum
    
    return np.concatenate([c_bound, c_overlap])

def objective(vars_flat):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars_flat[2*N:])

def get_bounds():
    """Returns variable bounds for x, y, r."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def force_relax_init(n, seed, steps=3000):
    """Generates a densely packed configuration via vectorized force-directed relaxation."""
    rng = np.random.default_rng(seed)
    centers = rng.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, 0.02)
    
    for _ in range(steps):
        radii *= 1.00008
        
        # Vectorized circle-circle repulsion
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, 1e9)
        
        req = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlap = np.maximum(0, req - dists)
        force_mag = overlap * 10.0
        
        safe_dists = np.where(dists < 1e-9, 1e-9, dists)
        dir_vec = diff / safe_dists[:, :, np.newaxis]
        forces = np.sum(force_mag[:, :, np.newaxis] * dir_vec, axis=1)
        
        # Boundary repulsion
        for i in range(n):
            if centers[i, 0] < radii[i]: forces[i, 0] += 20.0 * (radii[i] - centers[i, 0])
            if centers[i, 0] > 1.0 - radii[i]: forces[i, 0] -= 20.0 * (centers[i, 0] - (1.0 - radii[i]))
            if centers[i, 1] < radii[i]: forces[i, 1] += 20.0 * (radii[i] - centers[i, 1])
            if centers[i, 1] > 1.0 - radii[i]: forces[i, 1] -= 20.0 * (centers[i, 1] - (1.0 - radii[i]))
            
        centers += forces * 0.005
        centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
        
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_vars = None
    best_obj = -np.inf
    candidates = []
    
    # 1. Generate candidates via force relaxation (finds tight local minima)
    for s in range(15):
        try:
            c, r = force_relax_init(N, seed=s)
            # Slightly shrink to guarantee strict initial feasibility for SLSQP
            candidates.append(np.concatenate([c.flatten(), r * 0.98]))
        except Exception:
            pass
            
    # 2. Generate candidates via structured hexagonal grids
    row_configs = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [6, 6, 5, 5, 4], [4, 6, 6, 6, 4], [5, 6, 6, 5, 4]
    ]
    for cfg in row_configs:
        for s in range(3):
            try:
                rng = np.random.default_rng(s)
                pts = []
                s_grid = 0.17
                y = 0.10
                for r_idx, count in enumerate(cfg):
                    shift = (s_grid / 2.0) if r_idx % 2 == 1 else 0.0
                    x = 0.10 + shift
                    for _ in range(count):
                        pts.append([x + rng.normal(0, 0.001), y + rng.normal(0, 0.001)])
                        x += s_grid
                    y += s_grid * math.sqrt(3) / 2.0
                c = np.array(pts[:N])
                
                # Compute strictly feasible initial radii
                r_bound = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                                     np.minimum(c[:, 1], 1.0 - c[:, 1]))
                dists = np.sqrt(np.sum((c[:, None, :] - c[None, :, :])**2, axis=2))
                np.fill_diagonal(dists, np.inf)
                r_pair = 0.5 * np.min(dists, axis=1)
                r = np.minimum(r_bound, r_pair) * 0.98
                candidates.append(np.concatenate([c.flatten(), r]))
            except Exception:
                pass
                
    # 3. Primary Optimization Phase
    for x0 in candidates:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if res.success:
                c_vals = compute_constraints(res.x)
                if np.all(c_vals >= -1e-8):
                    curr_obj = -res.fun
                    if curr_obj > best_obj:
                        best_obj = curr_obj
                        best_vars = res.x.copy()
        except Exception:
            continue
            
    # 4. Perturbation & Refinement Phase (Basin-Hopping style)
    if best_vars is not None:
        rng = np.random.default_rng(42)
        curr = best_vars.copy()
        for _ in range(60):
            pert = curr.copy()
            pert[:2*N] += rng.normal(0, 0.0005, 2*N)
            pert[2*N:] *= rng.uniform(0.998, 1.002, N)
            pert[:2*N] = np.clip(pert[:2*N], 1e-4, 1.0 - 1e-4)
            pert[2*N:] = np.maximum(pert[2*N:], 1e-6)
            
            try:
                res = minimize(objective, pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
                c_vals = compute_constraints(res.x)
                if np.all(c_vals >= -1e-8):
                    curr_obj = -res.fun
                    if curr_obj > best_obj:
                        best_obj = curr_obj
                        best_vars = res.x.copy()
                        curr = best_vars.copy()
            except Exception:
                pass
                
    # Fallback if optimization completely fails
    if best_vars is None:
        best_vars = candidates[0]
        
    # Final safety clamp to strictly satisfy validator tolerance
    for _ in range(30):
        c_check = compute_constraints(best_vars)
        if np.min(c_check) >= -1e-9:
            break
        best_vars[2*N:] *= 0.9998
        
    centers = best_vars[:2*N].reshape(N, 2)
    radii = best_vars[2*N:]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
