import numpy as np
from scipy.optimize import minimize

def compute_radii(centers):
    """
    Computes the maximum valid radius for each circle given fixed centers.
    r_i = min(dist to boundary, 0.5 * dist to nearest neighbor)
    """
    n = centers.shape[0]
    # Distance to boundaries: min(x, 1-x, y, 1-y)
    xb = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    yb = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    r_bound = np.minimum(xb, yb)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # Ignore self-distance
    
    # Radius limited by half the distance to the nearest neighbor
    r_pair = 0.5 * np.min(dists, axis=1)
    
    return np.minimum(r_bound, r_pair)

def objective(x):
    """
    Objective function for minimization: maximize sum of radii.
    """
    centers = x.reshape(-1, 2)
    # Strictly keep centers inside the unit square to ensure valid radii
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    radii = compute_radii(centers)
    return -np.sum(radii)

def hex_init(n, r, seed):
    """Generates a hexagonal lattice initialization with specified base radius."""
    rng = np.random.default_rng(seed)
    pts = []
    y = r
    row = 0
    while len(pts) < n:
        shift = r if row % 2 == 1 else 0.0
        x = r + shift
        while x <= 1.0 - r and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3.0) * r
        row += 1
    pts = np.array(pts[:n])
    # Add small random noise to break symmetry
    pts += rng.normal(0, 0.005, pts.shape)
    return np.clip(pts, 0.05, 0.95).flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize sum of radii."""
    np.random.seed(42)
    n = 26
    best_val = np.inf
    best_x = None
    
    starts = []
    
    # 1. Hexagonal lattices with various densities
    for r in [0.09, 0.095, 0.10, 0.105, 0.11, 0.115]:
        for s in range(4):
            starts.append(hex_init(n, r, s))
            
    # 2. Random starts (good for exploring non-lattice optima)
    rng = np.random.default_rng(123)
    for _ in range(20):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)).flatten())
        
    # 3. Corner-clustered configurations (helps fill corners efficiently)
    corners = np.array([[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]])
    for _ in range(15):
        c = rng.uniform(0.2, 0.8, (n-4, 2))
        c = np.vstack([corners, c])
        c += rng.normal(0, 0.01, c.shape)
        starts.append(np.clip(c, 0.05, 0.95).flatten())

    # Phase 1: Global search over diverse starts
    for x0 in starts:
        try:
            res = minimize(objective, x0, method='Powell', 
                           options={'maxiter': 4000, 'ftol': 1e-12, 'xtol': 1e-12})
            if res.fun < best_val:
                best_val = res.fun
                best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Iterative refinement (Basin-Hopping style) to escape local minima
    if best_x is not None:
        curr_x = best_x.copy()
        for step in range(40):
            noise_scale = 0.003 * (1.0 - step/50.0)
            pert = curr_x + rng.normal(0, noise_scale, curr_x.shape)
            pert = np.clip(pert, 1e-5, 1.0 - 1e-5)
            
            try:
                res = minimize(objective, pert, method='Powell',
                               options={'maxiter': 2000, 'ftol': 1e-12, 'xtol': 1e-12})
                if res.fun < best_val:
                    best_val = res.fun
                    best_x = res.x.copy()
                    curr_x = best_x.copy()
            except Exception:
                pass

    # Final extraction and safety adjustments
    centers = best_x.reshape(-1, 2)
    radii = compute_radii(centers)
    radii = np.maximum(radii, 0.0)
    sum_r = float(np.sum(radii))
    
    return centers, radii, sum_r
