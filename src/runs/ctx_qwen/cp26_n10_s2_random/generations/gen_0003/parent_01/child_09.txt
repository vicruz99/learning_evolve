import numpy as np
from scipy.optimize import minimize

N = 26

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def compute_constraints(vars_flat):
    """Computes all boundary and non-overlap constraints for the packing."""
    X = vars_flat.reshape(N, 3)
    xs = X[:, 0]
    ys = X[:, 1]
    rs = X[:, 2]

    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c1 = xs - rs
    c2 = 1.0 - xs - rs
    c3 = ys - rs
    c4 = 1.0 - ys - rs

    # Overlap constraints: (x_i-x_j)^2 + (y_i-y_j)^2 >= (r_i+r_j)^2
    idx = np.triu_indices(N, k=1)
    dx = xs[idx[0]] - xs[idx[1]]
    dy = ys[idx[0]] - ys[idx[1]]
    dr = rs[idx[0]] + rs[idx[1]]
    c5 = dx**2 + dy**2 - dr**2

    return np.concatenate([c1, c2, c3, c4, c5])

def compute_objective(vars_flat):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars_flat[2::3])

def hex_init(rows, r_guess, seed):
    """Generates a hexagonal lattice initialization with specified row counts."""
    rng = np.random.default_rng(seed)
    centers = []
    y = r_guess
    for r_idx, count in enumerate(rows):
        x_start = r_guess + (r_idx % 2) * r_guess
        for c in range(count):
            x = x_start + c * (2 * r_guess)
            centers.append([x, y])
        y += r_guess * np.sqrt(3)
    centers = np.array(centers[:N])
    # Add small noise to break symmetry
    centers += rng.normal(0, 0.002, (N, 2))
    centers = np.clip(centers, 0.01, 0.99)
    
    vars_flat = np.zeros(N * 3)
    vars_flat[0::3] = centers[:, 0]
    vars_flat[1::3] = centers[:, 1]
    vars_flat[2::3] = r_guess
    return vars_flat

def random_init(seed):
    """Generates a random initial configuration."""
    rng = np.random.default_rng(seed)
    centers = rng.uniform(0.1, 0.9, (N, 2))
    vars_flat = np.zeros(N * 3)
    vars_flat[0::3] = centers[:, 0]
    vars_flat[1::3] = centers[:, 1]
    vars_flat[2::3] = 0.08
    return vars_flat

                    valid = False

    sum_r = float(np.sum(radii))
    return centers, radii, sum_r
