import numpy as np
from scipy.optimize import minimize

N = 26

def compute_constraints(vars):
    """Computes all boundary and non-overlap constraints for the packing."""
    c = vars.reshape(N, 3)
    x = c[:, 0]
    y = c[:, 1]
    r = c[:, 2]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c_bnd = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c_ov = dist[mask] - r_sum[mask]
    
    return np.concatenate([c_bnd, c_ov])

def compute_objective(vars):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars[2::3])

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return b

def hex_init(seed, r_base, perturb):
    """Generates a hexagonal lattice initialization."""
    rng = np.random.default_rng(seed)
    pts = []
    y = r_base
    row = 0
    while len(pts) < N:
        x_start = r_base + (row % 2) * r_base
        x = x_start
        while x <= 1.0 - r_base and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r_base
        y += np.sqrt(3.0) * r_base
        row += 1
        
    pts = np.array(pts[:N])
    pts += rng.normal(0, perturb, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    vars = np.zeros(3 * N)
    vars[0::3] = pts[:, 0]
    vars[1::3] = pts[:, 1]
    vars[2::3] = r_base
    return vars

def random_init(seed):
    """Generates a random initial configuration."""
    rng = np.random.default_rng(seed)
    pts = rng.uniform(0.15, 0.85, (N, 2))
    vars = np.zeros(3 * N)
    vars[0::3] = pts[:, 0]
    vars[1::3] = pts[:, 1]
    vars[2::3] = 0.05
    return vars

def corner_init(seed):
    """Generates a corner-focused initial configuration."""
    rng = np.random.default_rng(seed)
    pts = np.array([[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]])
    rest = rng.uniform(0.2, 0.8, (N-4, 2))
    pts = np.vstack([pts, rest])
    vars = np.zeros(3 * N)
    vars[0::3] = pts[:, 0]
    vars[1::3] = pts[:, 1]
    vars[2::3] = 0.07
    return vars

def run_optimization(x0, bounds):
    """Runs SLSQP optimization and returns optimized variables and objective value."""
    try:
        res = minimize(
            compute_objective, x0,
            method='SLSQP', bounds=bounds,
            constraints={'type': 'ineq', 'fun': compute_constraints},
            options={'maxiter': 8000, 'ftol': 1e-11, 'disp': False}
        )
        return res.x, -res.fun
    except Exception:
        return x0, -np.inf

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize sum of radii."""
    bounds = get_bounds()
    best_vars = None
    best_sum = -1.0
    
    # Generate diverse initial configurations
    inits = []
    for r in [0.090, 0.095, 0.100, 0.108]:
        for s in range(4):
            inits.append(hex_init(s * 500, r, 0.001))
    for s in range(10):
        inits.append(random_init(s))
    for s in range(5):
        inits.append(corner_init(s))
        
    # Phase 1: Multi-start optimization
    for x0 in inits:
        x0_safe = x0.copy()
        x0_safe[2::3] *= 0.85  # Ensure initial feasibility
        x_opt, curr_sum = run_optimization(x0_safe, bounds)
        
        # Verify constraints with tolerance
        if np.all(compute_constraints(x_opt) >= -1e-8) and curr_sum > best_sum:
            best_sum = curr_sum
            best_vars = x_opt.copy()
            
    if best_vars is None:
        best_vars = inits[0]
        best_sum = 0.0
        
    # Phase 2: Basin-Hopping Refinement
    rng = np.random.default_rng(42)
    curr_vars = best_vars.copy()
    for _ in range(50):
        p = curr_vars.copy()
        # Perturb centers
        p[:2 * N] += rng.normal(0, 0.0012, 2 * N)
        # Perturb radii
        p[2 * N:] *= rng.uniform(0.99, 1.01, N)
        p = np.clip(p, 0.0, 1.0)
        p[2 * N:] = np.clip(p[2 * N:], 1e-8, 0.5)
        
        x_opt, curr_sum = run_optimization(p, bounds)
        if np.all(compute_constraints(x_opt) >= -1e-8) and curr_sum > best_sum:
            best_sum = curr_sum
            best_vars = x_opt.copy()
            curr_vars = best_vars.copy()
            
    # Phase 3: Final Feasibility Repair
    for _ in range(30):
        c_check = compute_constraints(best_vars)
        if np.min(c_check) >= -1e-9:
            break
        best_vars[2::3] *= 0.9999  # Conservative shrink to satisfy 1e-12 tolerance
        
    centers = best_vars.reshape(N, 3)[:, :2]
    radii = best_vars.reshape(N, 3)[:, 2]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
