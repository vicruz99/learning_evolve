import numpy as np
from scipy.optimize import minimize

N = 26

def obj_var(vars):
    """Objective: minimize negative sum of radii."""
    X = vars.reshape(N, 3)
    return -np.sum(X[:, 2])

def cons_var(vars):
    """Computes all boundary and non-overlap constraints."""
    X = vars.reshape(N, 3)
    con = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    con.append(X[:, 0] - X[:, 2])
    con.append(1.0 - X[:, 0] - X[:, 2])
    con.append(X[:, 1] - X[:, 2])
    con.append(1.0 - X[:, 1] - X[:, 2])
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    idx = np.triu_indices(N, 1)
    d = np.sqrt(np.sum((X[idx[0], :2] - X[idx[1], :2])**2, axis=1))
    con.append(d - (X[idx[0], 2] + X[idx[1], 2]))
    return np.concatenate(con)

def force_relax_init(seed, steps=2500):
    """Generates a densely packed configuration via force-directed relaxation."""
    rng = np.random.default_rng(seed)
    c = rng.uniform(0.15, 0.85, (N, 2))
    r = np.full(N, 0.035)
    
    for _ in range(steps):
        r *= 1.0004
        f = np.zeros_like(c)
        
        # Circle-circle repulsion
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
                req = r[i] + r[j]
                if d < req:
                    push = (req - d) * 8.0
                    if d > 1e-9:
                        dx, dy = c[i,0]-c[j,0], c[i,1]-c[j,1]
                        f[i,0] += dx/d * push
                        f[i,1] += dy/d * push
                        f[j,0] -= dx/d * push
                        f[j,1] -= dy/d * push
                        
        # Boundary repulsion
        for i in range(N):
            if c[i,0] < r[i]: f[i,0] += 60.0*(r[i]-c[i,0])
            if c[i,0] > 1.0-r[i]: f[i,0] -= 60.0*(c[i,0]-(1.0-r[i]))
            if c[i,1] < r[i]: f[i,1] += 60.0*(r[i]-c[i,1])
            if c[i,1] > 1.0-r[i]: f[i,1] -= 60.0*(c[i,1]-(1.0-r[i]))
            
        c += f * 0.004
        c = np.clip(c, 0.02, 0.98)
    return c, r

def make_hex_init_var(counts, noise=0.0):
    """Generates a hexagonal lattice initialization with specified row counts."""
    rng = np.random.default_rng(42)
    pts = []
    r_est = 0.105
    dy = r_est * np.sqrt(3)
    y = r_est
    for i, cnt in enumerate(counts):
        x_start = r_est + (0.5 * r_est if i % 2 == 1 else 0.0)
        for _ in range(cnt):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += dy
    pts = np.array(pts[:N])
    if noise > 0:
        pts += rng.normal(0, noise, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    X = np.zeros((N, 3))
    X[:, :2] = pts
    X[:, 2] = 0.095
    return X.flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-7, 0.5)] * N
    
    best_vars = None
    best_val = -np.inf
    
    candidates = []

        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],

        candidates.append(X.flatten())

        try:

                           options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})

                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_vars = res.x.copy()
        except Exception:
            pass

    # Perturbation & refinement phase to escape local minima
    if best_vars is not None:
        rng = np.random.default_rng(123)
        for _ in range(30):
            x_curr = best_vars.copy()
            x_curr += rng.normal(0, 0.0007, x_curr.shape)
            x_curr = np.clip(x_curr, 0.0, 1.0)

                        best_vars = res.x.copy()

                pass
                
    # Fallback if optimization fails
    if best_vars is None:
        best_vars = candidates[0]
        
    X = best_vars.reshape(N, 3)

                    X[i,2] -= sh
                    X[j,2] -= sh
                    changed = True

                if X[i,k] < X[i,2] + 1e-11:
                    X[i,2] = X[i,k] - 1e-11
                    changed = True

                    X[i,2] = 1.0 - X[i,k] - 1e-11
                    changed = True
        if not changed:
            break
            
    X[:, 2] = np.maximum(X[:, 2], 0.0)
    centers = X[:, :2]
    radii = X[:, 2]
    return centers, radii, float(np.sum(radii))
