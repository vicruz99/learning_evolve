from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx_lp = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx_lp, i] = 1.0
        A_LP[idx_lp, j] = 1.0
        PAIR_IDX.append((i, j))
        idx_lp += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diffs = centers[:, None, :] - centers[None, :, :]

        b[k] = dists[i, j]
        k += 1

        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1

        duals = res.marginals.ineqlin
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:

def obj_lp(x):
    centers = x.reshape(N, 2)
    _, s, g = solve_lp_and_grad(centers)
    return -s, -g.flatten()

def coord_obj(x, centers, idx):
    temp = centers.copy()
    temp[idx] = np.clip(x, 1e-5, 1.0 - 1e-5)
    _, s, _ = solve_lp_and_grad(temp)
    return -s

def coord_opt(centers, idx):
    bounds = [(0.001, 0.999), (0.001, 0.999)]
    res = minimize(coord_obj, centers[idx], args=(centers, idx), method='L-BFGS-B', bounds=bounds, options={'maxiter': 500, 'ftol': 1e-12})
    return res.x

def force_init(rng):
    c = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(1500):
        f = np.zeros_like(c)
        for i in range(N):

                dv = c[i] - c[j]
                d = np.linalg.norm(dv)

                    f[i] += dv / d * push
                    f[j] -= dv / d * push
        c += f * 0.1
        c = np.clip(c, 0.02, 0.98)
    return c

def repair(centers, radii):

        changed = False

                changed = True

def constraints_joint(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)

    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def objective_joint(v):
