N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute constant LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1

def solve_lp_and_grad(centers):

def coord_refine(centers, cycles=3):
    """Coordinate-wise optimization of centers."""
    c = centers.copy()
    for _ in range(cycles):
        for i in range(N):
            others = np.delete(c, i, axis=0)
            def obj_2d(xy):

                _, s, _ = solve_lp_and_grad(temp)
                return -s

                res = minimize(obj_2d, c[i], method='Nelder-Mead',

                if res.success and res.fun < -np.sum(solve_lp_and_grad(c)[1]) - 1e-8:
                    c[i] = res.x
            except Exception:
                pass
    return c

def constraints_joint(v):
    """Boundary and pairwise non-overlap constraints for SLSQP."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    dx = c[TRIU_I, 0] - c[TRIU_J, 0]
    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def objective_joint(v):
    return -np.sum(v[2*N:])

def slsqp_polish(c0, r0, maxiter=8000):
    """Joint SLSQP optimization with high precision."""
    v0 = np.concatenate([c0.flatten(), r0])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(objective_joint, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constraints_joint},
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if np.min(constraints_joint(res.x)) >= -1e-8:
            return res.x[:2*N].reshape(N, 2), res.x[2*N:], np.sum(res.x[2*N:])
    except Exception:
        pass
    return c0, r0, np.sum(r0)

def force_init(rng):
    """Generates well-spaced configuration via repulsive forces."""
    c = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(600):
        f = np.zeros_like(c)
        for i in range(N):
            for j in range(i+1, N):
                dv = c[i] - c[j]
                d = np.linalg.norm(dv)
                if d < 0.22 and d > 1e-4:
                    push = (0.22 - d) * 0.04 / (d + 1e-4)
                    f[i] += dv / d * push
                    f[j] -= dv / d * push
        c += f * 0.6
        c = np.clip(c, 0.05, 0.95)
    return c

                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
