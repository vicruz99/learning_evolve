from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute constant LP constraint matrix structure for speed
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
k = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[k, i] = 1.0
        A_LP[k, j] = 1.0

        k += 1
for i in range(N):

def solve_lp_and_grad(centers):

                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))

    c = x.reshape(N, 2)
    r, s, g, d = solve_lp_and_grad(c)
    return -s, -g.flatten()

def local_opt(c0):
    """Runs L-BFGS-B local optimization on centers."""
    c = np.clip(c0, 0.02, 0.98)
    try:
        res = minimize(obj_and_grad, c.flatten(), method='L-BFGS-B', jac=True,
                       bounds=[(0.02, 0.98)] * (2 * N),
                       options={'maxiter': 3000, 'ftol': 1e-13, 'gtol': 1e-10})
        return res.x.reshape(N, 2), -res.fun
    except Exception:
        return c, 0.0

def slsqp_obj(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def slsqp_cons(v):
    """Constraints for joint SLSQP: boundaries and non-overlap (squared)."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    dx = c[TRIU_I, 0] - c[TRIU_J, 0]
    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def slsqp_polish(c, r):
    """Joint SLSQP optimization of centers and radii."""
    v0 = np.concatenate([c.flatten(), r])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

        pass

def repair(centers, radii):

                changed = True
        if not changed:

    return np.maximum(radii, 0.0)

        c = np.clip(c, 0.02, 0.98)
    return c
