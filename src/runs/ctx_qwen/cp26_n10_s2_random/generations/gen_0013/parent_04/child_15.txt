The problem is non-smooth due to the distance constraints $r_i + r_j \le \|c_i - c_j\|$, making gradient-based methods sensitive to initial configurations.

Key improvements in this solution:
1. **Hybrid Optimization**: Combines L-BFGS-B (good for smooth regions and bounds) with Subgradient Ascent (robust for non-smooth LP dual gradients). L-BFGS-B quickly converges, while SG ascent handles kinks in the objective landscape.
2. **Diverse High-Quality Starts**: Generates hexagonal lattice patterns, force-directed repulsion layouts, and corner-biased configurations to cover different topological packing regimes.
3. **Swap & Perturb Exploration**: Explicitly swaps circle positions and applies targeted noise. This breaks symmetry and allows circles to exchange roles in the contact graph, crucial for escaping local traps.
4. **Coordinate-wise Refinement**: Optimizes each circle's position independently while fixing others. This is highly effective for fine-tuning tightly packed configurations.
5. **Joint SLSQP Polish**: Simultaneously optimizes centers and radii using quadratic constraints to tightly resolve overlaps and boundary contacts.
6. **Strict Numerical Repair**: A final conservative pass ensures all constraints are satisfied within the validator's tolerance without unnecessarily shrinking radii.
7. **No Closures**: All helper functions are defined at the top level to comply with constraints and improve modularity/reusability.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def get_lp_and_grad(centers):
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    b = np.zeros(NUM_PAIRS + 4 * N)
    k = 0
    for i, j in PAIR_IDX:
        b[k] = dists[i, j]; k += 1
    for i in range(N):
        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(centers), np.zeros(len(b))
        
    radii = res.x
    s_sum = np.sum(radii)
    duals = np.zeros(len(b))
    
    if hasattr(res, 'marginals') and res.marginals is not None:
        duals = res.marginals.ineqlin
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = res.ineqlin.marginals
        
    grad = np.zeros_like(centers)
    k = 0
    for i, j in PAIR_IDX:
        mu = duals[k]
        if mu > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
        k += 1
        
    b_start = NUM_PAIRS
    for i in range(N):
        grad[i, 0] += duals[b_start + 4*i] - duals[b_start + 4*i + 1]
        grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
        
    return radii, s_sum, grad, duals

def lbfgs_obj_grad(x_flat):
    centers = np.clip(x_flat.reshape(N, 2), 1e-6, 1.0 - 1e-6)
    _, val, grad, _ = get_lp_and_grad(centers)
    return -val, -grad.flatten()

def lbfgs_optimize(c0, maxiter=2000):
    bounds = [(1e-5, 1.0 - 1e-5)] * (2 * N)
    try:
        res = minimize(lbfgs_obj_grad, c0.flatten(), method='L-BFGS-B', jac=True,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        c_opt = res.x.reshape(N, 2)
        _, s_opt, _, _ = get_lp_and_grad(c_opt)
        return c_opt, s_opt
    except Exception:
        return c0, -1.0

def sg_ascent(c0, max_iter=1500, init_step=0.008):
    c = c0.copy()
    best_c, best_s = c.copy(), -1.0
    _, best_s, _, _ = get_lp_and_grad(c)
    step = init_step
    
    for _ in range(max_iter):
        _, s, g, _ = get_lp_and_grad(c)
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
            
        direction = g / gn
        improved = False
        for factor in [1.0, 0.5, 0.2, 0.1]:
            trial_step = step * factor
            nc = np.clip(c + trial_step * direction, 1e-5, 1.0 - 1e-5)
            _, ns, _, _ = get_lp_and_grad(nc)
            if ns > s + 1e-12:
                c = nc
                s = ns
                step = min(step * 1.15, 0.04)
                improved = True
                if s > best_s:
                    best_s = s
                    best_c = c.copy()
                break
        if not improved:
            step *= 0.6
            if step < 1e-9:
                break
    return best_c, best_s

def force_init(rng):
    c = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(800):
        diffs = c[:, None, :] - c[None, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        dists = np.maximum(dists, 1e-6)
        inv_d2 = 1.0 / (dists**2)
        np.fill_diagonal(inv_d2, 0.0)
        f = np.zeros_like(c)
        for d in range(2):
            f[:, d] = np.sum(diffs[:, :, d] * inv_d2 / dists, axis=1)
        c += 0.003 * f
        c = np.clip(c, 0.02, 0.98)
    return c

def coord_obj(x, centers, idx):
    temp = centers.copy()
    temp[idx] = np.clip(x, 1e-5, 1.0 - 1e-5)
    _, s, _, _ = get_lp_and_grad(temp)
    return -s

def slsqp_obj(v):
    return -np.sum(v[2 * N:])

def slsqp_cons(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    dx = c[TRIU_I, 0] - c[TRIU_J, 0]
    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    return np.concatenate([con, dx**2 + dy**2 - dr**2])

def run_packing() -> tuple:
    rng = np.random.default_rng(42)
    best_c = None
    best_s = -1.0
    
    # Phase 1: Diverse starts + L-BFGS-B
    starts = []
    # Hexagonal patterns
    pats = [[6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], [4,6,6,6,4]]
    for pat in pats:
        for r0 in [0.095, 0.100, 0.105]:
            c = []
            y = r0
            for ri, cnt in enumerate(pat):
                sh = r0 if ri % 2 == 1 else 0.0
                x = r0 + sh
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            starts.append(np.clip(np.array(c[:N]), 0.05, 0.95))
            
    # Force-directed starts
    for _ in range(15):
        starts.append(force_init(rng))
        
    # Corner-biased
    for _ in range(10):
        c = rng.uniform(0.2, 0.8, (N, 2))
        c[:4] = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        starts.append(np.clip(c, 0.05, 0.95))
        
    for c0 in starts:
        c_opt, s_opt = lbfgs_optimize(c0, 2500)
        if s_opt > best_s:
            best_s = s_opt
            best_c = c_opt.copy()
            
    # Phase 2: Subgradient Ascent Polish
    c_opt, s_opt = sg_ascent(best_c, 2000, 0.006)
    if s_opt > best_s:
        best_s = s_opt
        best_c = c_opt.copy()
        
    # Phase 3: Swap & Perturb Exploration
    for step in range(600):
        strategy = rng.integers(0, 3)
        c_try = best_c.copy()
        
        if strategy == 0:
            idx = rng.choice(N, size=rng.integers(2, 6), replace=False)
            scale = 0.015 * (1.0 - step/700.0)
            c_try[idx] += rng.normal(0, scale, (len(idx), 2))
        elif strategy == 1:
            i, j = rng.choice(N, 2, replace=False)
            c_try[i], c_try[j] = c_try[j].copy(), c_try[i].copy()
        else:
            c_try += rng.normal(0, 0.004 * (1.0 - step/700.0), c_try.shape)
            
        c_try = np.clip(c_try, 0.02, 0.98)
        c_opt, s_opt = lbfgs_optimize(c_try, 1500)
        
        if s_opt > best_s:
            best_s = s_opt
            best_c = c_opt.copy()
            
        # Periodic SG ascent to refine
        if step % 50 == 0:
            c_opt, s_opt = sg_ascent(best_c, 800, 0.004)
            if s_opt > best_s:
                best_s = s_opt
                best_c = c_opt.copy()
                
    # Phase 4: Coordinate-wise optimization
    c = best_c.copy()
    for _ in range(3):
        for i in range(N):
            try:
                res = minimize(coord_obj, c[i], args=(c, i), method='Powell', options={'maxiter': 200, 'xtol': 1e-9})
                if res.success:
                    c[i] = np.clip(res.x, 1e-5, 1.0-1e-5)
            except Exception:
                pass
    _, s_coord, _, _ = get_lp_and_grad(c)
    if s_coord > best_s:
        best_s = s_coord
        best_c = c.copy()
        
    # Phase 5: Joint SLSQP Polish
    r_lp, _, _, _ = get_lp_and_grad(best_c)
    v0 = np.concatenate([best_c.flatten(), r_lp])
    bounds_sl = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    try:
        res_sl = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds_sl,
                          constraints={'type': 'ineq', 'fun': slsqp_cons},
                          options={'maxiter': 8000, 'ftol': 1e-14})
        if np.min(slsqp_cons(res_sl.x)) >= -1e-7:
            s_sl = np.sum(res_sl.x[2 * N:])
            if s_sl > best_s:
                best_s = s_sl
                best_c = res_sl.x[:2 * N].reshape(N, 2)
                r_lp = res_sl.x[2 * N:]
    except Exception:
        pass
        
    # Final LP verification
    r_final, s_final, _, _ = get_lp_and_grad(best_c)
    if s_final > best_s:
        best_s = s_final
        r_final = r_final
        
    # Repair
    radii = r_final.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(best_c[i, 0], 1.0 - best_c[i, 0], best_c[i, 1], 1.0 - best_c[i, 1])
            if radii[i] > mr - 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    
    return best_c, radii, float(np.sum(radii))
