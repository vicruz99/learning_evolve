import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0


def solve_lp_and_grad(centers):
    """Solves LP for maximal radii and computes gradient via dual variables."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)

    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-20)

    b = np.zeros(NUM_PAIRS + 4 * N)
    k = 0
    for i, j in PAIR_IDX:
        b[k] = dists[i, j]
        k += 1
    for i in range(N):
        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1

    bounds = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, bounds=bounds, method='highs')

    if res.success:
        try:
            duals = res.marginals.ineqlin
        except AttributeError:
            try:
                duals = res.ineqlin.marginals
            except Exception:
                duals = np.zeros(len(b))
    else:
        duals = np.zeros(len(b))

    radii = res.x if res.success else np.full(N, 1e-6)
    s_sum = np.sum(radii)

    grad = np.zeros_like(centers)
    k = 0
    for i, j in PAIR_IDX:
        mu = duals[k]
        if mu > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
        k += 1

    b_start = NUM_PAIRS
    for i in range(N):
        grad[i, 0] += duals[b_start + 4 * i] - duals[b_start + 4 * i + 1]
        grad[i, 1] += duals[b_start + 4 * i + 2] - duals[b_start + 4 * i + 3]

    return radii, s_sum, grad


def lbfgs_obj_grad(x_flat):
    """Objective and gradient for L-BFGS-B: minimize negative sum of radii."""
    c = np.clip(x_flat.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    _, val, grad = solve_lp_and_grad(c)
    return -val, -grad.flatten()


def lbfgs_optimize(c0, maxiter=3000):
    """Run L-BFGS-B optimization on centers."""
    bounds = [(0.01, 0.99)] * (2 * N)
    try:
        res = minimize(lbfgs_obj_grad, c0.flatten(), method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'disp': False})
        return res.x.reshape(N, 2), -res.fun
    except Exception:
        r, s, _ = solve_lp_and_grad(c0)
        return c0.copy(), s


def gradient_ascent(centers0, max_iter=2000, init_step=0.005):
    """Gradient ascent on centers using LP dual subgradients."""
    centers = centers0.copy()
    best_c = centers.copy()
    best_s = -1.0
    step = init_step
    momentum = np.zeros_like(centers)

    _, best_s, _ = solve_lp_and_grad(centers)

    for it in range(max_iter):
        r, s, g, = solve_lp_and_grad(centers)
        if s > best_s:
            best_s = s
            best_c = centers.copy()

        gn = np.linalg.norm(g)
        if gn < 1e-11:
            break

        g_dir = g / gn
        momentum = 0.3 * momentum + step * g_dir

        nc = centers + momentum
        nc = np.clip(nc, 1e-5, 1.0 - 1e-5)

        _, ns, _ = solve_lp_and_grad(nc)

        if ns > s + 1e-13:
            centers = nc
            step = min(step * 1.06, 0.04)
            if ns > best_s:
                best_s = ns
                best_c = centers.copy()
        else:
            step *= 0.6
            momentum *= 0.3
            if step < 1e-12:
                break

    return best_c, best_s


def force_repel(centers, steps=500, strength=0.015):
    """Spread circles apart using repulsive forces."""
    c = centers.copy()
    for _ in range(steps):
        diffs = c[:, None, :] - c[None, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        dists_safe = np.maximum(dists, 1e-6)
        np.fill_diagonal(dists_safe, np.inf)

        force_mag = np.where(dists < 0.30, strength / (dists_safe**2), 0.0)
        f = np.zeros_like(c)
        for d in range(2):
            f[:, d] = np.sum(diffs[:, :, d] * force_mag / dists_safe, axis=1)

        mask_low = c < 0.03
        mask_high = c > 0.97
        f[mask_low] += strength * 10.0
        f[mask_high] -= strength * 10.0

        c += f * 0.003
        c = np.clip(c, 0.02, 0.98)
    return c


def slsqp_obj(v):
    return -np.sum(v[2 * N:])


def slsqp_cons(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    dx = c[TRIU_I, 0] - c[TRIU_J, 0]
    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    con = np.concatenate([con, dx ** 2 + dy ** 2 - dr ** 2])
    return con


def slsqp_polish(centers, radii, maxiter=8000):
    """Joint SLSQP optimization of centers and radii."""
    v0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if np.min(slsqp_cons(res.x)) >= -1e-7:
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:], -res.fun
    except Exception:
        pass
    return centers.copy(), radii.copy(), np.sum(radii)


def coord_optimize(centers, cycles=3):
    """Coordinate-wise optimization of each circle's position."""
    c = centers.copy()
    for cycle in range(cycles):
        for i in range(N):
            def coord_obj(x):
                temp = c.copy()
                temp[i] = np.clip(x, 1e-4, 1.0 - 1e-4)
                _, s, _ = solve_lp_and_grad(temp)
                return -s

            try:
                res = minimize(coord_obj, c[i], method='Powell',
                               options={'maxiter': 300, 'xtol': 1e-9})
                if res.success:
                    c[i] = np.clip(res.x, 1e-4, 1.0 - 1e-4)
            except Exception:
                pass
    return c


def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)


def generate_hex_pattern(rng, pattern, r_est, angle=0.0):
    """Generate hexagonal lattice with optional rotation."""
    c = []
    y = r_est
    for ri, cnt in enumerate(pattern):
        sh = r_est if ri % 2 == 1 else 0.0
        x = r_est + sh
        for _ in range(cnt):
            if len(c) < N:
                c.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)

    result = np.array(c[:N])

    if angle != 0.0:
        cos_a = np.cos(angle)
        sin_a = np.sin(angle)
        cx, cy = 0.5, 0.5
        xc = result[:, 0] - cx
        yc = result[:, 1] - cy
        result[:, 0] = cos_a * xc - sin_a * yc + cx
        result[:, 1] = sin_a * xc + cos_a * yc + cy

    return np.clip(result, 0.05, 0.95)


def generate_starts(rng):

    starts = []

        [5, 5, 5, 6, 4], [6, 4, 5, 6, 5]
    ]

    # Hexagonal patterns with various densities and rotations
    for pat in patterns:
        for r_est in [0.090, 0.095, 0.100, 0.105, 0.110]:
            starts.append(generate_hex_pattern(rng, pat, r_est, angle=0.0))
            # Add rotated variants
            for angle in [np.pi / 6, np.pi / 4, np.pi / 3]:
                s = generate_hex_pattern(rng, pat, r_est, angle=angle)
                starts.append(s)

    # Force-directed layouts
    for _ in range(15):

        c = force_repel(c, steps=800, strength=0.018)
        starts.append(c)

    # Corner-biased starts
    for _ in range(12):
        c = rng.uniform(0.15, 0.85, (N, 2))
        # Place some circles near corners
        corners = [[0.07, 0.07], [0.93, 0.07], [0.07, 0.93], [0.93, 0.93]]
        for k in range(4):

                    corners[k][1] + rng.normal(0, 0.01)]
        starts.append(np.clip(c, 0.03, 0.97))

    # Edge-biased starts
    for _ in range(8):
        c = rng.uniform(0.1, 0.9, (N, 2))
        # Some circles near edges
        n_edge = rng.integers(4, 10)
        for k in range(n_edge):
            edge = rng.integers(0, 4)
            if edge == 0:
                c[k] = [rng.uniform(0.05, 0.95), 0.05 + rng.normal(0, 0.01)]

            else:
                c[k] = [0.95 + rng.normal(0, 0.01), rng.uniform(0.05, 0.95)]
