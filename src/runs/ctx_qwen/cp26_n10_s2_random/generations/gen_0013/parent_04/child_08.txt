N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute constant LP constraint matrix structure for speed
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):

    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii given fixed centers and computes gradient via duals."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    b = np.zeros(NUM_PAIRS + 4 * N)
    k = 0
    for i, j in PAIR_IDX:
        b[k] = dists[i, j]
        k += 1

        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1

            try:
                duals = res.marginals.ineqlin
            except AttributeError:
                duals = np.zeros(len(b))
        else:
            duals = np.zeros(len(b))
    except Exception:
        duals = np.zeros(len(b))
        res = None
        
    radii = res.x if res is not None and res.success else np.zeros(N)
    s_sum = np.sum(radii)

    k = 0

        k += 1

    return radii, s_sum, grad, duals

def lbfgs_obj_grad(x_flat):
    """Objective and exact gradient for L-BFGS-B optimization of centers."""
    centers = x_flat.reshape(N, 2)
    _, s, g, _ = solve_lp_and_grad(centers)
    return -s, -g.flatten()

def force_repel(centers, steps=500, strength=0.015):
    """Spreads circles apart using repulsive forces to avoid clustering."""
    c = centers.copy()
    for _ in range(steps):
        diffs = c[:, None, :] - c[None, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        dists = np.maximum(dists, 1e-6)
        force_mag = np.where(dists < 0.25, strength / dists**2, 0.0)
        f = np.zeros_like(c)
        for d in range(2):
            f[:, d] = np.sum(diffs[:, :, d] * force_mag / dists, axis=1)
        # Boundary repulsion
        mask_low = c < 0.03
        mask_high = c > 0.97
        f[mask_low] += strength * 10.0
        f[mask_high] -= strength * 10.0
        c += f * 0.005
        c = np.clip(c, 0.01, 0.99)
    return c

def slsqp_obj(v):
    return -np.sum(v[2 * N:])

def slsqp_cons(v):
    """Inequality constraints for SLSQP: boundaries and non-overlap (squared)."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    dx = c[TRIU_I, 0] - c[TRIU_J, 0]
    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    con = np.concatenate([con, dx**2 + dy**2 - dr**2])
    return con

def slsqp_polish(centers, radii, maxiter=8000):
    """Joint SLSQP optimization of centers and radii."""
    v0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if np.min(slsqp_cons(res.x)) >= -1e-7:
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:], -res.fun
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(100):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-12:
                radii[i] = mr
                changed = True
        # Resolve pairwise overlaps by symmetric shrinking
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])

                    shrink = (req - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
