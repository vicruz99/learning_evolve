1. **Joint Optimization (SLSQP)**: First, we optimize centers and radii jointly using SLSQP. This leverages gradient information to quickly navigate the feasible space. We use the direct distance constraint `dist >= r_i + r_j` for better conditioning than squared distances.
2. **LP Refinement**: For any fixed set of centers, the optimal radii that maximize the sum while satisfying non-overlap and boundary constraints can be found exactly via a Linear Program (LP). We use `scipy.optimize.linprog` with the 'highs' solver for this. This step guarantees that for any center configuration, we are extracting the absolute maximum possible sum of radii.
3. **Multi-Start & Perturbation**: We initialize from diverse configurations (hexagonal lattice, random dense, perturbed grid) and run multiple SLSQP optimizations. After finding a local optimum, we perturb it slightly and re-optimize to escape shallow local minima.
4. **Powell Local Search on Centers**: The joint objective landscape is non-convex. We run a derivative-free Powell optimizer on the centers alone, using the LP-sum as the objective. This helps fine-tune center positions to relieve tight constraints, often squeezing out additional radius sum that gradient-based methods miss.
5. **Safety Validation**: Finally, we apply a strict feasibility check against the validator's tolerances, shrinking radii minimally if necessary to guarantee a valid output.

This combination of gradient-based search, exact LP projection, and derivative-free local refinement is highly effective for continuous circle packing problems.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_lp_radii(centers):
    """
    Given fixed centers, compute the optimal radii that maximize sum(r)
    using Linear Programming.
    """
    n = centers.shape[0]
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    b = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    b = np.maximum(b, 0.0)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    
    # LP formulation:
    # Maximize sum(r_i)  <=>  Minimize -sum(r_i)
    # Subject to:
    #   r_i + r_j <= dist[i, j]  for all i < j
    #   r_i <= b[i]              for all i
    #   r_i >= 0                 for all i
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist[i, j])
            
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(b[i])
        
    res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                  bounds=(0.0, None), method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def objective_joint(vars_flat):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def constraints_joint(vars_flat):
    """
    Inequality constraints for SLSQP:
    Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    Overlap: dist(i,j) >= r_i + r_j
    All must be >= 0.
    """
    X = vars_flat.reshape(N, 3)
    xs, ys, rs = X[:, 0], X[:, 1], X[:, 2]
    c = []
    
    # Boundary constraints
    c.append(xs - rs)
    c.append(1.0 - xs - rs)
    c.append(ys - rs)
    c.append(1.0 - ys - rs)
    
    # Overlap constraints
    idx = np.triu_indices(N, k=1)
    dx = xs[idx[0]] - xs[idx[1]]
    dy = ys[idx[0]] - ys[idx[1]]
    dr = rs[idx[0]] + rs[idx[1]]
    dist = np.sqrt(dx**2 + dy**2)
    c.append(dist - dr)
    
    return np.concatenate(c)

def get_initial_configs(rng):
    """Generates diverse feasible initial configurations."""
    configs = []
    
    # 1. Hexagonal lattice
    hex_pts = []
    r0 = 0.095
    y = r0
    row = 0
    while len(hex_pts) < N:
        x_start = r0 + (row % 2) * r0
        x = x_start
        while x + r0 <= 1.0 + 1e-9 and len(hex_pts) < N:
            hex_pts.append([x, y])
            x += 2 * r0
        y += r0 * np.sqrt(3)
        row += 1
    hex_pts = np.array(hex_pts[:N])
    hex_pts += rng.normal(0, 0.005, hex_pts.shape)
    hex_pts = np.clip(hex_pts, 0.05, 0.95)
    configs.append(hex_pts)
    
    # 2. Random dense
    rand_pts = rng.uniform(0.1, 0.9, (N, 2))
    configs.append(rand_pts)
    
    # 3. Perturbed 5x5 grid + 1 center
    grid_pts = []
    for i in range(5):
        for j in range(5):
            grid_pts.append([0.1 + 0.2*i, 0.1 + 0.2*j])
    grid_pts.append([0.5, 0.5])
    grid_pts = np.array(grid_pts[:N])
    grid_pts += rng.normal(0, 0.01, grid_pts.shape)
    grid_pts = np.clip(grid_pts, 0.05, 0.95)
    configs.append(grid_pts)
    
    # Convert to joint vars [x, y, r]
    vars_list = []
    for pts in configs:
        v = np.zeros(N * 3)
        v[0::3] = pts[:, 0]
        v[1::3] = pts[:, 1]
        v[2::3] = 0.04  # Initial safe radius ensures feasibility
        vars_list.append(v)
        
    return vars_list

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_joint}
    
    best_vars = None
    best_sum_lp = 0.0
    best_centers_lp = None
    best_radii_lp = None
    
    # Generate initial configurations
    inits = get_initial_configs(rng)
    
    # Primary SLSQP optimization
    for x0 in inits:
        try:
            res = minimize(objective_joint, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if np.all(constraints_joint(res.x) >= -1e-7):
                s_lp, r_lp = get_lp_radii(res.x[:2*N].reshape(N, 2))
                if s_lp > best_sum_lp:
                    best_sum_lp = s_lp
                    best_centers_lp = res.x[:2*N].reshape(N, 2)
                    best_radii_lp = r_lp
                    best_vars = res.x.copy()
        except Exception:
            continue
            
    # Refinement: perturb best solution and re-optimize to escape local minima
    if best_vars is not None:
        for _ in range(15):
            x_pert = best_vars + rng.normal(0, 0.0005, best_vars.shape)
            x_pert = np.clip(x_pert, 0.0, 1.0)
            x_pert[2::3] = np.clip(x_pert[2::3], 1e-6, 0.5)
            
            try:
                res = minimize(objective_joint, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                if np.all(constraints_joint(res.x) >= -1e-7):
                    s_lp, r_lp = get_lp_radii(res.x[:2*N].reshape(N, 2))
                    if s_lp > best_sum_lp:
                        best_sum_lp = s_lp
                        best_centers_lp = res.x[:2*N].reshape(N, 2)
                        best_radii_lp = r_lp
                        best_vars = res.x.copy()
            except Exception:
                continue
                
    # Local search on centers using Powell + LP objective
    # This decoupled search often finds tighter packings by relieving constraint bottlenecks
    if best_centers_lp is not None:
        def obj_powell(x):
            c = x.reshape(N, 2)
            _, s = get_lp_radii(c)
            return -s
            
        res_pw = minimize(obj_powell, best_centers_lp.flatten(), method='Powell',
                          options={'maxiter': 150, 'xtol': 1e-8, 'ftol': 1e-10, 'disp': False})
        c_pw = res_pw.x.reshape(N, 2)
        s_pw, r_pw = get_lp_radii(c_pw)
        if s_pw > best_sum_lp:
            best_sum_lp = s_pw
            best_centers_lp = c_pw
            best_radii_lp = r_pw
            
    # Final safety adjustment to strictly satisfy validator tolerance
    centers = best_centers_lp
    radii = best_radii_lp
    
    for _ in range(5):
        valid = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            if x - r < -1e-12 or x + r > 1.0 + 1e-12 or y - r < -1e-12 or y + r > 1.0 + 1e-12:
                valid = False
                break
        if not valid:
            break
            
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
                
        if valid:
            break
        radii *= 0.999
        
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
