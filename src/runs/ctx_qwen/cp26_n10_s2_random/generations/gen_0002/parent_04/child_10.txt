import numpy as np
from scipy.optimize import minimize, differential_evolution

N = 26

def compute_max_radii(centers):
    """Compute maximum feasible radii for a given set of centers."""
    x = centers[:, 0]
    y = centers[:, 1]
    # Distance to boundaries
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    r_pair = np.min(dists, axis=1) / 2.0
    
    return np.minimum(r_bound, r_pair)

def objective_centers(c_flat):
    """Objective for DE: minimize negative sum of radii."""
    centers = c_flat.reshape(N, 2)
    radii = compute_max_radii(centers)
    return -np.sum(radii)

def compute_constraints(vars_flat):
    """Constraints for SLSQP: boundaries and non-overlap."""
    X = vars_flat.reshape(N, 3)
    xs = X[:, 0]
    ys = X[:, 1]
    rs = X[:, 2]
    
    c = []
    # Boundaries: x >= r, 1-x >= r, y >= r, 1-y >= r
    c.append(xs - rs)
    c.append(1.0 - xs - rs)
    c.append(ys - rs)
    c.append(1.0 - ys - rs)
    
    # Overlaps: dist^2 >= (r_i + r_j)^2
    idx = np.triu_indices(N, k=1)
    dx = xs[idx[0]] - xs[idx[1]]
    dy = ys[idx[0]] - ys[idx[1]]
    dr = rs[idx[0]] + rs[idx[1]]
    c.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(c)

def objective_combined(vars_flat):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(vars_flat[2::3])

def force_directed_init(seed):
    """Generate a tight packing using force-directed relaxation."""
    rng = np.random.default_rng(seed)
    centers = []
    r_init = 0.09
    # Hexagonal lattice generation
    for row in range(6):
        y = r_init + row * r_init * np.sqrt(3)
        offset = (row % 2) * r_init
        x = r_init + offset
        while x + r_init <= 1.0 and len(centers) < N:
            centers.append([x + rng.normal(0, 0.005), y + rng.normal(0, 0.005)])
            x += 2 * r_init
    centers = np.array(centers[:N])
    radii = np.full(N, 0.03)
    velocities = np.zeros_like(centers)
    
    stiffness = 800.0
    damping = 0.85
    step_size = 0.0008
    
    for step in range(6000):
        forces = np.zeros_like(centers)
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-12)
        np.fill_diagonal(dists, np.inf)
        sums_r = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlaps = np.maximum(0, sums_r - dists)
        dirs = diffs / dists[:, :, np.newaxis]
        forces += np.sum(overlaps[:, :, np.newaxis] * dirs * stiffness, axis=1)
        
        # Boundary repulsion
        for i in range(N):
            if centers[i, 0] < radii[i]: forces[i, 0] += stiffness * (radii[i] - centers[i, 0])
            elif centers[i, 0] > 1.0 - radii[i]: forces[i, 0] -= stiffness * (centers[i, 0] - (1.0 - radii[i]))
            if centers[i, 1] < radii[i]: forces[i, 1] += stiffness * (radii[i] - centers[i, 1])
            elif centers[i, 1] > 1.0 - radii[i]: forces[i, 1] -= stiffness * (centers[i, 1] - (1.0 - radii[i]))
            
        velocities = damping * velocities + forces * step_size
        centers += velocities
        
        max_ov = np.max(overlaps[np.triu_indices_from(overlaps, k=1)])
        if max_ov < 1e-4:
            radii *= 1.0003
            velocities *= 0.3
            
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_vars = None
    best_val = -np.inf
    
    # Strategy 1: Differential Evolution on centers
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    try:
        res_de = differential_evolution(objective_centers, bounds_centers, seed=42,

        c_opt = res_de.x.reshape(N, 2)
        r_opt = compute_max_radii(c_opt)
        val = -res_de.fun
        if val > best_val:
            best_val = val
            best_vars = np.zeros(N * 3)
            best_vars[0::3] = c_opt[:, 0]
            best_vars[1::3] = c_opt[:, 1]
            best_vars[2::3] = r_opt
    except Exception:
        pass

    # Strategy 2: Force-directed multiple seeds
    for seed in range(10):
        c_fd, r_fd = force_directed_init(seed)
        val = np.sum(r_fd)
        if val > best_val:
            best_val = val
            best_vars = np.zeros(N * 3)
            best_vars[0::3] = c_fd[:, 0]
            best_vars[1::3] = c_fd[:, 1]
            best_vars[2::3] = r_fd

    # Strategy 3: SLSQP Refinement with perturbations
    if best_vars is not None:
        bounds_full = [(0.0, 1.0)] * (2 * N) + [(1e-7, 0.5)] * N
        cons = {'type': 'ineq', 'fun': compute_constraints}
        
        rng = np.random.default_rng(99)
        starts = [best_vars.copy()]
        # Add perturbations to escape local minima

            p = best_vars.copy()
            p += rng.normal(0, 0.001, p.shape)
            starts.append(p)
            
        for x0 in starts:
            try:
                res = minimize(objective_combined, x0, method='SLSQP', bounds=bounds_full,

                c_vals = compute_constraints(res.x)
                if np.all(c_vals >= -1e-5):
                    curr_val = -res.fun
                    if curr_val > best_val:
                        best_val = curr_val
                        best_vars = res.x.copy()
            except Exception:
                continue

    # Fallback if optimization fails entirely
    if best_vars is None:
        best_vars = np.zeros(N * 3)
        best_vars[0::3] = np.linspace(0.1, 0.9, N)
        best_vars[1::3] = np.linspace(0.1, 0.9, N)
        best_vars[2::3] = 0.05

    # Strict Feasibility Repair to guarantee validator passes
    centers = best_vars.reshape(N, 3)[:, :2]
    radii = best_vars.reshape(N, 3)[:, 2].copy()
    
    for _ in range(50):
        changed = False
        # Boundary repair
        for i in range(N):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx + 1e-10:
                radii[i] = max(mx - 1e-11, 0.0)
                changed = True
                
        # Overlap repair
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-12
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
