import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars_flat):
    """Minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def constraints(vars_flat):
    """Compute all boundary and non-overlap constraints (must be >= 0)."""
    x = vars_flat.reshape(N, 3)
    xs = x[:, 0]
    ys = x[:, 1]
    rs = x[:, 2]
    
    # Boundary constraints
    c = [xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs]
    
    # Pairwise non-overlap constraints (squared distance formulation)
    i, j = np.triu_indices(N, k=1)
    dx = xs[i] - xs[j]
    dy = ys[i] - ys[j]
    dr = rs[i] + rs[j]
    c.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(c)

def get_bounds():
    """Define variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    b = []

        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
