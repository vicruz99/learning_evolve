import numpy as np
from scipy.optimize import differential_evolution, minimize, linprog

N = 26

def compute_max_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r)."""
    n = centers.shape[0]
    cx, cy = centers[:, 0], centers[:, 1]
    b_min = np.minimum(np.minimum(cx, 1.0 - cx), np.minimum(cy, 1.0 - cy))
    b_min = np.maximum(b_min, 0.0)
    
    dx = cx[:, None] - cx
    dy = cy[:, None] - cy
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    
    # Build LP: max sum(r) s.t. r_i + r_j <= dists[i,j], r_i <= b_min[i], r_i >= 0
    m = n * (n - 1) // 2 + n
    A = np.zeros((m, n))
    b = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        A[idx, i] = 1.0
        b[idx] = b_min[i]
        idx += 1
        
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        return -res.fun, res.x
    else:
        return 0.0, np.full(n, 1e-5)

def de_objective(centers_flat):
    """Objective for DE: maximize sum of radii (minimize negative sum)."""
    centers = np.clip(centers_flat.reshape(N, 2), 1e-6, 1.0 - 1e-6)
    s, _ = compute_max_radii_lp(centers)
    return -s

def constraints_func(vars):
    """SLSQP constraints: boundary and non-overlap (squared)."""
    X = vars.reshape(N, 3)
    xs, ys, rs = X[:, 0], X[:, 1], X[:, 2]
    c = []
    c.extend(xs - rs)
    c.extend(1.0 - xs - rs)
    c.extend(ys - rs)
    c.extend(1.0 - ys - rs)
    idx = np.triu_indices(N, k=1)
    dx = xs[idx[0]] - xs[idx[1]]
    dy = ys[idx[0]] - ys[idx[1]]
    dr = rs[idx[0]] + rs[idx[1]]
    c.extend(dx**2 + dy**2 - dr**2)
    return np.array(c)

def obj_full(vars):
    """Objective for local optimizer: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def run_packing() -> tuple:
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    
    # 1. Global search: DE optimizes centers, LP computes max radii
    best_res = differential_evolution(
        de_objective, bounds_centers, popsize=30, maxiter=180,
        mutation=(0.5, 1.0), recombination=0.9, seed=42,
        tol=1e-7, polish=False, updating='deferred'
    )
    
    best_centers = best_res.x.reshape(N, 2)
    best_sum, best_radii = compute_max_radii_lp(best_centers)
    
    # 2. Prepare variables for local refinement
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = best_radii
    
    cons = {'type': 'ineq', 'fun': constraints_func}
    bounds_full = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    
    final_best_sum = best_sum
    final_best_vars = x0.copy()
    
    rng = np.random.default_rng(123)
    
    # 3. Local refinement with multiple perturbations
    for _ in range(6):
        x_curr = x0 + rng.normal(0, 1e-4, x0.shape)
        try:
            res = minimize(
                obj_full, x_curr, method='SLSQP', bounds=bounds_full,
                constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13}
            )
            if -res.fun > final_best_sum:
                c_val = constraints_func(res.x)
                if np.all(c_val >= -1e-8):
                    final_best_sum = -res.fun
                    final_best_vars = res.x.copy()
        except Exception:
            continue
            
    # 4. Extract and repair final configuration
    centers_opt = final_best_vars[:2 * N].reshape(N, 2)
    radii_opt = final_best_vars[2 * N:].copy()
    
    # Deterministic repair to strictly satisfy validator tolerance
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(centers_opt[i, 0] - centers_opt[j, 0], 
                         centers_opt[i, 1] - centers_opt[j, 1])
            if d < radii_opt[i] + radii_opt[j] - 1e-10:
                shrink = (radii_opt[i] + radii_opt[j] - d) / 2.0 + 1e-10
                radii_opt[i] -= shrink
                radii_opt[j] -= shrink
        for k in range(2):
            if centers_opt[i, k] - radii_opt[i] < -1e-10:
                radii_opt[i] = centers_opt[i, k]
            if centers_opt[i, k] + radii_opt[i] > 1.0 + 1e-10:
                radii_opt[i] = 1.0 - centers_opt[i, k]
                
    radii_opt = np.maximum(radii_opt, 0.0)
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
