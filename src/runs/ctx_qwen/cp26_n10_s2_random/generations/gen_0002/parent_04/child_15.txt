1. **Exact Radius Optimization via Linear Programming**: For any fixed set of centers, finding the radii that maximize $\sum r_i$ subject to non-overlap ($r_i + r_j \le d_{ij}$) and boundary ($r_i \le \text{dist}(c_i, \partial)$) constraints is a Linear Program. We use `scipy.optimize.linprog` (method='highs') to solve this exactly and efficiently. This guarantees feasibility and provides a precise objective value for any center configuration.

2. **Derivative-Free Center Optimization**: The objective function over centers, defined by the LP result, is non-smooth due to the $\min$ interactions in packing constraints. We optimize centers using `Powell` and `Nelder-Mead` methods, which are robust for non-smooth, multi-modal landscapes.

3. **Diverse Initialization & Perturbation**: We start from multiple structured patterns (hexagonal, grid, corner-focused, random) and apply iterative random perturbations to the best found configuration. This helps escape local minima that plague standard gradient-based packings.

4. **Guaranteed Validity**: Because radii are always computed via an LP that enforces all geometric constraints, the final output is mathematically guaranteed to pass the validation function without needing post-hoc shrinkage or repair steps.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_max_radii_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii.
    Constraints: r_i + r_j <= dist(i,j) and r_i <= dist to boundary.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise non-overlap constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pairs = np.argwhere(mask)
    
    n_pair = len(pairs)
    n_bound = n * 4
    A_ub = np.zeros((n_pair + n_bound, n))
    b_ub = np.zeros(n_pair + n_bound)
    
    # r_i + r_j <= d_ij
    A_ub[:n_pair, pairs[:, 0]] = 1.0
    A_ub[:n_pair, pairs[:, 1]] = 1.0
    b_ub[:n_pair] = dists[mask]
    
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        idx_start = n_pair + i * 4
        bounds_vals = [centers[i, 0], 1.0 - centers[i, 0], 
                       centers[i, 1], 1.0 - centers[i, 1]]
        for k in range(4):
            A_ub[idx_start + k, i] = 1.0
            b_ub[idx_start + k] = bounds_vals[k]
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0.0, None), method='highs')
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def objective(centers_flat):
    """Objective: maximize LP-optimized sum of radii."""
    centers = centers_flat.reshape(N, 2)
    # Keep centers strictly inside to avoid degenerate LP boundaries
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    val, _ = get_max_radii_lp(centers)
    return -val

def generate_initial_configs():
    """Generate diverse starting configurations for center optimization."""
    configs = []
    
    # 1. Hexagonal lattice pattern
    pts = []
    r0 = 0.095
    y = r0
    row = 0
    while len(pts) < N:
        x = r0 + (row % 2) * r0
        while x <= 1.0 - r0 and len(pts) < N:
            pts.append([x, y])
            x += 2 * r0
        y += r0 * np.sqrt(3)
        row += 1
    configs.append(np.array(pts[:N]))
    
    # 2. Perturbed square grid
    pts2 = []
    for i in range(5):
        for j in range(5):
            pts2.append([0.12 + i*0.19, 0.12 + j*0.19])
    pts2.append([0.5, 0.06])
    configs.append(np.array(pts2[:N]))
    
    # 3. Dense random
    np.random.seed(42)
    configs.append(np.random.uniform(0.15, 0.85, (N, 2)))
    
    # 4. Corner-focused layout
    pts4 = np.array([[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]])
    pts4 = np.vstack([pts4, np.random.uniform(0.2, 0.8, (N-4, 2))])
    configs.append(pts4)
    
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(123)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = generate_initial_configs()
    
    # Phase 1: Optimize from diverse starts
    for c0 in inits:
        try:
            res = minimize(objective, c0.flatten(), method='Powell', 
                           options={'maxiter': 2000, 'ftol': 1e-12, 'xtol': 1e-9, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
                _, best_radii = get_max_radii_lp(best_centers)
        except Exception:
            pass
            
        try:
            res2 = minimize(objective, c0.flatten(), method='Nelder-Mead', 
                            options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-12, 'disp': False})
            if -res2.fun > best_sum:
                best_sum = -res2.fun
                best_centers = res2.x.reshape(N, 2)
                _, best_radii = get_max_radii_lp(best_centers)
        except Exception:
            pass

    # Phase 2: Perturb and refine best solution to escape local minima
    if best_centers is not None:
        for _ in range(15):
            pert = best_centers + np.random.normal(0, 0.005, best_centers.shape)
            pert = np.clip(pert, 0.05, 0.95)
            try:
                res3 = minimize(objective, pert.flatten(), method='Powell',
                                options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
                if -res3.fun > best_sum:
                    best_sum = -res3.fun
                    best_centers = res3.x.reshape(N, 2)
                    _, best_radii = get_max_radii_lp(best_centers)
            except Exception:
                pass
                
    # Fallback (should not trigger with robust LP)
    if best_centers is None:
        best_centers = inits[0]
        _, best_radii = get_max_radii_lp(best_centers)
        best_sum = float(np.sum(best_radii))
        
    return best_centers, best_radii, float(best_sum)
