The key insight is that for any fixed set of centers, the optimal radii can be computed exactly and efficiently by solving a Linear Program (LP). This reduces the optimization space from 78 variables (centers + radii) to 52 variables (centers only), where the objective function `sum_radii(centers)` is well-defined and amenable to derivative-free global optimization.

1. **LP-based Radius Computation**: For fixed centers, solve `max ∑r_i` subject to `r_i + r_j ≤ dist(i,j)` and boundary constraints. This gives the exact maximum sum of radii for any configuration.
2. **Derivative-Free Center Optimization**: Use `scipy.optimize.minimize` with the `Powell` method on the centers. Powell's method handles non-smooth objectives effectively and escapes shallow local minima better than SLSQP when radii are implicitly optimized.
3. **Multi-Start & Perturbation**: Run Powell from diverse initial configurations (hexagonal, grid, corner-focused, random) and apply small random perturbations to the best found solution to explore the landscape further.
4. **Final SLSQP Polish**: Use the best centers and LP-computed radii as a high-quality starting point for SLSQP on the full 78 variables. SLSQP excels at precise local polishing when initialized close to the optimum.
5. **Numerical Safety**: Apply strict boundary clamping and a final validation pass to guarantee the output satisfies the checker's tolerances.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def compute_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Preallocate constraint matrix and vector
    n_pair = n * (n - 1) // 2
    n_bound = 4 * n
    A_ub = np.zeros((n_pair + n_bound, n))
    b_ub = np.zeros(n_pair + n_bound)
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        cx, cy = centers[i]
        limits = [cx, 1.0 - cx, cy, 1.0 - cy]
        for lim in limits:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0.0, None), method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-6)

def objective_centers(centers_flat):
    """Objective for center optimization: maximize sum of radii."""
    centers = np.clip(centers_flat.reshape(-1, 2), 1e-4, 1.0 - 1e-4)
    radii = compute_lp_radii(centers)
    return -np.sum(radii)

def generate_initial_centers(method, rng):
    """Generates diverse initial center configurations."""
    if method == 'hex':
        pts = []
        r_est = 0.105
        dy = r_est * np.sqrt(3)
        dx = 2 * r_est
        row = 0
        while len(pts) < N:
            y = r_est + row * dy
            if y + r_est > 0.99: break
            offset = (row % 2) * (dx / 2.0)
            x = r_est + offset
            while x + r_est <= 1.0 - 1e-9 and len(pts) < N:
                pts.append([x, y])
                x += dx
            row += 1
        while len(pts) < N:
            pts.append(rng.uniform(0.2, 0.8, 2))
        return np.array(pts[:N])
    elif method == 'grid':
        pts = []
        for i in range(6):
            for j in range(5):
                pts.append([0.12 + i*0.16, 0.12 + j*0.20])
        return np.array(pts[:N]) + rng.normal(0, 0.01, (N, 2))
    elif method == 'rand':
        return rng.uniform(0.1, 0.9, (N, 2))
    elif method == 'corner':
        pts = np.array([[0.15,0.15], [0.85,0.15], [0.15,0.85], [0.85,0.85]])
        rem = N - 4
        pts = np.vstack([pts, rng.uniform(0.3, 0.7, (rem, 2))])
        return pts
    return rng.uniform(0.2, 0.8, (N, 2))

def slsqp_constraints(vars):
    """Vectorized constraints for SLSQP polishing."""
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    con = []
    con.extend(xs - rs)
    con.extend(1.0 - xs - rs)
    con.extend(ys - rs)
    con.extend(1.0 - ys - rs)
    
    dx = xs[:, np.newaxis] - xs[np.newaxis, :]
    dy = ys[:, np.newaxis] - ys[np.newaxis, :]
    dr = rs[:, np.newaxis] + rs[np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    con.extend(dists[mask] - dr[mask])
    return np.array(con)

def slsqp_obj(vars):
    return -np.sum(vars[2::3])

def get_bounds():
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_val = np.inf
    best_centers_flat = None
    
    configs_methods = ['hex', 'grid', 'corner', 'rand']
    
    # Phase 1: Multi-start derivative-free optimization on centers
    for method in configs_methods:
        for _ in range(4):
            c0 = generate_initial_centers(method, rng)
            c0 = np.clip(c0, 0.01, 0.99)
            try:
                res = minimize(objective_centers, c0.flatten(), method='Powell',
                               options={'maxiter': 3000, 'ftol': 1e-12, 'xtol': 1e-12})
                if res.fun < best_val:
                    best_val = res.fun
                    best_centers_flat = res.x.copy()
            except Exception:
                pass

    # Phase 2: Perturbation refinement to escape local minima
    if best_centers_flat is not None:
        c_best = best_centers_flat.copy()
        for _ in range(12):
            pert = c_best + rng.normal(0, 0.005, c_best.shape)
            pert = np.clip(pert, 0.01, 0.99)
            try:
                res = minimize(objective_centers, pert, method='Powell',
                               options={'maxiter': 1500, 'ftol': 1e-12})
                if res.fun < best_val:
                    best_val = res.fun
                    best_centers_flat = res.x.copy()
            except Exception:
                pass

    # Phase 3: SLSQP polish on full (centers + radii) space
    centers_opt = np.clip(best_centers_flat.reshape(-1, 2), 1e-6, 1.0-1e-6)
    radii_opt = compute_lp_radii(centers_opt)
    
    vars0 = np.zeros(N * 3)
    for i in range(N):
        vars0[3*i] = centers_opt[i, 0]
        vars0[3*i+1] = centers_opt[i, 1]
        vars0[3*i+2] = radii_opt[i]
        
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': slsqp_constraints}
    
    try:
        res_final = minimize(slsqp_obj, vars0, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12})
        if -res_final.fun > -best_val:
            best_centers_flat = res_final.x[:2*N]
            radii_opt = res_final.x[2*N:]
        else:
            radii_opt = compute_lp_radii(np.clip(best_centers_flat.reshape(-1, 2), 1e-6, 1.0-1e-6))
    except Exception:
        pass
        
    centers = best_centers_flat.reshape(-1, 2)
    centers = np.clip(centers, 1e-9, 1.0-1e-9)
    radii = np.maximum(radii_opt, 0.0)
    
    # Final safety validation pass
    for _ in range(5):
        valid = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            if x - r < -1e-9 or x + r > 1.0 + 1e-9 or y - r < -1e-9 or y + r > 1.0 + 1e-9:
                valid = False
                break
        if not valid: break
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    valid = False
                    break
            if not valid: break
        if valid: break
        radii *= 0.9999
        
    return centers, radii, float(np.sum(radii))
