import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_constraints(vars_flat):
    """Computes boundary and non-overlap constraints. Must be >= 0."""
    X = vars_flat.reshape(N, 3)
    xs = X[:, 0]
    ys = X[:, 1]
    rs = X[:, 2]

    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = []
    c.append(xs - rs)
    c.append(1.0 - xs - rs)
    c.append(ys - rs)
    c.append(1.0 - ys - rs)

    # Overlap constraints: dist(i,j) >= r_i + r_j
    # Vectorized distance calculation
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dists = np.sqrt(dx**2 + dy**2 + 1e-20)  # Tiny epsilon for numerical stability
    
    # Extract upper triangle (i < j)
    triu_idx = np.triu_indices(N, k=1)
    dists_pair = dists[triu_idx]
    r_sums = rs[triu_idx[0]] + rs[triu_idx[1]]
    
    c.append(dists_pair - r_sums)
    
    return np.concatenate(c)

def compute_objective(vars_flat):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]"""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return b

def generate_initial_configs():
    """Generates diverse feasible initial center configurations."""
    configs = []
    
    # 1. Hexagonal lattice (5,5,5,5,6 pattern)
    r0 = 0.102
    pts = []
    y = r0
    row = 0
    while len(pts) < N:
        offset = r0 if row % 2 == 1 else 0.0
        x = r0 + offset
        while x + r0 <= 1.0 and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
        row += 1
    configs.append(np.array(pts[:N]))
    
    # 2. Perturbed hexagonal
    configs.append(configs[0] + np.random.normal(0, 0.008, (N, 2)))
    
    # 3. Dense random
    configs.append(np.random.uniform(0.15, 0.85, (N, 2)))
    
    # 4. Structured grid 5x5 + 1 center
    g = []
    for i in range(5):
        for j in range(5):
            g.append([0.12 + i * 0.2, 0.12 + j * 0.2])
    g.append([0.5, 0.95])
    configs.append(np.array(g[:N]))
    
    # Normalize and clip all configs to stay safely inside [0,1]
    normalized = []
    for c in configs:
        c = c - c.min(axis=0)
        span = c.max(axis=0) - c.min(axis=0)
        span[span < 1e-9] = 1.0
        c = (c / span) * 0.9 + 0.05
        normalized.append(np.clip(c, 0.01, 0.99))
        
    return normalized

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_vars = None
    best_obj = -np.inf
    
    # Generate base configurations and add random perturbations
    base_configs = generate_initial_configs()
    inits = base_configs.copy()
    
    for _ in range(15):
        base = base_configs[np.random.randint(len(base_configs))]
        pert = base + np.random.normal(0, 0.006, base.shape)
        inits.append(np.clip(pert, 0.02, 0.98))
        
    # Primary optimization phase
    for centers_init in inits:
        vars_flat = np.zeros(N * 3)
        vars_flat[0::3] = centers_init[:, 0]
        vars_flat[1::3] = centers_init[:, 1]
        vars_flat[2::3] = 0.075  # Safe initial radius
        
        try:
            res = minimize(compute_objective, vars_flat, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13})
            
            # Check feasibility with tolerance
            c_vals = compute_constraints(res.x)
            if np.all(c_vals >= -1e-8):
                curr_obj = -res.fun
                if curr_obj > best_obj:
                    best_obj = curr_obj
                    best_vars = res.x.copy()
        except Exception:
            continue

    # Fallback if optimization failed
    if best_vars is None:
        best_vars = np.zeros(N * 3)
        best_vars[0::3] = np.random.uniform(0.15, 0.85, N)
        best_vars[1::3] = np.random.uniform(0.15, 0.85, N)
        best_vars[2::3] = 0.05
        
    # Refinement phase: perturb best solution and re-optimize
    for _ in range(10):
        x_curr = best_vars.copy()
        x_curr += np.random.normal(0, 0.0004, x_curr.shape)
        x_curr = np.clip(x_curr, 0.0, 1.0)
        x_curr[2::3] = np.clip(x_curr[2::3], 1e-8, 0.5)
        
        try:
            res = minimize(compute_objective, x_curr, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13})
            
            c_vals = compute_constraints(res.x)
            if np.all(c_vals >= -1e-8):
                curr_obj = -res.fun
                if curr_obj > best_obj:
                    best_obj = curr_obj
                    best_vars = res.x.copy()
        except Exception:
            continue

    # Extract optimized centers
    centers_opt = best_vars.reshape(N, 3)[:, :2]
    
    # LP Refinement: Given fixed centers, exactly maximize sum of radii
    # This guarantees feasibility and extracts maximum possible radii
    c_obj_lp = -np.ones(N)
    A_ub = []
    b_ub = []
    
    # Pairwise distance constraints: r_i + r_j <= dist_ij
    diff = centers_opt[:, None, :] - centers_opt[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(N):
        for j in range(i + 1, N):
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(N):
        bounds_vals = [centers_opt[i, 0], 1.0 - centers_opt[i, 0], 
                       centers_opt[i, 1], 1.0 - centers_opt[i, 1]]
        for b in bounds_vals:
            row = np.zeros(N)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    try:
        res_lp = linprog(c_obj_lp, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                         bounds=(0.0, None), method='highs')
        if res_lp.success:
            radii_opt = res_lp.x
        else:
            radii_opt = best_vars[2::3]
    except Exception:
        radii_opt = best_vars[2::3]
        
    # Apply negligible shrink to strictly satisfy validator tolerance (1e-12)
    radii_opt *= 0.999999
    centers_final = centers_opt.copy()
    
    sum_r = float(np.sum(radii_opt))
    return centers_final, radii_opt, sum_r
