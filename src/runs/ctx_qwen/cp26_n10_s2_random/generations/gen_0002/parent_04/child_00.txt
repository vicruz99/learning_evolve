

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we decouple the problem into two stages:
1. **Inner Optimization (LP)**: For any fixed set of centers, finding the optimal radii that maximize $\sum r_i$ subject to non-overlap ($r_i + r_j \le \|C_i - C_j\|$) and boundary constraints ($r_i \le \text{dist\_to\_boundary}$) is a Linear Programming problem. This avoids the non-smooth, non-convex inequality constraints in the outer optimizer.
2. **Outer Optimization (Global + Local Search)**: We optimize the 52 center coordinates to maximize the LP objective value. We use `differential_evolution` for robust global exploration, followed by `Powell` local refinement. We seed the search with diverse initial configurations (hexagonal lattices of varying densities and force-relaxed random packs) to ensure we escape poor local minima.

This separation makes the outer objective continuous and much easier for derivative-free optimizers to handle. Precomputing the LP constraint matrix avoids repeated allocation overhead, ensuring the solution runs efficiently within time limits.
</strategy>

```python
import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
M_PAIR = len(IDX_I)
A_PAIR = np.zeros((M_PAIR, N))
A_PAIR[np.arange(M_PAIR), IDX_I] = 1.0
A_PAIR[np.arange(M_PAIR), IDX_J] = 1.0
A_UB_FULL = np.vstack([A_PAIR, np.eye(N)])

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii given fixed centers."""
    lim = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                     np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    dx = centers[IDX_I, 0] - centers[IDX_J, 0]
    dy = centers[IDX_I, 1] - centers[IDX_J, 1]
    b_pair = np.sqrt(dx**2 + dy**2)
    
    b_ub = np.empty(M_PAIR + N)
    b_ub[:M_PAIR] = b_pair
    b_ub[M_PAIR:] = lim
    
    res = linprog(-np.ones(N), A_ub=A_UB_FULL, b_ub=b_ub, bounds=(0.0, None), method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def objective(centers_flat):
    """Outer objective: negative max sum of radii for given centers."""
    centers = np.clip(centers_flat.reshape(N, 2), 1e-7, 1.0 - 1e-7)
    s, _ = solve_lp_radii(centers)
    return -s

def relax_config(centers, iterations=300):
    """Force-directed relaxation to generate feasible starting configurations."""
    n = centers.shape[0]
    r = 0.03 * np.ones(n)
    for _ in range(iterations):
        diffs = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-16)
        min_d = r[:, None] + r[None, :]
        overlap = np.maximum(0, min_d - dists)
        force_mag = overlap * 100.0
        dir_ = diffs / dists[:, :, None]
        forces = np.sum(force_mag[:, :, None] * dir_, axis=1)
        
        for i in range(n):
            c = centers[i]
            ri = r[i]
            if c[0] < ri: forces[i,0] += 200.0*(ri-c[0])
            elif c[0] > 1-ri: forces[i,0] -= 200.0*(c[0]-(1-ri))
            if c[1] < ri: forces[i,1] += 200.0*(ri-c[1])
            elif c[1] > 1-ri: forces[i,1] -= 200.0*(c[1]-(1-ri))
            
        centers += forces * 0.001
        centers = np.clip(centers, 0.0, 1.0)
        r *= 1.0005
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    inits = []
    
    # 1. Hexagonal patterns with different densities
    for scale in [0.08, 0.09, 0.10, 0.11]:
        pos = []
        y = scale
        row = 0
        while len(pos) < N:
            x = scale + (row % 2) * scale
            while x + scale <= 1.0 - 1e-9 and len(pos) < N:
                pos.append([x, y])
                x += 2 * scale
            y += scale * np.sqrt(3)
            row += 1
        pos = np.array(pos[:N]) + rng.normal(0, 0.005, (N, 2))
        inits.append(np.clip(pos, 0.02, 0.98))
        
    # 2. Relaxed random configurations
    for _ in range(4):
        c = rng.uniform(0.15, 0.85, (N, 2))
        inits.append(relax_config(c, 200))
        
    best_val = -np.inf
    best_c = None
    
    # 3. Global Search
    bounds = [(0.05, 0.95)] * (2 * N)
    try:
        de_res = differential_evolution(objective, bounds, seed=42, popsize=8, 
                                        maxiter=100, mutation=(0.5, 1.0), recombination=0.8,
                                        tol=1e-6, polish=False)
        if -de_res.fun > best_val:
            best_val = -de_res.fun
            best_c = de_res.x.reshape(N, 2)
    except Exception:
        pass
        
    # 4. Local Refinement
    candidates = [best_c.flatten()] if best_c is not None else []
    candidates.extend([c.flatten() for c in inits])
    
    for c0 in candidates:
        try:
            res = minimize(objective, c0, method='Powell', 
                           options={'maxiter': 500, 'ftol': 1e-12, 'xtol': 1e-10})
            if -res.fun > best_val:
                best_val = -res.fun
                best_c = res.x.reshape(N, 2)
        except Exception:
            pass
            
    if best_c is None:
        best_c = inits[0]
        
    # 5. Final radii computation and safety adjustments
    _, radii = solve_lp_radii(best_c)
    best_c = np.clip(best_c, radii, 1.0 - radii)
    radii *= 0.99999  # Tiny shrink to guarantee strict validator compliance
    
    return best_c, radii, float(np.sum(radii))
```
