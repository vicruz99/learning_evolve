Key improvements in this approach:
1. **Hybrid Optimization (LP + NLP)**: For any fixed set of centers, the problem of maximizing the sum of radii is a *Linear Programming* problem. Solving this LP exactly gives the true maximum radii for those centers, which is far more efficient and accurate than optimizing radii as continuous variables in a non-linear solver.
2. **Iterative Refinement Loop**: We alternate between optimizing centers (using SLSQP) and solving the LP for radii. import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRI_IDX = np.triu_indices(N, k=1)

def compute_constraints(vars_flat):
    """Computes boundary and non-overlap constraints. Returns array >= 0 for feasibility."""
    X = vars_flat.reshape(N, 3)
    xs, ys, rs = X[:, 0], X[:, 1], X[:, 2]
    
    c = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(xs - rs)
    c.append(1.0 - xs - rs)
    c.append(ys - rs)
    c.append(1.0 - ys - rs)
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = xs[TRI_IDX[0]] - xs[TRI_IDX[1]]
    dy = ys[TRI_IDX[0]] - ys[TRI_IDX[1]]
    dr = rs[TRI_IDX[0]] + rs[TRI_IDX[1]]
    c.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(c)

def compute_objective(vars_flat):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars_flat[2::3])

def get_bounds():
    """Variable bounds for x, y in [0,1] and r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to find exact maximum feasible radii."""
    c_obj = -np.ones(N)
    
    i_idx, j_idx = TRI_IDX
    A_pair = np.zeros((len(i_idx), N))
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    b_pair = dists[i_idx, j_idx]
    A_pair[np.arange(len(i_idx)), i_idx] = 1.0
    A_pair[np.arange(len(i_idx)), j_idx] = 1.0
    
    A_bnd = np.zeros((4*N, N))
    b_bnd = np.zeros(4*N)
    for i in range(N):
        d = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for k in range(4):
            A_bnd[4*i+k, i] = 1.0
            b_bnd[4*i+k] = d
            
    A_ub = np.vstack([A_pair, A_bnd])
    b_ub = np.concatenate([b_pair, b_bnd])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return res.x if res.success else np.full(N, 0.01)

def make_init(row_counts, shift, seed):
    """Generates a feasible initial configuration from a hexagonal-like grid."""
    rng = np.random.default_rng(seed)
    pts = []
    y = 0.1
    dy = 0.165
    dx = 0.185
    
    for r_idx, count in enumerate(row_counts):
        shift_x = shift[r_idx] * dx / 2.0
        x = 0.1 + shift_x
        for _ in range(count):
            pts.append([x + rng.uniform(-0.005, 0.005), y + rng.uniform(-0.005, 0.005)])
            x += dx
        y += dy
        
    pts = np.array(pts[:N])
    
    # Compute strictly feasible initial radii
    rs = np.full(N, 0.5)
    dists = np.linalg.norm(pts[:, None, :] - pts[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    for i in range(N):
        d_b = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
        rs[i] = min(d_b, min_dists[i] * 0.5) * 0.95
        
    x0 = np.zeros(N*3)
    for i in range(N):
        x0[3*i] = pts[i,0]
        x0[3*i+1] = pts[i,1]
        x0[3*i+2] = rs[i]
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_val = -np.inf
    best_x = None
    
    # Diverse hexagonal row configurations
    configs = [
        [5, 5, 5, 5, 5, 1], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4],
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 5, 6, 6],
        [6, 6, 5, 5, 4], [4, 5, 5, 6, 6], [5, 5, 5, 6, 5], [6, 5, 5, 5, 5]
    ]
    
    seeds = np.arange(20)
    
    # Phase 1: Multi-start optimization
    for seed in seeds:
        for cfg in configs:
            for shift in [[i%2 for i in range(len(cfg))], [(i+1)%2 for i in range(len(cfg))]]:
                x0 = make_init(cfg, shift, seed)
                try:
                    res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': 1500, 'ftol': 1e-12})
                    if np.all(compute_constraints(res.x) >= -1e-6):
                        val = -res.fun
                        if val > best_val:
                            best_val = val
                            best_x = res.x.copy()
                except Exception:
                    pass
                    
    if best_x is None:
        best_x = make_init(configs[0], [0]*len(configs[0]), 42)
        
    # Phase 2: Iterative refinement (LP + Grow & Relax)
    for step in range(15):
        centers = best_x.reshape(N, 3)[:, :2]
        radii = solve_lp_radii(centers)
        
        # If LP finds a better sum, re-optimize centers around this new radius distribution
        lp_sum = np.sum(radii)
        if lp_sum > best_val + 1e-7:
            best_val = lp_sum
            x0_lp = np.zeros(N*3)
            for i in range(N):
                x0_lp[3*i] = centers[i,0]
                x0_lp[3*i+1] = centers[i,1]
                x0_lp[3*i+2] = radii[i]
                
            try:
                res = minimize(compute_objective, x0_lp, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 1000, 'ftol': 1e-12})
                if np.all(compute_constraints(res.x) >= -1e-6):
                    val = -res.fun
                    if val > best_val:
                        best_val = val
                        best_x = res.x.copy()
            except Exception:
                pass
                
        # Slightly grow radii and perturb centers to escape local minima
        new_x = best_x.copy()
        new_x[2::3] *= 1.001
        new_x[:2*N] += np.random.normal(0, 0.0003, 2*N)
        
        try:
            res = minimize(compute_objective, new_x, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 800, 'ftol': 1e-12})
            if np.all(compute_constraints(res.x) >= -1e-6):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Final exact radius computation for the best centers found
    centers = best_x.reshape(N, 3)[:, :2]
    radii = solve_lp_radii(centers)
    sum_r = np.sum(radii)
    
    return centers, radii, float(sum_r)
