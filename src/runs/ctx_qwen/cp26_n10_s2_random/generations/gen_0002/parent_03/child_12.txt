import numpy as np
from scipy.optimize import minimize

N = 26
TRI_U = np.triu_indices(N, k=1)

def compute_constraints(v):
    """
    Computes all boundary and non-overlap constraints.
    Returns an array where each element must be >= 0.
    """
    C = v[:2*N].reshape(N, 2)
    R = v[2*N:]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = [C[:, 0] - R, 1.0 - C[:, 0] - R, C[:, 1] - R, 1.0 - C[:, 1] - R]
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    idx_i, idx_j = TRI_U
    dx = C[idx_i, 0] - C[idx_j, 0]
    dy = C[idx_i, 1] - C[idx_j, 1]
    dr = R[idx_i] + R[idx_j]
    c.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(c)

def compute_objective(v):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2*N:])

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def make_hex_init(seed, density=1.0, noise=0.005):
    """Generates a feasible initial configuration based on a hexagonal-like grid."""
    rng = np.random.default_rng(seed)
    pts = []
    
    # Hex grid parameters tuned for unit square
    dx = 0.19 * density
    dy = dx * np.sqrt(3) / 2.0
    
    y = 0.1
    row = 0
    while len(pts) < N and y + 0.1 < 1.0:
        offset = (dx / 2.0) if (row % 2 == 1) else 0.0
        x = 0.1 + offset
        while x + 0.1 < 1.0 and len(pts) < N:
            pts.append([x, y])
            x += dx
        y += dy
        row += 1
        
    pts = np.array(pts[:N])
    pts += rng.normal(0, noise, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    # Initial radii: safe starting point that strictly satisfies constraints
    rs = np.full(N, 0.04)
    return np.concatenate([pts.flatten(), rs])

    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}

    best_x = None

    densities = [0.85, 0.9, 0.95, 1.0, 1.05, 1.1]
    seeds = list(range(40))

                    val = -res.fun
                    if val > best_val:
                        best_val = val
                        best_x = res.x.copy()
            except Exception:
                pass
                
    if best_x is None:
        best_x = make_hex_init(0)
        
    # Phase 2: Radius Growth & Refinement (Homotopy continuation)
    # Gradually increase radii and re-optimize positions to escape local minima
    # This pushes the solution to the true boundary of the feasible region
    curr_x = best_x.copy()
    for step in range(30):
        # Grow radii slightly
        rs = curr_x[2*N:]
        growth = 1.0012
        curr_x[2*N:] = np.clip(rs * growth, 1e-7, 0.5)
        
        # Perturb centers slightly to help resolver overlaps and escape plateaus
        curr_x[:2*N] += np.random.normal(0, 0.0003, 2*N)

    return centers, radii, sum_r
