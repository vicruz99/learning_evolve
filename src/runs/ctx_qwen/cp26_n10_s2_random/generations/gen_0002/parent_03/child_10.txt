To improve the circle packing and reach a higher sum of radii (~2.636+), I will employ a two-phase optimization strategy:

1. **Global Multi-Start Optimization**: Initialize circles in a dense hexagonal-like lattice. Run `SLSQP` with multiple random perturbations to escape poor local minima early. The constraints are formulated using squared distances to maintain smoothness and avoid `sqrt` operations in the gradient calculations.

2. **Block Coordinate Descent (BCD)**: After the global phase, sequentially optimize each circle's position and radius while keeping all others fixed. This is highly effective for packing problems because it monotonically increases each radius until it hits the "tightest" feasible configuration relative to its neighbors and boundaries. By cycling through all circles multiple times, the packing naturally converges to a highly compressed, optimal local arrangement where every circle is pushed to its maximum feasible size.

Key implementation details:
- All helper functions are top-level with no closures or lambdas to comply with constraints.
- Vectorized constraint evaluation speeds up the global phase.
- Strict feasibility checks ensure the optimizer doesn't drift into invalid regions.
- The squared-distance constraint `dist^2 >= (r_i + r_j)^2` is numerically stable and differentiable.
- BCD phase uses small sub-problem optimizations (3 variables, ~100 constraints each) which solve extremely fast but yield significant packing improvements.

import numpy as np
from scipy.optimize import minimize

N = 26
# Precompute upper triangle indices for pairwise constraints
TRI_U_IDX = np.triu_indices(N, k=1)

def compute_initial_radii(centers):
    """Compute strictly feasible initial radii for a given set of centers."""
    n = centers.shape[0]
    rs = np.full(n, 0.5)
    for i in range(n):
        xb, yb = centers[i]
        d_bound = min(xb, 1.0 - xb, yb, 1.0 - yb)
        diffs = centers - centers[i]
        dists = np.sqrt(np.sum(diffs**2, axis=1))
        dists[i] = np.inf
        d_neigh = np.min(dists)
        # Scale down slightly to guarantee strict feasibility initially
        rs[i] = min(d_bound, 0.5 * d_neigh) * 0.95
    return rs

def obj_all(v):
    """Global objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def cons_all(v):
    """Global constraints: boundaries and pairwise non-overlaps."""
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.extend(xs - rs)
    c.extend(1.0 - xs - rs)
    c.extend(ys - rs)
    c.extend(1.0 - ys - rs)
    c.extend(rs)  # Non-negative radii
    
    # Pairwise constraints: dist^2 >= (ri + rj)^2
    dx = xs[TRI_U_IDX[0]] - xs[TRI_U_IDX[1]]
    dy = ys[TRI_U_IDX[0]] - ys[TRI_U_IDX[1]]
    dr = rs[TRI_U_IDX[0]] + rs[TRI_U_IDX[1]]
    c.extend(dx**2 + dy**2 - dr**2)
    return np.array(c)

def obj_single(p):
    """Local objective for BCD: maximize current circle's radius."""
    return -p[2]

def cons_single(p, others_c, others_r):
    """Local constraints for BCD."""
    xi, yi, ri = p
    c = [xi - ri, 1.0 - xi - ri, yi - ri, 1.0 - yi - ri, ri]
    dx = xi - others_c[:, 0]
    dy = yi - others_c[:, 1]
    dr = ri + others_r
    c.extend(dx**2 + dy**2 - dr**2)
    return np.array(c)

def optimize_single_circle(i, centers, radii):
    """Optimize (x, y, r) for circle i while fixing all others."""
    others_idx = np.delete(np.arange(N), i)
    others_c = centers[others_idx]
    others_r = radii[others_idx]
    
    x0 = [centers[i, 0], centers[i, 1], radii[i]]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)]
    cons = {'type': 'ineq', 'fun': cons_single, 'args': (others_c, others_r)}
    
    res = minimize(obj_single, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 500, 'ftol': 1e-12})
    return np.array(res.x)

def run_packing():
    np.random.seed(42)
    
    # 1. Initialization: Dense hexagonal-like grid
    centers = np.zeros((N, 2))
    idx = 0
    y = 0.08
    dy = 0.165
    dx = 0.18
    # Row pattern summing to 26: 5 + 6 + 5 + 6 + 4
    row_counts = [5, 6, 5, 6, 4]
    for r_idx, count in enumerate(row_counts):
        shift = (dx / 2.0) if (r_idx % 2 == 1) else 0.0
        x = 0.08 + shift
        for _ in range(count):
            if idx < N:
                centers[idx] = [x, y]
                idx += 1
            x += dx
        y += dy
        
    radii = compute_initial_radii(centers)
    
    # Flatten to [x1, y1, r1, x2, y2, r2, ...]
    vars_all = np.zeros(N * 3)
    for i in range(N):
        vars_all[3*i] = centers[i, 0]
        vars_all[3*i+1] = centers[i, 1]
        vars_all[3*i+2] = radii[i]
        
    bounds_all = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_all}
    
    best_vars = vars_all.copy()
    best_sum = np.sum(radii)
    
    # 2. Global Multi-Start Optimization
    for seed in range(10):
        np.random.seed(seed)
        v0 = vars_all.copy()
        v0[:2*N] += np.random.normal(0, 0.008, 2*N)
        
        try:
            res = minimize(obj_all, v0, method='SLSQP', bounds=bounds_all,
                           constraints=cons_dict, options={'maxiter': 5000, 'ftol': 1e-12})
            if np.all(cons_all(res.x) >= -1e-6):
                s = np.sum(res.x[2::3])
                if s > best_sum:
                    best_sum = s
                    best_vars = res.x.copy()
        except Exception:
            pass
            
    centers = best_vars[:2*N].reshape(N, 2)
    radii = best_vars[2*N:]
    
    # 3. Block Coordinate Descent (BCD)
    # Sequentially maximize each circle's radius given fixed neighbors.
    # This phase tightly packs the configuration and typically yields significant gains.
    for cycle in range(25):
        improved = False
        for i in range(N):
            old_r = radii[i]
            opt_p = optimize_single_circle(i, centers, radii)
            centers[i] = opt_p[:2]
            radii[i] = opt_p[2]
            if radii[i] > old_r + 1e-9:
                improved = True
        if not improved:
            break
            
    # Final safety clamping
    radii = np.maximum(radii, 0.0)
    centers = np.clip(centers, 0.0, 1.0)
    
    return centers, radii, float(np.sum(radii))
