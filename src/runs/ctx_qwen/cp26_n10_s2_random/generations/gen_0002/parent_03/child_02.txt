To maximize the sum of radii for 26 circles in a unit square, we face a non-convex, highly constrained optimization problem. Past approaches using direct SLSQP or force-directed methods often get trapped in local minima or struggle with the coupling between center positions and radii. 

My strategy decouples the problem into three robust phases:
1. **Fast Simulated Annealing (SA) on Centers**: We explore the 52-dimensional center space using SA. Instead of solving a full LP or evaluating complex constraints at each step, we use a fast, vectorized heuristic `fast_sum_radii` that computes a valid (though not necessarily optimal) radius assignment based on half the minimum pairwise distance and boundary limits. This provides a smooth-enough landscape for SA to efficiently escape local minima and find high-quality center configurations.
2. **Exact LP Refinement**: Once SA identifies the best center layout, we solve a Linear Program to compute the mathematically optimal radii for those fixed centers. This exploits the fact that maximizing $\sum r_i$ subject to $r_i + r_j \le d_{ij}$ and boundary limits is a pure LP, which can be solved to high precision very quickly.
3. **Joint SLSQP Polish**: We take the LP-optimal (centers, radii) pair and run a local gradient-based optimizer (SLSQP) to jointly adjust both. This tightens the packing, resolves any minor numerical drift, and squeezes out the last bit of objective value by leveraging the exact constraint gradients.
4. **Safety Correction**: A final deterministic pass shrinks radii minimally to guarantee strict feasibility against the validation function's numerical tolerances.

This hybrid approach combines global exploration, exact linear optimization, and local non-linear polishing, reliably surpassing previous best scores.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def fast_sum_radii(centers):
    """
    Computes a valid sum of radii for fixed centers using a fast heuristic.
    r_i = min(distance_to_boundary, 0.5 * min_distance_to_other_circles)
    This is a feasible lower bound used as the SA objective.
    """
    b = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(d, np.inf)
    p = 0.5 * np.min(d, axis=1)
    return np.sum(np.minimum(b, p))

def compute_slsqp_obj(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N_CIRCLES:])

def compute_slsqp_cons(v):
    """Computes inequality constraints for SLSQP (must be >= 0)."""
    c = v[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    r = v[2*N_CIRCLES:]
    out = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    
    d2 = (c[:, np.newaxis, :] - c[np.newaxis, :, :])**2
    dists = np.sqrt(np.sum(d2, axis=2))
    np.fill_diagonal(dists, np.inf)
    mask = np.triu_indices(N_CIRCLES, k=1)
    out.append(dists[mask] - (r[mask[0]] + r[mask[1]]))
    return np.concatenate(out)

def run_packing():
    rng = np.random.default_rng(42)
    N = N_CIRCLES
    
    # 1. Initial Configuration: Dense hexagonal lattice
    centers = np.zeros((N, 2))
    r_est = 0.1
    dy = r_est * np.sqrt(3)
    row = 0
    cnt = 0
    while cnt < N:
        y = r_est + row * dy
        if y + r_est > 1.0: break
        offset = r_est if row % 2 == 1 else 0.0
        x = r_est + offset
        while x + r_est <= 1.0 and cnt < N:
            centers[cnt] = [x, y]
            x += 2 * r_est
            cnt += 1
        row += 1
    while cnt < N:
        centers[cnt] = rng.uniform(0.1, 0.9, 2)
        cnt += 1
        
    best_centers = centers.copy()
    best_fast = fast_sum_radii(centers)
    
    # 2. Simulated Annealing on centers using fast heuristic
    for restart in range(4):
        curr = best_centers.copy() + rng.normal(0, 0.005, (N, 2))
        curr = np.clip(curr, 0.0, 1.0)
        curr_val = fast_sum_radii(curr)
        
        temp = 0.01
        step = 0.015
        
        for _ in range(30000):
            idx = rng.integers(N)
            new_c = curr.copy()
            new_c[idx, 0] += rng.uniform(-step, step)
            new_c[idx, 1] += rng.uniform(-step, step)
            new_c[idx] = np.clip(new_c[idx], 0.0, 1.0)
            
            new_val = fast_sum_radii(new_c)
            delta = new_val - curr_val
            
            if delta > 0 or rng.random() < np.exp(delta / temp):
                curr = new_c
                curr_val = new_val
                if curr_val > best_fast:
                    best_fast = curr_val
                    best_centers = curr.copy()
            temp *= 0.9995
            if temp < 1e-7: temp = 1e-7
            
    # 3. Exact LP refinement for optimal radii given best centers
    n_pairs = N*(N-1)//2
    n_bnd = 4*N
    A_ub = np.zeros((n_pairs+n_bnd, N))
    idx = 0
    for i in range(N):
        for j in range(i+1, N):
            A_ub[idx, i] = 1.0; A_ub[idx, j] = 1.0; idx += 1
    for i in range(N):
        for _ in range(4):
            A_ub[idx, i] = 1.0; idx += 1
    c_obj = -np.ones(N)
            
    b_ub = np.zeros(n_pairs+n_bnd)
    idx = 0
    for i in range(N):
        for j in range(i+1, N):
            b_ub[idx] = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
            idx += 1
    for i in range(N):
        x, y = best_centers[i]
        b_ub[idx]=x; idx+=1
        b_ub[idx]=1.0-x; idx+=1
        b_ub[idx]=y; idx+=1
        b_ub[idx]=1.0-y; idx+=1
        
    res_lp = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    best_radii = res_lp.x if res_lp.success else np.full(N, 0.08)
    
    # 4. Joint SLSQP polish (centers + radii)
    x0 = np.concatenate([best_centers.flatten(), best_radii])
    bounds = [(0,1)]*(2*N) + [(0,0.5)]*N
    try:
        res = minimize(compute_slsqp_obj, x0, method='SLSQP', bounds=bounds, 
                       constraints={'type':'ineq','fun':compute_slsqp_cons},
                       options={'maxiter':3000, 'ftol':1e-12})
        if -res.fun > np.sum(best_radii):
            best_centers = res.x[:2*N].reshape(N, 2)
            best_radii = res.x[2*N:]
    except Exception:
        pass
        
    # 5. Final safety shrink to guarantee validation tolerance
    for _ in range(5):
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                if d < best_radii[i]+best_radii[j]-1e-12:
                    sh = (best_radii[i]+best_radii[j]-d)/2.0+1e-9
                    best_radii[i] -= sh; best_radii[j] -= sh
        for i in range(N):
            mx = min(best_centers[i,0], 1.0-best_centers[i,0], best_centers[i,1], 1.0-best_centers[i,1])
            if best_radii[i] > mx + 1e-12: best_radii[i] = mx
            
    return best_centers, best_radii, float(np.sum(best_radii))
