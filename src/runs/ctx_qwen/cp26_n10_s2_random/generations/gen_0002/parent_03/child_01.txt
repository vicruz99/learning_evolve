import numpy as np
from scipy.optimize import linprog, minimize

N = 26
# Precompute pairwise indices for SLSQP constraints
TRI_U = np.triu_indices(N, k=1)

def get_radii_lp(centers):
    """
    Given fixed centers, solve an LP to maximize sum of radii.
    Constraints: r_i + r_j <= dist(i,j), r_i <= min(x, 1-x, y, 1-y)
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    m_pairs = n * (n - 1) // 2
    m_bound = 4 * n
    A_ub = np.zeros((m_pairs + m_bound, n))
    b_ub = np.zeros(m_pairs + m_bound)
    
    idx = 0
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints
    for i in range(n):
        bounds = [centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1]]
        for b in bounds:
            A_ub[idx, i] = 1.0
            b_ub[idx] = b
            idx += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0.0, None), method='highs')
    return res.x if res.success else np.full(n, 0.001)

def compute_constraints(vars_flat):
    """Computes boundary and non-overlap constraints for SLSQP."""
    X = vars_flat.reshape(N, 3)
    xs, ys, rs = X[:, 0], X[:, 1], X[:, 2]
    
    c = []
    # Boundary: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(xs - rs)
    c.append(1.0 - xs - rs)
    c.append(ys - rs)
    c.append(1.0 - ys - rs)
    
    # Overlap: dist^2 >= (r_i + r_j)^2
    idx_i, idx_j = TRI_U
    dx = xs[idx_i] - xs[idx_j]
    dy = ys[idx_i] - ys[idx_j]
    dr = rs[idx_i] + rs[idx_j]
    c.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(c)

def compute_objective(vars_flat):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars_flat[2::3])

def run_fd_and_refine(seed):
    """Runs Force-Directed + LP expansion, followed by SLSQP refinement."""
    rng = np.random.default_rng(seed)
    
    # Hexagonal initialization
    centers = []
    r_init = 0.09
    y = r_init
    row = 0
    while len(centers) < N:
        x = r_init if row % 2 == 0 else 2 * r_init
        while x + r_init <= 1.0 and len(centers) < N:
            centers.append([x, y])
            x += 2 * r_init
        y += np.sqrt(3) * r_init
        row += 1
    centers = np.array(centers[:N])
    centers += rng.normal(0, 0.005, centers.shape)
    
    radii = get_radii_lp(centers)
    velocities = np.zeros_like(centers)
    
    # Simulation parameters
    k_rep = 120.0
    k_bound = 120.0
    dt = 0.008
    alpha = 0.9
    steps = 2000
    
    best_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for step in range(steps):
        temp = 1.0 - step / steps
        curr_dt = dt * (0.1 + 0.9 * temp)
        curr_alpha = 0.3 + 0.6 * temp
        
        # Vectorized pairwise forces
        diffs = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-12)
        safe_dists = np.maximum(dists, 1e-6)
        overlaps = radii[:, None] + radii[None, :] - dists
        overlaps = np.maximum(0, overlaps)
        
        dirs = diffs / safe_dists[:, :, None]
        force_mag = overlaps * k_rep
        force_vecs = dirs * force_mag[:, :, None]
        
        forces = np.sum(force_vecs, axis=1)
        
        # Boundary repulsion
        for i in range(N):
            r = radii[i]
            cx, cy = centers[i]
            if cx < r: forces[i, 0] += k_bound * (r - cx)
            elif cx > 1 - r: forces[i, 0] -= k_bound * (cx - (1 - r))
            if cy < r: forces[i, 1] += k_bound * (r - cy)
            elif cy > 1 - r: forces[i, 1] -= k_bound * (cy - (1 - r))
            
        velocities = curr_alpha * velocities + curr_dt * forces
        centers += velocities
        centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
        
        # Periodically update radii via exact LP
        if step % 3 == 0:
            radii = get_radii_lp(centers)
            curr_sum = np.sum(radii)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                
    # SLSQP local refinement
    x0 = np.zeros(N * 3)
    for i in range(N):
        x0[3 * i] = best_centers[i, 0]
        x0[3 * i + 1] = best_centers[i, 1]
        x0[3 * i + 2] = best_radii[i]
        
    # Small perturbation to help SLSQP explore neighborhood
    x0 += rng.normal(0, 1e-5, x0.shape)
    
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    try:
        res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12})
        if np.all(compute_constraints(res.x) >= -1e-7):
            opt_centers = res.x[:2 * N].reshape(N, 2)
            opt_radii = res.x[2 * N:]
            if np.sum(opt_radii) > best_sum:
                best_centers = opt_centers
                best_radii = opt_radii
                best_sum = np.sum(opt_radii)
    except Exception:
        pass
        
    # Final LP pass to guarantee consistency and strict feasibility
    final_radii = get_radii_lp(best_centers) * 0.9999
    return best_centers, final_radii, float(np.sum(final_radii))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main entry point. Runs multiple restarts and returns the best valid packing."""
    best_overall_sum = -1.0
    best_overall_centers = None
    best_overall_radii = None
    
    # Multiple restarts to escape local minima
    for seed in range(5):
        c, r, s = run_fd_and_refine(seed)
        if s > best_overall_sum:
            best_overall_sum = s
            best_overall_centers = c
            best_overall_radii = r
            
    return best_overall_centers, best_overall_radii, float(best_overall_sum)
