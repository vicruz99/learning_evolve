import numpy as np
from scipy.optimize import minimize

N = 26
TRI_MASK = np.triu(np.ones((N, N), dtype=bool), k=1)

def compute_constraints(v):
    """
    Computes all boundary and non-overlap constraints.
    Returns a 1D array where each element must be >= 0.
    """
    x = v.reshape(N, 3)
    xs, ys, rs = x[:, 0], x[:, 1], x[:, 2]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.append(xs - rs)
    c.append(1.0 - xs - rs)
    c.append(ys - rs)
    c.append(1.0 - ys - rs)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    diff_x = xs[:, np.newaxis] - xs[np.newaxis, :]
    diff_y = ys[:, np.newaxis] - ys[np.newaxis, :]
    sum_r = rs[:, np.newaxis] + rs[np.newaxis, :]
    
    c.append((diff_x**2 + diff_y**2 - sum_r**2)[TRI_MASK])
    
    return np.concatenate(c)

def obj_func(v):

    return -np.sum(v[2::3])

    b = []

    return b

        row += 1

    v0[2::3] = rs
