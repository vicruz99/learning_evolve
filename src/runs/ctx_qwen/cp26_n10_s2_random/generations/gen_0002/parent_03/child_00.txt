import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars_flat):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def compute_constraints(vars_flat):
    """
    Computes all boundary and non-overlap constraints.
    Returns a 1D array where each element must be >= 0.
    """
    X = vars_flat[:2*N].reshape(N, 2)
    R = vars_flat[2*N:]
    
    c = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(X[:, 0] - R)
    c.append(1.0 - X[:, 0] - R)
    c.append(X[:, 1] - R)
    c.append(1.0 - X[:, 1] - R)
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    dx = X[:, 0, None] - X[None, :, 0]
    dy = X[:, 1, None] - X[None, :, 1]
    dist = np.sqrt(dx**2 + dy**2 + 1e-16)
    
    idx_i, idx_j = np.triu_indices(N, k=1)
    c.append(dist[idx_i, idx_j] - (R[idx_i] + R[idx_j]))
    
    return np.concatenate(c)

def get_bounds():
    """Returns variable bounds for x, y in [0,1] and r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def hex_init(row_counts, seed, r0):
    """
    Generates a strictly feasible initial configuration from a hexagonal grid.
    """
    rng = np.random.default_rng(seed)
    pts = []
    y = r0
    for row_idx, count in enumerate(row_counts):
        shift = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(count):
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
        
    pts = np.array(pts[:N])
    # Add small perturbation to break symmetry and ensure strict feasibility
    pts += rng.normal(0, 0.001, pts.shape)
    pts = np.clip(pts, 0.001, 0.999)
    
    x0 = np.zeros(3*N)
    x0[:2*N] = pts.flatten()
    # Start with radii slightly smaller than center spacing to satisfy constraints strictly
    x0[2*N:] = r0 * 0.94
    return x0

    """
    Iteratively shrinks radii to guarantee strict feasibility within tolerance.
    """
    radii = radii.copy()
    for _ in range(100):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-10:
                radii[i] = max_r
                changed = True

            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

    bounds = get_bounds()
