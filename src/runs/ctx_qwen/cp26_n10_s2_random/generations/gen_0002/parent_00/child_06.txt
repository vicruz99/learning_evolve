from scipy.optimize import differential_evolution, minimize
import math

def get_max_radii(centers):
    """Compute maximum feasible radii for fixed centers respecting boundaries and non-overlap."""
    n = centers.shape[0]
    # Distance to boundaries
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    # Radius limited by half the distance to nearest neighbor
    rp = 0.5 * np.min(dists, axis=1)
    return np.minimum(rb, rp)

    """Objective for optimization: minimize negative sum of radii."""
    n = 26
    centers = x.reshape(-1, 2)
    radii = get_max_radii(centers)
    return -np.sum(radii)
