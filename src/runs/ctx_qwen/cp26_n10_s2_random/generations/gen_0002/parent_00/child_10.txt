import numpy as np
from scipy.optimize import differential_evolution, minimize

def compute_radii(centers):
    """
    Computes the maximum valid radii for a given set of centers.
    Each radius is limited by the distance to boundaries and half the distance to nearest neighbors.
    This guarantees non-overlap and boundary constraints are satisfied by construction.
    """
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    
    # Distance to boundaries: min(x, 1-x, y, 1-y)
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # Ignore self-distances
    
    # Radius constrained by half the minimum distance to avoid overlap
    min_dists = np.min(dists, axis=1)
    r_pair = 0.5 * min_dists
    
    # The actual radius is the tighter constraint between boundary and neighbors
    return np.minimum(r_bound, r_pair)

def objective(centers_flat):
    """
    Objective function for optimizers. 
    Minimizes the negative sum of radii, which is equivalent to maximizing the sum of radii.
    """
    centers = centers_flat.reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)

def get_hex_centers(n, seed):
    """
    Generates an initial configuration based on a hexagonal lattice,
    perturbed slightly to break symmetry and escape degenerate local minima.
    """
    rng = np.random.default_rng(seed)
    r_est = 0.105
    dy = r_est * np.sqrt(3)
    dx = 2 * r_est
    pts = []
    row = 0
    while len(pts) < n:
        y = r_est + row * dy
        if y + r_est > 0.99:
            break
        offset = (dx / 2.0) if row % 2 == 1 else 0.0
        x = r_est + offset
        while x + r_est <= 0.99 and len(pts) < n:
            pts.append([x, y])
            x += dx
        row += 1
    while len(pts) < n:
        pts.append([0.5, 0.5])
    pts = np.array(pts[:n])
    pts += rng.normal(0, 0.005, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Bounds for center coordinates. Keeping them slightly inside avoids degenerate boundary cases.
    bounds = [(0.001, 0.999)] * (2 * n)
    best_sum = 0.0
    best_centers = None
    
    # 1. Global Search with Differential Evolution
    # DE is robust for non-smooth objective functions like this packing problem.
    res_de = differential_evolution(
        objective, bounds,
        seed=42,
        maxiter=300,
        popsize=20,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-8,
        polish=False
    )
    best_sum = -res_de.fun
    best_centers = res_de.x.reshape(-1, 2)
    
    # 2. Local Polishing with Powell's Method
    # Powell handles non-differentiable plateaus effectively for fine-tuning.
    res_powell = minimize(objective, best_centers.flatten(), method='Powell',
                          options={'maxiter': 2000, 'ftol': 1e-12, 'xtol': 1e-12})
    if -res_powell.fun > best_sum:
        best_sum = -res_powell.fun
        best_centers = res_powell.x.reshape(-1, 2)
        
    # 3. Multi-Start from Structured Hexagonal Configurations
    # Provides diverse high-quality starting points to ensure we find the global optimum.
    for seed in range(10):
        c0 = get_hex_centers(n, seed).flatten()
        res = minimize(objective, c0, method='Powell', options={'maxiter': 1500, 'ftol': 1e-12})
        if -res.fun > best_sum:
            best_sum = -res.fun
            best_centers = res.x.reshape(-1, 2)
            
    # 4. Compute Final Radii and Ensure Numerical Safety
    radii = compute_radii(best_centers)
    # Slight shrink to prevent floating-point precision issues in the strict validator
    radii = np.maximum(radii - 1e-9, 1e-9)
    
    return best_centers, radii, float(np.sum(radii))
