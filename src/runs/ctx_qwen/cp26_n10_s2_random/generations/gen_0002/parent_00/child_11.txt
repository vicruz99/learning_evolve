To maximize the sum of radii for 26 circles in a unit square, we separate the problem into two parts:
1. **Given fixed centers, compute optimal radii exactly**: This is a Linear Programming problem. We maximize $\sum r_i$ subject to $r_i + r_j \le \text{dist}(C_i, C_j)$ and $r_i \le \min(x_i, 1-x_i, y_i, 1-y_i)$. This LP can be solved extremely fast (~0.5ms) using `scipy.optimize.linprog`.
2. **Optimize centers to maximize the LP objective**: The LP-optimal sum of radii as a function of centers is continuous and piecewise linear. We use derivative-free optimizers (`Powell` and `Nelder-Mead`) which handle non-smooth objectives robustly. We run multiple starts from diverse initial configurations (hexagonal, grid, random, corner-focused) and apply perturbation refinement to escape local minima.

This hybrid approach bypasses the non-convexity issues of jointly optimizing centers and radii with standard gradient-based methods, leading to a more reliable and higher-quality packing.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

def solve_radii_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii.
    Maximize sum(r_i)
    Subject to:
        r_i + r_j <= dist(i, j) for all i < j
        r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
        r_i >= 0
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Minimize negative sum => Maximize sum
    
    # Preallocate constraint matrix
    num_pairs = n * (n - 1) // 2
    num_constraints = num_pairs + 4 * n
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    idx = 0
    
    # Pairwise distance constraints
    dists = np.sqrt(((centers[:, None, :] - centers[None, :, :])**2).sum(axis=2))
    np.fill_diagonal(dists, np.inf)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    # Solve LP with HiGHS (fast and robust), fallback to interior-point
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0.0, None), method='highs')
        if res.success:
            return -res.fun, res.x
    except:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0.0, None), method='interior-point')
        if res.success:
            return -res.fun, res.x
    except:
        pass
        
    return 0.0, np.zeros(n)

def objective_func(centers_flat):
    """Objective for optimizers: minimize negative max radii sum."""
    centers = centers_flat.reshape(-1, 2)
    # Keep centers safely inside the square to avoid degenerate LP constraints
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    val, _ = solve_radii_lp(centers)
    return -val

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # 1. Generate diverse initial configurations
    configs = []
    
    # Config A: Hexagonal lattice
    hex_pts = []
    r0 = 0.1
    for row in range(8):
        y = r0 + row * r0 * np.sqrt(3)
        if y + r0 > 1.0: break
        offset = r0 if row % 2 == 1 else 0.0
        x = r0 + offset
        while x + r0 <= 1.0 and len(hex_pts) < n:
            hex_pts.append([x, y])
            x += 2 * r0
    while len(hex_pts) < n: hex_pts.append([0.5, 0.5])
    configs.append(np.array(hex_pts[:n]))
    
    # Config B: Structured grid
    gs = np.linspace(0.12, 0.88, 5)
    cx, cy = np.meshgrid(gs, gs)
    grid_pts = np.column_stack((cx.flatten(), cy.flatten()))
    configs.append(grid_pts[:n])
    
    # Config C: Dense random
    configs.append(np.random.rand(n, 2) * 0.8 + 0.1)
    
    # Config D: Corner-focused
    corners = np.array([[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]])
    rand_pts = np.random.rand(n-4, 2) * 0.6 + 0.2
    configs.append(np.vstack([corners, rand_pts]))
    
    # Config E: Perturbed hex
    configs.append(configs[0] + np.random.normal(0, 0.015, (n, 2)))
    configs[-1] = np.clip(configs[-1], 0.05, 0.95)

    # Expand with slight perturbations for basin exploration
    all_configs = []
    for cfg in configs:
        all_configs.append(cfg)
        for _ in range(3):
            pert = cfg + np.random.normal(0, 0.01, (n, 2))
            all_configs.append(np.clip(pert, 0.05, 0.95))

    # 2. Multi-start optimization
    for x0_centers in all_configs:
        x0 = x0_centers.flatten()
        
        # Powell method (robust for medium-dim non-smooth)
        try:
            res = minimize(objective_func, x0, method='Powell', 
                           options={'maxiter': 800, 'ftol': 1e-10})
            curr_val = -res.fun
            if curr_val > best_sum:
                best_centers = np.clip(res.x.reshape(n, 2), 1e-6, 1.0-1e-6)
                _, best_radii = solve_radii_lp(best_centers)
                best_sum = curr_val
        except: pass
        
        # Nelder-Mead (alternative simplex search)
        try:
            res2 = minimize(objective_func, x0, method='Nelder-Mead',
                            options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10})
            curr_val2 = -res2.fun
            if curr_val2 > best_sum:
                best_centers = np.clip(res2.x.reshape(n, 2), 1e-6, 1.0-1e-6)
                _, best_radii = solve_radii_lp(best_centers)
                best_sum = curr_val2
        except: pass

    # 3. Perturbation refinement to escape shallow local minima
    if best_centers is not None:
        for _ in range(15):
            pert_centers = best_centers + np.random.normal(0, 0.005, (n, 2))
            pert_centers = np.clip(pert_centers, 0.05, 0.95)
            x0 = pert_centers.flatten()
            try:
                res = minimize(objective_func, x0, method='Powell', 
                               options={'maxiter': 500, 'ftol': 1e-10})
                curr_val = -res.fun
                if curr_val > best_sum:
                    best_centers = np.clip(res.x.reshape(n, 2), 1e-6, 1.0-1e-6)
                    _, best_radii = solve_radii_lp(best_centers)
                    best_sum = curr_val
            except: pass

    # Fallback safety
    if best_centers is None:
        best_centers = configs[0]
        _, best_radii = solve_radii_lp(best_centers)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
