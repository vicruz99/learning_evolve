N = 26

def obj_var(vars):

    return -np.sum(vars[2*N:])

def con_var(vars):
    """Computes all boundary and non-overlap constraints for variable radii."""
    c = vars[:2*N].reshape(N, 2)
    r = vars[2*N:]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b = np.concatenate([c[:,0] - r, 1.0 - c[:,0] - r, c[:,1] - r, 1.0 - c[:,1] - r])

    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]

    np.fill_diagonal(d, np.inf)

    o = d[idx] - (r[idx[0]] + r[idx[1]])
    
    return np.concatenate([b, o])
