1. **Fix Centers, Optimize Radii via Linear Programming**: Given fixed center positions, the problem of maximizing $\sum r_i$ subject to $r_i + r_j \le \|c_i - c_j\|$ and boundary constraints $r_i \le \min(x_i, 1-x_i, y_i, 1-y_i)$ is a Linear Programming problem. 3. **Multi-Restart Strategy**: Run the SA+Refinement pipeline multiple times from perturbed hexagonal lattice initializations to explore the configuration space thoroughly.
4. **Numerical Safety**: Apply a tiny shrinkage factor to the final radii to guarantee strict compliance with the validation function's tolerance (`1e-12`), avoiding false rejection due to floating-point precision limits.

This hybrid LP + SA approach is highly robust for circle packing problems and typically yields denser, more optimal configurations than joint gradient-based methods.

```python
import numpy as np
from scipy.optimize import linprog, minimize
import math

N = 26

# Precompute constant parts of the LP to speed up evaluations
PAIR_INDICES = [(i, j) for i in range(N) for j in range(i + 1, N)]
NUM_PAIRS = len(PAIR_INDICES)

# A_ub matrix for pairwise constraints: r_i + r_j <= d_ij
A_UB = np.zeros((NUM_PAIRS, N))
for k, (i, j) in enumerate(PAIR_INDICES):
    A_UB[k, i] = 1.0
    A_UB[k, j] = 1.0

def compute_max_radii(centers):
    """
    Given fixed centers, solve LP to find radii that maximize sum(radii)
    subject to non-overlap and boundary constraints.
    Returns (sum_radii, radii_array)
    """
    # Compute pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Right-hand side for pairwise constraints
    b_ub = dists[np.triu_indices(N, k=1)]
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    b_min = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    b_min = np.maximum(b_min, 0.0)  # Ensure non-negative bounds
    
    # LP: min -sum(r)  s.t. A_ub @ r <= b_ub, 0 <= r <= b_min
    c_obj = -np.ones(N)
    bounds = [(0.0, b_min[i]) for i in range(N)]
    
    res = linprog(c_obj, A_ub=A_UB, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    
    # Fallback direct resolver if LP fails (rare, usually due to extreme configurations)
    radii = b_min.copy()
    for _ in range(10):
        changed = False
        for k, (i, j) in enumerate(PAIR_INDICES):
            if radii[i] + radii[j] > dists[i, j]:
                overlap = radii[i] + radii[j] - dists[i, j]
                radii[i] -= overlap / 2.0
                radii[j] -= overlap / 2.0
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    return np.sum(radii), radii

def objective_sa(centers_flat):
    """Objective for optimization: minimize negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    s, _ = compute_max_radii(centers)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # 1. Generate initial hexagonal lattice configuration
    r_est = 0.105
    dy = r_est * math.sqrt(3)
    dx = 2.0 * r_est
    centers_init = []
    row = 0
    while len(centers_init) < N:
        y = r_est + row * dy
        if y + r_est > 1.0:
            break
        offset = (dx / 2.0) if row % 2 == 1 else 0.0
        x = r_est + offset
        while x + r_est <= 1.0 and len(centers_init) < N:
            centers_init.append([x, y])
            x += dx
        row += 1
    while len(centers_init) < N:
        centers_init.append([np.random.rand(), np.random.rand()])
    centers_init = np.array(centers_init[:N])

    # 2. Multi-start Simulated Annealing + Local Refinement
    for seed in range(6):
        np.random.seed(seed)
        curr_centers = centers_init.copy()
        # Add noise to escape initial symmetry
        curr_centers += np.random.normal(0, 0.025, curr_centers.shape)
        curr_centers = np.clip(curr_centers, 0.05, 0.95)
        
        curr_flat = curr_centers.flatten()
        curr_obj = objective_sa(curr_flat)
        
        # Simulated Annealing parameters
        temp = 0.008
        for step in range(4000):
            idx = np.random.randint(N)
            new_flat = curr_flat.copy()
            # Perturb one circle
            new_flat[2*idx] += np.random.normal(0, temp)
            new_flat[2*idx+1] += np.random.normal(0, temp)
            
            # Strict bounds clipping
            new_flat[2*idx] = np.clip(new_flat[2*idx], 0.01, 0.99)
            new_flat[2*idx+1] = np.clip(new_flat[2*idx+1], 0.01, 0.99)
            
            new_obj = objective_sa(new_flat)
            delta = new_obj - curr_obj
            
            # Acceptance criterion (minimization problem)
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                curr_flat = new_flat
                curr_obj = new_obj
                
            # Cooling schedule
            temp *= 0.9990
            
        # Local refinement using Nelder-Mead (robust for non-smooth LP objective)
        try:
            res = minimize(objective_sa, curr_flat, method='Nelder-Mead', 
                           options={'maxiter': 3000, 'xatol': 1e-7, 'fatol': 1e-9})
            curr_flat = res.x
            curr_obj = objective_sa(curr_flat)
        except Exception:
            pass
            
        final_centers = curr_flat.reshape(N, 2)
        final_sum, final_radii = compute_max_radii(final_centers)
        
        if final_sum > best_sum:
            best_sum = final_sum
            best_centers = final_centers.copy()
            best_radii = final_radii.copy()

    # 3. Final Safety Adjustment
    centers = best_centers if best_centers is not None else np.random.rand(N, 2)
    radii = best_radii if best_radii is not None else np.full(N, 0.08)
    
    # Shrink radii slightly to guarantee strict compliance with validator's 1e-12 tolerance
    radii *= 0.9999
    
    sum_radii = float(np.sum(radii))
    return centers, radii, sum_radii
```
