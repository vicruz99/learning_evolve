from scipy.optimize import minimize
import math

N = 26
UPPER_TRIU_INDICES = np.triu_indices(N, k=1)

def objective_func(vars_flat):
    radii = vars_flat[2*N:]
    return -np.sum(radii)

def constraint_func(vars_flat):
    centers = vars_flat[:2*N].reshape(N, 2)
    radii = vars_flat[2*N:]
    
    cons = []
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    cons.append(centers[:, 0] - radii)
    cons.append(1.0 - centers[:, 0] - radii)
    cons.append(centers[:, 1] - radii)
    cons.append(1.0 - centers[:, 1] - radii)
    
    # Overlap constraints: dist_sq >= (r_i + r_j)^2
    # Using squared distances avoids sqrt singularities and improves gradient flow
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    cons.append(dist_sq[UPPER_TRIU_INDICES] - r_sum_sq[UPPER_TRIU_INDICES])
    
    return np.concatenate([c.flatten() for c in cons])

def generate_initial_configs(num_configs):

    np.random.seed(42)
    
    # Hexagonal lattices with varying spacing to cover different density regimes
    spacings = [0.16, 0.18, 0.20, 0.22, 0.24]

            np.random.seed(seed + sp*100)
            centers = []

                while x <= 1.0 - sp and len(centers) < N:
                    centers.append([x, y])
                    x += sp * 2.0
                y += sp * math.sqrt(3)
                row += 1

            c += np.random.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.05, 0.95)

        c = np.random.rand(N, 2) * 0.7 + 0.15
        r = np.full(N, 0.04)
        configs.append(np.concatenate([c.flatten(), r]))
        
    return configs
