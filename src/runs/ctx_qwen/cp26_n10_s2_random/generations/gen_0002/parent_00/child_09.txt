

<strategy>
The strategy focuses on a hybrid pipeline combining force-directed relaxation with gradient-based constrained optimization:
1. **Diverse Initialization**: Generate multiple starting configurations (hexagonal lattice, square grid, and random dense packing) to cover different structural basins.
2. **Force-Directed Relaxation**: For each initial configuration, run a simulation that applies repulsive forces between overlapping circles and boundary forces, while slowly and uniformly growing all radii. This quickly resolves overlaps and pushes the packing towards high density, effectively acting as a global pre-optimizer that avoids getting stuck in poor local minima.
3. **Feasibility Projection**: After relaxation, compute the exact maximum feasible radii for the resulting centers. This guarantees a strictly feasible starting point for the subsequent gradient-based optimizer.
4. **Constrained Optimization (SLSQP)**: Use `scipy.optimize.minimize` with the SLSQP method to jointly optimize centers and radii. The objective maximizes the sum of radii. Constraints are formulated using squared distances (`(dx^2 + dy^2) - (r_i + r_j)^2 >= 0`) to ensure smoothness and better conditioning than using square roots. Multiple restarts with small perturbations are applied to the best configuration.
5. **Strict Validation & Repair**: A final iterative pass clamps radii to ensure absolute compliance with the validation function's numerical tolerances (1e-12), shrinking radii proportionally if any microscopic overlaps or boundary violations remain due to floating-point drift.
This approach leverages the speed of physical simulation for global exploration and the precision of SLSQP for local exploitation, consistently yielding high-quality packings.

import numpy as np
from scipy.optimize import minimize
import math

N = 26

def get_feasible_radii(centers):
    """Compute maximum valid radii for a fixed set of centers."""
    n = centers.shape[0]
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    rp = 0.5 * np.min(dists, axis=1)
    return np.minimum(rb, rp)

def compute_forces(centers, radii):
    """Compute repulsive forces for overlaps and boundary violations."""
    n = len(centers)
    forces = np.zeros_like(centers)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    dist = np.sqrt(dist_sq + 1e-12)
    min_dist = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    overlap = np.maximum(0, min_dist - dist)
    force_mag = overlap * 5000.0
    force_dir = diff / (dist[:, :, np.newaxis] + 1e-12)
    forces += np.sum(force_mag[:, :, np.newaxis] * force_dir, axis=1)
    
    forces[:, 0] += 2000.0 * (np.maximum(0, radii - centers[:, 0]) - np.maximum(0, centers[:, 0] + radii - 1.0))
    forces[:, 1] += 2000.0 * (np.maximum(0, radii - centers[:, 1]) - np.maximum(0, centers[:, 1] + radii - 1.0))
    return forces

def relax_config(centers, radii, steps=20000):
    """Run force-directed relaxation with uniform radius growth."""
    for s in range(steps):
        forces = compute_forces(centers, radii)
        centers += forces * 0.00015
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        if s % 50 == 0:
            radii *= 1.00005
    return centers, radii

def obj_fun(vars):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2*N:])

def constraint_fun(vars):
    """Constraints: boundary containment and non-overlap (squared distances)."""
    c = vars[:2*N].reshape(N, 2)
    r = vars[2*N:]
    
    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    
    dx = c[:, 0, np.newaxis] - c[np.newaxis, :, 0]
    dy = c[:, 1, np.newaxis] - c[np.newaxis, :, 1]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    con = np.concatenate([con, (dx**2 + dy**2)[mask] - dr[mask]**2])
    return con

def run_packing() -> tuple:
    best_sum = -1.0
    best_c = None
    best_r = None
    
    np.random.seed(42)
    
    configs = []
    
    # Config 1: Hexagonal lattice
    r0 = 0.09
    dy = r0 * math.sqrt(3)
    cx, cy = [], []
    row = 0
    while len(cx) < N:
        y = r0 + row * dy
        if y + r0 > 1.0: break
        offset = r0 if row % 2 == 1 else 0.0
        x = r0 + offset
        while x + r0 <= 1.0 and len(cx) < N:
            cx.append(x)
            cy.append(y)
            x += 2 * r0
        row += 1
    while len(cx) < N:
        cx.append(0.5)
        cy.append(0.5)
    configs.append(np.column_stack((cx[:N], cy[:N])))
    
    # Config 2: Structured Grid
    xg = np.linspace(0.15, 0.85, 5)
    yg = np.linspace(0.15, 0.85, 5)
    gx, gy = np.meshgrid(xg, yg)
    configs.append(np.column_stack((gx.flatten()[:N], gy.flatten()[:N])))
    
    # Config 3: Dense Random
    configs.append(np.random.rand(N, 2) * 0.7 + 0.15)
    
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    constraint = {'type': 'ineq', 'fun': constraint_fun}
    
    for cfg in configs:
        for seed in range(5):
            np.random.seed(seed)
            c_init = cfg.copy() + np.random.normal(0, 0.005, cfg.shape)
            c_init = np.clip(c_init, 0.1, 0.9)
            r_init = np.full(N, 0.06)
            
            c_rel, r_rel = relax_config(c_init, r_init, steps=20000)
            r_feas = get_feasible_radii(c_rel)
            
            x0 = np.concatenate([c_rel.flatten(), r_feas])
            
            try:
                res = minimize(obj_fun, x0, method='SLSQP', bounds=bounds,
                               constraints=constraint, options={'maxiter': 4000, 'ftol': 1e-13})
                
                c_opt = res.x[:2*N].reshape(N, 2)
                r_opt = res.x[2*N:]
                
                r_final = get_feasible_radii(c_opt)
                s = np.sum(r_final)
                
                if s > best_sum:
                    best_sum = s
                    best_c = c_opt.copy()
                    best_r = r_final.copy()
            except Exception:
                pass
                
    centers = best_c if best_c is not None else np.random.rand(N, 2)
    radii = best_r if best_r is not None else np.full(N, 0.05)
    
    # Strict validation repair
    for _ in range(10):
        valid = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            if x - r < -1e-12 or x + r > 1.0 + 1e-12 or y - r < -1e-12 or y + r > 1.0 + 1e-12:
                valid = False
                break
        if not valid:
            break
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            break
        radii *= 0.999
        
    return centers, radii, float(np.sum(radii))
