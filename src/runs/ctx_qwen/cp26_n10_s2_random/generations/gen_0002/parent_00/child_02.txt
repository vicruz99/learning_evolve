import numpy as np
from scipy.optimize import differential_evolution, minimize
import math

def get_max_radii(centers):
    """Computes the maximum valid radius for each circle given fixed centers."""
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    
    # Limit by boundaries
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Limit by nearest neighbor distance
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_pair = 0.5 * np.min(dists, axis=1)
    
    return np.minimum(r_bound, r_pair)

def objective(centers_flat):
    """Objective function: maximize sum of radii => minimize negative sum."""
    centers = centers_flat.reshape(-1, 2)
    # Keep centers strictly inside to avoid degenerate boundary cases
    centers = np.clip(centers, 1e-8, 1.0 - 1e-8)
    radii = get_max_radii(centers)
    return -np.sum(radii)

def get_hex_centers(n):
    """Generates a hexagonal lattice initialization."""
    centers = []
    r_est = 0.095
    dy = r_est * math.sqrt(3)
    dx = 2 * r_est
    row = 0
    while len(centers) < n:
        y = r_est + row * dy
        if y + r_est > 1.0:
            break
        offset = (dx / 2.0) if row % 2 == 1 else 0.0
        x = r_est + offset
        while x + r_est <= 1.0 and len(centers) < n:
            centers.append([x, y])
            x += dx
        row += 1
    # Fill remaining if needed
    while len(centers) < n:
        centers.append([np.random.rand(), np.random.rand()])
    return np.array(centers[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n)
    
    best_centers = None
    best_sum = -1.0
    
    # 1. Hexagonal initialization + local search
    np.random.seed(42)
    hex_c = get_hex_centers(n)
    hex_c += np.random.normal(0, 0.005, hex_c.shape)
    hex_c = np.clip(hex_c, 0.02, 0.98)
    
    res_hex = minimize(objective, hex_c.flatten(), method='Powell', 
                       options={'maxiter': 3000, 'ftol': 1e-12})
    if -res_hex.fun > best_sum:
        best_centers = res_hex.x.reshape(n, 2)
        best_sum = -res_hex.fun
        
    # 2. Differential Evolution for global search
    de_res = differential_evolution(
        objective, 
        bounds, 
        seed=42, 
        maxiter=300, 
        popsize=2, 
        mutation=(0.5, 1.0), 
        recombination=0.9,
        tol=1e-9,
        polish=False,
        init='lhs'
    )
    
    if -de_res.fun > best_sum:
        best_centers = de_res.x.reshape(n, 2)
        best_sum = -de_res.fun
        
    # Polish DE result
    res_de_pw = minimize(objective, de_res.x, method='Powell', 
                         options={'maxiter': 3000, 'ftol': 1e-12})
    if -res_de_pw.fun > best_sum:
        best_centers = res_de_pw.x.reshape(n, 2)
        best_sum = -res_de_pw.fun
        
    # 3. Perturbation search to escape local minima
    for _ in range(40):
        pert = best_centers + np.random.normal(0, 0.002, best_centers.shape)
        pert = np.clip(pert, 0.01, 0.99)
        res = minimize(objective, pert.flatten(), method='Powell', 
                       options={'maxiter': 1500, 'ftol': 1e-12})
        if -res.fun > best_sum:
            best_centers = res.x.reshape(n, 2)
            best_sum = -res.fun
            
    final_radii = get_max_radii(best_centers)
    return best_centers, final_radii, float(np.sum(final_radii))
