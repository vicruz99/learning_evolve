To improve the packing and reach the target sum of radii (~2.636), we need a more robust global search combined with precise local refinement. The previous approaches using only SLSQP with random restarts tend to get trapped in local minima due to the highly non-convex nature of the constraint landscape.

Key improvements in this strategy:
1. **Optimize Centers Only**: Instead of optimizing centers and radii jointly, we treat radii as a deterministic function of centers: `r_i = min(dist_to_boundary, 0.5 * min_j(dist(i, j)))`. This reduces the dimensionality from 78 to 52 and avoids constraint-handling overhead during optimization.
2. **Differential Evolution (DE)**: DE is highly effective for non-smooth, multi-modal objectives like circle packing. It explores the global landscape well and finds dense configurations that gradient-based methods miss.
3. **Powell Local Refinement**: After DE finds a promising basin, we use the Powell method (derivative-free, robust to non-smoothness) to polish the solution to high precision.
4. **Diverse Initialization & Perturbation**: We supplement DE with multiple hexagonal lattice starts and random perturbations around the best found configuration to escape shallow local minima.
5. **Strict Feasibility Repair**: A final post-processing step clamps centers to valid regions, caps radii by boundaries, and iteratively resolves any numerical overlaps to guarantee the validation function passes cleanly.

import numpy as np
from scipy.optimize import minimize, differential_evolution

def compute_max_radii(centers):
    """Computes the maximum valid radius for each circle given fixed centers."""
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Distance to unit square boundaries
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    
    # Minimum distance to any other circle
    min_dist = np.min(dist, axis=1)
    r_pair = 0.5 * min_dist
    
    # Radius is limited by the tighter of boundary or neighbor constraint
    return np.minimum(r_bound, r_pair)

def objective(centers_flat):
    """Objective function to minimize: negative sum of max valid radii."""
    centers = centers_flat.reshape(-1, 2)
    # Clip to stay strictly inside [0,1] to avoid degenerate radius calculations
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    radii = compute_max_radii(centers)
    return -np.sum(radii)

def generate_hex_init(n, r=0.095, scale=0.85, shift=0.075, noise=0.005):
    """Generates a perturbed hexagonal lattice configuration for local optimization starts."""
    centers = []
    dx = 2.0 * r
    dy = np.sqrt(3.0) * r
    idx = 0
    y = r
    row = 0
    while idx < n:
        x = r if row % 2 == 0 else 2.0 * r
        while x < 1.0 - r and idx < n:
            centers.append([x, y])
            idx += 1
            x += dx
        y += dy
        row += 1
        
    centers = np.array(centers[:n])
    if centers.size == 0:
        return np.random.rand(n, 2)
        
    # Normalize to [0,1] and scale/shift to desired density
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0) + 1e-12)
    centers = centers * scale + shift
    centers += np.random.normal(0, noise, centers.shape)
    return np.clip(centers, 0.05, 0.95)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n)
    best_centers = None
    best_sum = -np.inf
    
    np.random.seed(42)
    
    # 1. Global search with Differential Evolution
    de_res = differential_evolution(
        objective,
        bounds,
        seed=42,
        maxiter=250,
        popsize=25,
        mutation=(0.5, 1.5),
        recombination=0.9,
        tol=1e-8,
        polish=False
    )
    
    # 2. High-precision local refinement of DE result
    de_centers = de_res.x.reshape(-1, 2)
    res_powell = minimize(
        objective, de_centers.flatten(), method='Powell',
        bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-13, 'xtol': 1e-13}
    )
    best_centers = res_powell.x.reshape(-1, 2)
    best_sum = -res_powell.fun
    
    # 3. Multiple hexagonal starts to cover alternative dense packings
    for seed in [0, 5, 10, 15, 20, 25, 30]:
        np.random.seed(seed)
        c0 = generate_hex_init(n, r=0.095, scale=0.85, shift=0.075, noise=0.005)
        res = minimize(
            objective, c0.flatten(), method='Powell', bounds=bounds,
            options={'maxiter': 1500, 'ftol': 1e-13, 'xtol': 1e-13}
        )
        curr_sum = -res.fun
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = res.x.reshape(-1, 2)
            
    # 4. Perturbation search around current best to escape shallow minima
    for _ in range(10):
        c_pert = best_centers + np.random.normal(0, 0.002, best_centers.shape)
        c_pert = np.clip(c_pert, 0.05, 0.95)
        res = minimize(
            objective, c_pert.flatten(), method='Powell', bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-13, 'xtol': 1e-13}
        )
        curr_sum = -res.fun
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = res.x.reshape(-1, 2)
            
    # 5. Compute final radii for the optimal centers
    final_radii = compute_max_radii(best_centers)
    
    # Strict boundary clamping
    for i in range(n):
        cx, cy = best_centers[i]
        r = final_radii[i]
        max_r = min(cx, 1.0 - cx, cy, 1.0 - cy)
        if r > max_r + 1e-10:
            final_radii[i] = max_r
            
    # Resolve any numerical overlaps iteratively
    for _ in range(10):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(
                    best_centers[i,0] - best_centers[j,0], 
                    best_centers[i,1] - best_centers[j,1]
                )
                if d < final_radii[i] + final_radii[j] - 1e-10:
                    shrink = (final_radii[i] + final_radii[j] - d) / 2.0 + 1e-10
                    final_radii[i] -= shrink
                    final_radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    # Ensure non-negative radii
    final_radii = np.maximum(final_radii, 1e-9)
    final_sum = float(np.sum(final_radii))
    
    return best_centers, final_radii, final_sum
