import numpy as np
from scipy.optimize import differential_evolution, minimize, linprog

def compute_max_radii(centers):
    """Solves an LP to find radii that maximize sum(r_i) for fixed centers."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    num_constraints = n*(n-1)//2 + 4*n
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    idx = 0
    
    # Pairwise non-overlap: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        vals = (centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        for v in vals:
            A_ub[idx, i] = 1.0
            b_ub[idx] = v
            idx += 1
            
    # Minimize -sum(r_i) subject to A_ub @ r <= b_ub, r >= 0
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0.0, None), method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-6)

def objective(centers_flat):
    """Objective for optimizers: negative sum of maximum feasible radii."""
    centers = centers_flat.reshape(26, 2)
    radii = compute_max_radii(centers)
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Keep centers slightly away from exact boundaries during optimization to maintain LP conditioning
    bounds = [(1e-4, 0.9999)] * (2 * n)
    
    # Global search using Differential Evolution
    de_res = differential_evolution(
        objective, bounds, popsize=15, maxiter=300, 
        mutation=(0.5, 1.0), recombination=0.9, seed=42, polish=False
    )
    
    best_centers = de_res.x.reshape(n, 2)
    
    # Local refinement using derivative-free Powell method
    res_local = minimize(
        objective, best_centers.flatten(), method='Powell',
        options={'maxiter': 2000, 'ftol': 1e-12, 'xtol': 1e-12}
    )
    best_centers = res_local.x.reshape(n, 2)
    
    # Final exact radius computation and safety clamping
    final_radii = compute_max_radii(best_centers)
    final_radii = np.maximum(final_radii, 0.0)
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    return best_centers, final_radii, float(np.sum(final_radii))
