import numpy as np
from scipy.optimize import minimize

def get_max_radii(centers):
    """
    Computes the maximum valid radius for each circle given fixed centers.
    r_i = min(dist to boundary, 0.5 * min(dist to other centers))
    """
    C = np.clip(centers, 0.0, 1.0)
    x, y = C[:, 0], C[:, 1]
    
    # Distance to boundaries
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise Euclidean distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Ignore self-distances
    np.fill_diagonal(dists, np.inf)
    
    # Radius constrained by half the minimum distance to avoid overlap
    r_pair = 0.5 * np.min(dists, axis=1)
    
    # The actual radius is the limiting factor between boundary and neighbors
    return np.minimum(r_bound, r_pair)

def objective_function(centers_flat):
    """
    Objective function for minimizers. 
    Minimizers seek to minimize, so we return the negative sum of radii.
    """
    centers = centers_flat.reshape(-1, 2)
    radii = get_max_radii(centers)
    return -np.sum(radii)

def generate_initial_configs(n, rng):
    """Generates a diverse set of initial center configurations."""
    configs = []
    
    # 1. Perturbed Hexagonal Lattices
    for _ in range(6):
        pts = []
        y = rng.uniform(0.08, 0.12)
        row = 0
        while len(pts) < n:
            x = rng.uniform(0.08, 0.12)
            if row % 2 == 1:
                x += 0.10
            while x <= 0.92 and len(pts) < n:
                pts.append([x, y])
                x += 0.20
            y += 0.1732
            row += 1
        configs.append(np.array(pts[:n]))
        
    # 2. Perturbed Square Grids
    for _ in range(4):
        pts = []
        y = rng.uniform(0.08, 0.12)
        while y <= 0.92 and len(pts) < n:
            x = rng.uniform(0.08, 0.12)
            while x <= 0.92 and len(pts) < n:
                pts.append([x, y])
                x += 0.20
            y += 0.20
        configs.append(np.array(pts[:n]))
        
    # 3. Dense Random Placements
    for _ in range(5):
        configs.append(rng.uniform(0.05, 0.95, (n, 2)))
        
    # 4. Corner & Center Focused
    for _ in range(4):
        base = rng.uniform(0.0, 1.0, (n, 2))
        # Force a few points near corners to encourage boundary usage
        base[0] = [rng.uniform(0.05, 0.15), rng.uniform(0.05, 0.15)]
        base[1] = [rng.uniform(0.85, 0.95), rng.uniform(0.05, 0.15)]
        base[2] = [rng.uniform(0.05, 0.15), rng.uniform(0.85, 0.95)]
        base[3] = [rng.uniform(0.85, 0.95), rng.uniform(0.85, 0.95)]
        base[4] = [rng.uniform(0.45, 0.55), rng.uniform(0.45, 0.55)]
        configs.append(base)
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    rng = np.random.default_rng(42)
    best_centers = None
    best_sum = -1.0
    
    initial_configs = generate_initial_configs(n, rng)
    
    # Phase 1: Local optimization from diverse starts using Powell
    for init_c in initial_configs:
        # Add tiny noise to break exact symmetries
        init_c += rng.normal(0, 1e-4, init_c.shape)
        
        try:
            res = minimize(objective_function, init_c.flatten(), method='Powell',
                           options={'maxiter': 8000, 'ftol': 1e-12, 'xtol': 1e-12})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(-1, 2)
        except Exception:
            continue
            
    # Phase 2: Final polish with Nelder-Mead on the best found configuration
    if best_centers is not None:
        try:
            res_polish = minimize(objective_function, best_centers.flatten(), method='Nelder-Mead',
                                  options={'maxiter': 30000, 'fatol': 1e-13, 'xatol': 1e-13})
            if -res_polish.fun > best_sum:
                best_sum = -res_polish.fun
                best_centers = res_polish.x.reshape(-1, 2)
        except Exception:
            pass
            
    # Fallback if all optimizations failed
    if best_centers is None:
        best_centers = initial_configs[0]
        
    # Compute exact radii for the optimal centers
    radii = get_max_radii(best_centers)
    final_sum = float(np.sum(radii))
    
    return best_centers, radii, final_sum
