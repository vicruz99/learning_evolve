from scipy.optimize import minimize

N = 26

def objective_func(vars):

    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Computes all boundary and non-overlap constraints (must be >= 0)."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints
    c_bound = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints (vectorized)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]

    dist_sq = dx**2 + dy**2
    sum_r_sq = dr**2
    
    return np.concatenate([c_bound, dist_sq[mask] - sum_r_sq[mask]])

def get_initial_config(seed):
