import numpy as np
from scipy.optimize import minimize
import math

def compute_objective(v):
    n = len(v) // 3
    return -np.sum(v[2*n:])

def compute_constraints(v):
    n = len(v) // 3
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    con = []
    
    # Boundary constraints
    con.append(centers[:, 0] - radii)
    con.append(1.0 - centers[:, 0] - radii)
    con.append(centers[:, 1] - radii)
    con.append(1.0 - centers[:, 1] - radii)
    
    # Pairwise non-overlap constraints (squared distances)
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dr = radii[:, None] + radii[None, :]
    dist_sq = dx**2 + dy**2
    sum_r_sq = dr**2
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    con.append(dist_sq[mask] - sum_r_sq[mask])
    
    return np.concatenate(con)

def get_bounds(n):
    b = []
    for _ in range(n):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def generate_hex_init(n, r0, noise):
    centers = np.zeros((n, 2))
    idx = 0
    y = r0
    row = 0
    while idx < n:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x < 1.0 - r0 and idx < n:
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
            x += 2.0 * r0
        y += r0 * math.sqrt(3.0)
        row += 1
        
    while idx < n:
        centers[idx, 0] = np.random.uniform(r0, 1.0 - r0)
        centers[idx, 1] = np.random.uniform(r0, 1.0 - r0)
        idx += 1
        
    if noise > 0:
        centers += np.random.normal(0, noise, centers.shape)
        centers = np.clip(centers, r0, 1.0 - r0)
        
    return centers

def force_directed_relax(centers, radii, steps, growth_rate):
    n = len(radii)
    velocities = np.zeros_like(centers)
    
    for _ in range(steps):
        forces = np.zeros_like(centers)
        max_overlap = 0.0
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = math.hypot(dx, dy)
                min_d = radii[i] + radii[j]
                if d < min_d and d > 1e-9:
                    overlap = min_d - d
                    if overlap > max_overlap:
                        max_overlap = overlap
                    f = overlap * 500.0 / d
                    fx, fy = dx * f, dy * f
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
                    
        for i in range(n):
            cx, cy = centers[i]
            r = radii[i]
            if cx < r:
                forces[i, 0] += (r - cx) * 1000.0
                max_overlap = max(max_overlap, r - cx)
            elif cx > 1.0 - r:
                forces[i, 0] -= (cx - (1.0 - r)) * 1000.0
                max_overlap = max(max_overlap, cx - (1.0 - r))
            if cy < r:
                forces[i, 1] += (r - cy) * 1000.0
                max_overlap = max(max_overlap, r - cy)
            elif cy > 1.0 - r:
                forces[i, 1] -= (cy - (1.0 - r)) * 1000.0
                max_overlap = max(max_overlap, cy - (1.0 - r))
                
        velocities = 0.2 * velocities + forces * 0.005
        centers += velocities
        
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
        if max_overlap < 1e-8:
            radii *= growth_rate
            
    return centers, radii

def refine_solution(centers, radii, n):
    v0 = np.concatenate([centers.flatten(), radii])
    bounds = get_bounds(n)
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    try:
        res = minimize(compute_objective, v0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
        if np.all(compute_constraints(res.x) >= -1e-6):
            return res.x[:2*n].reshape(n, 2), res.x[2*n:]
    except Exception:
        pass
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    for seed in range(15):
        np.random.seed(seed)
        r0 = 0.02
        centers = generate_hex_init(n, r0, noise=0.002)
        radii = np.full(n, r0)
        
        # Stage 1: Force-directed relaxation to find a dense, feasible packing
        centers, radii = force_directed_relax(centers, radii, steps=50000, growth_rate=1.000008)
        
        # Stage 2: Gradient-based refinement to maximize sum of radii precisely
        centers, radii = refine_solution(centers, radii, n)
        
        # Stage 3: Deterministic repair to guarantee strict feasibility
        for _ in range(10):
            changed = False
            for i in range(n):
                cx, cy = centers[i]
                r = radii[i]
                mx = min(cx, 1.0 - cx, cy, 1.0 - cy)
                if r > mx + 1e-9:
                    radii[i] = mx
                    changed = True
            for i in range(n):
                for j in range(i + 1, n):
                    d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                        radii[i] -= shrink
                        radii[j] -= shrink
                        changed = True
            if not changed:
                break
                
        radii = np.maximum(radii, 0.0)
        current_sum = np.sum(radii)
        
        # Quick validation check
        valid = True
        for i in range(n):
            cx, cy = centers[i]
            r = radii[i]
            if cx < r - 1e-9 or cx > 1.0 - r + 1e-9 or cy < r - 1e-9 or cy > 1.0 - r + 1e-9:
                valid = False
                break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
                    
        if valid and current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    # Fallback configuration
    if best_centers is None:
        best_centers = generate_hex_init(n, 0.08, 0.0)
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
