import numpy as np
from scipy.optimize import minimize
import math

def get_initial_configs(n, num_seeds):
    """Generates diverse initial center configurations using perturbed hexagonal grids."""
    configs = []
    for seed in range(num_seeds):
        rng = np.random.default_rng(seed)
        centers = np.zeros((n, 2))
        
        # Hexagonal packing parameters
        r_est = 0.095 + rng.uniform(-0.01, 0.01)
        y = r_est
        row = 0
        idx = 0
        while idx < n:
            x = r_est + (row % 2) * r_est
            while x < 1.0 - r_est and idx < n:
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
                x += 2.0 * r_est
            y += math.sqrt(3.0) * r_est
            row += 1
            
        # Fill remaining if any
        while idx < n:
            centers[idx, 0] = rng.uniform(0.1, 0.9)
            centers[idx, 1] = rng.uniform(0.1, 0.9)
            idx += 1
            
        # Add small noise to break symmetry
        centers += rng.normal(0, 0.005, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        configs.append(centers)
        
    return configs

def force_relax(centers, radii, steps=1500):
    """Vectorized force-directed relaxation to resolve overlaps and pack tightly."""
    n = centers.shape[0]
    for _ in range(steps):
        forces = np.zeros_like(centers)
        
        # Repulsion between circles
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        np.fill_diagonal(dists, np.inf)
        
        target_d = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlap = np.maximum(0, target_d - dists)
        safe_dists = np.where(dists < 1e-9, 1e-9, dists)
        repulsion = overlap / safe_dists
        forces += np.sum(repulsion[:, :, np.newaxis] * diffs / safe_dists[:, :, np.newaxis], axis=1)
        
        # Boundary repulsion
        for i in range(n):
            r = radii[i]
            cx, cy = centers[i]
            if cx < r + 0.005:
                forces[i, 0] += (r + 0.005 - cx) * 500.0
            if cx > 1.0 - r - 0.005:
                forces[i, 0] -= (cx - (1.0 - r - 0.005)) * 500.0
            if cy < r + 0.005:
                forces[i, 1] += (r + 0.005 - cy) * 500.0
            if cy > 1.0 - r - 0.005:
                forces[i, 1] -= (cy - (1.0 - r - 0.005)) * 500.0
                
        # Update positions
        centers += 0.002 * forces
        centers = np.clip(centers, 0.0, 1.0)
        
        # Gradually grow radii to explore tighter packings
        radii *= 1.00015
        
    return centers, radii

def compute_constraints(x, n):
    """Computes all boundary and non-overlap constraint values."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    cons = []
    # Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    cons.extend(cx - cr)
    cons.extend(1.0 - cx - cr)
    cons.extend(cy - cr)
    cons.extend(1.0 - cy - cr)
    
    # Overlap: dist^2 >= (r_i + r_j)^2
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    sum_r = cr[:, np.newaxis] + cr[np.newaxis, :]
    sum_r_sq = sum_r**2
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cons.extend(dist_sq[mask] - sum_r_sq[mask])
    
    return np.array(cons)

def constraint_fun(x):
    """Wrapper for constraint function to avoid lambda/closure issues."""
    return compute_constraints(x, 26)

def objective_func(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def repair_solution(centers, radii, n):
    """Deterministic repair to guarantee strict feasibility."""
    radii = radii.copy()
    for _ in range(30):
        changed = False
        # Clamp to boundaries
        for i in range(n):
            x, y = centers[i]
            mx = min(x, 1.0 - x, y, 1.0 - y)
            if radii[i] > mx:
                radii[i] = mx
                changed = True
                
        # Resolve overlaps
        for i in range(n):
            for j in range(i + 1, n):
                d = math.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if d < radii[i] + radii[j] - 1e-12:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
