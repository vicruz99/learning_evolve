import numpy as np
from scipy.optimize import minimize, linprog
import math

def get_initial_hex_centers(n=26):
    """Generates a structured hexagonal initialization (6-5-6-5-4 pattern)."""
    counts = [6, 5, 6, 5, 4]
    centers = []
    r = 0.1
    dy = math.sqrt(3) * r
    y = r
    for row_idx, count in enumerate(counts):
        dx = 2.0 * r
        # Center rows horizontally
        x_start = (1.0 - (count - 1) * dx) / 2.0
        if row_idx % 2 == 1:
            x_start += dx / 2.0
        for col in range(count):
            centers.append([x_start + col * dx, y])
        y += dy
    pts = np.array(centers[:n])
    # Normalize to fit comfortably within [0.05, 0.95]
    min_p, max_p = pts.min(axis=0), pts.max(axis=0)
    pts = (pts - min_p) / (max_p - min_p + 1e-12) * 0.9 + 0.05
    return pts, np.full(n, 0.08)

def run_physics(centers, radii):
    """Force-directed relaxation with slow radius growth to find a dense packing."""
    n = centers.shape[0]
    dt = 0.005
    rep_k = 50.0
    bnd_k = 100.0
    
    for step in range(8000):
        # Slowly grow radii to push circles apart
        radii *= (1.0 + 0.00002 * np.exp(-step / 2000))
        forces = np.zeros_like(centers)
        
        # Vectorized inter-circle repulsion
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-10)
        min_dists = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlap = np.maximum(0, min_dists - dists)
        dirs = diffs / (dists[:, :, np.newaxis] + 1e-10)
        forces = np.sum(overlap[:, :, np.newaxis] * dirs * rep_k, axis=1)
        
        # Boundary repulsion
        for i in range(n):
            r = radii[i]
            if centers[i, 0] < r: forces[i, 0] += bnd_k * (r - centers[i, 0])
            if centers[i, 0] > 1 - r: forces[i, 0] -= bnd_k * (centers[i, 0] - (1 - r))
            if centers[i, 1] < r: forces[i, 1] += bnd_k * (r - centers[i, 1])
            if centers[i, 1] > 1 - r: forces[i, 1] -= bnd_k * (centers[i, 1] - (1 - r))
            
        centers += dt * forces
        centers = np.clip(centers, 0.0, 1.0)
    return centers, radii

def objective_func(v):
    """Objective: minimize negative sum of radii."""
    n = 26
    return -np.sum(v[2*n:])

def compute_constraints(v):
    """Computes boundary and non-overlap constraints (must be >= 0)."""
    n = 26
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    cons = []
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    # Non-overlap constraints
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    sum_r = r[:, np.newaxis] + r[np.newaxis, :]
    mask = np.triu_indices(n, k=1)
    cons.append(dist[mask] - sum_r[mask])
    # Non-negative radii
    cons.append(r)
    return np.concatenate(cons)

def get_optimal_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints: r_i <= min(dist to walls)
    for i in range(n):
        cx, cy = centers[i]
        b_val = min(cx, 1.0 - cx, cy, 1.0 - cy)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(b_val)
        
    res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                  bounds=[(0, None)] * n, method='highs')
    return res.x if res.success else np.full(n, 0.05)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        np.random.seed(seed)

        v0[2 * n:] = radii + np.random.normal(0, 0.001, n)
        v0[2 * n:] = np.maximum(v0[2 * n:], 0.01)
        v0[:2 * n] = np.clip(v0[:2 * n], 0.0, 1.0)
        
        try:
            res = minimize(objective_func, v0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': compute_constraints},
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            
            # Check strict feasibility
            if np.all(compute_constraints(res.x) >= -1e-6):
                curr_sum = np.sum(res.x[2 * n:])
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Fallback
    if best_v is None:
        best_v = np.concatenate([centers.flatten(), radii])
        
    centers_opt = best_v[:2 * n].reshape(n, 2)

    radii_opt = get_optimal_radii_lp(centers_opt)

                radii_opt[i] = mx
                changed = True

                    radii_opt[i] -= shrink
                    radii_opt[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii_opt = np.maximum(radii_opt, 0.0)
    return centers_opt, radii_opt, float(np.sum(radii_opt))
