import numpy as np
from scipy.optimize import minimize

N = 26

def get_max_radii(centers):
    """Computes the maximum valid radius for each circle given fixed centers."""
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Distance to boundaries: min(x, 1-x, y, 1-y)
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise Euclidean distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist = np.sqrt(dx**2 + dy**2)
    
    # Ignore self-distances
    np.fill_diagonal(dist, np.inf)
    min_dist = np.min(dist, axis=1)
    
    # Radius constrained by half the minimum distance to avoid overlap
    return np.minimum(r_bound, 0.5 * min_dist)

def obj_centers(centers_flat):
    """Objective for centers-only optimization: minimize negative sum of radii."""
    centers = centers_flat.reshape(-1, 2)
    return -np.sum(get_max_radii(centers))

def obj_full(vars_flat):
    """Objective for full optimization: minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def con_full(vars_flat):
    """Computes all boundary and non-overlap constraints (must be >= 0)."""
    cx = vars_flat[0::3]
    cy = vars_flat[1::3]
    cr = vars_flat[2::3]
    
    con = []
    # Boundary constraints
    con.append(cx - cr)
    con.append(1.0 - cx - cr)
    con.append(cy - cr)
    con.append(1.0 - cy - cr)
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dist = np.sqrt(dx**2 + dy**2)
    sr = cr[:, np.newaxis] + cr[np.newaxis, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    con.append(dist[mask] - sr[mask])
    
    return np.concatenate(con)

def get_bounds():
    """Returns variable bounds for x, y in [0,1] and r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

    Uses a two-stage approach: centers-only search followed by full constrained refinement.
    """

    best_sum = -1.0
    best_full_vars = None
    bounds = get_bounds()
    constraints = {'type': 'ineq', 'fun': con_full}

    configs = []
    
    # 1. Hexagonal lattice variations
    for r in [0.095, 0.100, 0.105, 0.110]:

            x = r if row % 2 == 0 else 2 * r
            while x < 1.0 - r and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r
            y += np.sqrt(3.0) * r
            row += 1
        configs.append(np.array(pts[:N]))
        
    # 2. Structured grid variations
    for spacing in [0.18, 0.19, 0.20]:
        pts = []
        y = spacing
        while y < 1.0 - spacing and len(pts) < N:
            x = spacing
            while x < 1.0 - spacing and len(pts) < N:
                pts.append([x, y])
                x += spacing
            y += spacing
        configs.append(np.array(pts[:N]))
        
    # 3. Random configurations
    for _ in range(15):
        configs.append(np.random.rand(N, 2) * 0.8 + 0.1)

    # Stage 1: Optimize centers only using derivative-free methods
    for cfg in configs:
        # Add tiny deterministic noise to break symmetry
        cfg_perturbed = cfg + np.random.normal(0, 1e-4, cfg.shape)

        try:
            res_c = minimize(obj_centers, cfg_perturbed.flatten(), method='Powell',

            centers_opt = cfg_perturbed

                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 1e-9)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
