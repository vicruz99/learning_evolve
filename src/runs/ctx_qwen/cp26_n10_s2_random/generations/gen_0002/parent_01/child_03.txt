import numpy as np
from scipy.optimize import minimize
import math

N = 26

def get_max_radii(centers):
    """Computes the maximum valid radius for each circle given fixed centers."""
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    
    # Distance to boundaries: min(x, 1-x, y, 1-y)
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    # Radius constrained by half the minimum distance to avoid overlap
    r_pair = 0.5 * min_dists
    
    # The actual radius is the limiting factor between boundary and neighbors
    return np.minimum(r_bound, r_pair)

def obj_centros(centers_flat):
    """Objective function for center optimization: minimize negative sum of radii."""
    centers = centers_flat.reshape(-1, 2)
    radii = get_max_radii(centers)
    return -np.sum(radii)

def run_force_relaxation(centers, radii, steps=3500):
    """Physical simulation to pack circles tightly and grow radii."""
    n = len(centers)
    centers = centers.copy().astype(float)
    radii = radii.copy().astype(float)
    velocities = np.zeros_like(centers)
    
    k_rep = 1200.0      # Repulsion stiffness
    k_bnd = 2500.0      # Boundary stiffness
    damping = 0.65      # Velocity damping
    growth_rate = 1.00006 # Radius growth factor when stable
    
    for _ in range(steps):
        forces = np.zeros_like(centers)
        
        # Pairwise repulsion
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        min_req = radii[:, np.newaxis] + radii[np.newaxis, :]
        
        overlap = np.maximum(0.0, min_req - dists)
        dirs = diff / dists[:, :, np.newaxis]
        forces += np.sum(overlap[:, :, np.newaxis] * k_rep * dirs, axis=1)
        
        # Boundary repulsion
        for i in range(n):
            r = radii[i]
            if centers[i, 0] < r: forces[i, 0] += k_bnd * (r - centers[i, 0])
            if centers[i, 0] > 1 - r: forces[i, 0] -= k_bnd * (centers[i, 0] - (1 - r))
            if centers[i, 1] < r: forces[i, 1] += k_bnd * (r - centers[i, 1])
            if centers[i, 1] > 1 - r: forces[i, 1] -= k_bnd * (centers[i, 1] - (1 - r))
            
        # Update dynamics
        velocities = damping * velocities + forces * 0.00015
        centers += velocities
        centers = np.clip(centers, 1e-5, 1 - 1e-5)
        
        # Grow radii if packing is stable (minimal overlaps)
        if np.max(overlap) < 1e-6:
            radii *= growth_rate
            
    return centers, radii

def generate_hex_init(seed):
    """Generates a perturbed hexagonal lattice initial configuration."""
    np.random.seed(seed)
    centers = []
    r0 = 0.095
    y = r0
    row = 0
    while len(centers) < N:
        x = r0 if row % 2 == 0 else 2 * r0
        while x <= 1 - r0 and len(centers) < N:
            centers.append([x, y])
            x += 2 * r0
        y += math.sqrt(3) * r0
        row += 1
    centers = np.array(centers[:N])
    # Add small random noise to break symmetry
    centers += np.random.normal(0, 0.004, centers.shape)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize sum of radii."""
    best_sum = -1.0
    best_centers = None
    
    # Multi-start optimization with diverse initial perturbations
    for seed in range(12):
        c0 = generate_hex_init(seed)
        r0 = np.full(N, 0.045)
        
        # Stage 1: Force-directed relaxation to find a dense local packing
        c_rel, r_rel = run_force_relaxation(c0, r0, steps=3000)
        
        # Stage 2: Powell optimization on centers to maximize sum of radii
        x0 = c_rel.flatten()
        try:
            res = minimize(obj_centros, x0, method='Powell', 
                           options={'maxiter': 2500, 'ftol': 1e-13, 'disp': False})
            c_opt = res.x.reshape(-1, 2)
            r_opt = get_max_radii(c_opt)
            s = np.sum(r_opt)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
        except Exception:
            pass
            
    # Fallback if all optimizations fail
    if best_centers is None:
        best_centers = generate_hex_init(0)
        
    # Final exact radius computation and safety shrink to guarantee validation tolerance
    final_radii = get_max_radii(best_centers)
    final_radii *= 0.9999999  # Tiny shrink to absorb floating-point edge cases
    final_sum = float(np.sum(final_radii))
    
    return best_centers, final_radii, final_sum
