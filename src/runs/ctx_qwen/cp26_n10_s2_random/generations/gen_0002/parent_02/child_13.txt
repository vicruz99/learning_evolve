import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution
import math

N = 26

# Precompute LP structure at module level for maximum evaluation speed
A_ub_rows = []
pair_indices = []

# Boundary constraints: 4 per circle (x, 1-x, y, 1-y)
for i in range(N):
    for _ in range(4):
        row = np.zeros(N)
        row[i] = 1.0
        A_ub_rows.append(row)

# Pairwise constraints: r_i + r_j <= dist(i, j)
for i in range(N):
    for j in range(i + 1, N):
        row = np.zeros(N)
        row[i] = 1.0
        row[j] = 1.0
        A_ub_rows.append(row)
        pair_indices.append((i, j))

A_ub = np.array(A_ub_rows)
b_ub = np.zeros(len(A_ub_rows))
pair_start = 4 * N

def get_max_radii_lp_fast(centers):
    """Computes optimal radii for fixed centers using a highly optimized LP."""
    # Update boundary constraints
    for i in range(N):
        x, y = centers[i]
        b_ub[4*i] = x
        b_ub[4*i+1] = 1.0 - x
        b_ub[4*i+2] = y
        b_ub[4*i+3] = 1.0 - y
        
    # Update pairwise distance constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    idx = pair_start
    for i, j in pair_indices:
        b_ub[idx] = dists[i, j]
        idx += 1
        
    c_obj = -np.ones(N)
    bounds = [(0, None)] * N
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    # Fallback heuristic if LP fails
    r = np.zeros(N)
    for i in range(N):
        r[i] = min(centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1])
    return r

def objective_centers(centers_flat):
    """Objective: minimize negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    radii = get_max_radii_lp_fast(centers)
    return -np.sum(radii)

def greedy_init(n):
    """Generates initial centers using a greedy maximin placement."""
    centers = []
    xs = np.linspace(0, 1, 60)
    ys = np.linspace(0, 1, 60)
    xv, yv = np.meshgrid(xs, ys)
    grid = np.column_stack((xv.ravel(), yv.ravel()))
    
    for _ in range(n):
        scores = np.full(len(grid), 1.0)
        for cx, cy in centers:
            d = np.sqrt((grid[:,0]-cx)**2 + (grid[:,1]-cy)**2)
            scores = np.minimum(scores, d)
        bd = np.minimum(np.minimum(grid[:,0], 1-grid[:,0]), np.minimum(grid[:,1], 1-grid[:,1]))
        scores = np.minimum(scores, bd)
        idx = np.argmax(scores)
        centers.append(grid[idx])
    return np.array(centers)

def hex_init(n, r=0.1, noise=0.0):
    """Generates initial centers using a hexagonal lattice pattern."""
    centers = []
    y = r
    row = 0
    while len(centers) < n:
        x = r if row % 2 == 0 else 2*r
        while x + r <= 1.0 and len(centers) < n:
            centers.append([x, y])
            x += 2*r
        y += np.sqrt(3)*r
        row += 1
    centers = np.array(centers[:n])
    if noise > 0:
        centers += np.random.normal(0, noise, centers.shape)
        centers = np.clip(centers, 0.0, 1.0)
    return centers

def run_packing():
    """Main packing routine maximizing sum of radii for 26 circles."""
    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    
    # Diverse initial configurations
    inits = [
        greedy_init(N),
        hex_init(N, r=0.1, noise=0.0),
        hex_init(N, r=0.1, noise=0.02),
        hex_init(N, r=0.09, noise=0.05),
        np.random.uniform(0.1, 0.9, (N, 2)),
        np.random.uniform(0.05, 0.95, (N, 2)),
    ]
    
    # Phase 1: Global exploration via Differential Evolution
    try:
        bounds_c = [(0.0, 1.0)] * (2 * N)
        res_de = differential_evolution(
            objective_centers, bounds_c, seed=42, 
            maxiter=80, popsize=20, tol=1e-7, 
            mutation=(0.5, 1.0), recombination=0.9
        )
        c_de = res_de.x.reshape(N, 2)
        s_de = -res_de.fun
        if s_de > best_sum:
            best_sum = s_de
            best_centers = c_de
    except Exception:
        pass

    # Phase 2: Local refinement from multiple starts
    for init_c in inits:
        try:
            res = minimize(objective_centers, init_c.flatten(), method='Powell', 
                           options={'maxiter': 3000, 'ftol': 1e-9, 'xtol': 1e-9})
            c_opt = res.x.reshape(N, 2)
            s_opt = -res.fun
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt
        except Exception:
            pass
            
    # Phase 3: Refine the best found configuration further
    if best_centers is not None:
        try:
            res2 = minimize(objective_centers, best_centers.flatten(), method='Powell',
                            options={'maxiter': 3000, 'ftol': 1e-9, 'xtol': 1e-9})
            c_opt2 = res2.x.reshape(N, 2)
            s_opt2 = -res2.fun
            if s_opt2 > best_sum:
                best_sum = s_opt2
                best_centers = c_opt2
        except Exception:
            pass

    # Fallback safety
    if best_centers is None:
        best_centers = greedy_init(N)

    # Phase 4: Final LP for exact radii and strict validation repair
    radii = get_max_radii_lp_fast(best_centers)
    
    # Enforce boundaries strictly
    for i in range(N):
        x, y = best_centers[i]
        r = radii[i]
        best_centers[i, 0] = np.clip(x, r, 1.0 - r)
        best_centers[i, 1] = np.clip(y, r, 1.0 - y)
        
    # Resolve microscopic overlaps from numerical precision
    for _ in range(10):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(best_centers[i] - best_centers[j])
                if d < radii[i] + radii[j] - 1e-10:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return best_centers, radii, final_sum
