import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def compute_lp_radii(centers):
    """
    Given fixed centers, solves an LP to find radii that maximize the sum of radii
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Objective: maximize sum(r) <=> minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # 1. Pairwise non-overlap: r_i + r_j <= dist(i, j)
    # 2. Boundary containment: r_i <= x_i, 1-x_i, y_i, 1-y_i
    num_pairwise = (n * (n - 1)) // 2
    num_bound = 4 * n
    A_ub = np.zeros((num_pairwise + num_bound, n))
    b_ub = np.zeros(num_pairwise + num_bound)
    
    idx = 0
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.hypot(x[i] - x[j], y[i] - y[j])
            idx += 1
            
    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = x[i]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x[i]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y[i]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y[i]; idx += 1
        
    # Solve LP
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0.0, None), method='highs')
    
    if res.success:
        return res.x, -res.fun
    
    # Fallback direct calculation if LP fails
    r = np.full(n, 1.0)
    for i in range(n):
        r[i] = min(x[i], 1.0 - x[i], y[i], 1.0 - y[i])
        for j in range(n):
            if i != j:
                d = np.hypot(x[i] - x[j], y[i] - y[j])
                if d < r[i] * 2.0:
                    r[i] = d * 0.5
    return r, np.sum(r)

def objective_func(p):
    """Objective function for center optimization: negative sum of optimal radii."""
    centers = p.reshape(-1, 2)
    _, s = compute_lp_radii(centers)
    return -s

def run_packing() -> tuple:
    np.random.seed(42)
    n = N_CIRCLES
    bounds = [(0.0, 1.0)] * (2 * n)
    
    best_sum = 0.0
    best_p = None
    
    # Generate diverse initial configurations
    configs = []
    
    # 1. Hexagonal lattice
    pts = []
    r0 = 0.108
    y = r0
    row = 0
    while len(pts) < n:
        x = r0 if row % 2 == 0 else 2 * r0
        while x + r0 <= 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
    configs.append(np.array(pts[:n]))
    
    # 2. Perturbed hexagonal
    for _ in range(5):
        configs.append(configs[0] + np.random.normal(0, 0.025, (n, 2)))
        
    # 3. Random dense
    for _ in range(3):
        configs.append(np.random.uniform(0.15, 0.85, (n, 2)))
        
    # 4. Structured grid
    pts_g = []
    for i in range(5):
        for j in range(6):
            if len(pts_g) < n:
                pts_g.append([0.12 + j * 0.14, 0.12 + i * 0.19])
    configs.append(np.array(pts_g[:n]))
    
    best_p = configs[0].flatten()
    
    # Basin-hopping style optimization
    for trial in range(70):
        if trial < len(configs):
            p0 = configs[trial].flatten()
        else:
            # Perturb best solution found so far to escape local minima
            scale = 0.04 * (0.88 ** (trial - len(configs)))
            p0 = best_p + np.random.normal(0, scale, 2 * n)
            
        # Ensure centers stay strictly inside the square
        p0 = np.clip(p0, 1e-4, 1.0 - 1e-4)
        
        try:
            # L-BFGS-B handles box constraints efficiently
            res = minimize(objective_func, p0, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 1500, 'ftol': 1e-12})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_p = res.x.copy()
        except Exception:
            continue
            
    best_centers = best_p.reshape(-1, 2)
    radii, final_sum = compute_lp_radii(best_centers)
    
    # Microscopic shrink to guarantee strict compliance with validator tolerance
    radii *= 0.9999999
    
    return best_centers, radii, float(np.sum(radii))
