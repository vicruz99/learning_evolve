import numpy as np
import scipy.optimize as opt

N = 26
iu, ju = np.triu_indices(N, k=1)

def objective(p):
    """Objective: Minimize negative sum of radii."""
    return -np.sum(p[2::3])

def constraint_func(p):
    """Computes all inequality constraints (must be >= 0)."""
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Boundary constraints: x>=r, 1-x>=r, y>=r, 1-y>=r
    c_bnd = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[iu] - x[ju]
    dy = y[iu] - y[ju]
    dr = r[iu] + r[ju]
    c_ovl = dx**2 + dy**2 - dr**2
    
    return np.concatenate([c_bnd, c_ovl])

def make_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def hex_init(rows, r=0.095, noise=0.0):
    """Generates initial parameters using a hexagonal lattice pattern."""
    p = np.zeros(N * 3)
    idx = 0
    y = r
    for row, count in enumerate(rows):
        start_x = r if row % 2 == 0 else 2 * r
        x = start_x
        for _ in range(count):
            if idx >= N:
                break
            p[3 * idx] = x
            p[3 * idx + 1] = y
            p[3 * idx + 2] = r
            idx += 1
            x += 2 * r
        y += np.sqrt(3) * r
        
    if noise > 0:
        rng = np.random.default_rng(sum(rows) + int(noise * 1000))
        p[0::3] += rng.normal(0, noise, N)
        p[1::3] += rng.normal(0, noise, N)
        p[0::3] = np.clip(p[0::3], 0.02, 0.98)
        p[1::3] = np.clip(p[1::3], 0.02, 0.98)
    return p

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = make_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_p = None
    best_sum = -1.0
    
    # Diverse hexagonal patterns summing to 26
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 5, 5, 6], 
        [4, 6, 6, 6, 4], [5, 5, 6, 5, 5], [6, 6, 5, 5, 4],
        [7, 5, 6, 5, 3], [5, 7, 5, 6, 3], [4, 5, 7, 5, 5]
    ]
    
    inits = []
    for pat in patterns:
        for n in [0.0, 0.005, 0.01, 0.02, 0.03]:
            inits.append(hex_init(pat, r=0.095, noise=n))
            
    # Random dense starts
    for seed in range(20):
        rng = np.random.default_rng(seed)
        p = np.zeros(N * 3)
        p[0::3] = rng.uniform(0.15, 0.85, N)
        p[1::3] = rng.uniform(0.15, 0.85, N)
        p[2::3] = 0.085
        inits.append(p)
        
    # Phase 1: Multi-start optimization
    for p0 in inits:
        try:
            res = opt.minimize(
                objective, p0, method='SLSQP', bounds=bounds,
                constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False}
            )
            if res.success:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            pass
            
    if best_p is None:
        best_p = hex_init([6, 5, 6, 5, 4], 0.09, 0.01)
        
    # Phase 2: Perturbation & Expansion
    for _ in range(50):
        # Perturb centers to escape local minima
        p_trial = best_p.copy()
        p_trial[:2*N] += np.random.normal(0, 0.003, 2*N)
        p_trial[:2*N] = np.clip(p_trial[:2*N], 0.05, 0.95)
        
        try:
            res = opt.minimize(
                objective, p_trial, method='SLSQP', bounds=bounds,
                constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13}
            )
            if res.success and -res.fun > best_sum:
                best_sum = -res.fun
                best_p = res.x.copy()
        except Exception:
            pass
            
        # Try to expand radii to find tighter packings
        p_expand = best_p.copy()
        p_expand[2::3] *= 1.005
        p_expand[2::3] = np.clip(p_expand[2::3], 1e-6, 0.49)
        
        try:
            res = opt.minimize(
                objective, p_expand, method='SLSQP', bounds=bounds,
                constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13}
            )
            if res.success and -res.fun > best_sum:
                best_sum = -res.fun
                best_p = res.x.copy()
        except Exception:
            pass

    # Extract and prepare final solution
    centers = best_p[:2*N].reshape(N, 2)
    radii = best_p[2*N:].copy()
    
    # Strict repair phase
    radii = np.maximum(radii, 1e-9)
    centers = np.clip(centers, radii[:, np.newaxis], 1.0 - radii[:, np.newaxis])
    
    for _ in range(20):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < radii[i] + radii[j] - 1e-12:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
