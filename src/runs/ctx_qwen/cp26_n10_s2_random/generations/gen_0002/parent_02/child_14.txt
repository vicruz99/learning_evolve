import numpy as np
from scipy.optimize import minimize

N = 26

def get_sum_radii_from_centers(centers):
    """Given fixed centers, compute the maximum valid radii and their sum."""
    x, y = centers[:, 0], centers[:, 1]
    # Distance to boundaries
    rb = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    min_dist = np.min(dist, axis=1)
    
    # Radius constrained by half the minimum distance to avoid overlap
    rp = 0.5 * min_dist
    
    # The actual radius is the limiting factor between boundary and neighbors
    radii = np.minimum(rb, rp)
    return radii, np.sum(radii)

def obj_centers(x_flat):
    """Objective for derivative-free optimization on centers only."""
    centers = x_flat.reshape(N, 2)
    _, s = get_sum_radii_from_centers(centers)
    return -s

def obj_slq(p):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(p[2::3])

def slsqp_constraints(p):
    """Computes all inequality constraint values (must be >= 0)."""
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Boundary constraints
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    
    # Pairwise non-overlap constraints
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    mask = np.triu(np.ones((N, N)), k=1).astype(bool)
    c5 = dist[mask] - dr[mask]
    
    return np.concatenate([c1, c2, c3, c4, c5])

def get_hex_init(scale=1.0, angle=0.0):
    """Generates an initial hexagonal lattice configuration, rotated and scaled."""
    pts = []
    r = 0.095
    y = r
    row = 0
    while len(pts) < N:
        x = r if row % 2 == 0 else 2 * r
        while x <= 1.0 - r and len(pts) < N:
            pts.append([x, y])
            x += 2 * r
        y += r * np.sqrt(3)
        row += 1
    pts = np.array(pts[:N])
    
    # Rotate and scale around center
    c = np.mean(pts, axis=0)
    pts = pts - c
    ca, sa = np.cos(angle), np.sin(angle)
    rot = np.array([[ca, -sa], [sa, ca]])
    pts = (rot @ pts.T).T * scale
    
    # Normalize to fit inside [0,1] with margins
    pts = pts - np.min(pts, axis=0)
    span = np.max(pts, axis=0) - np.min(pts, axis=0)
    pts = pts / span * 0.85 + 0.075
    return np.clip(pts, 0.02, 0.98)

        p0[2::3] = r0
