import numpy as np
from scipy.optimize import differential_evolution, minimize

N = 26
IU, JU = np.triu_indices(N, k=1)

def get_radii_for_centers(centers):
    """Computes the maximum valid radius for each circle given fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    return np.minimum(r_bound, 0.5 * min_dists)

def objective_centers(x):
    """Objective for DE: minimize negative sum of radii."""
    centers = x.reshape(N, 2)
    radii = get_radii_for_centers(centers)
    return -np.sum(radii)

def obj_full(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_boundary(v):
    """Boundary constraints: circles must stay inside [0,1]x[0,1]."""
    cx = v[:N]
    cy = v[N:2*N]
    cr = v[2*N:]
    return np.concatenate([
        cx - cr, 1.0 - cx - cr,
        cy - cr, 1.0 - cy - cr
    ])

def cons_overlap(v):
    """Non-overlap constraints: distance between centers >= sum of radii."""
    cx = v[:N]
    cy = v[N:2*N]
    cr = v[2*N:]
    dx = cx[IU] - cx[JU]
    dy = cy[IU] - cy[JU]
    dist = np.sqrt(dx**2 + dy**2)
    sr = cr[IU] + cr[JU]
    return dist - sr

def run_packing() -> tuple:
    np.random.seed(42)
    
    # Phase 1: Global optimization of centers using Differential Evolution
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    
    best_x = None
    best_val = np.inf
    
    for seed in range(3):
        np.random.seed(seed)
        res = differential_evolution(
            objective_centers,
            bounds_centers,
            seed=seed,
            maxiter=1200,
            popsize=25,
            mutation=(0.5, 1.5),
            recombination=0.9,
            tol=1e-10,
            polish=False
        )
        if res.fun < best_val:
            best_val = res.fun
            best_x = res.x.copy()
            
    centers_best = best_x.reshape(N, 2)
    radii_best = get_radii_for_centers(centers_best)
    
    # Phase 2: Local refinement using SLSQP on (centers, radii)
    x0 = np.concatenate([centers_best.flatten(), radii_best])
    bounds_full = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
    constraints = [
        {'type': 'ineq', 'fun': cons_boundary},
        {'type': 'ineq', 'fun': cons_overlap}
    ]
    
    best_sum = -best_val
    best_centers_out = centers_best.copy()
    best_radii_out = radii_best.copy()
    
    # Multi-start local optimization to escape local minima
    for restart in range(15):
        x_start = x0.copy()
        if restart > 0:
            x_start += np.random.normal(0, 0.002, x_start.shape)
            x_start = np.clip(x_start, 0.0, 1.0)
            
        try:
            res = minimize(
                obj_full,
                x_start,
                method='SLSQP',
                bounds=bounds_full,
                constraints=constraints,
                options={'maxiter': 3000, 'ftol': 1e-12}
            )
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers_out = res.x[:2*N].reshape(N, 2)
                best_radii_out = res.x[2*N:].copy()
        except Exception:
            continue
            
    # Phase 3: Numerical repair to guarantee validation passes
    radii = best_radii_out.copy()
    centers = best_centers_out.copy()
    
    # Resolve overlaps by proportional shrinking
    for _ in range(20):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < radii[i] + radii[j] - 1e-12:
                    overlap = radii[i] + radii[j] - d
                    total = radii[i] + radii[j]
                    if total > 1e-12:
                        s1 = overlap * radii[i] / total
                        s2 = overlap * radii[j] / total
                    else:
                        s1 = s2 = overlap / 2.0
                    radii[i] -= s1
                    radii[j] -= s2
                    changed = True
        if not changed:
            break
            
    # Clamp to boundaries
    for i in range(N):
        r_max = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
        if radii[i] > r_max:
            radii[i] = r_max
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
