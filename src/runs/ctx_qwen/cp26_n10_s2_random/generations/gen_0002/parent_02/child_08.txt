import numpy as np
from scipy.optimize import differential_evolution, minimize

N_CIRCLES = 26

def compute_max_radii(centers):
    """
    Given fixed centers, compute the maximum valid radius for each circle.
    r_i = min(distance to boundaries, 0.5 * distance to nearest neighbor)
    """
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Distance to square boundaries
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    
    # Ignore self-distances
    np.fill_diagonal(dist, np.inf)
    min_dist = np.min(dist, axis=1)
    
    # Radius limited by half the distance to nearest neighbor
    r_pair = 0.5 * min_dist
    
    # Final radius is the tighter constraint
    return np.minimum(r_bound, r_pair)

def objective_function(centers_flat):
    """
    Objective for optimizers: minimize negative sum of radii.
    Centers are flattened to 1D array.
    """
    centers = centers_flat.reshape(-1, 2)
    radii = compute_max_radii(centers)
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # Bounds for centers: keep them slightly inside to avoid degenerate zero-radius cases
    bounds = [(0.001, 0.999)] * (2 * N_CIRCLES)
    
    # Phase 1: Global exploration with Differential Evolution
    de_res = differential_evolution(
        objective_function,
        bounds,
        seed=42,
        maxiter=2000,
        popsize=40,
        mutation=(0.5, 1.2),
        recombination=0.9,
        tol=1e-9,
        polish=False
    )
    
    best_centers = de_res.x.reshape(-1, 2)
    best_radii = compute_max_radii(best_centers)
    best_sum = np.sum(best_radii)
    
    # Phase 2: Local refinement with Powell (handles non-smooth objective well)
    res_local = minimize(
        objective_function,
        best_centers.flatten(),
        method='Powell',
        options={'maxiter': 10000, 'ftol': 1e-14, 'xtol': 1e-14}
    )
    if -res_local.fun > best_sum:
        best_centers = res_local.x.reshape(-1, 2)
        best_radii = compute_max_radii(best_centers)
        best_sum = np.sum(best_radii)
        
    # Phase 3: Multi-start perturbation to escape shallow local minima
    for seed in range(50):
        np.random.seed(seed)
        pert_centers = best_centers + np.random.normal(0, 0.002, (N_CIRCLES, 2))
        pert_centers = np.clip(pert_centers, 0.001, 0.999)
        
        res_pert = minimize(
            objective_function, 
            pert_centers.flatten(), 
            method='Powell', 
            options={'maxiter': 2000, 'ftol': 1e-12}
        )
        
        if -res_pert.fun > best_sum:
            best_centers = res_pert.x.reshape(-1, 2)
            best_radii = compute_max_radii(best_centers)
            best_sum = np.sum(best_radii)
            
    # Final safety clipping (should already be valid by construction)
    best_centers = np.clip(best_centers, 0.0, 1.0)
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_sum)
