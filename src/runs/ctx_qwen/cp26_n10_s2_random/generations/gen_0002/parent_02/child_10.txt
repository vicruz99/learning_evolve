import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_constraints(p):
    """Computes all inequality constraints (must be >= 0)."""
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    c_bound = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dist = np.sqrt(dx**2 + dy**2)
    c_pair = dist - (r[i_idx] + r[j_idx])
    
    return np.concatenate([c_bound, c_pair])

def objective(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[2::3])

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Distance to boundaries for each circle
    bounds_vals = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    bounds_vals = np.minimum(bounds_vals, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Pairwise distances
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    
    A_ub = []
    b_ub = []
    
    # Constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Solve LP
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0.0, None)] * n, method='highs')
    if res.success:
        return res.x
    return np.full(n, 0.01)

def generate_initial_configs(num_configs):

    configs = []
    
    # 1. Hexagonal patterns with varying densities
    for r_init in np.linspace(0.075, 0.115, 8):

        y = r_init
        row = 0
        idx = 0
        while idx < N:
            x = r_init if row % 2 == 0 else 2 * r_init

                pts.append([x, y])
                x += 2 * r_init
                idx += 1
            y += np.sqrt(3) * r_init
            row += 1
        configs.append((np.array(pts[:N]), np.full(N, r_init * 0.85)))

        configs.append((centers, radii))

                    pts.append([x, y])
                col += 1
            row += 1
        configs.append((np.array(pts[:N]), np.full(N, 0.06)))
        
    return configs[:num_configs]
