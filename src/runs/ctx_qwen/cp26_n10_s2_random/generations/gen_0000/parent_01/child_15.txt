    np.fill_diagonal(diff, 0)
    
    # Penalty for overlap (only if diff > 0)
    # Using a soft threshold or just max(0, diff)
    overlap = np.maximum(0, diff)
    penalty = np.sum(overlap**2)
    
    # Boundary penalty
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    pen_boundary = 0.0
    
    # Left: x < r
    diff_left = r - x
    mask_left = diff_left > 0
    if np.any(mask_left):
        pen_boundary += np.sum(diff_left[mask_left]**2)
        
    # Right: x > 1 - r  => x + r > 1 => r > 1 - x => r - (1-x) > 0
    diff_right = r - (1 - x)
    mask_right = diff_right > 0
    if np.any(mask_right):
        pen_boundary += np.sum(diff_right[mask_right]**2)

    mask_bottom = diff_bottom > 0

    mask_top = diff_top > 0
    if np.any(mask_top):
        pen_boundary += np.sum(diff_top[mask_top]**2)

    # We want to maximize sum(radii).
