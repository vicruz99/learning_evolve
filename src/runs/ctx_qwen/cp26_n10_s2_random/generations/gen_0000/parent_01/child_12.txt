def run_packing() -> tuple:

    centers_init, radii_init = get_initial_guess(n)

    x0 = np.zeros(3 * n)

        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]

    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = []

        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # Constraints
    constraints = []
    
    # 1. Boundary constraints:
    # x_i >= r_i  => r_i - x_i <= 0
    # x_i <= 1 - r_i => r_i + x_i - 1 <= 0
    # y_i >= r_i
    # y_i <= 1 - r_i

        idx_x = 3*i
        idx_y = 3*i+1
        idx_r = 3*i+2
        
        # r_i - x_i <= 0
        def make_bc_lower(x, idx_r, idx_x):
            return x[idx_r] - x[idx_x]
        constraints.append({'type': 'ineq', 'fun': lambda x, ir=idx_r, ix=idx_x: x[ix] - x[ir]}) 
        # Wait, inequality in scipy is func(x) >= 0.
        # So we need x[ix] - x[ir] >= 0 => x >= r.
        
        # x_i + r_i <= 1 => 1 - x_i - r_i >= 0
        def make_bc_upper_x(x, ix, ir):
            return 1.0 - x[ix] - x[ir]
        constraints.append({'type': 'ineq', 'fun': make_bc_upper_x, 'args': (idx_x, idx_r)})
        
        # y constraints
        def make_bc_lower_y(x, iy, ir):
            return x[iy] - x[ir]
        constraints.append({'type': 'ineq', 'fun': make_bc_lower_y, 'args': (idx_y, idx_r)})
        
        def make_bc_upper_y(x, iy, ir):
            return 1.0 - x[iy] - x[ir]
        constraints.append({'type': 'ineq', 'fun': make_bc_upper_y, 'args': (idx_y, idx_r)})
        
    # 2. Non-overlap constraints:
    # dist(i, j) >= r_i + r_j
    # sqrt((xi-xj)^2 + (yi-yj)^2) - ri - rj >= 0
    
    for i in range(n):

            ix, iy, ir = 3*i, 3*i+1, 3*i+2

                dx = x[ix] - x[jx]
                dy = x[iy] - x[jy]
                dist = np.sqrt(dx*dx + dy*dy)
                return dist - (x[ir] + x[jr])
            
            constraints.append({'type': 'ineq', 'fun': make_overlap, 'args': (ix, iy, ir, jx, jy, jr)})

        s = 0.0
        for i in range(n):
            s += x[3*i+2]
        return -s

    # Using SLSQP which handles bounds and constraints
    res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,

        return final_centers, final_radii, sum_r
    else:

        # (Should not happen with good guess)
        return centers_init, radii_init, np.sum(radii_init)
