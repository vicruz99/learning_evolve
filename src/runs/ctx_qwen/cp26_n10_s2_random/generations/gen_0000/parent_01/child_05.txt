import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    N = 26
    n_restarts = 50
    n_iterations = 400
    best_centers = None
    max_radius_sum = 0.0

    def solve_lp_radii(centers: np.ndarray) -> tuple[np.ndarray, float]:
        """Solves the LP to maximize sum of radii for fixed centers."""
        n = centers.shape[0]
        c_obj = -np.ones(n) # Minimize negative sum

        A_ub = []
        b_ub = []
        
        # Pairwise constraints: r_i + r_j <= dist_ij
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                row = [0.0] * n
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dist)

        # Boundary constraints: r_i <= dist_to_boundary
        for i in range(n):
            x, y = centers[i]
            max_r = min(x, 1 - x, y, 1 - y)
            # Ensure max_r is non-negative to avoid infeasible bounds
            if max_r < 0: 
                max_r = 0.0 
            row = [0.0] * n
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(max_r)

        bounds = [(0, None) for _ in range(n)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                radii = res.x
                return radii, -res.fun
            else:
                return None, 0.0
        except Exception:
            return None, 0.0

    # --- Phase 1: Force-directed center optimization ---
    for restart in range(n_restarts):
        centers = np.random.rand(N, 2)
        
        # Run force-directed relaxation
        for step in range(n_iterations):
            forces = np.zeros((N, 2))
            min_sep = 1.0
            
            for i in range(N):
                # Repulsion from other circles
                for j in range(i + 1, N):
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    if dist < 1e-6:
                        dist = 1e-6
                        diff = np.random.rand(2) * 1e-3
                    if dist < 0.2:
                        repulsion = (0.2 - dist) / (dist ** 2)
                        forces[i] += diff / dist * repulsion
                        forces[j] -= diff / dist * repulsion
                    min_sep = min(min_sep, dist)

                # Repulsion from boundaries (walls)
                x, y = centers[i]
                if x < 0.05: forces[i, 0] += (0.05 - x) * 10
                if x > 0.95: forces[i, 0] -= (x - 0.95) * 10
                if y < 0.05: forces[i, 1] += (0.05 - y) * 10
                if y > 0.95: forces[i, 1] -= (y - 0.95) * 10
            
            # Apply forces
            step_size = 0.01 * (1.0 - step / n_iterations)
            centers += forces * step_size
            
            # Clip to valid range
            centers = np.clip(centers, 0.01, 0.99)

        # --- Phase 2: LP Optimization for Radii ---
        radii, r_sum = solve_lp_radii(centers)
        if radii is not None and r_sum > max_radius_sum:
            max_radius_sum = r_sum
            best_centers = centers.copy()

    # Final validation and clean-up of the best configuration found
    radii, sum_radii = solve_lp_radii(best_centers)
    if radii is not None:
        return best_centers, radii, sum_radii
    else:
        # Fallback to equal circles if LP fails
        return best_centers, np.ones(26) * 0.05, 1.3

print(f"Sum of radii: {run_packing()[2]:.4f}")
