import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # --- Initialization ---
    # Create a grid of points to initialize centers
    # 6 columns, 5 rows = 30 points. We will use the first 26.
    # Spacing allows for potential expansion.
    x_coords = np.linspace(0.1, 0.9, 6)
    y_coords = np.linspace(0.1, 0.9, 5)
    
    centers_init = []
    idx = 0
    for y in y_coords:
        for x in x_coords:
            if idx < n:
                centers_init.append([x, y])
                idx += 1
            else:
                break
        if idx >= n:
            break
            
    centers_init = np.array(centers_init)
    radii_init = np.full(n, 0.05) # Small initial radius
    
    # Flatten variables: [x0...x25, y0...y25, r0...r25]
    x0 = np.concatenate([centers_init[:, 0], centers_init[:, 1], radii_init])
    
    # --- Bounds ---
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [
        (0, 1) for _ in range(n) # x
    ] + [
        (0, 1) for _ in range(n) # y
    ] + [
        (0, 0.5) for _ in range(n) # r
    ]
    
    # --- Constraints ---
    # 1. Boundary containment: x >= r, 1-x >= r, y >= r, 1-y >= r
    #    Implemented as: x - r >= 0, 1 - x - r >= 0, etc.
    
    def con_boundary_x_min(params):
        return params[:n] - params[2*n:] # x - r >= 0

    def con_boundary_x_max(params):
        return (1 - params[:n]) - params[2*n:] # 1 - x - r >= 0

    def con_boundary_y_min(params):
        return params[n:2*n] - params[2*n:] # y - r >= 0

    def con_boundary_y_max(params):
        return (1 - params[n:2*n]) - params[2*n:] # 1 - y - r >= 0

    # 2. Non-overlap: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    #    Implemented as: dist_sq - rad_sum_sq >= 0
    #    Since scipy handles inequality constraints as fun(x) >= 0
    
    def con_non_overlap(params):
        x = params[:n]
        y = params[n:2*n]
        r = params[2*n:]
        
        violations = []
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = (x[i] - x[j])**2 + (y[i] - y[j])**2
                rad_sum = r[i] + r[j]
                violations.append(dist_sq - rad_sum**2)
        return np.array(violations)

    # Define constraints for optimizer
    constraints = [
        {'type': 'ineq', 'fun': con_boundary_x_min},
        {'type': 'ineq', 'fun': con_boundary_x_max},
        {'type': 'ineq', 'fun': con_boundary_y_min},
        {'type': 'ineq', 'fun': con_boundary_y_max},
        {'type': 'ineq', 'fun': con_non_overlap}
    ]

    # --- Objective Function ---
    # Maximize sum(radii) => Minimize -sum(radii)
    def objective(params):
        return -np.sum(params[2*n:])

    # --- Optimization ---
    # Using SLSQP which handles non-linear constraints
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )

    if result.success:
        params_opt = result.x
    else:
        # Fallback to initial if optimization fails (unlikely with small init)
        params_opt = x0

    # Extract results
    centers_opt = np.column_stack((params_opt[:n], params_opt[n:2*n]))
    radii_opt = params_opt[2*n:]
    
    # Ensure non-negative radii (should be guaranteed by bounds)
    radii_opt = np.maximum(radii_opt, 0)
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
