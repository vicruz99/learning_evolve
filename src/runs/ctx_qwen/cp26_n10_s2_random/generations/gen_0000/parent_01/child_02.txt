def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def objective_func(vars, n):
    """
    Objective: Minimize -sum(radii)
    vars layout: [x1, y1, r1, x2, y2, r2, ...]
    """
    radii = vars[2::3]
    return -np.sum(radii)

def boundary_constraints(vars, n):
    """
    Constraints: r <= x <= 1-r, r <= y <= 1-r
    vars layout: [x1, y1, r1, ...]
    Returns array of constraint values (should be >= 0)
    """
    con = []
    for i in range(n):
        x = vars[3*i]
        y = vars[3*i + 1]
        r = vars[3*i + 2]
        # x - r >= 0
        con.append(x - r)
        # 1 - x - r >= 0 => 1 - x >= r => 1 - x - r >= 0
        con.append(1 - x - r)
        # y - r >= 0
        con.append(y - r)
        # 1 - y - r >= 0
        con.append(1 - y - r)
    return np.array(con)

def overlap_constraints(vars, n):
    """
    Constraints: dist(i, j) >= r_i + r_j
    vars layout: [x1, y1, r1, ...]
    Returns array of constraint values (should be >= 0)
    """
    con = []
    for i in range(n):
        xi = vars[3*i]
        yi = vars[3*i + 1]
        ri = vars[3*i + 2]
        for j in range(i + 1, n):
            xj = vars[3*j]
            yj = vars[3*j + 1]
            rj = vars[3*j + 2]
            # dist^2 >= (ri + rj)^2
            # To avoid sqrt in constraint function for better conditioning? 
            # Actually sqrt is fine, but squared is smooth.
            # However, constraint must be >= 0.
            # dist - (ri + rj) >= 0
            dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
            con.append(dist - (ri + rj))
    return np.array(con)
