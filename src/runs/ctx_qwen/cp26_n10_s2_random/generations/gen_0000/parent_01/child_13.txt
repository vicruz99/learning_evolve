import numpy as np
from scipy.optimize import minimize
import math


def compute_violations(centers, radii, n):
    """Compute total squared constraint violations."""
    v = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        v += max(0.0, r - x) ** 2
        v += max(0.0, r - (1.0 - x)) ** 2
        v += max(0.0, r - y) ** 2
        v += max(0.0, r - (1.0 - y)) ** 2
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = math.sqrt(dx * dx + dy * dy)
            v += max(0.0, radii[i] + radii[j] - d) ** 2
    return v


def full_objective(params, n, alpha):
    """Objective: negative sum of radii plus penalty for violations."""
    centers = params[:2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n:], 1e-12)
    return -np.sum(radii) + alpha * compute_violations(centers, radii, n)


def hexagonal_init(n):
    """Create hexagonal grid initialization."""
    centers = np.zeros((n, 2))
    idx = 0
    s = 0.19
    for row in range(7):
        y = 0.07 + row * s * math.sqrt(3) / 2.0
        if y > 0.93:
            break
        offset = (s / 2.0) if row % 2 == 1 else 0.0
        col = 0
        while idx < n:
            x = 0.07 + offset + col * s
            if x > 0.93:
                break
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
            col += 1
    while idx < n:
        centers[idx, 0] = 0.5
        centers[idx, 1] = 0.5
        idx += 1
    return centers


def grid_init(n):
    """Create uniform grid initialization."""
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(6):
        for col in range(5):
            if idx < n:
                centers[idx, 0] = 0.08 + col * 0.19
                centers[idx, 1] = 0.08 + row * 0.17
                idx += 1
    while idx < n:
        centers[idx, 0] = 0.5
        centers[idx, 1] = 0.5
        idx += 1
    return centers


def optimize_from(centers_init, radii_init, n):
    """Run multi-stage optimization from given initial state."""
    params = np.concatenate([centers_init.flatten(), radii_init])
    
    bounds = [(0.001, 0.999)] * (2 * n) + [(0.001, 0.5)] * n
    
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    
    for alpha in alphas:
        result = minimize(
            full_objective,
            params,
            args=(n, alpha),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 5000, 'ftol': 1e-16, 'gtol': 1e-10, 'disp': False}
        )
        params = result.x.copy()
    
    centers_opt = params[:2 * n].reshape(n, 2)
    radii_opt = np.maximum(params[2 * n:], 0.0)
    
    return centers_opt, radii_opt


def random_init(n, seed):
    """Create random initialization."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.1, 0.9, (n, 2))
    radii = np.full(n, 0.04)
    return centers, radii


def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Strategy 1: Hexagonal grid
    centers_h = hexagonal_init(n)
    radii_h = np.full(n, 0.04)
    c_opt, r_opt = optimize_from(centers_h, radii_h, n)
    s_opt = np.sum(r_opt)
    if s_opt > best_sum:
        best_sum = s_opt
        best_centers = c_opt.copy()
        best_radii = r_opt.copy()
    
    # Strategy 2: Uniform grid
    centers_g = grid_init(n)
    radii_g = np.full(n, 0.04)
    c_opt, r_opt = optimize_from(centers_g, radii_g, n)
    s_opt = np.sum(r_opt)
    if s_opt > best_sum:
        best_sum = s_opt
        best_centers = c_opt.copy()
        best_radii = r_opt.copy()
    
    # Strategy 3-12: Multiple random restarts
    for seed in range(10):
        centers_r, radii_r = random_init(n, seed)
        c_opt, r_opt = optimize_from(centers_r, radii_r, n)
        s_opt = np.sum(r_opt)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
    
    # Post-processing: slightly scale down radii to ensure strict feasibility
    # The validation allows 1e-12 tolerance, so we need minimal gap
    scale = 0.99999
    best_radii *= scale
    
    # Clip centers to be safely within bounds
    for i in range(n):
        r = best_radii[i]
        best_centers[i, 0] = max(r + 1e-10, min(1.0 - r - 1e-10, best_centers[i, 0]))
        best_centers[i, 1] = max(r + 1e-10, min(1.0 - r - 1e-10, best_centers[i, 1]))
    
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(final_sum)
