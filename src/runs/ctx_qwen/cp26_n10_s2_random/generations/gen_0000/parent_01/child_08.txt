import numpy as np
from scipy.optimize import minimize


def circle_overlap(x, i, j, n):
    cx1, cy1 = x[2 * i], x[2 * i + 1]
    cx2, cy2 = x[2 * j], x[2 * j + 1]
    r1, r2 = x[2 * n + i], x[2 * n + j]
    dist = np.sqrt((cx1 - cx2) ** 2 + (cy1 - cy2) ** 2)
    return dist - r1 - r2


def bnd_x_low(x, i, n):
    return x[2 * i] - x[2 * n + i]


def bnd_x_high(x, i, n):
    return 1 - x[2 * i] - x[2 * n + i]


def bnd_y_low(x, i, n):
    return x[2 * i + 1] - x[2 * n + i]


def bnd_y_high(x, i, n):
    return 1 - x[2 * i + 1] - x[2 * n + i]


def neg_sum_radii(x, n):
    return -np.sum(x[2 * n:])


def init_hexagonal(n, seed):
    rng = np.random.RandomState(seed)
    centers = []
    row_counts = [5, 5, 5, 5, 6]
    y_base = 1.0 / 6

    count = 0
    for row in range(5):
        y = (row + 1) * y_base
        offset = 0.06 if row % 2 == 1 else 0
        num_in_row = row_counts[row]
        x_base = 1.0 / (num_in_row + 1)

        for col in range(num_in_row):
            if count < n:
                x = (col + 1) * x_base + offset
                x += rng.uniform(-0.02, 0.02)
                y += rng.uniform(-0.02, 0.02)
                x = np.clip(x, 0.05, 0.95)
                y = np.clip(y, 0.05, 0.95)
                centers.append([x, y])
                count += 1

    centers = np.array(centers[:n])
    radii = np.ones(n) * 0.04
    return centers, radii


def init_strategic(n, seed):
    rng = np.random.RandomState(seed)
    centers = []
    radii_init = []

    # 4 large corner circles
    cr = 0.22 + rng.uniform(-0.02, 0.02)
    centers.append([cr, cr])
    radii_init.append(0.15)
    centers.append([1 - cr, cr])
    radii_init.append(0.15)
    centers.append([cr, 1 - cr])
    radii_init.append(0.15)
    centers.append([1 - cr, 1 - cr])
    radii_init.append(0.15)

    # 4 medium edge circles
    er = 0.15 + rng.uniform(-0.02, 0.02)
    centers.append([0.5 + rng.uniform(-0.05, 0.05), er])
    radii_init.append(0.1)
    centers.append([0.5 + rng.uniform(-0.05, 0.05), 1 - er])
    radii_init.append(0.1)
    centers.append([er, 0.5 + rng.uniform(-0.05, 0.05)])
    radii_init.append(0.1)
    centers.append([1 - er, 0.5 + rng.uniform(-0.05, 0.05)])
    radii_init.append(0.1)

    # Fill remaining with interior grid
    remaining = n - 8
    grid_n = int(np.ceil(np.sqrt(remaining)))
    if grid_n > 1:
        spacing = 0.35 / (grid_n - 1)
    else:
        spacing = 0.1
    start_x = 0.3
    start_y = 0.3

    idx = 0
    for i in range(grid_n):
        for j in range(grid_n):
            if idx < remaining:
                x = start_x + i * spacing + rng.uniform(-0.02, 0.02)
                y = start_y + j * spacing + rng.uniform(-0.02, 0.02)
                x = np.clip(x, 0.15, 0.85)
                y = np.clip(y, 0.15, 0.85)
                centers.append([x, y])
                radii_init.append(0.06)
                idx += 1

    centers = np.array(centers[:n])
    radii = np.array(radii_init[:n])
    return centers, radii


def init_dense_grid(n, seed):
    rng = np.random.RandomState(seed)
    centers = []
    
    # 6x5 grid pattern with perturbation
    for i in range(5):
        for j in range(6):
            if len(centers) < n:
                x = (j + 1) * (1.0 / 7) + rng.uniform(-0.02, 0.02)
                y = (i + 1) * (1.0 / 6) + rng.uniform(-0.02, 0.02)
                # Offset every other row for hexagonal
                if i % 2 == 1:
                    x += 1.0 / 14
                x = np.clip(x, 0.05, 0.95)
                y = np.clip(y, 0.05, 0.95)
                centers.append([x, y])
    
    centers = np.array(centers[:n])
    radii = np.ones(n) * 0.04
    return centers, radii


def build_constraints(n):
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': bnd_x_low, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': bnd_x_high, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': bnd_y_low, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': bnd_y_high, 'args': (i, n)})
    
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': circle_overlap, 'args': (i, j, n)})
    
    return cons


def try_optimization(x0, n, bounds, cons, method='SLSQP'):
    try:
        if method == 'SLSQP':
            result = minimize(
                neg_sum_radii,
                x0,
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 8000, 'ftol': 1e-15}
            )
        else:
            result = minimize(
                neg_sum_radii,
                x0,
                args=(n,),
                method='trust-constr',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 5000, 'gtol': 1e-12}
            )
        return result
    except Exception:
        return None


def run_packing():
    n = 26
    best_sum = -1
    best_centers = None
    best_radii = None

    cons = build_constraints(n)
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    # Strategy 1: Hexagonal grid with multiple seeds
    for seed in range(12):
        centers, radii = init_hexagonal(n, seed=seed * 17 + 3)
        x0 = np.concatenate([centers.flatten(), radii])
        
        result = try_optimization(x0, n, bounds, cons, 'SLSQP')
        if result is not None and result.success:
            sum_r = np.sum(result.x[2 * n:])
            if sum_r > best_sum:
                best_sum = sum_r
                best_centers = result.x[:2 * n].reshape(n, 2)
                best_radii = result.x[2 * n:].copy()

    # Strategy 2: Strategic placement (large corners, medium edges, small interior)
    for seed in range(8):
        centers, radii = init_strategic(n, seed=seed * 31 + 7)
        x0 = np.concatenate([centers.flatten(), radii])
        
        result = try_optimization(x0, n, bounds, cons, 'SLSQP')

            sum_r = np.sum(result.x[2 * n:])

                best_centers = result.x[:2 * n].reshape(n, 2)

    # Strategy 3: Dense grid with perturbation
    for seed in range(8):
        centers, radii = init_dense_grid(n, seed=seed * 43 + 11)
        x0 = np.concatenate([centers.flatten(), radii])
        
        result = try_optimization(x0, n, bounds, cons, 'SLSQP')

            sum_r = np.sum(result.x[2 * n:])

                best_centers = result.x[:2 * n].reshape(n, 2)
                best_radii = result.x[2 * n:].copy()

    # Strategy 4: Refine best solution with trust-constr
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii])
        result = try_optimization(x0, n, bounds, cons, 'trust-constr')
        if result is not None and result.success:
            sum_r = np.sum(result.x[2 * n:])

    # Final refinement with SLSQP again
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii])
        result = try_optimization(x0, n, bounds, cons, 'SLSQP')
        if result is not None and result.success:
            sum_r = np.sum(result.x[2 * n:])
            if sum_r > best_sum:
                best_sum = sum_r
                best_centers = result.x[:2 * n].reshape(n, 2)
                best_radii = result.x[2 * n:].copy()

    # Ensure validity
    if best_centers is not None:
        best_radii = np.maximum(best_radii, 0)
        best_centers = np.clip(best_centers, 0, 1)
        # Enforce boundary constraints strictly
        for i in range(n):
            r = best_radii[i]
            best_centers[i, 0] = np.clip(best_centers[i, 0], r, 1 - r)
            best_centers[i, 1] = np.clip(best_centers[i, 1], r, 1 - r)

    return best_centers, best_radii, np.sum(best_radii)
