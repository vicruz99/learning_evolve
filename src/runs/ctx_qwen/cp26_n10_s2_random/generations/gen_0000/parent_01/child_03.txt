import numpy as np
import scipy.optimize as opt

def objective(vars, n):
    """Objective function: minimize negative sum of radii."""
    # vars layout: [x1, y1, r1, x2, y2, r2, ...]
    radii = vars[2::3]
    return -np.sum(radii)

def constraints(vars, n):
    """
    Constraint function.
    Returns an array of constraint values. All must be >= 0.
    Includes boundary and overlap constraints.
    """
    V = vars.reshape(n, 3)
    x, y, r = V[:, 0], V[:, 1], V[:, 2]
    
    cons = []
    
    # Boundary constraints: r <= x <= 1-r  =>  x-r >= 0, 1-x-r >= 0
    # Same for y
    cons.extend(x - r)
    cons.extend(1.0 - x - r)
    cons.extend(y - r)
    cons.extend(1.0 - y - r)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist2 = dx*dx + dy*dy
            sum_r = r[i] + r[j]
            cons.append(dist2 - sum_r**2)
            
    return np.array(cons)

def get_hex_init(n):
    """Generate initial guess based on hexagonal packing."""
    V = np.zeros((n, 3))
    count = 0
    r_approx = 0.1
    # Row height in hex packing is sqrt(3)/2 * 2r = sqrt(3)r ~ 1.732*r
    row_h = np.sqrt(3) * r_approx
    # Col width is 2r
    col_w = 2 * r_approx
    
    y = r_approx
    row_idx = 0
    while count < n:
        x = r_approx
        # Shift every other row
        if row_idx % 2 == 1:
            x += col_w / 2.0
            
        while x + r_approx <= 1.0 and count < n:
            V[count] = [x, y, r_approx * 0.8] # Start slightly smaller
            count += 1
            x += col_w
        
        y += row_h
        row_idx += 1
        
    # If not enough spots (unlikely for 26), fill with random
    if count < n:
        for k in range(count, n):
            V[k] = [0.5, 0.5, 0.05]
            
    return V

def get_grid_init(n):

    V = np.zeros((n, 3))
    # Try to fit in 5x6 grid
    rows = 5
    cols = 6
    step_x = 1.0 / (cols + 1)
    step_y = 1.0 / (rows + 1)
    
    count = 0
    for i in range(rows):
        for j in range(cols):
            if count < n:
                x = step_x * (j + 1)
                y = step_y * (i + 1)
                V[count] = [x, y, 0.08]
                count += 1
    return V

def get_random_init(n, seed):
    """Generate random initial guess."""
    np.random.seed(seed)
    V = np.random.rand(n, 3)
    V[:, 2] *= 0.05 # Small initial radii
    return V

    }
