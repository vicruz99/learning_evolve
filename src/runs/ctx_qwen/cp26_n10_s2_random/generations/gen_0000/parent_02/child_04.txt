import numpy as np
from scipy.optimize import minimize

def compute_cost_and_grad(centers_flat, r, n, pair_indices):
    centers_mat = centers_flat.reshape((n, 2))
    cost = 0.0
    grad = np.zeros_like(centers_flat)
    
    for i, j in pair_indices:
        dx = centers_mat[i, 0] - centers_mat[j, 0]
        dy = centers_mat[i, 1] - centers_mat[j, 1]
        dist = np.sqrt(dx*dx + dy*dy)
        gap = 2*r - dist
        if gap > 0:
            cost += gap * gap
            if dist > 1e-12:
                term = -2.0 * gap / dist
                grad[2*i] += term * dx
                grad[2*i+1] += term * dy
                grad[2*j] -= term * dx
                grad[2*j+1] -= term * dy
    return cost, grad

def objective_wrapper(x, r, n, pair_indices):
    return compute_cost_and_grad(x, r, n, pair_indices)

def run_packing():
    n = 26
    pair_indices = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    # Initial staggered grid configuration
    centers = np.zeros((n, 2))
    idx = 0
    row_counts = [6, 5, 5, 5, 5]
    for i, count in enumerate(row_counts):
        y = (i + 1) / 6.0
        x_step = 1.0 / (count + 1)
        for j in range(count):
            x = (j + 1) * x_step
            centers[idx] = [x, y]
            idx += 1
            
    best_centers = centers.copy()
    r = 0.05
    step = 0.002
    args = (n, pair_indices)
    
    # Phase 1: Coarse continuation
    for _ in range(300):
        bounds = [(r, 1.0 - r)] * (2 * n)
        res = minimize(
            fun=objective_wrapper,
            args=(r, *args),
            jac=True,
            x0=best_centers.flatten(),
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-12, 'gtol': 1e-8, 'maxiter': 300}
        )
        if res.fun < 1e-7:
            best_centers = res.x.reshape((n, 2))
            r += step
        else:
            step /= 1.5
            r -= step
            if step < 1e-6:
                break
                
    # Phase 2: Fine continuation
    step = 5e-5
    for _ in range(200):
        bounds = [(r, 1.0 - r)] * (2 * n)
        res = minimize(
            fun=objective_wrapper,
            args=(r, *args),
            jac=True,
            x0=best_centers.flatten(),

            options={'ftol': 1e-14, 'gtol': 1e-10, 'maxiter': 500}
        )

            best_centers = res.x.reshape((n, 2))
            r += step
        else:

        best_r = r - step
    else:
        best_r = r
        
    radii_opt = np.full(n, best_r)
    return best_centers, radii_opt, np.sum(radii_opt)
