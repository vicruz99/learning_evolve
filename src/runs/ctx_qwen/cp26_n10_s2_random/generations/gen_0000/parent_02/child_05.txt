import numpy as np
import scipy.optimize as opt

def check_constraints(centers, radii):
    """
    Returns a list of constraint violations. 
    For optimization, we want these to be >= 0.
    1. Boundary constraints:
       x - r >= 0  => x - r >= 0
       1 - x - r >= 0 => 1 - x - r >= 0
       y - r >= 0
       1 - y - r >= 0
    2. Overlap constraints:
       dist(i, j) - (r_i + r_j) >= 0
    """
    constraints = []
    n = len(radii)
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r)
        constraints.append(1.0 - x - r)
        constraints.append(y - r)
        constraints.append(1.0 - y - r)
        
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(dist - (radii[i] + radii[j]))
            
    return np.array(constraints)

def objective(params, n):
    """
    Objective function to minimize: -sum(radii)
    params: [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = params[3*i]
        centers[i, 1] = params[3*i + 1]
        radii[i] = params[3*i + 2]
    return -np.sum(radii)

def constraint_func(params, n):
    """
    Returns array of constraint values.
    Optimizer expects constraints >= 0 for 'ineq'.
    """
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = params[3*i]
        centers[i, 1] = params[3*i + 1]
        radii[i] = params[3*i + 2]
    return check_constraints(centers, radii)

def generate_hexagonal_packing(n):

    Generates a hexagonal packing of n circles in a unit square.
    Returns centers and radii.
    """

    # Approximate number of circles along side

    radii = []

    y = r_guess

        x = r_guess + row_offset * r_guess # offset for hexagonal

            centers.append([x, y])
            radii.append(r_guess)
            count += 1
            x += 2 * r_guess
        y += np.sqrt(3) * r_guess
        row_offset = 1 - row_offset
        
    return np.array(centers), np.array(radii)
