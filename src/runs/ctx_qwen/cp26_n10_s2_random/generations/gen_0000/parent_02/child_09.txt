import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    return -np.sum(x[2::3])

def boundary_x_lower(x, i):
    return x[3*i] - x[3*i+2]

def boundary_x_upper(x, i):
    return 1.0 - x[3*i] - x[3*i+2]

def boundary_y_lower(x, i):
    return x[3*i+1] - x[3*i+2]

def boundary_y_upper(x, i):
    return 1.0 - x[3*i+1] - x[3*i+2]

def overlap(x, i, j):
    dx = x[3*i] - x[3*j]
    dy = x[3*i+1] - x[3*j+1]
    return np.sqrt(dx*dx + dy*dy) - x[3*i+2] - x[3*j+2]

def run_packing():
    np.random.seed(42)
    
    # Initialize in a hexagonal lattice
    centers = []
    y = 0.1
    row = 0
    while len(centers) < N_CIRCLES and y <= 0.9:
        x = 0.1
        shift = 0.09 if row % 2 == 1 else 0.0
        while x <= 0.9 and len(centers) < N_CIRCLES:
            centers.append([x + shift, y])
            x += 0.18
        y += 0.1558845
        row += 1
        
    centers = np.array(centers[:N_CIRCLES])
    # Small perturbation to break symmetry
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    r_init = np.full(N_CIRCLES, 0.04)
    x0 = np.hstack([centers.flatten(), r_init])
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES
    
    cons = []
    for i in range(N_CIRCLES):
        cons.append({'type': 'ineq', 'fun': boundary_x_lower, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': boundary_x_upper, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': boundary_y_lower, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': boundary_y_upper, 'args': (i,)})
        
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            cons.append({'type': 'ineq', 'fun': overlap, 'args': (i, j)})
            
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 3000, 'ftol': 1e-9, 'disp': False})
                   
    c_opt = res.x[:2*N_CIRCLES].reshape((N_CIRCLES, 2))
    r_opt = res.x[2*N_CIRCLES:]
    
    # Strict feasibility enforcement
    for i in range(N_CIRCLES):
        r_opt[i] = min(r_opt[i], c_opt[i,0], 1.0-c_opt[i,0], c_opt[i,1], 1.0-c_opt[i,1])
        r_opt[i] = max(0.0, r_opt[i])
        c_opt[i,0] = np.clip(c_opt[i,0], r_opt[i], 1.0-r_opt[i])
        c_opt[i,1] = np.clip(c_opt[i,1], r_opt[i], 1.0-r_opt[i])
        
    # Resolve numerical overlaps
    for _ in range(30):
        max_viol = 0.0
        for i in range(N_CIRCLES):
            for j in range(i+1, N_CIRCLES):
                d = np.sqrt(np.sum((c_opt[i] - c_opt[j])**2))
                req = r_opt[i] + r_opt[j]
                if d < req:
                    shrink = (req - d) / 2.0 + 1e-7
                    r_opt[i] -= shrink
                    r_opt[j] -= shrink
                    max_viol = max(max_viol, shrink)
        if max_viol < 1e-8:
            break
            
    return c_opt, r_opt, np.sum(r_opt)
