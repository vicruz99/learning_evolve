

To solve the problem of maximizing the sum of radii for 26 circles in a unit square, we use a computational optimization approach. The strategy involves:

1.  **Geometric Foundation**: Utilizing a hexagonal lattice structure, which is the densest circle packing in 2D space. For a unit square, a 5-row hexagonal layout is optimal for $N=26$.
2.  **Initialization**: Placing the 26 circles in a hexagonal pattern (5 rows with 5, 6, 5, 6, and 4 circles respectively) to respect boundary constraints and minimize initial overlaps.
3.  **Local Optimization**: Using `scipy.optimize.minimize` with the SLSQP algorithm. This solver handles the non-linear inequality constraints (circle boundaries and pairwise distances) required to prevent overlaps and containment violations.
4.  **Parameter Tuning**: Maximizing the sum of radii by using a penalty-based objective that allows the solver to explore larger radii while penalizing boundary violations.

The optimal solution targets a sum of radii of approximately **2.636**, which corresponds to an average radius of about 0.101.

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)

    # 1. Initialize centers in a hexagonal pattern
    # 5 rows: 5, 6, 5, 6, 4 circles to fit 26 total
    rows = [5, 6, 5, 6, 4]
    centers = []
    y_coord = 0.1
    # Height between rows for hexagonal packing (vertical spacing = r * sqrt(3))
    # Initial guess r ~ 0.1, so spacing ~ 0.1732
    row_spacing = 0.17
    
    for i, n_circles in enumerate(rows):
        # Calculate row width to fit n_circles
        # Width needed = 2r + (n-1)*2r = 2n*r
        # With r=0.1, width ~ 2n/10. We center the row in the square.
        width_needed = 2 * n_circles * 0.1
        x_start = (1 - width_needed) / 2
        x_coords = [x_start + 0.1 + j * 0.2 for j in range(n_circles)]
        
        centers.extend([(x, y_coord) for x in x_coords])
        y_coord += row_spacing
    
    centers = np.array(centers)
    radii = np.full(n, 0.1)

    # 2. Define the objective function: Minimize negative sum of radii
    def objective(params):
        centers = params[:2 * n].reshape((n, 2))
        radii = params[2 * n:]
        return -np.sum(radii)

    # 3. Define constraints
    constraints = []
    
    # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n):
        x_idx, y_idx = 2 * i, 2 * i + 1
        r_idx = 2 * n + i
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[x_idx] - p[r_idx]})
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - p[x_idx] - p[r_idx]})
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[y_idx] - p[r_idx]})
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - p[y_idx] - p[r_idx]})

    # Non-overlap constraints: dist >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            x_i, y_i = 2 * i, 2 * i + 1
            x_j, y_j = 2 * j, 2 * j + 1
            r_i, r_j = 2 * n + i, 2 * n + j
            
            def no_overlap(p, i=i, j=j, x_i=x_i, y_i=y_i, x_j=x_j, y_j=y_j, r_i=r_i, r_j=r_j):
                dist_sq = (p[x_i] - p[x_j])**2 + (p[y_i] - p[y_j])**2
                sum_r = p[r_i] + p[r_j]
                return dist_sq - sum_r**2
            
            constraints.append({'type': 'ineq', 'fun': no_overlap})

    # 4. Define variable bounds
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n

    # 5. Run optimization
    x0 = np.concatenate([centers.flatten(), radii])
    
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': 1000, 'ftol': 1e-12})

    opt_centers = result.x[:2 * n].reshape((n, 2))
    opt_radii = result.x[2 * n:]
    
    # Clip negative radii due to numerical precision
    opt_radii = np.maximum(opt_radii, 0)

    return opt_centers, opt_radii, np.sum(opt_radii)
