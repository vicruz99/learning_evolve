import numpy as np
from scipy.optimize import minimize

def compute_overlap_violations(centers, radii, n):
    """Compute total violation of non-overlap constraints."""
    total_violation = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                total_violation += (min_dist - dist) ** 2
    return total_violation


def force_based_relaxation(centers, radii, n, num_iterations=3000):
    """Use force-directed simulation to relax the packing and increase radii."""
    for iteration in range(num_iterations):
        forces = np.zeros_like(centers)
        force_radii = np.zeros(n)
        
        # Repulsive forces between overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff ** 2)
                dist = np.sqrt(dist_sq + 1e-15)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    if dist > 1e-10:
                        force = overlap * diff / dist
                        forces[i] += force
                        forces[j] -= force
                        force_radii[i] -= overlap * 0.5
                        force_radii[j] -= overlap * 0.5
        
        # Boundary repulsion forces
        for i in range(n):
            for dim in range(2):
                if centers[i][dim] < radii[i] + 1e-10:
                    violation = radii[i] - centers[i][dim]
                    forces[i][dim] += violation * 2.0
                    force_radii[i] -= violation * 0.5
                elif centers[i][dim] > 1 - radii[i] - 1e-10:
                    violation = centers[i][dim] - (1 - radii[i])
                    forces[i][dim] -= violation * 2.0
                    force_radii[i] -= violation * 0.5
        
        # Gradient of objective: maximize sum of radii
        force_radii += 1.0
        
        # Adaptive step sizes
        step_center = 0.005 / (1 + iteration * 0.0005)
        step_radius = 0.002 / (1 + iteration * 0.0005)
        
        centers += step_center * forces
        radii += step_radius * force_radii
        
        # Clip to valid ranges
        for i in range(n):
            centers[i][0] = max(radii[i] + 1e-8, min(1 - radii[i] - 1e-8, centers[i][0]))
            centers[i][1] = max(radii[i] + 1e-8, min(1 - radii[i] - 1e-8, centers[i][1]))
            radii[i] = max(1e-8, min(0.5, radii[i]))
    
    return centers, radii


def slsqp_optimize(centers, radii, n):
    """Optimize using SLSQP."""
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i][0]
        x0[3 * i + 1] = centers[i][1]
        x0[3 * i + 2] = radii[i]
    
    def objective(x):
        return -np.sum(x[2::3])
    
    def get_constraints(x):
        c_opt = x[:2 * n].reshape((n, 2))
        r_opt = x[2 * n:]
        cons = []
        
        # Non-overlap constraints (all pairs)
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((c_opt[i] - c_opt[j]) ** 2)
                min_dist = r_opt[i] + r_opt[j]
                cons.append(dist_sq - min_dist * min_dist)
        
        # Boundary constraints
        for i in range(n):
            cons.append(c_opt[i][0] - r_opt[i])
            cons.append(1 - c_opt[i][0] - r_opt[i])
            cons.append(c_opt[i][1] - r_opt[i])
            cons.append(1 - c_opt[i][1] - r_opt[i])
        
        # Non-negativity of radii
        for i in range(n):
            cons.append(r_opt[i])
        
        return np.array(cons)
    
    bounds = [(0, 1) for _ in range(2 * n)] + [(1e-6, 0.5) for _ in range(n)]
    
    result = minimize(objective, x0, method='SLSQP',
                     constraints={'type': 'ineq', 'fun': get_constraints},
                     bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False})
    
    if result.success:
        centers_opt = result.x[:2 * n].reshape((n, 2))
        radii_opt = result.x[2 * n:]
    else:
        centers_opt = centers
        radii_opt = radii
    
    return centers_opt, radii_opt


def generate_hexagonal_config(n, seed=0):
    """Generate an initial hexagonal configuration."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # 5+5+6+5+5 = 26 circles in hexagonal pattern
    rows = [5, 5, 6, 5, 5]
    r_init = 0.105
    
    idx = 0
    for row_idx, num_cols in enumerate(rows):
        y = r_init + row_idx * np.sqrt(3) * r_init
        for col_idx in range(num_cols):
            x = r_init + col_idx * 2 * r_init + (r_init if row_idx % 2 == 1 else 0)
            # Add small random perturbation
            x += np.random.uniform(-0.01, 0.01)
            y += np.random.uniform(-0.01, 0.01)
            centers[idx] = [x, y]
            radii[idx] = r_init
            idx += 1
    
    return centers, radii


def run_packing():

    best_sum = 0

            best_sum = current_sum
