

Looking at this circle packing problem, I need to pack 26 circles in a unit square to maximize the sum of radii.

<strategy>
Key observations:
1. For 25 circles in a 5×5 grid, r=0.1 gives sum=2.5. Target is 2.636, so I need ~5.4% improvement.
2. Equal-radius packing is generally optimal for maximizing sum of radii (convexity argument).
3. A 5×5 grid can't directly hold 26 circles at r=0.1, but a perturbed grid or mixed arrangement might.
4. I'll use a force-directed optimization with multiple strategies:
   - Start from various initial configurations (grid, hexagonal, random)
   - Use repulsive forces to push overlapping circles apart
   - Gradually increase radii while maintaining feasibility
   - Use simulated annealing concepts to escape local optima
5. I'll also try unequal radii configurations where larger circles fill the center and smaller circles fill gaps.

The force-directed approach with careful cooling and multiple restarts should find a near-optimal packing.

import numpy as np
from scipy.optimize import minimize


def objective(params, n):
    """Maximize sum of radii (minimize negative)."""
    return -np.sum(params[2 * n :])


def get_constraints(n):
    """Build constraint list for scipy optimizer."""
    constraints = []
    
    # Boundary constraints (vectorized as single functions returning arrays)
    def bnd_xmin(p):
        x = p[: 2 * n].reshape(n, 2)[:, 0]
        r = p[2 * n :]
        return x - r
    
    def bnd_xmax(p):
        x = p[: 2 * n].reshape(n, 2)[:, 0]
        r = p[2 * n :]
        return 1 - x - r
    
    def bnd_ymin(p):
        y = p[: 2 * n].reshape(n, 2)[:, 1]
        r = p[2 * n :]
        return y - r
    
    def bnd_ymax(p):
        y = p[: 2 * n].reshape(n, 2)[:, 1]
        r = p[2 * n :]
        return 1 - y - r
    
    constraints.append({"type": "ineq", "fun": bnd_xmin})
    constraints.append({"type": "ineq", "fun": bnd_xmax})
    constraints.append({"type": "ineq", "fun": bnd_ymin})
    constraints.append({"type": "ineq", "fun": bnd_ymax})
    
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            ii = i
            jj = j
    
            def overlap(p, i=ii, j=jj):
                c = p[: 2 * n].reshape(n, 2)
                r = p[2 * n :]
                dist_sq = np.sum((c[i] - c[j]) ** 2)
                return dist_sq - (r[i] + r[j]) ** 2
    
            constraints.append({"type": "ineq", "fun": overlap})
    
    return constraints


def init_grid_plus(n):
    """5x5 grid + 1 extra circle."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08
    
    for i in range(5):
        for j in range(5):
            idx = i * 5 + j
            centers[idx] = [0.1 + 0.2 * j, 0.1 + 0.2 * i]
    
    centers[25] = [0.2, 0.2]
    return centers, radii


def init_hexagonal(n):
    """Hexagonal arrangement."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.07
    
    idx = 0
    rows = [5, 5, 5, 5, 4, 2]
    r_est = 0.09
    
    for row_idx, num_circles in enumerate(rows):
        y = r_est + row_idx * r_est * np.sqrt(3)
        offset = r_est if row_idx % 2 == 1 else 0
        for col in range(num_circles):
            if idx < n:
                x = r_est + offset + col * 2 * r_est
                centers[idx] = [x, y]
                idx += 1
    
    return centers, radii


def init_random(n):
    """Random initial configuration."""
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.ones(n) * 0.05
    return centers, radii


def force_directed_optimize(centers, radii, n, steps=8000):
    """Force-directed optimization with gradual radius increase."""
    centers = centers.copy()
    radii = radii.copy()
    
    lr = 0.005
    radius_increment = 0.00003
    
    for iteration in range(steps):
        forces = np.zeros((n, 2))
        
        # Circle-circle repulsive forces
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff ** 2)
                dist = np.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-12:
                    overlap = min_dist - dist
                    force_mag = overlap / (dist + 1e-12) * 2.0
                    forces[i] += force_mag * diff
                    forces[j] -= force_mag * diff
        
        # Boundary forces (stronger push)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            if x - r < 0:
                forces[i, 0] += 5.0 * (r - x)
            if x + r > 1:
                forces[i, 0] -= 5.0 * (x + r - 1)
            if y - r < 0:
                forces[i, 1] += 5.0 * (r - y)
            if y + r > 1:
                forces[i, 1] -= 5.0 * (y + r - 1)
        
        centers += lr * forces
        centers = np.clip(centers, 1e-8, 1 - 1e-8)
        
        # Gradually increase radii
        radii += radius_increment
        
        # Cool learning rate
        lr *= 0.9998
    
    return centers, radii


def is_valid(centers, radii, n):
    """Check if packing is valid."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0 or np.isnan(r):
            return False
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True


def fix_violations(centers, radii, n, iterations=1000):
    """Fix constraint violations by shrinking radii and repositioning."""
    centers = centers.copy()
    radii = radii.copy()
    
    for _ in range(iterations):
        max_violation = 0
        
        # Fix boundary violations
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                if y - r < 0:
                    centers[i] = [r, r]
                else:
                    centers[i] = [r, y]
            if x + r > 1:
                if y - r < 0:
                    centers[i] = [1 - r, r]
                else:
                    centers[i] = [1 - r, y]
            if y - r < 0:
                centers[i, 1] = r
            if y + r > 1:
                centers[i, 1] = 1 - r
        
        # Fix overlap violations
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    if dist > 1e-12:
                        push = (min_dist - dist) * 0.51
                        d = diff / dist
                        centers[i] -= push * d
                        centers[j] += push * d
                        max_violation = max(max_violation, min_dist - dist)
                    else:
                        # Circles at same position, separate them
                        centers[i] -= [0.001, 0.001]
                        centers[j] += [0.001, 0.001]
                        max_violation = 1
        
        if max_violation < 1e-10:
            break
    
    return centers, radii


def scipy_optimize(centers, radii, n, constraints):
    """Use scipy optimizer to refine."""
    x0 = np.zeros(3 * n)
    x0[: 2 * n] = centers.flatten()
    x0[2 * n :] = radii
    
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    result = minimize(
        objective,
        x0,
        args=(n,),
        constraints=constraints,
        bounds=bounds,
        method="SLSQP",
        options={"maxiter": 1000, "ftol": 1e-15, "disp": False},
    )
    
    if result.success or result.fun < 0:
        new_centers = result.x[: 2 * n].reshape(n, 2)
        new_radii = result.x[2 * n :]
        return new_centers, new_radii, -result.fun
    
    return centers, radii, np.sum(radii)


def run_packing():

        centers, radii = force_directed_optimize(centers, radii, n, steps=6000)

            current_sum = np.sum(radii)
        else:

            radii *= scale
            centers, radii = fix_violations(centers, radii, n)

                current_sum = np.sum(radii)

                current_sum = 0
        
        if current_sum > best_sum:

            scale = 0.998
            radii *= scale
