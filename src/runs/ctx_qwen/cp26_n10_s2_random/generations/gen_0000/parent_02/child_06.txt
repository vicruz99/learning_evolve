import numpy as np
from scipy.optimize import minimize
import math

N_CIRCLES = 26


def compute_repulsive_forces(centers, radii, n):
    """Compute repulsive forces between overlapping circles."""
    forces = np.zeros((n, 2))
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[j][0] - centers[i][0]
            dy = centers[j][1] - centers[i][1]
            dist_sq = dx * dx + dy * dy
            dist = math.sqrt(dist_sq) if dist_sq > 1e-15 else 1e-15
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                force_mag = overlap
                fx = force_mag * dx / dist
                fy = force_mag * dy / dist
                forces[i][0] -= fx
                forces[i][1] -= fy
                forces[j][0] += fx
                forces[j][1] += fy
    return forces


def compute_boundary_forces(centers, radii, n):
    """Compute forces pushing circles away from boundaries."""
    forces = np.zeros((n, 2))
    k = 100.0
    for i in range(n):
        for axis in range(2):
            if centers[i][axis] - radii[i] < 0:
                forces[i][axis] += k * (radii[i] - centers[i][axis])
            if centers[i][axis] + radii[i] > 1:
                forces[i][axis] -= k * (centers[i][axis] + radii[i] - 1)
    return forces


def create_hexagonal_initial(n, r_init, row_config):
    """Create initial hexagonal placement."""
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(len(row_config)):
        size = row_config[row]
        y = r_init + row * r_init * math.sqrt(3)
        offset = r_init if row % 2 == 1 else 0
        x_start = r_init + offset
        for col in range(size):
            if idx < n:
                centers[idx][0] = x_start + col * 2 * r_init
                centers[idx][1] = y
                idx += 1
    return centers


def physics_optimize(centers, radii, n):
    """Run physics-based optimization."""
    velocities = np.zeros_like(centers)
    dt = 0.005
    damping = 0.7
    k_growth = 0.0002
    iterations = 15000
    
    growth_factor = 1.0
    stabilization_threshold = 0.01
    stable_counter = 0
    
    for iteration in range(iterations):
        # Compute forces
        repulsion = compute_repulsive_forces(centers, radii, n)
        boundary = compute_boundary_forces(centers, radii, n)
        forces = repulsion + boundary
        
        # Update velocities and positions
        velocities = damping * velocities + dt * forces
        centers += dt * velocities
        
        # Enforce boundary constraints strictly
        for i in range(n):
            for axis in range(2):
                centers[i][axis] = max(radii[i], min(1 - radii[i], centers[i][axis]))
        
        # Gradually grow radii
        radii += k_growth * growth_factor
        
        # Check stability
        total_force = np.sum(np.abs(repulsion))
        if total_force < stabilization_threshold:
            stable_counter += 1
            if stable_counter > 100:
                k_growth *= 1.1
                stable_counter = 50
        else:
            stable_counter = 0
    
    return centers, radii


def refine_positions(centers, radii, n):
    """Refine positions using gradient-based optimization on penalties."""
    x = np.concatenate([centers.flatten(), radii])
    
    def penalty_objective(x):
        c = x[:2 * n].reshape((n, 2))
        r = x[2 * n:]
        penalty = 0.0
        
        # Boundary penalties
        for i in range(n):
            xi, yi = c[i]
            ri = r[i]
            if xi - ri < 0:
                penalty += 1000 * (xi - ri) ** 2
            if xi + ri > 1:
                penalty += 1000 * (xi + ri - 1) ** 2
            if yi - ri < 0:
                penalty += 1000 * (yi - ri) ** 2
            if yi + ri > 1:
                penalty += 1000 * (yi + ri - 1) ** 2
        
        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i][0] - c[j][0]
                dy = c[i][1] - c[j][1]
                dist = math.sqrt(dx * dx + dy * dy)
                min_d = r[i] + r[j]
                if dist < min_d:
                    penalty += 1000 * (min_d - dist) ** 2
        
        return penalty
    
    bounds = [(0, 1)] * (2 * n) + [(0.001, 0.5)] * n
    
    result = minimize(penalty_objective, x, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-10})
    
    return result.x[:2 * n].reshape((n, 2)), result.x[2 * n:]


def fix_overlaps(centers, radii, n):
    """Reduce radii slightly to eliminate any remaining overlaps."""
    for _ in range(200):
        max_violation = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = math.sqrt(dx * dx + dy * dy)
                overlap = radii[i] + radii[j] - dist
                if overlap > max_violation:
                    max_violation = overlap
        
        # Boundary violations
        for i in range(n):
            for axis in range(2):
                if centers[i][axis] - radii[i] < 0:
                    max_violation = max(max_violation, -centers[i][axis] + radii[i])
                if centers[i][axis] + radii[i] > 1:
                    max_violation = max(max_violation, centers[i][axis] + radii[i] - 1)
        
        if max_violation <= 1e-12:
            break
        
        radii *= (1 - 0.005)
    
    return centers, radii


def try_configuration(row_config, r_init):
    """Try a specific hexagonal configuration."""
    n = N_CIRCLES
    centers = create_hexagonal_initial(n, r_init, row_config)
    radii = np.full(n, r_init)
    
    # Phase 1: Physics-based optimization
    centers, radii = physics_optimize(centers, radii, n)
    
    # Phase 2: Gradient-based refinement
    centers, radii = refine_positions(centers, radii, n)
    
    # Phase 3: Fix any remaining overlaps
    centers, radii = fix_overlaps(centers, radii, n)
    
    # Phase 4: Another round of growth and refinement
    for cycle in range(3):
        radii *= 1.001
        centers, radii = refine_positions(centers, radii, n)
        centers, radii = fix_overlaps(centers, radii, n)
    
    return centers, radii, np.sum(radii)


def run_packing():
    """Main function to pack 26 circles in unit square."""
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Try different hexagonal configurations
    configs = [
        # (row_sizes, initial_radius)

        ([4, 6, 5, 6, 5], 0.06),
        ([5, 4, 6, 5, 6], 0.06),
        ([6, 6, 6, 6, 2], 0.05),
        ([5, 5, 5, 5, 5, 1], 0.06),
    ]
    
    for row_config, r_init in configs:
        total_circles = sum(row_config)
        if total_circles != N_CIRCLES:
            continue
        
        centers, radii, s = try_configuration(row_config, r_init)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    # Final validation fix
    best_centers, best_radii = fix_overlaps(best_centers, best_radii, N_CIRCLES)
    
    # Clip centers to valid range
    for i in range(N_CIRCLES):
        for axis in range(2):
            best_centers[i][axis] = max(best_radii[i], min(1 - best_radii[i], best_centers[i][axis]))
    
    return best_centers, best_radii, np.sum(best_radii)
