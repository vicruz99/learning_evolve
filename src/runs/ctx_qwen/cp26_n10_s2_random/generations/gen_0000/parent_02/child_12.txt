import numpy as np
from scipy.optimize import linprog

def generate_initial_centers(n):
    """
    Generate an initial configuration of centers based on a hexagonal lattice.
    """
    # Target density approximation
    # We want to fit n circles.

    # Area per circle approx 1/n. Radius approx sqrt(1/(n*pi)).
    # But we need to fit them in 1x1.
    
    # Let's try a grid of points and select the best n, or generate a lattice.
    # A simple hexagonal lattice generation:
    # Rows shifted by half step.
    
    # Estimate spacing
    # Area = 1. n circles. Area per circle ~ 1/n.
    # pi * r^2 ~ 1/n => r ~ 1/sqrt(n*pi)
    # Diameter ~ 2/sqrt(n*pi)
    # Spacing s ~ Diameter
    s = 2.0 / np.sqrt(n * np.pi)
    
    centers = []
    y = s / 2 # Start with half spacing to center vertically? 
    # Actually, let's just fill the square.
    
    # We will generate a grid of potential centers and pick the best n?
    # Or just generate n centers on a lattice.
    
    # Let's try to fit a lattice inside [0,1]x[0,1]
    # Number of columns approx 1/s
    # Number of rows approx 1/(s * sqrt(3)/2)
    
    # Let's try a simpler approach: Random initialization with a bit of structure
    # or just a dense grid sampling and pick random subset?
    # No, let's try to construct a lattice.
    
    # Rows
    row_height = s * np.sqrt(3) / 2
    # y coordinates
    ys = np.arange(0, 1, row_height)
    
    centers_list = []
    for y_coord in ys:
        if y_coord < 0 or y_coord > 1:
            continue
        # x coordinates
        # Alternate offset
        # For even rows, offset 0. For odd rows, offset s/2
        row_idx = int(round(y_coord / row_height))
        offset = (row_idx % 2) * (s / 2)
        
        xs = np.arange(offset, 1, s)
        for x_coord in xs:
            if x_coord >= 0 and x_coord <= 1:
                centers_list.append([x_coord, y_coord])
    
    # If we have more than n, select n that are most central?
    # Or just take first n.
    # If fewer, add random?
    
    if len(centers_list) >= n:
        # Pick n points. Maybe pick those with max distance to neighbors?
        # Or just random sample to avoid bias?

        centers = np.array(centers_list[:n])
    else:
        # If lattice didn't produce enough, fill with random
        random_centers = np.random.rand(n - len(centers_list), 2)
        centers = np.vstack([np.array(centers_list), random_centers])
        
    return centers

def solve_radii_lp(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= ||c_i - c_j|| for all i < j
      r_i <= x_i
      r_i <= 1 - x_i
      r_i <= y_i
      r_i <= 1 - y_i
      r_i >= 0
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints matrix A_ub and vector b_ub
    # Each row corresponds to a constraint sum(a_k * r_k) <= b
    
    constraints_A = []
    constraints_b = []
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    # This is O(n^2)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            # Row with 1 at i and 1 at j
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints_A.append(row)
            constraints_b.append(dist)
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(x)
        
        # r_i <= 1 - x
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(1.0 - x)
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(y)
        
        # r_i <= 1 - y
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(1.0 - y)

    A_ub = np.array(constraints_A)
    b_ub = np.array(constraints_b)
    
    # Bounds for r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # Method 'highs' is efficient
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = res.x
        return radii, -res.fun
    else:
        # Fallback to small radii if LP fails (shouldn't happen with valid centers)
        small_r = 0.001
        radii = np.full(n, small_r)
        return radii, n * small_r
