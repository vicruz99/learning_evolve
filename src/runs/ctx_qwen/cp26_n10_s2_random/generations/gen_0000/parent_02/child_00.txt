def hexagonal_grid(n):
    """Initialize circles in a hexagonal pattern."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)
    
    idx = 0
    margin = 0.08
    col_spacing = 0.19
    row_spacing = 0.16
    
    for row in range(7):
        for col in range(5):
            if idx >= n:
                break
            x = margin + col * col_spacing
            y = margin + row * row_spacing
            if row % 2 == 1:
                x += col_spacing * 0.5
            centers[idx] = [x, y]
            idx += 1
    
    return centers, radii


def square_grid(n):

def penalty_objective(x, n):

    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    # Primary objective: maximize sum of radii (negate for minimization)
    obj = -np.sum(radii)
    
    # Boundary constraint penalties
    penalty_weight = 5000.0
    for i in range(n):
        # Left boundary
        violation = max(0.0, radii[i] - centers[i, 0])
        obj += penalty_weight * violation * violation
        # Right boundary
        violation = max(0.0, centers[i, 0] + radii[i] - 1.0)
        obj += penalty_weight * violation * violation
        # Bottom boundary
        violation = max(0.0, radii[i] - centers[i, 1])
        obj += penalty_weight * violation * violation
        # Top boundary
        violation = max(0.0, centers[i, 1] + radii[i] - 1.0)
        obj += penalty_weight * violation * violation
    
    # Non-overlap penalties
    for i in range(n):
        for j in range(i + 1, n):

            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-16)
            overlap = max(0.0, radii[i] + radii[j] - dist)
            obj += penalty_weight * overlap * overlap
    
    return obj


def compute_gradient(x, n):
    """Numerical gradient computation."""
    obj_val = penalty_objective(x, n)
    grad = np.zeros_like(x)
    eps = 1e-7
    
    for k in range(len(x)):
        x_plus = x.copy()
        x_plus[k] += eps
        obj_plus = penalty_objective(x_plus, n)
        grad[k] = (obj_plus - obj_val) / eps
    
    return grad


def optimize_with_basinhopping(centers_init, radii_init, n):

    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    bounds = []

        bounds.append((0.001, 0.999))

        bounds.append((0.001, 0.45))

        return penalty_objective(x, n)
    
    minimizer_kwargs = {
        'method': 'L-BFGS-B',
        'bounds': bounds,
        'args': (n,),
        'options': {'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-10}
    }
    
    try:
        result = basinhopping(
            objective_wrapper,

                    dist = np.sqrt(dx*dx + dy*dy + 1e-16)

                violation = max(0.0, radii[i] - centers_temp[i, 1])

def refine_with_slscp(centers, radii, n):
    """Final refinement using SLSQP with explicit constraints."""
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = []
    for i in range(n):
        bounds.append((0.0001, 0.9999))
        bounds.append((0.0001, 0.9999))
        bounds.append((0.0001, 0.5))
    
    def objective(x):
        return -np.sum(x[2*n:])
    
    def constraint_fun(x):
        c = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        cons = []
        
        # Boundary constraints
        for i in range(n):
            cons.append(c[i, 0] - r[i] - 1e-10)
            cons.append(1.0 - c[i, 0] - r[i] - 1e-10)
            cons.append(c[i, 1] - r[i] - 1e-10)
            cons.append(1.0 - c[i, 1] - r[i] - 1e-10)
        
        # Non-overlap constraints (squared form for stability)
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist_sq = dx * dx + dy * dy
                min_dist = r[i] + r[j] + 1e-10
                cons.append(dist_sq - min_dist * min_dist)
        
        return np.array(cons)
    
    constraint = {'type': 'ineq', 'fun': constraint_fun}
    
    try:
        result = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False}
        )
        x_opt = result.x
    except Exception:
        x_opt = x0
    
    centers_opt = x_opt[:2*n].reshape(n, 2)
    radii_opt = x_opt[2*n:]
    
    return centers_opt, radii_opt


def ensure_validity(centers, radii, n):
    """Ensure all constraints are satisfied."""
    centers = centers.copy()
    radii = radii.copy()
    
    # Clip radii to boundary limits
    for i in range(n):
        max_r = min(
            centers[i, 0],
            1.0 - centers[i, 0],
            centers[i, 1],
            1.0 - centers[i, 1],
            radii[i]
        )
        radii[i] = max(max_r, 1e-8)
    
    # Resolve remaining overlaps by reducing smaller circle
    for iteration in range(50):
        max_overlap = 0
        max_pair = None
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = radii[i] + radii[j] - dist
                if overlap > max_overlap:
                    max_overlap = overlap
                    max_pair = (i, j)
        
        if max_overlap <= 1e-10 or max_pair is None:
            break
        
        i, j = max_pair
        reduction = max_overlap * 0.5 + 1e-10
        if radii[i] < radii[j]:
            radii[i] = max(radii[i] - reduction, 1e-8)
        else:
            radii[j] = max(radii[j] - reduction, 1e-8)
    
    # Final boundary check
    for i in range(n):
        max_r = min(
            centers[i, 0],
            1.0 - centers[i, 0],
            centers[i, 1],
            1.0 - centers[i, 1],
            radii[i]
        )
        radii[i] = max(max_r, 1e-8)
    
    return centers, radii
