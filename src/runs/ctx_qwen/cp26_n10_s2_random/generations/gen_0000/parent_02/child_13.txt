import numpy as np
from scipy.optimize import linprog
import random
import math

def solve_radii_lp(centers):
    """
    Solves the Linear Programming problem to find optimal radii for fixed centers.
    Maximizes sum(r_i) subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    if n == 0:
        return np.array([]), 0.0

    # Variables: r_0, r_1, ..., r_{n-1}
    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    c_obj = -np.ones(n)

    # Constraints: A_ub * r <= b_ub
    
    # 1. Non-overlap constraints: r_i + r_j <= dist(i, j)
    # We construct the matrix row by row
    A_ub = []
    b_ub = []
    
    for i in range(n):
        for j in range(i + 1, n):
            # Calculate distance between centers i and j
            dist = np.linalg.norm(centers[i] - centers[j])
            
            # Constraint row: 0...1...1...0
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)

    # 2. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        row_x = np.zeros(n)
        row_x[i] = 1.0
        A_ub.append(row_x)
        b_ub.append(x)
        
        # r_i <= 1 - x
        row_1mx = np.zeros(n)
        row_1mx[i] = 1.0
        A_ub.append(row_1mx)
        b_ub.append(1.0 - x)
        
        # r_i <= y
        row_y = np.zeros(n)
        row_y[i] = 1.0
        A_ub.append(row_y)
        b_ub.append(y)
        
        # r_i <= 1 - y
        row_1my = np.zeros(n)
        row_1my[i] = 1.0
        A_ub.append(row_1my)
        b_ub.append(1.0 - y)

    # Convert to numpy arrays
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i: 0 <= r_i
    bounds = [(0, None) for _ in range(n)]

    try:
        # Solve LP
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            radii = res.x
            return radii, -res.fun
        else:
            # Fallback or error handling
            # If LP fails, return small valid radii (e.g., 0) to avoid crash, 
            # though ideally this shouldn't happen with valid centers.
            return np.zeros(n), 0.0
            
    except Exception:
        return np.zeros(n), 0.0

def get_initial_centers_grid(n, seed_offset=0):
    """Generates a grid-like initial configuration."""
    centers = []
    # Try to fit in a grid
    cols = int(math.ceil(math.sqrt(n)))
    rows = int(math.ceil(n / cols))
    
    # Add some randomness to break symmetry
    np.random.seed(42 + seed_offset)
    
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = (j + 0.5) / cols + np.random.uniform(-0.05, 0.05)
            y = (i + 0.5) / rows + np.random.uniform(-0.05, 0.05)
            x = np.clip(x, 0.1, 0.9)
            y = np.clip(y, 0.1, 0.9)
            centers.append([x, y])
        if len(centers) >= n:
            break
            
    # Fill remaining if any
    while len(centers) < n:
        centers.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
        
    return np.array(centers[:n])

def get_initial_centers_random(n, seed_offset=0):
    """Generates a random initial configuration."""
    np.random.seed(123 + seed_offset)
    return np.random.uniform(0.1, 0.9, size=(n, 2))

def run_packing():
    """
    Runs the optimization to pack 26 circles in a unit square.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Helper to run optimization from a specific start
    def optimize_from(centers_init, max_iter=1000, temp_start=1.0):
        nonlocal best_sum, best_centers, best_radii
        
        centers = centers_init.copy()
        current_radii, current_sum = solve_radii_lp(centers)
        
        # Simulated Annealing parameters
        temp = temp_start
        cooling_rate = 0.995
        step_size = 0.02
        
        # Pre-allocate for efficiency if needed, but n is small
        
        for k in range(max_iter):
            # Pick a random circle to move
            idx = np.random.randint(n)
            
            # Propose a move
            delta_x = np.random.uniform(-step_size, step_size)
            delta_y = np.random.uniform(-step_size, step_size)
            
            new_x = centers[idx, 0] + delta_x
            new_y = centers[idx, 1] + delta_y
            
            # Boundary check for centers (loose check, LP handles strict)
            if new_x < 0 or new_x > 1 or new_y < 0 or new_y > 1:
                continue
                
            # Apply move temporarily
            old_pos = centers[idx].copy()
            centers[idx] = [new_x, new_y]
            
            # Solve for new radii
            new_radii, new_sum = solve_radii_lp(centers)
            
            # Acceptance criteria
            if new_sum > current_sum:
                # Always accept improvement
                current_radii = new_radii
                current_sum = new_sum
                
                # Update global best
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers.copy()
                    best_radii = current_radii.copy()
            else:
                # Probabilistically accept worse solution
                delta_e = current_sum - new_sum # positive if worse
                prob = math.exp(-delta_e / temp)
                if np.random.random() < prob:
                    current_radii = new_radii
                    current_sum = new_sum
                else:
                    # Revert move
                    centers[idx] = old_pos
            
            # Cooling
            if k % 50 == 0:
                temp *= cooling_rate
                step_size *= 0.95 # Reduce step size slowly

    # Strategy: Run multiple attempts with different initializations
    attempts = 10
    
    for i in range(attempts):
        if i < 3:
            # Grid based starts
            centers_init = get_initial_centers_grid(n, seed_offset=i)
        else:
            # Random starts
            centers_init = get_initial_centers_random(n, seed_offset=i)
            
        optimize_from(centers_init, max_iter=1500, temp_start=0.5)
        
    # If best_radii is still None (should not happen), fallback
    if best_radii is None:
        centers_init = get_initial_centers_grid(n)
        best_radii, best_sum = solve_radii_lp(centers_init)
        best_centers = centers_init

    return best_centers, best_radii, best_sum
