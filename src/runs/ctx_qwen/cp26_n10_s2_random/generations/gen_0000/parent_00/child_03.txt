import numpy as np
from scipy.optimize import minimize

def get_initial_config(n=26):
    """Generate a hexagonal initial configuration."""
    pts = []
    r_unit = 1.0
    y = 0
    row_idx = 0
    while len(pts) < n:
        n_row = 6 if row_idx % 2 == 0 else 5
        x_start = r_unit if row_idx % 2 == 1 else 0
        for i in range(n_row):
            if len(pts) >= n: break
            pts.append((x_start + i * 2 * r_unit, y))
        y += np.sqrt(3) * r_unit
        row_idx += 1
        
    pts = np.array(pts[:n])
    min_val = pts.min()
    max_val = pts.max()
    span = max_val - min_val
    # Scale to [0.1, 0.9] to leave room for expansion
    pts = (pts - min_val) / span * 0.8 + 0.1
    
    # Add slight random jitter to break symmetry and avoid local minima
    pts += np.random.uniform(-0.01, 0.01, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    return pts, np.full(n, 0.08)

def extract_vars(vars, n=26):
    """Extract centers and radii from interleaved variables."""
    c = np.empty((n, 2))
    c[:, 0] = vars[0::3]
    c[:, 1] = vars[1::3]
    r = vars[2::3]
    return c, r

def obj(vars):
    """Objective: minimize negative sum of radii."""
    _, r = extract_vars(vars)
    return -np.sum(r)

def con(vars):
    """Constraint function: returns values that must be >= 0."""
    n = 26
    c, r = extract_vars(vars)
    
    M = 4 * n + n * (n - 1) // 2
    con_vals = np.empty(M)
    
    # Boundary constraints
    con_vals[:n] = c[:, 0] - r
    con_vals[n:2*n] = 1 - c[:, 0] - r
    con_vals[2*n:3*n] = c[:, 1] - r
    con_vals[3*n:4*n] = 1 - c[:, 1] - r
    
    # Pairwise constraints: dist^2 - (r_i + r_j)^2 >= 0
    diffs = c[:, None, :] - c[None, :, :]
    dists_sq = np.sum(diffs**2, axis=2)
    r_sums_sq = (r[:, None] + r[None, :])**2
    pairwise = dists_sq - r_sums_sq
    con_vals[4*n:] = pairwise[np.triu_indices(n, k=1)]
    
    return con_vals

def con_jac(vars):
    """Analytical Jacobian of constraints."""
    n = 26
    c, r = extract_vars(vars)
    
    M = 4 * n + n * (n - 1) // 2
    J = np.zeros((M, 3 * n))
    
    # Boundary constraints Jacobian
    # x_i - r_i >= 0
    J[:n, 0::3] = 1.0
    J[:n, 2::3] = -1.0
    # 1 - x_i - r_i >= 0
    J[n:2*n, 0::3] = -1.0
    J[n:2*n, 2::3] = -1.0
    # y_i - r_i >= 0
    J[2*n:3*n, 1::3] = 1.0
    J[2*n:3*n, 2::3] = -1.0
    # 1 - y_i - r_i >= 0
    J[3*n:4*n, 1::3] = -1.0
    J[3*n:4*n, 2::3] = -1.0
    
    # Pairwise constraints Jacobian
    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dr = r[i] + r[j]
            
            # df/dx_i, df/dy_i, df/dr_i
            J[idx, 3*i] = 2 * dx
            J[idx, 3*i + 1] = 2 * dy
            J[idx, 3*i + 2] = -2 * dr
            
            # df/dx_j, df/dy_j, df/dr_j
            J[idx, 3*j] = -2 * dx
            J[idx, 3*j + 1] = -2 * dy
            J[idx, 3*j + 2] = -2 * dr
            
            idx += 1
            
    return J

def run_packing():
    n = 26
    centers, radii = get_initial_config(n)
    
    # Construct initial variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.empty(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    
    cons = {'type': 'ineq', 'fun': con, 'jac': con_jac}
    
    # Run optimization
    res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
    
    best_vars = res.x
    
    # Extract results
    best_centers = np.empty((n, 2))
    best_centers[:, 0] = best_vars[0::3]
    best_centers[:, 1] = best_vars[1::3]
    best_radii = best_vars[2::3]
    
    # Ensure validity within numerical tolerance
    best_centers = np.clip(best_centers, 0, 1)
    best_radii = np.maximum(best_radii, 0)
    
    sum_r = np.sum(best_radii)
    return best_centers, best_radii, sum_r
