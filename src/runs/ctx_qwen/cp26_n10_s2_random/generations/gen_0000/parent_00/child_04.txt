import numpy as np
from scipy.optimize import minimize
import math

def compute_sum_radii(vars, n=26):
    """
    Calculates the sum of radii from the flattened variable array.
    vars structure: [x0, y0, x1, y1, ..., r0, r1, ...]
    Actually, let's use a simpler structure: [x0, y0, r0, x1, y1, r1, ...]
    """
    # Reshape to (n, 3) -> (x, y, r)
    points = np.array(vars).reshape(n, 3)
    return np.sum(points[:, 2])

def get_constraints(n=26):
    """
    Returns a list of constraint dictionaries for scipy.optimize.
    Constraints must be >= 0.
    """
    constraints = []
    
    # Boundary constraints for each circle
    # x - r >= 0  => x - r
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    for i in range(n):
        idx = i * 3
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx] - v[idx + 2]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx] - v[idx + 2]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx + 1] - v[idx + 2]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx + 1] - v[idx + 2]
        })
        # r >= 0 (handled by bounds or ineq)
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx + 2]
        })

    # Non-overlap constraints: dist(c_i, c_j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = i * 3
            idx_j = j * 3
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=idx_i, j=idx_j: \
                    (v[i] - v[j])**2 + (v[i+1] - v[j+1])**2 - (v[i+2] + v[j+2])**2
            })

    return constraints

def generate_hexagonal_init(n=26):
    """
    Generates an initial configuration of n circles in a hexagonal pattern.
    """
    centers = []
    radii = []
    
    # Estimate radius based on hexagonal packing density
    # Area ~ n * pi * r^2 <= 1 * 0.9069
    # r_est = sqrt(0.9069 / (n * pi))
    r_est = math.sqrt(0.9069 / (n * math.pi))
    
    # Adjust r_est to fit in square roughly
    # For 26 circles, a grid-like or hex layout works.
    # Let's try a simple hex grid generation.
    
    rows = int(math.ceil(math.sqrt(n * math.sqrt(3) / 2))) + 1
    # Better heuristic for 26: 6 rows, alternating 5,4,5,4,5,3 or similar
    
    current_n = 0
    y = r_est
    row_idx = 0
    
    while current_n < n and y <= 1.0 - r_est + 1e-6:
        # Determine number of circles in this row
        # Max circles in width 1 with radius r_est is floor(1 / (2*r_est))
        # But with offset, we can fit more?
        # Standard hex: row 0 has k, row 1 has k-1 or k.
        
        # Simple approach: fill row with circles spaced by 2*r_est
        # x_start = r_est
        # x_end = 1 - r_est
        
        if row_idx % 2 == 0:
            x_start = r_est
        else:
            x_start = 2 * r_est # Offset by one diameter? No, offset by r_est (radius) for hex?
            # In hex packing, offset is r (horizontal distance between rows centers is r)
            # Wait, distance between centers is 2r. Horizontal shift is r.
            # So x_start for odd row is 2r (if even row starts at r).
            x_start = 2 * r_est 
        
        x = x_start
        while x <= 1.0 - r_est + 1e-6 and current_n < n:
            centers.append([x, y])
            radii.append(r_est)
            current_n += 1
            x += 2 * r_est
        
        y += math.sqrt(3) * r_est # Vertical distance between rows
        row_idx += 1
        
    # If we didn't get n circles, or got too many, adjust
    if len(centers) < n:
        # Fill remaining randomly
        for _ in range(n - len(centers)):
            centers.append([np.random.rand(), np.random.rand()])
            radii.append(r_est * 0.5)
    
    # If we got more than n, trim
    centers = centers[:n]
    radii = radii[:n]
    
    # Reshape to variables vector [x, y, r, x, y, r, ...]
    vars = []
    for i in range(n):
        vars.extend([centers[i][0], centers[i][1], radii[i]])
        
    return np.array(vars)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    constraints = get_constraints(n)
    
    best_sum_radii = -1.0
    best_vars = None
    
    # Multiple restarts to avoid local minima
    # 1. Hexagonal init
    # 2. Random init
    
    inits = []
    inits.append(generate_hexagonal_init(n))
    
    # Add a few random initializations
    for _ in range(5):
        random_vars = np.random.uniform(0.05, 0.95, size=n * 3)
        # Set radii small initially to avoid constraint violation
        for i in range(n):
            random_vars[i*3 + 2] = 0.02 
        inits.append(random_vars)

    for init_vars in inits:
        try:
            res = minimize(
                compute_sum_radii, 
                init_vars, 
                args=(n,),
                method='SLSQP',
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9}
            )
            
            if res.success or (res.fun > best_sum_radii):
                # Verify constraints manually with tolerance
                vars_opt = res.x
                centers_opt = np.array(vars_opt).reshape(n, 3)[:, :2]
                radii_opt = np.array(vars_opt).reshape(n, 3)[:, 2]
                
                # Check validity
                valid = True
                # Boundary
                if np.any(centers_opt - radii_opt[:, None] < -1e-5) or \
                   np.any(centers_opt + radii_opt[:, None] > 1.0 + 1e-5):
                    valid = False
                
                # Overlap
                if valid:
                    for i in range(n):
                        for j in range(i+1, n):
                            dist = np.linalg.norm(centers_opt[i] - centers_opt[j])
                            if dist < radii_opt[i] + radii_opt[j] - 1e-5:
                                valid = False
                                break
                        if not valid: break
                
                if valid and res.fun > best_sum_radii:
                    best_sum_radii = res.fun
                    best_vars = vars_opt.copy()
        except Exception:
            continue

    if best_vars is None:
        # Fallback to simple grid if optimization fails
        # 5x5 grid is 25, we need 26.

                radii.append(0.1)

        radii.append(0.04)
        
        best_vars = []
        for i in range(n):
            best_vars.extend([centers[i][0], centers[i][1], radii[i]])
        best_vars = np.array(best_vars)
        best_sum_radii = np.sum(radii)

    centers = np.array(best_vars).reshape(n, 3)[:, :2]
    radii = np.array(best_vars).reshape(n, 3)[:, 2]
    
    # Final validation check (sanity)
    # The problem requires returning centers and radii.
    # We ensure they are valid.
    
    return centers, radii, float(np.sum(radii))
