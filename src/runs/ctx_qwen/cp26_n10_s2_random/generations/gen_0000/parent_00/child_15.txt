import numpy as np
from scipy.optimize import minimize

def run_packing():

    Solves the circle packing problem for 26 circles in a unit square.
    Maximizes the sum of radii.

    n = 26

    # We start with a grid-like configuration. 
    # A 6x5 grid gives 30 points, we take 26.
    # Coordinates are placed to be somewhat centered.
    
    cols = 6
    rows = 5
    
    # Create grid points
    # Spacing: to fit 6 items, maybe step 1/7? 
    # Let's use linspace from 0.1 to 0.9 to keep margin for radius.
    x_grid = np.linspace(0.1, 0.9, cols)
    y_grid = np.linspace(0.1, 0.9, rows)
    
    centers_list = []
    count = 0
    for y in y_grid:
        for x in x_grid:
            centers_list.append((x, y))
            count += 1
            if count == n:
                break
        if count == n:
            break
            
    centers_init = np.array(centers_list)
    
    # Initial radii: small enough to be valid.

    r_init = np.full(n, 0.05)
    
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = r_init[i]

    # x, y in [0, 1]

        # Sum of radii
        # vars_flat indices: 3*i + 2 is radius
        radii = vars_flat[2::3]
        return -np.sum(radii) # Minimize negative sum

            c.append(yi - ri)
            c.append(1 - yi - ri)
        return np.array(c)
