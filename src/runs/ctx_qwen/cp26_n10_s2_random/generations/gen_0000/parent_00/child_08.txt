from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    centers = np.zeros((n, 2))

    row_lengths = [6, 5, 6, 5, 4]
    current_y = 0.15
    y_spacing = 0.21 # Approx sqrt(3)/2 * diameter
    
    for r_len in row_lengths:
        row_x_spacing = (1.0 - 0.3) / (r_len - 1) if r_len > 1 else 0.5
        # Centering the row
        row_width = (r_len - 1) * row_x_spacing
        start_x = 0.15

            if idx < n:
                centers[idx] = [start_x + c * row_x_spacing, current_y]
                idx += 1
        current_y += y_spacing

    # Initialize radii
    radii = np.ones(n) * 0.10

    x0 = np.concatenate([centers.flatten(), radii])

    def objective(x):
        r = x[2*n:]
        return -np.sum(r)

        centers = x[:2*n].reshape(n, 2)

        bnd_x_lower = centers[:, 0] - radii
        bnd_x_upper = 1.0 - centers[:, 0] - radii
        bnd_y_lower = centers[:, 1] - radii
        bnd_y_upper = 1.0 - centers[:, 1] - radii

        r_sum_sq = r_sum**2

        mask = np.triu(np.ones((n, n), dtype=bool), k=1)

        return np.concatenate([overlaps, bnd_x_lower, bnd_x_upper, bnd_y_lower, bnd_y_upper])

        bounds.append((0.0, 1.0)) # xi

    res = minimize(objective, x0, method='SLSQP', bounds=bounds,

    best_centers = res.x[:2*n].reshape(n, 2)

    best_radii = np.maximum(best_radii, 0)
    
    return best_centers, best_radii, sum_radii
