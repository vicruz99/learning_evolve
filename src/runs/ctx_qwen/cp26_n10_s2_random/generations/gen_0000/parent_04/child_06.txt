import numpy as np
from scipy.optimize import minimize

def generate_hexagonal_lattice(n, padding=0.01):
    """
    Generates n points in a hexagonal lattice pattern inside the unit square.
    """
    points = []
    # Estimate grid size
    # Area per point approx sqrt(3)/2 * (2r)^2 ~ 1.732 r^2. 
    # If density is high, r ~ 0.1. 
    # We just generate a grid and pick points.
    
    # Try to fit points in a hexagonal grid
    # y spacing: sqrt(3) * spacing
    # x spacing: spacing
    
    # Heuristic: start with a grid density
    # 26 points. sqrt(26) ~ 5.
    # Let's try rows of 5 or 6.
    
    rows = 6
    cols = 5
    
    # Spacing parameters
    # We want points in [padding, 1-padding]
    width = 1.0 - 2 * padding
    height = 1.0 - 2 * padding
    
    # Adjust spacing to fill the square
    # For hexagonal, vertical pitch is sqrt(3)/2 * horizontal_pitch
    # Let horizontal_pitch = dx, vertical_pitch = dy = sqrt(3)/2 * dx
    
    # We need to fit 'rows' rows vertically and 'cols' cols horizontally (roughly)
    # But hexagonal rows shift.
    
    # Let's just generate a lot of points and pick the first n
    pts = []
    
    # Estimate dx based on width and cols
    # If we have cols circles, width needed is (cols-1)*dx + 2*radius? 
    # But we don't know radius. We just want centers.
    # Let's assume centers are distributed in [0,1].
    
    # A simple way to get a hexagonal distribution:
    # Iterate y, iterate x with offset
    
    # Let's fix a spacing 's'
    # Max y ~ 1. Max x ~ 1.
    # s_x = 1.0 / (cols - 1) if cols > 1 else 1.0
    # s_y = s_x * np.sqrt(3) / 2
    
    # Better: use linspace
    x_coords = np.linspace(padding, 1-padding, cols)
    y_coords = np.linspace(padding, 1-padding, rows)
    
    for r_idx, y in enumerate(y_coords):
        # Offset every other row
        offset = 0.0
        if r_idx % 2 == 1:
            offset = x_coords[1] - x_coords[0] if len(x_coords) > 1 else 0.0
        
        # Generate x positions
        # Shifted row has one less or same number?
        # Let's just generate within bounds
        current_x = padding + offset
        while current_x <= 1 - padding:
            pts.append((current_x, y))
            current_x += (x_coords[1] - x_coords[0])
            
    # Shuffle and pick n
    np.random.shuffle(pts)
    selected = pts[:n]
    
    # Convert to array
    centers = np.array(selected)
    
    # Ensure centers are strictly inside [0,1] for initialization (radius starts small)
    centers = np.clip(centers, 0.0, 1.0)
    
    return centers

def calculate_penalty(centers, radii):
    """
    Calculates a penalty for constraint violations.
    Returns 0 if valid, positive value otherwise.
    """
    n = len(radii)
    penalty = 0.0
    
    # Boundary constraints
    # x - r >= 0  => r - x <= 0
    # x + r <= 1  => x + r - 1 <= 0
    # Same for y
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Boundary violations
        viol = 0.0
        if x - r < 0:
            viol += (x - r)**2
        if x + r > 1:
            viol += (x + r - 1)**2
        if y - r < 0:
            viol += (y - r)**2
        if y + r > 1:
            viol += (y + r - 1)**2
        penalty += viol * 1000.0 # High weight for boundaries
        
    # Overlap constraints
    # dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # Check pairwise
    
    # Vectorized distance check might be faster but O(N^2) is fine for N=26
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            dist = np.sqrt(dist_sq)
            
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += overlap**2 * 5000.0 # Very high weight for overlaps
                
    return penalty

def objective(params, n):
    """
    Objective function to minimize.
    params: flattened array of [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = params[3*i]
        centers[i, 1] = params[3*i + 1]
        radii[i] = params[3*i + 2]
        
    # We want to maximize sum(radii), so minimize -sum(radii)
    obj_val = -np.sum(radii)
    
    # Add penalty
    pen = calculate_penalty(centers, radii)
    
    return obj_val + pen
