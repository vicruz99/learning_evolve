import numpy as np
from scipy.optimize import minimize

# Constants
N_CIRCLES = 26

def hexagonal_init(n, rng_seed=42):
    """
    Initialize centers in a hexagonal pattern.
    """
    rng = np.random.default_rng(rng_seed)
    
    # We want to place n circles. 
    # A hexagonal packing density is higher than square.
    # Let's try to fill rows.
    # Approximate radius r ~ 0.1. 
    # Vertical spacing ~ sqrt(3)*r ~ 0.173. 
    # Number of rows ~ 1/0.173 ~ 6.
    
    centers = np.zeros((n, 2))
    count = 0
    
    # Try to generate points on a hex grid
    # We can scale the grid to fit in [0,1] roughly
    # But since we don't know optimal r, let's just place them 
    # with a density that allows optimization to expand.
    
    # Let's create a pool of hex points and pick n.
    # Grid spacing 0.15 (approx)
    step = 0.12
    h_step = step * np.sqrt(3)
    
    points = []
    y = step # Start with some margin
    row = 0
    while y < 1.0:
        x = step
        offset = (row % 2) * (step / 2.0)
        x += offset
        while x < 1.0:
            points.append([x, y])
            x += step
        y += h_step
        row += 1
        
    points = np.array(points)
    
    # If we have more points than needed, select a subset?
    # Or just take the first n.
    # Better: select points that are well distributed.
    if len(points) > n:
        # Random shuffle and take first n to ensure diversity in restarts?
        # But for deterministic init, just take first n.
        # Actually, we want them to cover the square.
        # The grid generation might leave corners empty.
        # Let's just take the first n.
        centers[:n] = points[:n]
    else:
        centers[:len(points)] = points
        # Fill rest randomly if needed (unlikely)
        remaining = n - len(points)
        random_centers = rng.random((remaining, 2)) * 0.8 + 0.1
        centers[len(points):] = random_centers

    return centers

def objective_unequal(v, n):
    """
    Objective: Maximize sum of radii => Minimize -sum(radii)
    v structure: [x1..xn, y1..yn, r1..rn]
    """
    radii = v[2*n:]
    return -np.sum(radii)

def boundary_constraints(v, n):
    """
    Constraints:
    r_i <= x_i <= 1 - r_i  => x_i - r_i >= 0, 1 - x_i - r_i >= 0
    r_i <= y_i <= 1 - r_i  => y_i - r_i >= 0, 1 - y_i - r_i >= 0
    Returns array of constraint values (should be >= 0)
    """
    xs = v[:n]
    ys = v[n:2*n]
    rs = v[2*n:]
    
    # x - r >= 0
    c1 = xs - rs
    # 1 - x - r >= 0
    c2 = 1.0 - xs - rs
    # y - r >= 0
    c3 = ys - rs
    # 1 - y - r >= 0
    c4 = 1.0 - ys - rs
    
    return np.concatenate([c1, c2, c3, c4])

def overlap_constraints(v, n):
    """
    Constraints: dist(c_i, c_j) >= r_i + r_j
    Equivalent to: dist^2 - (r_i + r_j)^2 >= 0
    Returns array of constraint values
    """
    xs = v[:n]
    ys = v[n:2*n]
    rs = v[2*n:]
    
    constraints = []
    # Vectorized distance calculation might be faster but loop is clear for small n
    # To avoid O(N^2) python loop overhead if n was large, but 26 is small.
    # Let's do it efficiently.
    
    # Compute pairwise squared distances
    # xs is (N,), we can make it (N, 1)
    xs_col = xs[:, np.newaxis]
    ys_col = ys[:, np.newaxis]
    rs_col = rs[:, np.newaxis]
    
    # Diff matrices
    dx = xs_col - xs_col.T # (N, N)
    dy = ys_col - ys_col.T
    dr = rs_col + rs_col.T # (N, N)
    
    dist_sq = dx**2 + dy**2
    req_dist_sq = dr**2
    
    # We need dist_sq >= req_dist_sq for i < j
    # Create a mask for upper triangle (i < j)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    # Flatten the constraints corresponding to the mask
    vals = (dist_sq - req_dist_sq)[mask]
    
    return vals

def run_packing():
    n = N_CIRCLES
    
    best_sum = -1.0
    best_v = None
    
    # Try multiple initializations to find a good local optimum
    # Seed changes the random initialization if we used random, 
    # but here we use hex init. Let's perturb hex init.
    
    for seed in range(10): # 10 restarts
        # Generate initial centers
        centers = hexagonal_init(n, rng_seed=seed*100 + 1)
        
        # Initial radii: small enough to be valid, large enough to grow
        # Max possible radius for 26 circles is approx 0.1
        # Start with 0.02
        radii = np.full(n, 0.02)
        
        # Stack variables
        v0 = np.concatenate([centers.flatten(), radii])
        
        # Bounds for optimization
        # x, y in [0, 1]
        # r >= 0 (and effectively <= 0.5)
        bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
        
        # Define constraints
        # Boundary
        cons_boundary = {
            'type': 'ineq',
            'fun': lambda v, _n=n: boundary_constraints(v, _n)
        }
        
        # Overlap
        cons_overlap = {
            'type': 'ineq',
            'fun': lambda v, _n=n: overlap_constraints(v, _n)
        }
        
        constraints = [cons_boundary, cons_overlap]
        
        # Optimization
        # SLSQP is suitable for this
        try:
            res = minimize(
                objective_unequal,
                v0,
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
            )
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_v = res.x
        except Exception:
            continue

    if best_v is not None:
        final_centers = best_v[:2*n].reshape(n, 2)
        final_radii = best_v[2*n:]
        return final_centers, final_radii, np.sum(final_radii)
    
    # Fallback if optimization fails (should not happen)
    # Return a valid trivial solution
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    radii = np.full(n, 0.01)
    return centers, radii, np.sum(radii)

def overlap_func(v):

    rs = v[52:]
