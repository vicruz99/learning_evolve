import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(x):
    """Objective: minimize negative sum of radii"""
    return -np.sum(x[2*N_CIRCLES:])

def compute_constraints(x):
    """
    Computes all inequality constraints as a single flattened array.
    Returns values that must be >= 0.
    """
    n = N_CIRCLES
    cx = x[:n]
    cy = x[n:2*n]
    r = x[2*n:]
    
    c = []
    # Boundary constraints: cx >= r, cx <= 1-r, cy >= r, cy <= 1-r
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Pairwise non-overlap: dist^2 >= (r1 + r2)^2
    cx_diff = cx[:, None] - cx[None, :]
    cy_diff = cy[:, None] - cy[None, :]
    r_sum = r[:, None] + r[None, :]
    
    dist_sq = cx_diff**2 + cy_diff**2
    val = dist_sq - r_sum**2
    
    # Extract upper triangle (i < j) to avoid duplicates and self-checks
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.append(val[mask].flatten())
    
    return np.concatenate(c)

def run_packing():
    n = N_CIRCLES
    centers_init = []
    r_init = 0.09
    spacing = 0.19
    y = 0.1
    count = 0
    row_idx = 0
    
    # Hexagonal lattice initialization
    while count < n:
        cols = 6 if row_idx % 2 == 0 else 5
        if count + cols > n:
            cols = n - count
        x_start = 0.1 + (row_idx % 2) * spacing / 2
        for j in range(cols):
            centers_init.append([x_start + j * spacing, y])
            count += 1
        y += spacing * np.sqrt(3) / 2
        row_idx += 1

    centers_init = np.array(centers_init)
    # Small deterministic perturbation to break symmetry
    np.random.seed(42)
    centers_init += np.random.uniform(-0.005, 0.005, centers_init.shape)

    # Flatten variables: [x0, y0, ..., x25, y25, r0, ..., r25]
    x0 = np.concatenate([centers_init.flatten(), np.full(n, r_init)])
    bounds = [(0, 1)]*(2*n) + [(1e-4, 0.5)]*n

    cons = {'type': 'ineq', 'fun': compute_constraints}

    res = minimize(
        fun=compute_objective,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[cons],
        options={'maxiter': 3000, 'ftol': 1e-9, 'disp': False}
    )

    centers = res.x[:2*n].reshape(n, 2)
    radii = res.x[2*n:]
    
    # Clamp to strict bounds to handle numerical precision
    centers = np.clip(centers, radii, 1.0 - radii)
    radii = np.clip(radii, 1e-12, 0.5)
    
    return centers, radii, np.sum(radii)
