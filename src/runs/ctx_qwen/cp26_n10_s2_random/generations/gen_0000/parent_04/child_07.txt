import numpy as np
from scipy.optimize import minimize
import itertools

def compute_objective(params, n=26):
    """
    Objective function: -sum(radii) + penalty for overlaps and boundary violations.
    Vectorized for performance.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Boundary constraints: r <= min(x, 1-x, y, 1-y)
    d_bound = np.minimum(centers, 1.0 - centers)
    overlap_bound = np.maximum(0.0, radii - d_bound.min(axis=1))
    bound_penalty = np.sum(overlap_bound**2)
    
    # Pairwise constraints: r_i + r_j <= distance(i, j)
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_sum = radii[:, None] + radii[None, :]
    overlap_pair = np.maximum(0.0, r_sum - dists)
    
    # Sum squared violations over upper triangle only
    tri_u = np.triu_indices(n, k=1)
    pair_penalty = np.sum(overlap_pair[tri_u]**2)
    
    return -np.sum(radii) + 2000.0 * pair_penalty + 2000.0 * bound_penalty

def generate_initial_configs(n=26, num_random=12):
    """Generate diverse starting configurations to aid global search."""
    configs = []
    
    # 1. Grid-like configuration
    xs = np.linspace(0.1, 0.9, 5)
    ys = np.linspace(0.1, 0.9, 5)
    grid_centers = np.array(list(itertools.product(xs, ys)))
    extra = np.random.uniform(0.15, 0.85, size=(1, 2))
    centers_grid = np.vstack([grid_centers, extra])
    configs.append(np.hstack([centers_grid.flatten(), np.full(n, 0.05)]))
    
    # 2. Hexagonal lattice configuration
    hex_centers = []
    for i in range(5):
        for j in range(5):
            x = 0.1 + i * 0.2
            y = 0.1 + j * 0.1732 + (0.0866 if i % 2 == 1 else 0.0)
            if x < 0.95 and y < 0.95:
                hex_centers.append([x, y])
    while len(hex_centers) < n:
        hex_centers.append(np.random.uniform(0.1, 0.9, 2))
    centers_hex = np.array(hex_centers[:n])
    configs.append(np.hstack([centers_hex.flatten(), np.full(n, 0.05)]))
    
    # 3. Random configurations
    for _ in range(num_random):
        centers_rand = np.random.uniform(0.05, 0.95, size=(n, 2))
        configs.append(np.hstack([centers_rand.flatten(), np.full(n, 0.04)]))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    configs = generate_initial_configs(n, num_random=12)
    
    best_loss = np.inf
    best_params = None
    
    # Run optimization from each starting configuration
    for cfg in configs:
        res = minimize(compute_objective, cfg, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 800, 'ftol': 1e-10, 'gtol': 1e-8})
        if res.fun < best_loss:
            best_loss = res.fun
            best_params = res.x
            
    centers = best_params[:2*n].reshape(n, 2)
    radii = best_params[2*n:]
    
    # Post-processing: ensure strict validity within 1e-12 tolerance
    for _ in range(50):
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        r_sum = radii[:, None] + radii[None, :]
        max_viol_pair = np.max(r_sum - dists)
        
        d_bound = np.minimum(centers, 1.0 - centers)
        max_viol_bound = np.max(radii - d_bound.min(axis=1))
        
        max_viol = max(max_viol_pair, max_viol_bound)
        if max_viol <= 1e-10:
            break
            
        # Reduce all radii uniformly to clear violations safely
        radii = np.maximum(radii - max_viol / 2.0, 1e-7)
        
    return centers, radii, float(np.sum(radii))
