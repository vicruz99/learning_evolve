import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

N_CIRCLES = 26

def obj_fun(x):
    return -np.sum(x[:N_CIRCLES])

def boundary_constraints(x, n):
    radii = x[:n]
    cx = x[n:2*n]
    cy = x[2*n:3*n]
    con = np.zeros(4*n)
    for i in range(n):
        con[4*i] = cx[i] - radii[i]
        con[4*i + 1] = 1.0 - cx[i] - radii[i]
        con[4*i + 2] = cy[i] - radii[i]
        con[4*i + 3] = 1.0 - cy[i] - radii[i]
    return con

def overlap_constraints(x, n):
    radii = x[:n]
    cx = x[n:2*n]
    cy = x[2*n:3*n]
    m = n * (n - 1) // 2
    con = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            con[idx] = np.sqrt(dx * dx + dy * dy) - radii[i] - radii[j]
            idx += 1
    return con

def rad_pos_constraints(x, n):
    return x[:n] - 1e-8

def bc_fun(x):
    return boundary_constraints(x, N_CIRCLES)

def oc_fun(x):
    return overlap_constraints(x, N_CIRCLES)

def rc_fun(x):
    return rad_pos_constraints(x, N_CIRCLES)

def make_hex_centers(r, n, stretch):
    centers = []
    for row in range(8):
        y = r + row * r * np.sqrt(3) * stretch
        if y + r > 1.0:
            break
        if row % 2 == 0:
            x_start = r
        else:
            x_start = r + r * 0.5
        x = x_start
        while x + r <= 1.0 and len(centers) < n:
            centers.append([x, y])
            x += 2 * r
    while len(centers) < n:
        centers.append([0.5, 0.5])
    return centers[:n]

def make_grid_centers(r, n, cols, rows):
    centers = []
    dx = (1.0 - 2 * r) / (cols - 1) if cols > 1 else 0
    dy = (1.0 - 2 * r) / (rows - 1) if rows > 1 else 0
    for row in range(rows):
        for col in range(cols):
            if len(centers) >= n:
                return centers
            x = r + col * dx
            y = r + row * dy
            centers.append([x, y])
    while len(centers) < n:
        centers.append([0.5, 0.5])
    return centers[:n]

def run_optimization(x0, n):
    dim = 3 * n
    bc = NonlinearConstraint(bc_fun, 0, np.inf)
    oc = NonlinearConstraint(oc_fun, 0, np.inf)
    rc = NonlinearConstraint(rc_fun, 0, np.inf)
    
    bounds = [(1e-8, 0.5)] * n + [(0.0, 1.0)] * n + [(0.0, 1.0)] * n
    
    best_result = None
    best_val = np.inf
    
    for method in ['SLSQP', 'trust-constr']:
        try:
            result = minimize(obj_fun, x0.copy(), method=method,
                             constraints=[bc, oc, rc], bounds=bounds,
                             options={'maxiter': 10000, 'ftol': 1e-15, 'xtol': 1e-15})
            if result.fun < best_val:
                best_val = result.fun
                best_result = result
        except Exception:
            pass
    
    if best_result is None:
        return x0, np.inf
    
    return best_result.x, best_result.fun

    sum_radii = float(np.sum(radii))
