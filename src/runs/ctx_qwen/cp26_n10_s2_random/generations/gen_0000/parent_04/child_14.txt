    centers = np.random.rand(n, 2)

    num_iterations = 5000

    decay = 0.9995

                    force[i] += diff * 10.0
                    force[j] -= diff * 10.0

                    # Gentle repulsion for proximity
                    force[i] += diff
                    force[j] -= diff

            x, y = centers[i]
            r = radii[i]
