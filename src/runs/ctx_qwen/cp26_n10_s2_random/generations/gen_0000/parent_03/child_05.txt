import numpy as np
from scipy.optimize import minimize

def compute_obj(v):
    """Objective function: minimize negative sum of radii."""
    n = len(v) // 3
    return -np.sum(v[2::3])

def compute_constraints(v):
    """Compute all inequality constraints: boundary and non-overlap."""
    n = len(v) // 3
    P = v.reshape(n, 3)
    C = P[:, :2]
    R = P[:, 2]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    boundary = np.concatenate([
        C[:, 0] - R,
        1.0 - C[:, 0] - R,
        C[:, 1] - R,
        1.0 - C[:, 1] - R
    ])
    
    # Pairwise non-overlap constraints: dist^2 - (r1+r2)^2 >= 0
    diff = C[:, np.newaxis, :] - C[np.newaxis, :, :]
    D2 = np.sum(diff**2, axis=2)
    R_sum = R[:, np.newaxis] + R[np.newaxis, :]
    
    idx = np.triu_indices(n, k=1)
    pairwise = D2[idx] - R_sum[idx]**2
    
    return np.concatenate([boundary, pairwise])

def run_packing():
    n = 26
    best_sum = -1.0
    best_v = None
    
    # Generate initial hexagonal lattice
    centers = []
    y = 0
    while len(centers) < n:
        x = 0
        while x <= 1 and len(centers) < n:
            centers.append([x, y])
            x += np.sqrt(3)/2
        y += 1
    centers = np.array(centers[:n])
    
    # Scale and shift to fit comfortably inside [0.1, 0.9]
    c_min = centers.min(axis=0)
    c_max = centers.max(axis=0)
    spread = max(c_max[0] - c_min[0], c_max[1] - c_min[1])
    scale = 0.8 / spread if spread > 0 else 1.0
    centers = (centers - c_min) * scale + 0.1
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    constr = {'type': 'ineq', 'fun': compute_constraints}
    
    # Multi-start optimization to escape local minima
    for seed in range(10):
        np.random.seed(42 + seed)
        pert_centers = centers + np.random.uniform(-0.02, 0.02, centers.shape)
        pert_centers = np.clip(pert_centers, 0.1, 0.9)
        
        v0 = np.zeros(3*n)
        for i in range(n):
            v0[3*i] = pert_centers[i, 0]
            v0[3*i+1] = pert_centers[i, 1]
            v0[3*i+2] = 0.06  # Initial radius
            
        res = minimize(compute_obj, v0, method='SLSQP', bounds=bounds, constraints=[constr],
                       options={'maxiter': 500, 'ftol': 1e-10})
                       
        cur_sum = -res.fun
        if cur_sum > best_sum:
            # Verify feasibility with a slight tolerance for numerical errors
            if np.all(compute_constraints(res.x) >= -1e-6):
                best_sum = cur_sum
                best_v = res.x
                
    if best_v is not None:
        P = best_v.reshape(n, 3)
        final_centers = P[:, :2]
        final_radii = np.maximum(P[:, 2], 0.0)
        return final_centers, final_radii, best_sum
        
    # Fallback solution
    return centers, np.full(n, 0.06), 0.06 * n
