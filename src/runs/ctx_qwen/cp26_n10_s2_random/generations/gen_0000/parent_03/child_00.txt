def objective(vars_arr):

    radii = vars_arr[2::3]
    return -np.sum(radii)

def constraints_all(vars_arr):
    n = 26
    c = []
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    for i in range(n):
        x = vars_arr[3 * i]
        y = vars_arr[3 * i + 1]
        r = vars_arr[3 * i + 2]
        c.append(x - r)
        c.append(1.0 - x - r)
        c.append(y - r)
        c.append(1.0 - y - r)
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            xi = vars_arr[3 * i]
            yi = vars_arr[3 * i + 1]
            ri = vars_arr[3 * i + 2]
            xj = vars_arr[3 * j]
            yj = vars_arr[3 * j + 1]
            rj = vars_arr[3 * j + 2]
            dx = xi - xj
            dy = yi - yj
            min_dist = ri + rj
            c.append(dx * dx + dy * dy - min_dist * min_dist)
    return np.array(c)

def create_initial_vars(centers, radii):

def hexagonal_init():

    idx = 0
    y = 0.07
    row = 0

        while x <= 0.95 and idx < n:
            centers[idx] = [x, y]

            x += 0.13

def random_init(seed):
    rng = np.random.RandomState(seed)
    n = 26
    centers = rng.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, 0.05)

def force_optimize(centers, radii, steps=20000):
    n = 26
    centers = centers.copy()
    radii = radii.copy()
    
    for step in range(steps):
        forces = np.zeros((n, 2))
        
        # Repulsive forces between overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    force_mag = overlap * 200.0
                    inv_dist = 1.0 / dist
                    forces[i, 0] -= dx * inv_dist * force_mag
                    forces[i, 1] -= dy * inv_dist * force_mag
                    forces[j, 0] += dx * inv_dist * force_mag
                    forces[j, 1] += dy * inv_dist * force_mag
        
        # Boundary repulsion
        for i in range(n):
            x = centers[i, 0]
            y = centers[i, 1]
            r = radii[i]
            if x - r < 0:
                forces[i, 0] += (r - x) * 300.0
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) * 300.0
            if y - r < 0:
                forces[i, 1] += (r - y) * 300.0
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) * 300.0
        
        # Apply forces with decaying learning rate
        lr = 0.005 * (1.0 - step / steps) ** 0.5
        centers += forces * lr
        centers = np.clip(centers, 1e-6, 1 - 1e-6)
        
        # Gradually expand radii
        radii *= 1.0 + 0.00002
    
    return centers, radii

def is_valid(centers, radii):
    n = centers.shape[0]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < -1e-12:
            return False
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True
