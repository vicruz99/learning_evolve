import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize sum of radii.
    """
    n = 26
    best_sum_r = 0.0
    best_centers = None
    best_radii = None
    
    # Number of random restarts to find a good local optimum
    num_restarts = 10
    
    for attempt in range(num_restarts):
        # Generate random initial centers
        # Using a fixed seed based on attempt for reproducibility if needed, 
        # but varying seed allows exploring different landscapes.
        rng = np.random.default_rng(attempt * 42)
        centers = rng.random((n, 2))
        
        # Initialize radii to a small value to ensure initial feasibility
        # (random points in unit square are typically separated by > 0.1)
        radii = np.full(n, 0.01)
        
        # Flatten variables into a single vector: [x1, y1, r1, x2, y2, r2, ...]
        x0 = []
        for i in range(n):
            x0.extend([centers[i, 0], centers[i, 1], radii[i]])
        x0 = np.array(x0)
        
        # Define bounds: x, y in [0, 1], r in [0, 0.5]
        bounds = []
        for _ in range(n):
            bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
        # Objective function: minimize negative sum of radii
        def objective(vars_vec):
            # Extract radii (every 3rd element starting from index 2)
            r = vars_vec[2::3]
            return -np.sum(r)
        
        # Constraint function
        # Returns a concatenated array of constraint values.
        # All constraints must be >= 0.
        def get_constraints(vars_vec):
            # Reshape variables into (n, 3) matrix: [x, y, r]
            data = vars_vec.reshape(n, 3)
            c = data[:, :2] # Centers (n, 2)
            r = data[:, 2]  # Radii (n,)
            
            constraints = []
            
            # 1. Boundary constraints
            # Circle must be inside [0,1]x[0,1]
            # x - r >= 0
            constraints.append(c[:, 0] - r)
            # 1 - x - r >= 0
            constraints.append(1.0 - c[:, 0] - r)
            # y - r >= 0
            constraints.append(c[:, 1] - r)
            # 1 - y - r >= 0
            constraints.append(1.0 - c[:, 1] - r)
            
            # 2. Non-overlap constraints
            # Distance squared between centers >= (r_i + r_j)^2
            # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
            
            # Compute pairwise difference matrix (n, n, 2)
            # c[i] - c[j]
            diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
            
            # Compute squared Euclidean distances (n, n)
            dist_sq = np.sum(diff**2, axis=2)
            
            # Compute squared sum of radii (n, n)
            # r_sum[i, j] = r_i + r_j
            r_sum = r[:, np.newaxis] + r[np.newaxis, :]
            r_sum_sq = r_sum**2
            
            # We only need constraints for i < j (upper triangle)
            iu, ju = np.triu_indices(n, k=1)
            overlap_constraints = dist_sq[iu, ju] - r_sum_sq[iu, ju]
            
            constraints.append(overlap_constraints)
            
            return np.concatenate(constraints)
        
        constraint_dict = {
            'type': 'ineq',
            'fun': get_constraints
        }
        
        # Run SLSQP optimization
        # ftol sets tolerance for change in function value
        # maxiter limits iterations
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraint_dict, 
                           options={'ftol': 1e-10, 'maxiter': 2000, 'disp': False})
            
            # Check if optimization was successful or at least gave a result
            # SLSQP might return with exit status 1 or 2 if it converges to a bound or tolerance
            if not np.isnan(res.fun):
                current_sum_r = -res.fun
                
                # Extract results
                res_data = res.x.reshape(n, 3)
                res_centers = res_data[:, :2]
                res_radii = res_data[:, 2]
                
                # Basic validation check (approximate)
                # Check for negative radii (shouldn't happen due to bounds but good to be safe)
                if np.any(res_radii < -1e-9):
                    continue
                
                # Check sum is better

                    best_sum_r = current_sum_r
                    best_centers = res_centers.copy()
                    best_radii = res_radii.copy()
                    
        except Exception:
            continue

    # If no valid solution found (unlikely), fallback to simple grid
    if best_centers is None:
        # Fallback: 5x5 grid + 1
        centers_fallback = []
        radii_fallback = []
        count = 0

        # Add one small circle

        best_centers = np.array(centers_fallback)
        best_radii = np.array(radii_fallback)
        best_sum_r = np.sum(best_radii)

    return best_centers, best_radii, best_sum_r
