import numpy as np
from scipy.optimize import minimize

def get_hexagonal_init(n):
    """
    Generates a hexagonal lattice initialization for n circles in a unit square.
    """
    centers = []
    r_init = 0.05
    
    # Estimate radius to fit n circles roughly
    # Area of square = 1. Density ~ 0.9. 
    # n * pi * r^2 * 0.9 approx 1 => r approx sqrt(1 / (0.9 * n * pi))
    r_est = np.sqrt(1 / (0.9 * n * np.pi))
    
    # We try to fit a hexagonal grid
    # Row spacing: sqrt(3)/2 * diameter
    # We determine number of rows and cols based on estimated r
    # Let's try to place them
    
    # Simple approach: iterate y, then x
    # y starts at r_est, increments by sqrt(3)/2 * 2*r_est = sqrt(3)*r_est
    # x starts at r_est, increments by 2*r_est
    # Offset alternate rows by r_est
    
    # Let's pick a slightly larger r to fill space, optimizer will shrink if needed
    # But r_est is a good guide. Let's use a slightly smaller r for safe start
    r_start = r_est * 0.8
    
    # Grid search for positions
    y = r_start
    row = 0
    while y + r_start <= 1.0:
        x = r_start
        # Check if row is offset (hexagonal packing)
        if row % 2 == 1:
            x += r_start # Shift by radius (half diameter)
        
        while x + r_start <= 1.0:
            centers.append([x, y])
            if len(centers) >= n:
                break
            x += 2 * r_start
        if len(centers) >= n:
            break
        y += np.sqrt(3) * r_start
        row += 1
    
    # If we don't have enough, fill with random or reset
    if len(centers) < n:
        # Fallback to random
        centers = np.random.rand(n, 2) * 0.8 + 0.1
        r_start = 0.05
    
    return np.array(centers[:n]), r_start

def objective(vars, n):
    """
    Objective function to minimize: -sum(radii)
    vars: [x0, y0, r0, x1, y1, r1, ...]
    """
    radii = vars[2::3]
    return -np.sum(radii)

def constraints(vars, n):
    """
    Returns a list of constraint dictionaries for SLSQP.
    Constraints must be >= 0.
    """
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = vars[3*i]
        centers[i, 1] = vars[3*i+1]
        radii[i] = vars[3*i+2]
    
    cons = []
    
    # Boundary constraints
    for i in range(n):
        # x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
        # 1 - x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]})
        # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
        # 1 - y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]})
        
        # r >= 0 (handled by bounds, but good for safety)
        # cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+2]})

    # Non-overlap constraints
    # dist^2 >= (r_i + r_j)^2  =>  dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            def non_overlap(v, i=i, j=j):
                xi, yi = v[3*i], v[3*i+1]
                xj, yj = v[3*j], v[3*j+1]
                ri, rj = v[3*i+2], v[3*j+2]
                
                dx = xi - xj
                dy = yi - yj
                dist_sq = dx*dx + dy*dy
                sum_r = ri + rj
                return dist_sq - sum_r*sum_r
            
            cons.append({'type': 'ineq', 'fun': non_overlap})
            
    return cons
