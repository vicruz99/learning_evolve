import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns a packing of 26 circles in a unit square that maximizes the sum of radii.
    Uses simulated annealing to optimize an irregular, high-density configuration.
    """
    np.random.seed(42)
    n = 26
    num_iter = 5000
    
    # Start with a grid and add noise to escape the rigid structure
    # A 5x5 grid is too tight for 26 circles, so we add the 26th near the center
    # and perturb everything to allow for denser packing.
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < n:
                centers[idx] = [0.1 + i * 0.2, 0.1 + j * 0.2]
                idx += 1
    
    # Add noise to all centers to break symmetry and allow optimization
    centers += np.random.uniform(-0.05, 0.05, size=centers.shape)
    
    # Initialize with a radius that is valid for the perturbed grid
    current_r = 0.095 
    best_r = current_r
    best_centers = centers.copy()
    
    # Simulated Annealing Parameters
    temperature = 0.05
    temp_min = 1e-6
    cool_rate = 0.995
    expansion_step = 2e-5
    
    # Run optimization
    for _ in range(num_iter):
        # Increase radius slowly (expansion pressure)
        current_r += expansion_step
        
        # Attempt to move a random circle
        i = np.random.randint(n)
        old_pos = centers[i].copy()
        
        # Proposed move: random direction scaled by temperature
        centers[i] += np.random.uniform(-temperature, temperature, size=2)
        
        # Check validity (overlap and boundaries)
        valid = True
        # Boundary check
        if np.any(centers[i] - current_r < -1e-9) or np.any(centers[i] + current_r > 1.0 + 1e-9):
            valid = False
        
        # Overlap check
        if valid:
            dists = np.linalg.norm(centers[i] - centers, axis=1)
            dists[i] = float('inf') # Ignore self
            if np.any(dists < 2 * current_r - 1e-9):
                valid = False
        
        if valid:
            # Move accepted
            if current_r > best_r:
                best_r = current_r
                best_centers = centers.copy()
        else:
            # Move rejected, revert
            centers[i] = old_pos
            
        # Cool down
        temperature *= cool_rate
        if temperature < temp_min:
            temperature = temp_min

    return best_centers, np.full(n, best_r), best_r * n
