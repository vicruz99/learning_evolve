The problem is to maximize $\sum r_i$ subject to boundary and non-overlap constraints. This is a non-convex optimization problem. A robust approach is:
1. **Formulate as NLP**: Variables are $(r_i, x_i, y_i)$ for all $i$. Objective is $-\sum r_i$. Constraints enforce boundaries and pairwise distances $\ge r_i + r_j$.
2. **Initialization**: Start with a structured grid layout perturbed by small random noise to break symmetry and avoid degenerate local minima.
3. **NLP Solver**: Use `scipy.optimize.minimize` with SLSQP, which handles inequality constraints well.
4. **LP Refinement**: After finding good centers, the optimal radii for fixed centers are given by a Linear Program: maximize $\sum r_i$ s.t. $r_i + r_j \le d_{ij}$ and $r_i \le \text{dist-to-boundary}_i$. Solving this LP guarantees strict feasibility and extracts the maximum possible sum for the computed geometry.
5. **Implementation Details**: Vectorize constraint evaluation, use tight tolerances, and ensure all helper functions are top-level as requested.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(v):
    # Minimize negative sum of radii to maximize sum
    return -np.sum(v[0::3])

def constraints(v):
    r = v[0::3]
    x = v[1::3]
    y = v[2::3]
    
    # Boundary constraints: x >= r, x + r <= 1, y >= r, y + r <= 1
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Non-overlap constraints: dist(i,j) >= r_i + r_j
    coords = np.column_stack((x, y))
    diff = coords[:, np.newaxis, :] - coords[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    iu, ju = np.triu_indices(N_CIRCLES, k=1)
    c = np.concatenate([c, dists[iu, ju] - (r[iu] + r[ju])])
    return c

def run_packing():
    # Initial guess: Grid with small random perturbation
    v0 = np.zeros(3 * N_CIRCLES)
    pts = []
    for i in range(5):
        for j in range(5):
            if len(pts) == N_CIRCLES:
                break
            pts.append((0.1 + j * 0.2, 0.1 + i * 0.2))
            
    rng = np.random.default_rng(42)
    for i, (px, py) in enumerate(pts):
        v0[3*i] = 0.06
        v0[3*i+1] = np.clip(px + rng.uniform(-0.02, 0.02), 0.05, 0.95)
        v0[3*i+2] = np.clip(py + rng.uniform(-0.02, 0.02), 0.05, 0.95)
        
    bounds = [(0, 0.5) if i % 3 == 0 else (0, 1) for i in range(3 * N_CIRCLES)]
    
    # NLP optimization for centers and radii
    res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                   constraints={'type': 'ineq', 'fun': constraints},
                   options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
                   
    centers = np.column_stack((res.x[1::3], res.x[2::3]))
    radii = res.x[0::3]
    
    # LP refinement to exactly maximize radii for fixed centers
    # This guarantees strict feasibility and often improves the sum
    c_lp = -np.ones(N_CIRCLES)
    A_ub = []
    b_ub = []
    
    # Boundary constraints for radii: r_i <= min(x, 1-x, y, 1-y)
    bounds_i = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                          np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds_i = np.maximum(0.0, bounds_i)
    
    A_ub.append(np.eye(N_CIRCLES))
    b_ub.append(bounds_i)
    
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    iu, ju = np.triu_indices(N_CIRCLES, k=1)
    
    for i, j in zip(iu, ju):
        row = np.zeros(N_CIRCLES)
        row[i] = 1.0
        row[j] = 1.0
        A_ub.append(row)
        b_ub.append(dists[i, j])
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    res_lp = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, bounds=(0, 0.5), method='highs')
    if res_lp.success:
        radii = res_lp.x
        
    return centers, radii, np.sum(radii)
