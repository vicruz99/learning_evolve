

Looking at this problem, I need to pack 26 circles in a unit square to maximize the sum of radii, targeting 2.636.

## Strategy

The key insight is that this is a non-convex optimization problem with 78 variables (26 centers × 2 coords + 26 radii) and ~429 inequality constraints. A direct SLSQP approach may struggle with this many constraints and get stuck in local optima.

My strategy:
1. **Initial Configuration**: Start from a structured hexagonal-like packing (more efficient than square grid), with small radii to ensure feasibility
2. **Optimization**: Use scipy's SLSQP with careful constraint formulation (squared distances to avoid sqrt, small tolerance for numerical stability)
3. **Multiple Restarts**: Try different initial arrangements to escape local optima
4. **Force-based Refinement**: After optimization, apply a physics-inspired relaxation to squeeze out more space

import numpy as np
from scipy.optimize import minimize
from functools import partial


def objective(x, n):
    return -np.sum(x[2 * n:])


def constraint_boundary_left(x, n, i):
    return x[2 * i] - x[2 * n + i]


def constraint_boundary_right(x, n, i):
    return 1 - x[2 * i] - x[2 * n + i]


def constraint_boundary_bottom(x, n, i):
    return x[2 * i + 1] - x[2 * n + i]


def constraint_boundary_top(x, n, i):
    return 1 - x[2 * i + 1] - x[2 * n + i]


def constraint_no_overlap(x, n, i, j):
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist_sq = dx * dx + dy * dy
    r_sum = x[2 * n + i] + x[2 * n + j]
    return dist_sq - r_sum * r_sum + 1e-10


def build_constraints(n):
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': partial(constraint_boundary_left, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_boundary_right, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_boundary_bottom, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_boundary_top, n=n, i=i)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': partial(constraint_no_overlap, n=n, i=i, j=j)})

    return constraints


def hexagonal_init(n, seed_val):
    np.random.seed(seed_val)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.025

    idx = 0
    for row in range(8):
        if row % 2 == 0:
            ncols = 4
            x_start = 0.1
        else:
            ncols = 5
            x_start = 0.15

        for col in range(ncols):
            if idx < n:
                centers[idx, 0] = x_start + col * 0.18 + np.random.uniform(-0.01, 0.01)
                centers[idx, 1] = 0.08 + row * 0.12 + np.random.uniform(-0.01, 0.01)
                idx += 1

    centers = np.clip(centers, 0.05, 0.95)
    return centers, radii


def dense_grid_init(n, seed_val):
    np.random.seed(seed_val + 100)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.02

    # 5x5 grid with perturbation
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < n:
                centers[idx, 0] = 0.1 + j * 0.2 + np.random.uniform(-0.03, 0.03)
                centers[idx, 1] = 0.1 + i * 0.2 + np.random.uniform(-0.03, 0.03)
                idx += 1
    if idx < n:
        centers[idx, 0] = 0.5 + np.random.uniform(-0.05, 0.05)
        centers[idx, 1] = 0.5 + np.random.uniform(-0.05, 0.05)

    centers = np.clip(centers, 0.05, 0.95)
    return centers, radii


def corner_biased_init(n, seed_val):
    np.random.seed(seed_val + 200)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.025

    # Place some circles near corners and edges
    corners = [(0.08, 0.08), (0.92, 0.08), (0.08, 0.92), (0.92, 0.92)]
    idx = 0
    for cx, cy in corners:
        for dx in range(3):
            for dy in range(3):
                if idx < n:
                    centers[idx, 0] = cx + dx * 0.12 + np.random.uniform(-0.01, 0.01)
                    centers[idx, 1] = cy + dy * 0.12 + np.random.uniform(-0.01, 0.01)
                    idx += 1

    # Fill remaining in center
    while idx < n:
        centers[idx, 0] = 0.5 + np.random.uniform(-0.15, 0.15)
        centers[idx, 1] = 0.5 + np.random.uniform(-0.15, 0.15)
        idx += 1

    centers = np.clip(centers, 0.05, 0.95)
    return centers, radii


def force_based_refine(centers, radii, n, iterations=500):
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(iterations):
        # Try to grow radii slightly
        growth_rate = 0.00005 * (1 - iteration / iterations)
        radii += growth_rate

        # Apply repulsion forces for overlaps
        for _ in range(20):
            violated = False
            # Boundary corrections
            for i in range(n):
                for coord in range(2):
                    if centers[i, coord] - radii[i] < 0:
                        centers[i, coord] = radii[i]
                        violated = True
                    if centers[i, coord] + radii[i] > 1:
                        centers[i, coord] = 1 - radii[i]
                        violated = True

            # Overlap corrections
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    min_dist = radii[i] + radii[j]

                    if dist < min_dist and dist > 1e-12:
                        overlap = min_dist - dist
                        nx = dx / dist
                        ny = dy / dist
                        shift = overlap * 0.5 + 1e-8
                        centers[i, 0] += nx * shift
                        centers[i, 1] += ny * shift
                        centers[j, 0] -= nx * shift
                        centers[j, 1] -= ny * shift
                        violated = True

            if not violated:
                break

    return centers, radii


def run_packing():
    n = 26
    best_sum = 0
    best_centers = None
    best_radii = None

    constraints = build_constraints(n)
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    init_funcs = [hexagonal_init, dense_grid_init, corner_biased_init]

    for init_idx in range(len(init_funcs)):
        for trial in range(3):
            seed_val = init_idx * 50 + trial * 7 + 42
            centers, radii = init_funcs[init_idx](n, seed_val)

            x0 = np.concatenate([centers.flatten(), radii])

            result = minimize(
                objective, x0, args=(n,),
                bounds=bounds, constraints=constraints,
                method='SLSQP',
                options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
            )

            sum_r = -result.fun
            if sum_r > best_sum:
                best_sum = sum_r
                best_centers = result.x[:2 * n].reshape(n, 2).copy()
                best_radii = result.x[2 * n:].copy()

    # Force-based refinement on best result
    best_centers, best_radii = force_based_refine(best_centers, best_radii, n, iterations=800)

    # Second round of optimization from refined position
    x0 = np.concatenate([best_centers.flatten(), best_radii])
    result = minimize(
        objective, x0, args=(n,),
        bounds=bounds, constraints=constraints,
        method='SLSQP',
        options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
    )

    if -result.fun > best_sum:
        best_centers = result.x[:2 * n].reshape(n, 2).copy()
        best_radii = result.x[2 * n:].copy()
        best_sum = -result.fun

    # One more force-based refinement
    best_centers, best_radii = force_based_refine(best_centers, best_radii, n, iterations=300)
    best_sum = np.sum(best_radii)

    # Final optimization round
    x0 = np.concatenate([best_centers.flatten(), best_radii])
    result = minimize(
        objective, x0, args=(n,),
        bounds=bounds, constraints=constraints,
        method='SLSQP',
        options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
    )

    if -result.fun > best_sum:
        best_centers = result.x[:2 * n].reshape(n, 2).copy()
        best_radii = result.x[2 * n:].copy()
        best_sum = -result.fun

    best_radii = np.maximum(best_radii, 0)
    best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
