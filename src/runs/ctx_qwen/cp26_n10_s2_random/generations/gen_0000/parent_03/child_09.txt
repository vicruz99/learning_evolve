import numpy as np
from scipy.optimize import minimize

N = 26

def compute_penalty(vars, weight):
    cx = vars[:2*N]
    centers = cx.reshape(N, 2)
    r = vars[-1]
    pen = 0.0
    # Boundary constraints
    for i in range(N):
        x, y = centers[i]
        if x < r: 
            pen += (r - x)**2
        if x > 1 - r: 
            pen += (x - (1 - r))**2
        if y < r: 
            pen += (r - y)**2
        if y > 1 - r: 
            pen += (y - (1 - r))**2
    # Pairwise non-overlap constraints
    for i in range(N):
        for j in range(i + 1, N):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if d < 2 * r:
                pen += (2 * r - d)**2
    # Objective: maximize r (minimize -r) + penalty
    return -r + weight * pen

def run_packing():
    np.random.seed(42)
    
    # Initial guess: grid layout with small random perturbations
    r0 = 0.09
    x_vals = np.linspace(0.1, 0.9, 6)
    y_vals = np.linspace(0.1, 0.9, 5)
    cx, cy = np.meshgrid(x_vals, y_vals)
    centers0 = np.column_stack([cx.ravel(), cy.ravel()])[:N]
    centers0 += np.random.uniform(-0.02, 0.02, centers0.shape)
    
    x_init = np.concatenate([centers0.flatten(), [r0]])
    bounds = [(0, 1)] * (2*N) + [(0, 0.5)]
    
    # Stage 1: Coarse optimization to escape bad initial configurations
    res1 = minimize(compute_penalty, x_init, args=(100.0,), method='L-BFGS-B', bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-6})
    
    # Stage 2: Fine optimization with high penalty to tightly enforce constraints and maximize radius
    res2 = minimize(compute_penalty, res1.x, args=(1000.0,), method='L-BFGS-B', bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-9})
    
    centers_opt = res2.x[:2*N].reshape(N, 2)
    
    # Compute the strict maximum valid radius for the optimized positions
    r_valid = 1.0
    for i in range(N):
        x, y = centers_opt[i]
        r_valid = min(r_valid, x, 1 - x, y, 1 - y)
        for j in range(i + 1, N):
            d = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
            r_valid = min(r_valid, d / 2)
            
    radii_opt = np.full(N, r_valid)
    return centers_opt, radii_opt, np.sum(radii_opt)
