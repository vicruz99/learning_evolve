import numpy as np
from scipy.optimize import minimize

def objective(x, n):
    """Negative sum of radii (to be minimized)"""
    return -np.sum(x[2*n:])

def constraints(x, n):
    """Returns array of constraint values that must be >= 0"""
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    c_list = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c_list.extend(centers[:, 0] - radii)
    c_list.extend(1 - centers[:, 0] - radii)
    c_list.extend(centers[:, 1] - radii)
    c_list.extend(1 - centers[:, 1] - radii)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2 for i < j
    diff = centers[:, None, :] - centers[None, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum = radii[:, None] + radii[None, :]
    overlap = dist_sq - r_sum**2
    c_list.extend(overlap[np.triu_indices(n, k=1)])
    
    return np.array(c_list)

def run_packing():
    n = 26
    # Bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    best_sum = 0.0
    best_x = None
    
    starts = []
    
    # 1. Random initial configurations
    for i in range(10):
        np.random.seed(i)
        centers = np.random.uniform(0.1, 0.9, (n, 2))
        radii = np.ones(n) * 0.03  # Small initial radii to ensure feasibility
        starts.append(np.concatenate([centers.flatten(), radii]))
        
    # 2. Hexagonal lattice initial configuration
    centers = np.zeros((n, 2))
    y = 0.1
    idx = 0
    rows = 0
    while idx < n:
        n_cols = 5 if rows % 2 == 0 else 4
        x = 0.1 + (0.1 * (rows % 2))
        for _ in range(n_cols):
            if idx >= n: break
            centers[idx] = [x, y]
            x += 0.2
            idx += 1
        y += 0.1 * np.sqrt(3)
        rows += 1
    radii = np.ones(n) * 0.05
    starts.append(np.concatenate([centers.flatten(), radii]))

    # Optimization loop
    for x0 in starts:
        try:
            res = minimize(objective, x0, args=(n,), method='SLSQP', 
                          bounds=bounds, 
                          constraints={'type': 'ineq', 'fun': constraints, 'args': (n,)},
                          options={'maxiter': 1500, 'ftol': 1e-9})
            
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_x = res.x
        except Exception:
            continue
            
    if best_x is not None:
        centers = best_x[:2*n].reshape(n, 2)
        radii = best_x[2*n:]
        radii = np.maximum(radii, 0.0)
        return centers, radii, float(np.sum(radii))
    
    # Fallback (should not be reached)
    centers = np.random.uniform(0, 1, (n, 2))
    radii = np.ones(n) * 0.01
    return centers, radii, float(np.sum(radii))
