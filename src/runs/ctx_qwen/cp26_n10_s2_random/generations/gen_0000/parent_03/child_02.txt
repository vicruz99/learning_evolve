import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def objective(vars):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(vars[2::3])

def con_bounds(vars):
    """Boundary constraints: circles inside [0,1]x[0,1]"""
    n = len(vars) // 3
    vals = np.zeros(4 * n)
    for i in range(n):
        x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
        vals[4*i] = x - r          # x >= r
        vals[4*i+1] = 1.0 - x - r  # x + r <= 1
        vals[4*i+2] = y - r        # y >= r
        vals[4*i+3] = 1.0 - y - r  # y + r <= 1
    return vals

def con_pairs(vars):
    """Pairwise non-overlap constraints: dist^2 >= (r1+r2)^2"""
    n = len(vars) // 3
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    dist_sq = dx**2 + dy**2
    req_sq = dr**2
    
    # Only need upper triangle (i < j)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    return dist_sq[mask] - req_sq[mask]

def run_packing():
    n = 26
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    
    con1 = NonlinearConstraint(con_bounds, 0, np.inf)
    con2 = NonlinearConstraint(con_pairs, 0, np.inf)
    
    best_sum = -np.inf
    best_vars = None
    
    # --- Initialization 1: Hexagonal Lattice ---
    centers1 = np.zeros((n, 2))
    row_counts = [5, 6, 5, 6, 4]
    y_vals = np.linspace(0.15, 0.85, len(row_counts))
    idx = 0
    for y, cnt in zip(y_vals, row_counts):
        x_start = (1.0 - cnt * 0.2) / 2
        for k in range(cnt):
            centers1[idx] = [x_start + k * 0.2, y]
            idx += 1
            
    x0_1 = np.zeros(3 * n)
    for i in range(n):
        x0_1[3*i] = centers1[i, 0]
        x0_1[3*i+1] = centers1[i, 1]
        x0_1[3*i+2] = 0.09  # Strictly feasible initial radius
        
    res1 = minimize(objective, x0_1, method='trust-constr', bounds=bounds, 
                    constraints=[con1, con2], options={'maxiter': 1500, 'verbose': 0})
    if res1.success and -res1.fun > best_sum:
        best_sum = -res1.fun
        best_vars = res1.x
        
    # --- Initialization 2: Perturbed Hexagonal ---
    np.random.seed(42)
    centers2 = centers1 + np.random.normal(0, 0.015, centers1.shape)
    x0_2 = np.zeros(3 * n)
    for i in range(n):
        x0_2[3*i] = np.clip(centers2[i, 0], 0.05, 0.95)
        x0_2[3*i+1] = np.clip(centers2[i, 1], 0.05, 0.95)
        x0_2[3*i+2] = 0.09
        
    res2 = minimize(objective, x0_2, method='trust-constr', bounds=bounds, 
                    constraints=[con1, con2], options={'maxiter': 1500, 'verbose': 0})
    if res2.success and -res2.fun > best_sum:
        best_sum = -res2.fun
        best_vars = res2.x
        
    # --- Fallback if optimization fails ---
    if best_vars is None:
        best_vars = x0_1
        
    centers = best_vars[:2*n].reshape((n, 2))
    radii = best_vars[2::3]
    
    # Ensure strict validity for the checker
    radii = np.maximum(radii, 1e-13)
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
        
    return centers, radii, np.sum(radii)
