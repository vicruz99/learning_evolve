import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute constant LP constraint matrix structure for pairwise constraints
# r_i + r_j <= dist(i,j)  =>  row has 1 at i and j
PAIR_INDICES = []
A_LP = np.zeros((N * (N - 1) // 2, N))
_lp_row = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[_lp_row, i] = 1.0
        A_LP[_lp_row, j] = 1.0
        PAIR_INDICES.append((i, j))
        _lp_row += 1

def solve_lp_and_gradient(centers):
    """
    Solves LP to find max sum of radii for fixed centers.
    Returns radii, sum_radii, and gradient of the objective w.r.t centers.
    """
    n = centers.shape[0]
    
    # Upper bounds from boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    np.fill_diagonal(dists, 1e9) # Avoid zero distances
    
    b_ub = dists[np.triu_indices(n, k=1)]
    
    # Solve LP: max sum(r) s.t. A r <= b, 0 <= r <= ub
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(n), 0.0, np.zeros_like(centers)
        
    radii = res.x
    duals = res.ineqlin.marginals
    
    # Compute gradient using dual variables
    # grad[i] = sum_j lambda_ij * (c_i - c_j) / ||c_i - c_j||
    dual_mat = np.zeros((n, n))
    idx = 0
    for i, j in PAIR_INDICES:
        val = duals[idx]
        if val > 1e-10:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                dual_mat[i, j] = val
                dual_mat[j, i] = -val
        idx += 1
        
    unit_diff = diff / dists[..., np.newaxis]
    grad = np.einsum('ij,ijk->ik', dual_mat, unit_diff)
    
    return radii, np.sum(radii), grad

def generate_initial_configs(rng):

def local_coordinate_optimize(centers, radii, rng):
    """Optimize each circle's position individually while fixing others."""
    c = centers.copy()
    r = radii.copy()
    n = c.shape[0]
    
    for _ in range(2): # Multiple passes
        for i in range(n):
            other_c = c[np.arange(n) != i]
            other_r = r[np.arange(n) != i]
            
            def obj_loc(x):
                ci = np.array([x[0], x[1]])
                # Boundary constraint
                rb = min(ci[0], 1.0 - ci[0], ci[1], 1.0 - ci[1])
                # Neighbor constraint
                dists = np.linalg.norm(other_c - ci, axis=1)
                rn = np.min(dists - other_r)
                return -max(rb, rn)
            
            x0 = c[i]
            try:
                res = minimize(obj_loc, x0, method='Nelder-Mead', 
                               options={'xatol': 1e-8, 'fatol': 1e-8, 'maxiter': 400})
                if res.fun < obj_loc(x0) - 1e-8:
                    c[i] = res.x
                    # Update radius for i
                    rb = min(c[i, 0], 1.0 - c[i, 0], c[i, 1], 1.0 - c[i, 1])
                    dists = np.linalg.norm(c - c[i], axis=1)
                    dists[i] = 1e9
                    rn = np.min(dists - r)
                    r[i] = max(rb, rn)
            except Exception:
                pass
                
    return c, r

def joint_slsqp_refinement(centers, radii, rng):
    """Polishes centers and radii simultaneously using SLSQP."""
    n = centers.shape[0]
    v0 = np.concatenate([centers.flatten(), radii])
    
    def obj(v):
        return -np.sum(v[2 * n:])
        
    def cons(v):
        cc = v[:2 * n].reshape(n, 2)
        rr = v[2 * n:]
        c_list = [
            cc[:, 0] - rr,
            1.0 - cc[:, 0] - rr,
            cc[:, 1] - rr,
            1.0 - cc[:, 1] - rr
        ]
        idx = np.triu_indices(n, 1)
        d = np.linalg.norm(cc[idx[0]] - cc[idx[1]], axis=1)
        c_list.append(d - (rr[idx[0]] + rr[idx[1]]))
        return np.concatenate(c_list)
        
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
        if np.min(cons(res.x)) >= -1e-8:
            return res.x[:2 * n].reshape(n, 2), res.x[2 * n:], np.sum(res.x[2 * n:])
    except Exception:
        pass
    return centers, radii, np.sum(radii)
