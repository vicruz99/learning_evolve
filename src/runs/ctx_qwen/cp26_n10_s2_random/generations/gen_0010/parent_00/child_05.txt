import numpy as np
from scipy.optimize import linprog, minimize

N = 26
PAIR_INDICES = [(i, j) for i in range(N) for j in range(i + 1, N)]
A_LP = np.zeros((len(PAIR_INDICES), N))
for k, (i, j) in enumerate(PAIR_INDICES):
    A_LP[k, i] = 1.0
    A_LP[k, j] = 1.0

def compute_lp_obj_grad(x_flat):
    """Computes the sum of radii and its gradient w.r.t centers for a given configuration."""
    c = x_flat.reshape(N, 2)
    
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    b_ub = dists[np.triu_indices(N, k=1)]
    
    # Solve LP: max sum(r) s.t. r_i + r_j <= dist(i,j), 0 <= r_i <= ub[i]
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub,
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return 0.0, np.zeros_like(x_flat)
        
    radii = res.x
    duals = res.ineqlin.marginals
    
    # Gradient accumulation
    grad = np.zeros_like(c)
    idx = 0
    for i, j in PAIR_INDICES:
        lam = duals[idx]
        if lam > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    # Boundary gradient contribution: if r_i is constrained by a wall, push towards it
    tol = 1e-7
    for i in range(N):
        x, y = c[i]
        r = radii[i]
        if x - r < tol: grad[i, 0] += 1.0
        if (1.0 - x) - r < tol: grad[i, 0] -= 1.0
        if y - r < tol: grad[i, 1] += 1.0
        if (1.0 - y) - r < tol: grad[i, 1] -= 1.0
        
    return np.sum(radii), -grad.flatten()

def generate_starts(rng):
    """Generates a diverse set of initial center configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
        [7, 5, 5, 5, 4], [5, 7, 4, 5, 5], [4, 5, 7, 5, 5],
        [6, 4, 6, 6, 4], [5, 6, 6, 5, 4], [5, 5, 4, 6, 6]
    ]
    for pat in patterns:
        for shift in [0.0, 0.5]:
            c = []
            r0 = 0.09
            y = r0
            for r_idx, cnt in enumerate(pat):
                x_start = r0 + (shift * r0 if r_idx % 2 == 1 else 0.0)
                x = x_start
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            while len(c) < N:
                c.append(rng.uniform(0.1, 0.9, 2))
            c = np.array(c[:N])
            c += rng.normal(0, 0.005, c.shape)
            c = np.clip(c, 0.02, 0.98)
            starts.append(c.flatten())
            
    for _ in range(20):
        c = rng.uniform(0.1, 0.9, (N, 2))
        starts.append(c.flatten())
        
    # Corner-biased starts to encourage boundary sticking
    for _ in range(15):
        c = rng.uniform(0.1, 0.9, (N, 2))
        corners = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        c[:4] = np.array(corners) + rng.normal(0, 0.01, (4, 2))
        c = np.clip(c, 0.02, 0.98)
        starts.append(c.flatten())
        
    return starts

def repair_packing(centers, radii):
    """Deterministically shrinks radii to guarantee strict validation compliance."""
    radii = radii.copy()
    n = centers.shape[0]
    for _ in range(100):
        changed = False
        for i in range(n):
            mr = min(centers[i, 0], 1.0 - centers[i, 0],
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-11:
                radii[i] = mr
                changed = True
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds = [(0.005, 0.995)] * (2 * N)
    
    best_sum = -np.inf
    best_c = None
    best_r = None
    
    starts = generate_starts(rng)
    
    # Phase 1: Multi-start L-BFGS-B optimization
    for x0 in starts:
        try:
            res = minimize(lambda x: compute_lp_obj_grad(x)[0],
                          x0,
                          jac=lambda x: compute_lp_obj_grad(x)[1],
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 1500, 'ftol': 1e-12, 'gtol': 1e-10})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            continue
            
    # Phase 2: Basin Hopping / Perturbation Search to escape local optima
    if best_c is not None:
        for step in range(60):
            noise_scale = 0.006 * (0.92 ** (step // 6))
            c_pert = best_c + rng.normal(0, noise_scale, best_c.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            
            try:
                res = minimize(lambda x: compute_lp_obj_grad(x)[0],
                              c_pert.flatten(),
                              jac=lambda x: compute_lp_obj_grad(x)[1],
                              method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-10})
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_c = res.x.reshape(N, 2).copy()
            except Exception:
                continue
                
    # Phase 3: Final LP solve to get exact radii for best centers
    if best_c is not None:
        _, best_r = compute_lp_obj_grad(best_c.flatten())
        best_r = -best_r # Negate back to positive radii gradient was negative
        # Actually compute_lp_obj_grad returns (sum, -grad). We need radii directly.
        # Let's solve LP one last time cleanly
        ub = np.minimum(np.minimum(best_c[:, 0], 1.0 - best_c[:, 0]),
                        np.minimum(best_c[:, 1], 1.0 - best_c[:, 1]))
        ub = np.maximum(ub, 1e-9)
        diff = best_c[:, np.newaxis, :] - best_c[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, 0.0)
        b_ub = dists[np.triu_indices(N, k=1)]
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub,
                      bounds=[(0.0, u) for u in ub], method='highs')
        if res.success:
            best_r = res.x
        else:
            best_r = np.full(N, 1e-6)
            
    # Fallback safety
    if best_c is None:
        best_c = rng.uniform(0.1, 0.9, (N, 2))
        best_r = np.full(N, 0.05)
        
    # Strict numerical repair
    best_r = repair_packing(best_c, best_r)
    final_sum = float(np.sum(best_r))
    
    return best_c, best_r, final_sum
