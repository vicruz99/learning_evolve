import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_IDX = np.triu_indices(N, k=1)
PAIR_INDICES = list(zip(TRIU_IDX[0], TRIU_IDX[1]))
NUM_PAIRS = len(PAIR_INDICES)

# Precompute LP constraint matrix A_ub for r_i + r_j <= dist_ij
A_LP = np.zeros((NUM_PAIRS, N))
for k, (i, j) in enumerate(PAIR_INDICES):
    A_LP[k, i] = 1.0
    A_LP[k, j] = 1.0

def solve_lp(centers):
    n = centers.shape[0]
    # Upper bounds from boundaries: r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[TRIU_IDX]
    
    # Solve LP: max sum(r) s.t. A r <= b, 0 <= r <= ub
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if res.success:
        # Safely extract duals across scipy versions
        try:
            duals = res.ineqlin.marginals
        except AttributeError:
            duals = np.zeros(NUM_PAIRS)
        return res.x, duals
    return np.zeros(n), np.zeros(NUM_PAIRS)

def get_gradient(centers):
    radii, duals = solve_lp(centers)
    n = centers.shape[0]
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1.0)  # Prevent division by zero
    unit_diff = diff / dists[..., None]
    
    grad = np.zeros_like(centers)
    idx = 0
    for i, j in PAIR_INDICES:
        lam = duals[idx]
        if lam > 1e-10:
            grad[i] += lam * unit_diff[i, j]
            grad[j] -= lam * unit_diff[i, j]
        idx += 1
        
    return radii, np.sum(radii), grad

def local_optimization(centers, rng, max_iter=1500, init_step=0.006):
    c = centers.copy()
    best_c, best_sum = c.copy(), -1.0
    step = init_step
    
    radii, curr_sum, grad = get_gradient(c)
    best_sum = curr_sum
    
    for it in range(max_iter):
        gn = np.linalg.norm(grad)
        if gn < 1e-11:
            break
            
        # Gradient ascent step
        c_new = c + step * grad / gn
        c_new = np.clip(c_new, 0.005, 0.995)
        
        radii_new, new_sum, grad_new = get_gradient(c_new)
        
        if new_sum > curr_sum + 1e-12:
            c = c_new
            curr_sum = new_sum
            grad = grad_new
            step = min(step * 1.08, 0.04)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_c = c.copy()
        else:
            step *= 0.75
            
        if step < 1e-9:
            break
            
        # Periodic jitter to escape plateaus
        if it % 300 == 0 and it > 0:
            c += rng.normal(0, 0.0015, c.shape)
            c = np.clip(c, 0.005, 0.995)
            
    return best_c, best_sum

def generate_initial_configs(rng):
    configs = []
    
    # 1. Hexagonal lattices with varying parameters
    patterns = [
        [5, 5, 5, 5, 6], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
        [4, 5, 6, 5, 6], [6, 5, 5, 5, 5], [7, 7, 6, 6], [6, 7, 6, 7]
    ]
    for pat in patterns:
        for r0 in [0.092, 0.098, 0.105, 0.112]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            while len(c) < N:
                c.append(rng.uniform(0.1, 0.9, 2))
            c = np.array(c[:N])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.05, 0.95)
            configs.append(c)
            
    # 2. Force-directed repulsion starts
    for _ in range(15):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(600):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if 0.0 < dist < 0.18:
                        push = (0.18 - dist) * 0.06
                        forces[i] += d_vec / dist * push
                        forces[j] -= d_vec / dist * push
            c += forces
            c = np.clip(c, 0.05, 0.95)
            c += rng.normal(0, 0.001, c.shape)
            c = np.clip(c, 0.05, 0.95)
        configs.append(c)
        
    # 3. Corner-biased starts
    for _ in range(10):
        c = rng.uniform(0.15, 0.85, (N, 2))
        corners = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        idx = rng.choice(N, 4, replace=False)
        for i, corner in zip(idx, corners):
            c[i] = corner + rng.normal(0, 0.02, 2)
        c = np.clip(c, 0.05, 0.95)
        configs.append(c)
        
    return configs

def joint_slsqp_polish(centers, radii, rng):
    n = centers.shape[0]
    v0 = np.concatenate([centers.flatten(), radii])
    
    def obj(v):
        return -np.sum(v[2 * n:])
        
    def cons(v):
        cc = v[:2 * n].reshape(n, 2)
        rr = v[2 * n:]
        c_list = [
            cc[:, 0] - rr,
            1.0 - cc[:, 0] - rr,
            cc[:, 1] - rr,
            1.0 - cc[:, 1] - rr
        ]
        idx_i, idx_j = TRIU_IDX
        d = np.linalg.norm(cc[idx_i] - cc[idx_j], axis=1)
        c_list.append(d - (rr[idx_i] + rr[idx_j]))
        return np.concatenate(c_list)
        
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
        if np.min(cons(res.x)) >= -1e-9:
            return res.x[:2 * n].reshape(n, 2), res.x[2 * n:], np.sum(res.x[2 * n:])
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def repair_packing(centers, radii):
    radii = radii.copy()
    n = centers.shape[0]
    
    for _ in range(100):
        changed = False
        # Boundary clamp
        for i in range(n):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = max(mr, 0.0)
                changed = True
                
        # Pairwise overlap resolution
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    configs = generate_initial_configs(rng)
    
    # Phase 1: Local optimization from diverse starts
    for c0 in configs:
        c_opt, s_opt = local_optimization(c0, rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            
    if best_centers is not None:
        _, best_radii, _ = get_gradient(best_centers)
        
    # Phase 2: Basin Hopping / Perturbation Search
    if best_centers is not None:
        for iter_step in range(80):
            noise_scale = 0.012 * (0.93 ** (iter_step // 4))
            c_p = best_centers + rng.normal(0, noise_scale, best_centers.shape)
            c_p = np.clip(c_p, 0.02, 0.98)
            
            # Random swap to break symmetry traps
            swap_idx = rng.choice(N, 2, replace=False)
            c_p[swap_idx] = c_p[swap_idx[::-1]]
            
            c_opt, s_opt = local_optimization(c_p, rng, max_iter=1000, init_step=0.004)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                _, best_radii, _ = get_gradient(best_centers)
                
        # Phase 3: Joint SLSQP Polish on the best found
        if best_radii is not None:
            c_pol, r_pol, s_pol = joint_slsqp_polish(best_centers, best_radii, rng)
            if s_pol > best_sum:
                best_sum = s_pol
                best_centers = c_pol
                best_radii = r_pol
                
        # Final repair to guarantee validator compliance
        best_radii = repair_packing(best_centers, best_radii)
        
    # Fallback safety
    if best_centers is None:
        best_centers = rng.uniform(0.1, 0.9, (N, 2))
        best_radii = np.full(N, 0.05)
        best_sum = np.sum(best_radii)
        
    final_sum = float(np.sum(best_radii))
    return best_centers, best_radii, final_sum
