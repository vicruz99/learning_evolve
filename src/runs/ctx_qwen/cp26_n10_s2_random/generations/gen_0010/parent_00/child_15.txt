PAIR_INDICES = []
A_LP = np.zeros((N * (N - 1) // 2, N))
_lp_row = 0

def solve_lp_and_grad(centers):
    """Solves LP for max sum of radii given fixed centers. Returns radii, sum_radii, and gradient."""
    n = centers.shape[0]
    
    # Boundary upper bounds: r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    b_ub = dists[np.triu_indices(n, k=1)]
    
    # Solve LP: min -sum(r) s.t. A r <= b, 0 <= r <= ub
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(n), 0.0, np.zeros_like(centers)

    unit_diff = diff / safe_dists[..., np.newaxis]
    grad = np.einsum('ij,ijk->ik', dual_mat, unit_diff)
    
    return radii, np.sum(radii), grad

def lbfgs_optimize(c0):
    """Optimizes centers using L-BFGS-B."""
    bounds = [(0.01, 0.99)] * (2 * N)

        res = minimize(obj_and_grad_flat, c0.flatten(), method='L-BFGS-B', jac=True,
                       bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-15})
        return res.x.reshape(N, 2), -res.fun
    except Exception:
        return c0, -1.0

            best_c = c.copy()

def slsqp_joint(centers, radii, rng):

            cc[:, 0] - rr,
            1.0 - cc[:, 0] - rr,
            cc[:, 1] - rr,
            1.0 - cc[:, 1] - rr
        ]

        d = np.linalg.norm(cc[idx[0]] - cc[idx[1]], axis=1)
        c_list.append(d - (rr[idx[0]] + rr[idx[1]]))
        return np.concatenate(c_list)
        
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    try:

                       constraints={'type': 'ineq', 'fun': cons},

def repair_packing(centers, radii):

    n = centers.shape[0]
    
    for _ in range(100):
        changed = False
        # Boundary clamp

                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-11:
                radii[i] = mr
                changed = True
                
        # Pairwise overlap resolution
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    return np.maximum(radii, 0.0)
