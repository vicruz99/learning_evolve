import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute LP constraint matrix structure: r_i + r_j <= dist(i, j)
PAIR_INDICES = [(i, j) for i in range(N) for j in range(i + 1, N)]
A_LP = np.zeros((len(PAIR_INDICES), N))
for idx, (i, j) in enumerate(PAIR_INDICES):
    A_LP[idx, i] = 1.0
    A_LP[idx, j] = 1.0

def solve_lp_and_gradient(centers):
    """Solves LP for max sum of radii and computes exact gradient w.r.t centers."""
    # Boundary upper bounds: r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    b_ub = dists[np.triu_indices(N, k=1)]
    
    # Solve: max sum(r) s.t. A r <= b_ub, 0 <= r <= ub
    # scipy minimizes, so c = -1
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(centers)
        
    radii = res.x
    
    # Extract dual marginals (active constraint multipliers)
    try:
        duals = res.marginals.ineqlin
    except AttributeError:
        try:
            duals = res.ineqlin.marginals
        except AttributeError:
            duals = np.zeros_like(b_ub)
            
    # Gradient accumulation: grad[i] += sum_j lambda_ij * (c_i - c_j) / dist_ij
    grad = np.zeros_like(centers)
    idx = 0
    for i, j in PAIR_INDICES:
        lam = duals[idx]
        if lam > 1e-10:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    return radii, np.sum(radii), grad

def obj_and_grad_vector(x_flat):
    """Objective and gradient for scipy.optimize.minimize."""
    c = x_flat.reshape(N, 2)
    _, s, grad = solve_lp_and_gradient(c)
    return -s, -grad.flatten()

def optimize_from_start(c0, rng):
    """Runs L-BFGS-B optimization from a given configuration, with manual fallback."""
    c0 = np.clip(c0, 0.005, 0.995)
    bounds = [(0.005, 0.995)] * (2 * N)
    
    # Primary: L-BFGS-B
    try:
        res = minimize(obj_and_grad_vector, c0.flatten(), method='L-BFGS-B', jac=True,
                       bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-13, 'gtol': 1e-11})
        c_opt = res.x.reshape(N, 2)
        _, s_opt, _ = solve_lp_and_gradient(c_opt)
        return c_opt, s_opt
    except Exception:
        pass
        
    # Fallback: Adaptive Gradient Ascent
    c = c0.copy()
    best_c, best_s = c.copy(), -1.0
    step = 0.012
    for k in range(3500):
        radii, s, grad = solve_lp_and_gradient(c)
        if s > best_s:
            best_s = s
            best_c = c.copy()
            step = min(step * 1.08, 0.06)
        else:
            step *= 0.91
            
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-10:
            break
            
        c += step * grad / g_norm
        c = np.clip(c, 0.005, 0.995)
        if step < 1e-9:
            break
            
    return best_c, best_s

def generate_initial_configs(rng):
    """Generates a diverse set of starting configurations."""
    configs = []
    
    # 1. Hexagonal lattice patterns with varying densities
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
        [7, 7, 6, 6], [6, 7, 6, 7], [5, 7, 7, 7]
    ]
    for pat in patterns:
        for r0 in [0.090, 0.098, 0.105, 0.112]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            while len(c) < N:
                c.append(rng.uniform(0.1, 0.9, 2))
            c = np.array(c[:N])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.05, 0.95)
            configs.append(c)
            
    # 2. Force-directed repulsion (uniform spread)
    for _ in range(10):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(600):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if 0.0 < dist < 0.18:
                        push = (0.18 - dist) * 0.05
                        forces[i] += d_vec / dist * push
                        forces[j] -= d_vec / dist * push
            c += forces * 0.02
            c = np.clip(c, 0.05, 0.95)
        configs.append(c)
        
    # 3. Corner and edge biased starts (often yields better boundary utilization)
    for _ in range(8):
        c = rng.uniform(0.15, 0.85, (N, 2))
        corners = np.array([[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]])
        c[:4] = corners + rng.normal(0, 0.015, corners.shape)
        c = np.clip(c, 0.02, 0.98)
        configs.append(c)
        
    # 4. Pure random dense starts
    for _ in range(10):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return configs

def repair_packing(centers, radii):
    """Deterministically shrinks radii to guarantee strict validation compliance."""
    radii = radii.copy()
    n = centers.shape[0]
    
    for _ in range(80):
        changed = False
        # Boundary clamp
        for i in range(n):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
                
        # Pairwise overlap resolution
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    configs = generate_initial_configs(rng)
    
    # Phase 1: Global search from diverse starts
    for c0 in configs:
        c_opt, s_opt = optimize_from_start(c0, rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            _, best_radii, _ = solve_lp_and_gradient(c_opt)
            
    # Phase 2: Basin Hopping with decaying noise
    if best_centers is not None:
        for step_idx in range(60):
            noise_scale = 0.006 * (0.88 ** (step_idx // 4))
            c_pert = best_centers + rng.normal(0, noise_scale, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            
            # Random index swap to break rigid symmetries
            swap_idx = rng.choice(N, 2, replace=False)
            c_pert[swap_idx] = c_pert[swap_idx[::-1]]
            
            c_opt, s_opt = optimize_from_start(c_pert, rng)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                _, best_radii, _ = solve_lp_and_gradient(c_opt)
                
    # Fallback safety
    if best_centers is None:
        best_centers = rng.uniform(0.1, 0.9, (N, 2))
        best_radii = np.full(N, 0.05)
        best_sum = np.sum(best_radii)
        
    # Final strict repair to guarantee validator passes
    best_radii = repair_packing(best_centers, best_radii)
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
