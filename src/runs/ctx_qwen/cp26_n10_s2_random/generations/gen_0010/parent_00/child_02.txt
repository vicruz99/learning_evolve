import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute constant LP constraint matrix structure
M_PAIR = N * (N - 1) // 2
M_BOUND = 4 * N
A_LP = np.zeros((M_PAIR + M_BOUND, N))
PAIR_IDX = []
k = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[k, i] = 1.0
        A_LP[k, j] = 1.0
        PAIR_IDX.append((i, j))
        k += 1
for i in range(N):
    A_LP[M_PAIR + 4*i, i] = 1.0
    A_LP[M_PAIR + 4*i + 1, i] = 1.0
    A_LP[M_PAIR + 4*i + 2, i] = 1.0
    A_LP[M_PAIR + 4*i + 3, i] = 1.0

def solve_lp_and_gradient(centers):
    """Solves LP for max sum of radii and computes exact gradient via duals."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(A_LP.shape[0])
    b_ub[:M_PAIR] = dists[np.triu_indices(n, k=1)]
    for i in range(n):
        b_ub[M_PAIR + 4*i] = centers[i, 0]
        b_ub[M_PAIR + 4*i + 1] = 1.0 - centers[i, 0]
        b_ub[M_PAIR + 4*i + 2] = centers[i, 1]
        b_ub[M_PAIR + 4*i + 3] = 1.0 - centers[i, 1]
        
    try:
        res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, 
                      bounds=(0, None), method='highs')
        if not res.success:
            return np.zeros(n), 0.0, np.zeros_like(centers)
        
        radii = res.x
        duals = getattr(getattr(res, 'ineqlin', None), 'marginals', None)
        if duals is None:
            return radii, np.sum(radii), np.zeros_like(centers)
    except Exception:
        return np.zeros(n), 0.0, np.zeros_like(centers)
        
    grad = np.zeros_like(centers)
    idx = 0
    for i, j in PAIR_IDX:
        lam = duals[idx]
        if lam > 1e-10:
            d = dists[i, j]
            if d > 1e-10:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    base = M_PAIR
    for i in range(n):
        grad[i, 0] += duals[base + 4*i] - duals[base + 4*i + 1]
        grad[i, 1] += duals[base + 4*i + 2] - duals[base + 4*i + 3]
        
    return radii, np.sum(radii), grad

def generate_starts(rng):

def gradient_ascent(c0, rng, steps=1500):
    """Performs gradient ascent on centers to maximize sum of radii."""
    c = c0.copy()
    vel = np.zeros_like(c)
    best_c, best_r, best_s = c.copy(), None, -1.0
    lr = 0.012
    
    for k in range(steps):
        radii, s, grad = solve_lp_and_gradient(c)
        if s > best_s:
            best_s = s
            best_c = c.copy()
            best_r = radii.copy()
            
        gn = np.linalg.norm(grad)
        if gn < 1e-10:
            grad = rng.normal(0, 0.001, c.shape)
            gn = np.linalg.norm(grad)
            
        # Momentum update
        vel = 0.75 * vel + 0.25 * grad
        c += lr * vel / gn
        c = np.clip(c, 0.005, 0.995)
        
        # Adaptive learning rate decay
        if k % 150 == 0 and k > 0:
            lr *= 0.92
            
        # Periodic jitter to escape plateaus
        if k % 400 == 0 and k > 0:
            c += rng.normal(0, 0.002 * np.exp(-k/1500), c.shape)
            c = np.clip(c, 0.005, 0.995)
            
    return best_c, best_r, best_s

def slsqp_joint_polish(c0, r0, rng):
    """Refines centers and radii simultaneously using SLSQP."""
    n = c0.shape[0]
    v0 = np.concatenate([c0.flatten(), r0])
    
    def obj(v):
        return -np.sum(v[2*n:])
        
    def cons(v):
        cc = v[:2*n].reshape(n, 2)
        rr = v[2*n:]
        c_list = [
            cc[:, 0] - rr, 1.0 - cc[:, 0] - rr,
            cc[:, 1] - rr, 1.0 - cc[:, 1] - rr
        ]
        idx = np.triu_indices(n, 1)
        d = np.linalg.norm(cc[idx[0]] - cc[idx[1]], axis=1)
        c_list.append(d - (rr[idx[0]] + rr[idx[1]]))
        return np.concatenate(c_list)
        
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
        if np.min(cons(res.x)) >= -1e-8:
            return res.x[:2*n].reshape(n, 2), res.x[2*n:], np.sum(res.x[2*n:])
    except Exception:
        pass
    return c0, r0, np.sum(r0)

def repair_packing(centers, radii):
    """Deterministically shrinks radii to guarantee strict validation compliance."""
    radii = radii.copy()
    n = centers.shape[0]
    for _ in range(40):
        changed = False
        for i in range(n):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
