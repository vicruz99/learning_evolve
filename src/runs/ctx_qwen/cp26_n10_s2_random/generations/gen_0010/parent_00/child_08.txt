def push_apart_repair(centers, radii, iters=150):

    r = radii.copy()
    n = c.shape[0]
    
    for _ in range(iters):
        changed = False
        # Resolve pairwise overlaps
        for i in range(n):
            for j in range(i + 1, n):
                d_vec = c[i] - c[j]
                dist = np.linalg.norm(d_vec)
                if dist < r[i] + r[j] - 1e-12:
                    overlap = r[i] + r[j] - dist
                    if dist > 1e-12:
                        shift = overlap * 0.5 * d_vec / dist
                        c[i] += shift
                        c[j] -= shift
                    changed = True
                    
        # Clamp to boundaries and adjust radii if necessary
        for i in range(n):
            c[i, 0] = np.clip(c[i, 0], r[i], 1.0 - r[i])
            c[i, 1] = np.clip(c[i, 1], r[i], 1.0 - r[i])

                r[i] = mr
                changed = True

    return c, np.maximum(r, 0.0)

def joint_slsqp_refinement(centers, radii, rng):

            cc[:, 0] - rr,
            1.0 - cc[:, 0] - rr,
            cc[:, 1] - rr,
            1.0 - cc[:, 1] - rr
        ]

        d = np.linalg.norm(cc[idx[0]] - cc[idx[1]], axis=1)
        c_list.append(d - (rr[idx[0]] + rr[idx[1]]))
        return np.concatenate(c_list)

        pass
