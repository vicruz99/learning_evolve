import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute constant LP constraint matrix structure
PAIR_INDICES = []
A_LP = np.zeros((N * (N - 1) // 2, N))
_lp_row = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[_lp_row, i] = 1.0
        A_LP[_lp_row, j] = 1.0
        PAIR_INDICES.append((i, j))
        _lp_row += 1

TRIU_IND = np.triu_indices(N, 1)

def solve_lp_and_gradient(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns radii, sum_radii, and gradient of the objective w.r.t centers.
    """
    n = centers.shape[0]
    
    # Upper bounds from boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    b_ub = dists[TRIU_IND]
    
    # Solve LP: max sum(r) s.t. A r <= b, 0 <= r <= ub
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(n), 0.0, np.zeros_like(centers)
        
    radii = res.x
    
    # Extract duals safely
    duals = np.zeros_like(b_ub)
    try:
        if hasattr(res, 'ineqlin') and res.ineqlin is not None:
            duals = res.ineqlin.marginals
        elif hasattr(res, 'marginals') and res.marginals is not None:
            duals = res.marginals.ineqlin
    except Exception:
        pass
        
    # Compute gradient using dual variables for pairwise constraints
    grad = np.zeros_like(centers)
    idx = 0
    for i, j in PAIR_INDICES:
        mu = duals[idx]
        if mu > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
        idx += 1
        
    # Explicit boundary forces to push centers inward if radii are constrained by walls
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        margin = 1e-5
        if r > x - margin:
            grad[i, 0] += 10.0 * (r - x + margin)
        if r > (1.0 - x) - margin:
            grad[i, 0] -= 10.0 * (r - (1.0 - x) + margin)
        if r > y - margin:
            grad[i, 1] += 10.0 * (r - y + margin)
        if r > (1.0 - y) - margin:
            grad[i, 1] -= 10.0 * (r - (1.0 - y) + margin)
            
    return radii, np.sum(radii), grad

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    
    # 1. Hexagonal lattice patterns with varying densities
    patterns = [
        [5, 5, 5, 5, 6], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
        [4, 5, 6, 5, 6], [6, 5, 5, 5, 5]
    ]
    for pat in patterns:
        for r0 in [0.095, 0.100, 0.105]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            while len(c) < N:
                c.append(rng.uniform(0.1, 0.9, 2))
            c = np.array(c[:N]) + rng.normal(0, 0.002, (N, 2))
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    # 2. Force-directed repulsion starts
    for _ in range(10):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(600):
            for i in range(N):
                for j in range(i + 1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if 0.0 < dist < 0.18:
                        push = (0.18 - dist) * 0.05
                        c[i] += d_vec / dist * push
                        c[j] -= d_vec / dist * push
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    # 3. Pure random starts
    for _ in range(8):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return starts

def optimize_hill_climb(c0, rng):
    """Runs gradient ascent with simulated annealing to escape local minima."""
    c = c0.copy()
    radii, s, _ = solve_lp_and_gradient(c)
    best_c, best_r, best_s = c.copy(), radii.copy(), s
    step = 0.006
    
    for k in range(4000):
        radii, s, grad = solve_lp_and_gradient(c)
        if s > best_s + 1e-9:
            best_c, best_r, best_s = c.copy(), radii.copy(), s
            
        gn = np.linalg.norm(grad)
        if gn < 1e-11:
            break
            
        c_new = c + step * (grad / gn)
        c_new = np.clip(c_new, 0.005, 0.995)
        
        _, s_new, _ = solve_lp_and_gradient(c_new)
        
        # Accept better moves, or worse moves with SA probability
        if s_new > s:
            c = c_new
            step = min(step * 1.03, 0.025)
        else:
            step *= 0.90
            temp = 0.001 * (1.0 - k / 4000.0)
            if temp < 1e-6:
                temp = 1e-6
            if rng.random() < np.exp((s_new - s) / temp):
                c = c_new
                
        if step < 1e-6:
            c += rng.normal(0, 0.004, c.shape)
            c = np.clip(c, 0.05, 0.95)
            step = 0.006
            
    return best_c, best_r, best_s

def slsqp_polish(centers, radii):
    """Refines centers and radii jointly using SLSQP."""
    n = centers.shape[0]
    v0 = np.concatenate([centers.flatten(), radii])
    
    def obj(v):
        return -np.sum(v[2 * n:])
        
    def cons(v):
        cc = v[:2 * n].reshape(n, 2)
        rr = v[2 * n:]
        c_list = [
            cc[:, 0] - rr,
            1.0 - cc[:, 0] - rr,
            cc[:, 1] - rr,
            1.0 - cc[:, 1] - rr
        ]
        idx = np.triu_indices(n, 1)
        d = np.linalg.norm(cc[idx[0]] - cc[idx[1]], axis=1)
        c_list.append(d - (rr[idx[0]] + rr[idx[1]]))
        return np.concatenate(c_list)
        
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
        if np.min(cons(res.x)) >= -1e-8:
            return res.x[:2 * n].reshape(n, 2), res.x[2 * n:], np.sum(res.x[2 * n:])
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def repair_packing(centers, radii):
    """Deterministically shrinks radii to guarantee strict validation compliance."""
    radii = radii.copy()
    n = centers.shape[0]
    
    for _ in range(100):
        changed = False
        for i in range(n):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
                
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_c, best_r, best_s = None, None, -1.0
    
    starts = generate_starts(rng)
    
    # Phase 1: Gradient ascent from diverse starts
    for c0 in starts:
        c_opt, r_opt, s_opt = optimize_hill_climb(c0, rng)
        if s_opt > best_s:
            best_c, best_r, best_s = c_opt, r_opt, s_opt
            
        # Occasional SLSQP polish during search
        if rng.random() < 0.4:
            c_p, r_p, s_p = slsqp_polish(c_opt, r_opt)
            if s_p > best_s:
                best_c, best_r, best_s = c_p, r_p, s_p
                
    # Phase 2: Basin hopping around the current global best
    if best_c is not None:
        for _ in range(25):
            c_jit = best_c + rng.normal(0, 0.008, best_c.shape)
            c_jit = np.clip(c_jit, 0.05, 0.95)
            c_opt, r_opt, s_opt = optimize_hill_climb(c_jit, rng)
            if s_opt > best_s:
                best_c, best_r, best_s = c_opt, r_opt, s_opt
                
            c_p, r_p, s_p = slsqp_polish(c_opt, r_opt)
            if s_p > best_s:
                best_c, best_r, best_s = c_p, r_p, s_p
                
    # Fallback safety
    if best_c is None:
        best_c = rng.uniform(0.1, 0.9, (N, 2))
        best_r = np.full(N, 0.05)
        best_s = np.sum(best_r)
        
    # Final strict repair
    best_r = repair_packing(best_c, best_r)
    final_sum = float(np.sum(best_r))
    
    return best_c, best_r, final_sum
