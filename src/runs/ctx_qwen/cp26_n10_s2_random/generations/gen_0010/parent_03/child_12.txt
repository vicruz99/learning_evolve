import numpy as np
from scipy.optimize import linprog, minimize

N = 26
NUM_PAIRS = N * (N - 1) // 2
NUM_BOUND = 4 * N

def build_lp_matrix():
    """Precomputes the constant structure of the LP constraint matrix."""
    A = np.zeros((NUM_PAIRS + NUM_BOUND, N))
    pair_idx = []
    k = 0
    for i in range(N):
        for j in range(i + 1, N):
            A[k, i] = 1.0
            A[k, j] = 1.0
            pair_idx.append((i, j))
            k += 1
    for i in range(N):
        base = NUM_PAIRS + 4 * i
        A[base, i] = 1.0
        A[base + 1, i] = 1.0
        A[base + 2, i] = 1.0
        A[base + 3, i] = 1.0
    return A, pair_idx

A_LP, PAIR_IDX = build_lp_matrix()

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, and dual multipliers."""
    c = np.clip(centers, 1e-8, 1.0 - 1e-8)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b = np.zeros(A_LP.shape[0])
    idx = 0
    for i, j in PAIR_IDX:
        b[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b[idx] = c[i, 0]; idx += 1
        b[idx] = 1.0 - c[i, 0]; idx += 1
        b[idx] = c[i, 1]; idx += 1
        b[idx] = 1.0 - c[i, 1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:
        try:
            duals = np.asarray(res.marginals.ineqlin)
        except AttributeError:
            try:
                duals = np.asarray(res.ineqlin.marginals)
            except AttributeError:
                duals = np.zeros(b.shape[0])
        return res.x, np.sum(res.x), duals
    return None, 0.0, np.zeros(b.shape[0])

def get_gradient(centers, duals):
    """Computes gradient of sum of radii w.r.t centers using LP duals."""
    grad = np.zeros_like(centers)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    idx = 0
    for i, j in PAIR_IDX:
        mu = duals[idx]
        if mu > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
        idx += 1
        
    bound_start = NUM_PAIRS
    for i in range(N):
        mu_L = duals[bound_start + 4 * i]
        mu_R = duals[bound_start + 4 * i + 1]
        mu_B = duals[bound_start + 4 * i + 2]
        mu_T = duals[bound_start + 4 * i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return grad

def gradient_ascent(c0, steps=1200):
    """Performs gradient ascent on centers to maximize LP-optimal sum of radii."""
    rng = np.random.default_rng(np.random.randint(0, 2**31))
    c = c0.copy()
    best_c = c.copy()
    best_s = -1.0
    step = 0.008
    no_improve = 0
    
    for k in range(steps):
        r, s, duals = solve_lp(c)
        if r is None: break

            best_s = s
            best_c = c.copy()
            no_improve = 0
        else:
            no_improve += 1

            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.01, 0.99)
            
    return best_c, best_s

def obj_joint(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Constraints for SLSQP: boundaries and non-overlap (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    return np.concatenate(con)

def slsqp_polish(c0, r0, max_iter=3000):
    """Refines centers and radii jointly using SLSQP."""
    v0 = np.concatenate([c0.flatten(), r0])
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False})
        if np.min(cons_joint(res.x)) >= -1e-8:
            return res.x[:2*N].reshape(N, 2), res.x[2*N:], np.sum(res.x[2*N:])
    except Exception:
        pass
    return c0, r0, np.sum(r0)

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True

            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if radii[i] + radii[j] > d - 1e-11:
                    shrink = (radii[i] + radii[j] - d) * 0.5 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed: break
    return np.maximum(radii, 0.0)
