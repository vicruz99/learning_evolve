import numpy as np
from scipy.optimize import linprog, minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure (constant for fixed N)
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
k = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[k, i] = 1.0
        A_LP[k, j] = 1.0
        PAIR_IDX.append((i, j))
        k += 1
for i in range(N):
    for _ in range(4):
        A_LP[k, i] = 1.0
        k += 1

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii and dual multipliers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    b = np.zeros(A_LP.shape[0])
    k = 0
    for i, j in PAIR_IDX:
        b[k] = dists[i, j]
        k += 1
    for i in range(N):
        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, 
                  bounds=[(0, u) for u in ub], method='highs')
    if res.success:
        try:
            duals = np.asarray(res.ineqlin.marginals)
        except AttributeError:
            try:
                duals = np.asarray(res.marginals.ineqlin)
            except AttributeError:
                duals = np.zeros(A_LP.shape[0])
        return res.x, duals
    return None, None

def get_gradient(centers, duals):
    """Computes gradient of sum of radii w.r.t centers using LP duals."""
    grad = np.zeros_like(centers)
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    k = 0
    for i, j in PAIR_IDX:
        lam = duals[k]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        k += 1
        
    bound_start = NUM_PAIRS
    for i in range(N):
        mu_x0 = duals[bound_start + 4 * i]
        mu_x1 = duals[bound_start + 4 * i + 1]
        mu_y0 = duals[bound_start + 4 * i + 2]
        mu_y1 = duals[bound_start + 4 * i + 3]
        grad[i, 0] += mu_x0 - mu_x1
        grad[i, 1] += mu_y0 - mu_y1
        
    return grad

def gradient_ascent(c0, max_iter=1500, step_init=0.004):
    """Gradient ascent on centers to maximize sum of radii with adaptive step."""
    c = c0.copy()
    best_c = c.copy()
    best_s = -1.0
    step = step_init
    
    radii, duals = solve_lp(c)
    if radii is None: return c0, -1.0
    best_s = np.sum(radii)
    
    for _ in range(max_iter):
        radii, duals = solve_lp(c)
        if radii is None: break
        s = np.sum(radii)
        if s > best_s:
            best_s = s
            best_c = c.copy()

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-10: break

    return best_c, best_s

def physics_expand(c0, steps=800):
    """Repulsive physics simulation to spread centers and make room for larger radii."""
    c = c0.copy()
    for _ in range(steps):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                diff = c[i] - c[j]
                d = np.linalg.norm(diff)
                if d < 0.25 and d > 1e-5:
                    push = (0.25 - d) * 0.02
                    forces[i] += diff / d * push
                    forces[j] -= diff / d * push
        # Gentle pressure away from boundaries to center mass
        forces[:, 0] += (0.5 - c[:, 0]) * 0.01
        forces[:, 1] += (0.5 - c[:, 1]) * 0.01
        c += forces * 0.15
        c = np.clip(c, 0.05, 0.95)
    return c

def slsqp_obj(v):
    return -np.sum(v[2*N:])

def slsqp_cons(v):
    """SLSQP constraints: boundary and pairwise non-overlap (squared, must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    i, j = IDX_I, IDX_J
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def polish(c_init, rng):
    """Refines centers and radii jointly using SLSQP."""
    radii, _ = solve_lp(c_init)
    if radii is None: return c_init, radii, -1.0
    s = np.sum(radii)
    v0 = np.concatenate([c_init.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
        if np.min(slsqp_cons(res.x)) >= -1e-7:
            c_opt = res.x[:2*N].reshape(N, 2)
            r_opt = res.x[2*N:]
            return c_opt, r_opt, np.sum(r_opt)
    except Exception:
        pass
    return c_init, radii, s

def generate_inits(rng):
    """Generates diverse initial configurations."""
    inits = []
    pats = [[5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [4,6,6,6,4], [5,5,6,5,5], [5,4,6,6,5]]
    for pat in pats:

            y = r0

                shift = r0 if ri % 2 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.05, 0.95)
            inits.append(c)
            
    # Random starts
    for _ in range(10):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
    return inits

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(80):
        changed = False
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], 
                     centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], 
                             centers[i,1]-centers[j,1])
                if radii[i] + radii[j] > d - 1e-11:
                    shrink = (radii[i] + radii[j] - d) * 0.5 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
