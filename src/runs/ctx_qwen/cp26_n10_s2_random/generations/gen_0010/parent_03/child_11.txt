def obj_single_circle(pos, oc, or_):
    """Objective for single circle optimization: maximize its radius."""
    d = np.sqrt(np.sum((pos - oc)**2, axis=1))
    rb = min(pos[0], 1.0 - pos[0], pos[1], 1.0 - pos[1])
    rp = np.min(0.5 * d - or_)
    return -min(rb, rp)
