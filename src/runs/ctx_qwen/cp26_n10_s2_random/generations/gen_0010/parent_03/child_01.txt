import numpy as np
from scipy.optimize import linprog, minimize

N = 26
NUM_PAIRS = N * (N - 1) // 2

# Precompute constant LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = np.zeros((NUM_PAIRS, 2), dtype=int)
k = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[k, i] = 1.0
        A_LP[k, j] = 1.0
        PAIR_IDX[k] = [i, j]
        k += 1
for i in range(N):
    A_LP[k, i] = 1.0; k += 1
    A_LP[k, i] = 1.0; k += 1
    A_LP[k, i] = 1.0; k += 1
    A_LP[k, i] = 1.0; k += 1

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii and computes exact gradient via dual variables."""
    c = centers
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)

    b_ub = np.empty(NUM_PAIRS + 4 * N)
    b_ub[:NUM_PAIRS] = dists[PAIR_IDX[:, 0], PAIR_IDX[:, 1]]
    b_ub[NUM_PAIRS:] = np.concatenate([
        c[:, 0], 1.0 - c[:, 0],
        c[:, 1], 1.0 - c[:, 1]
    ])

    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=[(0, None)] * N, method='highs')
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(b_ub)

    radii = res.x
    s = np.sum(radii)

    # Extract dual marginals safely across scipy versions
    duals = np.zeros_like(b_ub)
    if hasattr(res, 'marginals') and res.marginals is not None:
        duals = res.marginals.ineqlin
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = res.ineqlin.marginals

    grad = np.zeros_like(c)
    # Pairwise repulsion forces from active distance constraints
    for k_idx in range(NUM_PAIRS):
        lam = duals[k_idx]
        if lam > 1e-9:
            i, j = PAIR_IDX[k_idx]
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec

    # Boundary forces from active wall constraints
    base = NUM_PAIRS
    for i in range(N):
        grad[i, 0] += duals[base + 4 * i] - duals[base + 4 * i + 1]
        grad[i, 1] += duals[base + 4 * i + 2] - duals[base + 4 * i + 3]

    return radii, s, grad

def gradient_ascent(c0, max_iter=2000):
    """Gradient ascent on centers to maximize LP-optimal sum of radii."""
    c = c0.copy()
    best_c = c.copy()
    best_s = -1.0
    step = 0.005
    stagnation = 0

    for it in range(max_iter):
        radii, s, grad = solve_lp_and_grad(c)
        if s > best_s:
            best_s = s
            best_c = c.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 80:
            step *= 0.85
        if step < 1e-9:
            break

        gn = np.linalg.norm(grad)
        if gn > 1e-9:
            c += step * grad / gn
            c = np.clip(c, 1e-4, 1.0 - 1e-4)

        # Periodic jitter to escape plateaus
        if it % 200 == 0:
            c += np.random.normal(0, 0.002, c.shape)
            c = np.clip(c, 1e-4, 1.0 - 1e-4)

    return best_c, best_s

def simulated_annealing(c0, max_iter=2500):
    """Simulated annealing to escape local minima using LP objective."""
    c = c0.copy()
    _, s, _ = solve_lp_and_grad(c)
    best_c = c.copy()
    best_s = s
    T = 0.018

    for _ in range(max_iter):
        T *= 0.996
        delta = np.random.normal(0, T, c.shape)
        c_new = np.clip(c + delta, 1e-4, 1.0 - 1e-4)
        _, s_new, _ = solve_lp_and_grad(c_new)
        
        # Acceptance criterion for maximization
        if s_new > s or np.random.rand() < np.exp((s_new - s) / max(T, 1e-9)):
            c = c_new
            s = s_new
            if s > best_s:
                best_s = s
                best_c = c.copy()
                
    return best_c, best_s

def slsqp_polish(c0, r0):
    """Refines centers and radii jointly using SLSQP."""
    v0 = np.concatenate([c0.flatten(), r0])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    def obj(v):
        return -np.sum(v[2 * N:])
        
    def con(v):
        cc = v[:2 * N].reshape(N, 2)
        rr = v[2 * N:]
        con_list = [cc[:, 0] - rr, 1.0 - cc[:, 0] - rr, cc[:, 1] - rr, 1.0 - cc[:, 1] - rr]
        idx = np.triu_indices(N, 1)
        d = np.linalg.norm(cc[idx[0]] - cc[idx[1]], axis=1)
        con_list.append(d - (rr[idx[0]] + rr[idx[1]]))
        return np.concatenate(con_list)

    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': con},
                       options={'maxiter': 3000, 'ftol': 1e-13})
        if np.min(con(res.x)) >= -1e-8:
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:], np.sum(res.x[2 * N:])
    except Exception:
        pass
    return c0, r0, np.sum(r0)

def force_init(seed):
    """Generates an evenly spread initial configuration using repulsive forces."""
    rng = np.random.default_rng(seed)
    c = rng.uniform(0.2, 0.8, (N, 2))
    for _ in range(800):
        f = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                d_vec = c[i] - c[j]
                d = np.linalg.norm(d_vec)
                if d < 0.22 and d > 1e-6:
                    push = (0.22 - d) * 0.05
                    f[i] += d_vec / d * push
                    f[j] -= d_vec / d * push
        c += f
        c = np.clip(c, 0.1, 0.9)
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_c, best_r, best_sum = None, None, -1.0

    # Diverse initial configurations
    starts = []
    # Hexagonal lattice patterns
    pats = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], [4, 6, 6, 6, 4]]
    for pat in pats:
        c = []
        y = 0.1
        for ri, cnt in enumerate(pat):
            shift = 0.1 if ri % 2 else 0.0
            x = 0.1 + shift
            for _ in range(cnt):
                if len(c) < N:
                    c.append([x, y])
                x += 0.2
            y += 0.1 * np.sqrt(3)
        while len(c) < N:
            c.append(np.random.uniform(0.2, 0.8, 2))
        starts.append(np.array(c[:N]))

    # Force-directed spreads
    for s in range(10):
        starts.append(force_init(s))

    # Optimization Pipeline
    for c0 in starts:
        # 1. Gradient Ascent (fast local improvement)
        c_ga, s_ga = gradient_ascent(c0)
        r_ga, _, _ = solve_lp_and_grad(c_ga)

        # 2. Simulated Annealing (global exploration)
        c_sa, s_sa = simulated_annealing(c_ga)
        r_sa, _, _ = solve_lp_and_grad(c_sa)

        # 3. SLSQP Joint Polish (high precision)
        c_pol, r_pol, s_pol = slsqp_polish(c_sa, r_sa)

        if s_pol > best_sum:
            best_sum = s_pol
            best_c = c_pol.copy()
            best_r = r_pol.copy()

    # Strict Numerical Repair for validator compliance
    radii = best_r.copy()
    centers = best_c.copy()
    for _ in range(60):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break

    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
