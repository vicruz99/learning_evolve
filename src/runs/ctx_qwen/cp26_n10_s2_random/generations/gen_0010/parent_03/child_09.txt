import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute constant constraint matrix for the LP
NUM_PAIRS = N * (N - 1) // 2
NUM_BOUND = 4 * N
A_LP = np.zeros((NUM_PAIRS + NUM_BOUND, N))
PAIR_IDX = []
k = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[k, i] = 1.0
        A_LP[k, j] = 1.0
        PAIR_IDX.append((i, j))
        k += 1
for i in range(N):
    base = k + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii and dual multipliers."""
    c = np.clip(centers, 1e-8, 1.0 - 1e-8)
    
    # Compute pairwise distances
    dx = c[:, 0, None] - c[None, :, 0]
    dy = c[:, 1, None] - c[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Build RHS vector b_ub
    b_ub = np.zeros(NUM_PAIRS + NUM_BOUND)
    for idx, (i, j) in enumerate(PAIR_IDX):
        b_ub[idx] = dists[i, j]
        
    for i in range(N):
        base = NUM_PAIRS + 4 * i
        b_ub[base] = c[i, 0]
        b_ub[base + 1] = 1.0 - c[i, 0]
        b_ub[base + 2] = c[i, 1]
        b_ub[base + 3] = 1.0 - c[i, 1]
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, 
                  bounds=[(0.0, None)] * N, method='highs')
    
    if not res.success:
        return None, None
        
    radii = res.x
    
    # Robust dual extraction across scipy versions
    duals = np.zeros(NUM_PAIRS + NUM_BOUND)
    try:
        if hasattr(res, 'marginals') and res.marginals is not None:
            duals = res.marginals.ineqlin
        elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
            duals = res.ineqlin.marginals
    except Exception:
        pass
        
    return radii, duals

def compute_grad(centers, duals):
    """Computes gradient of sum of radii w.r.t centers using LP duals."""
    grad = np.zeros_like(centers)
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Pairwise repulsion from active distance constraints
    for k, (i, j) in enumerate(PAIR_IDX):
        lam = duals[k]
        if lam > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
                
    # Boundary forces from active wall constraints
    for i in range(N):
        base = NUM_PAIRS + 4 * i
        mu_L = duals[base]
        mu_R = duals[base + 1]
        mu_B = duals[base + 2]
        mu_T = duals[base + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return grad

def gradient_ascent(c0, max_iter=3500, rng=None):
    """Gradient ascent on centers to maximize LP-optimal sum of radii with adaptive step."""
    if rng is None:
        rng = np.random.default_rng(0)
        
    centers = c0.copy()
    best_c = centers.copy()
    best_s = -1.0
    step = 0.008
    radii, duals = solve_lp(centers)
    if radii is None:
        return c0, -1.0
    best_s = np.sum(radii)
    no_improve = 0
    
    for k in range(max_iter):
        grad = compute_grad(centers, duals)
        g_norm = np.linalg.norm(grad)
        
        if g_norm < 1e-10:
            no_improve += 1
        else:
            # Adaptive step direction
            direction = grad / g_norm
            c_try = centers + step * direction
            c_try = np.clip(c_try, 0.005, 0.995)
            
            r_try, d_try = solve_lp(c_try)
            if r_try is not None:
                s_try = np.sum(r_try)
                if s_try > best_s + 1e-12:
                    centers = c_try
                    duals = d_try
                    step = min(step * 1.15, 0.04)
                    no_improve = 0
                else:
                    step *= 0.6
                    no_improve += 1
            else:
                step *= 0.5
                no_improve += 1
                
        if s_try > best_s:
            best_s = s_try
            best_c = centers.copy()
            
        if step < 1e-10:
            break
            
        # Periodic jitter to escape plateaus
        if k % 250 == 0 and k > 0:
            jitter = rng.normal(0, 0.0015, centers.shape)
            centers += jitter
            centers = np.clip(centers, 0.005, 0.995)
            
    return best_c, best_s

def slsqp_obj(v):
    return -np.sum(v[2 * N:])

def slsqp_cons(v):
    """SLSQP constraints using squared distances for smooth gradients."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    
    cons = []
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap (squared)
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    cons.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(cons)

def force_init(rng):
    """Generates evenly spread initial configuration."""
    c = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(1000):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                d_vec = c[i] - c[j]
                d = np.linalg.norm(d_vec)
                if d < 0.28 and d > 1e-6:
                    push = (0.28 - d) * 0.05
                    forces[i] += d_vec / d * push
                    forces[j] -= d_vec / d * push
        c += forces
        c = np.clip(c, 0.1, 0.9)
    return c

def hex_init(rng, pat, r0):
    """Generates hexagonal lattice initialization."""
    c = []
    y = r0
    for r_idx, cnt in enumerate(pat):
        shift = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(c) < N:
                c.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
    while len(c) < N:
        c.append(rng.uniform(0.1, 0.9, 2))
    return np.array(c[:N])

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(150):
        changed = False
        # Fix boundary
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True
        # Fix overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) * 0.5 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_opt = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # Generate diverse initial configurations
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], 
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 5, 6, 5, 5],
        [5, 4, 6, 6, 5], [6, 6, 5, 5, 4], [5, 5, 4, 6, 6]
    ]
    
    for pat in patterns:
        for r0 in [0.095, 0.100, 0.105, 0.110]:
            starts.append(hex_init(rng, pat, r0))
            
    for _ in range(15):
        starts.append(force_init(rng))
        
    for _ in range(10):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # Phase 1: Gradient Ascent from diverse starts
    for c0 in starts:
        c_opt, s_opt = gradient_ascent(c0, rng=rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            
    if best_c is None:
        best_c = starts[0]
        best_r, _ = solve_lp(best_c)
        best_sum = np.sum(best_r)
    else:
        best_r, _ = solve_lp(best_c)
        
    # Phase 2: SLSQP Joint Polish
    v0 = np.concatenate([best_c.flatten(), best_r])
    for _ in range(4):
        try:
            res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds_opt,
                          constraints={'type': 'ineq', 'fun': slsqp_cons},
                          options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if np.min(slsqp_cons(res.x)) >= -1e-8:
                s = np.sum(res.x[2 * N:])
                if s > best_sum:
                    best_sum = s
                    best_c = res.x[:2 * N].reshape(N, 2).copy()
                    best_r = res.x[2 * N:].copy()
                    v0 = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Perturbation & Refinement Loop
    for step_idx in range(40):
        noise_scale = 0.006 * (0.92 ** (step_idx // 4))
        c_pert = best_c + rng.normal(0, noise_scale, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        
        c_pert, s_pert = gradient_ascent(c_pert, max_iter=800, rng=rng)
        r_pert, _ = solve_lp(c_pert)
        if r_pert is None:
            continue
            
        v0_pert = np.concatenate([c_pert.flatten(), r_pert])
        try:
            res = minimize(slsqp_obj, v0_pert, method='SLSQP', bounds=bounds_opt,
                          constraints={'type': 'ineq', 'fun': slsqp_cons},
                          options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
            if np.min(slsqp_cons(res.x)) >= -1e-8:
                s = np.sum(res.x[2 * N:])
                if s > best_sum:
                    best_sum = s
                    best_c = res.x[:2 * N].reshape(N, 2).copy()
                    best_r = res.x[2 * N:].copy()
        except Exception:
            continue
            
    # Final LP verification for optimal radii on best centers
    lp_r, _ = solve_lp(best_c)
    if lp_r is not None and np.sum(lp_r) > best_sum:
        best_r = lp_r
        best_sum = np.sum(lp_r)
        
    # Strict Numerical Repair
    radii = repair(best_c.copy(), best_r.copy())
    
    return best_c, radii, float(np.sum(radii))
