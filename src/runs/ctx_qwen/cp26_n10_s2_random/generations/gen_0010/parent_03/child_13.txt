import numpy as np
from scipy.optimize import linprog, minimize

N = 26
PAIR_INDICES = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIR_INDICES)

# Precompute constant LP constraint matrix structure
A_UB = np.zeros((N_PAIRS + 4 * N, N))
for k, (i, j) in enumerate(PAIR_INDICES):
    A_UB[k, i] = 1.0
    A_UB[k, j] = 1.0
for i in range(N):
    base = N_PAIRS + 4 * i
    A_UB[base, i] = 1.0
    A_UB[base + 1, i] = 1.0
    A_UB[base + 2, i] = 1.0
    A_UB[base + 3, i] = 1.0

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii and dual multipliers."""
    c = np.clip(centers, 1e-7, 1.0 - 1e-7)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    dx = c[:, 0, None] - c[None, :, 0]
    dy = c[:, 1, None] - c[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    b_ub = np.zeros(N_PAIRS + 4 * N)
    k = 0
    for i, j in PAIR_INDICES:
        b_ub[k] = dists[i, j]
        k += 1
    for i in range(N):
        b_ub[k] = c[i, 0]; k += 1
        b_ub[k] = 1.0 - c[i, 0]; k += 1
        b_ub[k] = c[i, 1]; k += 1
        b_ub[k] = 1.0 - c[i, 1]; k += 1
        
    res = linprog(-np.ones(N), A_ub=A_UB, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:
        try:
            duals = np.asarray(res.ineqlin.marginals)
        except AttributeError:
            duals = np.zeros_like(b_ub)
        return res.x, duals
    return None, None

def compute_grad(centers, duals):
    """Computes gradient of sum of radii w.r.t centers using LP duals."""
    c = centers
    dx = c[:, 0, None] - c[None, :, 0]
    dy = c[:, 1, None] - c[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    grad = np.zeros_like(c)
    k = 0
    for i, j in PAIR_INDICES:
        lam = duals[k]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        k += 1
        
    bound_start = N_PAIRS
    for i in range(N):
        mu_x0 = duals[bound_start + 4 * i]
        mu_x1 = duals[bound_start + 4 * i + 1]
        mu_y0 = duals[bound_start + 4 * i + 2]
        mu_y1 = duals[bound_start + 4 * i + 3]
        grad[i, 0] += mu_x0 - mu_x1
        grad[i, 1] += mu_y0 - mu_y1
        
    return grad

def gradient_ascent_step(centers, step_size, max_steps=25):
    """Short burst of gradient ascent to polish centers."""
    c = centers.copy()
    for _ in range(max_steps):
        r, duals = solve_lp(c)
        if r is None: break
        grad = compute_grad(c, duals)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-9: break
        c += step_size * grad / g_norm
        c = np.clip(c, 0.01, 0.99)
    return c

def simulated_annealing(centers0, rng, steps=3000, init_temp=0.028):
    """Simulated Annealing on centers with periodic gradient polishing."""
    c = centers0.copy()
    r, _ = solve_lp(c)
    if r is None: r = np.zeros(N)
    best_c = c.copy()
    best_r = r.copy()
    best_sum = np.sum(r)
    
    T = init_temp
    for step in range(steps):
        # Perturb a random circle
        i = rng.integers(N)
        noise = rng.normal(0, T, 2)
        c_new = c.copy()
        c_new[i] = np.clip(c_new[i] + noise, 0.02, 0.98)
        
        r_new, _ = solve_lp(c_new)
        if r_new is None: s_new = 0.0
        else: s_new = np.sum(r_new)
        
        if s_new > best_sum:
            best_c = c_new.copy()
            best_r = r_new.copy()
            best_sum = s_new
            
        # Accept/reject
        if s_new > best_sum or rng.random() < np.exp((s_new - best_sum) / max(T * 6.0, 1e-8)):
            c = c_new
            
        # Decay temperature
        T *= 0.9985
        
        # Periodic gradient polish to climb local peaks
        if step % 180 == 0 and step > 0:
            c = gradient_ascent_step(c, 0.0025, max_steps=30)
            r_poly, _ = solve_lp(c)
            if r_poly is not None and np.sum(r_poly) > best_sum:
                best_c = c.copy()
                best_r = r_poly.copy()
                best_sum = np.sum(r_poly)
                
    return best_c, best_r, best_sum

def obj_joint(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def cons_joint(v):
    """Constraints for SLSQP: boundaries and non-overlap."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    idx = np.triu_indices(N, 1)
    dx = c[idx[0], 0] - c[idx[1], 0]
    dy = c[idx[0], 1] - c[idx[1], 1]
    dr = r[idx[0]] + r[idx[1]]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [6, 4, 6, 5, 5], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6]
    ]
    
    for pat in patterns:
        c = []
        y = 0.095
        for r_idx, cnt in enumerate(pat):
            shift = 0.095 if r_idx % 2 == 1 else 0.0
            x = 0.095 + shift
            for _ in range(cnt):
                if len(c) < N:
                    c.append([x, y])
                x += 0.19
            y += 0.095 * np.sqrt(3.0)
        c = np.array(c[:N])
        c += rng.normal(0, 0.003, c.shape)
        c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    # Random and corner-biased
    for _ in range(10):
        c = rng.uniform(0.15, 0.85, (N, 2))
        starts.append(c)
        
    for _ in range(8):
        c = rng.uniform(0.1, 0.9, (N, 2))
        corners = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        for i, cor in zip(rng.choice(N, 4, replace=False), corners):
            c[i] = cor + rng.normal(0, 0.015, 2)
        starts.append(np.clip(c, 0.05, 0.95))
        
    return starts

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(80):
        changed = False
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], 
                     centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], 
                             centers[i,1]-centers[j,1])
                if radii[i] + radii[j] > d - 1e-11:
                    shrink = (radii[i] + radii[j] - d) * 0.5 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_opt = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = generate_starts(rng)
    
    # Phase 1: Simulated Annealing + Gradient Ascent from diverse starts
    for i, c0 in enumerate(starts):
        c_opt, r_opt, s_opt = simulated_annealing(c0, rng, steps=2500, init_temp=0.03)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            best_r = r_opt.copy()
            
    if best_c is None:
        best_c = starts[0]
        best_r, _ = solve_lp(best_c)
        best_sum = np.sum(best_r)
        
    # Phase 2: SLSQP Joint Polish
    v0 = np.concatenate([best_c.flatten(), best_r])
    for _ in range(4):
        try:
            res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_opt,
                          constraints={'type': 'ineq', 'fun': cons_joint},
                          options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            if np.min(cons_joint(res.x)) >= -1e-9:
                s = np.sum(res.x[2 * N:])
                if s > best_sum:
                    best_sum = s
                    best_c = res.x[:2 * N].reshape(N, 2).copy()
                    best_r = res.x[2 * N:].copy()
                    v0 = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Iterative Perturbation & Re-polish to escape remaining local minima
    for step in range(20):
        noise_scale = 0.006 * (0.92 ** step)
        c_pert = best_c + rng.normal(0, noise_scale, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        
        c_pert, r_pert, s_pert = simulated_annealing(c_pert, rng, steps=1200, init_temp=0.012)
        if s_pert > best_sum:
            best_sum = s_pert
            best_c = c_pert.copy()
            best_r = r_pert.copy()
            
        # Quick SLSQP on perturbation
        v0_p = np.concatenate([c_pert.flatten(), r_pert])
        try:
            res = minimize(obj_joint, v0_p, method='SLSQP', bounds=bounds_opt,
                          constraints={'type': 'ineq', 'fun': cons_joint},
                          options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
            if np.min(cons_joint(res.x)) >= -1e-9:
                s = np.sum(res.x[2 * N:])
                if s > best_sum:
                    best_sum = s
                    best_c = res.x[:2 * N].reshape(N, 2).copy()
                    best_r = res.x[2 * N:].copy()
        except Exception:
            continue
            
    # Final LP verification for optimal radii on best centers
    lp_r, _ = solve_lp(best_c)
    if lp_r is not None and np.sum(lp_r) > best_sum:
        best_r = lp_r
        best_sum = np.sum(lp_r)
        
    # Strict Numerical Repair
    radii = repair(best_c.copy(), best_r.copy())
    return best_c, radii, float(np.sum(radii))
