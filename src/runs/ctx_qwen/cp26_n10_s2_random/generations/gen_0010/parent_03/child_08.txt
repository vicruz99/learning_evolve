N = 26

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii given fixed centers and returns radii, sum, and gradient w.r.t centers."""
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))

    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    num_pairs = N * (N - 1) // 2
    A_ub = np.zeros((num_pairs + 4 * N, N))
    b_ub = np.zeros(num_pairs + 4 * N)

    for i in range(N):
        for j in range(i + 1, N):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
    for i in range(N):
        b_ub[k] = c[i, 0]; k += 1
        b_ub[k] = 1.0 - c[i, 0]; k += 1
        b_ub[k] = c[i, 1]; k += 1
        b_ub[k] = 1.0 - c[i, 1]; k += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*N, method='highs')

        return np.zeros(N), 0.0, np.zeros_like(c)

def obj_joint(v):
    return -np.sum(v[2 * N:])

def cons_joint(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dists = np.sqrt(dx**2 + dy**2)
    con.append(dists - (r[idx_i] + r[idx_j]))
    return np.concatenate(con)

def repair(centers, radii):
    radii = radii.copy()
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-12:
                radii[i] = max(mr, 0.0)
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
