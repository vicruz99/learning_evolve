import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)

def get_lp_matrix(n):
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    A_ub = np.zeros((num_pairs + num_bound, n))
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            k += 1
    for i in range(n):
        A_ub[k, i] = 1.0; k += 1
        A_ub[k, i] = 1.0; k += 1
        A_ub[k, i] = 1.0; k += 1
        A_ub[k, i] = 1.0; k += 1
    return A_ub

A_LP = get_lp_matrix(N)
NUM_PAIRS = N * (N - 1) // 2

def solve_lp_and_grad(centers):
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    b_ub = np.zeros(NUM_PAIRS + 4 * n)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            b_ub[k] = dists[i, j]
            k += 1
    for i in range(n):
        b_ub[k] = centers[i, 0]; k += 1
        b_ub[k] = 1.0 - centers[i, 0]; k += 1
        b_ub[k] = centers[i, 1]; k += 1
        b_ub[k] = 1.0 - centers[i, 1]; k += 1
        
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if not res.success:
        return None, None, None
        
    radii = res.x
    try:
        duals = res.marginals.ineqlin
    except AttributeError:
        try:
            duals = res.ineqlin.marginals
        except AttributeError:
            duals = np.zeros_like(b_ub)
            
    grad = np.zeros_like(centers)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[k]
            if lam > 1e-8:
                d = dists[i, j]
                if d > 1e-9:
                    vec = (centers[i] - centers[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
            k += 1
            
    for i in range(n):
        mu_L = duals[NUM_PAIRS + 4 * i]
        mu_R = duals[NUM_PAIRS + 4 * i + 1]
        mu_B = duals[NUM_PAIRS + 4 * i + 2]
        mu_T = duals[NUM_PAIRS + 4 * i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, np.sum(radii), grad

def gradient_ascent(c0, max_iter=600, step_init=0.006):
    c = c0.copy()
    best_c = c.copy()
    best_s = -1.0
    step = step_init
    no_imp = 0
    
    _, s, _ = solve_lp_and_grad(c)

    for k in range(max_iter):

            best_s = s
            best_c = c.copy()
            no_imp = 0
        else:
            no_imp += 1

            step *= 0.82
        elif no_imp > 30:
            step *= 0.94
        if step < 1e-10:
            break

        if g_norm > 1e-10:

def generate_inits(rng):
    inits = []
    pats = [[5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [4,6,6,6,4], [6,6,5,5,4], [5,5,6,5,5]]

                        c.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
                    x += 2.0 * r0

            inits.append(np.array(c[:N]))
            
    for _ in range(12):
        c = rng.uniform(0.2, 0.8, (N, 2))

                    d = np.linalg.norm(c[i]-c[j])

                        push = (0.22 - d) * 0.045
                        f[i] += (c[i]-c[j])/d * push
                        f[j] -= (c[i]-c[j])/d * push
            c += f
            c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    for _ in range(8):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return inits

def slsqp_obj(v):
    return -np.sum(v[2*N:])

def slsqp_cons(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    diff = c[TRIU_I] - c[TRIU_J]
    d = np.sqrt(np.sum(diff**2, axis=1))
    con.append(d - (r[TRIU_I] + r[TRIU_J]))
    return np.concatenate(con)

def polish_slsqp(c0, r0, rng):
    v0 = np.concatenate([c0.flatten(), r0])
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': 6000, 'ftol': 1e-13})
        if np.min(slsqp_cons(res.x)) >= -1e-7:
            return res.x[:2*N].reshape(N, 2), res.x[2*N:], np.sum(res.x[2*N:])
    except Exception:
        pass
    return c0, r0, np.sum(r0)

def repair(centers, radii):
    radii = radii.copy()
    for _ in range(120):
        changed = False
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if radii[i] + radii[j] > d - 1e-11:
                    shrink = (radii[i] + radii[j] - d) * 0.5 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
