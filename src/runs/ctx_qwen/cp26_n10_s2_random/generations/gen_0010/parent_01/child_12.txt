import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_IND = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

def build_lp_matrices(n):
    """Pre-construct the sparse structure of the LP constraint matrix."""
    A = np.zeros((NUM_PAIRS + 4 * n, n))
    pairs = []
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            pairs.append((i, j))
            k += 1
    for i in range(n):
        base = NUM_PAIRS + 4 * i
        A[base, i] = 1.0
        A[base + 1, i] = 1.0
        A[base + 2, i] = 1.0
        A[base + 3, i] = 1.0
    return A, pairs

A_LP, LP_PAIRS = build_lp_matrices(N)

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers. Returns radii, sum, and dual multipliers."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    b = np.zeros(A_LP.shape[0])
    k = 0
    for i, j in LP_PAIRS:
        b[k] = dists[i, j]
        k += 1
    for i in range(n):
        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1
        
    bounds = [(0.0, u) for u in ub]
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b, bounds=bounds, method='highs')
    
    if res.success:
        radii = res.x
        s_sum = np.sum(radii)
        try:
            duals = np.asarray(res.ineqlin.marginals)
        except AttributeError:
            try:
                duals = np.asarray(res.marginals)
            except AttributeError:
                duals = np.zeros(A_LP.shape[0])
        return radii, s_sum, duals
    return np.zeros(n), 0.0, np.zeros(A_LP.shape[0])

def compute_grad(centers, duals):
    """Computes exact gradient of sum of radii w.r.t centers using LP duals."""
    n = centers.shape[0]
    grad = np.zeros_like(centers)
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    k = 0
    for i, j in LP_PAIRS:
        lam = duals[k]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        k += 1
        
    bound_start = NUM_PAIRS
    for i in range(n):
        grad[i, 0] += duals[bound_start + 4 * i] - duals[bound_start + 4 * i + 1]
        grad[i, 1] += duals[bound_start + 4 * i + 2] - duals[bound_start + 4 * i + 3]
    return grad

def gradient_ascent(c0, max_iter=2000, step_init=0.006):
    """Runs gradient ascent on centers to maximize sum of radii."""
    c = c0.copy()
    best_c = c.copy()
    best_s = -1.0
    step = step_init
    
    _, s, duals = solve_lp(c)
    best_s = s
    
    for _ in range(max_iter):
        radii, s, duals = solve_lp(c)
        if s > best_s:
            best_s = s
            best_c = c.copy()
            
        grad = compute_grad(c, duals)
        gn = np.linalg.norm(grad)
        if gn < 1e-10:
            break
            
        c_new = c + step * grad / gn
        c_new = np.clip(c_new, 1e-5, 1.0 - 1e-5)
        
        _, s_new, _ = solve_lp(c_new)
        if s_new > s + 1e-12:
            c = c_new
            step = min(step * 1.02, 0.04)
        else:
            step *= 0.85
        if step < 1e-10:
            break
            
    return best_c, best_s

def simulated_annealing(c0, max_iter=8000, T_init=0.006, T_min=1e-6, rng=None):
    """Simulated Annealing on centers to escape local packing minima."""
    if rng is None:
        rng = np.random.default_rng(0)
        
    c = c0.copy()
    best_c = c.copy()
    _, best_s, _ = solve_lp(c)
    
    _, s_curr, _ = solve_lp(c)
    T = T_init
    
    for step in range(max_iter):
        T = max(T_min, T * 0.9992)
        noise_scale = T
        c_new = c + rng.normal(0, noise_scale, c.shape)
        c_new = np.clip(c_new, 0.01, 0.99)
        
        _, s_new, _ = solve_lp(c_new)
        
        delta = s_new - s_curr
        if delta > 0 or rng.random() < np.exp(delta / max(T, 1e-8)):
            c = c_new
            s_curr = s_new
            if s_curr > best_s:
                best_s = s_curr
                best_c = c.copy()
                
    return best_c, best_s

def slsqp_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def slsqp_cons(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    dx = c[TRIU_IND[0], 0] - c[TRIU_IND[1], 0]
    dy = c[TRIU_IND[0], 1] - c[TRIU_IND[1], 1]
    dr = r[TRIU_IND[0]] + r[TRIU_IND[1]]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

SLSQP_BOUNDS = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

def slsqp_polish(c_init, r_init, maxiter=7000):
    """Joint SLSQP optimization of centers and radii."""
    v0 = np.concatenate([c_init.flatten(), r_init])
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=SLSQP_BOUNDS,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if np.min(slsqp_cons(res.x)) >= -1e-7:
            c_opt = res.x[:2*N].reshape(N, 2)
            r_opt = res.x[2*N:]
            return c_opt, r_opt, np.sum(r_opt)
    except Exception:
        pass
    return c_init, r_init, 0.0

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(200):
        changed = False
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d - 1e-11:
                    shrink = (radii[i] + radii[j] - d) * 0.5 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def generate_inits(rng, n_inits=30):
    """Generates diverse initial configurations."""
    inits = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
        [5, 4, 6, 6, 5], [6, 5, 5, 5, 5], [5, 5, 5, 6, 5],
        [5, 7, 5, 5, 4], [7, 6, 6, 7], [6, 7, 6, 7]
    ]
    for pat in patterns:
        for r0 in [0.092, 0.098, 0.104, 0.110]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            inits.append(np.array(c[:N]))
            
    # Force-directed random starts
    for _ in range(8):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(400):
            f = np.zeros_like(c)
            for i in range(N):
                for j in range(i+1, N):
                    dv = c[i] - c[j]
                    d = np.linalg.norm(dv)
                    if d < 0.22 and d > 1e-6:
                        push = (0.22 - d) / d * 0.015
                        f[i] += dv * push
                        f[j] -= dv * push
            c += f
            c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    return inits[:n_inits]

def run_packing() -> tuple:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    rng = np.random.default_rng(42)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    inits = generate_inits(rng, 30)
    
    # Phase 1: Gradient Ascent from diverse starts
    for c0 in inits:
        c_ga, s_ga = gradient_ascent(c0, max_iter=2500, step_init=0.006)
        if s_ga > best_sum:
            best_sum = s_ga
            best_c = c_ga.copy()
            
    if best_c is not None:
        best_r, _, _ = solve_lp(best_c)
    else:
        best_c = inits[0]
        best_r, best_sum, _ = solve_lp(best_c)
        
    # Phase 2: Simulated Annealing on best and perturbed versions
    candidates_for_sa = [best_c.copy()]
    for _ in range(5):
        c_pert = best_c + rng.normal(0, 0.015, best_c.shape)
        c_pert = np.clip(c_pert, 0.05, 0.95)
        candidates_for_sa.append(c_pert)
        
    for c_sa0 in candidates_for_sa:
        c_sa, s_sa = simulated_annealing(c_sa0, max_iter=8000, T_init=0.006, rng=rng)
        if s_sa > best_sum:
            best_sum = s_sa
            best_c = c_sa.copy()
            best_r, _, _ = solve_lp(best_c)
            
    # Phase 3: SLSQP Joint Polish
    for _ in range(4):
        c_p, r_p, s_p = slsqp_polish(best_c, best_r, maxiter=8000)
        if s_p > best_sum:
            best_sum = s_p
            best_c = c_p.copy()
            best_r = r_p.copy()
        c_pert = best_c + rng.normal(0, 0.004, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        r_lp, _, _ = solve_lp(c_pert)
        c_p2, r_p2, s_p2 = slsqp_polish(c_pert, r_lp, maxiter=6000)
        if s_p2 > best_sum:
            best_sum = s_p2
            best_c = c_p2.copy()
            best_r = r_p2.copy()
            
    # Phase 4: Final LP & Repair
    r_lp, s_lp, _ = solve_lp(best_c)
    if s_lp > best_sum:
        best_sum = s_lp
        best_r = r_lp
        
    radii = repair(best_c.copy(), best_r.copy())
    return best_c, radii, float(np.sum(radii))
