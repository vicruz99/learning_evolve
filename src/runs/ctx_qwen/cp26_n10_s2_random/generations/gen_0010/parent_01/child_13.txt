import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_IND = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

def build_lp_structure(n):
    """Pre-construct the LP constraint matrix structure."""
    A = np.zeros((NUM_PAIRS + 4 * n, n))
    pairs = []
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            pairs.append((i, j))
            k += 1
    for i in range(n):
        base = NUM_PAIRS + 4 * i
        A[base, i] = 1.0
        A[base + 1, i] = 1.0
        A[base + 2, i] = 1.0
        A[base + 3, i] = 1.0
    return A, pairs

A_LP, LP_PAIRS = build_lp_structure(N)

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii given fixed centers and computes gradient via duals."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b = np.zeros(A_LP.shape[0])
    idx = 0
    for i, j in LP_PAIRS:
        b[idx] = dists[i, j]
        idx += 1
    for i in range(n):
        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(n), 0.0, np.zeros_like(centers)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    duals = np.zeros(b.shape[0])
    try:
        duals = res.ineqlin.marginals
    except AttributeError:
        try:
            duals = res.marginals
        except AttributeError:
            pass
            
    grad = np.zeros_like(centers)
    
    # Pairwise repulsion forces
    idx = 0
    for i, j in LP_PAIRS:
        mu = duals[idx]
        if mu > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
        idx += 1
        
    # Boundary forces
    bound_start = NUM_PAIRS
    for i in range(n):
        base = bound_start + 4 * i
        grad[i, 0] += duals[base] - duals[base + 1]
        grad[i, 1] += duals[base + 2] - duals[base + 3]
        
    return radii, s_sum, grad

def gradient_ascent(centers0, max_iter=4000, init_step=0.008, rng=None):
    """Robust gradient ascent on centers to maximize sum of radii."""
    if rng is None:
        rng = np.random.default_rng(0)
    centers = centers0.copy()
    best_centers = centers.copy()
    best_sum = -1.0
    step = init_step
    momentum = np.zeros_like(centers)
    beta = 0.6
    
    _, curr_sum, grad = solve_lp_and_grad(centers)
    best_sum = curr_sum
    
    for k in range(max_iter):
        radii, curr_sum, grad = solve_lp_and_grad(centers)
        
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            
        # Update momentum and step
        momentum = beta * momentum + (1 - beta) * grad
        g_norm = np.linalg.norm(momentum)
        if g_norm < 1e-10:
            break
            
        step_dir = momentum / g_norm
        centers_new = centers + step * step_dir
        centers_new = np.clip(centers_new, 1e-5, 1.0 - 1e-5)
        
        _, sum_new, _ = solve_lp_and_grad(centers_new)
        
        if sum_new > curr_sum + 1e-12:
            centers = centers_new
            step = min(step * 1.08, 0.025)
        else:
            step *= 0.75
            if step < 1e-12:
                break
                
        # Periodic jitter to escape plateaus

            noise_scale = 0.003 * np.exp(-k / 4000.0)
            centers += rng.normal(0, noise_scale, centers.shape)
            centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
            
    return best_centers, best_sum

def slsqp_obj(v):
    return -np.sum(v[2*N:])

def slsqp_cons(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    dx = c[TRIU_IND[0], 0] - c[TRIU_IND[1], 0]
    dy = c[TRIU_IND[0], 1] - c[TRIU_IND[1], 1]
    dr = r[TRIU_IND[0]] + r[TRIU_IND[1]]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

SLSQP_BOUNDS = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

def slsqp_optimize(c_init, r_init, maxiter=6000):
    v0 = np.concatenate([c_init.flatten(), r_init])
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=SLSQP_BOUNDS,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if np.min(slsqp_cons(res.x)) >= -1e-7:
            return res.x[:2*N].reshape(N, 2), res.x[2*N:], -res.fun
    except Exception:
        pass
    return c_init, r_init, 0.0

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def generate_starts(rng, n_starts=45):

                    x += 2.0 * r0

            c = np.array(c[:N])

            c = np.clip(c, 0.05, 0.95)

    # Force-directed starts
    for _ in range(12):
        c = rng.uniform(0.2, 0.8, (N, 2))

            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c[i] - c[j]
                    d = np.linalg.norm(d_vec)
                    if d < 0.22 and d > 1e-6:
                        f = (0.22 - d) / d * 0.015
                        forces[i] += d_vec * f
                        forces[j] -= d_vec * f
            c += forces * 0.05
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    return starts[:n_starts]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_c = None
