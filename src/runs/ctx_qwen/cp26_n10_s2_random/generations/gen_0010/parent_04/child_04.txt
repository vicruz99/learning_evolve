N = 26
NUM_PAIRS = N * (N - 1) // 2
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
_k = 0
for _i in range(N):
    for _j in range(_i + 1, N):
        A_LP[_k, _i] = 1.0
        A_LP[_k, _j] = 1.0

for _i in range(N):

def solve_lp(centers):

                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

    k = 0

        b[k] = dists[i, j]
        k += 1

        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1

        return np.full(N, 1e-9), 0.0, np.zeros_like(centers)
        
    radii = res.x
    duals = np.zeros(A_LP.shape[0])

                grad[i] += lam * vec
                grad[j] -= lam * vec
        k += 1

        grad[i, 0] += mu_x0 - mu_x1

def gradient_ascent(centers, max_iter=1500, init_step=0.006, rng=None):

def slsqp_polish(centers, radii):
    v0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    def obj(v):
        return -np.sum(v[2 * N:])
        
    def cons(v):
        c = v[:2 * N].reshape(N, 2)
        r = v[2 * N:]
        con = []
        con.append(c[:, 0] - r)
        con.append(1.0 - c[:, 0] - r)
        con.append(c[:, 1] - r)
        con.append(1.0 - c[:, 1] - r)
        i, j = np.triu_indices(N, 1)
        dx = c[i, 0] - c[j, 0]
        dy = c[i, 1] - c[j, 1]
        dr = r[i] + r[j]
        con.append(dx ** 2 + dy ** 2 - dr ** 2)
        return np.concatenate(con)
        
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 6000, 'ftol': 1e-14})
        if np.min(cons(res.x)) >= -1e-8:
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:], np.sum(res.x[2 * N:])
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def repair(centers, radii):
    radii = radii.copy()
    for _ in range(60):
        changed = False
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0],
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0],
                             centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d + 1e-11:
                    shrink = (radii[i] + radii[j] - d) * 0.5 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
