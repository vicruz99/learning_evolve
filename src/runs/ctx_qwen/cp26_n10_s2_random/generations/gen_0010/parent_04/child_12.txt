def slsqp_objective(v):
    return -np.sum(v[2 * N:])

def slsqp_constraints(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    i_idx, j_idx = np.triu_indices(N, 1)
    dx = c[i_idx, 0] - c[j_idx, 0]
    dy = c[i_idx, 1] - c[j_idx, 1]
    dists = np.sqrt(dx**2 + dy**2 + 1e-12)
    con.append(dists - (r[i_idx] + r[j_idx]))
    return np.concatenate(con)

def repair_packing(centers, radii):
    radii = radii.copy()

                     centers[i, 1], 1.0 - centers[i, 1])

                radii[i] = max(mr, 0.0)
                changed = True

            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d + 1e-12:
                    shrink = (radii[i] + radii[j] - d) * 0.5 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
