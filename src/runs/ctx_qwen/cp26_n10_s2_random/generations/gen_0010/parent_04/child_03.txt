import numpy as np
from scipy.optimize import linprog, minimize

N = 26
A_LP = None
PAIR_IDX = None

def setup_lp():
    """Precompute the sparse structure of the LP constraint matrix."""
    global A_LP, PAIR_IDX
    num_pairs = N * (N - 1) // 2
    A_LP = np.zeros((num_pairs + 4 * N, N))
    PAIR_IDX = []
    k = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_LP[k, i] = 1.0
            A_LP[k, j] = 1.0
            PAIR_IDX.append((i, j))
            k += 1
    for i in range(N):
        A_LP[k, i] = 1.0; k += 1
        A_LP[k, i] = 1.0; k += 1
        A_LP[k, i] = 1.0; k += 1
        A_LP[k, i] = 1.0; k += 1

setup_lp()

def solve_lp(c):
    """Solves LP for optimal radii given fixed centers and computes exact gradient via duals."""
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b = np.zeros(A_LP.shape[0])
    k = 0
    for i, j in PAIR_IDX:
        b[k] = dists[i, j]; k += 1
    for i in range(N):
        b[k] = c[i, 0]; k += 1
        b[k] = 1.0 - c[i, 0]; k += 1
        b[k] = c[i, 1]; k += 1
        b[k] = 1.0 - c[i, 1]; k += 1

                      bounds=[(0, u) for u in ub], method='highs', options={'disp': False})

            return np.full(N, 1e-9), 0.0, np.zeros_like(c), None
        radii = res.x
        try:
            duals = res.ineqlin.marginals
        except AttributeError:
            duals = np.zeros_like(b)
    except Exception:
        return np.full(N, 1e-9), 0.0, np.zeros_like(c), None
        
    grad = np.zeros_like(c)
    k = 0
    for i, j in PAIR_IDX:
        lam = duals[k]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:

                grad[i] += lam * v
                grad[j] -= lam * v
        k += 1
        
    bs = len(PAIR_IDX)

        grad[i, 0] += duals[bs + 4*i] - duals[bs + 4*i + 1]
        grad[i, 1] += duals[bs + 4*i + 2] - duals[bs + 4*i + 3]
        
    return radii, np.sum(radii), grad, duals

def gradient_ascent(c0, max_steps, rng):

def slsqp_polish(c0, r0):
    """Joint SLSQP optimization of centers and radii."""
    v0 = np.concatenate([c0.flatten(), r0])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    def obj(v):
        return -np.sum(v[2 * N:])
        
    def cons(v):
        cc = v[:2 * N].reshape(N, 2)
        rr = v[2 * N:]
        con = [cc[:, 0] - rr, 1.0 - cc[:, 0] - rr, cc[:, 1] - rr, 1.0 - cc[:, 1] - rr]
        i, j = np.triu_indices(N, 1)
        con.append((cc[i, 0] - cc[j, 0])**2 + **(cc[i, 1] - cc[j, 1])2 - (rr[i] + rr[j])**2)
        return np.concatenate(con)
        
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 6000, 'ftol': 1e-13})
        c_vals = cons(res.x)
        if np.min(c_vals) >= -1e-6:
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:], np.sum(res.x[2 * N:])
    except Exception:
        pass
    return c0, r0, np.sum(r0)

def generate_starts(rng):

def repair_packing(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(60):
        changed = False
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d - 1e-11:
                    shrink = (radii[i] + radii[j] - d) * 0.5 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        best_c = c_sl

        best_r = r_lp
        
    # Phase 5: Final numerical repair
    best_r = repair_packing(best_c, best_r)
    
    return best_c, best_r, float(np.sum(best_r))
