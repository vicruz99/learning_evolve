import numpy as np
from scipy.optimize import linprog, minimize

N = 26
NUM_PAIRS = N * (N - 1) // 2
TRIU_I, TRIU_J = np.triu_indices(N, 1)

# Precompute constant LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
k = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[k, i] = 1.0
        A_LP[k, j] = 1.0
        k += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns radii, sum_radii, and gradient w.r.t centers using dual multipliers.
    """
    # Distance to boundaries
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Build LP RHS
    b = np.zeros(NUM_PAIRS + 4 * N)
    k = 0
    for i in range(N):
        for j in range(i + 1, N):
            b[k] = dists[i, j]
            k += 1
    for i in range(N):
        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1
        
    # Solve LP
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(centers)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    # Extract duals safely
    duals = np.zeros(NUM_PAIRS + 4 * N)
    try:
        duals = np.asarray(res.marginals.ineqlin)
    except AttributeError:
        try:
            duals = np.asarray(res.ineqlin.marginals)
        except AttributeError:
            pass
            
    # Compute gradient using duals
    # Pairwise repulsion forces
    dirs = diff / (dists[:, :, np.newaxis] + 1e-12)
    mu_mat = np.zeros((N, N))
    mu_mat[TRIU_I, TRIU_J] = duals[:NUM_PAIRS]
    mu_mat += mu_mat.T  # Symmetric
    grad_pair = np.sum(mu_mat[:, :, np.newaxis] * dirs, axis=1)
    
    # Boundary forces
    grad_bound = np.zeros_like(centers)
    bs = NUM_PAIRS
    grad_bound[:, 0] += duals[bs + 0::4] - duals[bs + 1::4]
    grad_bound[:, 1] += duals[bs + 2::4] - duals[bs + 3::4]
    
    return radii, s_sum, grad_pair + grad_bound

def gradient_ascent(c0, rng, max_iter=1500):
    """Gradient ascent with adaptive step size and momentum."""
    c = c0.copy()
    best_c = c.copy()
    best_s = -1.0
    step = 0.005
    momentum = np.zeros_like(c)
    
    _, curr_s, grad = solve_lp_and_grad(c)
    best_s = curr_s
    
    for _ in range(max_iter):
        # Update momentum
        momentum = 0.5 * momentum + grad
        g_norm = np.linalg.norm(momentum)
        if g_norm < 1e-12:
            break
            
        move = step * (momentum / g_norm)
        c_new = np.clip(c + move, 1e-5, 1.0 - 1e-5)
        
        _, s_new, grad = solve_lp_and_grad(c_new)
        
        if s_new > curr_s + 1e-10:
            c = c_new
            curr_s = s_new
            step = min(step * 1.08, 0.025)
        else:
            step *= 0.65
            # Try half step if full step failed
            if step > 1e-6:
                c_half = np.clip(c + step * (momentum / g_norm), 1e-5, 1.0 - 1e-5)
                _, s_half, _ = solve_lp_and_grad(c_half)
                if s_half > curr_s + 1e-10:
                    c = c_half
                    curr_s = s_half
                    
        if curr_s > best_s:
            best_s = curr_s
            best_c = c.copy()
            
        if step < 1e-9:
            break
            
    return best_c, best_s

def simulated_annealing(c0, rng, steps=2500):
    """Simulated Annealing to explore center configurations."""
    c = c0.copy()
    best_c = c.copy()
    best_s = -1.0
    _, curr_s, _ = solve_lp_and_grad(c)
    best_s = curr_s
    
    T = 0.008
    for _ in range(steps):
        scale = T * 0.5
        c_new = c + rng.normal(0, scale, c.shape)
        c_new = np.clip(c_new, 0.02, 0.98)
        
        _, s_new, _ = solve_lp_and_grad(c_new)
        
        if s_new > curr_s or rng.random() < np.exp((s_new - curr_s) / max(T, 1e-7)):
            c = c_new
            curr_s = s_new
            if curr_s > best_s:
                best_s = curr_s
                best_c = c.copy()
        T *= 0.9965
        
    return best_c, best_s

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    
    # 1. Hexagonal patterns
    pats = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], 
            [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
            [7, 7, 6, 6], [6, 7, 6, 7]]
    for pat in pats:
        for r0 in [0.092, 0.098, 0.105, 0.112]:
            c = []
            y = r0
            for ri, cnt in enumerate(pat):
                sh = r0 if ri % 2 == 1 else 0.0
                x = r0 + sh
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            starts.append(np.clip(np.array(c[:N]), 0.05, 0.95))
            
    # 2. Force-directed repulsion
    for _ in range(15):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(400):
            f = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    d_vec = c[i] - c[j]
                    d = np.linalg.norm(d_vec)
                    if d < 0.20 and d > 1e-4:
                        push = (0.20 - d) * 0.05
                        f[i] += d_vec / d * push
                        f[j] -= d_vec / d * push
            c += f
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    # 3. Corner/Edge biased
    for _ in range(12):
        c = rng.uniform(0.1, 0.9, (N, 2))
        # Push 8 circles toward edges
        edges = [[0.08, 0.5], [0.92, 0.5], [0.5, 0.08], [0.5, 0.92],
                 [0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        for i, e in enumerate(edges):
            c[i] = e + rng.normal(0, 0.03, 2)
        starts.append(np.clip(c, 0.02, 0.98))
        
    # 4. Random dense
    for _ in range(20):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    return starts

def slsqp_joint_polish(c0, r0, rng):
    """Joint SLSQP optimization for final fine-tuning."""
    v0 = np.concatenate([c0.flatten(), r0])
    
    def obj(v):
        return -np.sum(v[2 * N:])

    def cons(v):
        c = v[:2 * N].reshape(N, 2)
        r = v[2 * N:]
        con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
        dx = c[TRIU_I, 0] - c[TRIU_J, 0]
        dy = c[TRIU_I, 1] - c[TRIU_J, 1]
        dr = r[TRIU_I] + r[TRIU_J]
        con.append(dx**2 + dy**2 - dr**2)
        return np.concatenate(con)
        
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
        if np.min(cons(res.x)) >= -1e-8:
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:], -res.fun
    except Exception:
        pass
    return c0, r0, 0.0

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) * 0.5 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_c = None
    best_r = None

    # Phase 1 & 2: Coarse SA + GA on diverse starts
    candidates = []
    for c_init in starts:

        # GA polish
        c_ga, s_ga = gradient_ascent(c_sa, rng, max_iter=600)
        candidates.append((c_ga, s_ga))
        
    # Sort and take top 15
    candidates.sort(key=lambda x: x[1], reverse=True)
    top_candidates = candidates[:15]
    
    # Phase 3: Deep GA on top candidates
    for c_init, _ in top_candidates:
        c_opt, s_opt = gradient_ascent(c_init, rng, max_iter=2000)
        if s_opt > best_s:
            best_s = s_opt
            best_c = c_opt.copy()
            
    if best_c is None:
        best_c = top_candidates[0][0]
        _, best_s, _ = solve_lp_and_grad(best_c)
        
    best_r, _, _ = solve_lp_and_grad(best_c)
    
    # Phase 4: Targeted Kicks & Re-optimization
    for _ in range(40):
        c_kick = best_c.copy()
        idx = rng.choice(N, size=rng.integers(3, 8), replace=False)
        c_kick[idx] += rng.normal(0, 0.012, (len(idx), 2))
        c_kick = np.clip(c_kick, 0.05, 0.95)
        
        c_k_opt, s_k_opt = gradient_ascent(c_kick, rng, max_iter=1000)
        if s_k_opt > best_s:
            best_s = s_k_opt
            best_c = c_k_opt.copy()
            best_r, _, _ = solve_lp_and_grad(best_c)
            
    # Phase 5: Long SA on best
    c_bh, s_bh = simulated_annealing(best_c, rng, steps=3000)
    if s_bh > best_s:
        best_s = s_bh
        best_c = c_bh.copy()
        best_r, _, _ = solve_lp_and_grad(best_c)
        
    # Phase 6: SLSQP Joint Polish
    ub = np.minimum(np.minimum(best_c[:, 0], 1.0 - best_c[:, 0]), 
                    np.minimum(best_c[:, 1], 1.0 - best_c[:, 1]))
    dists = np.linalg.norm(best_c[:, None, :] - best_c[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    r_sl_init = np.minimum(ub, 0.5 * np.min(dists, axis=1)) * 0.95

    if s_sl > best_s:

        best_c = c_sl.copy()
        best_r = r_sl.copy()
        
    # Final LP match
    r_final, s_final, _ = solve_lp_and_grad(best_c)
    if s_final > best_s:
        best_s = s_final
        best_r = r_final
        
    # Strict repair
    radii = repair(best_c, best_r)
    return best_c, radii, float(np.sum(radii))
