import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_U, TRIU_V = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """
    Computes inequality constraints (must be >= 0).
    Uses squared distances for smooth optimization.
    """
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    c = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c.append(xs - rs)
    c.append(1.0 - xs - rs)
    c.append(ys - rs)
    c.append(1.0 - ys - rs)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = xs[TRIU_U] - xs[TRIU_V]
    dy = ys[TRIU_U] - ys[TRIU_V]
    dr = rs[TRIU_U] + rs[TRIU_V]
    c.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(c)

def force_grow(seed):
    """Dynamic force-directed packing simulation."""
    rng = np.random.default_rng(seed)
    
    # Initialize on hexagonal lattice
    pts = []
    r0 = 0.08
    y = r0
    row = 0
    while len(pts) < N:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x + r0 <= 1.0 and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
        
    centers = np.array(pts[:N]) + rng.normal(0, 0.005, (N, 2))
    centers = np.clip(centers, 0.05, 0.95)
    radii = np.full(N, 0.04)
    forces = np.zeros((N, 2))
    
    for step in range(3000):
        # Slowly grow circles
        radii *= 1.0001
        forces *= 0.0
        
        # Vectorized pairwise repulsion
        diffs = centers[:, None, :] - centers[None, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        req = radii[:, None] + radii[None, :]
        
        d_uv = dists[TRIU_U, TRIU_V]
        r_uv = req[TRIU_U, TRIU_V]
        overlaps = np.maximum(r_uv - d_uv, 0.0)
        
        dir_uv = diffs[TRIU_U, TRIU_V, :]
        dir_uv = np.where(d_uv[:, None] > 1e-9, dir_uv / d_uv[:, None], 0.0)
        
        f_mag = overlaps * 8.0
        f_u = dir_uv * f_mag[:, None]
        f_v = -f_u
        
        np.add.at(forces, TRIU_U, f_u)
        np.add.at(forces, TRIU_V, f_v)
        
        # Boundary repulsion
        for i in range(N):
            r = radii[i]
            if centers[i, 0] < r: forces[i, 0] += (r - centers[i, 0]) * 15.0
            if centers[i, 0] > 1.0 - r: forces[i, 0] -= (centers[i, 0] - (1.0 - r)) * 15.0
            if centers[i, 1] < r: forces[i, 1] += (r - centers[i, 1]) * 15.0
            if centers[i, 1] > 1.0 - r: forces[i, 1] -= (centers[i, 1] - (1.0 - r)) * 15.0
            
        # Adaptive step size (cooling)
        step_size = 0.005 / (1.0 + 0.001 * step)
        centers += forces * step_size
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
        
    return centers, radii

def polish(centers, radii):
    """Refine configuration using SLSQP."""
    v0 = np.zeros(3 * N)
    v0[0::3] = centers[:, 0]
    v0[1::3] = centers[:, 1]
    # Scale down slightly to ensure strict initial feasibility
    v0[2::3] = radii * 0.95
    
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    try:
        res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        # Check if constraints are satisfied within tolerance
        if np.min(constraints(res.x)) > -1e-5:
            return res.x
    except Exception:
        pass
    return None

def refine_radii_lp(centers):
    """Extract maximum possible radii for fixed centers using LP."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        bds = [centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1]]
        for b in bds:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return res.x if res.success else None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    best_v = None
    best_sum = -np.inf
    
    # Run multiple independent simulations with different seeds
    seeds = list(range(15))
    for s in seeds:
        c, r = force_grow(s)
        v = polish(c, r)
        if v is not None:
            s_val = -objective(v)
            if s_val > best_sum:
                best_sum = s_val
                best_v = v
                
    # Fallback if all optimizations fail
    if best_v is None:
        c, r = force_grow(0)
        best_v = np.zeros(3 * N)
        best_v[0::3] = c[:, 0]
        best_v[1::3] = c[:, 1]
        best_v[2::3] = r
        
    centers = best_v[:2 * N].reshape(N, 2)
    
    # LP refinement to squeeze maximum radii for the optimized layout
    r_lp = refine_radii_lp(centers)
    radii = r_lp if r_lp is not None else best_v[2 * N:].copy()
    
    # Final strict numerical repair to guarantee validator tolerance
    for _ in range(50):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            x, y = centers[i]
            mx = min(x, 1.0 - x, y, 1.0 - y)
            if radii[i] > mx + 1e-10:
                radii[i] = mx
                changed = True
                
        # Resolve overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-10:
                    shrink = (req - d) / 2.0 + 1e-11
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
