

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we exploit the fact that for any fixed set of centers, the optimal radii can be found exactly via Linear Programming (LP). This transforms the problem into optimizing the center coordinates to maximize the LP objective.

Strategy:
1. **LP for Radii**: Given fixed centers, solving `max sum(r_i)` subject to `r_i + r_j <= ||c_i - c_j||` and boundary constraints is a small LP (26 variables, ~430 constraints). Using `scipy.optimize.linprog` with the `highs` solver makes this extremely fast.
2. **Center Optimization**: We define an objective function `f(C) = -sum(optimal_radii(C))` and optimize it over center coordinates using `Powell`'s method. Powell is derivative-free and highly robust for non-smooth, multimodal landscapes like packing.
3. **Diverse Initialization**: Generate ~50 high-quality starting configurations (hexagonal lattices at various densities, square grids, random dense, and corner-biased packs) to ensure global coverage.
4. **Iterative Perturbation**: After finding a strong local optimum, run a targeted coordinate-wise perturbation search to escape shallow local minima.
5. **Joint Polish & Repair**: Use SLSQP on the full variable set (centers + radii) for final precision, followed by a strict numerical repair pass to guarantee validation compliance within `1e-12` tolerance.

This bi-level approach (LP for radii, derivative-free optimization for centers) is mathematically rigorous and computationally efficient, consistently outperforming naive joint optimization.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26

# Precompute the fixed pairwise constraint matrix for the LP
# Constraints: r_i + r_j <= dist_ij  =>  [0..1..1..0] @ r <= dist_ij
NUM_PAIRS = N * (N - 1) // 2
A_ub_pair = np.zeros((NUM_PAIRS, N))
_pair_idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub_pair[_pair_idx, i] = 1.0
        A_ub_pair[_pair_idx, j] = 1.0
        _pair_idx += 1

def get_optimal_radii(centers):
    """Solves LP to find radii maximizing sum for fixed centers."""
    c_obj = -np.ones(N)
    
    # Pairwise distances upper triangle
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    b_ub_pair = dists[np.triu_indices(N, k=1)]
    
    # Boundary constraints: r_i <= dist to boundary
    bounds_r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                          np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds_r = np.maximum(bounds_r, 1e-9)
    
    var_bounds = [(0.0, b) for b in bounds_r]
    
    res = linprog(c_obj, A_ub=A_ub_pair, b_ub=b_ub_pair, bounds=var_bounds, method='highs')
    if res.success:
        return res.x
    return np.zeros(N)

def objective_centers(x_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = x_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    radii = get_optimal_radii(centers)
    return -np.sum(radii)

def generate_inits(num_inits):
    """Generates diverse initial center configurations."""
    inits = []
    rng = np.random.RandomState(42)
    
    # 1. Hexagonal lattices with varying densities
    for density in np.linspace(0.085, 0.11, 8):
        pts = []
        y = density
        row = 0
        while len(pts) < N:
            shift = density if row % 2 == 1 else 0.0
            x = density + shift
            while x + density <= 1.0 - 1e-7 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * density
            y += np.sqrt(3) * density
            row += 1
        inits.append(np.array(pts[:N]))
        
    # 2. Square lattices
    for density in np.linspace(0.08, 0.105, 6):
        pts = []
        y = density
        row = 0
        while len(pts) < N:
            x = density
            while x + density <= 1.0 - 1e-7 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * density
            y += 2.0 * density
            row += 1
        inits.append(np.array(pts[:N]))
        
    # 3. Random dense configurations
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        inits.append(c)
        
    # 4. Corner/boundary biased configurations
    for _ in range(10):
        c = rng.uniform(0.2, 0.8, (N, 2))
        c[:4] = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]
        c += rng.normal(0, 0.02, c.shape)
        c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    inits = generate_inits(50)
    
    # Phase 1: Powell optimization on centers
    bounds = [(1e-5, 1.0 - 1e-5)] * (2 * N)
    
    for i, c0 in enumerate(inits):
        c0 = c0 + np.random.normal(0, 0.001, c0.shape)
        c0 = np.clip(c0, 1e-4, 1.0 - 1e-4)
        
        try:
            res = minimize(objective_centers, c0.flatten(), method='Powell',
                           bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-11, 'xtol': 1e-11})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = res.x.reshape(N, 2).copy()
        except Exception:
            continue
            
    if best_centers is None:
        best_centers = inits[0]
        
    best_radii = get_optimal_radii(best_centers)
    
    # Phase 2: Iterative perturbation search to escape local minima
    step = 0.004
    for epoch in range(80):
        improved = False
        for i in range(N):
            current_sum = np.sum(best_radii)
            best_local_c = best_centers[i].copy()
            best_local_sum = current_sum
            
            for _ in range(15):
                direction = np.random.randn(2) * step
                new_c = best_centers.copy()
                new_c[i] += direction
                new_c[i] = np.clip(new_c[i], 1e-5, 1.0 - 1e-5)
                
                r_new = get_optimal_radii(new_c)
                s_new = np.sum(r_new)
                
                if s_new > best_local_sum + 1e-9:
                    best_local_sum = s_new
                    best_local_c = new_c[i].copy()
                    improved = True
                    
            if best_local_sum > current_sum + 1e-9:
                best_centers[i] = best_local_c
                
        best_radii = get_optimal_radii(best_centers)
        
        if not improved:
            step *= 0.92
        if step < 1e-6:
            break
            
    # Phase 3: Joint SLSQP polish for high-precision convergence
    v0 = np.concatenate([best_centers.flatten(), best_radii])
    
    def obj_full(v):
        return -np.sum(v[2*N:])
        
    def cons_full(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        con = []
        con.append(c[:, 0] - r)
        con.append(1.0 - c[:, 0] - r)
        con.append(c[:, 1] - r)
        con.append(1.0 - c[:, 1] - r)
        
        dx = c[:, 0, np.newaxis] - c[np.newaxis, :, 0]
        dy = c[:, 1, np.newaxis] - c[np.newaxis, :, 1]
        dists = np.sqrt(dx**2 + dy**2)
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        
        idx_i, idx_j = np.triu_indices(N, k=1)
        con.append(dists[idx_i, idx_j] - r_sum[idx_i, idx_j])
        return np.concatenate(con)
        
    bounds_full = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
    try:
        res_full = minimize(obj_full, v0, method='SLSQP', bounds=bounds_full,
                            constraints={'type': 'ineq', 'fun': cons_full},
                            options={'maxiter': 2000, 'ftol': 1e-13})
        if -res_full.fun > best_sum:
            best_centers = res_full.x[:2*N].reshape(N, 2)
            best_radii = res_full.x[2*N:]
            best_sum = -res_full.fun
    except Exception:
        pass
        
    # Phase 4: Strict validation repair to guarantee tolerance compliance
    for _ in range(30):
        changed = False
        for i in range(N):
            x, y = best_centers[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if best_radii[i] > max_r + 1e-12:
                best_radii[i] = max_r
                changed = True
                
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                req = best_radii[i] + best_radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-10
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        best_radii = np.maximum(best_radii, 0.0)
        if not changed:
            break
            
    final_sum = float(np.sum(best_radii))
    return best_centers, best_radii, final_sum
