2. **Inner Level (Radii)**: For any fixed set of centers, computing the maximum possible radii is a Linear Programming (LP) problem. This is highly efficient and guarantees the exact optimal radii for those centers.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_IDX = np.triu_indices(N, k=1)

def setup_lp_matrix():
    """Precomputes the constraint matrix for the radius LP."""
    n = N
    num_pairs = n * (n - 1) // 2
    num_bounds = 4 * n
    A_ub = np.zeros((num_pairs + num_bounds, n))
    
    idx = 0
    # Pairwise non-overlap constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
            
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        for _ in range(4):
            A_ub[idx, i] = 1.0
            idx += 1
    return A_ub

A_LP = setup_lp_matrix()

def get_max_sum_radii(centers):
    """
    Given fixed centers, solves an LP to find radii that maximize sum(r_i).
    Falls back to analytical bounds if LP fails.
    """
    n = N
    x, y = centers[:, 0], centers[:, 1]
    
    # Pairwise distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Construct RHS vector for LP
    b_pairs = dists[TRIU_IDX]
    b_bounds = np.repeat(np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y)), 4)
    b_ub = np.concatenate([b_pairs, b_bounds])
    
    try:
        # Maximize sum(r) <=> Minimize -sum(r)
        res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, 
                      bounds=[(0, None)]*n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    # Analytical fallback (always feasible, though not necessarily sum-optimal)
    np.fill_diagonal(dists, np.inf)
    r_pair = 0.5 * np.min(dists, axis=1)
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    return np.minimum(r_bound, r_pair)

def objective_centers(v):
    """Objective for center optimization: minimize negative sum of optimal radii."""
    centers = v.reshape(N, 2)
    # Strictly keep centers inside the unit square
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    radii = get_max_sum_radii(centers)
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_c = [(0.0, 1.0)] * (2 * N)
    
    best_centers = None
    best_sum = -np.inf
    
    inits = []
    
    # 1. Hexagonal lattice patterns with diverse topologies
    row_patterns = [
        [5,5,5,5,6], [6,5,6,5,4], [5,6,5,6,4], 
        [5,5,5,6,5], [6,6,5,5,4], [5,4,6,5,6], [4,6,6,5,5]
    ]
    for rc in row_patterns:
        for r_est in [0.08, 0.09, 0.095, 0.10, 0.105]:
            pts = []
            y = r_est
            row = 0
            for cnt in rc:
                shift = r_est if row % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += np.sqrt(3) * r_est
                row += 1
            if len(pts) == N:
                c = np.array(pts)
                c = np.clip(c, 0.02, 0.98)
                inits.append(c.flatten())

    # 2. Dense random starts
    for _ in range(20):
        inits.append((np.random.rand(N, 2) * 0.8 + 0.1).flatten())
        
    # 3. Corner/Edge focused starts
    for _ in range(10):
        c = np.random.rand(N, 2) * 0.6 + 0.2
        c[:4] = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        inits.append(c.flatten())

    # Phase 1: Multi-start Powell optimization
    for v0 in inits:
        try:
            res = minimize(objective_centers, v0, method='Powell', bounds=bounds_c,
                           options={'maxiter': 2000, 'ftol': 1e-14})
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape(N, 2).copy()
        except Exception:
            continue

    # Phase 2: Perturbation search to escape local minima
    if best_centers is not None:
        curr = best_centers.copy()
        step = 0.005
        for _ in range(60):
            pert = curr + np.random.normal(0, step, (N, 2))
            pert = np.clip(pert, 1e-5, 1.0 - 1e-5)
            try:
                res = minimize(objective_centers, pert.flatten(), method='Powell', bounds=bounds_c,
                               options={'maxiter': 1000, 'ftol': 1e-13})
                s = -res.fun
                if s > best_sum + 1e-8:
                    best_sum = s
                    best_centers = res.x.reshape(N, 2).copy()
                    step = max(step * 0.75, 0.0001)
                else:
                    step *= 0.92
            except Exception:
                continue

    # Phase 3: Exact final radii computation
    final_radii = get_max_sum_radii(best_centers)
    # Slight shrinkage to guarantee strict compliance with validator tolerance
    final_radii *= 0.9999995
    
    # Phase 4: Strict validation repair
    for _ in range(30):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], final_radii[i]
            mr = min(x, 1.0 - x, y, 1.0 - y)
            if r > mr + 1e-11:
                final_radii[i] = mr
                changed = True
        # Resolve overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i,0] - best_centers[j,0], 
                             best_centers[i,1] - best_centers[j,1])
                req = final_radii[i] + final_radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-11
                    final_radii[i] -= shrink
                    final_radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    final_radii = np.maximum(final_radii, 0.0)
    return best_centers, final_radii, float(np.sum(final_radii))
