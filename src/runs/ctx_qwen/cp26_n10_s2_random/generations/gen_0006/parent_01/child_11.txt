import numpy as np
from scipy.optimize import minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)

def compute_max_radii(centers):
    """Computes the maximum valid radius for each circle given fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.hypot(diff[:, :, 0], diff[:, :, 1])
    np.fill_diagonal(dists, np.inf)
    r_pair = 0.5 * np.min(dists, axis=1)
    
    return np.minimum(r_bound, r_pair)

def objective_centers(v):
    """Objective for center-only optimization: maximize sum of max feasible radii."""
    centers = v.reshape(N, 2)
    return -np.sum(compute_max_radii(centers))

def generate_initial_configs():
    """Generates diverse initial center configurations."""
    configs = []
    
    # 1. Hexagonal row patterns
    patterns = [
        [5, 5, 5, 5, 6], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4],
        [4, 6, 6, 6, 4], [5, 5, 6, 5, 5], [6, 4, 6, 5, 5],
        [5, 4, 6, 6, 5], [4, 5, 6, 5, 6], [5, 5, 5, 6, 5],
        [6, 5, 5, 5, 5], [5, 5, 5, 5, 5, 1], [4, 5, 5, 5, 6]
    ]
    for pat in patterns:
        for r0 in [0.085, 0.095, 0.105]:
            pts = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            configs.append(np.array(pts[:N]))
            
    # 2. Dense random configurations
    rng = np.random.default_rng(42)
    for _ in range(15):
        configs.append(np.clip(rng.rand(N, 2), 0.05, 0.95))
        
    # 3. Rotated hexagonal lattices
    for rot in [0.0, 0.1, 0.2, 0.3, 0.4]:
        pts = []
        for i in range(-5, 10):
            for j in range(-5, 10):
                px = i * 0.1 + (j % 2) * 0.05
                py = j * 0.0866025
                pts.append([px, py])
        pts = np.array(pts[:N])
        c, s = np.cos(rot), np.sin(rot)
        rot_pts = pts @ np.array([[c, -s], [s, c]])
        rot_pts -= rot_pts.mean(axis=0)
        scale = np.max(np.abs(rot_pts))
        if scale > 1e-9:
            rot_pts = rot_pts / scale * 0.45 + 0.5
        configs.append(np.clip(rot_pts, 0.02, 0.98))
        
    return configs

def cons_joint(v):
    """Computes boundary and non-overlap constraints for SLSQP (must be >= 0)."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = []
    
    # Boundary constraints
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    # Overlap constraints: dist >= r_i + r_j
    dx = c[:, 0, np.newaxis] - c[np.newaxis, :, 0]
    dy = c[:, 1, np.newaxis] - c[np.newaxis, :, 1]
    dists = np.hypot(dx, dy)
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    con.append(dists[TRIU_I, TRIU_J] - rs[TRIU_I, TRIU_J])
    
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    center_bounds = [(0.0, 1.0)] * (2 * N)
    joint_bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    best_v_centers = None
    best_obj = np.inf
    
    configs = generate_initial_configs()
    
    # Phase 1: Powell optimization on diverse starts
    for c0 in configs:
        try:
            res = minimize(objective_centers, c0.flatten(), method='Powell', bounds=center_bounds,
                           options={'maxiter': 6000, 'ftol': 1e-14, 'xtol': 1e-14})
            if res.fun < best_obj:
                best_obj = res.fun
                best_v_centers = res.x.copy()
        except Exception:
            continue
            
    if best_v_centers is None:
        best_v_centers = configs[0].flatten()
        
    centers = best_v_centers.reshape(N, 2)
    radii = compute_max_radii(centers)
    
    # Phase 2: Joint SLSQP polish
    v_joint = np.concatenate([centers.flatten(), radii])
    
    def obj_joint(v):
        return -np.sum(v[2 * N:])
        
    try:
        res_j = minimize(obj_joint, v_joint, method='SLSQP', bounds=joint_bounds,
                         constraints={'type': 'ineq', 'fun': cons_joint},
                         options={'maxiter': 5000, 'ftol': 1e-14})
        if -res_j.fun > np.sum(radii):
            centers = res_j.x[:2 * N].reshape(N, 2)
            radii = res_j.x[2 * N:]
    except Exception:
        pass
        
    # Phase 3: Perturbation search to escape local minima
    rng = np.random.default_rng(123)
    for _ in range(60):
        c_pert = centers.copy()
        mask = rng.random(N) < 0.3
        if mask.sum() > 0:
            c_pert[mask] += rng.normal(0, 0.003, size=(int(mask.sum()), 2))
        c_pert = np.clip(c_pert, 0.02, 0.98)
        
        try:
            res_p = minimize(objective_centers, c_pert.flatten(), method='Powell', bounds=center_bounds,
                             options={'maxiter': 2000, 'ftol': 1e-13})
            if -res_p.fun > np.sum(compute_max_radii(centers)):
                centers = res_p.x.reshape(N, 2)
                radii = compute_max_radii(centers)
        except Exception:
            continue
            
    # Phase 4: Strict validation repair
    for _ in range(30):
        valid = True
        
        # Fix boundary violations
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            mr = min(x, 1.0 - x, y, 1.0 - y)
            if r > mr + 1e-12:
                radii[i] = mr
                valid = False
                
        # Fix overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    valid = False
                    
        radii = np.maximum(radii, 0.0)
        if valid:
            break
            
    return centers, radii, float(np.sum(radii))
