import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(n*(n-1)//2 + 4*n)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            idx += 1
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-9), 0.0

def obj_centers(v):
    """Objective for optimizing centers: maximize sum of radii via LP."""
    centers = v.reshape(N, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    _, s = compute_lp_radii(centers)
    return -s

def relax_forces(centers, steps=200):
    """Simple force-directed relaxation to tighten initial packing."""
    c = centers.copy()
    for _ in range(steps):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                dx = c[i,0] - c[j,0]
                dy = c[i,1] - c[j,1]
                d = np.hypot(dx, dy)
                if d < 0.3 and d > 1e-9:
                    f = (0.3 - d) / d
                    forces[i,0] += dx * f
                    forces[i,1] += dy * f
                    forces[j,0] -= dx * f
                    forces[j,1] -= dy * f
            x, y = c[i]
            if x < 0.05: forces[i,0] += 0.05 - x
            if x > 0.95: forces[i,0] -= x - 0.95
            if y < 0.05: forces[i,1] += 0.05 - y
            if y > 0.95: forces[i,1] -= y - 0.95
        c += forces * 0.01
        c = np.clip(c, 0.02, 0.98)
    return c

def generate_starts():
    """Generates diverse initial center configurations."""
    starts = []
    rng = np.random.default_rng(42)
    
    # Hexagonal grids with rotations
    for r0 in [0.08, 0.09, 0.10, 0.11]:
        for rot in [0.0, np.pi/12, np.pi/6]:
            pts = []
            y = r0
            row = 0
            while len(pts) < N:
                x = r0 if row % 2 == 0 else 2 * r0
                while x + r0 <= 1.0 and len(pts) < N:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
                row += 1
            pts = np.array(pts[:N])
            cc, ss = np.cos(rot), np.sin(rot)
            pts = pts @ np.array([[cc, -ss], [ss, cc]])
            pts -= pts.mean(axis=0)
            mx = np.max(np.abs(pts))
            if mx > 1e-9:
                pts = pts / mx * 0.4 + 0.5
            starts.append(relax_forces(np.clip(pts, 0.05, 0.95), steps=100))

    # Square grids
    for r0 in [0.08, 0.09, 0.10]:
        pts = []
        y = r0
        row = 0
        while len(pts) < N:
            x = r0
            while x + r0 <= 1.0 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r0
            y += 2.0 * r0
            row += 1
        starts.append(relax_forces(np.clip(np.array(pts[:N]), 0.05, 0.95), steps=100))
        
    # Random dense starts
    for _ in range(10):
        starts.append(relax_forces(rng.uniform(0.15, 0.85, (N, 2)), steps=150))
        
    # Specific row patterns known to be dense
    patterns = [[5,5,5,5,5,1], [6,5,6,5,4], [5,6,5,6,4]]
    for pat in patterns:
        pts = []
        y = 0.1
        r_est = 0.09
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3) * r_est
        starts.append(relax_forces(np.clip(np.array(pts[:N]), 0.05, 0.95), steps=100))

    return starts[:20]

def obj_full(v):
    """Objective for full variable optimization."""
    return -np.sum(v[2*N:])

def con_full(v):
    """Constraints for full variable optimization."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    d = np.sqrt(np.sum(diff**2, axis=2))
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    mask = np.triu_indices(N, k=1)
    con.append(d[mask] - rs[mask])
    return np.concatenate(con)

def run_packing():
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_centers = None
    best_sum = -np.inf
    starts = generate_starts()
    bounds_c = [(0.0, 1.0)] * (2 * N)
    
    # Phase 1: Optimize centers using derivative-free method with LP radii evaluation
    for c_init in starts:
        try:
            res = minimize(obj_centers, c_init.flatten(), method='Powell', 
                           bounds=bounds_c, options={'maxiter': 2500, 'ftol': 1e-12, 'xtol': 1e-12})
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Phase 2: Perturbation search to escape local minima
    if best_centers is not None:
        for _ in range(15):
            c_pert = best_centers + np.random.normal(0, 0.002, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            try:
                res = minimize(obj_centers, c_pert.flatten(), method='Powell',
                               bounds=bounds_c, options={'maxiter': 1500, 'ftol': 1e-12})
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_centers = res.x.reshape(N, 2)
            except Exception:
                pass
                
    # Phase 3: SLSQP polish on full variables (centers + radii)
    if best_centers is not None:
        radii_init, _ = compute_lp_radii(best_centers)
        v0 = np.concatenate([best_centers.flatten(), radii_init])
        bounds_full = [(0.0, 1.0)]*(2*N) + [(1e-7, 0.5)]*N
        try:
            res = minimize(obj_full, v0, method='SLSQP', bounds=bounds_full,
                           constraints={'type': 'ineq', 'fun': con_full},
                           options={'maxiter': 5000, 'ftol': 1e-13})
            if np.min(con_full(res.x)) > -1e-6:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_centers = res.x[:2*N].reshape(N, 2)
        except Exception:
            pass

    # Final exact LP radius computation
    radii, final_sum = compute_lp_radii(best_centers)
    
    # Strict safety repair to guarantee validation compliance
    for _ in range(50):
        changed = False
        for i in range(N):
            x, y, r = best_centers[i,0], best_centers[i,1], radii[i]
            mx = min(x, 1.0-x, y, 1.0-y)
            if r > mx + 1e-12:
                radii[i] = mx
                changed = True
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                if d < radii[i] + radii[j] - 1e-12:
                    shrink = (radii[i] + radii[j] - d)/2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    
    return best_centers, radii, float(np.sum(radii))
