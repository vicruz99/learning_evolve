This LP approach avoids the non-differentiable `min` operator often used in heuristic objectives and provides exact gradients implicitly.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute LP constraint matrix structure for efficiency
def get_lp_matrix():
    num_constraints = N*(N-1)//2 + N
    A = np.zeros((num_constraints, N))
    idx = 0
    for i in range(N):
        for j in range(i+1, N):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
    for i in range(N):
        A[idx, i] = 1.0
        idx += 1
    return A

A_LP = get_lp_matrix()

def solve_lp_for_radii(centers):
    """Given fixed centers, solve LP to exactly maximize sum of radii."""
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    b_ub = np.zeros(A_LP.shape[0])
    idx = 0
    for i in range(N):
        for j in range(i+1, N):
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    x, y = centers[:, 0], centers[:, 1]
    bnds = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    bnds = np.maximum(bnds, 1e-9)  # Prevent negative bounds
    b_ub[idx:] = bnds
    
    # LP: max sum(r_i) s.t. A_ub @ r <= b_ub, r >= 1e-9
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=(1e-9, None), method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(N, 1e-9), 0.0

def obj_func(v):
    """Objective for center optimization: minimize negative sum of radii."""
    c = v.reshape(N, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    _, s = solve_lp_for_radii(c)
    return -s

def force_init(seed, steps=250):
    """Generate spread-out configuration via repulsive forces."""
    rng = np.random.default_rng(seed)
    c = rng.uniform(0.2, 0.8, (N, 2))
    for _ in range(steps):
        f = np.zeros_like(c)
        for i in range(N):
            for j in range(i+1, N):
                d = c[i] - c[j]
                dist = np.linalg.norm(d)
                if dist < 0.25 and dist > 1e-6:
                    force = (0.25 - dist) * 10.0 / dist
                    f[i] += d * force
                    f[j] -= d * force
        c += f * 0.002
        c = np.clip(c, 0.05, 0.95)
    return c

def hex_init(r0, shift_x=0, shift_y=0):
    """Generate hexagonal lattice configuration."""
    pts = []
    y = r0 + shift_y
    row = 0
    while len(pts) < N:
        x = r0 + shift_x if row % 2 == 0 else 2*r0 + shift_x
        while x + r0 <= 1.0 - 1e-7 and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = [(1e-6, 1.0 - 1e-6)] * (2 * N)
    
    best_sum = -np.inf
    best_c = None
    
    starts = []
    
    # 1. Hex lattices with variations
    for r0 in [0.085, 0.09, 0.095, 0.10, 0.105, 0.11]:
        for sx in [-0.02, 0.0, 0.02]:
            for sy in [-0.02, 0.0, 0.02]:
                try:
                    starts.append(hex_init(r0, sx, sy))
                except:
                    pass
                    
    # 2. Force directed layouts
    for s in range(20):
        starts.append(force_init(s))
        
    # 3. Square grids
    for sp in [0.2, 0.22, 0.24, 0.25]:
        pts = []
        y = sp
        while y + sp <= 1.0 - 1e-7 and len(pts) < N:
            x = sp
            while x + sp <= 1.0 - 1e-7 and len(pts) < N:
                pts.append([x, y])
                x += sp
            y += sp
        starts.append(np.array(pts[:N]))
        
    # Phase 1: Multi-start Powell optimization
    for c0 in starts:
        c0 = np.clip(c0, 1e-6, 1.0 - 1e-6)
        # Add tiny perturbation to break exact symmetry
        c0 += np.random.normal(0, 0.002, c0.shape)
        c0 = np.clip(c0, 1e-6, 1.0 - 1e-6)
        
        try:
            res = minimize(obj_func, c0.flatten(), method='Powell', bounds=bounds,
                           options={'maxiter': 1500, 'ftol': 1e-12, 'xtol': 1e-12})
            if -res.fun > best_sum - 1e-7:
                best_sum = -res.fun
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    # Phase 2: Local perturbation search to escape local minima
    if best_c is not None:
        curr_c = best_c.copy()
        for step in range(60):
            noise = 0.004 * (0.9 ** step)
            c_pert = curr_c + np.random.normal(0, noise, curr_c.shape)
            c_pert = np.clip(c_pert, 1e-6, 1.0 - 1e-6)
            try:
                res = minimize(obj_func, c_pert.flatten(), method='Powell', bounds=bounds,
                               options={'maxiter': 800, 'ftol': 1e-12, 'xtol': 1e-12})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_c = res.x.reshape(N, 2).copy()
                    curr_c = best_c
            except Exception:
                pass
                
    # Phase 3: Exact LP resolution for final centers
    best_r, _ = solve_lp_for_radii(best_c)
    
    # Safety shrink to guarantee validation tolerance compliance
    best_r *= 0.9999999
    
    # Phase 4: Strict iterative repair for numerical robustness
    for _ in range(10):
        valid = True
        # Fix boundary violations
        for i in range(N):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                best_r[i] = max_r
                valid = False
        # Fix overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                req = best_r[i] + best_r[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-10
                    best_r[i] -= shrink
                    best_r[j] -= shrink
                    valid = False
        best_r = np.maximum(best_r, 1e-9)
        if valid:
            break
            
    return best_c, best_r, float(np.sum(best_r))
