import numpy as np
from scipy.optimize import minimize, linprog

N = 26
NUM_PAIRS = N * (N - 1) // 2

def compute_radii_fast(centers):
    """Computes maximum feasible radius for each circle given fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_pair = 0.5 * np.min(dists, axis=1)
    
    return np.minimum(r_bound, r_pair)

def objective_centers(v):
    """Objective: maximize sum of radii -> minimize negative sum."""
    c = v.reshape(N, 2)
    c = np.clip(c, 1e-7, 1.0 - 1e-7)
    return -np.sum(compute_radii_fast(c))

def solve_lp_radii(centers):
    """Solves exact LP to maximize sum of radii for fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Build LP constraints: r_i + r_j <= dist_ij, r_i <= r_bound_i, r_i >= 0
    A_ub = np.zeros((NUM_PAIRS + N, N))
    b_ub = np.zeros(NUM_PAIRS + N)
    
    idx = 0
    iu, ju = np.triu_indices(N, k=1)
    for i, j in zip(iu, ju):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        b_ub[idx] = dists[i, j]
        idx += 1
        
    for i in range(N):
        A_ub[idx, i] = 1.0
        b_ub[idx] = r_bound[i]
        idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*N, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0), -res.fun
    return np.full(N, 0.01), 0.26

def generate_starts(num_starts):
    """Generates diverse initial center configurations."""
    starts = []
    rng = np.random.default_rng(42)
    
    # 1. Hexagonal lattices with varying base radii
    for r0 in [0.085, 0.090, 0.095, 0.100, 0.105]:
        pts = []
        y = r0
        row = 0
        while len(pts) < N:
            x = r0 if row % 2 == 0 else 2 * r0
            while x + r0 <= 1.0 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        starts.append(np.array(pts[:N]))
        
    # 2. Rotated hex lattices
    for ang in [0.15, 0.3, -0.15, -0.3]:
        base = starts[0].copy()
        c, s = np.cos(ang), np.sin(ang)
        rot = base - 0.5
        rot = rot @ np.array([[c, -s], [s, c]]) + 0.5
        starts.append(np.clip(rot, 0.02, 0.98))
        
    # 3. Dense random configurations
    for _ in range(10):
        starts.append(rng.uniform(0.12, 0.88, (N, 2)))
        
    # 4. Structured square grids
    for sp in [0.16, 0.18, 0.20]:
        pts = []
        y = sp
        while len(pts) < N and y <= 1.0 - sp:
            x = sp
            while len(pts) < N and x <= 1.0 - sp:
                pts.append([x, y])
                x += sp
            y += sp
        if len(pts) >= N:
            starts.append(np.array(pts[:N]))
            
    return starts[:num_starts]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    
    best_v = None
    best_sum = -np.inf
    bounds = [(0.001, 0.999)] * (2 * N)
    
    starts = generate_starts(30)
    
    # Phase 1: Multi-start Powell optimization
    for s in starts:
        s_pert = s + np.random.normal(0, 0.005, s.shape)
        s_pert = np.clip(s_pert, 0.01, 0.99)
        
        try:
            res = minimize(objective_centers, s_pert.flatten(), method='Powell',
                           bounds=bounds, options={'maxiter': 2500, 'ftol': 1e-14, 'xtol': 1e-14})
            if res.fun < best_sum:
                best_sum = res.fun
                best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = starts[0].flatten()
        
    # Phase 2: Iterative perturbation search to escape local minima
    curr_v = best_v.copy()
    step = 0.008
    for epoch in range(60):
        improved = False
        for _ in range(8):
            v_pert = curr_v + np.random.normal(0, step, curr_v.shape)
            v_pert = np.clip(v_pert, 0.01, 0.99)
            
            try:
                res = minimize(objective_centers, v_pert, method='Powell',
                               bounds=bounds, options={'maxiter': 1200, 'ftol': 1e-14, 'xtol': 1e-14})
                if res.fun < best_sum - 1e-9:
                    best_sum = res.fun
                    best_v = res.x.copy()
                    curr_v = best_v.copy()
                    improved = True
            except Exception:
                pass
                
        if not improved:
            step *= 0.88
        if step < 1e-6:
            break
            
    # Phase 3: Exact LP polish for optimal radii given best centers
    centers = best_v.reshape(N, 2)
    radii, final_sum = solve_lp_radii(centers)
    
    # Phase 4: Strict validation repair to guarantee compliance with 1e-12 tolerance
    for _ in range(30):
        changed = False
        
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r_bound = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r_bound + 1e-12:
                radii[i] = max_r_bound
                changed = True
                
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        radii = np.maximum(radii, 0.0)
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
