import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, k=1)

def compute_constraints(v):
    """
    Computes boundary and non-overlap constraints.
    Returns a 1D array where all elements must be >= 0 for feasibility.
    """
    centers = v[:2 * N].reshape(N, 2)
    radii = v[2 * N:]
    
    con = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con.append(centers[:, 0] - radii)
    con.append(1.0 - centers[:, 0] - radii)
    con.append(centers[:, 1] - radii)
    con.append(1.0 - centers[:, 1] - radii)
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    con.append(dists[TRIU_I, TRIU_J] - radii_sum[TRIU_I, TRIU_J])
    
    return np.concatenate(con)

def objective_func(v):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2 * N:])

def get_lp_radii(centers):
    """Given fixed centers, solve LP to exactly maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        ub = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(ub)
        
    res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                  bounds=(0, None), method='highs')
    return res.x if res.success else np.zeros(n)

def generate_starts():
    """Generates diverse hexagonal and random initial configurations."""
    starts = []
    # Common row distributions for dense hexagonal packings of ~26 items
    pats = [[6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,5,5,6], 
            [6,5,5,5,5], [4,6,6,6,4], [5,6,6,5,4], [6,4,6,5,5]]
    for pat in pats:
        while sum(pat) < N: pat.append(1)
        if sum(pat) > N: pat[-1] = N - sum(pat[:-1])
        if sum(pat) != N: continue
        
        for s in [0.085, 0.09, 0.095, 0.10, 0.105]:
            pts = []
            y = s
            for r_idx, cnt in enumerate(pat):
                shift = s if r_idx % 2 == 1 else 0.0
                x = s + shift
                for _ in range(cnt):
                    pts.append([x, y])
                    x += 2.0 * s
                y += np.sqrt(3) * s
            if len(pts) == N:
                pts = np.array(pts[:N])
                starts.append(pts)
                
    # Add random dense starts to cover non-hexagonal topologies
    for _ in range(8):
        c = np.random.rand(N, 2) * 0.8 + 0.1
        starts.append(c)
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-7, 0.5)] * N
    constraints = {'type': 'ineq', 'fun': compute_constraints}
    
    best_sum = -np.inf
    best_v = None
    
    starts = generate_starts()
    
    # Phase 1: Multi-start SLSQP optimization
    for c_init in starts:
        # Compute a strictly feasible initial radius guess
        r_init = np.full(N, 0.08)
        for i in range(N):
            db = min(c_init[i,0], 1.0-c_init[i,0], c_init[i,1], 1.0-c_init[i,1])
            dn = min([np.hypot(c_init[i,0]-c_init[j,0], c_init[i,1]-c_init[j,1]) for j in range(N) if i!=j])/2.0
            r_init[i] = min(db, dn) * 0.95
        r_init = np.maximum(r_init, 1e-6)
        
        v0 = np.concatenate([c_init.flatten(), r_init])
        try:
            res = minimize(objective_func, v0, method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            # Verify feasibility with a comfortable tolerance
            if np.min(compute_constraints(res.x)) > -1e-5:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation search to escape local minima
    if best_v is not None:
        curr_v = best_v.copy()
        for step in range(60):
            noise = 0.006 * (0.92**step)
            v_pert = curr_v + np.random.normal(0, noise, curr_v.shape)
            v_pert = np.clip(v_pert, 1e-6, 1.0-1e-6)
            v_pert[2*N:] = np.clip(v_pert[2*N:], 1e-6, 0.5)
            try:
                res = minimize(objective_func, v_pert, method='SLSQP', bounds=bounds,
                               constraints=constraints, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                if np.min(compute_constraints(res.x)) > -1e-5:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_v = res.x.copy()
                        curr_v = best_v
            except Exception:
                pass
                
    # Extract optimized centers
    centers = best_v[:2*N].reshape(N, 2)
    
    # Phase 3: Exact LP refinement for radii given the optimized geometry
    radii = get_lp_radii(centers)
    
    # Phase 4: Strict numerical repair to guarantee validator compliance
    for _ in range(30):
        changed = False
        # Resolve overlaps by shrinking radii proportionally
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d)/2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-9:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
