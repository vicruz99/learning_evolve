To improve the sum of radii for packing 26 circles in a unit square, we shift from joint optimization of centers and radii to a more robust alternating approach:
1. **Center Optimization with LP Radii**: Fix centers, then solve a Linear Program (LP) to find the exact optimal radii that maximize $\sum r_i$ subject to $r_i + r_j \leq d_{ij}$ and boundary constraints. This LP subproblem is solved exactly and efficiently.
2. **Derivative-Free Center Search**: Since the LP solution is piecewise-linear and non-smooth with respect to centers, we use `scipy.optimize.minimize` with the `Powell` method. Powell handles non-smooth objectives well and avoids gradient computation issues.
3. **Diverse Initialization**: Generate multiple hexagonal lattice patterns with varying densities and row counts, plus force-dispersed random configurations, to ensure we explore different topological basins.
4. **Local Perturbation & Polish**: After Powell finds a local optimum, run a coordinate-wise perturbation search to escape shallow local minima, followed by a final Powell polish.
5. **Strict Validation Safety**: Multiply final radii by a factor slightly less than 1 to guarantee strict non-overlap within floating-point tolerances required by the validator.

This strategy leverages the mathematical structure of the problem (LP for radii is exact and fast for fixed centers) and avoids the non-convex constraint pitfalls of joint SLSQP optimization.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_COUNT = N * (N - 1) // 2
IU, JU = np.triu_indices(N, k=1)

# Precompute the constant inequality matrix A_ub for the LP
# Constraints: r_i + r_j <= d_ij  (PAIR_COUNT rows)
#              r_i <= dist_to_boundary (N rows)
A_ub = np.zeros((PAIR_COUNT + N, N))
for k, (i, j) in enumerate(zip(IU, JU)):
    A_ub[k, i] = 1.0
    A_ub[k, j] = 1.0
for i in range(N):
    A_ub[PAIR_COUNT + i, i] = 1.0

def get_max_sum_radii(centers):
    """
    Solves the LP to find optimal radii for fixed centers.
    Maximizes sum(r_i) subject to non-overlap and boundary constraints.
    """
    xs = centers[:, 0]
    ys = centers[:, 1]
    
    # Pairwise distances
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dists = np.hypot(dx, dy)
    
    # Update b_ub vector
    b_ub = np.zeros(PAIR_COUNT + N)
    b_ub[:PAIR_COUNT] = dists[IU, JU]
    b_ub[PAIR_COUNT:] = np.minimum(np.minimum(xs, 1.0 - xs), np.minimum(ys, 1.0 - ys))
    
    # Solve LP: minimize -sum(r) => maximize sum(r)
    c_obj = -np.ones(N)
    bounds = [(0.0, None)] * N
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def obj_centers(v):
    """Objective for center-only optimization: maximize LP sum of radii."""
    c = np.clip(v.reshape(N, 2), 1e-7, 1.0 - 1e-7)
    s, _ = get_max_sum_radii(c)
    return -s

def run_packing():
    np.random.seed(42)
    best_sum = 0.0
    best_centers = None
    
    configs = []
    
    # 1. Hexagonal lattice patterns with varying densities
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5],
        [6, 6, 5, 5, 4], [4, 6, 6, 6, 4], [5, 5, 5, 5, 6],
        [6, 5, 5, 5, 5], [5, 5, 6, 4, 6]
    ]
    
    for pat in patterns:
        for r_est in [0.088, 0.094, 0.100, 0.106, 0.112]:
            pts = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                x_start = r_est + (r_idx % 2) * r_est
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x_start, y])
                    x_start += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            if len(pts) == N:
                c = np.array(pts)
                # Small perturbation to break symmetry and avoid degenerate starts
                c += np.random.normal(0, 0.0012, c.shape)
                c = np.clip(c, 0.02, 0.98)
                configs.append(c)
                
    # 2. Force-dispersed random starts
    rng = np.random.default_rng(42)
    for _ in range(6):
        c = rng.uniform(0.1, 0.9, (N, 2))
        # Quick repulsion pass to spread circles out
        for _ in range(60):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
                    if d < 0.18 and d > 1e-8:
                        f = (0.18 - d) * 8.0 / d
                        fx, fy = c[i, 0] - c[j, 0], c[i, 1] - c[j, 1]
                        forces[i] += [fx, fy] * f
                        forces[j] -= [fx, fy] * f
                # Boundary repulsion
                for d_val, d_sign, ax in [(c[i, 0], -1, 0), (1.0 - c[i, 0], 1, 0), 
                                          (c[i, 1], -1, 1), (1.0 - c[i, 1], 1, 1)]:
                    if d_val < 0.06:
                        forces[i, ax] += d_sign * (0.06 - d_val) * 15.0
            c += forces * 0.0004
            c = np.clip(c, 0.02, 0.98)
        configs.append(c)
        
    bounds_center = [(0.0, 1.0)] * (2 * N)
    
    # Phase 1: Powell Optimization on Centers
    for c_init in configs:
        try:
            res = minimize(obj_centers, c_init.flatten(), method='Powell',
                           bounds=bounds_center,
                           options={'maxiter': 2500, 'ftol': 1e-11, 'xtol': 1e-11})
            s, _ = get_max_sum_radii(res.x.reshape(N, 2))
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Phase 2: Local Perturbation Search to escape local minima
    rng_p = np.random.default_rng(123)
    step = 0.003
    for _ in range(250):
        c_trial = best_centers.copy()
        i = rng_p.integers(N)
        c_trial[i] += rng_p.uniform(-step, step, 2)
        c_trial[i] = np.clip(c_trial[i], 0.01, 0.99)
        
        s, _ = get_max_sum_radii(c_trial)
        if s > best_sum:
            best_sum = s
            best_centers = c_trial.copy()
            step = min(step * 1.08, 0.008)
        else:
            step *= 0.985
            
    # Phase 3: Final Powell Polish
    try:
        res = minimize(obj_centers, best_centers.flatten(), method='Powell',
                       bounds=bounds_center, options={'maxiter': 1500, 'ftol': 1e-12})
        s, _ = get_max_sum_radii(res.x.reshape(N, 2))
        if s > best_sum:
            best_sum = s
            best_centers = res.x.reshape(N, 2)
    except Exception:
        pass
        
    # Compute final exact radii
    final_radii, _ = get_max_sum_radii(best_centers)
    
    # Strict safety shrink to guarantee validator tolerance (1e-12)
    final_radii *= 0.999999999
    
    # Ensure non-negativity
    final_radii = np.maximum(final_radii, 0.0)
    
    return best_centers, final_radii, float(np.sum(final_radii))
