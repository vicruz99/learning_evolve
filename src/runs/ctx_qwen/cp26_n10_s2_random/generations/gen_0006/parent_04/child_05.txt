import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute LP inequality matrix structure: r_i + r_j <= d_ij  and  r_i <= b_i
_PAIRS = [(i, j) for i in range(N) for j in range(i + 1, N)]
_AUB_ROWS = N * (N - 1) // 2 + N
_AUB = np.zeros((_AUB_ROWS, N))
_idx = 0
for i, j in _PAIRS:
    _AUB[_idx, i] = 1.0
    _AUB[_idx, j] = 1.0
    _idx += 1
for i in range(N):
    _AUB[_idx, i] = 1.0
    _idx += 1

def get_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    bnds = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
                      
    b_ub = np.zeros(_AUB_ROWS)
    for k, (i, j) in enumerate(_PAIRS):
        b_ub[k] = dists[i, j]
    for i in range(N):
        b_ub[len(_PAIRS) + i] = bnds[i]
        
    res = linprog(-np.ones(N), A_ub=_AUB, b_ub=b_ub, 
                  bounds=[(0.0, None)] * N, method='highs')
    if res.success:
        return res.x
    return np.full(N, 1e-6)

def obj_func(c_flat):
    """Objective for center optimization: maximize sum of radii."""
    c = np.clip(c_flat.reshape(N, 2), 1e-5, 0.99)
    return -np.sum(get_lp_radii(c))

def hex_config(rows, r0):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r0
    for r_idx, cnt in enumerate(rows):
        x = r0 + (r_idx % 2) * r0
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_val = np.inf
    best_c = None
    
    starts = []
    pats = [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [6,6,5,5,4], [4,6,6,6,4]]
    for p in pats:
        for r in [0.085, 0.095, 0.105, 0.115]:
            c = hex_config(p, r)
            c += np.random.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.02, 0.98)
            starts.append(c.flatten())
            
    for _ in range(8):
        c = np.random.rand(N, 2) * 0.8 + 0.1
        starts.append(c.flatten())
        
    for x0 in starts:
        try:
            res = minimize(obj_func, x0, method='Powell',
                           options={'maxiter': 3000, 'ftol': 1e-13, 'xtol': 1e-13})
            if res.fun < best_val:
                best_val = res.fun
                best_c = res.x.reshape(N, 2)
        except Exception:
            pass
            
    if best_c is None:
        best_c = starts[0].reshape(N, 2)
        
    # Adaptive coordinate descent local search
    c = best_c.copy()
    step = 0.005
    for _ in range(300):
        improved = False
        for i in range(N):
            best_local = -np.sum(get_lp_radii(c))
            for dx, dy in [(step,0), (-step,0), (0,step), (0,-step),
                           (step,step), (-step,step), (step,-step), (-step,-step)]:
                nc = c.copy()
                nc[i] += [dx, dy]
                nc[i] = np.clip(nc[i], 1e-5, 0.99)
                val = -np.sum(get_lp_radii(nc))
                if val < best_local - 1e-11:
                    c = nc.copy()
                    best_local = val
                    improved = True
        if not improved:
            step *= 0.85
        if step < 1e-6: break
            
    radii = get_lp_radii(c)
    
    # Strict repair pass to guarantee validator tolerance
    for _ in range(50):
        changed = False
        for i in range(N):
            x, y = c[i]
            mx = min(x, 1.0-x, y, 1.0-y)
            if radii[i] > mx + 1e-10:
                radii[i] = mx
                changed = True

                    shr = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shr
                    radii[j] -= shr
                    changed = True
        if not changed: break
    radii = np.maximum(radii, 0.0)
    
    return c, radii, float(np.sum(radii))
