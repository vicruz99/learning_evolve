from scipy.optimize import minimize, linprog

N = 26
IU, JU = np.triu_indices(N, k=1)

def compute_radii_fast(centers):
    """Computes maximum feasible radii for fixed centers."""
    n = centers.shape[0]
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    rp = 0.5 * np.min(dists, axis=1)
    return np.minimum(rb, rp)

def solve_lp_radii(centers):

    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))

    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)

    if res.success:
        return res.x
    return np.maximum(ub * 0.9, 0.0)

def obj_centers(v):
    """Objective for center-only optimization: maximize sum of radii."""
    centers = v.reshape(N, 2)
    r = compute_radii_fast(centers)
    return -np.sum(r)

def cons_slscp(v):
    """Inequality constraints for SLSQP (must be >= 0)."""
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    con = [xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs]

    dy = ys[IU] - ys[JU]
    dr = rs[IU] + rs[JU]
    con.append(np.sqrt(dx**2 + dy**2 + 1e-15) - dr)
    return np.concatenate(con)

def obj_slscp(v):
    """Objective for full variable optimization."""
    return -np.sum(v[2::3])
