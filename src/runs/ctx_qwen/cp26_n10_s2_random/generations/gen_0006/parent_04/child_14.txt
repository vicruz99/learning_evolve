import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_radii(centers):
    """Computes the maximum valid radius for each circle given fixed centers."""
    n = centers.shape[0]
    # Distance to boundaries
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    rp = 0.5 * np.min(dists, axis=1)
    return np.minimum(rb, rp)

def obj_func_centers(c_flat):
    """Objective for center-only optimization: minimize negative sum of radii."""
    c = c_flat.reshape(N, 2)
    # Strictly keep centers inside the unit square to ensure valid radii
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    r = compute_radii(c)
    return -np.sum(r)

def get_lp_radii(centers):
    """Given fixed centers, solve LP to exactly maximize sum of radii."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints: r_i <= ub[i]
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(ub[i])
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if res.success:
        return res.x
    return np.maximum(ub * 0.5, 1e-9)

def compute_constraints_joint(v):
    """Computes all boundary and non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r, r])
    
    # Pairwise non-overlap constraints: dist(i,j) - (r_i + r_j) >= 0
    idx = np.triu_indices(N, k=1)
    dx = x[idx[0]] - x[idx[1]]
    dy = y[idx[0]] - y[idx[1]]
    dr = r[idx[0]] + r[idx[1]]
    c = np.concatenate([c, np.hypot(dx, dy) - dr])
    
    return c

def get_bounds_joint():
    """Variable bounds for joint optimization."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-9, 0.5)])
    return b

def hex_init(n, r, rows):
    """Generates hexagonal lattice positions based on row counts."""
    pos = []
    y = r
    for r_idx, count in enumerate(rows):
        x_start = r + (r_idx % 2) * r
        for c in range(count):
            x = x_start + c * (2.0 * r)
            pos.append([x, y])
        y += r * np.sqrt(3.0)
    pts = np.array(pos[:n])
    
    # Normalize to fit comfortably inside [0,1]^2
    if pts.size > 0:
        pts -= pts.min(axis=0)
        span = pts.max(axis=0) - pts.min(axis=0)
        if np.any(span > 1e-9):
            pts = pts / span * 0.8 + 0.1
    return pts

            break
            
    best_radii = np.maximum(best_radii, 0.0)
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
