import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_constraints(v):
    """
    Computes boundary and non-overlap constraints.
    Returns a 1D array where each element must be >= 0.
    Uses squared distances for better gradient conditioning.
    """
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    
    cons = []
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap: ||c_i - c_j||^2 >= (r_i + r_j)^2
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    d2 = np.sum(diff**2, axis=2)
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    idx = np.triu_indices(N, k=1)
    cons.append(d2[idx] - rs[idx]**2)
    
    return np.concatenate(cons)

def objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def get_lp_radii(centers):
    """
    Given fixed centers, solve an LP to exactly maximize the sum of radii.
    Constraints: r_i + r_j <= dist(i,j), r_i <= dist(i, boundary)
    """
    c_obj = -np.ones(N)
    A_ub = []
    b_ub = []
    
    # Pairwise constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    for i in range(N):
        for j in range(i + 1, N):
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints
    for i in range(N):
        bounds_i = [centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1]]
        for b_val in bounds_i:
            row = np.zeros(N)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b_val)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None

def generate_inits():
    """Generates a diverse set of initial configurations."""
    inits = []
    rng = np.random.default_rng(42)
    
    # Hexagonal lattice patterns (row counts summing to ~26)
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [6, 5, 5, 6, 4],
        [5, 7, 5, 5, 4], [6, 4, 6, 6, 4], [5, 6, 6, 5, 4],
        [7, 5, 6, 5, 3], [5, 5, 5, 6, 5], [6, 6, 6, 4, 4]
    ]
    
    for pat in patterns:
        if sum(pat) < N:
            pat.append(N - sum(pat))
        for r0 in [0.092, 0.098, 0.105, 0.112, 0.118]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift

                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:N])

            c += rng.normal(0, 0.004, c.shape)
            c = np.clip(c, 0.02, 0.98)
            
            # Compute safe initial radii

            dists = np.linalg.norm(c[:,None,:] - c[None,:,:], axis=2)
            np.fill_diagonal(dists, np.inf)
            rp = 0.5 * np.min(dists, axis=1)
            r_init = np.minimum(rb, rp) * 0.96
            
            v = np.zeros(N*3)
            v[:N] = c[:,0]
            v[N:2*N] = c[:,1]
            v[2*N:] = r_init
            inits.append(v)
            
    # Random dense starts
    for _ in range(12):
        c = rng.uniform(0.15, 0.85, (N, 2))
        rb = np.minimum(np.minimum(c[:,0], 1.0-c[:,0]), np.minimum(c[:,1], 1.0-c[:,1]))
        dists = np.linalg.norm(c[:,None,:] - c[None,:,:], axis=2)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_v = None
    best_sum = -np.inf
    inits = generate_inits()
    
    # Phase 1: Multi-start SLSQP
    for v0 in inits:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-12, 'disp': False})
            if np.all(compute_constraints(res.x) >= -1e-8):
                s = -res.fun

                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = inits[0]

            try:

                        best_v = res.x.copy()

                        improved = True
                        break
            except Exception:
                pass
                
        if not improved:
            step *= 0.92
            if step < 1e-6:
                break

    c_best = best_v[:2*N].reshape(N, 2)
    r_lp = get_lp_radii(c_best)
    if r_lp is not None:
        sum_lp = np.sum(r_lp)
        if sum_lp > best_sum:
            best_sum = sum_lp
            best_v[:2*N] = c_best.flatten()
            best_v[2*N:] = r_lp
            
    # Phase 4: Final strict repair to guarantee validator tolerance
    for _ in range(40):
        c_vals = compute_constraints(best_v)
        if np.min(c_vals) >= -1e-9:
            break
        best_v[2*N:] *= 0.9996
        
    centers = best_v[:2*N].reshape(N, 2)
    radii = np.maximum(best_v[2*N:], 0.0)
    
    # Explicit overlap/boundary repair pass
    for _ in range(50):
        changed = False
        # Boundary clamp
        for i in range(N):
            x, y, r = centers[i,0], centers[i,1], radii[i]
            mx = min(x, 1.0-x, y, 1.0-y)
            if r > mx - 1e-12:
                radii[i] = mx
                changed = True
        # Overlap shrink
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-12:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-12
                    radii[i] -= shrink * 0.5
                    radii[j] -= shrink * 0.5
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
