from scipy.optimize import minimize, linprog

N = 26

def get_lp_radii(centers):
    """Given fixed centers, solve LP to exactly maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Pairwise distance constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(centers[i, 0])
        A_ub.append(row.copy())
        b_ub.append(1.0 - centers[i, 0])
        A_ub.append(row.copy())
        b_ub.append(centers[i, 1])
        A_ub.append(row.copy())
        b_ub.append(1.0 - centers[i, 1])
        
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                      bounds=(0.0, None), method='highs')

            return res.x
    except Exception:
        pass
    return np.full(n, 1e-9)

    c = v.reshape(N, 2)
    c = np.clip(c, 1e-9, 1.0 - 1e-9)

    return -np.sum(r)

    pts = []
    y = r

        x = r + (r_idx % 2) * r
        for _ in range(count):
            pts.append([x, y])
            x += 2.0 * r
        y += r * np.sqrt(3.0)

    # Normalize to fit within [0.075, 0.925] preserving aspect ratio
    pts -= pts.min(axis=0)
    dim_max = np.max(pts.max(axis=0) - pts.min(axis=0))
    if dim_max > 1e-9:
        scale = 0.85 / dim_max
        pts = pts * scale
    pts += 0.075
    
    # Rotate around center if needed
    if abs(rot) > 1e-6:
        c, s = np.cos(rot), np.sin(rot)
        center = np.array([0.5, 0.5])
        pts = pts - center
        rotated = pts @ np.array([[c, -s], [s, c]])
        pts = rotated + center
        
    return np.clip(pts, 0.01, 0.99)

def force_relax(centers, steps=150):
    """Simple force-directed repulsion to spread out overlapping/random points."""
    c = centers.copy()
    for _ in range(steps):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = np.sqrt(dx*dx + dy*dy)
                if d < 0.22 and d > 1e-6:
                    f = (0.22 - d) * 0.4 / d
                    forces[i, 0] += dx * f
                    forces[i, 1] += dy * f
                    forces[j, 0] -= dx * f
                    forces[j, 1] -= dy * f
            cx, cy = c[i]

            if cy > 0.88: forces[i, 1] -= (cy - 0.88) * 2.0
        c += forces * 0.008
        c = np.clip(c, 0.05, 0.95)
    return c
