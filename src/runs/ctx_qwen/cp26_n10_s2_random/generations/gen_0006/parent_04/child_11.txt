import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def get_lp_radii(centers):
    """
    Solves the Linear Program to find optimal radii for fixed centers.
    Maximizes sum(r_i) subject to r_i + r_j <= dist(i,j) and boundary constraints.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs + 4 * n, n))
    b_ub = np.zeros(num_pairs + 4 * n)
    
    idx = 0
    # Pairwise non-overlap constraints: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b_ub[idx] = np.sqrt(dx*dx + dy*dy)
            idx += 1
            
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        vals = [centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1]]
        for val in vals:
            A_ub[idx, i] = 1.0
            b_ub[idx] = val
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 0.01), 0.26

def objective_func(x_flat):
    """Objective for center optimization: maximize sum of optimal radii."""
    c = x_flat.reshape(N, 2)
    # Clip to prevent invalid LP bounds while allowing optimization to explore
    c = np.clip(c, 1e-3, 1.0 - 1e-3)
    _, s = get_lp_radii(c)
    return -s
