import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r)."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Distance to boundaries
    bnd = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                     np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Build LP matrices: A_ub x <= b_ub
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs + n, n))
    b_ub = np.zeros(num_pairs + n)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ub[idx] = bnd[i]
        idx += 1
        
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0.0, 0.5), method='highs')
    if res.success:
        return res.x
    return np.full(n, 0.01)

def obj_centers(v):
    """Objective for center-only optimization: minimize negative sum of LP radii."""
    c = v.reshape(N, 2)
    # Keep strictly inside to avoid degenerate LP
    c = np.clip(c, 1e-5, 1.0 - 1e-5)
    r = solve_lp_radii(c)
    return -np.sum(r)

    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
