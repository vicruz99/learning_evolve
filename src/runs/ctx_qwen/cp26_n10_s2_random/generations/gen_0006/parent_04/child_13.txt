2. **Center Optimization with LP Radii**: The sum of radii for fixed centers is a linear programming problem. We use `scipy.optimize.linprog` to compute the exact optimal radii for any given center configuration. This transforms the problem into optimizing only the 52 center coordinates.
3. **Derivative-Free Optimization**: The objective function (sum of LP-optimized radii) is continuous but non-smooth. `Powell`'s method is highly effective for such landscapes. We run multi-start Powell optimization to find strong local maxima.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    idx = np.triu_indices(n, k=1)
    for i, j in zip(idx[0], idx[1]):
        row = np.zeros(n)
        row[i] = 1.0
        row[j] = 1.0
        A_ub.append(row)
        b_ub.append(dists[i, j])
        
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        bounds_vals = [x, 1.0 - x, y, 1.0 - y]
        for b in bounds_vals:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(max(0.0, b))
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    # Fallback heuristic if LP fails
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    np.fill_diagonal(dists, np.inf)
    rp = 0.5 * np.min(dists, axis=1)
    return np.minimum(rb, rp) * 0.99

def objective_centers(x):
    """Objective for center optimization: maximize sum of radii."""
    c = x.reshape(N, 2)
    r = compute_lp_radii(c)
    return -np.sum(r)

def hex_grid(n, rows, r_est, shift_x=0.0, shift_y=0.0, rot=0.0):
    """Generates a hexagonal lattice configuration."""
    pts = []
    y = r_est + shift_y
    for r_idx, count in enumerate(rows):
        x_start = r_est + (r_idx % 2) * r_est + shift_x
        for _ in range(count):
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:n])
    
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
        
    # Normalize to fit within [0,1]x[0,1] with margin
    if pts.shape[0] > 1:
        pts -= pts.min(axis=0)
        span = pts.max(axis=0) - pts.min(axis=0)
        span[span == 0] = 1.0
        pts /= span
        pts *= 0.88
        pts += 0.06
    return np.clip(pts, 0.01, 0.99)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_centers = None
    best_sum = -np.inf
    
    inits = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5], [6, 6, 5, 5, 4], 
        [4, 6, 6, 6, 4], [5, 5, 5, 6, 5], [6, 5, 5, 5, 5], [5, 7, 5, 5, 4],
        [6, 5, 5, 6, 4], [5, 6, 5, 5, 5]
    ]
    
    for r_est in [0.095, 0.10, 0.105, 0.11]:
        for pat in patterns:
            if sum(pat) < N:
                pat = pat + [N - sum(pat)]
            c = hex_grid(N, pat, r_est)
            c += np.random.normal(0, 0.004, c.shape)
            inits.append(c.copy())
            
    # Add some rotated grids
    for rot in [0.1, -0.1, 0.2]:
        c = hex_grid(N, [6, 5, 6, 5, 4], 0.102, rot=rot)
        c += np.random.normal(0, 0.003, c.shape)
        inits.append(c.copy())
        
    # Random starts
    for _ in range(5):
        c = np.random.uniform(0.12, 0.88, (N, 2))
        inits.append(c)
        
    center_bounds = [(0.0, 1.0)] * (2 * N)
    
    # Phase 1: Powell optimization on centers using LP for exact radii
    for init_c in inits:
        try:
            res = minimize(objective_centers, init_c.flatten(), method='Powell',
                           bounds=center_bounds, 
                           options={'maxiter': 1500, 'ftol': 1e-13, 'xtol': 1e-13})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            continue
            
    # Phase 2: Perturbation & Powell to escape local minima
    for _ in range(15):
        pert = best_centers + np.random.normal(0, 0.0015, best_centers.shape)
        pert = np.clip(pert, 0.01, 0.99)
        try:
            res = minimize(objective_centers, pert.flatten(), method='Powell',
                           bounds=center_bounds,
                           options={'maxiter': 1000, 'ftol': 1e-13, 'xtol': 1e-13})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            continue
            
    # Phase 3: SLSQP polish on full variables (centers + radii)
    r_init = compute_lp_radii(best_centers)
    x0 = np.concatenate([best_centers.flatten(), r_init])
    
    def obj_full(v):
        return -np.sum(v[2 * N:])
        
    def cons_full(v):
        c = v[:2 * N].reshape(N, 2)
        r = v[2 * N:]
        # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
        con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, 
                              c[:, 1] - r, 1.0 - c[:, 1] - r])
        # Overlap constraints: dist - (r_i + r_j) >= 0
        idx = np.triu_indices(N, k=1)
        dx = c[idx[0], 0] - c[idx[1], 0]
        dy = c[idx[0], 1] - c[idx[1], 1]
        dr = r[idx[0]] + r[idx[1]]
        dists = np.sqrt(dx**2 + dy**2)
        con = np.concatenate([con, dists - dr])
        return con
        
    bounds_full = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': cons_full}
    
    try:
        res = minimize(obj_full, x0, method='SLSQP', bounds=bounds_full,
                       constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14})
        if -res.fun > best_sum:
            best_centers = res.x[:2 * N].reshape(N, 2)
            best_radii = res.x[2 * N:]
            best_sum = -res.fun
        else:
            best_radii = compute_lp_radii(best_centers)
    except Exception:
        best_radii = compute_lp_radii(best_centers)
        
    # Final strict repair to guarantee validator tolerance
    for _ in range(50):
        changed = False
        for i in range(N):
            x, y = best_centers[i]
            mx = min(x, 1.0 - x, y, 1.0 - y)
            if best_radii[i] > mx - 1e-10:
                best_radii[i] = mx
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.sqrt((best_centers[i, 0] - best_centers[j, 0])**2 + 
                            (best_centers[i, 1] - best_centers[j, 1])**2)
                if d < best_radii[i] + best_radii[j] - 1e-10:
                    shrink = (best_radii[i] + best_radii[j] - d) / 2.0
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
