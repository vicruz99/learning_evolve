import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        vals = [centers[i, 0], 1.0 - centers[i, 0], 
                centers[i, 1], 1.0 - centers[i, 1]]
        for v in vals:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(v)
            
    # Solve LP
    res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), 
                  bounds=[(1e-9, None)] * n, method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-5)

def objective_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints_joint(v):
    """Computes all inequality constraints (must be >= 0)."""
    n = len(v) // 3
    X = v.reshape(n, 3)
    xs, ys, rs = X[:, 0], X[:, 1], X[:, 2]
    
    # Boundary constraints
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    idx = np.triu_indices(n, k=1)
    dx = xs[idx[0]] - xs[idx[1]]
    dy = ys[idx[0]] - ys[idx[1]]
    dr = rs[idx[0]] + rs[idx[1]]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def bounds_joint():
    """Variable bounds: x,y in [0,1], r in [1e-9, 0.5]"""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-9, 0.5)])
    return b

def generate_hex_starts(n, num_starts):
    """Generates diverse hexagonal lattice configurations."""
    starts = []
    base_rows = [6, 5, 6, 5, 4]  # Sums to 26
    
    for i in range(num_starts):
        r0 = 0.09 + i * 0.0025
        pts = []
        y = r0
        for r_idx, cnt in enumerate(base_rows):
            x_start = r0 + (r_idx % 2) * r0
            for _ in range(cnt):
                pts.append([x_start, y])
                x_start += 2.0 * r0
            y += r0 * np.sqrt(3.0)
            
        pts = np.array(pts[:n])
        
        # Apply random affine transforms to diversify starts
        shift = np.random.uniform(-0.03, 0.03, 2)
        pts += shift
        scale = 0.92 + np.random.uniform(-0.06, 0.06)
        pts = (pts - 0.5) * scale + 0.5
        
        # Small random jitter to break symmetry
        pts += np.random.normal(0, 0.003, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        starts.append(pts)
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = -1.0
    best_centers = None
    best_radii = None

        hex_starts.append(c)

        centers = init_c.copy()
        
        # Solve LP for optimal radii given these centers
        radii = solve_lp_radii(centers)
        
        # Assemble initial vector for SLSQP

            continue

                valid = False
                break

            best_radii *= 0.9999

    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
