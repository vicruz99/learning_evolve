def compute_constraints(v):
    """Computes boundary and non-overlap constraints. Must return >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = np.empty(4*N + NUM_PAIRS)
    idx = 0
    c[idx:idx+N] = x - r; idx += N

def compute_objective(v):

    return -np.sum(v[2::3])

    c_obj = -np.ones(n)
    A_ub = np.zeros((NUM_PAIRS + 4*n, n))
    b_ub = np.zeros(NUM_PAIRS + 4*n)
