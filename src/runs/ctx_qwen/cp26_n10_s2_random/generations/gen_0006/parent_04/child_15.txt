2. **Center Optimization**: We treat the maximum sum of radii as a function of the centers and optimize it using derivative-free methods (Powell). Since the objective is non-smooth (it's the value function of an LP), gradient-based methods struggle, but direct search methods excel here.

Multi-start Powell optimization navigates the center space, while the LP guarantees the best possible radii for each configuration. import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist

N = 26

# Precompute A_ub structure for the LP
# Constraints: r_i + r_j <= dist_ij (N*(N-1)/2 constraints)
#              r_i <= boundary_i (N constraints)
num_pairs = N * (N - 1) // 2
A_ub_rows = np.zeros((num_pairs + N, N), dtype=float)
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub_rows[idx, i] = 1.0
        A_ub_rows[idx, j] = 1.0
        idx += 1
for i in range(N):
    A_ub_rows[idx, i] = 1.0
    idx += 1

def get_sum_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    # pdist returns condensed distances in the exact order needed for A_ub_rows
    b_pairs = pdist(centers)
    
    # Boundary limits: distance to nearest edge
    xb = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    yb = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    b_bound = np.minimum(xb, yb)
    
    b_ub = np.concatenate([b_pairs, b_bound])
    
    # Maximize sum(r_i) => Minimize -sum(r_i)
    res = linprog(-np.ones(N), A_ub=A_ub_rows, b_ub=b_ub, 
                  bounds=[(0.0, None)]*N, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def obj(centers_flat):
    """Objective for center optimization: minimize negative sum of optimal radii."""
    # Clip to strictly inside to avoid degenerate distances
    centers = np.clip(centers_flat.reshape(N, 2), 1e-6, 1.0 - 1e-6)
    s, _ = get_sum_radii(centers)
    return -s

def run_packing():
    np.random.seed(42)
    best_sum = 0.0
    best_c = None
    best_r = None
    
    configs = []
    
    # 1. Hexagonal lattices with varying densities
    for r0 in [0.090, 0.100, 0.110, 0.120]:
        c = []
        y = r0
        row = 0
        while len(c) < N:
            x = r0 + (row % 2) * r0
            while x + r0 <= 1.0 and len(c) < N:
                c.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
            row += 1
        c = np.array(c[:N])
        c += np.random.normal(0, 0.002, c.shape)
        c = np.clip(c, 0.05, 0.95)
        configs.append(c.flatten())
        
    # 2. Corner & Edge focused configuration
    c_corner = np.array([
        [0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85],
        [0.5, 0.15], [0.5, 0.85], [0.15, 0.5], [0.85, 0.5],
        [0.5, 0.5]
    ])
    rng_corner = np.random.default_rng(10)
    attempts = 0
    while c_corner.shape[0] < N and attempts < 500:
        pt = rng_corner.uniform(0.2, 0.8, 2)
        if np.min(np.linalg.norm(c_corner - pt, axis=1)) > 0.12:
            c_corner = np.vstack([c_corner, pt])
        attempts += 1
    while c_corner.shape[0] < N:
        c_corner = np.vstack([c_corner, rng_corner.uniform(0.2, 0.8, 2)])
    c_corner += rng_corner.normal(0, 0.005, c_corner.shape)
    c_corner = np.clip(c_corner, 0.05, 0.95)
    configs.append(c_corner[:N].flatten())
    
    # 3. Random dense starts
    rng_rand = np.random.default_rng(20)
    for _ in range(8):
        c = rng_rand.rand(N, 2) * 0.7 + 0.15
        configs.append(c.flatten())
        
    bounds_c = [(0.0, 1.0)] * (2 * N)
    
    # Multi-start optimization
    for x0 in configs:
        try:
            res = minimize(obj, x0, method='Powell', bounds=bounds_c,
                           options={'maxiter': 2000, 'ftol': 1e-12, 'xtol': 1e-12})
            s, r = get_sum_radii(res.x.reshape(N, 2))
            if s > best_sum:
                best_sum = s
                best_c = res.x.reshape(N, 2).copy()
                best_r = r.copy()
        except Exception:
            continue
            
    if best_c is None:
        best_c = configs[0].reshape(N, 2)
        _, best_r = get_sum_radii(best_c)
        best_sum = np.sum(best_r)
        
    # Perturbation refinement to escape local minima
    rng_ref = np.random.default_rng(123)
    for _ in range(15):
        c_pert = best_c + rng_ref.normal(0, 0.001, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        try:
            res = minimize(obj, c_pert.flatten(), method='Powell', bounds=bounds_c,
                           options={'maxiter': 1000, 'ftol': 1e-12, 'xtol': 1e-12})
            s, r = get_sum_radii(res.x.reshape(N, 2))
            if s > best_sum:
                best_sum = s
                best_c = res.x.reshape(N, 2).copy()
                best_r = r.copy()
        except Exception:
            pass
            
    # Final validation safety repair
    for _ in range(30):
        valid = True
        for i in range(N):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            if x - r < -1e-10 or x + r > 1.0 + 1e-10 or y - r < -1e-10 or y + r > 1.0 + 1e-10:
                valid = False
                break
        if not valid: break
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_c[i,0] - best_c[j,0], best_c[i,1] - best_c[j,1])
                if d < best_r[i] + best_r[j] - 1e-10:
                    valid = False
                    break
            if not valid: break
        if valid: break
        best_r *= 0.99995
        
    best_r = np.maximum(best_r, 0.0)
    return best_c, best_r, float(np.sum(best_r))
