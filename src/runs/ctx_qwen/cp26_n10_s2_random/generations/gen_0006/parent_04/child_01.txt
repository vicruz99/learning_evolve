from scipy.optimize import minimize, linprog

N = 26
A_ub = None

def init_constraints():
    """Precomputes the sparse constraint matrix structure for the LP radius solver."""
    global A_ub
    n = N
    num_pairs = n * (n - 1) // 2
    A = np.zeros((num_pairs + 4 * n, n))
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
    for i in range(n):
        for k in range(4):
            A[idx, i] = 1.0
            idx += 1
    A_ub = A

def get_optimal_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    b_ub = np.zeros(A_ub.shape[0])
    idx = 0
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    for i in range(n):
        for j in range(i + 1, n):
            b_ub[idx] = dists[i, j]
            idx += 1
    for i in range(n):
        vals = [centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1]]
        for v in vals:
            b_ub[idx] = v
            idx += 1
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return res.x if res.success else np.full(n, 0.05)

def obj_centers(x):
    """Objective for center optimization: maximize sum of radii -> minimize negative sum."""
    centers = x.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    r = get_optimal_radii(centers)
    return -np.sum(r)

def hex_init(n, rows, r_est, rng):
    """Generates a hexagonal lattice initialization with specified row counts."""
    pts = []
    y = r_est
    for r_idx, cnt in enumerate(rows):
        shift = r_est if r_idx % 2 == 1 else 0.0
        x = r_est + shift
        for _ in cnt:
            if len(pts) < n:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:n])
    pts += rng.normal(0, 0.005, pts.shape)
    return np.clip(pts, 0.02, 0.98)

            break
        radii *= 0.9995
        
    radii = np.maximum(radii, 0.0)
    return best_c, radii, float(np.sum(radii))
