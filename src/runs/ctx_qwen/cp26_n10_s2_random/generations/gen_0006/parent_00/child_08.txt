import numpy as np
from scipy.optimize import minimize, linprog

N = 26
IU, JU = np.triu_indices(N, k=1)

def compute_lp_radii(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii.
    Constraints: r_i + r_j <= dist(i,j), r_i <= dist_to_boundary, r_i >= 0
    """
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    c_obj = -np.ones(n)
    num_pairs = N * (N - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dists[i, j]
            k += 1
            
    bounds_r = [(0.0, u) for u in ub]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success and res.fun is not None:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback for degenerate configurations
    safe_r = ub * 0.5
    return safe_r, np.sum(safe_r)

def lp_objective(x):
    """Objective for center-only optimization: minimize negative sum of LP radii."""
    return -compute_lp_radii(x.reshape(N, 2))[1]

def joint_objective(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def joint_constraints(v):
    """Computes all boundary and non-overlap constraints for SLSQP. Must be >= 0."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    d = np.linalg.norm(c[IU] - c[JU], axis=1)
    con = np.concatenate([con, d - (r[IU] + r[JU])])
    return con

def generate_starts():
    """Generates diverse initial center configurations."""
    starts = []
    rng = np.random.default_rng(42)
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [5, 5, 6, 5, 5], [6, 4, 6, 5, 5],
        [5, 4, 6, 6, 5], [4, 5, 6, 5, 6], [5, 5, 5, 6, 5],
        [5, 5, 4, 6, 6], [6, 6, 4, 5, 5], [4, 5, 5, 6, 6]
    ]
    for pat in patterns:
        for r_est in [0.095, 0.100, 0.105]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.02, 0.98)
            starts.append(c.flatten())
            
    for _ in range(15):
        c = rng.uniform(0.1, 0.9, (N, 2))
        starts.append(c.flatten())
        
    for _ in range(10):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(300):
            for i in range(N):
                for j in range(i + 1, N):
                    d_vec = c[i] - c[j]
                    d = np.linalg.norm(d_vec)
                    if d < 0.15 and d > 1e-6:
                        f = (0.15 - d) * 0.1 / d
                        c[i] += d_vec * f
                        c[j] -= d_vec * f
        c = np.clip(c, 0.05, 0.95)
        starts.append(c.flatten())
        
    return starts

def run_packing() -> tuple:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    bounds_c = [(0.0, 1.0)] * (2 * N)
    bounds_joint = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    starts = generate_starts()
    
    # Phase 1: Multi-start optimization
    for x0 in starts:
        # LP-based center optimization
        try:
            res = minimize(lp_objective, x0, method='Powell', bounds=bounds_c,
                           options={'maxiter': 1500, 'ftol': 1e-13})
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape(N, 2)
                best_radii, _ = compute_lp_radii(best_centers)
        except Exception:
            pass
            
        # Joint SLSQP refinement
        c_curr = best_centers if best_centers is not None else x0.reshape(N, 2)
        r_curr, _ = compute_lp_radii(c_curr)
        v0 = np.concatenate([c_curr.flatten(), r_curr])
        
        try:
            res2 = minimize(joint_objective, v0, method='SLSQP', bounds=bounds_joint,
                            constraints={'type': 'ineq', 'fun': joint_constraints},
                            options={'maxiter': 5000, 'ftol': 1e-14})
            s2 = -res2.fun
            if s2 > best_sum:
                best_sum = s2
                best_centers = res2.x[:2*N].reshape(N, 2)
                best_radii = res2.x[2*N:]
        except Exception:
            pass
            
    # Fallback
    if best_centers is None:
        best_centers = starts[0].reshape(N, 2)
        best_radii, best_sum = compute_lp_radii(best_centers)
        
    # Phase 2: Simulated Annealing Perturbation Search
    curr_c = best_centers.copy()
    curr_r, curr_s = compute_lp_radii(curr_c)
    best_c, best_r, best_s = curr_c.copy(), curr_r.copy(), curr_s
    
    rng = np.random.default_rng(99)
    for step in range(80):
        temp = 0.004 * (0.92 ** step)
        c_pert = curr_c + rng.normal(0, temp, curr_c.shape)
        c_pert = np.clip(c_pert, 0.01, 0.99)
        
        r_p, s_p = compute_lp_radii(c_pert)
        
        if s_p > curr_s or rng.random() < np.exp((s_p - curr_s) / max(temp, 1e-9)):
            curr_c, curr_r, curr_s = c_pert, r_p, s_p
            if curr_s > best_s:
                best_c, best_r, best_s = curr_c.copy(), curr_r.copy(), curr_s
                
        # Periodic local polish
        if step % 10 == 0:
            try:
                res_loc = minimize(lp_objective, curr_c.flatten(), method='Powell', bounds=bounds_c,
                                   options={'maxiter': 600, 'ftol': 1e-13})
                if -res_loc.fun > best_s:
                    best_s = -res_loc.fun
                    best_c = res_loc.x.reshape(N, 2)
                    best_r, _ = compute_lp_radii(best_c)
                    curr_c, curr_r, curr_s = best_c, best_r, best_s
            except Exception:
                pass
                
    # Update best if perturbation found better
    if best_s > best_sum:
        best_sum = best_s
        best_centers = best_c
        best_radii = best_r
        
    # Phase 3: Strict numerical repair to guarantee validation passes
    centers = best_centers
    radii = best_radii
    
    for _ in range(100):
        changed = False
        # Fix overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx + 1e-12:
                radii[i] = mx
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
