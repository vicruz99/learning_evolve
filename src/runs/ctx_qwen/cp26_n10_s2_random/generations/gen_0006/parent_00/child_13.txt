import numpy as np
from scipy.optimize import minimize, linprog

N = 26
N_PAIRS = N * (N - 1) // 2
IDX_I, IDX_J = np.triu_indices(N, k=1)

# Precompute pairwise constraint matrix structure
A_ub_pair = np.zeros((N_PAIRS, N))
A_ub_pair[np.arange(N_PAIRS), IDX_I] = 1.0
A_ub_pair[np.arange(N_PAIRS), IDX_J] = 1.0

def compute_lp_sum(centers_flat):
    """Solves LP for radii given fixed centers and returns the maximum sum of radii."""
    c = centers_flat.reshape(N, 2)
    
    # Boundary constraints: r_i <= dist to boundary
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    d_pairs = dists[IDX_I, IDX_J]
    
    # Construct full A_ub and b_ub
    A = np.vstack([A_ub_pair, np.eye(N)])
    b = np.concatenate([d_pairs, ub])
    c_obj = -np.ones(N)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(N)

def objective_centers(p):
    """Objective for center-only optimization: minimize negative LP sum of radii."""
    val, _ = compute_lp_sum(p)
    return -val

def compute_joint_constraints(v):
    """Computes boundary and non-overlap constraints for SLSQP. Must be >= 0."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    cons = []
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    # Overlap constraints
    d = np.sqrt(np.sum((c[IDX_I] - c[IDX_J])**2, axis=1))
    cons.append(d - (r[IDX_I] + r[IDX_J]))
    return np.concatenate(cons)

def objective_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def generate_initial_configs():

    np.random.seed(42)

    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [6,4,6,5,5],

        [4,5,6,5,6], [5,5,5,6,5], [5,5,4,6,6], [6,5,5,5,5]
    ]
    for pat in patterns:
        for r0 in [0.09, 0.095, 0.10, 0.105, 0.11]:
            pts = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                        x += 2.0 * r0
                y += np.sqrt(3) * r0
            pts = np.array(pts[:N])
            pts += np.random.normal(0, 0.003, pts.shape)
            pts = np.clip(pts, 0.02, 0.98)
            starts.append(pts.flatten())

    # 2. Force-directed random starts
    for _ in range(30):
        c = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(100):
            for i in range(N):

                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)

                        push = (0.16 - dist) * 0.6 / dist
                        c[i] += d_vec * push
                        c[j] -= d_vec * push
        c = np.clip(c, 0.02, 0.98)
        starts.append(c.flatten())

    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        pass

                radii[i] = max_r
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
