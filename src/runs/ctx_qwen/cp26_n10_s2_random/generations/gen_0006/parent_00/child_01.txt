import numpy as np
from scipy.optimize import linprog

N = 26
NUM_PAIRS = N * (N - 1) // 2
PAIR_INDICES = np.array([(i, j) for i in range(N) for j in range(i + 1, N)])

def solve_lp_and_get_grad(centers):
    """
    Solves the LP for optimal radii given fixed centers and computes the gradient
    of the sum of radii with respect to center positions using dual variables.
    """
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    A_ub = np.zeros((NUM_PAIRS + 4 * n, n))
    b_ub = np.zeros(NUM_PAIRS + 4 * n)
    
    k = 0
    for i, j in PAIR_INDICES:
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]
        k += 1
        
    for i in range(n):
        off = NUM_PAIRS + 4 * i
        A_ub[off, i] = 1.0; b_ub[off] = centers[i, 0]
        A_ub[off+1, i] = 1.0; b_ub[off+1] = 1.0 - centers[i, 0]
        A_ub[off+2, i] = 1.0; b_ub[off+2] = centers[i, 1]
        A_ub[off+3, i] = 1.0; b_ub[off+3] = 1.0 - centers[i, 1]
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            return None, None, None, None
            
        radii = res.x
        total_sum = -res.fun
        marg = res.ineqlin.marginals
        
        grad = np.zeros_like(centers)
        
        # Pairwise repulsive forces derived from dual variables
        for k in range(NUM_PAIRS):
            lam = abs(marg[k])
            if lam > 1e-10:
                i, j = PAIR_INDICES[k]
                d = dists[i, j]
                if d > 1e-9:
                    vec = centers[i] - centers[j]
                    f = (lam / d) * vec
                    grad[i] += f
                    grad[j] -= f
                    
        # Boundary forces: push centers towards walls where slack allows growth
        for i in range(n):
            off = NUM_PAIRS + 4 * i
            grad[i, 0] += abs(marg[off]) - abs(marg[off+1])
            grad[i, 1] += abs(marg[off+2]) - abs(marg[off+3])
            
        return radii, total_sum, grad, dists
    except Exception:
        return None, None, None, None

def repair(centers, radii):
    """Deterministically shrinks radii to strictly satisfy non-overlap and boundary constraints."""
    radii = radii.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < radii[i] + radii[j] - 1e-12:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx + 1e-12:
                radii[i] = mx
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Diverse hexagonal lattice row patterns known to be optimal/near-optimal for N=26
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [7, 6, 6, 7],
        [5, 5, 5, 6, 5], [6, 5, 5, 5, 5], [4, 5, 6, 5, 6]
    ]
    
    for pat in patterns:
        centers = []
        r0 = 0.1
        y = r0
        row = 0
        for cnt in pat:
            xs = r0 if row % 2 == 0 else 2 * r0
            for _ in range(cnt):
                centers.append([xs, y])
                xs += 2 * r0
            y += np.sqrt(3) * r0
            row += 1
        centers = np.array(centers[:N])
        c_min, c_max = centers.min(axis=0), centers.max(axis=0)
        centers = (centers - c_min) / (c_max - c_min) * 0.85 + 0.075
        centers = np.clip(centers, 0.01, 0.99)
        
        curr = centers.copy()
        step = 0.005
        for it in range(400):
            r, s, g, _ = solve_lp_and_get_grad(curr)
            if r is None: break
            if s > best_sum:
                best_sum = s
                best_centers = curr.copy()
                best_radii = r.copy()
                
            gn = np.linalg.norm(g)
            if gn > 1e-9:
                curr += step * g / gn
            curr = np.clip(curr, 0.001, 0.999)
            step *= 0.996
            
        # Local perturbations to escape local minima
        for _ in range(15):
            pert = best_centers + np.random.normal(0, 0.004, best_centers.shape)
            pert = np.clip(pert, 0.01, 0.99)
            c2 = pert.copy()
            s2 = 0.004
            for _ in range(200):
                r2, s2v, g2, _ = solve_lp_and_get_grad(c2)
                if r2 is None: break
                if s2v > best_sum:
                    best_sum = s2v
                    best_centers = c2.copy()
                    best_radii = r2.copy()
                gn2 = np.linalg.norm(g2)
                if gn2 > 1e-9:
                    c2 += s2 * g2 / gn2
                c2 = np.clip(c2, 0.001, 0.999)
                s2 *= 0.985
                
    # Final exact LP solve for the best geometry found
    if best_centers is not None:
        r_final, _, _, _ = solve_lp_and_get_grad(best_centers)
        if r_final is not None:
            best_radii = r_final
            best_sum = np.sum(best_radii)
            
    # Ensure strict compliance with validator tolerances
    best_radii = repair(best_centers, best_radii)
    return best_centers, best_radii, float(best_sum)
