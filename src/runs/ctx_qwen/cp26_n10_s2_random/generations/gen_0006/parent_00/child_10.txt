import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute pairwise indices for constraints
idx_pairs = np.triu_indices(N, k=1)
num_pairs = len(idx_pairs[0])
m = 4 * N + num_pairs  # Total constraints: 4 boundary per circle + pairwise

# Precompute the static part of the inequality constraint matrix A_ub
A_ub_base = np.zeros((m, N))

# Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
for i in range(N):
    A_ub_base[i, i] = 1.0
    A_ub_base[N + i, i] = 1.0
    A_ub_base[2 * N + i, i] = 1.0
    A_ub_base[3 * N + i, i] = 1.0

# Pairwise constraints: r_i + r_j <= dist(i,j)
for k, (i, j) in enumerate(zip(idx_pairs[0], idx_pairs[1])):
    A_ub_base[4 * N + k, i] = 1.0
    A_ub_base[4 * N + k, j] = 1.0

c_obj = -np.ones(N)  # Maximize sum(r) -> Minimize -sum(r)
bounds_r = [(0, None)] * N

def solve_lp(centers):
    """Solves LP to find optimal radii and dual variables for fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    b_ub = np.empty(m)
    b_ub[:N] = x
    b_ub[N:2*N] = 1.0 - x
    b_ub[2*N:3*N] = y
    b_ub[3*N:4*N] = 1.0 - y
    
    diff = centers[idx_pairs[0]] - centers[idx_pairs[1]]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    b_ub[4*N:] = dists
    
    try:
        res = linprog(c_obj, A_ub=A_ub_base, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            # Marginals are dual variables. Take abs to handle sign conventions safely.
            duals = np.abs(res.ineqlin.marginals)
            return res.x, -res.fun, duals
    except Exception:
        pass
    return np.zeros(N), 0.0, np.zeros(m)

def get_gradient(centers, duals):
    """Computes gradient of max sum of radii w.r.t center coordinates using duals."""
    grad = np.zeros((N, 2))
    
    # Boundary gradients
    grad[:, 0] = duals[:N] - duals[N:2*N]
    grad[:, 1] = duals[2*N:3*N] - duals[3*N:4*N]
    
    # Pairwise gradients (repulsive forces along contact directions)
    i_idx, j_idx = idx_pairs
    diff = centers[i_idx] - centers[j_idx]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    dists = np.maximum(dists, 1e-9)
    lambdas = duals[4*N:] / dists
    
    for k in range(num_pairs):
        f = lambdas[k] * diff[k]
        grad[i_idx[k]] += f
        grad[j_idx[k]] -= f
        
    return grad

def obj_sum_radii(c_flat):
    """Objective for derivative-free optimizers: negative sum of optimal radii."""
    c = c_flat.reshape(N, 2)
    _, val, _ = solve_lp(c)
    return -val

def run_gradient_ascent(c_init, max_iter=200):
    """Performs gradient ascent using LP duals with adaptive step size."""
    c = c_init.copy()
    best_c = c.copy()
    _, best_val, _ = solve_lp(c)
    
    for step in range(max_iter):
        r, val, duals = solve_lp(c)
        g = get_gradient(c, duals)
        
        # Adaptive step size decay
        step_size = 0.06 / (1.0 + 0.008 * step)
        
        # Forward step
        c_new = np.clip(c + step_size * g, 0.0, 1.0)
        _, val_new, _ = solve_lp(c_new)
        
        if val_new > val:
            c = c_new
            if val_new > best_val:
                best_val = val_new
                best_c = c.copy()
        else:
            # Backtrack with reduced step
            step_size *= 0.5
            c_new = np.clip(c + step_size * g, 0.0, 1.0)
            _, val_new, _ = solve_lp(c_new)
            
            if val_new > val:
                c = c_new
                if val_new > best_val:
                    best_val = val_new
                    best_c = c.copy()
            else:
                break
                
    return best_c, best_val

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5],
        [5, 5, 5, 5, 6], [6, 6, 4, 5, 5], [5, 6, 4, 5, 6],
        [5, 5, 5, 6, 4], [4, 5, 5, 6, 6], [6, 5, 4, 5, 6],
        [5, 6, 5, 5, 5], [6, 5, 5, 6, 4], [4, 6, 5, 5, 6]
    ]
    
    for pat in patterns:

                    c.append([x, y])
                    x += 2.0 * r_est

    for _ in range(30):

        starts.append(c)

            best_c = c_opt

        try:
            res = minimize(obj_sum_radii, c_opt.flatten(), method='Powell',

                best_c = res.x.reshape(N, 2)

            pass

    r_final, _, _ = solve_lp(best_c)

        # Resolve overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(best_c[i] - best_c[j])
                if d < r_final[i] + r_final[j] - 1e-12:
                    shrink = (r_final[i] + r_final[j] - d) / 2.0 + 1e-9
                    r_final[i] -= shrink
                    r_final[j] -= shrink

        # Enforce boundaries
        for i in range(N):
            mr = min(best_c[i, 0], 1.0 - best_c[i, 0],
                     best_c[i, 1], 1.0 - best_c[i, 1])
            if r_final[i] > mr + 1e-12:
                r_final[i] = mr
                changed = True
        if not changed:
            break
            
    r_final = np.maximum(r_final, 0.0)
    return best_c, r_final, float(np.sum(r_final))
