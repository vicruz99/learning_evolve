2. **Derivative-Free Center Optimization**: I will use the LP solution as the objective function for center optimization. By maximizing the LP sum over center coordinates using `Nelder-Mead`, we navigate the feasible space robustly without gradient issues.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_COUNT = N * (N - 1) // 2
CONST_COUNT = PAIR_COUNT + 4 * N

# Precompute constant parts of the LP matrix
A_ub_pairs = np.zeros((PAIR_COUNT, N))
pair_idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub_pairs[pair_idx, i] = 1.0
        A_ub_pairs[pair_idx, j] = 1.0
        pair_idx += 1

A_ub_bound = np.zeros((4 * N, N))
for i in range(N):
    A_ub_bound[4*i, i] = 1.0
    A_ub_bound[4*i+1, i] = 1.0
    A_ub_bound[4*i+2, i] = 1.0
    A_ub_bound[4*i+3, i] = 1.0

A_ub = np.vstack([A_ub_pairs, A_ub_bound])
c_obj = -np.ones(N)

def solve_lp_for_sum(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    c = np.clip(centers, 1e-5, 1.0 - 1e-5)
    
    # Pairwise distances
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Boundary limits
    xb = np.minimum(c[:, 0], 1.0 - c[:, 0])
    yb = np.minimum(c[:, 1], 1.0 - c[:, 1])
    ub = np.minimum(xb, yb)
    
    b_ub = np.empty(CONST_COUNT)
    b_ub[:PAIR_COUNT] = dists[np.triu_indices(N, k=1)]
    b_ub[PAIR_COUNT:PAIR_COUNT+N] = c[:, 0]
    b_ub[PAIR_COUNT+N:PAIR_COUNT+2*N] = 1.0 - c[:, 0]
    b_ub[PAIR_COUNT+2*N:PAIR_COUNT+3*N] = c[:, 1]
    b_ub[PAIR_COUNT+3*N:] = 1.0 - c[:, 1]
    
    bounds_r = [(0.0, u) for u in ub]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return -res.fun, res.x, c
    except Exception:
        pass
    return 0.0, np.zeros(N), c

def obj_centers_lp(c_flat):
    """Objective for center optimization: negative sum of radii from LP."""
    c = c_flat.reshape(N, 2)
    s, _, _ = solve_lp_for_sum(c)
    return -s

def compute_constraints(v):
    """Computes all boundary and non-overlap constraints for SLSQP."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    cons = []
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    cons.append(d - (r[idx[0]] + r[idx[1]]))
    return np.concatenate(cons)

def objective_joint(v):
    """Objective for joint optimization."""
    return -np.sum(v[2*N:])

def generate_starts(num_starts):
    """Generates diverse initial center configurations."""
    starts = []
    rng = np.random.default_rng(42)
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5]
    ]
    
    for pat in patterns:
        for r_est in [0.09, 0.095, 0.10, 0.105]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.004, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    for _ in range(15):
        c = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(50):
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if dist < 0.16 and dist > 1e-6:
                        push = (0.16 - dist) * 0.5 / dist
                        c[i] += d_vec * push
                        c[j] -= d_vec * push
        c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    return starts[:num_starts]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    starts = generate_starts(30)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Derivative-free optimization of centers using LP as oracle
    for c0 in starts:
        try:
            res = minimize(obj_centers_lp, c0.flatten(), method='Nelder-Mead',
                           options={'maxiter': 600, 'xatol': 1e-5, 'fatol': 1e-9})
            c_opt = res.x.reshape(N, 2)
            s, r_opt, c_clipped = solve_lp_for_sum(c_opt)
            
            if s > best_sum:
                best_sum = s
                best_centers = c_clipped.copy()
                best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation search to escape local minima
    if best_centers is not None:
        for step in range(40):
            noise = 0.004 * (0.85 ** step)
            c_pert = best_centers + np.random.normal(0, noise, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            
            try:
                res = minimize(obj_centers_lp, c_pert.flatten(), method='Nelder-Mead',
                               options={'maxiter': 300, 'xatol': 1e-5, 'fatol': 1e-8})
                c_pert_opt = res.x.reshape(N, 2)
                s, r_pert, _ = solve_lp_for_sum(c_pert_opt)
                
                if s > best_sum:
                    best_sum = s
                    best_centers = c_pert_opt.copy()
                    best_radii = r_pert.copy()
            except Exception:
                pass

    # Phase 3: SLSQP joint optimization to polish geometry and radii together
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    v0 = np.concatenate([best_centers.flatten(), best_radii])
    try:
        res = minimize(objective_joint, v0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
        c_curr = res.x[:2*N].reshape(N, 2)
        r_curr = res.x[2*N:]
        
        # Always check LP against SLSQP result for fixed centers
        s_lp, r_lp, _ = solve_lp_for_sum(c_curr)
        if s_lp > np.sum(r_curr):
            best_centers = c_curr
            best_radii = r_lp
            best_sum = s_lp
        else:
            best_centers = c_curr
            best_radii = r_curr
            best_sum = np.sum(r_curr)
    except Exception:
        pass
        
    # Phase 4: Strict numerical repair
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(best_centers[i] - best_centers[j])
                req = best_radii[i] + best_radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                best_radii[i] = max_r
                changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
