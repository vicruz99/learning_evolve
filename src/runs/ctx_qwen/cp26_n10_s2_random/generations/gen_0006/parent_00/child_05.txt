import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    num_pairs = N * (N - 1) // 2
    total_constraints = num_pairs + 4 * N
    
    # Pre-allocate A_ub for LP: A_ub x <= b_ub
    # Rows 0..num_pairs-1: r_i + r_j <= dist_ij
    # Rows num_pairs..num_pairs+N-1: r_i <= x_i
    # Rows num_pairs+N..num_pairs+2N-1: r_i <= 1-x_i
    # Rows num_pairs+2N..num_pairs+3N-1: r_i <= y_i
    # Rows num_pairs+3N..num_pairs+4N-1: r_i <= 1-y_i
    A_ub = np.zeros((total_constraints, N))
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
    for i in range(N):
        A_ub[idx, i] = 1.0; idx += 1
        A_ub[idx, i] = 1.0; idx += 1
        A_ub[idx, i] = 1.0; idx += 1
        A_ub[idx, i] = 1.0; idx += 1
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # 1. Generate Initial Configurations
    configs = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [5, 5, 6, 5, 5], [6, 4, 6, 5, 5],
        [5, 4, 6, 6, 5], [4, 5, 6, 5, 6], [5, 5, 5, 6, 5],
        [5, 5, 4, 6, 6], [6, 6, 4, 5, 5], [4, 5, 5, 6, 6],
        [5, 6, 4, 5, 6], [6, 5, 5, 5, 5]
    ]
    
    for pat in patterns:
        for r_est in [0.088, 0.092, 0.096, 0.100]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            c = np.array(c[:N])
            c += np.random.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.05, 0.95)
            configs.append(c)
            
    # Random starts with pairwise repulsion
    for _ in range(15):
        c = np.random.uniform(0.1, 0.9, (N, 2))
        for _ in range(600):
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if dist < 0.16 and dist > 1e-6:
                        push = (0.16 - dist) * 0.08 / dist
                        c[i] += d_vec * push
                        c[j] -= d_vec * push
        c = np.clip(c, 0.05, 0.95)
        configs.append(c)
        
    # 2. Gradient Ascent on Centers using LP Duals
    for init_c in configs:
        centers = init_c.copy()
        step = 0.008
        
        for iteration in range(1500):
            diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diffs**2, axis=2))
            np.fill_diagonal(dists, 1e9)
            
            b_ub = np.zeros(total_constraints)
            idx = 0
            for i in range(N):
                for j in range(i+1, N):
                    b_ub[idx] = dists[i, j]
                    idx += 1
            for i in range(N):
                b_ub[idx] = centers[i, 0]; idx += 1
                b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
                b_ub[idx] = centers[i, 1]; idx += 1
                b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
                
            try:
                res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
            except Exception:
                break
                
            if not res.success:
                break
                
            radii = res.x
            sum_r = -res.fun
            duals = res.ineqlin.marginals
            
            if sum_r > best_sum:
                best_sum = sum_r
                best_centers = centers.copy()
                best_radii = radii.copy()
                
            # Compute gradient of sum_r w.r.t centers using duals
            grad = np.zeros((N, 2))
            idx = 0
            for i in range(N):
                for j in range(i+1, N):
                    lam = duals[idx]
                    if lam > 1e-7:
                        d = dists[i, j]
                        if d > 1e-9:
                            vec = centers[i] - centers[j]
                            force = (lam / d) * vec
                            grad[i] += force
                            grad[j] -= force
                    idx += 1
                    
            for i in range(N):
                mu_x = duals[num_pairs + 4*i]
                mu_1x = duals[num_pairs + 4*i + 1]
                mu_y = duals[num_pairs + 4*i + 2]
                mu_1y = duals[num_pairs + 4*i + 3]
                grad[i, 0] += mu_x - mu_1x
                grad[i, 1] += mu_y - mu_1y
                
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-8:
                break
                
            grad = grad / grad_norm
            centers += step * grad
            centers = np.clip(centers, 0.001, 0.999)
            
            if iteration % 60 == 0:
                step *= 0.94
            if iteration % 120 == 0:
                centers += np.random.normal(0, 0.0008, centers.shape)
                centers = np.clip(centers, 0.05, 0.95)
                
    # 3. Final SLSQP Polish
    def objective(v):
        return -np.sum(v[2*N:])

    def constraints(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        cons = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
        idx = np.triu_indices(N, 1)
        d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
        cons.append(d - (r[idx[0]] + r[idx[1]]))
        return np.concatenate(cons)

    v0 = np.concatenate([best_centers.flatten(), best_radii])
    try:
        res = minimize(objective, v0, method='SLSQP', 
                       bounds=[(0,1)]*(2*N) + [(0,0.5)]*N,
                       constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': 5000, 'ftol': 1e-14})
        if hasattr(res, 'x') and res.success:
            c_opt = res.x[:2*N].reshape(N, 2)
            r_opt = res.x[2*N:]
            s_opt = np.sum(r_opt)
            if s_opt > best_sum:
                best_centers = c_opt
                best_radii = r_opt
                best_sum = s_opt
    except Exception:
        pass
        
    # 4. Final LP Refinement on Best Centers
    diffs = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    b_ub = np.zeros(total_constraints)
    idx = 0
    for i in range(N):
        for j in range(i+1, N):
            b_ub[idx] = dists[i, j]
            idx += 1
    for i in range(N):
        b_ub[idx] = best_centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - best_centers[i, 0]; idx += 1
        b_ub[idx] = best_centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - best_centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        best_radii = res.x
        best_sum = -res.fun
        
    # 5. Strict Numerical Repair
    radii = best_radii.copy()
    centers = best_centers.copy()
    for _ in range(60):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < radii[i] + radii[j] - 1e-10:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-10:
                radii[i] = mr
                changed = True
        if not changed: break
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
