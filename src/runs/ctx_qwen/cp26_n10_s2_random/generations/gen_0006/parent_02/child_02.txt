

<strategy>
The previous solutions relied heavily on joint SLSQP optimization of centers and radii. While effective, SLSQP can get trapped in local minima and struggles with the high dimensionality and non-smoothness of the radius constraints. 

A more robust strategy for maximizing the sum of radii is to decouple the problem:
1. **Fixed Centers -> LP for Radii**: For any fixed set of centers, finding the optimal radii is a Linear Programming problem. This can be solved exactly and efficiently using `scipy.optimize.linprog` with the `highs` solver. The constraints are simply $r_i + r_j \le d_{ij}$ and $r_i \le \text{dist}(c_i, \text{boundary})$.
2. **Optimize Centers**: We treat the negative sum of optimal radii as the objective function for the centers alone. Since this objective is non-smooth (LP basis changes combinatorially), derivative-free optimizers like `Powell` or `Nelder-Mead` work well.
3. **Multi-start & Basin Hopping**: We initialize with diverse hexagonal lattice patterns and random configurations, perturb them, and run the center optimizer. We also perform iterative perturbation on the best solution to escape shallow local minima.
4. **Joint Polish**: After finding a strong center configuration via the LP-based search, we run a final joint SLSQP optimization on both centers and radii to achieve high-precision constraint satisfaction and squeeze out remaining gains.
5. **Repair & Safety**: A deterministic repair loop strictly enforces the $10^{-12}$ tolerance required by the validator.

This hybrid approach leverages the exactness of LP for radii and global search capabilities for centers, typically outperforming pure joint constrained optimization.
</strategy>

```python
import numpy as np
import scipy.optimize as opt
import math

N = 26

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii exactly."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub_rows = []
    b_ub_vals = []
    
    # Pairwise non-overlap: r_i + r_j <= dist_ij
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(dists[i, j])
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        bounds_vals = [centers[i, 0], 1.0 - centers[i, 0], 
                       centers[i, 1], 1.0 - centers[i, 1]]
        for b in bounds_vals:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(b)
            
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    # Ensure b_ub is strictly positive to prevent linprog warnings
    b_ub = np.maximum(b_ub, 1e-9)
    
    res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return np.full(n, 0.01)

def lp_objective(x_flat):
    """Objective for center optimization: maximize sum of optimal radii."""
    centers = x_flat.reshape(N, 2)
    # Keep centers safely inside the square to ensure valid LP bounds
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    r = solve_lp_radii(centers)
    return -np.sum(r)

def generate_hex_patterns():
    """Generates diverse hexagonal lattice configurations."""
    patterns = [
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5],
        [5, 5, 6, 5, 5], [5, 5, 5, 6, 5], [5, 5, 5, 5, 5, 1],
        [6, 5, 5, 5, 4], [4, 6, 5, 5, 6], [5, 5, 6, 6, 4],
        [5, 4, 6, 5, 6], [6, 6, 5, 5, 4], [5, 7, 5, 5, 4]
    ]
    inits = []
    for pat in patterns:
        if sum(pat) != N: 
            continue
        centers = []
        r_est = 0.092
        dy = np.sqrt(3) * r_est
        for r_idx, count in enumerate(pat):
            y = r_est + r_idx * dy
            offset = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + offset
            for _ in range(count):
                centers.append([x, y])
                x += 2 * r_est
        inits.append(np.array(centers[:N]))
        
    # Add dense random configurations for robustness
    for _ in range(5):
        c = np.random.rand(N, 2) * 0.8 + 0.1
        inits.append(c)
    return inits

def repair(centers, radii):
    """Iteratively shrinks radii to strictly resolve overlaps and clamp to boundaries."""
    radii = radii.copy()
    for _ in range(50):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx - 1e-9:
                radii[i] = mx
                changed = True
        # Resolve overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds_centers = [(0.01, 0.99)] * (2 * N)
    
    best_centers = None
    best_sum = -np.inf
    
    inits = generate_hex_patterns()
    
    # Phase 1: LP-based center optimization with multiple starts
    for i, c0 in enumerate(inits):
        c0_flat = c0.flatten()
        # Perturb slightly for diversity across trials
        for trial in range(3):
            x0 = c0_flat.copy()
            if trial > 0:
                x0 += np.random.normal(0, 0.012, x0.shape)
                x0 = np.clip(x0, 0.02, 0.98)
                
            try:
                res = opt.minimize(lp_objective, x0, method='Powell',
                                   bounds=bounds_centers,
                                   options={'maxiter': 2500, 'ftol': 1e-12})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_centers = res.x.reshape(N, 2)
            except Exception:
                pass
                
    # Phase 2: Iterative perturbation to escape local minima
    if best_centers is not None:
        for _ in range(15):
            pert = best_centers + np.random.normal(0, 0.006, best_centers.shape)
            pert = np.clip(pert, 0.02, 0.98)
            try:
                res = opt.minimize(lp_objective, pert.flatten(), method='Powell',
                                   bounds=bounds_centers,
                                   options={'maxiter': 1500, 'ftol': 1e-12})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_centers = res.x.reshape(N, 2)
            except Exception:
                pass
                
    # Phase 3: Joint SLSQP polish for high precision and exact constraint satisfaction
    radii_init = solve_lp_radii(best_centers) * 0.998
    x0_joint = np.concatenate([best_centers.flatten(), radii_init])
    
    def obj_joint(v):
        return -np.sum(v[2*N:])
        
    def cons_joint(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        con = []
        # Boundary constraints
        con.append(c[:, 0] - r)
        con.append(1.0 - c[:, 0] - r)
        con.append(c[:, 1] - r)
        con.append(1.0 - c[:, 1] - r)
        # Pairwise distance constraints
        idx = np.triu_indices(N, 1)
        d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
        con.append(d - (r[idx[0]] + r[idx[1]]))
        return np.concatenate(con)

    bounds_joint = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
    try:
        res_j = opt.minimize(obj_joint, x0_joint, method='SLSQP', bounds=bounds_joint,
                             constraints={'type': 'ineq', 'fun': cons_joint},
                             options={'maxiter': 10000, 'ftol': 1e-14})
        c_vals = cons_joint(res_j.x)
        if np.min(c_vals) >= -1e-8:
            best_centers = res_j.x[:2*N].reshape(N, 2)
    except Exception:
        pass
        
    # Compute final exact radii and repair
    radii_final = solve_lp_radii(best_centers)
    radii_final = repair(best_centers, radii_final)
    
    # Final safety shrink to guarantee validator tolerance (1e-12)
    for _ in range(10):
        valid = True
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], radii_final[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False
                break
        if not valid:
            break
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_centers[i, 0] - best_centers[j, 0], 
                               best_centers[i, 1] - best_centers[j, 1])
                if d < radii_final[i] + radii_final[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            break
        radii_final *= 0.99995
        
    radii_final = np.maximum(radii_final, 0.0)
    return best_centers, radii_final, float(np.sum(radii_final))
```
