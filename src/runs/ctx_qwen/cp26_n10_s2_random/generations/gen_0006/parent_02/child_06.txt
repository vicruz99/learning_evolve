import numpy as np
import scipy.optimize as opt
import math

N = 26

def build_A_ub(n):
    """Pre-computes the constraint matrix for the LP formulation."""
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    A = np.zeros((num_pairs + num_bound, n))
    idx = 0
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        for k in range(4):
            A[idx, i] = 1.0
            idx += 1
    return A

A_ub = build_A_ub(N)

def compute_b_ub(centers):
    """Computes the RHS vector for the LP constraints given fixed centers."""
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    b = np.zeros(num_pairs + 4 * n)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    idx = 0
    for i in range(n):
        b[idx:idx + (n - 1 - i)] = dists[i, i+1:]
        idx += (n - 1 - i)
        
    # Boundaries: x, 1-x, y, 1-y
    b[idx:idx+n] = centers[:, 0]
    idx += n
    b[idx:idx+n] = 1.0 - centers[:, 0]
    idx += n
    b[idx:idx+n] = centers[:, 1]
    idx += n
    b[idx:idx+n] = 1.0 - centers[:, 1]
    return b

def solve_lp(centers):
    """Solves the LP to find optimal radii and dual variables for fixed centers."""
    b = compute_b_ub(centers)
    c_obj = -np.ones(N)
    bounds = [(0.0, None)] * N
    res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
    return res

def get_gradient(centers, res):
    """
    Computes the gradient of the sum of radii w.r.t centers using LP duals.
    Duals indicate how much the objective improves if a constraint is relaxed.
    """
    if not res.success:
        return np.zeros_like(centers)
        
    duals = res.ineqlin.marginals
    force = np.zeros((N, 2))
    
    n = N
    idx = 0
    # Pairwise forces: push centers apart proportionally to dual magnitude
    for i in range(n):
        for j in range(i + 1, n):
            lam = -duals[idx]  # Marginals are <= 0 for minimization, so -duals >= 0
            if lam > 1e-10:
                d_vec = centers[i] - centers[j]
                d = np.linalg.norm(d_vec)
                if d > 1e-9:
                    f = (lam / d) * d_vec
                    force[i] += f
                    force[j] -= f
            idx += 1
            
    # Boundary forces: push centers away from walls
    for i in range(n):
        lam_x = -duals[idx]; force[i, 0] += lam_x; idx += 1
        lam_x1 = -duals[idx]; force[i, 0] -= lam_x1; idx += 1
        lam_y = -duals[idx]; force[i, 1] += lam_y; idx += 1
        lam_y1 = -duals[idx]; force[i, 1] -= lam_y1; idx += 1
        
    return force

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    global_best_centers = None
    global_best_sum = -np.inf
    
    # 1. Generate diverse initial configurations
    configs = []
    patterns = [
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5],
        [5, 5, 6, 5, 5], [5, 5, 5, 6, 5], [6, 5, 5, 5, 4],
        [4, 6, 5, 5, 6], [5, 5, 6, 6, 4], [5, 5, 5, 5, 5, 1]
    ]
    for pat in patterns:
        if sum(pat) != N: continue
        c = np.zeros((N, 2))
        idx = 0
        r_est = 0.095
        dy = math.sqrt(3) * r_est
        for r_idx, count in enumerate(pat):
            y = r_est + r_idx * dy
            offset = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + offset
            for _ in range(count):
                if idx < N:
                    c[idx] = [x, y]
                    idx += 1
                x += 2 * r_est
        configs.append(c)
        
    for _ in range(8):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    gx = np.linspace(0.1, 0.9, 6)
    gy = np.linspace(0.1, 0.9, 5)
    cx, cy = np.meshgrid(gx, gy)
    grid = np.column_stack([cx.ravel(), cy.ravel()])[:N]
    configs.append(grid)
    
    # 2. Gradient Ascent on Centers using LP Duals
    for init_c in configs:
        c = init_c.copy()
        step = 0.012
        for it in range(700):
            res = solve_lp(c)
            if res.success:
                s = -res.fun
                g = get_gradient(c, res)
                c += step * g
                c = np.clip(c, 0.001, 0.999)
                step *= 0.997
                
                # Periodic jitter to escape local minima
                if it % 60 == 0 and it > 0:
                    c += rng.normal(0, 0.0008, c.shape)
                    c = np.clip(c, 0.001, 0.999)
                    
                if s > global_best_sum:
                    global_best_sum = s
                    global_best_centers = c.copy()
                    
    if global_best_centers is None:
        global_best_centers = configs[0]
        
    # 3. SLSQP Polish on full variables (centers + radii)
    radii_init = solve_lp(global_best_centers).x
    p0 = np.concatenate([global_best_centers.flatten(), radii_init])
    
    def obj(p): 
        return -np.sum(p[2*N:])
        
    def cons(p):
        c = p[:2*N].reshape(N, 2)
        r = p[2*N:]
        # Boundary constraints
        con = np.concatenate([c[:,0]-r, 1-c[:,0]-r, c[:,1]-r, 1-c[:,1]-r])
        # Pairwise non-overlap constraints
        diff = c[:, None, :] - c[None, :, :]
        d = np.linalg.norm(diff, axis=2)
        idx = np.triu_indices(N, 1)
        con = np.concatenate([con, d[idx] - (r[idx[0]] + r[idx[1]])])
        return con
        
    bnds = [(0,1)]*(2*N) + [(0, 0.5)]*N
    try:
        res_sl = opt.minimize(obj, p0, method='SLSQP', bounds=bnds,
                              constraints={'type': 'ineq', 'fun': cons},
                              options={'maxiter': 6000, 'ftol': 1e-14})
        if res_sl.success or -res_sl.fun > global_best_sum:
            p0 = res_sl.x
    except Exception:
        pass
        
    centers_out = p0[:2*N].reshape(N, 2)
    radii_out = p0[2*N:]
    
    # 4. Strict Repair Pass to guarantee validation compliance
    for _ in range(30):
        changed = False
        for i in range(N):
            mx = min(centers_out[i,0], 1-centers_out[i,0], 
                     centers_out[i,1], 1-centers_out[i,1])
            if radii_out[i] > mx + 1e-10:
                radii_out[i] = mx
                changed = True
        for i in range(N):
            for j in range(i+1, N):
                d = math.hypot(centers_out[i,0]-centers_out[j,0], 
                               centers_out[i,1]-centers_out[j,1])
                if d < radii_out[i] + radii_out[j] - 1e-10:
                    shrink = (radii_out[i] + radii_out[j] - d) / 2.0 + 1e-9
                    radii_out[i] -= shrink
                    radii_out[j] -= shrink
                    changed = True
        if not changed: 
            break
        
    radii_out = np.maximum(radii_out, 0.0)
    final_sum = float(np.sum(radii_out))
    return centers_out, radii_out, final_sum
