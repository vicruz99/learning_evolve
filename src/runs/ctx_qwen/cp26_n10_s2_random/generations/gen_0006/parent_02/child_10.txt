import numpy as np
import scipy.optimize as opt
import math

N = 26
i_idx, j_idx = np.triu_indices(N, k=1)

def get_bounds():
    """Creates variable bounds: x,y in [0,1], r in [1e-6, 0.5]"""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def objective(p):
    """Minimize negative sum of radii to maximize total radius."""
    return -np.sum(p[2::3])

def constraints(p):
    """
    Computes all boundary and non-overlap constraints.
    Returns a 1D array where each element must be >= 0.
    Uses squared distances for better gradient conditioning.
    """
    x, y, r = p[0::3], p[1::3], p[2::3]
    c = np.empty(4 * N + len(i_idx))
    idx = 0
    
    # Boundary constraints
    c[idx:idx+N] = x - r; idx += N
    c[idx:idx+N] = 1.0 - x - r; idx += N
    c[idx:idx+N] = y - r; idx += N
    c[idx:idx+N] = 1.0 - y - r; idx += N
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    c[idx:] = dx*dx + dy*dy - dr*dr
    return c

def relax(centers, steps=600):
    """
    Physics-based relaxation to resolve overlaps and find stable contacts.
    Grows radii slightly each step while applying repulsive forces.
    """
    centers = centers.copy()
    radii = np.full(N, 0.03)
    for step in range(steps):
        radii *= 1.0004
        forces = np.zeros_like(centers)
        
        # Boundary forces
        for i in range(N):
            cx, cy = centers[i]
            r = radii[i]
            if cx < r: forces[i, 0] += (r - cx) * 500
            if cx > 1-r: forces[i, 0] -= (cx - (1-r)) * 500
            if cy < r: forces[i, 1] += (r - cy) * 500
            if cy > 1-r: forces[i, 1] -= (cy - (1-r)) * 500
            
        # Pairwise forces
        for i in range(N):
            for j in range(i+1, N):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                d = math.hypot(dx, dy)
                min_d = radii[i] + radii[j]
                if d < min_d and d > 1e-8:
                    f = (min_d - d) * 300.0 / d
                    fx, fy = dx * f, dy * f
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
                    
        centers += forces * 0.0005
        centers = np.clip(centers, 0.001, 0.999)
    return centers

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to exactly maximize sum of radii."""
    b_ub = []
    A_ub = []
    
    # Pairwise constraints: r_i + r_j <= d_ij
    for i in range(N):
        for j in range(i+1, N):
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(math.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1]))
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(N):
        b = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        row = np.zeros(N)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(b)
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    res = opt.linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*N, method='highs')
    if res.success:
        return res.x
    return np.full(N, 0.01)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = -np.inf
    best_p = None
    bounds = get_bounds()
    
    # Diverse hexagonal row patterns
    patterns = [
        [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 5, 6, 5, 5],
        [5, 5, 5, 6, 5], [5, 5, 5, 5, 6], [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4], [4, 6, 6, 6, 4], [5, 5, 5, 5, 5, 1],
        [6, 4, 6, 5, 5], [5, 7, 4, 5, 5], [6, 6, 5, 5, 4]
    ]
    
    candidates = []
    for pat in patterns:
        if sum(pat) != N: continue
        for r_est in [0.09, 0.095, 0.10, 0.105]:
            centers = []
            y = r_est
            for r_idx, count in enumerate(pat):
                offset = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + offset
                for _ in range(count):
                    centers.append([x, y])
                    x += 2 * r_est
                y += math.sqrt(3) * r_est
            c = np.array(centers[:N])
            c += np.random.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.05, 0.95)
            candidates.append(c)
            
    # Add dense random starts for robustness
    for _ in range(15):
        candidates.append(np.random.rand(N, 2) * 0.8 + 0.1)

    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Relaxation + Multi-Start SLSQP
    for c_init in candidates:
        c_rel = relax(c_init)
        r_rel = np.full(N, 0.08)
        p0 = np.zeros(N*3)
        p0[0::3] = c_rel[:, 0]
        p0[1::3] = c_rel[:, 1]
        p0[2::3] = r_rel
        
        for seed in range(4):
            p_start = p0.copy()
            p_start[0::3] += np.random.normal(0, 0.002, N)
            p_start[1::3] += np.random.normal(0, 0.002, N)
            p_start[2::3] += np.random.normal(0, 0.001, N)
            p_start[0::3] = np.clip(p_start[0::3], 0.01, 0.99)
            p_start[1::3] = np.clip(p_start[1::3], 0.01, 0.99)
            
            try:
                res = opt.minimize(objective, p_start, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13})
                if np.all(constraints(res.x) >= -1e-7):
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_p = res.x.copy()
            except Exception:
                pass

    # Fallback if optimization failed entirely
    if best_p is None:
        best_p = np.zeros(N*3)
        best_p[0::3] = np.random.rand(N)*0.8+0.1
        best_p[1::3] = np.random.rand(N)*0.8+0.1
        best_p[2::3] = 0.05
        
    # Phase 2: LP Polish (exact radii for fixed centers)
    c_lp = best_p[:2*N].reshape(N, 2)
    r_lp = solve_lp_radii(c_lp)
    if np.sum(r_lp) > best_sum:
        best_sum = np.sum(r_lp)
        best_p[2::3] = r_lp
        
    # Phase 3: Final high-precision SLSQP polish
    try:
        res_final = opt.minimize(objective, best_p, method='SLSQP', bounds=bounds,
                                 constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
        if np.all(constraints(res_final.x) >= -1e-8):
            best_p = res_final.x
            best_sum = -res_final.fun
    except Exception:
        pass
        
    centers = np.column_stack((best_p[0::3], best_p[1::3]))
    radii = best_p[2::3].copy()
    
    # Phase 4: Strict safety repair to guarantee validator tolerance
    for _ in range(40):
        changed = False
        for i in range(N):
            mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mx - 1e-9:
                radii[i] = mx
                changed = True
        for i in range(N):
            for j in range(i+1, N):
                d = math.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
