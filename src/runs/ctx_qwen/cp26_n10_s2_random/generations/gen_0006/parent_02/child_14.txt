2. **Powell Optimization on Centers**: We wrap the LP solver in an objective function `obj(centers)` that returns the negative sum of radii. import numpy as np
from scipy.optimize import linprog, minimize
import math

N = 26

def lp_sum_radii(centers):
    """Solves LP to maximize sum of radii given fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    n_pairs = n * (n - 1) // 2
    A_pair = np.zeros((n_pairs, n))
    b_pair = np.zeros(n_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_pair[idx, i] = 1.0
            A_pair[idx, j] = 1.0
            b_pair[idx] = dists[i, j]
            idx += 1
            
    # Boundary limits
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    A_bound = np.eye(n)
    b_bound = ub
    
    A_ub = np.vstack([A_pair, A_bound])
    b_ub = np.concatenate([b_pair, b_bound])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def obj(centers_flat):
    """Objective for center optimization: minimize negative LP sum of radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    val, _ = lp_sum_radii(centers)
    return -val

def repair(centers, radii):
    """Iteratively shrinks radii to strictly resolve overlaps and clamp to boundaries."""
    radii = radii.copy()
    for _ in range(50):
        changed = False
        for i in range(N):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx - 1e-10:
                radii[i] = mx
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def generate_inits():
    """Generates diverse initial center configurations."""
    inits = []
    rng = np.random.default_rng(42)
    
    patterns = [
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5],
        [5, 5, 6, 5, 5], [5, 5, 5, 6, 5], [5, 4, 5, 4, 5, 3],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 5, 4, 6, 6],
        [6, 6, 5, 5, 4], [5, 7, 5, 5, 4], [6, 6, 6, 5, 3],
        [5, 5, 5, 5, 5, 1], [5, 5, 5, 6, 5]
    ]
    
    for pat in patterns:
        if sum(pat) != N: continue
        centers = []
        r_est = 0.09
        dy = np.sqrt(3) * r_est
        for r_idx, cnt in enumerate(pat):
            y = r_est + r_idx * dy
            offset = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + offset
            for _ in range(cnt):
                centers.append([x, y])
                x += 2 * r_est
        if len(centers) == N:
            c = np.array(centers)
            c += rng.normal(0, 0.005, c.shape)
            c = np.clip(c, 0.02, 0.98)
            inits.append(c.flatten())
            
    for _ in range(10):
        c = rng.uniform(0.05, 0.95, (N, 2))
        inits.append(c.flatten())
        
    for _ in range(5):
        gx = np.linspace(0.1, 0.9, 5)
        gy = np.linspace(0.1, 0.9, 5)
        cx, cy = np.meshgrid(gx, gy)
        c = np.column_stack((cx.flatten(), cy.flatten()))
        c = np.vstack([c[:25], [0.5, 0.5]])
        c += rng.normal(0, 0.02, c.shape)
        c = np.clip(c, 0.02, 0.98)
        inits.append(c.flatten())
        
    return inits[:20]

def obj_full(p):
    """Objective for full joint optimization."""
    return -np.sum(p[2*N:])

def cons_full(p):
    """Constraints for full joint optimization."""
    c = p[:2*N].reshape(N, 2)
    r = p[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    inits = generate_inits()
    
    best_sum = 0.0
    best_centers = None
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    
    # Phase 1: Multi-start Powell optimization on centers
    for x0 in inits:
        try:
            res = minimize(obj, x0, method='Powell', 
                           bounds=bounds_centers,
                           options={'maxiter': 2500, 'ftol': 1e-12, 'xtol': 1e-12})
            val = -res.fun
            if val > best_sum:
                best_sum = val
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    if best_centers is None:
        best_centers = np.random.rand(N, 2) * 0.8 + 0.1
        
    # Phase 2: Basin hopping refinement
    rng = np.random.default_rng(123)
    for _ in range(12):
        pert = best_centers + rng.normal(0, 0.0015, best_centers.shape)
        pert = np.clip(pert, 0.02, 0.98)
        try:
            res = minimize(obj, pert.flatten(), method='Powell',
                           bounds=bounds_centers,
                           options={'maxiter': 1500, 'ftol': 1e-12})
            val = -res.fun
            if val > best_sum:
                best_sum = val
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Phase 3: SLSQP joint polish
    _, radii_init = lp_sum_radii(best_centers)
    x0_full = np.concatenate([best_centers.flatten(), radii_init])
    
    bounds_full = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
    try:
        res_full = minimize(obj_full, x0_full, method='SLSQP', bounds=bounds_full,
                            constraints={'type': 'ineq', 'fun': cons_full},
                            options={'maxiter': 5000, 'ftol': 1e-13})
        if -res_full.fun > best_sum:
            best_sum = -res_full.fun
            best_centers = res_full.x[:2*N].reshape(N, 2)
    except Exception:
        pass
        
    # Phase 4: Final LP radii computation and strict repair
    final_radii = repair(best_centers, lp_sum_radii(best_centers)[1])
    
    # Safety shrink to guarantee validator tolerance compliance
    for _ in range(10):
        valid = True
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], final_radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False
                break
        if not valid:
            break
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_centers[i, 0] - best_centers[j, 0], 
                               best_centers[i, 1] - best_centers[j, 1])
                if d < final_radii[i] + final_radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            break
        final_radii *= 0.99995
        
    final_radii = np.maximum(final_radii, 0.0)
    final_sum = float(np.sum(final_radii))
    return best_centers, final_radii, final_sum
