

<strategy>
To improve the packing, we shift from joint non-linear optimization of centers and radii to an alternating optimization strategy. The key insight is that for any fixed set of centers, finding the optimal radii that maximize the sum is a Linear Programming (LP) problem. This LP can be solved exactly and very quickly using `scipy.optimize.linprog` with the 'highs' solver.

By treating the LP solution's objective value as a function of the centers, we can optimize the centers directly. This function is continuous but non-differentiable, making derivative-free optimizers like Powell's method highly suitable. We will:
1. Precompute the constant constraint matrix `A_ub` for the LP.
2. Define an objective function that evaluates the maximum sum of radii via LP for any given center configuration.
3. Run Powell's method from diverse initial configurations (hexagonal lattices with varying row counts, dense random, and corner-focused layouts).
4. Perform local perturbations around the best found solution to escape local minima.
5. Extract the final optimal radii from the LP and apply strict numerical repairs to guarantee validation compliance.

This approach decouples the geometry optimization from the radius sizing, allowing the optimizer to focus purely on finding a dense, well-contacted center arrangement while the LP instantly computes the best possible radii for that arrangement.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
IU, JU = np.triu_indices(N, k=1)
NUM_PAIRS = len(IU)
NUM_BOUNDS = 4 * N

# Precompute constant constraint matrix for LP
# Constraints: r_i + r_j <= d_ij  (pairwise)
#              r_i <= boundary_dist_i  (boundary)
A_ub = np.zeros((NUM_PAIRS + NUM_BOUNDS, N))
for k in range(NUM_PAIRS):
    A_ub[k, IU[k]] = 1.0
    A_ub[k, JU[k]] = 1.0
idx = NUM_PAIRS
for i in range(N):
    A_ub[idx, i] = 1.0; idx += 1
    A_ub[idx, i] = 1.0; idx += 1
    A_ub[idx, i] = 1.0; idx += 1
    A_ub[idx, i] = 1.0; idx += 1

c_obj = -np.ones(N)  # Minimize -sum(r) => Maximize sum(r)
bounds_r = [(0.0, None)] * N

def lp_sum_radii(centers):
    """Solves LP to find max sum of radii for given centers."""
    centers = np.clip(centers, 1e-8, 1.0 - 1e-8)
    
    # Compute pairwise distances efficiently
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(diff[:,:,0]**2 + diff[:,:,1]**2)
    
    # Build b_ub dynamically
    b_ub = np.empty(NUM_PAIRS + NUM_BOUNDS)
    b_ub[:NUM_PAIRS] = dists[IU, JU]
    
    xb = centers[:, 0]
    yb = centers[:, 1]
    b_ub[NUM_PAIRS : NUM_PAIRS+N] = xb
    b_ub[NUM_PAIRS+N : 2*NUM_PAIRS+2*N] = 1.0 - xb
    b_ub[2*NUM_PAIRS+2*N : 3*NUM_PAIRS+3*N] = yb
    b_ub[3*NUM_PAIRS+3*N :] = 1.0 - yb
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def obj_func(centers_flat):
    """Objective for center optimization: minimize negative LP sum of radii."""
    c = centers_flat.reshape(N, 2)
    val, _ = lp_sum_radii(c)
    return -val

def generate_starts():
    """Generates diverse initial center configurations."""
    starts = []
    rng = np.random.default_rng(42)
    
    # Hexagonal lattice patterns summing to 26
    patterns = [
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], 
        [5, 5, 6, 5, 5], [5, 5, 5, 6, 5], [5, 5, 5, 5, 5, 1],
        [6, 5, 5, 5, 4], [4, 6, 5, 5, 6], [5, 5, 6, 6, 4],
        [5, 4, 6, 5, 6], [6, 6, 5, 5, 4], [4, 5, 6, 6, 5]
    ]
    
    for pat in patterns:
        if sum(pat) != N: continue
        pts = []
        r_est = 0.092
        dy = np.sqrt(3) * r_est
        y = r_est
        for r_idx, count in enumerate(pat):
            offset = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + offset
            for _ in range(count):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2 * r_est
            y += dy
        starts.append(np.array(pts[:N]))
        
    # Dense random starts
    for _ in range(15):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # Corner-focused starts to encourage boundary contacts
    for _ in range(10):
        c = rng.uniform(0.05, 0.95, (N, 2))
        c[:4] = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        starts.append(c)
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_centers = None
    best_sum = 0.0
    
    starts = generate_starts()
    
    # Phase 1: Multi-start Powell optimization
    for c0 in starts:
        try:
            res = minimize(obj_func, c0.flatten(), method='Powell',
                           bounds=[(0.0, 1.0)] * (2 * N),
                           options={'maxiter': 2500, 'ftol': 1e-12})
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    # Phase 2: Local perturbations to escape local minima
    if best_centers is not None:
        rng = np.random.default_rng(123)
        for _ in range(15):
            c_pert = best_centers + rng.normal(0, 0.004, best_centers.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            try:
                res = minimize(obj_func, c_pert.flatten(), method='Powell',
                               bounds=[(0.0, 1.0)] * (2 * N),
                               options={'maxiter': 2000, 'ftol': 1e-12})
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_centers = res.x.reshape(N, 2).copy()
            except Exception:
                pass

    # Fallback
    if best_centers is None:
        best_centers = np.random.rand(N, 2) * 0.8 + 0.1
        
    # Phase 3: Final LP extraction and numerical repair
    _, radii = lp_sum_radii(best_centers)
    radii = np.maximum(radii, 0.0)
    
    # Strict overlap and boundary resolution
    for _ in range(40):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            mx = min(best_centers[i, 0], 1.0 - best_centers[i, 0], 
                     best_centers[i, 1], 1.0 - best_centers[i, 1])
            if radii[i] > mx - 1e-9:
                radii[i] = mx
                changed = True
        # Resolve overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_centers[i, 0] - best_centers[j, 0], 
                               best_centers[i, 1] - best_centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
            
    # Final validation safety shrink
    for _ in range(10):
        valid = True
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False
                break
        if not valid:
            break
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_centers[i, 0] - best_centers[j, 0], 
                               best_centers[i, 1] - best_centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            break
        radii *= 0.99995
        
    radii = np.maximum(radii, 0.0)
    return best_centers, radii, float(np.sum(radii))
```
