2. **Derivative-Free Center Optimization**: Since the radius function is non-smooth (active constraints change as circles move), gradient-based methods often get stuck. We use `Powell`'s method (derivative-free) combined with a stochastic perturbation search to navigate the center space globally.

import numpy as np
import scipy.optimize as opt
import math

N = 26
NUM_PAIRS = N * (N - 1) // 2
i_idx, j_idx = np.triu_indices(N, k=1)

def setup_lp_matrix():
    """Precomputes the constant inequality constraint matrix for the LP radius solver."""
    A_ub = np.zeros((NUM_PAIRS + 4 * N, N))
    k = 0
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            k += 1
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(N):
        A_ub[k, i] = 1.0; k += 1
        A_ub[k, i] = 1.0; k += 1
        A_ub[k, i] = 1.0; k += 1
        A_ub[k, i] = 1.0; k += 1
    return A_ub

A_MAT = setup_lp_matrix()

def solve_radii(centers):
    """Given fixed centers, solves LP to maximize sum of radii."""
    n = centers.shape[0]
    b_ub = np.zeros(NUM_PAIRS + 4 * n)
    
    # Compute pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    # Fill LP bounds vector
    b_ub[:NUM_PAIRS] = dists[i_idx, j_idx]
    b_ub[NUM_PAIRS : NUM_PAIRS + n] = centers[:, 0]
    b_ub[NUM_PAIRS + n : NUM_PAIRS + 2*n] = 1.0 - centers[:, 0]
    b_ub[NUM_PAIRS + 2*n : NUM_PAIRS + 3*n] = centers[:, 1]
    b_ub[NUM_PAIRS + 3*n : NUM_PAIRS + 4*n] = 1.0 - centers[:, 1]
    
    # Ensure feasibility tolerance
    b_ub = np.maximum(b_ub, 1e-9)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    res = opt.linprog(c_obj, A_ub=A_MAT, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-9), 0.0

def obj_centers(x):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = x.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, val = solve_radii(centers)
    return -val

def generate_starts():
    """Generates diverse initial center configurations."""
    starts = []
    # Hexagonal lattice patterns
    pats = [[5,5,5,5,6], [6,5,5,5,5], [5,6,5,5,5], [5,5,6,5,5], [5,5,5,6,5], 
            [4,6,5,5,6], [6,4,5,6,5], [5,4,6,5,6], [6,5,6,4,5], [5,5,5,5,5,1],
            [5,5,5,4,7], [7,5,5,4,5], [5,6,6,4,5], [6,5,5,6,4]]
    for pat in pats:
        if sum(pat) != N: continue
        c = np.zeros((N, 2))
        idx = 0
        y = 0.085
        for r_idx, cnt in enumerate(pat):
            shift = 0.085 if r_idx % 2 == 1 else 0.0
            x = 0.085 + shift
            for _ in range(cnt):
                if idx < N:
                    c[idx] = [x, y]
                    idx += 1
                x += 0.17
            y += 0.085 * math.sqrt(3)
        starts.append(c.flatten())
        
    # Random dense starts
    rng = np.random.default_rng(42)
    for _ in range(15):
        c = rng.uniform(0.1, 0.9, (N, 2))
        starts.append(c.flatten())
    return starts

def repair(centers, radii):
    """Iteratively shrinks radii to strictly resolve overlaps and clamp to boundaries."""
    radii = radii.copy()
    for _ in range(30):
        changed = False
        for i in range(N):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx - 1e-9:
                radii[i] = mx
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds_c = [(1e-4, 1.0 - 1e-4)] * (2 * N)
    
    best_sum = 0.0
    best_centers = None
    
    starts = generate_starts()
    
    # Phase 1: Multi-start Powell optimization on centers
    for x0 in starts:
        try:
            res = opt.minimize(obj_centers, x0, method='Powell', bounds=bounds_c,
                               options={'maxiter': 3000, 'ftol': 1e-12, 'xtol': 1e-12})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Phase 2: Perturbation search to escape local minima
    rng = np.random.default_rng(123)
    current_sum = best_sum
    current_centers = best_centers
    step = 0.015
    
    for _ in range(800):
        trial = current_centers + rng.normal(0, step, (N, 2))
        trial = np.clip(trial, 1e-4, 1.0 - 1e-4)
        _, trial_sum = solve_radii(trial)
        
        if trial_sum > current_sum + 1e-9:
            current_sum = trial_sum
            current_centers = trial.copy()
            best_sum = current_sum
            best_centers = trial.copy()
            
        step *= 0.998
        
    # Phase 3: Local Powell refinement around best
    for _ in range(5):
        pert = best_centers + rng.normal(0, 0.003, (N, 2))
        pert = np.clip(pert, 1e-4, 1.0 - 1e-4)
        try:
            res_p = opt.minimize(obj_centers, pert.flatten(), method='Powell', bounds=bounds_c,
                                 options={'maxiter': 1500, 'ftol': 1e-13})
            if -res_p.fun > best_sum:
                best_sum = -res_p.fun
                best_centers = res_p.x.reshape(N, 2)
        except Exception:
            pass
            
    # Phase 4: Joint SLSQP polish for maximum precision
    _, radii_init = solve_radii(best_centers)
    p0 = np.zeros(N * 3)
    p0[0::3] = best_centers[:, 0]
    p0[1::3] = best_centers[:, 1]
    p0[2::3] = radii_init * 0.9995
    
    def obj_joint(p):
        return -np.sum(p[2::3])
        
    def cons_joint(p):
        c = p[:2*N].reshape(N, 2)
        r = p[2*N:]
        con = np.empty(4*N + NUM_PAIRS)
        idx = 0
        con[idx:idx+N] = c[:, 0] - r; idx += N
        con[idx:idx+N] = 1.0 - c[:, 0] - r; idx += N
        con[idx:idx+N] = c[:, 1] - r; idx += N
        con[idx:idx+N] = 1.0 - c[:, 1] - r; idx += N
        
        c1 = c[i_idx] - c[j_idx]
        dr = r[i_idx] + r[j_idx]
        con[idx:] = np.sqrt(np.sum(c1**2, axis=1)) - dr
        return con

    bounds_j = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
    try:
        res_j = opt.minimize(obj_joint, p0, method='SLSQP', bounds=bounds_j,
                             constraints={'type': 'ineq', 'fun': cons_joint},
                             options={'maxiter': 8000, 'ftol': 1e-14})
        if -res_j.fun > best_sum:
            best_sum = -res_j.fun
            best_centers = res_j.x[:2*N].reshape(N, 2)
    except Exception:
        pass
        
    # Final exact radii calculation and repair
    final_radii, _ = solve_radii(best_centers)
    final_radii = repair(best_centers, final_radii)
    
    # Safety shrink loop to guarantee validator tolerance
    for _ in range(10):
        valid = True
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], final_radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False; break
        if not valid: break
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
                if d < final_radii[i] + final_radii[j] - 1e-12:
                    valid = False; break
            if not valid: break
        if valid: break
        final_radii *= 0.99999
        
    final_radii = np.maximum(final_radii, 0.0)
    return best_centers, final_radii, float(np.sum(final_radii))
