

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we need to escape local minima that plague standard gradient-based optimizers. The dual gradient ascent method is highly effective for circle packing: by solving an LP for radii given fixed centers, we obtain Lagrange multipliers (dual variables) that represent the "contact forces" between circles and between circles and boundaries. Moving centers along these forces naturally pushes circles into dense, stable configurations where the contact graph is optimized.

Strategy:
1. **Precompute LP Structure**: The constraint matrix for maximizing radii is constant. We precompute it to speed up iterative LP solves.
2. **Dual-Based Force Relaxation**: For each initial configuration, iteratively solve the LP, extract dual variables, compute forces, and update centers. This efficiently navigates the non-smooth landscape of the packing problem.
3. **Diverse Initializations**: Generate multiple hexagonal lattice patterns with different row counts and offsets, plus random dense starts. This covers different topological arrangements.
4. **Perturbation Search**: After relaxation, apply controlled random perturbations and continue force ascent to escape shallow local minima.
5. **SLSQP Polish**: Use a high-precision constrained optimizer on the full variable set (centers + radii) starting from the best relaxed configuration to squeeze out remaining numerical slack.
6. **Strict Repair & Validation**: Iteratively shrink radii to guarantee strict compliance with the validator's 1e-12 tolerance, handling boundary and overlap constraints separately.

This hybrid approach combines the robustness of LP duals, the exploration of multi-start perturbations, and the precision of SLSQP, targeting a sum of radii > 2.636.

```python
import numpy as np
from scipy.optimize import linprog, minimize
import math

N = 26
IU, JU = np.triu_indices(N, k=1)
NUM_PAIRS = len(IU)

# Precompute constant parts of the LP constraint matrix
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        idx += 1
for i in range(N):
    for _ in range(4):
        A_LP[idx, i] = 1.0
        idx += 1

C_OBJ_LP = -np.ones(N)
BOUNDS_LP = [(0, None)] * N

def solve_lp_and_dists(centers):
    """Solves LP for max sum of radii given fixed centers. Returns radii, duals, objective value, and distance matrix."""
    n = centers.shape[0]
    b_ub = np.zeros(NUM_PAIRS + 4 * n)
    dists = np.zeros((n, n))
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub[idx] = d
            dists[i, j] = d
            dists[j, i] = d
            idx += 1
            
    for i in range(n):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(C_OBJ_LP, A_ub=A_LP, b_ub=b_ub, bounds=BOUNDS_LP, method='highs')
    
    if res.success:
        # Fallback for older scipy versions if attribute missing
        marginals = getattr(res.ineqlin, 'marginals', np.zeros_like(b_ub))
        return res.x, marginals, res.fun, dists
    return None, None, np.inf, dists

def compute_forces(centers, duals, dists):
    """Computes gradient ascent forces on centers from LP dual variables."""
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    idx = 0
    
    # Pairwise repulsion forces
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]
            idx += 1
            if lam > 1e-7 and dists[i, j] > 1e-7:
                vec = centers[i] - centers[j]
                f = (lam / dists[i, j]) * vec
                forces[i] += f
                forces[j] -= f
                
    # Boundary repulsion forces
    b_start = NUM_PAIRS
    for i in range(n):
        forces[i, 0] += duals[b_start + 4*i] - duals[b_start + 4*i + 1]
        forces[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
        
    return forces

def obj_slsqp(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_slsqp(v):
    """Constraints for SLSQP: boundary and non-overlap."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    d = np.linalg.norm(c[IU] - c[JU], axis=1)
    con = np.concatenate([con, d - (r[IU] + r[JU])])
    return con

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_centers = None
    best_sum = -np.inf
    best_radii = None
    
    # Diverse hexagonal patterns summing to 26
    patterns = [
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5],
        [5, 5, 6, 5, 5], [5, 5, 5, 6, 5], [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4], [4, 6, 5, 5, 6], [6, 6, 5, 5, 4],
        [5, 5, 5, 6, 5], [5, 4, 6, 5, 6], [6, 5, 4, 6, 5]
    ]
    
    inits = []
    for pat in patterns:
        if sum(pat) != N:
            continue
        pts = []
        r_est = 0.10
        dy = np.sqrt(3) * r_est
        for r_idx, cnt in enumerate(pat):
            y = r_est + r_idx * dy
            offset = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + offset
            for _ in range(cnt):
                pts.append([x, y])
                x += 2 * r_est
        pts = np.array(pts[:N])
        # Normalize to [0.075, 0.925] range
        pts -= pts.min(axis=0)
        pts /= pts.max(axis=0)
        pts = pts * 0.85 + 0.075
        inits.append(pts)
        
    # Add random dense starts
    for _ in range(8):
        inits.append(np.random.rand(N, 2) * 0.8 + 0.1)
        
    # Phase 1 & 2: Dual-based relaxation and perturbation search
    for c0 in inits:
        centers = c0.copy()
        
        # Main relaxation
        for step in range(600):
            r, duals, obj_val, dists = solve_lp_and_dists(centers)
            if r is None:
                break
                
            forces = compute_forces(centers, duals, dists)
            step_size = 0.004 / (1.0 + step * 0.005)
            centers += step_size * forces
            centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
            
            s = -obj_val
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = r.copy()
                
        # Perturbation escape search
        for _ in range(30):
            pert = best_centers + np.random.normal(0, 0.008, best_centers.shape)
            pert = np.clip(pert, 0.02, 0.98)
            r_p, d_p, obj_p, dists_p = solve_lp_and_dists(pert)
            
            if r_p is not None and -obj_p > best_sum:
                best_centers = pert
                best_sum = -obj_p
                best_radii = r_p
                
                # Local descent from perturbation
                for step in range(400):
                    r_loc, duals_loc, obj_loc, dists_loc = solve_lp_and_dists(best_centers)
                    if r_loc is None:
                        break
                    f_loc = compute_forces(best_centers, duals_loc, dists_loc)
                    ss = 0.002 / (1.0 + step * 0.01)
                    best_centers += ss * f_loc
                    best_centers = np.clip(best_centers, 1e-5, 1.0 - 1e-5)
                    if -obj_loc > best_sum:
                        best_sum = -obj_loc
                        best_radii = r_loc.copy()

    # Phase 3: SLSQP High-Precision Polish
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii])
        bounds_slsqp = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        try:
            res = minimize(obj_slsqp, x0, method='SLSQP', bounds=bounds_slsqp,
                           constraints={'type': 'ineq', 'fun': cons_slsqp},
                           options={'maxiter': 5000, 'ftol': 1e-13})
            if -res.fun > best_sum:
                best_centers = res.x[:2 * N].reshape(N, 2)
                best_radii = res.x[2 * N:]
                best_sum = -res.fun
        except Exception:
            pass

    # Phase 4: Strict Validation Repair
    radii = best_radii.copy()
    for _ in range(50):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            mx = min(best_centers[i, 0], 1.0 - best_centers[i, 0],
                     best_centers[i, 1], 1.0 - best_centers[i, 1])
            if radii[i] > mx - 1e-9:
                radii[i] = mx
                changed = True
        # Resolve overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_centers[i, 0] - best_centers[j, 0],
                               best_centers[i, 1] - best_centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return best_centers, radii, float(np.sum(radii))
```
