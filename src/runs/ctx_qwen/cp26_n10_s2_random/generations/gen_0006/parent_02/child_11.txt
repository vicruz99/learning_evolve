1. **Center Optimization with LP-based Radii**: For any fixed set of centers, the optimal radii that maximize the sum while satisfying non-overlap and boundary constraints can be found exactly via a Linear Program (LP). This transforms the problem into optimizing only the 52 center coordinates, where the objective function evaluates the LP solution. We use `scipy.optimize.minimize` with the `Powell` method (derivative-free) from multiple diverse initial configurations (hexagonal lattices with varying row patterns + random dense starts). Powell handles the non-smooth objective landscape effectively.
2. **Constrained Polish & Repair**: After finding the best centers, we compute the exact LP radii. A deterministic repair routine clamps radii to boundaries and resolves any residual overlaps to guarantee the validation function passes within tolerance.

This approach avoids the non-convex quadratic constraints that often trap gradient-based optimizers in local minima, leveraging the efficiency of modern LP solvers (`highs`) to evaluate the objective accurately and rapidly.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
i_idx, j_idx = np.triu_indices(N, k=1)
PAIR_COUNT = len(i_idx)

# Precompute pairwise constraint matrix structure for LP
A_PAIRWISE = np.zeros((PAIR_COUNT, N))
for k, (i, j) in enumerate(zip(i_idx, j_idx)):
    A_PAIRWISE[k, i] = 1.0
    A_PAIRWISE[k, j] = 1.0

def compute_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    # Boundary upper bounds
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-7)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_pairwise = dists[i_idx, j_idx]
    
    # Solve LP: min -sum r  s.t. r_i + r_j <= dist_ij, 0 <= r_i <= ub_i
    res = linprog(-np.ones(n), A_ub=A_PAIRWISE, b_ub=b_pairwise,
                  bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:
        return res.x
    # Fallback if LP fails (rare)
    return ub * 0.95

def obj_sum_radii(p):
    """Objective for center-only optimization: minimize negative sum of radii."""
    centers = p.reshape(N, 2)
    centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
    radii = compute_lp_radii(centers)
    return -np.sum(radii)

def generate_inits():
    """Generates diverse initial center configurations."""
    inits = []
    patterns = [
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5],
        [5, 5, 6, 5, 5], [5, 5, 5, 6, 5], [4, 6, 6, 5, 5],
        [5, 4, 6, 6, 5], [6, 6, 5, 5, 4], [5, 5, 4, 6, 6],
        [4, 5, 6, 5, 6], [6, 5, 4, 5, 6], [5, 5, 5, 4, 7]
    ]
    
    rng = np.random.default_rng(42)
    for pat in patterns:
        if sum(pat) != N: 
            continue
        c = []
        r_est = 0.092
        dy = np.sqrt(3) * r_est
        for r_idx, count in enumerate(pat):
            y = r_est + r_idx * dy
            offset = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + offset
            for _ in range(count):
                c.append([x, y])
                x += 2 * r_est
        c = np.array(c[:N])
        c += rng.normal(0, 0.005, c.shape)
        c = np.clip(c, 0.02, 0.98)
        inits.append(c.flatten())
        
    # Add random dense starts to escape symmetry traps
    for _ in range(15):
        c = rng.uniform(0.05, 0.95, (N, 2))
        inits.append(c.flatten())
        
    return inits

def repair(centers, radii):
    """Iteratively shrinks radii to strictly resolve overlaps and clamp to boundaries."""
    radii = radii.copy()
    for _ in range(50):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx - 1e-9:
                radii[i] = mx
                changed = True
        # Resolve overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = -np.inf
    best_centers = None
    
    inits = generate_inits()
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    
    # Phase 1: Optimize centers using Powell with LP-based radii evaluation
    for p0 in inits:
        try:
            res = minimize(obj_sum_radii, p0, method='Powell', bounds=bounds_centers,
                           options={'maxiter': 2500, 'ftol': 1e-14, 'xtol': 1e-14})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            continue
            
    # Phase 2: Perturbation refinement to escape local minima
    rng = np.random.default_rng(123)
    if best_centers is not None:
        for _ in range(20):
            p_pert = best_centers.flatten() + rng.normal(0, 0.003, 2*N)
            p_pert = np.clip(p_pert, 0.01, 0.99)
            try:
                res = minimize(obj_sum_radii, p_pert, method='Powell', bounds=bounds_centers,
                               options={'maxiter': 2000, 'ftol': 1e-14, 'xtol': 1e-14})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_centers = res.x.reshape(N, 2)
            except Exception:
                continue
                
    # Compute final radii for best centers
    radii = compute_lp_radii(best_centers)
    
    # Phase 3: SLSQP polish on full variables (centers + radii) for precision
    p0_full = np.concatenate([best_centers.flatten(), radii])
    
    def obj_full(p):
        return -np.sum(p[2*N:])
        
    def cons_full(p):
        c = p[:2*N].reshape(N, 2)
        r = p[2*N:]
        # Boundary constraints
        con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
        # Overlap constraints
        dx = c[:, 0][:, None] - c[:, 0][None, :]
        dy = c[:, 1][:, None] - c[:, 1][None, :]
        d = np.hypot(dx, dy)
        d[np.arange(N), np.arange(N)] = np.inf
        con = np.concatenate([con, d[i_idx, j_idx] - (r[i_idx] + r[j_idx])])
        return con
        
    bounds_full = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    try:
        res_full = minimize(obj_full, p0_full, method='SLSQP', bounds=bounds_full,
                            constraints={'type': 'ineq', 'fun': cons_full},
                            options={'maxiter': 6000, 'ftol': 1e-14})
        if np.all(cons_full(res_full.x) >= -1e-9):
            best_centers = res_full.x[:2*N].reshape(N, 2)
            radii = res_full.x[2*N:]
    except Exception:
        pass
        
    # Strict repair pass
    radii = repair(best_centers, radii)
    
    # Final safety shrink to guarantee validator tolerance
    for _ in range(10):
        valid = True
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False
                break
        if not valid: 
            break
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid: 
                break
        if valid: 
            break
        radii *= 0.99995
        
    radii = np.maximum(radii, 0.0)
    return best_centers, radii, float(np.sum(radii))
