import numpy as np
import scipy.optimize as opt
import math

N = 26
NUM_PAIRS = N * (N - 1) // 2
PAIR_LIST = [(i, j) for i in range(N) for j in range(i + 1, N)]
I_IDX, J_IDX = zip(*PAIR_LIST)

# Precompute LP constraint matrix structure
A_PAIRS = np.zeros((NUM_PAIRS, N))
for k, (i, j) in enumerate(PAIR_LIST):
    A_PAIRS[k, i] = 1.0
    A_PAIRS[k, j] = 1.0

A_BOUND = np.eye(N)
A_ub = np.vstack([A_PAIRS, A_BOUND])

def get_lp_result(centers):
    """Solves LP to find optimal radii for fixed centers."""
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    b_pairs = dists[np.array(I_IDX), np.array(J_IDX)]
    b_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    b_ub = np.concatenate([b_pairs, b_bound])
    res = opt.linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return res.x if res.success else np.zeros(N)

def objective_centers(x):
    """Objective for center optimization: minimize negative sum of optimal radii."""
    c = x.reshape(N, 2)
    r = get_lp_result(c)
    return -np.sum(r)

def get_bounds_centers():
    return [(0.0, 1.0)] * (2 * N)

def generate_starts(n_starts):
    """Generates diverse initial center configurations."""
    starts = []
    rng = np.random.default_rng(123)
    patterns = [
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], 
        [5, 5, 6, 5, 5], [5, 5, 5, 6, 5], [5, 5, 5, 5, 5, 1],
        [6, 5, 5, 5, 4], [4, 6, 5, 5, 6], [5, 5, 6, 6, 4],
        [5, 4, 6, 5, 6], [4, 5, 6, 6, 5], [6, 6, 5, 5, 4]
    ]
    for pat in patterns:
        if sum(pat) != N: continue
        centers = []
        r_est = 0.095
        dy = np.sqrt(3) * r_est
        for r_idx, count in enumerate(pat):
            y = r_est + r_idx * dy
            offset = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + offset
            for _ in range(count):
                centers.append([x, y])
                x += 2 * r_est
        starts.append(np.array(centers[:N]))
    for _ in range(max(0, n_starts - len(starts))):
        c = rng.uniform(0.1, 0.9, (N, 2))
        starts.append(c)
    return starts

def repair(centers, radii):
    """Iteratively shrinks radii to strictly resolve overlaps and clamp to boundaries."""
    radii = radii.copy()
    for _ in range(50):
        changed = False
        for i in range(N):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx - 1e-10:
                radii[i] = mx
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    best_sum = -np.inf
    best_centers = None
    
    starts = generate_starts(15)
    center_bounds = get_bounds_centers()
    
    # Phase 1: Powell optimization on centers
    for c_init in starts:
        c_init = c_init + rng.normal(0, 0.005, c_init.shape)
        c_init = np.clip(c_init, 0.02, 0.98)
        try:
            res = opt.minimize(objective_centers, c_init.flatten(), method='Powell',
                               bounds=center_bounds, 
                               options={'maxiter': 3000, 'ftol': 1e-12, 'xtol': 1e-12})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    # Phase 2: Perturb & Polish to escape local minima
    if best_centers is not None:
        for _ in range(20):
            c_pert = best_centers + rng.normal(0, 0.002, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            try:
                res = opt.minimize(objective_centers, c_pert.flatten(), method='Powell',
                                   bounds=center_bounds, 
                                   options={'maxiter': 2000, 'ftol': 1e-12, 'xtol': 1e-12})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_centers = res.x.reshape(N, 2).copy()
            except Exception:
                pass
                
    # Compute final radii via LP
    best_radii = get_lp_result(best_centers)
    
    # Phase 3: SLSQP joint polish for precision
    x0 = np.concatenate([best_centers.flatten(), best_radii])
    def obj_full(v):
        return -np.sum(v[2*N:])
        
    def cons_full(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        con = np.concatenate([c[:,0]-r, 1-c[:,0]-r, c[:,1]-r, 1-c[:,1]-r])
        diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        idx = np.triu_indices(N, k=1)
        con = np.concatenate([con, dists[idx] - (r[idx[0]] + r[idx[1]])])
        return con
        
    bounds_full = center_bounds + [(0.0, 0.5)]*N
    cons_obj = {'type': 'ineq', 'fun': cons_full}
    try:
        res_full = opt.minimize(obj_full, x0, method='SLSQP', bounds=bounds_full,
                                constraints=cons_obj, options={'maxiter': 5000, 'ftol': 1e-13})
        if -res_full.fun > np.sum(best_radii) - 1e-7:
            best_centers = res_full.x[:2*N].reshape(N, 2)
            best_radii = res_full.x[2*N:]
    except Exception:
        pass
        
    # Final strict repair to guarantee validation tolerances
    best_radii = repair(best_centers, best_radii)
    for _ in range(10):
        valid = True
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False
                break
        if not valid:
            break
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_centers[i, 0] - best_centers[j, 0], 
                               best_centers[i, 1] - best_centers[j, 1])
                if d < best_radii[i] + best_radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            break
        best_radii *= 0.99995
        
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
