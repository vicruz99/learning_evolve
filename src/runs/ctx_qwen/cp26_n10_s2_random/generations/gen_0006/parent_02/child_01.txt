

```python
import numpy as np
import scipy.optimize as opt
import math

N = 26

def get_optimal_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e6)
    
    # Boundary limits
    b_lim = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    b_lim = np.maximum(b_lim, 1e-12)
    
    m = n*(n-1)//2 + n
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    A_ub[idx:, np.arange(n)] = 1.0
    b_ub[idx:] = b_lim
    
    bounds = [(0.0, None)] * n
    res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 0.001), 0.026

def obj_centers(c_flat):
    """Objective for center-only optimization: minimize negative LP sum of radii."""
    c = c_flat.reshape(N, 2)
    _, s = get_optimal_radii_lp(c)
    return -s

def force_directed_init(seed, steps=1200, growth=1.0008):
    """Fast vectorized force-directed growth to find dense topologies."""
    rng = np.random.default_rng(seed)
    centers = rng.uniform(0.1, 0.9, (N, 2))
    radii = np.full(N, 0.025)
    lr = 0.008
    
    for _ in range(steps):
        radii *= growth
        forces = np.zeros_like(centers)
        
        # Boundary forces
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            if x < r: forces[i, 0] += (r - x) * 200.0
            if x > 1-r: forces[i, 0] -= (x - (1-r)) * 200.0
            if y < r: forces[i, 1] += (r - y) * 200.0
            if y > 1-r: forces[i, 1] -= (y - (1-r)) * 200.0
            
        # Pairwise forces (vectorized)
        dx = centers[:, None, 0] - centers[None, :, 0]
        dy = centers[:, None, 1] - centers[None, :, 1]
        dist = np.hypot(dx, dy)
        np.fill_diagonal(dist, np.inf)
        
        min_dist = radii[:, None] + radii[None, :]
        overlap = np.maximum(min_dist - dist, 0.0)
        force_mag = overlap * 150.0 / (dist + 1e-9)
        
        dir_x = dx / (dist + 1e-9)
        dir_y = dy / (dist + 1e-9)
        
        forces[:, 0] += (force_mag * dir_x).sum(axis=1)
        forces[:, 1] += (force_mag * dir_y).sum(axis=1)
        
        centers += forces * lr
        lr *= 0.997
        centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
        
    return centers

def constraints_slsqp(p):
    """SLSQP constraints: boundary and non-overlap (dist >= r_i + r_j)."""
    x, y, r = p[0::3], p[1::3], p[2::3]
    c = np.empty(4*N + N*(N-1)//2)
    idx = 0
    
    c[idx:idx+N] = x - r; idx += N
    c[idx:idx+N] = 1.0 - x - r; idx += N
    c[idx:idx+N] = y - r; idx += N
    c[idx:idx+N] = 1.0 - y - r; idx += N
    
    I, J = np.triu_indices(N, k=1)
    dx = x[I] - x[J]
    dy = y[I] - y[J]
    dr = r[I] + r[J]
    c[idx:] = np.hypot(dx, dy) - dr
    return c

def repair(centers, radii):
    """Iteratively shrinks radii to strictly resolve overlaps and clamp to boundaries."""
    radii = radii.copy()
    for _ in range(60):
        changed = False
        for i in range(N):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx - 1e-9:
                radii[i] = mx
                changed = True
        for i in range(N):
            for j in range(i+1, N):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = -np.inf
    best_centers = None
    bounds_c = [(0.01, 0.99)] * (2 * N)
    
    # 1. Generate diverse initial configurations
    inits = []
    for s in range(12):
        inits.append(force_directed_init(s))
        
    # Hexagonal lattice patterns summing to 26
    patterns = [[5,5,5,5,6], [6,5,5,5,5], [5,6,5,5,5], [5,5,6,5,5], [5,5,5,6,5], [4,5,6,5,6]]
    for pat in patterns:
        if sum(pat) != N: continue
        c = np.zeros((N, 2))
        idx = 0
        y = 0.088
        for r_idx, cnt in enumerate(pat):
            offset = 0.088 if r_idx % 2 == 1 else 0.0
            x = 0.088 + offset
            for _ in range(cnt):
                if idx < N:
                    c[idx] = [x, y]
                    idx += 1
                x += 0.176
            y += 0.088 * np.sqrt(3)
        c += np.random.normal(0, 0.004, c.shape)
        c = np.clip(c, 0.02, 0.98)
        inits.append(c)

    # 2. Optimize centers using Powell (exploits exact LP evaluation)
    for c0 in inits:
        try:
            res = opt.minimize(obj_centers, c0.flatten(), method='Powell', 
                               bounds=bounds_c, options={'maxiter': 800, 'ftol': 1e-14})
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass

    # 3. Local perturbation search around best topology
    rng = np.random.default_rng(42)
    for _ in range(25):
        pert = best_centers + rng.normal(0, 0.0012, best_centers.shape)
        pert = np.clip(pert, 0.015, 0.985)
        try:
            res = opt.minimize(obj_centers, pert.flatten(), method='Powell',
                               bounds=bounds_c, options={'maxiter': 600, 'ftol': 1e-14})
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass

    # 4. High-precision SLSQP polish on full (centers, radii) space
    radii_init, _ = get_optimal_radii_lp(best_centers)
    p0 = np.zeros(N * 3)
    p0[0::3] = best_centers[:, 0]
    p0[1::3] = best_centers[:, 1]
    p0[2::3] = radii_init * 0.995 # Start slightly feasible
    
    bounds_full = [(0.0, 1.0)] * (2 * N) + [(1e-7, 0.5)] * N
    try:
        res_slsqp = opt.minimize(lambda p: -np.sum(p[2::3]), p0, method='SLSQP', 
                                 bounds=bounds_full, constraints={'type': 'ineq', 'fun': constraints_slsqp},
                                 options={'maxiter': 5000, 'ftol': 1e-13})
        if -res_slsqp.fun > best_sum - 1e-8:
            c_vals = constraints_slsqp(res_slsqp.x)
            if np.all(c_vals >= -1e-8):
                best_sum = -res_slsqp.fun
                best_centers = res_slsqp.x[:2*N].reshape(N, 2)
                radii_init = res_slsqp.x[2*N:]
    except Exception:
        pass

    # 5. Final exact LP radii and strict repair
    radii, final_sum = get_optimal_radii_lp(best_centers)
    radii = repair(best_centers, radii)
    
    # Infinitesimal shrink to guarantee validator tolerance (1e-12)
    radii *= 0.999999995
    final_sum = float(np.sum(radii))
    
    return best_centers, radii, final_sum
```
