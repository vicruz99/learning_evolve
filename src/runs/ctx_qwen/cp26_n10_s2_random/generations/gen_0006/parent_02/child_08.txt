import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def get_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    dists = np.linalg.norm(centers[:, None] - centers[None, :], axis=2)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(lims[i])
        
    res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 0.01), 0.26

def obj_centers(x):
    """Objective for center optimization: negative sum of LP-optimal radii."""
    c = x.reshape(N, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    _, s = get_lp_radii(c)
    return -s

def generate_hex_inits():
    """Generates diverse hexagonal and relaxed initial configurations."""
    inits = []
    patterns = [
        [5,5,5,5,6], [6,5,5,5,5], [5,6,5,5,5], [5,5,6,5,5], [5,5,5,6,5],
        [6,5,5,5,4], [4,6,5,5,6], [5,5,6,6,4], [5,4,6,5,6], [6,4,5,5,6],
        [5,5,5,5,5,1], [4,5,5,5,7], [7,5,5,5,4], [6,6,5,5,4]
    ]
    for pat in patterns:
        if sum(pat) != N: 
            continue
        pts = []
        r = 0.095
        dy = np.sqrt(3) * r
        for r_idx, cnt in enumerate(pat):
            y = r + r_idx * dy
            off = r if r_idx % 2 == 1 else 0.0
            x = r + off
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2 * r
        inits.append(np.array(pts[:N]))
        
    np.random.seed(42)
    for _ in range(8):
        c = np.random.rand(N, 2) * 0.8 + 0.1
        # Simple repulsion relaxation
        for _ in range(100):
            for i in range(N):
                for j in range(i+1, N):
                    d = np.linalg.norm(c[i]-c[j])
                    if d < 0.18:
                        push = (0.18 - d) * 0.5
                        v = (c[i]-c[j])/(d+1e-6)
                        c[i] += v*push
                        c[j] -= v*push
            c = np.clip(c, 0.05, 0.95)
        inits.append(c)
    return inits

def run_packing():
    bounds_c = [(0.005, 0.995)] * (2 * N)
    best_c = None
    best_s = 0.0
    
    inits = generate_hex_inits()
    
    # Phase 1: Powell optimization on centers (uses LP inside for exact radii)
    for c0 in inits:
        res = minimize(obj_centers, c0.flatten(), method='Powell',
                       bounds=bounds_c, options={'maxiter': 1500, 'ftol': 1e-13})
        if -res.fun > best_s:
            best_s = -res.fun
            best_c = res.x.reshape(N, 2)
            
        # Perturb best found so far to explore neighboring basins
        for _ in range(3):
            cp = best_c + np.random.normal(0, 0.003, (N, 2))
            cp = np.clip(cp, 0.02, 0.98)
            res2 = minimize(obj_centers, cp.flatten(), method='Powell',
                            bounds=bounds_c, options={'maxiter': 800, 'ftol': 1e-11})
            if -res2.fun > best_s:
                best_s = -res2.fun
                best_c = res2.x.reshape(N, 2)

    # Phase 2: SLSQP full-variable polish for gradient-based precision
    radii0, _ = get_lp_radii(best_c)
    # Format: [x0, x1, ..., y0, y1, ..., r0, r1, ...]
    x0 = np.concatenate([best_c[:, 0], best_c[:, 1], radii0])
    
    def obj_full(v):
        return -np.sum(v[2*N:])
        
    def cons_full(v):
        C = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        c = []
        # Boundary constraints
        c.append(C[:, 0] - r)
        c.append(1.0 - C[:, 0] - r)
        c.append(C[:, 1] - r)
        c.append(1.0 - C[:, 1] - r)
        # Overlap constraints (squared for smooth gradients)
        diff = C[:, None, :] - C[None, :, :]
        d2 = np.sum(diff**2, axis=2)
        r_sum = r[:, None] + r[None, :]
        mask = np.triu_indices(N, k=1)
        c.append(d2[mask] - r_sum[mask]**2)
        return np.concatenate(c)
        
    bounds_full = [(0,1)]*(2*N) + [(0, 0.5)]*N
    cons = {'type': 'ineq', 'fun': cons_full}
    
    res = minimize(obj_full, x0, method='SLSQP', bounds=bounds_full,
                   constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
    
    if -res.fun > best_s:
        best_c = res.x[:2*N].reshape(N, 2)
        best_s = -res.fun
        
    radii = res.x[2*N:].copy()
    
    # Phase 3: Deterministic repair to strictly satisfy validator tolerances
    for _ in range(30):
        chg = False
        for i in range(N):
            mx = min(best_c[i,0], 1.0-best_c[i,0], best_c[i,1], 1.0-best_c[i,1])
            if radii[i] > mx - 1e-9:
                radii[i] = mx
                chg = True
        for i in range(N):
            for j in range(i+1, N):
                d = math.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                if d < radii[i]+radii[j] - 1e-9:
                    ov = radii[i]+radii[j]-d
                    radii[i] -= ov*0.5
                    radii[j] -= ov*0.5
                    chg = True
        if not chg:
            break
            
    radii = np.maximum(radii, 0.0)
    return best_c, radii, float(np.sum(radii))
