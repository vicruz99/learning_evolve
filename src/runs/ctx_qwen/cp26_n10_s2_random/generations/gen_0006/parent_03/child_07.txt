import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_optimal_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    
    # Upper bounds from boundaries
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    # Pairwise distances
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # LP: max sum(r) s.t. r_i + r_j <= D_ij, 0 <= r_i <= B_i
    c_obj = -np.ones(n)
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds_r = [(0.0, u) for u in ub]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def objective_powell(x):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = x.reshape(N, 2)
    s, _ = compute_optimal_radii(centers)
    return -s

def obj_joint(v):
    return -np.sum(v[2*N:])

def cons_joint(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    c_list = []
    c_list.append(c[:, 0] - r)
    c_list.append(1.0 - c[:, 0] - r)
    c_list.append(c[:, 1] - r)
    c_list.append(1.0 - c[:, 1] - r)
    
    idx_i, idx_j = np.triu_indices(N, 1)
    dists = np.sqrt(np.sum((c[idx_i] - c[idx_j])**2, axis=1))
    c_list.append(dists - (r[idx_i] + r[idx_j]))
    return np.concatenate(c_list)

def generate_starts():
    starts = []
    rng = np.random.default_rng(42)
    
    # Diverse hexagonal row patterns summing to 26
    patterns = [
        [5, 5, 5, 5, 6], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], 
        [4, 6, 6, 6, 4], [5, 5, 5, 6, 5], [6, 4, 6, 5, 5],
        [5, 5, 6, 5, 5], [5, 6, 5, 5, 5]
    ]
    
    for pat in patterns:
        c = []
        r_est = 0.1
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(c) < N:
                    c.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        c = np.array(c[:N])
        starts.append(c)
        
        # Scaled and shifted variants to probe different basin shapes
        for scale in [0.95, 1.05]:
            c_p = c.copy() * scale
            c_p -= c_p.mean(axis=0)
            c_p += 0.5
            c_p += rng.normal(0, 0.01, c_p.shape)
            starts.append(np.clip(c_p, 0.05, 0.95))

    # Random dense starts
    for _ in range(8):
        c = rng.uniform(0.1, 0.9, (N, 2))
        starts.append(c)
        
    # Force-directed repulsion start to spread points evenly
    c = rng.uniform(0.2, 0.8, (N, 2))
    for _ in range(600):
        diff = c[:, None, :] - c[None, :, :]
        dists = np.linalg.norm(diff, axis=2)
        np.fill_diagonal(dists, 1.0)
        forces = np.sum(diff / (dists**2)[:, :, None], axis=1)
        c += 0.001 * forces
        c = np.clip(c, 0.05, 0.95)
    starts.append(c)
    
    return starts

def run_packing():
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    best_sum = 0.0
    best_c = None
    best_r = None
    
    starts = generate_starts()
    
    # Phase 1: Powell optimization on centers using LP for exact radii evaluation
    for c_init in starts:
        c_init = np.clip(c_init, 0.02, 0.98)
        try:
            res = minimize(objective_powell, c_init.flatten(), method='Powell',
                           bounds=bounds_centers, 
                           options={'maxiter': 1500, 'ftol': 1e-13, 'xtol': 1e-13})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_c = res.x.reshape(N, 2)
        except Exception:
            continue
            
    # Phase 2: Joint SLSQP refinement to polish and escape LP combinatorial plateaus
    if best_c is not None:
        _, best_r = compute_optimal_radii(best_c)
        v0 = np.concatenate([best_c.flatten(), best_r])
        
        bounds_joint = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        try:
            res2 = minimize(obj_joint, v0, method='SLSQP',
                            bounds=bounds_joint, constraints={'type': 'ineq', 'fun': cons_joint},
                            options={'maxiter': 5000, 'ftol': 1e-14})
            if np.min(cons_joint(res2.x)) >= -1e-8:
                s = np.sum(res2.x[2*N:])
                if s > best_sum:
                    best_sum = s
                    best_c = res2.x[:2*N].reshape(N, 2)
                    best_r = res2.x[2*N:]
        except Exception:
            pass
            
        # Final LP refinement on best centers
        lp_sum, lp_r = compute_optimal_radii(best_c)
        if lp_sum > best_sum:
            best_sum = lp_sum
            best_r = lp_r
            
    # Phase 3: Strict Numerical Repair to guarantee validation passes
    centers = best_c.copy()
    radii = best_r.copy()
    
    for _ in range(60):
        changed = False
        # Resolve overlaps by symmetric shrinking
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
