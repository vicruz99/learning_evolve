import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_optimal_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    
    # Upper bounds from boundaries
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    # Pairwise distances
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # LP: max sum(r) s.t. r_i + r_j <= D_ij, 0 <= r_i <= B_i
    c_obj = -np.ones(N)
    n_pairs = N * (N - 1) // 2
    A_ub = np.zeros((n_pairs, N))
    b_ub = np.zeros(n_pairs)
    
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds_r = [(0.0, u) for u in ub]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(N)

def objective_packing(centers_flat):
    """Objective: minimize negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    s, _ = get_optimal_radii(centers)
    return -s

def generate_inits(rng):
    """Generates diverse initial center configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 5, 6, 4, 5], 
        [5, 5, 5, 5, 6], [4, 6, 6, 6, 4], [5, 6, 6, 5, 4]
    ]
    
    for pat in patterns:
        c_hex = []
        r_est = 0.095
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(c_hex) < N:
                    c_hex.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        
        c_arr = np.array(c_hex[:N])
        # Add slight noise to break exact symmetry
        c_arr += rng.normal(0, 0.002, c_arr.shape)
        c_arr = np.clip(c_arr, 0.02, 0.98)
        inits.append(c_arr.flatten())
        
    # 2. Force-repulsion spread
    for _ in range(5):
        c_rep = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(600):
            diff = c_rep[:, None, :] - c_rep[None, :, :]
            dists = np.linalg.norm(diff, axis=2)
            np.fill_diagonal(dists, 1.0)
            # Inverse-square repulsion
            forces = np.sum(diff / (dists**2)[:, :, None], axis=1)
            c_rep += 0.0015 * forces
            c_rep = np.clip(c_rep, 0.05, 0.95)
        inits.append(c_rep.flatten())
        
    # 3. Random dense
    for _ in range(8):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)).flatten())
        
    return inits

def get_joint_constraints(v):
    """Computes boundary and non-overlap constraints for SLSQP."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    cons = []
    
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    # Overlap constraints
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    cons.append(np.sqrt(dx**2 + dy**2) - dr)
    
    return np.concatenate(cons)

def run_packing():
    rng = np.random.default_rng(42)
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    bounds_joint = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons_joint = {'type': 'ineq', 'fun': get_joint_constraints}
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # --- Phase 1: Derivative-Free Center Optimization (Powell) ---
    inits = generate_inits(rng)
    
    for v0 in inits:
        # Run Powell to optimize centers only
        try:
            res_pow = minimize(objective_packing, v0, method='Powell', 
                               bounds=bounds_centers,
                               options={'maxiter': 3000, 'ftol': 1e-14, 'xtol': 1e-14})
            curr_sum = -res_pow.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_c = res_pow.x.reshape(N, 2).copy()
        except Exception:
            continue
            
        # Get optimal radii for this center configuration
        lp_sum, lp_r = get_optimal_radii(best_c)
        if lp_sum > best_sum:
            best_sum = lp_sum
            best_r = lp_r.copy()
            
    # --- Phase 2: Basin Hopping / Perturbation Search ---
    if best_c is not None:
        for step in range(20):
            noise = 0.008 * (0.8 ** step)
            c_pert = best_c + rng.normal(0, noise, best_c.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            
            try:
                res_p = minimize(objective_packing, c_pert.flatten(), method='Powell',
                                 bounds=bounds_centers,
                                 options={'maxiter': 2000, 'ftol': 1e-13})
                curr_sum = -res_p.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_c = res_p.x.reshape(N, 2).copy()
                    lp_sum, best_r = get_optimal_radii(best_c)
                    if lp_sum > best_sum:
                        best_sum = lp_sum
            except Exception:
                pass
                
    # --- Phase 3: Joint SLSQP Polish ---
    if best_c is not None and best_r is not None:
        # Initialize joint variables
        v0_joint = np.concatenate([best_c.flatten(), best_r])
        
        # Multiple restarts with slight radius inflation to push constraints
        for _ in range(5):
            v_curr = v0_joint.copy()
            v_curr[2*N:] *= 1.002 # Slightly inflate radii to force optimizer to adjust centers
            v_curr[2*N:] = np.clip(v_curr[2*N:], 0.0, 0.5)
            
            try:
                res_slsqp = minimize(lambda v: -np.sum(v[2*N:]), v_curr, method='SLSQP',
                                     bounds=bounds_joint, constraints=cons_joint,
                                     options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
                
                cand_c = res_slsqp.x[:2*N].reshape(N, 2)
                cand_r = res_slsqp.x[2*N:]
                
                # Verify and clamp
                cand_c = np.clip(cand_c, cand_r[:, None], 1.0 - cand_r[:, None])
                cand_r = np.maximum(cand_r, 0.0)
                
                cand_sum = np.sum(cand_r)
                if cand_sum > best_sum:
                    best_sum = cand_sum
                    best_c = cand_c.copy()
                    best_r = cand_r.copy()
            except Exception:
                pass
                
    # --- Phase 4: Strict Numerical Repair ---
    centers = best_c.copy()
    radii = best_r.copy()
    
    for _ in range(80):
        changed = False
        
        # Resolve overlaps by symmetric shrinking
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
            # Ensure center is valid relative to radius
            centers[i, 0] = np.clip(x, r, 1.0 - r)
            centers[i, 1] = np.clip(y, r, 1.0 - r)
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
