import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def get_lp_and_grad(centers):
    """
    Solves LP to find optimal radii for fixed centers and computes 
    the gradient of the sum of radii w.r.t centers using dual variables.
    """
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    total_con = num_pairs + 4 * n
    
    A_ub = np.zeros((total_con, n))
    b_ub = np.zeros(total_con)
    
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
            
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        A_ub[num_pairs + 4*i, i] = 1.0
        A_ub[num_pairs + 4*i + 1, i] = 1.0
        A_ub[num_pairs + 4*i + 2, i] = 1.0
        A_ub[num_pairs + 4*i + 3, i] = 1.0
        
    # Compute distances and bounds
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        b_ub[num_pairs + 4*i] = centers[i, 0]
        b_ub[num_pairs + 4*i + 1] = 1.0 - centers[i, 0]
        b_ub[num_pairs + 4*i + 2] = centers[i, 1]
        b_ub[num_pairs + 4*i + 3] = 1.0 - centers[i, 1]
        
    # Solve LP: max sum(r) -> min -sum(r)
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0, None)]*n, method='highs')
    
    if not res.success:
        return np.zeros(n), 0.0, np.zeros_like(centers)
        
    radii = res.x
    duals = res.ineqlin.marginals
    obj_val = -res.fun
    
    # Compute gradient using duals
    grad = np.zeros_like(centers)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]
            if lam > 1e-9:
                d = dists[i, j]
                if d > 1e-9:
                    vec = centers[i] - centers[j]
                    f = (lam / d) * vec
                    grad[i] += f
                    grad[j] -= f
            idx += 1
            
    for i in range(n):
        grad[i, 0] += duals[num_pairs + 4*i] - duals[num_pairs + 4*i + 1]
        grad[i, 1] += duals[num_pairs + 4*i + 2] - duals[num_pairs + 4*i + 3]
        
    return radii, obj_val, grad

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    # --- Phase 1: Generate Diverse Initial Configurations ---
    inits = []
    pats = [[5,5,5,5,6], [6,5,5,5,5], [5,6,5,5,5], [5,5,6,5,5], [5,5,5,6,5], [4,5,6,6,5]]
    
    for pat in pats:
        c = []
        y = 0.0
        for r_idx, cnt in enumerate(pat):
            shift = 0.5 if r_idx % 2 == 1 else 0.0
            x = shift
            for _ in range(cnt):
                c.append([x, y])
                x += 1.0
            y += np.sqrt(3)/2
        c = np.array(c[:N])
        # Scale to fit comfortably inside [0,1]^2
        c = (c - c.min(axis=0)) / (c.max(axis=0) - c.min(axis=0)) * 0.85 + 0.075
        inits.append(c)
        
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # --- Phase 2: Gradient Ascent on Centers ---
    for c0 in inits:
        c = c0.copy()
        step = 0.03
        for it in range(1500):
            r, s_val, g = get_lp_and_grad(c)
            if s_val > best_sum:
                best_sum = s_val
                best_c = c.copy()
                best_r = r.copy()
                
            # Projected gradient ascent
            c += step * g
            c = np.clip(c, 1e-4, 1.0 - 1e-4)
            
            if it % 100 == 0:
                step *= 0.93
                
        # Perturb best local to probe neighborhood
        for _ in range(4):
            cp = best_c + rng.normal(0, 0.008, best_c.shape)
            cp = np.clip(cp, 0.05, 0.95)
            step = 0.015
            for it in range(400):
                r, s_val, g = get_lp_and_grad(cp)
                if s_val > best_sum:
                    best_sum = s_val
                    best_c = cp.copy()
                    best_r = r.copy()
                cp += step * g
                cp = np.clip(cp, 1e-4, 1.0 - 1e-4)
                if it % 50 == 0:
                    step *= 0.94
                    
    # --- Phase 3: SLSQP Joint Optimization Polish ---
    v0 = np.concatenate([best_c.flatten(), best_r])
    
    def obj(v):
        return -np.sum(v[2*N:])

    def cons(v):
        cx = v[:2*N].reshape(N, 2)
        cr = v[2*N:]
        c = [cx[:,0]-cr, 1.0-cx[:,0]-cr, cx[:,1]-cr, 1.0-cx[:,1]-cr]
        ii, jj = np.triu_indices(N, 1)
        c.append(np.linalg.norm(cx[ii]-cx[jj], axis=1) - (cr[ii]+cr[jj]))
        return np.concatenate(c)

    bnds = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bnds, 
                       constraints={'type':'ineq', 'fun':cons},
                       options={'maxiter': 3000, 'ftol': 1e-13})
        if -res.fun > best_sum:
            best_c = res.x[:2*N].reshape(N, 2)
            best_r = res.x[2*N:]
            best_sum = -res.fun
    except Exception:
        pass
        
    # Final LP to extract exact optimal radii for polished centers
    _, best_sum, _ = get_lp_and_grad(best_c)
    best_r, _, _ = get_lp_and_grad(best_c)
    
    # --- Phase 4: Strict Numerical Repair ---
    centers = best_c.copy()
    radii = best_r.copy()
    for _ in range(60):
        changed = False
        # Resolve overlaps symmetrically
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i]-centers[j])
                if d < radii[i]+radii[j] - 1e-12:
                    shrink = (radii[i]+radii[j] - d)/2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
