import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def lp_max_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    
    # Upper bounds from boundaries
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    # Pairwise distances
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # LP: max sum(r) s.t. r_i + r_j <= D_ij, 0 <= r_i <= B_i
    c_obj = -np.ones(n)
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds_r = [(0.0, u) for u in ub]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def obj_centers_powell(x_flat):
    """Objective for center optimization: negative LP sum of radii."""
    centers = x_flat.reshape(N, 2)
    centers = np.clip(centers, 0.02, 0.98)
    val, _ = lp_max_radii(centers)
    if val < 1e-6:
        return 100.0
    return -val

def get_joint_constraints(vars_flat):
    """Computes boundary and non-overlap constraints (must be >= 0)."""
    c = vars_flat[:2*N].reshape(N, 2)
    r = vars_flat[2*N:]
    cons = []
    
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    # Overlap constraints (squared for better conditioning): dist^2 >= (r_i + r_j)^2
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    cons.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(cons)

def obj_joint(vars_flat):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2*N:])

def generate_inits(rng):
    """Generates diverse initial configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns with varying row structures
    patterns = [
        [5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], 
        [5,5,6,5,5], [6,4,6,5,5], [5,4,6,6,5], [4,5,6,5,6],
        [7,5,5,5,4], [5,5,5,5,5,1], [6,6,6,6,2], [5,5,5,6,5],
        [5,7,5,6], [6,6,5,6], [4,7,7,5], [6,5,5,5,5]
    ]
    for pat in patterns:
        c = []
        r_est = 0.098
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(c) < N:
                    c.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        inits.append(np.array(c[:N]) + rng.normal(0, 0.004, (N, 2)))
        
    # 2. Corner-focused placements
    for _ in range(5):
        c = []
        for cx, cy in [(0.11, 0.11), (0.89, 0.11), (0.11, 0.89), (0.89, 0.89)]:
            c.append([cx, cy])
        y = 0.15
        row = 0
        while len(c) < N:
            shift = 0.10 if row % 2 == 1 else 0.0
            x = 0.12 + shift
            while x <= 0.88 and len(c) < N:
                c.append([x, y])
                x += 0.20
            y += 0.17
            row += 1
        inits.append(np.array(c[:N]) + rng.normal(0, 0.005, (N, 2)))
        
    # 3. Force-relaxed random distributions
    for _ in range(5):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(600):
            forces = np.zeros_like(c)
            diff = c[:, None, :] - c[None, :, :]
            dists = np.linalg.norm(diff, axis=2)
            np.fill_diagonal(dists, 1.0)
            f_mag = 1.0 / (dists**2 + 0.05)
            forces += np.sum(diff * f_mag[:, :, None], axis=1)
            c += 0.0015 * forces
            c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    return inits

def run_packing():
    rng = np.random.default_rng(42)
    best_c = None
    best_r = np.zeros(N)
    best_sum = -1.0
    
    inits = generate_inits(rng)
    center_bounds = [(0.05, 0.95)] * (2 * N)
    
    # --- Phase 1: Powell optimization on centers only ---
    for c_init in inits:
        c_init = np.clip(c_init, 0.06, 0.94)
        try:
            res = minimize(obj_centers_powell, c_init.flatten(), method='Powell',
                           bounds=center_bounds, options={'maxiter': 5000, 'ftol': 1e-12})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_c = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # --- Phase 2: Perturbation & Basin Hopping ---
    if best_c is not None:
        for step in range(12):
            noise = 0.007 * (0.82 ** step)
            c_pert = best_c.copy() + rng.normal(0, noise, best_c.shape)
            c_pert = np.clip(c_pert, 0.06, 0.94)
            try:
                res = minimize(obj_centers_powell, c_pert.flatten(), method='Powell',
                               bounds=center_bounds, options={'maxiter': 3000, 'ftol': 1e-12})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_c = res.x.reshape(N, 2)
            except Exception:
                pass
                
    # Compute radii for best centers found so far
    if best_c is not None:
        _, best_r = lp_max_radii(best_c)
        
    # --- Phase 3: Joint SLSQP refinement ---
    if best_c is not None:
        v0 = np.concatenate([best_c.flatten(), best_r])
        bounds_opt = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        cons_dict = {'type': 'ineq', 'fun': get_joint_constraints}
        
        for _ in range(6):
            v_curr = v0.copy()
            v_curr[:2*N] += rng.normal(0, 0.0015, 2*N)
            v_curr[:2*N] = np.clip(v_curr[:2*N], 0.01, 0.99)
            v_curr[2*N:] *= 0.9995
            
            try:
                res = minimize(obj_joint, v_curr, method='SLSQP', bounds=bounds_opt,
                              constraints=cons_dict, options={'maxiter': 12000, 'ftol': 1e-14})
                if np.min(get_joint_constraints(res.x)) >= -1e-6:
                    s = np.sum(res.x[2*N:])
                    if s > best_sum:
                        best_sum = s
                        best_c = res.x[:2*N].reshape(N, 2)
                        best_r = res.x[2*N:]
            except Exception:
                pass
                
        # Final LP polish to ensure radii are exactly optimal for centers
        lp_sum, lp_r = lp_max_radii(best_c)
        if lp_sum > best_sum:
            best_sum = lp_sum
            best_r = lp_r
            
    # --- Phase 4: Strict Numerical Repair ---
    centers = best_c.copy()
    radii = best_r.copy()
    
    for _ in range(80):
        changed = False
        # Resolve overlaps by symmetric shrinking
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
