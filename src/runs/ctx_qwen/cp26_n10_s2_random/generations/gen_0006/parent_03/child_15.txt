Our strategy combines three powerful techniques:
1. **LP-based Gradient Ascent on Centers**: For a fixed set of centers, we solve an LP to maximize $\sum r_i$ subject to non-overlap and boundary constraints. The dual variables (Lagrange multipliers) from the LP directly give the gradient of the sum of radii with respect to the centers. We perform gradient ascent on the centers to continuously improve the packing.
2. **Diverse Multi-Start Initialization**: We generate multiple initial configurations using hexagonal lattice patterns with varying row structures, along with perturbed versions and random dense starts. This helps escape poor local minima that plague greedy packings.
3. **SLSQP Polishing & Numerical Repair**: After the gradient ascent converges to a strong local optimum, we use `scipy.optimize.minimize` (SLSQP) to jointly refine centers and radii. Finally, a strict repair pass guarantees all constraints are satisfied within the validator's tolerance.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def solve_lp_and_grad(centers):
    """
    Solves LP to find optimal radii for fixed centers and computes the 
    gradient of the sum of radii with respect to the centers using dual variables.
    """
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    total_cons = num_pairs + num_bound
    
    c_obj = -np.ones(n)
    A_ub = np.zeros((total_cons, n))
    b_ub = np.zeros(total_cons)
    
    idx = 0
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return 0.0, np.zeros(n), np.zeros_like(centers)
            
        radii = res.x
        sum_r = -res.fun
        duals = res.ineqlin.marginals
        
        # Compute gradient of sum(r) w.r.t centers using dual variables
        grad = np.zeros_like(centers)
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                lam = duals[idx]
                if lam > 1e-9:
                    d = dists[i, j]
                    if d > 1e-9:
                        vec = (centers[i] - centers[j]) / d
                        grad[i] += lam * vec
                        grad[j] -= lam * vec
                idx += 1
                
        for i in range(n):
            mu_xL = duals[num_pairs + 4*i]
            mu_xR = duals[num_pairs + 4*i + 1]
            mu_yL = duals[num_pairs + 4*i + 2]
            mu_yR = duals[num_pairs + 4*i + 3]
            grad[i, 0] += mu_xL - mu_xR
            grad[i, 1] += mu_yL - mu_yR
            
        return sum_r, radii, grad
    except Exception:
        return 0.0, np.zeros(n), np.zeros_like(centers)

def generate_hex_init(pattern, noise=0.0, rng=None):
    """Generates a hexagonal lattice configuration based on row lengths."""
    pts = []
    r_est = 0.095
    y = r_est
    for r_idx, cnt in enumerate(pattern):
        shift = r_est if r_idx % 2 == 1 else 0.0
        x = r_est + shift
        for _ in range(cnt):
            if len(pts) < N:
                pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3)
    c = np.array(pts[:N])
    if noise > 0 and rng is not None:
        c += rng.normal(0, noise, c.shape)
    c = np.clip(c, 0.05, 0.95)
    return c

def run_packing():
    rng = np.random.default_rng(42)
    
    patterns = [
        [5, 5, 5, 5, 6], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4],
        [4, 6, 6, 6, 4], [5, 5, 5, 6, 5], [6, 4, 6, 5, 5]
    ]
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    starts = []
    for pat in patterns:
        for ns in [0.0, 0.005, 0.01, 0.02]:
            starts.append(generate_hex_init(pat, noise=ns, rng=rng))
    for _ in range(15):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # Phase 1: LP-based Gradient Ascent on Centers
    for c_init in starts:
        c = c_init.copy()
        
        for it in range(1500):
            s, r, grad = solve_lp_and_grad(c)
            if s > best_sum:
                best_sum = s
                best_c = c.copy()
                best_r = r.copy()
                
            # Adaptive step size that decays over iterations
            step = 0.008 / (1.0 + 0.003 * it)
            c += step * grad
            
            # Periodic random perturbation to escape local minima
            if it % 150 == 0 and it > 0:
                c += rng.normal(0, 0.002, c.shape)
                
            # Keep centers strictly inside the square to avoid LP degeneracies
            c = np.clip(c, 1e-4, 1.0 - 1e-4)

    # Phase 2: SLSQP Joint Polish
    if best_c is not None:
        def obj_sl(x):
            return -np.sum(x[2*N:])
            
        def cons_sl(x):
            c = x[:2*N].reshape(N, 2)
            r = x[2*N:]
            con = []
            con.append(c[:, 0] - r)
            con.append(1.0 - c[:, 0] - r)
            con.append(c[:, 1] - r)
            con.append(1.0 - c[:, 1] - r)
            idx_i, idx_j = np.triu_indices(N, 1)
            dx = c[idx_i, 0] - c[idx_j, 0]
            dy = c[idx_i, 1] - c[idx_j, 1]
            dr = r[idx_i] + r[idx_j]
            con.append(np.sqrt(dx**2 + dy**2) - dr)
            return np.concatenate(con)
            
        bounds_sl = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        v0 = np.concatenate([best_c.flatten(), best_r])
        
        try:
            res = minimize(obj_sl, v0, method='SLSQP', bounds=bounds_sl,
                          constraints={'type': 'ineq', 'fun': cons_sl},
                          options={'maxiter': 5000, 'ftol': 1e-13})
            if res.success:
                c_pol = res.x[:2*N].reshape(N, 2)
                r_pol = res.x[2*N:]
                s_pol = np.sum(r_pol)
                if s_pol > best_sum:
                    best_sum = s_pol
                    best_c = c_pol
                    best_r = r_pol
        except Exception:
            pass

    # Phase 3: Strict Numerical Repair
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(best_c[i] - best_c[j])
                req = best_r[i] + best_r[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    best_r[i] -= shrink
                    best_r[j] -= shrink
                    changed = True
        for i in range(N):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            mx = min(x, 1.0-x, y, 1.0-y)
            if r > mx + 1e-12:
                best_r[i] = mx
                changed = True
        if not changed:
            break
            
    best_r = np.maximum(best_r, 0.0)
    return best_c, best_r, float(np.sum(best_r))
