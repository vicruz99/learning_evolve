Strategy:
1. **LP-based Objective for Centers**: Instead of optimizing centers and radii jointly from the start, we treat radii as implicit variables solved exactly via Linear Programming for any fixed set of centers. This transforms the problem into a smooth-ish optimization over 52 center coordinates, which is more tractable.
2. **Diverse Structured Initializations**: Generate multiple high-quality starting configurations based on hexagonal lattices with varying row distributions, rotations, and corner-hugging patterns. This covers the geometry of known optimal packings.
3. **Derivative-Free Local Search**: Use `scipy.optimize.minimize` with the `Powell` method on the center coordinates. Powell handles the non-smooth nature of the LP objective better than gradient methods and respects bounds.
4. **Iterative Perturbation (Basin Hopping)**: Systematically perturb the best found configuration with decaying noise and re-optimize to escape local minima.
5. **Joint SLSQP Polish**: After finding excellent centers, run a final joint optimization of centers and radii using `SLSQP` to perfectly satisfy boundary and pairwise constraints while squeezing out marginal gains.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I_IDX, J_IDX = np.triu_indices(N, k=1)
n_pairs = len(I_IDX)

# Precompute LP constraint matrix structure: r_i + r_j <= dist_ij
A_ub = np.zeros((n_pairs, N))
for k in range(n_pairs):
    A_ub[k, I_IDX[k]] = 1.0
    A_ub[k, J_IDX[k]] = 1.0

def compute_lp(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    c = np.clip(centers, 1e-8, 1.0 - 1e-8)
    
    # Upper bounds from square boundaries
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    b_ub = dists[I_IDX, J_IDX]
    
    # Solve: max sum(r) s.t. r_i + r_j <= D_ij, 0 <= r_i <= B_i
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def obj_centers(x):
    """Objective for center optimization: minimize negative sum of LP-optimal radii."""
    c = x.reshape(N, 2)
    if np.any(c < 0) or np.any(c > 1):
        return 1e6
    s, _ = compute_lp(c)
    return -s

def cons_joint(v):
    """Inequality constraints for joint center/radius optimization (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ]
    dx = c[I_IDX, 0] - c[J_IDX, 0]
    dy = c[I_IDX, 1] - c[J_IDX, 1]
    dr = r[I_IDX] + r[J_IDX]
    con.append(np.sqrt(dx**2 + dy**2) - dr)
    return np.concatenate(con)

def objective_joint(v):
    """Objective for joint optimization: maximize sum of radii."""
    return -np.sum(v[2*N:])

def run_packing():
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # --- Phase 1: Generate Diverse Structured Starts ---
    starts = []
    
    # Hexagonal lattice patterns with different row distributions
    for pat in [[5,4,5,4,5,3], [5,5,5,5,6], [6,5,6,5,4], [4,6,6,6,4], [5,5,6,5,5], [6,4,6,5,5]]:
        c_hex = []
        r0 = 0.10
        y = r0
        for ri, cnt in enumerate(pat):
            shift = r0 if ri % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(c_hex) < N:
                    c_hex.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        starts.append(np.array(c_hex[:N]))
        
    # Rotated hex grids to explore different alignments
    for angle in [0.15, -0.15, 0.3, -0.3]:
        c_rot = starts[0].copy() - 0.5
        c_rot = c_rot @ [[np.cos(angle), -np.sin(angle)], [np.sin(angle), np.cos(angle)]]
        c_rot += 0.5
        starts.append(c_rot)
        
    # Corner-hugging + interior grid
    c_corner = np.zeros((N, 2))
    c_corner[:4] = [[0.06, 0.06], [0.94, 0.06], [0.06, 0.94], [0.94, 0.94]]
    gs = np.linspace(0.22, 0.78, 5)
    gx, gy = np.meshgrid(gs, gs)
    c_corner[4:] = np.column_stack([gx.flatten(), gy.flatten()])[:22]
    starts.append(c_corner)
    
    # Random dense starts
    for _ in range(10):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    bounds_c = [(0.0, 1.0)] * (2 * N)
    
    # --- Phase 2: Local Optimization on Centers ---
    for c_init in starts:
        for _ in range(3):
            c_curr = c_init + rng.normal(0, 0.008, c_init.shape)
            c_curr = np.clip(c_curr, 0.02, 0.98)
            
            res = minimize(obj_centers, c_curr.flatten(), method='Powell', bounds=bounds_c,
                           options={'maxiter': 1500, 'ftol': 1e-13, 'xtol': 1e-13})
            
            s, r = compute_lp(res.x.reshape(N, 2))
            if s > best_sum:
                best_sum = s
                best_c = res.x.reshape(N, 2).copy()
                best_r = r.copy()
                
    # --- Phase 3: Iterative Perturbation Search ---
    for step in range(40):
        noise = 0.006 * (0.88 ** step)
        c_pert = best_c + rng.normal(0, noise, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        
        res = minimize(obj_centers, c_pert.flatten(), method='Powell', bounds=bounds_c,
                       options={'maxiter': 1200, 'ftol': 1e-13})
                       
        s, r = compute_lp(res.x.reshape(N, 2))
        if s > best_sum:
            best_sum = s
            best_c = res.x.reshape(N, 2).copy()
            best_r = r.copy()
            
    # --- Phase 4: Joint SLSQP Polish ---
    v0 = np.concatenate([best_c.flatten(), best_r])
    bounds_opt = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    res_f = minimize(objective_joint, v0, method='SLSQP', bounds=bounds_opt,
                     constraints={'type': 'ineq', 'fun': cons_joint},
                     options={'maxiter': 8000, 'ftol': 1e-14})
                     
    best_c = res_f.x[:2*N].reshape(N, 2)
    best_r = res_f.x[2*N:]
    
    # --- Phase 5: Strict Numerical Repair ---
    for _ in range(100):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(best_c[i] - best_c[j])
                if d < best_r[i] + best_r[j] - 1e-12:
                    shr = (best_r[i] + best_r[j] - d) / 2.0 + 1e-9
                    best_r[i] -= shr
                    best_r[j] -= shr
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            mr = min(best_c[i, 0], 1.0 - best_c[i, 0], best_c[i, 1], 1.0 - best_c[i, 1])
            if best_r[i] > mr + 1e-12:
                best_r[i] = mr
                changed = True
                
        if not changed:
            break
            
    best_r = np.maximum(best_r, 0.0)
    return best_c, best_r, float(np.sum(best_r))
