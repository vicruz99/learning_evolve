3. **Efficient Optimization**: We iteratively solve the LP, compute the gradient from duals, update centers via gradient ascent with decaying step size, and periodically perturb to escape local minima. import numpy as np
from scipy.optimize import linprog

N = 26

def solve_lp_and_gradient(centers):
    """
    Solves the LP for optimal radii given fixed centers and computes the 
    gradient of the sum of radii with respect to center positions using dual variables.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_pairs = n * (n - 1) // 2
    
    # Constraint matrix and bounds
    A_ub = np.zeros((num_pairs + 4 * n, n))
    b_ub = np.zeros(num_pairs + 4 * n)
    
    idx = 0
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    # Pairwise non-overlap constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)]*n, method='highs')
    
    if not res.success:
        return None, 0.0, None, None, None
        
    radii = res.x
    curr_sum = np.sum(radii)
    
    # Extract dual variables (shadow prices)
    duals = np.zeros(num_pairs + 4 * n)
    if hasattr(res, 'ineqlin') and res.ineqlin is not None:
        marg = getattr(res.ineqlin, 'marginals', None)
        if marg is not None:
            # Marginals for minimization are typically <= 0 for active <= constraints.
            # We flip sign to get positive forces for expansion.
            duals = -np.array(marg, dtype=float)
            duals = np.maximum(duals, 0.0)
            
    # Compute gradient using duals
    grad = np.zeros_like(centers)
    p_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[p_idx]
            d = dists[i, j]
            if lam > 1e-8 and d > 1e-8:
                vec = centers[i] - centers[j]
                force = (lam / d) * vec
                grad[i] += force
                grad[j] -= force
            p_idx += 1
            
    b_start = num_pairs
    for i in range(n):
        mu_x = duals[b_start + 4*i]
        mu_1x = duals[b_start + 4*i + 1]
        mu_y = duals[b_start + 4*i + 2]
        mu_1y = duals[b_start + 4*i + 3]
        grad[i, 0] += mu_x - mu_1x
        grad[i, 1] += mu_y - mu_1y
        
    return radii, curr_sum, grad, dists, duals

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses LP-driven gradient ascent with multiple restarts.
    """
    rng = np.random.default_rng(42)
    n = N
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Generate diverse initial configurations
    inits = []
    patterns = [
        [5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], 
        [4,6,6,6,4], [5,5,5,6,5], [6,6,4,5,5]
    ]
    for pat in patterns:
        c = []
        r_est = 0.095
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(c) < n:
                    c.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        inits.append(np.array(c[:n]))
        
    # Add random starts
    for _ in range(10):
        c = rng.uniform(0.1, 0.9, (n, 2))
        inits.append(c)
        
    # Gradient ascent optimization over all starts
    for init_c in inits:
        centers = init_c.copy()
        alpha = 0.025
        
        for step in range(2500):
            radii, curr_sum, grad, dists, _ = solve_lp_and_gradient(centers)
            if radii is None:
                break
                
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                
            # Adaptive decaying step size
            step_size = alpha / (1.0 + 0.002 * step)
            centers += step_size * grad
            centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
            
            # Periodic perturbation to escape local minima
            if step % 120 == 0 and step > 0:
                centers += rng.normal(0, 0.002, centers.shape)
                centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
                
    # Fallback if optimization failed unexpectedly
    if best_centers is None:
        best_centers = inits[0]
        best_radii, _, _, _, _ = solve_lp_and_gradient(best_centers)
        
    # Strict numerical repair to guarantee validator passes
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for _ in range(100):
        changed = False
        # Resolve overlaps symmetrically
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Clamp to boundaries
        for i in range(n):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
