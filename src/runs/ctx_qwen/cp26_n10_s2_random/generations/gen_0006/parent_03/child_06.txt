import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute constant structure of the LP constraint matrix
def build_lp_matrix():
    n = N
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs + n, n))
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
    for i in range(n):
        A_ub[idx, i] = 1.0
        idx += 1
    return A_ub

A_LP = build_lp_matrix()
c_obj_LP = -np.ones(N)
bounds_LP = [(0.0, None)] * N

def compute_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    
    # Boundary upper bounds
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    
    # Pairwise distances
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Construct b_ub dynamically
    b_ub = np.zeros(N*(N-1)//2 + N)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            b_ub[idx] = dists[i, j]
            idx += 1
    for i in range(N):
        b_ub[idx] = ub[i]
        idx += 1
        
    res = linprog(c_obj_LP, A_ub=A_LP, b_ub=b_ub, bounds=bounds_LP, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def obj_centers(x):
    """Objective for center optimization: minimize negative sum of LP radii."""
    c = np.clip(x.reshape(N, 2), 0.01, 0.99)
    return -compute_lp_radii(c)[0]

def run_packing():
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # --- Phase 1: Generate Diverse Initial Configurations ---
    starts = []
    
    # Hexagonal lattice patterns with varying row distributions
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [5,5,5,5,6], [4,6,6,6,4], [6,6,5,5,4], [5,5,6,5,5]]
    for pat in patterns:
        c = []
        y = 0.075
        r_est = 0.092
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = 0.075 + shift
            for _ in range(cnt):
                if len(c) < N:
                    c.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        # Add slight noise to break exact symmetries
        starts.append(np.array(c[:N]) + rng.normal(0, 0.002, (N, 2)))
        
    # Force-repulsed random starts
    for _ in range(10):
        c = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(400):
            diff = c[:, None, :] - c[None, :, :]
            dists = np.linalg.norm(diff, axis=2)
            np.fill_diagonal(dists, 1.0)
            forces = np.sum(diff / (dists**2)[:, :, None], axis=1)
            c += 0.002 * forces
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    # --- Phase 2: Local Optimization via Powell ---
    for c0 in starts:
        try:
            res = minimize(obj_centers, c0.flatten(), method='Powell',
                           options={'maxiter': 2000, 'xtol': 1e-10, 'ftol': 1e-12})
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_c = res.x.reshape(N, 2)
                _, best_r = compute_lp_radii(best_c)
        except Exception:
            pass
            
    # --- Phase 3: Basin Hopping / Simulated Annealing Refinement ---
    if best_c is not None:
        curr_c = best_c.copy()
        curr_sum = best_sum
        step = 0.02
        
        for _ in range(100):
            # Annealing step size
            c_pert = curr_c + rng.normal(0, step, curr_c.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            
            s_pert, _ = compute_lp_radii(c_pert)
            
            # Accept if better, or with probability based on temperature

                # Local polish around accepted point

    centers = best_c.copy()

        for i in range(N):

                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
