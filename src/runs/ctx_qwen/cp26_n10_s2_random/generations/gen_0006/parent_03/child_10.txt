

<strategy>
The core challenge in maximizing the sum of radii for 26 circles in a unit square is that the problem is highly non-convex with complex interaction constraints. Direct non-linear optimizers like SLSQP often get trapped in local minima or struggle with the non-smooth `min` and `sqrt` functions in constraints.

Instead, we exploit the mathematical structure: for **fixed centers**, finding the optimal radii is a **Linear Programming (LP)** problem. More importantly, the dual variables (shadow prices) from this LP give us the exact gradient of the maximum achievable sum of radii with respect to the centers. 

**Strategy:**
1. **LP-Dual Gradient Ascent:** We iterate between solving an LP for radii and updating centers using the gradient derived from LP duals.
   - Primal LP: Maximize $\sum r_i$ subject to $r_i + r_j \le \|c_i - c_j\|$ and $0 \le r_i \le \text{dist}(c_i, \text{boundary})$.
   - Dual variables $\lambda_{ij}$ indicate how much the objective increases if we increase the distance between $i$ and $j$.
   - Gradient for $c_i$ is $\sum_{j} \lambda_{ij} \frac{c_i - c_j}{\|c_i - c_j\|}$ plus boundary terms. This naturally pushes overlapping or tight circles apart and pulls boundary-hugging circles inward.
2. **Projected Gradient Descent:** We update centers along this gradient, project back to $[0,1]^2$, and use a decaying step size schedule with periodic random noise to escape local optima.
3. **Diverse Initializations:** We seed the process with hexagonal lattices (varying densities), random clouds, and perturbed versions to cover different packing topologies.
4. **Strict Repair:** A final deterministic pass ensures all constraints are met within the $10^{-12}$ tolerance of the validator by symmetrically shrinking overlapping radii and clamping boundary violations.

This approach is significantly more stable and efficient than joint non-linear optimization, converging to high-quality packings reliably.
</strategy>

```python
import numpy as np
from scipy.optimize import linprog

N = 26
M_PAIRS = N * (N - 1) // 2
M_BOUND = 4 * N
M_TOTAL = M_PAIRS + M_BOUND

# Precompute constant parts of the LP to avoid reconstruction overhead
A_ub_pairs = np.zeros((M_PAIRS, N))
idx_pairs = []
b_idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub_pairs[b_idx, i] = 1.0
        A_ub_pairs[b_idx, j] = 1.0
        idx_pairs.append((i, j))
        b_idx += 1

A_ub_bound = np.zeros((M_BOUND, N))
for i in range(N):
    A_ub_bound[4*i, i] = 1.0
    A_ub_bound[4*i+1, i] = 1.0
    A_ub_bound[4*i+2, i] = 1.0
    A_ub_bound[4*i+3, i] = 1.0

A_ub = np.vstack([A_ub_pairs, A_ub_bound])
c_obj = -np.ones(N)
bounds_r = [(0.0, None)] * N
triu_idx = np.triu_indices(N, k=1)

def solve_lp_and_grad(centers):
    """
    Solves the LP for optimal radii given fixed centers, and computes the 
    gradient of the max sum of radii w.r.t. centers using dual variables.
    """
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.empty(M_TOTAL)
    b_ub[:M_PAIRS] = dists[triu_idx]
    for i in range(N):
        b_ub[M_PAIRS + 4*i] = centers[i, 0]
        b_ub[M_PAIRS + 4*i + 1] = 1.0 - centers[i, 0]
        b_ub[M_PAIRS + 4*i + 2] = centers[i, 1]
        b_ub[M_PAIRS + 4*i + 3] = 1.0 - centers[i, 1]
        
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if not res.success:
        return None, None, None
        
    radii = res.x
    sum_r = -res.fun
    marginals = res.ineqlin.marginals
    
    grad = np.zeros((N, 2))
    
    # Pairwise gradients from contact forces
    for k, (i, j) in enumerate(idx_pairs):
        lam = max(marginals[k], 0.0)
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-8:
                vec = centers[i] - centers[j]
                f = lam / d
                grad[i] += f * vec
                grad[j] -= f * vec
                
    # Boundary gradients from wall forces
    for i in range(N):
        base = M_PAIRS + 4 * i
        mu_L = max(marginals[base], 0.0)
        mu_R = max(marginals[base + 1], 0.0)
        mu_B = max(marginals[base + 2], 0.0)
        mu_T = max(marginals[base + 3], 0.0)
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, sum_r, grad

def run_packing():
    rng = np.random.default_rng(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    configs = []
    # 1. Hexagonal lattice starts with varying densities
    for r0 in [0.09, 0.10, 0.11, 0.12]:
        c = []
        y = r0
        row = 0
        while len(c) < N:
            x = r0 if row % 2 == 0 else 2 * r0
            while x + r0 <= 1.0 and len(c) < N:
                c.append([x, y])
                x += 2 * r0
            y += r0 * np.sqrt(3)
            row += 1
        configs.append(np.array(c[:N]))
        
    # 2. Random dense starts
    for _ in range(10):
        configs.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # 3. Perturbed lattice starts to break symmetries
    for c_hex in configs[:4]:
        c_pert = c_hex + rng.normal(0, 0.008, (N, 2))
        c_pert = np.clip(c_pert, 0.02, 0.98)
        configs.append(c_pert)

    # Main optimization loop
    for init_c in configs:
        centers = init_c.copy()
        step_size = 0.02
        for it in range(2500):
            radii, curr_sum, grad = solve_lp_and_grad(centers)
            if radii is None:
                break
                
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                
            # Projected gradient ascent
            centers += step_size * grad
            centers = np.clip(centers, 0.0, 1.0)
            step_size *= 0.9995
            
            # Periodic noise injection to escape local minima
            if it % 400 == 399:
                centers += rng.normal(0, 0.004, (N, 2))
                centers = np.clip(centers, 0.05, 0.95)
                step_size = max(step_size, 0.003)

    # High-precision polishing phase
    centers = best_centers
    for _ in range(1500):
        radii, curr_sum, grad = solve_lp_and_grad(centers)
        if radii is None: break
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
        centers += 0.0005 * grad
        centers = np.clip(centers, 0.0, 1.0)
        
    # Strict numerical repair to guarantee validator tolerance
    centers = best_centers
    radii = best_radii
    
    for _ in range(40):
        changed = False
        # Clamp radii to boundary distances
        for i in range(N):
            mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mx + 1e-12:
                radii[i] = mx
                changed = True
        # Resolve pairwise overlaps symmetrically
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-12:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
```
