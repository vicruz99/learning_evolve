from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure for speed
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
lp_idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[lp_idx, i] = 1.0
        A_LP[lp_idx, j] = 1.0
        PAIR_IDX.append((i, j))
        lp_idx += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):

    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),

        b[k] = dists[i, j]
        k += 1

        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1

        res = None

    return radii, s_sum, grad

def lp_wrapper(x_flat):
    """Wrapper for scipy minimization: returns objective and gradient."""
    c = np.clip(x_flat.reshape(N, 2), 1e-6, 1.0 - 1e-6)
    _, val, grad = solve_lp_and_grad(c)
    return -val, -grad.flatten()

def lp_obj_only(x_flat):
    """Objective only wrapper for derivative-free optimizers."""
    c = np.clip(x_flat.reshape(N, 2), 1e-6, 1.0 - 1e-6)
    _, val, _ = solve_lp_and_grad(c)
    return -val

def force_sim_init(rng, n_iter=2000):
    """Generates a well-spaced configuration via repulsive force growth and boundary attraction."""
    c = rng.uniform(0.15, 0.85, (N, 2))
    r = np.full(N, 0.025)
    for _ in range(n_iter):
        r *= 1.0004

def hex_init(r_est, rng, pattern=None):
    """Generates hexagonal lattice initialization."""
    if pattern is None:
        pattern = [5, 6, 5, 6, 4]
    c = []
    y = r_est

                c.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    c = np.array(c[:N])
    c += rng.normal(0, 0.004, c.shape)
    return np.clip(c, 0.02, 0.98)

def slsqp_obj(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def slsqp_cons(v):
    """Constraints for joint SLSQP: boundary and non-overlap."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    dx = c[TRIU_I, 0] - c[TRIU_J, 0]
    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def run_slsqp(c0, r0):
    """Joint SLSQP optimization of centers and radii."""
    v0 = np.concatenate([c0.flatten(), r0])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
        c_val = slsqp_cons(res.x)
        if np.min(c_val) >= -1e-7:
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:], -res.fun
    except Exception:
        pass
    return c0, r0, np.sum(r0)

def repair(centers, radii):

                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True

            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
