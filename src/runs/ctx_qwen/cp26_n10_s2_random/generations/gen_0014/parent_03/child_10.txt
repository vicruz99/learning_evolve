import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute constant LP constraint matrix structure globally
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii given fixed centers and computes exact subgradient via duals."""
    centers = np.clip(centers, 1e-9, 1.0 - 1e-9)
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-16)
    
    b = np.zeros(NUM_PAIRS + 4 * N)
    k = 0
    for i, j in PAIR_IDX:
        b[k] = dists[i, j]
        k += 1
    for i in range(N):
        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(centers)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    duals = np.zeros(len(b))
    if hasattr(res, 'marginals') and res.marginals is not None:
        duals = np.maximum(res.marginals.ineqlin, 0.0)
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = np.maximum(res.ineqlin.marginals, 0.0)
        
    grad = np.zeros_like(centers)
    k = 0
    for i, j in PAIR_IDX:
        mu = duals[k]
        if mu > 1e-10:
            d = dists[i, j]
            if d > 1e-10:
                vec = (centers[i] - centers[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
        k += 1
        
    b_start = NUM_PAIRS
    for i in range(N):
        grad[i, 0] += duals[b_start + 4*i] - duals[b_start + 4*i + 1]
        grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
        
    return radii, s_sum, grad

def obj_lp_centers(x):
    """Objective and gradient for L-BFGS-B: minimizes negative sum of radii."""
    centers = x.reshape(N, 2)
    _, s, g = solve_lp_and_grad(centers)
    return -s, -g.flatten()

def constraints_joint(v):
    """Computes boundary and non-overlap constraints for joint SLSQP optimization."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    dx = c[TRIU_I, 0] - c[TRIU_J, 0]
    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def objective_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def grow_circles_init(rng, growth_rate=0.00035, steps=2500):
    """Generates a dense configuration via repulsive force growth simulation."""
    c = rng.uniform(0.15, 0.85, (N, 2))
    r = np.full(N, 0.015)
    dt = 0.04
    
    # Pre-allocate arrays for vectorized forces
    for _ in range(steps):
        r += growth_rate
        
        # Vectorized pairwise forces
        diff = c[:, None, :] - c[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dist, 1e9)
        
        overlap = np.maximum(r[:, None] + r[None, :] - dist, 0.0)
        push = overlap / (dist + 1e-6)
        dir_x = diff[:, :, 0] / (dist + 1e-6)
        dir_y = diff[:, :, 1] / (dist + 1e-6)
        
        fx = np.sum(push * dir_x, axis=1)
        fy = np.sum(push * dir_y, axis=1)
        
        # Boundary forces
        margin_x_low = np.maximum(r - c[:, 0], 0.0)
        margin_x_high = np.maximum(c[:, 0] - (1.0 - r), 0.0)
        margin_y_low = np.maximum(r - c[:, 1], 0.0)
        margin_y_high = np.maximum(c[:, 1] - (1.0 - r), 0.0)
        
        fx += (margin_x_low - margin_x_high) * 15.0
        fy += (margin_y_low - margin_y_high) * 15.0
        
        c[:, 0] += dt * fx
        c[:, 1] += dt * fy
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
        
    return c

def hex_init(rng, r_est, pattern=None):
    """Generates hexagonal lattice initialization with slight noise."""
    if pattern is None:
        pattern = [5, 6, 5, 6, 4]
    c = []
    y = r_est
    for r_idx, cnt in enumerate(pattern):
        shift = r_est if r_idx % 2 == 1 else 0.0
        x = r_est + shift
        for _ in range(cnt):
            if len(c) < N:
                c.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    c = np.array(c[:N])
    c += rng.normal(0, 0.003, c.shape)
    return np.clip(c, 0.02, 0.98)

def corner_edge_init(rng):
    """Places circles strategically in corners and along edges."""
    c = np.zeros((N, 2))
    corners = [[0.07, 0.07], [0.93, 0.07], [0.07, 0.93], [0.93, 0.93]]
    c[:4] = corners
    edge_pts = []
    for _ in range(4):
        edge_pts.append([rng.uniform(0.12, 0.88), 0.07])
        edge_pts.append([rng.uniform(0.12, 0.88), 0.93])
        edge_pts.append([0.07, rng.uniform(0.12, 0.88)])
        edge_pts.append([0.93, rng.uniform(0.12, 0.88)])
    c[4:20] = np.array(edge_pts)
    c[20:] = rng.uniform(0.3, 0.7, (6, 2))
    c += rng.normal(0, 0.004, c.shape)
    return np.clip(c, 0.02, 0.98)

def optimize_pipeline(c0, rng):
    """Runs L-BFGS-B followed by Joint SLSQP polish."""
    bounds_c = [(0.001, 0.999)] * (2 * N)
    bounds_joint = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    # Phase 1: L-BFGS-B on centers
    try:
        res1 = minimize(obj_lp_centers, c0.flatten(), jac=True, method='L-BFGS-B', 
                        bounds=bounds_c, options={'maxiter': 6000, 'ftol': 1e-15})
        c_opt = res1.x.reshape(N, 2)
        r_opt, s_opt, _ = solve_lp_and_grad(c_opt)
    except Exception:
        c_opt = c0
        r_opt, s_opt, _ = solve_lp_and_grad(c_opt)
        
    # Phase 2: Joint SLSQP Polish
    try:
        ub = np.minimum(np.minimum(c_opt[:, 0], 1.0 - c_opt[:, 0]), 
                        np.minimum(c_opt[:, 1], 1.0 - c_opt[:, 1]))
        dists = np.linalg.norm(c_opt[:, None, :] - c_opt[None, :, :], axis=2)
        np.fill_diagonal(dists, np.inf)
        rp = 0.5 * np.min(dists, axis=1)
        r_init = np.minimum(ub, rp) * 0.90
        
        v0 = np.concatenate([c_opt.flatten(), r_init])
        res2 = minimize(objective_joint, v0, method='SLSQP', bounds=bounds_joint,
                        constraints={'type': 'ineq', 'fun': constraints_joint},
                        options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        if np.min(constraints_joint(res2.x)) >= -1e-7:
            c_opt = res2.x[:2*N].reshape(N, 2)
            r_opt = res2.x[2*N:]
            s_opt = np.sum(r_opt)
    except Exception:
        pass
        
    return c_opt, r_opt, s_opt

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(200):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing():
    rng = np.random.default_rng(42)
    
    best_c = None
    best_r = None
    best_s = -1.0
    
    starts = []
    
    # 1. Growing circles inits
    for _ in range(8):
        starts.append(grow_circles_init(rng))
        
    # 2. Hexagonal patterns
    pats = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], 
            [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5]]
    for pat in pats:
        for r0 in [0.092, 0.098, 0.105]:
            starts.append(hex_init(rng, r0, pat))
            
    # 3. Corner/Edge biased
    for _ in range(8):
        starts.append(corner_edge_init(rng))
        
    # 4. Pure random
    for _ in range(6):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # Phase 1: Multi-start optimization
    for c_init in starts:
        c_opt, r_opt, s_opt = optimize_pipeline(c_init, rng)
        if s_opt > best_s:
            best_s = s_opt
            best_c = c_opt.copy()
            best_r = r_opt.copy()
            
    # Phase 2: Iterative perturbation to escape local minima
    for step in range(40):
        scale = 0.018 * (0.93 ** (step // 5))
        c_k = best_c.copy()
        idx = rng.choice(N, size=rng.integers(4, 9), replace=False)
        c_k[idx] += rng.normal(0, scale, (len(idx), 2))
        c_k = np.clip(c_k, 0.02, 0.98)
        
        c_opt, r_opt, s_opt = optimize_pipeline(c_k, rng)
        if s_opt > best_s:
            best_s = s_opt
            best_c = c_opt.copy()
            best_r = r_opt.copy()
            
        # Swap two random circles to break symmetry
        if step % 3 == 0:
            swap_idx = rng.choice(N, 2, replace=False)
            c_swap = best_c.copy()
            c_swap[swap_idx] = c_swap[swap_idx[::-1]]
            c_opt, r_opt, s_opt = optimize_pipeline(c_swap, rng)
            if s_opt > best_s:
                best_s = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
                
    # Phase 3: Final high-precision joint polish
    try:
        v0_final = np.concatenate([best_c.flatten(), best_r])
        bounds_joint = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        res_final = minimize(objective_joint, v0_final, method='SLSQP', bounds=bounds_joint,
                             constraints={'type': 'ineq', 'fun': constraints_joint},
                             options={'maxiter': 15000, 'ftol': 1e-15, 'disp': False})
        if np.min(constraints_joint(res_final.x)) >= -1e-8:
            s_final = np.sum(res_final.x[2*N:])
            if s_final > best_s:
                best_c = res_final.x[:2*N].reshape(N, 2)
                best_r = res_final.x[2*N:]
                best_s = s_final
    except Exception:
        pass
        
    # Ensure radii are consistent with centers via LP
    lp_r, lp_s, _ = solve_lp_and_grad(best_c)
    if lp_s > best_s - 1e-9:
        best_r = lp_r
        best_s = lp_s
        
    radii = repair(best_c, best_r)
    return best_c, radii, float(np.sum(radii))
