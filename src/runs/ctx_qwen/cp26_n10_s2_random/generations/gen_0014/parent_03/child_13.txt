To push the sum of radii beyond 2.636, we need a more aggressive global search strategy combined with high-precision local refinement. The key bottlenecks in circle packing are:
1. **Contact Graph Topology**: The optimal packing depends heavily on which circles touch which. L-BFGS-B gets trapped in local minima defined by a suboptimal contact graph.
2. **Boundary Utilization**: Circles near corners/edges can be larger. We must ensure the optimizer explores configurations that heavily utilize boundaries.
3. **Numerical Stability**: The LP gradient is exact but discontinuous when the active set changes. We need robust handling and multiple restarts.

**Improvements:**
- **Diverse Initializations**: Generate ~60 high-quality starts combining hexagonal lattices, force-directed layouts, and explicit corner/edge biases.
- **Multi-Stage Local Search**: 
  1. Run L-BFGS-B from all starts.
  2. Implement a "Kick & Optimize" loop that perturbs random subsets of circles heavily to jump between basins.
  3. Run Simulated Annealing with periodic local optimization to escape shallow minima.
  4. Apply Joint SLSQP for final micro-adjustments of centers and radii simultaneously.
- **Efficient LP Solver**: Precompute constraint matrices and use `highs` method for speed. Vectorize distance and gradient calculations.
- **Robust Repair**: Iteratively shrink overlapping circles and clamp to boundaries to guarantee strict validation compliance.
- **Coordinate Descent**: After SA, run a few passes of independent circle optimization using Powell's method to squeeze out last increments.

This pipeline balances exploration (kicks, SA, diverse starts) with exploitation (L-BFGS-B, SLSQP, coordinate descent), systematically breaking poor contact graphs and refining promising configurations.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute constant LP constraint matrix structure for speed
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii given fixed centers and computes exact subgradient via duals."""
    centers = np.clip(centers, 1e-9, 1.0 - 1e-9)
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    
    b_ub = np.zeros(A_LP.shape[0])
    k = 0
    for i, j in PAIR_IDX:
        b_ub[k] = dists[i, j]
        k += 1
    for i in range(N):
        b_ub[k] = centers[i, 0]; k += 1
        b_ub[k] = 1.0 - centers[i, 0]; k += 1
        b_ub[k] = centers[i, 1]; k += 1
        b_ub[k] = 1.0 - centers[i, 1]; k += 1
        
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub,
                      bounds=[(0.0, u) for u in ub], method='highs')
        if not res.success:
            return ub * 0.5, np.sum(ub) * 0.5, np.zeros_like(centers)
    except Exception:
        return ub * 0.5, np.sum(ub) * 0.5, np.zeros_like(centers)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    duals = np.zeros_like(b_ub)
    if hasattr(res, 'marginals') and res.marginals is not None:
        duals = np.maximum(res.marginals.ineqlin, 0.0)
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = np.maximum(res.ineqlin.marginals, 0.0)
        
    grad = np.zeros_like(centers)
    k = 0
    for i, j in PAIR_IDX:
        mu = duals[k]
        if mu > 1e-10:
            d = max(dists[i, j], 1e-10)
            vec = (centers[i] - centers[j]) / d
            grad[i] += mu * vec
            grad[j] -= mu * vec
        k += 1
        
    b_start = NUM_PAIRS
    for i in range(N):
        b = b_start + 4 * i
        grad[i, 0] += duals[b] - duals[b+1]
        grad[i, 1] += duals[b+2] - duals[b+3]
        
    return radii, s_sum, grad

def obj_grad(x):
    """Objective and gradient wrapper for L-BFGS-B."""
    c = x.reshape(N, 2)
    _, s, g = solve_lp_and_grad(c)
    return -s, -g.flatten()

def constraints_joint(v):
    """Computes boundary and non-overlap constraints for joint SLSQP optimization."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    dx = c[TRIU_I, 0] - c[TRIU_J, 0]
    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def objective_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 4, 6, 6, 5],
        [5, 5, 6, 5, 5], [4, 5, 6, 5, 6], [6, 5, 5, 6, 4],
        [5, 6, 4, 5, 6], [6, 4, 5, 6, 5], [5, 5, 4, 6, 6]
    ]
    
    for pat in patterns:
        for r_est in [0.088, 0.094, 0.100, 0.106, 0.112]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            starts.append(np.clip(np.array(c[:N]), 0.05, 0.95))
            
    # Force-directed layouts
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(500):
            f = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    dv = c[i] - c[j]
                    d = np.linalg.norm(dv)
                    if d < 0.22 and d > 1e-4:
                        push = (0.22 - d) * 0.04 / (d + 1e-4)
                        f[i] += dv / d * push
                        f[j] -= dv / d * push
            c += f * 0.3
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    # Corner/Edge biased starts
    for _ in range(12):
        c = rng.uniform(0.2, 0.8, (N, 2))
        corners = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        c[:4] = corners
        edges = [[0.5, 0.08], [0.5, 0.92], [0.08, 0.5], [0.92, 0.5]]
        c[4:8] = edges
        c += rng.normal(0, 0.008, c.shape)
        starts.append(np.clip(c, 0.02, 0.98))
        
    # Random
    for _ in range(10):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return starts

def coordinate_descent(centers, rng, cycles=2):
    """Refines each circle's position independently."""
    c = centers.copy()
    for _ in range(cycles):
        for i in range(N):
            def local_obj(xy):
                temp = c.copy()
                temp[i] = np.clip(xy, 1e-5, 1.0 - 1e-5)
                _, s, _ = solve_lp_and_grad(temp)
                return -s
            try:
                res = minimize(local_obj, c[i], method='Powell', 
                               options={'maxiter': 200, 'xtol': 1e-8, 'ftol': 1e-12})
                if res.success:
                    c[i] = np.clip(res.x, 1e-5, 1.0 - 1e-5)
            except Exception:
                pass
    return c

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d - 1e-11:
                    shrink = (radii[i] + radii[j] - d) * 0.5 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_c = [(0.001, 0.999)] * (2 * N)
    bounds_joint = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    best_c = None
    best_r = None
    best_s = -1.0
    
    starts = generate_starts(rng)
    
    # Phase 1: Multi-start L-BFGS-B
    for c_init in starts:
        try:
            res = minimize(obj_grad, c_init.flatten(), method='L-BFGS-B', 
                           bounds=bounds_c, jac=True,
                           options={'maxiter': 5000, 'ftol': 1e-14, 'gtol': 1e-10})
            _, s, _ = solve_lp_and_grad(res.x.reshape(N, 2))
            if s > best_s:
                best_s = s
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = starts[0]
    best_r, best_s, _ = solve_lp_and_grad(best_c)
    
    # Phase 2: Aggressive Kick & Optimize to escape local minima
    for step in range(40):
        scale = 0.025 * (0.93 ** (step // 8))
        c_kick = best_c.copy()
        # Perturb a random subset of circles
        n_kick = rng.integers(5, 12)
        idx_kick = rng.choice(N, size=n_kick, replace=False)
        c_kick[idx_kick] += rng.normal(0, scale, (n_kick, 2))
        c_kick = np.clip(c_kick, 0.05, 0.95)
        
        try:
            res = minimize(obj_grad, c_kick.flatten(), method='L-BFGS-B', 
                           bounds=bounds_c, jac=True,
                           options={'maxiter': 3000, 'ftol': 1e-14})
            _, s, _ = solve_lp_and_grad(res.x.reshape(N, 2))
            if s > best_s:
                best_s = s
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    # Phase 3: Coordinate Descent Refinement
    best_c = coordinate_descent(best_c, rng, cycles=2)
    best_r, best_s, _ = solve_lp_and_grad(best_c)
    
    # Phase 4: Simulated Annealing with local settling
    c_curr = best_c.copy()
    s_curr = best_s
    T = 0.007
    for step in range(800):
        noise = T * 0.6
        c_try = c_curr + rng.normal(0, noise, c_curr.shape)
        c_try = np.clip(c_try, 0.02, 0.98)
        _, s_try, _ = solve_lp_and_grad(c_try)
        
        delta = s_try - s_curr
        if delta > 0 or rng.random() < np.exp(delta / max(T, 1e-9)):
            c_curr = c_try
            s_curr = s_try
            if s_curr > best_s:
                best_s = s_curr
                best_c = c_curr.copy()
        T *= 0.995
        
        # Periodic local optimization during SA
        if step % 30 == 0 and step > 0:
            try:
                res = minimize(obj_grad, c_curr.flatten(), method='L-BFGS-B',
                               bounds=bounds_c, jac=True,
                               options={'maxiter': 1500, 'ftol': 1e-13})
                _, s_loc, _ = solve_lp_and_grad(res.x.reshape(N, 2))
                if s_loc > s_curr:
                    c_curr = res.x.reshape(N, 2)
                    s_curr = s_loc
                    if s_curr > best_s:
                        best_s = s_curr
                        best_c = c_curr.copy()
            except Exception:
                pass
                
    best_r, _, _ = solve_lp_and_grad(best_c)
    
    # Phase 5: Joint SLSQP Polish for high-precision center-radius tuning
    try:
        v0 = np.concatenate([best_c.flatten(), best_r])
        res_sl = minimize(objective_joint, v0, method='SLSQP', bounds=bounds_joint,
                          constraints={'type': 'ineq', 'fun': constraints_joint},
                          options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if np.min(constraints_joint(res_sl.x)) >= -1e-7:
            s_sl = np.sum(res_sl.x[2*N:])
            if s_sl > best_s:
                best_s = s_sl
                best_c = res_sl.x[:2*N].reshape(N, 2).copy()
                best_r = res_sl.x[2*N:].copy()
    except Exception:
        pass
        
    # Phase 6: Final LP sync & Strict Repair
    best_r, best_s, _ = solve_lp_and_grad(best_c)
    radii = repair(best_c.copy(), best_r.copy())
    final_sum = float(np.sum(radii))
    
    return best_c, radii, final_sum
