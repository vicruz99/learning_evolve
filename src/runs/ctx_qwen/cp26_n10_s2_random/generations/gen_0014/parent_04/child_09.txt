import numpy as np
from scipy.optimize import minimize, linprog

N = 26
NUM_PAIRS = N * (N - 1) // 2
NUM_BOUND = 4 * N

# Precompute LP constraint matrix structure globally
A_LP = np.zeros((NUM_PAIRS + NUM_BOUND, N))
PAIR_I = np.zeros(NUM_PAIRS, dtype=int)
PAIR_J = np.zeros(NUM_PAIRS, dtype=int)
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_I[idx] = i
        PAIR_J[idx] = j
        idx += 1
for i in range(N):
    A_LP[NUM_PAIRS + 4*i, i] = 1.0
    A_LP[NUM_PAIRS + 4*i + 1, i] = 1.0
    A_LP[NUM_PAIRS + 4*i + 2, i] = 1.0
    A_LP[NUM_PAIRS + 4*i + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii and computes exact gradient w.r.t centers using duals."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(NUM_PAIRS + NUM_BOUND)
    b_ub[:NUM_PAIRS] = dists[np.triu_indices(n, 1)]
    for i in range(n):
        b_ub[NUM_PAIRS + 4*i] = centers[i, 0]
        b_ub[NUM_PAIRS + 4*i + 1] = 1.0 - centers[i, 0]
        b_ub[NUM_PAIRS + 4*i + 2] = centers[i, 1]
        b_ub[NUM_PAIRS + 4*i + 3] = 1.0 - centers[i, 1]
        
    try:
        res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, 
                      bounds=[(0.0, u) for u in ub], method='highs')
        if not res.success:
            return np.zeros(n), 0.0, np.zeros_like(centers)
            
        radii = res.x
        duals = np.zeros(len(b_ub))
        try:
            duals = np.asarray(res.marginals.ineqlin)
        except AttributeError:
            try:
                duals = np.asarray(res.ineqlin.marginals)
            except Exception:
                pass
                
        grad = np.zeros_like(centers)
        
        # Pairwise gradients
        active_pairs = duals[:NUM_PAIRS] > 1e-9
        if np.any(active_pairs):
            i_idx = PAIR_I[active_pairs]
            j_idx = PAIR_J[active_pairs]
            d = dists[i_idx, j_idx]
            safe_d = np.where(d > 1e-9, d, 1e-9)
            vec = (centers[i_idx] - centers[j_idx]) / safe_d[:, np.newaxis]
            lam = np.maximum(duals[:NUM_PAIRS][active_pairs], 0.0)[:, np.newaxis]
            grad[i_idx] += vec * lam
            grad[j_idx] -= vec * lam
            
        # Boundary gradients
        idx_base = NUM_PAIRS + 4 * np.arange(n)
        grad[:, 0] += np.maximum(duals[idx_base], 0.0) - np.maximum(duals[idx_base + 1], 0.0)
        grad[:, 1] += np.maximum(duals[idx_base + 2], 0.0) - np.maximum(duals[idx_base + 3], 0.0)
            
        return radii, np.sum(radii), grad
    except Exception:
        return np.zeros(n), 0.0, np.zeros_like(centers)

def lbfgs_obj_grad(x):
    """Objective and gradient for L-BFGS-B."""
    c = np.clip(x.reshape(N, 2), 1e-5, 1.0 - 1e-5)
    _, val, grad = solve_lp_and_grad(c)
    return -val, -grad.flatten()

def generate_inits(rng):
    """Generates diverse initial configurations."""
    inits = []
    
    # Hexagonal lattices with various patterns and rotations
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 5, 5, 6],
        [5, 4, 6, 6, 5], [4, 5, 6, 5, 6], [6, 5, 5, 6, 4]
    ]
    for pat in patterns:
        for r_est in [0.095, 0.100, 0.105, 0.110]:
            for angle in [0.0, 0.05, 0.1]:
                c = []
                y = r_est
                for r_idx, cnt in enumerate(pat):
                    shift = r_est if r_idx % 2 == 1 else 0.0
                    x = r_est + shift
                    for _ in range(cnt):
                        if len(c) < N:
                            # Apply rotation around center
                            cx, cy = x - 0.5, y - 0.5
                            ca, sa = np.cos(angle), np.sin(angle)
                            rot_x = cx * ca - cy * sa + 0.5
                            rot_y = cx * sa + cy * ca + 0.5
                            c.append([rot_x, rot_y])
                        x += 2.0 * r_est
                    y += r_est * np.sqrt(3.0)
                c = np.array(c[:N])
                c += rng.normal(0, 0.002, c.shape)
                c = np.clip(c, 0.05, 0.95)
                inits.append(c)
                
    # Force-directed spreads
    for _ in range(10):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(600):
            forces = np.zeros_like(c)
            diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-9
            rep = np.where(dists < 0.28, (0.28 - dists) * 5.0 / dists, 0.0)
            forces += np.sum(diff * rep[:, :, np.newaxis], axis=1)
            # Boundary repulsion
            for i in range(N):
                if c[i, 0] < 0.1: forces[i, 0] += (0.1 - c[i, 0]) * 10
                if c[i, 0] > 0.9: forces[i, 0] -= (c[i, 0] - 0.9) * 10
                if c[i, 1] < 0.1: forces[i, 1] += (0.1 - c[i, 1]) * 10
                if c[i, 1] > 0.9: forces[i, 1] -= (c[i, 1] - 0.9) * 10
            c += forces * 0.005
            c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    # Random with corner bias
    for _ in range(8):
        c = rng.uniform(0.15, 0.85, (N, 2))
        corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        c[:4] = corners
        c += rng.normal(0, 0.02, c.shape)
        c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    return inits

def coordinate_descent(centers, rng):
    """Refines each circle's position independently using Nelder-Mead."""
    c = centers.copy()
    _, best_s, _ = solve_lp_and_grad(c)
    
    for _ in range(4):
        for i in range(N):
            def obj_2d(xy):
                temp = c.copy()
                temp[i] = np.clip(xy, 0.01, 0.99)
                _, s, _ = solve_lp_and_grad(temp)
                return -s
                
            try:
                res = minimize(obj_2d, c[i], method='Nelder-Mead', 
                               options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-12})
                if -res.fun > best_s + 1e-9:
                    c[i] = np.clip(res.x, 0.01, 0.99)
                    _, best_s, _ = solve_lp_and_grad(c)
            except Exception:
                pass
    return c

def basin_hopping_manual(centers, rng):
    """Manual basin hopping to escape local minima with decaying scale."""
    c = centers.copy()
    _, s_best, _ = solve_lp_and_grad(c)
    
    # Adaptive perturbation schedule
    scales = [0.02, 0.015, 0.01, 0.005, 0.002]
    for scale in scales:
        for step in range(60):
            c_trial = c + rng.normal(0, scale, c.shape)
            c_trial = np.clip(c_trial, 0.02, 0.98)
            _, s_trial, _ = solve_lp_and_grad(c_trial)
            if s_trial > s_best:
                c = c_trial.copy()
                s_best = s_trial
    return c

def joint_slsqp_polish(centers, radii):
    """Joint SLSQP polish for centers and radii to tighten constraints."""
    v0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    def obj(v):
        return -np.sum(v[2 * N:])
        
    def cons(v):
        c = v[:2 * N].reshape(N, 2)
        r = v[2 * N:]
        con = []
        con.append(c[:, 0] - r)
        con.append(1.0 - c[:, 0] - r)
        con.append(c[:, 1] - r)
        con.append(1.0 - c[:, 1] - r)
        i, j = np.triu_indices(N, 1)
        dx = c[i, 0] - c[j, 0]
        dy = c[i, 1] - c[j, 1]
        dr = r[i] + r[j]
        con.append(dx**2 + dy**2 - dr**2)
        return np.concatenate(con)
        
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if np.min(cons(res.x)) >= -1e-7:
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:]
    except Exception:
        pass
    return centers, radii

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_lbfgs = [(0.01, 0.99)] * (2 * N)
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    inits = generate_inits(rng)
    
    # Phase 1: Multi-start L-BFGS-B with exact LP gradients
    for c0 in inits:
        try:
            res = minimize(lbfgs_obj_grad, c0.flatten(), method='L-BFGS-B', jac=True,
                           bounds=bounds_lbfgs, options={'maxiter': 4000, 'ftol': 1e-13})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt, _ = solve_lp_and_grad(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum, _ = solve_lp_and_grad(best_c)
        
    # Phase 2: Coordinate Descent Refinement to break symmetry
    best_c = coordinate_descent(best_c, rng)
    best_r, best_sum, _ = solve_lp_and_grad(best_c)
    
    # Phase 3: Adaptive Basin Hopping
    best_c = basin_hopping_manual(best_c, rng)
    best_r, best_sum, _ = solve_lp_and_grad(best_c)
    
    # Phase 4: L-BFGS-B Polish after basin hopping
    try:
        res = minimize(lbfgs_obj_grad, best_c.flatten(), method='L-BFGS-B', jac=True,
                       bounds=bounds_lbfgs, options={'maxiter': 5000, 'ftol': 1e-13})
        c_pol = res.x.reshape(N, 2)
        r_pol, s_pol, _ = solve_lp_and_grad(c_pol)
        if s_pol > best_sum:
            best_sum = s_pol
            best_c = c_pol.copy()
            best_r = r_pol.copy()
    except Exception:
        pass
        
    # Phase 5: Joint SLSQP Polish for high-precision constraint satisfaction
    best_c, best_r = joint_slsqp_polish(best_c, best_r)
    best_r, best_sum, _ = solve_lp_and_grad(best_c)
    
    # Phase 6: Final LP Verification & Strict Repair
    lp_r, final_s, _ = solve_lp_and_grad(best_c)
    if final_s > best_sum:
        best_r = lp_r
        best_sum = final_s
        
    final_radii = repair(best_c.copy(), best_r.copy())
    return best_c, final_radii, float(np.sum(final_radii))
