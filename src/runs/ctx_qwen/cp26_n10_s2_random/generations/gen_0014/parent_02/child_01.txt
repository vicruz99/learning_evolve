N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))

    A_LP[base + 1, i] = 1.0

    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),

def lbfgs_obj_grad(x_flat):
    c = x_flat.reshape(N, 2)
    _, s, g = solve_lp_and_grad(c)
    return -s, -g.flatten()

def lbfgs_optimize(c0, maxiter=5000):
    bounds = [(1e-5, 1.0 - 1e-5)] * (2 * N)
    try:
        res = minimize(lbfgs_obj_grad, c0.flatten(), method='L-BFGS-B',
                       jac=True, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-11})
        c_opt = res.x.reshape(N, 2)
        _, s_opt, _ = solve_lp_and_grad(c_opt)
        return c_opt, s_opt
    except Exception:
        _, s0, _ = solve_lp_and_grad(c0)
        return c0.copy(), s0

def slsqp_obj(v):
    return -np.sum(v[2 * N:])

def slsqp_cons(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r,
                          c[:, 1] - r, 1.0 - c[:, 1] - r])
    dx = c[TRIU_I, 0] - c[TRIU_J, 0]
    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    con = np.concatenate([con, dx**2 + dy**2 - dr**2])
    return con

def slsqp_polish(centers, radii, maxiter=10000):

    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

                       constraints={'type': 'ineq', 'fun': slsqp_cons},

def repair(centers, radii):

        changed = False

                    radii[j] -= shrink
                    changed = True

            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])

                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def force_repel(centers, steps=600, strength=0.012):
    c = centers.copy()
    for _ in range(steps):
        diffs = c[:, None, :] - c[None, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        dists_safe = np.maximum(dists, 1e-6)
        inv_d = 1.0 / dists_safe
        np.fill_diagonal(inv_d, 0.0)
        force_mag = np.where(dists < 0.28, strength * inv_d**2, 0.0)
        f = np.zeros_like(c)
        for d_dim in range(2):
            f[:, d_dim] = np.sum(diffs[:, :, d_dim] * force_mag / dists_safe, axis=1)
        mask_low = c < 0.04

        f[mask_low] += strength * 8.0
        f[mask_high] -= strength * 8.0
        c += f * 0.004
        c = np.clip(c, 0.01, 0.99)
    return c

def coordinate_descent(c0, num_rounds=3):

        y += r0 * np.sqrt(3.0)
