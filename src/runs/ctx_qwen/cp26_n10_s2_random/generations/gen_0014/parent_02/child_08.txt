import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute constant LP constraint matrix structure for speed
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):

    centers = np.clip(centers, 1e-7, 1.0 - 1e-7)

    k = 0

        b_ub[k] = dists[i, j]
        k += 1

        b_ub[k] = 1.0 - centers[i, 0]; k += 1
        b_ub[k] = centers[i, 1]; k += 1
        b_ub[k] = 1.0 - centers[i, 1]; k += 1
        
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub,

            return np.full(N, 0.08), 0.0, np.zeros_like(centers)
    except Exception:
        return np.full(N, 0.08), 0.0, np.zeros_like(centers)

    except Exception:
        pass

        k += 1

        b = b_start + 4 * i
        grad[i, 0] += duals[b] - duals[b+1]
        grad[i, 1] += duals[b+2] - duals[b+3]
        
    return radii, s_sum, grad

def obj_grad_lbfgs(v):
    """Objective and gradient wrapper for L-BFGS-B."""
    c = v.reshape(N, 2)
    _, s, g = solve_lp_and_grad(c)
    return -s, -g.flatten()

def slsqp_obj(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def slsqp_cons(v):
    """Constraints for joint SLSQP: boundary and non-overlap (squared)."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    dx = c[TRIU_I, 0] - c[TRIU_J, 0]
    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    con = np.concatenate([con, dx**2 + dy**2 - dr**2])
    return con

def coordinate_refine(centers, rng):
    """Refines each circle's position independently using Nelder-Mead."""
    c = centers.copy()
    _, best_s, _ = solve_lp_and_grad(c)
    
    for _ in range(3):
        order = rng.permutation(N)
        for i in order:

                temp[i] = np.clip(xy, 1e-4, 1.0 - 1e-4)
                _, s, _ = solve_lp_and_grad(temp)
                return -s
                
            try:
                res = minimize(obj_2d, c[i], method='Nelder-Mead', 
                               options={'maxiter': 400, 'xatol': 1e-8, 'fatol': 1e-12})

                    c[i] = np.clip(res.x, 1e-4, 1.0 - 1e-4)
                    _, best_s, _ = solve_lp_and_grad(c)
            except Exception:
                pass
    return c

def force_layout_growth(rng):

    r = np.full(N, 0.02)

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(150):
        changed = False
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
