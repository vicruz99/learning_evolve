from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure globally for speed
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii given fixed centers and computes exact subgradient via duals."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    b = np.zeros(NUM_PAIRS + 4 * N)
    k = 0

        b[k] = dists[i, j]
        k += 1

        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1

        return np.full(N, 1e-7), 1e-5, np.zeros_like(centers)

        return np.full(N, 1e-7), 1e-5, np.zeros_like(centers)

    s_sum = np.sum(radii)
    
    duals = np.zeros(len(b))

    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = res.ineqlin.marginals

        grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
    return radii, s_sum, grad

def obj_grad_lbfgsb(x_flat):
    """Objective and gradient wrapper for L-BFGS-B."""
    c = np.clip(x_flat.reshape(N, 2), 1e-5, 1.0 - 1e-5)
    _, val, grad = solve_lp_and_grad(c)
    return -val, -grad.flatten()

def optimize_lbfgsb(c0, max_iter=3000):
    """Optimizes circle positions using L-BFGS-B with exact gradient."""
    bounds = [(1e-5, 1.0 - 1e-5)] * (2 * N)
    try:
        res = minimize(obj_grad_lbfgsb, c0.flatten(), method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
        return res.x.reshape(N, 2), -res.fun
    except Exception:
        return c0, 0.0

def obj_2d_wrapper(xy, fixed_c, idx):
    """Objective for coordinate-wise optimization of a single circle."""
    temp = fixed_c.copy()
    temp[idx] = np.clip(xy, 1e-5, 1.0 - 1e-5)
    _, v, _ = solve_lp_and_grad(temp)
    return -v

def coordinate_descent(c, rng):
    """Refines each circle's position sequentially using Nelder-Mead."""
    c = c.copy()
    for _ in range(3):
        for i in range(N):
            best_val, _ = solve_lp_and_grad(c)
            try:
                res = minimize(obj_2d_wrapper, c[i], args=(c, i), method='Nelder-Mead',
                               options={'maxiter': 1000, 'xatol': 1e-9, 'fatol': 1e-12})
                if -res.fun > best_val + 1e-10:
                    c[i] = np.clip(res.x, 1e-5, 1.0 - 1e-5)
            except Exception:
                pass
    return c

def grow_pack(rng):
    """Force-directed growth packing to find dense initial configurations."""
    c = rng.uniform(0.2, 0.8, (N, 2))
    r = np.full(N, 0.01)
    for _ in range(2000):
        r *= 1.0005
        forces = np.zeros_like(c)
        margin = 0.005
        for i in range(N):
            if c[i, 0] < r[i] + margin: forces[i, 0] += (r[i] + margin - c[i, 0]) * 100
            elif c[i, 0] > 1.0 - r[i] - margin: forces[i, 0] -= (c[i, 0] - (1.0 - r[i] - margin)) * 100
            if c[i, 1] < r[i] + margin: forces[i, 1] += (r[i] + margin - c[i, 1]) * 100

        dy = c[:, None, 1] - c[None, :, 1]

        min_dist = r[:, None] + r[None, :] + margin

        dir_x = dx / (dist + 1e-6)
        dir_y = dy / (dist + 1e-6)
        forces[:, 0] += (repulsion * dir_x).sum(axis=1)
        forces[:, 1] += (repulsion * dir_y).sum(axis=1)
        
        c += forces * 0.0005
        c = np.clip(c, 1e-5, 1.0 - 1e-5)
    return c

def obj_joint_func(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint_func(v):
    """Constraints for joint SLSQP: boundary and non-overlap (squared for smoothness)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    dx = c[TRIU_I, 0] - c[TRIU_J, 0]
    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def slsqp_polish(c0, r0):
    """Joint SLSQP optimization of centers and radii."""
    v0 = np.concatenate([c0.flatten(), r0])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        
    try:
        res = minimize(obj_joint_func, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons_joint_func},
                       options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
        c_val = cons_joint_func(res.x)
        if np.min(c_val) >= -1e-6:
            return res.x[:2*N].reshape(N, 2), res.x[2*N:], -res.fun
    except Exception:
        pass
    return c0, r0, np.sum(r0)

def repair(centers, radii):
    """Deterministic repair to ensure strict validation compliance."""
    radii = radii.copy()
    for _ in range(200):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
