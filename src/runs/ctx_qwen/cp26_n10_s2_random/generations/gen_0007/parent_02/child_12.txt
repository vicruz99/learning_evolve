import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_INDICES = np.triu_indices(N, k=1)
n_pairs = N*(N-1)//2
n_bnd = 4*N

# Precompute LP constraint matrix structure globally for speed
A_ub_mat = np.zeros((n_pairs + n_bnd, N))
pair_indices = []
idx = 0
for i in range(N):
    for j in range(i+1, N):
        A_ub_mat[idx, i] = 1.0
        A_ub_mat[idx, j] = 1.0
        pair_indices.append((i, j))
        idx += 1
for i in range(N):
    for _ in range(4):
        A_ub_mat[idx, i] = 1.0
        idx += 1

def solve_lp_and_gradient(centers):
    """Solves LP for radii and computes exact gradient of sum of radii w.r.t centers."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)

    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

    b_ub = np.zeros(n_pairs + n_bnd)
    idx = 0
    for i, j in pair_indices:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(n):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1

    c_obj = -np.ones(n)
    bounds_r = [(0.0, u) for u in ub]

    res = linprog(c_obj, A_ub=A_ub_mat, b_ub=b_ub, bounds=bounds_r, method='highs')
    if not res.success:
        return None, None, None

    radii = res.x
    duals = res.ineqlin.marginals
    if duals is None:
        return radii, -res.fun, np.zeros_like(centers)

    grad = np.zeros_like(centers)
    idx = 0
    for i, j in pair_indices:
        lam = duals[idx]
        if lam > 1e-7:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1

    b_start = n_pairs
    for i in range(n):
        mu_L = duals[b_start + 4*i]
        mu_R = duals[b_start + 4*i + 1]
        mu_B = duals[b_start + 4*i + 2]
        mu_T = duals[b_start + 4*i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T

    return radii, -res.fun, grad

def lp_objective(x_flat):
    """Objective for center-only optimization: negative sum of radii."""
    centers = x_flat.reshape(N, 2)
    _, s, _ = solve_lp_and_gradient(centers)
    if s is None:
        return 0.0
    return -s

def generate_initial_configs(rng):
    """Generates diverse initial center configurations."""
    configs = []
    pats = [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], [5,5,5,6,5], [5,5,5,5,5,1]]
    for pat in pats:
        c = []
        r_est = 0.095
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(c) < N:
                    c.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
                x += 2.0 * r_est
            y += r_est * 1.73205
        configs.append(np.array(c[:N]))

    for _ in range(15):
        c = rng.uniform(0.12, 0.88, (N, 2))
        configs.append(c)

    gx, gy = np.meshgrid(np.linspace(0.1, 0.9, 5), np.linspace(0.1, 0.9, 5))
    c_grid = np.column_stack([gx.ravel(), gy.ravel()])
    c_grid = np.vstack([c_grid, [0.5, 0.5]])
    configs.append(c_grid[:N])

    return configs

def gradient_ascent_run(centers0, rng):
    """Runs gradient ascent on centers using LP duals."""
    c = centers0.copy()
    best_c = c.copy()
    best_sum = -1.0
    max_iter = 1500
    no_improve = 0

    for k in range(max_iter):
        r, s, grad = solve_lp_and_gradient(c)
        if r is None:
            break
        if s > best_sum:
            best_sum = s
            best_c = c.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size with decay
        step = 0.005 * (0.997 ** k)
        if no_improve > 150:
            step *= 0.5
            
        c += step * grad
        c = np.clip(c, 0.02, 0.98)

        # Perturb to escape plateaus
        if k % 200 == 0 and k > 0:
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.02, 0.98)

    return best_c, best_sum

def obj_joint(v):
    """Objective for joint SLSQP optimization."""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Constraints for joint SLSQP optimization."""
    cc = v[:2*N].reshape(N, 2)
    rr = v[2*N:]
    c_list = [cc[:,0]-rr, 1.0-cc[:,0]-rr, cc[:,1]-rr, 1.0-cc[:,1]-rr]
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = cc[idx_i, 0] - cc[idx_j, 0]
    dy = cc[idx_i, 1] - cc[idx_j, 1]
    dr = rr[idx_i] + rr[idx_j]
    c_list.append(np.sqrt(dx**2 + dy**2) - dr)
    return np.concatenate(c_list)

def run_packing():
    rng = np.random.default_rng(42)
    configs = generate_initial_configs(rng)

    best_centers = None
    best_sum = -1.0

    # Phase 1: Gradient Ascent from multiple starts
    for cfg in configs:

        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt

    # Phase 2: Powell refinement on the best centers
    x0 = best_centers.flatten()
    bounds_c = [(0.01, 0.99)] * (2*N)
    try:
        res_pw = minimize(lp_objective, x0, method='Powell', bounds=bounds_c,
                          options={'maxiter': 2500, 'ftol': 1e-12})
        c_pw = res_pw.x.reshape(N, 2)
        _, s_pw, _ = solve_lp_and_gradient(c_pw)
        if s_pw > best_sum:
            best_sum = s_pw
            best_centers = c_pw
    except Exception:
        pass

    # Get exact LP radii for the optimized centers
    r_lp, s_lp, _ = solve_lp_and_gradient(best_centers)
    best_radii = r_lp if r_lp is not None else np.zeros(N)
    best_sum = s_lp if s_lp is not None else best_sum

    # Phase 3: Joint SLSQP polish
    v0 = np.concatenate([best_centers.flatten(), best_radii])
    bounds_j = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': 5000, 'ftol': 1e-13})
        if np.min(cons_joint(res.x)) >= -1e-8:
            s_new = np.sum(res.x[2*N:])
            if s_new > best_sum:
                best_centers = res.x[:2*N].reshape(N, 2)
                best_radii = res.x[2*N:]
                best_sum = s_new
    except Exception:
        pass

    # Phase 4: Strict numerical repair to guarantee validation tolerance
    centers = best_centers.copy()
    radii = best_radii.copy()
    for _ in range(60):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    sh = (req - d)/2.0 + 1e-9
                    radii[i] -= sh
                    radii[j] -= sh
                    changed = True
        for i in range(N):
            mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mx + 1e-12:
                radii[i] = mx
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)

    return centers, radii, float(np.sum(radii))
