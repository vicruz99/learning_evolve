import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    c = np.clip(centers, 1e-8, 1.0 - 1e-8)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    n_pairs = N * (N - 1) // 2
    A_ub = np.zeros((n_pairs, N))
    b_ub = np.zeros(n_pairs)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, np.sum(res.x)
    return np.full(N, 1e-5), 0.0

def objective_lp_centers(x):
    _, s = solve_lp_radii(x.reshape(N, 2))
    return -s

def generate_starts():
    starts = []
    rng = np.random.default_rng(42)
    
    patterns = [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], [6,6,5,5,4], [5,5,5,6,5]]
    for pat in patterns:
        c = []
        r_est = 0.095
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(c) < N:
                    c.append([x, y])
                x += 2.0 * r_est
            y += r_est * 1.7320508
        starts.append(np.array(c[:N]))
        
    for _ in range(10):
        c = rng.uniform(0.05, 0.95, (N, 2))
        starts.append(c)
        
    for _ in range(5):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(400):
            f = np.zeros_like(c)
            diff = c[:, None, :] - c[None, :, :]
            dist = np.linalg.norm(diff, axis=2) + 1e-4
            f += np.sum(diff / (dist**2)[:, :, None], axis=1)
            c += 0.005 * f
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    return starts

def gradient_ascent_step(centers, step_size):
    c = centers.copy()
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    n_pairs = N * (N - 1) // 2
    A_ub = np.zeros((n_pairs, N))
    b_ub = np.zeros(n_pairs)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    bounds = [(0.0, max(u, 1e-9)) for u in ub]
    
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return centers
        
    duals = res.ineqlin.marginals
    if duals is None or len(duals) == 0:
        return centers
    
    grad = np.zeros_like(c)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            lam = duals[idx]
            if lam > 1e-7:
                d = dists[i, j]
                if d > 1e-9:
                    vec = (c[i] - c[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
            idx += 1
            
    c += step_size * grad
    return np.clip(c, 1e-4, 1.0 - 1e-4)

def obj_joint(v):
    return -np.sum(v[2*N:])

def cons_joint(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    con.append(np.sqrt(dx**2 + dy**2) - dr)
    return np.concatenate(con)

def run_packing() -> tuple:
    rng = np.random.default_rng(42)
    starts = generate_starts()
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    bounds_c = [(0.0, 1.0)] * (2*N)
    
    # Phase 1: Powell optimization from diverse starts
    for c0 in starts:
        try:
            res = minimize(objective_lp_centers, c0.flatten(), method='Powell', 
                          bounds=bounds_c, options={'maxiter': 800, 'ftol': 1e-12})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt = solve_lp_radii(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            pass
            
    # Phase 2: LP-Dual Guided Gradient Ascent
    if best_c is not None:
        c_curr = best_c.copy()
        step = 0.003
        for _ in range(120):
            c_new = gradient_ascent_step(c_curr, step)
            r_new, s_new = solve_lp_radii(c_new)
            if s_new > best_sum:
                best_sum = s_new
                best_c = c_new.copy()
                best_r = r_new.copy()
                c_curr = c_new.copy()
            else:
                step *= 0.95
                c_curr = best_c.copy()
            if step < 1e-7:
                break
                
        # Phase 3: Local Powell refinement after gradient ascent
        try:
            res2 = minimize(objective_lp_centers, best_c.flatten(), method='Powell',
                           bounds=bounds_c, options={'maxiter': 600, 'ftol': 1e-12})
            c2 = res2.x.reshape(N, 2)
            r2, s2 = solve_lp_radii(c2)
            if s2 > best_sum:
                best_sum = s2
                best_c = c2
                best_r = r2
        except Exception:
            pass
            
        # Phase 4: Stochastic perturbations to escape shallow basins
        for _ in range(6):
            c_pert = best_c + rng.normal(0, 0.003, best_c.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            try:
                res3 = minimize(objective_lp_centers, c_pert.flatten(), method='Powell',
                               bounds=bounds_c, options={'maxiter': 500, 'ftol': 1e-12})
                c3 = res3.x.reshape(N, 2)
                r3, s3 = solve_lp_radii(c3)
                if s3 > best_sum:
                    best_sum = s3
                    best_c = c3
                    best_r = r3
            except Exception:
                pass
                
        # Phase 5: Joint SLSQP optimization on centers and radii
        v0 = np.concatenate([best_c.flatten(), best_r])
        bounds_joint = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        try:
            res_j = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                            constraints={'type': 'ineq', 'fun': cons_joint},
                            options={'maxiter': 5000, 'ftol': 1e-13})
            if np.sum(res_j.x[2*N:]) > best_sum:
                best_c = res_j.x[:2*N].reshape(N, 2)
                best_r = res_j.x[2*N:]
                best_sum = np.sum(best_r)
        except Exception:
            pass
            
    # Final strict numerical repair to guarantee validator compliance
    centers = best_c.copy()
    radii = best_r.copy()
    
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d)/2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
