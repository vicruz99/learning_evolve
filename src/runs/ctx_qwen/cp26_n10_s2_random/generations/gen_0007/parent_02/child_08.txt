from scipy.optimize import linprog, minimize

N = 26

# Precompute LP constraint structure for pairwise distances
A_ub_pair = np.zeros((N*(N-1)//2, N))
pair_idx = []
idx = 0
for i in range(N):
    for j in range(i+1, N):
        A_ub_pair[idx, i] = 1.0
        A_ub_pair[idx, j] = 1.0
        pair_idx.append((i, j))
        idx += 1

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[np.triu_indices(N, k=1)]
    
    res = linprog(-np.ones(N), A_ub=A_ub_pair, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:
        return res.x, -res.fun, res.ineqlin.marginals
    return np.zeros(N), 0.0, None

def obj_centers(x):
    """Objective: minimize negative LP sum of radii."""
    c = x.reshape(N, 2)
    _, s, _ = solve_lp(c)
    return -s

                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
    return grad
