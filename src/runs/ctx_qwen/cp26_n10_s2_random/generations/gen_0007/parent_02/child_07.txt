import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_and_grad(centers):
    """
    Solves LP to maximize sum of radii for fixed centers and computes the 
    exact gradient of the objective with respect to center positions using LP duals.
    """
    n = centers.shape[0]
    # Distance to boundaries
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    A_ub = np.zeros((num_pairs + num_bound, n))
    b_ub = np.zeros(num_pairs + num_bound)
    
    idx = 0
    pair_indices = []
    # Pairwise constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            pair_indices.append((i, j))
            idx += 1
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    # Solve LP: max sum(r_i) <=> min -sum(r_i)
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return None, None, None
        
    radii = res.x
    duals = np.abs(res.ineqlin.marginals)
    
    grad = np.zeros_like(centers)
    lam = np.zeros((n, n))
    idx = 0
    for i, j in pair_indices:
        lam[i, j] = duals[idx]
        lam[j, i] = duals[idx]
        idx += 1
        
    # Pairwise repulsive forces: grad_i += sum_j lambda_ij * (c_i - c_j) / d_ij
    d_safe = np.where(dists > 1e-9, dists, 1.0)
    forces = lam * (diff / d_safe[:, :, None])
    grad += np.sum(forces, axis=1)
    
    # Boundary forces from wall constraints
    b_start = num_pairs
    for i in range(n):
        grad[i, 0] += duals[b_start + 4*i]     # left wall
        grad[i, 0] -= duals[b_start + 4*i + 1] # right wall
        grad[i, 1] += duals[b_start + 4*i + 2] # bottom wall
        grad[i, 1] -= duals[b_start + 4*i + 3] # top wall
        
    return radii, np.sum(radii), grad

def obj_func(x):
    """Objective function for L-BFGS-B"""
    r, s, _ = solve_lp_and_grad(x.reshape(N, 2))
    return s if r is not None else -1.0

def obj_jac(x):
    """Gradient of the objective for L-BFGS-B"""
    r, _, g = solve_lp_and_grad(x.reshape(N, 2))
    if r is None:
        return np.zeros_like(x)
    # We want to maximize s, so gradient of -s is -g
    return -g.flatten()

def hex_init(row_counts, r0, rng):
    """Generates initial hexagonal lattice configuration"""
    c = []
    y = r0
    for r_idx, cnt in enumerate(row_counts):
        shift = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            c.append([x + rng.normal(0, 0.001), y + rng.normal(0, 0.001)])
            x += 2.0 * r0
        y += r0 * 1.7320508
    return np.array(c[:N])

def run_packing() -> tuple:
    rng = np.random.default_rng(42)
    bounds = [(0.02, 0.98)] * (2 * N)
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = []
    # Diverse hexagonal patterns covering likely optimal topologies for N=26
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [6,6,5,5,4], [4,5,6,5,6],
        [5,6,4,6,5], [6,4,6,6,4], [5,5,6,5,5], [4,6,6,5,5], [5,4,6,6,5]
    ]
    
    for pat in patterns:
        for r0 in [0.095, 0.100, 0.105, 0.108]:
            c = hex_init(pat, r0, rng)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    # Force-repulsion spread configurations to escape lattice biases
    for _ in range(12):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(300):
            f = np.zeros_like(c)
            diff = c[:, None, :] - c[None, :, :]
            d = np.linalg.norm(diff, axis=2)
            d = np.maximum(d, 1e-4)
            f += np.sum(diff / d**2[:, :, None], axis=1)
            c += 0.005 * f
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    # Phase 1: Analytical Gradient Optimization from multiple starts
    for c0 in starts:
        x0 = c0.flatten()
        try:
            res = minimize(obj_func, x0, jac=obj_jac, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 4000, 'ftol': 1e-13, 'gtol': 1e-12})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt, _ = solve_lp_and_grad(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Strict Numerical Repair to guarantee validator compliance
    centers = best_c.copy()
    radii = best_r.copy()
    
    for _ in range(60):
        changed = False
        # Resolve overlaps by symmetric shrinking
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d)/2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
