To maximize the sum of radii for 26 circles in a unit square, we exploit the fact that for fixed centers, the optimal radii are found by solving a Linear Program (LP). The objective function (sum of radii) is a concave-like function of the centers, and its gradient can be computed exactly using the dual variables (shadow prices) of the LP. 

Strategy:
1. **LP Formulation**: Maximize $\sum r_i$ subject to $r_i + r_j \le \|c_i - c_j\|$ and $r_i \le \text{dist}(c_i, \text{boundary})$. The `highs` solver efficiently computes both optimal radii and dual variables.
2. **Dual-Gradient Ascent**: The gradient of the optimal sum w.r.t centers is $\nabla_{c_i} V = \sum_j \lambda_{ij} \frac{c_i - c_j}{\|c_i - c_j\|} + \sum_{walls} \mu_{i,w} \nabla \text{dist}$. We perform gradient ascent on centers to "push" circles apart and towards boundaries optimally.
3. **Hybrid Optimization**: Combine gradient ascent (fast, follows constraint geometry) with Powell's method (robust for non-smooth landscapes) and multi-start initializations (hexagonal lattices, force-directed spreads, random).
4. **Refinement & Repair**: After finding the best centers, solve the LP exactly for radii, then apply a strict iterative repair to guarantee validator tolerance compliance ($10^{-12}$ slack).

This approach systematically explores the configuration space, leverages exact sensitivity analysis for rapid convergence, and ensures numerical validity.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def get_lp_matrices(n):
    """Precompute sparse structure for LP constraint matrix."""
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    A_ub = np.zeros((num_pairs + num_bound, n))
    pair_indices = []
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            pair_indices.append((i, j))
            idx += 1
    for i in range(n):
        for _ in range(4):
            A_ub[idx, i] = 1.0
            idx += 1
    return A_ub, pair_indices

A_ub_global, pair_indices_global = get_lp_matrices(N)

def solve_lp_and_grad(centers):
    """
    Solves LP for optimal radii given fixed centers and computes the gradient 
    of the sum of radii w.r.t. center positions using LP duals.
    Returns (radii, sum_radii, gradient_centers)
    """
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(A_ub_global.shape[0])
    idx = 0
    for i, j in pair_indices_global:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(n):
        b_ub[len(pair_indices_global) + 4*i] = centers[i, 0]
        b_ub[len(pair_indices_global) + 4*i + 1] = 1.0 - centers[i, 0]
        b_ub[len(pair_indices_global) + 4*i + 2] = centers[i, 1]
        b_ub[len(pair_indices_global) + 4*i + 3] = 1.0 - centers[i, 1]
        
    bounds_r = [(0.0, u) for u in ub]
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_global, b_ub=b_ub, 
                      bounds=bounds_r, method='highs')
        if not res.success:
            return np.zeros(n), 0.0, np.zeros((n, 2))
    except Exception:
        return np.zeros(n), 0.0, np.zeros((n, 2))
        
    radii = res.x
    duals = res.ineqlin.marginals
    
    grad = np.zeros_like(centers)
    idx = 0
    for i, j in pair_indices_global:
        lam = duals[idx]
        if lam > 1e-7:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    bs = len(pair_indices_global)
    for i in range(n):
        grad[i, 0] += duals[bs + 4*i] - duals[bs + 4*i + 1]
        grad[i, 1] += duals[bs + 4*i + 2] - duals[bs + 4*i + 3]
        
    return radii, np.sum(radii), grad

def gradient_ascent(centers0, max_iter=1200):
    """Gradient ascent on centers to maximize LP sum of radii."""
    centers = centers0.copy()
    best_c = centers.copy()
    best_sum = -1.0
    step = 0.005
    no_improve = 0
    
    for k in range(max_iter):
        _, curr_sum, grad = solve_lp_and_grad(centers)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_c = centers.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-6:
            break
            
        centers += step * grad / (g_norm + 1e-8)
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
        
        if no_improve > 40:
            step *= 0.8
            centers += np.random.normal(0, step/5.0, centers.shape)
            centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
            no_improve = 0
        if step < 1e-7:
            break
            
    return best_c, best_sum

def run_packing() -> tuple:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_c = None
    best_sum = -1.0
    
    starts = []
    
    # 1. Hexagonal lattice patterns (various densities & row counts)
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [5,5,5,6,5], 
                [4,6,6,6,4], [5,4,6,5,6], [6,6,5,5,4]]
    for pat in patterns:
        c = []
        r_est = 0.095
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(c) < N:
                    c.append([x, y])
                x += 2.0 * r_est
            y += r_est * 1.7320508
        starts.append(np.array(c[:N]))
        
    # 2. Random uniform starts
    for _ in range(8):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # 3. Force-repulsion spread starts
    for _ in range(5):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(400):
            f = np.zeros_like(c)
            diff = c[:, None, :] - c[None, :, :]
            dist = np.linalg.norm(diff, axis=2) + 1e-4
            f += np.sum(diff / dist[:, :, None]**2, axis=1)
            c += 0.0025 * f
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    # Phase 1: Gradient ascent from diverse starts
    for c_init in starts:
        c_opt, s_opt = gradient_ascent(c_init)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            
    # Phase 2: Powell refinement (handles non-smooth LP objective robustly)
    def lp_obj(x):
        _, s = solve_lp_and_grad(x.reshape(N, 2))
        return -s
        
    x0 = best_c.flatten()
    bounds_c = [(0.0, 1.0)] * (2*N)
    try:
        res_p = minimize(lp_obj, x0, method='Powell', bounds=bounds_c, 
                        options={'maxiter': 1500, 'ftol': 1e-12})
        c_powell = res_p.x.reshape(N, 2)
        _, s_powell = solve_lp_and_grad(c_powell)
        if s_powell > best_sum:
            best_sum = s_powell
            best_c = c_powell.copy()
    except Exception:
        pass
        
    # Phase 3: Gradient ascent again to polish gradient-aligned directions
    c_opt2, s_opt2 = gradient_ascent(best_c, max_iter=800)
    if s_opt2 > best_sum:
        best_sum = s_opt2
        best_c = c_opt2.copy()
        
    # Phase 4: Perturb & Powell to escape shallow basins
    for _ in range(6):
        c_pert = best_c + rng.normal(0, 0.0015, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        try:
            res_p2 = minimize(lp_obj, c_pert.flatten(), method='Powell', bounds=bounds_c,
                             options={'maxiter': 700, 'ftol': 1e-12})
            c2 = res_p2.x.reshape(N, 2)
            _, s2 = solve_lp_and_grad(c2)
            if s2 > best_sum:
                best_sum = s2
                best_c = c2.copy()
        except Exception:
            pass

    # Final exact LP radii computation
    radii, final_sum = solve_lp_and_grad(best_c)
    
    # Strict numerical repair to guarantee validator compliance
    centers = best_c.copy()
    radii_final = radii.copy()
    
    for _ in range(80):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii_final[i] + radii_final[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii_final[i] -= shrink
                    radii_final[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii_final[i] > mr + 1e-12:
                radii_final[i] = mr
                changed = True
        if not changed:
            break
            
    radii_final = np.maximum(radii_final, 0.0)
    return centers, radii_final, float(np.sum(radii_final))
