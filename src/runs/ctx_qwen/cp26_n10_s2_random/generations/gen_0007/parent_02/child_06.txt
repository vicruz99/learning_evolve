import numpy as np

N = 26

# Precompute LP constraint matrix structure (reused for speed)
A_LP = np.zeros((N * (N - 1) // 2, N))
_ridx = 0
for _i in range(N):
    for _j in range(_i + 1, N):
        A_LP[_ridx, _i] = 1.0
        A_LP[_ridx, _j] = 1.0
        _ridx += 1


def solve_lp(centers):
    """Solves LP for maximum sum of radii given fixed centers."""
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-10)

    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    b = np.zeros(N * (N - 1) // 2)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            b[idx] = dists[i, j]
            idx += 1

    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b,
                  bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:
        return res.x.copy(), float(np.sum(res.x))
    return np.full(N, 1e-10), 0.0


def lp_objective(x):
    """Negative of optimal sum of radii for given centers."""
    _, s = solve_lp(x.reshape(N, 2))
    return -s


def safe_radii(centers, scale=0.93):
    """Compute strictly feasible initial radii for given centers."""
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)
    rp = 0.5 * np.min(dists, axis=1)
    return np.minimum(rb, rp) * scale


def hex_init(row_counts, r0, seed=0):
    """Generate hexagonal lattice pattern."""
    rng = np.random.default_rng(seed)
    pts = []
    y = r0
    for ri, cnt in enumerate(row_counts):
        shift = r0 if ri % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            pts.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    result = np.array(pts[:N])
    result = np.clip(result, 0.02, 0.98)
    return result

def force_repulsion_init(seed):
    """Generate well-spread configuration via force repulsion."""
    rng = np.random.default_rng(seed)
    c = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(800):
        diff = c[:, None, :] - c[None, :, :]
        dist = np.linalg.norm(diff, axis=2)
        dist_safe = np.maximum(dist, 1e-4)
        forces = np.sum(diff / (dist_safe ** 2)[:, :, None], axis=1)
        c += 0.002 * forces
        # Wall repulsion
        c = np.clip(c, 0.05, 0.95)
        margin = 0.08
        c[:, 0] = np.clip(c[:, 0], margin, 1 - margin)
        c[:, 1] = np.clip(c[:, 1], margin, 1 - margin)
    return c


def joint_objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2 * N:])


def joint_constraints(v):
    """Boundary and non-overlap constraints (>= 0). Uses squared distance to avoid sqrt singularity."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]

    cons = [
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r,
    ]

    i_idx, j_idx = np.triu_indices(N, 1)
    dx = c[i_idx, 0] - c[j_idx, 0]
    dy = c[i_idx, 1] - c[j_idx, 1]
    dr = r[i_idx] + r[j_idx]
    cons.append(dx ** 2 + dy ** 2 - dr ** 2)

    return np.concatenate(cons)


def boundary_push(centers, radii, strength=0.003):
    """Push centers toward boundaries to potentially grow radii."""
    c = centers.copy()
    r = radii.copy()

    for i in range(N):
        x, y, ri = c[i, 0], c[i, 1], r[i]
        # Push toward nearest wall
        dists_walls = [x, 1 - x, y, 1 - y]
        min_wall = min(dists_walls)
        idx_wall = dists_walls.index(min_wall)

        if min_wall < 0.4:  # Only push if reasonably close to a wall
            if idx_wall == 0:
                c[i, 0] -= strength
            elif idx_wall == 1:
                c[i, 0] += strength
            elif idx_wall == 2:
                c[i, 1] -= strength
            else:
                c[i, 1] += strength

    return np.clip(c, 0.02, 0.98)
