import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_bounds():
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0)])
    for _ in range(N):
        b.append((0.0, 0.5))
    return b

def constraints_func(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    i, j = np.triu_indices(N, 1)
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def objective_func(v):
    return -np.sum(v[2*N:])

def solve_lp(centers):
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    A_ub = np.zeros((N*(N-1)//2, N))
    b_ub = np.zeros(N*(N-1)//2)
    idx = 0
    for i in range(N):
        for j in range(i+1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0, u) for u in ub], method='highs')
    if res.success:
        return res.x, np.sum(res.x)
    return np.zeros(N), 0.0

def lp_obj(x):
    _, s = solve_lp(x.reshape(N, 2))
    return -s

def generate_starts(rng):
    starts = []
    patterns = [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], [6,5,5,5,5], [5,5,5,6,5], [6,6,4,5,5], [5,6,4,5,6]]
    for pat in patterns:
        for r0 in [0.09, 0.095, 0.10, 0.105]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
                    x += 2.0 * r0
                y += r0 * 1.7320508
            starts.append(np.array(c[:N]))
    for _ in range(15):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
    return starts

def repair(centers, radii):
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                req = radii[i] + radii[j]
                if d < req - 1e-10:
                    shrink = (req - d)/2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-10:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple:
    rng = np.random.default_rng(42)
    bounds = get_bounds()
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = generate_starts(rng)
    
    # Phase 1: SLSQP from multiple starts
    for c_init in starts:
        rb = np.minimum(np.minimum(c_init[:,0], 1-c_init[:,0]), np.minimum(c_init[:,1], 1-c_init[:,1]))
        dists = np.linalg.norm(c_init[:,None,:] - c_init[None,:, :], axis=2)
        np.fill_diagonal(dists, np.inf)
        rp = 0.5 * np.min(dists, axis=1)
        r_init = np.minimum(rb, rp) * 0.85
        
        v0 = np.concatenate([c_init.flatten(), r_init])
        
        for _ in range(2):
            v_curr = v0 + rng.normal(0, 0.0015, v0.shape)
            v_curr = np.clip(v_curr, 0.02, 0.98)
            v_curr[2*N:] = np.clip(v_curr[2*N:], 0.02, 0.4)
            
            try:
                res = minimize(objective_func, v_curr, method='SLSQP', bounds=bounds,
                              constraints={'type': 'ineq', 'fun': constraints_func},
                              options={'maxiter': 5000, 'ftol': 1e-13})
                if np.min(constraints_func(res.x)) >= -1e-7:
                    s = np.sum(res.x[2*N:])
                    if s > best_sum:
                        best_sum = s
                        best_c = res.x[:2*N].reshape(N, 2).copy()
                        best_r = res.x[2*N:].copy()
            except:
                pass
                
    # Phase 2: LP refinement & Powell center optimization
    if best_c is not None:
        r_lp, s_lp = solve_lp(best_c)
        if s_lp > best_sum:
            best_sum = s_lp
            best_r = r_lp
            
        try:
            res_p = minimize(lp_obj, best_c.flatten(), method='Powell', bounds=[(0,1)]*(2*N),

            c_p = res_p.x.reshape(N, 2)
            r_p, s_p = solve_lp(c_p)
            if s_p > best_sum:
                best_sum = s_p
                best_c = c_p
                best_r = r_p
        except:
            pass
            
        for _ in range(4):
            c_per = best_c + rng.normal(0, 0.005, best_c.shape)
            c_per = np.clip(c_per, 0.02, 0.98)
            try:
                res_pp = minimize(lp_obj, c_per.flatten(), method='Powell', bounds=[(0,1)]*(2*N),
                                 options={'maxiter': 800, 'ftol': 1e-12})
                c_pp = res_pp.x.reshape(N, 2)
                r_pp, s_pp = solve_lp(c_pp)
                if s_pp > best_sum:
                    best_sum = s_pp
                    best_c = c_pp
                    best_r = r_pp
            except:
                pass
                
        v0_polish = np.concatenate([best_c.flatten(), best_r])
        try:
            res_pol = minimize(objective_func, v0_polish, method='SLSQP', bounds=bounds,
                              constraints={'type': 'ineq', 'fun': constraints_func},
                              options={'maxiter': 3000, 'ftol': 1e-14})
            if np.min(constraints_func(res_pol.x)) >= -1e-8:
                s_pol = np.sum(res_pol.x[2*N:])
                if s_pol > best_sum:
                    best_c = res_pol.x[:2*N].reshape(N, 2).copy()
                    best_r = res_pol.x[2*N:].copy()
        except:
            pass
            
    centers = best_c.copy()
    radii = repair(centers, best_r.copy())
    return centers, radii, float(np.sum(radii))
