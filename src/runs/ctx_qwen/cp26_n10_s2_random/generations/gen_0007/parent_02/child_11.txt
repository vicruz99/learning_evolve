import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    n_pairs = N * (N - 1) // 2
    A_ub = np.zeros((n_pairs, N))
    b_ub = np.zeros(n_pairs)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:
        return res.x, np.sum(res.x)
    return np.zeros(N), 0.0

def lp_obj(centers_flat):
    """Objective for centers-only optimization: minimize negative LP sum of radii."""
    centers = centers_flat.reshape(N, 2)
    _, s = solve_lp_radii(centers)
    return -s

def joint_obj(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def joint_cons(v):
    """Computes boundary and non-overlap constraints (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    
    # Boundary constraints
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    # Overlap constraints using squared distances for smooth gradients
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    con.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(con)

def force_spread(centers, steps, step_size):
    """Spreads centers using repulsive forces to avoid clustering."""
    c = centers.copy()
    for _ in range(steps):
        f = np.zeros_like(c)
        diff = c[:, None, :] - c[None, :, :]
        dist = np.linalg.norm(diff, axis=2)
        dist = np.maximum(dist, 1e-4)
        force = diff / (dist**2)[:, :, None]
        f += np.sum(force, axis=1)
        f -= 0.5 * (c - 0.5)  # Attract to center to keep in bounds
        c += step_size * f
        c = np.clip(c, 0.05, 0.95)
    return c

def generate_inits(rng):
    """Generates diverse initial configurations for optimization."""
    inits = []
    
    # Hexagonal lattice patterns with varying row structures
    patterns = [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], [6,4,6,6,6]]
    for pat in patterns:
        for r_est in [0.09, 0.095, 0.10, 0.105]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * 1.7320508
            inits.append(np.array(c[:N]))
            
    # Perturbed grid patterns
    for s in range(5):
        gx, gy = np.meshgrid(np.linspace(0.1, 0.9, 6), np.linspace(0.1, 0.9, 6))
        c = np.column_stack((gx.flatten(), gy.flatten()))[:N]
        c += rng.normal(0, 0.02, c.shape)
        c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    # Force-spread random configurations
    for _ in range(5):
        c = rng.uniform(0.2, 0.8, (N, 2))
        c = force_spread(c, steps=300, step_size=0.008)
        inits.append(c)
        
    # Rotated lattice variants to escape symmetry traps
    for angle in [0.05, 0.15, -0.1]:
        c = inits[0].copy()
        c = c - 0.5
        rot = np.array([[np.cos(angle), -np.sin(angle)], [np.sin(angle), np.cos(angle)]])
        c = c @ rot.T
        c = c + 0.5
        c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    return inits

def run_packing() -> tuple:
    rng = np.random.default_rng(42)
    inits = generate_inits(rng)
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # --- Phase 1: Center Optimization via Powell ---
    for c_init in inits:
        x0 = c_init.flatten()
        try:
            res = minimize(lp_obj, x0, method='Powell',
                          options={'maxiter': 1000, 'ftol': 1e-12, 'xtol': 1e-12})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt = solve_lp_radii(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            pass
            
        # Local perturbations to escape shallow basins
        for scale in [0.005, 0.003, 0.001]:
            for _ in range(3):
                c_pert = c_opt + rng.normal(0, scale, c_opt.shape)
                c_pert = np.clip(c_pert, 0.02, 0.98)
                try:
                    res2 = minimize(lp_obj, c_pert.flatten(), method='Powell',
                                   options={'maxiter': 600, 'ftol': 1e-12})
                    c_opt2 = res2.x.reshape(N, 2)
                    r_opt2, s_opt2 = solve_lp_radii(c_opt2)
                    if s_opt2 > best_sum:
                        best_sum = s_opt2
                        best_c = c_opt2.copy()
                        best_r = r_opt2.copy()
                except Exception:
                    pass
                    
    # --- Phase 2: Joint SLSQP Polish ---
    if best_c is not None:
        v0 = np.concatenate([best_c.flatten(), best_r])
        bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        try:
            res_sl = minimize(joint_obj, v0, method='SLSQP', bounds=bounds,
                             constraints={'type': 'ineq', 'fun': joint_cons},
                             options={'maxiter': 8000, 'ftol': 1e-13})
            c_sl = res_sl.x[:2*N].reshape(N, 2)
            r_sl = res_sl.x[2*N:]
            c_check = joint_cons(res_sl.x)
            if np.min(c_check) >= -1e-7 and np.sum(r_sl) > best_sum:
                best_sum = np.sum(r_sl)
                best_c = c_sl
                best_r = r_sl
        except Exception:
            pass
            
    # Recompute exact radii for the best centers found
    if best_c is not None:
        r_exact, s_exact = solve_lp_radii(best_c)
        if s_exact > best_sum:
            best_sum = s_exact
            best_r = r_exact

    # --- Phase 3: Strict Numerical Repair ---
    centers = best_c.copy()
    radii = best_r.copy()
    
    for _ in range(60):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            mx = min(x, 1.0 - x, y, 1.0 - y)
            if r > mx + 1e-12:
                radii[i] = mx
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
