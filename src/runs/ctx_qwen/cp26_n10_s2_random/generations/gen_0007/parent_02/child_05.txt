N = 26

def solve_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    n_pairs = N * (N - 1) // 2
    A_ub = np.zeros((n_pairs, N))
    b_ub = np.zeros(n_pairs)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:
        return res.x, np.sum(res.x), res.ineqlin.marginals
    return np.zeros(N), 0.0, None


def compute_lp_gradient(centers):

        grad[i, 1] += mu_B - mu_T

def generate_hex_init(r_est, pat, rng, noise_scale=0.003):
    """Generate hexagonal initial configuration."""
    centers = []
    y = r_est
    for r_idx, cnt in enumerate(pat):
        shift = r_est if r_idx % 2 == 1 else 0.0
        x = r_est + shift
        for _ in range(cnt):
            if len(centers) < N:
                centers.append([x + rng.normal(0, noise_scale), y + rng.normal(0, noise_scale)])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3)
    centers = np.array(centers[:N])
    centers = np.clip(centers, 0.05, 0.95)
    return centers


def get_feasible_radii(centers):
    """Compute feasible initial radii for given centers."""
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    rp = 0.5 * np.min(dists, axis=1)
    return np.minimum(rb, rp) * 0.92


def force_repulsion_init(rng):
    """Generate configuration via force repulsion."""
    c = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(800):
        f = np.zeros_like(c)
        diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dist = np.linalg.norm(diff, axis=2)
        dist = np.maximum(dist, 0.02)
        weight = 1.0 / (dist ** 2)
        f += np.sum(diff * weight[:, :, np.newaxis], axis=1)
        c += 0.004 * f
        c = np.clip(c, 0.05, 0.95)
    return c


def joint_constraints(v):
    """Computes boundary and non-overlap constraints (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    cons = []
    
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    cons.append(np.sqrt(dx**2 + dy**2) - dr)
    
    return np.concatenate(cons)


def lp_objective_for_centers(x_flat):

    _, s, _ = solve_lp_radii(x_flat.reshape(N, 2))
    return -s
