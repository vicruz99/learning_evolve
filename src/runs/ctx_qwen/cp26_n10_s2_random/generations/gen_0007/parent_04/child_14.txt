from scipy.optimize import minimize, linprog

N = 26
TRIU_IDX = np.triu_indices(N, k=1)

def compute_slqp_obj(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def compute_slqp_cons(v):
    """Computes boundary and pairwise non-overlap constraints for SLSQP."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    c_pair = dx**2 + dy**2 - dr**2
    
    return np.concatenate([c, c_pair[TRIU_IDX]])

def solve_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    np.fill_diagonal(dists, np.inf)
    
    M = n * (n - 1) // 2 + 4 * n
    A_ub = np.zeros((M, n))
    b_ub = np.zeros(M)
    idx = 0

            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        x, y = centers[i]
        b_ub[idx] = x; idx += 1
        b_ub[idx] = 1.0 - x; idx += 1
        b_ub[idx] = y; idx += 1
        b_ub[idx] = 1.0 - y; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return res.x if res.success else np.zeros(n)

def lp_obj_func(x_flat):
    """Objective for derivative-free optimization: negative LP sum of radii."""
    c = np.clip(x_flat.reshape(N, 2), 1e-5, 1.0 - 1e-5)
    r = solve_lp(c)
    return -np.sum(r)

    centers = []
    y = r0
    for r_idx, count in enumerate(rows):
