

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we need to escape local optima that plague gradient-based methods. The problem can be decoupled: for fixed centers, finding the optimal radii is a Linear Programming (LP) problem. This allows us to use a global optimizer on the centers while always knowing the exact maximum sum of radii achievable.

Strategy:
1. **LP for Radii**: Precompute the constraint matrix structure. For any given center configuration, solve a tiny LP to find the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints.
2. **Global Search (DE)**: Use Differential Evolution to optimize the 52 center coordinates. The objective is the negative LP-optimal sum of radii. This effectively searches for center configurations that allow the largest circles.
3. **Local Refinement (Powell)**: After DE finds a promising basin, use Powell's method (derivative-free) to fine-tune the centers, as the LP objective landscape is non-smooth.
4. **Joint Polish (SLSQP)**: Extract the optimal radii from the LP and perform a constrained joint optimization (centers + radii) using SLSQP to squeeze out microscopic improvements where gradients can help.
5. **Strict Repair**: Iteratively shrink radii to guarantee the solution passes the validation function's $10^{-12}$ tolerance strictly.

This hybrid approach combines the robustness of LP, the global reach of DE, and the precision of SLSQP, systematically pushing beyond previous local maxima.
</strategy>

```python
import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

N = 26
TRIU = np.triu_indices(N, k=1)
NUM_PAIRS = N*(N-1)//2
NUM_BND = 4*N

# Precompute the inequality constraint matrix structure for the LP
# Each row corresponds to r_i + r_j <= dist(i,j) or r_i <= boundary_dist
A_ub = np.zeros((NUM_PAIRS + NUM_BND, N))
idx = 0
for i, j in zip(TRIU[0], TRIU[1]):
    A_ub[idx, i] = 1.0
    A_ub[idx, j] = 1.0
    idx += 1
for i in range(N):
    for _ in range(4):
        A_ub[idx, i] = 1.0
        idx += 1

def solve_lp_fast(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    # Compute pairwise distances
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Fill RHS vector b_ub
    b_ub = np.empty(NUM_PAIRS + NUM_BND)
    b_ub[:NUM_PAIRS] = dists[TRIU]
    b_ub[NUM_PAIRS : NUM_PAIRS+N] = centers[:, 0]
    b_ub[NUM_PAIRS+N : 2*NUM_PAIRS+N] = 1.0 - centers[:, 0]
    b_ub[2*NUM_PAIRS+N : 3*NUM_PAIRS+N] = centers[:, 1]
    b_ub[3*NUM_PAIRS+N : ] = 1.0 - centers[:, 1]
    
    # Minimize -sum(r) => Maximize sum(r)
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.fun, res.x
    return -1.0, np.zeros(N)

def obj_de(x_flat):
    """Objective for Differential Evolution: minimize negative optimal sum of radii."""
    c = np.clip(x_flat.reshape(N, 2), 1e-5, 1.0 - 1e-5)
    neg_sum, _ = solve_lp_fast(c)
    return neg_sum

def slsqp_obj(v):
    """Objective for joint SLSQP optimization."""
    return -np.sum(v[2*N:])

def slsqp_cons(v):
    """Boundary and non-overlap constraints for SLSQP (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    
    # Boundary constraints
    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    
    # Pairwise non-overlap constraints
    diff = c[:, None, :] - c[None, :, :]
    d = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(d, np.inf)
    con = np.concatenate([con, d[TRIU] - (r[TRIU[0]] + r[TRIU[1]])])
    return con

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds_de = [(0.05, 0.95)] * (2 * N)
    
    # Phase 1: Global search on centers using Differential Evolution
    res_de = differential_evolution(obj_de, bounds_de, popsize=30, maxiter=150, 
                                    mutation=0.9, recombination=0.7, 
                                    seed=42, workers=1, tol=1e-8, polish=False)
    best_centers = res_de.x.reshape(N, 2)
    
    # Phase 2: Local refinement on centers using Powell's method
    res_powell = minimize(obj_de, best_centers.flatten(), method='Powell', 
                          options={'maxiter': 3000, 'ftol': 1e-13, 'xtol': 1e-13})
    best_centers = res_powell.x.reshape(N, 2)
    
    # Phase 3: Compute exact optimal radii via Linear Programming
    _, best_radii = solve_lp_fast(best_centers)
    
    # Phase 4: Joint SLSQP polish for final micro-adjustments
    x0 = np.concatenate([best_centers.flatten(), best_radii])
    bounds_sl = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    try:
        res_sl = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds_sl,
                          constraints={'type': 'ineq', 'fun': slsqp_cons},
                          options={'maxiter': 5000, 'ftol': 1e-13})
        if -res_sl.fun > np.sum(best_radii) - 1e-6:
            best_centers = res_sl.x[:2*N].reshape(N, 2)
            best_radii = res_sl.x[2*N:]
    except Exception:
        pass
        
    # Phase 5: Deterministic repair to strictly satisfy validator tolerance
    centers = best_centers
    radii = best_radii
    for _ in range(50):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i]+radii[j]-1e-10:
                    sh = (radii[i]+radii[j]-d)/2.0 + 1e-10
                    radii[i] -= sh
                    radii[j] -= sh
                    changed = True
        # Resolve boundary violations
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-10:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
```
