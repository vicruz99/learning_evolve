from scipy.optimize import minimize

N = 26
# Precompute pairwise indices for constraint evaluation
MASK_I, MASK_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        x - r, 1.0 - x - r,
        y - r, 1.0 - y - r
    ])

    dx = x[MASK_I] - x[MASK_J]
    dy = y[MASK_I] - y[MASK_J]
    dr = r[MASK_I] + r[MASK_J]
    dist = np.hypot(dx, dy)
    c = np.concatenate([c, dist - dr])
    return c

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

    y = r0

            pts.append([xs, y])
            xs += 2.0 * r0

        row += 1
    pts = np.array(pts[:N])
    if perturb > 0:
        pts += rng.normal(0, perturb, pts.shape)
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
    return pts
