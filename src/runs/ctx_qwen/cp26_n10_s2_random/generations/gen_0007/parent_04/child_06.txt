import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

N = 26
N_PAIRS = N * (N - 1) // 2

# Precompute the constant inequality matrix for the LP
# Constraints: r_i + r_j <= d_ij  (N_PAIRS constraints)
#              r_i <= b_i         (N boundary constraints)
A_UB = np.zeros((N_PAIRS + N, N))
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_UB[idx, i] = 1.0
        A_UB[idx, j] = 1.0
        idx += 1
for i in range(N):
    A_UB[idx, i] = 1.0
    idx += 1

TRIU_INDICES = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    b_ub = np.zeros(N_PAIRS + N)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            b_ub[idx] = dists[i, j]
            idx += 1
    for i in range(N):
        b_ub[idx] = min(centers[i, 0], 1.0 - centers[i, 0], 
                        centers[i, 1], 1.0 - centers[i, 1])
        idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_UB, b_ub=b_ub, bounds=(0, None), method='highs')
    return res.x if res.success else np.full(N, 0.01)

def obj_centers(x_flat):
    """Objective for center optimization: minimize negative sum of LP-optimal radii."""
    centers = x_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    radii = solve_lp_radii(centers)
    return -np.sum(radii)

def init_hex(row_counts, r0, rng):
    """Generates an initial configuration from a perturbed hexagonal lattice."""
    centers = []
    y = r0
    for row_idx, count in enumerate(row_counts):
        offset = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + offset
        for _ in range(count):
            centers.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(centers[:N])

def compute_slsqp_obj(v):
    """Objective for joint SLSQP optimization."""
    return -np.sum(v[2 * N:])

def compute_slsqp_cons(v):
    """Computes inequality constraints for SLSQP (must be >= 0)."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    
    # Boundary constraints
    out = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    
    # Pairwise non-overlap (squared distance for smoother gradients)
    dx = c[:, 0, np.newaxis] - c[:, np.newaxis, 0]
    dy = c[:, 1, np.newaxis] - c[:, np.newaxis, 1]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    out.append((dx**2 + dy**2 - dr**2)[mask])
    return np.concatenate(out)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize sum of radii."""
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_centers = None
    best_sum = -1e9
    
    # Phase 1: Global search with Differential Evolution
    bounds_de = [(0.02, 0.98)] * (2 * N)
    res_de = differential_evolution(obj_centers, bounds_de, popsize=10, maxiter=80, 
                                    seed=42, workers=1, tol=1e-7, polish=False)
    best_centers = res_de.x.reshape(N, 2)
    best_sum = -res_de.fun

                          options={'maxiter': 5000, 'ftol': 1e-12, 'xtol': 1e-12})
    if -res_powell.fun > best_sum:
        best_centers = res_powell.x.reshape(N, 2)
        best_sum = -res_powell.fun
        
    # Phase 3: Multi-start from diverse hex patterns
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [5, 7, 5, 5, 4], [4, 6, 6, 6, 4],

                           options={'maxiter': 15000, 'fatol': 1e-12, 'xatol': 1e-12})

                best_sum = -res.fun

            best_radii = res.x[2 * N:]
            final_sum = np.sum(best_radii)
    except Exception:
        pass

        changed = False

            for j in range(i + 1, N):

                    sh = (best_radii[i] + best_radii[j] - d) / 2.0 + 1e-11

                    best_radii[j] -= sh
                    changed = True

            mx = min(best_centers[i, 0], 1.0 - best_centers[i, 0], 
                     best_centers[i, 1], 1.0 - best_centers[i, 1])
            if best_radii[i] > mx + 1e-11:
                best_radii[i] = mx
                changed = True
                
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
