import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
TRIU_IND = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    n_pairs = n * (n - 1) // 2
    n_bnd = 4 * n
    A_ub = np.zeros((n_pairs + n_bnd, n))
    b_ub = np.zeros(n_pairs + n_bnd)
    
    idx = 0
    i_idx, j_idx = TRIU_IND
    for ii, jj in zip(i_idx, j_idx):
        A_ub[idx, ii] = 1.0
        A_ub[idx, jj] = 1.0
        b_ub[idx] = np.hypot(centers[ii, 0] - centers[jj, 0], centers[ii, 1] - centers[jj, 1])
        idx += 1
        
    for i in range(n):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def obj_centers(x):
    """Objective for center optimization: minimize negative sum of radii."""
    return -solve_lp_radii(x.reshape(N, 2))[1]

def joint_obj(v):
    """Objective for joint SLSQP optimization."""
    return -np.sum(v[2 * N:])

def joint_cons(v):
    """Computes inequality constraints for SLSQP (must be >= 0)."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    
    dx = c[:, 0][:, None] - c[:, 0][None, :]
    dy = c[:, 1][:, None] - c[:, 1][None, :]
    dr = r[:, None] + r[None, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    con = np.concatenate([con, np.hypot(dx[mask], dy[mask]) - dr[mask]])
    return con

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # Phase 1: Global search with Differential Evolution on centers
    bounds_de = [(0.05, 0.95)] * (2 * N)
    res_de = differential_evolution(obj_centers, bounds_de, popsize=15, maxiter=150, 
                                    seed=42, tol=1e-8, polish=False, workers=1)
    best_centers = res_de.x.reshape(N, 2)
    
    # Phase 2: Local refinement with Powell's method
    res_powell = minimize(obj_centers, best_centers.flatten(), method='Powell', 
                          options={'maxiter': 3000, 'ftol': 1e-12, 'xtol': 1e-12})
    best_centers = res_powell.x.reshape(N, 2)
    best_radii, best_sum = solve_lp_radii(best_centers)
    
    # Phase 3: Joint SLSQP polish (centers + radii) for tight contact convergence
    x0 = np.concatenate([best_centers.flatten(), best_radii])
    bounds_slsqp = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(joint_obj, x0, method='SLSQP', bounds=bounds_slsqp, 
                       constraints={'type': 'ineq', 'fun': joint_cons},
                       options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
        if -res.fun > best_sum - 1e-6:
            best_centers = res.x[:2*N].reshape(N, 2)
            best_radii = res.x[2*N:]
            best_sum = -res.fun
    except Exception:
        pass
        
    # Phase 4: Adaptive perturbation search to escape local minima
    rng = np.random.default_rng(123)
    for _ in range(60):
        c_trial = best_centers.copy()
        c_trial += rng.normal(0, 0.0025, c_trial.shape)
        c_trial = np.clip(c_trial, 0.02, 0.98)
        
        _, s_trial = solve_lp_radii(c_trial)
        if s_trial > best_sum + 1e-7:
            best_sum = s_trial
            best_centers = c_trial.copy()
            best_radii, _ = solve_lp_radii(best_centers)
            
            # Quick local refine from new promising point

            best_radii, best_sum = solve_lp_radii(best_centers)

    final_radii, _ = solve_lp_radii(best_centers)

                    final_radii[i] -= shrink
                    final_radii[j] -= shrink
                    changed = True

            mr = min(best_centers[i, 0], 1.0 - best_centers[i, 0], 
                     best_centers[i, 1], 1.0 - best_centers[i, 1])
            if final_radii[i] > mr + 1e-12:
                final_radii[i] = mr
                changed = True
        if not changed:
            break
            
    final_radii = np.maximum(final_radii, 0.0)
    return best_centers, final_radii, float(np.sum(final_radii))
