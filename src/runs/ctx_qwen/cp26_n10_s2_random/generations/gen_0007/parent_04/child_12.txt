N = 26
MASK = np.triu_indices(N, k=1)

_LP_A = np.zeros((N*(N-1)//2 + 4*N, N))
_idx = 0
for i in range(N):
    for j in range(i+1, N):
        _LP_A[_idx, i] = 1.0
        _LP_A[_idx, j] = 1.0
        _idx += 1

def compute_lp_radii(centers):

    n = centers.shape[0]
    b = np.zeros(N*(N-1)//2 + 4*n)
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            b[idx] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            idx += 1
    for i in range(n):
        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1

    if res.success:
        return res.x
    return np.full(n, 0.01)

def obj_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def cons_joint(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    return np.concatenate([c, (dx**2 + dy**2 - dr**2)[MASK]])

            centers.append([x, y])
            x += 2.0 * r0

    centers = np.array(centers[:N])
