import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def get_pairwise_indices():
    return np.triu_indices(N, k=1)

PAIR_INDICES = get_pairwise_indices()
NUM_PAIRS = len(PAIR_INDICES[0])

def build_A_ub():
    M = NUM_PAIRS + 4 * N
    A = np.zeros((M, N))
    idx = 0
    for i, j in zip(PAIR_INDICES[0], PAIR_INDICES[1]):
        A[idx, i] = 1.0
        A[idx, j] = 1.0
        idx += 1
    for i in range(N):
        A[idx, i] = 1.0; idx += 1
        A[idx, i] = 1.0; idx += 1
        A[idx, i] = 1.0; idx += 1
        A[idx, i] = 1.0; idx += 1
    return A

A_UB = build_A_ub()

def solve_lp_and_grad(centers):
    b = np.zeros(NUM_PAIRS + 4 * N)
    idx = 0
    for i, j in zip(PAIR_INDICES[0], PAIR_INDICES[1]):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b[idx] = np.hypot(dx, dy)
        idx += 1
    for i in range(N):
        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1

    res = linprog(-np.ones(N), A_ub=A_UB, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        return None, 0.0, np.zeros_like(centers)

    radii = res.x
    duals = np.asarray(res.ineqlin.marginals).ravel()

    grad = np.zeros_like(centers)
    idx = 0
    for i, j in zip(PAIR_INDICES[0], PAIR_INDICES[1]):
        lam = max(0.0, duals[idx])
        if lam > 1e-9:
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.hypot(dx, dy)
            if d > 1e-12:
                fx = lam * dx / d
                fy = lam * dy / d
                grad[i, 0] += fx
                grad[i, 1] += fy
                grad[j, 0] -= fx
                grad[j, 1] -= fy
        idx += 1

    for i in range(N):
        mu_x = max(0.0, duals[idx])
        mu_1x = max(0.0, duals[idx+1])
        mu_y = max(0.0, duals[idx+2])
        mu_1y = max(0.0, duals[idx+3])
        grad[i, 0] += mu_x - mu_1x
        grad[i, 1] += mu_y - mu_1y
        idx += 4

    return radii, np.sum(radii), grad

def init_hex(row_counts, r0, rng):
    centers = np.zeros((N, 2))
    y = r0
    count = 0
    for row_idx, cnt in enumerate(row_counts):
        offset = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + offset
        for _ in range(cnt):
            if count < N:
                centers[count] = [x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)]
                x += 2.0 * r0
                count += 1
        y += np.sqrt(3) * r0
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_centers = np.zeros((N, 2))
    best_radii = np.zeros(N)
    best_sum = 0.0

    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 5, 5, 5, 5], [5, 7, 5, 5, 4], [4, 6, 6, 6, 4],
        [5, 5, 6, 5, 5], [6, 4, 6, 5, 5]
    ]

    for pat in patterns:
        for r0 in [0.085, 0.090, 0.095, 0.100, 0.105, 0.110]:
            c = init_hex(pat, r0, rng)
            c = np.clip(c, 0.05, 0.95)
            
            curr_c = c.copy()
            step = 0.005
            noise_scale = 0.0015
            
            for it in range(2000):
                radii, s, grad = solve_lp_and_grad(curr_c)
                if s > best_sum:
                    best_sum = s
                    best_centers = curr_c.copy()
                    best_radii = radii.copy()
                
                g_norm = np.linalg.norm(grad)
                if g_norm > 1e-12:
                    curr_c += step * grad / g_norm
                
                noise = rng.normal(0, noise_scale * (1.0 - 0.6 * it / 2000.0), curr_c.shape)
                curr_c += noise
                curr_c = np.clip(curr_c, 1e-5, 1.0 - 1e-5)
                
                step *= 0.9992
                noise_scale *= 0.9995

    def obj(v):
        return -np.sum(v[2::3])

    def cons(v):
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
        dx = x[:, None] - x[None, :]
        dy = y[:, None] - y[None, :]
        dr = r[:, None] + r[None, :]
        return np.concatenate([c, (dx**2 + dy**2 - dr**2)[PAIR_INDICES]])

    v0 = np.zeros(3 * N)
    v0[0::3] = best_centers[:, 0]
    v0[1::3] = best_centers[:, 1]
    v0[2::3] = best_radii
    
    try:
        res = minimize(obj, v0, method='SLSQP', 
                       bounds=[(0.0, 1.0)] * (2 * N) + [(1e-7, 0.5)] * N,
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        if -res.fun > best_sum - 1e-8:
            best_centers = res.x[:2 * N].reshape(N, 2)
            best_radii = res.x[2 * N:]
            best_sum = -res.fun
    except Exception:
        pass

    radii = np.maximum(best_radii, 0.0)
    for _ in range(100):
        changed = False
        for i in range(N):
            mr = min(best_centers[i, 0], 1.0 - best_centers[i, 0], 
                     best_centers[i, 1], 1.0 - best_centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                if d < radii[i] + radii[j] - 1e-11:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-11
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break

    return best_centers, radii, float(np.sum(radii))
