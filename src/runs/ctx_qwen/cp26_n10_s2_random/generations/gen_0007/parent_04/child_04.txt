import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

N = 26
PAIR_INDICES = np.triu_indices(N, k=1)
n_pairs = len(PAIR_INDICES[0])

# Precompute LP constraint matrix structure (constant)
A_ub_pairs = np.zeros((n_pairs, N))
for idx, (i, j) in enumerate(zip(*PAIR_INDICES)):
    A_ub_pairs[idx, i] = 1.0
    A_ub_pairs[idx, j] = 1.0

A_ub_bounds = np.zeros((4 * N, N))
for i in range(N):
    A_ub_bounds[4 * i, i] = 1.0
    A_ub_bounds[4 * i + 1, i] = 1.0
    A_ub_bounds[4 * i + 2, i] = 1.0
    A_ub_bounds[4 * i + 3, i] = 1.0

A_ub = np.vstack([A_ub_pairs, A_ub_bounds])

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    b_ub = np.empty(n_pairs + 4 * N)
    b_ub[:n_pairs] = dists[PAIR_INDICES]
    
    cx, cy = centers[:, 0], centers[:, 1]
    idx = n_pairs
    for i in range(N):
        b_ub[idx] = cx[i]; idx += 1
        b_ub[idx] = 1.0 - cx[i]; idx += 1
        b_ub[idx] = cy[i]; idx += 1
        b_ub[idx] = 1.0 - cy[i]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(N), 0.0

def obj_centers_lp(x_flat):
    """Objective for center optimization: maximize LP-optimal sum of radii."""
    c = x_flat.reshape(N, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    _, s = solve_lp_radii(c)
    return -s

def compute_joint_cons(v):
    """Computes boundary and pairwise non-overlap constraints for joint optimization."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    c.append((dx**2 + dy**2 - dr**2)[PAIR_INDICES])
    return np.concatenate(c)

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds_de = [(0.05, 0.95)] * (2 * N)
    
    # Phase 1: Global search on centers using Differential Evolution
    res_de = differential_evolution(obj_centers_lp, bounds_de, popsize=20, maxiter=80, 
                                    seed=42, workers=1, tol=1e-7, polish=False)
    best_centers = res_de.x.reshape(N, 2)
    
    # Phase 2: Local refinement with Powell's method on the LP objective
    res_powell = minimize(obj_centers_lp, best_centers.flatten(), method='Powell', 
                          options={'maxiter': 5000, 'ftol': 1e-12, 'xtol': 1e-12})
    if -res_powell.fun < -res_de.fun:
        best_centers = res_powell.x.reshape(N, 2)
        
    # Phase 3: Compute exact optimal radii via LP
    radii, _ = solve_lp_radii(best_centers)
    
    # Phase 4: Joint SLSQP polish (centers + radii) for micro-adjustments
    x0_slsqp = np.concatenate([best_centers.flatten(), radii])
    bounds_slsqp = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    try:
        res_slsqp = minimize(obj_joint, x0_slsqp, method='SLSQP', bounds=bounds_slsqp,
                             constraints={'type': 'ineq', 'fun': compute_joint_cons},
                             options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
        if -res_slsqp.fun > np.sum(radii) - 1e-6:
            best_centers = res_slsqp.x[:2 * N].reshape(N, 2)
            radii = res_slsqp.x[2 * N:]
    except Exception:
        pass
        
    # Phase 5: Final LP to ensure radii are mathematically optimal for final centers
    radii, _ = solve_lp_radii(best_centers)
    
    # Phase 6: Strict numerical repair to guarantee validation tolerance compliance
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mx = min(best_centers[i, 0], 1.0 - best_centers[i, 0], 
                     best_centers[i, 1], 1.0 - best_centers[i, 1])
            if radii[i] > mx + 1e-12:
                radii[i] = mx
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return best_centers, radii, float(np.sum(radii))
