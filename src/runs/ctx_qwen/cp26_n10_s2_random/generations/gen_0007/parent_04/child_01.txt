import numpy as np
from scipy.optimize import minimize, linprog

N = 26
# Precompute upper triangular indices for pairwise constraints
MASK = np.triu_indices(N, k=1)

def compute_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def compute_cons(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    return np.concatenate([c, (dx**2 + dy**2 - dr**2)[MASK]])

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Solves LP to find exact maximum sum of radii for fixed centers."""
    # Boundary limits
    b_lim = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # LP formulation: max sum(r_i) s.t. r_i + r_j <= d_ij, r_i <= b_lim_i, r_i >= 0
    n_pairs = N * (N - 1) // 2
    n_bnd = N
    A_ub = np.zeros((n_pairs + n_bnd, N))
    idx = 0
    for i, j in zip(MASK[0], MASK[1]):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        idx += 1
    A_ub[idx:, :] = np.eye(N)
    
    b_ub = np.concatenate([dists[MASK], b_lim])
    
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return res.x if res.success else np.full(N, 0.05)

def generate_hex_start(pat, r0, rng):
    """Generates a perturbed hexagonal lattice configuration."""
    centers = []
    y = r0
    for ri, cnt in enumerate(pat):
        offset = r0 if ri % 2 == 1 else 0.0
        x = r0 + offset
        for _ in range(cnt):
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        
    centers = np.array(centers[:N])
    centers += rng.normal(0, 0.002, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_cons}
    rng = np.random.default_rng(42)
    
    best_v = None
    best_sum = -1e9
    
    # Phase 1: Diverse initial configurations
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [6, 6, 4, 6, 4], [4, 6, 6, 6, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [5, 7, 4, 5, 5]
    ]
    
    for pat in patterns:
        for r0 in [0.095, 0.098, 0.100, 0.102, 0.105, 0.108]:
            c = generate_hex_start(pat, r0, rng)
            r_init = solve_lp_radii(c)
            v = np.zeros(3 * N)
            v[0::3] = c[:, 0]
            v[1::3] = c[:, 1]
            v[2::3] = r_init
            starts.append(v)
            
    # Add random starts for diversity
    for _ in range(12):
        c = rng.uniform(0.15, 0.85, (N, 2))
        r_init = solve_lp_radii(c)
        v = np.zeros(3 * N)
        v[0::3] = c[:, 0]
        v[1::3] = c[:, 1]
        v[2::3] = r_init
        starts.append(v)
        
    # Phase 2: Multi-start SLSQP optimization
    for v0 in starts:
        try:
            res = minimize(compute_obj, v0, method='SLSQP', bounds=bounds,

            s_val = -res.fun
            if s_val > best_sum:
                c_vals = compute_cons(res.x)
                if np.min(c_vals) >= -1e-6:
                    best_sum = s_val
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Adaptive perturbation search to escape local minima
    if best_v is not None:
        for step in range(120):
            scale = 0.006 * (0.955 ** step)
            v_trial = best_v.copy()
            v_trial[0::3] += rng.normal(0, scale, N)
            v_trial[1::3] += rng.normal(0, scale, N)
            v_trial = np.clip(v_trial, 0.01, 0.99)
            v_trial[2::3] = np.clip(v_trial[2::3], 0.0, 0.5)

            except Exception:
                pass

    centers = best_v.reshape(N, 3)[:, :2]

    for _ in range(50):

        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Resolve boundary violations
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-10:
                radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
