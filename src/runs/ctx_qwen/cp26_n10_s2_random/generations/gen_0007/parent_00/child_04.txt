from scipy.optimize import minimize, linprog, differential_evolution

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
PAIR_COUNT = len(IDX_I)

# Precompute static LP constraint matrix for pairwise distances: r_i + r_j <= dist_ij
A_LP = np.zeros((PAIR_COUNT, N))
_lp_idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[_lp_idx, i] = 1.0
        A_LP[_lp_idx, j] = 1.0
        _lp_idx += 1

def solve_lp_for_radii(centers):

    Given fixed centers, solve LP to exactly maximize sum of radii.
    Constraints: r_i + r_j <= dist(i,j), 0 <= r_i <= dist_to_boundary
    """

    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-7)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]

    return np.full(n, 1e-7), 26e-7

    c = x.reshape(N, 2)
    _, s = solve_lp_for_radii(c)
    return -s

    return -np.sum(v[2*N:])

    Computes all boundary and non-overlap constraints for SLSQP.
    Returns a 1D array where each element must be >= 0.

    out = []
    
    # Boundary constraints
    out.append(c[:, 0] - r)
    out.append(1.0 - c[:, 0] - r)
    out.append(c[:, 1] - r)
