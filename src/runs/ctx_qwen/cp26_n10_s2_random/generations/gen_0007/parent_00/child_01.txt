import numpy as np
from scipy.optimize import minimize, linprog

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
PAIR_COUNT = len(IDX_I)

# Precompute static LP constraint matrix for pairwise distances: r_i + r_j <= dist_ij
A_LP = np.zeros((PAIR_COUNT, N))
for k in range(PAIR_COUNT):
    i, j = IDX_I[k], IDX_J[k]
    A_LP[k, i] = 1.0
    A_LP[k, j] = 1.0

def solve_lp_and_grad(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns radii, sum_radii, and gradient of sum_radii w.r.t centers.
    """
    n = centers.shape[0]
    
    # Upper bounds from square boundaries
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    b_ub = dists[IDX_I, IDX_J]
    
    # Solve LP: max sum(r) s.t. r_i + r_j <= dist_ij, 0 <= r_i <= ub_i
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(n), 0.0, np.zeros_like(centers)
        
    radii = res.x
    
    # Compute gradient using dual variables (shadow prices)
    grad = np.zeros_like(centers)
    duals = np.abs(res.ineqlin.marginals)  # Shadow prices are non-negative
    
    # Pairwise constraint gradients
    for k in range(PAIR_COUNT):
        lam = duals[k]
        if lam > 1e-10:
            i, j = IDX_I[k], IDX_J[k]
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
                
    # Boundary constraint gradients (computed heuristically from active status)
    for i in range(n):
        if radii[i] > 0.999 * ub[i]:
            # Active boundary constraint
            if radii[i] > 0.999 * centers[i, 0]:
                grad[i, 0] += 1.0
            if radii[i] > 0.999 * (1.0 - centers[i, 0]):
                grad[i, 0] -= 1.0
            if radii[i] > 0.999 * centers[i, 1]:
                grad[i, 1] += 1.0
            if radii[i] > 0.999 * (1.0 - centers[i, 1]):
                grad[i, 1] -= 1.0
                
    return radii, np.sum(radii), grad

def objective_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints_joint(v):
    """Computes all boundary and non-overlap constraints for SLSQP."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    cons = []
    
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap: dist >= r_i + r_j
    dx = c[IDX_I, 0] - c[IDX_J, 0]
    dy = c[IDX_I, 1] - c[IDX_J, 1]
    cons.append(np.hypot(dx, dy) - (r[IDX_I] + r[IDX_J]))
    
    return np.concatenate(cons)

def generate_starts(rng):
    """Generates diverse initial center configurations."""
    starts = []
    
    # 1. Hexagonal lattice patterns with varying row structures
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [6,4,6,5,5],
        [4,6,6,6,4], [5,4,6,6,5], [6,6,5,5,4], [5,5,6,5,5],
        [4,5,6,5,6], [5,5,5,6,5], [5,5,4,6,6], [6,5,5,5,5],
        [5,5,5,5,6], [6,6,4,5,5], [5,6,4,5,6], [6,5,5,6,4],
        [5,7,5,5,4], [5,5,7,5,4], [6,6,6,4,4], [5,5,5,7,4]
    ]
    
    for pat in patterns:
        for r_est in [0.09, 0.095, 0.10, 0.105]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    # 2. Force-directed layouts to encourage natural spacing
    for _ in range(30):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(100):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c[i] - c[j]
                    d = np.linalg.norm(d_vec)
                    if d < 0.25 and d > 1e-6:
                        f = (0.25 - d) * 0.8 / d
                        forces[i] += d_vec * f
                        forces[j] -= d_vec * f
            c += forces * 0.015
        c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    starts = generate_starts(rng)
    
    bounds_j = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_joint}
    
    # Phase 1: Multi-start SLSQP + LP refinement
    for c_init in starts:
        r_init = np.full(N, 0.07)
        v0 = np.concatenate([c_init.flatten(), r_init])
        
        try:
            res = minimize(objective_joint, v0, method='SLSQP', bounds=bounds_j,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
            if hasattr(res, 'x'):
                c_curr = res.x[:2*N].reshape(N, 2)
                r_lp, s_lp = solve_lp_and_grad(c_curr)[0:2]
                
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c_curr.copy()
                    best_radii = r_lp.copy()
        except Exception:
            pass

    # Fallback
    if best_centers is None:
        c0 = rng.uniform(0.2, 0.8, (N, 2))
        best_radii, best_sum = solve_lp_and_grad(c0)[0:2]
        best_centers = c0

    # Phase 2: Gradient Ascent on Centers using LP Duals
    curr_centers = best_centers.copy()
    step_size = 0.002
    for _ in range(150):
        _, curr_sum, grad = solve_lp_and_grad(curr_centers)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-9:
            curr_centers += step_size * grad / g_norm
        curr_centers = np.clip(curr_centers, 1e-5, 1.0 - 1e-5)
        step_size *= 0.997
        
    # Check if gradient ascent improved
    r_lp, s_lp = solve_lp_and_grad(curr_centers)[0:2]
    if s_lp > best_sum:
        best_sum = s_lp
        best_centers = curr_centers.copy()
        best_radii = r_lp.copy()

    # Phase 3: Final SLSQP Polish from improved centers
    r_init = best_radii * 0.95
    v0 = np.concatenate([best_centers.flatten(), r_init])
    try:
        res = minimize(objective_joint, v0, method='SLSQP', bounds=bounds_j,
                       constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
        if hasattr(res, 'x'):
            c_curr = res.x[:2*N].reshape(N, 2)
            r_lp, s_lp = solve_lp_and_grad(c_curr)[0:2]
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = c_curr.copy()
                best_radii = r_lp.copy()
    except Exception:
        pass

    # Phase 4: Strict Numerical Repair
    for _ in range(50):
        changed = False
        # Fix boundary violations
        for i in range(N):
            mx = min(best_centers[i,0], 1.0-best_centers[i,0], 
                     best_centers[i,1], 1.0-best_centers[i,1])
            if best_radii[i] > mx - 1e-12:
                best_radii[i] = mx
                changed = True
        # Fix overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], 
                             best_centers[i,1]-best_centers[j,1])
                if d < best_radii[i] + best_radii[j] - 1e-12:
                    shrink = (best_radii[i] + best_radii[j] - d) / 2.0 + 1e-9
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
