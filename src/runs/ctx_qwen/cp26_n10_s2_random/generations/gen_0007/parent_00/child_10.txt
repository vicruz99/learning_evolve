

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we use a decoupled optimization strategy:
1. **LP for Radii**: For any fixed set of centers, the maximum sum of radii is found exactly via Linear Programming (`scipy.optimize.linprog`). This transforms the non-convex joint optimization into a smooth optimization over centers only.
2. **Analytical Gradient**: We compute the gradient of the sum of radii w.r.t. centers using LP dual variables (shadow prices). This allows L-BFGS-B to converge rapidly without expensive finite-difference approximations.
3. **Multi-Start & Perturbation**: We initialize from diverse configurations (hexagonal lattices, corner-biased layouts, force-relaxed random positions) and run L-BFGS-B from each. After finding a local optimum, we apply controlled perturbations to escape shallow basins.
4. **Strict Repair**: A final deterministic pass ensures all circles strictly satisfy boundary and non-overlap constraints within the validator's tolerance.

This approach efficiently navigates the high-dimensional constraint landscape and pushes the sum of radii beyond previous local optima.
</strategy>

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
PAIR_COUNT = len(IDX_I)

# Precompute constant LP constraint matrix A_ub: r_i + r_j <= d_ij and r_i <= boundary
A_LP = np.zeros((PAIR_COUNT + 4 * N, N))
for k in range(PAIR_COUNT):
    A_LP[k, IDX_I[k]] = 1.0
    A_LP[k, IDX_J[k]] = 1.0
for i in range(N):
    A_LP[PAIR_COUNT + 4 * i, i] = 1.0
    A_LP[PAIR_COUNT + 4 * i + 1, i] = 1.0
    A_LP[PAIR_COUNT + 4 * i + 2, i] = 1.0
    A_LP[PAIR_COUNT + 4 * i + 3, i] = 1.0

C_OBJ = -np.ones(N)
BOUNDS_LP = [(0.0, None)] * N

def solve_lp_and_grad(x_flat):
    """Solves LP for radii and computes analytical gradient of sum of radii w.r.t centers."""
    centers = x_flat.reshape(N, 2)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_sq = np.sum(diff**2, axis=2)
    dists = np.sqrt(np.maximum(dists_sq, 1e-12))
    np.fill_diagonal(dists, 1.0)
    
    # Build b_ub dynamically
    b_ub = np.empty(PAIR_COUNT + 4 * N)
    b_ub[:PAIR_COUNT] = dists[IDX_I, IDX_J]
    for i in range(N):
        x, y = centers[i]
        b_ub[PAIR_COUNT + 4 * i] = x
        b_ub[PAIR_COUNT + 4 * i + 1] = 1.0 - x
        b_ub[PAIR_COUNT + 4 * i + 2] = y
        b_ub[PAIR_COUNT + 4 * i + 3] = 1.0 - y
        
    res = linprog(C_OBJ, A_ub=A_LP, b_ub=b_ub, bounds=BOUNDS_LP, method='highs')
    if not res.success:
        return 0.0, np.zeros_like(x_flat)
        
    s = -res.fun
    marginals = res.ineqlin.marginals
    grad = np.zeros_like(centers)
    
    # Pairwise gradient terms: sum_j lambda_ij * (c_i - c_j) / d_ij
    lambdas = -marginals[:PAIR_COUNT]
    lambdas_mat = np.zeros((N, N))
    lambdas_mat[IDX_I, IDX_J] = lambdas
    lambdas_mat += lambdas_mat.T
    
    vec = diff / dists[..., np.newaxis]
    np.fill_diagonal(vec[:, :, 0], 0.0)
    np.fill_diagonal(vec[:, :, 1], 0.0)
    grad += np.einsum('ij,ijk->ik', lambdas_mat, vec)
    
    # Boundary gradient terms
    idx_base = PAIR_COUNT
    for i in range(N):
        # dS/dx_i = lambda_x - lambda_{1-x} = -m_x + m_{1x}
        grad[i, 0] += -marginals[idx_base + 4 * i] + marginals[idx_base + 4 * i + 1]
        grad[i, 1] += -marginals[idx_base + 4 * i + 2] + marginals[idx_base + 4 * i + 3]
        
    return -s, grad.reshape(-1)

def generate_inits(rng):
    inits = []
    # 1. Hexagonal patterns
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [6,6,5,5,4], [4,6,6,6,4]]
    for pat in patterns:
        for r_est in [0.09, 0.098, 0.105]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            c = np.array(c[:N]) + rng.normal(0, 0.003, c.shape)
            inits.append(np.clip(c, 0.02, 0.98))
            
    # 2. Corner/Edge biased
    for _ in range(10):
        c = rng.uniform(0.05, 0.95, (N, 2))
        # Pull some circles to corners/edges
        c[:4] = rng.uniform(0.02, 0.15, (4, 2))
        c[4:8] = rng.uniform(0.85, 0.98, (4, 2))
        c[8:12] = rng.uniform(0.02, 0.15, 4) * np.array([1, 0]) + rng.uniform(0.02, 0.98, 4) * np.array([0, 1])
        inits.append(c)
        
    # 3. Force-relaxed random
    for _ in range(15):
        c = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(50):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c[i] - c[j]
                    d = np.linalg.norm(d_vec)
                    if d < 0.25 and d > 1e-5:
                        f = (0.25 - d) / d
                        forces[i] += d_vec * f
                        forces[j] -= d_vec * f
            c += forces * 0.005
        inits.append(np.clip(c, 0.02, 0.98))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    bounds_opt = [(0.005, 0.995)] * (2 * N)
    best_val = -np.inf
    best_centers = None
    best_radii = None
    
    inits = generate_inits(rng)
    
    # Phase 1: Multi-start L-BFGS-B
    for c_init in inits:
        x0 = c_init.flatten()
        try:
            res = minimize(solve_lp_and_grad, x0, method='L-BFGS-B', jac=True, bounds=bounds_opt,
                           options={'maxiter': 2000, 'ftol': 1e-14})
            if -res.fun > best_val:
                best_val = -res.fun
                best_centers = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation & Escape
    if best_centers is not None:
        for step in range(30):
            noise = 0.006 * (0.85 ** step)
            c_pert = best_centers + rng.normal(0, noise, best_centers.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            try:
                res = minimize(solve_lp_and_grad, c_pert.flatten(), method='L-BFGS-B', jac=True, 
                               bounds=bounds_opt, options={'maxiter': 1500, 'ftol': 1e-14})
                if -res.fun > best_val:
                    best_val = -res.fun
                    best_centers = res.x.reshape(N, 2).copy()
            except Exception:
                pass
                
    # Phase 3: Exact LP radii & Strict Repair
    _, best_radii = solve_lp(best_centers)
    
    for _ in range(60):
        changed = False
        # Boundary clamp
        for i in range(N):
            mx = min(best_centers[i,0], 1.0-best_centers[i,0], 
                     best_centers[i,1], 1.0-best_centers[i,1])
            if best_radii[i] > mx - 1e-12:
                best_radii[i] = mx
                changed = True
        # Overlap shrink
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], 
                             best_centers[i,1]-best_centers[j,1])
                if d < best_radii[i] + best_radii[j] - 1e-12:
                    shrink = (best_radii[i] + best_radii[j] - d) / 2.0 + 1e-9
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))

# Helper standalone for Phase 3 to avoid closure issues if called externally
def solve_lp(centers):
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    b_ub = np.empty(PAIR_COUNT + 4 * N)
    b_ub[:PAIR_COUNT] = dists[IDX_I, IDX_J]
    for i in range(N):
        x, y = centers[i]
        b_ub[PAIR_COUNT + 4*i] = x
        b_ub[PAIR_COUNT + 4*i + 1] = 1.0 - x
        b_ub[PAIR_COUNT + 4*i + 2] = y
        b_ub[PAIR_COUNT + 4*i + 3] = 1.0 - y
    res = linprog(C_OBJ, A_ub=A_LP, b_ub=b_ub, bounds=BOUNDS_LP, method='highs')
    return res.x, -res.fun
