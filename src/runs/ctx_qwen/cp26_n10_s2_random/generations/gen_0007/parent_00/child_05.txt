import numpy as np
from scipy.optimize import minimize, linprog

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
# Precompute static LP constraint matrix: r_i + r_j <= dist(i,j)
A_LP = np.zeros((len(IDX_I), N))
A_LP[np.arange(len(IDX_I)), IDX_I] = 1.0
A_LP[np.arange(len(IDX_I)), IDX_J] = 1.0

def compute_lp(centers):
    """Solves LP to find optimal radii and their sum for fixed centers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    b_ub = dists[IDX_I, IDX_J]
    
    bounds = [(0.0, u) for u in ub]
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(N)

def obj_func(x):
    """Objective for center optimization: minimize negative LP sum."""
    return -compute_lp(x.reshape(N, 2))[0]

def relax(centers, steps=80):
    """Force-directed relaxation to spread circles and avoid initial overlaps."""
    c = centers.copy()
    t = 0.02
    for _ in range(steps):
        f = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                d = c[i] - c[j]
                dist = np.linalg.norm(d)
                if dist < 0.25 and dist > 1e-5:
                    force = (0.25 - dist) / dist * 0.4
                    f[i] += d * force
                    f[j] -= d * force
        c += f * t
        c = np.clip(c, 0.02, 0.98)
        t *= 0.99
    return c

def push_boundaries(centers):
    """Iteratively nudge circles towards boundaries/corners to maximize space utilization."""
    c = centers.copy()
    s0, _ = compute_lp(c)
    improved = True
    while improved:
        improved = False
        for i in range(N):
            for dx, dy in [(0.005, 0), (-0.005, 0), (0, 0.005), (0, -0.005),
                           (0.005, 0.005), (0.005, -0.005), (-0.005, 0.005), (-0.005, -0.005)]:
                c_new = c.copy()
                c_new[i] += [dx, dy]
                c_new = np.clip(c_new, 0.0, 1.0)
                s_new, _ = compute_lp(c_new)
                if s_new > s0 + 1e-7:
                    c = c_new
                    s0 = s_new
                    improved = True
                    break
            if improved:
                break
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # 1. Generate diverse initial candidates
    candidates = []
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [4,6,6,6,4],
        [5,4,6,6,5], [6,6,5,5,4], [5,5,6,5,5], [4,5,6,5,6],
        [6,6,4,6,4], [5,5,5,6,5], [6,4,6,5,5], [5,6,4,5,6]
    ]
    
    for pat in patterns:
        for r0 in [0.085, 0.095, 0.105]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.005, c.shape)
            candidates.append(relax(c, steps=60))
            
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        candidates.append(relax(c, steps=100))
        
    bounds_opt = [(0.0, 1.0)] * (2 * N)
    
    # 2. Multi-start Powell optimization
    for c0 in candidates:
        try:
            res = minimize(obj_func, c0.flatten(), method='Powell',
                           bounds=bounds_opt,
                           options={'maxiter': 2000, 'ftol': 1e-12})
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_c = res.x.reshape(N, 2)
                _, best_r = compute_lp(best_c)
        except Exception:
            pass
            
    # 3. Refine with boundary push
    if best_c is not None:
        best_c = push_boundaries(best_c)
        _, best_r = compute_lp(best_c)
        best_sum = np.sum(best_r)

                curr_s = new_s

        best_c = push_boundaries(best_c)

                best_r[i] = mx
                changed = True
        for i in range(N):

                d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])

                    shrink = (best_r[i] + best_r[j] - d) / 2.0 + 1e-9
                    best_r[i] -= shrink
                    best_r[j] -= shrink
                    changed = True
        if not changed:
            break
    best_r = np.maximum(best_r, 0.0)
    
    return best_c, best_r, float(np.sum(best_r))
