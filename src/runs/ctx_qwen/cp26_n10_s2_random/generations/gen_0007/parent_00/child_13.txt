2. **Inner Problem**: For fixed centers, exactly compute the maximum possible radii. This is a Linear Programming (LP) problem: maximize $\sum r_i$ subject to $r_i + r_j \le d_{ij}$ and $0 \le r_i \le \text{dist}(\text{center}_i, \text{boundary})$.

- The DE fitness function calls the LP solver to get the exact maximum sum of radii for the proposed centers.

import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(IDX_I)

# Precompute the static inequality constraint matrix for the LP: r_i + r_j <= dist_ij
A_LP = np.zeros((NUM_PAIRS, N))
A_LP[np.arange(NUM_PAIRS), IDX_I] = 1.0
A_LP[np.arange(NUM_PAIRS), IDX_J] = 1.0

def solve_radii_lp(centers):
    """
    Given fixed centers, solve LP to exactly maximize sum of radii.
    Constraints: r_i + r_j <= dist(i,j), 0 <= r_i <= dist_to_boundary
    """
    n = centers.shape[0]
    
    # Upper bound from boundaries
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    b_ub = dists[IDX_I, IDX_J]
    
    c_obj = -np.ones(N)
    bounds = [(0.0, u) for u in ub]
    
    try:
        res = linprog(c_obj, A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(N), 0.0

def obj_de(x):
    """Objective for Differential Evolution: minimize negative sum of radii."""
    _, val = solve_radii_lp(x.reshape(N, 2))
    return -val

def obj_powell(x):
    """Objective for Powell refinement: minimize negative sum of radii."""
    return -solve_radii_lp(x.reshape(N, 2))[1]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    bounds = [(0.0, 1.0)] * (2 * N)
    
    # Initialize DE population with diverse configurations
    pop = []
    
    # 1. Perturbed hexagonal lattice patterns
    for i in range(20):
        r_base = 0.10 + rng.normal(0, 0.006)
        pts = []
        y = r_base
        row = 0
        while len(pts) < N:
            x_start = r_base if row % 2 == 0 else 2.0 * r_base
            x = x_start
            while x <= 1.0 - r_base and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r_base
            y += r_base * np.sqrt(3.0)
            row += 1
        pts = np.array(pts[:N])
        pts += rng.normal(0, 0.012, pts.shape)
        pts = np.clip(pts, 0.0, 1.0)
        pop.append(pts.flatten())
        
    # 2. Random starts to ensure global coverage
    for i in range(10):
        c = rng.uniform(0.1, 0.9, (N, 2))
        pop.append(c.flatten())
        
    pop = np.array(pop)
    
    # Phase 1: Global search with Differential Evolution
    res_de = differential_evolution(
        obj_de, bounds, popsize=pop.shape[0], maxiter=120,
        seed=42, workers=1, tol=1e-8, polish=False,
        initializing=pop
    )
    
    best_c = res_de.x.reshape(N, 2)
    best_r, best_sum = solve_radii_lp(best_c)
    
    # Phase 2: Local refinement with Powell's method
    res_pw = minimize(
        obj_powell, best_c.flatten(), method='Powell',
        bounds=bounds, options={'maxiter': 10000, 'ftol': 1e-13, 'xtol': 1e-13}
    )
    if -res_pw.fun > best_sum:
        best_c = res_pw.x.reshape(N, 2)
        best_r, best_sum = solve_radii_lp(best_c)
        
    # Phase 3: Adaptive perturbation search to escape remaining local minima
    for step in range(60):
        noise_scale = 0.005 * (0.93 ** step)
        c_pert = best_c + rng.normal(0, noise_scale, best_c.shape)
        c_pert = np.clip(c_pert, 0.0, 1.0)
        
        res_pert = minimize(
            obj_powell, c_pert.flatten(), method='Powell',
            bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-13}
        )
        if -res_pert.fun > best_sum:
            best_c = res_pert.x.reshape(N, 2)
            best_r, best_sum = solve_radii_lp(best_c)
            
    # Phase 4: Strict numerical repair to guarantee validator compliance
    for _ in range(100):
        changed = False
        # Fix boundary violations
        for i in range(N):
            mx = min(best_c[i, 0], 1.0 - best_c[i, 0], 
                     best_c[i, 1], 1.0 - best_c[i, 1])
            if best_r[i] > mx + 1e-12:
                best_r[i] = mx
                changed = True
                
        # Fix overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_c[i, 0] - best_c[j, 0], 
                             best_c[i, 1] - best_c[j, 1])
                if d < best_r[i] + best_r[j] - 1e-12:
                    shrink = (best_r[i] + best_r[j] - d) / 2.0 + 1e-10
                    best_r[i] -= shrink
                    best_r[j] -= shrink
                    changed = True
        if not changed:
            break
            
    best_r = np.maximum(best_r, 0.0)
    final_sum = float(np.sum(best_r))
    
    return best_c, best_r, final_sum
