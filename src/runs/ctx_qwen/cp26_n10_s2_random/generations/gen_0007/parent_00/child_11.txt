To maximize the sum of radii for 26 circles in a unit square, we leverage a hybrid optimization strategy:
1. **Exact Radius Computation via LP**: For any fixed set of centers, the optimal radii are found by solving a linear program that maximizes $\sum r_i$ subject to $r_i + r_j \leq d_{ij}$ (non-overlap) and $r_i \leq \text{dist\_to\_boundary}$ (containment). This is exact and fast using `scipy.optimize.linprog` with the `highs` solver.
2. **Center Optimization**: We treat the sum of radii as a function of centers only. We optimize centers using `Powell`'s method, which handles the non-smooth nature of the LP objective well.
3. **Diverse Initialization**: We generate multiple hexagonal lattice patterns with varying row distributions and base radii, plus random starts, to cover the configuration space thoroughly. Known optimal packings for N=26 are closely related to perturbed hexagonal lattices.
4. **Perturbation Search**: After finding a local optimum, we run a decaying noise perturbation loop to escape shallow local minima and fine-tune positions.
5. **Strict Numerical Repair**: Finally, we apply a deterministic repair step to guarantee all constraints strictly satisfy the validator's tolerance (`1e-12`), ensuring the solution passes validation without sacrificing meaningful radius sum.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
PAIR_COUNT = len(IDX_I)

# Precompute static LP constraint matrix: r_i + r_j <= dist_ij
A_LP = np.zeros((PAIR_COUNT, N))
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        idx += 1

def solve_lp_for_radii(centers):
    """
    Given fixed centers, solve LP to exactly maximize sum of radii.
    Constraints: r_i + r_j <= dist(i,j), 0 <= r_i <= dist_to_boundary
    """
    n = centers.shape[0]
    
    # Upper bound from boundaries
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    b_ub = dists[IDX_I, IDX_J]
    
    c_obj = -np.ones(n)
    bounds = [(0.0, u) for u in ub]
    
    try:
        res = linprog(c_obj, A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def obj_centers(x):
    """Objective: minimize negative sum of radii computed via LP."""
    c = x.reshape(N, 2)
    c = np.clip(c, 1e-7, 1.0 - 1e-7)
    _, s = solve_lp_for_radii(c)
    return -s

def hex_init(pat, r_est, rng):
    """Generates initial configuration from a perturbed hexagonal lattice."""
    c = []
    y = r_est
    for r_idx, cnt in enumerate(pat):
        shift = r_est if r_idx % 2 == 1 else 0.0
        x = r_est + shift
        for _ in range(cnt):
            c.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3)
    c = np.array(c[:N])
    c += rng.normal(0, 0.003, c.shape)
    return np.clip(c, 0.05, 0.95)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_sum = 0.0
    best_centers = None
    
    # Diverse hexagonal patterns known to be effective for N=26
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [6,6,5,5,4],
        [4,6,6,6,4], [5,4,6,6,5], [6,5,5,5,5], [5,5,5,6,5],
        [4,5,6,5,6], [5,5,4,6,6], [5,6,4,5,6], [6,5,5,6,4],
        [5,7,5,5,4], [5,5,7,5,4], [6,6,6,4,4], [5,5,5,7,4],
        [6,6,5,4,5], [5,6,5,5,5], [4,5,5,6,6], [5,5,6,6,4]
    ]
    
    starts = []
    for pat in patterns:
        for r_est in [0.092, 0.098, 0.102, 0.108]:
            starts.append(hex_init(pat, r_est, rng))
            
    for _ in range(20):
        c = rng.uniform(0.1, 0.9, (N, 2))
        starts.append(c)
        
    # Phase 1: Powell optimization on centers for each start
    for c_init in starts:
        try:
            res = minimize(obj_centers, c_init.flatten(), method='Powell',
                           options={'maxiter': 2500, 'ftol': 1e-15, 'xtol': 1e-15})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Phase 2: Perturbation search to escape local minima and fine-tune
    if best_centers is not None:
        for step in range(80):
            noise = 0.005 * (0.91 ** step)
            c_pert = best_centers + rng.normal(0, noise, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            
            res = minimize(obj_centers, c_pert.flatten(), method='Powell',
                           options={'maxiter': 1500, 'ftol': 1e-15})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
                
    # Phase 3: Final exact LP radii computation
    best_radii, final_sum = solve_lp_for_radii(best_centers)
    
    # Phase 4: Strict numerical repair to guarantee validation passes
    for _ in range(100):
        changed = False
        # Fix boundary violations
        for i in range(N):
            mx = min(best_centers[i,0], 1.0-best_centers[i,0], 
                     best_centers[i,1], 1.0-best_centers[i,1])
            if best_radii[i] > mx + 1e-13:
                best_radii[i] = mx
                changed = True
        # Fix overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], 
                             best_centers[i,1]-best_centers[j,1])
                if d < best_radii[i] + best_radii[j] - 1e-13:
                    shrink = (best_radii[i] + best_radii[j] - d) / 2.0 + 1e-10
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
