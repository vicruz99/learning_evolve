import numpy as np
from scipy.optimize import linprog, minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)

def solve_lp_and_grad(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns radii, sum_radii, and gradient of sum_radii w.r.t centers.
    """
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    num_pairs = n*(n-1)//2
    num_bnd = 4*n
    A = np.zeros((num_pairs + num_bnd, n))
    b = np.zeros(num_pairs + num_bnd)
    
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        x, y = centers[i]
        A[idx, i] = 1.0; b[idx] = x; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - x; idx += 1
        A[idx, i] = 1.0; b[idx] = y; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - y; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs')

        return np.zeros(n), 0.0, np.zeros_like(centers)

                    grad[i] += lam * vec
                    grad[j] -= lam * vec
            idx += 1
            
    b_start = num_pairs

        mu_x = duals[b_start + 4*i]

        grad[i, 0] += mu_x - mu_1x
        grad[i, 1] += mu_y - mu_1y
        
    return radii, np.sum(radii), grad
