N = 26

def get_lp_structure():
    M = N*(N-1)//2 + 4*N
    A = np.zeros((M, N))
    idx = 0
    pairs = []

        for j in range(i+1, N):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            pairs.append((i, j))
            idx += 1
    for i in range(N):
        for _ in range(4):
            A[idx, i] = 1.0
            idx += 1
    return A, pairs

A_ub, PAIRS = get_lp_structure()

    return -obj_and_grad(v.reshape(N, 2))[0]

def grad_func(v):

    return -grad.flatten()
