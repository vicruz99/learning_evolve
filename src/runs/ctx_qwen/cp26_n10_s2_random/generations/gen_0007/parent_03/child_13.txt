import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
TRIU_IND = np.triu_indices(N, k=1)

def get_lp_data(n):
    """Pre-construct the sparse structure of the LP constraint matrix."""
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs + 4 * n, n))
    idx = 0
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            pairs.append((i, j))
            idx += 1
    for i in range(n):
        A_ub[idx, i] = 1.0; idx += 1
        A_ub[idx, i] = 1.0; idx += 1
        A_ub[idx, i] = 1.0; idx += 1
        A_ub[idx, i] = 1.0; idx += 1
    return A_ub, pairs

A_ub_global, pairs_global = get_lp_data(N)

def eval_sum_radii(centers):
    """Solves LP for optimal radii given fixed centers and returns radii + sum."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b = np.zeros(A_ub_global.shape[0])
    idx = 0
    for i, j in pairs_global:
        b[idx] = dists[i, j]
        idx += 1
    for i in range(n):
        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub_global, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def obj_centers(x):
    """Objective for center optimization: minimize negative sum of radii."""
    c = x.reshape(N, 2)
    _, s = eval_sum_radii(c)
    return -s

def generate_starts(n, rng):

    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],

                x = r_est + shift

                    x += 2.0 * r_est

        c = rng.uniform(0.15, 0.85, (n, 2))
        starts.append(c.flatten())

                changed = True

    return best_c, best_r, float(np.sum(best_r))
