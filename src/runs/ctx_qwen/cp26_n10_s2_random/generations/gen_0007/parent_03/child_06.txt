from scipy.optimize import minimize, linprog

N = 26

def compute_constraints(v):
    """Computes boundary and pairwise non-overlap constraints. Must return >= 0."""
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = []
    c.append(xs - rs)
    c.append(1.0 - xs - rs)
    c.append(ys - rs)
    c.append(1.0 - ys - rs)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append((dx**2 + dy**2 - dr**2)[mask])
    
    return np.concatenate(c)

def objective(v):

    return -np.sum(v[2::3])

def get_bounds():

    return b

def get_feasible_radii(centers):

    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    dists = np.linalg.norm(centers[:, None] - centers[None, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    rn = 0.5 * np.min(dists, axis=1)
    # Scale down slightly to ensure strict initial feasibility
    return np.minimum(rb, rn) * 0.92
