import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs + 4 * n, n))
    b_ub = np.zeros(num_pairs + 4 * n)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:
        return res.x
    return np.full(n, 0.05)

def objective_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def constraints_joint(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    
    c1 = c[:, 0] - r
    c2 = 1.0 - c[:, 0] - r
    c3 = c[:, 1] - r
    c4 = 1.0 - c[:, 1] - r
    
    idx = np.triu_indices(N, k=1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    c5 = d - (r[idx[0]] + r[idx[1]])
    
    return np.concatenate([c1, c2, c3, c4, c5])

def get_bounds():
    """Returns variable bounds for x, y in [0,1] and r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def hex_init(rows, r0, rng):
    """Generates hexagonal lattice positions based on row counts."""
    centers = []
    y = r0
    for r_idx, cnt in enumerate(rows):
        shift = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    centers = np.array(centers[:N])
    centers += rng.normal(0, 0.004, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_v = None
    best_sum = -np.inf
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_joint}
    
    # Phase 1: Generate diverse initial configurations
    starts = []
    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 5, 5, 5, 5], [5, 5, 5, 5, 6], [5, 5, 5, 6, 5],
        [7, 5, 6, 5, 3], [6, 4, 6, 5, 5], [5, 6, 4, 5, 6],
        [5, 5, 4, 6, 6], [4, 5, 6, 5, 6]
    ]
    
    for pat in row_patterns:
        for r0 in [0.088, 0.092, 0.096, 0.100, 0.104]:
            c = hex_init(pat, r0, rng)
            r = solve_lp_radii(c) * 0.99  # Strictly feasible start
            v = np.concatenate([c.flatten(), r])
            starts.append(v)
            
    for _ in range(25):
        c = rng.uniform(0.12, 0.88, (N, 2))
        r = solve_lp_radii(c) * 0.99
        v = np.concatenate([c.flatten(), r])
        starts.append(v)
        
    # Phase 2: Multi-start SLSQP optimization
    for v0 in starts:
        try:
            res = minimize(objective_joint, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            curr_sum = -res.fun
            c_vals = constraints_joint(res.x)
            if np.min(c_vals) >= -1e-8 and curr_sum > best_sum:
                best_sum = curr_sum
                best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Iterative perturbation to escape local minima
    if best_v is not None:
        base_perturbation = 0.004
        for it in range(120):
            v_trial = best_v.copy()
            # Perturb centers
            noise = rng.normal(0, base_perturbation * (1.0 - it / 150.0), 2 * N)
            v_trial[:2 * N] += noise
            v_trial[:2 * N] = np.clip(v_trial[:2 * N], 0.02, 0.98)
            
            # Recompute feasible radii via LP
            c_trial = v_trial[:2 * N].reshape(N, 2)
            r_trial = solve_lp_radii(c_trial) * 0.99
            v_trial[2 * N:] = r_trial
            
            try:
                res = minimize(objective_joint, v_trial, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 2500, 'ftol': 1e-13, 'disp': False})
                if -res.fun > best_sum:
                    c_vals = constraints_joint(res.x)
                    if np.min(c_vals) >= -1e-8:

                        best_v = res.x.copy()

                pass
                
    # Fallback if optimization failed completely
    if best_v is None:
        c = hex_init([5, 6, 5, 6, 4], 0.095, rng)
        r = solve_lp_radii(c) * 0.99
        best_v = np.concatenate([c.flatten(), r])
        
    # Phase 4: Extract and compute exact optimal radii for final centers
    centers = best_v[:2 * N].reshape(N, 2)
    radii = solve_lp_radii(centers)
    
    # Phase 5: Strict numerical repair to guarantee validator compliance
    for _ in range(150):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            mr = min(x, 1.0 - x, y, 1.0 - y)
            if r > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
