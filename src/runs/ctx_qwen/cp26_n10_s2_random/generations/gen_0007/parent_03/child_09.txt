import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_safe_radii(centers):
    n = centers.shape[0]
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    rp = 0.5 * np.min(dists, axis=1)
    return np.minimum(rb, rp) * 0.95

def compute_constraints(v):
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = xs[idx_i] - xs[idx_j]
    dy = ys[idx_i] - ys[idx_j]
    dr = rs[idx_i] + rs[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def objective(v):
    return -np.sum(v[2::3])

def hex_init(rng, rows, r0):
    centers = []
    y = r0
    for r_idx, cnt in enumerate(rows):
        shift = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    centers = np.array(centers[:N])
    centers += rng.normal(0, 0.002, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_v = None
    best_sum = -np.inf
    
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [6,4,6,5,5],
        [4,6,6,6,4], [5,5,5,5,6], [6,6,5,5,4], [5,4,6,6,5],
        [5,5,5,6,5], [6,5,5,5,5], [4,5,6,5,6], [5,6,4,5,6]
    ]
    
    candidates = []
    for pat in patterns:
        for scale in [0.92, 0.96, 1.0, 1.04, 1.08]:
            r0 = 0.10 * scale
            c = hex_init(rng, pat, r0)
            r_safe = get_safe_radii(c)
            v = np.zeros(3*N)
            v[0::3] = c[:, 0]
            v[1::3] = c[:, 1]
            v[2::3] = r_safe
            candidates.append(v)

        c = rng.uniform(0.15, 0.85, (N, 2))
        r_safe = get_safe_radii(c)
        v = np.zeros(3*N)
        v[0::3] = c[:, 0]
        v[1::3] = c[:, 1]
        v[2::3] = r_safe
        candidates.append(v)

            res = minimize(objective, x0, method='SLSQP', bounds=bounds,

            c_vals = compute_constraints(res.x)
            if np.min(c_vals) >= -1e-8:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = candidates[0]
        
    for step in range(40):

        x_trial = best_v + rng.normal(0, noise, best_v.shape)

        try:

            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
