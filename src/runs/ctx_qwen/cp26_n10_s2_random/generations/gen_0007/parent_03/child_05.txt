import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute LP constraint matrix structure for speed
pair_indices = []
for i in range(N):
    for j in range(i + 1, N):
        pair_indices.append((i, j))
NUM_PAIRS = len(pair_indices)
NUM_BOUND = 4 * N
A_UB = np.zeros((NUM_PAIRS + NUM_BOUND, N))

for idx, (i, j) in enumerate(pair_indices):
    A_UB[idx, i] = 1.0
    A_UB[idx, j] = 1.0

for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_UB[base, i] = 1.0
    A_UB[base + 1, i] = 1.0
    A_UB[base + 2, i] = 1.0
    A_UB[base + 3, i] = 1.0

def solve_lp_and_gradient(centers):
    """
    Solves LP for optimal radii given fixed centers and computes the gradient 
    of the sum of radii with respect to center positions using LP duals.
    """
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(NUM_PAIRS + NUM_BOUND)
    idx = 0
    for i, j in pair_indices:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(n):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    bounds_r = [(0, u) for u in ub]
    res = linprog(-np.ones(n), A_ub=A_UB, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if not res.success:
        return None, -1.0, np.zeros_like(centers)
        
    radii = res.x
    duals = res.ineqlin.marginals
    grad = np.zeros_like(centers)
    
    # Pairwise forces from distance constraints
    idx = 0
    for i, j in pair_indices:
        lam = duals[idx]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    # Boundary forces from wall constraints
    bound_start = NUM_PAIRS
    for i in range(n):
        mu_L = duals[bound_start + 4*i]
        mu_R = duals[bound_start + 4*i + 1]
        mu_B = duals[bound_start + 4*i + 2]
        mu_T = duals[bound_start + 4*i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, np.sum(radii), grad

def optimize_gradient(centers0, steps=2500, init_step=0.012):
    """Runs gradient ascent on centers to maximize sum of radii."""
    centers = centers0.copy()
    best_centers = centers.copy()
    best_sum = -1.0
    step = init_step
    no_improve = 0
    
    for k in range(steps):
        radii, curr_sum, grad = solve_lp_and_gradient(centers)

            break
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        # Adaptive step size decay to refine precision
        if no_improve > 60:
            step *= 0.65
        elif no_improve > 25:
            step *= 0.85
            
        if step < 1e-10:
            break
            
        # Occasional jitter to escape local subgradient plateaus
        jitter = np.zeros_like(centers)

        centers += step * (grad + jitter)
        centers = np.clip(centers, 0.03, 0.97)
        
    return best_centers, best_sum

def generate_hex_starts(n, rng):
    """Generates diverse initial configurations based on hexagonal lattices and random seeds."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],

        starts.append(c)
        
    return starts

def compute_slsqp_obj(v):
    """Objective for joint SLSQP optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def compute_slsqp_cons(v):
    """Computes inequality constraints for SLSQP (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    # Use squared distances for smooth gradients in SLSQP
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

            best_radii = r_lp

        best_radii = r_lp

                changed = True
        if not changed:

    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
