import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

N = 26
num_pairs = N * (N - 1) // 2
num_bound = 4 * N

# Precompute LP constraint matrix structure globally
A_ub_global = np.zeros((num_pairs + num_bound, N))
pair_indices_global = []
_idx = 0
for _i in range(N):
    for _j in range(_i + 1, N):
        A_ub_global[_idx, _i] = 1.0
        A_ub_global[_idx, _j] = 1.0
        pair_indices_global.append((_i, _j))
        _idx += 1
for _i in range(N):
    for _ in range(4):
        A_ub_global[_idx, _i] = 1.0
        _idx += 1

def eval_sum_radii(centers):
    """Computes the exact maximum sum of radii for fixed centers via LP."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(num_pairs + num_bound)
    k = 0
    for i, j in pair_indices_global:
        b_ub[k] = dists[i, j]; k += 1
    for i in range(N):
        b_ub[k] = centers[i, 0]; k += 1
        b_ub[k] = 1.0 - centers[i, 0]; k += 1
        b_ub[k] = centers[i, 1]; k += 1
        b_ub[k] = 1.0 - centers[i, 1]; k += 1
        
    bounds_r = [(0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_ub_global, b_ub=b_ub, bounds=bounds_r, method='highs')
    return np.sum(res.x) if res.success else 0.0

def objective(x):
    """Objective for center optimization: minimize negative sum of radii."""
    c = np.clip(x.reshape(N, 2), 1e-6, 1.0 - 1e-6)
    val = -eval_sum_radii(c)
    return val if np.isfinite(val) else 1e6

def generate_hex_starts(r0, patterns):
    """Generates hexagonal lattice initial configurations."""
    starts = []
    for pat in patterns:
        c = []
        y = r0
        for r_idx, cnt in enumerate(pat):
            shift = r0 if r_idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                c.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        c = np.array(c[:N])
        c = np.clip(c, 0.05, 0.95)
        starts.append(c)
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_centers = None
    best_sum = -1.0

                                        tol=1e-7, seed=42, workers=1, polish=False)
        best_centers = res_de.x.reshape(N, 2)
    except Exception:
        pass
        
    # Phase 2: Local polish from DE result
    if best_centers is not None:
        res_pw = minimize(objective, best_centers.flatten(), method='Powell',

        best_centers = res_pw.x.reshape(N, 2)
        best_sum = eval_sum_radii(best_centers)
        
    # Phase 3: Multi-start from hex patterns and random configurations
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [6,5,5,5,5], [5,5,5,6,5]]
    hex_starts = generate_hex_starts(0.095, patterns)
    
    rng = np.random.default_rng(42)
    candidates = hex_starts + [rng.uniform(0.1, 0.9, (N, 2)) for _ in range(10)]

        c_new = res.x.reshape(N, 2)
        s_new = eval_sum_radii(c_new)
        if s_new > best_sum:
            best_sum = s_new
            best_centers = c_new

    for _ in range(30):
        c_trial = best_centers.copy() + rng.normal(0, 0.003, best_centers.shape)
        c_trial = np.clip(c_trial, 0.02, 0.98)
        res = minimize(objective, c_trial.flatten(), method='Powell',
                       options={'maxiter': 1000, 'ftol': 1e-12, 'xtol': 1e-12})
        c_new = res.x.reshape(N, 2)
        s_new = eval_sum_radii(c_new)
        if s_new > best_sum:
            best_sum = s_new

        b_ub[k] = 1.0 - best_centers[i, 1]; k += 1

                req = best_radii[i] + best_radii[j]

                    best_radii[j] -= shrink
                    changed = True

            mr = min(x, 1.0-x, y, 1.0-y)
            if r > mr + 1e-12:
                best_radii[i] = mr
                changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
