import numpy as np
from scipy.optimize import minimize, linprog

N = 26
NUM_PAIRS = N * (N - 1) // 2

def get_A_LP():
    """Precomputes the inequality constraint matrix for the radii LP."""
    A = np.zeros((NUM_PAIRS + 4 * N, N))
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
    for i in range(N):
        A[idx + 4 * i, i] = 1.0
        A[idx + 4 * i + 1, i] = 1.0
        A[idx + 4 * i + 2, i] = 1.0
        A[idx + 4 * i + 3, i] = 1.0
    return A

A_LP = get_A_LP()

def solve_lp_and_grad(centers):
    """Solves LP for radii and computes subgradient w.r.t centers using duals."""
    n = centers.shape[0]
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    
    b_ub = np.empty(NUM_PAIRS + 4 * n)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            b_ub[idx] = dists[i, j]
            idx += 1
    for i in range(n):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1

    if not res.success:
        return 0.0, np.zeros(n), np.zeros_like(centers)
        
    radii = res.x
    duals = res.ineqlin.marginals
    grad = np.zeros_like(centers)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]
            if lam > 1e-7:
                d = dists[i, j]
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
            idx += 1
            
    for i in range(n):
        mu_x = duals[idx]; idx += 1
        mu_xm = duals[idx]; idx += 1
        mu_y = duals[idx]; idx += 1
        mu_ym = duals[idx]; idx += 1
        grad[i, 0] += mu_x - mu_xm
        grad[i, 1] += mu_y - mu_ym
        
    return -res.fun, radii, grad

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def cons_joint(v):
    """Computes boundary and non-overlap constraints (must be >= 0)."""
    c_ = v[:2 * N].reshape(N, 2)
    r_ = v[2 * N:]
    con = [c_[:, 0] - r_, 1.0 - c_[:, 0] - r_, c_[:, 1] - r_, 1.0 - c_[:, 1] - r_]
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c_[idx_i, 0] - c_[idx_j, 0]
    dy = c_[idx_i, 1] - c_[idx_j, 1]
    dr = r_[idx_i] + r_[idx_j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def optimize_gradient(centers0, steps=700, rng=None):
    """Runs gradient ascent on centers to maximize sum of radii."""
    c = centers0.copy()
    best_c = c.copy()
    best_sum = -1.0
    
    s_curr, _, _ = solve_lp_and_grad(c)
    best_sum = s_curr
    
    step_size = 0.005
    for k in range(steps):
        s_curr, _, grad = solve_lp_and_grad(c)
        if s_curr > best_sum:
            best_sum = s_curr
            best_c = c.copy()
            
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-9:
            break
            
        c_new = c + step_size * grad / norm_g
        c_new = np.clip(c_new, 0.02, 0.98)
        
        s_new, _, _ = solve_lp_and_grad(c_new)
        if s_new > s_curr:

            step_size *= 0.85

            c += rng.normal(0, 0.0015, c.shape)
            c = np.clip(c, 0.02, 0.98)
            
    return best_c, best_sum
