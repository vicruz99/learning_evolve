import numpy as np
from scipy.optimize import minimize, linprog

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(IDX_I)

# Precompute LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS + N, N))
_idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[_idx, i] = 1.0
        A_LP[_idx, j] = 1.0
        _idx += 1
for i in range(N):
    A_LP[_idx + i, i] = 1.0

def solve_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    c = np.clip(centers, 1e-6, 1.0 - 1e-6)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    b = np.empty(NUM_PAIRS + N)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            b[idx] = dists[i, j]
            idx += 1
    b[idx:] = ub
    
    bounds_r = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, bounds=bounds_r, method='highs')
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def obj_centers(x):
    """Objective for center optimization: minimize negative sum of radii."""
    return -solve_lp_radii(x.reshape(N, 2))[0]

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def cons_joint(v):
    """Computes boundary and non-overlap constraints (must be >= 0)."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    
    cons = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    
    dx = c[IDX_I, 0] - c[IDX_J, 0]
    dy = c[IDX_I, 1] - c[IDX_J, 1]
    dr = r[IDX_I] + r[IDX_J]
    cons.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(cons)

def force_spread(centers, steps=400):
    """Simple repulsion simulation to spread clusters."""
    c = centers.copy()
    temp = 0.015
    for _ in range(steps):
        diff = c[:, None, :] - c[None, :, :]
        dists = np.linalg.norm(diff, axis=2)
        np.fill_diagonal(dists, 1.0)
        f_mag = temp / (dists**2 + 0.0001)
        forces = np.sum(diff * f_mag[:, :, None], axis=1)
        c += 0.002 * forces
        c = np.clip(c, 0.02, 0.98)
        temp *= 0.997
    return c

def generate_starts(rng):
    """Generates diverse initial center configurations."""
    starts = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4], [5,5,5,5,6], [6,4,6,5,5]]
    
    for pat in patterns:
        for scale in [0.90, 0.95, 1.0, 1.05]:
            pts = []
            y = 0.09 * scale
            r_est = 0.09 * scale
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            c = np.array(pts[:N])
            c += rng.normal(0, 0.004, c.shape)
            c = np.clip(c, 0.02, 0.98)
            starts.append(c)
            
    for _ in range(12):
        c = rng.uniform(0.15, 0.85, (N, 2))
        starts.append(force_spread(c))
        
    for _ in range(8):
        starts.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    bounds_joint = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': cons_joint}
    
    starts = generate_starts(rng)
    
    # Phase 1: Multi-start center optimization with Powell
    for c_init in starts:
        try:
            res = minimize(obj_centers, c_init.flatten(), method='Powell', 
                           options={'maxiter': 6000, 'ftol': 1e-11, 'xtol': 1e-11})
            c_opt = res.x.reshape(N, 2)
            s_lp, r_lp = solve_lp_radii(c_opt)
            
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = c_opt.copy()
                best_radii = r_lp.copy()
        except Exception:
            pass

    # Phase 2: Perturbation search to escape local minima
    if best_centers is not None:
        curr_c = best_centers.copy()
        noise_scale = 0.005
        for _ in range(40):
            c_trial = curr_c + rng.normal(0, noise_scale, curr_c.shape)
            c_trial = np.clip(c_trial, 0.02, 0.98)
            
            try:
                res = minimize(obj_centers, c_trial.flatten(), method='Powell',
                               options={'maxiter': 3000, 'ftol': 1e-10})
                s_lp, r_lp = solve_lp_radii(res.x.reshape(N, 2))
                
                if s_lp > best_sum + 1e-9:
                    best_sum = s_lp
                    best_centers = res.x.reshape(N, 2).copy()
                    best_radii = r_lp.copy()
                    curr_c = best_centers.copy()
            except Exception:
                pass
            noise_scale *= 0.95

    # Phase 3: Joint SLSQP polish for micro-adjustments
    r_init = np.maximum(best_radii * 0.94, 0.001)
    v0 = np.concatenate([best_centers.flatten(), r_init])
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                       constraints=cons_dict, options={'maxiter': 20000, 'ftol': 1e-13, 'catol': 1e-13})
        
        if np.min(cons_joint(res.x)) >= -1e-9:
            c_opt = res.x[:2 * N].reshape(N, 2)
            s_lp, r_lp = solve_lp_radii(c_opt)
            if s_lp > best_sum:
                best_sum = s_lp
                best_centers = c_opt.copy()
                best_radii = r_lp.copy()
    except Exception:
        pass

    # Phase 4: Strict numerical repair to guarantee validation tolerance compliance
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for _ in range(100):
        changed = False
        
        # Resolve overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-12:
                radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
