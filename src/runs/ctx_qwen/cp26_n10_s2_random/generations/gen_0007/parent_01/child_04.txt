To improve the packing sum of radii beyond 2.627, we will use a gradient ascent approach on circle centers, leveraging the exact dual variables (shadow prices) from a Linear Programming (LP) solver. 

Key insights:
1. **LP for Radii**: For fixed centers, maximizing the sum of radii subject to non-overlap and boundary constraints is a linear program. `scipy.optimize.linprog` with the `highs` method solves this extremely fast.
2. **Exact Gradient via Duals**: The gradient of the optimal sum of radii with respect to center positions can be computed directly from the LP dual variables. Each active pairwise constraint $(r_i + r_j \le d_{ij})$ contributes a repulsive force proportional to the dual multiplier $\lambda_{ij}$ directed along the line connecting the centers. Boundary constraints similarly contribute forces pushing circles away from walls.
3. **Gradient Ascent**: We iteratively update centers in the direction of this gradient to maximize the sum of radii. This is significantly more stable and directed than generic nonlinear optimizers (like SLSQP) for this problem.
4. **Robust Search Strategy**: We run multiple trajectories from diverse initial configurations (hexagonal lattices, force-repulsion layouts, random starts). Each trajectory uses adaptive step-size decay and periodic random kicks to escape local minima. A final local basin-hopping phase polishes the best result.
5. **Strict Repair**: A deterministic post-processing step symmetrically shrinks overlapping circles and clamps boundary violations to guarantee the validation function passes without losing significant radius sum.

This approach directly targets the geometry of the optimal packing, typically converging to high-density configurations efficiently.

import numpy as np
from scipy.optimize import linprog

N = 26
PAIR_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_IDX[0])

# Precompute static LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
p_idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[p_idx, i] = 1.0
        A_LP[p_idx, j] = 1.0
        p_idx += 1
for i in range(N):
    A_LP[NUM_PAIRS + 4 * i, i] = 1.0
    A_LP[NUM_PAIRS + 4 * i + 1, i] = 1.0
    A_LP[NUM_PAIRS + 4 * i + 2, i] = 1.0
    A_LP[NUM_PAIRS + 4 * i + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for max sum of radii and computes exact gradient w.r.t centers."""
    c = np.clip(centers, 1e-6, 1.0 - 1e-6)
    
    # Pairwise distances
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    np.fill_diagonal(dists, np.inf)
    
    # Build RHS for LP
    b_ub = np.empty(NUM_PAIRS + 4 * N)
    idx = 0
    for i, j in zip(PAIR_IDX[0], PAIR_IDX[1]):
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b_ub[idx] = c[i, 0]; idx += 1
        b_ub[idx] = 1.0 - c[i, 0]; idx += 1
        b_ub[idx] = c[i, 1]; idx += 1
        b_ub[idx] = 1.0 - c[i, 1]; idx += 1
        
    bounds = [(0.0, 0.5)] * N
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return 0.0, np.zeros(N), np.zeros((N, 2))
        
    radii = res.x
    total_sum = -res.fun
    
    # Extract duals (shadow prices)
    duals = res.ineqlin.marginals if res.ineqlin.marginals is not None else np.zeros(NUM_PAIRS + 4 * N)
    
    # Compute gradient of sum_radii w.r.t centers using duals
    grad = np.zeros((N, 2))
    
    # Pairwise repulsive forces from distance constraints
    lam = np.maximum(duals[:NUM_PAIRS], 0.0)
    i_idx, j_idx = PAIR_IDX
    d_vals = dists[i_idx, j_idx]
    d_vals[d_vals < 1e-9] = 1e-9  # Prevent division by zero
    vecs = diff[i_idx, j_idx] / d_vals[:, np.newaxis]
    forces = lam[:, np.newaxis] * vecs
    grad[i_idx] += np.sum(forces, axis=0)
    grad[j_idx] -= np.sum(forces, axis=0)
    
    # Boundary forces from wall constraints
    for i in range(N):
        base = NUM_PAIRS + 4 * i
        grad[i, 0] += duals[base] - duals[base + 1]
        grad[i, 1] += duals[base + 2] - duals[base + 3]
        
    return total_sum, radii, grad

def optimize_trajectory(c_init, rng, max_iter=3000):
    """Runs gradient ascent on centers to maximize sum of radii."""
    c = c_init.copy()
    best_c = c.copy()
    best_sum = -1.0
    step = 0.008
    
    for k in range(max_iter):
        curr_sum, radii, grad = solve_lp_and_grad(c)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_c = c.copy()
            
        # Adaptive step decay
        if k > max_iter * 0.4:
            step *= 0.997
            
        c += step * grad
        c = np.clip(c, 0.005, 0.995)
        
        # Periodic random kick to escape local minima
        if k % 250 == 0 and k > 0:
            c += rng.normal(0, 0.0015, c.shape)
            c = np.clip(c, 0.005, 0.995)
            
    return best_c, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_centers = None
    best_sum = -1.0
    
    starts = []
    
    # 1. Hexagonal lattice patterns
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,6,5,5], [6,4,6,5,5], [4,6,6,6,4]]
    for pat in patterns:
        pts = []
        y = 0.09
        r_est = 0.09
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        starts.append(np.array(pts[:N]))
        
    # 2. Random starts
    for _ in range(8):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # 3. Force-repulsion spread starts
    for _ in range(5):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(400):
            diff = c[:, None, :] - c[None, :, :]
            dists = np.linalg.norm(diff, axis=2)
            np.fill_diagonal(dists, 1.0)
            f_mag = 0.004 / (dists**2 + 1e-4)
            forces = np.sum(diff * f_mag[:, :, None], axis=1)
            c += 0.008 * forces
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    # Phase 1: Gradient ascent from diverse starts
    for c_init in starts:
        c_opt, s_opt = optimize_trajectory(c_init, rng, max_iter=2500)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            
    # Phase 2: Local basin-hopping refinement
    if best_centers is not None:
        curr_c = best_centers.copy()
        for _ in range(200):
            c_trial = curr_c + rng.normal(0, 0.0008, curr_c.shape)
            c_trial = np.clip(c_trial, 0.01, 0.99)
            s_trial, _, _ = solve_lp_and_grad(c_trial)
            if s_trial > best_sum + 1e-9:
                best_sum = s_trial
                best_centers = c_trial.copy()
                curr_c = c_trial.copy()
            elif rng.random() < 0.05:
                curr_c = c_trial.copy()
                
    # Final LP evaluation
    final_sum, radii, _ = solve_lp_and_grad(best_centers)
    
    # Phase 3: Strict numerical repair to guarantee validator compliance
    centers = best_centers.copy()
    for _ in range(80):
        changed = False
        
        # Resolve pairwise overlaps symmetrically
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-12:
                radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
