import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26

def get_lp_matrices():
    n = N
    num_pairs = n * (n - 1) // 2
    A = np.zeros((num_pairs, n))
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
    return A

A_LP = get_lp_matrices()
TRIU_I, TRIU_J = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    n = centers.shape[0]
    c = np.clip(centers, 1e-7, 1.0 - 1e-7)
    
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = dists[TRIU_I, TRIU_J]
    
    bounds_r = [(0.0, u) for u in ub]
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def obj_centers(x):
    return -solve_lp_radii(x.reshape(N, 2))[0]

def generate_inits():
    inits = []
    rng = np.random.default_rng(42)
    
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [5,5,6,5,5], [4,6,6,6,4], [6,4,6,5,5]]
    for pat in patterns:
        pts = []
        y = 0.09
        r_est = 0.09
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        inits.append(np.array(pts[:N]))
        
    for s in range(8):
        rng.seed(s)
        cx, cy = np.meshgrid(np.linspace(0.1, 0.9, 5), np.linspace(0.1, 0.9, 5))
        pts = np.column_stack((cx.flatten(), cy.flatten()))
        pts += rng.normal(0, 0.015, pts.shape)
        pts = np.clip(pts, 0.05, 0.95)
        rng.shuffle(pts)
        inits.append(pts[:N])
        
    for s in range(10):
        rng.seed(s)
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return inits

def run_packing():
    best_sum = -1.0
    best_centers = None
    
    inits = generate_inits()
    bounds_c = [(0.0, 1.0)] * (2 * N)
    
    # Phase 1: Local optimization from diverse starts
    for c0 in inits:
        c0 = np.clip(c0, 0.01, 0.99)
        try:
            res = minimize(obj_centers, c0.flatten(), method='Nelder-Mead', 
                           options={'maxiter': 15000, 'xatol': 1e-8, 'fatol': 1e-10})
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    # Phase 2: Global search with Differential Evolution
    try:
        de_res = differential_evolution(obj_centers, bounds_c, popsize=15, maxiter=120, 
                                        seed=42, tol=1e-7, polish=False, workers=1)
        s = -de_res.fun
        if s > best_sum:
            best_sum = s
            best_centers = de_res.x.reshape(N, 2).copy()
    except Exception:
        pass
        
    # Phase 3: Perturbation refinement to escape local minima
    rng = np.random.default_rng(123)
    curr_c = best_centers.copy()
    for step in range(120):
        scale = 0.006 * (1.0 - step / 120.0)
        c_trial = curr_c + rng.normal(0, scale, curr_c.shape)
        c_trial = np.clip(c_trial, 0.01, 0.99)
        s_trial, _ = solve_lp_radii(c_trial)
        if s_trial > best_sum + 1e-9:
            best_sum = s_trial
            best_centers = c_trial.copy()
            curr_c = c_trial.copy()
        elif rng.random() < 0.03:
            curr_c = c_trial.copy()
            
    # Phase 4: Gradient-free local polish
    try:
        res = minimize(obj_centers, best_centers.flatten(), method='Powell', 
                       options={'maxiter': 10000, 'ftol': 1e-12, 'xtol': 1e-12})
        s = -res.fun
        if s > best_sum:
            best_sum = s
            best_centers = res.x.reshape(N, 2).copy()
    except Exception:
        pass
        
    # Phase 5: Joint SLSQP polish (centers + radii) for micro-adjustments
    def obj_joint(v):
        return -np.sum(v[2 * N:])
        
    def cons_joint(v):
        c = v[:2 * N].reshape(N, 2)
        r = v[2 * N:]
        out = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
        dx = c[TRIU_I, 0] - c[TRIU_J, 0]
        dy = c[TRIU_I, 1] - c[TRIU_J, 1]
        dr = r[TRIU_I] + r[TRIU_J]
        out.append(dx**2 + dy**2 - dr**2)
        return np.concatenate(out)
        
    _, best_radii = solve_lp_radii(best_centers)
    r_init = best_radii * 0.95
    v0 = np.concatenate([best_centers.flatten(), r_init])
    bounds_j = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j, 
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': 8000, 'ftol': 1e-13})
        if np.sum(res.x[2 * N:]) > best_sum - 1e-6:
            best_centers = res.x[:2 * N].reshape(N, 2).copy()
            best_radii = res.x[2 * N:].copy()
            best_sum = np.sum(best_radii)
    except Exception:
        pass
        
    # Phase 6: Final LP computation & strict numerical repair
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < radii[i] + radii[j] - 1e-12:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
