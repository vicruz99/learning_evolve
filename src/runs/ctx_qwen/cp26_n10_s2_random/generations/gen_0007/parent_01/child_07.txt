import numpy as np
from scipy.optimize import differential_evolution, minimize, linprog

N = 26
NUM_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure (fixed regardless of center positions)
A_LP = np.zeros((NUM_PAIRS + N, N))
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        idx += 1
for i in range(N):
    A_LP[idx + i, i] = 1.0

def solve_lp(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    b_ub = np.empty(NUM_PAIRS + n)
    b_ub[:NUM_PAIRS] = dists[np.triu_indices(n, k=1)]
    b_ub[NUM_PAIRS:] = ub
    
    bounds_r = [(0.0, u) for u in ub]
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def obj_centers(x):
    """Objective for center optimization: minimize negative sum of radii."""
    return -solve_lp(x.reshape(N, 2))[0]

def cons_joint(v):
    """Computes boundary and non-overlap constraints (must be >= 0). Uses squared distances."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    
    cons = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    
    idx_i, idx_j = np.triu_indices(N, k=1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    cons.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(cons)

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_c = [(0.0, 1.0)] * (2 * N)
    
    best_c = None
    best_sum = -1.0
    
    # --- Phase 1: Differential Evolution Global Search ---
    res_de = differential_evolution(obj_centers, bounds_c, popsize=20, maxiter=150,
                                    seed=42, workers=1, tol=1e-8, polish=False,
                                    mutation=(0.6, 1.4))
    best_c = res_de.x.reshape(N, 2)
    
    # --- Phase 2: L-BFGS-B Local Refinement ---
    res_lb = minimize(obj_centers, best_c.flatten(), method='L-BFGS-B', bounds=bounds_c,
                      options={'maxiter': 5000, 'ftol': 1e-13})
    best_c = res_lb.x.reshape(N, 2)
    best_sum, best_r = solve_lp(best_c)
    
    # --- Phase 3: Simulated Annealing on Centers ---
    curr_c = best_c.copy()
    step = 0.008
    for _ in range(300):
        trial_c = curr_c + rng.normal(0, step, curr_c.shape)
        trial_c = np.clip(trial_c, 0.01, 0.99)
        trial_sum, _ = solve_lp(trial_c)
        
        if trial_sum > best_sum + 1e-9:
            best_sum = trial_sum
            best_c = trial_c.copy()
            curr_c = trial_c.copy()
        elif rng.random() < np.exp(-(best_sum - trial_sum) / max(step, 1e-6)):
            curr_c = trial_c.copy()
        step *= 0.995
        
    # --- Phase 4: Joint SLSQP Polish ---
    _, init_r = solve_lp(best_c)
    init_r = np.maximum(init_r * 0.92, 0.005)  # Ensure strict initial feasibility
    v0 = np.concatenate([best_c.flatten(), init_r])
    bounds_joint = bounds_c + [(0.0, 0.5)] * N
    
    try:
        res_slsqp = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                             constraints={'type': 'ineq', 'fun': cons_joint},
                             options={'maxiter': 20000, 'ftol': 1e-14})
        if np.min(cons_joint(res_slsqp.x)) >= -1e-8:
            best_c = res_slsqp.x[:2*N].reshape(N, 2)
            _, best_r = solve_lp(best_c)  # Recompute radii via LP for exactness
            best_sum = np.sum(best_r)
    except Exception:
        pass
        
    # --- Phase 5: Final LP and Strict Numerical Repair ---
    _, best_r = solve_lp(best_c)
    centers = best_c.copy()
    radii = best_r.copy()
    
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
