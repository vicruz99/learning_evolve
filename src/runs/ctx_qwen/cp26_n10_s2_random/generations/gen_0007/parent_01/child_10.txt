import numpy as np
from scipy.optimize import linprog, minimize

N = 26
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(PAIR_I)

def solve_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    # Upper bounds from boundaries: r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # LP: max sum(r) s.t. r_i + r_j <= d_ij, 0 <= r_i <= ub_i
    A_ub = np.zeros((NUM_PAIRS, n))
    A_ub[np.arange(NUM_PAIRS), PAIR_I] = 1.0
    A_ub[np.arange(NUM_PAIRS), PAIR_J] = 1.0
    b_ub = dists[PAIR_I, PAIR_J]
    
    c_obj = -np.ones(n)
    bounds = [(0.0, u) for u in ub]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def obj_centers(x):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = x.reshape(N, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    s, _ = solve_lp_radii(centers)
    return -s

def generate_starts(rng):
    """Generates diverse initial center configurations."""
    starts = []
    
    # 1. Hexagonal lattice patterns
    patterns = [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], [6,4,6,5,5]]
    for pat in patterns:
        pts = []
        y = 0.085
        r_est = 0.085
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        starts.append(np.array(pts[:N]))
        
    # 2. Random dense starts
    for _ in range(12):
        starts.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    # 3. Repulsive spread starts
    for _ in range(6):
        c = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(400):
            forces = np.zeros_like(c)
            diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
            dists = np.linalg.norm(diff, axis=2)
            np.fill_diagonal(dists, 1.0)
            f_mag = 0.02 / (dists**2 + 1e-4)
            forces += np.sum(diff * f_mag[:, :, np.newaxis], axis=1)
            c += 0.01 * forces
            c = np.clip(c, 0.02, 0.98)
        starts.append(c)
        
    return starts

def run_packing():
    rng = np.random.default_rng(42)
    best_c = None
    best_s = -1.0
    best_r = None
    
    starts = generate_starts(rng)
    
    # Phase 1: Powell optimization from each start
    for c0 in starts:
        try:
            res = minimize(obj_centers, c0.flatten(), method='Powell',
                           options={'maxiter': 2500, 'ftol': 1e-12, 'xtol': 1e-12})
            s = -res.fun
            if s > best_s:
                best_s = s
                best_c = res.x.reshape(N, 2).copy()
                _, best_r = solve_lp_radii(best_c)
        except Exception:
            pass
            
    # Phase 2: Basin Hopping / Simulated Annealing on centers
    if best_c is not None:
        curr_c = best_c.copy()
        curr_s = best_s
        noise = 0.006
        
        for step in range(800):
            c_trial = curr_c + rng.normal(0, noise, curr_c.shape)
            c_trial = np.clip(c_trial, 0.01, 0.99)
            
            s_trial, _ = solve_lp_radii(c_trial)
            
            if s_trial > curr_s:
                curr_s = s_trial
                curr_c = c_trial.copy()
                if s_trial > best_s:
                    best_s = s_trial
                    best_c = c_trial.copy()
                    _, best_r = solve_lp_radii(best_c)
            else:
                # Accept worse with probability exp(-delta/T)
                T = 0.0005 * max(0.0, 1.0 - step/800.0)
                if T > 1e-7 and rng.random() < np.exp((s_trial - curr_s) / T):
                    curr_c = c_trial.copy()
                    curr_s = s_trial
                    
            # Decay noise
            noise = max(0.0001, noise * 0.995)
            
        # Ensure best_r matches best_c
        _, best_r = solve_lp_radii(best_c)
        
    # Phase 3: Joint SLSQP polish
    if best_c is not None and best_r is not None:
        def obj_joint(v):
            return -np.sum(v[2 * N:])
            
        def cons_joint(v):
            c = v[:2 * N].reshape(N, 2)
            r = v[2 * N:]
            con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
            dx = c[PAIR_I, 0] - c[PAIR_J, 0]
            dy = c[PAIR_I, 1] - c[PAIR_J, 1]
            dr = r[PAIR_I] + r[PAIR_J]
            con.append(dx**2 + dy**2 - dr**2)
            return np.concatenate(con)
            
        v0 = np.concatenate([best_c.flatten(), best_r * 0.95])
        bounds_joint = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        try:
            res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                           constraints={'type': 'ineq', 'fun': cons_joint},
                           options={'maxiter': 15000, 'ftol': 1e-13})
            if np.min(cons_joint(res.x)) >= -1e-8:
                c_opt = res.x[:2 * N].reshape(N, 2)
                r_opt = res.x[2 * N:]
                # Re-optimize radii exactly for these centers
                s_lp, r_lp = solve_lp_radii(c_opt)
                if s_lp > best_s:
                    best_s = s_lp
                    best_c = c_opt.copy()
                    best_r = r_lp.copy()
        except Exception:
            pass
            
    # Phase 4: Strict Numerical Repair
    centers = best_c.copy()
    radii = best_r.copy()
    
    for _ in range(100):
        changed = False
        
        # Resolve overlaps by symmetric shrinking
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-12:
                radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
