To maximize the sum of radii for 26 circles in a unit square, we need to efficiently navigate a highly non-convex optimization landscape. The key insight is that for any fixed set of centers, the optimal radii are determined exactly by a Linear Programming (LP) problem. This allows us to separate the optimization into two stages:

1. **Center Optimization**: We treat the sum of optimal radii (computed via LP) as an objective function of the centers. This function is continuous but non-smooth. We use Powell's method (a derivative-free local optimizer) starting from a diverse set of high-quality initial configurations (hexagonal lattices, grids, force-spread distributions, and random starts). This reliably converges to deep local optima.
2. **Perturbation & Joint Polish**: After finding the best center configuration, we perform a decaying-step perturbation search to escape shallow plateaus. Finally, we polish the geometry using a joint SLSQP optimization on both centers and radii, which fine-tunes contact constraints and boundary adherence.
3. **Strict Repair**: A deterministic shrinkage loop guarantees that all pairwise distances and boundary constraints strictly satisfy the validator's tolerance (`1e-12`), preventing false overlaps due to numerical drift.

This hybrid approach leverages the exactness of LP for radius assignment, the robustness of Powell for center placement, and the precision of SLSQP for final polishing, consistently pushing past previous benchmarks.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRI_I, TRI_J = np.triu_indices(N, k=1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS + N, N))
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        idx += 1
for i in range(N):
    A_LP[idx + i, i] = 1.0

def compute_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    c = np.clip(centers, 1e-7, 1.0 - 1e-7)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b = np.empty(NUM_PAIRS + N)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            b[idx] = dists[i, j]
            idx += 1
    b[idx:] = ub
    
    # Solve: max sum(r) s.t. A_ub r <= b, 0 <= r <= ub
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, bounds=[(0, u) for u in ub], method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def obj_centers(x):
    """Objective for center optimization: minimize negative sum of radii."""
    return -compute_lp_radii(x.reshape(N, 2))[0]

def generate_starts(rng):
    """Generates diverse initial center configurations."""
    starts = []
    
    # 1. Hexagonal lattice patterns with various row structures
    patterns = [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], 
                [5,5,6,5,5], [6,4,6,5,5], [5,4,6,5,6]]
    for pat in patterns:
        for r0 in [0.09, 0.095, 0.10, 0.105]:
            pts = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) < N:
                        pts.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(pts[:N])
            c += rng.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.02, 0.98)
            starts.append(c)
            
    # 2. Perturbed grid patterns
    for s in range(5):
        gx = np.linspace(0.1, 0.9, 5)
        gy = np.linspace(0.1, 0.9, 5)
        cx, cy = np.meshgrid(gx, gy)
        c = np.column_stack((cx.ravel(), cy.ravel()))[:N]
        c += rng.normal(0, 0.005, c.shape)
        c = np.clip(c, 0.02, 0.98)
        starts.append(c)
        
    # 3. Force-repulsion spread (simulates dense packing)
    for _ in range(10):
        c = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(500):
            diff = c[:, None, :] - c[None, :, :]
            dists = np.linalg.norm(diff, axis=2)
            np.fill_diagonal(dists, 1.0)
            f = diff / (dists**2 + 0.001)[:, :, None]
            c += 0.005 * np.sum(f, axis=1)
            c = np.clip(c, 0.02, 0.98)
        starts.append(c)
        
    # 4. Random dense starts
    for _ in range(15):
        starts.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    return starts

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def cons_joint(v):
    """Computes boundary and non-overlap constraints (must be >= 0). Uses squared distances."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    
    cons = []
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    # Overlap constraints (squared to avoid sqrt gradient issues)
    dx = c[TRI_I, 0] - c[TRI_J, 0]
    dy = c[TRI_I, 1] - c[TRI_J, 1]
    dr = r[TRI_I] + r[TRI_J]
    cons.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(cons)

def run_packing():
    rng = np.random.default_rng(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    starts = generate_starts(rng)
    
    # Phase 1: Multi-start local optimization on centers using Powell
    for c_init in starts:
        try:
            res = minimize(obj_centers, c_init.flatten(), method='Powell', 
                           options={'maxiter': 2500, 'ftol': 1e-12, 'xtol': 1e-12})
            s, r = compute_lp_radii(res.x.reshape(N, 2))
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape(N, 2)
                best_radii = r
        except Exception:
            pass
            
    # Phase 2: Greedy Basin Hopping on Centers (LP Objective)
    if best_centers is not None:
        curr_c = best_centers.copy()
        for step in range(200):
            scale = 0.006 * (1.0 - step / 200.0)
            c_trial = curr_c + rng.normal(0, scale, curr_c.shape)
            c_trial = np.clip(c_trial, 0.01, 0.99)
            
            s_trial, _ = compute_lp_radii(c_trial)
            
            # Greedy accept
            if s_trial > best_sum + 1e-9:
                best_sum = s_trial
                best_centers = c_trial.copy()
                best_radii, _ = compute_lp_radii(best_centers)
                curr_c = c_trial.copy()
            # Occasionally accept neutral moves to escape plateaus
            elif rng.random() < 0.02:
                curr_c = c_trial.copy()
        
    # Phase 3: Joint SLSQP polish on the best geometry found
    r_init = best_radii * 0.93
    v0 = np.concatenate([best_centers.flatten(), r_init])
    bounds_j = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': 15000, 'ftol': 1e-13})
        if np.min(cons_joint(res.x)) >= -1e-7:
            c_opt = res.x[:2 * N].reshape(N, 2)
            s_lp, r_lp = compute_lp_radii(c_opt)
            if s_lp > best_sum:
                best_centers = c_opt
                best_radii = r_lp
                best_sum = s_lp
    except Exception:
        pass
        
    # Phase 4: Strict Numerical Repair to guarantee validation tolerance compliance
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for _ in range(60):
        changed = False
        
        # Resolve overlaps by symmetric shrinking
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    sh = (req - d) / 2.0 + 1e-9
                    radii[i] -= sh
                    radii[j] -= sh
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-12:
                radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
