import numpy as np
from scipy.optimize import linprog, basinhopping, minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(IDX_I)

# Precompute LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS + N, N))
for k, (i, j) in enumerate(zip(IDX_I, IDX_J)):
    A_LP[k, i] = 1.0
    A_LP[k, j] = 1.0
for i in range(N):
    A_LP[NUM_PAIRS + i, i] = 1.0

def solve_lp(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    b_ub = np.empty(NUM_PAIRS + N)
    b_ub[:NUM_PAIRS] = dists[IDX_I, IDX_J]
    b_ub[NUM_PAIRS:] = ub
    
    bounds_r = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def obj_func(x):
    """Objective for basin hopping: minimize negative sum of radii."""
    return -solve_lp(x.reshape(N, 2))[0]

def generate_starts():
    """Generates diverse initial configurations."""
    starts = []
    # Hexagonal lattice patterns known to be near-optimal for N=26
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [4,6,5,6,5], [5,5,5,5,6]]
    for pat in patterns:
        pts = []
        y = 0.1
        r_est = 0.085
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = 0.1 + shift
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        starts.append(np.array(pts[:N]).flatten())
        
    np.random.seed(42)
    for _ in range(4):
        starts.append(np.random.uniform(0.15, 0.85, (N, 2)).flatten())
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_x = None
    best_val = np.inf
    
    starts = generate_starts()
    rng = np.random.default_rng(123)
    
    # Phase 1: Basin Hopping Global Search on Centers
    for x0 in starts:
        x0 = np.clip(x0.reshape(N, 2), 0.02, 0.98).flatten()
        
        bp_minimizer = {'method': 'Powell', 'options': {'ftol': 1e-9, 'xtol': 1e-9, 'maxiter': 600}}
        try:
            bp = basinhopping(obj_func, x0, minimizer_kwargs=bp_minimizer, stepsize=0.014, 
                              T=0.015, niter=35, interval=50, disp=False)
            if bp.fun < best_val:
                best_val = bp.fun
                best_x = bp.x.copy()
        except Exception:
            pass
            
    c_best = best_x.reshape(N, 2)
    s_best, r_best = solve_lp(c_best)
    
    # Phase 2: Joint SLSQP Polish (Centers + Radii)
    def obj_joint(v):
        return -np.sum(v[2*N:])
        
    def cons_joint(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        cons = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
        dx = c[IDX_I, 0] - c[IDX_J, 0]
        dy = c[IDX_I, 1] - c[IDX_J, 1]
        dr = r[IDX_I] + r[IDX_J]
        cons.append(dx**2 + dy**2 - dr**2)
        return np.concatenate(cons)
        
    bounds_j = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    v0 = np.concatenate([c_best.flatten(), r_best * 0.96])
    
    try:
        res_sl = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                          constraints={'type': 'ineq', 'fun': cons_joint},
                          options={'maxiter': 20000, 'ftol': 1e-12})
        if np.min(cons_joint(res_sl.x)) >= -1e-8:
            s_new = -res_sl.fun
            if s_new > s_best:
                c_best = res_sl.x[:2*N].reshape(N, 2)
                r_best = res_sl.x[2*N:].copy()
                s_best = s_new

        pass

    for _ in range(100):

                    r_best[j] -= sh
                    changed = True

            if r_best[i] > mr - 1e-12:
                r_best[i] = mr
                changed = True
        if not changed:
            break
            
    r_best = np.maximum(r_best, 0.0)
    return c_best, r_best, float(np.sum(r_best))
