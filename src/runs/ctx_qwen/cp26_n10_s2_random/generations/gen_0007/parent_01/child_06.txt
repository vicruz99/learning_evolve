import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N_CIRCLES = 26
TRIU_INDICES = np.triu_indices(N_CIRCLES, k=1)
NUM_PAIRS = N_CIRCLES * (N_CIRCLES - 1) // 2

# Precompute the constant inequality constraint matrix for the radii LP
A_LP_GLOBAL = np.zeros((NUM_PAIRS + N_CIRCLES, N_CIRCLES))
for idx, (i, j) in enumerate(zip(TRIU_INDICES[0], TRIU_INDICES[1])):
    A_LP_GLOBAL[idx, i] = 1.0
    A_LP_GLOBAL[idx, j] = 1.0
for i in range(N_CIRCLES):
    A_LP_GLOBAL[NUM_PAIRS + i, i] = 1.0

def evaluate_lp(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    c = np.clip(centers, 1e-6, 1.0 - 1e-6)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.empty(NUM_PAIRS + N_CIRCLES)
    b_ub[:NUM_PAIRS] = dists[TRIU_INDICES]
    b_ub[NUM_PAIRS:] = ub
    
    bounds_r = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N_CIRCLES), A_ub=A_LP_GLOBAL, b_ub=b_ub, 
                  bounds=bounds_r, method='highs')
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N_CIRCLES)

def objective_func(x_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    s, _ = evaluate_lp(x_flat.reshape(N_CIRCLES, 2))
    return -s

def generate_initial_starts(rng):
    """Generates diverse initial center configurations."""
    starts = []
    patterns = [[6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 5, 5, 6], 
                [4, 6, 6, 6, 4], [6, 4, 6, 5, 5], [5, 5, 6, 5, 5]]
    
    for pat in patterns:
        for scale in [0.95, 1.0, 1.05, 1.1]:
            pts = []
            r0 = 0.10 / scale
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(pts) < N_CIRCLES:
                        pts.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            pts = np.array(pts[:N_CIRCLES])
            pts += rng.normal(0, 0.004, pts.shape)
            starts.append(np.clip(pts, 0.02, 0.98))
            
    for _ in range(20):
        starts.append(rng.uniform(0.05, 0.95, (N_CIRCLES, 2)))
    return starts

def run_packing():
    rng = np.random.default_rng(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # 1. Differential Evolution for robust global search on centers
    bounds_de = [(0.05, 0.95)] * (2 * N_CIRCLES)
    res_de = differential_evolution(objective_func, bounds_de, popsize=15, maxiter=250, 
                                    seed=42, workers=1, tol=1e-7, polish=False)
    s_de, r_de = evaluate_lp(res_de.x.reshape(N_CIRCLES, 2))
    best_sum = s_de
    best_centers = res_de.x.reshape(N_CIRCLES, 2).copy()
    best_radii = r_de.copy()
    
    # 2. Multi-start Powell optimization to escape local minima
    starts = generate_initial_starts(rng)
    all_starts = [best_centers] + starts
    
    for c0 in all_starts:
        res_pw = minimize(objective_func, c0.flatten(), method='Powell', 
                          options={'maxiter': 3000, 'ftol': 1e-12, 'xtol': 1e-12})
        s_pw, r_pw = evaluate_lp(res_pw.x.reshape(N_CIRCLES, 2))
        if s_pw > best_sum:
            best_sum = s_pw
            best_centers = res_pw.x.reshape(N_CIRCLES, 2).copy()
            best_radii = r_pw.copy()
            
    # 3. Simulated Annealing refinement for high-quality geometry
    temp = 0.004
    for _ in range(800):
        c_trial = best_centers + rng.normal(0, temp, best_centers.shape)
        c_trial = np.clip(c_trial, 0.01, 0.99)
        s_trial, r_trial = evaluate_lp(c_trial)
        
        if s_trial > best_sum:
            best_sum = s_trial
            best_centers = c_trial.copy()
            best_radii = r_trial.copy()
        elif rng.random() < np.exp((s_trial - best_sum) / max(temp, 1e-9)):
            best_centers = c_trial.copy()
            best_radii = r_trial.copy()
            
        temp *= 0.998
        
    # 4. Joint SLSQP Polish (centers + radii) for final micro-adjustments
    def obj_joint(v):
        return -np.sum(v[2 * N_CIRCLES:])
        
    def cons_joint(v):
        c = v[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
        r = v[2 * N_CIRCLES:]
        con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
        dx = c[TRIU_INDICES[0], 0] - c[TRIU_INDICES[1], 0]
        dy = c[TRIU_INDICES[0], 1] - c[TRIU_INDICES[1], 1]
        dr = r[TRIU_INDICES[0]] + r[TRIU_INDICES[1]]
        con.append(dx**2 + dy**2 - dr**2)
        return np.concatenate(con)
        
    v0 = np.concatenate([best_centers.flatten(), best_radii * 0.995])
    bounds_j = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    try:
        res_slsqp = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                             constraints={'type': 'ineq', 'fun': cons_joint},
                             options={'maxiter': 15000, 'ftol': 1e-13, 'catol': 1e-13})
        if -res_slsqp.fun > best_sum - 1e-7:
            best_centers = res_slsqp.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
            best_radii = res_slsqp.x[2 * N_CIRCLES:]
            best_sum = -res_slsqp.fun
    except Exception:
        pass
        
    # 5. Strict Numerical Repair to guarantee validation tolerance compliance
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for _ in range(150):
        changed = False
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N_CIRCLES):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
