

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we employ a hybrid optimization strategy that separates the center positioning from radius determination, while still allowing joint refinement:

1. **Exact LP for Radii**: For any fixed set of center coordinates, the problem of maximizing the sum of radii subject to non-overlap and boundary constraints is a linear program. We precompute the LP constraint matrix and use `scipy.optimize.linprog` (HiGHS solver) to compute optimal radii instantaneously. This gives us a precise objective function `f(centers) = sum(LP_radii(centers))`.

2. **Global Search (Differential Evolution)**: We run `differential_evolution` on the 52-dimensional center space. DE is robust for non-convex landscapes and effectively explores diverse packing configurations (hexagonal, random, boundary-hugging) without getting trapped in shallow local minima.

3. **Local Refinement (Powell)**: Multiple diverse initial configurations (hexagonal lattices, force-repulsion spreads, random) are refined using Powell's method, which works well for derivative-free optimization of our LP-based objective.

4. **Basin Hopping**: A perturbation phase with decaying noise is applied to the best configuration found so far. This helps escape residual local optima by exploring neighboring center geometries and accepting improvements or occasional neutral moves.

5. **Joint SLSQP Polish**: Finally, we perform a constrained joint optimization of centers and radii using SLSQP. This phase handles the non-smooth transitions where circles make/break contact, squeezing out the last bits of precision.

6. **Strict Numerical Repair**: An iterative shrinking/clamping pass guarantees that the returned configuration strictly satisfies the validator's tolerance constraints (`1e-12`), preventing rejection due to floating-point drift.

```python
import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
NUM_PAIRS = len(IDX_I)

# Precompute LP constraint matrix structure
# Rows 0 to NUM_PAIRS-1: pairwise distance constraints r_i + r_j <= dist(i,j)
# Rows NUM_PAIRS to NUM_PAIRS+N-1: boundary constraints r_i <= min(x,1-x,y,1-y)
A_LP = np.zeros((NUM_PAIRS + N, N))
p_idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[p_idx, i] = 1.0
        A_LP[p_idx, j] = 1.0
        p_idx += 1
for i in range(N):
    A_LP[p_idx + i, i] = 1.0

def fast_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    c = np.clip(centers, 1e-5, 1.0 - 1e-5)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b = np.empty(NUM_PAIRS + N)
    b[:NUM_PAIRS] = dists[IDX_I, IDX_J]
    b[NUM_PAIRS:] = ub
    
    bounds_r = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, bounds=bounds_r, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(N), 0.0

def obj_centers(x):
    """Objective for center optimization: minimize negative sum of LP radii."""
    return -fast_lp_radii(x.reshape(N, 2))[1]

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def cons_joint(v):
    """Computes boundary and non-overlap constraints for joint optimization."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    
    cons = []
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    dx = c[IDX_I, 0] - c[IDX_J, 0]
    dy = c[IDX_I, 1] - c[IDX_J, 1]
    dr = r[IDX_I] + r[IDX_J]
    cons.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(cons)

def generate_starts(rng):
    """Generates diverse initial center configurations."""
    starts = []
    
    # 1. Hexagonal lattice patterns
    patterns = [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], [5,5,6,5,5]]
    for pat in patterns:
        pts = []
        y = 0.09
        r_est = 0.09
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        starts.append(np.array(pts[:N]))
        
    # 2. Force-repulsion spread (simulates dense packing)
    for _ in range(5):
        c = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(500):
            forces = np.zeros_like(c)
            diff = c[:, None, :] - c[None, :, :]
            dists = np.linalg.norm(diff, axis=2)
            np.fill_diagonal(dists, 1.0)
            f_mag = 0.005 / (dists**2 + 0.0001)
            forces += np.sum(diff * f_mag[:, :, None], axis=1)
            c += 0.002 * forces
            c = np.clip(c, 0.02, 0.98)
        starts.append(c)
        
    # 3. Random dense starts
    for _ in range(10):
        starts.append(rng.uniform(0.05, 0.95, (N, 2)))
    return starts

def run_packing():
    rng = np.random.default_rng(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # --- Phase 1: Global optimization with Differential Evolution ---
    bounds_de = [(0.02, 0.98)] * (2 * N)
    try:
        res_de = differential_evolution(obj_centers, bounds_de, popsize=30, maxiter=1200, 
                                        seed=42, workers=1, tol=1e-7, polish=False)
        c_de = res_de.x.reshape(N, 2)
        _, s_de = fast_lp_radii(c_de)
        if s_de > best_sum:
            best_sum = s_de
            best_centers = c_de
            best_radii, _ = fast_lp_radii(best_centers)
    except Exception:
        pass
        
    # --- Phase 2: Local refinement from diverse starts ---
    starts = generate_starts(rng)
    if best_centers is not None:
        starts.append(best_centers)
        
    for c_init in starts:
        c_init = np.clip(c_init, 0.02, 0.98)
        res_powell = minimize(obj_centers, c_init.flatten(), method='Powell', 
                              options={'maxiter': 2000, 'ftol': 1e-12, 'xtol': 1e-12})
        c_opt = res_powell.x.reshape(N, 2)
        _, s_opt = fast_lp_radii(c_opt)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            best_radii, _ = fast_lp_radii(best_centers)
            
    # --- Phase 3: Basin Hopping / Perturbation Search ---
    if best_centers is not None:
        curr_c = best_centers.copy()
        for pass_idx in range(3):
            noise_scale = 0.005 * (0.7 ** pass_idx)
            for step in range(100):
                c_trial = curr_c + rng.normal(0, noise_scale, curr_c.shape)
                c_trial = np.clip(c_trial, 0.01, 0.99)
                _, s_trial = fast_lp_radii(c_trial)
                if s_trial > best_sum + 1e-9:
                    best_sum = s_trial
                    best_centers = c_trial
                    best_radii, _ = fast_lp_radii(best_centers)
                    curr_c = c_trial
                elif rng.random() < 0.1:
                    curr_c = c_trial
                    
        # Final SLSQP joint polish to handle non-smooth contact constraints
        r_init = best_radii * 0.93
        v0 = np.concatenate([best_centers.flatten(), r_init])
        bounds_joint = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        cons_dict = {'type': 'ineq', 'fun': cons_joint}
        
        try:
            res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                           constraints=cons_dict, options={'maxiter': 20000, 'ftol': 1e-14})
            if np.min(cons_joint(res.x)) >= -1e-8:
                c_opt = res.x[:2 * N].reshape(N, 2)
                s_lp, r_lp = fast_lp_radii(c_opt)
                if s_lp > best_sum:
                    best_sum = s_lp
                    best_centers = c_opt
                    best_radii = r_lp
        except Exception:
            pass

    # --- Phase 4: Strict Numerical Repair ---
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for _ in range(100):
        changed = False
        # Resolve overlaps by symmetric shrinking
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-12:
                radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
```
