import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute LP constraint matrix structure globally
A_ub_lp = np.zeros((NUM_PAIRS + 4 * N, N))
idx = 0
for i, j in zip(I, J):
    A_ub_lp[idx, i] = 1.0
    A_ub_lp[idx, j] = 1.0
    idx += 1
for i in range(N):
    for _ in range(4):
        A_ub_lp[idx, i] = 1.0
        idx += 1

def get_lp_radii_and_grad(centers):
    """Solves LP for optimal radii given fixed centers and computes gradient w.r.t centers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-7)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b = np.zeros(NUM_PAIRS + 4 * N)
    idx = 0
    for i, j in zip(I, J):
        b[idx] = dists[i, j]
        idx += 1

        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub_lp, b_ub=b, 
                  bounds=[(0, u) for u in ub], method='highs')
    if not res.success:
        return None, None, None
        
    radii = res.x
    duals = res.ineqlin.marginals
    grad = np.zeros_like(centers)

    idx = 0
    for i, j in zip(I, J):
        lam = duals[idx]
        if lam > 1e-8:

            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    # Boundary forces from wall constraints
    bound_start = NUM_PAIRS
    for i in range(N):

        mu_B = duals[bound_start + 4 * i + 2]
        mu_T = duals[bound_start + 4 * i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, np.sum(radii), grad

def gradient_ascent(c0, rng, steps=1200, init_step=0.012):
    """Performs gradient ascent on centers using LP duals."""
    centers = c0.copy()
    best_c = centers.copy()
    best_s = -1.0
    step = init_step
    no_imp = 0
    
    for k in range(steps):
        r, s, g = get_lp_radii_and_grad(centers)
        if r is None:
            break
            
        if s > best_s:
            best_s = s
            best_c = centers.copy()
            no_imp = 0
        else:
            no_imp += 1
            
        if no_imp > 60:
            step *= 0.75
        elif no_imp > 20:
            step *= 0.9
        if step < 1e-10:
            break
            
        # Jitter to escape subgradient plateaus
        jitter = np.zeros_like(centers)
        if k % 50 == 0:
            jitter = rng.normal(0, step * 0.15, centers.shape)
            
        centers += step * (g + jitter)
        centers = np.clip(centers, 0.001, 0.999)
        
    return best_c, best_s

def slsqp_polish(centers, radii, rng):
    """Joints SLSQP optimization on centers and radii."""
    def obj(v):
        return -np.sum(v[2 * N:])
        
    def cons(v):
        c = v[:2 * N].reshape(N, 2)
        r = v[2 * N:]
        con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
        dx = c[I, 0] - c[J, 0]
        dy = c[I, 1] - c[J, 1]
        dr = r[I] + r[J]
        # Squared constraints for better conditioning
        con.append(dx**2 + dy**2 - dr**2)
        return np.concatenate(con)
        
    v0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 5000, 'ftol': 1e-14})
        if np.all(cons(res.x) >= -1e-8):
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:]
    except Exception:
        pass
    return centers, radii

def generate_starts(n, rng):

    patterns = [

                x = r0 + shift

                    c.append([x, y])
                    x += 2.0 * r0

            c = np.array(c[:n])

            c = np.clip(c, 0.02, 0.98)
