import numpy as np
from scipy.optimize import linprog, minimize

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute LP constraint matrix structure globally
A_ub_lp = np.zeros((NUM_PAIRS + 4 * N, N))
idx = 0
for i, j in zip(I, J):
    A_ub_lp[idx, i] = 1.0
    A_ub_lp[idx, j] = 1.0
    idx += 1
for i in range(N):
    for _ in range(4):
        A_ub_lp[idx, i] = 1.0
        idx += 1

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii given fixed centers and computes gradient via duals."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b = np.zeros(NUM_PAIRS + 4 * N)
    p_idx = 0
    for i, j in zip(I, J):
        b[p_idx] = dists[i, j]
        p_idx += 1
    for i in range(N):
        b[p_idx] = centers[i, 0]; p_idx += 1
        b[p_idx] = 1.0 - centers[i, 0]; p_idx += 1
        b[p_idx] = centers[i, 1]; p_idx += 1
        b[p_idx] = 1.0 - centers[i, 1]; p_idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub_lp, b_ub=b, 
                  bounds=[(0, u) for u in ub], method='highs')
    
    if not res.success:
        return None, -1.0, np.zeros_like(centers)
        
    radii = res.x
    duals = res.ineqlin.marginals
    
    grad = np.zeros_like(centers)
    p_idx = 0
    for i, j in zip(I, J):
        lam = duals[p_idx]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        p_idx += 1
        
    b_start = NUM_PAIRS
    for i in range(N):
        mu_L = duals[b_start + 4 * i]
        mu_R = duals[b_start + 4 * i + 1]
        mu_B = duals[b_start + 4 * i + 2]
        mu_T = duals[b_start + 4 * i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, np.sum(radii), grad

def gradient_ascent(centers0, steps=3000, init_step=0.015):
    """Runs gradient ascent on centers using LP dual forces."""
    centers = centers0.copy()
    best_centers = centers.copy()
    best_sum = -1.0
    step = init_step
    no_improve = 0
    
    for k in range(steps):
        radii, curr_sum, grad = solve_lp_and_grad(centers)
        if radii is None:
            break
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        # Adaptive step decay
        if no_improve > 100:
            step *= 0.5
        elif no_improve > 40:
            step *= 0.8
            
        if step < 1e-10:
            break
            
        # Occasional jitter to escape discrete constraint plateaus
        if k % 150 == 0:
            centers += np.random.normal(0, step * 0.2, centers.shape)
            
        centers += step * grad
        centers = np.clip(centers, 0.01, 0.99)
        
    return best_centers, best_sum

def generate_starts(n, rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5]
    ]
    for pat in patterns:
        for r0 in [0.09, 0.095, 0.10, 0.105]:
            c = []
            y = r0

                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:n])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.02, 0.98)
            starts.append(c)
            
    for _ in range(12):
        starts.append(rng.uniform(0.15, 0.85, (n, 2)))
    return starts

def joint_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def joint_cons(v):
    """Computes all boundary and non-overlap constraints for SLSQP."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    dx = c[I, 0] - c[J, 0]
    dy = c[I, 1] - c[J, 1]
    con.append(np.sqrt(dx**2 + dy**2) - (r[I] + r[J]))
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

            best_sum = s_opt
            best_centers = c_opt.copy()

    r_lp, _, _ = solve_lp_and_grad(best_centers)
    v0 = np.concatenate([best_centers.flatten(), r_lp])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': joint_cons}
    
    try:
        res = minimize(joint_obj, v0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
        c_vals = joint_cons(res.x)

            best_centers = res.x[:2 * N].reshape(N, 2)
            best_sum = -res.fun
    except Exception:
        pass
        
    # Phase 4: Exact LP resolution for final radii
    best_radii, final_sum, _ = solve_lp_and_grad(best_centers)
    
    # Phase 5: Strict numerical repair to guarantee validator tolerance
    for _ in range(150):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                req = best_radii[i] + best_radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            mr = min(best_centers[i, 0], 1.0 - best_centers[i, 0],

            if best_radii[i] > mr + 1e-12:
                best_radii[i] = mr
                changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    final_sum = float(np.sum(best_radii))
    return best_centers, best_radii, final_sum
