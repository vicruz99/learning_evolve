import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute LP constraint matrix structure globally
A_ub_lp = np.zeros((NUM_PAIRS + 4*N, N))
idx = 0
for i, j in zip(I, J):
    A_ub_lp[idx, i] = 1.0
    A_ub_lp[idx, j] = 1.0
    idx += 1
for i in range(N):
    for _ in range(4):
        A_ub_lp[idx, i] = 1.0
        idx += 1

def solve_lp(centers):
    """Solves LP to find optimal radii for fixed centers and returns duals."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b = np.zeros(NUM_PAIRS + 4*N)
    idx = 0
    for i, j in zip(I, J):
        b[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub_lp, b_ub=b, bounds=[(0, u) for u in ub], method='highs')
    if res.success:
        return res.x, res.ineqlin.marginals, -res.fun
    return ub * 0.05, np.zeros(NUM_PAIRS + 4*N), -np.inf

def grad_ascent(centers0, steps=1500, lr0=0.015):
    """Gradient ascent on centers using LP duals to maximize sum of radii."""
    rng = np.random.default_rng(42)
    centers = centers0.copy()
    best_sum = -np.inf
    best_centers = centers.copy()
    no_improve = 0
    
    for step in range(steps):
        radii, duals, curr_sum = solve_lp(centers)
        if curr_sum == -np.inf:
            break
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        grad = np.zeros_like(centers)
        idx = 0
        # Pairwise repulsion from active distance constraints
        for i, j in zip(I, J):
            lam = duals[idx]
            if lam > 1e-8:
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d > 1e-9:
                    vec = (centers[i] - centers[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
            idx += 1
            
        # Boundary attraction from active wall constraints
        bound_start = NUM_PAIRS
        for i in range(N):
            grad[i, 0] += duals[bound_start + 4*i] - duals[bound_start + 4*i + 1]
            grad[i, 1] += duals[bound_start + 4*i + 2] - duals[bound_start + 4*i + 3]
            
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-9:
            grad /= grad_norm
            
        # Adaptive decaying learning rate
        lr = lr0 / (1.0 + step * 0.0008)
        centers += lr * grad
        centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
        
        # Periodic jitter to escape subgradient plateaus
        if step % 300 == 0 and step > 0:
            centers += rng.normal(0, 0.0006, centers.shape)
            centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
            
        if no_improve > 250 and lr < 1e-5:
            break
            
    return best_centers, best_sum

def compute_obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def compute_cons(v):
    """Computes all boundary and non-overlap constraints for SLSQP."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    dx = c[I, 0] - c[J, 0]
    dy = c[I, 1] - c[J, 1]
    con.append(np.hypot(dx, dy) - (r[I] + r[J]))
    return np.concatenate(con)

def generate_hex_start(pat, r0, rng):
    """Generates a hexagonal lattice initialization."""
    c = []
    y = r0
    for r_idx, cnt in enumerate(pat):
        shift = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(c) < N:
                c.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    c = np.array(c[:N])
    c += rng.normal(0, 0.002, c.shape)
    c = np.clip(c, 0.02, 0.98)
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    
    best_c = None
    best_sum = -np.inf
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5]
    ]
    
    candidates = []
    for pat in patterns:
        for r0 in [0.092, 0.096, 0.100, 0.104, 0.108]:
            candidates.append(generate_hex_start(pat, r0, rng))
    for _ in range(10):
        candidates.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # Phase 1: Gradient Ascent from diverse starts
    for c0 in candidates:
        c_opt, s_opt = grad_ascent(c0)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            
    # Phase 2: SLSQP Joint Polish
    r0, _, _ = solve_lp(best_c)
    v0 = np.concatenate([best_c.flatten(), r0])
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_cons}
    
    try:
        res = minimize(compute_obj, v0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 6000, 'ftol': 1e-14})
        if -res.fun > best_sum:
            best_c = res.x[:2*N].reshape(N, 2)
            best_sum = -res.fun
    except Exception:
        pass
        
    # Phase 3: Perturbed SLSQP to escape local plateaus
    v0_pert = best_c.flatten().copy()
    v0_pert += rng.normal(0, 0.0003, 2*N)
    v0_pert = np.clip(v0_pert, 0.01, 0.99)
    r_pert, _, _ = solve_lp(v0_pert.reshape(N, 2))
    v0_pert_full = np.concatenate([v0_pert, r_pert])
    
    try:
        res2 = minimize(compute_obj, v0_pert_full, method='SLSQP', bounds=bounds,
                        constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14})
        if -res2.fun > best_sum:
            best_c = res2.x[:2*N].reshape(N, 2)
            best_sum = -res2.fun
    except Exception:
        pass
        
    # Phase 4: Final exact LP resolution
    best_radii, _, _ = solve_lp(best_c)
    
    # Phase 5: Strict numerical repair for validator tolerance
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                if d < best_radii[i] + best_radii[j] - 1e-12:
                    shrink = (best_radii[i] + best_radii[j] - d) / 2.0 + 1e-9
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(best_c[i,0], 1.0-best_c[i,0], best_c[i,1], 1.0-best_c[i,1])
            if best_radii[i] > mr + 1e-12:
                best_radii[i] = mr
                changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_c, best_radii, float(np.sum(best_radii))
