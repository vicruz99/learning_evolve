import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute LP constraint matrix structure globally for speed
A_ub_lp = np.zeros((NUM_PAIRS + 4*N, N))
idx = 0
for i, j in zip(I, J):
    A_ub_lp[idx, i] = 1.0
    A_ub_lp[idx, j] = 1.0
    idx += 1
for i in range(N):
    for _ in range(4):
        A_ub_lp[idx, i] = 1.0
        idx += 1

def get_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b = np.zeros(NUM_PAIRS + 4*N)
    idx = 0
    for i, j in zip(I, J):
        b[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_lp, b_ub=b, bounds=[(0, u) for u in ub], method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return ub * 0.5

def obj_centers(x):
    """Objective for center-only optimization: maximize sum of LP radii."""
    c = x.reshape(N, 2)
    return -np.sum(get_lp_radii(c))

def cons_joint(v):
    """Computes all boundary and non-overlap constraints for SLSQP."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    dx = c[I, 0] - c[J, 0]
    dy = c[I, 1] - c[J, 1]
    con.append(np.sqrt(dx**2 + dy**2) - (r[I] + r[J]))
    return np.concatenate(con)

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def generate_starts(n, rng):

    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5]
    ]
    
    # Standard hexagonal patterns
    for pat in patterns:
        for r0 in [0.08, 0.085, 0.09, 0.095, 0.10]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < n:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:n])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.02, 0.98)
            starts.append(c)
            
    # Rotated hexagonal patterns (often fit square boundaries better)
    for pat in patterns[:4]:
        for rot in [0.05, 0.1, 0.15, 0.2]:
            c_base = []
            r0 = 0.09
            y = r0

                shift = r0 if r_idx % 2 == 1 else 0.0

                    x += 2.0 * r0

            c_mat, s_mat = np.cos(rot), np.sin(rot)
            rot_mat = np.array([[c_mat, -s_mat], [s_mat, c_mat]])
            c_rot = c_base @ rot_mat.T
            scale = 0.9 / (2 * np.max(np.abs(c_rot)))
            c_rot *= scale
            c_rot += 0.5
            c_rot += rng.normal(0, 0.003, c_rot.shape)
            c_rot = np.clip(c_rot, 0.02, 0.98)
            starts.append(c_rot)

                        diff = c[i] - c[j]
                        forces[i] += diff * f
                        forces[j] -= diff * f

                    elif c[i, d_idx] > 0.9: forces[i, d_idx] -= (c[i, d_idx] - 0.9) * 10.0
            c += forces * 0.005
            c = np.clip(c, 0.02, 0.98)
        starts.append(c)
        
    # Dense random starts
    for _ in range(8):
        starts.append(rng.uniform(0.1, 0.9, (n, 2)))
    return starts

    rng = np.random.default_rng(42)

    best_sum = -np.inf

    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
