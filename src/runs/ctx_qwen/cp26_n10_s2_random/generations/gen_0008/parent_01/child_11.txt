

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute LP constraint matrix structure globally
A_ub_lp = np.zeros((NUM_PAIRS + 4*N, N))
idx = 0
for i, j in zip(I, J):
    A_ub_lp[idx, i] = 1.0
    A_ub_lp[idx, j] = 1.0
    idx += 1
for i in range(N):
    for _ in range(4):
        A_ub_lp[idx, i] = 1.0
        idx += 1

def solve_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b = np.zeros(NUM_PAIRS + 4*N)
    idx = 0
    for i, j in zip(I, J):
        b[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub_lp, b_ub=b, bounds=[(0, u) for u in ub], method='highs')
    if res.success:
        return res.x, -res.fun
    return ub * 0.5, 0.0

def lp_obj_flat(centers_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    return -solve_lp_radii(centers_flat.reshape(N, 2))[1]

def generate_starts(n, rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5]
    ]
    for pat in patterns:
        for r0 in [0.085, 0.090, 0.095, 0.100, 0.105]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:n])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.01, 0.99)
            starts.append(c)
            
    for _ in range(15):
        starts.append(rng.uniform(0.05, 0.95, (n, 2)))
        
    # Corner-anchored starts to encourage larger boundary circles
    for _ in range(5):
        c = rng.uniform(0.2, 0.8, (n, 2))
        c[0] = [0.12, 0.12]
        c[1] = [0.88, 0.12]
        c[2] = [0.12, 0.88]
        c[3] = [0.88, 0.88]
        starts.append(c)
    return starts

def obj_joint(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Computes all boundary and non-overlap constraints for SLSQP."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    dx = c[I,0]-c[J,0]; dy = c[I,1]-c[J,1]
    con.append(np.sqrt(dx**2+dy**2)-(r[I]+r[J]))
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_centers = None
    best_sum = -1.0
    
    starts = generate_starts(N, rng)
    
    # Phase 1: Powell optimization on centers using LP objective
    bounds_c = [(0.005, 0.995)] * (2*N)
    for c0 in starts:
        try:
            res = minimize(lp_obj_flat, c0.flatten(), method='Powell', bounds=bounds_c,
                           options={'maxiter': 1500, 'ftol': 1e-12, 'xtol': 1e-12})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt = solve_lp_radii(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
        except Exception:
            pass
            
    if best_centers is None:
        best_centers = starts[0]
        _, best_sum = solve_lp_radii(best_centers)
        
    # Phase 2: Iterative Perturbation & Local Search to escape local minima
    c_curr = best_centers.copy()
    s_curr = best_sum
    step = 0.006
    for epoch in range(30):
        improved = False
        for i in range(N):
            best_local = s_curr
            best_pos = c_curr[i].copy()
            # Try multiple perturbations per circle
            for _ in range(20):
                nc = c_curr.copy()
                nc[i] += rng.normal(0, step, 2)
                nc[i] = np.clip(nc[i], 0.02, 0.98)
                _, s_new = solve_lp_radii(nc)
                if s_new > best_local:
                    best_local = s_new
                    best_pos = nc[i].copy()
            if best_local > s_curr + 1e-10:
                c_curr[i] = best_pos
                improved = True
        if improved:
            _, s_curr = solve_lp_radii(c_curr)
            if s_curr > best_sum:
                best_sum = s_curr
                best_centers = c_curr.copy()
        else:
            step *= 0.85
            
    # Phase 3: SLSQP Joint Polish for micro-adjustments
    r_lp, _ = solve_lp_radii(best_centers)
    v0 = np.concatenate([best_centers.flatten(), r_lp])
    bounds_j = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                       constraints={'type':'ineq', 'fun':cons_joint},
                       options={'maxiter':6000, 'ftol':1e-13})
        if np.sum(res.x[2*N:]) > best_sum:
            best_centers = res.x[:2*N].reshape(N, 2)
            best_sum = np.sum(res.x[2*N:])
    except Exception:
        pass
        
    # Final exact LP resolution for radii
    best_radii = solve_lp_radii(best_centers)[0]
    
    # Strict numerical repair to guarantee validator tolerance compliance
    for _ in range(60):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                if d < best_radii[i] + best_radii[j] - 1e-12:
                    shrink = (best_radii[i] + best_radii[j] - d)/2.0 + 1e-9
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(best_centers[i,0], 1.0-best_centers[i,0], best_centers[i,1], 1.0-best_centers[i,1])
            if best_radii[i] > mr + 1e-12:
                best_radii[i] = mr
                changed = True
        if not changed: break
        
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
```
