def generate_hex_config(row_counts, r0, rot=0.0):
    """Generate hexagonal lattice configuration."""
    pts = []
    y = r0
    for r_idx, cnt in enumerate(row_counts):
        shift = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    
    pts = np.array(pts[:N])
    
    if rot != 0:
        c, s = np.cos(rot), np.sin(rot)
        center = pts.mean(axis=0)
        pts_centered = pts - center
        pts_rotated = pts_centered @ np.array([[c, -s], [s, c]])
        pts = pts_rotated + center
    
    # Scale to fit in unit square with margin
    min_pt = pts.min(axis=0)
    max_pt = pts.max(axis=0)
    pts = (pts - min_pt) / (max_pt - min_pt + 1e-9)
    pts = pts * 0.9 + 0.05
    
    return np.clip(pts, 0.01, 0.99)

def compute_cons(v):
