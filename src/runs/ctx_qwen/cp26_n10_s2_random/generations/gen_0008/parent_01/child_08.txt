from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)
NUM_CONSTRAINTS = NUM_PAIRS + 4 * N

# Precompute LP constraint matrix structure globally for speed
A_ub_lp = np.zeros((NUM_CONSTRAINTS, N))
pair_indices = []
idx = 0
for i, j in zip(I, J):
    A_ub_lp[idx, i] = 1.0
    A_ub_lp[idx, j] = 1.0
    pair_indices.append((i, j))
    idx += 1
for i in range(N):
    for _ in range(4):
        A_ub_lp[idx, i] = 1.0
        idx += 1

def solve_lp_and_forces(centers):

                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-10)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]

        c = np.clip(c, 1e-5, 1.0 - 1e-5)

def obj_joint(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Computes boundary and non-overlap constraints for SLSQP."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)

    dy = c[I, 1] - c[J, 1]
    con.append(np.sqrt(dx**2 + dy**2) - (r[I] + r[J]))
    return np.concatenate(con)
