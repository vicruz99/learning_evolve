1. **Decouple Centers and Radii via LP**: For fixed centers, finding optimal radii is a linear program. This creates an objective function over centers only: `f(c) = max_sum_radii(c)`. While `f(c)` is non-smooth (piecewise linear), it's well-suited for derivative-free optimizers.
2. **Powell's Method for Centers**: Powell's method doesn't require gradients and handles the non-smooth LP objective much better than gradient-based methods like SLSQP. We run it from many diverse hexagonal lattice starts.
3. **Basin-Hopping / Simulated Annealing**: After finding a local optimum with Powell, we perturb centers with controlled Gaussian noise that cools over time. This helps escape shallow local basins that trap standard multi-start approaches.
4. **Joint SLSQP Polish**: Once centers are highly optimized, we switch to a joint optimization of `(centers, radii)` using SLSQP. We use squared-distance constraints (`dx^2 + dy^2 >= (r_i + r_j)^2`) for better numerical conditioning and smoothness near contact points.
5. **Strict Numerical Repair**: Finally, we iteratively clamp radii to boundary limits and shrink overlapping pairs by the exact overlap amount plus a tiny safety margin (`1e-10`) to strictly satisfy the validator's `1e-12` tolerance without sacrificing meaningful radius sum.
6. **Precision Tuning**: We set extremely tight tolerances (`ftol=1e-14`, `xtol=1e-14`) and increase iteration limits to ensure the optimizer converges to the mathematical limit of each basin.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute constant LP constraint matrix structure
A_ub_lp = np.zeros((NUM_PAIRS + 4*N, N))
idx = 0
for i, j in zip(I, J):
    A_ub_lp[idx, i] = 1.0
    A_ub_lp[idx, j] = 1.0
    idx += 1
for i in range(N):
    for _ in range(4):
        A_ub_lp[idx, i] = 1.0
        idx += 1

def get_lp_radii_sum(centers):
    """Solves LP to find optimal radii and their sum for fixed centers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b = np.zeros(NUM_PAIRS + 4*N)
    idx = 0
    for i, j in zip(I, J):
        b[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub_lp, b_ub=b, 
                  bounds=[(0, u) for u in ub], method='highs')
    if res.success:
        return res.x, -res.fun
    return ub, np.sum(ub)

def obj_func(v):
    """Objective for center optimization: minimize negative LP sum of radii."""
    return -get_lp_radii_sum(v.reshape(N, 2))[1]

def compute_cons_joint(v):
    """Computes all boundary and non-overlap constraints for joint optimization."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    cons = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    dx = c[I, 0] - c[J, 0]
    dy = c[I, 1] - c[J, 1]
    dr = r[I] + r[J]
    # Squared distance constraint for smooth gradients
    cons.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(cons)

def compute_obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def generate_starts(n, rng):
    """Generates diverse hexagonal and random initial configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5],
        [5, 6, 4, 6, 5], [6, 5, 4, 6, 5], [4, 6, 5, 6, 5]
    ]
    for pat in patterns:
        for r0 in [0.090, 0.095, 0.100, 0.105, 0.110]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:n])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.01, 0.99)
            starts.append(c)
            
    for _ in range(20):
        starts.append(rng.uniform(0.05, 0.95, (n, 2)))
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_c = None
    best_s = -np.inf
    
    starts = generate_starts(N, rng)
    bounds_c = [(0.001, 0.999)] * (2*N)
    
    # Phase 1: Multi-start Powell optimization on centers
    for c0 in starts:
        try:
            res = minimize(obj_func, c0.flatten(), method='Powell', bounds=bounds_c,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'xtol': 1e-14})
            if -res.fun > best_s:
                best_s = -res.fun
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    # Phase 2: Basin-hopping style perturbations to escape local minima
    if best_c is not None:
        c_curr = best_c.copy()
        s_curr = best_s
        T = 0.005
        for step in range(60):
            T *= 0.98
            c_trial = c_curr + rng.normal(0, T, c_curr.shape)
            c_trial = np.clip(c_trial, 0.01, 0.99)
            try:
                res = minimize(obj_func, c_trial.flatten(), method='Powell', bounds=bounds_c,
                               options={'maxiter': 3000, 'ftol': 1e-13, 'xtol': 1e-13})
                s_new = -res.fun
                if s_new > s_curr or rng.random() < np.exp((s_new - s_curr) / max(T*2, 1e-7)):
                    c_curr = res.x.reshape(N, 2)
                    s_curr = s_new
                    if s_curr > best_s:
                        best_s = s_curr
                        best_c = c_curr.copy()
            except Exception:
                pass

    # Phase 3: Joint SLSQP polish with squared constraints for precision
    if best_c is not None:
        r_lp, _ = get_lp_radii_sum(best_c)
        v0 = np.concatenate([best_c.flatten(), r_lp])
        bounds_j = [(0.0, 1.0)] * (2*N) + [(1e-8, 0.5)] * N
        cons = {'type': 'ineq', 'fun': compute_cons_joint}
        
        for _ in range(5):
            v_trial = v0 + rng.normal(0, 1e-6, v0.shape)
            v_trial[:2*N] = np.clip(v_trial[:2*N], 0.0, 1.0)
            v_trial[2*N:] = np.clip(v_trial[2*N:], 1e-8, 0.5)
            try:
                res = minimize(compute_obj_joint, v_trial, method='SLSQP',
                               bounds=bounds_j, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-14})
                if np.all(compute_cons_joint(res.x) > -1e-8):
                    s = np.sum(res.x[2*N:])
                    if s > best_s:
                        best_s = s
                        best_c = res.x[:2*N].reshape(N, 2).copy()
                        r_lp = res.x[2*N:].copy()
            except Exception:
                pass
                
        # Final LP ensures radii are mathematically optimal for the polished centers
        r_lp, _ = get_lp_radii_sum(best_c)
    else:
        best_c = rng.uniform(0.1, 0.9, (N, 2))
        r_lp, _ = get_lp_radii_sum(best_c)

    # Phase 4: Strict numerical repair to guarantee validator tolerance compliance
    centers = best_c.copy()
    radii = r_lp.copy()
    
    for _ in range(100):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
