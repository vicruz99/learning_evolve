2. **Derivative-Free Global Search**: We use `scipy.optimize.minimize` with the `Powell` method on the center coordinates, using the negative LP sum as the objective. Powell's method is robust for non-smooth landscapes and doesn't require gradients.
3. **Diverse Initializations**: We generate multiple hexagonal lattice configurations with varying scales, row patterns, and small random perturbations, alongside random and grid starts. This covers different topological arrangements that might lead to higher density.
4. **Joint Constraint Optimization**: After finding good center configurations, we polish the solution using `SLSQP` on joint variables (centers and radii) with explicit boundary and distance constraints. This refines positions and radii simultaneously.
5. **Adaptive Local Search**: A coordinate descent routine perturbs each circle's position, accepts moves that increase the LP sum, and adaptively shrinks the step size. This escapes shallow local minima.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
I, J = np.triu_indices(N, k=1)
NUM_PAIRS = len(I)

# Precompute LP constraint matrix structure globally
A_ub_lp = np.zeros((NUM_PAIRS + 4*N, N))
idx = 0
for i, j in zip(I, J):
    A_ub_lp[idx, i] = 1.0
    A_ub_lp[idx, j] = 1.0
    idx += 1
for i in range(N):
    for _ in range(4):
        A_ub_lp[idx, i] = 1.0
        idx += 1

def get_lp_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.clip(ub, 1e-8, 0.5)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b = np.zeros(NUM_PAIRS + 4*N)
    idx = 0
    for i, j in zip(I, J):
        b[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub_lp, b_ub=b, bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:
        return res.x, np.sum(res.x)
    return ub * 0.1, 0.0

def obj_centers(v):
    """Objective for center optimization: minimize negative LP sum of radii."""
    _, s = get_lp_radii(v.reshape(N, 2))
    return -s

def compute_obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def compute_cons_joint(v):
    """Computes all boundary and non-overlap constraints for SLSQP."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    dx = c[I, 0] - c[J, 0]
    dy = c[I, 1] - c[J, 1]
    con.append(np.sqrt(dx**2 + dy**2) - (r[I] + r[J]))
    return np.concatenate(con)

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5]
    ]
    for pat in patterns:
        for r0 in [0.085, 0.095, 0.105, 0.115]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.004, c.shape)
            c = np.clip(c, 0.02, 0.98)
            starts.append(c)
            
    # Random and grid starts
    for _ in range(10):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    for sp in [5, 6]:
        x = np.linspace(0.12, 0.88, sp)
        y = x.copy()
        cx, cy = np.meshgrid(x, y)
        g = np.column_stack([cx.flatten(), cy.flatten()])
        if len(g) > N:
            g = g[:N]
        starts.append(g)
        
    return starts

def local_search(centers, rng, max_steps=250):
    """Adaptive coordinate descent to refine center positions."""
    c = centers.copy()
    _, best_s = get_lp_radii(c)
    step_size = 0.012
    no_improve = 0
    
    for _ in range(max_steps):
        improved = False
        for i in range(N):
            for _ in range(6):
                dc = rng.normal(0, step_size, 2)
                nc = c.copy()
                nc[i] = np.clip(nc[i] + dc, 0.001, 0.999)
                _, s = get_lp_radii(nc)
                if s > best_s + 1e-10:
                    c = nc
                    best_s = s
                    improved = True
        if not improved:
            step_size *= 0.82
            no_improve += 1
        else:
            no_improve = 0
        if no_improve > 12 or step_size < 1e-7:
            break
    return c, best_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    bounds_centers = [(0.0, 1.0)] * (2*N)
    bounds_joint = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    
    starts = generate_starts(rng)
    
    # Phase 1: Powell on centers with LP objective
    for c0 in starts:
        try:
            res = minimize(obj_centers, c0.flatten(), method='Powell', 
                           bounds=bounds_centers, 
                           options={'maxiter': 1500, 'ftol': 1e-12, 'xtol': 1e-12})
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    # Phase 2: SLSQP joint polish
    if best_centers is not None:
        r_lp, _ = get_lp_radii(best_centers)
        v0 = np.concatenate([best_centers.flatten(), r_lp])
        try:
            res = minimize(compute_obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                           constraints={'type': 'ineq', 'fun': compute_cons_joint},
                           options={'maxiter': 4000, 'ftol': 1e-13})
            if -res.fun > best_sum:
                best_centers = res.x[:2*N].reshape(N, 2)
                best_radii = res.x[2*N:]
                best_sum = -res.fun
        except Exception:
            pass

    # Phase 3: Local search refinement
    if best_centers is not None:
        c_ref, s_ref = local_search(best_centers, rng)
        if s_ref > best_sum:
            best_centers = c_ref
            best_sum = s_ref
            
    # Phase 4: Final LP resolution for exact radii
    if best_centers is not None:
        best_radii, best_sum = get_lp_radii(best_centers)
        
    # Fallback if optimization failed entirely
    if best_centers is None:
        c_init = rng.uniform(0.2, 0.8, (N, 2))
        best_radii, best_sum = get_lp_radii(c_init)
        best_centers = c_init
        
    # Phase 5: Strict numerical repair to guarantee validator tolerance compliance
    for _ in range(60):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                req = best_radii[i] + best_radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            mr = min(best_centers[i,0], 1.0-best_centers[i,0], best_centers[i,1], 1.0-best_centers[i,1])
            if best_radii[i] > mr + 1e-12:
                best_radii[i] = mr
                changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
