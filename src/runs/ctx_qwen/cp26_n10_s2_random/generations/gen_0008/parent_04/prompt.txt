def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

import numpy as np

N = 26

def compute_constraints(p):
    """Vectorized constraint computation for SLSQP."""
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    n_constraints = 4 * N + N * (N - 1) // 2
    c = np.empty(n_constraints)
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist_sq - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    dist_sq = dx * dx + dy * dy
    sum_r_sq = dr * dr
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c[4*N:] = (dist_sq - sum_r_sq)[mask]
    
    return c

def objective(p):

def get_lp_radii(centers):

    b_ub = np.zeros(n_pairs + 4 * N)
    
    idx = 0
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.hypot(centers[i, 0] - centers[j, 0], 
                                 centers[i, 1] - centers[j, 1])
            idx += 1
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(N):
        vals = [centers[i, 0], 1.0 - centers[i, 0], 
                centers[i, 1], 1.0 - centers[i, 1]]
        for v in vals:
            A_ub[idx, i] = 1.0
            b_ub[idx] = v
            idx += 1
            
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0, None)] * N, method='highs')

def compute_radii(centers):

    X = vars_flat.reshape(N, 3)
    xs = X[:, 0]
    ys = X[:, 1]
    rs = X[:, 2]
    
    c = []
    # Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    c.append(xs - rs)
    c.append(1.0 - xs - rs)
    c.append(ys - rs)
    c.append(1.0 - ys - rs)
    
    # Overlap: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    dist_sq = dx**2 + dy**2
    dr_sq = dr**2
    
    c.append(dist_sq[TRI_U_IDX] - dr_sq[TRI_U_IDX])
    return np.concatenate(c)
