import numpy as np
from scipy.optimize import minimize, linprog

N = 26
A_LP = None
PAIRS = None

def setup_lp_structure():
    """Precompute the constant constraint matrix for the LP solver."""
    global A_LP, PAIRS
    n_pairs = N * (N - 1) // 2
    n_bound = 4 * N
    A_LP = np.zeros((n_pairs + n_bound, N))
    PAIRS = []
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_LP[idx, i] = 1.0
            A_LP[idx, j] = 1.0
            PAIRS.append((i, j))
            idx += 1
    for i in range(N):
        for _ in range(4):
            A_LP[idx, i] = 1.0
            idx += 1

def solve_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    ub = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    ub = np.maximum(ub, 1e-9)
    
    b_ub = np.zeros(A_LP.shape[0])
    idx = 0
    for i, j in PAIRS:
        b_ub[idx] = np.hypot(x[i] - x[j], y[i] - y[j])
        idx += 1
    for i in range(N):
        b_ub[idx] = x[i]; idx += 1
        b_ub[idx] = 1.0 - x[i]; idx += 1
        b_ub[idx] = y[i]; idx += 1
        b_ub[idx] = 1.0 - y[i]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(N, 1e-7), 0.0

def obj_centers(v):
    """Objective for center optimization: minimize negative LP sum of radii."""
    return -solve_lp_radii(v.reshape(N, 2))[1]

def generate_starts():
    """Generates diverse initial configurations."""
    starts = []
    rng = np.random.default_rng(42)
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 5, 5, 6],
        [6, 5, 5, 5, 5], [4, 6, 6, 6, 4], [6, 6, 5, 4, 5],
        [5, 7, 5, 5, 4], [6, 4, 6, 5, 5], [5, 5, 6, 5, 5],
        [5, 5, 6, 6, 4], [6, 6, 6, 4, 4], [4, 6, 5, 6, 5]
    ]
    
    for pat in patterns:
        for r0 in np.linspace(0.085, 0.115, 6):
            pts = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    pts.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            pts = np.array(pts[:N])
            
            rot = rng.uniform(-0.2, 0.2)
            c, s = np.cos(rot), np.sin(rot)
            pts_rot = pts @ [[c, -s], [s, c]]
            pts_rot -= pts_rot.min(axis=0)
            span = pts_rot.max(axis=0) - pts_rot.min(axis=0)
            scale = 0.92 / (np.max(span) + 1e-9)
            pts_rot *= scale
            pts_rot += 0.04
            pts_rot += rng.normal(0, 0.003, pts_rot.shape)
            pts_rot = np.clip(pts_rot, 0.02, 0.98)
            starts.append(pts_rot.flatten())
            
    for _ in range(12):
        c = rng.uniform(0.15, 0.85, (N, 2))
        starts.append(c.flatten())
        
    return starts

def coordinate_search(centers, epochs=60):
    """Iteratively moves each circle to locally maximize the LP objective."""
    c = centers.copy()
    best_sum = solve_lp_radii(c)[1]
    step = 0.006
    directions = np.array([
        [step, 0], [-step, 0], [0, step], [0, -step],
        [step, step], [-step, step], [step, -step], [-step, -step]
    ]) * 0.5 / np.sqrt(2)
    directions[0] = [step, 0]
    directions[1] = [-step, 0]
    directions[2] = [0, step]
    directions[3] = [0, -step]
    directions[4] = [step, step] / np.sqrt(2)
    directions[5] = [-step, step] / np.sqrt(2)
    directions[6] = [step, -step] / np.sqrt(2)
    directions[7] = [-step, -step] / np.sqrt(2)

    rng = np.random.default_rng(77)
    for ep in range(epochs):
        improved = False
        order = rng.permutation(N)
        for i in order:
            curr_sum = best_sum
            best_pos = c[i].copy()
            for dx, dy in directions:
                nc = c.copy()
                nc[i] = np.clip(c[i] + [dx, dy], 0.005, 0.995)
                _, s = solve_lp_radii(nc)
                if s > curr_sum + 1e-9:
                    curr_sum = s
                    best_pos = nc[i].copy()
            if curr_sum > best_sum + 1e-9:
                c[i] = best_pos
                best_sum = curr_sum
                improved = True
        if not improved:
            step *= 0.85
            if step < 1e-6:
                break
    return c, best_sum

def simulated_annealing(centers, steps=1500):
    """SA on centers to escape local minima."""
    c = centers.copy()
    _, curr_sum = solve_lp_radii(c)
    best_c = c.copy()
    best_sum = curr_sum
    T = 0.006
    rng = np.random.default_rng(88)
    
    for _ in range(steps):
        T *= 0.9995
        c_new = c + rng.normal(0, T, c.shape)
        c_new = np.clip(c_new, 0.02, 0.98)
        _, new_sum = solve_lp_radii(c_new)
        
        if new_sum > curr_sum:
            c = c_new
            curr_sum = new_sum
            if new_sum > best_sum:
                best_c = c.copy()
                best_sum = new_sum
        else:
            if rng.random() < np.exp((new_sum - curr_sum) / max(T * 4.0, 1e-9)):
                c = c_new
                curr_sum = new_sum
    return best_c, best_sum

def joint_obj(v):
    return -np.sum(v[2*N:])

def joint_cons(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    idx_i, idx_j = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx_i] - c[idx_j], axis=1)
    con.append(d - (r[idx_i] + r[idx_j]))
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    setup_lp_structure()
    bounds_c = [(0.0, 1.0)] * (2 * N)
    bounds_j = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    
    starts = generate_starts()
    best_c = None
    best_sum = -1.0
    
    # Phase 1: Multi-start Powell
    for v0 in starts:
        try:
            res = minimize(obj_centers, v0, method='Powell', bounds=bounds_c,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'xtol': 1e-13})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = starts[0].reshape(N, 2)
        best_sum = solve_lp_radii(best_c)[1]

    # Phase 2: Coordinate Search Refinement
    best_c, best_sum = coordinate_search(best_c, epochs=80)
    
    # Phase 3: Simulated Annealing
    best_c, best_sum = simulated_annealing(best_c, steps=1800)
    
    # Phase 4: Joint SLSQP Polish
    r_lp, _ = solve_lp_radii(best_c)
    v0_joint = np.concatenate([best_c.flatten(), r_lp])
    
    try:
        res_j = minimize(joint_obj, v0_joint, method='SLSQP', bounds=bounds_j,
                         constraints={'type': 'ineq', 'fun': joint_cons},
                         options={'maxiter': 6000, 'ftol': 1e-14})
        if np.min(joint_cons(res_j.x)) >= -1e-8:
            curr_sum = -res_j.fun
            if curr_sum > best_sum:
                best_c = res_j.x[:2*N].reshape(N, 2)
                best_sum = curr_sum
    except Exception:
        pass

    # Final LP resolution
    final_r, final_sum = solve_lp_radii(best_c)
    
    # Phase 5: Strict Deterministic Repair
    centers = best_c.copy()
    radii = final_r.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
