import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute LP constraint matrix structure (constant for fixed N)
def setup_lp():
    n_pairs = N * (N - 1) // 2
    n_bound = 4 * N
    A_ub = np.zeros((n_pairs + n_bound, N))
    pairs = []
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            pairs.append((i, j))
            idx += 1
    for i in range(N):
        for _ in range(4):
            A_ub[idx, i] = 1.0
            idx += 1
    return A_ub, pairs

A_LP, PAIRS = setup_lp()

def solve_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(A_LP.shape[0])
    idx = 0
    for i, j in PAIRS:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    bounds_r = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(N, 0.01), 0.0

def obj_func(v):
    """Objective for centers-only optimization: minimize negative LP sum of radii."""
    return -solve_lp_radii(v.reshape(N, 2))[1]

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5, 5, 5, 5, 6], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4],
        [4, 6, 6, 6, 4], [5, 5, 5, 6, 5], [6, 6, 5, 5, 4],
        [5, 4, 6, 6, 5], [6, 5, 5, 5, 5], [5, 5, 4, 6, 6],
        [4, 5, 6, 5, 6], [6, 4, 6, 5, 5], [5, 5, 6, 5, 5],
        [5, 7, 5, 5, 4], [4, 5, 5, 6, 6], [6, 6, 4, 5, 5],
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5]
    ]
    for pat in patterns:
        for r0 in [0.085, 0.090, 0.095, 0.100, 0.105]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.02, 0.98)
            starts.append(c)
            
    for _ in range(20):
        c = rng.uniform(0.1, 0.9, (N, 2))
        starts.append(c)
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    starts = generate_starts(rng)
    
    best_c = None
    best_sum = -np.inf
    bounds_c = [(0.0, 1.0)] * (2 * N)
    
    # Phase 1: Powell optimization from multiple starts
    for c_init in starts:
        try:
            res = minimize(obj_func, c_init.flatten(), method='Powell', bounds=bounds_c,
                           options={'maxiter': 8000, 'ftol': 1e-12, 'xtol': 1e-12})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass

    if best_c is None:
        best_c = starts[0]
        
    # Phase 2: Simulated Annealing on centers to escape local minima

    s_curr = best_sum

        c_new = c_curr + rng.normal(0, T, c_curr.shape)
        c_new = np.clip(c_new, 0.02, 0.98)
        r_new, s_new = solve_lp_radii(c_new)

            c_curr = c_new
            s_curr = s_new
            if s_curr > best_sum:
                best_sum = s_curr
                best_c = c_curr.copy()

                best_sum = best_local_s

    try:

        _, best_sum = solve_lp_radii(best_c)
    except Exception:
        pass

    best_r, _ = solve_lp_radii(best_c)
    
    # Phase 6: Strict Deterministic Repair for Validator Compliance
    for _ in range(100):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(best_c[i] - best_c[j])
                req = best_r[i] + best_r[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    best_r[i] -= shrink
                    best_r[j] -= shrink
                    changed = True
        # Resolve boundary violations
        for i in range(N):
            mr = min(best_c[i,0], 1.0-best_c[i,0], best_c[i,1], 1.0-best_c[i,1])
            if best_r[i] > mr + 1e-12:
                best_r[i] = mr
                changed = True
        if not changed:
            break
            
    best_r = np.maximum(best_r, 0.0)
    return best_c, best_r, float(np.sum(best_r))
