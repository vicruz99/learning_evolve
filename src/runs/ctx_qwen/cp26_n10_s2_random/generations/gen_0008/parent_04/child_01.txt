pair_idx = []
idx = 0

def solve_lp_radii(centers):

    return -solve_lp_radii(v.reshape(N, 2))[1]

    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]

    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    return np.concatenate([c, (dx**2 + dy**2 - dr**2)[MASK]])

def compute_obj(v):
