To maximize the sum of radii for 26 circles in a unit square, we need an optimization strategy that efficiently navigates the highly non-smooth objective landscape. Direct joint optimization of centers and radii often gets trapped in shallow local minima. 

The key insight is to decouple the problem: fix centers, and solve a Linear Program (LP) to find the optimal radii. The LP not only gives the maximum sum of radii but also provides dual variables (marginals) that exactly represent the gradient of the sum of radii with respect to the center positions. By performing gradient ascent on the centers using these exact dual-derived gradients, we can converge much faster and more precisely than derivative-free methods.

The strategy combines:
1. **LP-based Gradient Ascent**: Use LP duals to compute exact forces pushing circles apart and toward boundaries. Normalize and apply these forces to optimize centers.
2. **Diverse Initializations**: Hexagonal lattices, grids, and random starts to cover different basin attractors.
3. **Simulated Annealing**: To escape local optima found by gradient ascent.
4. **Perturbation Restarts**: Jitter the best configuration and re-optimize.
5. **SLSQP Joint Polish**: Fine-tune both centers and radii simultaneously near convergence.
6. **Strict Numerical Repair**: Ensure the final output strictly satisfies all constraints within machine epsilon tolerances.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
N_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure for efficiency
def build_lp_matrices():
    A_ub = np.zeros((N_PAIRS + 4 * N, N))
    pair_indices = []
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            pair_indices.append((i, j))
            idx += 1
    for i in range(N):
        for _ in range(4):
            A_ub[idx, i] = 1.0
            idx += 1
    return A_ub, pair_indices

A_LP, PAIR_INDICES = build_lp_matrices()

def solve_lp_and_grad(centers):
    """Solves LP for radii given fixed centers and computes exact gradient via duals."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(A_LP.shape[0])
    idx = 0
    for i, j in PAIR_INDICES:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(n):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    bounds_r = [(0.0, u) for u in ub]
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if not res.success:
        return None, None, None
        
    radii = res.x
    sum_r = -res.fun
    duals = res.ineqlin.marginals
    
    grad = np.zeros_like(centers)
    
    # Pairwise repulsion forces from distance constraints
    idx = 0
    for i, j in PAIR_INDICES:
        lam = duals[idx]
        if lam > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    # Boundary attraction forces
    b_start = N_PAIRS
    for i in range(n):
        mu_L = duals[b_start + 4*i]
        mu_R = duals[b_start + 4*i + 1]
        mu_B = duals[b_start + 4*i + 2]
        mu_T = duals[b_start + 4*i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, sum_r, grad

def gradient_ascent(centers0, max_iter=3000, init_step=0.005, rng=None):
    """Optimizes centers using exact gradients from LP duals."""
    centers = centers0.copy()
    best_centers = centers.copy()
    best_sum = -1.0
    step = init_step
    no_improve = 0
    
    for k in range(max_iter):
        _, curr_sum, grad = solve_lp_and_grad(centers)
        if grad is None:
            break
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        # Adaptive step size decay
        if no_improve > 50:
            step *= 0.7
        elif no_improve > 20:
            step *= 0.85
            
        if step < 1e-8:
            break
            
        g_norm = np.linalg.norm(grad)
        if g_norm > 0:
            centers += step * grad / g_norm
        centers = np.clip(centers, 0.001, 0.999)
        
        # Occasional jitter to escape flat regions
        if rng is not None and k % 400 == 0 and k > 0:
            centers += rng.normal(0, 0.002, centers.shape)
            centers = np.clip(centers, 0.001, 0.999)
            
    return best_centers, best_sum

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], [5,5,5,6,5],
                [6,6,5,5,4], [5,4,6,6,5], [6,5,5,5,5], [5,5,4,6,6], [4,5,6,5,6]]
    for pat in patterns:
        for r0 in [0.08, 0.09, 0.095, 0.10, 0.105, 0.11]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            starts.append(np.array(c[:N]))
            
    for sp in [5, 6]:
        x = np.linspace(0.1, 0.9, sp)
        y = x.copy()
        cx, cy = np.meshgrid(x, y)
        g = np.column_stack([cx.flatten(), cy.flatten()])
        if len(g) > N:
            g = g[:N]
        starts.append(g)
        
    for _ in range(15):
        starts.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = generate_starts(rng)
    
    # Phase 1: Gradient ascent from multiple starts
    for c0 in starts:
        c_opt, s_opt = gradient_ascent(c0, rng=rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            
    # Phase 2: Simulated Annealing on centers to escape local minima
    c_curr = best_c.copy()
    s_curr = best_sum
    T = 0.006
    for step in range(3000):
        T *= 0.9993
        c_new = c_curr + rng.normal(0, T, c_curr.shape)
        c_new = np.clip(c_new, 0.01, 0.99)
        r_new, s_new, _ = solve_lp_and_grad(c_new)
        
        if s_new is None:
            continue
            
        if s_new > s_curr:
            c_curr = c_new
            s_curr = s_new
            if s_curr > best_sum:
                best_sum = s_curr
                best_c = c_curr.copy()
        else:
            if rng.random() < np.exp((s_new - s_curr) / max(T * 4.0, 1e-7)):
                c_curr = c_new
                s_curr = s_new
                
    # Phase 3: Perturbation runs to explore nearby basins
    for _ in range(5):
        c_pert = best_c + rng.normal(0, 0.003, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        c_opt, s_opt = gradient_ascent(c_pert, max_iter=1500, rng=rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            
    # Phase 4: Final LP to extract optimal radii
    best_r, _, _ = solve_lp_and_grad(best_c)
    if best_r is None:
        ub = np.minimum(np.minimum(best_c[:, 0], 1.0 - best_c[:, 0]),
                        np.minimum(best_c[:, 1], 1.0 - best_c[:, 1]))
        dists = np.linalg.norm(best_c[:, np.newaxis, :] - best_c[np.newaxis, :, :], axis=2)
        np.fill_diagonal(dists, np.inf)
        best_r = np.minimum(ub, 0.5 * np.min(dists, axis=1))
        
    # Phase 5: Joint SLSQP Polish for fine-grained adjustment
    bounds_j = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    
    def obj_joint(v):
        return -np.sum(v[2*N:])
        
    def cons_joint(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        cons = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
        idx_i, idx_j = np.triu_indices(N, 1)
        dx = c[idx_i, 0] - c[idx_j, 0]
        dy = c[idx_i, 1] - c[idx_j, 1]
        dr = r[idx_i] + r[idx_j]
        cons.append(np.sqrt(dx**2 + dy**2) - dr)
        return np.concatenate(cons)
        
    v0 = np.concatenate([best_c.flatten(), best_r])
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                      constraints={'type': 'ineq', 'fun': cons_joint},
                      options={'maxiter': 10000, 'ftol': 1e-13})
        if np.min(cons_joint(res.x)) >= -1e-9:
            s = np.sum(res.x[2*N:])
            if s > best_sum:
                best_sum = s
                best_c = res.x[:2*N].reshape(N, 2).copy()
                best_r = res.x[2*N:].copy()
    except Exception:
        pass

    # Phase 6: Strict Numerical Repair to guarantee validator compliance
    centers = best_c.copy()
    radii = best_r.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
