import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute pairwise constraint matrix structure (constant for fixed N)
def get_pairwise_lp_matrix():
    n_pairs = N * (N - 1) // 2
    A_ub = np.zeros((n_pairs, N))
    pairs = []
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            pairs.append((i, j))
            idx += 1
    return A_ub, pairs

A_PAIRWISE, PAIR_INDICES = get_pairwise_lp_matrix()

def solve_lp_radii(centers):
    """
    Solves LP to find maximum sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j) and 0 <= r_i <= dist_to_boundary(i)
    """
    x, y = centers[:, 0], centers[:, 1]
    
    # Distance to boundaries for each circle
    ub = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    ub = np.maximum(ub, 1e-9)
    
    # Compute pairwise distances for b_ub
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[np.triu_indices(N, k=1)]
    
    bounds_r = [(0.0, u) for u in ub]
    
    # Maximize sum(r) => Minimize -sum(r)
    res = linprog(-np.ones(N), A_ub=A_PAIRWISE, b_ub=b_ub, 
                  bounds=bounds_r, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.full(N, 1e-7), 0.0

def lp_objective(centers_flat):
    """Objective for center-only optimization: minimize negative LP sum of radii."""
    centers = centers_flat.reshape(N, 2)
    _, s = solve_lp_radii(centers)
    return -s

def generate_hex_starts(rng):

    starts = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
        [5, 5, 5, 5, 6], [4, 6, 6, 6, 4], [6, 4, 6, 5, 5],
        [5, 7, 5, 5, 4], [6, 5, 5, 5, 5], [5, 5, 6, 5, 5]
    ]
    
    for pat in patterns:

            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            starts.append(np.array(c[:N]))
            
    return starts

        starts.append(g)

    centers = best_c.copy()

                    shrink = (req - d) / 2.0 + 1e-11
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
