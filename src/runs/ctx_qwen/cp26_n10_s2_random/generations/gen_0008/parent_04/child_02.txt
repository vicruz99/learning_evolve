import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_INDICES = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIR_INDICES)

# Precompute LP constraint matrix structure for speed
A_ub_struct = np.zeros((N_PAIRS + 4 * N, N))
p_idx = 0
for i, j in PAIR_INDICES:
    A_ub_struct[p_idx, i] = 1.0
    A_ub_struct[p_idx, j] = 1.0
    p_idx += 1
for i in range(N):
    for _ in range(4):
        A_ub_struct[p_idx, i] = 1.0
        p_idx += 1

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii given fixed centers and computes exact gradient via duals."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.empty(N_PAIRS + 4 * n)
    p_idx = 0
    for i, j in PAIR_INDICES:
        b_ub[p_idx] = dists[i, j]
        p_idx += 1
    for i in range(n):
        b_ub[p_idx] = centers[i, 0]; p_idx += 1
        b_ub[p_idx] = 1.0 - centers[i, 0]; p_idx += 1
        b_ub[p_idx] = centers[i, 1]; p_idx += 1
        b_ub[p_idx] = 1.0 - centers[i, 1]; p_idx += 1
        
    bounds_r = [(0.0, None)] * n
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_struct, b_ub=b_ub, 
                      bounds=bounds_r, method='highs', options={'presolve': True})
        if not res.success:
            return np.full(n, 1e-6), 0.0, np.zeros_like(centers)
            
        radii = res.x
        duals = res.ineqlin.marginals
        
        grad = np.zeros_like(centers)
        p_idx = 0
        for i, j in PAIR_INDICES:
            lam = -duals[p_idx]
            if lam > 1e-9:
                d = dists[i, j]
                if d > 1e-9:
                    vec = (centers[i] - centers[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
            p_idx += 1
            
        b_start = N_PAIRS
        for i in range(n):
            mu_L = -duals[b_start + 4*i]
            mu_R = -duals[b_start + 4*i + 1]
            mu_B = -duals[b_start + 4*i + 2]
            mu_T = -duals[b_start + 4*i + 3]
            grad[i, 0] += mu_L - mu_R
            grad[i, 1] += mu_B - mu_T
            
        return radii, np.sum(radii), grad
    except Exception:
        return np.full(n, 1e-6), 0.0, np.zeros_like(centers)

def obj_func(v):
    _, s, _ = solve_lp_and_grad(v.reshape(N, 2))
    return -s

def grad_func(v):
    _, _, g = solve_lp_and_grad(v.reshape(N, 2))
    return -g.flatten()

def generate_hex_init(row_counts, r0, rng):
    centers = []
    y = r0
    for row_idx, count in enumerate(row_counts):
        offset = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + offset
        for _ in range(count):
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        
    centers = np.array(centers[:N])
    centers += rng.normal(0, 0.002, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    return centers

def force_directed_init(rng):
    c = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(200):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(c[i] - c[j])
                if d < 0.15 and d > 1e-6:
                    f = (0.15 - d) / d
                    diff = c[i] - c[j]
                    forces[i] += diff * f
                    forces[j] -= diff * f
        c += forces * 0.002
        c = np.clip(c, 0.01, 0.99)
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_centers = None
    best_sum = -np.inf
    
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 5, 5, 6],
        [6, 5, 5, 5, 5], [4, 6, 6, 6, 4], [5, 7, 5, 5, 4],
        [6, 4, 6, 5, 5], [5, 5, 6, 5, 5], [6, 6, 5, 4, 5],
        [5, 5, 4, 6, 6], [4, 5, 6, 5, 6], [6, 6, 5, 5, 4]
    ]
    
    starts = []
    for pat in patterns:
        for r0 in [0.085, 0.09, 0.095, 0.10, 0.105]:
            starts.append(generate_hex_init(pat, r0, rng))
            
    for _ in range(10):
        starts.append(force_directed_init(rng))
        
    for _ in range(8):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))

    bounds_c = [(0.001, 0.999)] * (2 * N)
    
    # Phase 1: Gradient Ascent on Centers using LP duals
    for c0 in starts:
        try:
            res = minimize(obj_func, c0.flatten(), method='L-BFGS-B', jac=grad_func,
                           bounds=bounds_c, options={'maxiter': 1500, 'ftol': 1e-12, 'gtol': 1e-8})
            _, s, _ = solve_lp_and_grad(res.x.reshape(N, 2))
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape(N, 2).copy()
        except Exception:
            pass

    # Phase 2: Perturbation Search to escape local minima
    for _ in range(40):
        c_pert = best_centers.copy()
        noise = rng.normal(0, 0.006, c_pert.shape)
        c_pert += noise
        c_pert = np.clip(c_pert, 0.02, 0.98)
        try:
            res = minimize(obj_func, c_pert.flatten(), method='L-BFGS-B', jac=grad_func,
                           bounds=bounds_c, options={'maxiter': 1000, 'ftol': 1e-12})
            _, s, _ = solve_lp_and_grad(res.x.reshape(N, 2))
            if s > best_sum + 1e-7:
                best_sum = s
                best_centers = res.x.reshape(N, 2).copy()
        except Exception:
            pass

    # Phase 3: Joint SLSQP Polish on (x, y, r)
    r_lp, s_lp, _ = solve_lp_and_grad(best_centers)
    v0 = np.concatenate([best_centers.flatten(), r_lp])
    
    def obj_joint(v):
        return -np.sum(v[2*N:])
        
    def cons_joint(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
        idx_i, idx_j = np.triu_indices(N, 1)
        d = np.linalg.norm(c[idx_i] - c[idx_j], axis=1)
        con.append(d - (r[idx_i] + r[idx_j]))
        return np.concatenate(con)
        
    bounds_j = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    try:
        res_j = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                         constraints={'type': 'ineq', 'fun': cons_joint},
                         options={'maxiter': 15000, 'ftol': 1e-14})
        if -res_j.fun > best_sum - 1e-8:
            best_centers = res_j.x[:2*N].reshape(N, 2)
            best_radii = res_j.x[2*N:]
            best_sum = -res_j.fun
        else:
            best_radii = r_lp
    except Exception:
        best_radii = r_lp
        
    # Phase 4: Final Deterministic Repair for Validator Compliance
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-10:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-10:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
