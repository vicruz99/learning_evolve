1. **LP-based Radius Calculation**: For any fixed set of centers, the maximum sum of radii is found by solving an LP. Constraints ensure circles don't overlap ($r_i + r_j \leq d_{ij}$) and stay inside boundaries ($r_i \leq \text{dist}(c_i, \text{boundary})$). This gives an exact, piecewise-smooth objective function over center positions.
2. **Diverse Initialization**: We generate starting configurations using hexagonal lattices (various densities and shifts), corner-focused patterns (to leverage boundary space), random distributions, and force-directed layouts. This covers both regular and irregular packing topologies.
3. **Center Optimization (Powell)**: We optimize centers using Powell's method, which handles the non-smooth nature of the LP-derived objective robustly without requiring gradients. Multiple restarts from each initial configuration are performed.
4. **Basin Hopping / Perturbation**: To escape local minima, we iteratively perturb the best-found centers with decaying noise and re-optimize. This allows the configuration to shift into tighter, higher-density arrangements.
5. **Joint SLSQP Polish**: After finding optimal centers, we perform a final joint optimization of centers and radii using SLSQP. This handles constraint boundaries precisely and can tweak radii individually where the LP solution might be constrained by non-active pairs.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRI_IDX = np.triu_indices(N, k=1)
N_PAIRS = len(TRI_IDX[0])

# Precompute LP constraint matrix structure for pairwise distances
A_ub_pairs = np.zeros((N_PAIRS, N))
A_ub_pairs[np.arange(N_PAIRS), TRI_IDX[0]] = 1.0
A_ub_pairs[np.arange(N_PAIRS), TRI_IDX[1]] = 1.0

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[TRI_IDX]
    
    bounds = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_ub_pairs, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    # Fallback if LP fails
    r = np.minimum(ub, 0.5 * np.min(dists, axis=1))
    return r, np.sum(r)

def obj_func(v):
    """Objective for center optimization: minimize negative LP sum of radii."""
    return -solve_lp_radii(v.reshape(N, 2))[1]

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 5, 5, 6],
        [6, 6, 5, 5, 4], [4, 6, 6, 6, 4], [5, 5, 6, 5, 5],
        [6, 5, 5, 5, 5], [5, 4, 6, 6, 5], [5, 6, 4, 5, 6],
        [4, 5, 6, 5, 6], [6, 5, 6, 4, 5], [5, 5, 4, 6, 6],
        [5, 5, 5, 6, 5], [6, 6, 4, 5, 5], [5, 7, 5, 5, 4]
    ]
    
    # Hexagonal lattice starts with varying scales
    for pat in patterns:
        for r0 in [0.085, 0.090, 0.095, 0.100, 0.105, 0.110]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.02, 0.98)
            starts.append(c)
            
    # Random dense starts
    for _ in range(15):
        starts.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    # Corner-focused starts
    corners = [[0.06, 0.06], [0.94, 0.06], [0.06, 0.94], [0.94, 0.94]]
    for _ in range(10):
        c = rng.uniform(0.2, 0.8, (N-4, 2))
        c_full = np.vstack([corners, c])
        np.random.shuffle(c_full)
        starts.append(c_full)
        
    return starts

def obj_joint(v):
    """Joint optimization objective."""
    return -np.sum(v[2*N:])
    
def cons_joint(v):
    """Joint optimization constraints."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    d = np.linalg.norm(c[TRI_IDX[0]] - c[TRI_IDX[1]], axis=1)
    con.append(d - (r[TRI_IDX[0]] + r[TRI_IDX[1]]))
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_c = None
    best_sum = -1.0
    
    starts = generate_starts(rng)
    bounds_c = [(0.01, 0.99)] * (2 * N)
    
    # Phase 1: Powell optimization on centers from multiple starts
    for c0 in starts:
        try:
            res = minimize(obj_func, c0.flatten(), method='Powell', 
                           bounds=bounds_c, options={'maxiter': 3000, 'ftol': 1e-12})
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = starts[0]
        _, best_sum = solve_lp_radii(best_c)
        
    # Phase 2: Perturbation search to escape local minima
    for epoch in range(50):
        c_pert = best_c + rng.normal(0, 0.006 * (0.95**epoch), best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        try:
            res = minimize(obj_func, c_pert.flatten(), method='Powell',
                           bounds=bounds_c, options={'maxiter': 1500, 'ftol': 1e-11})
            s_new = -res.fun
            if s_new > best_sum:
                best_sum = s_new
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    # Phase 3: Joint SLSQP polish for precision
    r_best, _ = solve_lp_radii(best_c)
    v0 = np.concatenate([best_c.flatten(), r_best])
    bounds_j = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': 6000, 'ftol': 1e-13})
        if np.min(cons_joint(res.x)) >= -1e-8:
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_c = res.x[:2*N].reshape(N, 2)
                best_r = res.x[2*N:]
            else:
                best_r, _ = solve_lp_radii(best_c)
        else:
            best_r, _ = solve_lp_radii(best_c)
    except Exception:
        best_r, _ = solve_lp_radii(best_c)
        
    # Phase 4: Strict Deterministic Repair for Validator Compliance
    centers = best_c
    radii = best_r.copy()
    
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
