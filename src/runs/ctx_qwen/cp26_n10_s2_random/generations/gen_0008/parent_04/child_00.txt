

<strategy>
The optimization problem is to maximize $\sum r_i$ subject to boundary and pairwise non-overlap constraints. This is a convex optimization problem in terms of radii for fixed centers, but non-convex in centers. 
Strategy:
1. Use Linear Programming (LP) to compute optimal radii and their dual variables (Lagrange multipliers) for any fixed center configuration. The dual variables directly provide the gradient of the objective function with respect to the centers ($\nabla_C \sum r_i$).
2. Perform gradient-based optimization (L-BFGS-B) on the centers to maximize the sum of radii. This efficiently navigates the landscape by pushing circles apart and towards boundaries where they can grow.
3. Use diverse initial configurations (hexagonal lattices with varying densities/shifts, random placements) to avoid poor local minima.
4. Polish the best configuration using a joint SLSQP optimization on both centers and radii to handle constraints precisely.
5. Apply iterative perturbation and refinement to escape shallow local optima.
6. Implement a strict deterministic repair phase to ensure all constraints are satisfied within the $10^{-12}$ tolerance required by the validator.
</strategy>
```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute LP constraint matrix structure
def _get_lp_structure():
    num_pairs = N * (N - 1) // 2
    num_bound = 4 * N
    A_ub = np.zeros((num_pairs + num_bound, N))
    pair_indices = []
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            pair_indices.append((i, j))
            idx += 1
    for i in range(N):
        for _ in range(4):
            A_ub[idx, i] = 1.0
            idx += 1
    return A_ub, pair_indices, num_pairs

A_LP, PAIRS, NUM_PAIRS = _get_lp_structure()

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii and computes gradient w.r.t centers using duals."""
    x, y = centers[:, 0], centers[:, 1]
    n = centers.shape[0]
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Build RHS vector for LP
    b_ub = np.zeros(A_LP.shape[0])
    idx = 0
    for i, j in PAIRS:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(n):
        b_ub[idx] = x[i]; idx += 1
        b_ub[idx] = 1.0 - x[i]; idx += 1
        b_ub[idx] = y[i]; idx += 1
        b_ub[idx] = 1.0 - y[i]; idx += 1
        
    # Upper bounds for radii (from boundaries)
    ub_r = np.maximum(np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y)), 1e-9)
    bounds_r = [(0.0, u) for u in ub_r]
    
    # Solve LP: max sum(r) => min -sum(r)
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, bounds=bounds_r, method='highs')
    if not res.success:
        return np.full(n, 1e-9), 0.0, np.zeros_like(centers)
        
    radii = res.x
    duals = res.ineqlin.marginals
    sum_r = np.sum(radii)
    
    # Compute gradient using dual variables
    grad = np.zeros_like(centers)
    idx = 0
    for i, j in PAIRS:
        lam = max(duals[idx], 0.0)
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    b_start = NUM_PAIRS
    for i in range(n):
        # Boundary duals: r <= x, r <= 1-x, r <= y, r <= 1-y
        grad[i, 0] += max(duals[b_start + 4*i], 0.0) - max(duals[b_start + 4*i + 1], 0.0)
        grad[i, 1] += max(duals[b_start + 4*i + 2], 0.0) - max(duals[b_start + 4*i + 3], 0.0)
        
    return radii, sum_r, grad

def obj_func_centers(v):
    c = v.reshape(N, 2)
    _, s, _ = solve_lp_and_grad(c)
    return -s

def grad_func_centers(v):
    c = v.reshape(N, 2)
    _, _, g = solve_lp_and_grad(c)
    return -g.flatten()

def generate_initial_configs(rng):
    starts = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 4, 6, 6], [6, 5, 5, 5, 5]
    ]
    for pat in patterns:
        for r0 in [0.085, 0.09, 0.095, 0.10, 0.105, 0.11]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.01, 0.99)
            starts.append(c.flatten())
            
    # Random starts
    for _ in range(12):
        c = rng.uniform(0.05, 0.95, (N, 2))
        starts.append(c.flatten())
        
    return starts

def run_packing():
    rng = np.random.default_rng(42)
    bounds_c = [(0.001, 0.999)] * (2 * N)
    
    best_c = None
    best_sum = -np.inf
    
    candidates = generate_initial_configs(rng)
    
    # Phase 1: Gradient-based optimization on centers from multiple starts
    for v0 in candidates:
        try:
            res = minimize(obj_func_centers, v0, jac=grad_func_centers, method='L-BFGS-B', 
                           bounds=bounds_c, options={'maxiter': 3000, 'ftol': 1e-12})
            c_opt = res.x.reshape(N, 2)
            _, s_opt, _ = solve_lp_and_grad(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = candidates[0].reshape(N, 2)
        _, best_sum, _ = solve_lp_and_grad(best_c)

    # Phase 2: Joint SLSQP Polish
    def obj_joint(v):
        return -np.sum(v[2*N:])
        
    def cons_joint(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
        idx_i, idx_j = np.triu_indices(N, 1)
        d = np.linalg.norm(c[idx_i] - c[idx_j], axis=1)
        con.append(d - (r[idx_i] + r[idx_j]))
        return np.concatenate(con)
        
    _, r_lp, _ = solve_lp_and_grad(best_c)
    v0_joint = np.concatenate([best_c.flatten(), r_lp])
    bounds_j = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    
    try:
        res = minimize(obj_joint, v0_joint, method='SLSQP', bounds=bounds_j,
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': 6000, 'ftol': 1e-13})
        if -res.fun > best_sum - 1e-6:
            best_c = res.x[:2*N].reshape(N, 2)
            best_sum = -res.fun
            r_lp = res.x[2*N:]
    except Exception:
        pass

    # Phase 3: Perturbation & Refinement to escape local minima
    for trial in range(30):
        step = 0.003 * (0.95 ** trial)
        v_pert = best_c.flatten().copy() + rng.normal(0, step, 2*N)
        v_pert = np.clip(v_pert, 0.02, 0.98)
        try:
            res = minimize(obj_func_centers, v_pert, jac=grad_func_centers, method='L-BFGS-B',
                           bounds=bounds_c, options={'maxiter': 2000, 'ftol': 1e-12})
            c_p = res.x.reshape(N, 2)
            _, s_p, _ = solve_lp_and_grad(c_p)
            if s_p > best_sum:
                best_sum = s_p
                best_c = c_p.copy()
        except Exception:
            pass
            
    # Phase 4: Final LP Radii Computation
    radii, final_sum, _ = solve_lp_and_grad(best_c)
    
    # Phase 5: Strict Numerical Repair
    centers = best_c.copy()
    radii = radii.copy()
    for _ in range(100):
        changed = False
        # Boundary clamping
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        # Pairwise overlap resolution
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d)/2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
```
