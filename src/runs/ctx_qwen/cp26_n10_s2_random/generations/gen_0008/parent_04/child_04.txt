import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute LP constraint matrix structure globally
PAIRS = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIRS)
A_ub_pre = np.zeros((N_PAIRS + 4 * N, N))

# Pairwise constraints: r_i + r_j <= dist_ij
for k, (i, j) in enumerate(PAIRS):
    A_ub_pre[k, i] = 1.0
    A_ub_pre[k, j] = 1.0

# Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
for i in range(N):
    A_ub_pre[N_PAIRS + 4 * i, i] = 1.0
    A_ub_pre[N_PAIRS + 4 * i + 1, i] = 1.0
    A_ub_pre[N_PAIRS + 4 * i + 2, i] = 1.0
    A_ub_pre[N_PAIRS + 4 * i + 3, i] = 1.0


def solve_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    ub = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    ub = np.maximum(ub, 1e-10)

    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    b_ub = np.zeros(A_ub_pre.shape[0])
    b_ub[:N_PAIRS] = np.array([dists[i, j] for i, j in PAIRS])
    b_ub[N_PAIRS:N_PAIRS + N] = x
    b_ub[N_PAIRS + N:N_PAIRS + 2 * N] = 1.0 - x
    b_ub[N_PAIRS + 2 * N:N_PAIRS + 3 * N] = y
    b_ub[N_PAIRS + 3 * N:] = 1.0 - y

    bounds = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_ub_pre, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(N, 1e-6), 0.0


def lp_obj(x_flat):
    """Objective for centers-only optimization: minimize negative LP sum of radii."""
    centers = x_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    _, s = solve_lp_radii(centers)
    return -s


def force_directed_init(seed, iters=1500):
    """Generates a repulsion-based initial configuration."""
    rng = np.random.default_rng(seed)
    c = rng.uniform(0.1, 0.9, (N, 2))
    r = np.full(N, 0.04)

    for _ in range(iters):
        forces = np.zeros_like(c)
        for i in range(N):
            # Boundary repulsion
            for d in [0, 1]:
                if c[i, d] < r[i]:
                    forces[i, d] += (r[i] - c[i, d]) * 100.0
                elif c[i, d] > 1.0 - r[i]:
                    forces[i, d] -= (c[i, d] - (1.0 - r[i])) * 100.0

            # Inter-circle repulsion
            for j in range(i + 1, N):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < r[i] + r[j] and dist > 1e-9:
                    f = (r[i] + r[j] - dist) / dist
                    fx, fy = dx * f, dy * f
                    forces[i, 0] += fx * 10.0
                    forces[i, 1] += fy * 10.0
                    forces[j, 0] -= fx * 10.0
                    forces[j, 1] -= fy * 10.0

        c += forces * 0.002
        c = np.clip(c, 0.01, 0.99)
        r *= 1.0005

    return c


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize sum of radii."""
    rng = np.random.default_rng(123)
    best_c = None
    best_r = None
    best_sum = -1.0

    # --- Phase 1: Generate diverse initial configurations ---
    starts = []
    patterns = [
        [5, 5, 5, 5, 6], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [4, 6, 6, 6, 4],
        [5, 5, 5, 6, 5], [6, 6, 5, 5, 4], [5, 4, 6, 6, 5], [6, 5, 5, 5, 5],
        [5, 5, 4, 6, 6], [4, 5, 6, 5, 6], [6, 6, 6, 5, 3], [5, 5, 7, 5, 4]
    ]

    for pat in patterns:
        for r0 in [0.085, 0.09, 0.095, 0.10, 0.105, 0.11]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            starts.append(np.array(c[:N]))

    for i in range(8):
        starts.append(force_directed_init(i))

    for _ in range(10):
        starts.append(rng.uniform(0.05, 0.95, (N, 2)))

    bounds_c = [(0.01, 0.99)] * (2 * N)

    # --- Phase 2: Powell Optimization on Centers ---
    for c_init in starts:
        try:
            res = minimize(lp_obj, c_init.flatten(), method='Powell', bounds=bounds_c,
                           options={'maxiter': 10000, 'ftol': 1e-14, 'xtol': 1e-14})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt = solve_lp_radii(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            pass

    # --- Phase 3: Simulated Annealing on Centers ---
    if best_c is not None:
        c_curr = best_c.copy()
        s_curr = best_sum
        r_curr = best_r.copy()
        T = 0.007

        for step in range(2500):
            T *= 0.9994
            c_new = c_curr + rng.normal(0, T, c_curr.shape)
            c_new = np.clip(c_new, 0.02, 0.98)
            r_new, s_new = solve_lp_radii(c_new)

            delta = s_new - s_curr
            if delta > 0 or rng.random() < np.exp(delta / max(T * 5.0, 1e-7)):
                c_curr = c_new
                s_curr = s_new
                r_curr = r_new
                if s_curr > best_sum:
                    best_sum = s_curr
                    best_c = c_curr.copy()
                    best_r = r_curr.copy()

        # Random restarts from best to escape deep local minima
        for _ in range(10):
            c_trial = best_c + rng.normal(0, 0.005, best_c.shape)
            c_trial = np.clip(c_trial, 0.02, 0.98)
            try:
                res = minimize(lp_obj, c_trial.flatten(), method='Powell', bounds=bounds_c,
                               options={'maxiter': 4000, 'ftol': 1e-14})
                c_t = res.x.reshape(N, 2)
                r_t, s_t = solve_lp_radii(c_t)
                if s_t > best_sum:
                    best_sum = s_t
                    best_c = c_t.copy()
                    best_r = r_t.copy()
            except Exception:
                pass

    # --- Phase 4: Joint SLSQP Polish ---
    def obj_joint(v):
        return -np.sum(v[2 * N:])

    def cons_joint(v):
        c = v[:2 * N].reshape(N, 2)
        r = v[2 * N:]
        con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
        idx_i, idx_j = np.triu_indices(N, 1)
        d = np.linalg.norm(c[idx_i] - c[idx_j], axis=1)
        con.append(d - (r[idx_i] + r[idx_j]))
        return np.concatenate(con)

    bounds_j = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    v0 = np.concatenate([best_c.flatten(), best_r])

    for _ in range(4):
        v0 = v0 + rng.normal(0, 0.001, v0.shape)
        v0[:2 * N] = np.clip(v0[:2 * N], 0.01, 0.99)
        v0[2 * N:] = np.clip(v0[2 * N:], 0.01, 0.4)
        try:
            res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                           constraints={'type': 'ineq', 'fun': cons_joint},
                           options={'maxiter': 8000, 'ftol': 1e-13})
            if np.min(cons_joint(res.x)) >= -1e-8:
                s = np.sum(res.x[2 * N:])
                if s > best_sum:
                    best_sum = s
                    best_c = res.x[:2 * N].reshape(N, 2).copy()
                    best_r = res.x[2 * N:].copy()
                    v0 = res.x.copy()
        except Exception:
            pass

    # --- Phase 5: Strict Numerical Repair ---
    centers = best_c.copy()
    radii = best_r.copy()

    for _ in range(50):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Resolve boundary violations
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break

    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
