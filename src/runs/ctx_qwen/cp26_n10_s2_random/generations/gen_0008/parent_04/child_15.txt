import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute LP constraint structure (constant for fixed N)
PAIR_INDICES = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIR_INDICES)

A_ub_pairs = np.zeros((N_PAIRS, N))
for k, (i, j) in enumerate(PAIR_INDICES):
    A_ub_pairs[k, i] = 1.0
    A_ub_pairs[k, j] = 1.0

A_ub_bounds = np.zeros((4 * N, N))
for i in range(N):
    A_ub_bounds[4 * i, i] = 1.0
    A_ub_bounds[4 * i + 1, i] = 1.0
    A_ub_bounds[4 * i + 2, i] = 1.0
    A_ub_bounds[4 * i + 3, i] = 1.0

A_ub_total = np.vstack([A_ub_pairs, A_ub_bounds])


def solve_lp_and_grad(centers):
    """Solves LP for optimal radii given centers and computes exact gradient via duals."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub_pairs = dists[np.triu_indices(N, k=1)]
    b_ub_bounds = np.zeros(4 * N)
    idx = 0
    for i in range(N):
        b_ub_bounds[idx] = centers[i, 0]; idx += 1
        b_ub_bounds[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub_bounds[idx] = centers[i, 1]; idx += 1
        b_ub_bounds[idx] = 1.0 - centers[i, 1]; idx += 1
        
    b_ub = np.concatenate([b_ub_pairs, b_ub_bounds])
    c_obj = -np.ones(N)
    bounds_r = [(0, u) for u in ub]
    
    try:
        res = linprog(c_obj, A_ub=A_ub_total, b_ub=b_ub, bounds=bounds_r, method='highs')
        if not res.success:
            return None, None, None
            
        radii = res.x
        duals = res.ineqlin.marginals
        sum_r = np.sum(radii)
        
        grad = np.zeros_like(centers)
        
        # Pairwise gradients from distance constraints
        idx = 0
        for i, j in PAIR_INDICES:
            lam = duals[idx]
            if lam > 1e-8:
                d = dists[i, j]
                if d > 1e-9:
                    vec = (centers[i] - centers[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
            idx += 1
            
        # Boundary gradients from wall constraints
        b_start = N_PAIRS
        for i in range(N):
            grad[i, 0] += duals[b_start + 4 * i] - duals[b_start + 4 * i + 1]
            grad[i, 1] += duals[b_start + 4 * i + 2] - duals[b_start + 4 * i + 3]
            
        return radii, sum_r, grad
    except Exception:
        return None, None, None


def make_hex_start(pat, scale, rot, rng):
    """Generates a perturbed, normalized hexagonal lattice configuration."""
    pts = []
    y = scale
    for r_idx, cnt in enumerate(pat):
        shift = scale if r_idx % 2 == 1 else 0.0
        x = scale + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * scale
        y += np.sqrt(3) * scale
        
    pts = np.array(pts[:N])
    if rot != 0.0:
        c, s = np.cos(rot), np.sin(rot)
        pts = pts @ np.array([[c, -s], [s, c]])
        
    # Normalize to [0.1, 0.9] region
    pts -= pts.min(axis=0)
    span = pts.max(axis=0) - pts.min(axis=0)
    span[span < 1e-9] = 1.0
    pts = pts / span * 0.8 + 0.1
    pts += rng.normal(0, 0.002, pts.shape)
    return np.clip(pts, 0.02, 0.98)


def gradient_ascent(centers0, rng):
    """Performs gradient ascent on centers using LP-derived gradients."""
    centers = centers0.copy()
    best_c = centers.copy()
    _, best_s, _ = solve_lp_and_grad(best_c)
    if best_s is None:
        best_s = -1.0
        
    step = 0.006
    no_imp = 0
    
    for step_iter in range(2500):
        radii, s, grad = solve_lp_and_grad(centers)
        if radii is None:
            break
            
        if s > best_s + 1e-7:
            best_s = s
            best_c = centers.copy()
            no_imp = 0
        else:
            no_imp += 1
            
        if no_imp > 60:
            step *= 0.65
        elif no_imp > 25:
            step *= 0.85
        if step < 1e-9:
            break
            
        centers += step * grad
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
        
        if step_iter % 400 == 0 and step_iter > 0:
            centers += rng.normal(0, 0.003, centers.shape)
            centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
            
    return best_c, best_s


def sim_annealing(centers0, rng):
    """Simulated annealing on centers using LP evaluation."""
    c_curr = centers0.copy()
    radii_curr, s_curr, _ = solve_lp_and_grad(c_curr)
    if s_curr is None:
        s_curr = -1.0
        
    best_c = c_curr.copy()
    best_s = s_curr
    T = 0.004
    
    for step in range(3000):
        T *= 0.9992
        c_new = c_curr + rng.normal(0, max(T, 1e-5), c_curr.shape)
        c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
        
        radii_new, s_new, _ = solve_lp_and_grad(c_new)
        if s_new is None:
            continue
            
        delta = s_new - s_curr
        if delta > 0 or rng.random() < np.exp(delta / max(T * 10.0, 1e-7)):
            c_curr = c_new
            s_curr = s_new
            if s_new > best_s:
                best_s = s_new
                best_c = c_new.copy()
                
    return best_c, best_s


def obj_joint(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])


def cons_joint(v):
    """Boundary and non-overlap constraints for SLSQP (must be >= 0)."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    idx_i, idx_j = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx_i] - c[idx_j], axis=1)
    con.append(d - (r[idx_i] + r[idx_j]))
    return np.concatenate(con)


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize sum of radii."""
    rng = np.random.default_rng(42)
    
    # Diverse initial configurations
    patterns = [
        [5, 5, 5, 5, 6], [6, 5, 6, 5, 4], [5, 6, 5, 6, 4],
        [4, 6, 6, 6, 4], [5, 5, 5, 6, 5], [6, 6, 5, 5, 4],
        [5, 4, 6, 6, 5], [6, 5, 5, 5, 5], [5, 5, 4, 6, 6],
        [4, 5, 6, 5, 6], [5, 6, 4, 5, 6], [6, 4, 6, 5, 5]
    ]
    
    starts = []
    for pat in patterns:
        for scale in [0.09, 0.095, 0.10, 0.105]:
            for rot in [0.0, 0.05, 0.1, 0.15, 0.2]:
                starts.append(make_hex_start(pat, scale, rot, rng))
                
    for _ in range(12):
        c = rng.uniform(0.15, 0.85, (N, 2))
        starts.append(c)
        
    best_c = None
    best_s = -1.0
    
    # Phase 1: Gradient ascent from multiple starts
    for c0 in starts:
        c_opt, s_opt = gradient_ascent(c0, rng)
        if s_opt > best_s:
            best_s = s_opt
            best_c = c_opt.copy()
            
    # Phase 2: Simulated Annealing to escape local minima
    if best_c is not None:
        c_sa, s_sa = sim_annealing(best_c, rng)
        if s_sa > best_s:
            best_s = s_sa
            best_c = c_sa.copy()
            
    # Fallback if optimization failed entirely
    if best_c is None:
        best_c = starts[0]
        _, best_s, _ = solve_lp_and_grad(best_c)
        if best_s is None:
            best_s = 0.0
            
    # Phase 3: Joint SLSQP Polish
    radii_lp, _, _ = solve_lp_and_grad(best_c)
    if radii_lp is None:
        radii_lp = np.full(N, 0.05)
        
    v0 = np.concatenate([best_c.flatten(), radii_lp])
    bounds_joint = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': 8000, 'ftol': 1e-13})
        if np.min(cons_joint(res.x)) >= -1e-6:
            curr_s = -res.fun
            if curr_s > best_s:
                best_s = curr_s
                best_c = res.x[:2 * N].reshape(N, 2).copy()
                radii_lp = res.x[2 * N:].copy()
    except Exception:
        pass
        
    # Phase 4: Exact LP refinement for final radii
    radii_final, best_s, _ = solve_lp_and_grad(best_c)
    if radii_final is not None:
        radii_lp = radii_final
        
    # Phase 5: Strict Deterministic Repair for Validator Compliance
    centers = best_c.copy()
    radii = radii_lp.copy()
    
    for _ in range(100):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-10:
                    shrink = (req - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Resolve boundary violations
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0],
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-10:
                radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
