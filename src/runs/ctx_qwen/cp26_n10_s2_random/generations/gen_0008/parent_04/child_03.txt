To maximize the sum of radii for 26 circles in a unit square, we use a hybrid optimization strategy:
1. **Linear Programming for Radii**: Given fixed centers, the maximum feasible radii and their sum can be found exactly and efficiently using LP (`scipy.optimize.linprog`). This decouples the non-smooth radius constraints from center optimization.
2. **Derivative-Free Center Optimization**: The LP objective function (sum of radii as a function of centers) is continuous but piecewise-smooth. We optimize centers using `Powell`'s method, which handles bounds and non-smoothness robustly.
3. **Diverse Initialization**: We generate starting configurations from perturbed hexagonal lattices (varying row counts, scales, and shifts), random dense packs, and force-directed layouts to cover the configuration space widely.
4. **Multi-Start & Perturbation**: We run Powell from ~60 diverse starts, then perform a targeted perturbation search around the best result to escape shallow local minima.
5. **Joint SLSQP Polish**: Finally, we refine both centers and radii simultaneously using `SLSQP` on the original non-linear constraints to squeeze out the last bit of precision.
6. **Strict Repair**: A deterministic repair pass resolves any microscopic constraint violations within the validator's tolerance (`1e-12`).

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
num_pairs = N * (N - 1) // 2
num_bound = 4 * N
A_ub = np.zeros((num_pairs + num_bound, N))
pair_idx = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        pair_idx.append((i, j))
        idx += 1
for i in range(N):
    A_ub[idx, i] = 1.0; idx += 1  # r <= x
    A_ub[idx, i] = 1.0; idx += 1  # r <= 1-x
    A_ub[idx, i] = 1.0; idx += 1  # r <= y
    A_ub[idx, i] = 1.0; idx += 1  # r <= 1-y

TRIU_INDICES = np.triu_indices(N, k=1)

def solve_lp(centers):
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-10)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    
    b_ub = np.zeros(num_pairs + num_bound)
    b_ub[:num_pairs] = dists[TRIU_INDICES]
    
    for i in range(N):
        b_ub[num_pairs + 4 * i] = centers[i, 0]
        b_ub[num_pairs + 4 * i + 1] = 1.0 - centers[i, 0]
        b_ub[num_pairs + 4 * i + 2] = centers[i, 1]
        b_ub[num_pairs + 4 * i + 3] = 1.0 - centers[i, 1]
        
    bounds = [(0.0, 1.0)] * N
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    # Fallback to analytical caps if LP fails
    r_fall = np.minimum(ub, 0.5 * np.min(np.where(np.eye(N) == 0, dists, np.inf), axis=1))
    return r_fall, np.sum(r_fall)

def obj_centers(v):
    c = np.clip(v.reshape(N, 2), 1e-9, 1.0 - 1e-9)
    _, s = solve_lp(c)
    return -s

def obj_joint(v):
    return -np.sum(v[2 * N:])

def cons_joint(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    idx_i, idx_j = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx_i] - c[idx_j], axis=1)
    con.append(d - (r[idx_i] + r[idx_j]))
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_c = None
    best_s = -1.0
    bounds_c = [(0.0, 1.0)] * (2 * N)
    
    starts = []
    
    # Hexagonal lattice patterns with various row structures
    patterns = [
        [6,5,6,5,4], [5,6,5,6,4], [5,5,5,5,6], [6,5,5,5,5], [4,6,6,6,4],
        [6,6,5,5,4], [5,7,4,5,5], [4,5,6,5,6], [5,5,6,5,5], [6,4,6,5,5],
        [5,5,5,6,5], [6,5,4,6,5], [5,6,4,5,6], [4,6,5,6,5], [6,6,4,5,5],
        [5,4,6,6,5], [6,5,5,6,4], [5,6,6,4,5], [4,5,5,6,6], [5,5,4,6,6]
    ]
    
    for pat in patterns:
        for r0 in [0.085, 0.090, 0.095, 0.100, 0.105, 0.110]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.0025, c.shape)
            c = np.clip(c, 0.02, 0.98)
            starts.append(c.flatten())
            
    # Random dense starts
    for _ in range(25):
        c = rng.uniform(0.1, 0.9, (N, 2))
        starts.append(c.flatten())
        
    # Force-directed relaxation starts
    for _ in range(10):
        c = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(120):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.linalg.norm(c[i] - c[j])
                    if d < 0.14 and d > 1e-6:
                        f = (0.14 - d) / d
                        diff = c[i] - c[j]
                        forces[i] += diff * f
                        forces[j] -= diff * f
            c += forces * 0.0012
            c = np.clip(c, 0.02, 0.98)
        starts.append(c.flatten())
        
    # Phase 1: Powell optimization from diverse starts
    for v0 in starts:
        try:
            res = minimize(obj_centers, v0, method='Powell', bounds=bounds_c,
                           options={'maxiter': 2500, 'ftol': 1e-14, 'xtol': 1e-14})
            if -res.fun > best_s:
                best_s = -res.fun
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation search to escape local minima
    for _ in range(40):
        v_pert = best_c.flatten().copy()
        v_pert += rng.normal(0, 0.0015, v_pert.shape)
        v_pert = np.clip(v_pert, 0.02, 0.98)
        try:
            res = minimize(obj_centers, v_pert, method='Powell', bounds=bounds_c,
                           options={'maxiter': 1500, 'ftol': 1e-14})
            if -res.fun > best_s:
                best_s = -res.fun
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    # Phase 3: Joint SLSQP polish on centers and radii
    r_lp, _ = solve_lp(best_c)
    v0_joint = np.concatenate([best_c.flatten(), r_lp])
    bounds_j = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    try:
        res = minimize(obj_joint, v0_joint, method='SLSQP', bounds=bounds_j,
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': 10000, 'ftol': 1e-14})
        if -res.fun > best_s - 1e-7:
            best_c = res.x[:2 * N].reshape(N, 2)
            r_lp = res.x[2 * N:]
            best_s = -res.fun
    except Exception:
        pass
        
    # Phase 4: Strict Deterministic Repair for Validator Compliance
    centers = best_c.copy()
    radii = r_lp.copy()
    
    for _ in range(60):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-12
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Resolve boundary violations
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-12:
                radii[i] = mr - 1e-12
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
