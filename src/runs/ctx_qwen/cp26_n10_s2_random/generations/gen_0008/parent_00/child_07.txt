import numpy as np
from scipy.optimize import linprog

N = 26

# Precompute LP constraint matrix structure for speed
PAIR_INDICES = []
for i in range(N):
    for j in range(i + 1, N):
        PAIR_INDICES.append((i, j))
N_PAIRS = len(PAIR_INDICES)

# Constraints: r_i + r_j <= dist(i,j)  and  r_i <= boundaries
A_ub = np.zeros((N_PAIRS + 4 * N, N))
idx = 0
for i, j in PAIR_INDICES:
    A_ub[idx, i] = 1.0
    A_ub[idx, j] = 1.0
    idx += 1
for i in range(N):
    A_ub[idx, i] = 1.0; idx += 1  # r <= x
    A_ub[idx, i] = 1.0; idx += 1  # r <= 1-x
    A_ub[idx, i] = 1.0; idx += 1  # r <= y
    A_ub[idx, i] = 1.0; idx += 1  # r <= 1-y

def solve_lp_and_gradient(centers):

    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(N_PAIRS + 4 * N)
    idx = 0
    for i, j in PAIR_INDICES:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(N)
    bounds_r = [(0.0, None)] * N
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if not res.success:
            return np.zeros(N), 0.0, np.zeros_like(centers)
    except Exception:
        return np.zeros(N), 0.0, np.zeros_like(centers)
        
    radii = res.x
    total_sum = -res.fun
    
    # Extract dual marginals (handles scipy version differences)
    marginals = None
    if hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
        marginals = res.marginals.ineqlin
    elif hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
        marginals = res.ineqlin.marginals
    else:
        marginals = np.zeros(N_PAIRS + 4 * N)

    idx = 0
    
    # Pairwise gradient contributions

                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1

    for i in range(N):
        mu_L = marginals[idx]; idx += 1  # r <= x
        mu_R = marginals[idx]; idx += 1  # r <= 1-x
        mu_B = marginals[idx]; idx += 1  # r <= y
        mu_T = marginals[idx]; idx += 1  # r <= 1-y
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, total_sum, grad

def run_gradient_ascent(centers0, steps=1500, step_size=0.005):
    """Ascend gradient of sum of radii w.r.t center positions."""
    centers = centers0.copy()
    best_centers = centers.copy()
    best_sum = -1.0
    
    step = step_size

            best_centers = centers.copy()
