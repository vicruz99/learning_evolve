import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute indices for LP constraints to speed up repeated solves
PAIR_INDICES = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        PAIR_INDICES.append((i, j))
N_PAIRS = len(PAIR_INDICES)

# Structure of the inequality constraint matrix A_ub for the LP
# Rows 0..N_PAIRS-1: r_i + r_j <= dist(i,j)
# Rows N_PAIRS..end: boundary constraints r_i <= dist(i, wall)
A_UB_STRUCT = np.zeros((N_PAIRS + 4 * N, N))
k = 0
for i, j in PAIR_INDICES:
    A_UB_STRUCT[k, i] = 1.0
    A_UB_STRUCT[k, j] = 1.0
    k += 1
for i in range(N):
    A_UB_STRUCT[k, i] = 1.0; k += 1
    A_UB_STRUCT[k+1, i] = 1.0; k += 1
    A_UB_STRUCT[k+2, i] = 1.0; k += 1
    A_UB_STRUCT[k+3, i] = 1.0; k += 1

def solve_lp_and_grad(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns optimal radii, sum of radii, and gradient w.r.t centers using LP duals.
    """
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(N_PAIRS + 4 * N)
    k = 0
    for i, j in PAIR_INDICES:
        b_ub[k] = dists[i, j]
        k += 1
    for i in range(N):
        b_ub[k] = centers[i, 0]; k += 1
        b_ub[k] = 1.0 - centers[i, 0]; k += 1
        b_ub[k] = centers[i, 1]; k += 1
        b_ub[k] = 1.0 - centers[i, 1]; k += 1
        
    bounds_r = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_UB_STRUCT, b_ub=b_ub, 
                  bounds=bounds_r, method='highs')
    
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(centers)
        
    radii = res.x
    duals = res.ineqlin.marginals
    grad = np.zeros_like(centers)
    
    # Pairwise repulsion forces from active distance constraints
    k = 0
    for i, j in PAIR_INDICES:
        lam = duals[k]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        k += 1
        
    # Boundary attraction/repulsion forces from active wall constraints
    bound_start = N_PAIRS
    for i in range(N):
        mu_x0 = duals[bound_start + 4*i]
        mu_x1 = duals[bound_start + 4*i + 1]
        mu_y0 = duals[bound_start + 4*i + 2]
        mu_y1 = duals[bound_start + 4*i + 3]
        grad[i, 0] += mu_x0 - mu_x1
        grad[i, 1] += mu_y0 - mu_y1
        
    return radii, np.sum(radii), grad

def optimize_gradient(centers0, steps=2500, init_step=0.015):
    """Runs gradient ascent on centers to maximize sum of radii."""
    centers = centers0.copy()
    best_centers = centers.copy()
    best_sum = -1.0
    step = init_step
    no_improve = 0
    
    for k in range(steps):
        radii, curr_sum, grad = solve_lp_and_grad(centers)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        # Adaptive step decay
        if no_improve > 120:
            step *= 0.5
        elif no_improve > 50:
            step *= 0.8
            
        if step < 1e-11:
            break
            
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-12:
            centers += step * (grad / g_norm)
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
        
        # Periodic jitter to escape plateaus
        if k % 300 == 0 and k > 0:
            centers += np.random.normal(0, step * 0.4, centers.shape)
            centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
            
    return best_centers, best_sum

def generate_starts():
    """Generates diverse initial configurations."""
    starts = []
    rng = np.random.default_rng(42)
    
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [4,6,6,6,4], 
                [6,6,5,5,4], [5,4,6,6,5], [6,5,5,5,5], [5,5,4,6,6]]
    for pat in patterns:
        for r0 in [0.09, 0.095, 0.10, 0.105, 0.11]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            starts.append(np.array(c[:N]))
            
    for _ in range(12):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return starts

def obj_joint(v):
    return -np.sum(v[2*N:])

def cons_joint(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    idx_i, idx_j = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx_i] - c[idx_j], axis=1)
    con.append(d - (r[idx_i] + r[idx_j]))
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(123)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = generate_starts()
    
    # Phase 1: Gradient Ascent from multiple starts
    for c0 in starts:
        c_init = c0 + rng.normal(0, 0.002, c0.shape)
        c_init = np.clip(c_init, 0.05, 0.95)
        c_opt, s_opt = optimize_gradient(c_init, steps=2000, init_step=0.012)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            
    # Phase 2: Deep refinement of best found so far
    if best_c is not None:
        c_ref, s_ref = optimize_gradient(best_c, steps=3000, init_step=0.004)
        if s_ref > best_sum:
            best_sum = s_ref
            best_c = c_ref.copy()
            
        # Perturbation restarts around the best configuration
        for _ in range(6):
            c_pert = best_c + rng.normal(0, 0.0015, best_c.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            c_ref2, s_ref2 = optimize_gradient(c_pert, steps=1500, init_step=0.003)
            if s_ref2 > best_sum:
                best_sum = s_ref2
                best_c = c_ref2.copy()
                
    # Phase 3: SLSQP Joint Polish (centers + radii)
    r_lp, _, _ = solve_lp_and_grad(best_c)
    v0 = np.concatenate([best_c.flatten(), r_lp])
    bounds_j = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': 5000, 'ftol': 1e-14})
        if np.min(cons_joint(res.x)) >= -1e-10:
            s = np.sum(res.x[2*N:])
            if s > best_sum:
                best_sum = s
                best_c = res.x[:2*N].reshape(N, 2).copy()
                best_r = res.x[2*N:].copy()
    except Exception:
        pass
        
    if best_r is None:
        best_r, _, _ = solve_lp_and_grad(best_c)
        
    # Phase 4: Strict Numerical Repair for Validator Tolerance
    centers = best_c.copy()
    radii = best_r.copy()
    for _ in range(150):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Resolve boundary violations
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
