import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute LP constraint matrix structure globally for speed
PAIR_INDICES = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIR_INDICES)
A_ub_full = np.zeros((N_PAIRS + 4 * N, N))

# Pairwise constraints: r_i + r_j <= dist(i, j)
for k, (i, j) in enumerate(PAIR_INDICES):
    A_ub_full[k, i] = 1.0
    A_ub_full[k, j] = 1.0

# Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
for i in range(N):
    base = N_PAIRS + 4 * i
    A_ub_full[base, i] = 1.0
    A_ub_full[base + 1, i] = 1.0
    A_ub_full[base + 2, i] = 1.0
    A_ub_full[base + 3, i] = 1.0


def solve_lp_and_grad(centers):
    """Solves LP for optimal radii and computes exact gradient w.r.t centers."""
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    
    b_ub = np.zeros(N_PAIRS + 4 * N)
    idx = 0
    for i, j in PAIR_INDICES:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b_ub[idx] = c[i, 0]; idx += 1
        b_ub[idx] = 1.0 - c[i, 0]; idx += 1
        b_ub[idx] = c[i, 1]; idx += 1
        b_ub[idx] = 1.0 - c[i, 1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub_full, b_ub=b_ub, 
                  bounds=[(0.0, None)] * N, method='highs')
                  
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(c)
        
    radii = res.x
    duals = res.ineqlin.marginals
    grad = np.zeros_like(c)
    
    # Pairwise gradient contributions
    idx = 0
    for i, j in PAIR_INDICES:
        lam = duals[idx]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    # Boundary gradient contributions
    bound_start = N_PAIRS
    for i in range(N):
        mu_L = duals[bound_start + 4 * i]
        mu_R = duals[bound_start + 4 * i + 1]
        mu_B = duals[bound_start + 4 * i + 2]
        mu_T = duals[bound_start + 4 * i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, np.sum(radii), grad

def grad_ascent(centers0, steps=2000, init_step=0.015):
    """Runs gradient ascent on centers to maximize sum of radii."""

    best_c = centers.copy()

            no_improve += 1

            break

            step *= 0.7

def lp_obj(x):
    """Objective for derivative-free optimization: minimize negative LP sum."""
    _, s = solve_lp_and_grad(x.reshape(N, 2))
    return -s


def obj_joint(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])


def cons_joint(v):
    """Computes boundary and non-overlap constraints for SLSQP."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    cons = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    cons.append(np.sqrt(dx ** 2 + dy ** 2) - dr)
    return np.concatenate(cons)


def repair_packing(centers, radii):
    """Strict numerical repair to guarantee validator tolerance."""
    radii = radii.copy()
    for _ in range(150):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
