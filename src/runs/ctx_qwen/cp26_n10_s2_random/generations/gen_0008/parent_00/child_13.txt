import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute constant constraint matrix structure for LP
N_PAIRS = N * (N - 1) // 2
A_ub_const = np.zeros((N_PAIRS + 4 * N, N))

idx = 0
PAIR_LIST = []
for i in range(N):
    for j in range(i + 1, N):
        A_ub_const[idx, i] = 1.0
        A_ub_const[idx, j] = 1.0
        PAIR_LIST.append((i, j))
        idx += 1

for i in range(N):
    for _ in range(4):
        A_ub_const[idx, i] = 1.0
        idx += 1

def solve_lp_and_gradient(centers):
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.empty(N_PAIRS + 4 * N)
    for k, (i, j) in enumerate(PAIR_LIST):
        b_ub[k] = dists[i, j]
        
    for i in range(N):
        x, y = centers[i]
        b_ub[N_PAIRS + 4*i] = x
        b_ub[N_PAIRS + 4*i + 1] = 1.0 - x
        b_ub[N_PAIRS + 4*i + 2] = y
        b_ub[N_PAIRS + 4*i + 3] = 1.0 - y
        
    bounds_r = [(0.0, None)] * N
    res = linprog(-np.ones(N), A_ub=A_ub_const, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(centers)
        
    radii = res.x
    sum_r = np.sum(radii)
    duals = res.ineqlin.marginals
    
    grad = np.zeros_like(centers)
    
    for k, (i, j) in enumerate(PAIR_LIST):
        mu = duals[k]
        if mu > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
                
    bound_start = N_PAIRS
    for i in range(N):
        grad[i, 0] += duals[bound_start + 4*i] - duals[bound_start + 4*i + 1]
        grad[i, 1] += duals[bound_start + 4*i + 2] - duals[bound_start + 4*i + 3]
        
    return radii, sum_r, grad

            centers = c_new
            curr_s = s_new
            grad = g_new

            radii, curr_s, grad = solve_lp_and_gradient(centers)
            
    return best_c, best_s

def lp_obj_func(x):
    centers = x.reshape(N, 2)
    _, s, _ = solve_lp_and_gradient(centers)
    return -s

def obj_joint(v):
    return -np.sum(v[2*N:])

def cons_joint(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    cons = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    cons.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(cons)
