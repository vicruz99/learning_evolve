To push the sum of radii past 2.636, we need a highly robust optimization pipeline that combines:
1. **LP-based Gradient Ascent**: The sum of radii for fixed centers is a linear program. Its gradient with respect to center positions can be computed exactly using LP dual variables (marginals). This gives a principled direction to move centers to increase radii.
2. **Diverse Initialization**: Hexagonal lattice patterns with various scales, offsets, and random perturbations cover the configuration space well. Circle packing landscapes have many local optima; diverse starts are crucial.
3. **Powell Optimization on LP Objective**: Since the LP objective is non-smooth, derivative-free methods like Powell's work exceptionally well for fine-tuning centers.
4. **Simulated Annealing**: Helps escape shallow local minima during the fine-tuning phase by occasionally accepting worse configurations.
5. **Joint SLSQP Polish**: Optimizes centers and radii simultaneously with explicit constraints to squeeze out the last bits of precision.
6. **Strict Numerical Repair**: Iteratively resolves microscopic overlaps and boundary violations to guarantee the validator's `1e-12` tolerance is met without unnecessarily shrinking radii.

The pipeline runs gradient ascent from multiple starts, keeps the best, polishes with Powell, applies SA, finishes with SLSQP, and repairs numerically. This combination has consistently broken through local optima barriers in similar packing problems.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute LP constraint structure (constant for fixed N)
PAIR_IDX = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIR_IDX)

A_ub = np.zeros((N_PAIRS + 4 * N, N))
for k, (i, j) in enumerate(PAIR_IDX):
    A_ub[k, i] = 1.0
    A_ub[k, j] = 1.0
for i in range(N):
    base = N_PAIRS + 4 * i
    A_ub[base, i] = 1.0      # r <= x
    A_ub[base + 1, i] = 1.0  # r <= 1 - x
    A_ub[base + 2, i] = 1.0  # r <= y
    A_ub[base + 3, i] = 1.0  # r <= 1 - y

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii given fixed centers and returns gradient of sum of radii."""
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    
    b_ub = np.zeros(N_PAIRS + 4 * N)
    for k, (i, j) in enumerate(PAIR_IDX):
        b_ub[k] = dists[i, j]
    for i in range(N):
        base = N_PAIRS + 4 * i
        b_ub[base] = c[i, 0]
        b_ub[base + 1] = 1.0 - c[i, 0]
        b_ub[base + 2] = c[i, 1]
        b_ub[base + 3] = 1.0 - c[i, 1]
        
    bounds = [(0.0, u) for u in ub]
    c_obj = -np.ones(N)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(c)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    grad = np.zeros_like(c)
    marginals = res.ineqlin.marginals
    
    # Pairwise distance constraints gradient
    for k, (i, j) in enumerate(PAIR_IDX):
        lam = marginals[k]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
                
    # Boundary constraints gradient
    for i in range(N):
        base = N_PAIRS + 4 * i
        grad[i, 0] += marginals[base] - marginals[base + 1]
        grad[i, 1] += marginals[base + 2] - marginals[base + 3]
        
    return radii, s_sum, grad

def gradient_ascent(c0, rng, steps=1500, init_step=0.005):
    """Runs gradient ascent on centers to maximize LP sum of radii."""
    c = c0.copy()
    best_c = c.copy()
    best_s = -1.0
    step = init_step
    no_imp = 0
    
    for k in range(steps):
        r, s, g = solve_lp_and_grad(c)
        if s > best_s:
            best_s = s
            best_c = c.copy()
            no_imp = 0
        else:
            no_imp += 1
            
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-9:
            break
            
        if no_imp > 60:
            step *= 0.65
        elif no_imp > 25:
            step *= 0.85
            
        if step < 1e-10:
            break
            
        c += step * (g / g_norm)
        c = np.clip(c, 0.02, 0.98)
        
        # Occasional jitter to escape plateaus
        if k % 200 == 100:
            c += rng.normal(0, 0.001 * step, c.shape)
            c = np.clip(c, 0.02, 0.98)
            
    return best_c, best_s

def lp_obj(x):
    """Objective for center optimization: minimize negative LP sum of radii."""
    _, s, _ = solve_lp_and_grad(x.reshape(N, 2))
    return -s

def generate_starts():
    """Generates diverse initial configurations."""
    starts = []
    rng = np.random.default_rng(42)
    
    # Hexagonal lattice patterns with varying row structures
    patterns = [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], [5,5,5,6,5], 
                [6,6,5,5,4], [5,4,6,6,5], [6,5,5,5,5], [5,5,4,6,6], [4,5,6,5,6]]
    for pat in patterns:
        for r0 in [0.085, 0.090, 0.095, 0.100, 0.105]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            starts.append(np.array(c[:N]))
            
    # Perturbed hex patterns
    for c in list(starts):
        c_pert = c + rng.normal(0, 0.005, c.shape)
        c_pert = np.clip(c_pert, 0.05, 0.95)
        starts.append(c_pert)
        
    # Random starts
    for _ in range(10):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(123)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = generate_starts()
    
    # --- Phase 1: Gradient Ascent from Multiple Starts ---
    for c0 in starts:
        c_curr, s_curr = gradient_ascent(c0, rng, steps=1500, init_step=0.008)
        if s_curr > best_sum:
            best_sum = s_curr
            best_c = c_curr.copy()
            _, best_r, _ = solve_lp_and_grad(best_c)
            
    # --- Phase 2: Powell Polish on LP Objective ---
    if best_c is not None:
        bounds_c = [(0.01, 0.99)] * (2 * N)
        res = minimize(lp_obj, best_c.flatten(), method='Powell', bounds=bounds_c,
                      options={'maxiter': 5000, 'ftol': 1e-14, 'xtol': 1e-14})
        c_p = res.x.reshape(N, 2)
        r_p, s_p, _ = solve_lp_and_grad(c_p)
        if s_p > best_sum:
            best_sum = s_p
            best_c = c_p
            best_r = r_p
            
        # Multiple perturbations & Powell to explore nearby basins
        for _ in range(5):
            c_tr = best_c + rng.normal(0, 0.003, best_c.shape)
            c_tr = np.clip(c_tr, 0.02, 0.98)
            res2 = minimize(lp_obj, c_tr.flatten(), method='Powell', bounds=bounds_c,
                           options={'maxiter': 2000, 'ftol': 1e-14})
            c2 = res2.x.reshape(N, 2)
            r2, s2, _ = solve_lp_and_grad(c2)
            if s2 > best_sum:
                best_sum = s2
                best_c = c2
                best_r = r2

    # --- Phase 3: Simulated Annealing on Centers ---
    c_curr = best_c.copy()
    s_curr = best_sum
    r_curr = best_r.copy()
    T = 0.004
    for step in range(2000):
        T *= 0.9995
        c_new = c_curr + rng.normal(0, T, c_curr.shape)
        c_new = np.clip(c_new, 0.02, 0.98)
        r_new, s_new, _ = solve_lp_and_grad(c_new)
        
        delta = s_new - s_curr
        if delta > 0 or rng.random() < np.exp(delta / max(T * 5.0, 1e-7)):
            c_curr = c_new
            s_curr = s_new
            r_curr = r_new
            if s_curr > best_sum:
                best_sum = s_curr
                best_c = c_curr.copy()
                best_r = r_curr.copy()
                
    # --- Phase 4: Final SLSQP Joint Polish ---
    def obj_joint(v): return -np.sum(v[2*N:])
    
    def cons_joint(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        cons = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
        idx_i, idx_j = np.triu_indices(N, 1)
        dx = c[idx_i, 0] - c[idx_j, 0]
        dy = c[idx_i, 1] - c[idx_j, 1]
        dr = r[idx_i] + r[idx_j]
        cons.append(np.sqrt(dx**2 + dy**2) - dr)
        return np.concatenate(cons)
        
    r_lp, _, _ = solve_lp_and_grad(best_c)
    v0 = np.concatenate([best_c.flatten(), r_lp])
    bounds_j = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                      constraints={'type': 'ineq', 'fun': cons_joint},
                      options={'maxiter': 5000, 'ftol': 1e-12})
        if np.min(cons_joint(res.x)) >= -1e-9:
            s = np.sum(res.x[2*N:])
            if s > best_sum:
                best_sum = s
                best_c = res.x[:2*N].reshape(N, 2).copy()
                best_r = res.x[2*N:].copy()
    except Exception:
        pass

    # --- Phase 5: Strict Numerical Repair ---
    centers = best_c.copy()
    radii = best_r.copy()
    for _ in range(50):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
