import numpy as np
from scipy.optimize import linprog

N = 26

def solve_lp_and_grad(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers and computes 
    the exact gradient of the objective with respect to center positions 
    using LP dual variables.
    """
    n_pairs = N * (N - 1) // 2
    n_bound = 4 * N
    A_ub = np.zeros((n_pairs + n_bound, N))
    b_ub = np.zeros(n_pairs + n_bound)
    
    idx = 0
    pair_data = []
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub[idx] = d
            pair_data.append((i, j, idx, d))
            idx += 1
            
    bound_data = []
    for i in range(N):
        x, y = centers[i]
        # r <= x (left wall)
        A_ub[idx, i] = 1.0; b_ub[idx] = x; bound_data.append((i, idx, 0)); idx += 1
        # r <= 1-x (right wall)
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; bound_data.append((i, idx, 1)); idx += 1
        # r <= y (bottom wall)
        A_ub[idx, i] = 1.0; b_ub[idx] = y; bound_data.append((i, idx, 2)); idx += 1
        # r <= 1-y (top wall)
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; bound_data.append((i, idx, 3)); idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0.0, None), method='highs')
    if not res.success:
        return np.zeros(N), 0.0, np.zeros((N, 2))
        
    radii = res.x
    duals = res.ineqlin.marginals
    grad = np.zeros((N, 2))

            grad[j] -= lam * v

            elif b_type == 3: grad[i, 1] -= mu    # Top wall: push down
            
    return radii, np.sum(radii), grad
