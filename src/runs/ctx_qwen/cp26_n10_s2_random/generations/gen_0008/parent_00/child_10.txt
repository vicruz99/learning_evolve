import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute pair indices and constraint matrix structure for speed
PAIR_INDICES = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        PAIR_INDICES.append((i, j))
N_PAIRS = len(PAIR_INDICES)
A_ub_pairs = np.zeros((N_PAIRS, N))
for k, (i, j) in enumerate(PAIR_INDICES):
    A_ub_pairs[k, i] = 1.0
    A_ub_pairs[k, j] = 1.0

def solve_lp(centers):
    """Solves LP to find maximum sum of radii for fixed centers. Returns radii, sum, duals, distances."""
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(N_PAIRS)
    for k, (i, j) in enumerate(PAIR_INDICES):
        b_ub[k] = dists[i, j]
        
    bounds = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_ub_pairs, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun, res.ineqlin.marginals, dists
    return np.zeros(N), 0.0, np.zeros(N_PAIRS), np.zeros((N, N))

def compute_gradient(c, duals, dists):
    """Computes gradient of sum of radii w.r.t center positions using LP duals."""
    grad = np.zeros_like(c)
    k = 0
    for i, j in PAIR_INDICES:
        lam = duals[k]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        k += 1
    return grad

def optimize_gradient(c0, max_iter=1200, rng=None):
    """Runs gradient ascent on centers to maximize sum of radii."""
    if rng is None:
        rng = np.random.default_rng()
    c = c0.copy()

        radii, s, duals, dists = solve_lp(c)
        if s > best_s:
            best_s = s
            best_c = c.copy()

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-10:
            break

        c_new = np.clip(c_new, 0.02, 0.98)

        if s_new > s + 1e-12:
            c = c_new
            s = s_new
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.8
            if step < 1e-9:
                break
                
        if it % 200 == 100:
            c += rng.normal(0, 0.0015, c.shape)
            c = np.clip(c, 0.02, 0.98)
            
    return best_c, best_s

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Computes boundary and non-overlap constraints (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    con.append(np.sqrt(dx**2 + dy**2) - dr)
    return np.concatenate(con)

def generate_starts(rng):

    return centers, radii, float(np.sum(radii))
