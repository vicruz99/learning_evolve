import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute LP constraint matrix structure for speed
PAIRS = []
for i in range(N):
    for j in range(i + 1, N):
        PAIRS.append((i, j))
N_PAIRS = len(PAIRS)
A_ub_pairs = np.zeros((N_PAIRS, N))
for k, (i, j) in enumerate(PAIRS):
    A_ub_pairs[k, i] = 1.0
    A_ub_pairs[k, j] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii given centers and computes gradient using duals."""
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[np.triu_indices(N, k=1)]
    
    bounds = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_ub_pairs, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, -1.0, np.zeros_like(c)
        
    radii = res.x
    duals = res.ineqlin.marginals
    grad = np.zeros_like(c)
    
    idx = 0
    for i, j in PAIRS:
        lam = duals[idx]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
    return radii, np.sum(radii), grad

def lp_obj_func(x):
    """Objective for derivative-free optimization: negative LP sum of radii."""
    _, s, _ = solve_lp_and_grad(x.reshape(N, 2))
    return -s

def gradient_ascent(c0, steps=600, init_step=0.008):
    """Gradient ascent on centers using LP duals with adaptive step size."""
    c = c0.copy()
    best_c = c.copy()
    best_sum = -1.0
    step = init_step
    
    for _ in range(steps):
        radii, s, grad = solve_lp_and_grad(c)
        if radii is None:
            break
        if s > best_sum:
            best_sum = s
            best_c = c.copy()
            
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-10:
            break
            
        c_new = c + step * (grad / g_norm)
        c_new = np.clip(c_new, 0.02, 0.98)
        
        _, s_new, _ = solve_lp_and_grad(c_new)
        if s_new > s:
            c = c_new
        else:
            step *= 0.85
    return best_c, best_sum

def coordinate_descent(c0, iters=5):
    """Optimizes each circle's position sequentially to escape local traps."""
    c = c0.copy()
    
    for _ in range(iters):
        for i in range(N):
            def obj_1d(p):
                c[i] = np.clip(p, 1e-5, 1.0 - 1e-5)
                _, s, _ = solve_lp_and_grad(c)
                return -s
                
            res = minimize(obj_1d, c[i], method='Nelder-Mead', 
                          options={'xatol': 1e-6, 'fatol': 1e-8, 'maxiter': 300})
            c[i] = res.x
    _, s, _ = solve_lp_and_grad(c)
    return c, s

def obj_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Boundary and non-overlap constraints for SLSQP."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    cons = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    cons.append(np.sqrt(dx**2 + dy**2) - dr)
    return np.concatenate(cons)

def generate_starts():
    """Generates diverse initial configurations."""
    starts = []
    rng = np.random.default_rng(42)
    
    # Hexagonal lattice patterns
    pats = [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], [6,6,5,5,4], 
            [5,5,5,6,5], [5,4,6,6,5], [6,5,5,5,5], [5,5,4,6,6], [4,5,6,5,6],
            [5,6,6,4,5], [6,5,5,6,4], [5,5,6,5,5]]
    for pat in pats:
        for r0 in [0.09, 0.095, 0.10, 0.105, 0.11]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            starts.append(np.array(c[:N]))
            
    # Grid patterns
    for sp in [5, 6]:
        x = np.linspace(0.08, 0.92, sp)
        y = x.copy()
        cx, cy = np.meshgrid(x, y)
        g = np.column_stack([cx.flatten(), cy.flatten()])
        if len(g) > N:
            g = g[:N]
        starts.append(g)
        
    # Random starts
    for _ in range(10):
        starts.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    return starts

def run_packing() -> tuple:
    rng = np.random.default_rng(123)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = generate_starts()
    
    # Phase 1: Gradient Ascent + Powell + Coordinate Descent from multiple starts
    for c_init in starts:
        c_opt, s_opt = gradient_ascent(c_init)
        r_opt, _, _ = solve_lp_and_grad(c_opt)
        
        # Powell refinement
        try:
            res = minimize(lp_obj_func, c_opt.flatten(), method='Powell', 
                          bounds=[(0.0, 1.0)]*(2*N), 
                          options={'maxiter': 1500, 'ftol': 1e-12, 'xtol': 1e-12})
            c_pow = res.x.reshape(N, 2)
            r_pow, s_pow, _ = solve_lp_and_grad(c_pow)
            if s_pow > s_opt:
                c_opt = c_pow
                s_opt = s_pow
                r_opt = r_pow
        except Exception:
            pass
            
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            best_r = r_opt.copy()
            
        # Coordinate descent refinement
        c_cd, s_cd = coordinate_descent(c_opt)
        r_cd, _, _ = solve_lp_and_grad(c_cd)
        if s_cd > best_sum:
            best_sum = s_cd
            best_c = c_cd.copy()
            best_r = r_cd.copy()

    # Phase 2: Joint SLSQP Polish
    r_lp, _, _ = solve_lp_and_grad(best_c)
    v0 = np.concatenate([best_c.flatten(), r_lp])
    bounds_j = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                      constraints={'type': 'ineq', 'fun': cons_joint},
                      options={'maxiter': 5000, 'ftol': 1e-12})
        if np.min(cons_joint(res.x)) >= -1e-9:
            s = np.sum(res.x[2*N:])
            if s > best_sum:
                best_sum = s
                best_c = res.x[:2*N].reshape(N, 2).copy()
                best_r = res.x[2*N:].copy()
    except Exception:
        pass

    # Phase 3: Perturbation & Restart from Best
    for _ in range(5):
        c_pert = best_c + rng.normal(0, 0.004, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        c_opt, s_opt = gradient_ascent(c_pert, steps=400)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            best_r = solve_lp_and_grad(c_opt)[0].copy()

    # Phase 4: Strict Numerical Repair
    centers = best_c.copy()
    radii = best_r.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
