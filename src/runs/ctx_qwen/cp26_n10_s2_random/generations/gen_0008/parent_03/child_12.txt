import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRI_U_IDX = np.triu_indices(N, k=1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure
A_ub_pre = np.zeros((NUM_PAIRS + 4 * N, N))
pair_indices = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub_pre[idx, i] = 1.0
        A_ub_pre[idx, j] = 1.0
        pair_indices.append((i, j))
        idx += 1
for i in range(N):
    for _ in range(4):
        A_ub_pre[idx, i] = 1.0
        idx += 1

def solve_lp_and_grad(centers):
    """
    Solves LP for optimal radii given centers and computes gradient of sum(r) w.r.t centers.
    Returns (radii, sum_r, grad) or (None, None, None) on failure.
    """
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    
    ub = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    ub = np.maximum(ub, 1e-8)
    
    # Pairwise distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.hypot(dx, dy)
    
    b_ub = np.zeros(A_ub_pre.shape[0])
    k = 0
    for i, j in pair_indices:
        b_ub[k] = dists[i, j]
        k += 1
    for i in range(n):
        b_ub[k] = x[i]; k += 1
        b_ub[k] = 1.0 - x[i]; k += 1
        b_ub[k] = y[i]; k += 1
        b_ub[k] = 1.0 - y[i]; k += 1
        
    bounds_r = [(0.0, u) for u in ub]
    res = linprog(-np.ones(n), A_ub=A_ub_pre, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if not res.success:
        return None, None, None
        
    radii = res.x
    duals = res.ineqlin.marginals
    
    grad = np.zeros_like(centers)
    k = 0
    for i, j in pair_indices:
        lam = duals[k]
        if lam > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        k += 1
        
    b_start = NUM_PAIRS
    for i in range(n):
        # Constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
        grad[i, 0] += duals[b_start + 4*i] - duals[b_start + 4*i + 1]
        grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
        
    return radii, np.sum(radii), grad

def lbfgs_wrapper(v):
    """Objective and gradient for L-BFGS-B"""
    centers = v.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, val, grad = solve_lp_and_grad(centers)
    if val is None:
        return 0.0, np.zeros_like(v)
    return -val, -grad.reshape(-1)

def obj_joint(v):
    """Objective for SLSQP: minimize negative sum of radii"""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Constraints for SLSQP: boundary and non-overlap (squared)"""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    dx = c[TRI_U_IDX[0], 0] - c[TRI_U_IDX[1], 0]
    dy = c[TRI_U_IDX[0], 1] - c[TRI_U_IDX[1], 1]
    dr = r[TRI_U_IDX[0]] + r[TRI_U_IDX[1]]
    con = np.concatenate([con, dx**2 + dy**2 - dr**2])
    return con

def generate_starts(rng):
    """Generates diverse initial configurations"""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5],
        [5, 5, 5, 5, 5, 1], [5, 6, 4, 6, 5]
    ]
    
    for pat in patterns:
        for r_est in [0.085, 0.095, 0.105, 0.115]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    if len(c) >= N: break
                    c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.004, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    for _ in range(15):
        c = rng.uniform(0.1, 0.9, (N, 2))
        starts.append(c)
        
    # Corner focused
    for _ in range(8):
        c = rng.uniform(0.2, 0.8, (N, 2))
        c[:4] = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        c += rng.normal(0, 0.01, c.shape)
        c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    return starts

                changed = True
        if not changed:

    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
