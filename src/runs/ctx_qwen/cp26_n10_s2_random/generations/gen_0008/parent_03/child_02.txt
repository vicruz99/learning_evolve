My strategy combines:
1. **Diverse Initialization**: Generate high-quality starting configurations using hexagonal lattice patterns with varying densities/shifts, purely random starts, and force-directed relaxations that naturally push circles apart to find feasible dense packings.
2. **Exact Gradient Ascent on Centers**: Use `L-BFGS-B` on circle centers while evaluating radii via Linear Programming (LP). The gradient is computed exactly using LP dual variables (shadow prices), which tells us how moving a center changes the maximum possible sum of radii. This is highly efficient for this problem structure.
3. **Simulated Annealing (SA)**: After gradient ascent, run SA on centers to escape shallow local minima. SA evaluates the LP sum of radii and accepts worse moves probabilistically based on temperature.
4. **Joint Non-Linear Polish**: Use `SLSQP` to optimize centers and radii simultaneously. This handles the non-smoothness of the LP envelope and fine-tunes boundary/pairwise contacts. We run this on the best configuration and perturbed variants.
5. **Strict Numerical Repair**: Iteratively shrink radii to guarantee non-overlap and boundary constraints within the validator's `1e-12` tolerance, while preserving as much radius sum as possible.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
A_ub_pre = None
pair_indices = None

def init_lp_structure():
    """Pre-construct the sparse structure of the LP constraint matrix."""
    global A_ub_pre, pair_indices
    num_pairs = N * (N - 1) // 2
    num_bound = 4 * N
    A = np.zeros((num_pairs + num_bound, N))
    pairs = []
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            pairs.append((i, j))
            idx += 1
    for i in range(N):
        for _ in range(4):
            A[idx, i] = 1.0
            idx += 1
    A_ub_pre = A
    pair_indices = pairs

init_lp_structure()

def solve_lp_and_grad(centers):
    """
    Solves LP for optimal radii given fixed centers and computes the gradient 
    of the sum of radii with respect to center positions using LP duals.
    """
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    
    b_ub = np.zeros(A_ub_pre.shape[0])
    idx = 0
    for i, j in pair_indices:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(n):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub_pre, b_ub=b_ub, 
                  bounds=[(0, u) for u in ub], method='highs')
    
    if not res.success:
        return None, None, None
        
    radii = res.x
    duals = res.ineqlin.marginals
    
    grad = np.zeros_like(centers)
    idx = 0
    for i, j in pair_indices:
        lam = duals[idx]
        if lam > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    boundary_start = len(pair_indices)
    for i in range(n):
        grad[i, 0] += duals[boundary_start + 4*i] - duals[boundary_start + 4*i + 1]
        grad[i, 1] += duals[boundary_start + 4*i + 2] - duals[boundary_start + 4*i + 3]
        
    return radii, np.sum(radii), grad

def lbfgs_wrapper(v):
    """Objective and gradient wrapper for L-BFGS-B."""
    centers = v.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, val, grad = solve_lp_and_grad(centers)
    if val is None:
        return 0.0, np.zeros_like(v)
    return -val, -grad.reshape(-1)

def obj_joint(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Constraints for joint SLSQP: boundary and pairwise non-overlap (squared)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    idx = np.triu_indices(N, 1)
    dx = c[idx[0], 0] - c[idx[1], 0]
    dy = c[idx[0], 1] - c[idx[1], 1]
    dr = r[idx[0]] + r[idx[1]]
    con = np.concatenate([con, dx**2 + dy**2 - dr**2])
    return con

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5],
        [5, 5, 6, 4, 6], [6, 5, 4, 6, 5], [4, 5, 5, 6, 6]
    ]
    
    for pat in patterns:
        for r_est in [0.08, 0.09, 0.10, 0.11]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        starts.append(c)
        
    # Force-directed starts to find naturally relaxed configurations
    for seed in range(5):
        rng_fd = np.random.default_rng(seed)
        c = rng_fd.uniform(0.2, 0.8, (N, 2))
        r = np.full(N, 0.05)
        for _ in range(200):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i+1, N):
                    d = np.linalg.norm(c[i] - c[j])
                    if d < r[i] + r[j] and d > 1e-5:
                        f = (r[i] + r[j] - d) / d
                        diff = c[i] - c[j]
                        forces[i] += diff * f
                        forces[j] -= diff * f
                for axis in [0, 1]:
                    if c[i, axis] < r[i]: forces[i, axis] += (r[i] - c[i, axis]) * 5
                    if c[i, axis] > 1 - r[i]: forces[i, axis] -= (c[i, axis] - (1 - r[i])) * 5
            c += forces * 0.01
            c = np.clip(c, 0.01, 0.99)
            r *= 1.002
        starts.append(c)
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_centers = None
    best_sum = -1.0
    bounds_xy = [(0.0, 1.0)] * (2 * N)
    
    starts = generate_starts(rng)
    
    # Phase 1: L-BFGS-B optimization on centers using exact LP gradient
    for c0 in starts:
        v0 = c0.flatten()
        try:
            res = minimize(lbfgs_wrapper, v0, jac=True,
                          method='L-BFGS-B', bounds=bounds_xy,
                          options={'maxiter': 800, 'ftol': 1e-13})
            c_opt = res.x.reshape(N, 2)
            _, s_opt, _ = solve_lp_and_grad(c_opt)
            if s_opt is not None and s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Simulated Annealing on centers to escape local minima
    if best_centers is not None:
        c_curr = best_centers.copy()
        s_curr = best_sum
        T = 0.005
        for step in range(1500):
            T *= 0.9995
            c_new = c_curr + rng.normal(0, T, c_curr.shape)
            c_new = np.clip(c_new, 0.02, 0.98)
            _, s_new, _ = solve_lp_and_grad(c_new)
            if s_new is None: s_new = -1e9
            
            if s_new > s_curr or rng.random() < np.exp((s_new - s_curr) / max(T, 1e-7)):
                c_curr = c_new
                s_curr = s_new
                if s_curr > best_sum:
                    best_sum = s_curr
                    best_centers = c_curr.copy()
                    
    # Phase 3: SLSQP joint polish on centers and radii
    r_lp, _, _ = solve_lp_and_grad(best_centers)
    v0 = np.concatenate([best_centers.flatten(), r_lp])
    bounds_joint = bounds_xy + [(0.0, 0.5)] * N
    
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                       constraints={'type': 'ineq', 'fun': cons_joint},
                       options={'maxiter': 8000, 'ftol': 1e-14})
        if np.sum(res.x[2*N:]) > best_sum:
            best_centers = res.x[:2*N].reshape(N, 2)
            best_radii = res.x[2*N:]
        else:
            best_radii = r_lp
    except Exception:
        best_radii = r_lp
        
    # Phase 4: Perturbation + SLSQP to explore nearby high-quality basins
    for _ in range(10):
        c_tr = best_centers.copy()
        c_tr += rng.normal(0, 0.003, c_tr.shape)
        c_tr = np.clip(c_tr, 0.02, 0.98)
        _, r_tr, _ = solve_lp_and_grad(c_tr)
        v_tr = np.concatenate([c_tr.flatten(), r_tr])
        try:
            res = minimize(obj_joint, v_tr, method='SLSQP', bounds=bounds_joint,
                           constraints={'type': 'ineq', 'fun': cons_joint},
                           options={'maxiter': 4000, 'ftol': 1e-14})
            if np.sum(res.x[2*N:]) > best_sum:
                best_sum = np.sum(res.x[2*N:])
                best_centers = res.x[:2*N].reshape(N, 2)
                best_radii = res.x[2*N:]
        except Exception:
            pass
            
    # Phase 5: Strict numerical repair to guarantee validation passes within 1e-12
    for _ in range(100):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(best_centers[i] - best_centers[j])
                req = best_radii[i] + best_radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            mr = min(x, 1.0-x, y, 1.0-y)
            if r > mr + 1e-12:
                best_radii[i] = mr
                changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
