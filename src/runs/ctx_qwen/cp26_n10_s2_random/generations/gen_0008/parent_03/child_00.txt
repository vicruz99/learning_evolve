import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute LP constraint matrix structure (constant for fixed N)
NUM_PAIRS = N * (N - 1) // 2
NUM_BOUND = 4 * N
A_ub_pre = np.zeros((NUM_PAIRS + NUM_BOUND, N))
PAIR_INDICES = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub_pre[idx, i] = 1.0
        A_ub_pre[idx, j] = 1.0
        PAIR_INDICES.append((i, j))
        idx += 1
for i in range(N):
    for _ in range(4):
        A_ub_pre[idx, i] = 1.0
        idx += 1

def solve_lp_and_gradient(centers):
    """Solves LP for optimal radii given fixed centers and computes exact gradient via duals."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    
    b_ub = np.zeros(NUM_PAIRS + NUM_BOUND)
    k = 0
    for i, j in PAIR_INDICES:
        b_ub[k] = dists[i, j]
        k += 1
    for i in range(n):
        b_ub[k] = centers[i, 0]; k += 1
        b_ub[k] = 1.0 - centers[i, 0]; k += 1
        b_ub[k] = centers[i, 1]; k += 1
        b_ub[k] = 1.0 - centers[i, 1]; k += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub_pre, b_ub=b_ub,
                  bounds=[(0.0, u) for u in ub], method='highs', 
                  options={'presolve': True, 'time_limit': 10.0})
    if not res.success:
        return None, None, None
        
    radii = res.x
    duals = res.ineqlin.marginals
    grad = np.zeros_like(centers)
    
    # Pairwise gradient from distance constraints
    k = 0
    for i, j in PAIR_INDICES:
        lam = duals[k]
        if lam > 1e-7:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        k += 1
        
    # Boundary gradient from wall constraints
    bound_start = NUM_PAIRS
    for i in range(n):
        grad[i, 0] += duals[bound_start + 4*i] - duals[bound_start + 4*i + 1]
        grad[i, 1] += duals[bound_start + 4*i + 2] - duals[bound_start + 4*i + 3]
        
    return radii, np.sum(radii), grad

def lbfgs_wrapper(v):
    """Objective and gradient wrapper for L-BFGS-B (minimizes negative sum)."""
    centers = v.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, val, grad = solve_lp_and_gradient(centers)
    if val is None:
        return 0.0, np.zeros_like(v)
    return -val, -grad.reshape(-1)

def obj_joint(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Constraints for joint SLSQP: boundary and pairwise non-overlap (squared for smoothness)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    idx = np.triu_indices(N, 1)
    dx = c[idx[0], 0] - c[idx[1], 0]
    dy = c[idx[0], 1] - c[idx[1], 1]
    con = np.concatenate([con, dx**2 + dy**2 - (r[idx[0]] + r[idx[1]])**2])
    return con

def force_directed_init(rng, iters=600):
    """Generates a repulsion-based initial configuration."""
    c = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(iters):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(c[i] - c[j])
                if d < 0.15 and d > 1e-9:
                    f = (0.15 - d) * 20.0 / d
                    diff = c[i] - c[j]
                    forces[i] += diff * f
                    forces[j] -= diff * f
        c += forces * 0.0005
        c = np.clip(c, 0.02, 0.98)
    return c

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],

    ]
    
    for pat in patterns:
        for r_est in [0.085, 0.095, 0.105, 0.115]:
            c = []
            y = r_est

                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift

                    c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)

        starts.append(force_directed_init(rng))
        
    for _ in range(10):
        c = rng.uniform(0.15, 0.85, (N, 2))
        starts.append(c)
        
    return starts

def optimize_lbbfgs(c0, bounds, max_iter=600):
    """Local optimization of centers using L-BFGS-B."""
    try:
        res = minimize(lbfgs_wrapper, c0.flatten(), jac=True,
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        return res.x.reshape(N, 2), -res.fun
    except Exception:
        return c0, 0.0

def simulated_annealing(centers, bounds, rng, steps=1500):
    """Simulated Annealing on centers to escape local minima."""
    c_curr = centers.copy()
    _, s_curr, _ = solve_lp_and_gradient(c_curr)
    if s_curr is None: s_curr = 0.0
    best_c = c_curr.copy()
    best_s = s_curr
    T = 0.006
    
    for step in range(steps):
        T *= 0.998
        c_new = c_curr + rng.normal(0, max(T, 1e-4), c_curr.shape)
        c_new = np.clip(c_new, bounds[:, 0], bounds[:, 1])
        _, s_new, _ = solve_lp_and_gradient(c_new)
        if s_new is None: s_new = 0.0
        
        delta = s_new - s_curr
        if delta > 0 or rng.random() < np.exp(delta / max(T * 5.0, 1e-7)):
            c_curr = c_new
            s_curr = s_new
            if s_curr > best_s:
                best_s = s_curr
                best_c = c_curr.copy()
                
    return best_c, best_s

def repair(centers, radii, tol=1e-9):
    """Deterministic repair to ensure strict validity."""
    radii = radii.copy()
    for _ in range(50):
        changed = False
        for i in range(N):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx - tol:
                radii[i] = max(mx, 0.0)
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d - tol:
                    shrink = (radii[i] + radii[j] - d) / 2.0
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_centers = None

    final_sum = float(np.sum(best_radii))
    return best_centers, best_radii, final_sum
