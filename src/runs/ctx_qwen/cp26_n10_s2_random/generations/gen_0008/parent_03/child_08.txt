import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute LP constraint matrix structure for efficiency
LP_NUM_PAIRS = N * (N - 1) // 2
LP_NUM_CONST = LP_NUM_PAIRS + 4 * N
LP_A = np.zeros((LP_NUM_CONST, N))
LP_PAIRS = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        LP_A[idx, i] = 1.0
        LP_A[idx, j] = 1.0
        LP_PAIRS.append((i, j))
        idx += 1
for i in range(N):
    for _ in range(4):
        LP_A[idx, i] = 1.0
        idx += 1

def solve_lp_radii(centers):
    """Solves LP to find maximal radii for fixed centers."""
    n = centers.shape[0]
    b_ub = np.zeros(LP_NUM_CONST)
    
    # Pairwise distance constraints
    idx = 0
    for i, j in LP_PAIRS:
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        b_ub[idx] = np.hypot(dx, dy)
        idx += 1
        
    # Boundary constraints
    for i in range(n):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    try:
        res = linprog(-np.ones(n), A_ub=LP_A, b_ub=b_ub, 
                      bounds=[(0.0, 0.5)] * n, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 1e-7), 1e-7 * n

def obj_centers(c_flat):
    """Objective for center optimization: minimize negative LP sum of radii."""
    c = np.clip(c_flat.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    _, s = solve_lp_radii(c)
    return -s

    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
