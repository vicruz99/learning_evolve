import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute LP constraint matrix structure (constant for fixed N)
def get_lp_structure():
    n_pairs = N * (N - 1) // 2
    A = np.zeros((n_pairs + 4 * N, N))
    pairs = []
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            pairs.append((i, j))
            idx += 1
    for i in range(N):
        for _ in range(4):
            A[idx, i] = 1.0
            idx += 1
    return A, pairs

A_ub_pre, PAIRS = get_lp_structure()

def solve_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(A_ub_pre.shape[0])
    idx = 0
    for i, j in PAIRS:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    bounds = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_ub_pre, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
        
    # Fallback to analytical caps if LP fails
    r_bound = ub
    r_pair = 0.5 * np.min(dists, axis=1)
    r_fallback = np.minimum(r_bound, r_pair)
    return r_fallback, np.sum(r_fallback)

def lp_obj(x):
    """Objective for centers-only optimization: minimize negative LP sum of radii."""
    _, s = solve_lp_radii(x.reshape(N, 2))
    return -s

def obj_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Computes boundary and non-overlap constraints (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    cons = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    cons.append(np.sqrt(dx**2 + dy**2) - dr)
    return np.concatenate(cons)

def obj_i_wrapper(x_flat, c_curr, i):
    """Objective for coordinate-wise optimization of circle i."""
    nc = c_curr.copy()
    nc[i] = np.clip(x_flat, 0.01, 0.99)
    _, s = solve_lp_radii(nc)
    return -s

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], [5,5,5,6,5], 
                [6,6,5,5,4], [5,4,6,6,5], [6,5,5,5,5], [5,5,4,6,6], [4,5,6,5,6]]
                
    for pat in patterns:
        for r0 in [0.085, 0.095, 0.105, 0.115]:
            for rot in [0.0, 0.05, 0.1]:
                c = []
                y = r0
                for r_idx, cnt in enumerate(pat):
                    shift = r0 if r_idx % 2 == 1 else 0.0
                    x = r0 + shift
                    for _ in range(cnt):
                        if len(c) < N:
                            c.append([x, y])
                        x += 2.0 * r0
                    y += r0 * np.sqrt(3)
                c = np.array(c[:N])
                
                if rot != 0.0:
                    c_center = c.mean(axis=0)
                    c_rel = c - c_center
                    cs, sn = np.cos(rot), np.sin(rot)
                    c_rot = np.column_stack([c_rel[:,0]*cs - c_rel[:,1]*sn, 
                                             c_rel[:,0]*sn + c_rel[:,1]*cs])
                    c = c_rot + c_center
                    
                c = (c - c.min(axis=0)) / (c.max(axis=0) - c.min(axis=0) + 1e-9)
                c = c * 0.8 + 0.1
                c += rng.normal(0, 0.002, c.shape)
                c = np.clip(c, 0.02, 0.98)
                starts.append(c)
                
    for seed in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(200):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i+1, N):
                    d = np.linalg.norm(c[i] - c[j])
                    if d < 0.2:
                        f = (0.2 - d) * 5.0 / max(d, 1e-9)
                        diff = c[i] - c[j]
                        forces[i] += diff * f
                        forces[j] -= diff * f
            c += forces * 0.005
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_c = None
    best_r = None
    best_sum = -1.0
    
    bounds_j = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    bounds_c = [(0.0, 1.0)] * (2*N)

    # Phase 1: SLSQP Joint Optimization
    for c_init in starts:

                              constraints={'type': 'ineq', 'fun': cons_joint},

            except Exception:
                pass

    if best_c is not None:
        res_p = minimize(lp_obj, best_c.flatten(), method='Powell', bounds=bounds_c,
                        options={'maxiter': 2000, 'ftol': 1e-12})
        c_p = res_p.x.reshape(N, 2)
        r_p, s_p = solve_lp_radii(c_p)

            best_c = c_p

    s_curr = best_sum
    r_curr = best_r.copy()

            best_local_c = c_curr[i].copy()
            best_local_s = s_curr

                c_tmp[i] = nc_i

                pass

            s_curr = s_new
            r_curr = r_new

                best_sum = s_curr
                best_c = c_curr.copy()
                best_r = r_curr.copy()

                best_r = res.x[2*N:].copy()
    except Exception:
        pass

    # Phase 6: Strict Numerical Repair
    centers = best_c.copy()
    radii = best_r.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])

            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
