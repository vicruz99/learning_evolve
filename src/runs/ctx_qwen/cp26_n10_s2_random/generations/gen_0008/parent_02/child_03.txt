import numpy as np
from scipy.optimize import linprog

N = 26

# Precompute LP constraint matrix structure globally for efficiency
pair_idx = [(i, j) for i in range(N) for j in range(i + 1, N)]
NUM_PAIRS = len(pair_idx)
NUM_BOUND = 4 * N
A_UB_STRUCT = np.zeros((NUM_PAIRS + NUM_BOUND, N))
for idx, (i, j) in enumerate(pair_idx):
    A_UB_STRUCT[idx, i] = 1.0
    A_UB_STRUCT[idx, j] = 1.0
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_UB_STRUCT[base, i] = 1.0
    A_UB_STRUCT[base + 1, i] = 1.0
    A_UB_STRUCT[base + 2, i] = 1.0
    A_UB_STRUCT[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii and computes exact gradient w.r.t centers using duals."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.empty(NUM_PAIRS + NUM_BOUND)
    idx = 0
    for i, j in pair_idx:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(n):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_UB_STRUCT, b_ub=b_ub, 
                  bounds=[(0, None)] * n, method='highs')
    if not res.success:
        return None, -1.0, None
        
    radii = res.x
    duals = res.ineqlin.marginals
    grad = np.zeros_like(centers)
    
    idx = 0
    for i, j in pair_idx:
        lam = duals[idx]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    b_start = NUM_PAIRS
    for i in range(n):
        grad[i, 0] += duals[b_start + 4*i] - duals[b_start + 4*i + 1]
        grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
        
    return radii, np.sum(radii), grad

def optimize_gradient(centers0, rng, max_iter=1200, init_step=0.007):
    """Gradient ascent on centers using LP duals with adaptive step and restarts."""
    centers = centers0.copy()
    best_centers = centers.copy()
    best_sum = -1.0
    step = init_step
    no_improve = 0
    
    for k in range(max_iter):
        radii, curr_sum, grad = solve_lp_and_grad(centers)
        if radii is None:
            break
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        if no_improve > 40:
            step *= 0.75
        elif no_improve > 15:
            step *= 0.9
            
        if step < 1e-10:
            break
            
        # Periodic restart from best to escape shallow basins
        if k % 150 == 100 and no_improve > 20:
            centers = best_centers + rng.normal(0, step * 2.0, best_centers.shape)
            centers = np.clip(centers, 0.02, 0.98)
            no_improve = 0
            
        centers += step * grad
        centers = np.clip(centers, 0.005, 0.995)
        
    return best_centers, best_sum

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5],
        [5, 5, 5, 5, 6], [6, 6, 4, 5, 5], [5, 6, 4, 5, 6],
        [5, 5, 5, 5, 5, 1], [6, 5, 5, 5, 5], [4, 5, 6, 6, 5]
    ]
    
    for pat in patterns:
        for r0 in [0.085, 0.090, 0.095, 0.100, 0.105]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            c = np.array(c[:N])
            c += rng.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    for _ in range(25):
        c = rng.uniform(0.1, 0.9, (N, 2))
        starts.append(c)
        
    for _ in range(10):
        c = rng.uniform(0.2, 0.8, (N, 2))
        c[:4] = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        starts.append(c)
        
    return starts

def repair(centers, radii):
    """Strictly enforces non-overlap and boundary constraints within validator tolerance."""
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d)/2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple:
    rng = np.random.default_rng(42)
    
    best_c = None
    best_sum = -1.0
    
    starts = generate_starts(rng)
    
    # Phase 1: Gradient Ascent from diverse starts
    for c0 in starts:
        c_opt, s_opt = optimize_gradient(c0, rng, max_iter=1000, init_step=0.006)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            
    # Phase 2: Perturbation & Refinement to escape local minima
    if best_c is not None:
        for _ in range(8):
            c_per = best_c + rng.normal(0, 0.004, best_c.shape)
            c_per = np.clip(c_per, 0.02, 0.98)
            c_opt2, s_opt2 = optimize_gradient(c_per, rng, max_iter=800, init_step=0.004)
            if s_opt2 > best_sum:
                best_sum = s_opt2
                best_c = c_opt2.copy()
                
        # Local search: try moving individual circles
        for _ in range(3):
            c_local = best_c.copy()
            for _ in range(200):
                i = rng.integers(0, N)
                old_pos = c_local[i].copy()
                dir = rng.normal(0, 1, 2)
                for step in [0.005, 0.002, 0.0005]:
                    c_local[i] = np.clip(old_pos + dir * step, 0.01, 0.99)
                    _, s_local, _ = solve_lp_and_grad(c_local)
                    if s_local > best_sum:
                        best_sum = s_local
                        best_c = c_local.copy()
                        break
                    c_local[i] = old_pos
                    
    # Phase 3: Final exact LP solve
    best_r, best_sum, _ = solve_lp_and_grad(best_c)
    
    centers = best_c
    radii = repair(centers, best_r)
    return centers, radii, float(np.sum(radii))
