import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIU_IJ = np.triu_indices(N, 1)

def get_lp_structure():
    n_pairs = N * (N - 1) // 2
    n_bound = 4 * N
    A_ub = np.zeros((n_pairs + n_bound, N))
    pair_idx = []
    k = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            pair_idx.append((i, j))
            k += 1
    for i in range(N):
        base = n_pairs + 4 * i
        A_ub[base, i] = 1.0
        A_ub[base+1, i] = 1.0
        A_ub[base+2, i] = 1.0
        A_ub[base+3, i] = 1.0
    return A_ub, pair_idx

A_LU, PAIR_IDX = get_lp_structure()

def solve_lp_and_grad(centers):
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    dists[np.arange(n), np.arange(n)] = 1e9
    
    b_ub = np.zeros(A_LU.shape[0])
    k = 0
    for i, j in PAIR_IDX:
        b_ub[k] = dists[i, j]
        k += 1
    for i in range(n):
        base = N*(N-1)//2 + 4*i
        b_ub[base] = centers[i, 0]
        b_ub[base+1] = 1.0 - centers[i, 0]
        b_ub[base+2] = centers[i, 1]
        b_ub[base+3] = 1.0 - centers[i, 1]
        
    res = linprog(-np.ones(n), A_ub=A_LU, b_ub=b_ub, 
                  bounds=[(0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.zeros(n), 0.0, np.zeros_like(centers)
        
    radii = res.x
    duals = res.ineqlin.marginals
    grad = np.zeros_like(centers)
    
    k = 0
    for i, j in PAIR_IDX:
        lam = duals[k]
        if lam > 1e-7:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        k += 1
        
    base = N*(N-1)//2
    for i in range(n):
        mu_L = duals[base + 4*i]
        mu_R = duals[base + 4*i + 1]
        mu_B = duals[base + 4*i + 2]
        mu_T = duals[base + 4*i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, np.sum(radii), grad

def gradient_ascent(c0, steps=1500, init_step=0.008, rng=None):
    c = c0.copy()
    best_c = c.copy()
    best_sum = -1.0
    step = init_step
    no_improve = 0
    
    for k in range(steps):
        radii, s, grad = solve_lp_and_grad(c)
        if s > best_sum:
            best_sum = s
            best_c = c.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        if no_improve > 60:
            step *= 0.8
        elif no_improve > 25:
            step *= 0.9
            
        if step < 1e-11:
            break
            
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-9:
            c += step * (grad / g_norm)
        elif rng is not None:
            c += rng.normal(0, step * 0.2, c.shape)
            
        c = np.clip(c, 0.001, 0.999)
        
    return best_c, best_sum

def generate_hex_config(rows, r_est, rng):
    pts = []
    y = r_est
    for r_idx, cnt in enumerate(rows):
        shift = r_est if r_idx % 2 == 1 else 0.0
        x = r_est + shift
        for _ in range(cnt):
            pts.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3.0)
    pts = np.array(pts[:N])
    pts += rng.normal(0, 0.005, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def obj_joint(v):
    return -np.sum(v[2*N:])
    
def cons_joint(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    idx_i, idx_j = TRIU_IJ
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def run_packing() -> tuple:
    rng = np.random.default_rng(123)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = []
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [6,6,5,5,4], 
        [4,6,6,6,4], [5,5,6,5,5], [6,5,5,5,5], [5,6,4,5,6],
        [4,5,6,5,6], [6,4,6,5,5]
    ]
    
    for pat in patterns:
        for r0 in [0.09, 0.095, 0.10, 0.105]:
            starts.append(generate_hex_config(pat, r0, rng))
            
    for _ in range(10):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    for c_init in starts:
        c_opt, s_opt = gradient_ascent(c_init, steps=1500, init_step=0.008, rng=rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            
    if best_c is not None:
        r_lp, _, _ = solve_lp_and_grad(best_c)
        v0 = np.concatenate([best_c.flatten(), r_lp])
        
        bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        
        for _ in range(3):
            try:
                res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds,
                               constraints={'type': 'ineq', 'fun': cons_joint},
                               options={'maxiter': 5000, 'ftol': 1e-14})
                if np.sum(res.x[2*N:]) > best_sum:
                    best_c = res.x[:2*N].reshape(N, 2).copy()
                    best_r = res.x[2*N:].copy()
                    best_sum = np.sum(best_r)
                v0 = best_c.flatten()
                v0[-N:] = best_r
            except:
                break
                
        for _ in range(4):
            c_per = best_c + rng.normal(0, 0.003, best_c.shape)
            c_per = np.clip(c_per, 0.02, 0.98)
            r_per, s_per, _ = solve_lp_and_grad(c_per)
            v0 = np.concatenate([c_per.flatten(), r_per])
            try:
                res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds,
                               constraints={'type': 'ineq', 'fun': cons_joint},
                               options={'maxiter': 3000, 'ftol': 1e-14})
                if np.sum(res.x[2*N:]) > best_sum:
                    best_c = res.x[:2*N].reshape(N, 2).copy()
                    best_r = res.x[2*N:].copy()
                    best_sum = np.sum(best_r)
            except:
                pass

    final_r, final_s, _ = solve_lp_and_grad(best_c)
    best_r = final_r
    best_sum = final_s
    
    radii = best_r.copy()
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d)/2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(best_c[i,0], 1.0-best_c[i,0], best_c[i,1], 1.0-best_c[i,1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    
    return best_c, radii, float(np.sum(radii))
