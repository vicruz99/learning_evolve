import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def get_lp_structure():
    """Precompute the sparse structure of the LP constraint matrix."""
    num_pairs = N * (N - 1) // 2
    num_bound = 4 * N
    A_ub = np.zeros((num_pairs + num_bound, N))
    pair_idx = []
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            pair_idx.append((i, j))
            idx += 1
    for i in range(N):
        base = num_pairs + 4 * i
        A_ub[base, i] = 1.0
        A_ub[base+1, i] = 1.0
        A_ub[base+2, i] = 1.0
        A_ub[base+3, i] = 1.0
    return A_ub, pair_idx, num_pairs

A_LP, PAIR_IDX, NUM_PAIRS = get_lp_structure()

def solve_lp_and_grad(centers):
    """
    Solves LP for optimal radii given fixed centers and computes the exact gradient 
    of the sum of radii with respect to center positions using LP dual variables.
    """
    ub_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                          np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(A_LP.shape[0])
    idx = 0
    for i, j in PAIR_IDX:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub,
                  bounds=[(0, 0.5)] * N, method='highs')
    
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(centers)
        
    radii = res.x
    duals = res.ineqlin.marginals if hasattr(res, 'ineqlin') and res.ineqlin is not None else None
    grad = np.zeros_like(centers)
    
    if duals is not None:
        # Pairwise forces from distance constraints
        idx = 0
        for i, j in PAIR_IDX:
            mu = duals[idx]
            if mu > 1e-8:
                d = dists[i, j]
                if d > 1e-9:
                    vec = (centers[i] - centers[j]) / d
                    grad[i] += mu * vec
                    grad[j] -= mu * vec
            idx += 1
            
        # Boundary forces from wall constraints
        bound_start = NUM_PAIRS
        for i in range(N):
            mu_L = duals[bound_start + 4*i]
            mu_R = duals[bound_start + 4*i + 1]
            mu_B = duals[bound_start + 4*i + 2]
            mu_T = duals[bound_start + 4*i + 3]
            grad[i, 0] += mu_L - mu_R
            grad[i, 1] += mu_B - mu_T
            
    return radii, np.sum(radii), grad

def optimize_gradient(centers0, steps=3000, init_step=0.01, seed=0):
    """Runs gradient ascent on centers to maximize sum of radii."""
    rng = np.random.default_rng(seed)
    centers = centers0.copy()
    best_centers = centers.copy()
    best_sum = -1.0
    step = init_step
    vel = np.zeros_like(centers)
    
    for k in range(steps):
        radii, curr_sum, grad = solve_lp_and_grad(centers)
        
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            
        # Adaptive step decay
        if k % 300 == 0 and k > 0:
            step *= 0.93
            
        # Momentum-based update
        vel = 0.45 * vel + step * grad
        centers += vel
        centers = np.clip(centers, 0.005, 0.995)
        
        # Periodic jitter to escape plateaus
        if k % 600 == 0 and k > 0:
            centers += rng.normal(0, 0.0012 * step, centers.shape)
            centers = np.clip(centers, 0.005, 0.995)
            
    return best_centers, best_sum

def generate_starts(rng):
    """Generates diverse initial configurations based on hexagonal lattices and random seeds."""
    starts = []
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [6,6,5,5,4], 
        [4,6,6,6,4], [5,5,6,5,5], [5,4,6,5,6], [6,5,5,5,5]
    ]
    
    for pat in patterns:
        for r0 in [0.090, 0.096, 0.102]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            c = np.array(c[:N])
            c += rng.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    # Add purely random starts
    for _ in range(12):
        starts.append(rng.uniform(0.12, 0.88, (N, 2)))
        
    return starts

def repair(centers, radii):
    """Deterministic repair to ensure strict validity."""
    radii = radii.copy()
    for _ in range(60):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                req = radii[i] + radii[j]
                if d < req - 1e-10:
                    shrink = (req - d)/2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-10:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = generate_starts(rng)
    
    # Phase 1: Gradient ascent from multiple diverse starts
    for i, c0 in enumerate(starts):
        c_opt, s_opt = optimize_gradient(c0, steps=3500, init_step=0.012, seed=i*17)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            
    # Phase 2: Perturb best configuration and re-optimize to escape local minima
    for _ in range(10):
        c_pert = best_c + rng.normal(0, 0.006, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)

        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            
    # Phase 3: Coordinate descent / Local search for micro-adjustments
    c = best_c.copy()
    step_local = 0.006
    for epoch in range(120):
        improved = False
        for i in range(N):
            _, curr_s, _ = solve_lp_and_grad(c)
            directions = [(step_local, 0), (-step_local, 0), (0, step_local), (0, -step_local)]
            for dx, dy in directions:
                nc = c.copy()
                nc[i, 0] += dx
                nc[i, 1] += dy
                nc[i] = np.clip(nc[i], 0.005, 0.995)
                _, new_s, _ = solve_lp_and_grad(nc)
                if new_s > curr_s + 1e-9:
                    c = nc
                    improved = True
                    break
        if not improved:
            step_local *= 0.87
        if step_local < 1e-7:
            break
            
    best_c = c
    radii, best_sum, _ = solve_lp_and_grad(best_c)
    best_r = radii
    
    # Phase 4: Joint SLSQP polish for final precision
    v0 = np.concatenate([best_c.flatten(), best_r])
    
    def obj_sl(v):
        return -np.sum(v[2*N:])
        
    def cons_sl(v):
        cc = v[:2*N].reshape(N, 2)
        rr = v[2*N:]
        con = []
        con.append(cc[:, 0] - rr)
        con.append(1.0 - cc[:, 0] - rr)
        con.append(cc[:, 1] - rr)
        con.append(1.0 - cc[:, 1] - rr)
        idx_i, idx_j = np.triu_indices(N, 1)
        d = np.linalg.norm(cc[idx_i] - cc[idx_j], axis=1)
        con.append(d - (rr[idx_i] + rr[idx_j]))
        return np.concatenate(con)
        
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    try:
        res_sl = minimize(obj_sl, v0, method='SLSQP', bounds=bounds,
                          constraints={'type': 'ineq', 'fun': cons_sl},
                          options={'maxiter': 4000, 'ftol': 1e-14})

            best_c = res_sl.x[:2*N].reshape(N, 2)
            best_r = res_sl.x[2*N:]
            best_sum = np.sum(best_r)
    except Exception:
        pass
        
    # Phase 5: Strict numerical safety repair
    best_r = repair(best_c, best_r)
    final_sum = float(np.sum(best_r))
    return best_c, best_r, final_sum
