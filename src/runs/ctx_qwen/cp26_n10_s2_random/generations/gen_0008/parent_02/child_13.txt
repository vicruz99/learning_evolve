TRIU_IND = np.triu_indices(N, k=1)

def get_constraints(x):

    R = x[2*N:]
    c = []
    c.append(X[:, 0] - R)
    c.append(1.0 - X[:, 0] - R)
    c.append(X[:, 1] - R)
    c.append(1.0 - X[:, 1] - R)

def solve_lp(centers):

            b_ub[idx] = dists[i, j]
            idx += 1
