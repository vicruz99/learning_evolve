import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii subject to non-overlap and boundary constraints."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-10)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    n_pairs = N * (N - 1) // 2
    A_ub = np.zeros((n_pairs + N, N))
    b_ub = np.zeros(n_pairs + N)
    
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
    for i in range(N):
        A_ub[idx, i] = 1.0
        b_ub[idx] = ub[i]
        idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x, np.sum(res.x)
    return np.zeros(N), 0.0

def lp_neg_sum(centers_flat):
    """Objective for center optimization: negative sum of optimal radii."""
    c = centers_flat.reshape(N, 2)
    _, s = solve_lp(c)
    return -s

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 5, 5, 6], [6, 4, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [6, 5, 5, 5, 5],
        [5, 4, 6, 5, 6], [4, 5, 6, 5, 6]
    ]
    for pat in patterns:
        for r0 in [0.08, 0.09, 0.10, 0.11]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 * 0.5 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.01, 0.99)
            starts.append(c)
            
    for _ in range(20):
        starts.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    for _ in range(10):
        c = rng.uniform(0.05, 0.95, (N, 2))
        c[0] = [0.12, 0.12]
        c[1] = [0.88, 0.12]
        c[2] = [0.12, 0.88]
        c[3] = [0.88, 0.88]
        starts.append(c)
        
    return starts

def obj_joint(v):
    return -np.sum(v[2*N:])

def cons_joint(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    i, j = np.triu_indices(N, 1)
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def run_packing() -> tuple:
    rng = np.random.default_rng(42)
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = generate_starts(rng)
    bounds = [(0.0, 1.0)] * (2 * N)
    
    # Phase 1: Powell optimization on centers using exact LP evaluation
    for c_init in starts:
        try:
            res = minimize(lp_neg_sum, c_init.flatten(), method='Powell', bounds=bounds,
                           options={'maxiter': 600, 'ftol': 1e-14, 'xtol': 1e-14})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt = solve_lp(c_opt)
            
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except:
            pass
            
    # Phase 2: Jitter & Refine to escape local minima
    if best_c is not None:
        for scale in [0.025, 0.012, 0.005, 0.002]:
            for _ in range(10):
                c_pert = best_c + rng.normal(0, scale, best_c.shape)
                c_pert = np.clip(c_pert, 0.01, 0.99)
                try:
                    res = minimize(lp_neg_sum, c_pert.flatten(), method='Powell', bounds=bounds,
                                   options={'maxiter': 400, 'ftol': 1e-14})
                    c_opt = res.x.reshape(N, 2)
                    r_opt, s_opt = solve_lp(c_opt)
                    if s_opt > best_sum:
                        best_sum = s_opt
                        best_c = c_opt.copy()
                        best_r = r_opt.copy()
                except:
                    pass
                    
    # Phase 3: Joint SLSQP polish for centers and radii
    if best_c is not None:
        v0 = np.concatenate([best_c.flatten(), best_r])
        bounds_joint = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        
        for _ in range(3):
            try:
                res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                               constraints={'type': 'ineq', 'fun': cons_joint},
                               options={'maxiter': 6000, 'ftol': 1e-14})
                if np.min(cons_joint(res.x)) >= -1e-8:
                    s = np.sum(res.x[2*N:])
                    if s > best_sum:
                        best_sum = s
                        best_c = res.x[:2*N].reshape(N, 2).copy()
                        best_r = res.x[2*N:].copy()
                        v0 = res.x.copy()
            except:
                break
                
    # Phase 4: Strict numerical repair to guarantee validator tolerance compliance
    centers = best_c.copy()
    radii = best_r.copy()
    
    for _ in range(60):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
