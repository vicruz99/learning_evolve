idx = 0

    centers = x.reshape(N, 2)
    _, s, g = solve_lp_and_grad(centers)
    return -s, -g.flatten()

def obj_lp_no_grad(x):
    """Objective without gradient for Powell/Nelder-Mead."""
    centers = x.reshape(N, 2)
    _, s, _ = solve_lp_and_grad(centers)
    return -s

def joint_obj(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def joint_cons(v):
