N = 26

NUM_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):

def obj_joint(v):

def cons_joint(v):
    c = v[:2*N].reshape(N, 2)

    con.append(c[:, 0] - r)
