from scipy.optimize import linprog, minimize

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute constant LP constraint matrix structure
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii given fixed centers and computes exact subgradient via duals."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-15)
    
    b = np.zeros(NUM_PAIRS + 4 * N)
    k = 0
    for i, j in PAIR_IDX:
        b[k] = dists[i, j]
        k += 1

        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1

    try:

        return np.zeros(N), 0.0, np.zeros_like(centers)
        
    if res.success:

        s = np.sum(radii)

            duals = np.zeros(len(b))
            
        grad = np.zeros_like(centers)
        k = 0

            k += 1

    return np.zeros(N), 0.0, np.zeros_like(centers)

def obj_grad(x_flat):
    """Objective and gradient wrapper for scipy optimizer."""
    c = np.clip(x_flat.reshape(N, 2), 1e-6, 1.0 - 1e-6)
    _, s, g = solve_lp_and_grad(c)
    return -s, -g.flatten()

def optimize_lbfgsb(c0):
    """Runs L-BFGS-B to optimize centers maximizing sum of radii."""
    bounds = [(1e-5, 1.0 - 1e-5)] * (2 * N)
    try:
        res = minimize(obj_grad, c0.flatten(), method='L-BFGS-B', jac=True,
                       bounds=bounds, options={'maxiter': 8000, 'ftol': 1e-14, 'gtol': 1e-12})
        return res.x.reshape(N, 2), -res.fun
    except Exception:
        return c0, 0.0

def generate_starts(rng):

    # 1. Hexagonal lattice patterns
    pats = [[6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [5,5,6,5,5], [4,6,6,6,4], [5,4,6,6,5]]

        starts.append(rng.uniform(0.1, 0.9, (N, 2)))

def coord_refine_step(centers, idx, rng):
    """Optimizes a single circle's position while others are fixed."""
    best_c = centers.copy()
    _, best_s, _ = solve_lp_and_grad(best_c)
    
    # Nelder-Mead on the 2D coordinates of circle idx
    def obj_2d(xy):
        temp = best_c.copy()
        temp[idx] = np.clip(xy, 0.001, 0.999)
        _, s, _ = solve_lp_and_grad(temp)
        return -s

        res = minimize(obj_2d, best_c[idx], method='Nelder-Mead',
                       options={'maxiter': 600, 'xatol': 1e-9, 'fatol': 1e-13})
        if -res.fun > best_s + 1e-9:
            best_c[idx] = np.clip(res.x, 0.001, 0.999)
    except Exception:
        pass
    return best_c

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(200):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):

                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0],

            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
