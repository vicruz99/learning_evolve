

<strategy>
1. **LP-Based Radii & Exact Gradients**: Solve a linear program to find the maximal radii for fixed centers. Extract dual variables to compute the exact subgradient of the objective with respect to center positions. This provides a mathematically sound direction for ascent.
2. **L-BFGS-B Optimization**: Use `scipy.optimize.minimize` with the `L-BFGS-B` method and the exact gradient. This efficiently handles bounds and adapts step sizes, converging faster than manual gradient ascent.
3. **Diverse Initialization**: Generate starting configurations using hexagonal lattices (varying densities/shifts), force-directed repulsion, and corner-biased layouts to cover different regions of the configuration space.
4. **Coordinate-wise Refinement**: Apply Nelder-Mead optimization to each circle's position independently while holding others fixed. This local search excels at navigating the non-smooth landscape created by active constraint switches.
5. **Iterative Perturbation & Basin Hopping**: Systematically perturb the best configuration and re-optimize to escape local minima. Combine with a simulated annealing schedule to accept temporarily worse solutions that lead to better basins.
6. **Joint SLSQP Polish**: Run a constrained non-linear optimizer on both centers and radii jointly to resolve tight numerical tolerances and smooth out residual constraint violations.
7. **Strict Deterministic Repair**: Iteratively shrink overlapping radii and clamp to boundaries to guarantee the validator passes within the `1e-12` tolerance, preserving the maximal valid sum.
8. **No Closures/Lambdas**: All helper functions are defined at the top level, and nested functions are eliminated to strictly comply with the rules.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
NUM_PAIRS = N * (N - 1) // 2
TRIU_I, TRIU_J = np.triu_indices(N, 1)

# Precompute LP matrices
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    b = np.zeros(NUM_PAIRS + 4 * N)
    k = 0
    for i, j in PAIR_IDX:
        b[k] = dists[i, j]
        k += 1
    for i in range(N):
        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1
        
    bounds = [(0.0, u) for u in ub]
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, bounds=bounds, method='highs')
    except Exception:
        return np.zeros(N), 0.0, np.zeros_like(centers)
        
    if res.success:
        try:
            duals = res.marginals.ineqlin
        except AttributeError:
            duals = np.zeros(len(b))
        
        grad = np.zeros_like(centers)
        k = 0
        for i, j in PAIR_IDX:
            mu = duals[k]
            if mu > 1e-9:
                d = dists[i, j]
                if d > 1e-9:
                    vec = (centers[i] - centers[j]) / d
                    grad[i] += mu * vec
                    grad[j] -= mu * vec
            k += 1
            
        b_start = NUM_PAIRS
        for i in range(N):
            grad[i, 0] += duals[b_start + 4*i] - duals[b_start + 4*i + 1]
            grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
            
        return res.x, np.sum(res.x), grad
    return np.zeros(N), 0.0, np.zeros_like(centers)

def obj_grad_lbfgs(v):
    centers = np.clip(v.reshape(N, 2), 1e-6, 1.0 - 1e-6)
    r, s, g = solve_lp_and_grad(centers)
    return -s, -g.flatten()

def force_directed_init(rng):
    c = rng.uniform(0.15, 0.85, (N, 2))
    for _ in range(600):
        diffs = c[:, None, :] - c[None, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        mask = dists < 0.25
        rep = np.zeros_like(dists)
        rep[mask] = 0.02 / (dists[mask] ** 2 + 1e-4)
        
        forces = np.zeros_like(c)
        forces[:, 0] = np.sum(diffs[:, :, 0] * rep, axis=1)
        forces[:, 1] = np.sum(diffs[:, :, 1] * rep, axis=1)
        
        c += forces * 0.005
        c = np.clip(c, 0.05, 0.95)
    return c

def generate_starts(rng):
    starts = []
    pats = [[6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4], 
            [5, 5, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5]]
    for pat in pats:
        for r0 in [0.090, 0.095, 0.100, 0.105]:
            c = []
            y = r0
            for ri, cnt in enumerate(pat):
                sh = r0 if ri % 2 == 1 else 0.0
                x = r0 + sh
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            starts.append(np.clip(np.array(c[:N]), 0.02, 0.98))
            
    for _ in range(10):
        starts.append(force_directed_init(rng))
        
    for _ in range(10):
        c = rng.uniform(0.1, 0.9, (N, 2))
        c[:4] = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        starts.append(np.clip(c, 0.02, 0.98))
        
    return starts

def optimize_lbfgs(c0, maxiter=5000):
    bounds = [(1e-5, 1.0 - 1e-5)] * (2 * N)
    try:
        res = minimize(obj_grad_lbfgs, c0.flatten(), method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-11})
        return res.x.reshape(N, 2), -res.fun
    except Exception:
        return c0, 0.0

def obj_coord(xy, centers, i):
    temp = centers.copy()
    temp[i] = np.clip(xy, 1e-5, 1.0 - 1e-5)
    _, s, _ = solve_lp_and_grad(temp)
    return -s

def coordinate_refine(centers, rng):
    c = centers.copy()
    _, best_s, _ = solve_lp_and_grad(c)
    for _ in range(3):
        for i in range(N):
            try:
                res = minimize(obj_coord, c[i], args=(c, i), method='Nelder-Mead',
                               options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-12})
                if -res.fun > best_s + 1e-10:
                    c[i] = np.clip(res.x, 1e-5, 1.0 - 1e-5)
                    _, best_s, _ = solve_lp_and_grad(c)
            except Exception:
                pass
    return c

def slsqp_obj(v):
    return -np.sum(v[2 * N:])

def slsqp_cons(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    dx = c[TRIU_I, 0] - c[TRIU_J, 0]
    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def slsqp_polish(c0, r0):
    v0 = np.concatenate([c0.flatten(), r0])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
        if np.min(slsqp_cons(res.x)) >= -1e-7:
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:], -res.fun
    except Exception:
        pass
    return c0, r0, 0.0

def repair(centers, radii):
    radii = radii.copy()
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple:
    rng = np.random.default_rng(42)
    best_c = None
    best_r = None
    best_s = -1.0
    
    starts = generate_starts(rng)
    
    # Phase 1: L-BFGS-B from diverse starts
    for c_init in starts:
        c_opt, s_opt = optimize_lbfgs(c_init, maxiter=4000)
        if s_opt > best_s:
            best_s = s_opt
            best_c = c_opt.copy()
            best_r, _, _ = solve_lp_and_grad(best_c)
            
    # Phase 2: Coordinate-wise refinement
    if best_c is not None:
        best_c = coordinate_refine(best_c, rng)
        _, best_s, _ = solve_lp_and_grad(best_c)
        best_r, _, _ = solve_lp_and_grad(best_c)
        
    # Phase 3: Perturbation & Re-optimization
    if best_c is not None:
        for step in range(60):
            noise = 0.012 * (0.92 ** (step // 10))
            c_pert = best_c.copy()
            idx = rng.choice(N, size=N, replace=True)
            c_pert[idx] += rng.normal(0, noise, (N, 2))
            c_pert = np.clip(c_pert, 0.02, 0.98)
            
            c_opt, s_opt = optimize_lbfgs(c_pert, maxiter=3000)
            if s_opt > best_s:
                best_s = s_opt
                best_c = c_opt.copy()
                best_r, _, _ = solve_lp_and_grad(best_c)
                
    # Phase 4: Basin Hopping
    if best_c is not None:
        c_bh = best_c.copy()
        s_bh = best_s
        T = 0.005
        for step in range(300):
            c_try = c_bh + rng.normal(0, 0.004, c_bh.shape)
            c_try = np.clip(c_try, 0.02, 0.98)
            _, s_try, _ = solve_lp_and_grad(c_try)
            
            if s_try > s_bh or np.exp((s_try - s_bh) / max(T, 1e-8)) > rng.random():
                c_bh, s_bh = c_try, s_try
                if s_bh > best_s:
                    best_s = s_bh
                    best_c = c_bh.copy()
                    best_r, _, _ = solve_lp_and_grad(best_c)
            T *= 0.993
            
    # Phase 5: SLSQP Joint Polish
    if best_c is not None:
        c_sl, r_sl, s_sl = slsqp_polish(best_c, best_r)
        if s_sl > best_s:
            best_s = s_sl
            best_c = c_sl
            best_r = r_sl
            
    # Final LP polish
    if best_c is not None:
        r_final, s_final, _ = solve_lp_and_grad(best_c)
        if s_final > best_s:
            best_s = s_final
            best_r = r_final
            
    # Repair
    radii = repair(best_c, best_r)
    return best_c, radii, float(np.sum(radii))
