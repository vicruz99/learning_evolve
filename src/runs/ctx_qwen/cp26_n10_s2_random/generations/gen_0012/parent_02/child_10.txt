import numpy as np
from scipy.optimize import minimize, linprog

N = 26
NUM_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure globally
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    A_LP[NUM_PAIRS + 4 * i, i] = 1.0
    A_LP[NUM_PAIRS + 4 * i + 1, i] = 1.0
    A_LP[NUM_PAIRS + 4 * i + 2, i] = 1.0
    A_LP[NUM_PAIRS + 4 * i + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii given fixed centers and computes exact subgradient via duals."""
    c = np.clip(centers, 1e-7, 1.0 - 1e-7)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    
    b_ub = np.zeros(A_LP.shape[0])
    k = 0
    for i, j in PAIR_IDX:
        b_ub[k] = dists[i, j]
        k += 1
    for i in range(N):
        b_ub[k] = c[i, 0]; k += 1
        b_ub[k] = 1.0 - c[i, 0]; k += 1
        b_ub[k] = c[i, 1]; k += 1
        b_ub[k] = 1.0 - c[i, 1]; k += 1
        
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, 
                      bounds=[(0.0, None)] * N, method='highs')
        if not res.success:
            return np.full(N, 0.04), 1.04, np.zeros_like(c)
    except Exception:
        return np.full(N, 0.04), 1.04, np.zeros_like(c)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    # Extract duals safely
    duals = np.zeros_like(b_ub)
    if hasattr(res, 'marginals') and res.marginals is not None:
        duals[:] = res.marginals.ineqlin
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals[:] = res.ineqlin.marginals
    duals = np.clip(duals, 0.0, 1e6) # Prevent runaway gradients
    
    grad = np.zeros_like(c)
    k = 0
    for i, j in PAIR_IDX:
        mu = duals[k]
        if mu > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
        k += 1
        
    b_start = NUM_PAIRS
    for i in range(N):
        grad[i, 0] += duals[b_start + 4*i] - duals[b_start + 4*i + 1]
        grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
        
    return radii, s_sum, grad

def lbfgs_obj_and_grad(v_flat):
    """Objective and gradient wrapper for L-BFGS-B."""
    centers = v_flat.reshape(N, 2)
    _, val, grad = solve_lp_and_grad(centers)
    # Minimize negative sum
    return -val, -grad.flatten()

def generate_starts(rng):
    """Generates diverse initial configurations from hexagonal patterns and random spreads."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 5, 5, 6],
        [5, 4, 6, 6, 5], [4, 5, 6, 5, 6], [6, 5, 5, 6, 4],
        [7, 6, 6, 7], [6, 7, 7, 6], [8, 6, 6, 6], [9, 8, 9]
    ]
    
    for pat in patterns:
        for r_est in [0.092, 0.098, 0.104, 0.110]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            c = np.array(c[:N])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
            # Rotated versions (15 degrees)
            angle = np.radians(15)
            rot = np.array([[np.cos(angle), -np.sin(angle)], [np.sin(angle), np.cos(angle)]])
            cr = c @ rot.T
            cr = (cr - cr.min(axis=0)) / (cr.max(axis=0) - cr.min(axis=0)) * 0.8 + 0.1
            starts.append(np.clip(cr, 0.05, 0.95))
            
    # Corner/Edge biased
    for _ in range(15):
        c = rng.uniform(0.1, 0.9, (N, 2))
        c[:4] = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
        c += rng.normal(0, 0.015, c.shape)
        starts.append(np.clip(c, 0.05, 0.95))
        
    # Force-directed spreads
    for _ in range(10):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(500):
            f = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    dv = c[i] - c[j]
                    d = np.linalg.norm(dv) + 1e-5
                    if d < 0.22:
                        push = (0.22 - d) * 0.02 / d
                        f[i] += dv * push
                        f[j] -= dv * push
            c += f * 0.05
            c = np.clip(c, 0.08, 0.92)
        starts.append(c)
        
    return starts

def obj_joint(v):
    return -np.sum(v[2 * N:])

def cons_joint(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    i, j = np.triu_indices(N, 1)
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) * 0.5 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_lbfgs = [(0.005, 0.995)] * (2 * N)
    bounds_slsqp = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # Phase 1: Multi-start L-BFGS-B
    inits = generate_starts(rng)
    for c0 in inits:
        try:
            res = minimize(lbfgs_obj_and_grad, c0.flatten(), method='L-BFGS-B', jac=True,
                           bounds=bounds_lbfgs, options={'maxiter': 5000, 'ftol': 1e-14, 'gtol': 1e-12})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt, _ = solve_lp_and_grad(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = inits[0]
        best_r, best_sum, _ = solve_lp_and_grad(best_c)
        
    # Phase 2: Structured Perturbations & Basin Hopping
    for step in range(60):
        c_trial = best_c.copy()
        
        # Random perturbation type
        pert_type = rng.integers(0, 4)
        scale = 0.008 * (0.92 ** (step // 8))
        
        if pert_type == 0:
            # Gaussian noise
            c_trial += rng.normal(0, scale, c_trial.shape)
        elif pert_type == 1:
            # Swap two circles
            idx = rng.choice(N, 2, replace=False)
            c_trial[idx] = c_trial[idx[::-1]]
            c_trial += rng.normal(0, scale * 0.5, c_trial.shape)
        elif pert_type == 2:
            # Push random circle to boundary
            idx = rng.integers(N)
            corner = rng.choice([[0,0], [1,0], [0,1], [1,1]])
            c_trial[idx] = corner * (1.0 - scale) + rng.normal(0, scale * 0.2, 2)
        else:
            # Global reflection/shift
            c_trial[:, 0] = rng.uniform(0.9, 1.1) * c_trial[:, 0]
            c_trial += rng.normal(0, scale * 0.3, c_trial.shape)
            
        c_trial = np.clip(c_trial, 0.01, 0.99)
        
        try:
            res = minimize(lbfgs_obj_and_grad, c_trial.flatten(), method='L-BFGS-B', jac=True,
                           bounds=bounds_lbfgs, options={'maxiter': 2000, 'ftol': 1e-13})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt, _ = solve_lp_and_grad(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            pass
            
    # Phase 3: SLSQP Joint Polish
    v0 = np.concatenate([best_c.flatten(), best_r])
    for _ in range(5):
        try:
            res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_slsqp,
                          constraints={'type': 'ineq', 'fun': cons_joint},
                          options={'maxiter': 8000, 'ftol': 1e-14})
            if np.min(cons_joint(res.x)) >= -1e-8:
                s = np.sum(res.x[2 * N:])
                if s > best_sum:
                    best_sum = s
                    best_c = res.x[:2 * N].reshape(N, 2).copy()
                    best_r = res.x[2 * N:].copy()
                    v0 = res.x.copy()
        except Exception:
            pass
            
    # Final LP verification & strict repair
    lp_r, lp_s, _ = solve_lp_and_grad(best_c)
    if lp_s > best_sum:
        best_r = lp_r
        best_sum = lp_s
        
    final_radii = repair(best_c.copy(), best_r.copy())
    return best_c, final_radii, float(np.sum(final_radii))
