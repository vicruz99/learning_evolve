import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii given fixed centers and computes exact subgradient via duals."""
    c = np.clip(centers, 1e-7, 1.0 - 1e-7)
    n = c.shape[0]
    
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-24)
    np.fill_diagonal(dists, np.inf)
    
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    num_con = num_pairs + num_bound
    
    A_ub = np.zeros((num_con, n))
    b_ub = np.zeros(num_con)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = c[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - c[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = c[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - c[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if not res.success:
            return np.ones(n) * 0.04, 0.0, np.zeros_like(c)
    except Exception:
        return np.ones(n) * 0.04, 0.0, np.zeros_like(c)
        
    radii = res.x
    
    # Extract duals safely
    duals = np.zeros(num_con)
    if hasattr(res, 'marginals') and res.marginals is not None:
        try:
            duals = np.asarray(res.marginals.ineqlin)
        except (AttributeError, TypeError):
            pass
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        try:
            duals = np.asarray(res.ineqlin.marginals)
        except (AttributeError, TypeError):
            pass
            
    grad = np.zeros_like(c)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]

                d = dists[i, j]
                if d > 1e-9:
                    vec = (c[i] - c[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
            idx += 1
            
    b_start = num_pairs
    for i in range(n):
        mu_L = duals[b_start + 4 * i]
        mu_R = duals[b_start + 4 * i + 1]
        mu_B = duals[b_start + 4 * i + 2]
        mu_T = duals[b_start + 4 * i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, np.sum(radii), grad

def lbfgs_wrapper(v_flat):
    """Objective and gradient wrapper for L-BFGS-B."""
    centers = v_flat.reshape(N, 2)
    _, val, grad = solve_lp_and_grad(centers)
    # We minimize -sum(radii), so gradient is -grad
    return -val, -grad.flatten()

def generate_inits(rng):
    """Generates diverse initial configurations."""
    inits = []
    
    # Hexagonal patterns with varying row counts
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 5, 5, 6],
        [5, 4, 6, 6, 5], [4, 5, 6, 5, 6], [6, 5, 5, 6, 4],
        [5, 7, 5, 4, 5], [4, 7, 6, 4, 5], [6, 6, 6, 4, 4]
    ]
    for pat in patterns:
        for r_est in [0.095, 0.100, 0.105, 0.110]:

                x = r_est + shift

                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            c = np.array(c[:N])
            c += rng.normal(0, 0.004, c.shape)
            c = np.clip(c, 0.05, 0.95)
            inits.append(c)

    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        corners = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
        c[:4] = corners
        c += rng.normal(0, 0.025, c.shape)
        c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    # Force-directed random spreads

            forces = np.zeros_like(c)

                    d_vec = c[i] - c[j]
                    d = np.linalg.norm(d_vec)
                    if d < 0.22 and d > 1e-6:
                        push = (0.22 - d) * 0.06
                        forces[i] += d_vec / d * push
                        forces[j] -= d_vec / d * push
            c += forces
            c = np.clip(c, 0.1, 0.9)
        inits.append(c)
        
    return inits

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(200):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def obj_joint(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def cons_joint_sq(v):
    """Constraints for SLSQP: boundaries and non-overlap using squared distances."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    i, j = np.triu_indices(N, 1)
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def slsqp_polish(c0, r0):
    """Joint SLSQP optimization of centers and radii."""
    v0 = np.concatenate([c0.flatten(), r0])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons_joint_sq},
                       options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
        if np.min(cons_joint_sq(res.x)) >= -1e-8:
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:], np.sum(res.x[2 * N:])
    except Exception:
        pass
    return c0, r0, np.sum(r0)
