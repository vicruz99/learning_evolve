import numpy as np
from scipy.optimize import minimize, linprog

N = 26
NUM_PAIRS = N * (N - 1) // 2

# Precompute constant LP constraint matrix structure
A_ub_pre = np.zeros((NUM_PAIRS + 4 * N, N))
pair_indices = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub_pre[idx, i] = 1.0
        A_ub_pre[idx, j] = 1.0
        pair_indices.append((i, j))
        idx += 1
for i in range(N):
    A_ub_pre[NUM_PAIRS + 4 * i, i] = 1.0
    A_ub_pre[NUM_PAIRS + 4 * i + 1, i] = 1.0
    A_ub_pre[NUM_PAIRS + 4 * i + 2, i] = 1.0
    A_ub_pre[NUM_PAIRS + 4 * i + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii given fixed centers and computes exact gradient via duals."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-16)
    
    b = np.zeros(NUM_PAIRS + 4 * N)
    k = 0
    for i, j in pair_indices:
        b[k] = dists[i, j]
        k += 1
    for i in range(N):
        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1
        
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_pre, b_ub=b, 
                      bounds=[(0.0, None)] * N, method='highs')
        if not res.success:
            return np.full(N, 1e-6), 0.0, np.zeros_like(centers)
    except Exception:
        return np.full(N, 1e-6), 0.0, np.zeros_like(centers)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    # Extract duals safely
    duals = np.zeros_like(b)
    try:
        duals = res.marginals.ineqlin
    except AttributeError:
        try:
            duals = res.ineqlin.marginals
        except AttributeError:
            pass
            
    grad = np.zeros_like(centers)
    k = 0
    for i, j in pair_indices:
        mu = duals[k]
        if mu > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
        k += 1
        
    b_start = NUM_PAIRS
    for i in range(N):
        grad[i, 0] += duals[b_start + 4*i] - duals[b_start + 4*i + 1]
        grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
        
    return radii, s_sum, grad

def obj_grad_lbfgsb(x_flat):
    """Objective and gradient wrapper for L-BFGS-B."""
    c = np.clip(x_flat.reshape(N, 2), 0.005, 0.995)
    _, val, grad = solve_lp_and_grad(c)
    # We minimize -sum(radii)
    return -val, -grad.flatten()

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
        [4, 5, 6, 5, 6], [5, 4, 6, 6, 5], [6, 5, 5, 6, 4]
    ]
    
    for pat in patterns:
        for r0 in [0.092, 0.098, 0.105, 0.112]:
            c = []
            y = r0
            for ri, cnt in enumerate(pat):
                sh = r0 if ri % 2 == 1 else 0.0
                x = r0 + sh
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            while len(c) < N:
                c.append(rng.uniform(0.2, 0.8, 2))
            starts.append(np.clip(np.array(c[:N]), 0.02, 0.98))
            
    # Force-directed spreads
    for _ in range(12):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(600):
            f = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    dv = c[i] - c[j]
                    d = np.linalg.norm(dv)
                    if d < 0.20 and d > 1e-4:
                        push = (0.20 - d) * 0.05 / (d + 1e-4)
                        f[i] += dv / d * push
                        f[j] -= dv / d * push
            c += f * 0.01
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    # Corner/Edge biased starts
    for _ in range(8):
        c = rng.uniform(0.1, 0.9, (N, 2))
        corners = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        mid_edges = [[0.5, 0.08], [0.5, 0.92], [0.08, 0.5], [0.92, 0.5]]
        c[:4] = corners
        c[4:8] = mid_edges
        starts.append(c)
        
    return starts

def obj_joint_slsqp(v):
    return -np.sum(v[2 * N:])

def cons_joint_slsqp(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    i, j = np.triu_indices(N, 1)
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def slsqp_polish(centers, radii):
    """Joint SLSQP optimization for final precision."""
    v0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(obj_joint_slsqp, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': cons_joint_slsqp},
                       options={'maxiter': 8000, 'ftol': 1e-15, 'disp': False})
        if np.min(cons_joint_slsqp(res.x)) >= -1e-8:
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:], np.sum(res.x[2 * N:])
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(150):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_lbfgs = [(0.005, 0.995)] * (2 * N)
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = generate_starts(rng)
    
    # Phase 1: Multi-start L-BFGS-B
    for c0 in starts:

                           bounds=bounds_lbfgs, options={'maxiter': 5000, 'ftol': 1e-14})
            val = -res.fun
            if val > best_sum:
                best_sum = val
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = starts[0]
    best_r, _, best_sum = solve_lp_and_grad(best_c)
    
    # Phase 2: Basin-Hopping / Simulated Annealing Perturbation
    curr_c = best_c.copy()
    curr_sum = best_sum
    T = 0.006
    
    for step in range(120):
        noise_scale = 0.008 * (1.0 + 0.3 * np.exp(-step / 20.0))
        c_pert = curr_c + rng.normal(0, noise_scale, curr_c.shape)
        c_pert = np.clip(c_pert, 0.01, 0.99)
        
        # Run local optimization from perturbation
        try:
            res = minimize(obj_grad_lbfgsb, c_pert.flatten(), method='L-BFGS-B', jac=True,
                           bounds=bounds_lbfgs, options={'maxiter': 2000, 'ftol': 1e-13})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt, _ = solve_lp_and_grad(c_opt)
            
            delta = s_opt - curr_sum
            if delta > 0 or rng.random() < np.exp(delta / max(T, 1e-8)):
                curr_c = c_opt.copy()
                curr_sum = s_opt
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
                    
            # Cool down temperature if no improvement for a while, else keep it
            if delta > 1e-5:
                T = min(T * 1.1, 0.015)
            else:
                T *= 0.97
                
        except Exception:
            continue
            
    # Phase 3: SLSQP Joint Polish for final precision
    c_sl, r_sl, s_sl = slsqp_polish(best_c, best_r)
    if s_sl > best_sum:
        best_sum = s_sl
        best_c = c_sl
        best_r = r_sl
        
    # Phase 4: Final LP verification & Strict Repair
    lp_r, lp_s, _ = solve_lp_and_grad(best_c)
    if lp_s > best_sum:
        best_r = lp_r
        best_sum = lp_s
        
    radii = repair(best_c.copy(), best_r.copy())
    return best_c, radii, float(np.sum(radii))
