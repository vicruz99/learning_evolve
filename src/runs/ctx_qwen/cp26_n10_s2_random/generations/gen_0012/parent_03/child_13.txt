import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_lp_structure():
    """Precompute the constant structure of the LP constraint matrix."""
    num_pairs = N * (N - 1) // 2
    num_bound = 4 * N
    A_ub = np.zeros((num_pairs + num_bound, N))
    pair_indices = []
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            pair_indices.append((i, j))
            idx += 1
    for i in range(N):
        for _ in range(4):
            A_ub[idx, i] = 1.0
            idx += 1
    return A_ub, pair_indices

A_LP, PAIR_IDX = get_lp_structure()

def solve_lp_and_grad(centers):
    """
    Solves LP for maximal radii given fixed centers.
    Returns radii, duals, sum_radii, and gradient w.r.t centers.
    """
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    
    b_ub = np.zeros(A_LP.shape[0])
    k = 0
    for i, j in PAIR_IDX:
        b_ub[k] = dists[i, j]
        k += 1
    for i in range(n):
        b_ub[k] = centers[i, 0]; k += 1
        b_ub[k] = 1.0 - centers[i, 0]; k += 1
        b_ub[k] = centers[i, 1]; k += 1
        b_ub[k] = 1.0 - centers[i, 1]; k += 1
        
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.full(n, 1e-4), np.zeros(len(b_ub)), 0.0, np.zeros_like(centers)
        
    radii = res.x
    sum_r = np.sum(radii)
    
    duals = np.zeros(len(b_ub))
    try:
        duals = res.marginals.ineqlin
    except AttributeError:
        try:
            duals = res.ineqlin.marginals
        except AttributeError:
            pass
            
    grad = np.zeros_like(centers)
    k = 0
    for i, j in PAIR_IDX:
        mu = duals[k]
        if mu > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
        k += 1
        
    b_start = len(PAIR_IDX)
    for i in range(n):
        grad[i, 0] += duals[b_start + 4*i] - duals[b_start + 4*i + 1]
        grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
        
    return radii, duals, sum_r, grad

def lbfgs_wrapper(x_flat):
    centers = x_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, _, sum_r, grad = solve_lp_and_grad(centers)
    # Minimize negative sum
    return -sum_r, -grad.flatten()

def generate_starts(rng):
    starts = []
    
    # Hexagonal patterns
    patterns = [
        [5, 5, 5, 5, 6], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], 
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], 
        [4, 5, 6, 5, 6], [5, 4, 6, 6, 5], [6, 5, 5, 6, 4],
        [5, 5, 4, 6, 6], [4, 6, 5, 5, 6], [6, 4, 5, 6, 5]
    ]
    
    for pat in patterns:
        for r_est in [0.092, 0.096, 0.100, 0.104, 0.108]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            while len(c) < N:
                c.append(rng.uniform(0.2, 0.8, 2))
            starts.append(np.array(c[:N]))
            
    # Random starts
    for _ in range(20):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    # Boundary/Corner biased
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        c[0] = [0.12, 0.12]
        c[1] = [0.88, 0.12]
        c[2] = [0.12, 0.88]
        c[3] = [0.88, 0.88]
        starts.append(c)
        
    return starts

def obj_joint(v):
    return -np.sum(v[2 * N:])

def cons_joint(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    i, j = np.triu_indices(N, 1)
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def repair(centers, radii):
    radii = radii.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-12:
                radii[i] = max(mr, 0.0)
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d - 1e-12:
                    shrink = (radii[i] + radii[j] - d) * 0.5 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_lbfgs = [(0.01, 0.99)] * (2 * N)
    bounds_slsqp = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = generate_starts(rng)
    
    # Phase 1: Multi-start L-BFGS-B
    print("Phase 1: Multi-start L-BFGS-B...")
    for c0 in starts:
        try:
            res = minimize(lbfgs_wrapper, c0.flatten(), method='L-BFGS-B', jac=True,
                           bounds=bounds_lbfgs, options={'maxiter': 4000, 'ftol': 1e-14})
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = starts[0]
    best_r, _, best_sum = solve_lp_and_grad(best_c)
    print(f"Phase 1 best sum: {best_sum:.6f}")
    
    # Phase 2: Perturbation & Simulated Annealing
    print("Phase 2: Perturbation & SA...")
    T = 0.006
    for step in range(120):
        # Adaptive noise scale
        noise_scale = 0.005 * (1.0 + 0.4 * np.exp(-step / 15.0))
        
        c_pert = best_c + rng.normal(0, noise_scale, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        
        # Random swap perturbation to break symmetry
        if rng.random() < 0.3:
            idx = rng.choice(N, 2, replace=False)
            c_pert[idx] = c_pert[idx[::-1]]
            
        try:
            res = minimize(lbfgs_wrapper, c_pert.flatten(), method='L-BFGS-B', jac=True,
                           bounds=bounds_lbfgs, options={'maxiter': 2000, 'ftol': 1e-14})
            c_pert = res.x.reshape(N, 2)
            r_pert, _, s_pert = solve_lp_and_grad(c_pert)
            
            delta = s_pert - best_sum
            if delta > 0 or rng.random() < np.exp(delta / max(T, 1e-6)):
                best_sum = s_pert
                best_c = c_pert.copy()
                best_r = r_pert.copy()
                if delta > 0:
                    T = min(T * 1.08, 0.015)
                else:
                    T *= 0.95
        except Exception:
            continue
            
    print(f"Phase 2 best sum: {best_sum:.6f}")
    
    # Phase 3: Joint SLSQP Polish
    print("Phase 3: Joint SLSQP Polish...")
    v0 = np.concatenate([best_c.flatten(), best_r])
    for _ in range(4):
        try:
            res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_slsqp,
                          constraints={'type': 'ineq', 'fun': cons_joint},
                          options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if np.min(cons_joint(res.x)) >= -1e-8:
                s = np.sum(res.x[2 * N:])
                if s > best_sum:
                    best_sum = s
                    best_c = res.x[:2 * N].reshape(N, 2).copy()
                    best_r = res.x[2 * N:].copy()
                    v0 = res.x.copy()
        except Exception:
            pass
            
    # Final LP verification
    lp_r, _, lp_sum = solve_lp_and_grad(best_c)
    if lp_sum > best_sum:
        best_r = lp_r
        best_sum = lp_sum
        
    # Strict numerical repair
    radii = repair(best_c.copy(), best_r.copy())
    final_sum = float(np.sum(radii))
    print(f"Final sum: {final_sum:.6f}")
    
    return best_c, radii, final_sum
