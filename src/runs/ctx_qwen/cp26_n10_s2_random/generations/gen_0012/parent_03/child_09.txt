import numpy as np
from scipy.optimize import minimize, linprog

N = 26
NUM_PAIRS = N * (N - 1) // 2
NUM_CONSTRAINTS = NUM_PAIRS + 4 * N

# Precompute indices for constraints to avoid repeated allocations
PAIR_IDX = [(i, j) for i in range(N) for j in range(i + 1, N)]
PAIR_I = np.array([p[0] for p in PAIR_IDX])
PAIR_J = np.array([p[1] for p in PAIR_IDX])

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii given fixed centers and computes exact gradient via duals."""
    c = centers
    n = c.shape[0]
    
    # Pairwise distances
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    np.fill_diagonal(dists, np.inf)
    
    # Prepare LP matrices
    A_ub = np.zeros((NUM_CONSTRAINTS, n))
    b_ub = np.zeros(NUM_CONSTRAINTS)
    
    # Pair constraints: r_i + r_j <= d_ij
    A_ub[:NUM_PAIRS, PAIR_I] = 1.0
    A_ub[:NUM_PAIRS, PAIR_J] = 1.0
    b_ub[:NUM_PAIRS] = dists[PAIR_I, PAIR_J]
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        offset = NUM_PAIRS + 4 * i
        A_ub[offset, i] = 1.0; b_ub[offset] = c[i, 0]
        A_ub[offset + 1, i] = 1.0; b_ub[offset + 1] = 1.0 - c[i, 0]
        A_ub[offset + 2, i] = 1.0; b_ub[offset + 2] = c[i, 1]
        A_ub[offset + 3, i] = 1.0; b_ub[offset + 3] = 1.0 - c[i, 1]
        
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if not res.success:
            return np.full(n, 1e-5), np.zeros(NUM_CONSTRAINTS), np.zeros_like(c), 0.0
            
        radii = res.x
        sum_r = np.sum(radii)
        
        # Extract duals safely
        duals = np.zeros(NUM_CONSTRAINTS)
        if hasattr(res, 'marginals') and res.marginals is not None:
            duals = res.marginals.ineqlin
        elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
            duals = res.ineqlin.marginals
            
        grad = np.zeros_like(c)
        
        # Pair gradients
        for idx in range(NUM_PAIRS):
            lam = duals[idx]
            if lam > 1e-9:
                i, j = PAIR_IDX[idx]
                d = dists[i, j]
                if d > 1e-9:
                    vec = (c[i] - c[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
                    
        # Boundary gradients
        for i in range(n):
            offset = NUM_PAIRS + 4 * i
            grad[i, 0] += duals[offset] - duals[offset + 1]
            grad[i, 1] += duals[offset + 2] - duals[offset + 3]
            
        return radii, duals, grad, sum_r
    except Exception:
        return np.full(n, 1e-5), np.zeros(NUM_CONSTRAINTS), np.zeros_like(c), 0.0

def obj_and_grad(v):
    """Objective and gradient wrapper for L-BFGS-B."""
    c = v.reshape(N, 2)
    _, _, grad, sum_r = solve_lp_and_grad(c)
    return -sum_r, -grad.flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_c = None

    starts = []

        [4, 5, 6, 5, 6], [5, 4, 6, 6, 5], [6, 5, 5, 6, 4],

        r = v[2 * N:]

                changed = True

    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
