N = 26

# Precompute LP constraint matrix structure globally for speed
NUM_PAIRS = N * (N - 1) // 2
NUM_BOUND = 4 * N
A_ub_const = np.zeros((NUM_PAIRS + NUM_BOUND, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub_const[idx, i] = 1.0
        A_ub_const[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    A_ub_const[NUM_PAIRS + 4*i, i] = 1.0
    A_ub_const[NUM_PAIRS + 4*i + 1, i] = 1.0
    A_ub_const[NUM_PAIRS + 4*i + 2, i] = 1.0
    A_ub_const[NUM_PAIRS + 4*i + 3, i] = 1.0

def solve_lp_and_grad(centers):

    c = np.clip(centers, 1e-8, 1.0 - 1e-8)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))

    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

    idx = 0
    for i, j in PAIR_IDX:
        b_ub[idx] = dists[i, j]
        idx += 1

        b_ub[idx] = c[i, 0]; idx += 1
        b_ub[idx] = 1.0 - c[i, 0]; idx += 1
        b_ub[idx] = c[i, 1]; idx += 1
        b_ub[idx] = 1.0 - c[i, 1]; idx += 1
        
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_const, b_ub=b_ub, 
                      bounds=[(0.0, u) for u in ub], method='highs')
        if not res.success:
            return np.full(N, 1e-4), 0.0, np.zeros_like(c)
    except Exception:
        return np.full(N, 1e-4), 0.0, np.zeros_like(c)
        
    radii = res.x
    sum_r = np.sum(radii)
    
    duals = np.zeros(NUM_PAIRS + NUM_BOUND)
    try:
        duals = res.marginals.ineqlin
    except AttributeError:
        try:
            duals = res.ineqlin.marginals
        except AttributeError:
            pass
            
    grad = np.zeros_like(c)
    idx = 0
    for i, j in PAIR_IDX:
        lam = duals[idx]
        if lam > 1e-7:
            d = dists[i, j]

                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1

        grad[i, 0] += duals[b_start + 4*i] - duals[b_start + 4*i + 1]
        grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
        
    return radii, sum_r, grad

def gradient_ascent(c0, max_iter, rng):

def generate_starts(rng):

    return starts

def obj_joint(v):
    return -np.sum(v[2 * N:])

def cons_joint(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    i, j = np.triu_indices(N, 1)
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)
