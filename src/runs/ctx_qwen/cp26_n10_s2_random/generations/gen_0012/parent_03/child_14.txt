import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii given fixed centers and computes exact gradient via duals."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    
    A_ub = np.zeros((num_pairs + num_bound, n))
    b_ub = np.zeros(num_pairs + num_bound)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if not res.success:
            return np.full(n, 1e-4), np.zeros_like(centers), 0.0
            
        radii = res.x
        sum_r = np.sum(radii)
        
        duals = np.zeros_like(b_ub)
        try:
            duals = np.asarray(res.marginals.ineqlin)
        except AttributeError:
            try:
                duals = np.asarray(res.ineqlin.marginals)
            except AttributeError:
                pass
                
        grad = np.zeros_like(centers)
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                lam = duals[idx]
                if lam > 1e-8:
                    d = dists[i, j]
                    vec = (centers[i] - centers[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
                idx += 1
                
        for i in range(n):
            mu_L = duals[num_pairs + 4 * i]
            mu_R = duals[num_pairs + 4 * i + 1]
            mu_B = duals[num_pairs + 4 * i + 2]
            mu_T = duals[num_pairs + 4 * i + 3]
            grad[i, 0] += mu_L - mu_R
            grad[i, 1] += mu_B - mu_T
            
        return radii, grad, sum_r
    except Exception:
        return np.full(n, 1e-4), np.zeros_like(centers), 0.0

def lbfgs_obj_grad(x_flat):
    """Objective and exact gradient wrapper for L-BFGS-B."""
    centers = x_flat.reshape(N, 2)
    _, grad, sum_r = solve_lp_and_grad(centers)
    return -sum_r, -grad.flatten()

def generate_starts(rng):
    """Generates diverse initial configurations including hex lattices, force-spreads, and random."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 6, 5, 5, 4], [5, 4, 6, 6, 5], [4, 6, 6, 6, 4],
        [7, 6, 6, 7], [6, 7, 7, 6], [8, 8, 8, 2], [4, 5, 6, 5, 6]
    ]
    
    for pat in patterns:
        for r_est in [0.090, 0.095, 0.100, 0.105, 0.110]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            while len(c) < N:
                c.append(rng.uniform(0.2, 0.8, 2))
            starts.append(np.array(c[:N]))
            
    for _ in range(10):
        c = rng.uniform(0.1, 0.9, (N, 2))
        starts.append(c)
        
    for _ in range(8):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(600):
            f = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    dv = c[i] - c[j]
                    d = np.linalg.norm(dv)
                    if d < 0.20 and d > 1e-4:
                        push = (0.20 - d) * 0.05 / (d + 1e-4)
                        f[i] += dv / d * push
                        f[j] -= dv / d * push
            c += f * 0.01
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    return starts

def repair(centers, radii):
    """Deterministic repair to ensure strict validation compliance."""
    radii = radii.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_xy = [(0.001, 0.999)] * (2 * N)
    
    best_c = None
    best_sum = -1.0
    best_r = None
    
    starts = generate_starts(rng)
    
    # Phase 1: L-BFGS-B from diverse starts
    for c0 in starts:
        try:
            res = minimize(lbfgs_obj_grad, c0.flatten(), method='L-BFGS-B', jac=True,
                           bounds=bounds_xy, options={'maxiter': 5000, 'ftol': 1e-14})
            s_val = -res.fun
            if s_val > best_sum:
                best_sum = s_val
                best_c = res.x.reshape(N, 2).copy()
        except Exception:
            pass
            
    if best_c is None:
        best_c = starts[0]
    best_r, _, best_sum = solve_lp_and_grad(best_c)
    
    # Phase 2: Simulated Annealing with symmetry-breaking swaps
    T = 0.006
    for step in range(80):
        noise_scale = 0.005 * (1.0 + 0.5 * np.exp(-step / 15.0))
        c_pert = best_c + rng.normal(0, noise_scale, best_c.shape)
        c_pert = np.clip(c_pert, 0.01, 0.99)
        
        # Randomly swap a pair of circles to break rigid contact graphs
        i, j = rng.choice(N, 2, replace=False)
        c_pert[[i, j]] = c_pert[[j, i]]
        
        try:
            res = minimize(lbfgs_obj_grad, c_pert.flatten(), method='L-BFGS-B', jac=True,
                           bounds=bounds_xy, options={'maxiter': 2500, 'ftol': 1e-13})
            c_pert = res.x.reshape(N, 2)
            r_pert, _, s_pert = solve_lp_and_grad(c_pert)
            
            delta = s_pert - best_sum
            if delta > 0 or rng.random() < np.exp(delta / max(T, 1e-6)):
                best_sum = s_pert
                best_c = c_pert.copy()
                best_r = r_pert.copy()
                if delta > 0:
                    T = min(T * 1.02, 0.015)
                else:
                    T *= 0.97
        except Exception:
            continue
            
    # Phase 3: Targeted subset perturbation (local surgery)
    for _ in range(40):
        num_pert = rng.integers(3, 9)
        idx = rng.choice(N, size=num_pert, replace=False)
        c_trial = best_c.copy()
        c_trial[idx] = rng.uniform(0.1, 0.9, (num_pert, 2))
        
        try:
            res = minimize(lbfgs_obj_grad, c_trial.flatten(), method='L-BFGS-B', jac=True,
                           bounds=bounds_xy, options={'maxiter': 2000, 'ftol': 1e-13})
            c_trial = res.x.reshape(N, 2)
            r_trial, _, s_trial = solve_lp_and_grad(c_trial)
            if s_trial > best_sum:
                best_sum = s_trial
                best_c = c_trial.copy()
                best_r = r_trial.copy()
        except Exception:
            pass
            
    # Final LP verification & strict repair
    lp_r, _, lp_sum = solve_lp_and_grad(best_c)
    if lp_sum > best_sum:
        best_r = lp_r
        best_sum = lp_sum
        
    radii = repair(best_c.copy(), best_r.copy())
    return best_c, radii, float(np.sum(radii))
