import numpy as np
from scipy.optimize import linprog, minimize

N = 26
PAIRS = [(i, j) for i in range(N) for j in range(i + 1, N)]
NUM_PAIRS = len(PAIRS)

# Precompute constant part of LP constraint matrix A_ub
A_ub_pre = np.zeros((NUM_PAIRS + 4 * N, N))
k = 0
for i, j in PAIRS:
    A_ub_pre[k, i] = 1.0
    A_ub_pre[k, j] = 1.0
    k += 1
for i in range(N):
    A_ub_pre[k, i] = 1.0; k += 1
    A_ub_pre[k, i] = 1.0; k += 1
    A_ub_pre[k, i] = 1.0; k += 1
    A_ub_pre[k, i] = 1.0; k += 1

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii given fixed centers and computes exact subgradient."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    
    b_ub = np.zeros(NUM_PAIRS + 4 * N)
    k = 0
    for i, j in PAIRS:
        b_ub[k] = dists[i, j]
        k += 1
    for i in range(N):
        b_ub[k] = centers[i, 0]; k += 1
        b_ub[k] = 1.0 - centers[i, 0]; k += 1
        b_ub[k] = centers[i, 1]; k += 1
        b_ub[k] = 1.0 - centers[i, 1]; k += 1
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub_pre, b_ub=b_ub, 
                      bounds=[(0.0, u) for u in ub], method='highs')
        if not res.success:
            return np.full(n, 1e-4), 0.0, np.zeros_like(centers)
    except Exception:
        return np.full(n, 1e-4), 0.0, np.zeros_like(centers)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    duals = np.zeros(len(b_ub))
    try:
        duals = res.marginals.ineqlin
    except AttributeError:
        try:
            duals = res.ineqlin.marginals
        except AttributeError:
            pass
            
    grad = np.zeros_like(centers)
    k = 0
    for i, j in PAIRS:
        lam = duals[k]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        k += 1
        
    b_start = NUM_PAIRS
    for i in range(N):
        grad[i, 0] += duals[b_start + 4 * i] - duals[b_start + 4 * i + 1]
        grad[i, 1] += duals[b_start + 4 * i + 2] - duals[b_start + 4 * i + 3]
        
    return radii, s_sum, grad

def lp_obj_and_grad(v):
    """Objective and gradient wrapper for L-BFGS-B."""
    c = np.clip(v.reshape(N, 2), 0.005, 0.995)
    _, val, grad = solve_lp_and_grad(c)
    return -val, -grad.flatten()

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 4, 6, 6], [6, 6, 4, 5, 5], [5, 6, 4, 5, 6]
    ]
    
    for pat in patterns:
        for r_est in [0.095, 0.100, 0.105, 0.110]:
            c = []
            y = r_est

                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            while len(c) < N:
                c.append(rng.uniform(0.2, 0.8, 2))
            starts.append(np.array(c[:N]))
            
    # Force-directed spreads
    for _ in range(10):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(600):

                    d_vec = c[i] - c[j]
                    d = np.linalg.norm(d_vec)
                    if d < 0.22 and d > 1e-6:
                        push = (0.22 - d) * 0.04 / (d + 1e-6)
                        forces[i] += d_vec * push
                        forces[j] -= d_vec * push
            c += forces * 0.1
            c = np.clip(c, 0.1, 0.9)
        starts.append(c)
        
    # Corner/Edge biased starts
    for _ in range(8):
        c = rng.uniform(0.15, 0.85, (N, 2))
        c[0] = [0.12, 0.12]
        c[1] = [0.88, 0.12]
        c[2] = [0.12, 0.88]
        c[3] = [0.88, 0.88]
        c += rng.normal(0, 0.01, c.shape)
        c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    return starts

def obj_joint(v):
    return -np.sum(v[2 * N:])

def cons_joint(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    i, j = np.triu_indices(N, 1)
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def repair(centers, radii):
    radii = radii.copy()
    for _ in range(200):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) * 0.5 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = max(mr, 0.0)
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

            s_val = -res.fun

                best_c = c_pert.copy()
                best_r = r_pert.copy()
                if delta > 0:
                    T = min(T * 1.08, 0.015)
                else:
                    T *= 0.95

            continue

    for epoch in range(3):
        for i in range(N):
            c_trial = best_c.copy()

    return best_c, final_radii, final_sum
