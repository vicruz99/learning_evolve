import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_I, TRIU_J = np.triu_indices(N, 1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute LP constraint matrix structure globally for speed
A_LP = np.zeros((NUM_PAIRS + 4 * N, N))
PAIR_IDX = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        PAIR_IDX.append((i, j))
        idx += 1
for i in range(N):
    base = NUM_PAIRS + 4 * i
    A_LP[base, i] = 1.0      # r_i <= x_i
    A_LP[base + 1, i] = 1.0  # r_i <= 1 - x_i
    A_LP[base + 2, i] = 1.0  # r_i <= y_i
    A_LP[base + 3, i] = 1.0  # r_i <= 1 - y_i

def solve_lp_and_grad(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns (radii, sum_radii, gradient_of_sum_radii_wrt_centers).
    """
    c = centers
    n = c.shape[0]
    
    # Distance matrix
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)
    
    # Boundary upper bounds for radii (optional but helps stability)
    ub = np.minimum(
        np.minimum(c[:, 0], 1.0 - c[:, 0]),
        np.minimum(c[:, 1], 1.0 - c[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    # RHS of LP constraints
    b_ub = np.zeros(NUM_PAIRS + 4 * N)
    k = 0

        b_ub[k] = dists[i, j]
        k += 1
    for i in range(n):
        b_ub[k] = c[i, 0]; k += 1
        b_ub[k] = 1.0 - c[i, 0]; k += 1
        b_ub[k] = c[i, 1]; k += 1
        b_ub[k] = 1.0 - c[i, 1]; k += 1

        res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, 
                      bounds=[(0.0, u) for u in ub], method='highs')

            return np.full(n, 1e-4), 0.0, np.zeros_like(c)
    except Exception:
        return np.full(n, 1e-4), 0.0, np.zeros_like(c)

    sum_r = np.sum(radii)

    duals = np.zeros_like(b_ub)
    try:
        duals = res.marginals.ineqlin
    except AttributeError:
        try:
            duals = res.ineqlin.marginals
        except AttributeError:
            pass

    grad = np.zeros_like(c)
    k = 0

        lam = duals[k]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        k += 1
        
    bound_start = NUM_PAIRS

        mu_L = duals[bound_start + 4 * i]
        mu_R = duals[bound_start + 4 * i + 1]
        mu_B = duals[bound_start + 4 * i + 2]
        mu_T = duals[bound_start + 4 * i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, sum_r, grad

def obj_grad_lbfgsb(x_flat):
    """Wrapper for L-BFGS-B: minimizes -sum(radii)"""
    c = np.clip(x_flat.reshape(N, 2), 1e-5, 1.0 - 1e-5)
    _, val, grad = solve_lp_and_grad(c)
    return -val, -grad.flatten()

def optimize_centers_lbfgsb(c0):
    """Runs L-BFGS-B from a given configuration."""
    bounds = [(1e-5, 1.0 - 1e-5)] * (2 * N)
    try:
        res = minimize(obj_grad_lbfgsb, c0.flatten(), method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': 8000, 'ftol': 1e-14, 'gtol': 1e-11})
        return res.x.reshape(N, 2), -res.fun
    except Exception:
        return c0, 0.0

def slsqp_obj(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def slsqp_cons(v):
    """Constraints for joint SLSQP: boundary and non-overlap (squared for smoothness)."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    dx = c[TRIU_I, 0] - c[TRIU_J, 0]
    dy = c[TRIU_I, 1] - c[TRIU_J, 1]
    dr = r[TRIU_I] + r[TRIU_J]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def polish_joint(c0, r0):
    """Refines centers and radii jointly using SLSQP."""
    v0 = np.concatenate([c0.flatten(), r0])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if np.min(slsqp_cons(res.x)) >= -1e-8:
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:], np.sum(res.x[2 * N:])
    except Exception:
        pass
    return c0, r0, np.sum(r0)

def generate_starts(rng):

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(200):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) * 0.5 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
