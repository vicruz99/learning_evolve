from scipy.optimize import minimize, linprog

N = 26

# Precompute constant structure for LP constraints to speed up repeated calls
_NUM_PAIRS = N * (N - 1) // 2
_NUM_BOUND = 4 * N
_A_LP = np.zeros((_NUM_PAIRS + _NUM_BOUND, N))
_PAIR_IDX = []
_idx = 0
for i in range(N):
    for j in range(i + 1, N):
        _A_LP[_idx, i] = 1.0
        _A_LP[_idx, j] = 1.0
        _PAIR_IDX.append((i, j))
        _idx += 1
for i in range(N):
    for _ in range(4):
        _A_LP[_idx, i] = 1.0
        _idx += 1

def solve_lp_and_grad(centers):
    """Solves LP for maximal radii given fixed centers and computes exact gradient via duals."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-16)
    
    b_ub = np.zeros(_A_LP.shape[0])
    k = 0
    for i, j in _PAIR_IDX:
        b_ub[k] = dists[i, j]
        k += 1

        b_ub[k] = centers[i, 0]; k += 1
        b_ub[k] = 1.0 - centers[i, 0]; k += 1
        b_ub[k] = centers[i, 1]; k += 1
        b_ub[k] = 1.0 - centers[i, 1]; k += 1
        
    try:
        res = linprog(-np.ones(N), A_ub=_A_LP, b_ub=b_ub, 
                      bounds=[(0.0, u) for u in ub], method='highs')
        if not res.success:
            return np.full(N, 1e-4), np.zeros_like(centers), 0.0
    except Exception:
        return np.full(N, 1e-4), np.zeros_like(centers), 0.0
        
    radii = res.x
    sum_r = np.sum(radii)
    
    duals = np.zeros(_A_LP.shape[0])
    try:
        duals = np.asarray(res.marginals.ineqlin)
    except AttributeError:
        try:
            duals = np.asarray(res.ineqlin.marginals)
        except AttributeError:
            pass
            
    grad = np.zeros_like(centers)
    k = 0
    for i, j in _PAIR_IDX:
        mu = duals[k]
        if mu > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
        k += 1
        
    b_start = _NUM_PAIRS

        grad[i, 0] += duals[b_start + 4*i] - duals[b_start + 4*i + 1]
        grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
        
    return radii, grad, sum_r

def lbfgs_obj_grad(x_flat):
    """Objective and exact gradient for L-BFGS-B optimization of centers."""
    centers = x_flat.reshape(N, 2)
    # Keep slightly inside bounds to avoid degenerate LP constraints during optimization
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    _, grad, sum_r = solve_lp_and_grad(centers)
    # We minimize -sum_r, so gradient is -grad
    return -sum_r, -grad.flatten()

def obj_joint(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def cons_joint(v):
    """Constraints for joint SLSQP: boundary and non-overlap (squared for smoothness)."""
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = np.concatenate([c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r])
    i, j = np.triu_indices(N, 1)
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con = np.concatenate([con, dx**2 + dy**2 - dr**2])
    return con

def repair(centers, radii):
    """Deterministic repair to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(200):
        changed = False
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) * 0.5 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
