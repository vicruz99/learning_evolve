import numpy as np
from scipy.optimize import minimize, differential_evolution

N = 26

def compute_constraints(vars_flat):
    """Computes boundary and non-overlap constraints as >= 0."""
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    c = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    xi = x[:, None]
    yi = y[:, None]
    ri = r[:, None]
    
    dx = xi - xi.T
    dy = yi - yi.T
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    
    # Upper triangle constraints to reduce size
    mask = np.triu(np.ones((N, N)), k=1).astype(bool)
    c.append(dist[mask] - (ri[mask] + ri.T[mask]))
    
    return np.concatenate(c)

def compute_objective(vars_flat):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def get_bounds():
    """Returns variable bounds for x, y in [0.001, 0.999] and r in [1e-5, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.001, 0.999), (0.001, 0.999), (1e-5, 0.5)])
    return b

def centers_to_vars(centers, radii=None):
    """Converts centers array to flattened optimization variables, estimating radii if needed."""
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    if radii is not None:
        v[2::3] = radii
    else:
        x, y = centers[:, 0], centers[:, 1]
        r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
        dx = x[:, None] - x[None, :]
        dy = y[:, None] - y[None, :]
        dist = np.sqrt(dx**2 + dy**2)
        np.fill_diagonal(dist, np.inf)
        r_pair = 0.5 * np.min(dist, axis=1)
        v[2::3] = np.minimum(r_bound, r_pair) * 0.85
    return v

def run_slsqp(v0, maxiter=2000):
    """Runs SLSQP optimization and returns (optimized_vars, sum_radii) or (None, None)."""
    try:
        res = minimize(compute_objective, v0, method='SLSQP', bounds=get_bounds(),
                       constraints={'type': 'ineq', 'fun': compute_constraints},
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if np.all(compute_constraints(res.x) >= -1e-6):
            return res.x, -res.fun
    except Exception:
        pass
    return None, None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    bounds_c = [(0.001, 0.999)] * (2 * N)

        x, y = c[:, 0], c[:, 1]

        dx = x[:, None] - x[None, :]
        dy = y[:, None] - y[None, :]
        dist = np.sqrt(dx**2 + dy**2)
        np.fill_diagonal(dist, np.inf)
        rp = 0.5 * np.min(dist, axis=1)
        return -np.sum(np.minimum(rb, rp))

    res_de = differential_evolution(obj_centers, bounds_c, seed=42, 
                                    popsize=12, maxiter=500, tol=1e-7, polish=False)
    c_de = res_de.x.reshape(N, 2)
    v_de = centers_to_vars(c_de)
    v_res, val_res = run_slsqp(v_de, maxiter=3000)

    # 2. Hexagonal Grid Initializations

            y += scale * np.sqrt(3)
            row += 1

                radii[i] = max_r
                changed = True

        for i in range(N):
            for j in range(i + 1, N):

                if d < radii[i] + radii[j] - 1e-9:

                    radii[i] -= overlap
                    radii[j] -= overlap
                    changed = True
                    
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
