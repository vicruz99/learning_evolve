2. **Center Optimization via Powell**: I will optimize only the center coordinates using `scipy.optimize.minimize` with the `Powell` method (derivative-free, robust for non-smooth landscapes). The objective is to maximize the sum of radii returned by the LP.
3. **Diverse Multi-Start & Perturbation**: I'll generate multiple structured initial configurations (hexagonal lattices, grids, random) and run local optimization on each. A perturbation phase will escape local minima by jittering the best solution and re-optimizing.
4. **Strict Repair**: After optimization, I'll clamp radii to boundary limits and iteratively shrink overlapping circles proportionally to guarantee strict compliance with the validator's tolerances.

This decoupling of centers and radii leverages the exactness of LP for radii and the robustness of derivative-free optimization for positions, typically yielding significantly higher packing densities for moderate N.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute constraint matrix A_ub for the LP solver
# Constraints: r_i + r_j <= dist(i,j) for i<j
#              r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
A_LP = np.zeros((N*(N-1)//2 + 4*N, N))
idx = 0
for i in range(N):
    for j in range(i+1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        idx += 1
for i in range(N):
    for _ in range(4):
        A_LP[idx, i] = 1.0
        idx += 1

def get_radii_for_centers(centers):
    """Solves LP to find maximum radii for fixed centers."""
    n = centers.shape[0]
    num_pairs = n*(n-1)//2
    b_ub = np.zeros(num_pairs + 4*n)
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            b_ub[idx] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            idx += 1
        for val in [centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1]]:
            b_ub[idx] = val
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-6)

def obj_func_centers(centers_flat):
    """Objective function for center optimization: maximize sum of radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    radii = get_radii_for_centers(centers)
    return -np.sum(radii)

def generate_initial_configs():
    """Generates diverse initial center configurations."""
    configs = []
    rng = np.random.default_rng(42)
    
    # Hexagonal patterns
    patterns = [[5,5,5,5,5,1], [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4], [6,6,5,5,4]]
    for pat in patterns:
        c = []
        y = 0.08
        dy = 0.135
        dx = 0.15
        row_idx = 0
        for count in pat:
            shift = (0.5 if row_idx % 2 == 1 else 0.0) * dx
            x = 0.08 + shift
            for _ in range(count):
                c.append([x, y])
                x += dx
            y += dy
            row_idx += 1
        c = np.array(c[:N])
        c += rng.uniform(-0.01, 0.01, c.shape)
        c = np.clip(c, 0.05, 0.95)
        configs.append(c)
        
    # Random starts
    for _ in range(10):
        c = rng.uniform(0.1, 0.9, (N, 2))
        configs.append(c)
        
    # Grid starts
    for _ in range(5):
        x = np.linspace(0.12, 0.88, 5)
        y = np.linspace(0.12, 0.88, 5)
        cx, cy = np.meshgrid(x, y)
        c = np.column_stack((cx.flatten(), cy.flatten()))
        c = np.vstack([c, [0.5, 0.5]])
        c += rng.uniform(-0.02, 0.02, c.shape)
        c = np.clip(c, 0.05, 0.95)
        configs.append(c)
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    configs = generate_initial_configs()
    bounds = [(0.01, 0.99)] * (2 * N)
    
    best_val = -np.inf
    best_centers = None
    
    # Phase 1: Local optimization on centers using Powell
    for cfg in configs:
        try:
            res = minimize(obj_func_centers, cfg.flatten(), method='Powell', bounds=bounds,
                           options={'maxiter': 1500, 'ftol': 1e-12, 'xtol': 1e-12})
            if -res.fun > best_val:
                best_val = -res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Phase 2: Perturbation & Refinement to escape local minima
    if best_centers is not None:
        for _ in range(30):
            pert = best_centers.copy()
            pert += np.random.normal(0, 0.004, pert.shape)
            pert = np.clip(pert, 0.02, 0.98)
            try:
                res = minimize(obj_func_centers, pert.flatten(), method='Powell', bounds=bounds,
                               options={'maxiter': 800, 'ftol': 1e-12})
                if -res.fun > best_val:
                    best_val = -res.fun
                    best_centers = res.x.reshape(N, 2)
            except Exception:
                pass
                
    # Phase 3: Compute final radii via LP
    radii = get_radii_for_centers(best_centers)
    
    # Phase 4: Strict Repair for Validation Compliance
    # Clamp to boundaries
    for i in range(N):
        x, y = best_centers[i]
        max_r = min(x, 1.0-x, y, 1.0-y)
        if radii[i] > max_r + 1e-9:
            radii[i] = max_r
            
    # Resolve overlaps conservatively
    for _ in range(15):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d + 1e-9) / 2.0
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return best_centers, radii, float(np.sum(radii))
