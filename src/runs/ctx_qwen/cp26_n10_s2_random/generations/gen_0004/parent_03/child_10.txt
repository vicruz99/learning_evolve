1. **Constraint Formulation**: Use squared Euclidean distances for non-overlap constraints: `(x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0`. This avoids square roots and their associated gradient singularities, making the problem smoother for gradient-based optimizers like SLSQP. Boundary constraints are kept linear: `x - r >= 0`, etc.

2. **Diverse Initialization**: Generate multiple initial configurations using hexagonal lattices, perturbed grids, and random placements. Each configuration is passed through a `make_feasible` function that computes strictly feasible initial radii (scaled down by 85% to ensure interior feasibility), which significantly improves SLSQP convergence.

3. **Multi-Start Optimization**: Run SLSQP from each initial point. Keep the best feasible solution found.

4. **Sequential Expansion (Pressure Method)**: Iteratively scale up all radii by a small factor (1.0015), then re-optimize centers and radii to satisfy constraints. This "pressure" technique systematically pushes the packing towards its density limit. If infeasible, radii are backed down and centers perturbed to escape local minima.

5. **Perturbation Refinement & Repair**: After expansion, run random perturbations with local optimization to escape any remaining suboptimal basins. Finally, apply a deterministic repair step to strictly enforce boundary and non-overlap constraints within validator tolerances.

import numpy as np
from scipy.optimize import minimize

N = 26
IDX = np.triu_indices(N, k=1)

def obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constr(v):
    """Computes all boundary and non-overlap constraints as >= 0."""
    x, y, r = v[0::3], v[1::3], v[2::3]
    c = [
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ]
    xi, xj = x[IDX[0]], x[IDX[1]]
    yi, yj = y[IDX[0]], y[IDX[1]]
    ri, rj = r[IDX[0]], r[IDX[1]]
    c.append((xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2)
    return np.concatenate(c)

def get_bounds():
    """Returns variable bounds for x, y in [0,1] and r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def make_feasible(v):
    """Assigns feasible initial radii to centers based on distances."""
    centers = v[:2 * N].reshape(N, 2)
    rs = np.zeros(N)
    for i in range(N):
        d_bound = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        d_min = np.inf
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < d_min:
                    d_min = d
        rs[i] = min(d_bound, d_min / 2.0) * 0.85
    v[2::3] = np.maximum(rs, 1e-6)
    return v

def generate_inits():
    """Generates diverse initial configurations."""
    inits = []
    rng = np.random.default_rng(42)
    
    # Hexagonal lattice variations
    for seed in range(15):
        v = np.zeros(3 * N)
        centers = []
        r0 = 0.095
        y = r0
        row = 0
        while len(centers) < N:
            x = r0 + (row % 2) * r0
            while x + r0 <= 0.95 and len(centers) < N:
                centers.append([x, y])
                x += 2 * r0
            y += np.sqrt(3) * r0
            row += 1
        while len(centers) < N:
            centers.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])
        centers = np.array(centers[:N]) + rng.normal(0, 0.005, (N, 2))
        v[0::3] = centers[:, 0]
        v[1::3] = centers[:, 1]
        v[2::3] = 0.05
        inits.append(v)
        
    # Random dense configurations
    for seed in range(20):
        v = np.zeros(3 * N)
        centers = rng.uniform(0.1, 0.9, (N, 2))
        v[0::3] = centers[:, 0]
        v[1::3] = centers[:, 1]
        v[2::3] = 0.04
        inits.append(v)
        
    # Grid variations
    for scale in [0.9, 1.0, 1.1]:
        v = np.zeros(3 * N)
        centers = []
        for i in range(5):
            for j in range(5):
                centers.append([0.1 + i * 0.2, 0.1 + j * 0.2])
        centers.append([0.5, 0.5])
        centers = np.array(centers[:N]) * scale + (1 - scale) * 0.5
        v[0::3] = centers[:, 0]
        v[1::3] = centers[:, 1]
        v[2::3] = 0.05
        inits.append(v)

    # Ensure all starts are strictly feasible
    for v in inits:
        make_feasible(v)
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constr}
    
    best_v = None
    best_val = -np.inf
    
    inits = generate_inits()
    
    # Phase 1: Multi-start optimization
    for x0 in inits:
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 8000, 'ftol': 1e-12})
            if np.all(constr(res.x) >= -1e-5):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = inits[0]
        
    # Phase 2: Sequential Expansion (Pressure Method)
    curr_v = best_v.copy()
    for step in range(150):
        curr_v[2::3] *= 1.0015
        try:
            res = minimize(obj, curr_v, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 1500, 'ftol': 1e-13})
            if np.all(constr(res.x) >= -1e-5):
                curr_v = res.x.copy()
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = curr_v.copy()
            else:
                # Backtrack and perturb to escape local minimum
                curr_v[2::3] /= 1.0015
                curr_v[0::3] += np.random.normal(0, 0.001, N)
                curr_v[1::3] += np.random.normal(0, 0.001, N)
                curr_v[0::3] = np.clip(curr_v[0::3], 0.01, 0.99)
                curr_v[1::3] = np.clip(curr_v[1::3], 0.01, 0.99)
        except Exception:
            break
            
    # Phase 3: Perturbation Refinement
    for _ in range(50):
        pert_v = best_v.copy()
        pert_v[0::3] += np.random.normal(0, 0.001, N)
        pert_v[1::3] += np.random.normal(0, 0.001, N)
        pert_v[0::3] = np.clip(pert_v[0::3], 0.01, 0.99)
        pert_v[1::3] = np.clip(pert_v[1::3], 0.01, 0.99)
        try:
            res = minimize(obj, pert_v, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-13})
            if np.all(constr(res.x) >= -1e-5):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
        except Exception:
            pass

    centers = best_v.reshape(N, 3)[:, :2]
    radii = best_v.reshape(N, 3)[:, 2].copy()
    
    # Phase 4: Deterministic Repair for Strict Validation Compliance
    for _ in range(30):
        changed = False
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-9:
                radii[i] = max_r
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    overlap = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= overlap
                    radii[j] -= overlap
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
