import numpy as np
from scipy.optimize import linprog, minimize

# Problem dimensions
N = 26
NUM_PAIRS = N * (N - 1) // 2
NUM_BOUNDS = 4 * N
TOTAL_CONSTRAINTS = NUM_PAIRS + NUM_BOUNDS

# Precompute constraint matrix A_ub for LP: A_ub @ r <= b_ub
A_ub = np.zeros((TOTAL_CONSTRAINTS, N))
pair_idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub[pair_idx, i] = 1.0
        A_ub[pair_idx, j] = 1.0
        pair_idx += 1

for i in range(N):
    A_ub[NUM_PAIRS + 4*i, i] = 1.0
    A_ub[NUM_PAIRS + 4*i + 1, i] = 1.0
    A_ub[NUM_PAIRS + 4*i + 2, i] = 1.0
    A_ub[NUM_PAIRS + 4*i + 3, i] = 1.0

# Precompute triu indices for SLSQP overlap constraints
IDX_I, IDX_J = np.triu_indices(N, k=1)

def solve_lp_for_radii(centers):
    """Solves the LP to find optimal radii for fixed centers."""
    b_ub = np.zeros(TOTAL_CONSTRAINTS)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            b_ub[idx] = dists[i, j]
            idx += 1
            
    x = centers[:, 0]
    y = centers[:, 1]
    b_ub[NUM_PAIRS : NUM_PAIRS + N] = x
    b_ub[NUM_PAIRS + N : NUM_PAIRS + 2*N] = 1.0 - x
    b_ub[NUM_PAIRS + 2*N : NUM_PAIRS + 3*N] = y
    b_ub[NUM_PAIRS + 3*N : NUM_PAIRS + 4*N] = 1.0 - y
    
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x, res.ineqlin.marginals
    return None, None

def compute_gradient_from_duals(centers, duals):
    """Computes gradient of sum(radii) w.r.t centers using LP duals."""
    grad = np.zeros((N, 2))
    if duals is None:
        return grad
        
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            lam = duals[idx]
            if lam > 1e-8:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)
                if d > 1e-9:
                    fx = dx / d
                    fy = dy / d
                    grad[i, 0] += lam * fx
                    grad[i, 1] += lam * fy
                    grad[j, 0] -= lam * fx
                    grad[j, 1] -= lam * fy
            idx += 1
            
    for i in range(N):
        mu_x = duals[NUM_PAIRS + 4*i]
        mu_x1 = duals[NUM_PAIRS + 4*i + 1]
        mu_y = duals[NUM_PAIRS + 4*i + 2]
        mu_y1 = duals[NUM_PAIRS + 4*i + 3]
        grad[i, 0] += mu_x - mu_x1
        grad[i, 1] += mu_y - mu_y1
        
    return grad

def get_hex_init(r0, seed):
    """Generates a perturbed hexagonal lattice initialization."""
    rng = np.random.default_rng(seed)
    centers = []
    y = r0
    row = 0
    while len(centers) < N:
        x = r0 + (row % 2) * r0
        while x + r0 <= 1.0 and len(centers) < N:
            centers.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
    c = np.array(centers[:N]) + rng.normal(0, 0.002, (N, 2))
    return np.clip(c, 0.05, 0.95)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Phase 1: Generate diverse initial configurations
    starts = []
    for r0 in [0.08, 0.09, 0.10, 0.105, 0.11]:
        for s in range(6):
            starts.append(get_hex_init(r0, s))
    for s in range(10):
        c = np.random.rand(N, 2) * 0.8 + 0.1
        starts.append(c)
        
    # Phase 2: Gradient Ascent on Centers with LP Radius Update
    for init_c in starts:
        centers = init_c.copy()
        
        for step in range(500):
            radii, duals = solve_lp_for_radii(centers)
            if radii is None:
                break
            curr_sum = np.sum(radii)
            
            grad = compute_gradient_from_duals(centers, duals)
            
            # Adaptive step size with periodic jitter
            alpha = 0.025 * (0.996 ** step)
            if step % 60 == 0 and step > 0:
                alpha *= 3.0
                centers += np.random.normal(0, 0.001, centers.shape)
            
            centers += alpha * grad
            centers = np.clip(centers, 0.01, 0.99)
            
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                
    if best_centers is None:
        best_centers = starts[0]
        radii, _ = solve_lp_for_radii(best_centers)
        best_radii = radii if radii is not None else np.full(N, 0.05)
        best_sum = np.sum(best_radii)
        
    # Phase 3: SLSQP Polish on full (x, y, r) space
    def obj(vars):
        return -np.sum(vars[2::3])
        
    def cons(vars):
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]
        c = []
        c.append(x - r)
        c.append(1.0 - x - r)
        c.append(y - r)
        c.append(1.0 - y - r)
        
        xi, xj = x[IDX_I], x[IDX_J]
        yi, yj = y[IDX_I], y[IDX_J]
        ri, rj = r[IDX_I], r[IDX_J]
        dx = xi - xj
        dy = yi - yj
        dist = np.sqrt(dx**2 + dy**2)
        c.append(dist - (ri + rj))
        return np.concatenate(c)
        
    v0 = np.zeros(3*N)
    v0[0::3] = best_centers[:, 0]
    v0[1::3] = best_centers[:, 1]
    v0[2::3] = best_radii
    
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    for _ in range(5):
        try:
            res = minimize(obj, v0, method='SLSQP', bounds=bounds, constraints=cons_dict,
                           options={'maxiter': 5000, 'ftol': 1e-12})
            if np.all(cons(res.x) >= -1e-6):
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = res.x[:2*N].reshape(N, 2)
                    best_radii = res.x[2*N:]
                v0 = res.x.copy()
        except Exception:
            pass
            
    # Phase 4: Deterministic Repair for Strict Validation Compliance
    for _ in range(20):
        changed = False
        for i in range(N):
            x, y, r = best_centers[i,0], best_centers[i,1], best_radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-9:
                best_radii[i] = max_r
                changed = True
                
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                if d < best_radii[i] + best_radii[j] - 1e-9:
                    shrink = (best_radii[i] + best_radii[j] - d) / 2.0 + 1e-9
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    final_sum = float(np.sum(best_radii))
    return best_centers, best_radii, final_sum
