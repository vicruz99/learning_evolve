import numpy as np
from scipy.optimize import minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)

def compute_constraints(v):
    """Computes boundary and non-overlap constraints as >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    xi, xj = x[IDX_I], x[IDX_J]
    yi, yj = y[IDX_I], y[IDX_J]
    ri, rj = r[IDX_I], r[IDX_J]
    
    dx = xi - xj
    dy = yi - yj
    dr = ri + rj
    
    c_overlap = dx*dx + dy*dy - dr*dr
    return np.concatenate([c, c_overlap])

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def get_bounds():
    """Returns variable bounds for x, y in [0,1] and r in [1e-7, 0.5]."""
    return [(0.0, 1.0)] * (2 * N) + [(1e-7, 0.5)] * N

def make_feasible_init(centers, base_r=0.05):
    """Assigns strictly feasible initial radii to centers."""
    rs = np.full(N, base_r)
    for _ in range(10):
        for i in range(N):
            d_bound = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            d_neighbor = np.inf
            for j in range(N):
                if i != j:
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < d_neighbor: d_neighbor = d
            rs[i] = min(rs[i], d_bound * 0.95, (d_neighbor / 2.0) * 0.95)
    v = np.zeros(3*N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = rs
    return v

def generate_inits():
    """Generates diverse initial configurations."""
    inits = []
    np.random.seed(42)
    
    # Hexagonal patterns with perturbations
    for seed in range(15):
        rng = np.random.default_rng(seed)
        centers = []
        pats = [[5,5,5,5,5,1], [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4]]
        pat = pats[seed % len(pats)]
        y, dy, dx = 0.08, 0.14, 0.16
        row = 0
        for cnt in pat:
            shift = 0.5 * dx if row % 2 else 0.0
            x = 0.08 + shift
            for _ in range(cnt):
                centers.append([x + rng.uniform(-0.005, 0.005), y + rng.uniform(-0.005, 0.005)])
                x += dx
            y += dy
            row += 1
        inits.append(make_feasible_init(np.array(centers[:N])))
        
    # Dense random configurations
    for seed in range(20):
        rng = np.random.default_rng(seed)
        centers = rng.uniform(0.15, 0.85, (N, 2))
        inits.append(make_feasible_init(centers))
        
    # Perturbed square grids
    for seed in range(10):
        rng = np.random.default_rng(seed)
        xs = np.linspace(0.15, 0.85, 5)
        ys = np.linspace(0.15, 0.85, 5)
        cx, cy = np.meshgrid(xs, ys)
        centers = np.column_stack((cx.flatten(), cy.flatten()))[:N]
        centers += rng.normal(0, 0.01, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        inits.append(make_feasible_init(centers))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_v = None
    best_val = -np.inf
    
    # Phase 1: Multi-start optimization
    for x0 in generate_inits():
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,

                # Backtrack if infeasible and perturb to escape
                curr_v[2::3] /= 1.01
                curr_v[0::3] += np.random.normal(0, 0.001, N)

                curr_v[0::3] = np.clip(curr_v[0::3], 0.01, 0.99)

        except Exception:
            pass

    for _ in range(25):

                val = -res.fun

                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Extract final centers and radii
    centers = best_v.reshape(N, 3)[:, :2]
    radii = best_v.reshape(N, 3)[:, 2].copy()

    for _ in range(20):
        changed = False
        for i in range(N):

                    radii[j] -= overlap
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
