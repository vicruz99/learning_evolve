

<strategy>
To maximize the sum of radii for 26 circles in a unit square, pure gradient-based optimizers like SLSQP often get trapped in local minima or struggle with the highly non-convex pairwise distance constraints. A more robust and state-of-the-art approach for circle packing is **Alternating Optimization**:
1. **Fix Centers, Solve LP for Radii**: Given fixed center positions, finding the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints is a Linear Programming problem. This is solved exactly and efficiently using `scipy.optimize.linprog`.
2. **Compute Gradients via Dual Variables**: The LP solution provides dual variables (shadow prices) for each constraint. By the envelope theorem, these duals directly give the subgradient of the optimal objective value with respect to the centers. Constraints that are "tight" (circles touching or at boundaries) exert "forces" pushing centers apart or towards walls to allow larger radii.
3. **Gradient Ascent on Centers**: Update centers using these computed forces with an adaptive decaying step size. This naturally evolves the configuration towards a denser packing.
4. **Multi-Start & Perturbation**: Initialize from diverse high-quality layouts (hexagonal lattices, jittered grids, well-spaced random) and occasionally inject noise to escape local basins.
5. **SLSQP Polish**: Once the alternating method converges, hand the configuration to SLSQP for fine-grained boundary and overlap constraint enforcement, maximizing the remaining margin.
6. **Deterministic Repair**: Final clipping ensures strict compliance with the validator's tolerance thresholds.

This hybrid strategy combines the global search capability of multi-start initialization, the exact constraint handling of LP, the directional intelligence of dual gradients, and the precision of nonlinear programming, typically surpassing pure SLSQP approaches significantly.

```python
import numpy as np
from scipy.optimize import linprog, minimize

N = 26
# Precompute pairwise indices for efficiency
PAIRS_IDX = np.triu_indices(N, k=1)

def get_pairwise_dists(centers):
    """Computes Euclidean distances for all upper-triangle pairs."""
    c = centers
    return np.sqrt(np.sum((c[PAIRS_IDX[0]] - c[PAIRS_IDX[1]])**2, axis=1))

def solve_radii_lp(centers):
    """
    Given fixed centers, solves the LP to find radii that maximize sum(r_i)
    while satisfying all non-overlap and boundary constraints.
    Returns radii and dual variables (marginals).
    """
    x, y = centers[:, 0], centers[:, 1]
    n = N
    dists = get_pairwise_dists(centers)
    
    num_pairs = len(PAIRS_IDX[0])
    num_bounds = 4 * n
    A_ub = np.zeros((num_pairs + num_bounds, n))
    b_ub = np.zeros(num_pairs + num_bounds)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    i_idx, j_idx = PAIRS_IDX
    A_ub[:num_pairs, i_idx] = 1.0
    A_ub[:num_pairs, j_idx] = 1.0
    b_ub[:num_pairs] = dists
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    b_start = num_pairs
    A_ub[b_start:b_start+n, np.arange(n)] = 1.0; b_ub[b_start:b_start+n] = x
    A_ub[b_start+n:b_start+2*n, np.arange(n)] = 1.0; b_ub[b_start+n:b_start+2*n] = 1.0 - x
    A_ub[b_start+2*n:b_start+3*n, np.arange(n)] = 1.0; b_ub[b_start+2*n:b_start+3*n] = y
    A_ub[b_start+3*n:b_start+4*n, np.arange(n)] = 1.0; b_ub[b_start+3*n:b_start+4*n] = 1.0 - y
    
    # Minimize -sum(r_i)
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x, res.ineqlin.marginals
    return None, None

def compute_gradient(centers, duals):
    """
    Computes the gradient of the optimal sum of radii with respect to centers
    using the dual variables from the LP.
    """
    grad = np.zeros((N, 2))
    i_idx, j_idx = PAIRS_IDX
    n_pairs = len(i_idx)
    
    # Forces from pairwise non-overlap constraints
    for k in range(n_pairs):
        lam = duals[k]
        if lam > 1e-8:
            i, j = i_idx[k], j_idx[k]
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.hypot(dx, dy)
            if d > 1e-9:
                fx, fy = lam * dx / d, lam * dy / d
                grad[i, 0] += fx; grad[i, 1] += fy
                grad[j, 0] -= fx; grad[j, 1] -= fy
                
    # Forces from boundary constraints
    b_start = n_pairs
    for i in range(N):
        # mu_L: r_i <= x_i  (pushes x right)
        # mu_R: r_i <= 1-x_i (pushes x left)
        # mu_B: r_i <= y_i  (pushes y up)
        # mu_T: r_i <= 1-y_i (pushes y down)
        grad[i, 0] += duals[b_start + i] - duals[b_start + N + i]
        grad[i, 1] += duals[b_start + 2*N + i] - duals[b_start + 3*N + i]
    return grad

def sllsq_constraints(v):
    """Nonlinear constraints for SLSQP: boundary and non-overlap."""
    x = v[0::3]; y = v[1::3]; r = v[2::3]
    c = []
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    dx = x[PAIRS_IDX[0]] - x[PAIRS_IDX[1]]
    dy = y[PAIRS_IDX[0]] - y[PAIRS_IDX[1]]
    dr = r[PAIRS_IDX[0]] + r[PAIRS_IDX[1]]
    c.append(np.sqrt(dx**2 + dy**2) - dr)
    return np.concatenate(c)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    inits = []
    # 1. Hexagonal grids with varying row patterns
    for s in range(8):
        rng = np.random.default_rng(s)
        pts = []
        pats = [[5,5,5,5,5,1], [6,5,5,5,5], [5,6,5,5,5], [5,5,6,5,5], [5,5,5,6,5]]
        pat = pats[s % len(pats)]
        y = 0.1; dy = 0.14; dx = 0.17; ridx = 0
        for cnt in pat:
            shift = (0.5 if ridx % 2 else 0.0) * dx
            x = 0.1 + shift
            for _ in range(cnt):
                if len(pts) < N: 
                    pts.append([x + rng.uniform(-0.02, 0.02), y + rng.uniform(-0.02, 0.02)])
                x += dx
            y += dy; ridx += 1
        if len(pts) >= N: inits.append(np.array(pts[:N]))
        
    # 2. Well-spaced random points (Poisson-disk like)
    for s in range(8):
        rng = np.random.default_rng(s)
        pts = []
        for _ in range(N):
            while True:
                p = rng.uniform(0.05, 0.95, 2)
                if all(np.hypot(p[0]-c[0], p[1]-c[1]) > 0.12 for c in pts):
                    pts.append(p); break
        inits.append(np.array(pts))
        
    # 3. Perturbed square grids
    for s in range(4):
        rng = np.random.default_rng(s)
        gx = np.linspace(0.1, 0.9, 6); gy = np.linspace(0.1, 0.9, 5)
        pts = []
        for y in gy:
            for x in gx:
                if len(pts) < N: 
                    pts.append([x + rng.uniform(-0.02, 0.02), y + rng.uniform(-0.02, 0.02)])
        inits.append(np.array(pts))
        
    # Phase 1: Alternating LP + Gradient Ascent
    for init_c in inits:
        centers = init_c.copy()
        for step in range(400):
            radii, duals = solve_radii_lp(centers)
            if radii is None: break
            curr_sum = np.sum(radii)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                
            grad = compute_gradient(centers, duals)
            # Adaptive step size decaying over iterations
            step_size = 0.04 / (1.0 + 0.025 * step)
            centers += step_size * grad
            centers = np.clip(centers, 0.001, 0.999)
            
            # Occasional large perturbation to escape local minima
            if step % 60 == 59 and step < 250:
                centers += np.random.normal(0, 0.012, centers.shape)
                centers = np.clip(centers, 0.01, 0.99)
                
    # Phase 2: SLSQP Polish to tighten constraints and squeeze final gains
    if best_centers is not None:
        r0 = best_radii.copy()
        v0 = np.zeros(3 * N)
        v0[0::3] = best_centers[:, 0]
        v0[1::3] = best_centers[:, 1]
        v0[2::3] = r0
        
        bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
        cons = {'type': 'ineq', 'fun': sllsq_constraints}
        
        # Run SLSQP multiple times with slight perturbations for robustness
        for _ in range(3):
            v_start = v0.copy()
            v_start[:2*N] += np.random.normal(0, 0.001, 2*N)
            v_start[:2*N] = np.clip(v_start[:2*N], 0.005, 0.995)
            
            try:
                res = minimize(lambda v: -np.sum(v[2::3]), v_start, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                if np.all(sllsq_constraints(res.x) >= -1e-6):
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = res.x[:2*N].reshape(N, 2)
                        best_radii = res.x[2*N:].copy()
                        v0 = res.x.copy()
            except Exception:
                pass

    # Phase 3: Deterministic Repair for Strict Validation Compliance
    centers = best_centers
    radii = best_radii
    for _ in range(30):
        changed = False
        for i in range(N):
            mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mx + 1e-9:
                radii[i] = mx; changed = True
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    ov = (radii[i] + radii[j] - d + 1e-9) / 2.0
                    radii[i] -= ov; radii[j] -= ov; changed = True
        if not changed: break
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
```
