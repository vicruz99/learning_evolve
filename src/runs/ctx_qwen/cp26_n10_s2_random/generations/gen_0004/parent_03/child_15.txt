import numpy as np
from scipy.optimize import minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """Computes boundary and non-overlap constraints as >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = np.empty(4*N + N*(N-1)//2)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    dx = x[IDX_I] - x[IDX_J]
    dy = y[IDX_I] - y[IDX_J]
    dist = np.hypot(dx, dy)
    c[4*N:] = dist - (r[IDX_I] + r[IDX_J])
    return c

def get_bounds():
    """Returns variable bounds for x, y in [0,1] and r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def make_v_from_centers(centers, scale=0.9):
    """Assigns feasible initial radii to centers based on distances."""
    v = np.zeros(3*N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    
    rs = np.zeros(N)
    for i in range(N):
        d_bound = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        d_nb = np.inf
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < d_nb:
                    d_nb = d
        rs[i] = min(d_bound, d_nb * 0.5) * scale
    v[2::3] = rs
    return v

def generate_hex_init(seed):
    """Generates initial configuration from a perturbed hexagonal lattice."""
    rng = np.random.default_rng(seed)
    centers = []
    patterns = [
        [5,5,5,5,5,1], [6,5,5,5,5], [5,6,5,5,5], [5,5,6,5,5], [5,5,5,6,5], [5,5,5,5,6],
        [4,6,6,5,5], [5,4,6,6,5], [6,4,6,6,4], [5,5,4,6,6], [6,5,4,6,5],
        [5,5,5,5,4,2], [4,5,5,5,5,2], [5,4,5,5,5,2], [5,5,4,5,5,2], [5,5,5,4,5,2]
    ]
    pat = patterns[seed % len(patterns)]
    
    y = 0.08
    dy = 0.135
    dx = 0.175
    row = 0
    for cnt in pat:
        shift = (0.5 if row % 2 == 1 else 0.0) * dx
        x = 0.08 + shift
        for _ in range(cnt):
            centers.append([x + rng.uniform(-0.005, 0.005), y + rng.uniform(-0.005, 0.005)])
            x += dx
        y += dy
        row += 1
        
    centers = np.array(centers[:N])
    return make_v_from_centers(centers, scale=0.95)

def generate_corner_init(seed):
    """Generates initial configuration pushing circles toward corners."""
    rng = np.random.default_rng(seed)
    centers = []
    corners = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]
    centers.extend(corners)
    
    while len(centers) < N:
        pt = rng.uniform(0.2, 0.8, 2)
        ok = True
        for c in centers:
            if np.hypot(pt[0]-c[0], pt[1]-c[1]) < 0.12:
                ok = False
                break
        if ok:
            centers.append(pt)
        if len(centers) >= N:
            break
            
    centers = np.array(centers[:N])
    return make_v_from_centers(centers, scale=0.85)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_val = -np.inf
    
    # Phase 1: Diverse multi-start optimization
    inits = []
    for i in range(60):
        inits.append(generate_hex_init(i))
    for i in range(30):
        inits.append(generate_corner_init(i))
        
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-14})
            if np.min(constraints(res.x)) > -1e-6:
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = inits[0]
        
    # Phase 2: Iterative perturbation & basin hopping
    for _ in range(80):
        pv = best_v.copy()
        pv[0::3] += np.random.normal(0, 0.0015, N)
        pv[1::3] += np.random.normal(0, 0.0015, N)
        pv[0::3] = np.clip(pv[0::3], 0.01, 0.99)
        pv[1::3] = np.clip(pv[1::3], 0.01, 0.99)
        
        try:
            res = minimize(objective, pv, method='SLSQP', bounds=bds, constraints=cons,
                           options={'maxiter': 2500, 'ftol': 1e-14})
            if np.min(constraints(res.x)) > -1e-6:
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Sequential radius expansion
    curr_v = best_v.copy()
    for step in range(50):
        factor = 1.0 + 0.002 * (1.0 - step / 50.0)
        curr_v[2::3] *= factor
        try:
            res = minimize(objective, curr_v, method='SLSQP', bounds=bds, constraints=cons,

                    best_val = val
                    best_v = res.x.copy()
                curr_v = res.x.copy()
            else:
                curr_v[2::3] /= 1.005
                curr_v[0::3] += np.random.normal(0, 0.001, N)
                curr_v[1::3] += np.random.normal(0, 0.001, N)
                curr_v[0::3] = np.clip(curr_v[0::3], 0.01, 0.99)
                curr_v[1::3] = np.clip(curr_v[1::3], 0.01, 0.99)
        except Exception:
            pass

    # Extract final centers and radii
    centers = best_v.reshape(N, 3)[:, :2]
    radii = best_v.reshape(N, 3)[:, 2].copy()

    for _ in range(30):
        changed = False
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-9:
                radii[i] = max_r
                changed = True

            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
