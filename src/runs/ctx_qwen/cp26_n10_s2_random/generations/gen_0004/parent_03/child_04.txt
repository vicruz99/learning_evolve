import numpy as np
from scipy.optimize import minimize

N = 26

def compute_loss(v, P):
    """
    Computes the penalized objective function.
    Minimizes this is equivalent to maximizing sum of radii while penalizing violations.
    """
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    loss = 0.0
    # Boundary penalties: r <= x, r <= 1-x, r <= y, r <= 1-y
    loss += np.sum(np.maximum(0.0, r - x)**2)
    loss += np.sum(np.maximum(0.0, r - (1.0 - x))**2)
    loss += np.sum(np.maximum(0.0, r - y)**2)
    loss += np.sum(np.maximum(0.0, r - (1.0 - y))**2)
    
    # Overlap penalties: r_i + r_j <= distance(i, j)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, 1e9)
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    d_up = dist[mask]
    r_sum_up = (r[:, None] + r[None, :])[mask]
    
    loss += np.sum(np.maximum(0.0, r_sum_up - d_up)**2)
    
    # Objective: maximize sum(r) => minimize -sum(r) + P * penalty
    return -np.sum(r) + P * loss

def get_bounds():
    """Box bounds for x, y in [0, 1] and r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def generate_hex_init(seed):
    """Generates initial configuration from a perturbed hexagonal lattice."""
    rng = np.random.default_rng(seed)
    centers = []
    patterns = [[5,5,5,5,5,1], [6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [4,6,6,6,4], [6,6,5,5,4]]
    pat = patterns[seed % len(patterns)]
    
    y = 0.05
    dy = 0.13
    dx = 0.15
    row_idx = 0
    for count in pat:
        shift = (0.5 if row_idx % 2 == 1 else 0.0) * dx
        x = 0.05 + shift
        for _ in range(count):
            centers.append([x + rng.uniform(-0.01, 0.01), y + rng.uniform(-0.01, 0.01)])
            x += dx
        y += dy
        row_idx += 1
        
    v = np.zeros(3 * N)
    v[0::3] = np.array(centers[:N])[:, 0]
    v[1::3] = np.array(centers[:N])[:, 1]
    # Start with moderately small radii to ensure low initial penalty
    v[2::3] = 0.08 + rng.uniform(0, 0.01, N)
    return v

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = get_bounds()
    best_v = None
    best_sum = -np.inf
    
    # Phase 1: Multi-start continuation optimization
    for seed in range(30):
        v0 = generate_hex_init(seed)
        
        P = 50.0
        curr_v = v0.copy()
        
        # Continuation: gradually increase penalty strength
        for step in range(10):
            try:
                res = minimize(compute_loss, curr_v, args=(P,), method='L-BFGS-B', 
                               bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-13})
                curr_v = res.x.copy()
                P *= 2.0
            except Exception:
                break
                
        r = curr_v[2::3]
        s = np.sum(r)
        # Evaluate residual penalty to check feasibility
        pen = compute_loss(curr_v, P) + s
        
        if pen < 1e-5 and s > best_sum:
            best_sum = s
            best_v = curr_v.copy()
            
        # Phase 2: Basin hopping refinement for top candidates
        if best_v is not None and seed >= 15:
            v0_pert = best_v.copy()
            v0_pert += np.random.normal(0, 0.003, v0_pert.shape)
            v0_pert = np.clip(v0_pert, 0.0, 1.0)
            v0_pert[2::3] = np.clip(v0_pert[2::3], 0.0, 0.5)
            
            P = 200.0
            curr_v = v0_pert
            for _ in range(6):
                try:
                    res = minimize(compute_loss, curr_v, args=(P,), method='L-BFGS-B',
                                   bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-13})
                    curr_v = res.x.copy()
                    P *= 2.0
                except Exception:
                    pass
            r = curr_v[2::3]
            s = np.sum(r)
            pen = compute_loss(curr_v, P) + s
            if pen < 1e-5 and s > best_sum:
                best_sum = s
                best_v = curr_v.copy()

    # Fallback if optimization unexpectedly fails
    if best_v is None:
        best_v = generate_hex_init(0)
        
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3].copy()
    
    # Phase 3: Deterministic Repair for Strict Validation Compliance
    # Resolves any residual floating-point drift from the optimizer
    for _ in range(20):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i,0], centers[i,1], radii[i]
            mr = min(x, 1-x, y, 1-y)
            if r > mr + 1e-10:
                radii[i] = mr
                changed = True
                
        # Resolve overlaps by proportional shrinking
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if radii[i] + radii[j] > d - 1e-10:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
