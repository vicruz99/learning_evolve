import numpy as np
from scipy.optimize import minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)

def fast_radii(centers):
    """Computes feasible radii for fixed centers using min-distance logic."""
    xb = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    yb = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    r_bound = np.minimum(xb, yb)
    
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    r_pair = 0.5 * np.min(dists, axis=1)
    
    return np.minimum(r_bound, r_pair)

def obj_centers(v):
    """Objective for center-only optimization: minimize negative sum of radii."""
    centers = v.reshape(N, 2)
    radii = fast_radii(centers)
    return -np.sum(radii)

def compute_constraints(v):
    """Computes all boundary and non-overlap constraints as >= 0 for SLSQP."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Overlap constraints using precomputed indices
    xi, xj = x[IDX_I], x[IDX_J]
    yi, yj = y[IDX_I], y[IDX_J]
    ri, rj = r[IDX_I], r[IDX_J]
    
    dx = xi - xj
    dy = yi - yj
    dist = np.sqrt(dx*dx + dy*dy)
    c.append(dist - (ri + rj))
    
    return np.concatenate(c)

def obj_full(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def generate_hex_configs():
    """Generates diverse hexagonal lattice initializations."""
    configs = []
    patterns = [
        [5,5,5,5,5,1], [6,5,6,5,4], [5,6,5,6,4], 
        [4,6,6,6,4], [6,6,5,5,4], [5,5,5,6,5], [5,4,6,5,6]
    ]
    for pat in patterns:
        centers = []
        y = 0.08
        dy = 0.14
        dx = 0.16
        row = 0
        for count in pat:
            shift = (0.5 if row % 2 == 1 else 0.0) * dx
            x = 0.08 + shift
            for _ in range(count):
                centers.append([x, y])
                x += dx
            y += dy
            row += 1
        configs.append(np.array(centers[:N]))
    return configs

def repair_solution(centers, radii):
    """Deterministically shrinks radii to guarantee strict validation compliance."""
    radii = radii.copy()
    for _ in range(15):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-9:
                radii[i] = max_r
                changed = True
                
        # Resolve overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    overlap = (radii[i] + radii[j] - d + 1e-9) / 2.0
                    radii[i] -= overlap
                    radii[j] -= overlap
                    changed = True
                    
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    bounds_c = [(0.02, 0.98)] * (2 * N)
    bounds_full = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_v = None
    best_sum = -1.0
    
    # 1. Generate diverse starts
    starts = []
    for cfg in generate_hex_configs():
        for _ in range(3):
            c_pert = cfg + np.random.normal(0, 0.008, cfg.shape)
            starts.append(np.clip(c_pert, 0.03, 0.97))
            
    for _ in range(10):
        rand_c = np.random.uniform(0.15, 0.85, (N, 2))
        starts.append(rand_c)
        
    # 2. Phase A: Powell on centers -> Phase B: SLSQP on (x,y,r)
    for c0 in starts:
        try:
            res_powell = minimize(obj_centers, c0.flatten(), method='Powell', bounds=bounds_c,
                                  options={'maxiter': 3000, 'ftol': 1e-12, 'xtol': 1e-12})
            best_c = res_powell.x.reshape(N, 2)
            
            # Initialize radii for SLSQP
            r0 = fast_radii(best_c) * 0.95
            v0 = np.zeros(3 * N)
            v0[0::3] = best_c[:, 0]
            v0[1::3] = best_c[:, 1]
            v0[2::3] = r0
            
            res_slsqp = minimize(obj_full, v0, method='SLSQP', bounds=bounds_full,
                                 constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12})
                                 
            if np.all(compute_constraints(res_slsqp.x) >= -1e-7):
                curr_sum = np.sum(res_slsqp.x[2::3])

                    best_sum = curr_sum
                    best_v = res_slsqp.x.copy()
        except Exception:
            pass

        v0[0::3] = np.random.uniform(0.1, 0.9, N)

        except Exception:
            pass

        pert_v = best_v.copy()

                    best_sum = val
                    best_v = res.x.copy()
        except Exception:
            pass

    # 5. Extract and Strictly Repair
    centers = best_v.reshape(N, 3)[:, :2]
    radii = best_v.reshape(N, 3)[:, 2].copy()
    radii = repair_solution(centers, radii)
    
    # Final safety clamp
    centers = np.clip(centers, 0.0, 1.0)
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
