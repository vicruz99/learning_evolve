import numpy as np
from scipy.optimize import minimize, linprog

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)

def compute_constraints(vars_flat):
    """Computes all boundary and non-overlap constraints as >= 0."""
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    c = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    xi, xj = x[IDX_I], x[IDX_J]
    yi, yj = y[IDX_I], y[IDX_J]
    ri, rj = r[IDX_I], r[IDX_J]
    
    dx = xi - xj
    dy = yi - yj
    # Add tiny epsilon to prevent NaN gradients when points coincide
    dist = np.sqrt(dx*dx + dy*dy + 1e-15)
    c.append(dist - (ri + rj))
    
    return np.concatenate(c)

def compute_objective(vars_flat):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def get_bounds():
    """Returns variable bounds for x, y in [0,1] and r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def solve_lp_radii(centers):
    """Solves LP to find exact maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    num_constraints = num_pairs + 4 * n
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    # Pairwise constraints: r_i + r_j <= distance(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.hypot(centers[i, 0]-centers[j, 0], centers[i, 1]-centers[j, 1])
            idx += 1
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        vals = [centers[i, 0], 1.0-centers[i, 0], centers[i, 1], 1.0-centers[i, 1]]
        for v in vals:
            A_ub[idx, i] = 1.0
            b_ub[idx] = v
            idx += 1
            
    # Maximize sum(r) <=> Minimize -sum(r)
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return np.zeros(n)

def generate_init(seed, method='random'):

    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    
    if method == 'corner_hex':
        # Prioritize corners for large circles, fill rest with hex grid

        idx = 0
        for c in corners:
            centers[idx] = c
            idx += 1
            
        y = 0.2
        row = 0
        while idx < N:
            shift = 0.1 * (row % 2)
            x = 0.2 + shift
            while x <= 0.8 and idx < N:
                centers[idx] = [x, y]
                x += 0.2
                idx += 1
            y += 0.17
            row += 1
            
    elif method == 'random_dense':
        # Rejection sampling for uniform dense packing
        pts = []
        attempts = 0
        while len(pts) < N and attempts < 10000:
            pt = rng.uniform(0.05, 0.95, 2)
            ok = True
            for c in pts:
                if np.hypot(pt[0]-c[0], pt[1]-c[1]) < 0.13:
                    ok = False
                    break
            if ok:
                pts.append(pt)
            attempts += 1
        centers = np.array(pts)

    centers = np.clip(centers, 0.02, 0.98)
    
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = 0.07  # Safe initial radius
    
    return v

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}

            if np.all(compute_constraints(res.x) >= -1e-7):
                val = -res.fun

                    best_val = val
                    best_v = res.x.copy()
        except Exception:
            pass

        best_v = inits[0]

            pass

    centers = best_v.reshape(N, 3)[:, :2]

        # Clamp to boundaries

                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
