import numpy as np
from scipy.optimize import minimize

N = 26
PAIR_MASK = np.triu(np.ones((N, N), dtype=bool), k=1)

def compute_objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def compute_constraints(v):
    """Computes boundary and non-overlap constraints as >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = []
    # Boundary constraints
    c.extend(x - r)
    c.extend(1.0 - x - r)
    c.extend(y - r)
    c.extend(1.0 - y - r)
    
    # Overlap constraints (squared distance formulation for stability)
    xi = x[:, np.newaxis]
    xj = x[np.newaxis, :]
    yi = y[:, np.newaxis]
    yj = y[np.newaxis, :]
    ri = r[:, np.newaxis]
    rj = r[np.newaxis, :]
    
    dx = xi - xj
    dy = yi - yj
    dr = ri + rj
    
    dist_sq = dx**2 + dy**2
    sum_r_sq = dr**2
    
    c.extend((dist_sq - sum_r_sq)[PAIR_MASK])
    
    return np.concatenate(c)

def get_bounds():
    """Returns variable bounds for x, y in [0,1] and r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def generate_initialization(seed, method='hex'):
    """Generates diverse initial configurations."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    
    if method == 'hex':
        r_guess = 0.09 + rng.uniform(0, 0.025)
        y = r_guess
        row = 0
        idx = 0
        while idx < N:
            offset = (r_guess if row % 2 == 1 else 0.0)
            x = r_guess + offset
            while x + r_guess <= 1.0 and idx < N:
                centers[idx] = [x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)]
                x += 2 * r_guess
                idx += 1
            y += np.sqrt(3) * r_guess
            row += 1
        if idx < N:
            centers[idx:N] = rng.uniform(0.1, 0.9, (N - idx, 2))
            
    elif method == 'grid':
        xs = np.linspace(0.12, 0.88, 5)
        ys = np.linspace(0.12, 0.88, 5)
        cx, cy = np.meshgrid(xs, ys)
        centers[:25] = np.column_stack([cx.flatten(), cy.flatten()])
        centers[25] = [0.5, 0.5]
        centers += rng.normal(0, 0.015, (N, 2))
        centers = np.clip(centers, 0.05, 0.95)
        
    elif method == 'corners':
        corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        centers[:4] = corners
        sub_rng = np.random.default_rng(seed + 1000)
        centers[4:] = sub_rng.uniform(0.15, 0.85, (N - 4, 2))
        
    else:
        centers = rng.uniform(0.15, 0.85, (N, 2))
            
    # Compute feasible initial radii
    rs = np.zeros(N)
    for i in range(N):
        db = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        dists = np.sqrt(np.sum((centers[i] - centers)**2, axis=1))
        dists[i] = np.inf
        dn = np.min(dists)
        rs[i] = min(db, dn / 2.0) * 0.85
        
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = rs
    return v

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_v = None
    best_val = -np.inf
    
    # Phase 1: Multi-start optimization
    inits = []
    for i in range(25):
        inits.append(generate_initialization(i, 'hex'))
    for i in range(15):
        inits.append(generate_initialization(i, 'grid'))
    for i in range(10):
        inits.append(generate_initialization(i, 'corners'))
        
    for x0 in inits:
        try:
            res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-12})
            if np.all(compute_constraints(res.x) >= -1e-5):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = inits[0]
        
    # Phase 2: Sequential Inflation & Re-optimization
    # Iteratively grow radii to force denser packing
    curr_v = best_v.copy()
    for step in range(60):
        factor = 1.0 + 0.0035 * (1.0 - step / 60.0)
        curr_v[2::3] *= factor
        
        try:
            res = minimize(compute_objective, curr_v, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
            if np.all(compute_constraints(res.x) >= -1e-5):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
                curr_v = res.x.copy()
            else:
                # Backtrack radii slightly and perturb centers to escape
                curr_v[2::3] /= 1.005
                curr_v[0::3] += np.random.normal(0, 0.002, N)
                curr_v[1::3] += np.random.normal(0, 0.002, N)
                curr_v[0::3] = np.clip(curr_v[0::3], 0.01, 0.99)
                curr_v[1::3] = np.clip(curr_v[1::3], 0.01, 0.99)
        except Exception:
            pass
            
    # Phase 3: Perturbation Refinement (Basin Hopping)
    for _ in range(40):
        pert_v = best_v.copy()
        pert_v[0::3] += np.random.normal(0, 0.0015, N)
        pert_v[1::3] += np.random.normal(0, 0.0015, N)
        pert_v[0::3] = np.clip(pert_v[0::3], 0.01, 0.99)
        pert_v[1::3] = np.clip(pert_v[1::3], 0.01, 0.99)
        
        try:
            res = minimize(compute_objective, pert_v, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
            if np.all(compute_constraints(res.x) >= -1e-5):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
        except Exception:
            pass

    # Extract final centers and radii
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3].copy()
    
    # Phase 4: Deterministic Repair for Strict Validation Compliance
    for _ in range(25):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-9:
                radii[i] = max_r
                changed = True
                
        # Resolve overlaps by proportional shrinking
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    overlap = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= overlap
                    radii[j] -= overlap
                    changed = True
                    
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
