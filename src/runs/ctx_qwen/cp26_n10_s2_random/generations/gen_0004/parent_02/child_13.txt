To maximize the sum of radii for 26 circles in a unit square, we recognize that for any fixed set of centers, the optimal radii are given by a Linear Program (LP). The constraints are linear: each radius $r_i$ must satisfy boundary limits ($r_i \leq x_i, 1-x_i, y_i, 1-y_i$) and pairwise non-overlap limits ($r_i + r_j \leq \|c_i - c_j\|$). Maximizing $\sum r_i$ under these constraints is a standard LP that can be solved extremely quickly with `scipy.optimize.linprog`.

By decoupling the problem into:
1. Optimizing centers using a derivative-free method (`Powell`), where the objective function evaluates the LP to get the maximum achievable sum of radii.
2. Using multiple diverse initial configurations (hexagonal lattices, perturbed grids, random dense clusters) to escape poor local minima.
3. Applying a perturbation refinement phase after the best solution is found.

We guarantee a valid packing (with strict safety margins) while aggressively pushing the sum of radii towards the theoretical optimum. This LP-guided center optimization is significantly more robust than simultaneous non-linear optimization of positions and radii, which often suffers from non-convex constraint surfaces.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp(centers):
    """
    Solves the LP to find optimal radii for fixed centers.
    Maximizes sum(r_i) subject to boundary and pairwise non-overlap constraints.
    Returns (radii, sum_radii).
    """
    x, y = centers[:, 0], centers[:, 1]
    num_pairs = N * (N - 1) // 2
    num_cons = 4 * N + num_pairs
    
    A_ub = np.zeros((num_cons, N))
    b_ub = np.zeros(num_cons)
    
    idx = 0
    # Boundary constraints: r_i <= dist_to_edge
    for i in range(N):
        A_ub[idx, i] = 1.0; b_ub[idx] = x[i] - 1e-9; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x[i] - 1e-9; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y[i] - 1e-9; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y[i] - 1e-9; idx += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.hypot(x[i]-x[j], y[i]-y[j]) - 1e-9
            idx += 1
            
    c = -np.ones(N)  # Maximize sum(r_i) => Minimize -sum(r_i)
    bounds = [(0.0, None)] * N
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.zeros(N), 0.0

def objective(v):
    """Objective for center optimization: negative max sum of radii."""
    centers = v.reshape(N, 2)
    # Keep centers strictly inside to allow positive radii
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    _, s = solve_lp(centers)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_val = -np.inf
    best_c = None
    
    # Phase 1: Generate diverse initial configurations
    configs = []
    
    # Hexagonal lattices with varying scales
    for scale in [0.95, 1.0, 1.05, 1.1]:
        pts = []
        r = 0.1 * scale
        y = r
        row = 0
        while len(pts) < N + 5:
            x = r if row % 2 == 0 else 2*r
            while x <= 1.0 - r:
                pts.append([x, y])
                x += 2*r
            y += np.sqrt(3)*r
            row += 1
        pts = np.array(pts[:N])
        pts += np.random.normal(0, 0.005, pts.shape)
        configs.append(np.clip(pts, 0.05, 0.95).flatten())
        
    # Random dense configurations
    for _ in range(8):
        pts = np.random.rand(N, 2) * 0.8 + 0.1
        configs.append(pts.flatten())
        
    # Perturbed 5x5 grids (25 points) + 1 random
    for _ in range(4):
        xs = np.linspace(0.1, 0.9, 5)
        ys = np.linspace(0.1, 0.9, 5)
        cx, cy = np.meshgrid(xs, ys)
        grid = np.column_stack((cx.ravel(), cy.ravel()))
        grid = np.vstack([grid, np.random.rand(1, 2)*0.8+0.1])
        grid += np.random.normal(0, 0.02, grid.shape)
        configs.append(np.clip(grid, 0.05, 0.95).flatten())

    # Phase 2: Multi-start Powell optimization
    for v0 in configs:
        try:
            res = minimize(objective, v0, method='Powell',
                           options={'maxiter': 1200, 'ftol': 1e-10, 'xtol': 1e-10})
            val = -res.fun
            if val > best_val:
                best_val = val
                best_c = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Phase 3: Perturbation refinement to escape local minima
    if best_c is not None:
        for _ in range(12):
            v_trial = best_c.copy()
            v_trial += np.random.normal(0, 0.004, v_trial.shape)
            v_trial = np.clip(v_trial, 0.05, 0.95)
            try:
                res = minimize(objective, v_trial.flatten(), method='Powell',
                               options={'maxiter': 800, 'ftol': 1e-10, 'xtol': 1e-10})
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_c = res.x.reshape(N, 2)
            except Exception:
                pass
                
    # Phase 4: Final exact LP solution for the best centers
    radii, final_sum = solve_lp(best_c)
    
    return best_c, radii, float(final_sum)
