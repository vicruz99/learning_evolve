import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_lp_radii(centers):
    """Solves LP to find maximum feasible radii for fixed centers."""
    n = centers.shape[0]
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.hypot(dx, dy)
    
    num_pairs = n * (n - 1) // 2
    total_cons = num_pairs + 4 * n
    A_ub = np.zeros((total_cons, n))
    b_ub = np.zeros(total_cons)
    
    idx = 0
    mask = np.triu_indices(n, k=1)
    for i, j in zip(*mask):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        b_ub[idx] = dists[i, j] - 1e-9
        idx += 1
        
    for i in range(n):
        x, y = centers[i]
        vals = [x, 1.0 - x, y, 1.0 - y]
        for v in vals:
            A_ub[idx, i] = 1.0
            b_ub[idx] = v - 1e-9
            idx += 1
            
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return np.full(n, 0.01)

def objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """Boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    X = x[:, None] - x[None, :]
    Y = y[:, None] - y[None, :]
    R = r[:, None] + r[None, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (X**2 + Y**2)[mask] - R[mask]**2])
    return c

def get_bounds():
    """Variable bounds for x, y, r."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def init_hex(seed, scale=1.0):
    """Hexagonal lattice initialization."""
    rng = np.random.default_rng(seed)
    centers = []
    r0 = 0.10 * scale
    y = r0
    row = 0
    while len(centers) < N:
        x_start = r0 if row % 2 == 0 else 2 * r0
        x = x_start
        while x + r0 <= 1.0 + 1e-9 and len(centers) < N:
            centers.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
    while len(centers) < N:
        centers.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])
        
    centers = np.array(centers[:N])
    centers = np.clip(centers, 0.05, 0.95)
    rs = compute_lp_radii(centers)
    
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = rs
    return v

def init_corner(seed):
    """Corner-focused initialization."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    corners = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
    for i, c in enumerate(corners):
        centers[i] = c
    idx = 4
    r0 = 0.07
    y = 0.3
    row = 0
    while idx < N:
        x_start = 0.25 if row % 2 == 0 else 0.32
        x = x_start
        while x + r0 <= 0.75 and idx < N:
            centers[idx] = [x + rng.uniform(-0.01, 0.01), y + rng.uniform(-0.01, 0.01)]
            idx += 1
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
    centers = np.clip(centers, 0.05, 0.95)
    rs = compute_lp_radii(centers)
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = rs
    return v

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main optimization routine."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    for s in range(20):
        inits.append(init_hex(s, scale=0.95))
        inits.append(init_hex(s, scale=1.05))
    for s in range(10):
        inits.append(init_corner(s))
    for s in range(10):
        rng = np.random.default_rng(s)
        rand_centers = rng.uniform(0.1, 0.9, (N, 2))
        rs = compute_lp_radii(rand_centers)
        v = np.zeros(3*N)
        v[0::3] = rand_centers[:,0]
        v[1::3] = rand_centers[:,1]
        v[2::3] = rs
        inits.append(v)
        
    # Phase 1: Multi-start SLSQP + LP Polish
    for v0 in inits:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-12})
            v_opt = res.x
            centers_opt = np.column_stack((v_opt[0::3], v_opt[1::3]))
            
            # LP Polish: extract exact max radii for the optimized geometry
            radii_lp = compute_lp_radii(centers_opt)
            current_sum = np.sum(radii_lp)
            
            if current_sum > best_sum:
                best_sum = current_sum
                best_v = v_opt.copy()
                best_v[2::3] = radii_lp
        except Exception:
            pass
            
    # Phase 2: Perturbation & Refinement to escape local minima
    if best_v is not None:
        for _ in range(40):
            v_trial = best_v.copy()
            v_trial += np.random.normal(0, 0.0015, v_trial.shape)
            v_trial[0::3] = np.clip(v_trial[0::3], 0.02, 0.98)
            v_trial[1::3] = np.clip(v_trial[1::3], 0.02, 0.98)
            v_trial[2::3] = np.clip(v_trial[2::3], 0.01, 0.45)
            
            try:
                res = minimize(objective, v_trial, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12})
                centers_trial = np.column_stack((res.x[0::3], res.x[1::3]))
                radii_lp = compute_lp_radii(centers_trial)
                current_sum = np.sum(radii_lp)
                
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_v = res.x.copy()
                    best_v[2::3] = radii_lp
            except Exception:
                pass
                
    # Extract final results
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3].copy()
    
    # Phase 3: Deterministic Repair for Strict Validation Compliance
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-9:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
