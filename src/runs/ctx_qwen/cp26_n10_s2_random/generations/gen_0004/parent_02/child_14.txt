import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_radii_from_centers(centers):
    """Computes the largest feasible radii for a given set of centers."""
    x, y = centers[:, 0], centers[:, 1]
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    r_pair = 0.5 * np.min(dist, axis=1)
    
    return np.minimum(r_bound, r_pair)

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints: dist^2 >= (ri + rj)^2
    X = x[:, None] - x[None, :]
    Y = y[:, None] - y[None, :]
    R = r[:, None] + r[None, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (X**2 + Y**2)[mask] - R[mask]**2])
    return c

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def generate_starts():

    starts = []
    rng = np.random.default_rng(42)
    
    # Hexagonal lattices with varying scales
    for scale in [0.90, 0.95, 1.00, 1.05, 1.10]:
        r0 = 0.10 * scale
        centers = []
        y = r0
        row = 0
        while len(centers) < N:

                centers.append([x, y])
                x += 2 * r0
            y += np.sqrt(3) * r0
            row += 1
        centers = np.array(centers[:N])
        centers += rng.normal(0, 0.004, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        starts.append(centers)
        
    # Dense random configurations
    for _ in range(15):
        c = rng.random((N, 2)) * 0.8 + 0.1
        starts.append(c)
        
    # Staggered grid configurations
    for s in [0.18, 0.20, 0.22]:
        c = []
        for i in range(6):
            for j in range(5):
                if len(c) < N:
                    x = 0.05 + j * s + (i % 2) * s/2
                    y = 0.05 + i * s * np.sqrt(3)/2
                    if x <= 0.95 and y <= 0.95:
                        c.append([x, y])
        if len(c) == N:
            starts.append(np.array(c) + rng.normal(0, 0.003, (N, 2)))
            
    return starts

def polish_radii_lp(centers):
    """Uses Linear Programming to maximize sum of radii for fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    num_pairs = N * (N - 1) // 2
    num_cons = num_pairs + 4 * N
    A_ub = np.zeros((num_cons, N))
    b_ub = np.zeros(num_cons)
    
    idx = 0
    # Pairwise distance constraints: r_i + r_j <= dist(i,j) - margin
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.hypot(x[i]-x[j], y[i]-y[j]) - 1e-10
            idx += 1
            
    # Boundary constraints: r_i <= coord - margin
    for i in range(N):
        A_ub[idx, i] = 1.0; b_ub[idx] = x[i] - 1e-10; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x[i] - 1e-10; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y[i] - 1e-10; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y[i] - 1e-10; idx += 1
        
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, 
                      bounds=[(0.0, None)]*N, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return compute_radii_from_centers(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None

        v0[2::3] = r0

                c_vals = constraints(res.x)

                    best_sum = s_val
                    best_v = res.x.copy()
        except Exception:
            pass

            r_temp = compute_radii_from_centers(c_temp)
            v_trial[2::3] = r_temp
            
            try:

                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12})

                    c_vals = constraints(res.x)
                    if np.min(c_vals) >= -1e-6:
                        best_sum = -res.fun
                        best_v = res.x.copy()
            except Exception:

    if best_v is None:

    radii = polish_radii_lp(centers)

            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
