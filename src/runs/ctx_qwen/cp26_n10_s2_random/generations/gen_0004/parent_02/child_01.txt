import numpy as np
from scipy.optimize import minimize

N = 26

def compute_radii_from_centers(centers):
    """Computes maximum valid radii for fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    r_pair = 0.5 * np.min(dist, axis=1)
    return np.minimum(r_bound, r_pair)

def obj_centers(v):
    """Objective for center-only optimization: maximize sum of radii."""
    centers = v.reshape(N, 2)
    radii = compute_radii_from_centers(centers)
    return -np.sum(radii)

def full_objective(v):
    """Objective for full optimization: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def full_constraints(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    X = x[:, None] - x[None, :]
    Y = y[:, None] - y[None, :]
    R = r[:, None] + r[None, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (X**2 + Y**2)[mask] - R[mask]**2])
    return c

def get_bounds_full():
    """Returns variable bounds for x, y, r for each circle."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def create_centers_init(seed, pattern='hex'):
    """Generates initial center configurations."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    if pattern == 'hex':
        r0 = 0.105
        y = r0
        row = 0
        idx = 0
        while idx < N:
            x_start = r0 if row % 2 == 0 else 2 * r0
            x = x_start
            while x + r0 <= 1.0 and idx < N:
                centers[idx] = [x, y]
                idx += 1
                x += 2 * r0
            y += np.sqrt(3) * r0
            row += 1
    elif pattern == 'grid':
        idx = 0
        for r in range(6):
            for c in range(5):
                if idx >= N: break
                centers[idx] = [c * 1.0/4.9 + 0.02, r * 1.0/5.9 + 0.02]
                idx += 1
    elif pattern == 'rand':
        centers = rng.uniform(0.05, 0.95, (N, 2))
        
    centers += rng.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles in a unit square."""
    np.random.seed(42)
    bounds_c = [(0.01, 0.99)] * (2 * N)
    bounds_full = get_bounds_full()
    cons_full = {'type': 'ineq', 'fun': full_constraints}
    
    best_c = None
    best_s = -np.inf
    
    # Stage 1: Powell optimization on centers with diverse starts
    patterns = ['hex', 'grid', 'rand']
    inits = []
    for p in patterns:
        for s in range(10):
            inits.append(create_centers_init(s, p))
            
    for c0 in inits:
        try:
            res = minimize(obj_centers, c0.flatten(), method='Powell', bounds=bounds_c,
                           options={'maxiter': 8000, 'ftol': 1e-14, 'xtol': 1e-14})
            if -res.fun > best_s:
                best_s = -res.fun
                best_c = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Stage 2: Perturbation search to escape local minima
    if best_c is not None:
        for _ in range(40):
            c_trial = best_c.copy()
            c_trial += np.random.normal(0, 0.002, c_trial.shape)
            c_trial = np.clip(c_trial, 0.02, 0.98)
            try:
                res = minimize(obj_centers, c_trial.flatten(), method='Powell', bounds=bounds_c,

                    best_s = -res.fun
                    best_c = res.x.reshape(N, 2)
            except Exception:
                pass
                
    # Stage 3: SLSQP polish on full variables (centers + radii)
    centers = best_c if best_c is not None else create_centers_init(0, 'hex')
    radii = compute_radii_from_centers(centers)
    v0 = np.zeros(3 * N)
    v0[0::3] = centers[:, 0]
    v0[1::3] = centers[:, 1]
    v0[2::3] = radii
    
    try:
        res = minimize(full_objective, v0, method='SLSQP', bounds=bounds_full,
                       constraints=cons_full, options={'maxiter': 10000, 'ftol': 1e-13})
        if np.min(full_constraints(res.x)) >= -1e-7:

        else:
            radii = compute_radii_from_centers(centers)

        radii = compute_radii_from_centers(centers)

    for _ in range(50):

                if d < radii[i] + radii[j] - 1e-9:

                radii[i] = mr

        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
