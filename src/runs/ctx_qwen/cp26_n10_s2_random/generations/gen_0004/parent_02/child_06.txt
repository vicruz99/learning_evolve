import numpy as np
from scipy.optimize import minimize

N = 26

def compute_radii(centers):
    """Computes maximal valid radii for given centers."""
    x = centers[:, 0]
    y = centers[:, 1]
    # Distance to boundaries
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    r_bound = np.maximum(r_bound, 0.0)
    
    # Pairwise distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    
    # Radius limited by half distance to nearest neighbor
    r_pair = 0.5 * np.min(dist, axis=1)
    
    return np.minimum(r_bound, r_pair)

def obj_centers(v):
    """Objective: maximize sum of radii by optimizing centers only."""
    centers = v.reshape(N, 2)
    centers = np.clip(centers, 0.0, 1.0)
    radii = compute_radii(centers)
    return -np.sum(radii)

def obj_full(v):
    """Objective: maximize sum of radii for full variables."""
    return -np.sum(v[2::3])

def generate_hex_lattice(rows, noise=0.0):
    """Generates initial configuration from a hexagonal lattice pattern."""
    centers = []
    s = 0.20
    dy = s * np.sqrt(3) / 2
    y = s / 2
    for i, k in enumerate(rows):
        x_start = s/2 + (i % 2) * s/2
        for j in range(k):
            centers.append([x_start + j * s, y])
        y += dy
        
    c = np.array(centers[:N])
    # Normalize to [0.05, 0.95]
    c = (c - c.min(axis=0)) / (c.max(axis=0) - c.min(axis=0) + 1e-9)
    c = c * 0.9 + 0.05
    if noise > 0:
        rng = np.random.default_rng(np.random.randint(0, 2**31))
        c += rng.normal(0, noise, c.shape)
        c = np.clip(c, 0.05, 0.95)
    return c

def constraints_full(v):
    """Computes boundary and pairwise non-overlap constraints."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise constraints: dist^2 >= (ri + rj)^2
    idx_i, idx_j = np.triu_indices(N, k=1)
    dx = x[idx_i] - x[idx_j]
    dy = y[idx_i] - y[idx_j]
    dr = r[idx_i] + r[idx_j]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # Diverse row patterns summing to 26
    patterns = [
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5],
        [5, 5, 6, 5, 5], [5, 5, 5, 6, 5], [4, 6, 6, 6, 4],
        [6, 4, 6, 5, 5], [5, 4, 6, 5, 6], [4, 5, 5, 6, 6],
        [6, 6, 5, 5, 4], [5, 7, 5, 5, 4], [6, 5, 4, 6, 5],
        [5, 5, 4, 6, 6], [4, 6, 5, 6, 5], [6, 5, 6, 4, 5]
    ]
    
    best_val = np.inf
    best_centers = None
    
    # Phase 1: Multi-start Powell optimization on centers
    for pat in patterns:
        for noise in [0.0, 0.005, 0.01]:
            c0 = generate_hex_lattice(pat, noise=noise).flatten()
            try:
                res = minimize(obj_centers, c0, method='Powell',
                               options={'maxiter': 3000, 'ftol': 1e-13, 'xtol': 1e-13})
                if res.fun < best_val:
                    best_val = res.fun
                    best_centers = res.x.reshape(N, 2)
            except Exception:
                pass
                
    # Random starts
    for _ in range(15):
        c0 = np.random.rand(N, 2) * 0.8 + 0.1
        try:
            res = minimize(obj_centers, c0.flatten(), method='Powell',
                           options={'maxiter': 3000, 'ftol': 1e-13, 'xtol': 1e-13})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    if best_centers is None:
        best_centers = generate_hex_lattice([5, 5, 5, 5, 6])

    # Phase 2: Perturbation refinement
    for _ in range(30):
        c_pert = best_centers + np.random.normal(0, 0.0015, best_centers.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        try:
            res = minimize(obj_centers, c_pert.flatten(), method='Powell',
                           options={'maxiter': 2000, 'ftol': 1e-13})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass

    # Phase 3: SLSQP polish on full variables (centers + radii)
    radii = compute_radii(best_centers)
    v0 = np.zeros(3 * N)
    v0[0::3] = best_centers[:, 0]
    v0[1::3] = best_centers[:, 1]
    v0[2::3] = radii
    
    cons = {'type': 'ineq', 'fun': constraints_full}
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    try:
        res_full = minimize(obj_full, v0, method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': 5000, 'ftol': 1e-14})
        if np.min(constraints_full(res_full.x)) >= -1e-8:
            best_centers = np.column_stack((res_full.x[0::3], res_full.x[1::3]))
            radii = res_full.x[2::3]
    except Exception:
        pass

    # Final strict validation repair
    for _ in range(40):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(best_centers[i, 0], 1.0 - best_centers[i, 0], best_centers[i, 1], 1.0 - best_centers[i, 1])
            if radii[i] > mr - 1e-10:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return best_centers, radii, float(np.sum(radii))
