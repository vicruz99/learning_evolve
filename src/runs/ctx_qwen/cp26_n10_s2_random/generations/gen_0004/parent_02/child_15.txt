import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def compute_max_radii_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum(r_i) subject to non-overlap and boundary constraints.
    Returns radii array.
    """
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    m = num_pairs + num_bound
    
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            dist = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub[idx] = dist
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    bounds_r = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-7)

def objective_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraints_joint(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    X = x[:, None] - x[None, :]
    Y = y[:, None] - y[None, :]
    R = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (X**2 + Y**2)[mask] - R[mask]**2])
    return c

def get_bounds_joint():
    """Returns variable bounds for x, y, r."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def init_hex_grid(seed, scale=1.0, shift=0.0):
    """Generates initial configuration from a perturbed hexagonal lattice."""
    rng = np.random.default_rng(seed)
    centers = []
    r0 = 0.105 * scale
    y = r0
    row = 0
    while len(centers) < N:
        x_start = r0 + shift
        x = x_start
        while x + r0 <= 1.0 + 1e-9 and len(centers) < N:
            centers.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
    while len(centers) < N:
        centers.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])
        
    centers = np.array(centers[:N])
    centers = np.clip(centers, 0.02, 0.98)
    
    rs = compute_max_radii_lp(centers)
    
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = rs
    return v

def init_corner_focus(seed):
    """Generates initial configuration focused on corners and edges."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    # Corners
    centers[0] = [0.1, 0.1]
    centers[1] = [0.9, 0.1]
    centers[2] = [0.1, 0.9]
    centers[3] = [0.9, 0.9]
    # Mid-edges
    centers[4] = [0.5, 0.1]
    centers[5] = [0.5, 0.9]
    centers[6] = [0.1, 0.5]
    centers[7] = [0.9, 0.5]
    
    idx = 8

    y = 0.35
    row = 0
    while idx < N:
        x_start = 0.25 if row % 2 == 0 else 0.35
        x = x_start
        while x + r0 <= 0.75 and idx < N:
            centers[idx] = [x + rng.uniform(-0.01, 0.01), y + rng.uniform(-0.01, 0.01)]
            idx += 1
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
        
    centers = np.clip(centers, 0.02, 0.98)
    rs = compute_max_radii_lp(centers)
    
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = rs
    return v

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles in a unit square."""
    np.random.seed(42)
    bounds = get_bounds_joint()
    cons = {'type': 'ineq', 'fun': constraints_joint}
    
    best_v = None
    best_sum = -np.inf
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    for s in range(20):
        inits.append(init_hex_grid(s, scale=0.95, shift=0.0))
        inits.append(init_hex_grid(s, scale=1.0, shift=0.02))
        inits.append(init_hex_grid(s, scale=1.05, shift=0.0))
    for s in range(10):
        inits.append(init_corner_focus(s))
        
    # Phase 2: Multi-start optimization
    for i, v0 in enumerate(inits):
        try:
            res = minimize(objective_joint, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12})
            s_val = -res.fun
            if s_val > best_sum:
                c_vals = constraints_joint(res.x)
                if np.min(c_vals) >= -1e-7:
                    best_sum = s_val
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Perturbation & Re-optimization (Basin Hopping)
    if best_v is not None:
        perturbation_mag = 0.005
        for _ in range(50):
            v_trial = best_v.copy()
            v_trial += np.random.normal(0, perturbation_mag, v_trial.shape)
            v_trial[0::3] = np.clip(v_trial[0::3], 0.02, 0.98)
            v_trial[1::3] = np.clip(v_trial[1::3], 0.02, 0.98)
            v_trial[2::3] = np.clip(v_trial[2::3], 0.01, 0.45)
            
            try:
                res = minimize(objective_joint, v_trial, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12})
                if -res.fun > best_sum:
                    c_vals = constraints_joint(res.x)
                    if np.min(c_vals) >= -1e-7:
                        best_sum = -res.fun
                        best_v = res.x.copy()
            except Exception:
                pass
            perturbation_mag *= 0.95 # Anneal
            
    # Fallback
    if best_v is None:
        best_v = init_hex_grid(0)
        
    # Extract centers and radii
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    
    # Phase 4: LP-based Radius Recomputation for Guaranteed Validity & Maximality
    radii = compute_max_radii_lp(centers)
    
    # Phase 5: Strict Validation Repair
    for _ in range(50):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-9:
                radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    # Ensure non-negativity
    radii = np.maximum(radii, 0.0)
    
    sum_r = float(np.sum(radii))
    return centers, radii, sum_r
