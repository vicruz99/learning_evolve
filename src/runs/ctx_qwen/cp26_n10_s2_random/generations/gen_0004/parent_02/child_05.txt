    y = v[1::3]
    r = v[2::3]

    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])

    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    num_constraints = num_pairs + 4 * n
    A = np.zeros((num_constraints, n))
    b = np.zeros(num_constraints)
    
    idx = 0

            A[idx, j] = 1.0

            idx += 1
