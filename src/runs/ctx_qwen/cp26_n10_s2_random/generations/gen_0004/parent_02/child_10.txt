import numpy as np
from scipy.optimize import minimize, linprog

def compute_radii_fast(centers):
    """Fast analytical radius estimation for fixed centers."""
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    r_pair = 0.5 * np.min(dist, axis=1)
    
    radii = np.minimum(r_bound, r_pair)
    return np.maximum(radii, 0.0)

def objective_centers(v):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = v.reshape(-1, 2)
    return -np.sum(compute_radii_fast(centers))

def solve_lp_radii(centers):
    """Exact radius optimization via Linear Programming."""
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    A = []
    b = []
    
    # Pairwise constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A.append(row)
            d = np.hypot(x[i] - x[j], y[i] - y[j])
            b.append(d - 1e-9)
            
    # Boundary constraints: r_i <= dist_to_boundary
    for i in range(n):
        for val in [x[i], 1.0 - x[i], y[i], 1.0 - y[i]]:
            row = np.zeros(n)
            row[i] = 1.0
            A.append(row)
            b.append(val - 1e-9)
            
    A = np.array(A)
    b = np.array(b)
    
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-6)
