import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_bounds():
    """Creates variable bounds: x,y in [0,1], r in [1e-6, 0.5]"""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def objective(p):
    """Minimize negative sum of radii to maximize total radius."""
    return -np.sum(p[2::3])

def constraints(p):
    """
    Computes all boundary and non-overlap constraints.
    Returns a 1D array where each element must be >= 0.
    Uses direct Euclidean distance for better gradient conditioning near contact.
    """
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Boundary constraints
    c = np.empty(4 * N)
    c[0:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise distance constraints: dist(i,j) >= r_i + r_j
    xi = x[:, np.newaxis]
    yi = y[:, np.newaxis]
    ri = r[:, np.newaxis]
    
    dx = xi - x[np.newaxis, :]
    dy = yi - y[np.newaxis, :]
    dr = ri + r[np.newaxis, :]
    
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)  # Ignore self-distances
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    overlaps = dist[mask] - dr[mask]
    
    return np.concatenate([c, overlaps])

def solve_lp_radii(centers):
    """
    Given fixed centers, solves an LP to find radii that maximize sum(r_i)
    while strictly satisfying all non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    A = np.zeros((num_pairs + 4 * n, n))
    b = np.zeros(num_pairs + 4 * n)
    
    idx = 0
    # Pairwise constraints: r_i + r_j <= distance(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d - 1e-11  # Slight buffer for numerical safety
            idx += 1
            
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        vals = [centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1]]
        for val in vals:
            A[idx, i] = 1.0
            b[idx] = val - 1e-11
            idx += 1
            
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-6)

def generate_starts():
    """Generates a diverse set of initial center configurations."""
    starts = []
    rng = np.random.RandomState(42)
    
    # 1. Hexagonal lattice patterns with varying densities
    for scale in [0.085, 0.095, 0.105]:
        centers = []
        y = scale
        row = 0
        while len(centers) < N:
            x = scale if row % 2 == 0 else 2 * scale
            while x + scale <= 1.0 and len(centers) < N:
                centers.append([x, y])
                x += 2 * scale
            y += np.sqrt(3) * scale
            row += 1
        centers = np.array(centers[:N])
        for _ in range(4):
            c = centers + rng.normal(0, 0.004, centers.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    # 2. Square grid patterns
    for s in [0.09, 0.10]:
        x = np.linspace(s, 1 - s, 5)
        y = np.linspace(s, 1 - s, 6)  # 5x6=30 points, we take first 26
        cx, cy = np.meshgrid(x, y)
        c = np.column_stack((cx.flatten(), cy.flatten()))[:N]
        for _ in range(4):
            c = c + rng.normal(0, 0.004, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    # 3. Dense random configurations
    for _ in range(15):
        c = rng.rand(N, 2) * 0.8 + 0.1
        starts.append(c)
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_p = None
    best_sum = -np.inf
    
    starts = generate_starts()
    
    # Phase 1: Multi-start optimization with LP polishing
    for c0 in starts:
        p0 = np.zeros(N * 3)
        p0[0::3] = c0[:, 0]
        p0[1::3] = c0[:, 1]
        p0[2::3] = 0.08  # Initial radius guess
        
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13})
            # Check feasibility with tolerance
            if np.min(constraints(res.x)) >= -1e-9:
                centers = res.x[:2 * N].reshape(N, 2)
                # LP polish: find exact max radii for these centers
                radii = solve_lp_radii(centers)
                s = np.sum(radii)
                if s > best_sum:
                    best_sum = s
                    best_p = np.concatenate([centers.flatten(), radii])
        except Exception:
            pass
            
    # Fallback if all optimizations failed
    if best_p is None:
        c = np.random.rand(N, 2) * 0.8 + 0.1
        r = solve_lp_radii(c)
        best_p = np.concatenate([c.flatten(), r])
        best_sum = np.sum(r)
        
    # Phase 2: Iterative perturbation refinement to escape local minima
    rng = np.random.RandomState(123)
    for _ in range(30):
        c_curr = best_p[:2 * N].reshape(N, 2)
        r_curr = best_p[2 * N:]
        
        # Perturb centers slightly
        c_pert = c_curr + rng.normal(0, 0.0015, c_curr.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        
        p0 = np.zeros(N * 3)
        p0[0::3] = c_pert[:, 0]
        p0[1::3] = c_pert[:, 1]
        p0[2::3] = r_curr
        
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13})
            if np.min(constraints(res.x)) >= -1e-9:
                centers = res.x[:2 * N].reshape(N, 2)
                radii = solve_lp_radii(centers)
                s = np.sum(radii)
                if s > best_sum:
                    best_sum = s
                    best_p = np.concatenate([centers.flatten(), radii])
        except Exception:
            pass
            
    # Extract final results
    centers = best_p[:2 * N].reshape(N, 2)
    radii = best_p[2 * N:]
    
    # Phase 3: Final strict safety validation and minimal adjustment
    for _ in range(20):
        valid = True
        # Check boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            if x - r < -1e-12 or x + r > 1.0 + 1e-12 or y - r < -1e-12 or y + r > 1.0 + 1e-12:
                valid = False
                break
        if not valid:
            radii *= 0.99995
            continue
            
        # Check overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            break
        radii *= 0.99995
        
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
