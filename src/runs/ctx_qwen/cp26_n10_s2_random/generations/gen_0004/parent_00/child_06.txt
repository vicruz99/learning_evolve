To maximize the sum of radii for 26 circles in a unit square, we adopt a bi-level optimization strategy:
1. **Inner Level (Radii Optimization)**: For any fixed set of centers, finding the optimal radii is a Linear Programming (LP) problem. We maximize $\sum r_i$ subject to $r_i + r_j \le \|c_i - c_j\|$ and $r_i \le \text{dist}(c_i, \text{boundary})$. This LP is small and can be solved exactly and rapidly using `scipy.optimize.linprog` with the `highs` solver.
2. **Outer Level (Center Optimization)**: We optimize the 52 coordinates of the centers to maximize the optimal sum of radii returned by the LP. The objective function is continuous but non-smooth, making derivative-free methods ideal. We use `scipy.optimize.minimize` with the `Powell` method.
3. **Diverse Initialization & Basin Hopping**: To escape local minima, we generate multiple initial configurations (hexagonal lattices with varying scales, random dense packs, and corner-focused layouts). We run `Powell` from many perturbed starts, followed by an iterative perturbation refinement phase (basin hopping) to push the solution to higher density.
4. **Strict Validation Repair**: Floating-point arithmetic in LP and optimizers can cause microscopic constraint violations. We implement a deterministic shrink-and-check loop to guarantee the solution strictly satisfies the validator's `1e-12` tolerance without sacrificing meaningful radius sum.

This approach decouples the non-convex center placement from the convex radius allocation, providing a robust and highly effective solver for circle packing problems.

```python
import numpy as np
from scipy.optimize import linprog, minimize

N = 26
NUM_PAIRS = N * (N - 1) // 2

# Precompute the structure of the inequality matrix A_ub for the LP
# Constraints: r_i + r_j <= dist(i,j)  and  r_i <= boundary_dist(i)
A_ub = np.zeros((NUM_PAIRS + 4 * N, N))
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        idx += 1

for i in range(N):
    # r_i <= x_i
    A_ub[idx, i] = 1.0; idx += 1
    # r_i <= 1 - x_i
    A_ub[idx, i] = 1.0; idx += 1
    # r_i <= y_i
    A_ub[idx, i] = 1.0; idx += 1
    # r_i <= 1 - y_i
    A_ub[idx, i] = 1.0; idx += 1

def solve_radii_lp(centers):
    """
    Given fixed centers, solves the LP to find radii that maximize sum(r_i)
    subject to non-overlap and boundary constraints.
    """
    b_ub = np.zeros(NUM_PAIRS + 4 * N)
    idx = 0
    
    # Pairwise distances
    for i in range(N):
        for j in range(i + 1, N):
            b_ub[idx] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            idx += 1
            
    # Boundary distances
    for i in range(N):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    # Maximize sum(r_i) => Minimize -sum(r_i)
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * N, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.zeros(N), 0.0

def objective(v):
    """Objective function for center optimization."""
    centers = np.clip(v.reshape(N, 2), 1e-6, 1.0 - 1e-6)
    _, sum_r = solve_radii_lp(centers)
    return -sum_r

def generate_configs():
    """Generates diverse initial configurations for centers."""
    configs = []
    np.random.seed(42)
    
    # 1. Hexagonal lattices with varying scales
    for scale in [0.085, 0.090, 0.095, 0.100, 0.105]:
        pts = []
        y = scale
        row = 0
        while len(pts) < N:
            x = scale if row % 2 == 0 else 2 * scale
            while x <= 1.0 - scale and len(pts) < N:
                pts.append([x, y])
                x += 2 * scale
            y += np.sqrt(3) * scale
            row += 1
        configs.append(np.array(pts[:N]))
        
    # 2. Random uniform configurations
    for _ in range(30):
        configs.append(np.random.rand(N, 2) * 0.8 + 0.1)
        
    # 3. Corner-pushed configurations (often optimal for packings)
    for _ in range(20):
        c = np.random.rand(N, 2) * 0.8 + 0.1
        idx = np.random.choice(N, 4, replace=False)
        corners = [[0.05, 0.05], [0.95, 0.05], [0.05, 0.95], [0.95, 0.95]]
        for i, cor in zip(idx, corners):
            c[i] = cor
        configs.append(c)
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    configs = generate_configs()
    best_sum = -1.0
    best_centers = None
    
    # Phase 1: Multi-start local optimization with Powell
    for cfg in configs:
        for _ in range(3):
            pert = cfg + np.random.normal(0, 0.005, cfg.shape)
            pert = np.clip(pert, 0.0, 1.0)
            try:
                res = minimize(objective, pert.flatten(), method='Powell', 
                               options={'maxiter': 2000, 'ftol': 1e-12, 'xtol': 1e-12})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_centers = res.x.reshape(N, 2)
            except Exception:
                pass
                
    # Phase 2: Iterative Basin Hopping / Perturbation Refinement
    if best_centers is not None:
        for _ in range(50):
            # Perturb best solution
            pert = best_centers + np.random.normal(0, 0.002, best_centers.shape)
            pert = np.clip(pert, 0.0, 1.0)
            try:
                res = minimize(objective, pert.flatten(), method='Powell',
                               options={'maxiter': 1500, 'ftol': 1e-12})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_centers = res.x.reshape(N, 2)
            except Exception:
                pass
                
    # Compute final optimal radii for the best centers
    radii, final_sum = solve_radii_lp(best_centers)
    
    # Phase 3: Strict Validation Repair
    # Ensure the solution strictly satisfies the validator's tolerance (1e-12)
    for _ in range(30):
        valid = True
        # Check boundaries
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], radii[i]
            if x - r < -1e-12 or x + r > 1.0 + 1e-12 or y - r < -1e-12 or y + r > 1.0 + 1e-12:
                valid = False
                break
        if not valid:
            break
            
        # Check overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
                
        if valid:
            break
        # Slightly shrink all radii to resolve numerical violations
        radii *= 0.99995
        
    radii = np.maximum(radii, 0.0)
    return best_centers, radii, float(np.sum(radii))
```
