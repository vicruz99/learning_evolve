1. **LP-Driven Center Optimization**: For any fixed set of centers, the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints can be found exactly and efficiently via Linear Programming (LP). We formulate this LP once with a precomputed constraint matrix $A_{ub}$, updating only the bounds vector $b_{ub}$ which depends on inter-center distances and boundary distances.
2. **Global Search with Differential Evolution**: We use `scipy.optimize.differential_evolution` to search the 52-dimensional center space. The objective function evaluates the LP to get the maximum possible sum of radii for those centers. DE is robust for non-smooth objectives and explores the global landscape effectively.
3. **Local Refinement with SLSQP**: Once DE finds a promising center configuration, we extract the LP-optimal radii and initialize a full constrained nonlinear optimization (centers + radii) using `SLSQP`. This polishes the solution to the exact KKT conditions.
4. **Perturbation & Repair**: We run multiple perturbed local optimizations to escape residual local minima, followed by a deterministic repair step that iteratively resolves any numerical overlaps or boundary violations, ensuring strict compliance with the validator's tolerance.

This strategy leverages the mathematical exactness of LP for radii, the global exploration of DE for centers, and the precision of SLSQP for final polishing, systematically pushing the sum of radii higher.

import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

N = 26
NUM_PAIRS = N * (N - 1) // 2
NUM_BOUND = 4 * N
TOTAL_CONS = NUM_PAIRS + NUM_BOUND
UPPER_TRIU = np.triu_indices(N, k=1)

# Precompute A_ub for the LP: r_i + r_j <= d_ij and boundary constraints r_i <= boundary_dist
A_ub = np.zeros((TOTAL_CONS, N))
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        idx += 1
for i in range(N):
    A_ub[idx, i] = 1.0; idx += 1  # x
    A_ub[idx, i] = 1.0; idx += 1  # 1-x
    A_ub[idx, i] = 1.0; idx += 1  # y
    A_ub[idx, i] = 1.0; idx += 1  # 1-y

c_obj = -np.ones(N)
bounds_r = [(0, None)] * N

def solve_lp_for_radii(centers):
    """Solves LP to find optimal radii for fixed centers maximizing sum(r_i)."""
    b_ub = np.zeros(TOTAL_CONS)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub[:NUM_PAIRS] = dists[UPPER_TRIU]
    
    # Boundary distances
    x = centers[:, 0]
    y = centers[:, 1]
    idx = NUM_PAIRS
    b_ub[idx:idx+N] = x; idx += N
    b_ub[idx:idx+N] = 1.0 - x; idx += N
    b_ub[idx:idx+N] = y; idx += N
    b_ub[idx:idx+N] = 1.0 - y
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def objective_centers(centers_flat):
    """Objective for DE: minimize negative sum of optimal radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    neg_sum, _ = solve_lp_for_radii(centers)
    return neg_sum

def full_objective(p):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(p[2::3])

def full_constraints(p):
    """Constraints for SLSQP: boundary and non-overlap."""
    x, y, r = p[0::3], p[1::3], p[2::3]
    c = np.empty(4 * N + NUM_PAIRS)
    idx = 0
    c[idx:idx+N] = x - r; idx += N
    c[idx:idx+N] = 1.0 - x - r; idx += N
    c[idx:idx+N] = y - r; idx += N
    c[idx:idx+N] = 1.0 - y - r; idx += N
    
    i_idx, j_idx = UPPER_TRIU
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    c[idx:] = dx*dx + dy*dy - dr*dr
    return c

def repair(centers, radii):
    """Iteratively shrinks radii to resolve overlaps and clamp to boundaries."""
    for _ in range(30):
        changed = False
        for i in range(N):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > max_r - 1e-9:
                radii[i] = max_r
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # Phase 1: Global search for centers using Differential Evolution
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    
    try:
        res_de = differential_evolution(objective_centers, bounds_centers, popsize=15, 
                                        maxiter=600, tol=1e-7, atol=1e-7, seed=42, 
                                        workers=1, mutation=(0.5, 1.5), recombination=0.9)
        best_centers = res_de.x.reshape(N, 2)
        best_sum, best_radii = solve_lp_for_radii(best_centers)
    except Exception:
        best_centers = np.random.rand(N, 2)
        best_sum, best_radii = solve_lp_for_radii(best_centers)
        
    # Phase 2: Local refinement using SLSQP on full variables
    p0 = np.zeros(N * 3)
    p0[0::3] = best_centers[:, 0]
    p0[1::3] = best_centers[:, 1]
    p0[2::3] = best_radii
    
    full_bounds = [(0.0, 1.0)] * (2 * N) + [(1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': full_constraints}
    
    try:
        res_slsqp = minimize(full_objective, p0, method='SLSQP', bounds=full_bounds, 
                             constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13})
        if -res_slsqp.fun > best_sum:
            p0 = res_slsqp.x
            best_sum = -res_slsqp.fun
    except Exception:
        pass
        
    # Phase 3: Perturbation refinement to escape local minima
    best_p = p0.copy()
    best_sum_val = best_sum
    for _ in range(10):
        pert = best_p.copy()
        pert += np.random.normal(0, 0.0015, pert.shape)
        pert = np.clip(pert, 1e-6, 1.0)
        pert[2::3] = np.clip(pert[2::3], 1e-6, 0.5)
        try:
            res_pert = minimize(full_objective, pert, method='SLSQP', bounds=full_bounds, 
                                constraints=cons, options={'maxiter': 1500, 'ftol': 1e-13})
            if -res_pert.fun > best_sum_val:
                best_p = res_pert.x
                best_sum_val = -res_pert.fun
        except Exception:
            pass
    p0 = best_p
        
    # Extract final configuration
    centers = np.column_stack((p0[0::3], p0[1::3]))
    radii = p0[2::3].copy()
    
    # Deterministic repair
    radii = repair(centers, radii)
    radii = np.maximum(radii, 0.0)
    
    # Final validation shrink to strictly satisfy validator tolerance
    for _ in range(10):
        valid = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False
                break
        if not valid:
            break
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            break
        radii *= 0.99995
        
    return centers, radii, float(np.sum(radii))
