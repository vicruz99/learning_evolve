import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_COUNT = N * (N - 1) // 2
i_idx, j_idx = np.triu_indices(N, k=1)

def get_bounds():
    """Creates variable bounds: x,y in [0,1], r in [1e-7, 0.5]"""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def objective(p):
    """Minimize negative sum of radii to maximize total radius."""
    return -np.sum(p[2::3])

def constraints(p):
    """
    Computes all boundary and non-overlap constraints.
    Returns a 1D array where each element must be >= 0.
    Uses squared distances for better gradient conditioning.
    """
    x, y, r = p[0::3], p[1::3], p[2::3]
    c = np.empty(4 * N + PAIR_COUNT)
    idx = 0
    
    # Boundary constraints
    c[idx:idx+N] = x - r; idx += N
    c[idx:idx+N] = 1.0 - x - r; idx += N
    c[idx:idx+N] = y - r; idx += N
    c[idx:idx+N] = 1.0 - y - r; idx += N
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    c[idx:] = dx*dx + dy*dy - dr*dr
    return c

def solve_lp_radii(centers):
    """
    Given fixed centers, solves the LP to find radii that maximize sum(r_i)
    while satisfying all non-overlap and boundary constraints strictly.
    """
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    
    # Pairwise Euclidean distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    
    # LP Setup: Maximize sum(r) <=> Minimize -sum(r)
    c_obj = -np.ones(N)
    num_pairs = N * (N - 1) // 2
    A = np.zeros((num_pairs + 4 * N, N))
    b = np.zeros(num_pairs + 4 * N)
    
    idx = 0
    # Pairwise constraints: r_i + r_j <= distance(i,j)
    for i in range(N):
        for j in range(i + 1, N):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(N):
        max_r = min(x[i], 1.0 - x[i], y[i], 1.0 - y[i])
        A[idx, i] = 1.0
        b[idx] = max_r
        idx += 1
        
    bounds = [(0.0, None)] * N
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except:
        pass
    return np.full(N, 1e-6), 0.0

def generate_init_configs(rng):
    """Generates diverse initial configurations for multi-start optimization."""
    configs = []
    
    # 1. Hexagonal patterns with varying densities
    for base_r in [0.09, 0.095, 0.10, 0.105]:
        c = []
        y = base_r
        row = 0
        while len(c) < N:
            x = base_r + (row % 2) * base_r
            while x + base_r <= 1.0 + 1e-9 and len(c) < N:
                c.append([x, y])
                x += 2 * base_r
            y += np.sqrt(3) * base_r
            row += 1
        c = np.array(c[:N]) + rng.normal(0, 0.005, (N, 2))
        c = np.clip(c, 0.02, 0.98)
        configs.append(c)
        
    # 2. Square grid patterns
    for s in [5, 6]:
        xs = np.linspace(0.1, 0.9, s)
        ys = np.linspace(0.1, 0.9, s)
        cx, cy = np.meshgrid(xs, ys)
        c = np.column_stack((cx.flatten(), cy.flatten()))[:N]
        c += rng.normal(0, 0.01, c.shape)
        c = np.clip(c, 0.02, 0.98)
        configs.append(c)
        
    # 3. Random dense configurations
    for _ in range(10):
        c = rng.uniform(0.1, 0.9, (N, 2))
        configs.append(c)
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_p = None
    best_sum = -np.inf
    best_centers = None
    
    configs = generate_init_configs(rng)
    
    # Phase 1: Multi-start SLSQP Optimization
    for c0 in configs:
        p0 = np.zeros(N * 3)
        p0[0::3] = c0[:, 0]
        p0[1::3] = c0[:, 1]
        p0[2::3] = 0.08 + rng.uniform(0, 0.01, N)
        
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12})
            if res.success:
                curr_centers = res.x[:2*N].reshape(N, 2)
                # Extract exact optimal radii for these centers via LP
                lp_r, lp_sum = solve_lp_radii(curr_centers)
                if lp_sum > best_sum:
                    best_sum = lp_sum
                    best_p = res.x.copy()
                    best_centers = curr_centers.copy()
        except:
            continue

    # Fallback if all starts fail
    if best_p is None:
        p0 = np.zeros(N * 3)
        p0[0::3] = rng.uniform(0.1, 0.9, N)
        p0[1::3] = rng.uniform(0.1, 0.9, N)
        p0[2::3] = 0.08
        res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons)
        best_p = res.x.copy()
        best_centers = res.x[:2*N].reshape(N, 2)

    # Phase 2: Iterative Perturbation & LP Refinement
    # Perturb centers, solve LP for radii, re-optimize jointly. Escapes local minima.
    for _ in range(40):
        curr_c = best_centers.copy()
        curr_c += rng.normal(0, 0.0015, curr_c.shape)
        curr_c = np.clip(curr_c, 0.01, 0.99)
        
        lp_r, lp_sum = solve_lp_radii(curr_c)
        
        p0 = np.zeros(N * 3)
        p0[0::3] = curr_c[:, 0]
        p0[1::3] = curr_c[:, 1]
        p0[2::3] = lp_r * 0.995  # Slight shrink ensures feasible start for SLSQP
        
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12})
            if res.success:
                new_c = res.x[:2*N].reshape(N, 2)
                new_lp_r, new_lp_sum = solve_lp_radii(new_c)
                if new_lp_sum > best_sum:
                    best_sum = new_lp_sum
                    best_p = res.x.copy()
                    best_centers = new_c.copy()
        except:
            continue

    # Phase 3: Final High-Precision Polish
    try:
        res = minimize(objective, best_p, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
        if -res.fun > best_sum * 0.99:
            best_p = res.x.copy()
            best_centers = res.x[:2*N].reshape(N, 2)
    except:
        pass

    # Extract final radii via LP for maximum accuracy
    final_radii, final_sum = solve_lp_radii(best_centers)
    
    # Strict validation repair to guarantee validator tolerance compliance
    centers = best_centers.copy()
    radii = final_radii.copy()
    
    for _ in range(20):
        valid = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            if x - r < -1e-12 or x + r > 1.0 + 1e-12 or y - r < -1e-12 or y + r > 1.0 + 1e-12:
                valid = False
                break
        if not valid: break
        
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid: break
        if valid: break
        radii *= 0.9995  # Gentle shrink to resolve numerical overlaps
        
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
