import numpy as np
from scipy.optimize import minimize

N = 26
_MASK_I, _MASK_J = np.triu_indices(N, k=1)

def compute_constraints(vars_flat):
    """Computes inequality constraints for SLSQP. Returns array where each element >= 0 is feasible."""
    c = vars_flat[:2*N].reshape(N, 2)
    r = vars_flat[2*N:]
    
    # Boundary constraints: x >= r, x + r <= 1, y >= r, y + r <= 1
    cons = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    # Overlap constraints: dist(i, j) >= r_i + r_j
    dx = c[:, 0, np.newaxis] - c[:, 0]
    dy = c[:, 1, np.newaxis] - c[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Vectorized pairwise separation constraints
    cons = np.concatenate([cons, dists[_MASK_I, _MASK_J] - (r[_MASK_I] + r[_MASK_J])])
    return cons

def objective(vars_flat):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars_flat[2*N:])

def get_bounds():
    """Variable bounds: centers in [0,1], radii in [1e-6, 0.5]."""
    return [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N

def init_hex(rows, r, noise=0.0):
    """Generates a perturbed hexagonal lattice configuration with specified row counts."""
    centers = []
    y = r
    dy = np.sqrt(3.0) * r
    dx = 2.0 * r

        ox = (dx / 2.0) if (i % 2 == 1) else 0.0
        x = r + ox
        for _ in range(cnt):
            centers.append([x, y])
            x += dx
        y += dy
        
    c = np.array(centers[:N])
    if noise > 0:
        c += np.random.normal(0, noise, c.shape)
    return np.clip(c, 0.02, 0.98)
