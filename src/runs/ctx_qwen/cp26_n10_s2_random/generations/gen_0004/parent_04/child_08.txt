import numpy as np
from scipy.optimize import minimize

N = 26
_MASK_I, _MASK_J = np.triu_indices(N, k=1)

def compute_constraints(vars):
    """
    Computes inequality constraints for SLSQP.
    Returns array where each element must be >= 0 for a valid configuration.
    """
    c = vars[:2*N].reshape(N, 2)
    r = vars[2*N:]
    
    # Boundary constraints: x >= r, x + r <= 1, y >= r, y + r <= 1
    cons = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    # Overlap constraints: dist(i, j) >= r_i + r_j
    dx = c[:, 0, np.newaxis] - c[:, 0]
    dy = c[:, 1, np.newaxis] - c[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Append pairwise separation constraints using precomputed masks
    cons = np.concatenate([cons, dists[_MASK_I, _MASK_J] - (r[_MASK_I] + r[_MASK_J])])
    return cons

def objective(vars):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars[2*N:])

def get_bounds():
    """Variable bounds: centers in [0,1], radii in [1e-6, 0.5]."""
    return [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N

def generate_starts(rng):
    """Generates a diverse set of initial configurations."""
    starts = []
    
    # 1. Hexagonal lattice variations with different base radii and noise
    for r0 in [0.09, 0.095, 0.10, 0.105, 0.11]:
        c = []
        y = r0
        row = 0
        while len(c) < N:
            offset = (r0) if row % 2 == 1 else 0.0
            x = r0 + offset
            while x + r0 <= 1.0 + 1e-9 and len(c) < N:
                c.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
        c = np.array(c[:N])
        for noise in [0.0, 0.002, 0.005, 0.01]:
            c_pert = c + rng.normal(0, noise, c.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            r = np.full(N, r0 * 0.92)
            starts.append(np.concatenate([c_pert.flatten(), r]))
            
    # 2. Corner-focused configurations (corners allow larger circles)
    corners = np.array([[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]])
    for _ in range(8):
        rem = rng.uniform(0.15, 0.85, (N-4, 2))
        c_corner = np.vstack([corners, rem])
        c_pert = c_corner + rng.normal(0, 0.02, c_corner.shape)
        c_pert = np.clip(c_pert, 0.05, 0.95)
        r = np.full(N, 0.085)
        starts.append(np.concatenate([c_pert.flatten(), r]))
        
    # 3. Dense random starts
    for _ in range(12):
        c = rng.uniform(0.1, 0.9, (N, 2))
        r = np.full(N, 0.055)
        starts.append(np.concatenate([c.flatten(), r]))
        
    # 4. Square grid + center fill variation
    for _ in range(5):
        gx = np.linspace(0.1, 0.9, 5)
        gy = np.linspace(0.1, 0.9, 5)
        grid_centers = np.array(np.meshgrid(gx, gy)).T.reshape(-1, 2)
        grid_centers = np.vstack([grid_centers, [0.5, 0.5]])
        c_pert = grid_centers + rng.normal(0, 0.015, grid_centers.shape)
        c_pert = np.clip(c_pert, 0.05, 0.95)
        r = np.full(N, 0.08)
        starts.append(np.concatenate([c_pert.flatten(), r]))
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses multi-start SLSQP with iterative growth refinement and strict repair.
    """
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_val = -np.inf
    best_vars = None
    
    # Phase 1: Multi-Start Local Optimization
    starts = generate_starts(rng)
    
    for x0 in starts:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14})
            if np.all(compute_constraints(res.x) >= -1e-10):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_vars = res.x.copy()
        except Exception:
            continue
            
    if best_vars is None:
        # Fallback to a valid small packing
        c = rng.uniform(0.1, 0.9, (N, 2))
        r = np.full(N, 0.05)
        best_vars = np.concatenate([c.flatten(), r])
        best_val = np.sum(r)
        
    # Phase 2: Iterative Growth & Perturbation Refinement
    # Forces the solver to push past local maxima by scaling radii
    # up and re-optimizing positions with adaptive noise.
    curr = best_vars.copy()
    for step in range(120):
        # Adaptive noise schedule
        noise_scale = 0.0025 * np.exp(-step / 25.0)
        curr[:2*N] += rng.normal(0, noise_scale, 2*N)
        curr[:2*N] = np.clip(curr[:2*N], 0.02, 0.98)
        
        # Periodically kick a random circle to a new position to escape basins
        if step % 12 == 0:
            idx = rng.integers(N)
            curr[2*idx] = rng.uniform(0.15, 0.85)
            curr[2*idx+1] = rng.uniform(0.15, 0.85)
            
        # Slightly increase radii to create "pressure" for expansion
        curr[2*N:] *= 1.0004
        
        try:
            res = minimize(objective, curr, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-14})
            if np.all(compute_constraints(res.x) >= -1e-10):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 3: Deterministic Repair for Strict Validation
    centers = best_vars[:2*N].reshape(N, 2)
    radii = best_vars[2*N:].copy()
    
    for _ in range(50):
        changed = False
        # Resolve overlaps by proportional shrinking
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if d < radii[i] + radii[j] - 1e-12:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov / 2.0
                    radii[j] -= ov / 2.0
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            mx = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
            if radii[i] > mx + 1e-12:
                radii[i] = mx
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
