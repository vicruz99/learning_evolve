import numpy as np
from scipy.optimize import minimize, linprog

N = 26
_MASK_I, _MASK_J = np.triu_indices(N, k=1)

def compute_constraints(vars_flat):
    """Computes inequality constraints for SLSQP. All values must be >= 0."""
    c = vars_flat[:2*N].reshape(N, 2)
    r = vars_flat[2*N:]
    
    # Boundary constraints
    cons = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    # Pairwise non-overlap constraints
    dx = c[_MASK_I, 0] - c[_MASK_J, 0]
    dy = c[_MASK_I, 1] - c[_MASK_J, 1]
    dists = np.hypot(dx, dy)
    cons = np.concatenate([cons, dists - (r[_MASK_I] + r[_MASK_J])])
    return cons

def objective(vars_flat):
    """Minimize negative sum of radii."""
    return -np.sum(vars_flat[2*N:])

def get_bounds():
    """Variable bounds for x, y, r."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def generate_starts():

def solve_lp_radii(centers):
    """Solves LP to find maximum radii for fixed centers."""
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    num_constraints = num_pairs + 4 * n
    
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    # Pairwise: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub[idx] = d - 1e-10
            idx += 1
            
    # Boundary: r_i <= min(x, 1-x, y, 1-y)

        for val in [centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1]]:

            b_ub[idx] = val - 1e-10
            idx += 1
            
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)]*n, method='highs')
    if res.success:
        return res.x
    return np.full(n, 0.01)

def repair_solution(centers, radii):

        # Boundary clamping

                changed = True

                    radii[i] -= ov / 2.0
                    radii[j] -= ov / 2.0
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
