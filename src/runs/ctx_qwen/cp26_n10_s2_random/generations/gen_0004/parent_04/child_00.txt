import numpy as np
from scipy.optimize import minimize, differential_evolution

N = 26

def get_radii_from_centers(centers):
    """Computes the maximum valid radius for each circle given fixed centers."""
    c = centers.reshape(N, 2)
    # Distance to boundaries
    r_bound = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                         np.minimum(c[:, 1], 1.0 - c[:, 1]))
    # Pairwise distances
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    # Radius limited by half distance to nearest neighbor
    r_pair = 0.5 * np.min(dists, axis=1)
    return np.minimum(r_bound, r_pair)

def obj_centers(v):
    """Objective for global search on centers only: minimize negative sum of radii."""
    return -np.sum(get_radii_from_centers(v.reshape(N, 2)))

def obj_full(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_full(v):
    """Constraints for joint optimization: boundary and non-overlap."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    # Boundary constraints: x >= r, x + r <= 1, y >= r, y + r <= 1
    con = np.concatenate([
        c[:, 0] - r, 1.0 - c[:, 0] - r,
        c[:, 1] - r, 1.0 - c[:, 1] - r
    ])
    # Overlap constraints: dist(i, j) >= r_i + r_j
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    idx = np.triu_indices(N, k=1)
    con = np.concatenate([con, dists[idx] - (r[idx[0]] + r[idx[1]])])
    return con

def run_packing():
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    # Initialize with a valid hex grid fallback
    fallback_c = np.zeros((N, 2))
    idx = 0
    for row in range(6):
        for col in range(5):
            if idx >= N: break
            fallback_c[idx, 0] = 0.1 + col * 0.2
            fallback_c[idx, 1] = 0.1 + row * 0.17
            idx += 1
    fallback_r = get_radii_from_centers(fallback_c)
    
    best_sum = np.sum(fallback_r)
    best_c = fallback_c.copy()
    best_r = fallback_r.copy()

    # 1. Differential Evolution on centers for global exploration
    bounds_c = [(0.0, 1.0)] * (2 * N)
    try:
        res_de = differential_evolution(obj_centers, bounds_c, popsize=15, maxiter=500,
                                        mutation=0.8, recombination=0.9,
                                        tol=1e-7, seed=42, workers=1, 
                                        updating='deferred', polish=False)
        c_de = res_de.x.reshape(N, 2)
        r_de = get_radii_from_centers(c_de)
        s = np.sum(r_de)
        if s > best_sum:
            best_sum = s
            best_c = c_de.copy()
            best_r = r_de.copy()
    except Exception:
        pass

    # 2. Generate diverse starts for SLSQP
    starts = []
    
    # Hexagonal lattice variations
    for r0 in [0.09, 0.095, 0.10, 0.105]:
        centers = []
        rows = [5, 6, 5, 6, 4]
        y = r0
        dy = np.sqrt(3) * r0
        dx = 2 * r0
        for i, cnt in enumerate(rows):
            offset = dx/2 if i % 2 == 1 else 0
            x = r0 + offset
            for _ in range(cnt):
                centers.append([x, y])
                x += dx
            y += dy
        c = np.array(centers[:N])
        c += rng.normal(0, 0.008, c.shape)
        c = np.clip(c, 0.02, 0.98)
        r = np.full(N, r0 * 0.92)
        starts.append(np.concatenate([c.flatten(), r]))

    # Add DE best if found
    starts.append(np.concatenate([best_c.flatten(), best_r]))

    # Random starts
    for _ in range(15):
        c = rng.uniform(0.1, 0.9, (N, 2))
        r = np.full(N, 0.06)
        starts.append(np.concatenate([c.flatten(), r]))

    bounds_f = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.5)] * N
    cons = {'type': 'ineq', 'fun': cons_full}

    # 3. Multi-Start SLSQP Local Optimization
    for x0 in starts:
        try:
            res = minimize(obj_full, x0, method='SLSQP', bounds=bounds_f,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13})
            if np.all(cons_full(res.x) >= -1e-9):
                val = -res.fun
                if val > best_sum:
                    best_sum = val
                    best_c = res.x[:2*N].reshape(N, 2)
                    best_r = res.x[2*N:]
        except Exception:
            pass

    # 4. Iterative Perturbation Refinement
    for _ in range(25):
        pert = best_c + rng.normal(0, 0.0015, best_c.shape)
        pert = np.clip(pert, 0.01, 0.99)
        p_r = best_r * 0.998
        x0_p = np.concatenate([pert.flatten(), p_r])
        try:
            res_p = minimize(obj_full, x0_p, method='SLSQP', bounds=bounds_f,
                             constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
            if np.all(cons_full(res_p.x) >= -1e-9):
                val = -res_p.fun
                if val > best_sum:
                    best_sum = val
                    best_c = res_p.x[:2*N].reshape(N, 2)
                    best_r = res_p.x[2*N:]
        except Exception:
            pass

    # 5. Strict Validation Repair
    centers = best_c
    radii = best_r
    
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if d < radii[i] + radii[j] - 1e-12:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov / 2.0
                    radii[j] -= ov / 2.0
                    changed = True
        for i in range(N):
            mx = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
            if radii[i] > mx + 1e-12:
                radii[i] = mx
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
