import numpy as np
from scipy.optimize import minimize

N = 26
_MASK_I, _MASK_J = np.triu_indices(N, k=1)

def compute_constraints(v):
    """
    Computes inequality constraints for SLSQP.
    Returns 1D array where each element must be >= 0 for validity.
    """
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    
    # Boundary constraints: x >= r, x + r <= 1, y >= r, y + r <= 1
    b = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    # Overlap constraints: dist(i, j) >= r_i + r_j
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    d = np.sqrt(dx**2 + dy**2)
    ov = d[_MASK_I, _MASK_J] - (r[_MASK_I] + r[_MASK_J])
    
    return np.concatenate([b, ov])

def objective(v):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(v[2*N:])

def get_bounds():
    """Variable bounds: centers in [0,1], radii in [1e-6, 0.5]."""
    return [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N

def generate_starts():
    """Generates diverse initial configurations for multi-start optimization."""
    starts = []
    np.random.seed(42)
    
    # Hexagonal row patterns summing to 26
    patterns = [
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5],

    ]
    
    for pat in patterns:

            c = []

                x = r0 + offset

                    c.append([x, y])
                    x += 2.0 * r0

            c = np.array(c[:N])
            c = np.clip(c, 0.05, 0.95)
            r = np.full(N, r0 * 0.78)
            starts.append(np.concatenate([c.flatten(), r]))
            
    # Dense random starts
    for _ in range(25):
        c = np.random.rand(N, 2) * 0.8 + 0.1
        r = np.full(N, 0.045)
        starts.append(np.concatenate([c.flatten(), r]))

    for s in starts:

            # Allow slight numerical drift during optimization
            if np.all(compute_constraints(res.x) >= -1e-8):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_v = res.x.copy()
        except Exception:
            continue

    return centers, radii, sum_radii
