import numpy as np
from scipy.optimize import minimize

N = 26
_MASK_I, _MASK_J = np.triu_indices(N, k=1)

def compute_radii_from_centers(centers):
    x, y = centers[:, 0], centers[:, 1]
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)
    r_pair = 0.5 * np.min(dist, axis=1)
    return np.minimum(r_bound, r_pair)

def obj_centers(v):
    return -np.sum(compute_radii_from_centers(v.reshape(N, 2)))

def obj_full(v):
    return -np.sum(v[2*N:])

def cons_full(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = np.concatenate([
        c[:, 0] - r, 1.0 - c[:, 0] - r,
        c[:, 1] - r, 1.0 - c[:, 1] - r
    ])
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    con = np.concatenate([con, dists[_MASK_I, _MASK_J] - (r[_MASK_I] + r[_MASK_J])])
    return con

def get_bounds_full():
    return [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N

def get_hex_init(rows, r0=0.1, noise=0.0):
    centers = []
    y = r0
    dy = np.sqrt(3.0) * r0
    dx = 2.0 * r0
    for i, cnt in enumerate(rows):
        offset = (dx / 2.0) if (i % 2 == 1) else 0.0
        x = r0 + offset
        for _ in range(cnt):
            centers.append([x, y])
            x += dx
        y += dy
    centers = np.array(centers[:N])
    if noise > 0:
        centers += np.random.normal(0, noise, centers.shape)
    min_c = centers.min(axis=0)
    max_c = centers.max(axis=0)
    span = max_c - min_c
    if span.max() > 1e-9:
        scale = 0.94 / span
        centers = (centers - min_c) * scale + 0.03
    return np.clip(centers, 0.01, 0.99)

def greedy_init(rng, n_samples=2000):
    centers = []
    centers.append([0.5, 0.5])
    for _ in range(N - 1):
        pts = rng.uniform(0.05, 0.95, (n_samples, 2))
        valid = (pts[:, 0] >= 0.05) & (pts[:, 0] <= 0.95) & \
                (pts[:, 1] >= 0.05) & (pts[:, 1] <= 0.95)
        pts = pts[valid]
        if len(pts) == 0:
            pts = rng.uniform(0.1, 0.9, (100, 2))
            
        if len(centers) > 0:
            dists_to_centers = np.linalg.norm(pts[:, None, :] - centers[None, :, :], axis=2)
            min_dists = np.min(dists_to_centers, axis=1)
        else:
            min_dists = np.full(len(pts), 1.0)
            
        best_idx = np.argmax(min_dists)
        centers.append(pts[best_idx])
    return np.array(centers[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    bounds = get_bounds_full()
    cons = {'type': 'ineq', 'fun': cons_full}
    
    starts = []
    
    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 5, 5],

                starts.append(np.concatenate([c.flatten(), r]))

        r = np.full(N, 0.07)
        starts.append(np.concatenate([c.flatten(), r]))

        c = rng.uniform(0.1, 0.9, (N, 2))
        r = np.full(N, 0.06)
        starts.append(np.concatenate([c.flatten(), r]))

    bounds_c = [(0.01, 0.99)] * (2*N)

                changed = True
                
        if not changed:

    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
