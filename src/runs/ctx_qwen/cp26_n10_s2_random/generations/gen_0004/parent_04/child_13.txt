1. **LP-based Radius Optimization**: For any fixed set of center positions, finding the maximum valid radii is a Linear Programming problem. I will precompute the constraint matrix `A_ub` and efficiently update the right-hand side `b_ub` (pairwise distances and boundary distances) at each step. `scipy.optimize.linprog` solves this instantly.
2. **Gradient Ascent on Centers**: Using the envelope theorem, the gradient of the maximum sum of radii with respect to center positions can be computed directly from the LP dual variables (marginals). I will perform iterative gradient ascent to push centers into configurations that allow larger radii.
3. **Diverse Initialization & Basin Hopping**: I will test multiple starting configurations (hexagonal grids, perturbed square grids, and dense random layouts). During gradient ascent, I'll use a decaying step size and occasional perturbations to escape local minima.
4. **SLSQP Polish**: After identifying the best configuration from the LP+gradient phase, I will run a final SLSQP optimization on the full (centers, radii) variable space with explicit constraints to squeeze out the last bits of optimality and ensure strict feasibility.
5. **Deterministic Repair**: A final pass will resolve any numerical overlaps or boundary violations within tolerance, guaranteeing the validator passes.

This approach leverages the exactness of LP for radii, the directional guidance of dual gradients for centers, and the robustness of SLSQP for final polishing.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
NUM_PAIRS = N * (N - 1) // 2
TOTAL_CONS = NUM_PAIRS + 4 * N

# Precompute constant parts of the LP constraint matrix
A_ub = np.zeros((TOTAL_CONS, N))
_idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub[_idx, i] = 1.0
        A_ub[_idx, j] = 1.0
        _idx += 1
for i in range(N):
    A_ub[_idx, i] = 1.0; _idx += 1  # r <= x
    A_ub[_idx, i] = 1.0; _idx += 1  # r <= 1-x
    A_ub[_idx, i] = 1.0; _idx += 1  # r <= y
    A_ub[_idx, i] = 1.0; _idx += 1  # r <= 1-y

c_obj_lp = -np.ones(N)  # Minimize -sum(r) => Maximize sum(r)
bounds_r = [(0.0, None)] * N
_MASK_I, _MASK_J = np.triu_indices(N, k=1)

def solve_lp_for_radii(centers):
    """Solves LP to find optimal radii and duals for fixed centers."""
    x = centers[:, 0]
    y = centers[:, 1]
    
    b_ub = np.zeros(TOTAL_CONS)
    
    # Pairwise distances
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    b_ub[:NUM_PAIRS] = dists[_MASK_I, _MASK_J]
    
    # Boundary distances
    b_ub[NUM_PAIRS : NUM_PAIRS + N] = x
    b_ub[NUM_PAIRS + N : NUM_PAIRS + 2*N] = 1.0 - x
    b_ub[NUM_PAIRS + 2*N : NUM_PAIRS + 3*N] = y
    b_ub[NUM_PAIRS + 3*N : NUM_PAIRS + 4*N] = 1.0 - y
    
    try:
        res = linprog(c_obj_lp, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            duals = getattr(res.ineqlin, 'marginals', None)
            if duals is None:
                duals = getattr(res.ineqlin, 'shadow_prices', np.zeros(TOTAL_CONS))
            return res.x, duals, -res.fun
        else:
            return None, None, -np.inf
    except Exception:
        return None, None, -np.inf

def compute_gradient(centers, duals):
    """Computes gradient of max sum(r) w.r.t centers using LP duals."""
    grad = np.zeros((N, 2))
    idx = 0
    
    # Pairwise terms
    for i in range(N):
        for j in range(i + 1, N):
            lam = duals[idx]
            if lam > 1e-8:
                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if d > 1e-8:
                    vec = (centers[i] - centers[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
            idx += 1
            
    # Boundary terms
    base = NUM_PAIRS
    for i in range(N):
        mu_xL = duals[base + 4*i]
        mu_xR = duals[base + 4*i + 1]
        mu_yB = duals[base + 4*i + 2]
        mu_yT = duals[base + 4*i + 3]
        grad[i, 0] += mu_xL - mu_xR
        grad[i, 1] += mu_yB - mu_yT
    return grad

def generate_hex_init(r_guess, noise_scale=0.0):
    """Generates a perturbed hexagonal lattice configuration."""
    centers = []
    rows = [5, 6, 5, 6, 4]
    y = r_guess
    dy = np.sqrt(3.0) * r_guess
    dx = 2.0 * r_guess
    
    for i, cnt in enumerate(rows):
        offset = (dx / 2.0) if (i % 2 == 1) else 0.0
        x = r_guess + offset
        for _ in range(cnt):
            centers.append([x, y])
            x += dx
        y += dy
        
    centers = np.array(centers[:N])
    if noise_scale > 0:
        centers += np.random.normal(0, noise_scale, centers.shape)
    return np.clip(centers, 0.05, 0.95)

def compute_full_constraints(vars):
    """Standard inequality constraints for SLSQP polish."""
    c = vars[:2*N].reshape(N, 2)
    r = vars[2*N:]
    cons = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    dx = c[:, 0, np.newaxis] - c[:, 0]
    dy = c[:, 1, np.newaxis] - c[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    cons = np.concatenate([cons, dists[_MASK_I, _MASK_J] - (r[_MASK_I] + r[_MASK_J])])
    return cons

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Generate diverse starting configurations
    starts = []
    for r_init in [0.09, 0.095, 0.10, 0.105]:
        for noise in [0.0, 0.005, 0.01]:
            starts.append(generate_hex_init(r_init, noise))
            
    # Random dense starts
    for _ in range(8):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # Grid perturbed
    gx = np.linspace(0.15, 0.85, 5)
    gy = np.linspace(0.15, 0.85, 5)
    cx, cy = np.meshgrid(gx, gy)
    grid_c = np.column_stack((cx.flatten(), cy.flatten()))
    grid_c = np.vstack([grid_c, [0.5, 0.5]])
    for _ in range(5):
        starts.append(np.clip(grid_c + rng.normal(0, 0.015, grid_c.shape), 0.1, 0.9))

    # Phase 1: LP + Gradient Ascent
    for c0 in starts:
        centers = c0.copy()
        # Add tiny random perturbation to break exact symmetries
        centers += rng.normal(0, 1e-4, centers.shape)
        
        for step in range(500):
            radii, duals, curr_sum = solve_lp_for_radii(centers)
            if radii is None or curr_sum < 0:
                break
                
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                
            # Compute gradient
            grad = compute_gradient(centers, duals)
            
            # Decay step size
            alpha = 0.012 * np.exp(-step / 150.0)
            if step > 300:
                alpha *= 0.5
                
            centers += alpha * grad
            centers = np.clip(centers, 0.02, 0.98)
            
        # Occasional jitter to escape basins
        centers += rng.normal(0, 0.002, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)

    # Phase 2: SLSQP Polish on best found configuration
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii])
        bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
        cons = {'type': 'ineq', 'fun': compute_full_constraints}
        
        try:
            res = minimize(lambda v: -np.sum(v[2*N:]), x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
            if np.all(compute_full_constraints(res.x) >= -1e-9):
                val = -res.fun
                if val > best_sum:
                    best_centers = res.x[:2*N].reshape(N, 2)
                    best_radii = res.x[2*N:].copy()
                    best_sum = val
        except Exception:
            pass

    # Phase 3: Deterministic Repair
    if best_centers is None:
        best_centers = starts[0]
        best_radii, _, _ = solve_lp_for_radii(best_centers)
        
    centers = best_centers
    radii = best_radii
    
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if d < radii[i] + radii[j] - 1e-12:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov / 2.0
                    radii[j] -= ov / 2.0
                    changed = True
                    
        for i in range(N):
            mx = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
            if radii[i] > mx + 1e-12:
                radii[i] = mx
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
