import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def _build_A_ub(n):
    """Precomputes the constant structure of the LP constraint matrix."""
    num_pairs = n * (n - 1) // 2
    A = np.zeros((num_pairs + 4 * n, n))
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
    for i in range(n):
        A[idx, i] = 1.0; idx += 1
        A[idx, i] = 1.0; idx += 1
        A[idx, i] = 1.0; idx += 1
        A[idx, i] = 1.0; idx += 1
    return A

def _solve_radii(centers, A_ub):
    """Solves the LP to find maximum radii for fixed centers."""
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    
    # Pairwise distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.sqrt(dx * dx + dy * dy)
    np.fill_diagonal(dists, 1e9)
    
    b_ub = np.zeros(A_ub.shape[0])
    k = 0
    num_pairs = n * (n - 1) // 2
    
    # Overlap constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            b_ub[k] = dists[i, j]
            k += 1
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, etc.
    for i in range(n):
        b_ub[k] = x[i]; k += 1
        b_ub[k] = 1.0 - x[i]; k += 1
        b_ub[k] = y[i]; k += 1
        b_ub[k] = 1.0 - y[i]; k += 1
        
    # Maximize sum(r) <=> Minimize -sum(r)
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def _objective(centers_flat, A_ub):
    """Objective for outer optimizer: negative sum of optimal radii."""
    centers = np.clip(centers_flat.reshape(N_CIRCLES, 2), 1e-6, 1.0 - 1e-6)
    _, sum_r = _solve_radii(centers, A_ub)
    return -sum_r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    A_ub = _build_A_ub(N_CIRCLES)
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    rng = np.random.default_rng(42)

    inits = []
    
    # Hexagonal lattices with varying base radii
    for r0 in [0.09, 0.10, 0.11, 0.12]:
        pts = []
        y = r0
        row = 0
        while len(pts) < N_CIRCLES:
            offset = r0 if row % 2 else 0.0
            x = r0 + offset
            while x <= 1.0 - r0 + 1e-6 and len(pts) < N_CIRCLES:
                pts.append([x, y])
                x += 2 * r0
            y += r0 * np.sqrt(3)
            row += 1
        inits.append(np.array(pts[:N_CIRCLES]))
        
    # 5x5 Grid + Center fill
    g_pts = []
    for i in range(5):
        for j in range(5):
            g_pts.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    g_pts.append([0.5, 0.5])
    inits.append(np.array(g_pts))
    
    # Dense random starts
    for _ in range(10):
        inits.append(rng.uniform(0.1, 0.9, (N_CIRCLES, 2)))

    # Phase 1: Multi-start local optimization
    for c0 in inits:

        try:
            res = minimize(_objective, c0_noisy.flatten(), args=(A_ub,), method='Powell', bounds=bounds,

                best_sum = -res.fun
                best_centers = res.x.reshape(N_CIRCLES, 2)

        except Exception:
            pass

    # Phase 2: Iterative perturbation to escape local minima
    if best_centers is not None:

                radii[i] = mx
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
