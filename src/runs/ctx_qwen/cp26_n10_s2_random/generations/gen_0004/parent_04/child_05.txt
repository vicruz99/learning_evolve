import numpy as np
from scipy.optimize import minimize, linprog

N = 26
_MASK_I, _MASK_J = np.triu_indices(N, k=1)

def get_bounds():
    """Variable bounds: centers in [0,1], radii in [1e-7, 0.5]."""
    bounds = []
    for _ in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return bounds

def compute_constraints(vars):
    """
    Computes inequality constraints for SLSQP.
    Returns array where each element must be >= 0 for a valid configuration.
    Uses linear distance constraints for better gradient behavior.
    """
    c = vars[:2*N].reshape(N, 2)
    r = vars[2*N:]
    
    # Boundary constraints: x >= r, x + r <= 1, y >= r, y + r <= 1
    cons = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    # Overlap constraints: dist(i, j) >= r_i + r_j
    dx = c[:, 0, np.newaxis] - c[:, 0]
    dy = c[:, 1, np.newaxis] - c[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    cons = np.concatenate([cons, dists[_MASK_I, _MASK_J] - (r[_MASK_I] + r[_MASK_J])])
    return cons

def objective(vars):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars[2*N:])

def generate_hex_init(r0, pattern, rng):
    """Generates a perturbed hexagonal lattice configuration."""
    centers = []
    y = r0
    dy = np.sqrt(3.0) * r0
    dx = 2.0 * r0
    
    for i, cnt in enumerate(pattern):
        offset = (dx / 2.0) if (i % 2 == 1) else 0.0
        x = r0 + offset
        for _ in range(cnt):
            centers.append([x, y])
            x += dx
        y += dy
        
    centers = np.array(centers[:N])
    centers += rng.normal(0, 0.002, centers.shape)
    return np.clip(centers, 0.05, 0.95)

def solve_lp_radii(centers):
    """
    Given fixed centers, solves the LP to find radii that maximize sum(r_i)
    while satisfying all non-overlap and boundary constraints strictly.
    """
    x, y = centers[:, 0], centers[:, 1]
    n = N
    num_pairs = n * (n - 1) // 2
    m = num_pairs + 4 * n
    
    A = np.zeros((m, n))
    b = np.zeros(m)
    
    idx = 0
    # Pairwise constraints: r_i + r_j <= distance(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = np.hypot(x[i] - x[j], y[i] - y[j]) - 1e-9
            idx += 1
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        A[idx, i] = 1.0; b[idx] = x[i] - 1e-9; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - x[i] - 1e-9; idx += 1
        A[idx, i] = 1.0; b[idx] = y[i] - 1e-9; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - y[i] - 1e-9; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0.0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return np.full(n, 1e-6)

    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
