N = 26
_MASK_I, _MASK_J = np.triu_indices(N, k=1)
NUM_PAIRS = N * (N - 1) // 2

def compute_radii_from_centers(centers):
    """Computes the maximum valid radius for each circle given fixed centers."""
    x, y = centers[:, 0], centers[:, 1]
    # Distance to boundaries
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise distances using broadcasting
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Radius limited by half the distance to the nearest neighbor
    r_pair = 0.5 * np.min(dists, axis=1)
    
    return np.minimum(r_bound, r_pair)

def objective_centers(v):
    """Objective for centers-only optimization: minimize negative sum of radii."""
    centers = v.reshape(N, 2)
    # Soft penalty for out-of-bounds to keep Powell within [0,1] effectively
    penalty = np.sum(np.maximum(0.0, -centers)) + np.sum(np.maximum(0.0, centers - 1.0))
    radii = compute_radii_from_centers(centers)
    return -np.sum(radii) + 1e4 * penalty

def compute_constraints_joint(v):
    """Computes inequality constraints for SLSQP joint optimization."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    cons = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    # Overlap constraints: dist(i, j) >= r_i + r_j
    dx = c[_MASK_I, 0] - c[_MASK_J, 0]
    dy = c[_MASK_I, 1] - c[_MASK_J, 1]
    dists = np.sqrt(dx**2 + dy**2)
    cons = np.concatenate([cons, dists - (r[_MASK_I] + r[_MASK_J])])
    
    return cons

def objective_joint(v):

    centers = []
    r_guess = 0.10
    y = r_guess
    dy = np.sqrt(3.0) * r_guess
    dx = 2.0 * r_guess

        offset = (dx / 2.0) if (i % 2 == 1) else 0.0
        x = r_guess + offset
        for _ in range(cnt):
            centers.append([x * scale, y * scale])
            x += dx * scale
        y += dy * scale
        
    centers = np.array(centers[:N])
    if noise > 0:
        centers += np.random.normal(0, noise, centers.shape)
    return np.clip(centers, 0.02, 0.98)

def repair_solution(centers, radii):
    """Iteratively shrinks radii to strictly satisfy validation tolerances."""
    radii = radii.copy()
    for _ in range(30):
        changed = False
        # Resolve overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if d < radii[i] + radii[j] - 1e-12:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov / 2.0
                    radii[j] -= ov / 2.0
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            mx = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
            if radii[i] > mx + 1e-12:
                radii[i] = mx
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
