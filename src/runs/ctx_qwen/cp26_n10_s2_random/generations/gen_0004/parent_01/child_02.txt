The current SLSQP approach reaches ~2.6287, but struggles with the highly non-convex landscape of circle packing. To break through the 2.63 barrier, we switch to an **LP-based alternating optimization** strategy, which is theoretically superior for this problem:

1. **Fix Centers, Optimize Radii (LP)**: For any fixed set of centers, finding the maximum sum of radii is a Linear Program: maximize $\sum r_i$ subject to $r_i + r_j \le \|c_i - c_j\|$ and $r_i \le \text{dist}(c_i, \partial \Omega)$. This is solved exactly and instantly by `scipy.optimize.linprog`.
2. **Fix Radii, Optimize Centers (Gradient Ascent)**: The LP dual variables (marginals) directly give the gradient of the max sum of radii with respect to center positions. A positive dual $\lambda_{ij}$ for pair $(i,j)$ means increasing the distance $\|c_i - c_j\|$ relaxes the constraint and allows larger radii. We compute forces $F_i = \sum_{j} \lambda_{ij} \frac{c_i - c_j}{\|c_i - c_j\|}$ plus boundary repulsion forces, and perform gradient ascent: $c \leftarrow c + \alpha F$.
3. **Alternating Scheme**: We iterate between LP radius optimization and center gradient ascent with a decaying step size. This smoothly expands the configuration into high-value regions of the search space.
4. **Diverse Initialization**: We seed the process with random, hexagonal, and grid configurations to escape poor local minima.
5. **Strict Repair**: A final iterative shrink/clamp pass guarantees the solution passes the validation checker's tolerance thresholds.

This method is robust, differentiable almost everywhere, and directly targets the objective function rather than relying on constraint penalties.

import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    pairs = [(i, j) for i in range(N) for j in range(i+1, N)]
    M = len(pairs)
    
    # Precompute constant part of A_ub for LP:
    # Rows 0..M-1: r_i + r_j <= d_ij  -> A[k, i]=1, A[k, j]=1
    # Rows M..M+N-1: r_i <= b_i       -> A[M+i, i]=1
    A_ub = np.zeros((M + N, N))
    for k, (i, j) in enumerate(pairs):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
    for i in range(N):
        A_ub[M+i, i] = 1.0
        
    c_obj = -np.ones(N)  # Maximize sum(r) <=> Minimize -sum(r)
    bounds = [(0, None)] * N
    b_ub = np.zeros(M + N)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)
    inits = []
    
    # 1. Random spreads with quick repulsion initialization
    for _ in range(10):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(100):
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if dist < 0.18:
                        push = (0.18 - dist) * 0.5
                        c[i] += d_vec / dist * push
                        c[j] -= d_vec / dist * push
        inits.append(np.clip(c, 0.05, 0.95))
        
    # 2. Hexagonal lattice patterns
    for _ in range(5):
        c = []
        r_est = 0.095
        y = r_est
        row = 0
        while len(c) < N:
            x = r_est if row % 2 == 0 else 2 * r_est
            while x + r_est <= 1.0 and len(c) < N:
                c.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
            row += 1
        c = np.array(c[:N]) + rng.normal(0, 0.005, (N, 2))
        inits.append(np.clip(c, 0.05, 0.95))
        
    # 3. Grid patterns
    for _ in range(5):
        x = np.linspace(0.15, 0.85, 6)
        y = np.linspace(0.15, 0.85, 5)
        cx, cy = np.meshgrid(x, y)
        c = np.column_stack((cx.flatten(), cy.flatten()))
        c += rng.normal(0, 0.01, c.shape)
        inits.append(np.clip(c[:N], 0.05, 0.95))
        
    # Alternating LP + Gradient Ascent
    for c_init in inits:
        c = c_init.copy()
        alpha = 0.05
        
        for it in range(3000):
            # Pairwise distances
            diff = c[:, None, :] - c[None, :, :]
            d_mat = np.sqrt(np.sum(diff**2, axis=2))
            np.fill_diagonal(d_mat, np.inf)
            d_mat = np.maximum(d_mat, 1e-9)
            
            # Boundary distances
            xb = np.minimum(c[:, 0], 1.0 - c[:, 0])
            yb = np.minimum(c[:, 1], 1.0 - c[:, 1])
            b_i = np.maximum(np.minimum(xb, yb), 1e-9)
            
            # Update LP RHS
            for k, (i, j) in enumerate(pairs):
                b_ub[k] = d_mat[i, j]
            b_ub[M:] = b_i
            
            try:
                res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
                if not res.success:
                    break
                    
                r_opt = res.x
                curr_sum = np.sum(r_opt)
                
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = c.copy()
                    best_radii = r_opt.copy()
                    
                # Compute ascent forces from LP duals
                F = np.zeros_like(c)
                # marginals are <=0 for minimization; negate for ascent direction
                marginals = -getattr(getattr(res, 'marginals', None), 'ineqlin', np.zeros(M+N))
                
                # Pairwise repulsion forces
                for k, (i, j) in enumerate(pairs):
                    lam = marginals[k]
                    if lam > 1e-8:
                        dist = d_mat[i, j]
                        inv_dist = 1.0 / dist
                        dx = (c[i, 0] - c[j, 0]) * inv_dist
                        dy = (c[i, 1] - c[j, 1]) * inv_dist
                        F[i, 0] += lam * dx
                        F[j, 0] -= lam * dx
                        F[i, 1] += lam * dy
                        F[j, 1] -= lam * dy
                        
                # Boundary repulsion forces
                for i in range(N):
                    mu = marginals[M+i]
                    if mu > 1e-8:
                        xi, yi = c[i]
                        dists = [xi, 1.0-xi, yi, 1.0-yi]
                        min_idx = np.argmin(dists)
                        if min_idx == 0: F[i, 0] += mu
                        elif min_idx == 1: F[i, 0] -= mu
                        elif min_idx == 2: F[i, 1] += mu
                        else: F[i, 1] -= mu
                        
                # Update centers and decay step size
                c += alpha * F
                c = np.clip(c, 1e-4, 1.0 - 1e-4)
                alpha *= 0.998
                
            except Exception:
                break
                
    # Fallback safety
    if best_centers is None:
        best_centers = inits[0]
        best_radii = np.zeros(N)
        
    centers = best_centers
    radii = best_radii
    
    # Strict numerical repair to guarantee validation passes
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
