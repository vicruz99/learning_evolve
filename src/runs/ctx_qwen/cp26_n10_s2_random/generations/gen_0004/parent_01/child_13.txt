

<strategy>
To improve the packing of 26 circles in a unit square, we move beyond pure SLSQP optimization which often gets trapped in local minima. Instead, we use a hybrid approach combining:
1. **LP-based Gradient Ascent**: For fixed centers, the maximum sum of radii is a Linear Program. The dual variables (shadow prices) from this LP directly give the sensitivity of the objective to constraint bounds. This allows us to compute an exact gradient of the sum of radii with respect to circle centers. We iteratively move centers along this gradient, effectively "pushing" circles apart where constraints are tight. This is highly efficient and escapes many local optima that plague nonlinear solvers.
2. **Diverse Initialization**: We seed the process with hexagonal lattices of varying densities and random dense configurations to cover different topological basins.
3. **SLSQP Polishing**: After the gradient ascent converges, we run SLSQP on the joint (centers, radii) problem to fine-tune the solution to machine precision.
4. **Perturbation Search**: We apply decaying random perturbations around the best configuration and re-optimize to escape remaining shallow local minima.
5. **Strict Numerical Repair**: A final pass shrinks radii only if necessary to guarantee strict feasibility against the validation checker's tolerances.

This strategy leverages exact sensitivity analysis to guide the search, typically yielding significantly higher sum of radii than direct nonlinear optimization alone.

import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_for_radii(centers, A_ub, N, M_pair):
    """Solves LP to find optimal radii for fixed centers and returns duals."""
    M = M_pair + 4 * N
    b_ub = np.zeros(M)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            b_ub[idx] = d
            idx += 1
    for i in range(N):
        b_ub[idx] = centers[i, 0]
        idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]
        idx += 1
        b_ub[idx] = centers[i, 1]
        idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]
        
    c_obj = -np.ones(N)
    bounds_r = [(0, None)] * N
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if not res.success:
        return None, None, None
        
    # Extract duals (marginals or shadow_price depending on scipy version)
    duals = getattr(res.ineqlin, 'marginals', res.ineqlin.shadow_price)
    return res.x, -res.fun, duals

def compute_gradient(centers, duals, N, M_pair):
    """Computes gradient of sum of radii w.r.t centers using LP duals."""
    grad = np.zeros_like(centers)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            lam = duals[idx]
            if lam > 1e-9:
                d_vec = centers[i] - centers[j]
                d = np.linalg.norm(d_vec)
                if d > 1e-9:
                    force = (lam / d) * d_vec
                    grad[i] += force
                    grad[j] -= force
            idx += 1
            
    b_start = M_pair
    for i in range(N):
        mu_x1 = duals[b_start + 4*i]     # r <= x
        mu_x2 = duals[b_start + 4*i + 1] # r <= 1-x
        mu_y1 = duals[b_start + 4*i + 2] # r <= y
        mu_y2 = duals[b_start + 4*i + 3] # r <= 1-y
        grad[i, 0] += mu_x1 - mu_x2
        grad[i, 1] += mu_y1 - mu_y2
    return grad

def repair_and_validate(centers, radii, N):
    """Iteratively shrinks radii to guarantee strict feasibility."""
    radii = radii.copy()
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-10:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx + 1e-10:
                radii[i] = mx
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def obj_slsqp(v, N):
    return -np.sum(v[2*N:])

def cons_slsqp(v, N):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    return np.concatenate(con)

def run_packing():
    N = 26
    M_pair = N * (N - 1) // 2
    M = M_pair + 4 * N
    
    # Precompute constant A_ub matrix for LP
    A_ub = np.zeros((M, N))
    pair_idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[pair_idx, i] = 1.0
            A_ub[pair_idx, j] = 1.0
            pair_idx += 1
    for i in range(N):
        A_ub[M_pair + 4*i, i] = 1.0
        A_ub[M_pair + 4*i + 1, i] = 1.0
        A_ub[M_pair + 4*i + 2, i] = 1.0
        A_ub[M_pair + 4*i + 3, i] = 1.0
        
    np.random.seed(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Generate diverse initial configurations
    inits = []
    # Hexagonal lattices
    for r0 in [0.085, 0.095, 0.105]:
        c = []
        y = r0
        row = 0
        while len(c) < N:
            x = r0 if row % 2 == 0 else 2 * r0
            while x + r0 <= 1.0 and len(c) < N:
                c.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        inits.append(np.array(c[:N]))
        
    # Random dense starts
    for _ in range(5):
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))
        
    # Phase 1: LP Gradient Ascent
    for init_c in inits:
        centers = init_c.copy()
        step = 0.02
        for it in range(800):
            radii, curr_sum, duals = solve_lp_for_radii(centers, A_ub, N, M_pair)
            if radii is None:
                break
                
            grad = compute_gradient(centers, duals, N, M_pair)
            norm_g = np.linalg.norm(grad)
            if norm_g < 1e-8:
                break
                
            move = centers + step * grad / (norm_g + 1e-9)
            move = np.clip(move, 0.0, 1.0)
            
            new_r, new_s, _ = solve_lp_for_radii(move, A_ub, N, M_pair)
            if new_s is None or new_s < curr_sum - 1e-7:
                step *= 0.5
                if step < 1e-6:
                    break
                continue
                
            centers = move
            curr_sum = new_s
            step = min(step * 1.02, 0.05)
            
        radii = repair_and_validate(centers, radii, N)
        final_s = np.sum(radii)
        if final_s > best_sum:
            best_sum = final_s
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    # Phase 2: SLSQP Polish
    if best_centers is not None:
        v0 = np.concatenate([best_centers.flatten(), best_radii])
        bounds_sl = [(0, 1)] * 2*N + [(0, 0.5)] * N
        try:
            res = minimize(obj_slsqp, v0, args=(N,), method='SLSQP', bounds=bounds_sl,
                           constraints={'type': 'ineq', 'fun': cons_slsqp, 'args': (N,)},
                           options={'maxiter': 3000, 'ftol': 1e-13})
            v_opt = res.x
            c_opt = v_opt[:2*N].reshape(N, 2)
            r_opt = v_opt[2*N:]
            r_opt = repair_and_validate(c_opt, r_opt, N)
            s_opt = np.sum(r_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt
                best_radii = r_opt
        except Exception:
            pass
            
    # Phase 3: Perturbation Search
    if best_centers is not None:
        bounds_sl = [(0, 1)] * 2*N + [(0, 0.5)] * N
        for step_i in range(20):
            noise = 0.005 * (0.8 ** step_i)
            pert_c = best_centers + np.random.normal(0, noise, best_centers.shape)
            pert_c = np.clip(pert_c, 0.05, 0.95)
            v0 = np.concatenate([pert_c.flatten(), best_radii * 0.95])
            try:
                res = minimize(obj_slsqp, v0, args=(N,), method='SLSQP', bounds=bounds_sl,
                               constraints={'type': 'ineq', 'fun': cons_slsqp, 'args': (N,)},
                               options={'maxiter': 2000, 'ftol': 1e-13})
                v_opt = res.x
                c_opt = v_opt[:2*N].reshape(N, 2)
                r_opt = v_opt[2*N:]
                r_opt = repair_and_validate(c_opt, r_opt, N)
                s_opt = np.sum(r_opt)
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt
                    best_radii = r_opt
            except Exception:
                pass
                
    return best_centers, best_radii, float(best_sum)
