import numpy as np
from scipy.optimize import minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, 1)

def compute_radii(centers):
    """Computes the maximum feasible radius for each circle given fixed centers."""
    xb = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    yb = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    r_bound = np.minimum(xb, yb)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_pair = 0.5 * np.min(dists, axis=1)
    
    return np.minimum(r_bound, r_pair)

def obj_centers(c_flat):
    """Objective for center-only optimization: maximize sum of bottleneck radii."""
    c = c_flat.reshape(N, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    return -np.sum(compute_radii(c))

def compute_constraints(v):
    """Computes all boundary and non-overlap constraints for SLSQP."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    
    con = [
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r,
        np.linalg.norm(c[IDX_I] - c[IDX_J], axis=1) - (r[IDX_I] + r[IDX_J])
    ]
    return np.concatenate(con)

def obj_joint(v):
    """Objective for joint optimization: maximize sum of radii."""
    return -np.sum(v[2*N:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}

    best_v = None
    best_sum = -1.0

    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice patterns with varying row structures
    patterns = [[5,5,5,5,6], [6,5,6,5,4], [5,6,5,6,4], [4,6,6,6,4], [5,5,6,5,5]]
    for pat in patterns:
        c = []
        r_est = 0.095
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                c.append([x, y])
                x += 2.0 * r_est
            y += r_est * 1.7320508
        c = np.array(c[:N])
        c += rng.normal(0, 0.002, c.shape)
        c = np.clip(c, 0.1, 0.9)
        r_init = compute_radii(c) * 0.9
        inits.append(np.concatenate([c.flatten(), r_init]))

    # 2. Random starts with pairwise repulsion to ensure well-spaced points
    for _ in range(12):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(30):
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if dist < 0.16:
                        push = (0.16 - dist) * 0.5
                        c[i] += d_vec / dist * push
                        c[j] -= d_vec / dist * push
        c = np.clip(c, 0.1, 0.9)
        r_init = compute_radii(c) * 0.9
        inits.append(np.concatenate([c.flatten(), r_init]))

    # Phase 1: Multi-start constrained optimization
    for v0 in inits:
        try:
            res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
            if np.min(compute_constraints(res.x)) > -1e-7:
                s = np.sum(res.x[2*N:])
                if s > best_sum:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            continue

    if best_v is None:
        best_v = inits[0]

    best_centers = best_v[:2*N].reshape(N, 2)

    # Phase 2: Alternating refinement to escape local minima
    # Powell on centers handles non-smooth radius landscape, SLSQP tightens constraints
    for step in range(15):
        noise = 0.003 * (0.8 ** step)
        c_pert = best_centers + rng.normal(0, noise, best_centers.shape)
        c_pert = np.clip(c_pert, 1e-5, 1.0 - 1e-5)

        try:
            res_c = minimize(obj_centers, c_pert.flatten(), method='Powell',
                            options={'maxiter': 1500, 'ftol': 1e-12})
            best_centers = res_c.x.reshape(N, 2)
        except Exception:
            pass

        radii = compute_radii(best_centers)
        v0 = np.concatenate([best_centers.flatten(), radii])

        try:
            res_j = minimize(obj_joint, v0, method='SLSQP', bounds=bounds,
                            constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
            if np.min(compute_constraints(res_j.x)) > -1e-7:
                s = np.sum(res_j.x[2*N:])
                if s > best_sum:
                    best_sum = s
                    best_v = res_j.x.copy()
                    best_centers = best_v[:2*N].reshape(N, 2)
        except Exception:
            pass

    # Phase 3: Final extraction and strict numerical repair
    centers = best_v[:2*N].reshape(N, 2) if best_v is not None else best_centers
    radii = compute_radii(centers)

    # Guarantee validity against strict checker tolerances
    for _ in range(20):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):

            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break

    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
