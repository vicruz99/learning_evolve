import numpy as np
from scipy.optimize import minimize

N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)

def compute_constraints(v):
    """Computes boundary and non-overlap constraints. Returns array >= 0 for feasibility."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    
    con = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    c_i = c[IDX_I]
    c_j = c[IDX_J]
    r_i = r[IDX_I]
    r_j = r[IDX_J]
    
    dists = np.sqrt(np.sum((c_i - c_j)**2, axis=1))
    con.append(dists - (r_i + r_j))
    
    return np.concatenate(con)

def objective(v):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2*N:])

def get_safe_radii(c):
    """Computes strictly feasible initial radii for a given set of centers."""
    rb = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    dists = np.sqrt(np.sum((c[:, None, :] - c[None, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    rp = 0.5 * np.min(dists, axis=1)
    # Return 95% of max possible to ensure strict initial feasibility
    return np.minimum(rb, rp) * 0.95

def generate_inits(rng):
    """Generates diverse initial configurations."""
    inits = []
    
    # 1. Hexagonal lattice patterns with various row structures
    patterns = [
        [5, 5, 5, 5, 6], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4],
        [4, 6, 6, 6, 4], [5, 5, 6, 5, 5], [6, 4, 6, 5, 5],
        [5, 4, 6, 6, 5], [4, 5, 6, 5, 6], [5, 5, 5, 6, 5],
        [5, 5, 4, 6, 6], [6, 6, 4, 5, 5], [4, 5, 5, 6, 6]
    ]
    
    for pat in patterns:
        c = []
        r_est = 0.095
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                c.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        c = np.array(c[:N])
        # Add small noise to break symmetry and avoid exact grid degeneracies
        c += rng.normal(0, 0.004, c.shape)
        c = np.clip(c, 0.12, 0.88)
        r_init = get_safe_radii(c)
        inits.append(np.concatenate([c.flatten(), r_init]))
        
    # 2. Force-directed random starts to ensure good initial spacing
    for _ in range(25):
        c = rng.uniform(0.2, 0.8, (N, 2))
        # Simple repulsive force simulation to spread points evenly
        for _ in range(800):
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if dist < 0.18 and dist > 1e-6:
                        push = (0.18 - dist) * 0.08
                        c[i] += d_vec / dist * push
                        c[j] -= d_vec / dist * push
        c = np.clip(c, 0.1, 0.9)
        r_init = get_safe_radii(c)
        inits.append(np.concatenate([c.flatten(), r_init]))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    rng = np.random.default_rng(42)
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_v = None
    best_sum = -1.0
    
    inits = generate_inits(rng)
    
    # Phase 1: Multi-start constrained optimization
    for v0 in inits:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            c_vals = compute_constraints(res.x)
            if np.min(c_vals) >= -1e-7:
                s = np.sum(res.x[2*N:])
                if s > best_sum:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            continue
            
    # Fallback if all optimizations failed (highly unlikely)
    if best_v is None:
        best_v = inits[0]
        
    # Phase 2: Iterative perturbation search to escape local minima
    for step in range(40):
        noise = 0.006 * (0.85 ** step)
        v_pert = best_v + rng.normal(0, noise, best_v.shape)
        v_pert = np.clip(v_pert, 0.0, 1.0)
        v_pert[2*N:] = np.clip(v_pert[2*N:], 0.0, 0.5)
        
        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 8000, 'ftol': 1e-13})
            c_vals = compute_constraints(res.x)
            if np.min(c_vals) >= -1e-7:
                s = np.sum(res.x[2*N:])
                if s > best_sum:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            continue

    centers = best_v[:2*N].reshape(N, 2)
    radii = best_v[2*N:].copy()
    
    # Phase 3: Strict numerical repair to guarantee validation passes
    for _ in range(50):
        changed = False
        # Fix overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
