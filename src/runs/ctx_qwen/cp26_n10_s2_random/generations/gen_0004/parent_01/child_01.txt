import numpy as np
from scipy.optimize import minimize

N = 26

def obj_func(v):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2*N:])

def compute_constraints(v):
    """Computes all boundary and non-overlap constraints. Returns array >= 0 for feasibility."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    
    con = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    
    return np.concatenate(con)

def get_safe_radii(c):
    """Computes strictly feasible initial radii for a given set of centers."""
    # Distance to boundaries
    rb = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    # Distance to nearest neighbor
    dists = np.linalg.norm(c[:, None, :] - c[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    rp = 0.5 * np.min(dists, axis=1)
    # Take 95% of the maximum possible to ensure strict feasibility for optimizer
    return np.minimum(rb, rp) * 0.95

def generate_hex_init(row_counts, r_est, rng):
    """Generates a hexagonal lattice initialization with specified row counts."""
    c = []
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = r_est if r_idx % 2 == 1 else 0.0
        x = r_est + shift
        for _ in range(cnt):
            c.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3)
    c = np.array(c[:N])
    # Add small noise to break symmetry
    c += rng.normal(0, 0.004, c.shape)
    return np.clip(c, 0.08, 0.92)

def run_optimizer(v0):
    """Runs SLSQP optimization and returns optimized parameters and sum of radii."""
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    try:
        res = minimize(obj_func, v0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 8000, 'ftol': 1e-15, 'disp': False})
        c_vals = compute_constraints(res.x)
        if np.min(c_vals) >= -1e-7:
            return res.x, np.sum(res.x[2*N:])
    except Exception:
        pass
    return v0, -1.0
