import numpy as np
from scipy.optimize import minimize, NonlinearConstraint, Bounds

N = 26

def objective(x):
    """Maximize sum of radii -> minimize negative sum."""
    return -np.sum(x[2::3])

def compute_constraints(x):
    """
    Computes all boundary and non-overlap constraints.
    Returns an array of constraint values (must be >= 0).
    """
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    num_constraints = 5 * N + N * (N - 1) // 2
    c = np.empty(num_constraints)
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r, r >= 0
    c[0:N] = xs - rs
    c[N:2*N] = 1.0 - xs - rs
    c[2*N:3*N] = ys - rs
    c[3*N:4*N] = 1.0 - ys - rs
    c[4*N:5*N] = rs
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    idx_i, idx_j = np.triu_indices(N, k=1)
    dx = xs[idx_i] - xs[idx_j]
    dy = ys[idx_i] - ys[idx_j]
    dr = rs[idx_i] + rs[idx_j]
    c[5*N:] = dx**2 + dy**2 - dr**2
    
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]"""
    lower = [0.0] * (3 * N)
    upper = [1.0] * (2 * N) + [0.5] * N
    return Bounds(lower, upper)

def generate_initial_guess(seed):
    """Generates a feasible initial configuration."""
    rng = np.random.default_rng(seed)
    x0 = np.zeros(3 * N)
    
    strategy = seed % 4
    if strategy == 0:
        # Hexagonal lattice
        pts = []
        y = 0.06
        row = 0
        while len(pts) < N:
            x = 0.06 + (row % 2) * 0.085
            while x <= 0.94 and len(pts) < N:
                pts.append([x, y])
                x += 0.17
            y += 0.15
            row += 1
        for i, p in enumerate(pts[:N]):
            x0[3*i] = p[0]
            x0[3*i+1] = p[1]
    elif strategy == 1:
        # Uniform grid
        pts = []
        for i in range(6):
            for j in range(5):
                pts.append([0.08 + i * 0.16, 0.08 + j * 0.19])
        for i, p in enumerate(pts[:N]):
            x0[3*i] = p[0]
            x0[3*i+1] = p[1]
    elif strategy == 2:
        # Corner-focused + random fill
        pts = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]
        for _ in range(N - 4):
            pts.append(rng.uniform(0.2, 0.8, 2).tolist())
        for i, p in enumerate(pts[:N]):
            x0[3*i] = p[0]
            x0[3*i+1] = p[1]
    else:
        # Dense random
        for i in range(N):
            x0[3*i] = rng.uniform(0.05, 0.95)
            x0[3*i+1] = rng.uniform(0.05, 0.95)
            
    # Start with small feasible radii
    x0[2::3] = 0.04
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = get_bounds()
    cons = NonlinearConstraint(compute_constraints, 0, np.inf)
    
    best_val = -np.inf
    best_x = None
    
    # Phase 1: Multiple restarts from diverse initializations
    for s in range(20):
        x0 = generate_initial_guess(s)
        try:
            res = minimize(objective, x0, method='trust-constr', 
                           bounds=bounds, constraints=[cons],
                           options={'maxiter': 1500, 'verbose': 0})
            
            if -res.fun > best_val:
                c_vals = compute_constraints(res.x)
                if np.min(c_vals) >= -1e-8:
                    best_val = -res.fun
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Iterative perturbation to escape local optima
    if best_x is not None:
        current_x = best_x.copy()
        for step in range(10):
            # Perturb centers and radii slightly
            noise = np.random.normal(0, 0.002, current_x.shape)
            # Keep radii positive during noise
            noise[2::3] = np.abs(noise[2::3]) * 0.01
            
            x_perturbed = current_x + noise
            # Project back into bounds
            x_perturbed = np.clip(x_perturbed, bounds.lb, bounds.ub)
            
            try:
                res = minimize(objective, x_perturbed, method='trust-constr',
                               bounds=bounds, constraints=[cons],
                               options={'maxiter': 800, 'verbose': 0})
                
                if -res.fun > best_val:
                    c_vals = compute_constraints(res.x)
                    if np.min(c_vals) >= -1e-8:
                        best_val = -res.fun
                        best_x = res.x.copy()
                        current_x = best_x.copy()
            except Exception:
                pass

    # Fallback if optimization completely failed
    if best_x is None:
        best_x = generate_initial_guess(0)
        
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x.reshape(N, 3)[:, 2]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
