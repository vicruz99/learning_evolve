import numpy as np
from scipy.optimize import minimize

N = 26

def force_directed_init(seed):
    """Rapidly converges to a dense, feasible packing using repulsive forces and radius growth."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    idx = 0
    r_est = 0.09
    dy = r_est * np.sqrt(3)
    
    # Hexagonal lattice initialization
    for row in range(6):
        for col in range(6):
            x = col * 2 * r_est + (0.5 if row % 2 else 0) * 2 * r_est + 0.05
            y = row * dy + 0.05
            if x < 1.0 and y < 1.0 and idx < N:
                centers[idx] = [x, y]
                idx += 1
        if idx >= N: break
        
    # Fill any remaining slots randomly (shouldn't trigger for N=26)
    while idx < N:
        centers[idx] = rng.uniform(0.2, 0.8, 2)
        idx += 1
        
    radii = np.full(N, 0.07)
    
    # Simulation loop
    for step in range(4000):
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.linalg.norm(diff, axis=2)
        min_dists = radii[:, None] + radii[None, :]
        
        overlap = np.maximum(0.0, min_dists - dists)
        np.fill_diagonal(overlap, 0.0)
        max_ov = np.max(overlap)
        
        # Repulsion forces (proportional to overlap)
        force_dirs = np.where(dists < 1e-9, 0.0, diff / dists[:, :, None])
        force_mags = overlap * 800.0
        forces = np.sum(force_mags[:, :, None] * force_dirs, axis=1)
        
        # Boundary repulsion
        for i in range(N):
            r = radii[i]
            if centers[i,0] < r:
                forces[i,0] += (r - centers[i,0]) * 1500.0
            elif centers[i,0] > 1-r:
                forces[i,0] -= (centers[i,0] - (1-r)) * 1500.0
            if centers[i,1] < r:
                forces[i,1] += (r - centers[i,1]) * 1500.0
            elif centers[i,1] > 1-r:
                forces[i,1] -= (centers[i,1] - (1-r)) * 1500.0
                
        centers += forces * 0.0004
        centers = np.clip(centers, 0.0, 1.0)
        
        # Grow radii when packing is stable
        if max_ov < 1e-8:
            radii *= 1.00015
            radii = np.clip(radii, 0.0, 0.25)
            
    return centers, radii

def penalty_fn(x_flat, W):
    """Smooth penalty function for boundary and non-overlap constraints."""
    X = x_flat.reshape(N, 3)
    xs, ys, rs = X[:,0], X[:,1], X[:,2]
    
    p = 0.0
    # Boundary penalties
    p += np.sum(np.maximum(0.0, rs - xs)**2)
    p += np.sum(np.maximum(0.0, xs + rs - 1.0)**2)
    p += np.sum(np.maximum(0.0, rs - ys)**2)
    p += np.sum(np.maximum(0.0, ys + rs - 1.0)**2)
    
    # Overlap penalties
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    dr = rs[:, None] + rs[None, :]
    
    mask = np.triu(np.ones((N,N), bool), k=1)
    p += np.sum(np.maximum(0.0, dr[mask] - dist[mask])**2)
    
    # Objective: maximize sum(radii) => minimize -sum(radii) + penalty
    return -np.sum(rs) + W * p

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    best_x = None
    best_score = np.inf
    
    # Multiple seeds to explore different local minima
    seeds = [0, 7, 14, 21, 28, 35]
    for seed in seeds:
        c, r = force_directed_init(seed)
        x0 = np.hstack([c.flatten(), r])
        
        # Small perturbation to break symmetry before gradient descent
        x0 += np.random.default_rng(seed).normal(0, 1e-4, x0.shape)
        
        # Progressive penalty annealing
        for W in [500, 5000, 20000]:
            res = minimize(penalty_fn, x0, args=(W,), method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 1500, 'ftol': 1e-12, 'iprint': -1})
            x0 = res.x
            
        score = penalty_fn(x0, 20000)
        if score < best_score:
            best_score = score
            best_x = x0.copy()
            
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x.reshape(N, 3)[:, 2]
    radii = np.maximum(radii, 0.0)
    
    # Final strict validation & numerical correction
    for _ in range(15):
        valid = True
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    radii[i] *= 0.99995
                    radii[j] *= 0.99995
                    valid = False
            # Boundary corrections
            if centers[i,0] < radii[i] - 1e-9 or centers[i,0] > 1.0 - radii[i] + 1e-9:
                radii[i] = min(centers[i,0], 1.0 - centers[i,0])
                valid = False
            if centers[i,1] < radii[i] - 1e-9 or centers[i,1] > 1.0 - radii[i] + 1e-9:
                radii[i] = min(centers[i,1], 1.0 - centers[i,1])
                valid = False
        if valid: break
            
    return centers, radii, float(np.sum(radii))
