import numpy as np
from scipy.optimize import minimize

N = 26

def compute_forces(centers, radii):
    """Computes repulsive forces between circles and from boundaries."""
    forces = np.zeros_like(centers)
    
    # Pairwise repulsion
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-12)
    min_dists = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap = np.maximum(0.0, min_dists - dists)
    
    dirs = diffs / dists[:, :, np.newaxis]
    force_mag = overlap * 500.0
    forces += np.sum(force_mag[:, :, np.newaxis] * dirs, axis=1)
    
    # Boundary repulsion
    for i in range(N):
        r = radii[i]
        c = centers[i]
        if c[0] < r:
            forces[i, 0] += (r - c[0]) * 1000.0
        if c[0] > 1.0 - r:
            forces[i, 0] -= (c[0] - (1.0 - r)) * 1000.0
        if c[1] < r:
            forces[i, 1] += (r - c[1]) * 1000.0
        if c[1] > 1.0 - r:
            forces[i, 1] -= (c[1] - (1.0 - r)) * 1000.0
            
    return forces

def relax_packing(centers, radii, steps=1500):
    """Simulates circle expansion and repulsion to find a dense, valid packing."""
    centers = centers.copy()
    radii = radii.copy()
    vel = np.zeros_like(centers)
    
    for _ in range(steps):
        forces = compute_forces(centers, radii)
        vel = 0.2 * vel + 0.8 * forces * 0.0005
        centers += vel
        
        # Explicitly resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.sqrt(dx*dx + dy*dy)
                min_d = radii[i] + radii[j]
                if d < min_d and d > 1e-9:
                    overlap = min_d - d
                    shift = overlap * 0.51
                    nx, ny = dx / d, dy / d
                    centers[i, 0] -= nx * shift
                    centers[i, 1] -= ny * shift
                    centers[j, 0] += nx * shift
                    centers[j, 1] += ny * shift
                    
        # Enforce boundary constraints strictly
        for i in range(N):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
            
        # Gradually grow radii to pack tightly
        radii *= 1.00005
        
    return centers, radii

def objective_func(vars_flat):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars_flat[2::3])

def constraint_func(vars_flat):
    """Computes all boundary and non-overlap constraints."""
    centers = vars_flat[:2 * N].reshape(N, 2)
    radii = vars_flat[2 * N:]
    
    con = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    for i in range(N):
        con.append(centers[i, 0] - radii[i])
        con.append(1.0 - centers[i, 0] - radii[i])
        con.append(centers[i, 1] - radii[i])
        con.append(1.0 - centers[i, 1] - radii[i])
        
    # Overlap constraints: dist >= r_i + r_j
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            con.append(dist - radii[i] - radii[j])
            
    return np.array(con)

def get_bounds():
    """Returns variable bounds for x, y, r."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def generate_hex_init(seed):
    """Generates initial positions using a perturbed hexagonal lattice."""
    rng = np.random.default_rng(seed)
    centers = np.zeros((N, 2))
    idx = 0
    r_init = 0.09
    dy = r_init * np.sqrt(3)
    y = r_init
    row = 0
    
    while idx < N:
        x = r_init if row % 2 == 0 else 2 * r_init
        x_curr = x
        while x_curr <= 1.0 - r_init and idx < N:
            centers[idx, 0] = x_curr + rng.uniform(-0.005, 0.005)
            centers[idx, 1] = y + rng.uniform(-0.005, 0.005)
            idx += 1
            x_curr += 2 * r_init
        y += dy
        row += 1
        
    centers = np.clip(centers, 0.01, 0.99)
    radii = np.full(N, 0.035)
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_vars = None
    best_obj = -np.inf
    
    # Multiple restarts from different perturbed hexagonal lattices
    seeds = list(range(20))
    for seed in seeds:
        centers, radii = generate_hex_init(seed)
        
        # Relax to find a valid, dense configuration
        centers_rel, radii_rel = relax_packing(centers, radii, steps=1500)
        
        # Prepare initial vector for SLSQP
        # Slightly shrink radii to ensure strict feasibility for the optimizer start
        safe_radii = np.maximum(0.001, radii_rel - 0.005)
        x0 = np.zeros(N * 3)
        for i in range(N):
            x0[3 * i] = centers_rel[i, 0]
            x0[3 * i + 1] = centers_rel[i, 1]
            x0[3 * i + 2] = safe_radii[i]
            
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2500, 'ftol': 1e-13, 'disp': False})
            
            if res.success:
                c_vals = constraint_func(res.x)
                # Check feasibility with tolerance
                if np.all(c_vals >= -1e-8):
                    curr_obj = -res.fun
                    if curr_obj > best_obj:
                        best_obj = curr_obj
                        best_vars = res.x.copy()
        except Exception:
            pass
            
    # Fallback if optimization fails entirely
    if best_vars is None:
        centers_fb, radii_fb = generate_hex_init(0)
        centers_fb, radii_fb = relax_packing(centers_fb, radii_fb, steps=1500)
        best_vars = np.zeros(N * 3)
        for i in range(N):
            best_vars[3 * i] = centers_fb[i, 0]
            best_vars[3 * i + 1] = centers_fb[i, 1]
            best_vars[3 * i + 2] = radii_fb[i]

    centers = best_vars[:2 * N].reshape(N, 2)
    radii = best_vars[2 * N:]
    
    # Final safety check and minor shrink to guarantee validation passes
    c_vals = constraint_func(best_vars)
    min_c = np.min(c_vals)
    if min_c < -1e-9:
        # Scale radii down slightly if constraints are violated
        radii *= 0.9999
        centers = np.clip(centers, radii, 1.0 - radii)
        
    sum_r = float(np.sum(radii))
    return centers, radii, sum_r
