import numpy as np
from scipy.optimize import minimize
import itertools

N = 26

# Precompute pairwise indices for efficient constraint evaluation
PAIR_IDXS = np.array(list(itertools.combinations(range(N), 2))).T

def get_objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_constraints(x):
    """Compute all boundary and non-overlap constraints (must be >= 0)."""
    X = x.reshape(N, 3)
    xs, ys, rs = X[:, 0], X[:, 1], X[:, 2]
    
    c = np.empty(N*4 + 325)
    idx = 0
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c[idx:idx+N] = xs - rs; idx += N
    c[idx:idx+N] = 1.0 - xs - rs; idx += N
    c[idx:idx+N] = ys - rs; idx += N
    c[idx:idx+N] = 1.0 - ys - rs; idx += N
    
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    i_idx = PAIR_IDXS[0]
    j_idx = PAIR_IDXS[1]
    dx = xs[i_idx] - xs[j_idx]
    dy = ys[i_idx] - ys[j_idx]
    dr = rs[i_idx] + rs[j_idx]
    c[idx:] = dx**2 + dy**2 - dr**2
    
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': get_constraints}
    
    best_x = None
    best_val = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice
    hex_pts = []
    r0 = 0.09
    row = 0
    while len(hex_pts) < N:
        y = r0 + row * r0 * np.sqrt(3)
        if y + r0 > 1.0: break
        x_start = r0 if row % 2 == 0 else 2*r0
        x = x_start
        while x + r0 <= 1.0 and len(hex_pts) < N:
            hex_pts.append([x, y])
            x += 2*r0
        row += 1
    while len(hex_pts) < N:
        hex_pts.append([0.5, 0.5])
    inits.append(np.array(hex_pts[:N]))
    
    # 2. Random dense configurations
    for seed in range(30):
        np.random.seed(seed)
        pts = np.random.rand(N, 2) * 0.8 + 0.1
        inits.append(pts)
        
    # 3. Perturbed hexagonal
    for seed in range(15):
        np.random.seed(1000 + seed)
        p = inits[0].copy()
        p += np.random.normal(0, 0.025, p.shape)
        p = np.clip(p, 0.05, 0.95)
        inits.append(p)
        
    # Multi-start optimization
    for init_centers in inits:
        # Compute a safe initial radius to ensure feasible start
        min_gap = 1.0
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(init_centers[i,0]-init_centers[j,0], init_centers[i,1]-init_centers[j,1])
                if d < min_gap: min_gap = d
            d_b = min(init_centers[i,0], 1-init_centers[i,0], init_centers[i,1], 1-init_centers[i,1])
            if d_b < min_gap: min_gap = d_b
            
        r_start = max(0.005, 0.85 * min_gap / 2.0)
        
        x0 = np.zeros(N*3)
        for i in range(N):
            x0[3*i] = init_centers[i, 0]
            x0[3*i+1] = init_centers[i, 1]
            x0[3*i+2] = r_start
            
        try:
            res = minimize(get_objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons_dict,
                           options={'maxiter': 3000, 'ftol': 1e-11, 'disp': False})
            
            # Check if solution is valid and improves best
            if res.success or np.all(get_constraints(res.x) >= -1e-7):
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Local perturbation refinement to escape local minima
    if best_x is not None:
        for _ in range(10):
            pert_x = best_x.copy()
            # Perturb centers slightly
            pert_x[:2*N] += np.random.normal(0, 0.002, 2*N)
            pert_x[:2*N] = np.clip(pert_x[:2*N], 0.01, 0.99)
            
            try:
                res = minimize(get_objective, pert_x, method='SLSQP', bounds=bounds,
                               constraints=cons_dict,
                               options={'maxiter': 2000, 'ftol': 1e-11, 'disp': False})
                if res.success and -res.fun > best_val:
                    best_val = -res.fun
                    best_x = res.x.copy()
            except Exception:
                pass
                
    # Fallback to a safe grid configuration if optimization fails
    if best_x is None:
        x0 = np.zeros(N*3)
        idx = 0
        for i in range(5):
            for j in range(5):
                if idx < N:
                    x0[3*idx] = 0.1 + i*0.2
                    x0[3*idx+1] = 0.1 + j*0.2
                    x0[3*idx+2] = 0.09
                    idx += 1
        if idx < N:
            x0[3*idx] = 0.5; x0[3*idx+1] = 0.5; x0[3*idx+2] = 0.05
        best_x = x0
        best_val = np.sum(best_x[2::3])
        
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x.reshape(N, 3)[:, 2]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
