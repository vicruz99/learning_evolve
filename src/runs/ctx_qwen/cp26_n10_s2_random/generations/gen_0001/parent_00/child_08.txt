import numpy as np
from scipy.optimize import minimize

N = 26

def compute_constraints(vars_flat):
    """Computes boundary and non-overlap constraints for the packing."""
    X = vars_flat.reshape(N, 3)
    xs, ys, rs = X[:, 0], X[:, 1], X[:, 2]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c1 = xs - rs
    c2 = 1.0 - xs - rs
    c3 = ys - rs
    c4 = 1.0 - ys - rs
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    idx = np.triu_indices(N, k=1)
    dx = xs[idx[0]] - xs[idx[1]]
    dy = ys[idx[0]] - ys[idx[1]]
    dr = rs[idx[0]] + rs[idx[1]]
    c5 = dx**2 + dy**2 - dr**2
    
    return np.concatenate([c1, c2, c3, c4, c5])

def objective(vars_flat):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars_flat[2::3])

def get_bounds():
    """Returns variable bounds for x, y, r."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def generate_hex_init():
    """Generates a dense hexagonal lattice initialization."""
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.095)
    idx = 0
    y = 0.1
    row = 0
    while idx < N:
        shift = 0.1 if row % 2 == 1 else 0.0
        x = 0.1 + shift
        while x <= 0.9 and idx < N:
            centers[idx] = [x, y]
            idx += 1
            x += 0.2
        y += 0.1 * np.sqrt(3)
        row += 1
    return centers[:N], radii[:N]

def run_force_relaxation(centers, radii, steps=250):
    """Resolves overlaps using repulsive forces to ensure feasibility."""
    diffs = centers.reshape(N, 1, 2) - centers.reshape(1, N, 2)
    dists = np.linalg.norm(diffs, axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = radii[:, None] + radii[None, :]
    
    for _ in range(steps):
        safe_dists = np.where(dists < 1e-9, 1e-9, dists)
        overlap = np.maximum(0, min_dists - safe_dists)
        
        # Repulsion forces
        force_mag = overlap * 50.0
        force_dir = diffs / safe_dists[:, :, None]
        forces = np.sum(force_mag[:, :, None] * force_dir, axis=1)
        
        # Boundary repulsion
        for i in range(N):
            for d in range(2):
                if centers[i, d] < radii[i]:
                    forces[i, d] += (radii[i] - centers[i, d]) * 100.0
                elif centers[i, d] > 1.0 - radii[i]:
                    forces[i, d] -= (centers[i, d] - (1.0 - radii[i])) * 100.0
                    
        centers += forces * 0.005
        centers = np.clip(centers, radii, 1.0 - radii)
        
        # Update distances for next iteration
        diffs = centers.reshape(N, 1, 2) - centers.reshape(1, N, 2)
        dists = np.linalg.norm(diffs, axis=2)
        np.fill_diagonal(dists, np.inf)
        
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_vars = None
    best_obj = -np.inf
    
    # Strategy 1: Perturbed Hexagonal Lattices
    for seed in range(6):
        np.random.seed(seed)
        centers, radii = generate_hex_init()
        centers += np.random.normal(0, 0.008, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        
        x0 = np.zeros(N * 3)
        for i in range(N):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False})
            
            if np.all(compute_constraints(res.x) >= -1e-6):
                obj = -res.fun
                if obj > best_obj:
                    best_obj = obj
                    best_vars = res.x.copy()
        except Exception:
            pass

    # Strategy 2: Random Dense Starts + Relaxation
    for seed in range(4):
        np.random.seed(42 + seed)
        centers = np.random.rand(N, 2) * 0.8 + 0.1
        radii = np.full(N, 0.08)
        centers = run_force_relaxation(centers, radii, steps=200)
        
        x0 = np.zeros(N * 3)
        for i in range(N):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False})
            
            if np.all(compute_constraints(res.x) >= -1e-6):
                obj = -res.fun
                if obj > best_obj:
                    best_obj = obj
                    best_vars = res.x.copy()
        except Exception:
            pass

    # Strategy 3: Local Search around Best Found
    if best_vars is not None:
        for _ in range(5):
            x_pert = best_vars + np.random.normal(0, 0.0015, best_vars.shape)
            x_pert[2::3] = np.maximum(x_pert[2::3], 0.005) # Ensure positive radii
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
                
                if np.all(compute_constraints(res.x) >= -1e-6):
                    obj = -res.fun
                    if obj > best_obj:
                        best_obj = obj
                        best_vars = res.x.copy()
            except Exception:
                pass

    # Fallback if all optimizations failed
    if best_vars is None:
        c, r = generate_hex_init()
        best_vars = np.zeros(N * 3)
        for i in range(N):
            best_vars[3*i] = c[i, 0]
            best_vars[3*i+1] = c[i, 1]
            best_vars[3*i+2] = r[i]

    centers = best_vars.reshape(N, 3)[:, :2]
    radii = best_vars.reshape(N, 3)[:, 2]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
