import numpy as np
from scipy.optimize import minimize

N = 26

def compute_constraints(vars_flat):
    """Computes all boundary and non-overlap constraints for the packing."""
    X = vars_flat.reshape(N, 3)
    xs = X[:, 0]
    ys = X[:, 1]
    rs = X[:, 2]

    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c1 = xs - rs
    c2 = 1.0 - xs - rs
    c3 = ys - rs
    c4 = 1.0 - ys - rs

    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    idx = np.triu_indices(N, k=1)
    dx = xs[idx[0]] - xs[idx[1]]
    dy = ys[idx[0]] - ys[idx[1]]
    dr = rs[idx[0]] + rs[idx[1]]
    c5 = dx**2 + dy**2 - dr**2

    return np.concatenate([c1, c2, c3, c4, c5])

def compute_objective(vars_flat):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars_flat[2::3])

def get_bounds():
    """Returns variable bounds for x, y, r."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def generate_hex_init(scale_factor=1.0, jitter=0.0):
    """Generates a hexagonal lattice initial configuration."""
    r = 0.085 * scale_factor
    centers = []
    y = r
    # 5 rows: 5, 5, 5, 5, 6 circles = 26
    for row in range(5):
        count = 5 if row < 4 else 6
        shift = r if row % 2 == 1 else 0.0
        for col in range(count):
            x = r + col * 2 * r + shift
            centers.append([x, y])
        y += r * np.sqrt(3)
    
    centers = np.array(centers[:N])
    if jitter > 0:
        centers += np.random.normal(0, jitter, centers.shape)
        centers = np.clip(centers, 0.01, 0.99)
        
    init = np.zeros(N * 3)
    for i in range(N):
        init[3*i] = centers[i, 0]
        init[3*i+1] = centers[i, 1]
        init[3*i+2] = 0.06  # Conservative initial radius for feasibility
    return init

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_vars = None
    best_obj = -np.inf
    
    # Generate diverse initial configurations
    configs = []
    
    # 1. Hexagonal grid with varying compression/expansion
    for scale in [0.95, 1.0, 1.05, 1.1]:
        configs.append(generate_hex_init(scale_factor=scale, jitter=0.005))
        
    # 2. Random dense packings
    for seed in [42, 123, 789, 1024]:
        np.random.seed(seed)
        c = np.random.rand(N, 2) * 0.8 + 0.1
        init = np.zeros(N*3)
        for i in range(N):
            init[3*i] = c[i, 0]
            init[3*i+1] = c[i, 1]
            init[3*i+2] = 0.06
        configs.append(init)
        
    # 3. Regular grid initialization
    np.random.seed(0)
    grid_x = np.linspace(0.15, 0.85, 6)
    grid_y = np.linspace(0.15, 0.85, 5)
    cx, cy = np.meshgrid(grid_x, grid_y)
    grid = np.column_stack((cx.flatten(), cy.flatten()))[:N]
    init = np.zeros(N*3)
    for i in range(N):
        init[3*i] = grid[i, 0]
        init[3*i+1] = grid[i, 1]
        init[3*i+2] = 0.07
    configs.append(init)

    # Optimization across all configurations
    for x0 in configs:
        try:
            res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            
            # Verify strict feasibility within tolerance
            c_vals = compute_constraints(res.x)
            if np.min(c_vals) >= -1e-7:
                curr_obj = -res.fun
                if curr_obj > best_obj:
                    best_obj = curr_obj
                    best_vars = res.x.copy()
        except Exception:
            pass

    # Fallback if all optimizations fail
    if best_vars is None:
        best_vars = generate_hex_init(scale_factor=1.0, jitter=0.0)

    # Iterative expansion refinement to escape local optima
    radii = best_vars.reshape(N, 3)[:, 2]
    for _ in range(50):
        trial_radii = radii * 1.0015
        trial_vars = best_vars.copy()
        trial_vars[2::3] = trial_radii
        
        try:
            res = minimize(compute_objective, trial_vars, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
            
            if np.min(compute_constraints(res.x)) >= -1e-7:
                if -res.fun > best_obj:
                    best_obj = -res.fun
                    best_vars = res.x.copy()
                    radii = best_vars.reshape(N, 3)[:, 2]
                else:
                    break  # Cannot improve further
            else:
                break  # Overlaps cannot be resolved
        except Exception:
            break
            
    centers = best_vars.reshape(N, 2)
    radii = best_vars.reshape(N, 3)[:, 2]
    
    # Ensure non-negative radii and numerical safety
    radii = np.maximum(radii, 0.0)
    sum_r = float(np.sum(radii))
    
    return centers, radii, sum_r
