To maximize the sum of radii for 26 circles in a unit square, we need a robust optimization strategy that escapes local minima effectively. The problem is non-convex and highly constrained. A powerful approach for circle packing is an alternating optimization scheme:
1. **LP for Radii**: For fixed centers, maximizing the sum of radii subject to non-overlap and boundary constraints is a Linear Programming problem. This guarantees the optimal radii for any given configuration.
2. **Gradient Ascent for Centers**: The dual variables (marginals) from the LP solution directly give the repulsive forces needed to move centers apart to allow larger radii. We update centers using these forces.
3. **Hybrid Refinement**: After the LP/gradient phase converges to a strong local optimum, we use `SLSQP` for a final polynomial-time refinement to squeeze out the last bits of performance, handling the exact nonlinear constraints precisely.
4. **Multiple Restarts**: We run this process from multiple perturbed hexagonal initializations to ensure we find the global (or near-global) optimum.

This combination of exact LP subproblems, dual-driven gradient ascent, and nonlinear polishing is known to outperform standard black-box optimizers for packing problems.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
n_pairs = N * (N - 1) // 2
n_bnd = 4 * N
n_constraints = n_pairs + n_bnd

# Precompute LP structure
A_ub = np.zeros((n_constraints, N))
pair_indices = []
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        pair_indices.append((i, j))
        idx += 1
for i in range(N):
    A_ub[idx, i] = 1.0; idx += 1
    A_ub[idx, i] = 1.0; idx += 1
    A_ub[idx, i] = 1.0; idx += 1
    A_ub[idx, i] = 1.0; idx += 1

c_obj_lp = -np.ones(N)
bounds_r_lp = [(0.0, None)] * N

def solve_lp(centers):
    b_ub = np.zeros(n_constraints)
    k = 0
    for i, j in pair_indices:
        b_ub[k] = np.linalg.norm(centers[i] - centers[j])
        k += 1
    for i in range(N):
        b_ub[k] = centers[i, 0]; k += 1
        b_ub[k] = 1.0 - centers[i, 0]; k += 1
        b_ub[k] = centers[i, 1]; k += 1
        b_ub[k] = 1.0 - centers[i, 1]; k += 1
        
    res = linprog(c_obj_lp, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r_lp, method='highs')
    return res

def compute_grad(centers, res):
    grad = np.zeros((N, 2))
    if not (hasattr(res, 'success') and res.success and hasattr(res, 'ineqlin') and res.ineqlin is not None):
        return grad
    marg = res.ineqlin.marginals
    
    k = 0
    for i, j in pair_indices:
        lam = marg[k]
        if lam > 1e-8:
            d = np.linalg.norm(centers[i] - centers[j])
            if d > 1e-9:
                vec = centers[i] - centers[j]
                f = (lam / d) * vec
                grad[i] += f
                grad[j] -= f
        k += 1
        
    for i in range(N):
        lam_x = marg[n_pairs + 4*i]
        lam_1mx = marg[n_pairs + 4*i + 1]
        lam_y = marg[n_pairs + 4*i + 2]
        lam_1my = marg[n_pairs + 4*i + 3]
        grad[i, 0] += lam_x - lam_1mx
        grad[i, 1] += lam_y - lam_1my
    return grad

def get_init(seed):
    np.random.seed(seed)
    c = np.zeros((N, 2))
    idx = 0
    r = 0.1
    y = r
    row = 0
    while idx < N:
        sx = r if row % 2 == 0 else 2 * r
        x = sx
        while x + r <= 1.0 and idx < N:
            c[idx, 0] = x + np.random.normal(0, 0.005)
            c[idx, 1] = y + np.random.normal(0, 0.005)
            idx += 1
            x += 2 * r
        y += np.sqrt(3) * r
        row += 1
    while idx < N:
        c[idx, 0] = np.random.uniform(0.1, 0.9)
        c[idx, 1] = np.random.uniform(0.1, 0.9)
        idx += 1
    return np.clip(c, 0.01, 0.99)

def obj_slqp(p):
    return -np.sum(p[2::3])

def constr_overlap_slqp(p):
    x = p[0::3]; y = p[1::3]; r = p[2::3]
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    sum_r_sq = dr**2
    mask = np.triu(np.ones((N, N)), k=1).astype(bool)
    return (dist_sq - sum_r_sq)[mask]

def constr_boundary_slqp(p):
    x = p[0::3]; y = p[1::3]; r = p[2::3]
    return np.concatenate([x - r, 1 - x - r, y - r, 1 - y - r, r])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -1.0
    best_c = None
    best_r = None
    
    # Phase 1: LP + Gradient Ascent
    for seed in range(15):
        centers = get_init(seed)
        for step in range(2000):
            res = solve_lp(centers)
            if not res.success:
                break
            radii = res.x
            cur_sum = -res.fun
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_c = centers.copy()
                best_r = radii.copy()
                
            grad = compute_grad(centers, res)
            step_size = 0.05 * (0.999 ** step)
            if step_size < 0.0005:
                step_size = 0.0005
            centers += step_size * grad
            centers = np.clip(centers, 0.0, 1.0)
            
    # Phase 2: SLSQP Polish
    if best_c is not None:
        p0 = np.zeros(N * 3)
        for i in range(N):
            p0[3*i] = best_c[i, 0]
            p0[3*i+1] = best_c[i, 1]
            p0[3*i+2] = best_r[i]
            
        bnds = [(0, 1)]*(2*N) + [(0, 0.5)]*N
        cons = [{'type': 'ineq', 'fun': constr_overlap_slqp}, {'type': 'ineq', 'fun': constr_boundary_slqp}]
        try:
            res_s = minimize(obj_slqp, p0, method='SLSQP', bounds=bnds, constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13})
            p_opt = res_s.x
            c_f = np.array([[p_opt[3*i], p_opt[3*i+1]] for i in range(N)])
            r_f = p_opt[2::3].copy()
            if np.sum(r_f) > best_sum:
                best_sum = np.sum(r_f)
                best_c = c_f
                best_r = r_f
        except Exception:
            pass
            
    # Repair numerical violations
    for _ in range(200):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(best_c[i] - best_c[j])
                s = best_r[i] + best_r[j]
                if d < s - 1e-12:
                    sh = (s - d + 1e-12) / 2
                    best_r[i] -= sh
                    best_r[j] -= sh
                    changed = True
            x, y = best_c[i]
            mr = min(x, 1-x, y, 1-y)
            if best_r[i] > mr + 1e-12:
                best_r[i] = mr
                changed = True
            if best_r[i] < 0:
                best_r[i] = 0.0
                changed = True
        if not changed:
            break
    best_r = np.maximum(best_r, 0.0)
    return best_c, best_r, float(np.sum(best_r))
