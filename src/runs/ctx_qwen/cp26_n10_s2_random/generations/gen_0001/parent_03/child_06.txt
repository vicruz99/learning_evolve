import numpy as np
import scipy.optimize as opt

N = 26

def compute_constraints(p):
    """
    Computes all boundary and non-overlap constraints.
    Returns an array of constraint values that must be >= 0.
    p is a flat array of shape (N*3,) with [x1, y1, r1, x2, y2, r2, ...]
    """
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c_bound = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Non-overlap constraints: dist(i,j) >= r_i + r_j
    # Vectorized computation
    X = x[:, np.newaxis] - x[np.newaxis, :]
    Y = y[:, np.newaxis] - y[np.newaxis, :]
    R = r[:, np.newaxis] + r[np.newaxis, :]
    
    dist = np.sqrt(X**2 + Y**2)
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c_overlap = dist[mask] - R[mask]
    
    return np.concatenate([c_bound, c_overlap])

def objective(p):
    """Objective: Minimize -sum(radii)"""
    return -np.sum(p[2::3])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]"""
    bounds = []
    for _ in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return bounds

def generate_hex_init():
    """Generate a dense hexagonal lattice initialization."""
    p = np.zeros(N * 3)
    idx = 0
    r = 0.08  # Start safely inside
    y = r
    row = 0
    while idx < N:
        start_x = r if row % 2 == 0 else 2 * r
        x = start_x
        while x + r <= 1.0 + 1e-9 and idx < N:
            p[3*idx] = x
            p[3*idx+1] = y
            p[3*idx+2] = r
            idx += 1
            x += 2 * r
        y += np.sqrt(3) * r
        row += 1
    return p

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = get_bounds()
    cons = opt.NonlinearConstraint(compute_constraints, 0, np.inf)
    
    best_p = None
    best_val = -np.inf
    
    # Stage 1: Multiple initial configurations
    inits = [generate_hex_init()]
    
    # Add a few random/dense inits
    for _ in range(4):
        p_rand = np.zeros(N*3)
        for i in range(N):
            p_rand[3*i] = np.random.uniform(0.1, 0.9)
            p_rand[3*i+1] = np.random.uniform(0.1, 0.9)
            p_rand[3*i+2] = np.random.uniform(0.04, 0.08)
        inits.append(p_rand)
        
    # Stage 2: Optimization loop with perturbations
    all_configs = inits
    perturb_scale = 0.01
    
    for stage in range(15):
        for p_start in all_configs:
            # Apply perturbation
            p_pert = p_start.copy()
            p_pert[:2*N] += np.random.normal(0, perturb_scale, 2*N)
            p_pert[:2*N] = np.clip(p_pert[:2*N], 0.0, 1.0)
            # Ensure radii are valid for perturbed centers
            for i in range(N):
                x, y, r = p_pert[3*i], p_pert[3*i+1], p_pert[3*i+2]
                max_r = min(x, 1-x, y, 1-y)
                p_pert[3*i+2] = min(r, max_r)
            
            try:
                res = opt.minimize(
                    objective, p_pert, method='SLSQP', bounds=bounds,
                    constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False}
                )
                
                if res.success or res.fun < best_val:
                    val = -res.fun
                    # Check feasibility
                    c = compute_constraints(res.x)
                    if np.all(c >= -1e-8):
                        if val > best_val:
                            best_val = val
                            best_p = res.x.copy()
            except Exception:
                continue
        
        # Decay perturbation for finer search
        perturb_scale *= 0.7
        
        # If we found a good solution, add perturbed versions of it to the pool
        if best_p is not None:
            all_configs.append(best_p)
            
    if best_p is None:
        best_p = generate_hex_init()
        
    # Extract final results
    centers = np.array([[best_p[3*i], best_p[3*i+1]] for i in range(N)])
    radii = best_p[2::3].copy()
    
    # Final Repair: Ensure strict validity within tolerance
    for _ in range(100):
        changed = False
        # Fix boundary violations
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1-x, y, 1-y)
            if r > max_r + 1e-11:
                radii[i] = max_r
                changed = True
                
        # Fix overlaps by reducing radii slightly
        for i in range(N):
            for j in range(i+1, N):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                sum_r = radii[i] + radii[j]
                if dist < sum_r - 1e-11:
                    reduction = (sum_r - dist) * 0.51
                    radii[i] -= reduction
                    radii[j] -= reduction
                    changed = True
                    
        if not changed:
            break
            
    # Ensure non-negative
    radii = np.maximum(radii, 0.0)
    sum_r = float(np.sum(radii))
    
    return centers, radii, sum_r
