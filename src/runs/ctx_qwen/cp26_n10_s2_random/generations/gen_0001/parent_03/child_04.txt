import numpy as np
import scipy.optimize as opt

N = 26

def compute_constraints(p):
    """
    Computes all constraint values. Returns a 1D array where each element must be >= 0.
    Constraints:
    1. Boundary: r <= x <= 1-r, r <= y <= 1-r
    2. Non-overlap: dist(i,j)^2 >= (r_i + r_j)^2
    """
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0, r >= 0
    c_bnd = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r, r])
    
    # Overlap constraints using vectorized operations
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    dist2 = dx**2 + dy**2
    sum_r2 = dr**2
    
    # Upper triangle mask for unique pairs (i < j)
    mask = np.triu(np.ones((N, N)), k=1).astype(bool)
    c_ov = (dist2 - sum_r2)[mask]
    
    return np.concatenate([c_bnd, c_ov])

def objective(p):
    """Objective function: Minimize negative sum of radii."""
    return -np.sum(p[2::3])

def run_single_opt(p0, bnds, cons):
    """Runs a single optimization step and returns the result."""
    try:
        res = opt.minimize(objective, p0, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
        return res.x, -res.fun
    except Exception:
        # Fallback to initial state if optimizer fails
        return p0, -np.sum(p0[2::3])

def run_packing() -> tuple:

    best_sum = -1.0

    # 1. Hexagonal Lattice Initialization

    p_hex = np.zeros(3 * N)

    idx = 0
    y = r_init
    row = 0
    while idx < N:

        x = start_x
        while x + r_init <= 1.0 + 1e-9 and idx < N:
            p_hex[3 * idx] = x
            p_hex[3 * idx + 1] = y
            p_hex[3 * idx + 2] = r_init
            idx += 1
            x += 2.0 * r_init
        y += np.sqrt(3.0) * r_init
        row += 1

    p_opt, val = run_single_opt(p_hex, bnds, cons)
    if val > best_sum:
        best_sum = val
        best_p = p_opt.copy()

    # 2. Multi-Start with Iterative Perturbation
    # ---------------------------------------------------------
    for seed in range(20):

        # Random feasible start: small radii ensure no initial overlaps
        p0 = np.random.rand(3 * N)
        p0[2::3] = 0.02  

        curr_p = p0
        # Perform 3 optimization-perturbation cycles per seed
        for _ in range(3):
            curr_p, val = run_single_opt(curr_p, bnds, cons)
            if val > best_sum:
                best_sum = val
                best_p = curr_p.copy()

            # Shift centers slightly
            curr_p[0::3] += np.random.normal(0.0, 0.005, N)

            # Clamp centers to valid range

            curr_p[1::3] = np.clip(curr_p[1::3], 0.01, 0.99)

            curr_p[2::3] *= 0.9
