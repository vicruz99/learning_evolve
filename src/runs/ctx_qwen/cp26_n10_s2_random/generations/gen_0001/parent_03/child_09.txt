from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(p):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(p[2::3])

def compute_constraints(p):
    """Computes all boundary and non-overlap constraints (must be >= 0)."""
    n = N_CIRCLES
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    c = []
    # Boundary constraints
    c.extend(x - r)
    c.extend(1.0 - x - r)
    c.extend(y - r)
    c.extend(1.0 - y - r)
    
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    d2 = dx**2 + dy**2
    dr2 = dr**2
    triu_idx = np.triu_indices(n, k=1)
    c.extend(d2[triu_idx] - dr2[triu_idx])
    
    return np.array(c)

def run_optimizer(p0, bounds, cons):
    """Runs SLSQP optimization and returns result or None on failure."""
    try:
        res = minimize(compute_objective, p0, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12})
        return res
    except Exception:
        return None
