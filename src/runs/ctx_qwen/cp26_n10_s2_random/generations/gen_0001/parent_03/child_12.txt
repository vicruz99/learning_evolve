import numpy as np
import scipy.optimize as opt

def objective_func(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[2::3])

def constraint_func(p, n):
    """
    Computes all boundary and non-overlap constraints.
    Returns array of values that must be >= 0.
    """
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Boundary constraints: x-r>=0, 1-x-r>=0, y-r>=0, 1-y-r>=0, r>=0
    c_bound = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r,
        r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    sum_r_sq = dr**2
    
    # Upper triangle mask to avoid duplicates and self-comparison
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c_overlap = (dist_sq - sum_r_sq)[mask]
    
    return np.concatenate([c_bound, c_overlap])

def constraint_func_26(p):
    """Wrapper for constraint function with N=26 fixed."""
    return constraint_func(p, 26)

def run_packing() -> tuple:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Multiple restarts to escape local minima
    for seed in range(7):
        np.random.seed(seed)
        
        # 1. Hexagonal initialization with small noise
        centers = np.zeros((n, 2))
        idx = 0
        r_init = 0.04
        y = r_init
        row = 0
        while idx < n:
            x = r_init + (row % 2) * r_init * 0.85
            while x < 1.0 - r_init and idx < n:
                centers[idx] = [x + np.random.normal(0, 0.002), y + np.random.normal(0, 0.002)]
                idx += 1
                x += r_init * 1.72
            y += r_init * 1.48
            row += 1
            
        radii = np.full(n, r_init)
        
        # 2. Force-directed simulation with radius growth
        dt = 0.008
        growth = 8e-6
        repulsion = 800.0
        damping = 0.9
        steps = 12000
        
        velocities = np.zeros_like(centers)
        
        for step in range(steps):
            radii += growth
            
            # Vectorized pairwise repulsion
            diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            min_dists = radii[:, np.newaxis] + radii[np.newaxis, :]
            
            overlap = np.maximum(0, min_dists - dists)
            dists_safe = np.maximum(dists, 1e-8)
            dir_norm = diff / dists_safe[:, :, np.newaxis]
            
            force_mag = overlap * repulsion
            forces = np.sum(force_mag[:, :, np.newaxis] * dir_norm, axis=1)
            
            # Boundary repulsion
            for i in range(n):
                cx, cy = centers[i]
                r = radii[i]
                if cx - r < 0: forces[i, 0] += (r - cx) * repulsion * 2
                if cx + r > 1: forces[i, 0] -= (cx + r - 1) * repulsion * 2
                if cy - r < 0: forces[i, 1] += (r - cy) * repulsion * 2
                if cy + r > 1: forces[i, 1] -= (cy + r - 1) * repulsion * 2
                
            velocities = damping * velocities + forces * dt
            centers += velocities * dt
            
        # 3. SLSQP refinement
        # Shrink slightly to guarantee feasible start for optimizer
        radii *= 0.95
        centers = np.clip(centers, radii, 1.0 - radii)
        
        p0 = np.zeros(n * 3)
        for i in range(n):
            p0[3*i] = centers[i, 0]
            p0[3*i+1] = centers[i, 1]
            p0[3*i+2] = radii[i]
            
        bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
        cons = {'type': 'ineq', 'fun': constraint_func_26}
        
        try:
            res = opt.minimize(objective_func, p0, method='SLSQP', 
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 800, 'ftol': 1e-12})
            p_opt = res.x
            
            c_opt = np.array([[p_opt[3*i], p_opt[3*i+1]] for i in range(n)])
            r_opt = p_opt[2::3].copy()
            r_opt = np.maximum(r_opt, 0)
            
            # Strict validation
            valid = True
            for i in range(n):
                if r_opt[i] < 1e-9: valid = False; break
                if c_opt[i,0]-r_opt[i] < -1e-10 or c_opt[i,0]+r_opt[i] > 1+1e-10: valid = False; break
                if c_opt[i,1]-r_opt[i] < -1e-10 or c_opt[i,1]+r_opt[i] > 1+1e-10: valid = False; break
            
            if valid:
                for i in range(n):
                    for j in range(i+1, n):
                        d = np.sqrt((c_opt[i]-c_opt[j])**2)
                        if d < r_opt[i] + r_opt[j] - 1e-10:
                            valid = False
                            break
                    if not valid: break
                    
            if valid and np.sum(r_opt) > best_sum:
                best_sum = np.sum(r_opt)
                best_centers = c_opt
                best_radii = r_opt
                
        except Exception:
            pass
            
    # Fallback configuration
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.full(n, 0.075)
        idx = 0
        for r in range(6):
            for c in range(5):
                if idx < n:
                    best_centers[idx] = [0.075 + c*0.15, 0.075 + r*0.15]
                    idx += 1
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
