import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective_func(p):
    """Objective: Minimize negative sum of radii to maximize total radius."""
    return -np.sum(p[2::3])

def constraint_func(p):
    """
    Computes all boundary and non-overlap constraints.
    Constraints must be >= 0.
    """
    N = N_CIRCLES
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c_boundary = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    sum_r_sq = dr**2
    
    i_upper, j_upper = np.triu_indices(N, k=1)
    c_overlap = dist_sq[i_upper, j_upper] - sum_r_sq[i_upper, j_upper]
    
    return np.concatenate([c_boundary, c_overlap])

def generate_hex_init(seed):
    """Generates an initial configuration based on a perturbed hexagonal lattice."""
    np.random.seed(seed)
    N = N_CIRCLES
    centers = []
    r_init = 0.09
    y = r_init
    row = 0
    while len(centers) < N:
        x = r_init if row % 2 == 0 else 2 * r_init
        while x + r_init <= 1.0 and len(centers) < N:
            centers.append([x, y])
            x += 2 * r_init
        y += np.sqrt(3) * r_init
        row += 1
    centers = np.array(centers[:N])
    centers += np.random.normal(0, 0.01, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    r = np.full(N, r_init)
    return np.concatenate([centers.ravel(), r])

def run_packing():

    best_val = -np.inf

    cons = {'type': 'ineq', 'fun': constraint_func}

    for seed in range(60):

            p0 = generate_hex_init(seed)
        else:

            np.random.seed(seed)

            r = np.full(N, 0.05)
            p0 = np.concatenate([c.ravel(), r])

            if res.fun < best_val:
                p_cand = res.x

                    if centers[i,1] < radii[i] - 1e-9 or centers[i,1] > 1 - radii[i] + 1e-9: 
                        valid = False; break
