import numpy as np
from scipy.optimize import minimize

N = 26

def compute_constraints(x):
    """
    Computes all boundary and non-overlap constraints.
    Returns an array where each element must be >= 0 for a valid packing.
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c_bound = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2  =>  dist^2 - (r_i + r_j)^2 >= 0
    dx = cx[:, np.newaxis] - cx[np.newaxis, :]
    dy = cy[:, np.newaxis] - cy[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    
    d2 = dx**2 + dy**2
    r2 = dr**2
    
    # Only consider i < j to avoid duplicates and self-comparison
    mask = np.triu(np.ones((N, N)), k=1).astype(bool)
    c_overlap = d2[mask] - r2[mask]
    
    return np.concatenate([c_bound, c_overlap])

def compute_objective(x):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(x[2::3])

def hexagonal_init(perturb_scale):
    """Generates an initial configuration based on a hexagonal lattice."""
    r_est = 0.1
    pts = []
    y = r_est
    row = 0
    # Generate slightly more points than needed to ensure coverage
    while len(pts) < N + 5:
        x = r_est if row % 2 == 0 else 2.0 * r_est
        while x + r_est <= 1.0 + 1e-6:
            pts.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    centers = np.array(pts[:N])
    if perturb_scale > 0.0:
        centers += np.random.normal(0, perturb_scale, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        
    radii = np.full(N, 0.09)
    return np.concatenate([centers.flatten(), radii])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_params = None
    best_val = -np.inf
    
    np.random.seed(42)
    
    # Multiple restarts to escape local minima
    for i in range(50):
        if i < 15:
            # Low perturbation hex grid
            p0 = hexagonal_init(0.005 * (i + 1))
        elif i < 30:
            # Higher perturbation hex grid
            p0 = hexagonal_init(0.02 * (i + 1))
        else:
            # Random initialization
            c_rand = np.random.rand(N, 2) * 0.8 + 0.1
            r_rand = np.full(N, 0.08)
            p0 = np.concatenate([c_rand.flatten(), r_rand])
            
        try:
            res = minimize(compute_objective, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            
            # Accept if converged or sufficiently feasible
            if np.all(compute_constraints(res.x) >= -1e-5):
                current_val = -res.fun
                if current_val > best_val:
                    best_val = current_val
                    best_params = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization entirely fails
    if best_params is None:
        best_params = hexagonal_init(0.0)
        
    centers = best_params[:2 * N].reshape(N, 2)
    radii = best_params[2 * N:].copy()
    
    # Repair function to ensure strict validity within numerical tolerance
    for _ in range(200):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                sum_r = radii[i] + radii[j]
                if dist < sum_r - 1e-9:
                    shrink = (sum_r - dist) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True

            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if radii[i] > max_r + 1e-9:
                radii[i] = max_r
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
