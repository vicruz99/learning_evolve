import numpy as np
from scipy.optimize import minimize

N = 26

def objective(x):
    """Minimize negative sum of radii to maximize total sum."""
    return -np.sum(x[2::3])

def constraints(x):
    """
    Compute all inequality constraints.
    Format: [x1-r1, ..., xN-rN, 1-x1-r1, ..., 1-xN-rN, 
             y1-r1, ..., yN-rN, 1-y1-r1, ..., 1-yN-rN, 
             dist^2_ij - (ri+rj)^2 for i<j]
    All must be >= 0.
    """
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    
    c = []
    # Boundary constraints
    c.extend(xc - r)
    c.extend(1.0 - xc - r)
    c.extend(yc - r)
    c.extend(1.0 - yc - r)
    
    # Non-overlap constraints (squared distance formulation for smoothness)
    for i in range(N):
        for j in range(i + 1, N):
            dx = xc[i] - xc[j]
            dy = yc[i] - yc[j]
            dr = r[i] + r[j]
            c.append(dx * dx + dy * dy - dr * dr)
            
    return np.array(c)

def get_initial_guess(seed):
    """Generate a hexagonal lattice initialization with small random perturbation."""
    np.random.seed(seed)
    r0 = 0.10
    pts = []
    y = r0
    row = 0
    while len(pts) < N:
        shift = r0 if row % 2 == 1 else 0.0
        x = r0 + shift
        while x + r0 <= 1.0 + 1e-9:
            pts.append([x, y])
            if len(pts) >= N:
                break
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    centers = np.array(pts[:N])
    centers += np.random.normal(0, 0.001, centers.shape)
    
    x0 = np.zeros(3 * N)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = r0
    return x0

def run_packing() -> tuple:
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_x = None
    best_val = -np.inf
    
    # Multiple restarts from perturbed hexagonal lattices
    seeds = [0, 1, 2, 3, 4, 5, 10, 15, 20, 25]
    for s in seeds:
        x0 = get_initial_guess(s)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12})
            if -res.fun > best_val:
                best_val = -res.fun
                best_x = res.x.copy()
        except Exception:
            continue
            
    # Fallback to first guess if optimization completely failed
    if best_x is None:
        best_x = get_initial_guess(0)
        
    centers = best_x.reshape(N, 3)[:, :2].copy()
    radii = best_x.reshape(N, 3)[:, 2].copy()
    
    # Strict repair loop to guarantee validation passes within tolerance
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                sum_r = radii[i] + radii[j]
                if dist < sum_r - 1e-11:
                    overlap = sum_r - dist
                    radii[i] -= overlap / 2
                    radii[j] -= overlap / 2
                    changed = True
                    
        for i in range(N):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx - 1e-11:
                radii[i] = mx
                changed = True
        if not changed:
            break
            
    # Ensure non-negativity
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
