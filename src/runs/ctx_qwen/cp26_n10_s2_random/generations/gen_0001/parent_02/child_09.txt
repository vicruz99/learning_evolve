import numpy as np
from scipy.optimize import minimize
import math

def compute_objective(vars_flat):
    """Objective function to minimize (negative sum of radii)."""
    n = 26
    radii = np.zeros(n)
    for i in range(n):
        radii[i] = vars_flat[3*i + 2]
    return -np.sum(radii)

def compute_constraints(vars_flat):
    """Computes all boundary and non-overlap constraints."""
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = vars_flat[3*i]
        centers[i, 1] = vars_flat[3*i + 1]
        radii[i] = vars_flat[3*i + 2]
    
    con = []
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        con.append(x - r)
        con.append(1.0 - x - r)
        con.append(y - r)
        con.append(1.0 - y - r)
        
    # Non-overlap constraints: distance^2 >= (r_i + r_j)^2
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        for j in range(i + 1, n):
            xj, yj = centers[j]
            rj = radii[j]
            dx = xi - xj
            dy = yi - yj
            dist_sq = dx*dx + dy*dy
            sum_r = ri + rj
            con.append(dist_sq - sum_r**2)
            
    return np.array(con)

def get_bounds():
    """Returns variable bounds for x, y, r."""
    n = 26
    b = []
    for _ in range(n):
        b.append((0.0, 1.0))
        b.append((0.0, 1.0))
        b.append((0.0, 0.5))
    return b

def generate_initial_guess(seed):

    np.random.seed(seed)
    centers = np.zeros((n, 2))
    
    r_est = 0.10
    dx = 2.0 * r_est
    dy = math.sqrt(3.0) * r_est
    
    idx = 0
    y = r_est
    row = 0
    while idx < n:
        x = r_est
        if row % 2 == 1:
            x += r_est
            
        while x < 1.0 - r_est and idx < n:
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
            x += dx
        y += dy
        row += 1
        
    while idx < n:
        centers[idx, 0] = np.random.uniform(0.1, 0.9)
        centers[idx, 1] = np.random.uniform(0.1, 0.9)
        idx += 1

        vars_flat[3*i] = centers[i, 0]
        vars_flat[3*i + 1] = centers[i, 1]
        vars_flat[3*i + 2] = radii[i]
        
    return vars_flat

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """

    bounds = get_bounds()
    
    best_vars = None
    best_obj = -np.inf

    seeds = list(range(20))
    
    for seed in seeds:
        x0 = generate_initial_guess(seed)
        
        try:
            res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': compute_constraints},
                           options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})

    return centers, radii, sum_r
