import numpy as np
from scipy.optimize import minimize
import math

def objective_func(x, n):
    """Objective: Maximize sum of radii -> Minimize negative sum."""
    return -np.sum(x[2 * n:])

def constraint_func(x, n):
    """
    Computes all inequality constraints (must be >= 0).
    Layout of x: [x1, y1, ..., xn, yn, r1, r2, ..., rn]
    """
    c = x[:2 * n].reshape(n, 2)
    r = x[2 * n:]
    vals = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    vals.append(c[:, 0] - r)
    vals.append(1.0 - c[:, 0] - r)
    vals.append(c[:, 1] - r)
    vals.append(1.0 - c[:, 1] - r)
    
    # Non-overlap constraints: dist(i, j) >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            vals.append(math.hypot(dx, dy) - (r[i] + r[j]))
            
    return np.concatenate(vals)

def get_bounds(n):
    """Returns variable bounds for optimization."""
    b = []
    for _ in range(n):
        # x, y in [0, 1], r in [0, 0.5]
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = get_bounds(n)
    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}
    
    best_sum = -1.0
    best_x = None
    
    inits = []
    
    # 1. Hexagonal lattice initialization (dense packing structure)
    r0 = 0.095
    pts = []
    y = r0
    row = 0
    while len(pts) < n:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x < 1.0 - r0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r0
        y += r0 * math.sqrt(3.0)
        row += 1
    pts = np.array(pts[:n])
    inits.append(np.concatenate([pts.flatten(), np.full(n, 0.08)]))
    
    # 2. Random dense initializations to explore different local minima
    for i in range(25):
        np.random.seed(i)
        c = np.random.rand(n, 2) * 0.8 + 0.1
        r = np.full(n, 0.05)
        inits.append(np.concatenate([c.flatten(), r]))
        
    # 3. Grid perturbation initialization
    np.random.seed(100)
    gx, gy = np.meshgrid(np.linspace(0.12, 0.88, 5), np.linspace(0.12, 0.88, 5))
    c_g = np.column_stack([gx.flatten(), gy.flatten()])
    c_g = np.vstack([c_g, [0.5, 0.5]])
    c_g += np.random.normal(0, 0.015, c_g.shape)
    c_g = np.clip(c_g, 0.05, 0.95)
    inits.append(np.concatenate([c_g.flatten(), np.full(n, 0.07)]))
    
    # Main optimization loop
    for x0 in inits:
        try:
            res = minimize(objective_func, x0, args=(n,), method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 3000, 'ftol': 1e-13})
            if -res.fun > best_sum:
                con_vals = constraint_func(res.x, n)
                # Accept if constraints are satisfied within numerical tolerance
                if np.min(con_vals) >= -1e-8:
                    best_sum = -res.fun
                    best_x = res.x.copy()
        except Exception:
            continue
            
    if best_x is None:
        best_x = inits[0]
        
    # Refinement phase: Iteratively expand radii and re-optimize positions
    for _ in range(3):
        c_ref = best_x[:2 * n].reshape(n, 2)
        r_ref = best_x[2 * n:]
        
        min_gap = 1.0
        for i in range(n):
            # Gaps to boundaries
            min_gap = min(min_gap, c_ref[i, 0] - r_ref[i], 1.0 - c_ref[i, 0] - r_ref[i],
                               c_ref[i, 1] - r_ref[i], 1.0 - c_ref[i, 1] - r_ref[i])
            # Gaps to other circles

                d = math.hypot(c_ref[i, 0] - c_ref[j, 0], c_ref[i, 1] - c_ref[j, 1])

                               options={'maxiter': 1500, 'ftol': 1e-13})

                best_x = res_ref.x.copy()
                
    centers = best_x[:2 * n].reshape(n, 2)
    radii = best_x[2 * n:]
    return centers, radii, float(np.sum(radii))
