import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars_flat):
    """Minimize negative sum of radii."""
    r = vars_flat[2::3]
    return -np.sum(r)

def constraint_func(vars_flat, n):
    """Compute all boundary and non-overlap constraints."""
    c = vars_flat.reshape(n, 3)
    cx, cy, r = c[:, 0], c[:, 1], c[:, 2]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Overlap constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    dist_sq = dx**2 + dy**2
    
    idx = np.triu_indices(n, k=1)
    o = dist_sq[idx] - dr[idx]**2
    
    return np.concatenate([b, o])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Variable bounds: x,y in [0,1], r in [epsilon, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    # Constraint dictionary without lambdas
    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}
    
    num_trials = 15
    
    for trial in range(num_trials):
        np.random.seed(trial * 137 + 42)
        
        # Hexagonal initialization for high packing density
        centers = np.zeros((n, 2))
        idx = 0
        r_init = 0.095
        y = r_init
        row = 0
        while idx < n:
            x_start = r_init if row % 2 == 0 else 2 * r_init
            x = x_start
            while x <= 1.0 - r_init and idx < n:
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
                x += 2 * r_init
            y += r_init * np.sqrt(3)
            row += 1
            
        # Perturb to break symmetry and avoid degenerate gradients
        centers += np.random.normal(0, 0.005, centers.shape)
        centers = np.clip(centers, 0.02, 0.98)
        radii = np.full(n, 0.08)
        
        # Flatten to optimizer format: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3 * i] = centers[i, 0]
            x0[3 * i + 1] = centers[i, 1]
            x0[3 * i + 2] = radii[i]
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False})
            
            c_opt = res.x.reshape(n, 3)
            centers_opt = c_opt[:, :2]
            radii_opt = c_opt[:, 2]
            
            # Strict validity check
            valid = True
            for i in range(n):
                if radii_opt[i] < 1e-8:
                    valid = False
                    break
                if (centers_opt[i, 0] < radii_opt[i] - 1e-9 or centers_opt[i, 0] > 1.0 - radii_opt[i] + 1e-9 or
                    centers_opt[i, 1] < radii_opt[i] - 1e-9 or centers_opt[i, 1] > 1.0 - radii_opt[i] + 1e-9):
                    valid = False
                    break
            if valid:
                for i in range(n):
                    for j in range(i + 1, n):
                        d = np.hypot(centers_opt[i, 0] - centers_opt[j, 0], centers_opt[i, 1] - centers_opt[j, 1])
                        if d < radii_opt[i] + radii_opt[j] - 1e-9:
                            valid = False
                            break
                    if not valid:
                        break
                        
            if valid:
                s = np.sum(radii_opt)
                if s > best_sum:
                    best_sum = s
                    best_c = centers_opt.copy()
                    best_r = radii_opt.copy()
                    
        except Exception:
            continue

    # Fallback if optimization fails
    if best_c is None:
        best_c = centers
        best_r = radii
        best_sum = np.sum(radii)
        
    # Uniform growth phase: squeeze out remaining space
    for _ in range(100):
        min_slack_bnd = np.inf
        min_slack_pair = np.inf
        
        for i in range(n):
            cx, cy, r = best_c[i, 0], best_c[i, 1], best_r[i]
            min_slack_bnd = min(min_slack_bnd, cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r)
            for j in range(i + 1, n):
                d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                min_slack_pair = min(min_slack_pair, d - best_r[i] - best_r[j])
                
        # Adding delta to radius consumes 1*delta from boundary slack and 2*delta from pairwise slack
        delta = min(min_slack_bnd, min_slack_pair / 2.0)
        if delta > 1e-8:
            best_r += delta
            best_sum += n * delta
        else:
            break
            
    # Final safety clamp to guarantee validation passes within tolerance
    for _ in range(5):
        changed = False
        for i in range(n):
            cx, cy, r = best_c[i, 0], best_c[i, 1], best_r[i]
            best_c[i, 0] = np.clip(cx, r + 1e-9, 1.0 - r - 1e-9)
            best_c[i, 1] = np.clip(cy, r + 1e-9, 1.0 - r - 1e-9)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                req_d = best_r[i] + best_r[j]
                if d < req_d - 1e-9:
                    shrink = (req_d - d) * 0.51
                    best_r[i] -= shrink
                    best_r[j] -= shrink
                    changed = True
        if not changed:
            break
            
    best_sum = np.sum(best_r)
    return best_c, best_r, float(best_sum)
