import numpy as np
from scipy.optimize import minimize

N = 26

def obj_func(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def con_boundary(v):
    """Boundary constraints: r <= x <= 1-r, r <= y <= 1-r."""
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    return np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])

def con_overlap(v):
    """Non-overlap constraints: dist^2 >= (r_i + r_j)^2."""
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    i, j = np.triu_indices(N, k=1)
    dx = xs[i] - xs[j]
    dy = ys[i] - ys[j]
    dr = rs[i] + rs[j]
    return dx**2 + dy**2 - dr**2

def generate_hex_init(seed=0):
    """Generate a perturbed hexagonal lattice initialization."""
    rng = np.random.default_rng(seed)
    r_approx = 0.06  # Start small to guarantee initial feasibility
    centers = []
    y = r_approx
    row = 0
    while len(centers) < N:
        x_start = r_approx if row % 2 == 0 else 2 * r_approx
        x = x_start
        while x <= 1.0 - r_approx and len(centers) < N:
            centers.append([x, y])
            x += 2 * r_approx
        y += r_approx * np.sqrt(3)
        row += 1
        
    centers = np.array(centers[:N])
    centers += rng.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    
    v = np.zeros(N * 3)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = r_approx
    return v

def generate_random_init(seed=0):
    """Generate a random dense initialization."""
    rng = np.random.default_rng(seed)
    v = np.zeros(N * 3)
    v[0::3] = rng.uniform(0.15, 0.85, N)

    return v
