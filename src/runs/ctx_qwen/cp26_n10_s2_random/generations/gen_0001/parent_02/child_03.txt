import numpy as np
from scipy.optimize import minimize
import math

def obj_func(vars_flat):
    """Objective: maximize sum of radii => minimize negative sum."""
    n = 26
    radii = vars_flat[2*n:]
    return -np.sum(radii)

def constraint_fun(vars_flat):
    """Computes all boundary and non-overlap constraints."""
    n = 26
    centers = vars_flat[:2*n].reshape(n, 2)
    radii = vars_flat[2*n:]
    
    cons = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # Formulated as: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    cons.append(centers[:, 0] - radii)
    cons.append(1.0 - centers[:, 0] - radii)
    cons.append(centers[:, 1] - radii)
    cons.append(1.0 - centers[:, 1] - radii)
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    # Vectorized pairwise distance calculation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # Ignore self-distances
    
    # Extract upper triangle to avoid duplicates and self-comparison
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pair_dists = dists[mask]
    pair_radii_sum = (radii[:, np.newaxis] + radii[np.newaxis, :])[mask]
    
    cons.append(pair_dists - pair_radii_sum)
    
    return np.concatenate([c.flatten() for c in cons])
