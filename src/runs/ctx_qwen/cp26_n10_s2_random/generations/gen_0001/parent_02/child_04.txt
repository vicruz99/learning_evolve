import numpy as np
from scipy.optimize import minimize
import math

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii (equivalent to maximizing sum of radii)."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Compute all boundary and non-overlap constraints (must be >= 0)."""
    n = N_CIRCLES
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Boundary constraints: r <= x <= 1-r  =>  x-r >= 0, 1-x-r >= 0
    c_x_min = centers[:, 0] - radii
    c_x_max = 1.0 - centers[:, 0] - radii
    c_y_min = centers[:, 1] - radii
    c_y_max = 1.0 - centers[:, 1] - radii
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap_c = dist_sq - radii_sum**2
    
    # Extract upper triangle to avoid duplicates and self-comparison
    triu_indices = np.triu_indices(n, k=1)
    overlap_c = overlap_c[triu_indices]
    
    return np.concatenate([c_x_min, c_x_max, c_y_min, c_y_max, overlap_c])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraints}
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    options = {'maxiter': 3000, 'ftol': 1e-12, 'disp': False}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # --- Generate Initial Configurations ---
    hex_pts = []
    r_approx = 0.09
    y = r_approx
    row = 0
    while len(hex_pts) < n:
        x = r_approx + (row % 2) * r_approx
        while x <= 1.0 - r_approx and len(hex_pts) < n:

            x += 2 * r_approx
        y += r_approx * math.sqrt(3)
        row += 1
    base_hex = np.array(hex_pts[:n])
    
    initial_centers_list = [base_hex]

    for i in range(10):
        np.random.seed(i)
        pert = base_hex + np.random.normal(0, 0.015, base_hex.shape)

        initial_centers_list.append(pert)

        np.random.seed(100 + i)

        initial_centers_list.append(rand_pts)

        x0[1:2*n:3] = centers_init[:, 1]
        x0[2:2*n:3] = r_init

                           constraints=cons, options=options)

                cur_sum = -res.fun
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers = res.x[:2*n].reshape(n, 2).copy()
                    best_radii = res.x[2*n:].copy()
        except Exception:
            continue
