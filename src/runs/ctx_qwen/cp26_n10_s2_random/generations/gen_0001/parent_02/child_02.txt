import numpy as np
import math
from scipy.optimize import minimize

def generate_hex_init(n, seed):
    """Generates a feasible initial configuration using a hexagonal grid with perturbation."""
    np.random.seed(seed)
    r_est = 0.095
    centers = []
    y = r_est
    row = 0
    while len(centers) < n:
        x = r_est if row % 2 == 0 else 2 * r_est
        while x < 1.0 - r_est and len(centers) < n:
            centers.append([x, y])
            x += 2 * r_est
        y += r_est * math.sqrt(3)
        row += 1
        
    centers = np.array(centers[:n])
    centers += np.random.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, r_est + 0.005, 1.0 - r_est - 0.005)
    return centers, np.full(n, r_est)

def objective_func(x, n):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum of radii)."""
    return -np.sum(x[2*n:])

def constraint_func(x, n):
    """Computes all boundary and non-overlap constraints as a flattened array (must be >= 0)."""
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    # Boundary constraints
    c1 = centers[:, 0] - radii
    c2 = 1.0 - centers[:, 0] - radii
    c3 = centers[:, 1] - radii
    c4 = 1.0 - centers[:, 1] - radii
    
    # Pairwise distance constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    dists = np.maximum(dists, 1e-12)  # Prevent numerical issues at zero distance
    np.fill_diagonal(dists, 1.0)      # Dummy value for diagonal (not used in extraction)
    
    sum_r = radii[:, np.newaxis] + radii[np.newaxis, :]
    c_overlap = dists - sum_r
    
    # Extract upper triangle to avoid duplicate constraints and self-comparison
    idx = np.triu_indices(n, k=1)
    c_overlap = c_overlap[idx]
    
    return np.concatenate([c1, c2, c3, c4, c_overlap])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}
    
    # Multiple restarts to escape local minima
    for seed in range(25):
        centers_init, radii_init = generate_hex_init(n, seed=seed)
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        
        try:
            res = minimize(
                objective_func,
                x0,
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
            )
            
            curr_sum = -res.fun
            if curr_sum > best_sum:
                # Verify feasibility within tolerance
                cons_vals = constraint_func(res.x, n)
                if np.min(cons_vals) >= -1e-6:
                    best_sum = curr_sum
                    best_centers = res.x[:2*n].reshape(n, 2)
                    best_radii = res.x[2*n:].copy()
        except Exception:
            continue
            
    # Fallback if optimization fails to find any valid solution
    if best_centers is None:
        centers_init, radii_init = generate_hex_init(n, seed=0)
        best_centers = centers_init
        best_radii = radii_init
        best_sum = np.sum(radii_init)
        
    return best_centers, best_radii, float(best_sum)
