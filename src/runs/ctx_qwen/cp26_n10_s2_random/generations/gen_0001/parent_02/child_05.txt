import numpy as np
import math
from scipy.optimize import minimize

N = 26

def objective(vars_flat):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2::3])

def constraints(vars_flat):
    """Compute all constraint values (must be >= 0)."""
    cx = vars_flat[0::3]
    cy = vars_flat[1::3]
    r = vars_flat[2::3]
    
    c = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.append(cx - r)
    c.append(1.0 - cx - r)
    c.append(cy - r)
    c.append(1.0 - cy - r)
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    diff_x = cx[:, np.newaxis] - cx[np.newaxis, :]
    diff_y = cy[:, np.newaxis] - cy[np.newaxis, :]
    dist = np.sqrt(diff_x**2 + diff_y**2)
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    c.append(dist[mask] - r_sum[mask])
    
    return np.concatenate(c)

def get_bounds():
    """Returns variable bounds for x, y, r."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def generate_init(seed):
    """Generates a feasible initial configuration."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = 0.1
    row = 0
    # Hexagonal grid initialization
    while idx < N and y < 0.9:
        x_start = 0.1 + (row % 2) * 0.09
        x = x_start
        while idx < N and x < 0.9:
            centers[idx] = [x, y]
            idx += 1
            x += 0.18
        y += 0.15 * math.sqrt(3)
        row += 1
        
    # Fill remaining if any
    while idx < N:
        centers[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
        
    # Perturb slightly to avoid exact symmetry
    centers += np.random.normal(0, 0.008, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    # Compute feasible initial radii based on geometry
    r_vals = np.full(N, 1.0)
    for i in range(N):
        r_vals[i] = min(centers[i, 0], 1.0 - centers[i, 0], 
                        centers[i, 1], 1.0 - centers[i, 1])
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                r_vals[i] = min(r_vals[i], d / 2.0)
    
    # Scale down slightly to ensure strict feasibility
    r_vals *= 0.92
    # Add small random variation to break symmetry
    r_vals *= np.random.uniform(0.95, 1.05, N)
    r_vals = np.clip(r_vals, 0.01, 0.4)
    
    return centers, r_vals

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Run multiple optimizations with different seeds to escape local minima
    for seed in range(25):
        init_c, init_r = generate_init(seed)
        x0 = np.zeros(3 * N)
        for i in range(N):
            x0[3 * i] = init_c[i, 0]
            x0[3 * i + 1] = init_c[i, 1]
            x0[3 * i + 2] = init_r[i]
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, 
                           options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
            
            if res.success:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = res.x[:2 * N].reshape(N, 2).copy()
                    best_radii = res.x[2 * N:].copy()
        except Exception:
            continue
            
    if best_centers is None:
        # Fallback to first initialization if all fail
        best_centers, best_radii = generate_init(0)
        best_sum = np.sum(best_radii)
        
    # Ensure radii are non-negative
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_sum)
