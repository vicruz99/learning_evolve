import numpy as np
from scipy.optimize import minimize
import math

N = 26

def objective(x):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)."""
    radii = x[2::3]
    return -np.sum(radii)

def constraints(x):
    """
    Computes all constraint values. Must be >= 0 for a valid packing.
    Layout of x: [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    """
    centers = x[:2*N].reshape(N, 2)
    radii = x[2*N:]
    
    # Boundary constraints: circles must be inside [0,1]x[0,1]
    c1 = centers[:, 0] - radii
    c2 = 1.0 - centers[:, 0] - radii
    c3 = centers[:, 1] - radii
    c4 = 1.0 - centers[:, 1] - radii
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    # Using squared distances for smoothness: dist^2 >= (r_i + r_j)^2
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_sq = np.sum(diff**2, axis=2)
    
    radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    radii_sum_sq = radii_sum**2
    
    # Only need upper triangle to avoid duplicates and self-comparison
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c5 = dists_sq[mask] - radii_sum_sq[mask]
    
    return np.concatenate([c1, c2, c3, c4, c5])

def bounds_setup():
    """Defines variable bounds for the optimizer."""
    b = []
    for _ in range(N):
        b.append((0.0, 1.0))  # x
        b.append((0.0, 1.0))  # y
        b.append((0.0, 0.5))  # r
    return b

def init_config(seed):
    """Generates a hexagonal lattice initial guess with random perturbation."""
    rng = np.random.default_rng(seed)
    r0 = 0.10
    centers = []
    y = r0
    row = 0
    
    while len(centers) < N:
        x = r0 if row % 2 == 0 else 2 * r0
        while x <= 1.0 - r0 and len(centers) < N:
            centers.append([x, y])
            x += 2 * r0
        y += r0 * math.sqrt(3)
        row += 1
        
    centers = np.array(centers[:N])
    # Add small noise to break symmetry and explore nearby configurations
    centers += rng.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    x0 = np.zeros(N * 3)
    for i in range(N):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = r0
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_x = None
    best_obj = -np.inf
    bnds = bounds_setup()
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Run multiple restarts to find the global optimum
    for seed in range(15):
        x0 = init_config(seed)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
            
            if -res.fun > best_obj:
                # Verify feasibility within numerical tolerance
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-7:
                    best_obj = -res.fun
                    best_x = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization unexpectedly fails
    if best_x is None:
        best_x = init_config(0)
        best_obj = -objective(best_x)
        
    centers = best_x[:2*N].reshape(N, 2)
    radii = best_x[2*N:]
    
    # Final safety adjustment to guarantee strict validity against the validator
    c_vals = constraints(best_x)
    min_c = np.min(c_vals)
    
    if min_c < -1e-9:
        scale = 1.0
        for _ in range(5000):
            scale -= 0.0001
            r_tmp = radii * scale
            
            # Quick validation check
            diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
            dists_sq = np.sum(diff**2, axis=2)
            rs_sq = (r_tmp[:, np.newaxis] + r_tmp[np.newaxis, :])**2
            mask = np.triu(np.ones((N, N), dtype=bool), k=1)
            
            overlap_ok = np.min(dists_sq[mask] - rs_sq[mask]) >= -1e-9
            boundary_ok = (np.all(centers[:,0] - r_tmp >= -1e-9) and 
                           np.all(1 - centers[:,0] - r_tmp >= -1e-9) and
                           np.all(centers[:,1] - r_tmp >= -1e-9) and 
                           np.all(1 - centers[:,1] - r_tmp >= -1e-9))
                           
            if overlap_ok and boundary_ok:
                radii = r_tmp
                best_obj = np.sum(radii)
                break
                
    return centers, radii, float(best_obj)
