import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26

def compute_lp_radii(centers):
    """Compute optimal radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    # Boundary constraints: r_i <= dist(c_i, edge)
    for i in range(n):
        for v in [centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1]]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(v)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-9), 0.0

def simulated_annealing_centers(centers_init):
    """Optimize center positions using Simulated Annealing with LP evaluation."""
    n = N_CIRCLES
    curr = centers_init.copy()
    curr_r, curr_s = compute_lp_radii(curr)
    best_c, best_r, best_s = curr.copy(), curr_r.copy(), curr_s
    
    temp = 0.02
    step = 0.015
    for _ in range(1000):
        idx = np.random.randint(n)
        new_c = curr.copy()
        new_c[idx] += np.random.normal(0, step, 2)
        new_c[idx] = np.clip(new_c[idx], 0.005, 0.995)
        
        new_r, new_s = compute_lp_radii(new_c)
        delta = new_s - curr_s
        if delta > 0 or np.random.rand() < np.exp(delta / max(temp, 1e-6)):
            curr = new_c
            curr_r, curr_s = new_r, new_s
            if curr_s > best_s:
                best_c, best_r, best_s = curr.copy(), curr_r.copy(), curr_s
        temp *= 0.999
        step *= 0.9995
    return best_c, best_r, best_s

def slsqp_refine_packing(centers, radii):
    """Refine the configuration using SLSQP on joint (centers, radii) variables."""
    n = N_CIRCLES
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]
        
    def obj_func(x):
        return -np.sum(x[2::3])
        
    def constraint_func(x):
        con = []
        # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
        for i in range(n):
            cx, cy, cr = x[3*i], x[3*i+1], x[3*i+2]
            con.append(cx - cr)
            con.append(1.0 - cx - cr)
            con.append(cy - cr)
            con.append(1.0 - cy - cr)
        # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
        for i in range(n):
            for j in range(i + 1, n):
                dx = x[3*i] - x[3*j]
                dy = x[3*i+1] - x[3*j+1]
                dr = x[3*i+2] + x[3*j+2]
                con.append(dx*dx + dy*dy - dr*dr)
        return np.array(con)
        
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 500, 'ftol': 1e-12})
    
    if res.success or np.all(constraint_func(res.x) >= -1e-7):
        c_out = np.column_stack((res.x[::3], res.x[1::3]))
        r_out = res.x[2::3]
        return c_out, r_out, -res.fun
    return centers, radii, -obj_func(x0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)

    c2 = np.vstack([c2, [0.5, 0.5]])
    c2 += np.random.normal(0, 0.01, c2.shape)
    starts.append(np.clip(c2, 0.05, 0.95))

        starts.append(np.random.rand(n, 2) * 0.8 + 0.1)

    for sc in starts:
        c, r, s = simulated_annealing_centers(sc)
        if s > best_s:
            best_c, best_r, best_s = c, r, s
            
    # Polish best result with SLSQP
    best_c, best_r, best_s = slsqp_refine_packing(best_c, best_r)
    
    # Ensure numerical safety for the validator
    best_r = np.maximum(best_r, 0.0)
    best_r *= 0.99999  # Guarantees strict inequality against tolerance checks
    
    return best_c, best_r, float(np.sum(best_r))
