import numpy as np
from scipy.optimize import linprog, minimize

def get_lp_sum_radii(centers):
    """Solve LP to maximize sum of radii given fixed centers."""
    n = centers.shape[0]
    # Keep centers strictly inside to avoid degenerate LP bounds
    c = np.clip(centers, 1e-5, 1.0 - 1e-5)
    xs = c[:, 0]
    ys = c[:, 1]
    # Maximum radius allowed by boundaries
    bounds_r = np.minimum(np.minimum(xs, 1.0 - xs), np.minimum(ys, 1.0 - ys))
    bounds_r = np.maximum(bounds_r, 0.0)
    
    # Pairwise Euclidean distances
    diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Construct LP: Max sum(r) <=> Min -sum(r)
    # Constraints: r_i + r_j <= dist_ij
    idxs = np.triu_indices(n, k=1)
    m = len(idxs[0])
    A_ub = np.zeros((m, n))
    A_ub[np.arange(m), idxs[0]] = 1.0
    A_ub[np.arange(m), idxs[1]] = 1.0
    b_ub = dists[idxs[0], idxs[1]]
    
    bounds = [(0.0, b) for b in bounds_r]
    
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def obj_func(x):
    """Objective for center optimization: negative sum of radii."""
    centers = x.reshape(-1, 2)
    s, _ = get_lp_sum_radii(centers)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice pattern
    hex_c = []
    y = 0.1
    row = 0
    while len(hex_c) < n:
        x = 0.1 if row % 2 == 0 else 0.15
        while x < 0.9 and len(hex_c) < n:
            hex_c.append([x, y])
            x += 0.18
        y += 0.15
        row += 1
    inits.append(np.array(hex_c[:n]))
    
    # 2. Perturbed hexagonal
    inits.append(hex_c[:n] + np.random.normal(0, 0.02, size=(n, 2)))

    grid_c = np.column_stack((gx.flatten(), gy.flatten()))
    grid_c = np.vstack([grid_c, [0.5, 0.5]])
    inits.append(grid_c[:n])
    
    # 4. Dense random
    inits.append(np.random.rand(n, 2) * 0.8 + 0.1)
    
    # Optimize centers using Nelder-Mead from each start
    for idx, init_c in enumerate(inits):
        np.random.seed(42 + idx)
        x0 = np.clip(init_c, 0.05, 0.95).flatten()
        try:
            res = minimize(obj_func, x0, method='Nelder-Mead', 
                           options={'maxiter': 6000, 'xatol': 1e-6, 'fatol': 1e-9})
            cur_sum, cur_r = get_lp_sum_radii(res.x.reshape(-1, 2))
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = res.x.reshape(-1, 2)
                best_radii = cur_r
        except Exception:
            pass
            
    # Final refinement using SLSQP on full variable set (positions + radii)
    if best_centers is not None:
        v0 = np.zeros(n * 3)
        for i in range(n):
            v0[3 * i] = best_centers[i, 0]
            v0[3 * i + 1] = best_centers[i, 1]
            v0[3 * i + 2] = best_radii[i]

            return -np.sum(v[2::3])

            c = v[:2 * n].reshape(n, 2)

            con = []
            # Boundary constraints
            con.extend(c[:, 0] - r)
            con.extend(1.0 - c[:, 0] - r)
            con.extend(c[:, 1] - r)
            con.extend(1.0 - c[:, 1] - r)
            # Non-overlap constraints
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                    con.append(d - r[i] - r[j])
            return np.array(con)

            res_full = minimize(full_obj, v0, method='SLSQP', bounds=bounds,

        except Exception:
            pass

    best_radii = np.maximum(best_radii, 0.0)
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    return best_centers, best_radii, float(best_sum)
