import numpy as np
from scipy.optimize import linprog

def solve_radii_lp(centers, idx_i, idx_j):
    """
    Given fixed centers, solve LP to maximize sum of radii.
    Constraints: r_i + r_j <= dist(i,j) and boundary limits.
    """
    n = centers.shape[0]
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    b = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    d_vals = dists[idx_i, idx_j]
    
    # Construct constraint matrix A_ub x <= b_ub
    num_ineq = len(idx_i) + n
    A_ub = np.zeros((num_ineq, n))
    
    # Pairwise: r_i + r_j <= dist_ij
    A_ub[:len(idx_i), idx_i] = 1.0
    A_ub[:len(idx_i), idx_j] = 1.0
    
    # Boundary: r_i <= b_i
    A_ub[len(idx_i):, np.arange(n)] = 1.0
    
    b_ub = np.concatenate([d_vals, b])
    
    # Maximize sum(r) -> Minimize -sum(r)
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x, -res.fun
    else:
        # Fallback for numerical edge cases
        return np.full(n, 1e-6), 26e-6

    y = r_init

        while x <= 1 - r_init and len(centers) < n:
            centers.append([x, y])
            x += 2 * r_init
        y += r_init * np.sqrt(3)
        row += 1
        
    centers = np.array(centers[:n])
    centers += np.random.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)

    best_centers = centers.copy()
    best_sum_r = sum_r
    best_radii = radii.copy()
    
    # 3. Iterative Center Perturbation (Hill Climbing with Cooling)
    step_size = 0.05
    num_iterations = 30000
    
    for it in range(num_iterations):
        # Cool down step size
        step_size *= 0.99995
        
        # Perturb a random circle
        idx = np.random.randint(n)
        new_centers = centers.copy()
        delta = np.random.normal(0, step_size, 2)
        new_centers[idx] += delta
        new_centers[idx] = np.clip(new_centers[idx], 0.0, 1.0)
        
        # Evaluate new configuration
        new_radii, new_sum_r = solve_radii_lp(new_centers, idx_i, idx_j)
        
        # Track global best
        if new_sum_r > best_sum_r + 1e-9:
            best_sum_r = new_sum_r
            best_centers = new_centers.copy()
            best_radii = new_radii.copy()

        if new_sum_r > sum_r:
            centers = new_centers
            radii = new_radii
            sum_r = new_sum_r

    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_sum_r)
