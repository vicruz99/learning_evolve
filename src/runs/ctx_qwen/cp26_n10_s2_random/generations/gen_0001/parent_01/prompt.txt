def compute_sum_radii(centers):
    """Compute the maximum sum of radii for a given set of centers."""
    n = centers.shape[0]
    # Boundary constraints: distance to each side of the unit square
    r = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    r = np.minimum(r, np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Pairwise constraints: half the distance to the nearest other center
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    r = np.minimum(r, min_dists * 0.5)
    
    return np.sum(r)

def objective_func(vars, n):

    Objective: Minimize -sum(radii)

def boundary_constraints(vars, n):
    """
    Constraints: r <= x <= 1-r, r <= y <= 1-r
    vars layout: [x1, y1, r1, ...]
    Returns array of constraint values (should be >= 0)
    """
    con = []
    for i in range(n):
        x = vars[3*i]
        y = vars[3*i + 1]
        r = vars[3*i + 2]
        # x - r >= 0
        con.append(x - r)
        # 1 - x - r >= 0 => 1 - x >= r => 1 - x - r >= 0
        con.append(1 - x - r)
        # y - r >= 0
        con.append(y - r)
        # 1 - y - r >= 0
        con.append(1 - y - r)
    return np.array(con)

def overlap_constraints(vars, n):
    """
    Constraints: dist(i, j) >= r_i + r_j
    vars layout: [x1, y1, r1, ...]
    Returns array of constraint values (should be >= 0)
    """
    con = []
    for i in range(n):
        xi = vars[3*i]
        yi = vars[3*i + 1]
        ri = vars[3*i + 2]
        for j in range(i + 1, n):
            xj = vars[3*j]
            yj = vars[3*j + 1]
            rj = vars[3*j + 2]
            # dist^2 >= (ri + rj)^2
            # To avoid sqrt in constraint function for better conditioning? 
            # Actually sqrt is fine, but squared is smooth.
            # However, constraint must be >= 0.
            # dist - (ri + rj) >= 0
            dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)

    c4 = 1.0 - ys - rs

    dy = ys[idx[0]] - ys[idx[1]]

    dist = np.sqrt(np.sum(diff**2, axis=2))

    np.fill_diagonal(dist, np.inf)

    min_dist = np.min(dist, axis=1)
    
    # Radius constrained by half the minimum distance to avoid overlap
    r_pair = 0.5 * min_dist

    radii = np.minimum(r_bound, r_pair)
