The key insight is to decouple the problem into two stages:
1. **Inner Problem (Radii Optimization):** For any fixed set of centers, finding the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints is a **Linear Programming (LP)** problem. This can be solved exactly and very quickly using `scipy.optimize.linprog` with the `highs` solver.
2. **Outer Problem (Center Optimization):** We optimize the center coordinates to maximize the LP objective value. Since the LP value function is non-smooth and non-convex with respect to centers, we use `scipy.optimize.minimize` with the `Nelder-Mead` method (derivative-free, robust to discontinuities) across multiple randomized restarts.
3. **Polishing Phase:** After finding a good configuration via LP+Nelder-Mead, we run a final `SLSQP` optimization on the joint (centers, radii) variables. The LP solution provides an excellent feasible starting point, allowing SLSQP to fine-tune both positions and radii simultaneously, often pushing past local plateaus.

We precompute the constraint matrix structure for the LP to minimize overhead during the iterative optimization loop. This approach combines the mathematical exactness of LP for fixed geometry with robust global search techniques for the layout, reliably surpassing previous best results.

import numpy as np
from scipy.optimize import linprog, minimize

# Precompute constraint matrix structure for LP to speed up repeated calls
N_CIRCLES = 26
PAIR_IDX = []
for i in range(N_CIRCLES):
    for j in range(i + 1, N_CIRCLES):
        PAIR_IDX.append((i, j))
PAIR_IDX = np.array(PAIR_IDX)
M_PAIR = PAIR_IDX.shape[0]
A_PAIR = np.zeros((M_PAIR, N_CIRCLES))
for k, (i, j) in enumerate(PAIR_IDX):
    A_PAIR[k, i] = 1.0
    A_PAIR[k, j] = 1.0
A_BOUND = np.eye(N_CIRCLES)
A_FULL = np.vstack([A_PAIR, A_BOUND])

def compute_lp_sum_and_radii(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    b_i = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                     np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    b_i = np.maximum(b_i, 0.0)
    
    # Pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    dists_upper = dists[np.triu_indices(n, k=1)]
    
    # LP: max sum(r) s.t. r_i + r_j <= dists[i,j], r_i <= b_i, r_i >= 0
    b_ub = np.concatenate([dists_upper, b_i])
    
    try:
        res = linprog(-np.ones(n), A_ub=A_FULL, b_ub=b_ub, 
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def objective(centers_flat):
    """Objective for center optimization: negative max sum of radii."""
    centers = centers_flat.reshape(-1, 2)
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    val, _ = compute_lp_sum_and_radii(centers)
    return -val

def obj_poly(v):
    """Objective for SLSQP polish: minimize negative sum of radii."""
    return -np.sum(v[2*N_CIRCLES:])

def con_poly(v):
    """Constraints for SLSQP polish."""
    n = N_CIRCLES
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    
    # Boundary constraints
    con = np.concatenate([c[:,0] - r, 1.0 - c[:,0] - r, c[:,1] - r, 1.0 - c[:,1] - r])
    
    # Overlap constraints: dist >= r_i + r_j
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    con_ov = []
    for i in range(n):
        for j in range(i+1, n):
            con_ov.append(dists[i,j] - r[i] - r[j])
            
    return np.concatenate([con, con_ov, r])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    best_sum = 0.0
    best_centers = None
    
    np.random.seed(42)
    restarts = 12
    
    # Phase 1: Multi-start Nelder-Mead on centers with LP evaluation
    for seed in range(restarts):
        np.random.seed(seed)
        # Hexagonal lattice initialization
        centers = np.zeros((n, 2))
        idx = 0
        y = 0.12
        row = 0
        while idx < n:
            x = 0.12 + (row % 2) * 0.09
            while x < 0.92 and idx < n:
                centers[idx] = [x, y]
                idx += 1
                x += 0.18
            y += 0.13
            row += 1
            
        # Add perturbation to escape symmetric local minima
        centers += np.random.uniform(-0.02, 0.02, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        x0 = centers.flatten()
        
        try:
            res = minimize(objective, x0, method='Nelder-Mead', 
                           options={'maxiter': 1500, 'xatol': 1e-7, 'fatol': 1e-9})
            cur_sum = -res.fun
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = res.x.reshape(n, 2)
        except Exception:
            pass
            
    # Phase 2: Compute optimal radii for best centers found
    if best_centers is not None:
        _, best_radii = compute_lp_sum_and_radii(best_centers)
        
        # Phase 3: SLSQP Polish on joint variables to fine-tune
        x0_poly = np.concatenate([best_centers.flatten(), best_radii])
        cons_poly = {'type': 'ineq', 'fun': con_poly}
        bounds_poly = [(0,1)]*(2*n) + [(0, 0.5)]*n
        
        try:
            res_poly = minimize(obj_poly, x0_poly, method='SLSQP', bounds=bounds_poly,
                                constraints=cons_poly, options={'maxiter': 600, 'ftol': 1e-12})
            if res_poly.success:
                cur_sum_poly = -res_poly.fun
                if cur_sum_poly > best_sum:
                    best_centers = res_poly.x[:2*n].reshape(n, 2)
                    # Re-solve LP to ensure radii are exactly optimal for refined centers
                    _, best_radii = compute_lp_sum_and_radii(best_centers)
                    best_sum = np.sum(best_radii)
        except Exception:
            pass
    else:
        best_radii = np.zeros(n)
        
    return best_centers, best_radii, float(np.sum(best_radii))
