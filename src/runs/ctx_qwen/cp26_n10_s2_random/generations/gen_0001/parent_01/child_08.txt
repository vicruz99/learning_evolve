import math
from scipy.optimize import minimize

N_CIRCLES = 26
TRIU_IDX = np.triu_indices(N_CIRCLES, k=1)

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraint_fun(v):
    """Compute boundary and non-overlap constraints."""
    n = N_CIRCLES
    P = v.reshape(n, 3)
    C = P[:, :2]
    R = P[:, 2]
    
    # Boundary constraints: r <= min(x, 1-x, y, 1-y)  =>  min(...) - r >= 0
    b = np.minimum(np.minimum(C[:, 0], 1.0 - C[:, 0]), 
                   np.minimum(C[:, 1], 1.0 - C[:, 1])) - R
    
    # Overlap constraints: dist(i,j) >= r_i + r_j  =>  dist - (r_i + r_j) >= 0
    diff = C[:, np.newaxis, :] - C[np.newaxis, :, :]
    D = np.sqrt(np.sum(diff**2, axis=2))
    o = D[TRIU_IDX] - (R[TRIU_IDX[0]] + R[TRIU_IDX[1]])
    
    return np.concatenate([b, o])
