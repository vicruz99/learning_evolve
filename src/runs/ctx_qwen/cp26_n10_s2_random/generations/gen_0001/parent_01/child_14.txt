import numpy as np
from scipy.optimize import minimize
from scipy.optimize import linprog

# Global constants for N=26 circles
N = 26
IDX_I, IDX_J = np.triu_indices(N, k=1)
N_PAIRS = len(IDX_I)

def objective_func(v):
    """Minimize negative sum of radii (equivalent to maximizing sum of radii)."""
    return -np.sum(v[2::3])

def constraint_func(v):
    """Compute all inequality constraints: must be >= 0."""
    cx = v[0::3]
    cy = v[1::3]
    r = v[2::3]
    
    con = np.empty(4 * N + N_PAIRS)
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    con[:N] = cx - r
    con[N:2*N] = 1.0 - cx - r
    con[2*N:3*N] = cy - r
    con[3*N:4*N] = 1.0 - cy - r
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[IDX_I] - cx[IDX_J]
    dy = cy[IDX_I] - cy[IDX_J]
    dr = r[IDX_I] + r[IDX_J]
    con[4*N:] = dx*dx + dy*dy - dr*dr
    
    return con

def get_initial_guess(seed):
    """Generate an initial feasible configuration using a perturbed hexagonal lattice."""
    rng = np.random.RandomState(seed)
    pts = []
    y = 0.05
    row = 0
    # Hexagonal spacing approximation
    while len(pts) < N:
        x = 0.05 + (0.5 if row % 2 else 0.0) * 0.16
        while x < 0.95 and len(pts) < N:
            pts.append([x + rng.uniform(-0.015, 0.015), y + rng.uniform(-0.015, 0.015)])
            x += 0.18
        y += 0.155
        row += 1
        
    pts = np.array(pts[:N])
    pts = np.clip(pts, 0.05, 0.95)
    
    v = np.zeros(3 * N)
    v[0::3] = pts[:, 0]
    v[1::3] = pts[:, 1]
    v[2::3] = 0.04  # Small initial radius ensures initial feasibility
    
    return v

def refine_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Upper bounds from boundaries
    bounds_r = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            dist = np.hypot(x[i] - x[j], y[i] - y[j])
            A_ub.append(row)
            b_ub.append(dist)
            
    # Boundary constraints: r_i <= bounds_r[i]
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(bounds_r[i])
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Maximize sum(r) => Minimize -sum(r)
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, 
                      bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.full(n, 0.0), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)  # Reproducibility
    
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = 0.0
    best_v = None
    
    # Multi-start optimization
    for trial in range(60):
        seed = trial * 17 + 3
        v0 = get_initial_guess(seed)
        
        # Alternate between fresh starts and perturbing the best known solution
        if best_v is not None and trial % 2 == 0:
            v_pert = best_v + np.random.normal(0, 0.003, 3 * N)
            v_pert[0::3] = np.clip(v_pert[0::3], 0.01, 0.99)
            v_pert[1::3] = np.clip(v_pert[1::3], 0.01, 0.99)
            v_pert[2::3] = np.clip(v_pert[2::3], 0.001, 0.5)
            v0 = v_pert
            
        try:
            res = minimize(objective_func, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            
            if res.success or (res.fun < -best_sum):
                cur_sum = -res.fun
                if cur_sum > best_sum:
                    # Verify constraints are satisfied within tolerance
                    c_vals = constraint_func(res.x)
                    if np.all(c_vals >= -1e-7):
                        best_sum = cur_sum
                        best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is not None:
        # Extract best centers
        centers = np.column_stack((best_v[0::3], best_v[1::3]))
        
        # Final refinement: fix centers and solve LP for optimal radii
        radii, lp_sum = refine_radii_lp(centers)
        
        # Ensure non-negativity and numerical safety
        radii = np.maximum(radii, 0.0)
        
        # If LP found a better sum, update (it should)
        if lp_sum > best_sum:
            best_sum = lp_sum
            
        return centers, radii, float(best_sum)
        
    # Fallback (should rarely be reached)
    centers = np.random.rand(N, 2) * 0.8 + 0.1
    radii = np.full(N, 0.02)
    return centers, radii, float(np.sum(radii))
