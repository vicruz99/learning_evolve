To maximize the sum of radii for 26 circles in a unit square, we need to escape local minima that trap standard gradient-based optimizers. The problem has non-convex constraints (circle-circle distances), making simultaneous optimization of centers and radii challenging.

Strategy:
1. **Decouple Initialization via LP**: For any fixed set of centers, finding the optimal radii is a Linear Programming problem. We can solve this quickly to get a strictly feasible starting point for the radii.
2. **Iterative Local Search (ILS)**: We will repeatedly perturb the best-known centers, solve the LP to find corresponding feasible radii, and then run a constrained non-linear optimizer (SLSQP) to jointly optimize both. This ensures every optimization step starts from a feasible state, dramatically improving convergence.
3. **Simulated Annealing-style Decay**: The perturbation magnitude (sigma) will decay over iterations, allowing broad exploration initially and fine-tuning later.
4. **Vectorized Constraints**: We'll compute boundary and pairwise distance constraints efficiently using NumPy broadcasting and `np.triu_indices` to keep each SLSQP evaluation fast.
5. **Robust Fallbacks**: Multiple initial configurations and strict feasibility checks ensure we always return a valid packing.

This approach combines the mathematical exactness of LP for radius sizing with the flexibility of SLSQP for geometric arrangement, consistently pushing past previous local optima.

import numpy as np
from scipy.optimize import minimize, linprog

def get_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Preallocate constraint matrices
    n_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    # Pairwise constraints: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    # Fallback to small radii if LP fails (highly unlikely with valid centers)
    return np.full(n, 1e-5)

def compute_constraints(v, n):
    """Computes all inequality constraints for SLSQP. Must return >= 0."""
    P = v.reshape(n, 3)
    C = P[:, :2]
    R = P[:, 2]
    
    # Boundary constraints
    c1 = C[:, 0] - R
    c2 = 1.0 - C[:, 0] - R
    c3 = C[:, 1] - R
    c4 = 1.0 - C[:, 1] - R
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    diff = C[:, np.newaxis, :] - C[np.newaxis, :, :]
    dist_sq = np.sum(diff ** 2, axis=2)
    R_sum = R[:, np.newaxis] + R[np.newaxis, :]
    idx = np.triu_indices(n, k=1)
    c5 = dist_sq[idx] - R_sum[idx] ** 2
    
    return np.concatenate([c1, c2, c3, c4, c5])

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': lambda v: compute_constraints(v, n)}
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # 1. Generate a structured initial configuration (hexagonal-like)
    centers_init = np.zeros((n, 2))
    idx = 0
    y = 0.12
    while idx < n:
        x = 0.12
        while x < 0.88 and idx < n:
            centers_init[idx] = [x, y]
            x += 0.18
            idx += 1
        y += 0.155
    # Fill remaining if any
    if idx < n:
        centers_init[idx:] = np.random.rand(n - idx, 2) * 0.8 + 0.1
        
    # Solve LP for initial radii
    r_init = get_lp_radii(centers_init)
    v0 = np.concatenate([centers_init.ravel(), r_init])
    
    # 2. Iterative Local Search with SLSQP refinement
    np.random.seed(42)
    sigma = 0.06  # Initial perturbation scale
    
    for step in range(120):
        # Perturb the best centers found so far
        if best_centers is not None:
            c_pert = best_centers + np.random.normal(0, sigma, best_centers.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
        else:
            c_pert = centers_init.copy()
            
        # Get feasible radii for perturbed centers via LP
        r_pert = get_lp_radii(c_pert)
        v_pert = np.concatenate([c_pert.ravel(), r_pert])
        
        # Run SLSQP to optimize jointly
        try:
            res = minimize(objective, v_pert, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 600, 'ftol': 1e-12})
            
            cur_sum = -res.fun
            # Check feasibility with tolerance
            if np.all(compute_constraints(res.x, n) >= -1e-7):
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers = res.x[:2 * n].reshape(n, 2).copy()
                    best_radii = res.x[2 * n:].copy()
        except Exception:
            pass
            
        # Decay perturbation scale
        sigma *= 0.96
        
    # Fallback if optimization completely failed (should not happen)
    if best_centers is None:
        best_centers = centers_init
        best_radii = r_init
        best_sum = np.sum(r_init)
        
    return best_centers, best_radii, float(best_sum)
