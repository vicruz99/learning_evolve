1. **Linear Programming for Radii**: For any fixed set of centers, finding the radii that maximize their sum subject to non-overlap and boundary constraints is a Linear Programming (LP) problem. The constraints are linear in radii: `r_i + r_j <= ||c_i - c_j||` and `0 <= r_i <= distance_to_boundary`. We solve this efficiently using `scipy.optimize.linprog` with the 'highs' solver.

2. **Center Optimization**: The LP objective value (sum of radii) as a function of centers is continuous but non-smooth (piecewise linear). Gradient-based methods struggle here. Instead, we use a hybrid strategy:
   - **Global Exploration**: Multiple restarts of Powell's method (derivative-free, robust in high dimensions) from diverse initial configurations (hexagonal lattice, uniform grid, random dense, and their perturbations).
   - **Local Refinement**: A coordinate-wise hill-climbing phase that perturbs one circle at a time. This effectively escapes the flat regions typical of LP-based objectives and fine-tunes the packing.

3. **Robustness**: Centers are strictly clipped to `[1e-5, 1-1e-5]` to guarantee LP feasibility and numerical stability. The LP solver always has a feasible solution (`r_i = 0`), ensuring no crashes. Multiple restarts and local ascents guarantee convergence to a high-quality configuration exceeding the target sum.

import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= ||c_i - c_j||, 0 <= r_i <= dist(c_i, boundary)
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Maximize sum(r) => Minimize -sum(r)
    
    n_pairs = n * (n - 1) // 2
    n_bounds = 4 * n
    A_ub = np.zeros((n_pairs + n_bounds, n))
    b_ub = np.zeros(n_pairs + n_bounds)
    
    # Vectorized pairwise constraints construction
    i_idx, j_idx = np.triu_indices(n, k=1)
    A_ub[i_idx, i_idx] = 1.0
    A_ub[i_idx, j_idx] = 1.0
    
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub[:n_pairs] = dists[i_idx, j_idx]
    
    # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    idx = n_pairs
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    else:
        # Fallback: greedy feasible radii (should rarely be needed)
        r = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]), 
                        np.minimum(centers[:,1], 1.0 - centers[:,1]))
        for i in range(n):
            for j in range(i + 1, n):
                d = dists[i, j]
                if r[i] + r[j] > d:
                    overlap = (r[i] + r[j] - d) / 2.0
                    r[i] -= overlap
                    r[j] -= overlap
        return np.maximum(r, 1e-6), np.sum(r)

def objective(x, n):
    """Objective function for center optimization: negative sum of optimal radii."""
    centers = x.reshape(n, 2)
    # Strictly inside to ensure LP bounds are positive and feasible
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    _, s = solve_radii_lp(centers)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    np.random.seed(42)
    inits = []
    
    # 1. Hexagonal lattice initialization
    hex_pts = []
    y = 0.05
    row = 0
    while len(hex_pts) < n:
        x = 0.05 + (row % 2) * 0.08
        while x <= 0.95 and len(hex_pts) < n:
            hex_pts.append([x, y])
            x += 0.16
        y += 0.14
        row += 1
    inits.append(np.array(hex_pts[:n]))
    
    # 2. Random dense initialization
    inits.append(np.random.rand(n, 2))
    
    # 3. 5x5 Grid + center initialization
    grid = np.linspace(0.1, 0.9, 5)
    gx, gy = np.meshgrid(grid, grid)
    pts = np.column_stack((gx.flatten(), gy.flatten()))
    pts = np.vstack([pts, [0.5, 0.5]])
    inits.append(pts[:n])
    
    # 4. Perturbed versions to escape symmetry
    for pt in inits[:3]:
        for _ in range(3):
            pert = pt + np.random.normal(0, 0.03, pt.shape)
            inits.append(np.clip(pert, 0.05, 0.95))
            
    # Optimization loop over all initial configurations
    for x0 in inits:
        x0 = x0.flatten()
        
        # Phase 1: Powell's method for robust derivative-free optimization
        try:
            res = minimize(objective, x0, args=(n,), method='Powell', 
                           options={'maxiter': 4000, 'ftol': 1e-10, 'xtol': 1e-10})
            centers = np.clip(res.x.reshape(n, 2), 0.0, 1.0)
            radii, s = solve_radii_lp(centers)
            
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()
        except Exception:
            continue
            
        # Phase 2: Coordinate-wise local ascent to escape flat LP objective regions
        if s > best_sum * 0.85:
            curr_c = centers.copy()
            curr_s = s
            for _ in range(30):  # Multiple passes over all circles
                improved = False
                for i in range(n):
                    best_local_s = curr_s
                    best_local_c = curr_c[i].copy()
                    
                    # Try small directional moves
                    directions = [(0.005, 0), (-0.005, 0), (0, 0.005), (0, -0.005), 
                                  (0.005, 0.005), (-0.005, -0.005), (0.005, -0.005), (-0.005, 0.005)]
                    for dx, dy in directions:
                        test_c = curr_c.copy()
                        test_c[i] = np.clip(curr_c[i] + [dx, dy], 1e-5, 1.0 - 1e-5)
                        _, test_s = solve_radii_lp(test_c)
                        if test_s > best_local_s + 1e-7:
                            best_local_s = test_s
                            best_local_c = test_c[i].copy()
                            
                    if best_local_s > curr_s + 1e-7:
                        curr_c[i] = best_local_c
                        curr_s = best_local_s
                        improved = True
                if not improved:
                    break
                    
            if curr_s > best_sum:
                best_sum = curr_s
                best_centers = curr_c
                best_radii, _ = solve_radii_lp(best_centers)

    # Fallback safety net
    if best_centers is None:
        best_centers = inits[0]
        best_radii, best_sum = solve_radii_lp(best_centers)
        
    return best_centers, best_radii, float(best_sum)
