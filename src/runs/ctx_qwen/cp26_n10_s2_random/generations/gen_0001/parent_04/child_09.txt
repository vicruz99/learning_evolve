import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize
import itertools

def get_max_radii_sum(centers):
    """
    Given fixed centers, solve the LP to find radii that maximize sum(r_i)
    while satisfying non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    if n == 0:
        return np.array([]), 0.0
        
    # Objective: maximize sum(r) -> minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # 1. Boundary: r_i <= dist_to_side (4 constraints per circle)
    # 2. Pairwise: r_i + r_j <= dist_ij (n*(n-1)/2 constraints)
    n_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    row_idx = 0
    for i in range(n):
        x, y = centers[i]
        # Distance to each of the 4 walls
        for b in (x, 1.0 - x, y, 1.0 - y):
            A_ub[row_idx, i] = 1.0
            b_ub[row_idx] = b
            row_idx += 1
            
    for i, j in itertools.combinations(range(n), 2):
        dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        A_ub[row_idx, i] = 1.0
        A_ub[row_idx, j] = 1.0
        b_ub[row_idx] = dist
        row_idx += 1
        
    bounds = [(0, None)] * n
    
    try:
        # 'highs' is highly efficient for dense linear programs
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback for numerical edge cases
    return np.full(n, 0.001), 0.0

def objective(x, n):
    """Objective function for center optimization: minimize negative sum of radii."""
    centers = x.reshape(n, 2)
    # Clip to ensure numerical stability for LP boundary constraints
    centers = np.clip(centers, 0.001, 0.999)
    _, s = get_max_radii_sum(centers)
    return -s

def run_packing():
    n = 26
    # Bounds for centers: keep slightly inside [0,1] to avoid degenerate LP bounds
    bounds = [(0.005, 0.995)] * (2 * n)
    
    best_x = None
    best_score = np.inf
    
    # Global search using Differential Evolution with multiple seeds
    # DE is robust for non-smooth landscapes like this
    for seed in [0, 7, 42, 101, 2024]:
        try:
            res = differential_evolution(
                objective, bounds, args=(n,),
                popsize=12, maxiter=150, tol=1e-8, seed=seed,
                polishing=False  # We handle polishing manually below
            )
            if res.fun < best_score:
                best_score = res.fun
                best_x = res.x.copy()
        except Exception:
            pass
            
    if best_x is None:
        # Fallback to random initialization if DE fails
        best_x = np.random.rand(2 * n) * 0.9 + 0.05
        
    # Local refinement using Powell's method to precisely converge
    try:
        res_local = minimize(
            objective, best_x, args=(n,), method='Powell',
            options={'maxiter': 3000, 'ftol': 1e-12, 'xtol': 1e-12}
        )
        best_x = res_local.x
    except Exception:
        pass
        
    best_centers = best_x.reshape(n, 2)
    best_radii, best_sum = get_max_radii_sum(best_centers)
    
    # Ensure non-negative radii (LP guarantees this, but safeguard)
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_sum)
