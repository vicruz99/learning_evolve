import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_initial_hexagonal():
    """Create initial positions using a dense hexagonal packing pattern."""
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.085)
    idx = 0
    y = 0.12
    row = 0
    x_base = 0.12
    x_spacing = 0.18
    
    while idx < N:
        if row % 2 == 0:
            x = x_base
        else:
            x = x_base + x_spacing / 2.0
            
        while x <= 1.0 - 0.12 and idx < N:
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
            x += x_spacing
        y += x_spacing * np.sqrt(3) / 2.0
        row += 1
    return centers[:N], radii[:N]

def objective_func(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def constraint_func(v):
    """Compute all inequality constraints (must be >= 0)."""
    centers = v[:2 * N].reshape((N, 2))
    radii = v[2 * N:]
    
    cons = []
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    cons.extend(x - radii)
    cons.extend(1.0 - x - radii)
    cons.extend(y - radii)
    cons.extend(1.0 - y - radii)
    
    # Non-overlap constraints: distance >= r_i + r_j
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    # Add small epsilon to avoid gradient singularity at zero distance
    dist = np.sqrt(dx**2 + dy**2 + 1e-12)
    r_sum = radii[:, None] + radii[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    cons.extend(dist[mask] - r_sum[mask])
    
    # Non-negative radii
    cons.extend(radii)
    
    return np.array(cons)

def refine_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii exactly."""
    x = centers[:, 0]
    y = centers[:, 1]
    b = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    
    c_obj = -np.ones(N)
    num_con = N * (N - 1) // 2 + N
    A_ub = np.zeros((num_con, N))
    b_ub = np.zeros(num_con)
    
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
    for i in range(N):
        A_ub[idx, i] = 1.0
        b_ub[idx] = b[i]
        idx += 1
        
    bounds_r = [(0.0, None)] * N
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return centers, res.x
    # Fallback to greedy radius assignment if LP fails
    return centers, np.maximum(b, 0.0)

def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = 0.0
    best_v = None
    
    centers_init, radii_init = compute_initial_hexagonal()
    
    for trial in range(12):
        if trial == 0:
            v0 = np.concatenate([centers_init.flatten(), radii_init])
        elif trial == 1:
            # Grid initialization
            cx, cy = np.meshgrid(np.linspace(0.15, 0.85, 5), np.linspace(0.15, 0.85, 5))
            centers_grid = np.column_stack((cx.flatten(), cy.flatten()))
            centers_grid = np.vstack([centers_grid, [0.5, 0.5]])
            radii_grid = np.full(N, 0.08)
            v0 = np.concatenate([centers_grid.flatten(), radii_grid])
        elif trial == 2:
            # Random dense initialization
            v0 = np.random.uniform(0.1, 0.9, 2 * N)
            v0[2 * N:] = 0.06
        elif trial == 3:
            # Corner-focused initialization
            corners = np.array([[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]])
            v0 = np.concatenate([corners.flatten(), np.zeros(4), 
                                 centers_init[4:].flatten(), radii_init[4:]])
        else:
            # Perturbed versions of the best solution found so far
            if best_v is not None:
                v0 = best_v.copy()
                v0[:2*N] += np.random.normal(0, 0.008, 2*N)
                v0[:2*N] = np.clip(v0[:2*N], 0.02, 0.98)
            else:
                v0 = np.random.uniform(0.1, 0.9, 2 * N)
                v0[2 * N:] = 0.06
                
        try:
            res = minimize(objective_func, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12})
            cur_sum = -res.fun
            if cur_sum > best_sum:
                # Verify feasibility with tolerance for numerical errors
                if np.all(constraint_func(res.x) >= -1e-5):
                    best_sum = cur_sum
                    best_v = res.x.copy()
        except Exception:
            continue
            
    if best_v is None:
        best_v = np.concatenate([centers_init.flatten(), radii_init])
        best_sum = np.sum(radii_init)
        
    centers_opt = best_v[:2 * N].reshape((N, 2))
    radii_opt = best_v[2 * N:]
    
    # LP refinement guarantees maximum radii for the optimized centers
    centers_lp, radii_lp = refine_lp(centers_opt)
    radii_lp = np.maximum(radii_lp, 0.0)
    final_sum = np.sum(radii_lp)
    
    return centers_lp, radii_lp, float(final_sum)
