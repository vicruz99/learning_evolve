import numpy as np
from scipy.optimize import minimize, linprog
import math

def solve_lp_radii(centers):
    """
    Given fixed centers, solve the LP to find radii that maximize sum(r_i)
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    
    # Compute pairwise Euclidean distances efficiently
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Indices for upper triangle (i < j)
    rows_i, rows_j = np.triu_indices(n, k=1)
    n_pairs = len(rows_i)
    
    # Construct LP constraint matrix A_ub and vector b_ub
    # Constraints: 
    # 1) r_i + r_j <= dist_ij  for all i < j
    # 2) r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i
    A_ub = np.zeros((n_pairs + n, n))
    b_ub = np.zeros(n_pairs + n)
    
    A_ub[:n_pairs, rows_i] = 1.0
    A_ub[:n_pairs, rows_j] = 1.0
    b_ub[:n_pairs] = dists[rows_i, rows_j]
    
    for i in range(n):
        x, y = centers[i]
        A_ub[n_pairs + i, i] = 1.0
        b_ub[n_pairs + i] = min(x, 1.0 - x, y, 1.0 - y)
        
    bounds = [(0, None)] * n
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success and res.fun != -np.inf:
        return res.x, -res.fun
    return np.zeros(n), 0.0

    return -s
