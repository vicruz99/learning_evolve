import numpy as np
from scipy.optimize import minimize, linprog

def compute_optimal_radii(centers):
    """Solve LP to find maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    n_constraints = n * (n - 1) // 2 + 4 * n
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    # Pairwise non-overlap constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        limits = [x, 1.0 - x, y, 1.0 - y]
        for lim in limits:
            A_ub[idx, i] = 1.0
            b_ub[idx] = lim
            idx += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x, -res.fun
    else:
        return np.full(n, 1e-6), 0.0

def objective_centers(x):
    """Objective for center optimization: maximize sum of radii (computed via LP)."""
    n = 26
    centers = x.reshape(n, 2)
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    _, s = compute_optimal_radii(centers)
    return -s
