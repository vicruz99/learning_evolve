import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    Returns (radii, sum_radii).
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Maximize sum(r) => Minimize -sum(r)
    
    # Constraint counts: pairwise n(n-1)/2 + boundary 4n
    n_con = n * (n - 1) // 2 + 4 * n
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    idx = 0
    
    # Pairwise non-overlap: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        for val in [centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1]]:
            A_ub[idx, i] = 1.0
            b_ub[idx] = val
            idx += 1
            
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0


def objective(x, n):
    """Objective for center optimization: negative of max sum of radii."""
    centers = np.clip(x.reshape((n, 2)), 1e-6, 1.0 - 1e-6)
    _, val = solve_lp(centers)
    return -val


def generate_initial_configs(n, num_configs):
    """Generate diverse starting configurations for centers."""
    np.random.seed(42)
    configs = []
    
    # Hexagonal lattice variations
    for rho in [0.85, 0.95, 1.05, 1.15]:
        centers = np.zeros((n, 2))
        idx = 0
        y = 0.08
        row = 0
        while idx < n:
            x = 0.08 + (row % 2) * 0.5 * rho * 0.28
            while x <= 0.92 and idx < n:
                centers[idx] = [x, y]
                x += rho * 0.28
                idx += 1
            y += rho * 0.24
            row += 1
        configs.append(centers)
        
    # Grid-based variations
    for grid_n in [5, 6]:
        x = np.linspace(0.05, 0.95, grid_n)
        y = np.linspace(0.05, 0.95, grid_n)
        cx, cy = np.meshgrid(x, y)
        grid = np.column_stack((cx.ravel(), cy.ravel()))
        idx = np.random.choice(len(grid), n, replace=False)
        configs.append(grid[idx])
        
    # Random dense clusters
    for _ in range(4):
        configs.append(np.random.rand(n, 2) * 0.8 + 0.1)
        
    return configs[:num_configs]


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    best_val = -1.0
    best_centers = None
    
    # 1. Multi-start local optimization
    configs = generate_initial_configs(n, 15)
    for cfg in configs:
        x0 = cfg.flatten()
        try:
            res = minimize(objective, x0, args=(n,), method='Nelder-Mead', 
                           options={'maxiter': 4000, 'xatol': 1e-8, 'fatol': 1e-10})
            curr_val = -res.fun
            if curr_val > best_val:
                best_val = curr_val
                best_centers = res.x.reshape(n, 2)
        except Exception:
            continue
            
    # 2. Perturbation phase to escape local minima
    if best_centers is not None:
        for _ in range(10):
            perturbed = best_centers + np.random.normal(0, 0.008, best_centers.shape)
            perturbed = np.clip(perturbed, 0.02, 0.98)
            x0 = perturbed.flatten()
            try:
                res = minimize(objective, x0, args=(n,), method='Nelder-Mead', 
                               options={'maxiter': 2500, 'xatol': 1e-7, 'fatol': 1e-9})
                curr_val = -res.fun
                if curr_val > best_val:
                    best_val = curr_val
                    best_centers = res.x.reshape(n, 2)
            except Exception:
                pass
                
    # Fallback if optimization failed entirely
    if best_centers is None:
        best_centers = generate_initial_configs(n, 1)[0]
        
    best_centers = np.clip(best_centers, 0.0, 1.0)
    final_radii, final_sum = solve_lp(best_centers)
    
    return best_centers, final_radii, float(final_sum)
