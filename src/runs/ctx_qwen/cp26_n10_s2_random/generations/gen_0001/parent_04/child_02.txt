Instead, we adopt a **decomposition strategy**:
1. **Fix centers, optimize radii via Linear Programming (LP):** For any fixed set of centers, finding the maximum sum of radii is a convex LP problem. The constraints are linear: $r_i + r_j \le \|c_i - c_j\|$ and $0 \le r_i \le \text{dist}(c_i, \text{boundary})$. This can be solved exactly and extremely fast using `scipy.optimize.linprog` with the `highs` method.
2. **Global optimization of centers:** We define an objective function $F(\text{centers}) = -\sum r_i(\text{centers})$. Since this function is continuous but piecewise-linear (non-smooth), we use `differential_evolution` for robust global exploration, followed by `Nelder-Mead` for precise local refinement.
3. **Diverse restarts:** We combine DE with structured (hexagonal lattice) and randomized initial configurations to ensure coverage of the search space and escape shallow local minima.
4. This approach decouples the geometry search from the sizing problem, guaranteeing that for any center configuration, radii are optimally assigned, while the center optimizer drives the layout towards maximum density.

import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def solve_radii_lp(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Ensure centers are strictly inside (0,1) to avoid degenerate LP bounds
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    
    # Boundary constraints: 0 <= r_i <= min(x, 1-x, y, 1-y)
    b_i = np.minimum(np.minimum(c[:, 0], 1 - c[:, 0]), 
                     np.minimum(c[:, 1], 1 - c[:, 1]))
    bounds = [(0, bi) for bi in b_i]
    
    # Pairwise non-overlap constraints: r_i + r_j <= ||c_i - c_j||
    idx = np.triu_indices(n, k=1)
    A_ub = np.zeros((len(idx[0]), n))
    A_ub[(idx[0], idx[1])] = 1.0
    A_ub[(idx[1], idx[0])] = 1.0
    
    dx = c[idx[0], 0] - c[idx[1], 0]
    dy = c[idx[0], 1] - c[idx[1], 1]
    b_ub = np.sqrt(dx**2 + dy**2)
    
    # Maximize sum(r) <=> Minimize -sum(r)
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, res.x
    else:
        return 1e-9 * n, np.full(n, 1e-9)

def objective_func(x):
    """Objective for optimizers: negative sum of radii."""
    centers = x.reshape(-1, 2)
    s, _ = solve_radii_lp(centers)
    return -s

def generate_hex_init(n):
    """Generate a hexagonal lattice initialization."""
    centers = np.zeros((n, 2))
    idx = 0
    y = 0.09
    row = 0
    while idx < n:
        x = 0.09 + (row % 2) * 0.09
        while x <= 0.91 and idx < n:
            centers[idx] = [x, y]
            x += 0.18
            idx += 1
        y += 0.156
        row += 1
    return centers[:n]

def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n)
    best_score = -np.inf
    best_centers = None
    
    # Phase 1: Global search using Differential Evolution
    for seed in range(4):
        try:
            res = differential_evolution(objective_func, bounds, maxiter=200, popsize=12,
                                         mutation=(0.5, 1.0), recombination=0.9, 
                                         seed=seed, polishing=False)
            if -res.fun > best_score:
                best_score = -res.fun
                best_centers = res.x.reshape(n, 2).copy()
        except Exception:
            pass
            
    # Phase 2: Local refinement from best DE result
    if best_centers is not None:
        x0 = best_centers.flatten()
        res = minimize(objective_func, x0, method='Nelder-Mead',
                       options={'maxiter': 40000, 'xatol': 1e-7, 'fatol': 1e-11})
        if -res.fun > best_score:
            best_score = -res.fun
            best_centers = res.x.reshape(n, 2).copy()
            
    # Phase 3: Diverse local searches to escape potential DE local minima
    starts = [generate_hex_init(n).flatten()]
    np.random.seed(42)
    starts.append(np.clip(np.random.rand(2*n) * 0.8 + 0.1, 0.05, 0.95))
    starts.append(np.clip(np.random.rand(2*n) * 0.6 + 0.2, 0.1, 0.9))
    
    for s in starts:
        try:
            res = minimize(objective_func, s, method='Nelder-Mead',
                           options={'maxiter': 30000, 'xatol': 1e-7, 'fatol': 1e-11})
            if -res.fun > best_score:
                best_score = -res.fun
                best_centers = res.x.reshape(n, 2).copy()
        except Exception:
            pass
            
    # Fallback
    if best_centers is None:
        best_centers = generate_hex_init(n)
        
    # Final LP to compute exact optimal radii for the best centers
    _, radii = solve_radii_lp(best_centers)
    
    return best_centers, radii, float(best_score)
