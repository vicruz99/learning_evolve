import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_constraints(v):
    """Computes all inequality constraint values (must be >= 0)."""
    centers = v[:2*N].reshape(N, 2)
    radii = v[2*N:]
    
    c = []
    # Boundary constraints: circle inside [0,1]x[0,1]
    c.append(centers[:, 0] - radii)
    c.append(1.0 - centers[:, 0] - radii)
    c.append(centers[:, 1] - radii)
    c.append(1.0 - centers[:, 1] - radii)
    
    # Pairwise non-overlap constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d2 = np.sum(diff**2, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c.append(d2[mask] - r_sum[mask]**2)
    
    # Non-negative radii
    c.append(radii)
    
    return np.concatenate(c)

def compute_objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def hex_init(seed):
    """Generates an initial configuration based on a perturbed hexagonal lattice."""
    rng = np.random.default_rng(seed)
    pts = []
    r_est = 0.1
    dx = 2.0 * r_est
    dy = r_est * np.sqrt(3.0)
    y = r_est
    row = 0
    while len(pts) < N + 10:
        x = r_est + (row % 2) * (dx / 2.0)
        while x < 1.0 - r_est and len(pts) < N + 10:
            pts.append([x, y])
            x += dx
        y += dy
        row += 1
        
    pts = np.array(pts[:N+10])
    # Normalize and scale to fit comfortably inside the unit square
    min_p = pts.min(axis=0)
    max_p = pts.max(axis=0)
    pts = (pts - min_p) / (max_p - min_p + 1e-9)
    pts = pts * 0.8 + 0.1
    
    # Add Gaussian perturbation to break symmetry and explore configuration space
    pts += rng.normal(0, 0.02, pts.shape)
    pts = np.clip(pts, 0.01, 0.99)
    return pts[:N]

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii exactly."""
    c_obj = -np.ones(N)
    rows = []
    bounds_vec = []
    
    # Pairwise constraints: r_i + r_j <= distance
    for i in range(N):
        for j in range(i+1, N):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            bounds_vec.append(dist - 1e-9) # Small slack for numerical safety
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(N):
        dists = np.array([centers[i, 0], 1.0 - centers[i, 0], 
                          centers[i, 1], 1.0 - centers[i, 1]])
        max_r = np.min(dists)
        row = np.zeros(N)
        row[i] = 1.0
        rows.append(row)
        bounds_vec.append(max_r - 1e-9)
        
    A_ub = np.array(rows)
    b_ub = np.array(bounds_vec)
    bounds = [(0, None)] * N
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def run_packing():

    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
