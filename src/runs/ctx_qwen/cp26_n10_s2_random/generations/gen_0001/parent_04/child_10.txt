To maximize the sum of radii for 26 circles in a unit square, we decouple the problem into two stages:
1. **Center Optimization**: Treat the centers as the primary variables. For any fixed set of centers, the maximum sum of radii is found by solving a Linear Program (LP). This transforms the problem into maximizing a continuous (though non-smooth) function of the centers.
2. **LP for Radii**: Given centers, constraints are linear: $r_i + r_j \le \|c_i - c_j\|$ and $r_i \le \text{dist}(c_i, \partial)$. This LP is solved efficiently using `scipy.optimize.linprog` with the `highs` solver.
3. **Global Search**: We start from a dense hexagonal lattice (known to be near-optimal for packing density), generate multiple perturbed initial configurations, and optimize using derivative-free methods (`Powell` and `Nelder-Mead`).
4. **Simulated Annealing Refinement**: After local optimization, a controlled SA phase perturbs individual centers, accepting improvements and occasionally worse moves to escape local minima, with a decaying temperature.

This approach is robust, avoids the numerical pitfalls of joint non-linear constrained optimization, and naturally pushes circles apart to maximize the objective.

import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_radii(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii.
    Constraints: r_i + r_j <= dist(i,j), r_i <= boundary_dist, r_i >= 0
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Maximize sum(r) => Minimize -sum(r)
    
    A_ub_list = []
    b_ub_list = []
    
    # Pairwise non-overlap constraints: r_i + r_j <= ||c_i - c_j||
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dist)
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(b)
            
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    bounds = [(0.0, None)] * n
    
    # highs is the default and most efficient solver in recent scipy versions
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    # Fallback if LP fails (rare, usually due to infeasibility from bad centers)
    return 0.0, np.zeros(n)

def objective_for_centers(x):
    """Objective function for center optimization: negative max sum of radii."""
    centers = x.reshape(-1, 2)
    # Keep centers strictly inside to maintain feasible LP bounds
    centers = np.clip(centers, 0.001, 0.999)
    val, _ = solve_lp_radii(centers)
    return -val

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    best_val = -1.0
    best_centers = None
    best_radii = None
    
    # 1. Generate a strong initial configuration (Hexagonal lattice)
    centers_hex = np.zeros((n, 2))
    idx = 0
    y = 0.1
    row = 0
    while idx < n:
        shift = 0.0875 if row % 2 == 1 else 0.0
        x = 0.1 + shift
        while x <= 0.9125 and idx < n:
            centers_hex[idx] = [x, y]
            idx += 1
            x += 0.175
        y += 0.14
        row += 1
        
    inits = [centers_hex.ravel()]
    # Create multiple perturbed starting points to explore basin of attraction
    for _ in range(8):
        c = centers_hex.copy() + np.random.normal(0, 0.03, centers_hex.shape)
        inits.append(np.clip(c, 0.02, 0.98).ravel())
        
    # 2. Local optimization from each start
    for x0 in inits:
        # Powell's method is robust for non-smooth landscapes
        try:
            res = minimize(objective_for_centers, x0, method='Powell',
                           options={'maxiter': 2000, 'ftol': 1e-12, 'xtol': 1e-12})
            if -res.fun > best_val:
                best_val = -res.fun
                best_centers = res.x.reshape(n, 2)
        except Exception:
            pass
            
        # Nelder-Mead provides a different convergence path
        try:
            res = minimize(objective_for_centers, x0, method='Nelder-Mead',
                           options={'maxiter': 5000, 'fatol': 1e-12, 'xatol': 1e-12})
            if -res.fun > best_val:
                best_val = -res.fun
                best_centers = res.x.reshape(n, 2)
        except Exception:
            pass
            
    # 3. Simulated Annealing / Hill Climbing to escape local minima
    if best_centers is not None:
        current_centers = best_centers.copy()
        current_val = best_val
        step = 0.05
        
        for i in range(1500):
            temp = 0.04 * (1.0 - i / 1500.0)
            idx = np.random.randint(n)
            
            new_centers = current_centers.copy()
            new_centers[idx] += np.random.normal(0, step, 2)
            new_centers[idx] = np.clip(new_centers[idx], 0.001, 0.999)
            
            new_val, _ = solve_lp_radii(new_centers)
            delta = new_val - current_val
            
            # Metropolis criterion
            if delta > 0 or np.random.rand() < np.exp(delta / max(temp, 1e-6)):
                current_centers = new_centers
                current_val = new_val
                
            if current_val > best_val:
                best_val = current_val
                best_centers = current_centers.copy()
                
        # Extract optimal radii for the best centers found
        _, best_radii = solve_lp_radii(best_centers)
    else:
        # Fallback
        best_centers = centers_hex
        _, best_radii = solve_lp_radii(best_centers)
        best_val = np.sum(best_radii)
        
    # Ensure non-negative radii (LP guarantees this, but safe for numerical noise)
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_val)
