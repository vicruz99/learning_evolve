import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    Constraints:
      r_i + r_j <= ||c_i - c_j|| for all i < j
      r_i <= distance to boundary for all i
      r_i >= 0
    """
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    
    # Boundary limits for each circle
    b_lim = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    b_lim = np.maximum(b_lim, 0.0)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Indices for upper triangle (i < j)
    idx = np.triu_indices(n, k=1)
    n_pairs = len(idx[0])
    
    # Build constraint matrix A_ub and vector b_ub for: A_ub @ r <= b_ub
    # Pairwise constraints: r_i + r_j <= dist_ij
    A_pair = np.zeros((n_pairs, n))
    A_pair[np.arange(n_pairs), idx[0]] = 1.0
    A_pair[np.arange(n_pairs), idx[1]] = 1.0
    b_pair = dists[idx]
    
    # Boundary constraints: r_i <= b_lim_i
    A_bound = np.eye(n)
    b_bound = b_lim
    
    A_ub = np.vstack([A_pair, A_bound])
    b_ub = np.concatenate([b_pair, b_bound])
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        # highs is a highly efficient modern LP solver
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
        
    # Fallback if LP fails (rare)
    small_r = np.full(n, 0.001)
    return np.sum(small_r), small_r

def objective_centers(x, n):
    """Objective for outer optimizer: negative of max sum of radii."""
    centers = x.reshape(n, 2)
    # Keep centers strictly inside the unit square to ensure valid radii
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    val, _ = solve_radii_lp(centers)
    return -val

def generate_hex_init(n):
    """Generate a dense hexagonal lattice initialization."""
    centers = []
    r_guess = 0.12
    dy = r_guess * np.sqrt(3)
    y = r_guess
    row = 0
    while len(centers) < n and y <= 1.0 - r_guess:
        dx = 2 * r_guess
        x_start = r_guess + (row % 2) * r_guess
        x = x_start
        while x <= 1.0 - r_guess and len(centers) < n:
            centers.append([x, y])
            x += dx
        y += dy
        row += 1
    return np.array(centers[:n])

def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Base initialization using hexagonal packing
    c0 = generate_hex_init(n)
    
    np.random.seed(42)
    # Multiple restarts with Powell method to escape local optima
    for trial in range(15):
        if trial == 0:
            x0 = c0.flatten()
        else:
            if best_centers is not None:
                # Perturb the best solution found so far
                x0 = best_centers.flatten() + np.random.normal(0, 0.004, n * 2)
            else:
                # Random perturbation of initial grid
                x0 = c0.flatten() + np.random.normal(0, 0.015, n * 2)
                
        try:
            # Powell method is robust for non-smooth black-box objectives
            res = minimize(objective_centers, x0, args=(n,), method='Powell',
                           options={'maxiter': 2500, 'ftol': 1e-10, 'xtol': 1e-10})
            cur_sum = -res.fun
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = res.x.reshape(n, 2)
                _, best_radii = solve_radii_lp(best_centers)
        except Exception:
            continue
            
    # Final exact LP solve to ensure radii precisely match the optimal centers
    if best_centers is not None:
        final_sum, final_radii = solve_radii_lp(best_centers)
        best_sum = final_sum
        best_radii = final_radii
    else:
        best_centers, best_radii = c0, np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
