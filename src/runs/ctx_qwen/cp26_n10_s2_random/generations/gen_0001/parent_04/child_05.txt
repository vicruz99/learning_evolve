

<strategy>
The problem of packing circles to maximize the sum of radii can be formulated as an optimization problem where we alternate between:
1. **Fixing centers** and solving a Linear Programming (LP) problem to find optimal radii.
2. **Using the dual variables (Lagrange multipliers)** from the LP to compute forces that push centers apart and away from boundaries, then updating centers.

This approach is mathematically sound: the dual variables directly give the gradient of the maximum possible sum of radii with respect to the distances between circles and boundaries. By following these "forces", we perform a projected gradient ascent on the center positions.

Key improvements over previous solutions:
- **LP-based radius optimization**: Guarantees optimal radii for any fixed center configuration, avoiding heuristic shrinking/growing.
- **Dual-variable forces**: Provides exact gradient directions for moving centers to relax tight constraints, leading to faster and more reliable convergence.
- **Multiple diverse initializations**: Hexagonal, perturbed hexagonal, grid, and dense random starts to escape local minima.
- **Adaptive step size & patience**: Prevents overshooting and allows early stopping when convergence is reached.

The LP formulation:
- Maximize $\sum r_i$
- Subject to: $r_i + r_j \le \|c_i - c_j\|$ for all $i < j$
- Subject to: $r_i \le \text{dist}(c_i, \text{boundary})$
- $r_i \ge 0$

This is solved efficiently with `scipy.optimize.linprog` (`method='highs'`). The duals guide center updates: circles touching each other repel proportionally to their constraint duals, and circles touching walls are pushed inward.
</strategy>

import numpy as np
from scipy.optimize import linprog

def setup_lp_matrices(n):
    """Precompute constant constraint matrix for the LP."""
    n_pairs = n * (n - 1) // 2
    pair_indices = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    A_pairs = np.zeros((n_pairs, n))
    for k, (i, j) in enumerate(pair_indices):
        A_pairs[k, i] = 1.0
        A_pairs[k, j] = 1.0
        
    A_bounds = np.zeros((4 * n, n))
    for i in range(n):
        A_bounds[4 * i, i] = 1.0
        A_bounds[4 * i + 1, i] = 1.0
        A_bounds[4 * i + 2, i] = 1.0
        A_bounds[4 * i + 3, i] = 1.0
        
    return np.vstack([A_pairs, A_bounds]), pair_indices, n_pairs

def optimize_packing(n, A_ub, pair_indices, n_pairs, init_centers, seed=42):
    """Iteratively optimize centers using LP-derived forces."""
    np.random.seed(seed)
    centers = init_centers.copy()
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    
    best_sum = 0.0
    best_centers = centers.copy()
    best_radii = np.zeros(n)
    
    step = 0.005
    patience = 0
    
    for it in range(800):
        # Compute pairwise distances
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        
        # Build RHS vector for LP constraints
        b_pairs = dists[np.triu_indices(n, k=1)]
        b_bounds = np.concatenate([
            centers[:, 0],
            1.0 - centers[:, 0],
            centers[:, 1],
            1.0 - centers[:, 1]
        ])
        b_ub = np.concatenate([b_pairs, b_bounds])
        
        # Solve LP: maximize sum(radii)
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, 
                      bounds=[(0, None)] * n, method='highs')
        
        if res.success:
            radii = res.x
            curr_sum = np.sum(radii)
            
            if curr_sum > best_sum + 1e-12:
                best_sum = curr_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                patience = 0
            else:
                patience += 1
                
            # Extract dual variables (shadow prices)
            try:
                duals = res.ineqlin.marginals
            except AttributeError:
                duals = getattr(res, 'ineqfunc.marginals', np.zeros_like(b_ub))
                
            lambdas = np.maximum(duals[:n_pairs], 0.0)
            mus = np.maximum(duals[n_pairs:], 0.0)
            
            # Compute forces from duals
            forces = np.zeros_like(centers)
            for k, (i, j) in enumerate(pair_indices):
                lam = lambdas[k]
                if lam > 1e-8:
                    vec = centers[i] - centers[j]
                    d = dists[i, j]
                    if d > 1e-8:
                        f = (lam / d) * vec
                        forces[i] += f
                        forces[j] -= f
                        
            for i in range(n):
                forces[i, 0] += mus[4 * i] - mus[4 * i + 1]
                forces[i, 1] += mus[4 * i + 2] - mus[4 * i + 3]
                
            # Update centers
            centers += step * forces
            centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
            step *= 0.998
            
            if patience > 200:
                break
        else:
            break
            
    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    A_ub, pair_indices, n_pairs = setup_lp_matrices(n)
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice
    c = np.zeros((n, 2))
    idx = 0
    y = 0.1
    row = 0
    while idx < n:
        x = 0.1 + (row % 2) * 0.1
        while x < 0.9 and idx < n:
            c[idx] = [x, y]
            idx += 1
            x += 0.2
        y += 0.173205
        row += 1
    inits.append(c)
    
    # 2. Perturbed hexagonal
    inits.append(c + np.random.uniform(-0.015, 0.015, (n, 2)))
    
    # 3. Grid layout
    c2 = np.zeros((n, 2))
    idx = 0
    for r in range(5):
        for col in range(6):
            if idx >= n: break
            c2[idx] = [0.1 + col * 0.15, 0.1 + r * 0.18]
            idx += 1
    inits.append(c2)
    
    # 4. Dense random
    np.random.seed(123)
    inits.append(np.random.rand(n, 2) * 0.8 + 0.1)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Run optimization from each initialization
    for i, c_init in enumerate(inits):
        c_res, r_res, s_res = optimize_packing(n, A_ub, pair_indices, n_pairs, c_init, seed=42 + i)
        if s_res > best_sum:
            best_sum = s_res
            best_centers = c_res
            best_radii = r_res
            
    # Ensure non-negative radii (LP guarantees this, but clamp for safety)
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_sum)
