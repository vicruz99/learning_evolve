

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we need to efficiently explore configurations where circles tightly pack against each other and the boundaries. Past solutions using joint SLSQP on (centers, radii) tend to get trapped in local minima because the constraint landscape is highly non-convex and rugged.

A more effective strategy for this specific objective is to decouple center positions and radii:
1. **LP-based Radius Optimization**: For any fixed set of centers, the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints form a Linear Program. Solving this LP exactly gives the maximal sum for that geometry.
2. **Gradient Ascent on Centers**: The dual variables (marginals) from the LP solver directly give the sensitivity of the optimal objective value to changes in the constraint bounds. This translates to exact forces pushing centers apart or towards boundaries. We use these forces to perform gradient ascent on the center coordinates.
3. **Hybrid Refinement**: We combine this LP-gradient ascent with SLSQP polishing. The GA phase quickly finds high-quality basin locations, while SLSQP precisely tightens the packing geometry. Multiple random restarts, hexagonal templates, force-directed layouts, and corner-biased starts ensure we escape local minima.
4. **Strict Repair**: A final deterministic shrinking pass guarantees compliance with the $10^{-12}$ validation tolerance without sacrificing meaningful radius sum.

This approach directly targets the objective function's geometry and typically outperforms black-box joint optimizers for circle packing sum-of-radii problems.
</strategy>

```python
import numpy as np
from scipy.optimize import linprog, minimize

N = 26
# Precompute pairwise indices
PAIR_INDICES = [(i, j) for i in range(N) for j in range(i + 1, N)]
NUM_PAIRS = len(PAIR_INDICES)
NUM_BOUND = 4 * N
TOTAL_CONS = NUM_PAIRS + NUM_BOUND

# Precompute LP constraint matrix structure
A_LP = np.zeros((TOTAL_CONS, N))
for idx, (i, j) in enumerate(PAIR_INDICES):
    A_LP[idx, i] = 1.0
    A_LP[idx, j] = 1.0
for i in range(N):
    base = NUM_PAIRS + 4 * i
    for k in range(4):
        A_LP[base + k, i] = 1.0

def lp_and_grad(centers):
    """Solves LP for max sum of radii given fixed centers and computes gradient."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(TOTAL_CONS)
    idx = 0
    for i, j in PAIR_INDICES:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        x, y = centers[i]
        b_ub[NUM_PAIRS + 4*i] = x
        b_ub[NUM_PAIRS + 4*i + 1] = 1.0 - x
        b_ub[NUM_PAIRS + 4*i + 2] = y
        b_ub[NUM_PAIRS + 4*i + 3] = 1.0 - y
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=[(0, 1.0)]*N, method='highs')
    if not res.success:
        return np.zeros(N), -1.0, np.zeros_like(centers)
        
    radii = res.x
    duals = res.ineqlin.marginals
    
    grad = np.zeros_like(centers)
    idx = 0
    for i, j in PAIR_INDICES:
        lam = -duals[idx]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    for i in range(N):
        base = NUM_PAIRS + 4*i
        mu_L = -duals[base]
        mu_R = -duals[base+1]
        mu_B = -duals[base+2]
        mu_T = -duals[base+3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, np.sum(radii), grad

def run_ga(centers0, rng, steps=1200):
    """Gradient ascent on centers using LP dual gradients."""
    c = centers0.copy()
    best_c, best_sum = c.copy(), -1.0
    step = 0.008
    no_improve = 0
    
    for k in range(steps):
        r, s, g = lp_and_grad(c)
        if s > best_sum:
            best_sum = s
            best_c = c.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        if no_improve > 80:
            step *= 0.7
            c += rng.normal(0, 0.005, c.shape)
            c = np.clip(c, 0.02, 0.98)
            no_improve = 0
        elif no_improve > 30:
            step *= 0.9
            
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-9:
            c += step * g / g_norm
            c = np.clip(c, 0.01, 0.99)
            
        if k % 300 == 0 and k > 0:
            c += rng.normal(0, 0.002 * (0.85**(k//300)), c.shape)
            c = np.clip(c, 0.02, 0.98)
            
    return best_c, best_sum

def init_hex(r0, pat, rng):
    c = []
    y = r0
    for r_idx, cnt in enumerate(pat):
        shift = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(c) < N:
                c.append([x, y])
            x += 2.0 * r0
        y += r0 * np.sqrt(3)
    while len(c) < N:
        c.append(rng.uniform(0.1, 0.9, 2))
    c = np.array(c[:N])
    c += rng.normal(0, 0.003, c.shape)
    return np.clip(c, 0.02, 0.98)

def init_force(rng):
    c = rng.uniform(0.2, 0.8, (N, 2))
    for _ in range(500):
        for i in range(N):
            for j in range(i+1, N):
                d_vec = c[i] - c[j]
                dist = np.linalg.norm(d_vec)
                if dist < 0.15 and dist > 1e-6:
                    push = (0.15 - dist) * 0.05
                    c[i] += d_vec / dist * push
                    c[j] -= d_vec / dist * push
    return np.clip(c, 0.05, 0.95)

def slsqp_constraints(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def slsqp_objective(v):
    return -np.sum(v[2*N:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_opt = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': slsqp_constraints}
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    inits = []
    patterns = [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], 
                [6,6,5,5,4], [5,5,6,5,5], [6,4,6,5,5], [5,4,6,6,5]]
    for pat in patterns:
        for r0 in [0.09, 0.098, 0.105]:
            inits.append(init_hex(r0, pat, rng))
    for _ in range(10):
        inits.append(rng.uniform(0.1, 0.9, (N, 2)))
    for _ in range(8):
        inits.append(init_force(rng))
    for _ in range(8):
        c = rng.uniform(0.15, 0.85, (N, 2))
        c[:4] = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        inits.append(c)
        
    # Phase 1: LP-Gradient Ascent from diverse starts
    for c0 in inits:
        c_opt, s_opt = run_ga(c0, rng, steps=1000)
        r_opt, s_opt, _ = lp_and_grad(c_opt)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            best_r = r_opt.copy()
            
    # Phase 2: SLSQP Joint Polish
    if best_c is not None:
        v0 = np.concatenate([best_c.flatten(), best_r])
        for _ in range(3):
            try:
                res = minimize(slsqp_objective, v0, method='SLSQP', bounds=bounds_opt,
                              constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13})
                if np.min(slsqp_constraints(res.x)) >= -1e-7:
                    s = np.sum(res.x[2*N:])
                    if s > best_sum:
                        best_sum = s
                        best_c = res.x[:2*N].reshape(N, 2).copy()
                        best_r = res.x[2*N:].copy()
                        v0 = res.x.copy()
            except Exception:
                pass
                
    # Phase 3: Perturbation + GA + Polish loop to escape local minima
    for i in range(20):
        noise = 0.008 * (0.9**(i//5))
        c_pert = best_c + rng.normal(0, noise, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        c_opt2, s_opt2 = run_ga(c_pert, rng, steps=800)
        r_opt2, s_opt2, _ = lp_and_grad(c_opt2)
        if s_opt2 > best_sum:
            best_sum = s_opt2
            best_c = c_opt2.copy()
            best_r = r_opt2.copy()
            
        v0_p = np.concatenate([best_c.flatten(), best_r])
        try:
            res_p = minimize(slsqp_objective, v0_p, method='SLSQP', bounds=bounds_opt,
                            constraints=cons, options={'maxiter': 4000, 'ftol': 1e-12})
            if np.min(slsqp_constraints(res_p.x)) >= -1e-7:
                s_p = np.sum(res_p.x[2*N:])
                if s_p > best_sum:
                    best_sum = s_p
                    best_c = res_p.x[:2*N].reshape(N, 2).copy()
                    best_r = res_p.x[2*N:].copy()
        except Exception:
            pass

    # Fallback
    if best_c is None:
        best_c = rng.uniform(0.1, 0.9, (N, 2))
        best_r, best_sum, _ = lp_and_grad(best_c)

    # Strict Numerical Repair
    radii = best_r.copy()
    centers = best_c.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
```
