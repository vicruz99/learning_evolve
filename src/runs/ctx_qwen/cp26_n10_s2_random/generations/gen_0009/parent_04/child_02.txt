import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def compute_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # LP: max sum(r) s.t. r_i + r_j <= dist_ij, 0 <= r_i <= ub_i
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs + n, n))
    b_ub = np.zeros(num_pairs + n)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ub[idx] = ub[i]
        idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def obj_centers(v):
    """Objective for center optimization: maximize sum of LP radii."""
    c = v.reshape(N, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    s, _ = compute_lp_radii(c)
    return -s

def joint_constraints(v):
    """Constraints for joint SLSQP polish."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    return np.concatenate(con)

def obj_joint(v):
    return -np.sum(v[2*N:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_c = None
    best_r = None
    
    starts = []
    
    # 1. Hexagonal lattice patterns (high density baselines)
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [4,6,6,6,4], [6,6,5,5,4], [5,5,6,5,5]]
    for pat in patterns:
        c = []
        r0 = 0.105
        y = r0
        for r_idx, cnt in enumerate(pat):
            shift = r0 if r_idx % 2 == 1 else 0.0
            x = r0 + shift
            for _ in range(cnt):
                if len(c) < N:
                    c.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
        while len(c) < N:
            c.append(rng.uniform(0.15, 0.85, 2))
        starts.append(np.array(c[:N]))
        
    # 2. Diverse random starts
    for _ in range(30):
        starts.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    # 3. Corner & Edge focused (exploits boundary slack)
    for _ in range(20):
        c = rng.uniform(0.2, 0.8, (N, 2))
        c[0] = [0.1, 0.1]
        c[1] = [0.9, 0.1]
        c[2] = [0.1, 0.9]
        c[3] = [0.9, 0.9]
        starts.append(c)
        
    # Phase 1: Powell optimization on centers (decoupled from radii)
    for c0 in starts:
        try:
            res = minimize(obj_centers, c0.flatten(), method='Powell', 
                          options={'maxiter': 400, 'ftol': 1e-12})
            c_opt = np.clip(res.x.reshape(N, 2), 1e-6, 1.0-1e-6)
            s, r_opt = compute_lp_radii(c_opt)
            if s > best_sum:
                best_sum = s
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            pass

    # Phase 2: Local perturbation & swap strategy to escape local minima
    if best_c is not None:
        for i in range(60):
            noise = 0.018 * (0.92 ** (i // 6))
            c_pert = best_c + rng.normal(0, noise, best_c.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            
            # Swap two circles to break symmetry traps
            idx = rng.choice(N, 2, replace=False)
            c_pert[idx] = c_pert[idx[::-1]]
            
            try:
                res = minimize(obj_centers, c_pert.flatten(), method='Powell',
                              options={'maxiter': 250, 'ftol': 1e-12})
                c_opt = np.clip(res.x.reshape(N, 2), 1e-6, 1.0-1e-6)
                s, r_opt = compute_lp_radii(c_opt)
                if s > best_sum:
                    best_sum = s
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
            except Exception:
                pass

    # Phase 3: SLSQP Joint Polish (refines both centers and radii simultaneously)
    if best_c is not None:
        v0 = np.concatenate([best_c.flatten(), best_r])
        bounds_joint = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        cons = {'type': 'ineq', 'fun': joint_constraints}
        try:
            res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                          constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14})
            if np.min(joint_constraints(res.x)) >= -1e-7:
                s = np.sum(res.x[2*N:])
                if s > best_sum:
                    best_sum = s
                    best_c = res.x[:2*N].reshape(N, 2)
                    best_r = res.x[2*N:]
        except Exception:
            pass

    # Fallback guarantee
    if best_c is None:
        best_c = rng.uniform(0.1, 0.9, (N, 2))
        _, best_r = compute_lp_radii(best_c)
        best_sum = np.sum(best_r)

    # Phase 4: Strict Numerical Repair
    radii = best_r.copy()
    centers = best_c.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
