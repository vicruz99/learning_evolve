import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def solve_lp_and_grad(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns: radii, sum_radii, gradient_wrt_centers
    """
    n = centers.shape[0]
    
    # Boundary upper bounds for each circle
    ub_x0 = centers[:, 0]
    ub_x1 = 1.0 - centers[:, 0]
    ub_y0 = centers[:, 1]
    ub_y1 = 1.0 - centers[:, 1]
    
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    A_ub = np.zeros((num_pairs + num_bound, n))
    b_ub = np.zeros(num_pairs + num_bound)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = ub_x0[i]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = ub_x1[i]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = ub_y0[i]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = ub_y1[i]; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return np.zeros(n), 0.0, np.zeros_like(centers)
        
    radii = res.x
    duals = res.ineqlin.marginals
    
    grad = np.zeros_like(centers)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]
            if lam > 1e-9:
                d = dists[i, j]
                if d > 1e-9:
                    vec = (centers[i] - centers[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
            idx += 1
            
    b_start = num_pairs
    for i in range(n):
        mu_x0 = duals[b_start + 4*i]
        mu_x1 = duals[b_start + 4*i + 1]
        mu_y0 = duals[b_start + 4*i + 2]
        mu_y1 = duals[b_start + 4*i + 3]
        
        grad[i, 0] += mu_x0 - mu_x1
        grad[i, 1] += mu_y0 - mu_y1
        
    return radii, np.sum(radii), grad

def generate_starts(rng):
    starts = []
    
    # 1. Hexagonal lattice patterns with varying row counts
    patterns = [
        [5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], 
        [6,6,5,5,4], [5,5,6,5,5], [4,5,6,5,6], [6,5,5,5,5]
    ]
    for pat in patterns:
        c = []
        r_est = 0.095
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(c) < N:
                    c.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        while len(c) < N:
            c.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])
        starts.append(np.array(c[:N]) + rng.normal(0, 0.002, (N, 2)))
        
    # 2. Force-directed spread to find naturally balanced configurations
    for _ in range(12):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(300):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if dist < 0.28 and dist > 1e-6:
                        push = (0.28 - dist) * 0.06
                        forces[i] += d_vec / dist * push
                        forces[j] -= d_vec / dist * push
            c += forces
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    # 3. Pure random starts for diversity
    for _ in range(10):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    starts = generate_starts(rng)
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Phase 1: Gradient Ascent from diverse starts
    for c0 in starts:
        c = np.clip(c0, 0.01, 0.99)
        step = 0.005
        no_improve = 0
        
        for k in range(2000):
            radii, curr_sum, grad = solve_lp_and_grad(c)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = c.copy()
                best_radii = radii.copy()
                no_improve = 0
            else:
                no_improve += 1
                
            if no_improve > 100:
                step *= 0.7
                if step < 1e-9:
                    break
                    
            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-9:
                c += step * grad / g_norm
                c = np.clip(c, 0.005, 0.995)
                
            # Periodic noise to escape flat gradients
            if k % 400 == 0 and k > 0:
                c += rng.normal(0, 0.001, c.shape)
                c = np.clip(c, 0.01, 0.99)
                
    # Phase 2: Basin Hopping / Perturbation Search around best
    c_curr = best_centers.copy()
    for i in range(60):
        noise_scale = 0.006 * (0.88 ** (i // 6))
        c_pert = c_curr + rng.normal(0, noise_scale, c_curr.shape)
        c_pert = np.clip(c_pert, 0.01, 0.99)
        
        # Short gradient ascent from perturbation
        step = 0.004
        for k in range(400):
            radii, curr_sum, grad = solve_lp_and_grad(c_pert)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = c_pert.copy()
                best_radii = radii.copy()
                c_curr = c_pert.copy()
                
            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-9:
                c_pert += step * grad / g_norm
                c_pert = np.clip(c_pert, 0.005, 0.995)
                
        # Occasionally swap two circles to break symmetric traps
        if i % 4 == 0:
            idx = rng.choice(N, 2, replace=False)
            c_curr[idx] = c_curr[idx[::-1]]
            c_pert = c_curr.copy()
            
    # Phase 3: SLSQP Polish on (centers, radii) to finalize
    if best_centers is not None:
        def obj_joint(v): return -np.sum(v[2*N:])
        def cons_joint(v):
            c = v[:2*N].reshape(N, 2)
            r = v[2*N:]
            con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
            idx = np.triu_indices(N, 1)
            d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
            con.append(d - (r[idx[0]] + r[idx[1]]))
            return np.concatenate(con)
            
        bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        v0 = np.concatenate([best_centers.flatten(), best_radii])
        
        try:
            res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds,
                          constraints={'type': 'ineq', 'fun': cons_joint},
                          options={'maxiter': 8000, 'ftol': 1e-14})
            if np.sum(res.x[2*N:]) > best_sum:
                best_centers = res.x[:2*N].reshape(N, 2)
                best_radii = res.x[2*N:]
                best_sum = np.sum(best_radii)
        except Exception:
            pass
            
    # Strict Numerical Repair to guarantee validator compliance
    radii = best_radii.copy()
    centers = best_centers.copy()
    for _ in range(80):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Clamp to boundaries
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
