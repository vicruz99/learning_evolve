import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def get_lp_structure():
    """Precomputes the constant A_ub matrix structure for the LP radii solver."""
    num_pairs = N * (N - 1) // 2
    num_bound = 4 * N
    A_ub = np.zeros((num_pairs + num_bound, N))
    pair_indices = []
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            pair_indices.append((i, j))
            idx += 1
    for i in range(N):
        for _ in range(4):
            A_ub[idx, i] = 1.0
            idx += 1
    return A_ub, pair_indices

A_LP, PAIR_IDX = get_lp_structure()

def solve_lp_radii(centers):
    """Given fixed centers, solves an LP to maximize the sum of radii."""
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(A_LP.shape[0])
    idx = 0
    for i, j in PAIR_IDX:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        x, y = centers[i]
        b_ub[idx] = x; idx += 1
        b_ub[idx] = 1.0 - x; idx += 1
        b_ub[idx] = y; idx += 1
        b_ub[idx] = 1.0 - y; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=[(0, u) for u in ub], method='highs')
    if res.success:
        return res.x
    return np.full(N, 1e-4)

def constraints_sl(v):
    """Computes boundary and non-overlap constraints for SLSQP. Must be >= 0."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    return np.concatenate(con)

def obj_sl(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # 1. Generate diverse initial configurations
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5]
    ]
    for pat in patterns:
        for scale in [0.95, 1.0, 1.05]:
            c = []
            r_est = 0.1 * scale
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            while len(c) < N:
                c.append(rng.uniform(0.1, 0.9, 2))
            c = np.array(c[:N])
            c += rng.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    for _ in range(8):
        # Simple repulsion init to break grid symmetry
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(200):
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if dist < 0.15 and dist > 1e-6:
                        push = (0.15 - dist) * 0.05
                        c[i] += d_vec / dist * push
                        c[j] -= d_vec / dist * push
        c = np.clip(c, 0.05, 0.95)
        starts.append(c)

    bounds_sl = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N

    # 2. Simulated Annealing on Centers (evaluates LP radii as objective)
    for c0 in starts:
        c = c0.copy()
        r_curr = solve_lp_radii(c)
        sum_curr = np.sum(r_curr)
        T = 0.015
        
        for step in range(1200):
            i = rng.integers(N)
            new_c = c.copy()
            step_size = 0.02 * np.sqrt(T)
            new_c[i] += rng.normal(0, step_size, 2)
            new_c = np.clip(new_c, 0.01, 0.99)
            
            r_new = solve_lp_radii(new_c)
            sum_new = np.sum(r_new)
            
            delta = sum_new - sum_curr
            if delta > 0 or rng.random() < np.exp(delta / T):
                c = new_c
                r_curr = r_new
                sum_curr = sum_new
                
            T *= 0.997
            
        if sum_curr > best_sum:
            best_sum = sum_curr
            best_c = c.copy()
            best_r = r_curr.copy()

    if best_c is not None:
        v0 = np.concatenate([best_c.flatten(), best_r])
        try:

            if np.min(constraints_sl(res.x)) >= -1e-8:
                s = np.sum(res.x[2*N:])
                if s > best_sum:
                    best_sum = s
                    best_c = res.x[:2*N].reshape(N, 2).copy()
                    best_r = res.x[2*N:].copy()
        except Exception:
            pass
            
    # 4. Basin Hopping Refinement to escape final microscopic local minima
    if best_c is not None:
        for _ in range(15):
            c_pert = best_c + rng.normal(0, 0.004, best_c.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            r_pert = solve_lp_radii(c_pert) * 0.97
            v0 = np.concatenate([c_pert.flatten(), r_pert])
            
            try:
                res = minimize(obj_sl, v0, method='SLSQP', bounds=bounds_sl,
                              constraints={'type': 'ineq', 'fun': constraints_sl},
                              options={'maxiter': 4000, 'ftol': 1e-14})
                if np.min(constraints_sl(res.x)) >= -1e-8:
                    s = np.sum(res.x[2*N:])
                    if s > best_sum:
                        best_sum = s
                        best_c = res.x[:2*N].reshape(N, 2).copy()
                        best_r = res.x[2*N:].copy()
            except Exception:
                pass
                
    if best_c is None:
        best_c = rng.uniform(0.1, 0.9, (N, 2))
        best_r = solve_lp_radii(best_c)
        best_sum = np.sum(best_r)

    # 5. Strict Numerical Repair to guarantee validator compliance
    centers = best_c.copy()
    radii = best_r.copy()
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
