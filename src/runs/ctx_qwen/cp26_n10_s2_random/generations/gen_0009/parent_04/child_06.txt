import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_constraints(v):
    """Computes boundary and pairwise non-overlap constraints. Must return >= 0."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con = np.concatenate([
        c[:, 0] - r, 1.0 - c[:, 0] - r,
        c[:, 1] - r, 1.0 - c[:, 1] - r
    ])
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    idx = np.triu_indices(N, 1)
    dists = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con = np.concatenate([con, dists - (r[idx[0]] + r[idx[1]])])
    return con

def obj_func(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def get_safe_radii(centers):
    """Computes strictly feasible initial radii for a given set of centers."""
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    rp = 0.5 * np.min(dists, axis=1)
    # Scale down slightly to ensure strict initial feasibility
    return np.minimum(rb, rp) * 0.96

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_v = None
    best_sum = -np.inf
    
    # --- Phase 1: Generate Diverse Initial Configurations ---
    starts = []
    
    # Hexagonal lattice patterns with varying row structures and scales
    patterns = [
        [5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], 
        [4,6,6,6,4], [6,6,5,5,4], [5,5,6,5,5], [6,5,5,6,4]
    ]
    for pat in patterns:
        for r0 in [0.092, 0.098, 0.104, 0.110]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N: c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            while len(c) < N: c.append(rng.uniform(0.1, 0.9, 2))
            c = np.array(c[:N]) + rng.normal(0, 0.003, (N, 2))
            c = np.clip(c, 0.05, 0.95)
            r_s = get_safe_radii(c)
            starts.append(np.concatenate([c.flatten(), r_s]))
            
    # Dense random starts to explore non-lattice optima
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        r_s = get_safe_radii(c)
        starts.append(np.concatenate([c.flatten(), r_s]))
        
    # Corner-focused starts (exploits boundary space efficiently)
    corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    for _ in range(10):
        c = rng.uniform(0.2, 0.8, (N, 2))
        c[:4] = corners
        c = np.clip(c, 0.05, 0.95)
        r_s = get_safe_radii(c)
        starts.append(np.concatenate([c.flatten(), r_s]))
        
    # --- Phase 2: Multi-Start SLSQP Optimization ---
    for v0 in starts:
        try:
            res = minimize(obj_func, v0, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if np.min(compute_constraints(res.x)) >= -1e-7:
                s = np.sum(res.x[2*N:])
                if s > best_sum:
                    best_sum = s
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # --- Phase 3: Iterative Perturbation & Refinement ---
    if best_v is not None:
        for k in range(80):
            # Annealing schedule for noise magnitude
            noise = 0.007 * (0.94 ** (k // 4))
            v_p = best_v + rng.normal(0, noise, best_v.shape)
            v_p = np.clip(v_p, 0.01, 0.99)
            v_p[2*N:] = np.maximum(v_p[2*N:], 0.01)
            
            # Occasionally swap random circles to break symmetric traps
            if k % 10 == 0:
                swap_idx = rng.choice(N, 2, replace=False)
                v_p[swap_idx*3 : (swap_idx[0]+1)*3], v_p[swap_idx[1]*3 : (swap_idx[1]+1)*3] = \
                v_p[swap_idx[1]*3 : (swap_idx[1]+1)*3].copy(), v_p[swap_idx*3 : (swap_idx[0]+1)*3].copy()
                
            try:
                res = minimize(obj_func, v_p, method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
                if np.min(compute_constraints(res.x)) >= -1e-7:
                    s = np.sum(res.x[2*N:])
                    if s > best_sum:
                        best_sum = s
                        best_v = res.x.copy()
            except Exception:
                pass
                
    # --- Phase 4: LP Radius Polish ---
    # Given fixed centers, maximizing sum of radii is a linear program.
    # This extracts the exact theoretical maximum for the found topology.
    centers = best_v[:2*N].reshape(N, 2)
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    num_pairs = N * (N - 1) // 2
    A_ub = np.zeros((num_pairs, N))
    b_ub = np.zeros(num_pairs)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    res_lp = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, 
                     bounds=[(0, u) for u in ub], method='highs')
    if res_lp.success:
        lp_r = res_lp.x
        if np.sum(lp_r) > best_sum:
            best_sum = np.sum(lp_r)
            best_v[2*N:] = lp_r
            
    # --- Phase 5: Strict Numerical Repair ---
    c = best_v[:2*N].reshape(N, 2)
    r = best_v[2*N:].copy()
    for _ in range(100):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(c[i] - c[j])
                req = r[i] + r[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    r[i] -= shrink
                    r[j] -= shrink
                    changed = True
        # Clamp to boundaries
        for i in range(N):
            mr = min(c[i, 0], 1.0 - c[i, 0], c[i, 1], 1.0 - c[i, 1])
            if r[i] > mr + 1e-12:
                r[i] = mr
                changed = True
        if not changed:
            break
            
    r = np.maximum(r, 0.0)
    return c, r, float(np.sum(r))
