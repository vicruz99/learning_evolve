import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_lp_radii_and_duals(centers):
    """Given fixed centers, solve LP to maximize sum of radii and return duals."""
    n = centers.shape[0]
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs + n, n))
    b_ub = np.zeros(num_pairs + n)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ub[idx] = ub[i]
        idx += 1
        
    c_obj = -np.ones(n)
    bounds = [(0, u) for u in ub]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return np.zeros(n), np.zeros(A_ub.shape[0]), 0.0
        
    return res.x, res.ineqlin.marginals, np.sum(res.x)

def compute_gradient(centers, duals):
    """Computes gradient of sum of radii w.r.t centers using LP duals."""
    n = centers.shape[0]
    grad = np.zeros_like(centers)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]
            if lam > 1e-8:
                d = np.linalg.norm(centers[i] - centers[j])
                if d > 1e-9:
                    vec = (centers[i] - centers[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
            idx += 1
            
    b_start = n * (n - 1) // 2
    for i in range(n):
        mu = duals[b_start + i]
        if mu > 1e-8:
            x, y = centers[i]
            d_min = min(x, 1.0-x, y, 1.0-y)
            if abs(x - d_min) < 1e-6:
                grad[i, 0] += mu
            elif abs((1.0-x) - d_min) < 1e-6:
                grad[i, 0] -= mu
            if abs(y - d_min) < 1e-6:
                grad[i, 1] += mu
            elif abs((1.0-y) - d_min) < 1e-6:
                grad[i, 1] -= mu
    return grad

def force_directed_init(rng, n, iters=800):
    """Spreads points evenly using repulsive forces to create good initial layouts."""
    centers = rng.uniform(0.2, 0.8, (n, 2))
    for _ in range(iters):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i+1, n):
                d_vec = centers[i] - centers[j]
                d = np.linalg.norm(d_vec)
                if d < 0.15 and d > 1e-6:
                    f = (0.15 - d) / d * d_vec
                    forces[i] += f
                    forces[j] -= f
            for k in range(2):
                if centers[i, k] < 0.12:
                    forces[i, k] += (0.12 - centers[i, k]) * 15
                elif centers[i, k] > 0.88:
                    forces[i, k] -= (centers[i, k] - 0.88) * 15
        centers += forces * 0.008
        centers = np.clip(centers, 0.05, 0.95)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    starts = []
    # Hexagonal lattice patterns with various row distributions
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [4,6,6,6,4], 
                [6,6,5,5,4], [5,4,6,6,5], [5,5,6,5,5], [6,4,6,5,5]]
    for pat in patterns:
        c = []

        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(c) < N: c.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        while len(c) < N: c.append(rng.uniform(0.1, 0.9, 2))
        starts.append(np.array(c[:N]))

    for _ in range(12): starts.append(force_directed_init(rng, N, iters=600))

    for _ in range(10): starts.append(rng.uniform(0.15, 0.85, (N, 2)))
    
    for c0 in starts:
        centers = c0 + rng.normal(0, 0.003, c0.shape)
        centers = np.clip(centers, 0.02, 0.98)
        
        local_best_sum = -1.0
        local_best_c = centers.copy()
        local_best_r = None
        momentum = np.zeros_like(centers)

        for k in range(2500):

                local_best_sum = curr_sum
                local_best_c = centers.copy()
                local_best_r = radii.copy()

            if g_mom > 1e-10:
                grad_dir = momentum / g_mom

                r_new, _, s_new = compute_lp_radii_and_duals(centers_new)

                    centers = centers_new

            best_radii = local_best_r.copy()

                        centers = centers_new
                if k % 200 == 0: momentum *= 0.8
                    
            if local_best_sum > best_sum:
                best_sum = local_best_sum
                best_centers = local_best_c.copy()
                best_radii = local_best_r.copy()

    # Phase 3: SLSQP Joint Polish
    def obj(v): return -np.sum(v[2*N:])
    def cons(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
        idx = np.triu_indices(N, 1)
        d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
        con.append(d - (r[idx[0]] + r[idx[1]]))
        return np.concatenate(con)
        
    bounds_opt = [(0,1)]*(2*N) + [(0, 0.5)]*N
    v0 = np.concatenate([best_centers.flatten(), best_radii])
    
    for _ in range(3):
        try:
            res = minimize(obj, v0, method='SLSQP', bounds=bounds_opt, constraints={'type':'ineq', 'fun':cons},
                           options={'maxiter': 5000, 'ftol': 1e-14})
            if np.sum(res.x[2*N:]) > best_sum:
                best_centers = res.x[:2*N].reshape(N, 2)
                best_radii = res.x[2*N:]
                best_sum = np.sum(best_radii)
                v0 = res.x.copy()
        except Exception:
            pass
            
    # Phase 4: Strict Numerical Repair
    radii = best_radii.copy()
    centers = best_centers.copy()

        changed = False
        for i in range(N):

                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d)/2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed: break
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
