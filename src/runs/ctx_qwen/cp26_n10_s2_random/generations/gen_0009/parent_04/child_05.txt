import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def compute_lp_and_grad(centers):
    """Solves LP for optimal radii given fixed centers and computes gradient via LP duals."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    A_ub = np.zeros((num_pairs + num_bound, n))
    b_ub = np.zeros(num_pairs + num_bound)
    
    pair_info = []
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            d = np.linalg.norm(centers[i] - centers[j])
            b_ub[idx] = d
            pair_info.append((i, j, d))
            idx += 1
            
    for i in range(n):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, u) for u in ub], method='highs')
    if not res.success:
        return np.zeros(n), np.zeros(n), np.zeros((n, 2))
        
    radii = res.x
    duals = res.ineqlin.marginals
    
    grad = np.zeros((n, 2))
    idx = 0
    for i, j, d in pair_info:
        lam = duals[idx]
        if lam > 1e-8 and d > 1e-9:
            vec = (centers[i] - centers[j]) / d
            grad[i] += lam * vec
            grad[j] -= lam * vec
        idx += 1
        
    bound_start = num_pairs
    for i in range(n):
        mu_L = duals[bound_start + 4*i]
        mu_R = duals[bound_start + 4*i + 1]
        mu_B = duals[bound_start + 4*i + 2]
        mu_T = duals[bound_start + 4*i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, np.sum(radii), grad

def objective_joint(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def constraints_joint(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    return np.concatenate(con)

def generate_starts(rng):
    """Generates diverse initial configurations."""
    starts = []
    patterns = [
        [5, 5, 5, 5, 6], [5, 6, 5, 6, 4], [6, 5, 6, 5, 4],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5],
        [6, 4, 6, 5, 5], [5, 4, 6, 6, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 4, 6, 6], [6, 5, 5, 5, 5]
    ]
    for pat in patterns:
        for scale in [0.95, 1.0, 1.05]:
            c = []
            r_est = 0.1 * scale
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.02, 0.98)
            starts.append(c)
            
    for _ in range(20):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    corners = np.array([[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]])
    for _ in range(15):
        c = rng.uniform(0.2, 0.8, (N, 2))
        c[:4] = corners + rng.normal(0, 0.01, (4, 2))
        starts.append(c)
        
    return starts

def optimize_gradient_ascent(centers0, rng):
    """Runs gradient ascent on centers to maximize sum of radii."""
    centers = centers0.copy()
    best_centers = centers.copy()
    best_sum = -1.0
    step = 0.008
    no_improve = 0
    
    for k in range(2500):
        radii, curr_sum, grad = compute_lp_and_grad(centers)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        if no_improve > 80:
            step *= 0.8
        elif no_improve > 30:
            step *= 0.95
            
        if step < 1e-10:
            break
            
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-10:
            centers += step * grad / g_norm
        centers = np.clip(centers, 0.005, 0.995)
        
        if k % 150 == 0 and k > 0:
            noise_mag = 0.002 * (0.9 ** (k // 150))
            centers += rng.normal(0, noise_mag, centers.shape)
            centers = np.clip(centers, 0.01, 0.99)
            
    return best_centers, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    starts = generate_starts(rng)
    
    # Phase 1: Gradient ascent on diverse starts
    for c0 in starts:
        c_opt, s_opt = optimize_gradient_ascent(c0, rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            
    # Phase 2: SLSQP Joint Polish
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_joint}
    
    r_lp = compute_lp_and_grad(best_centers)[0]
    v0 = np.concatenate([best_centers.flatten(), r_lp])
    
    for _ in range(3):
        try:
            res = minimize(objective_joint, v0, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14})
            if np.min(constraints_joint(res.x)) >= -1e-8:
                s = np.sum(res.x[2*N:])
                if s > best_sum:
                    best_sum = s
                    best_centers = res.x[:2*N].reshape(N, 2).copy()
                    best_radii = res.x[2*N:].copy()
                    v0 = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Basin hopping / Perturbation search
    for i in range(60):
        noise = 0.008 * (0.96 ** i)
        c_pert = best_centers + rng.normal(0, noise, best_centers.shape)
        c_pert = np.clip(c_pert, 0.01, 0.99)
        
        # Swap two circles to break symmetry traps
        idx_swap = rng.choice(N, 2, replace=False)
        c_pert[idx_swap] = c_pert[idx_swap[::-1]]
        
        c_opt2, s_opt2 = optimize_gradient_ascent(c_pert, rng)
        if s_opt2 > best_sum:
            best_sum = s_opt2
            best_centers = c_opt2.copy()
            
        # Polish the new best
        r_lp2 = compute_lp_and_grad(best_centers)[0]
        v0_new = np.concatenate([best_centers.flatten(), r_lp2])
        try:
            res2 = minimize(objective_joint, v0_new, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14})
            if np.min(constraints_joint(res2.x)) >= -1e-8:
                s_new = np.sum(res2.x[2*N:])
                if s_new > best_sum:
                    best_sum = s_new
                    best_centers = res2.x[:2*N].reshape(N, 2).copy()
                    best_radii = res2.x[2*N:].copy()
        except Exception:
            pass
            
    if best_radii is None:
        best_radii = compute_lp_and_grad(best_centers)[0]
        
    # Phase 4: Strict Numerical Repair
    centers = best_centers.copy()
    radii = best_radii.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
