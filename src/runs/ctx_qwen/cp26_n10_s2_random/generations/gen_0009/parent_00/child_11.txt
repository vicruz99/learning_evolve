import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_INDICES = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIR_INDICES)
TRI_IDX_I, TRI_IDX_J = np.triu_indices(N, 1)

# Precompute constant structure of the LP constraint matrix
A_ub_structure = np.zeros((N_PAIRS + 4 * N, N))
for k, (i, j) in enumerate(PAIR_INDICES):
    A_ub_structure[k, i] = 1.0
    A_ub_structure[k, j] = 1.0
for i in range(N):
    base = N_PAIRS + 4 * i
    A_ub_structure[base, i] = 1.0
    A_ub_structure[base + 1, i] = 1.0
    A_ub_structure[base + 2, i] = 1.0
    A_ub_structure[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii given fixed centers and computes exact gradient via duals."""
    c = np.clip(centers, 1e-7, 1.0 - 1e-7)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(N_PAIRS + 4 * N)
    for k, (i, j) in enumerate(PAIR_INDICES):
        b_ub[k] = dists[i, j]
    for i in range(N):
        base = N_PAIRS + 4 * i
        b_ub[base] = c[i, 0]
        b_ub[base + 1] = 1.0 - c[i, 0]
        b_ub[base + 2] = c[i, 1]
        b_ub[base + 3] = 1.0 - c[i, 1]
        
    bounds = [(0.0, u) for u in ub]
    res = linprog(-np.ones(N), A_ub=A_ub_structure, b_ub=b_ub, 
                  bounds=bounds, method='highs')
    
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(c)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    duals = np.zeros(b_ub.shape[0])
    if hasattr(res, 'marginals') and res.marginals is not None:
        duals = res.marginals.ineqlin
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = res.ineqlin.marginals
        
    grad = np.zeros_like(c)
    
    for k, (i, j) in enumerate(PAIR_INDICES):
        mu = duals[k]
        if mu > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
                
    for i in range(N):
        base = N_PAIRS + 4 * i
        mu_L = duals[base]
        mu_R = duals[base + 1]
        mu_B = duals[base + 2]
        mu_T = duals[base + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, s_sum, grad

def obj_c(c_flat):
    c = c_flat.reshape(N, 2)
    _, s, _ = solve_lp_and_grad(c)
    return -s

def grad_c(c_flat):
    c = c_flat.reshape(N, 2)
    _, _, g = solve_lp_and_grad(c)
    return -g.flatten()

def optimize_centers(c0, max_iter=1000):
    """Optimizes centers using L-BFGS-B with exact LP gradient."""
    bounds_c = [(0.001, 0.999)] * (2 * N)
    try:
        res = minimize(obj_c, c0.flatten(), jac=grad_c, method='L-BFGS-B', 
                       bounds=bounds_c, options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
        return res.x.reshape(N, 2), -res.fun
    except Exception:
        return c0, -obj_c(c0.flatten())

def hex_init(pat, r0, rng):
    """Generates a perturbed hexagonal lattice."""
    c = []
    y = r0
    for r_idx, cnt in enumerate(pat):
        shift = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            c.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
    c = np.array(c[:N])
    return np.clip(c, 0.05, 0.95)

def force_directed_init(rng):
    """Generates a layout via iterative repulsive forces."""
    c = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(1500):
        forces = np.zeros_like(c)
        diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        
        rep = np.zeros_like(diff)
        mask = dists < 0.15
        mask = mask & (dists > 1e-5)
        rep[mask] = (0.15 - dists[mask]) / dists[mask]
        
        forces += np.sum(rep * diff, axis=1)
        
        # Boundary repulsion
        forces[:, 0] += np.where(c[:, 0] < 0.1, (0.1 - c[:, 0]) * 5.0, 0.0)
        forces[:, 0] -= np.where(c[:, 0] > 0.9, (c[:, 0] - 0.9) * 5.0, 0.0)
        forces[:, 1] += np.where(c[:, 1] < 0.1, (0.1 - c[:, 1]) * 5.0, 0.0)
        forces[:, 1] -= np.where(c[:, 1] > 0.9, (c[:, 1] - 0.9) * 5.0, 0.0)
        
        c += 0.015 * forces
        c = np.clip(c, 0.05, 0.95)
    return c

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Computes boundary and non-overlap constraints (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    dx = c[TRI_IDX_I, 0] - c[TRI_IDX_J, 0]
    dy = c[TRI_IDX_I, 1] - c[TRI_IDX_J, 1]
    dr = r[TRI_IDX_I] + r[TRI_IDX_J]
    dists = np.sqrt(dx**2 + dy**2 + 1e-12)
    con.append(dists - dr)
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    patterns = [
        [6,6,6,5,3], [7,6,6,4,3], [5,6,6,6,3], [6,5,6,6,3],

    ]
    
    inits = []

        for r0 in [0.09, 0.095, 0.10]:
            inits.append(hex_init(pat, r0, rng))
    for _ in range(8):
        inits.append(force_directed_init(rng))
    for _ in range(5):
        inits.append(rng.uniform(0.12, 0.88, (N, 2)))
        
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # Phase 1: L-BFGS-B on centers from diverse starts
    for c0 in inits:
        c_opt, s_opt = optimize_centers(c0, max_iter=800)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            
    if best_c is not None:
        best_r, _, _ = solve_lp_and_grad(best_c)
    else:
        best_c = inits[0]
        best_r, best_sum, _ = solve_lp_and_grad(best_c)
        
    # Phase 2: Perturbation & Restart to escape local minima
    for _ in range(6):
        c_pert = best_c + rng.normal(0, 0.008, best_c.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        c_opt, s_opt = optimize_centers(c_pert, max_iter=1000)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            best_r, _, _ = solve_lp_and_grad(best_c)
            
    # Phase 3: SLSQP Joint Polish
    v0 = np.concatenate([best_c.flatten(), best_r])
    bounds_j = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    
    for _ in range(5):
        v_pert = v0 + rng.normal(0, 0.001, v0.shape)
        v_pert = np.clip(v_pert, 0.01, 0.99)
        v_pert[2*N:] = np.clip(v_pert[2*N:], 0.01, 0.4)
        try:
            res = minimize(obj_joint, v_pert, method='SLSQP', bounds=bounds_j,
                          constraints={'type': 'ineq', 'fun': cons_joint},

                    v0 = res.x.copy()
        except Exception:
            pass
            
    # Phase 4: Simulated Annealing on Centers
    c_curr = best_c.copy()
    s_curr = best_sum
    r_curr = best_r.copy()

        c_new = c_curr + rng.normal(0, T, c_curr.shape)

            r_curr = r_new

                best_r = res.x[2*N:].copy()

        pass
        
    # Phase 6: Strict Numerical Repair
    centers = best_c.copy()
    radii = best_r.copy()

        changed = False
        for i in range(N):
            for j in range(i+1, N):

                if d < req - 1e-12:

                    radii[i] -= shrink
                    radii[j] -= shrink

            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
