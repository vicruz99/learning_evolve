import numpy as np
from scipy.optimize import linprog, minimize

N = 26
PAIR_INDICES = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIR_INDICES)

# Precompute constant structure of the LP constraint matrix
A_ub_const = np.zeros((N_PAIRS + 4 * N, N))
for k, (i, j) in enumerate(PAIR_INDICES):
    A_ub_const[k, i] = 1.0
    A_ub_const[k, j] = 1.0
for i in range(N):
    b = N_PAIRS + 4 * i
    A_ub_const[b + 0, i] = 1.0
    A_ub_const[b + 1, i] = 1.0
    A_ub_const[b + 2, i] = 1.0
    A_ub_const[b + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii given fixed centers and computes exact gradient via duals."""
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]

    b_ub = np.zeros(N_PAIRS + 4 * N)

        b_ub[k] = dists[i, j]

        b = N_PAIRS + 4 * i
        b_ub[b + 0] = c[i, 0]
        b_ub[b + 1] = 1.0 - c[i, 0]
        b_ub[b + 2] = c[i, 1]
        b_ub[b + 3] = 1.0 - c[i, 1]
        
    res = linprog(-np.ones(N), A_ub=A_ub_const, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if not res.success:
        return np.full(N, 1e-5), 0.0, np.zeros_like(c)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    # Extract dual marginals safely across scipy versions
    duals = np.zeros(b_ub.shape[0])
    if hasattr(res, 'marginals') and res.marginals is not None:
        duals = res.marginals.ineqlin
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = res.ineqlin.marginals
        
    grad = np.zeros_like(c)
    
    # Pairwise repulsion forces from active distance constraints
    for k, (i, j) in enumerate(PAIR_INDICES):
        mu = duals[k]
        if mu > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec

    for i in range(N):
        b = N_PAIRS + 4 * i
        mu_L = duals[b]
        mu_R = duals[b + 1]
        mu_B = duals[b + 2]
        mu_T = duals[b + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, s_sum, grad

def optimize_gradient_ascent(centers0, max_iter=2000, init_step=0.008, rng=None):

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Computes boundary and non-overlap constraints (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    con.append(np.sqrt(dx**2 + dy**2) - dr)
    return np.concatenate(con)
