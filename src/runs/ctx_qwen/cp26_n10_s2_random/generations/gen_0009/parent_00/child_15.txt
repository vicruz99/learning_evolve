import numpy as np
from scipy.optimize import linprog, minimize

N = 26
PAIR_INDICES = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIR_INDICES)

# Precompute constant structure of the LP constraint matrix
A_ub_structure = np.zeros((N_PAIRS + 4 * N, N))
for k, (i, j) in enumerate(PAIR_INDICES):
    A_ub_structure[k, i] = 1.0
    A_ub_structure[k, j] = 1.0
for i in range(N):
    base = N_PAIRS + 4 * i
    A_ub_structure[base, i] = 1.0
    A_ub_structure[base + 1, i] = 1.0
    A_ub_structure[base + 2, i] = 1.0
    A_ub_structure[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii given fixed centers and computes exact gradient via duals."""
    c = np.clip(centers, 1e-7, 1.0 - 1e-7)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(N_PAIRS + 4 * N)
    for k, (i, j) in enumerate(PAIR_INDICES):
        b_ub[k] = dists[i, j]
    for i in range(N):
        base = N_PAIRS + 4 * i
        b_ub[base] = c[i, 0]
        b_ub[base + 1] = 1.0 - c[i, 0]
        b_ub[base + 2] = c[i, 1]
        b_ub[base + 3] = 1.0 - c[i, 1]
        
    bounds = [(0.0, u) for u in ub]
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_structure, b_ub=b_ub, 
                      bounds=bounds, method='highs')
    except Exception:
        return np.zeros(N), 0.0, np.zeros_like(c)
        
    if not res.success:
        return np.zeros(N), 0.0, np.zeros_like(c)
        
    radii = res.x
    s_sum = np.sum(radii)
    
    duals = np.zeros(b_ub.shape[0])
    if hasattr(res, 'marginals') and res.marginals is not None:
        duals = res.marginals.ineqlin
    elif hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = res.ineqlin.marginals
        
    grad = np.zeros_like(c)
    
    # Pairwise repulsion forces from active distance constraints
    for k, (i, j) in enumerate(PAIR_INDICES):
        mu = duals[k]
        if mu > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (c[i] - c[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
                
    # Boundary forces from active wall constraints
    for i in range(N):
        base = N_PAIRS + 4 * i
        mu_L = duals[base]
        mu_R = duals[base + 1]
        mu_B = duals[base + 2]
        mu_T = duals[base + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, s_sum, grad

def obj_grad_wrapper(x):
    """Objective and gradient for L-BFGS-B minimization."""
    c = x.reshape(N, 2)
    _, s, g = solve_lp_and_grad(c)
    return -s, -g.flatten()

def force_spread(centers, iters=300):
    """Simple repulsive force simulation to evenly space initial points."""
    c = centers.copy()
    for _ in range(iters):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                d_vec = c[i] - c[j]
                d = np.linalg.norm(d_vec)
                if d < 0.2 and d > 1e-6:
                    rep = (0.2 - d) * 0.1 / d
                    forces[i] += d_vec * rep
                    forces[j] -= d_vec * rep
        c += forces
        c = np.clip(c, 0.05, 0.95)
    return c

def generate_starts(rng):
    """Generates a wide variety of initial configurations."""
    starts = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],

        [6, 6, 6, 6, 2], [5, 5, 5, 6, 5]
    ]
    
    for pat in patterns:
        for r_est in [0.088, 0.095, 0.102, 0.108]:
            c = []

                    x += 2.0 * r_est
                y += r_est * np.sqrt(3.0)
            c = np.array(c[:N])

            c = np.clip(c, 0.05, 0.95)
            starts.append(c)

        c_rand = rng.uniform(0.15, 0.85, (N, 2))
        starts.append(force_spread(c_rand, iters=200))
        
    return starts

def optimize_lbfgsb(c0):
    """Optimizes centers using L-BFGS-B with exact LP gradients."""
    c0 = np.clip(c0, 0.01, 0.99)
    bounds = [(0.01, 0.99)] * (2 * N)
    res = minimize(obj_grad_wrapper, c0.flatten(), method='L-BFGS-B', jac=True,
                   bounds=bounds, options={'maxiter': 1500, 'ftol': 1e-12})
    c_opt = res.x.reshape(N, 2)
    _, s_opt, _ = solve_lp_and_grad(c_opt)
    return c_opt, s_opt

def sim_annealing(c0, rng):
    """Simulated annealing on center coordinates to escape local minima."""
    c_curr = c0.copy()
    _, s_curr, _ = solve_lp_and_grad(c_curr)
    T = 0.008
    best_c = c_curr.copy()
    best_s = s_curr
    
    for step in range(800):
        T *= 0.996
        c_new = c_curr + rng.normal(0, T, c_curr.shape)
        c_new = np.clip(c_new, 0.02, 0.98)
        _, s_new, _ = solve_lp_and_grad(c_new)
        
        if s_new > s_curr:
            c_curr = c_new
            s_curr = s_new
            if s_curr > best_s:
                best_s = s_curr
                best_c = c_curr.copy()
        else:
            if rng.random() < np.exp((s_new - s_curr) / max(T * 4.0, 1e-8)):
                c_curr = c_new
                s_curr = s_new
                
    return best_c, best_s

        return -np.sum(v[2*N:])

        cc = v[:2*N].reshape(N, 2)

        idx_i, idx_j = np.triu_indices(N, 1)
        dx = cc[idx_i, 0] - cc[idx_j, 0]
        dy = cc[idx_i, 1] - cc[idx_j, 1]
        dr = rr[idx_i] + rr[idx_j]
        con.append(np.sqrt(dx**2 + dy**2) - dr)
        return np.concatenate(con)
        
    for _ in range(3):
        v_pert = v0 + rng.normal(0, 0.001, v0.shape)

        try:
            res = minimize(obj_sl, v_pert, method='SLSQP', bounds=bounds_j,
