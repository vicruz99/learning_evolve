from scipy.optimize import linprog, minimize

N = 26
PAIR_INDICES = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIR_INDICES)

# Precompute constant structure of the LP constraint matrix
A_ub_structure = np.zeros((N_PAIRS + 4 * N, N))
for k, (i, j) in enumerate(PAIR_INDICES):
    A_ub_structure[k, i] = 1.0
    A_ub_structure[k, j] = 1.0
for i in range(N):
    base = N_PAIRS + 4 * i
    A_ub_structure[base, i] = 1.0
    A_ub_structure[base + 1, i] = 1.0
    A_ub_structure[base + 2, i] = 1.0
    A_ub_structure[base + 3, i] = 1.0

def solve_lp_and_grad(centers):

    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),

        mu_R = duals[base + 1]
        mu_B = duals[base + 2]
        mu_T = duals[base + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T

def obj_and_grad_centers(c_flat):
    """Objective and gradient for L-BFGS-B optimization on centers."""
    c = np.clip(c_flat.reshape(N, 2), 1e-7, 1.0 - 1e-7)
    _, s, grad = solve_lp_and_grad(c)
    # minimize -sum(radii) => gradient is negative of ascent gradient
    return -s, -grad.flatten()

def optimize_l_bfgsb(c0):
    """Runs L-BFGS-B on centers to maximize sum of radii."""
    try:
        res = minimize(obj_and_grad_centers, c0.flatten(), method='L-BFGS-B',
                       bounds=[(0.01, 0.99)] * (2 * N), jac=True,
                       options={'maxiter': 800, 'ftol': 1e-14, 'gtol': 1e-11})
        return res.x.reshape(N, 2), -res.fun

        return c0, s

def coordinate_descent(centers, n_sweeps=5):
    """Optimizes one circle at a time using Nelder-Mead."""
    c = centers.copy()
    for _ in range(n_sweeps):
        for i in range(N):
            def obj_i(xy_flat):
                c_temp = c.copy()
                c_temp[i] = np.clip(xy_flat, 0.01, 0.99)
                _, s, _ = solve_lp_and_grad(c_temp)
                return -s
            
            try:
                res = minimize(obj_i, c[i], method='Nelder-Mead',
                               options={'maxiter': 400, 'xatol': 1e-6, 'fatol': 1e-6})
                if res.success:
                    c[i] = res.x
            except Exception:
                pass
    _, s, _ = solve_lp_and_grad(c)
    return c, s

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint_sq(v):
    """Computes boundary and squared non-overlap constraints (must be >= 0)."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    cons = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    # Squared distance constraint: dx^2 + dy^2 - (r_i+r_j)^2 >= 0
    cons.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(cons)
