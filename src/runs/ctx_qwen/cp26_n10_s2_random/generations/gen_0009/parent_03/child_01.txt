N = 26
_NUM_PAIRS = N * (N - 1) // 2
_A_LP = np.zeros((_NUM_PAIRS + 4 * N, N))
_PAIRS = []
_k = 0
for _i in range(N):

    _A_LP[base+2, _i] = 1.0
    _A_LP[base+3, _i] = 1.0

def solve_lp(centers):

    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diff = centers[:, None, :] - centers[None, :, :]

        idx += 1

        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1

    return np.zeros(n), 0.0, np.zeros_like(b)

def compute_grad(centers, duals):

    g = np.zeros_like(centers)
    diff = centers[:, None, :] - centers[None, :, :]

    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    _, s, _ = solve_lp(c)
    return -s

def slsqp_obj(v):
    return -np.sum(v[2*N:])

def slsqp_cons(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]

    dr = r[_TRI_U_IDX[0]] + r[_TRI_U_IDX[1]]
    con.append(dx**2 + dy**2 - dr**2)
