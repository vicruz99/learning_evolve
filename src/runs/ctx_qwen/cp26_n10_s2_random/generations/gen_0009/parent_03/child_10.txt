import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute LP constraint matrix structure
num_pairs = N * (N - 1) // 2
A_lp = np.zeros((num_pairs + 4 * N, N))
pair_idx = []
k = 0
for i in range(N):
    for j in range(i + 1, N):
        A_lp[k, i] = 1.0
        A_lp[k, j] = 1.0
        pair_idx.append((i, j))
        k += 1
for i in range(N):
    base = num_pairs + 4 * i
    A_lp[base, i] = 1.0
    A_lp[base + 1, i] = 1.0
    A_lp[base + 2, i] = 1.0
    A_lp[base + 3, i] = 1.0

# Precompute pairwise indices for SLSQP constraints
TRIU_IND = np.triu_indices(N, k=1)

def solve_lp_radii(centers):
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    b = np.zeros(A_lp.shape[0])
    idx = 0
    for i, j in pair_idx:
        b[idx] = dists[i, j]
        idx += 1
    for i in range(N):
        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_lp, b_ub=b, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:
        if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            duals = res.ineqlin.marginals
        elif hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
            duals = res.marginals.ineqlin
        else:
            duals = np.zeros_like(b)
        return res.x, np.sum(res.x), duals
    return np.zeros(N), 0.0, np.zeros_like(b)

def compute_gradient(centers, duals):
    n = N
    grad = np.zeros_like(centers)
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    idx = 0
    for i, j in pair_idx:
        mu = duals[idx]
        if mu > 1e-9:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += mu * vec
                grad[j] -= mu * vec
        idx += 1
        
    bound_start = len(pair_idx)
    for i in range(n):
        mu_L = duals[bound_start + 4*i]
        mu_R = duals[bound_start + 4*i + 1]
        mu_B = duals[bound_start + 4*i + 2]
        mu_T = duals[bound_start + 4*i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
    return grad

def obj_and_grad(x):
    c = x.reshape(N, 2)
    val, _, grad = solve_lp_radii(c)
    return -val, -grad.flatten()

def generate_starts(rng):
    starts = []
    pats = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [4,6,6,6,4],
        [6,6,5,5,4], [5,4,6,6,5], [6,5,5,5,5], [5,5,5,6,5],
        [4,5,6,5,6], [5,6,4,5,6], [6,4,6,5,5], [5,5,4,6,6],
        [5,5,5,5,5,1]
    ]
    for pat in pats:
        for r0 in [0.088, 0.092, 0.096, 0.100, 0.104, 0.108]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            c = np.array(c[:N])
            c += rng.normal(0, 0.002, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    # Corner/edge focused
    for _ in range(10):
        c = np.zeros((N, 2))
        c[0] = [0.1, 0.1]; c[1] = [0.9, 0.1]
        c[2] = [0.1, 0.9]; c[3] = [0.9, 0.9]
        c[4] = [0.5, 0.1]; c[5] = [0.5, 0.9]
        c[6] = [0.1, 0.5]; c[7] = [0.9, 0.5]
        idx = 8
        r0 = 0.09
        y = 0.28
        row = 0
        while idx < N:
            x_start = 0.3 if row % 2 == 0 else 0.39
            x = x_start
            while x <= 0.7 and idx < N:
                c[idx] = [x + rng.uniform(-0.02, 0.02), y + rng.uniform(-0.02, 0.02)]
                idx += 1
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    # Force-directed random
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(400):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i+1, N):
                    d = np.linalg.norm(c[i]-c[j])
                    if d < 0.22 and d > 1e-4:
                        f = (0.22 - d) / (d + 1e-4) * 0.006
                        diff = c[i] - c[j]
                        forces[i] += diff * f
                        forces[j] -= diff * f
            c += forces
            c = np.clip(c, 0.05, 0.95)
        starts.append(c)
    return starts

def slsqp_obj(v):
    return -np.sum(v[2*N:])

def slsqp_cons(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    dx = c[TRIU_IND[0], 0] - c[TRIU_IND[1], 0]
    dy = c[TRIU_IND[0], 1] - c[TRIU_IND[1], 1]
    dr = r[TRIU_IND[0]] + r[TRIU_IND[1]]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def slsqp_joint(c_init, r_init):
    v0 = np.concatenate([c_init.flatten(), r_init])
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': 6000, 'ftol': 1e-13})
        if np.min(slsqp_cons(res.x)) >= -1e-7:
            return res.x[:2*N].reshape(N, 2), res.x[2*N:], -res.fun
    except Exception:
        pass
    return c_init, r_init, 0.0

def repair(centers, radii):
    for _ in range(200):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple:
    rng = np.random.default_rng(42)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = generate_starts(rng)
    bounds_c = [(0.02, 0.98)] * (2 * N)
    
    # Phase 1: L-BFGS-B on centers using LP radii
    for c0 in starts:
        try:
            res = minimize(obj_and_grad, c0.flatten(), method='L-BFGS-B',
                           bounds=bounds_c, jac=True,
                           options={'maxiter': 2500, 'ftol': 1e-14, 'gtol': 1e-10})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt, _ = solve_lp_radii(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Joint SLSQP polish & Perturbation search
    if best_c is not None:
        c_p, r_p, s_p = slsqp_joint(best_c, best_r)
        if s_p > best_sum:
            best_sum = s_p
            best_c = c_p
            best_r = r_p
            
        for step in range(35):
            noise = 0.006 * (0.88 ** step)
            c_per = best_c + rng.normal(0, noise, best_c.shape)
            c_per = np.clip(c_per, 0.02, 0.98)
            
            r_per, s_per, _ = solve_lp_radii(c_per)
            c_opt, r_opt, s_opt = slsqp_joint(c_per, r_per)
            if s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt
                best_r = r_opt
                
        # Final LP update on best centers
        r_final, s_final, _ = solve_lp_radii(best_c)
        if s_final > best_sum:
            best_sum = s_final
            best_r = r_final
            
    radii = repair(best_c.copy(), best_r.copy())
    return best_c, radii, float(np.sum(radii))
