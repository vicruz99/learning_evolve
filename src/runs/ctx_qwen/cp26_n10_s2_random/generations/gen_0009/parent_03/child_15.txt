TRIU_I, TRIU_J = np.triu_indices(N, 1)

NUM_BOUND = 4 * N
A_LP = np.zeros((NUM_PAIRS + NUM_BOUND, N))
PAIR_IDX = []
idx = 0

    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0

        idx += 1

    base = NUM_PAIRS + 4 * i

    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_radii(centers):

    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    b = np.zeros(A_LP.shape[0])
    k = 0

        b[k] = dists[i, j]
        k += 1

        b[k] = centers[i, 0]; k += 1
        b[k] = 1.0 - centers[i, 0]; k += 1
        b[k] = centers[i, 1]; k += 1
        b[k] = 1.0 - centers[i, 1]; k += 1

                  bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:

    return np.zeros(N), 0.0, np.zeros_like(b)

def compute_lp_gradient(centers, duals):

    grad = np.zeros_like(centers)
    diffs = centers[:, None, :] - centers[None, :, :]

                grad[j] -= mu * vec
        k += 1

        mu_T = duals[bound_start + 4*i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return grad

def force_directed_init(rng):
    """Spread points using simple repulsive forces."""
    centers = rng.uniform(0.2, 0.8, (N, 2))
    for _ in range(500):
        forces = np.zeros_like(centers)
        for i in range(N):
            for j in range(i + 1, N):
                d_vec = centers[i] - centers[j]
                d = np.linalg.norm(d_vec)
                if d < 0.25 and d > 1e-6:
                    f = (0.25 - d) / d * 0.01
                    forces[i] += d_vec * f
                    forces[j] -= d_vec * f
        centers += forces
        centers = np.clip(centers, 0.05, 0.95)
    return centers

def grow_circles_init(rng):
    """Initialize by growing circles and resolving overlaps."""
    centers = rng.uniform(0.1, 0.9, (N, 2))
    radii = np.full(N, 0.01)

def hex_init(rng, pattern, r0):
    """Hexagonal lattice initialization."""
    centers = []
    y = r0
    for r_idx, cnt in enumerate(pattern):
        shift = r0 if r_idx % 2 == 1 else 0.0

                centers.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
            x += 2.0 * r0
        y += r0 * np.sqrt(3.0)
    centers = np.array(centers[:N])
    centers += rng.normal(0, 0.002, centers.shape)
    return np.clip(centers, 0.05, 0.95)

def slsqp_obj(v):
    return -np.sum(v[2*N:])

def slsqp_cons(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r,
        np.sum((c[TRIU_I] - c[TRIU_J])**2, axis=1) - (r[TRIU_I] + r[TRIU_J])**2
    ]
    return np.concatenate(con)

SLSQP_BOUNDS = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

def slsqp_optimize(c_init, r_init):
    v0 = np.concatenate([c_init.flatten(), r_init])
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=SLSQP_BOUNDS,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': 8000, 'ftol': 1e-14})
        con_vals = slsqp_cons(res.x)
        if np.min(con_vals) >= -1e-7:
            return res.x[:2*N].reshape(N, 2), res.x[2*N:], -res.fun
    except Exception:
        pass
    return c_init, r_init, 0.0

def repair(centers, radii):
    radii = radii.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):

                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)
