import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_IND = np.triu_indices(N, 1)

def get_lp_structure():
    num_pairs = N * (N - 1) // 2
    num_bound = 4 * N
    A = np.zeros((num_pairs + num_bound, N))
    pair_idx = []
    k = 0
    for i in range(N):
        for j in range(i + 1, N):
            A[k, i] = 1.0
            A[k, j] = 1.0
            pair_idx.append((i, j))
            k += 1
    for i in range(N):
        base = num_pairs + 4 * i
        A[base, i] = 1.0
        A[base + 1, i] = 1.0
        A[base + 2, i] = 1.0
        A[base + 3, i] = 1.0
    return A, pair_idx

A_LP, PAIR_IDX = get_lp_structure()

def solve_lp_radii(centers):
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    b = np.zeros(A_LP.shape[0])
    idx = 0
    for i, j in PAIR_IDX:
        b[idx] = dists[i, j]
        idx += 1
    for i in range(n):
        b[idx] = centers[i, 0]; idx += 1
        b[idx] = 1.0 - centers[i, 0]; idx += 1
        b[idx] = centers[i, 1]; idx += 1
        b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:
        return res.x, np.sum(res.x)
    return np.zeros(n), 0.0

def penalty_obj(v, mu):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    base = -np.sum(r)
    
    bounds_dist = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                             np.minimum(c[:, 1], 1.0 - c[:, 1]))
    pen_bound = np.sum(np.maximum(0.0, r - bounds_dist)**2)
    
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = r[:, None] + r[None, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    pen_pair = np.sum(np.maximum(0.0, r_sum - dists)[mask]**2)
    
    return base + mu * (pen_bound + pen_pair)

def slsqp_obj(v):
    return -np.sum(v[2*N:])

def slsqp_cons(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    dx = c[TRIU_IND[0], 0] - c[TRIU_IND[1], 0]
    dy = c[TRIU_IND[0], 1] - c[TRIU_IND[1], 1]
    dr = r[TRIU_IND[0]] + r[TRIU_IND[1]]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def get_bounds_opt():
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

OPT_BOUNDS = get_bounds_opt()

def force_directed_init(rng):
    pts = rng.uniform(0.2, 0.8, (N, 2))
    for _ in range(300):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(pts[i,0]-pts[j,0], pts[i,1]-pts[j,1])
                if d < 0.25 and d > 1e-6:
                    f = (0.25 - d) * 0.05
                    diff = pts[i] - pts[j]
                    forces[i] += diff / d * f
                    forces[j] -= diff / d * f
        pts += forces
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def generate_starts(rng):
    starts = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 5, 6, 5, 5]
    ]
    for pat in patterns:
        for r0 in [0.085, 0.095, 0.105]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3.0)
            starts.append(np.array(c[:N]))
            
    for _ in range(10):
        starts.append(force_directed_init(rng))
        
    for _ in range(15):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))
        
    return starts

def optimize_start(c_init, rng):
    ub = np.minimum(np.minimum(c_init[:, 0], 1.0 - c_init[:, 0]),
                    np.minimum(c_init[:, 1], 1.0 - c_init[:, 1]))
    dists = np.linalg.norm(c_init[:, None, :] - c_init[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    rp = 0.5 * np.min(dists, axis=1)
    r_init = np.minimum(ub, rp) * 0.8
    
    v0 = np.concatenate([c_init.flatten(), r_init])
    
    mu = 50.0
    v_curr = v0.copy()
    for _ in range(6):
        res = minimize(penalty_obj, v_curr, args=(mu,), method='L-BFGS-B',
                       bounds=OPT_BOUNDS, options={'maxiter': 1500, 'ftol': 1e-12})
        v_curr = res.x.copy()
        mu *= 1.8
        
        v_curr[:2*N] += rng.normal(0, 0.001, 2*N)
        v_curr = np.clip(v_curr, 0.0, 1.0)
        v_curr[2*N:] = np.clip(v_curr[2*N:], 0.0, 0.5)
        
    try:
        res2 = minimize(slsqp_obj, v_curr, method='SLSQP', bounds=OPT_BOUNDS,
                        constraints={'type': 'ineq', 'fun': slsqp_cons},
                        options={'maxiter': 5000, 'ftol': 1e-13})
        if np.min(slsqp_cons(res2.x)) >= -1e-7:
            v_curr = res2.x
    except Exception:
        pass
        
    c_final = v_curr[:2*N].reshape(N, 2)
    r_final, s_final = solve_lp_radii(c_final)
    
    return c_final, r_final, s_final

def repair_packing(centers, radii):
    radii = radii.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple:
    rng = np.random.default_rng(123)
    best_c = None
    best_r = None
    best_sum = -1.0
    
    starts = generate_starts(rng)
    
    for c_init in starts:
        c_opt, r_opt, s_opt = optimize_start(c_init, rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_c = c_opt.copy()
            best_r = r_opt.copy()
            
    final_radii = repair_packing(best_c.copy(), best_r.copy())
    return best_c, final_radii, float(np.sum(final_radii))
