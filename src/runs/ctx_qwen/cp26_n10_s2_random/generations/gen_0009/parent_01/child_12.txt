import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute LP constraint matrix structure and pair indices
PAIR_IDX = []
k = 0
for i in range(N):
    for j in range(i + 1, N):
        PAIR_IDX.append((i, j))
        k += 1
num_pairs = k

A_LP = np.zeros((num_pairs + 4 * N, N))
for idx, (i, j) in enumerate(PAIR_IDX):
    A_LP[idx, i] = 1.0
    A_LP[idx, j] = 1.0
    
for i in range(N):
    base = num_pairs + 4 * i
    A_LP[base + 0, i] = 1.0
    A_LP[base + 1, i] = 1.0
    A_LP[base + 2, i] = 1.0
    A_LP[base + 3, i] = 1.0

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii given fixed centers and computes exact gradient."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    b = np.zeros(A_LP.shape[0])
    for idx, (i, j) in enumerate(PAIR_IDX):
        b[idx] = dists[i, j]
        
    for i in range(N):
        base = num_pairs + 4 * i
        b[base + 0] = centers[i, 0]
        b[base + 1] = 1.0 - centers[i, 0]
        b[base + 2] = centers[i, 1]
        b[base + 3] = 1.0 - centers[i, 1]
        
    try:
        res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b, 
                      bounds=[(0, u) for u in ub], method='highs')
        if not res.success:
            return np.zeros(n), 0.0, np.zeros_like(centers)
    except Exception:
        return np.zeros(n), 0.0, np.zeros_like(centers)
        
    radii = res.x
    try:
        # Handle different scipy versions for marginals
        if hasattr(res, 'marginals'):
            duals = np.asarray(res.marginals.ineqlin)
        elif hasattr(res, 'ineqlin'):
            duals = np.asarray(res.ineqlin.marginals)
        else:
            duals = np.zeros(A_LP.shape[0])
    except Exception:
        duals = np.zeros(A_LP.shape[0])
        
    grad = np.zeros_like(centers)
    for idx, (i, j) in enumerate(PAIR_IDX):
        lam = duals[idx]
        if lam > 1e-8:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
                
    for i in range(N):
        base = num_pairs + 4 * i
        grad[i, 0] += duals[base + 0] - duals[base + 1]
        grad[i, 1] += duals[base + 2] - duals[base + 3]
        
    return radii, np.sum(radii), grad

def force_directed_init(rng, n_trials=400):
    """Generates a well-spaced initial configuration using repulsive forces."""
    c = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(n_trials):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                d_vec = c[i] - c[j]
                d = np.linalg.norm(d_vec)
                if d < 0.16 and d > 1e-6:
                    f = (0.16 - d) / d
                    forces[i] += d_vec * f
                    forces[j] -= d_vec * f
        c += forces * 0.003
        c = np.clip(c, 0.05, 0.95)
    return c

def optimize_centers(c0, max_iter=1500, rng=None):
    """Performs gradient ascent on centers using LP duals."""
    c = c0.copy()
    best_c = c.copy()
    best_sum = -1.0
    step = 0.006
    
    radii, s, grad = solve_lp_and_grad(c)
    best_sum = s
    
    for k in range(max_iter):
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-10:
            break
            
        step = min(step * 1.015, 0.035)
        c_new = c + step * grad / g_norm
        c_new = np.clip(c_new, 0.01, 0.99)
        
        _, s_new, grad_new = solve_lp_and_grad(c_new)
        
        if s_new > s + 1e-12:
            c = c_new
            s = s_new
            grad = grad_new
            if s > best_sum:
                best_sum = s
                best_c = c.copy()
        else:
            step *= 0.8
            
        if k > 0 and k % 200 == 0 and rng is not None:
            c += rng.normal(0, 0.004, c.shape)
            c = np.clip(c, 0.02, 0.98)
            _, s, grad = solve_lp_and_grad(c)
            
        if step < 1e-9:
            break
    return best_c, best_sum

def slsqp_obj(v):
    return -np.sum(v[2 * N:])

def slsqp_cons(v):
    c = v[:2 * N].reshape(N, 2)
    r = v[2 * N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    i, j = np.triu_indices(N, 1)
    dx = c[i, 0] - c[j, 0]
    dy = c[i, 1] - c[j, 1]
    dr = r[i] + r[j]
    con.append(dx**2 + dy**2 - dr**2)
    return np.concatenate(con)

def polish_slsqp(centers):
    """Refines centers and radii jointly using SLSQP."""
    radii, s, _ = solve_lp_and_grad(centers)
    v0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    try:
        res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': slsqp_cons},
                       options={'maxiter': 6000, 'ftol': 1e-14})
        if np.min(slsqp_cons(res.x)) >= -1e-8:
            return res.x[:2 * N].reshape(N, 2), res.x[2 * N:], np.sum(res.x[2 * N:])
    except Exception:
        pass
    return centers, radii, s

def repair(centers, radii):
    """Deterministic repair to ensure strict validation compliance."""
    radii = radii.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-11:
                radii[i] = max(mr, 0.0)
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d - 1e-11:
                    shrink = (radii[i] + radii[j] - d) * 0.5 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple:

        if s_opt > best_sum:
