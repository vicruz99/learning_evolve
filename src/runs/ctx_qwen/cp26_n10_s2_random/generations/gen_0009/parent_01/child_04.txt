from scipy.optimize import linprog, minimize

N = 26

def setup_lp():
    """Precompute the sparse structure of the LP constraint matrix."""
    num_pairs = N * (N - 1) // 2
    num_bound = 4 * N
    A = np.zeros((num_pairs + num_bound, N))
    p_idx = []
    k = 0
    for i in range(N):

            A[k, i] = 1.0
            A[k, j] = 1.0
            p_idx.append((i, j))
            k += 1
    for i in range(N):
        for _ in range(4):
            A[k, i] = 1.0
            k += 1
    return A, p_idx

A_LP, PAIR_IDX = setup_lp()

def solve_lp_and_grad(centers):

        return np.zeros(N), 0.0, np.zeros_like(centers)

def obj_grad(v):
    """Objective and gradient for scipy L-BFGS-B: minimizes negative sum of radii."""
    c = v.reshape(N, 2)
    _, s, g = solve_lp_and_grad(c)
    return -s, -g.flatten()

def generate_starts(rng):

                    changed = True

    return np.maximum(radii, 0.0)
