The LP dual variables (marginals) directly provide the gradient of the maximum sum of radii with respect to center positions. This allows us to perform gradient ascent on the centers efficiently, pushing them apart and towards boundaries to maximize available space.

4. **Joint SLSQP Polish**: The best centers and radii from gradient ascent are passed to `scipy.optimize.minimize` with SLSQP for a high-precision joint optimization of both variables simultaneously.
5. **Deterministic Repair**: A final strict numerical repair step shrinks radii minimally to guarantee all constraints are satisfied within the `1e-12` tolerance required by the validator.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def setup_lp_matrices():
    num_pairs = N * (N - 1) // 2
    num_bound = 4 * N
    A_ub = np.zeros((num_pairs + num_bound, N))
    pair_indices = []
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            pair_indices.append((i, j))
            idx += 1
    for i in range(N):
        for _ in range(4):
            A_ub[idx, i] = 1.0
            idx += 1
    return A_ub, pair_indices

def solve_lp_and_grad(centers, A_ub, pair_indices):
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-12)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(A_ub.shape[0])
    idx = 0
    for i, j in pair_indices:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(n):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, u) for u in ub], method='highs')
    if not res.success:
        return None, -1.0, np.zeros_like(centers)
        
    radii = res.x
    
    # Extract duals safely across scipy versions
    duals = np.zeros_like(b_ub)
    if hasattr(res, 'marginals') and hasattr(res.marginals, 'ineqlin'):
        duals = res.marginals.ineqlin
    elif hasattr(res, 'ineqlin'):
        duals = res.ineqlin.marginals
        
    grad = np.zeros_like(centers)
    idx = 0
    for i, j in pair_indices:
        lam = duals[idx]
        if lam > 1e-9:
            d = dists[i, j]
            if d > 1e-12:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    b_start = len(pair_indices)
    for i in range(n):
        grad[i, 0] += duals[b_start + 4*i] - duals[b_start + 4*i + 1]
        grad[i, 1] += duals[b_start + 4*i + 2] - duals[b_start + 4*i + 3]
        
    return radii, np.sum(radii), grad

def run_gradient_ascent(c0, A_ub, pair_indices, max_iter=3000, init_step=0.005, rng=None):
    centers = c0.copy()
    best_c = centers.copy()
    best_sum = -1.0
    step = init_step
    
    radii, curr_sum, grad = solve_lp_and_grad(centers, A_ub, pair_indices)
    if radii is None:
        return c0, -1.0, None
    best_sum = curr_sum
    
    for k in range(max_iter):
        gn = np.linalg.norm(grad)
        if gn > 1e-10:
            centers += step * grad / gn
            
        centers = np.clip(centers, 0.001, 0.999)
        
        radii, curr_sum, grad = solve_lp_and_grad(centers, A_ub, pair_indices)
        if radii is None:
            break
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_c = centers.copy()
            step = min(step * 1.03, 0.02)
        else:
            step *= 0.97
            
        if k > 0 and k % 200 == 0:
            noise = 0.002 * (0.5 ** (k // 200))
            if rng is not None:
                centers += rng.normal(0, noise, centers.shape)
            centers = np.clip(centers, 0.001, 0.999)
            radii, curr_sum, grad = solve_lp_and_grad(centers, A_ub, pair_indices)
            
        if step < 1e-8:
            break
            
    return best_c, best_sum, radii

def generate_starts(n_starts, rng):
    starts = []
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5],
        [6, 6, 5, 5, 4], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 4, 6, 5, 5], [5, 5, 5, 6, 5], [5, 5, 4, 6, 6],
        [7, 5, 5, 5, 4], [5, 7, 5, 5, 4], [6, 5, 5, 5, 5]
    ]
    
    for pat in patterns:
        for r0 in [0.085, 0.090, 0.095, 0.100, 0.105, 0.110]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                shift = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + shift
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += r0 * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    for _ in range(15):
        c = rng.uniform(0.1, 0.9, (N, 2))
        starts.append(c)
        
    for _ in range(5):
        c = np.zeros((N, 2))
        c[0] = [0.1, 0.1]; c[1] = [0.9, 0.1]
        c[2] = [0.1, 0.9]; c[3] = [0.9, 0.9]
        c[4:] = rng.uniform(0.2, 0.8, (N-4, 2))
        c += rng.normal(0, 0.01, c.shape)
        c = np.clip(c, 0.05, 0.95)
        starts.append(c)
        
    return starts[:n_starts]

def compute_constraints_joint(v):
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = np.concatenate([c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r])
    idx_i, idx_j = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx_i] - c[idx_j], axis=1)
    con = np.concatenate([con, d - (r[idx_i] + r[idx_j])])
    return con

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    A_ub, pair_indices = setup_lp_matrices()
    
    starts = generate_starts(50, rng)
    
    best_centers = None
    best_sum = -1.0
    best_radii = None
    
    # Phase 1: LP-Dual Gradient Ascent from multiple starts
    for i, c0 in enumerate(starts):
        c_opt, s_opt, r_opt = run_gradient_ascent(c0, A_ub, pair_indices, max_iter=3000, init_step=0.006, rng=rng)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    # Phase 2: SLSQP Joint Polish
    if best_centers is not None:
        bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        v0 = np.concatenate([best_centers.flatten(), best_radii])
        
        try:
            res = minimize(lambda v: -np.sum(v[2*N:]), v0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': compute_constraints_joint},
                           options={'maxiter': 15000, 'ftol': 1e-14})
            if np.min(compute_constraints_joint(res.x)) >= -1e-7:
                s_new = np.sum(res.x[2*N:])
                if s_new > best_sum:
                    best_sum = s_new
                    best_centers = res.x[:2*N].reshape(N, 2).copy()
                    best_radii = res.x[2*N:].copy()
        except Exception:
            pass
            
        # Phase 3: Perturbation & Re-polish to escape local minima
        for _ in range(20):
            c_pert = best_centers + rng.normal(0, 0.002, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            r_lp, s_lp, _ = solve_lp_and_grad(c_pert, A_ub, pair_indices)
            if r_lp is None: continue
            
            v_pert = np.concatenate([c_pert.flatten(), r_lp])
            try:
                res = minimize(lambda v: -np.sum(v[2*N:]), v_pert, method='SLSQP', bounds=bounds,
                               constraints={'type': 'ineq', 'fun': compute_constraints_joint},
                               options={'maxiter': 8000, 'ftol': 1e-13})
                if np.min(compute_constraints_joint(res.x)) >= -1e-7:
                    s_new = np.sum(res.x[2*N:])
                    if s_new > best_sum:
                        best_sum = s_new
                        best_centers = res.x[:2*N].reshape(N, 2).copy()
                        best_radii = res.x[2*N:].copy()
            except Exception:
                pass

    # Phase 4: Strict Numerical Repair
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                req = radii[i] + radii[j]
                if d < req - 1e-11:
                    shrink = (req - d) / 2.0 + 1e-11
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
