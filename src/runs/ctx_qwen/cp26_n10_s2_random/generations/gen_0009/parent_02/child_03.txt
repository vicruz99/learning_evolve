import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def setup_lp_matrices():
    """Pre-construct the sparse structure of the LP constraint matrix."""
    num_pairs = N * (N - 1) // 2
    num_bound = 4 * N
    A_ub = np.zeros((num_pairs + num_bound, N))
    pair_indices = []
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            pair_indices.append((i, j))
            idx += 1
    bound_start = idx
    for i in range(N):
        for _ in range(4):
            A_ub[idx, i] = 1.0
            idx += 1
    return A_ub, pair_indices, bound_start

def get_lp_radii(centers, A_ub, pair_indices, bound_start):
    """Solves LP to find maximum feasible radii for fixed centers."""
    n = centers.shape[0]
    ub_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                          np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub_bound, 1e-12)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(A_ub.shape[0])
    idx = 0
    for i, j in pair_indices:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(n):
        b_ub[idx] = centers[i, 0]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        b_ub[idx] = centers[i, 1]; idx += 1
        b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    bounds_r = [(0, u) for u in ub]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.full(n, 0.02), 0.52

def obj_centers(v, A_ub, pair_indices, bound_start):
    """Objective function for center optimization: maximize sum of LP radii."""
    c = np.clip(v.reshape(N, 2), 0.005, 0.995)
    _, s = get_lp_radii(c, A_ub, pair_indices, bound_start)
    return -s

def force_directed_init(rng):
    """Generates an initial configuration using repulsive forces."""
    c = rng.uniform(0.2, 0.8, (N, 2))
    for _ in range(500):
        for i in range(N):
            for j in range(i + 1, N):
                d_vec = c[i] - c[j]
                dist = np.linalg.norm(d_vec)
                if dist < 0.15 and dist > 1e-6:
                    push = (0.15 - dist) * 0.05
                    c[i] += d_vec / dist * push
                    c[j] -= d_vec / dist * push
        c = np.clip(c, 0.1, 0.9)
    return c

def hex_init(pat, r0, rng):
    """Generates an initial configuration from a perturbed hexagonal lattice."""
    c = []
    y = r0
    for r_idx, cnt in enumerate(pat):
        shift = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(c) < N:
                c.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    c = np.array(c[:N])
    c += rng.normal(0, 0.002, c.shape)
    c = np.clip(c, 0.05, 0.95)
    return c

def obj_joint(v):
    """Objective for joint SLSQP polish: minimize negative sum of radii."""
    return -np.sum(v[2 * N:])

def cons_joint(v):
    """Constraints for joint SLSQP polish: boundary and non-overlap."""
    cc = v[:2 * N].reshape(N, 2)
    rr = v[2 * N:]
    con = [cc[:, 0] - rr, 1.0 - cc[:, 0] - rr, cc[:, 1] - rr, 1.0 - cc[:, 1] - rr]
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(cc[idx[0]] - cc[idx[1]], axis=1)
    con.append(d - (rr[idx[0]] + rr[idx[1]]))
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize sum of radii."""
    np.random.seed(42)
    rng = np.random.default_rng(42)
    A_ub, pair_indices, bound_start = setup_lp_matrices()
    
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5],
        [6, 6, 5, 5, 4], [4, 6, 6, 6, 4], [5, 5, 5, 5, 6],
        [7, 5, 5, 5, 4], [5, 7, 5, 5, 4], [6, 4, 6, 5, 5],
        [4, 5, 6, 6, 5], [5, 4, 6, 6, 5], [6, 5, 5, 6, 4]
    ]
    
    inits = []
    for pat in patterns:
        for r0 in [0.085, 0.090, 0.095, 0.100, 0.105, 0.110]:
            inits.append(hex_init(pat, r0, rng))
            
    for _ in range(20):
        inits.append(force_directed_init(rng))
        
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        inits.append(c)
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds_centers = [(0.01, 0.99)] * (2 * N)
    
    # Phase 1: Global center optimization using Powell
    for i, c0 in enumerate(inits):
        try:
            res = minimize(obj_centers, c0.flatten(), args=(A_ub, pair_indices, bound_start),
                          method='Powell', bounds=bounds_centers,
                          options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            c_opt = np.clip(res.x.reshape(N, 2), 0.005, 0.995)
            radii, s = get_lp_radii(c_opt, A_ub, pair_indices, bound_start)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = radii.copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation refinement to escape local minima
    for _ in range(25):
        scale = 0.004 + rng.uniform(0, 0.003)
        c_pert = best_centers + rng.normal(0, scale, best_centers.shape)
        c_pert = np.clip(c_pert, 0.02, 0.98)
        try:
            res = minimize(obj_centers, c_pert.flatten(), args=(A_ub, pair_indices, bound_start),
                          method='Powell', bounds=bounds_centers,
                          options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False})
            c_opt = np.clip(res.x.reshape(N, 2), 0.005, 0.995)
            radii, s = get_lp_radii(c_opt, A_ub, pair_indices, bound_start)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = radii.copy()
        except Exception:
            pass
            
    # Phase 3: Joint SLSQP Polish
    v0 = np.concatenate([best_centers.flatten(), best_radii])
    bounds_joint = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    try:
        res_sl = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                         constraints={'type': 'ineq', 'fun': cons_joint},
                         options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if np.min(cons_joint(res_sl.x)) >= -1e-8:
            s_sl = np.sum(res_sl.x[2 * N:])
            if s_sl > best_sum:
                best_sum = s_sl
                best_centers = res_sl.x[:2 * N].reshape(N, 2)
                best_radii = res_sl.x[2 * N:]
    except Exception:
        pass
        
    # Phase 4: Strict Deterministic Repair
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0],
                             best_centers[i, 1] - best_centers[j, 1])
                if d < best_radii[i] + best_radii[j] - 1e-11:
                    shrink = (best_radii[i] + best_radii[j] - d) / 2.0 + 1e-11
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(best_centers[i, 0], 1.0 - best_centers[i, 0],
                     best_centers[i, 1], 1.0 - best_centers[i, 1])
            if best_radii[i] > mr + 1e-11:
                best_radii[i] = mr
                changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
