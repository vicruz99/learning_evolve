To maximize the sum of radii for 26 circles in a unit square, we leverage a hybrid optimization strategy that combines Linear Programming (LP) with gradient-based and constraint-based non-linear optimization:

1. **LP-Radii Maximization with Exact Gradient**: For any fixed set of centers, the maximum feasible radii can be found by solving a simple LP. Crucially, the dual variables (marginals) of this LP provide the exact gradient of the sum of radii with respect to the center positions. This allows us to efficiently perform gradient ascent on the centers alone.
2. **L-BFGS-B on Centers**: We run `L-BFGS-B` using the exact LP gradient to push centers into configurations that allow larger radii. We initialize this from multiple diverse starting points: perturbed hexagonal lattices, force-directed repulsive spreads, and random starts.
3. **Basin Hopping / Perturbation**: To escape local optima, we repeatedly perturb the best-found centers and re-run the L-BFGS-B optimization.
4. **Joint SLSQP Polish**: Once high-quality centers are found, we switch to a joint non-linear optimization (`SLSQP`) over both centers and radii. This phase handles the non-smooth constraint boundaries more directly and can extract a few more percentage points of radius sum.
5. **Deterministic Repair**: Finally, a strict iterative repair step shrinks radii minimally to guarantee all pairwise and boundary constraints are satisfied within the validator's `1e-12` tolerance, ensuring 100% validation pass rate without sacrificing the optimized sum.

This approach systematically explores the configuration space, uses mathematically exact gradients for fast convergence, and polishes the result to machine precision.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_IND = np.triu_indices(N, k=1)
NUM_PAIRS = N * (N - 1) // 2

# Precompute fixed structure for LP constraint matrix to speed up repeated solves
A_ub_lp = np.zeros((NUM_PAIRS + 4*N, N))
pair_idx = []
idx = 0
for i, j in TRIU_IND:
    A_ub_lp[idx, i] = 1.0
    A_ub_lp[idx, j] = 1.0
    pair_idx.append((i, j))
    idx += 1
for i in range(N):
    for _ in range(4):
        A_ub_lp[idx, i] = 1.0
        idx += 1

def solve_lp_and_grad(centers):
    """Solves LP for optimal radii given fixed centers and computes gradient w.r.t centers."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    b_ub = np.zeros(A_ub_lp.shape[0])
    idx = 0
    for i, j in pair_idx:
        b_ub[idx] = dists[i, j]
        idx += 1
    for i in range(n):
        x, y = centers[i]
        b_ub[idx] = x; idx += 1
        b_ub[idx] = 1.0 - x; idx += 1
        b_ub[idx] = y; idx += 1
        b_ub[idx] = 1.0 - y; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub_lp, b_ub=b_ub, bounds=[(0, None)]*n, 
                  method='highs', options={'presolve': True})
    if not res.success:
        return np.zeros(n), 0.0, np.zeros_like(centers)
        
    radii = res.x
    # Handle scipy version differences for dual variables
    if hasattr(res, 'marginals'):
        duals = res.marginals.ineqlin
    else:
        duals = res.ineqlin.marginals
        
    grad = np.zeros_like(centers)
    idx = 0
    for i, j in pair_idx:
        lam = duals[idx]
        if lam > 1e-7:
            d = dists[i, j]
            if d > 1e-9:
                vec = (centers[i] - centers[j]) / d
                grad[i] += lam * vec
                grad[j] -= lam * vec
        idx += 1
        
    b_start = NUM_PAIRS
    for i in range(n):
        mu_L = duals[b_start + 4*i]
        mu_R = duals[b_start + 4*i + 1]
        mu_B = duals[b_start + 4*i + 2]
        mu_T = duals[b_start + 4*i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return radii, np.sum(radii), grad

def obj_centers(v):
    """Objective for center optimization: minimize negative sum of radii."""
    return -solve_lp_and_grad(v.reshape(N, 2))[1]

def grad_centers(v):
    """Gradient for center optimization."""
    return -solve_lp_and_grad(v.reshape(N, 2))[2].flatten()

def cons_joint(v):
    """Boundary and pairwise non-overlap constraints for joint optimization."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = [c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r]
    idx = np.triu_indices(N, 1)
    d = np.hypot(c[idx[0],0]-c[idx[1],0], c[idx[0],1]-c[idx[1],1])
    con.append(d - (r[idx[0]] + r[idx[1]]))
    return np.concatenate(con)

def obj_joint(v):
    """Objective for joint optimization."""
    return -np.sum(v[2*N:])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    rng = np.random.default_rng(42)
    best_centers = None
    best_sum = -1.0
    best_radii = None
    
    # Generate diverse initial centers
    inits = []
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [5,5,6,5,5], [6,6,5,5,4], [4,6,6,6,4]]
    for pat in patterns:
        for r0 in [0.088, 0.093, 0.098]:
            c = []
            y = r0
            for ri, cnt in enumerate(pat):
                offset = r0 if ri % 2 == 1 else 0.0
                x = r0 + offset
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            c = np.array(c[:N])
            c += rng.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.05, 0.95)
            inits.append(c)
            
    # Force-directed repulsive spreads for non-lattice starts
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(300):
            f = np.zeros_like(c)
            for i in range(N):
                for j in range(i+1, N):
                    dv = c[i]-c[j]; d = np.linalg.norm(dv)
                    if d < 0.18 and d > 1e-4:
                        ft = (0.18-d)*0.015/d
                        f[i] += dv*ft; f[j] -= dv*ft
            c += f; c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    bounds_c = [(0.01, 0.99)]*(2*N)
    
    # Phase 1: L-BFGS-B on centers to maximize LP radii sum
    for c0 in inits:
        try:
            res = minimize(obj_centers, c0.flatten(), jac=grad_centers, method='L-BFGS-B',
                          bounds=bounds_c, options={'maxiter': 1500, 'ftol': 1e-13})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt, _ = solve_lp_and_grad(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation restarts to escape local minima
    for _ in range(40):
        c_pert = best_centers + rng.normal(0, 0.004, best_centers.shape)
        c_pert = np.clip(c_pert, 0.05, 0.95)
        try:
            res = minimize(obj_centers, c_pert.flatten(), jac=grad_centers, method='L-BFGS-B',
                          bounds=bounds_c, options={'maxiter': 1000, 'ftol': 1e-13})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt, _ = solve_lp_and_grad(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Phase 3: Joint SLSQP polish for exact feasibility and slight radius boost
    if best_centers is not None:
        v0 = np.concatenate([best_centers.flatten(), best_radii])
        bounds_j = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
        for _ in range(5):
            try:
                res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_j,
                              constraints={'type': 'ineq', 'fun': cons_joint},
                              options={'maxiter': 15000, 'ftol': 1e-14})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_centers = res.x[:2*N].reshape(N, 2).copy()
                    best_radii = res.x[2*N:].copy()
                    v0 = res.x.copy()
            except Exception:
                pass
                
    # Phase 4: Strict Deterministic Repair to guarantee validation passes
    for _ in range(100):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                if d < best_radii[i] + best_radii[j] - 1e-11:
                    shrink = (best_radii[i] + best_radii[j] - d)/2.0 + 1e-11
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        # Clamp to boundaries
        for i in range(N):
            mr = min(best_centers[i,0], 1.0-best_centers[i,0], best_centers[i,1], 1.0-best_centers[i,1])
            if best_radii[i] > mr - 1e-11:
                best_radii[i] = mr
                changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
