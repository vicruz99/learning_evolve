import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_pair_indices(n):
    return np.triu_indices(n, k=1)

def solve_lp_radii(centers):
    """Solves LP to find maximum feasible radii for fixed centers."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    num_pairs = n * (n - 1) // 2
    total_cons = num_pairs + 4 * n
    A_ub = np.zeros((total_cons, n))
    b_ub = np.zeros(total_cons)
    
    idx = 0
    for i, j in zip(*get_pair_indices(n)):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        b_ub[idx] = dists[i, j]
        idx += 1
    
    for i in range(n):
        x, y = centers[i]
        vals = [x, 1.0 - x, y, 1.0 - y]
        for v in vals:
            A_ub[idx, i] = 1.0
            b_ub[idx] = max(v, 1e-12)
            idx += 1
    
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        duals = np.zeros(total_cons)
        if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
            duals = res.ineqlin.marginals
        return res.x, duals
    return np.full(n, 1e-6), np.zeros(total_cons)

def lp_gradient_ascent(centers0, max_iter=3000, init_step=0.008, rng=None):
    """Gradient ascent on centers using LP duals."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    centers = centers0.copy()
    best_centers = centers.copy()
    best_sum = -1.0
    best_radii = None
    step = init_step
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    
    radii, duals = solve_lp_radii(centers)
    best_sum = np.sum(radii)
    best_radii = radii.copy()
    
    for k in range(max_iter):
        grad = np.zeros_like(centers)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        idx = 0
        for i, j in zip(*get_pair_indices(n)):
            lam = duals[idx]
            if lam > 1e-7:
                d = dists[i, j]
                if d > 1e-9:
                    vec = (centers[i] - centers[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
            idx += 1
        
        b_start = num_pairs
        for i in range(n):
            mu_L = duals[b_start + 4*i]
            mu_R = duals[b_start + 4*i + 1]
            mu_B = duals[b_start + 4*i + 2]
            mu_T = duals[b_start + 4*i + 3]
            grad[i, 0] += mu_L - mu_R
            grad[i, 1] += mu_B - mu_T
        
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-10:
            centers += step * grad / g_norm
        else:
            step *= 0.5
            if step < 1e-8:
                break
        
        centers = np.clip(centers, 0.002, 0.998)
        radii, duals = solve_lp_radii(centers)
        curr_sum = np.sum(radii)
        
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            step = min(step * 1.12, 0.025)
        else:
            step *= 0.88
        
        if k > 0 and k % 150 == 0:
            noise_scale = 0.005 * (0.75 ** (k // 150))
            centers += rng.normal(0, noise_scale, centers.shape)
            centers = np.clip(centers, 0.01, 0.99)
            radii, duals = solve_lp_radii(centers)
    
    return best_centers, best_radii, best_sum

def obj_lp(centers_flat):
    """Objective for optimizing centers only (LP for radii)."""
    centers = centers_flat.reshape(N, 2)
    radii, _ = solve_lp_radii(centers)
    return -np.sum(radii)

def compute_sl_cons(v):
    """SLSQP constraints for joint optimization."""
    centers = v[:2*N].reshape(N, 2)
    radii = v[2*N:]
    
    con = []
    con.append(centers[:, 0] - radii)
    con.append(1.0 - centers[:, 0] - radii)
    con.append(centers[:, 1] - radii)
    con.append(1.0 - centers[:, 1] - radii)
    
    i_idx, j_idx = get_pair_indices(N)
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    dr = radii[i_idx] + radii[j_idx]
    con.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(con)

def obj_sl(v):
    return -np.sum(v[2*N:])

def hex_init(row_counts, r0, rng):
    """Hexagonal lattice initialization."""
    centers = []
    y = r0
    for r_idx, count in enumerate(row_counts):
        offset = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + offset
        for _ in range(count):
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    
    centers = np.array(centers[:N])
    centers += rng.normal(0, 0.003, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def corner_init(rng, pattern_type=0):
    """Corner-focused initialization."""
    centers = np.zeros((N, 2))
    corner_r = 0.12
    
    # 4 corners
    centers[0] = [corner_r, corner_r]
    centers[1] = [1.0 - corner_r, corner_r]
    centers[2] = [corner_r, 1.0 - corner_r]
    centers[3] = [1.0 - corner_r, 1.0 - corner_r]
    
    # Edge midpoints
    centers[4] = [0.5, corner_r]
    centers[5] = [0.5, 1.0 - corner_r]
    centers[6] = [corner_r, 0.5]
    centers[7] = [1.0 - corner_r, 0.5]
    
    # Fill remaining with hexagonal pattern in interior
    idx = 8
    r0 = 0.075
    y = 0.25
    row = 0
    while idx < N:
        x_start = 0.30 + (row % 2) * r0
        x = x_start
        while x <= 0.70 and idx < N:
            centers[idx] = [x + rng.normal(0, 0.005), y + rng.normal(0, 0.005)]
            idx += 1
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    
    while idx < N:
        centers[idx] = rng.uniform(0.2, 0.8, 2)
        idx += 1
    
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def force_directed_init(rng):
    """Force-directed initialization to spread points evenly."""
    centers = rng.uniform(0.15, 0.85, (N, 2))
    
    for _ in range(1500):
        forces = np.zeros((N, 2))
        for i in range(N):
            for j in range(i + 1, N):
                d_vec = centers[i] - centers[j]
                dist = np.linalg.norm(d_vec)
                if dist < 0.15 and dist > 1e-6:
                    push = (0.15 - dist) * 0.1
                    forces[i] += d_vec / dist * push
                    forces[j] -= d_vec / dist * push
            # Push away from boundaries
            for d in range(2):
                if centers[i, d] < 0.1:
                    forces[i, d] += (0.1 - centers[i, d]) * 2.0
                elif centers[i, d] > 0.9:
                    forces[i, d] -= (centers[i, d] - 0.9) * 2.0
        
        centers += forces * 0.5
        centers = np.clip(centers, 0.05, 0.95)
    
    return centers

def generate_all_inits(rng):

def slsqp_optimize(v0, bounds, cons, maxiter=15000):
    """Run SLSQP optimization."""
    try:
        res = minimize(obj_sl, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': compute_sl_cons},
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        return res
    except Exception:
        return None

def repair(centers, radii):
    """Strict numerical repair."""
    radii = radii.copy()
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0],
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr + 1e-12:
                radii[i] = mr
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def get_safe_radii(centers):
    """Compute strictly feasible radii for given centers."""
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    rn = 0.5 * np.min(dists, axis=1)
    return np.minimum(rb, rn) * 0.93
