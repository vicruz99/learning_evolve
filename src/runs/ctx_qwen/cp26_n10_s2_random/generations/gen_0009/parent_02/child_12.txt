import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def build_lp_structure():
    """Precompute the constant structure of the LP constraint matrix."""
    n_pairs = N * (N - 1) // 2
    A_ub = np.zeros((n_pairs + 4 * N, N))
    
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
            
    for i in range(N):
        for _ in range(4):
            A_ub[idx, i] = 1.0
            idx += 1
    return A_ub

# Global precomputed LP structure
A_LP = build_lp_structure()

def compute_lp_and_grad(centers):
    """Solves LP for max radii and computes gradient w.r.t centers using duals."""
    x, y = centers[:, 0], centers[:, 1]
    
    # Pairwise distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.hypot(dx, dy)
    
    # Build RHS vector
    n_pairs = N * (N - 1) // 2
    b_ub = np.zeros(n_pairs + 4 * N)
    
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(N):
        b_ub[idx] = x[i]; idx += 1
        b_ub[idx] = 1.0 - x[i]; idx += 1
        b_ub[idx] = y[i]; idx += 1
        b_ub[idx] = 1.0 - y[i]; idx += 1
        
    # Solve LP
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b_ub, bounds=[(0.0, None)]*N, method='highs')
    if not res.success:
        return -1.0, np.zeros_like(centers), None
        
    radii = res.x
    duals = res.ineqlin.marginals
    
    # Compute gradient
    grad = np.zeros_like(centers)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            lam = duals[idx]
            if lam > 1e-8:
                d = dists[i, j]
                if d > 1e-9:
                    vec = (centers[i] - centers[j]) / d
                    grad[i] += lam * vec
                    grad[j] -= lam * vec
            idx += 1
            
    b_start = n_pairs
    for i in range(N):
        mu_L = duals[b_start + 4*i]
        mu_R = duals[b_start + 4*i + 1]
        mu_B = duals[b_start + 4*i + 2]
        mu_T = duals[b_start + 4*i + 3]
        grad[i, 0] += mu_L - mu_R
        grad[i, 1] += mu_B - mu_T
        
    return np.sum(radii), grad, radii

def obj_flat(v):
    c = v.reshape(N, 2)
    s, _, _ = compute_lp_and_grad(c)
    return -s

def jac_flat(v):
    c = v.reshape(N, 2)
    _, g, _ = compute_lp_and_grad(c)
    return g.flatten()

def force_spread(centers, steps=800):
    """Applies repulsive forces to spread centers evenly."""
    c = centers.copy()
    for _ in range(steps):
        for i in range(N):
            for j in range(i + 1, N):
                d_vec = c[i] - c[j]
                dist = np.linalg.norm(d_vec)
                if dist < 0.16 and dist > 1e-6:
                    push = (0.16 - dist) * 0.12
                    c[i] += d_vec / dist * push
                    c[j] -= d_vec / dist * push
        c = np.clip(c, 0.06, 0.94)
    return c

def generate_starts(rng):
    """Generates diverse initial center configurations."""
    starts = []
    
    # 1. Hexagonal lattices
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4], [6, 4, 6, 5, 5], [5, 4, 6, 6, 5]
    ]
    for pat in patterns:
        for r0 in [0.088, 0.095, 0.102]:
            c = []
            y = r0
            for r_idx, cnt in enumerate(pat):
                offset = r0 if r_idx % 2 == 1 else 0.0
                x = r0 + offset
                for _ in range(cnt):
                    if len(c) < N:
                        c.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3) * r0
            c = np.array(c[:N])
            c += rng.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.05, 0.95)
            starts.append(c)
            
    # 2. Force-directed random starts
    for _ in range(12):
        c = rng.uniform(0.15, 0.85, (N, 2))
        c = force_spread(c, steps=600)
        starts.append(c)
        
    # 3. Corner & Edge biased
    for _ in range(8):
        c = rng.uniform(0.08, 0.92, (N, 2))
        # Bias towards boundaries
        c[:, 0] = np.clip(c[:, 0] + rng.normal(0, 0.05, N), 0.05, 0.95)
        c[:, 1] = np.clip(c[:, 1] + rng.normal(0, 0.05, N), 0.05, 0.95)
        c = force_spread(c, steps=500)
        starts.append(c)
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    rng = np.random.default_rng(42)
    
    bounds_c = [(0.01, 0.99)] * (2 * N)

            s = -res.fun
            if s > best_sum:
                best_sum = s

        except Exception:
            pass
            
    # Phase 2: Perturbation Search to Escape Local Minima
    if best_centers is not None:

            c_try = c_curr + rng.normal(0, noise_scale, c_curr.shape)
            c_try = np.clip(c_try, 0.02, 0.98)
            
            v_try = c_try.flatten()
            try:
                res = minimize(obj_flat, v_try, jac=jac_flat, method='L-BFGS-B',

                s = -res.fun
                if s > best_sum + 1e-8:
                    best_sum = s
                    best_centers = res.x.reshape(N, 2).copy()

            except Exception:
                pass

        v_final = best_centers.flatten()
        try:

                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2).copy()
        except Exception:
            pass

    # Phase 3: Joint SLSQP Polish (centers + radii)
    _, _, r_lp = compute_lp_and_grad(best_centers)
    if r_lp is None:
        r_lp = np.full(N, 0.05)
        
    v_joint0 = np.concatenate([best_centers.flatten(), r_lp])
    bounds_joint = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    def obj_joint(v):
        return -np.sum(v[2*N:])
        
    def cons_joint(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        con = np.concatenate([c[:,0]-r, 1.0-c[:,0]-r, c[:,1]-r, 1.0-c[:,1]-r])
        idx_i, idx_j = np.triu_indices(N, 1)
        d = np.linalg.norm(c[idx_i] - c[idx_j], axis=1)
        con = np.concatenate([con, d - (r[idx_i] + r[idx_j])])

                         constraints={'type': 'ineq', 'fun': cons_joint},

        if np.min(cons_joint(res_j.x)) >= -1e-7:
            s_j = np.sum(res_j.x[2*N:])
            if s_j > best_sum:
                best_sum = s_j
                best_centers = res_j.x[:2*N].reshape(N, 2).copy()
                best_radii = res_j.x[2*N:].copy()
    except Exception:
        pass
        
    if best_radii is None:
        _, _, best_radii = compute_lp_and_grad(best_centers)

    # Phase 4: Strict Deterministic Repair
    centers = best_centers.copy()
    radii = best_radii.copy()

        changed = False
        # Pairwise overlaps

        # Boundary violations

            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-11:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
