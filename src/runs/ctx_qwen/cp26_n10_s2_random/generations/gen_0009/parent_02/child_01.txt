from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIU_IND = np.triu_indices(N, k=1)
NUM_PAIRS = N * (N - 1) // 2
NUM_CONS = NUM_PAIRS + 4 * N

# Precompute LP matrix structure for fast radius optimization
A_LINPROG = np.zeros((NUM_CONS, N))
PAIR_INDICES_LP = []
idx = 0
for i, j in zip(*TRIU_IND):
    A_LINPROG[idx, i] = 1.0
    A_LINPROG[idx, j] = 1.0
    PAIR_INDICES_LP.append((i, j))
    idx += 1
for i in range(N):
    A_LINPROG[idx:idx+4, i] = 1.0
    idx += 4

def solve_lp_radii(centers):
    """Solves LP to find maximum feasible radii for fixed centers."""
    b = np.zeros(NUM_CONS)
    idx = 0
    for i, j in PAIR_INDICES_LP:
        b[idx] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
        idx += 1
    for i in range(N):
        x, y = centers[i]
        b[idx] = x; idx += 1
        b[idx] = 1.0 - x; idx += 1
        b[idx] = y; idx += 1
        b[idx] = 1.0 - y; idx += 1
        
    res = linprog(-np.ones(N), A_ub=A_LINPROG, b_ub=b, bounds=[(0, 0.5)]*N, method='highs')
    if res.success:
        return res.x
    return np.full(N, 0.01)

def compute_cons(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = np.empty(NUM_CONS)
    idx = 0
    c[idx:idx+N] = x - r; idx += N
    c[idx:idx+N] = 1.0 - x - r; idx += N
    c[idx:idx+N] = y - r; idx += N
    c[idx:idx+N] = 1.0 - y - r; idx += N
    dx = x[TRIU_IND[0]] - x[TRIU_IND[1]]
    dy = y[TRIU_IND[0]] - y[TRIU_IND[1]]
    dr = r[TRIU_IND[0]] + r[TRIU_IND[1]]
    c[idx:] = dx**2 + dy**2 - dr**2
    return c

def obj(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

    c = []
    y = r0
    for r_idx, cnt in enumerate(pat):
        shift = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            c.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    c = np.array(c[:N])
    c += rng.normal(0, 0.002, c.shape)
    return np.clip(c, 0.05, 0.95)

                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
