Solving this exactly (using `scipy.optimize.linprog`) gives the true maximum sum of radii for those centers, avoiding the suboptimal greedy heuristic used previously.
2. **Global Center Search with Differential Evolution**: We optimize circle centers to maximize the LP result. DE is robust for non-smooth, multi-modal landscapes and will find high-quality center configurations globally.
3. **Joint Fine-Tuning with SLSQP**: Starting from the DE-found centers and LP-optimized radii, we run `SLSQP` on the full $(x, y, r)$ space. This joint optimization allows circles to shift and resize simultaneously to squeeze out the last fraction of sum of radii, escaping local plateaus.
4. **Diverse Multi-Start & Perturbation**: We seed the joint optimizer with multiple configurations (DE result, randomized, hexagonal lattices) to ensure we don't get trapped in a poor local minimum.
5. **Strict Numerical Repair**: A final pass ensures all constraints are satisfied within the validator's $10^{-12}$ tolerance by proportionally shrinking overlapping circles and clamping to boundaries.

import numpy as np
from scipy.optimize import minimize, differential_evolution, linprog

N = 26

def get_optimal_radii_lp(centers):
    """Solves LP to find exact maximum sum of radii for fixed centers."""
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    # Distance to boundaries
    b = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                   np.minimum(c[:, 1], 1.0 - c[:, 1]))
    b = np.maximum(b, 0.0)
    
    # Pairwise distances
    diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    A_ub = []
    b_ub = []
    
    # Overlap constraints: r_i + r_j <= d_ij
    for i in range(N):
        for j in range(i + 1, N):
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints: r_i <= b_i
    for i in range(N):
        row = np.zeros(N)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(b[i])
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c_obj = -np.ones(N)
    bounds_r = [(0, 1.0)] * N
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except:
        pass
        
    # Fallback heuristic if LP fails
    rp = 0.5 * np.min(dists, axis=1)
    radii = np.minimum(b, rp)
    return radii, np.sum(radii)

def de_objective(centers_flat):
    """Objective for DE: maximize LP result."""
    try:
        c = centers_flat.reshape(N, 2)
        _, s = get_optimal_radii_lp(c)
        if np.isnan(s) or s < 0:
            return 1e6
        return -s
    except:
        return 1e6

def obj_func_joint(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def compute_constraints_joint(v):
    """Computes boundary and non-overlap constraints. Must be >= 0."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    # Boundary constraints
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    # Overlap constraints
    idx = np.triu_indices(N, 1)
    dx = c[idx[0], 0] - c[idx[1], 0]
    dy = c[idx[0], 1] - c[idx[1], 1]
    d = np.sqrt(dx**2 + dy**2)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # Phase 1: Global search on centers using Differential Evolution
    bounds_de = [(0.02, 0.98)] * (2 * N)
    try:
        de_res = differential_evolution(de_objective, bounds_de, seed=42, maxiter=120, 
                                        popsize=15, tol=1e-10, mutation=(0.5, 1.0), 
                                        recombination=0.9, polish=False)
        c_de = de_res.x.reshape(N, 2)
        r_de, s_de = get_optimal_radii_lp(c_de)
        best_c, best_r, best_sum = c_de, r_de, s_de
    except Exception:
        pass

    if best_c is None:
        best_c = rng.uniform(0.1, 0.9, (N, 2))
        best_r, best_sum = get_optimal_radii_lp(best_c)

    # Phase 2: Multi-start Joint Optimization with SLSQP
    bounds_opt = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints_joint}
    
    starts = [np.concatenate([best_c.flatten(), best_r])]
    
    # Add diverse randomized starts
    for _ in range(8):
        c_rand = rng.uniform(0.1, 0.9, (N, 2))
        r_rand, _ = get_optimal_radii_lp(c_rand)
        starts.append(np.concatenate([c_rand.flatten(), r_rand]))
        
    # Add hexagonal lattice starts with shifts
    for shift in [0.0, 0.04, 0.08]:
        hex_c = []
        r_est = 0.10
        y = r_est + shift
        row = 0
        while len(hex_c) < N:
            x_start = r_est + shift
            if row % 2 == 1: x_start += r_est
            x = x_start
            while x < 1.0 - 0.01 and len(hex_c) < N:
                hex_c.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
            row += 1
        hex_c = np.array(hex_c[:N]) + rng.normal(0, 0.003, (N, 2))
        hex_c = np.clip(hex_c, 0.05, 0.95)
        r_hex, _ = get_optimal_radii_lp(hex_c)
        starts.append(np.concatenate([hex_c.flatten(), r_hex]))
        
    for s0 in starts:
        try:
            res = minimize(obj_func_joint, s0, method='SLSQP', bounds=bounds_opt,
                          constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
            
            # Verify constraints are satisfied within tolerance
            c_vals = compute_constraints_joint(res.x)
            if np.min(c_vals) >= -1e-7:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_c = res.x[:2*N].reshape(N, 2).copy()
                    best_r = res.x[2*N:].copy()
        except Exception:
            continue
            
    # Phase 3: Strict Numerical Repair to guarantee validator passes
    centers = best_c
    radii = best_r
    for _ in range(100):
        changed = False
        # Resolve overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
