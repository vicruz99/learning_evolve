import numpy as np
from scipy.optimize import minimize, differential_evolution, linprog

N = 26

def get_optimal_radii_lp(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c = centers.copy()
    c = np.clip(c, 1e-10, 1.0 - 1e-10)
    
    # Upper bounds from boundaries
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    # Pairwise distances
    dists = np.linalg.norm(c[:, None, :] - c[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    
    # LP: maximize sum(r_i) subject to r_i + r_j <= dists[i,j] and r_i <= ub[i]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, u) for u in ub]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def de_objective(centers_flat):
    """Objective for Differential Evolution."""
    c = centers_flat.reshape(N, 2)
    r, s = get_optimal_radii_lp(c)
    return -s

def generate_hex_init(spacing, shift_x=0, shift_y=0):
    """Generate hexagonal lattice initialization."""
    pts = []
    y = spacing * 0.5 + shift_y
    row = 0
    while len(pts) < N:
        x_start = spacing * 0.5 + (spacing if row % 2 == 1 else 0.0) + shift_x
        x = x_start
        while x < 1.0 and len(pts) < N:
            pts.append([x, y])
            x += spacing
        y += spacing * np.sqrt(3) / 2
        row += 1
    return np.array(pts[:N])

def constraints_func(v):
    """Constraints for SLSQP: must be >= 0."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    return np.concatenate(con)
