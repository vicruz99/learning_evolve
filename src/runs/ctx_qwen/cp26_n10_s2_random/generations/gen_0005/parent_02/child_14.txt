

<strategy>
The problem of packing circles to maximize the sum of radii is highly non-convex. Jointly optimizing centers and radii with gradient-based methods often gets stuck in local minima or violates constraints due to numerical precision. 

A more robust mathematical approach is to decouple the problem:
1. For any fixed set of centers, finding the optimal radii that maximize the sum is a **Linear Programming (LP)** problem: maximize $\sum r_i$ subject to $r_i + r_j \le d_{ij}$ and $r_i \le \text{dist}(c_i, \text{boundary})$. This LP is small (26 variables, ~350 constraints) and can be solved exactly and extremely fast.
2. We then treat the optimal LP value as a function of the centers, $F(C) = \max \sum r_i$. We optimize the centers to maximize $F(C)$.
3. Since $F(C)$ is the minimum of many linear functions, it is continuous but non-smooth (concave). We use **Differential Evolution** for global exploration, followed by **Nelder-Mead** and **Powell** for local refinement. These derivative-free methods handle non-smooth landscapes well.
4. Finally, we extract the exact optimal radii via LP and apply a strict numerical repair pass to guarantee the validation function passes within the $10^{-12}$ tolerance.

This hybrid LP + global/local search strategy bypasses the pitfalls of non-linear constrained optimization and consistently yields near-optimal packings.
</strategy>

```python
import numpy as np
from scipy.optimize import differential_evolution, minimize, linprog

N = 26
# Precompute constraint matrix structure for LP: r_i + r_j <= d_ij
tri_idx = np.triu_indices(N, 1)
A_ub_structure = np.zeros((len(tri_idx[0]), N))
A_ub_structure[np.arange(len(tri_idx[0])), tri_idx[0]] = 1.0
A_ub_structure[np.arange(len(tri_idx[0])), tri_idx[1]] = 1.0

def lp_max_radii_sum(centers):
    """Solves LP to find max sum of radii for fixed centers."""
    xb = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    yb = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    ub = np.minimum(xb, yb)
    if np.any(ub < 0):
        return -1e6
        
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[tri_idx]
    
    c_obj = -np.ones(N)
    bounds = [(0, u) for u in ub]
    
    try:
        res = linprog(c_obj, A_ub=A_ub_structure, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun
    except Exception:
        pass
    return -1e6

def objective_centers(centers_flat):
    """Objective for center optimization: maximize LP sum of radii."""
    centers = centers_flat.reshape(N, 2)
    if np.any(centers < 0) or np.any(centers > 1):
        return 1e6
    return -lp_max_radii_sum(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    
    # Phase 1: Global search with Differential Evolution
    bounds_de = [(0.0, 1.0)] * (2 * N)
    try:
        de_res = differential_evolution(
            objective_centers, 
            bounds_de, 
            seed=42, 
            maxiter=400, 
            popsize=20, 
            tol=1e-10, 
            mutation=(0.8, 1.2), 
            recombination=0.9, 
            polish=False
        )
        best_centers = de_res.x.reshape(N, 2)
        best_sum = -de_res.fun
    except Exception as e:
        print(f"DE failed: {e}")

    # Phase 2: Local refinement
    starts = []
    if best_centers is not None:
        starts.append(best_centers)
        
    # Hexagonal lattice initialization
    c_hex = []
    r_est = 0.10
    y = r_est
    row = 0
    while len(c_hex) < N:
        shift = r_est if row % 2 == 1 else 0.0
        x = r_est + shift
        while x + r_est <= 1.0 and len(c_hex) < N:
            c_hex.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3)
        row += 1
    starts.append(np.array(c_hex[:N]))
    
    # Grid initialization
    c_grid = []
    for i in range(5):
        for j in range(5):
            c_grid.append([0.1 + j*0.2, 0.1 + i*0.2])
    c_grid.append([0.5, 0.5])
    starts.append(np.array(c_grid[:N]))
    
    # Local search from multiple starts
    for c0 in starts:
        for _ in range(4):
            c_pert = c0 + rng.normal(0, 0.004, c0.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            for method in ['Nelder-Mead', 'Powell']:
                try:
                    opts = {'maxiter': 10000, 'xatol': 1e-10, 'fatol': 1e-12} if method=='Nelder-Mead' else {'maxiter': 10000, 'ftol': 1e-12, 'xtol': 1e-12}
                    res = minimize(objective_centers, c_pert.flatten(), method=method, options=opts)
                    if -res.fun > best_sum:
                        best_sum = -res.fun
                        best_centers = res.x.reshape(N, 2)
                except Exception:
                    pass

    # Phase 3: Extract final radii via LP and strict repair
    if best_centers is not None:
        centers = np.clip(best_centers, 1e-9, 1.0 - 1e-9)
        
        xb = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
        yb = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
        ub = np.minimum(xb, yb)
        
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        b_ub = dists[tri_idx]
        
        c_obj = -np.ones(N)
        bounds_lp = [(0, u) for u in ub]
        
        res_lp = linprog(c_obj, A_ub=A_ub_structure, b_ub=b_ub, bounds=bounds_lp, method='highs')
        radii = res_lp.x if res_lp.success else np.zeros(N)
        
        # Numerical repair to guarantee validation passes
        for _ in range(20):
            changed = False
            for i in range(N):
                for j in range(i+1, N):
                    d = np.linalg.norm(centers[i] - centers[j])
                    if d < radii[i] + radii[j] - 1e-12:
                        shrink = (radii[i] + radii[j] - d + 1e-13) / 2.0
                        radii[i] -= shrink
                        radii[j] -= shrink
                        changed = True
            for i in range(N):
                x, y = centers[i]
                max_r = min(x, 1.0-x, y, 1.0-y)
                if radii[i] > max_r + 1e-12:
                    radii[i] = max_r
                    changed = True
            if not changed:
                break
                
        radii = np.maximum(radii, 0.0)
        return centers, radii, float(np.sum(radii))
        
    # Fallback
    centers = rng.uniform(0.1, 0.9, (N, 2))
    radii = np.full(N, 0.05)
    return centers, radii, 1.3
```
