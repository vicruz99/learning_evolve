import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26

# Precompute indices for LP pairwise constraints to avoid rebuilding masks repeatedly
_PAIR_INDICES = np.triu_indices(N, k=1)
_PAIR_COUNT = _PAIR_INDICES[0].shape[0]

def compute_lp_radii(centers):
    """Computes exact maximum radii for fixed centers via Linear Programming."""
    centers = np.clip(centers, 1e-9, 1.0 - 1e-9)
    n = centers.shape[0]
    
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Build LP: max sum(r) s.t. r_i + r_j <= dists[i,j], r_i <= ub[i], r_i >= 0
    m = _PAIR_COUNT + n
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    i_idx, j_idx = _PAIR_INDICES
    A_ub[:_PAIR_COUNT, i_idx] = 1.0
    A_ub[:_PAIR_COUNT, j_idx] = 1.0
    b_ub[:_PAIR_COUNT] = dists[i_idx, j_idx]
    
    A_ub[_PAIR_COUNT:, np.arange(n)] = 1.0
    b_ub[_PAIR_COUNT:] = ub
    
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    # Fallback if LP fails (rare)
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    np.fill_diagonal(dists, np.inf)
    rp = 0.5 * np.min(dists, axis=1)
    return np.maximum(np.minimum(rb, rp), 0.0)

def de_objective(centers_flat):
    """Objective for Differential Evolution: maximize LP sum of radii."""
    c = centers_flat.reshape(N, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    r = compute_lp_radii(c)
    return -np.sum(r)

def obj_bound(v):
    """Objective for L-BFGS-B: maximize LP sum of radii with clipped centers."""
    c = v.reshape(N, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    r = compute_lp_radii(c)
    return -np.sum(r)

def force_repulsion_init(rng, iters=300):
    """Generates a well-spread initial configuration using force-directed repulsion."""
    pts = rng.uniform(0.15, 0.85, (N, 2))
    corners = np.array([[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]])
    pts[:4] = corners
    
    for _ in range(iters):
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, 1.0)
        forces = diff / (dists**2 + 1e-5)[:, :, np.newaxis]
        pts += 0.004 * np.sum(forces, axis=1)
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def obj_joint(v):
    """Objective for SLSQP: maximize sum of radii."""
    return -np.sum(v[2*N:])

def cons_joint(v):
    """Joint constraints for centers and radii."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    
    # 1. Generate diverse starts
    starts = []
    for r0 in [0.08, 0.09, 0.10, 0.11]:
        pts = []
        y = r0
        row = 0
        while len(pts) < N:
            x = r0 if row % 2 == 0 else 2.0 * r0
            while x + r0 <= 1.0 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
        starts.append(np.array(pts[:N]))
        
    for _ in range(5):
        starts.append(force_repulsion_init(rng))
        
    for _ in range(5):
        starts.append(np.clip(rng.uniform(0.05, 0.95, (N, 2)) + rng.normal(0, 0.02, (N, 2)), 0.01, 0.99))
        
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # 2. Evaluate starts with LP
    for s in starts:
        r = compute_lp_radii(s)
        s_sum = np.sum(r)
        if s_sum > best_sum:
            best_sum = s_sum
            best_c = s.copy()
            best_r = r.copy()
            
    # 3. Global search with Differential Evolution
    try:
        bounds_de = [(0.02, 0.98)] * (2 * N)
        de_res = differential_evolution(de_objective, bounds_de, seed=42, maxiter=50,
                                        popsize=10, tol=1e-9, mutation=(0.5, 1.0),
                                        recombination=0.9, polish=False)
        c_de = de_res.x.reshape(N, 2)
        r_de = compute_lp_radii(c_de)
        if np.sum(r_de) > best_sum:
            best_sum = np.sum(r_de)
            best_c = c_de.copy()
            best_r = r_de.copy()
    except Exception:
        pass
        
    # 4. Local refinement on centers (L-BFGS-B)
    bounds_powell = [(0.0, 1.0)] * (2 * N)
    for i in range(15):
        if i == 0:
            x0 = best_c.flatten()
        else:
            noise = 0.003 * (0.7 ** (i // 2))
            c_pert = best_c.copy() + rng.normal(0, noise, best_c.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            x0 = c_pert.flatten()
            
        try:
            res = minimize(obj_bound, x0, method='L-BFGS-B', bounds=bounds_powell,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-12})
            c_opt = np.clip(res.x.reshape(N, 2), 1e-6, 1.0 - 1e-6)
            r_opt = compute_lp_radii(c_opt)
            s = np.sum(r_opt)
            if s > best_sum:
                best_sum = s
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            continue
            
    # 5. Joint SLSQP refinement (centers + radii)
    bounds_joint = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': cons_joint}
    v0 = np.concatenate([best_c.flatten(), best_r])
    
    for i in range(10):
        if i == 0:
            v_curr = v0.copy()
        else:
            noise = 0.002 * (0.8 ** i)
            c_pert = best_c.copy() + rng.normal(0, noise, best_c.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            r_pert = compute_lp_radii(c_pert) * 0.99
            v_curr = np.concatenate([c_pert.flatten(), r_pert])
            
        try:
            res = minimize(obj_joint, v_curr, method='SLSQP', bounds=bounds_joint,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
            c_vals = cons_joint(res.x)
            if np.min(c_vals) >= -1e-9:
                s = np.sum(res.x[2*N:])
                if s > best_sum:
                    best_sum = s
                    best_c = res.x[:2*N].reshape(N, 2).copy()
                    best_r = res.x[2*N:].copy()
        except Exception:
            continue
            
    # 6. Final exact LP on best centers & numerical repair
    best_r = compute_lp_radii(best_c)
    radii = best_r.copy()
    centers = best_c.copy()
    
    for _ in range(30):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
