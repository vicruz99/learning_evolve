import numpy as np
from scipy.optimize import minimize

N = 26

def compute_constraints(v):
    """Computes boundary and non-overlap constraints. Must be >= 0."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    # Overlap constraints: dist(i,j) - (r_i + r_j) >= 0
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    return np.concatenate(con)

def obj_func(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def force_init(n, seed, iters=1200):
    """Generates a feasible initial configuration using force-directed expansion."""
    rng = np.random.default_rng(seed)
    c = rng.uniform(0.25, 0.75, (n, 2))
    r = np.full(n, 0.02)
    
    for _ in range(iters):
        # Gradually grow radii to simulate expansion
        r *= 1.0006
        
        forces = np.zeros_like(c)
        
        # Wall repulsion
        for i in range(n):
            for dim in range(2):
                if c[i, dim] < r[i]: 
                    forces[i, dim] += (r[i] - c[i, dim]) * 20.0
                if c[i, dim] > 1.0 - r[i]: 
                    forces[i, dim] -= (c[i, dim] - (1.0 - r[i])) * 20.0
                    
        # Inter-circle repulsion
        diffs = c[:, None, :] - c[None, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        sum_r = r[:, None] + r[None, :]
        
        overlap = np.maximum(0.0, sum_r - dists)
        safe_dists = np.where(dists > 1e-9, dists, 1.0)
        force_mag = overlap * 10.0 / safe_dists
        
        # Accumulate forces
        for i in range(n):
            dir_vecs = c[i] - c
            forces[i] += np.sum(force_mag[i, :, np.newaxis] * dir_vecs, axis=0)
            
        # Update positions
        c += forces * 0.0003
        c = np.clip(c, 1e-6, 1.0 - 1e-6)
        
    return c, r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    bounds_opt = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # Phase 1: Force-directed initialization + SLSQP
    for seed in range(15):
        c0, r0 = force_init(N, seed)
        v0 = np.concatenate([c0.flatten(), r0])
        
        try:
            res = minimize(obj_func, v0, method='SLSQP', bounds=bounds_opt,
                          constraints=cons, options={'maxiter': 4000, 'ftol': 1e-14})
            s = np.sum(res.x[2*N:])
            if s > best_sum:
                best_sum = s
                best_c = res.x[:2*N].reshape(N, 2).copy()
                best_r = res.x[2*N:].copy()
        except Exception:
            pass

    if best_c is None:
        best_c = rng.uniform(0.1, 0.9, (N, 2))
        best_r = np.full(N, 0.05)
        best_sum = np.sum(best_r)

    for step in range(30):
        noise = 0.005 * (0.85 ** (step // 5))

        c_pert = np.clip(c_pert, 0.01, 0.99)
        
        # Compute safe initial radii for perturbed centers
        rb = np.minimum(np.minimum(c_pert[:, 0], 1.0 - c_pert[:, 0]),

        dists = np.linalg.norm(c_pert[:, None, :] - c_pert[None, :, :], axis=2)

        r_pert = np.minimum(rb, rp) * 0.85
        
        v_curr = np.concatenate([c_pert.flatten(), r_pert])
        
        try:
            res = minimize(obj_func, v_curr, method='SLSQP', bounds=bounds_opt,

                best_c = res.x[:2*N].reshape(N, 2).copy()
                best_r = res.x[2*N:].copy()

            continue

    radii = best_r.copy()

                    radii[j] -= shrink
                    changed = True
        for i in range(N):

                radii[i] = max_r
                changed = True

    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
