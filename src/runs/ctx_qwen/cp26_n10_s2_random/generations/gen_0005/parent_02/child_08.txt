import numpy as np
from scipy.optimize import linprog

N = 26

def solve_lp_radii(centers):
    """
    Solves the LP to maximize sum of radii for a fixed set of centers.
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= ||c_i - c_j||  for all i < j
      r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
      r_i >= 0
    """
    c_obj = -np.ones(N)
    
    # Boundary upper bounds
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 0.0)
    
    # Pairwise distance constraints: r_i + r_j <= dist_ij
    # We build the constraint matrix efficiently
    idx = np.triu_indices(N, 1)
    n_pairs = len(idx[0])
    
    A_ub = np.zeros((n_pairs, N))
    A_ub[np.arange(n_pairs), idx[0]] = 1.0
    A_ub[np.arange(n_pairs), idx[1]] = 1.0
    
    dx = centers[idx[0], 0] - centers[idx[1], 0]
    dy = centers[idx[0], 1] - centers[idx[1], 1]
    b_ub = np.hypot(dx, dy)
    
    bounds = [(0.0, u) for u in ub]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
        
    return 0.0, np.zeros(N)

def generate_initial_configs(rng):
    """Generates diverse starting configurations for center placement."""
    configs = []
    
    # 1. Hexagonal lattices with varying densities
    for r0 in [0.085, 0.095, 0.105, 0.115]:
        pts = []
        y = r0
        row = 0
        while len(pts) < N:
            x = r0 if row % 2 == 0 else 2.0 * r0
            while x + r0 <= 1.0 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r0
            y += r0 * np.sqrt(3)
            row += 1
        configs.append(np.array(pts[:N]))
        
    # 2. Corner-anchored random configurations
    # Corners and edges naturally support larger circles
    for _ in range(12):
        c = rng.uniform(0.15, 0.85, (N, 2))
        corners = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        c[:4] = corners
        configs.append(c)
        
    # 3. Staggered rectangular grids
    for sx, sy in [(5, 6), (6, 5), (5, 5)]:
        xs = np.linspace(0.1, 0.9, sx)
        ys = np.linspace(0.1, 0.9, sy)
        gx, gy = np.meshgrid(xs, ys)
        c = np.column_stack((gx.flatten(), gy.flatten()))[:N]
        c += rng.normal(0, 0.01, c.shape)
        configs.append(np.clip(c, 0.05, 0.95))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    configs = generate_initial_configs(rng)
    
    best_sum = -1.0
    best_centers = None
    
    # Simulated Annealing on centers with LP oracle evaluation
    for init_c in configs:
        curr_c = init_c.copy()
        curr_sum, _ = solve_lp_radii(curr_c)
        
        temp = 0.05
        step = 0.04
        
        for it in range(3500):
            # Cooling schedule
            temp = 0.05 * (0.999 ** it)
            step = 0.04 * (0.998 ** it)
            if temp < 1e-6:
                temp = 1e-6
                
            # Perturb a random circle
            idx = rng.integers(N)
            new_c = curr_c.copy()
            new_c[idx] += rng.uniform(-step, step, 2)
            new_c[idx] = np.clip(new_c[idx], 1e-5, 1.0 - 1e-5)
            
            new_sum, _ = solve_lp_radii(new_c)
            delta = new_sum - curr_sum
            
            # Metropolis criterion
            if delta > 0 or rng.random() < np.exp(delta / temp):
                curr_c = new_c
                curr_sum = new_sum
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = curr_c.copy()
                    
    if best_centers is None:
        best_centers = configs[0]
        best_sum, _ = solve_lp_radii(best_centers)
        
    # Final LP to get optimal radii for the best centers found
    final_sum, final_radii = solve_lp_radii(best_centers)
    
    # Numerical repair to guarantee strict validity within 1e-12 tolerance
    centers = best_centers.copy()
    radii = final_radii.copy()
    
    for _ in range(30):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            max_r = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
            if radii[i] > max_r + 1e-12:
                radii[i] = max_r
                changed = True
                
        # Resolve overlaps by shrinking proportionally
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if d < radii[i] + radii[j] - 1e-12:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
