import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_max_radii_lp(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    c = np.clip(centers, 1e-9, 1.0 - 1e-9)
    
    # Upper bounds from boundaries
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    # Pairwise distances
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # LP: max sum(r) s.t. r_i + r_j <= D_ij, 0 <= r_i <= B_i
    c_obj = -np.ones(n)
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds_r = [(0.0, u) for u in ub]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def get_objective(vars_flat):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars_flat[2*N:])

def get_constraints(vars_flat):
    """Computes boundary and non-overlap constraints (must be >= 0)."""
    c = vars_flat[:2*N].reshape(N, 2)
    r = vars_flat[2*N:]
    cons = []
    
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1.0 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1.0 - c[:, 1] - r)
    
    # Overlap constraints: dist(i,j) - (r_i + r_j) >= 0
    idx_i, idx_j = np.triu_indices(N, 1)
    dx = c[idx_i, 0] - c[idx_j, 0]
    dy = c[idx_i, 1] - c[idx_j, 1]
    dr = r[idx_i] + r[idx_j]
    cons.append(np.sqrt(dx**2 + dy**2) - dr)
    
    return np.concatenate(cons)

def run_packing():
    rng = np.random.default_rng(42)
    bounds_opt = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons_dict = {'type': 'ineq', 'fun': get_constraints}
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # --- Phase 1: Generate Diverse Initial Configurations ---
    starts = []
    
    # 1. Hexagonal lattice patterns
    for pat in [[5,5,5,5,6], [5,6,5,6,4], [6,5,6,5,4], [4,6,6,6,4], [5,5,5,6,5]]:
        c_hex = []
        r_est = 0.095
        y = r_est
        for r_idx, cnt in enumerate(pat):
            shift = r_est if r_idx % 2 == 1 else 0.0
            x = r_est + shift
            for _ in range(cnt):
                if len(c_hex) < N:
                    c_hex.append([x, y])
                x += 2.0 * r_est
            y += r_est * np.sqrt(3)
        starts.append(np.array(c_hex[:N]))
        
    # 2. Corner-hugging + interior grid
    c_corner = np.zeros((N, 2))
    c_corner[:4] = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
    grid_pts = np.array(np.meshgrid(np.linspace(0.2, 0.8, 5), np.linspace(0.2, 0.8, 5))).T.reshape(-1, 2)
    c_corner[4:] = grid_pts[:22]
    starts.append(c_corner)
    
    # 3. Force-repulsion spread
    c_rep = rng.uniform(0.1, 0.9, (N, 2))
    for _ in range(800):
        forces = np.zeros_like(c_rep)
        diff = c_rep[:, None, :] - c_rep[None, :, :]
        dists = np.linalg.norm(diff, axis=2)
        np.fill_diagonal(dists, 1.0)
        forces += np.sum(diff / (dists**2)[:, :, None], axis=1)
        c_rep += 0.002 * forces
        c_rep = np.clip(c_rep, 0.05, 0.95)
    starts.append(c_rep)

    for _ in range(5):
        starts.append(rng.uniform(0.1, 0.9, (N, 2)))

                best_sum = lp_sum
                best_r = lp_r

    if best_c is not None:
        for step in range(15):
            noise_scale = 0.004 * (0.85 ** step)
            c_pert = best_c.copy() + rng.normal(0, noise_scale, best_c.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            
            # Use LP to get optimal radii for perturbed geometry
            _, r_pert = compute_max_radii_lp(c_pert)
            v_pert = np.concatenate([c_pert.flatten(), r_pert])
            
            try:
                res = minimize(get_objective, v_pert, method='SLSQP', bounds=bounds_opt,

                if np.min(get_constraints(res.x)) >= -1e-8:

                    if s > best_sum:
                        best_sum = s
                        best_c = res.x[:2*N].reshape(N, 2).copy()
                        best_r = res.x[2*N:].copy()
            except Exception:
                pass
                
            # Check LP again
            lp_sum, lp_r = compute_max_radii_lp(best_c)
            if lp_sum > best_sum:
                best_sum = lp_sum
                best_r = lp_r

    # --- Phase 4: Strict Numerical Repair ---
    radii = best_r.copy()
    centers = best_c.copy()
    
    for _ in range(60):
        changed = False
        
        # Resolve overlaps by symmetric shrinking
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
