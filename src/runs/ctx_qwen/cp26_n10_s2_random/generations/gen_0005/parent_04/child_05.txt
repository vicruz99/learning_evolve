import numpy as np
from scipy.optimize import linprog, minimize

N = 26

# Precompute indices for pairwise constraints to speed up LP construction
i_idx, j_idx = np.triu_indices(N, k=1)
NUM_PAIRS = len(i_idx)

def solve_lp_radii(centers):
    """Solves the LP to find maximum sum of radii for fixed centers."""
    c = np.clip(centers, 1e-5, 1.0 - 1e-5)
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_pairs = dists[i_idx, j_idx]
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    x, y = c[:, 0], c[:, 1]
    ub = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    ub = np.maximum(ub, 1e-9)
    
    # Construct A_ub and b_ub
    # Shape: (NUM_PAIRS + 4*N, N)
    A_ub = np.zeros((NUM_PAIRS + 4 * N, N))
    b_ub = np.zeros(NUM_PAIRS + 4 * N)
    
    # Fill pairwise part
    row_idx = np.arange(NUM_PAIRS)
    A_ub[row_idx, i_idx] = 1.0
    A_ub[row_idx, j_idx] = 1.0
    b_ub[:NUM_PAIRS] = b_pairs
    
    # Fill boundary part
    for k in range(4):
        row_start = NUM_PAIRS + k * N
        A_ub[row_start:row_start + N, :] = np.eye(N)
        b_ub[row_start:row_start + N] = ub
        
    # Solve LP: max sum(r) -> min -sum(r)
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * N, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: equal radii limited by nearest neighbor and boundary
        min_dist = np.min(dists, axis=1)
        return np.minimum(ub, 0.5 * min_dist)

def objective_centers(x_flat):
    """Objective for center optimization: maximize sum of radii."""
    centers = x_flat.reshape(N, 2)
    # Enforce bounds within the objective to guide Powell
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    radii = solve_lp_radii(centers)
    return -np.sum(radii)

def generate_initial_configs():
    """Generates diverse initial center configurations."""
    configs = []
    
    # 1. Hexagonal lattices with varying densities and shifts
    for spacing in np.linspace(0.15, 0.25, 6):
        for shift_x in [0.0, 0.02, 0.05]:
            for shift_y in [0.0, 0.02]:
                pts = []
                y = spacing * 0.5 + shift_y
                row = 0
                while len(pts) < N:
                    x_start = spacing * 0.5 + shift_x + (spacing if row % 2 == 1 else 0.0)
                    x = x_start
                    while x < 1.0 and len(pts) < N:
                        pts.append([x, y])
                        x += spacing
                    y += spacing * np.sqrt(3) / 2
                    row += 1
                configs.append(np.array(pts[:N]).flatten())
                
    # 2. Force-directed repulsion configurations
    np.random.seed(42)
    for _ in range(8):
        pts = np.random.rand(N, 2) * 0.8 + 0.1
        for _ in range(300):
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-9
            np.fill_diagonal(dists, 1.0)
            forces = diff / (dists**2)[:, :, np.newaxis]
            pts += 0.004 * np.sum(forces, axis=1)
            pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts.flatten())
        
    # 3. Corner-focused configurations
    for _ in range(5):
        pts = np.random.rand(N, 2) * 0.6 + 0.2
        corners = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        pts[:4] = corners
        configs.append(pts.flatten())
        
    return configs

def slesqp_constraints(p):
    """Constraints for joint SLSQP optimization: dist^2 >= (r_i + r_j)^2 and boundary."""
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise constraints: dx^2 + dy^2 - (ri + rj)^2 >= 0
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    c = np.concatenate([c, dx*dx + dy*dy - dr*dr])
    return c

def slesqp_objective(p):
    return -np.sum(p[2::3])

def run_packing() -> tuple:
    np.random.seed(42)
    configs = generate_initial_configs()
    
    best_val = np.inf
    best_centers_flat = None
    
    # Phase 1: Optimize centers using Powell (derivative-free, robust for non-smooth LP objective)
    bounds_centers = [(1e-4, 1.0 - 1e-4)] * (2 * N)
    
    for cfg in configs:
        try:
            res = minimize(objective_centers, cfg, method='Powell', 
                           bounds=bounds_centers, 
                           options={'maxiter': 1500, 'ftol': 1e-13, 'xtol': 1e-13})
            if res.fun < best_val:
                best_val = res.fun
                best_centers_flat = res.x.copy()
        except Exception:
            continue
            
    if best_centers_flat is None:
        # Fallback
        best_centers_flat = configs[0].copy()
        
    best_centers = best_centers_flat.reshape(N, 2)
    
    # Phase 2: Joint SLSQP refinement on (centers, radii)
    # Initialize radii from LP
    init_radii = solve_lp_radii(best_centers)
    p0 = np.zeros(3 * N)
    p0[0::3] = best_centers[:, 0]
    p0[1::3] = best_centers[:, 1]
    p0[2::3] = init_radii
    
    bounds_joint = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': slesqp_constraints}
    
    try:
        res_joint = minimize(slesqp_objective, p0, method='SLSQP', bounds=bounds_joint,
                             constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
        p0 = res_joint.x
    except Exception:
        pass
        
    # Extract and repair
    centers = np.column_stack((p0[0::3], p0[1::3]))
    radii = p0[2::3].copy()
    
    # Final LP pass to guarantee optimality for these centers
    radii = solve_lp_radii(centers)
    
    # Strict numerical repair
    for _ in range(50):
        changed = False
        for i in range(N):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0], 
                        centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > max_r - 1e-10:
                radii[i] = max_r
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
