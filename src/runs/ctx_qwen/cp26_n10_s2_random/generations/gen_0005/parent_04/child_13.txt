import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_radii_direct(centers):
    """Computes valid radii for fixed centers using direct distance constraints."""
    xb = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    yb = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    rb = np.minimum(xb, yb)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    dists[np.diag_indices(N)] = np.inf
    rp = 0.5 * np.min(dists, axis=1)
    
    return np.minimum(rb, rp)

def objective_func(centers_flat):
    """Objective: maximize sum of radii -> minimize negative sum."""
    centers = centers_flat.reshape(N, 2)
    r = compute_radii_direct(centers)
    return -np.sum(r)

def get_lp_radii(centers):
    """Solves LP to find exact maximal radii for fixed centers."""
    c_obj = -np.ones(N)
    num_pairs = N * (N - 1) // 2
    num_constraints = num_pairs + 4 * N
    A_ub = np.zeros((num_constraints, N))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub[idx] = d
            idx += 1
    for i in range(N):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * N, method='highs')
    if res.success:
        return res.x
    return compute_radii_direct(centers)

def repair_packing(centers, radii):
    """Ensures strict validity within numerical tolerance."""
    radii = radii.copy()
    for _ in range(20):
        changed = False
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-9:
                radii[i] = max_r
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def generate_configs():
    """Generates diverse initial center configurations."""
    configs = []
    rng = np.random.default_rng(42)
    
    # Hexagonal lattices with noise
    for seed in range(10):
        rng_local = np.random.default_rng(seed)
        centers = np.zeros((N, 2))
        idx = 0
        y = 0.095
        row = 0
        r = 0.095
        while idx < N:
            x_start = r if row % 2 == 0 else 2 * r
            x = x_start
            while x + r <= 1.0 + 1e-9 and idx < N:
                centers[idx] = [x, y]
                idx += 1
                x += 2 * r
            y += np.sqrt(3) * r
            row += 1
        centers += rng_local.normal(0, 0.005, centers.shape)
        centers = np.clip(centers, 0.02, 0.98)
        configs.append(centers.flatten())
        
    # Uniform random
    for _ in range(5):
        centers = rng.random((N, 2)) * 0.9 + 0.05
        configs.append(centers.flatten())
        
    # Corner-focused + random
    for _ in range(5):
        centers = rng.random((N, 2)) * 0.8 + 0.1
        corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        centers[:4] = corners
        configs.append(centers.flatten())
        
    return configs

def run_packing() -> tuple:
    bounds = [(0.0, 1.0)] * (2 * N)
    
    best_centers = None
    best_sum = -np.inf
    
    configs = generate_configs()
    
    # Phase 1: SLSQP optimization from multiple starts
    for cfg in configs:
        try:
            res = minimize(objective_func, cfg, method='SLSQP', bounds=bounds,
                           options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            continue
            
    if best_centers is None:
        rng = np.random.default_rng(0)
        best_centers = rng.random((N, 2)) * 0.8 + 0.1
        curr_r = compute_radii_direct(best_centers)
        best_sum = np.sum(curr_r)
        
    # Phase 2: Stochastic Coordinate Descent (Jiggle Phase)
    curr = best_centers.copy()
    curr_r = compute_radii_direct(curr)
    curr_sum = np.sum(curr_r)
    rng = np.random.default_rng(123)
    
    for _ in range(4000):
        i = rng.integers(N)
        best_local_pos = curr[i].copy()
        best_local_r = curr_r[i]
        
        # Explore local neighborhood
        for _ in range(20):
            pert = curr[i] + rng.normal(0, 0.008, 2)
            pert = np.clip(pert, 0.0, 1.0)
            temp = curr.copy()
            temp[i] = pert
            temp_r = compute_radii_direct(temp)
            if temp_r[i] > best_local_r:
                best_local_r = temp_r[i]
                best_local_pos = pert.copy()
                
        if best_local_r > curr_r[i] + 1e-9:
            curr[i] = best_local_pos
            curr_r = compute_radii_direct(curr)
            curr_sum = np.sum(curr_r)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = curr.copy()
                
    # Phase 3: Final SLSQP polish
    try:
        res = minimize(objective_func, best_centers.flatten(), method='SLSQP', bounds=bounds,
                       options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
        if -res.fun > best_sum:
            best_sum = -res.fun
            best_centers = res.x.reshape(N, 2)
    except Exception:
        pass
        
    # Phase 4: LP for exact maximal radii at optimal centers
    radii = get_lp_radii(best_centers)
    radii = repair_packing(best_centers, radii)
    
    # Final safety shrink to guarantee validator tolerance
    for _ in range(5):
        valid = True
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False
                break
        if not valid:
            break
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            break
        radii *= 0.99995
        
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return best_centers, radii, final_sum
