import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIR_COUNT = N * (N - 1) // 2
i_idx, j_idx = np.triu_indices(N, k=1)

def get_radii_lp(centers):
    """Solves LP to find optimal radii maximizing sum given fixed centers."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    A = np.zeros((PAIR_COUNT + n, n))
    b = np.zeros(PAIR_COUNT + n)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            b[k] = dists[i, j]
            k += 1
    for i in range(n):
        A[k, i] = 1.0
        b[k] = ub[i]
        k += 1
        
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def obj_centers(v):
    """Objective for center optimization: minimize negative sum of radii (via LP)."""
    centers = v.reshape(-1, 2)
    centers = np.clip(centers, 0.001, 0.999)
    _, s = get_radii_lp(centers)
    return -s

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def con_joint(v):
    """Smooth constraints for SLSQP: boundaries and squared distances."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    cons = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
    dx = c[i_idx, 0] - c[j_idx, 0]
    dy = c[i_idx, 1] - c[j_idx, 1]
    dr = r[i_idx] + r[j_idx]
    cons.append(dx*dx + dy*dy - dr*dr)
    return np.concatenate(cons)

def generate_inits():
    """Generates diverse initial configurations for centers."""
    inits = []
    # Hexagonal lattices with varying densities
    for r in [0.08, 0.09, 0.10, 0.105, 0.11]:
        pts = []
        y = r
        row = 0
        while len(pts) < N:

            while x + r <= 1.0 and len(pts) < N:
                pts.append([x, y])
                x += 2*r
            y += np.sqrt(3)*r
            row += 1
        inits.append(np.array(pts[:N]).flatten())
        
    # Force-directed repulsion spreads

        pts = np.random.rand(N, 2) * 0.8 + 0.1

            dists = np.sqrt(np.sum(diff**2, axis=2))
            np.fill_diagonal(dists, 1.0)
            f = diff / (dists**2 + 1e-4)[:, :, None]
            pts += 0.005 * np.sum(f, axis=1)
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts.flatten())
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_v = None
    best_sum = -np.inf
    
    inits = generate_inits()

            if -res.fun > best_sum:
                best_sum = -res.fun
                best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Joint refinement with SLSQP
    v0 = np.zeros(3*N)

            pass

    for _ in range(30):

                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
