import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    rows, rhs = [], []
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(dists[i, j])
            
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, u) for u in ub], method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-9)

def objective_func(v):
    centers = v.reshape(N, 2)
    radii = compute_lp_radii(centers)
    return -np.sum(radii)

def generate_candidates(rng, count=55):
    candidates = []
    # Hexagonal lattices with varied spacing and shifts
    for s in np.linspace(0.11, 0.19, 5):
        for sx in [-0.02, 0.0, 0.02]:
            for sy in [-0.02, 0.0, 0.02]:
                pts = []
                y = s / 2.0 + sy
                row = 0
                while len(pts) < N:
                    x = s / 2.0 + (s if row % 2 == 1 else 0.0) + sx
                    while x < 1.0 and len(pts) < N:
                        pts.append([x, y])
                        x += s
                    y += s * np.sqrt(3) / 2.0
                    row += 1
                pts = np.array(pts[:N]) + rng.normal(0, 0.005, (N, 2))
                candidates.append(np.clip(pts, 0.02, 0.98).flatten())
                
    # Random dense configurations
    for _ in range(10):
        candidates.append(rng.uniform(0.05, 0.95, (N, 2)).flatten())
        
    # Corner/edge focused configurations
    for _ in range(10):
        c = rng.uniform(0.1, 0.9, (N, 2))
        c[:4] = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        candidates.append(np.clip(c, 0.02, 0.98).flatten())
        
    return candidates

def slsqp_objective(p):
    return -np.sum(p[2::3])

def slsqp_constraints(p):
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (dx**2 + dy**2)[mask] - dr[mask]**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    rng = np.random.default_rng(42)
    bounds_opt = [(0.0, 1.0)] * (2 * N)
    
    best_val = np.inf
    best_centers = None
    
    candidates = generate_candidates(rng)
    
    # Phase 1: Global exploration with derivative-free Powell
    for v0 in candidates:
        try:
            res = minimize(objective_func, v0, method='Powell', bounds=bounds_opt,
                           options={'maxiter': 1500, 'ftol': 1e-14, 'xtol': 1e-14})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(N, 2).copy()
        except Exception:
            continue
            
    # Phase 2: Iterative perturbation to escape local minima
    if best_centers is not None:
        for i in range(25):
            sigma = 0.006 * (0.85 ** i)
            pert = best_centers + rng.normal(0, sigma, best_centers.shape)
            pert = np.clip(pert, 0.02, 0.98)
            try:
                res = minimize(objective_func, pert.flatten(), method='Powell', bounds=bounds_opt,
                               options={'maxiter': 1200, 'ftol': 1e-14})
                if res.fun < best_val:
                    best_val = res.fun
                    best_centers = res.x.reshape(N, 2).copy()
            except Exception:
                continue
                
    # Phase 3: Local refinement with SLSQP on smooth constraints
    radii = compute_lp_radii(best_centers)
    p0 = np.concatenate([best_centers.flatten(), radii])
    cons = {'type': 'ineq', 'fun': slsqp_constraints}
    bounds_slsqp = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    try:
        res = minimize(slsqp_objective, p0, method='SLSQP', bounds=bounds_slsqp,
                       constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
        if -res.fun > -best_val:
            best_centers = res.x[:2 * N].reshape(N, 2)
    except Exception:
        pass
        
    # Recompute exact optimal radii via LP for the refined centers
    final_radii = compute_lp_radii(best_centers)
    radii = final_radii.copy()
    
    # Phase 4: Strict numerical repair to guarantee validator tolerance
    for _ in range(50):
        changed = False
        for i in range(N):
            max_r = min(best_centers[i, 0], 1.0 - best_centers[i, 0], 
                        best_centers[i, 1], 1.0 - best_centers[i, 1])
            if radii[i] > max_r + 1e-9:
                radii[i] = max_r
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return best_centers, radii, final_sum
