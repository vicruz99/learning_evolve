import numpy as np
from scipy.optimize import minimize, linprog

N = 26
i_idx, j_idx = np.triu_indices(N, k=1)

def get_bounds():
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def objective(p):
    return -np.sum(p[2::3])

def constraints(p):
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c = np.empty(4 * N + N * (N - 1) // 2)
    idx = 0
    c[idx:idx+N] = x - r; idx += N
    c[idx:idx+N] = 1.0 - x - r; idx += N
    c[idx:idx+N] = y - r; idx += N
    c[idx:idx+N] = 1.0 - y - r; idx += N
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2  =>  dx^2 + dy^2 - (r_i+r_j)^2 >= 0
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    c[idx:] = dx*dx + dy*dy - dr*dr
    return c

def get_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Boundary limits
    b_lim = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    b_lim = np.maximum(b_lim, 0.0)
    
    # Pairwise distances
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)
    pair_dists = dists[i_idx, j_idx]
    
    num_pairs = N * (N - 1) // 2
    A = np.zeros((num_pairs + n, n))
    ub = np.zeros(num_pairs + n)
    
    # r_i + r_j <= dist_ij
    A[:num_pairs, i_idx] = 1.0
    A[:num_pairs, j_idx] = 1.0
    ub[:num_pairs] = pair_dists
    
    # r_i <= boundary_limit
    A[num_pairs:, :] = np.eye(n)
    ub[num_pairs:] = b_lim
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=ub, bounds=[(0, None)]*n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-6)

def repair(centers, radii):
    radii = radii.copy()
    for _ in range(50):
        changed = False
        # Clamp to boundaries
        for i in range(N):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0], 
                        centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > max_r + 1e-10:
                radii[i] = max_r
                changed = True
        # Resolve overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
    return radii

def force_relax(centers, iters=300):
    pts = centers.copy()
    rad = np.full(N, 0.04)
    for _ in range(iters):
        rad *= 1.002
        diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        
        # Repulsion force from overlaps
        overlap = rad[:, np.newaxis] + rad[np.newaxis, :] - dists
        overlap = np.maximum(overlap, 0)
        force_mag = overlap / (dists + 1e-9)
        forces = np.sum(force_mag[:, :, np.newaxis] * diff, axis=1)
        
        # Boundary repulsion
        forces += np.where(pts < rad[:, np.newaxis], (rad - pts) * 10.0, 0.0)
        forces += np.where(pts > 1.0 - rad[:, np.newaxis], (pts - (1.0 - rad)) * 10.0, 0.0)
        
        pts += forces * 0.008
        pts = np.clip(pts, 0.005, 0.995)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_p = None
    best_sum = -np.inf
    
    # Generate diverse starting configurations
    starts = []
    
    # 1. Hexagonal grids with varying densities
    for r_est in [0.08, 0.085, 0.09, 0.095, 0.10]:
        c = []
        y = r_est
        row = 0
        while len(c) < N:
            x = r_est if row % 2 == 0 else 2 * r_est
            while x + r_est <= 1.0 and len(c) < N:
                c.append([x, y])
                x += 2 * r_est
            y += np.sqrt(3) * r_est
            row += 1
        starts.append(np.array(c[:N]))
        
    # 2. Random dense clusters
    for _ in range(8):
        starts.append(np.random.rand(N, 2) * 0.8 + 0.1)
        
    # 3. Corner-focused
    for _ in range(5):
        c = np.random.rand(N, 2) * 0.6 + 0.2
        c[:4] = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        starts.append(c)

    for c0 in starts:
        # Force-directed relaxation to find dense feasible geometry
        pts = force_relax(c0 + np.random.normal(0, 0.003, c0.shape))
        
        p0 = np.zeros(3 * N)
        p0[0::3] = pts[:, 0]
        p0[1::3] = pts[:, 1]
        p0[2::3] = np.full(N, 0.08) # Start with safe radius
        
        # Primary SLSQP optimization
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13})
            c_vals = constraints(res.x)
            if np.all(c_vals >= -1e-8):
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
                    
                    # LP Reset: Optimize radii for these converged centers
                    centers_lp = best_p[:2*N].reshape(N, 2)
                    r_lp = get_lp_radii(centers_lp)
                    p_lp = np.concatenate([centers_lp.flatten(), r_lp])
                    p_lp[2::3] = np.clip(p_lp[2::3], 1e-6, 0.5)
                    
                    # Secondary SLSQP with LP-reset radii
                    res2 = minimize(objective, p_lp, method='SLSQP', bounds=bounds,
                                    constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13})
                    c_vals2 = constraints(res2.x)
                    if np.all(c_vals2 >= -1e-8):
                        s2 = -res2.fun
                        if s2 > best_sum:
                            best_sum = s2
                            best_p = res2.x.copy()
        except Exception:
            continue

    if best_p is not None:
        # Final perturbation & refinement to push out of local minima
        for _ in range(5):
            curr = best_p.copy()
            curr[2::3] *= 1.005
            curr[0::3] += np.random.normal(0, 0.0003, N)
            curr[1::3] += np.random.normal(0, 0.0003, N)
            curr[0::3] = np.clip(curr[0::3], 0.01, 0.99)
            curr[1::3] = np.clip(curr[1::3], 0.01, 0.99)
            try:
                res = minimize(objective, curr, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 2000, 'ftol': 1e-13})
                c_vals = constraints(res.x)
                if np.all(c_vals >= -1e-8):
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_p = res.x.copy()
            except Exception:
                pass
                
        # Final LP pass to guarantee optimal radii for best centers
        centers = best_p[:2*N].reshape(N, 2)
        radii = get_lp_radii(centers)
        radii = repair(centers, radii)
    else:
        # Fallback valid packing
        centers = np.random.rand(N, 2) * 0.8 + 0.1
        radii = np.full(N, 0.01)
        
    # Final safety shrink to strictly satisfy 1e-12 validator
    for _ in range(10):
        valid = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False; break
        if not valid: break
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False; break
            if not valid: break
        if valid: break
        radii *= 0.99995
        
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
