import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26

def get_lp_radii(centers):
    """
    Given fixed centers, computes the maximum feasible radii using Linear Programming.
    Constraints: r_i + r_j <= dist(i,j) and r_i <= dist_to_boundary(i)
    """
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Distance to boundaries
    b_dist = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Build LP matrices: A_ub @ r <= b_ub
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs + n, n))
    b_ub = np.zeros(num_pairs + n)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ub[idx] = b_dist[i]
        idx += 1
        
    bounds = [(0, None)] * n
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return np.full(n, 1e-6)

def obj_centers(centers_flat):
    """
    Objective function for center optimization: minimize negative sum of optimal radii.
    """
    centers = np.clip(centers_flat.reshape(N, 2), 1e-7, 1.0 - 1e-7)
    radii = get_lp_radii(centers)
    return -np.sum(radii)

def generate_structured_starts():
    """Generates diverse initial configurations for local optimization."""
    starts = []
    np.random.seed(42)
    
    # 1. Hexagonal lattices with varying densities
    for r_est in np.linspace(0.085, 0.115, 9):
        pts = []
        y = r_est
        row = 0
        while len(pts) < N:
            x_start = r_est if row % 2 == 0 else 2.0 * r_est
            x = x_start
            while x + r_est <= 1.0 + 1e-9 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r_est
            y += np.sqrt(3) * r_est
            row += 1
        starts.append(np.array(pts[:N]).flatten())
        
    # 2. Grid layouts
    xs = np.linspace(0.09, 0.91, 5)
    ys = np.linspace(0.09, 0.91, 5)
    grid = np.array(np.meshgrid(xs, ys)).T.reshape(-1, 2)
    extra = np.array([[0.04, 0.04], [0.96, 0.04], [0.04, 0.96], [0.96, 0.96]])
    # Add corner points and perturb slightly
    starts.append(np.vstack([grid, extra]).flatten())
    
    # 3. Random starts
    for _ in range(15):
        starts.append(np.random.uniform(0.05, 0.95, (N, 2)).flatten())
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    best_val = np.inf
    best_c = None
    
    # Phase 1: Global exploration with Differential Evolution
    bounds_de = [(0.02, 0.98)] * (2 * N)
    try:
        de_res = differential_evolution(
            obj_centers, bounds_de, seed=42, 
            maxiter=60, popsize=12, tol=1e-7, 
            mutation=(0.5, 1.0), recombination=0.9, polish=False
        )
        best_c = de_res.x.reshape(N, 2)
        best_val = de_res.fun
    except Exception:
        best_c = np.random.uniform(0.1, 0.9, (N, 2))
        best_val = obj_centers(best_c.flatten())
        
    # Phase 2: Local refinement on best DE result
    for method in ['Powell', 'Nelder-Mead']:
        try:
            res = minimize(obj_centers, best_c.flatten(), method=method,
                           options={'maxiter': 4000, 'xatol': 1e-9, 'fatol': 1e-11})
            if res.fun < best_val:
                best_val = res.fun
                best_c = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Phase 3: Multiple restarts from structured + random configurations
    starts = generate_structured_starts()
    for s in starts:
        try:
            res = minimize(obj_centers, s, method='Powell',
                           options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-11})
            if res.fun < best_val:
                best_val = res.fun
                best_c = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Phase 4: Perturbation & re-optimization to escape local minima
    for i in range(8):
        sigma = 0.015 * (0.8 ** i)
        pert = best_c.copy() + np.random.normal(0, sigma, best_c.shape)
        pert = np.clip(pert, 0.02, 0.98)
        try:
            res = minimize(obj_centers, pert.flatten(), method='Powell',
                           options={'maxiter': 2500, 'xatol': 1e-9})
            if res.fun < best_val:
                best_val = res.fun
                best_c = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Final center clipping and exact radius assignment
    centers = np.clip(best_c, 1e-7, 1.0 - 1e-7)
    radii = get_lp_radii(centers)
    
    # Phase 5: Strict numerical repair to satisfy validator tolerance
    for _ in range(60):
        changed = False
        for i in range(N):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0], 
                        centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > max_r - 1e-12:
                radii[i] = max_r
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-12:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
