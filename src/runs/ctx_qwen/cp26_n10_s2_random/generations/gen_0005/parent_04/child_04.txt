import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_constraints(p):
    """Vectorized constraint computation for SLSQP."""
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    n_constraints = 4 * N + N * (N - 1) // 2
    c = np.empty(n_constraints)
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Pairwise non-overlap: dist_sq - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    dist_sq = dx * dx + dy * dy
    sum_r_sq = dr * dr
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c[4*N:] = (dist_sq - sum_r_sq)[mask]
    
    return c

def objective(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[2::3])

def get_lp_radii(centers):
    """Given fixed centers, solve LP to find maximal radii satisfying all constraints."""
    n_pairs = N * (N - 1) // 2
    A_ub = np.zeros((n_pairs + 4 * N, N))
    b_ub = np.zeros(n_pairs + 4 * N)
    
    idx = 0
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.hypot(centers[i, 0] - centers[j, 0], 
                                 centers[i, 1] - centers[j, 1])
            idx += 1
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(N):
        vals = [centers[i, 0], 1.0 - centers[i, 0], 
                centers[i, 1], 1.0 - centers[i, 1]]
        for v in vals:
            A_ub[idx, i] = 1.0
            b_ub[idx] = v
            idx += 1
            
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0, None)] * N, method='highs')
    if res.success:
        return res.x
    return np.full(N, 1e-7)

def repair(centers, radii, tolerance=1e-9):
    """Deterministic repair to ensure strict validity."""
    radii = radii.copy()
    for _ in range(50):
        changed = False
        # Boundary clamping
        for i in range(N):
            mx = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mx - tolerance:
                radii[i] = max(mx, 0.0)
                changed = True
        # Pairwise shrinkage
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                if radii[i] + radii[j] > d - tolerance:
                    shrink = (radii[i] + radii[j] - d) / 2.0
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def force_directed_init(seed, iters=800):

    np.random.seed(seed)
    centers = np.random.rand(N, 2) * 0.8 + 0.1
    radii = np.full(N, 0.04)

        else:
