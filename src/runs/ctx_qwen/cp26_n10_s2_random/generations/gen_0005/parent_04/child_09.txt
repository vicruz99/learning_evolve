import numpy as np
from scipy.optimize import minimize, linprog

N = 26
i_idx, j_idx = np.triu_indices(N, k=1)
PAIR_COUNT = len(i_idx)

def get_lp_radii(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    A = np.zeros((PAIR_COUNT + 4 * N, N))
    b = np.zeros(PAIR_COUNT + 4 * N)
    
    # Pairwise non-overlap: r_i + r_j <= distance(i, j)
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    dists = np.sqrt(dx * dx + dy * dy)
    
    A[:PAIR_COUNT, i_idx] = 1.0
    A[:PAIR_COUNT, j_idx] = 1.0
    b[:PAIR_COUNT] = dists
    
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    off = PAIR_COUNT
    A[off:off+N, np.arange(N)] = 1.0; b[off:off+N] = centers[:, 0]; off += N
    A[off:off+N, np.arange(N)] = 1.0; b[off:off+N] = 1.0 - centers[:, 0]; off += N
    A[off:off+N, np.arange(N)] = 1.0; b[off:off+N] = centers[:, 1]; off += N
    A[off:off+N, np.arange(N)] = 1.0; b[off:off+N] = 1.0 - centers[:, 1]; off += N
    
    # Maximize sum(r) <=> Minimize -sum(r)
    res = linprog(-np.ones(N), A_ub=A, b_ub=b, bounds=[(0.0, None)] * N, 
                  method='highs', options={'disp': False})
    if res.success:
        return res.x
    return np.full(N, 1e-6)

def objective(centers_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    radii = get_lp_radii(centers)
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    inits = []
    
    # 1. Hexagonal lattice initializations with varying scales
    for scale in np.linspace(0.08, 0.15, 10):
        pts = []
        y = scale
        row = 0
        while len(pts) < N:
            x = scale if row % 2 == 0 else 2.0 * scale
            while x + scale <= 1.0 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * scale
            y += np.sqrt(3.0) * scale
            row += 1
        pts = np.array(pts[:N])
        pts = pts * 0.85 + 0.075
        pts += rng.normal(0, 0.005, pts.shape)
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts.flatten())
        
    # 2. Force-directed repulsion initializations
    for _ in range(12):
        pts = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(300):
            diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2)) + 1e-5
            forces = diff / (dists[:, :, np.newaxis]**2)
            pts += 0.008 * np.sum(forces, axis=1)
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts.flatten())
        
    # 3. Structured grid initializations
    for sx, sy in [(5, 5), (6, 5), (4, 7), (7, 4), (6, 6)]:
        if sx * sy < N:
            continue
        grid = np.linspace(0.08, 0.92, max(sx, sy))
        cx, cy = np.meshgrid(grid[:sx], grid[:sy])
        pts = np.column_stack((cx.flatten(), cy.flatten()))[:N]
        pts += rng.normal(0, 0.015, pts.shape)
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts.flatten())
        
    best_val = np.inf
    best_centers = None
    
    # Phase 1: Local optimization from diverse starts
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='Powell',
                           options={'maxiter': 2000, 'ftol': 1e-12, 'xtol': 1e-12})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            continue
            
    # Phase 2: Iterative perturbation to escape local minima
    if best_centers is not None:
        curr = best_centers.copy()
        for i in range(40):
            sigma = 0.015 * (0.9 ** (i // 4))
            pert = curr + rng.normal(0, sigma, curr.shape)
            pert = np.clip(pert, 0.02, 0.98)
            try:
                res = minimize(objective, pert.flatten(), method='Powell',
                               options={'maxiter': 1500, 'ftol': 1e-12, 'xtol': 1e-12})
                if res.fun < best_val:
                    best_val = res.fun
                    best_centers = res.x.reshape(N, 2)
            except Exception:
                pass
                
    # Phase 3: Compute exact optimal radii for the best centers
    radii = get_lp_radii(best_centers)
    
    # Phase 4: Strict numerical repair to guarantee validation passes
    for _ in range(100):
        changed = False
        for i in range(N):
            mx = min(best_centers[i, 0], 1.0 - best_centers[i, 0], 
                     best_centers[i, 1], 1.0 - best_centers[i, 1])
            if radii[i] > mx - 1e-9:
                radii[i] = mx
                changed = True
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                             best_centers[i, 1] - best_centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return best_centers, radii, final_sum
