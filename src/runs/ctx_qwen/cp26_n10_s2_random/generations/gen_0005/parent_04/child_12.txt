import numpy as np
from scipy.optimize import minimize, linprog

N = 26
idx_pairs = np.triu_indices(N, k=1)
NUM_PAIRS = len(idx_pairs[0])

# Precompute constraint matrix for LP to save time
A_PAIRS = np.zeros((NUM_PAIRS, N))
A_PAIRS[np.arange(NUM_PAIRS), idx_pairs[0]] = 1.0
A_PAIRS[np.arange(NUM_PAIRS), idx_pairs[1]] = 1.0
I_N = np.eye(N)
A_FULL = np.vstack([A_PAIRS, I_N])

def solve_lp_for_centers(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    c = centers.reshape(N, 2)
    
    # Distance to boundaries: min(x, 1-x, y, 1-y)
    b_dist = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                        np.minimum(c[:, 1], 1.0 - c[:, 1]))
    
    # Pairwise Euclidean distances
    dx = c[:, np.newaxis, 0] - c[np.newaxis, :, 0]
    dy = c[:, np.newaxis, 1] - c[np.newaxis, :, 1]
    dists = np.sqrt(dx*dx + dy*dy)
    b_pairs = dists[idx_pairs]
    
    # LP Right-hand side
    b_ub = np.concatenate([b_pairs, b_dist])
    
    # Solve: min -sum(r) s.t. A_ub @ r <= b_ub, r >= 0
    res = linprog(-np.ones(N), A_ub=A_FULL, b_ub=b_ub, 
                  bounds=[(0, None)]*N, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def objective_lp(centers_flat):
    """Objective for center optimization: maximize LP sum of radii."""
    c = centers_flat.reshape(N, 2)
    # Keep centers strictly inside to ensure feasible LP
    c = np.clip(c, 1e-5, 1.0 - 1e-5)
    val, _ = solve_lp_for_centers(c)
    return -val

def get_initial_configs():
    """Generates diverse starting configurations for centers."""
    configs = []
    np.random.seed(42)
    
    # 1. Hexagonal lattices at various densities
    for r in [0.085, 0.09, 0.10, 0.105, 0.11]:
        pts = []
        y, row = r, 0
        while len(pts) < N:
            x = r if row % 2 == 0 else 2*r
            while x + r <= 1.0 and len(pts) < N:
                pts.append([x, y])
                x += 2*r
            y += np.sqrt(3)*r
            row += 1
        configs.append(np.array(pts[:N]).flatten())
        
    # 2. Random dense configurations
    for _ in range(10):
        configs.append((np.random.rand(N, 2)*0.8 + 0.1).flatten())
        
    # 3. Force-directed repulsion configurations
    for _ in range(5):
        pts = np.random.rand(N, 2)*0.6 + 0.2
        for _ in range(400):
            diff = pts[:, None, :] - pts[None, :, :]
            d = np.linalg.norm(diff, axis=2) + 1e-6
            pts += 0.001 * np.sum(diff / d[:, :, None]**2, axis=1)
            pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts.flatten())
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize sum of radii."""
    bounds = [(1e-5, 1.0 - 1e-5)] * (2*N)
    configs = get_initial_configs()
    
    best_val = np.inf
    best_c = None
    
    # Phase 1: Multi-start Powell optimization
    for cfg in configs:
        try:
            res = minimize(objective_lp, cfg, method='Powell', bounds=bounds,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'xtol': 1e-14})
            if res.fun < best_val:
                best_val = res.fun
                best_c = res.x.reshape(N, 2)
        except Exception:
            continue
            
    # Phase 2: Perturbation refinement to escape local minima
    if best_c is not None:
        for _ in range(15):
            pc = best_c + np.random.normal(0, 0.0025, best_c.shape)
            pc = np.clip(pc, 1e-5, 1.0 - 1e-5)
            try:
                res = minimize(objective_lp, pc.flatten(), method='Powell', bounds=bounds,
                               options={'maxiter': 8000, 'ftol': 1e-14})
                if res.fun < best_val:
                    best_val = res.fun
                    best_c = res.x.reshape(N, 2)
            except Exception:
                continue

    # Fallback if optimization failed completely
    if best_c is None:
        best_c = np.random.rand(N, 2) * 0.8 + 0.1
        
    # Phase 3: Final exact LP radius computation
    _, radii = solve_lp_for_centers(best_c)
    radii = np.maximum(radii, 0.0)
    
    # Tiny shrinkage to guarantee strict compliance with 1e-12 validator tolerance
    radii *= 0.999995
    
    return best_c, radii, float(np.sum(radii))
