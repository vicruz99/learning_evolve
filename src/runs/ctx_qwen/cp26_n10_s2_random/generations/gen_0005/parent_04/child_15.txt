import numpy as np
from scipy.optimize import minimize, linprog

N = 26
# Precompute indices for pairwise constraints to save time
triu_i, triu_j = np.triu_indices(N, k=1)

def obj(p):
    """Objective: minimize negative sum of radii."""
    return -np.sum(p[2::3])

def cons_fn(p):
    """Computes all boundary and non-overlap constraints (must be >= 0)."""
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Boundary constraints
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    c5 = dx[triu_i, triu_j]**2 + dy[triu_i, triu_j]**2 - dr[triu_i, triu_j]**2
    
    return np.concatenate([c1, c2, c3, c4, c5])

def get_hex_init(seed, noise=0.005):
    """Generates a hexagonal lattice initialization."""
    np.random.seed(seed)
    p = np.zeros(3*N)
    idx = 0
    r0 = 0.095
    y = r0
    row = 0
    while idx < N:
        x_start = r0 if row % 2 == 0 else 2*r0
        x = x_start
        while x + r0 <= 1.0 and idx < N:
            p[3*idx] = x + np.random.normal(0, noise)
            p[3*idx+1] = y + np.random.normal(0, noise)
            p[3*idx+2] = r0 * 0.95  # Start slightly smaller to ensure feasibility
            idx += 1
            x += 2*r0
        y += np.sqrt(3)*r0
        row += 1
    p[0::3] = np.clip(p[0::3], 0.05, 0.95)
    p[1::3] = np.clip(p[1::3], 0.05, 0.95)
    p[2::3] = np.clip(p[2::3], 0.01, 0.5)
    return p

def get_force_init(seed):
    """Generates a configuration using force-directed repulsion."""
    np.random.seed(seed)
    centers = np.random.rand(N, 2) * 0.8 + 0.1
    for _ in range(400):
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.linalg.norm(diff, axis=2)
        np.fill_diagonal(dists, 1.0)
        forces = diff / (dists**2 + 1e-4)[:, :, None]
        centers += 0.008 * np.sum(forces, axis=1)
        centers = np.clip(centers, 0.05, 0.95)
    p = np.zeros(3*N)
    p[0::3] = centers[:, 0]
    p[1::3] = centers[:, 1]
    p[2::3] = 0.06  # Safe initial radius
    return p

def get_grid_init(seed, noise=0.005):
    """Generates a structured grid initialization."""
    np.random.seed(seed)
    p = np.zeros(3*N)
    idx = 0
    for r in range(6):
        for c in range(5):
            if idx >= N: break
            x = 0.1 + c * 0.2
            y = 0.1 + r * 0.18
            p[3*idx] = x + np.random.normal(0, noise)
            p[3*idx+1] = y + np.random.normal(0, noise)
            p[3*idx+2] = 0.08
            idx += 1
    p[0::3] = np.clip(p[0::3], 0.05, 0.95)
    p[1::3] = np.clip(p[1::3], 0.05, 0.95)
    return p

def get_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    A = np.zeros((num_pairs + 4*n, n))
    b = np.zeros(num_pairs + 4*n)
    idx = 0
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            idx += 1
            
    # Boundary constraints: r_i <= dist_to_boundary
    for i in range(n):
        for val in [centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1]]:
            A[idx, i] = 1.0
            b[idx] = val
            idx += 1
            
    # Minimize -sum(r) => Maximize sum(r)
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-6)

def repair(centers, radii):
    """Deterministically repairs radii to strictly satisfy constraints."""
    radii = radii.copy()
    for _ in range(50):
        changed = False
        # Boundary repair
        for i in range(N):
            mx = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mx - 1e-10:
                radii[i] = mx
                changed = True
        # Overlap repair
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-10:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov * 0.5
                    radii[j] -= ov * 0.5
                    changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': cons_fn}
    
    best_p = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    inits = []
    for s in range(12):
        inits.append(get_hex_init(s, noise=0.002 + s*0.0005))

    for s in range(6):
        inits.append(get_grid_init(s))
        
    # Phase 1: Local optimization from multiple starts
    for p0 in inits:
        try:
            res = minimize(obj, p0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-12})
            # Check constraint satisfaction
            if np.min(cons_fn(res.x)) >= -1e-8:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_p = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Perturbation refinement to escape local minima
    if best_p is not None:
        for _ in range(10):
            p_pert = best_p + np.random.normal(0, 0.001, best_p.shape)
            p_pert[0::3] = np.clip(p_pert[0::3], 0.02, 0.98)
            p_pert[1::3] = np.clip(p_pert[1::3], 0.02, 0.98)
            try:
                res = minimize(obj, p_pert, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-12})
                if np.min(cons_fn(res.x)) >= -1e-8:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_p = res.x.copy()
            except Exception:
                pass

    # Phase 3: Extract centers and solve LP for maximal radii
    centers = best_p[:2*N].reshape(N, 2)
    radii = get_lp_radii(centers)
    
    # Phase 4: Strict numerical repair
    radii = repair(centers, radii)
    
    # Final safety scaling to guarantee validator tolerance
    radii *= 0.99999
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
