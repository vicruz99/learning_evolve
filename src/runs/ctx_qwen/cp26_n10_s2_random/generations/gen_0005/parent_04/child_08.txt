import numpy as np
from scipy.optimize import minimize, linprog

N = 26
i_idx, j_idx = np.triu_indices(N, k=1)

def get_lp_radii(centers):
    """Solves LP to find maximal sum of radii for fixed centers."""
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    A = np.zeros((num_pairs + 4*n, n))
    b = np.zeros(num_pairs + 4*n)
    idx = 0
    
    # Pairwise distance constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            idx += 1
            
    # Boundary constraints: r_i <= dist to walls
    for i in range(n):
        A[idx, i] = 1.0; b[idx] = centers[i, 0]; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - centers[i, 0]; idx += 1
        A[idx, i] = 1.0; b[idx] = centers[i, 1]; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    return res.x if res.success else np.full(n, 1e-6)

def obj_powell(c_flat):
    """Objective for Powell: minimize negative sum of optimal radii."""
    c = c_flat.reshape(N, 2)
    c = np.clip(c, 1e-4, 1.0 - 1e-4)
    return -np.sum(get_lp_radii(c))

def joint_obj(p):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(p[2*N:])

def joint_cons(p):
    """Constraints for SLSQP: boundary and non-overlap (must be >= 0)."""
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    c = np.empty(4*N + len(i_idx))
    idx = 0
    c[idx:idx+N] = x - r; idx += N
    c[idx:idx+N] = 1.0 - x - r; idx += N
    c[idx:idx+N] = y - r; idx += N
    c[idx:idx+N] = 1.0 - y - r; idx += N
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    c[idx:] = dx*dx + dy*dy - dr*dr
    return c

def generate_inits():
    """Generates diverse initial center configurations."""
    inits = []
    np.random.seed(42)
    
    # 1. Hexagonal lattices with varying densities
    for r in [0.08, 0.09, 0.10, 0.11]:
        pts = []
        y = r
        row = 0
        while len(pts) < N:
            x = r if row % 2 == 0 else 2*r
            while x + r <= 1.0 and len(pts) < N:
                pts.append([x, y])
                x += 2*r
            y += np.sqrt(3)*r
            row += 1
        inits.append(np.array(pts[:N]))
        
    # 2. Force-directed repulsion layouts
    for _ in range(12):
        c = np.random.rand(N, 2) * 0.8 + 0.1
        for _ in range(600):
            diff = c[:, None, :] - c[None, :, :]
            dists = np.linalg.norm(diff, axis=2)
            np.fill_diagonal(dists, 1.0)
            forces = diff / (dists**2 + 1e-5)[:, :, None]
            c += 0.005 * np.sum(forces, axis=1)
            c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    # 3. Corner-focused placements
    for _ in range(6):
        c = np.random.rand(N, 2) * 0.6 + 0.2
        corners = np.array([[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]])
        c[:4] = corners
        inits.append(c)
        
    # 4. Uniform grid
    x = np.linspace(0.1, 0.9, 6)
    y = np.linspace(0.1, 0.9, 5)
    gx, gy = np.meshgrid(x, y)
    c = np.column_stack([gx.flatten(), gy.flatten()])[:N]
    inits.append(c)
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    inits = generate_inits()
    best_c = None
    best_sum = -1.0
    
    bounds_centers = [(0.0, 1.0)] * (2*N)
    
    # Phase 1: Derivative-free center optimization using Powell + LP objective
    for c0 in inits:
        try:
            res = minimize(obj_powell, c0.flatten(), method='Powell', bounds=bounds_centers,
                           options={'maxiter': 250, 'ftol': 1e-12, 'xtol': 1e-12})
            c_opt = res.x.reshape(N, 2)
            r_opt = get_lp_radii(c_opt)
            s = np.sum(r_opt)
            if s > best_sum:
                best_sum = s
                best_c = c_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Basin hopping to escape local minima
    if best_c is not None:
        c_curr = best_c.copy()
        for step in range(40):
            noise = 0.015 * (0.85 ** (step // 5))
            c_pert = c_curr + np.random.normal(0, noise, c_curr.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            try:
                res = minimize(obj_powell, c_pert.flatten(), method='Powell', bounds=bounds_centers,
                               options={'maxiter': 150, 'ftol': 1e-12})
                c_opt = res.x.reshape(N, 2)
                r_opt = get_lp_radii(c_opt)
                s = np.sum(r_opt)
                if s > best_sum:
                    best_sum = s
                    best_c = c_opt.copy()
                    c_curr = c_opt.copy()
            except Exception:
                pass
                
    # Phase 3: Constrained joint refinement with SLSQP
    if best_c is not None:
        r_init = get_lp_radii(best_c)
        x0 = np.concatenate([best_c.flatten(), r_init])
        
        bounds_joint = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
        cons = {'type': 'ineq', 'fun': joint_cons}
        
        try:
            res = minimize(joint_obj, x0, method='SLSQP', bounds=bounds_joint,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13})
            if -res.fun > best_sum:
                best_c = res.x[:2*N].reshape(N, 2)
                best_sum = -res.fun
        except Exception:
            pass
            
    # Phase 4: Numerical repair to guarantee strict validity
    radii = get_lp_radii(best_c)
    for _ in range(30):
        changed = False
        for i in range(N):
            mx = min(best_c[i,0], 1.0-best_c[i,0], best_c[i,1], 1.0-best_c[i,1])
            if radii[i] > mx - 1e-10:
                radii[i] = mx
                changed = True
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                if d < radii[i] + radii[j] - 1e-10:
                    ov = radii[i] + radii[j] - d
                    radii[i] -= ov*0.5
                    radii[j] -= ov*0.5
                    changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return best_c, radii, float(np.sum(radii))
