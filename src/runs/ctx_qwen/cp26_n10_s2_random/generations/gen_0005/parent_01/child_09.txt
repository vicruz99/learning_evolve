import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def compute_radii_lp(centers):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j), 0 <= r_i <= dist_to_boundary(i)
    """
    n = N
    c = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(c**2, axis=2))
    
    # Pairwise constraints: r_i + r_j <= dists[i,j]
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    m = num_pairs + num_bound
    
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.status == 0:
            return res.x
    except Exception:
        pass
        
    # Fallback: geometric safe radii
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    # Avoid inf on diagonal
    safe_dists = dists.copy()
    np.fill_diagonal(safe_dists, 10.0)
    rp = 0.5 * np.min(safe_dists, axis=1)
    return np.maximum(np.minimum(rb, rp), 1e-7)

def objective(centers_flat):
    """Objective function: maximize sum of radii => minimize negative sum."""
    centers = centers_flat.reshape(N, 2)
    # Keep strictly inside to avoid LP boundary degeneracy
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    radii = compute_radii_lp(centers)
    return -np.sum(radii)

def generate_initial_configs():
    """Generates diverse starting configurations for multi-start optimization."""
    configs = []
    np.random.seed(42)
    
    # 1. Hexagonal lattices with varying radii
    for r0 in [0.085, 0.095, 0.105, 0.115]:
        c = []
        y = r0
        row = 0
        while len(c) < N:
            x = r0 if row % 2 == 0 else 2.0 * r0
            while x + r0 <= 1.0 and len(c) < N:
                c.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
        configs.append(np.array(c[:N]).flatten())
        
    # 2. Corner-focused layouts
    for _ in range(8):
        c = np.random.rand(N, 2) * 0.6 + 0.2
        c[:4] = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
        configs.append(c.flatten())
        
    # 3. Random dense packings
    for _ in range(10):
        c = np.random.rand(N, 2) * 0.8 + 0.1
        configs.append(c.flatten())
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles in a unit square."""
    best_val = np.inf
    best_centers = None
    bounds_center = [(1e-5, 1.0 - 1e-5)] * (2 * N)
    
    configs = generate_initial_configs()
    
    # Phase 1: Multi-start Powell Optimization
    for cfg in configs:
        try:
            res = minimize(objective, cfg, method='Powell', bounds=bounds_center,
                           options={'maxiter': 4000, 'ftol': 1e-15, 'xtol': 1e-15})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            continue
            
    if best_centers is None:
        best_centers = configs[0].reshape(N, 2)
        
    # Phase 2: Coordinate Ascent Local Search
    current_centers = best_centers.copy()
    current_sum = -objective(current_centers.flatten())
    step = 0.004
    
    # Adaptive grid search for each circle
    for _ in range(8):
        improved = True
        while improved:
            improved = False
            for i in range(N):
                orig = current_centers[i].copy()
                best_local = current_centers.copy()
                best_local_sum = current_sum
                
                # 9 directions (including stay)

                    for dy in [-step, 0.0, step]:
                        if dx == 0.0 and dy == 0.0:
                            continue
                        cand = current_centers.copy()
                        cand[i, 0] = np.clip(orig[0] + dx, 1e-5, 1.0 - 1e-5)
                        cand[i, 1] = np.clip(orig[1] + dy, 1e-5, 1.0 - 1e-5)
                        cand_sum = -objective(cand.flatten())
                        if cand_sum > best_local_sum + 1e-9:
                            best_local_sum = cand_sum
                            best_local = cand.copy()
                            
                if best_local_sum > current_sum + 1e-9:
                    current_centers = best_local
                    current_sum = best_local_sum
                    improved = True
            step *= 0.75
            if step < 1e-6:
                break
                
    if current_sum > -best_val:
        best_val = -current_sum
        best_centers = current_centers
        
    # Phase 3: Perturbation & Re-optimization to escape local minima
    for trial in range(15):
        pert = best_centers + np.random.normal(0, 0.003, best_centers.shape)
        pert = np.clip(pert, 0.05, 0.95)
        try:

                           options={'maxiter': 3000, 'ftol': 1e-15, 'xtol': 1e-15})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            continue
            
    # Phase 4: Final Radius Computation & Validation Repair
    radii = compute_radii_lp(best_centers)
    
    # Deterministic shrinkage to guarantee 1e-12 tolerance compliance
    for _ in range(50):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_centers[i, 0] - best_centers[j, 0], 
                               best_centers[i, 1] - best_centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-9:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Fix boundary violations
        for i in range(N):
            mr = min(best_centers[i, 0], 1.0 - best_centers[i, 0], 
                     best_centers[i, 1], 1.0 - best_centers[i, 1])
            if radii[i] > mr - 1e-9:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return best_centers, radii, final_sum
