import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def obj_func(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def con_func(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    X = x[:, None] - x[None, :]
    Y = y[:, None] - y[None, :]
    R = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (X**2 + Y**2)[mask] - R[mask]**2])
    return c

def get_bounds():
    """Returns variable bounds for x, y, r."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def force_layout_init(c, r, steps=250):
    """Iteratively grows radii and resolves overlaps using repulsive forces."""
    n = N
    for _ in range(steps):
        r *= 1.0004
        forces = np.zeros_like(c)
        for i in range(n):
            # Boundary repulsion
            if c[i,0] - r[i] < 0: forces[i,0] += (r[i] - c[i,0]) * 10.0
            if c[i,0] + r[i] > 1: forces[i,0] -= (c[i,0] + r[i] - 1) * 10.0
            if c[i,1] - r[i] < 0: forces[i,1] += (r[i] - c[i,1]) * 10.0
            if c[i,1] + r[i] > 1: forces[i,1] -= (c[i,1] + r[i] - 1) * 10.0
            
            # Pairwise repulsion
            for j in range(i+1, n):
                dx, dy = c[i,0]-c[j,0], c[i,1]-c[j,1]
                d = math.hypot(dx, dy)
                min_d = r[i] + r[j]
                if d < min_d and d > 1e-9:
                    f = (min_d - d) * 5.0 / d
                    fx, fy = dx*f, dy*f
                    forces[i,0] += fx; forces[i,1] += fy
                    forces[j,0] -= fx; forces[j,1] -= fy
        c += forces * 0.001
        c = np.clip(c, 1e-5, 1-1e-5)
    return c, r

def compute_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum(r_i) subject to constraints."""
    n = N
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    m = num_pairs + num_bound
    
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            dist = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub[idx] = dist
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    bounds_r = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-7)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': con_func}
    
    best_v = None
    best_sum = -np.inf
    rng = np.random.default_rng(42)
    
    # --- Phase 1: Generate Diverse Initial Configurations ---
    starts = []
    
    # Hexagonal lattice patterns with variations
    for scale in [0.95, 1.0, 1.05]:
        for shift in [0.0, 0.02]:

            r0 = 0.105 * scale

            row = 0

                    x += 2 * r0
                y += np.sqrt(3) * r0
                row += 1
            while len(c) < N:
                c.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])
            c = np.array(c[:N])
            r_init = np.full(N, 0.05)
            c, r_init = force_layout_init(c, r_init, steps=200)
            v = np.zeros(3*N)
            v[0::3] = c[:,0]; v[1::3] = c[:,1]; v[2::3] = r_init
            starts.append(v)
            
    # Corner/Edge focused configurations
    for _ in range(10):
        c = np.zeros((N, 2))
        c[:4] = [[0.1,0.1], [0.9,0.1], [0.1,0.9], [0.9,0.9]]

            v_trial[[3*j, 3*j+1, 3*j+2]], v_trial[[3*i, 3*i+1, 3*i+2]]
            
        try:
            res = minimize(obj_func, v_trial, method='SLSQP', bounds=bounds,

                if np.min(c_vals) >= -1e-7:
                    best_sum = -res.fun
                    best_v = res.x.copy()

            pass
        pert_mag *= 0.95

    # --- Phase 4: LP-based Radius Maximization ---
    centers = np.column_stack((best_v[0::3], best_v[1::3]))

    for _ in range(50):

            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-9:
                radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    sum_r = float(np.sum(radii))
    return centers, radii, sum_r
