import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def compute_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum(r_i) subject to constraints."""
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    m = num_pairs + num_bound
    
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            dist = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub[idx] = dist
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    bounds_r = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-7)

def obj_func(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def cons_func(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise squared distance constraints: dist^2 >= (r_i + r_j)^2
    X = x[:, None] - x[None, :]
    Y = y[:, None] - y[None, :]
    R = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (X**2 + Y**2)[mask] - R[mask]**2])
    return c

def get_bounds():
    """Returns variable bounds for x, y, r."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def force_relax_init(seed, iterations=500):
    """Generates a well-spread configuration using repulsive forces."""
    rng = np.random.default_rng(seed)
    c = rng.uniform(0.2, 0.8, (N, 2))
    r = np.full(N, 0.06)
    
    for _ in range(iterations):
        forces = np.zeros_like(c)
        # Vectorized pairwise repulsion
        diff = c[:, None, :] - c[None, :, :]
        dists = np.linalg.norm(diff, axis=2)
        np.fill_diagonal(dists, 1e9)
        
        # Soft repulsion when dist < target
        target = r[:, None] + r[None, :]
        inv_dist = 1.0 / dists
        inv_dist[np.isinf(inv_dist)] = 0
        overlap = target - dists
        overlap = np.maximum(overlap, 0)
        
        # Apply forces
        force_vec = diff * inv_dist[:, :, None] * overlap[:, :, None]
        forces += np.sum(force_vec, axis=1)
        
        # Boundary repulsion
        for dim in range(2):
            low_viol = r - c[:, dim]
            high_viol = c[:, dim] - (1.0 - r)
            forces[:, dim] += np.maximum(low_viol, 0) * 10.0
            forces[:, dim] -= np.maximum(high_viol, 0) * 10.0
            
        c += forces * 0.02
        c = np.clip(c, 0.01, 0.99)
        r *= 1.0005  # Slowly grow radii
        
    return c, r

def generate_init_configs():

    configs = []
    np.random.seed(42)
    
    # 1. Force-relaxed starts

    # Ensure non-negativity and final sum
    radii = np.maximum(radii, 0.0)
    sum_r = float(np.sum(radii))
    
    return centers, radii, sum_r
