import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def compute_max_radii_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum(r_i) subject to non-overlap and boundary constraints.
    Returns radii array.
    """
    n = centers.shape[0]
    # Upper bounds from distance to boundaries
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 0.0)

    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)

    # LP formulation: maximize sum(r_i) s.t. r_i + r_j <= dists[i,j], 0 <= r_i <= ub[i]
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1

    c_obj = -np.ones(n)
    bounds_r = [(0.0, u) for u in ub]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-7)

def objective_centers(c_flat):
    """Objective for center optimization: maximize sum of LP-optimal radii."""
    c = c_flat.reshape(-1, 2)
    # Keep centers strictly inside to avoid degenerate LP bounds
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    r = compute_max_radii_lp(c)
    return -np.sum(r)

def generate_hex_init(row_counts, seed):
    """Generates initial configuration from a perturbed hexagonal lattice."""
    rng = np.random.default_rng(seed)
    centers = []
    r0 = 0.102
    y = r0
    row_idx = 0
    while len(centers) < N and row_idx < len(row_counts):
        cnt = row_counts[row_idx]
        shift = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(centers) >= N:
                break
            centers.append([x + rng.normal(0, 0.0015), y + rng.normal(0, 0.0015)])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row_idx += 1
        
    # Fill remaining if any (rare with correct row_counts)
    while len(centers) < N:
        centers.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
        
    return np.array(centers[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles in a unit square to maximize sum of radii."""
    np.random.seed(42)
    bounds_c = [(0.0, 1.0)] * (2 * N)
    rng = np.random.default_rng(42)
    
    best_sum = -1.0
    best_centers = None

    # Diverse hexagonal row patterns summing to >= 26
    hex_patterns = [
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5],
        [4, 6, 6, 6, 4], [5, 5, 5, 6, 5], [5, 5, 6, 5, 5],
        [6, 4, 6, 5, 5], [5, 7, 5, 5, 4], [4, 5, 7, 5, 5],
        [6, 5, 6, 5, 4], [5, 6, 6, 5, 4], [5, 5, 5, 5, 5, 1]
    ]

    candidates = []
    for pat in hex_patterns:
        if sum(pat) < N: 
            continue
        for s in range(4):
            c = generate_hex_init(pat, s)
            candidates.append(c)

    # Corner & Edge focused starts
    for s in range(12):
        c = rng.uniform(0.12, 0.88, (N, 2))
        c[0] = [0.11, 0.11]
        c[1] = [0.89, 0.11]
        c[2] = [0.11, 0.89]
        c[3] = [0.89, 0.89]
        c[4] = [0.5, 0.11]
        c[5] = [0.5, 0.89]
        c[6] = [0.11, 0.5]
        c[7] = [0.89, 0.5]
        candidates.append(c)

    # Pure random starts
    for _ in range(8):
        candidates.append(rng.uniform(0.08, 0.92, (N, 2)))

    # Phase 1: Multi-start Local Optimization
    for c0 in candidates:
        c0 = np.clip(c0, 0.02, 0.98)
        try:
            res = minimize(objective_centers, c0.flatten(), method='Powell',
                           bounds=bounds_c, options={'maxiter': 2500, 'ftol': 1e-13, 'xtol': 1e-13})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(-1, 2)
        except Exception:
            pass

    # Phase 2: Basin Hopping & Perturbation
    if best_centers is not None:
        current_c = best_centers.copy()
        current_sum = best_sum
        for step in range(50):
            noise = 0.0045 * (0.88 ** step)
            pert_c = current_c + rng.normal(0, noise, current_c.shape)
            pert_c = np.clip(pert_c, 0.02, 0.98)

            try:
                res = minimize(objective_centers, pert_c.flatten(), method='Powell',
                               bounds=bounds_c, options={'maxiter': 1500, 'ftol': 1e-13})
                if -res.fun > current_sum:
                    current_sum = -res.fun
                    current_c = res.x.reshape(-1, 2)
                    best_sum = current_sum
                    best_centers = current_c
            except Exception:
                pass

    # Phase 3: Symmetry-Breaking Swaps
    if best_centers is not None:

            idx = rng.choice(N, 2, replace=False)

            swap_c[idx] = swap_c[idx[::-1]]
            try:
                res = minimize(objective_centers, swap_c.flatten(), method='Powell',
                               bounds=bounds_c, options={'maxiter': 800, 'ftol': 1e-13})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_centers = res.x.reshape(-1, 2)
            except Exception:
                pass

    # Fallback safety
    if best_centers is None:
        best_centers = generate_hex_init([5, 5, 5, 5, 6], 0)

    # Phase 4: Exact LP Radius Extraction
    radii = compute_max_radii_lp(best_centers)

    # Phase 5: Strict Numerical Repair for Validator Tolerance
    for _ in range(40):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            mr = min(best_centers[i, 0], 1.0 - best_centers[i, 0], 
                     best_centers[i, 1], 1.0 - best_centers[i, 1])
            if radii[i] > mr - 1e-9:
                radii[i] = mr
                changed = True
                
        if not changed:
            break

    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return best_centers, radii, final_sum
