import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def get_lp_radii(centers):
    """
    Given fixed centers, solve LP to maximize sum(r_i) subject to non-overlap and boundary constraints.
    Returns optimal radii array.
    """
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    ub = np.maximum(ub, 0.0)
    
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Build LP constraints: r_i + r_j <= dists[i, j]
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Maximize sum(r_i) -> Minimize -sum(r_i)
    c_obj = -np.ones(n)
    bounds_r = [(0.0, u) for u in ub]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return res.x
    # Fallback if LP fails (numerical edge case)
    return np.full(n, 1e-7)

def objective(x):
    """Objective function for center optimization: maximize sum of LP radii."""
    centers = x.reshape(N, 2)
    # Ensure centers stay strictly inside the square for valid LP bounds
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    r = get_lp_radii(centers)
    return -np.sum(r)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles in a unit square."""
    np.random.seed(42)
    best_val = np.inf
    best_centers = None
    
    # 1. Generate diverse initial configurations
    starts = []
    
    # Hexagonal lattices with varying densities
    for r0 in [0.08, 0.09, 0.10, 0.11, 0.12]:
        c = []
        y = r0
        row = 0
        while len(c) < N:
            x_start = r0 if row % 2 == 0 else 2 * r0
            x = x_start
            while x + r0 <= 1.0 and len(c) < N:
                c.append([x, y])
                x += 2 * r0
            y += np.sqrt(3) * r0
            row += 1
        starts.append(np.array(c[:N]).flatten())
        
    # Square grids
    for s in [5, 6, 7]:
        xs = np.linspace(0.1, 0.9, s)
        ys = np.linspace(0.1, 0.9, s)
        cx, cy = np.meshgrid(xs, ys)
        grid = np.column_stack((cx.flatten(), cy.flatten()))
        starts.append(grid[:N].flatten())
        
    # Random dense configurations
    for _ in range(15):
        c = np.random.rand(N, 2) * 0.8 + 0.1
        starts.append(c.flatten())
        
    # Corner-focused configurations
    for _ in range(10):
        c = np.random.rand(N, 2)
        c[:4] = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        starts.append(c.flatten())
        
    # 2. Multi-start Powell Optimization
    bounds = [(0.0, 1.0)] * (2 * N)
    
    for i, s in enumerate(starts):
        s_pert = s.copy()
        s_pert += np.random.normal(0, 0.001, s_pert.shape)
        s_pert = np.clip(s_pert, 0.001, 0.999)
        
        try:
            res = minimize(objective, s_pert, method='Powell', bounds=bounds,
                           options={'maxiter': 4000, 'ftol': 1e-12, 'xtol': 1e-12})

                best_centers = res.x.reshape(N, 2)

            pass

                radii[i] = mr
                changed = True

            break
            
    # Ensure non-negativity
    radii = np.maximum(radii, 0.0)
    
    sum_r = float(np.sum(radii))
    return best_centers, radii, sum_r
