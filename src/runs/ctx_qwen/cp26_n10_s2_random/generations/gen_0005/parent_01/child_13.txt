import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum(r_i) subject to constraints."""
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    m = num_pairs + num_bound
    
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    bounds_r = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='interior-point')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-7)

def obj_centers(x_flat):
    """Objective for center optimization: negative sum of optimal radii."""
    c = x_flat.reshape(-1, 2)
    c = np.clip(c, 1e-5, 1.0 - 1e-5)
    r = solve_lp_radii(c)
    return -np.sum(r)

def obj_joint(v):
    """Objective for joint optimization: negative sum of radii."""
    return -np.sum(v[2::3])

def constraints_joint(v):
    """Boundary and pairwise non-overlap constraints for joint optimization."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    X = x[:, None] - x[None, :]
    Y = y[:, None] - y[None, :]
    R = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (X**2 + Y**2)[mask] - R[mask]**2])
    return c

def get_bounds_joint():
    """Returns variable bounds for x, y, r."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def init_grow_circles(rng):
    """Initialize via force-directed repulsion to find dense configurations."""
    c = rng.uniform(0.25, 0.75, (N, 2))
    r = np.full(N, 0.02)
    
    for _ in range(1500):
        r *= 1.0012
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                dx, dy = c[i, 0] - c[j, 0], c[i, 1] - c[j, 1]
                d = math.hypot(dx, dy)
                if d < r[i] + r[j] and d > 1e-9:
                    f = (r[i] + r[j] - d) * 4.0 / d
                    forces[i, 0] += f * dx
                    forces[i, 1] += f * dy
                    forces[j, 0] -= f * dx
                    forces[j, 1] -= f * dy
            # Boundary repulsion
            if c[i, 0] < r[i]: forces[i, 0] += (r[i] - c[i, 0]) * 15.0
            if c[i, 0] > 1.0 - r[i]: forces[i, 0] -= (c[i, 0] - (1.0 - r[i])) * 15.0
            if c[i, 1] < r[i]: forces[i, 1] += (r[i] - c[i, 1]) * 15.0
            if c[i, 1] > 1.0 - r[i]: forces[i, 1] -= (c[i, 1] - (1.0 - r[i])) * 15.0
            
        c += forces * 0.015
        c = np.clip(c, 0.01, 0.99)
    return c

def init_hex_grid(rng, density=1.0):
    """Hexagonal lattice initialization with perturbation."""
    c = []
    r0 = 0.105 * density
    y = r0
    row = 0
    while len(c) < N:
        x_start = r0 if row % 2 == 0 else 2 * r0
        x = x_start
        while x + r0 <= 1.0 + 1e-9 and len(c) < N:
            c.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
    while len(c) < N:
        c.append([rng.uniform(0.15, 0.85), rng.uniform(0.15, 0.85)])
    return np.clip(np.array(c[:N]), 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles in a unit square."""

    best_sum = -np.inf

    # Phase 2: Optimize centers using LP-guided Powell search
    for c0 in candidates:

            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum

        except Exception:
            pass
            
    # Phase 3: Perturbation, Swap & Local Refinement

                trial[i], trial[j] = trial[j], trial[i]

                pass

        best_c = init_grow_circles(rng)

        pass

    radii = solve_lp_radii(best_c)

                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Clamp to boundaries
        for i in range(N):
            mr = min(best_c[i, 0], 1.0 - best_c[i, 0], best_c[i, 1], 1.0 - best_c[i, 1])
            if radii[i] > mr - 1e-9:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return best_c, radii, final_sum
