import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def compute_radii_lp(centers):
    """Solves LP to maximize sum(r_i) subject to r_i + r_j <= dist(i,j) and boundary constraints."""
    n = N
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    m = num_pairs + num_bound
    
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    # Pairwise constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            dist = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub[idx] = dist
            idx += 1
            
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    bounds_r = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-7)

def obj_centers(c_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = c_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    radii = compute_radii_lp(centers)
    return -np.sum(radii)

def generate_hex_start(row_pattern, r0=0.095):
    """Generates a hexagonal-like initial configuration based on row counts."""
    centers = []
    y = r0
    for r_idx, cnt in enumerate(row_pattern):
        shift = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            if len(centers) >= N: break
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    centers = np.array(centers[:N])
    # Normalize to unit square with margin
    min_c = centers.min(axis=0)
    max_c = centers.max(axis=0)
    centers = (centers - min_c) / (max_c - min_c) * 0.8 + 0.1
    return centers

def generate_repulsive_start(rng):
    """Generates a well-spaced configuration using force repulsion."""
    c = rng.uniform(0.2, 0.8, (N, 2))
    for _ in range(500):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                d_vec = c[i] - c[j]
                dist = np.linalg.norm(d_vec)
                if dist < 0.2 and dist > 1e-5:
                    push = (0.2 - dist) * 0.5 / dist
                    f = d_vec * push
                    forces[i] += f
                    forces[j] -= f
        c += forces
        c = np.clip(c, 0.05, 0.95)
    return c

def optimize_single_circle(centers, idx, rng):
    """Optimizes position of circle idx while keeping others fixed."""
    best_sum = -obj_centers(centers.flatten())
    
    def local_obj(pos):
        temp = centers.copy()
        temp[idx] = np.clip(pos, 1e-5, 1.0 - 1e-5)
        return obj_centers(temp.flatten())
        
    try:
        res = minimize(local_obj, centers[idx], method='L-BFGS-B', 
                       bounds=[(1e-5, 1.0-1e-5), (1e-5, 1.0-1e-5)],
                       options={'maxiter': 100, 'ftol': 1e-10})
        if -res.fun > best_sum + 1e-9:
            centers[idx] = res.x
            return True
    except Exception:
        pass
    return False

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    
    return best_centers, radii, final_sum
