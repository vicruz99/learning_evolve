import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution
import math

N = 26

def compute_max_radii_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum(r_i) subject to non-overlap and boundary constraints.
    Returns radii array.
    """
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    m = num_pairs + num_bound
    
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            dist = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub[idx] = dist
            idx += 1
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    bounds_r = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-7)

def objective_de(centers_flat):
    """Objective for differential evolution: minimize negative sum of radii."""
    centers = centers_flat.reshape(-1, 2)
    radii = compute_max_radii_lp(centers)
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Phase 1: Global search with Differential Evolution

            objective_de,
            bounds,
            popsize=15,
            maxiter=200,
            mutation=(0.5, 1.5),
            recombination=0.9,
            seed=42,
            workers=1,
            polish=False
        )
        best_centers = de_result.x.reshape(N, 2)
        best_radii = compute_max_radii_lp(best_centers)
        best_sum = np.sum(best_radii)
    except Exception:
        # Fallback to hexagonal grid if DE fails or times out
        best_centers = np.zeros((N, 2))
        r0 = 0.105
        idx = 0
        y = r0
        row = 0

                best_centers[idx] = [x, y]
                idx += 1
                x += 2 * r0

            row += 1

        trial = best_centers.copy()

        trial[idx] = np.clip(trial[idx], 1e-5, 1.0 - 1e-5)

                best_radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    sum_r = float(np.sum(best_radii))
    
    return best_centers, best_radii, sum_r
