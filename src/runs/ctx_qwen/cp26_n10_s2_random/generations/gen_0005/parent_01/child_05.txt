import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def compute_max_radii_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum(r_i) subject to non-overlap and boundary constraints.
    Returns the optimal radii array.
    """
    n = N
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    m = num_pairs + num_bound
    
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b_ub[idx] = math.hypot(dx, dy)
            idx += 1
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    bounds_r = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-7)

def obj_centers(centers_flat):
    """
    Objective function for center optimization: maximize sum of radii.
    Returns negative sum for minimization.
    """
    centers = centers_flat.reshape(N, 2)
    # Strictly keep centers inside bounds to ensure LP feasibility
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    radii = compute_max_radii_lp(centers)
    return -np.sum(radii)

def get_init_hex():
    """Generates a hexagonal lattice initialization."""
    centers = []
    r0 = 0.105
    y = r0
    row = 0
    while len(centers) < N:
        x_start = r0 if row % 2 == 0 else 2 * r0
        x = x_start
        while x + r0 <= 1.0 + 1e-9 and len(centers) < N:
            centers.append([x, y])
            x += 2 * r0
        y += math.sqrt(3) * r0
        row += 1
    while len(centers) < N:
        centers.append([np.random.rand() * 0.8 + 0.1, np.random.rand() * 0.8 + 0.1])
    return np.array(centers[:N])
