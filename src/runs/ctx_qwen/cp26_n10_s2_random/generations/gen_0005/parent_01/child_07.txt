import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def solve_lp_radii(centers):
    """Solve LP to maximize sum of radii given fixed centers."""
    n = centers.shape[0]
    m_pairs = n * (n - 1) // 2
    m_bounds = 4 * n
    m = m_pairs + m_bounds
    
    A = np.zeros((m, n))
    b = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            idx += 1
            
    for i in range(n):
        A[idx, i] = 1.0; b[idx] = centers[i, 0]; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - centers[i, 0]; idx += 1
        A[idx, i] = 1.0; b[idx] = centers[i, 1]; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    bounds_r = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds_r, method='highs')
        if res.success:
            # Extract duals carefully for different scipy versions
            duals = None
            if hasattr(res, 'ineqlin'):
                duals = getattr(res.ineqlin, 'marginals', None)
            if duals is None and hasattr(res, 'marginals'):
                duals = getattr(res.marginals, 'ineqlin', None)
            if duals is None:
                duals = np.zeros(m)
            return res.x, duals
    except Exception:
        pass
    return np.full(n, 1e-7), np.zeros(m)

def objective_centers(c_flat):
    """Objective: maximize sum of radii -> minimize negative sum."""
    centers = c_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    radii, _ = solve_lp_radii(centers)
    return -np.sum(radii)

def force_refinement(centers, iterations=4000, lr_init=1e-4):
    """Gradient ascent on centers using LP dual variables."""
    best_sum = -np.inf
    best_c = centers.copy()
    best_radii = None
    
    for it in range(iterations):
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
        radii, duals = solve_lp_radii(centers)
        curr_sum = np.sum(radii)
        
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_c = centers.copy()
            best_radii = radii.copy()
            
        lr = lr_init * (0.9995 ** it)
        forces = np.zeros_like(centers)
        
        if len(duals) < 4 * N:
            continue
            
        pair_idx = 0
        for i in range(N):
            for j in range(i + 1, N):
                mu = duals[pair_idx]
                if mu > 1e-7:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = math.hypot(dx, dy)
                    if d > 1e-9:
                        fx = mu * dx / d
                        fy = mu * dy / d
                        forces[i] += [fx, fy]
                        forces[j] -= [fx, fy]
                pair_idx += 1
                
        bound_idx = pair_idx
        for i in range(N):
            forces[i, 0] += duals[bound_idx]
            bound_idx += 1
            forces[i, 0] -= duals[bound_idx]
            bound_idx += 1
            forces[i, 1] += duals[bound_idx]
            bound_idx += 1
            forces[i, 1] -= duals[bound_idx]
            bound_idx += 1
            
        centers += lr * forces
        
    return best_c, best_radii, best_sum

def init_configs():
    """Generate diverse initial configurations."""
    configs = []
    np.random.seed(42)
    
    # Hexagonal grids with varying densities
    for r0 in [0.08, 0.09, 0.10, 0.105, 0.11]:
        c = []
        y = r0
        row = 0
        while len(c) < N:
            x = r0 if row % 2 == 0 else 2 * r0
            while x + r0 <= 1.0 and len(c) < N:
                c.append([x, y])
                x += 2 * r0
            y += np.sqrt(3) * r0
            row += 1
        configs.append(np.array(c[:N]))
        
    # Perturbed hex grids
    for i, c_hex in enumerate(configs[:3]):
        c_pert = c_hex + np.random.normal(0, 0.015, c_hex.shape)
        c_pert = np.clip(c_pert, 0.05, 0.95)
        configs.append(c_pert)
        
    # Random + repulsion to find non-lattice optima
    for _ in range(5):
        c = np.random.uniform(0.1, 0.9, (N, 2))
        for _ in range(800):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i+1, N):
                    d = np.linalg.norm(c[i] - c[j])
                    if d < 0.22 and d > 1e-5:
                        f = (0.22 - d) / d
                        forces[i] += (c[i] - c[j]) * f
                        forces[j] -= (c[i] - c[j]) * f
            c += forces * 0.015
            c = np.clip(c, 0.05, 0.95)
        configs.append(c)
        
    # Corner/Edge focused placement
    c_corners = np.array([
        [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9],
        [0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5]
    ])
    rem = np.random.uniform(0.2, 0.8, (N - 8, 2))
    configs.append(np.vstack([c_corners, rem]))
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles in a unit square."""
    np.random.seed(42)
    bounds = [(0.001, 0.999)] * (2 * N)
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    inits = init_configs()
    
    # Phase 1: Local optimization on centers using SLSQP
    for c0 in inits:
        try:
            res = minimize(objective_centers, c0.flatten(), method='SLSQP',
                           bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-12})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Phase 2: Dual-based force refinement to escape local minima and fine-tune
    candidates = [best_centers]
    candidates.append(best_centers + np.random.normal(0, 0.005, best_centers.shape))
    candidates.append(best_centers + np.random.normal(0, 0.01, best_centers.shape))
    
    for cand_c in candidates:
        cand_c = np.clip(cand_c, 0.01, 0.99)
        ref_c, ref_r, ref_s = force_refinement(cand_c, iterations=3000, lr_init=1e-4)
        if ref_s > best_sum:
            best_sum = ref_s
            best_centers = ref_c
            best_radii = ref_r
            
    if best_radii is None:
        best_radii, _ = solve_lp_radii(best_centers)
        
    # Phase 3: Strict Validation Repair
    for _ in range(50):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
                if d < best_radii[i] + best_radii[j] - 1e-9:
                    shrink = (best_radii[i] + best_radii[j] - d) / 2.0 + 1e-9
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            mr = min(best_centers[i, 0], 1.0 - best_centers[i, 0], best_centers[i, 1], 1.0 - best_centers[i, 1])
            if best_radii[i] > mr - 1e-9:
                best_radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    # Ensure non-negativity
    best_radii = np.maximum(best_radii, 0.0)
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
