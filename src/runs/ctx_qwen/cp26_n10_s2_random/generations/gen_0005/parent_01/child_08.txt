import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def compute_max_radii_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum(r_i) subject to non-overlap and boundary constraints.
    Returns radii array.
    """
    n = centers.shape[0]
    num_pairs = n * (n - 1) // 2
    num_bound = 4 * n
    m = num_pairs + num_bound
    
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b_ub[idx] = math.hypot(dx, dy)
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    c_obj = -np.ones(n)
    bounds_r = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-7)

def obj_centers_only(c_flat):
    """Objective for center optimization: maximize sum of LP-optimized radii."""
    c = np.clip(c_flat.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    r = compute_max_radii_lp(c)
    return -np.sum(r)

def force_repulsion_init(rng, n, iters=600):
    """Initializes centers using force-directed repulsion, heavily favoring corners/edges."""
    c = rng.uniform(0.25, 0.75, (n, 2))
    corners = np.array([[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]])
    c[:4] = corners + rng.normal(0, 0.02, (4, 2))
    
    for _ in range(iters):
        forces = np.zeros_like(c)
        for i in range(n):
            for j in range(i + 1, n):
                d_vec = c[i] - c[j]
                dist = np.linalg.norm(d_vec)
                if dist < 0.22 and dist > 1e-5:
                    push = (0.22 - dist) * 3.0
                    f = (d_vec / dist) * push
                    forces[i] += f
                    forces[j] -= f
            for k in range(2):
                if c[i, k] < 0.08: forces[i, k] += (0.08 - c[i, k]) * 15.0
                if c[i, k] > 0.92: forces[i, k] -= (c[i, k] - 0.92) * 15.0
        c += forces * 0.012
        c = np.clip(c, 0.02, 0.98)
    return c

def hex_init(rng, n, r0=0.105):
    """Generates a perturbed hexagonal lattice initialization."""
    c = []
    y = r0
    row = 0
    while len(c) < n:
        x = r0 if row % 2 == 0 else 2 * r0
        while x + r0 <= 1.0 and len(c) < n:
            c.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
    c = np.array(c[:n])
    c += rng.normal(0, 0.006, c.shape)
    c = np.clip(c, 0.02, 0.98)
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles in a unit square."""
    np.random.seed(42)
    rng = np.random.default_rng(42)
    
    best_c = None
    best_r = None
    best_sum = -np.inf
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    for _ in range(12):
        inits.append(force_repulsion_init(rng, N))
    for r0 in [0.09, 0.10, 0.11, 0.12, 0.13]:
        inits.append(hex_init(rng, N, r0))
    for _ in range(8):
        inits.append(rng.uniform(0.15, 0.85, (N, 2)))
        
    # Phase 2: Optimize centers using Powell on LP objective
    for c0 in inits:
        try:
            res = minimize(obj_centers_only, c0.flatten(), method='Powell', 
                           options={'maxiter': 2500, 'ftol': 1e-11, 'xtol': 1e-11})
            c_opt = np.clip(res.x.reshape(N, 2), 0.01, 0.99)
            r_opt = compute_max_radii_lp(c_opt)
            s = np.sum(r_opt)
            if s > best_sum:
                best_sum = s
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            pass
            
    # Phase 3: Basin Hopping with Swaps & Perturbations
    for step in range(45):
        c_pert = best_c.copy()
        noise = 0.004 * (0.92 ** (step // 3))
        c_pert += rng.normal(0, noise, c_pert.shape)
        
        # Random swap to explore different topological arrangements
        idx = rng.choice(N, 2, replace=False)
        c_pert[idx] = c_pert[idx[::-1]]
        
        c_pert = np.clip(c_pert, 0.02, 0.98)
        
        try:
            res = minimize(obj_centers_only, c_pert.flatten(), method='Nelder-Mead',
                           options={'maxiter': 1500, 'xatol': 1e-9, 'fatol': 1e-11})
            c_new = np.clip(res.x.reshape(N, 2), 0.01, 0.99)
            r_new = compute_max_radii_lp(c_new)
            s_new = np.sum(r_new)
            if s_new > best_sum:
                best_sum = s_new
                best_c = c_new.copy()
                best_r = r_new.copy()
        except Exception:
            pass
            
    # Phase 4: Joint SLSQP refinement for final polishing
    v0 = np.zeros(3 * N)
    v0[0::3] = best_c[:, 0]
    v0[1::3] = best_c[:, 1]
    v0[2::3] = best_r
    
    def joint_obj(v):
        return -np.sum(v[2::3])
        
    def joint_cons(v):
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
        X = x[:, None] - x[None, :]
        Y = y[:, None] - y[None, :]
        R = r[:, None] + r[None, :]
        mask = np.triu(np.ones((N, N), dtype=bool), k=1)
        c = np.concatenate([c, (X**2 + Y**2)[mask] - R[mask]**2])
        return c

    bounds_j = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons_j = {'type': 'ineq', 'fun': joint_cons}
    
    try:
        res = minimize(joint_obj, v0, method='SLSQP', bounds=bounds_j, 
                       constraints=cons_j, options={'maxiter': 6000, 'ftol': 1e-13})
        if -res.fun > best_sum:
            best_c = np.column_stack((res.x[0::3], res.x[1::3]))
            best_r = res.x[2::3].copy()
            best_sum = -res.fun
    except Exception:
        pass
        
    # Phase 5: Final LP recomputation & Strict Numerical Repair
    final_r = compute_max_radii_lp(best_c)
    
    for _ in range(60):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                if d < final_r[i] + final_r[j] - 1e-9:
                    shrink = (final_r[i] + final_r[j] - d) / 2.0 + 1e-9
                    final_r[i] -= shrink
                    final_r[j] -= shrink
                    changed = True
        # Clamp to boundaries
        for i in range(N):
            mr = min(best_c[i, 0], 1.0 - best_c[i, 0], best_c[i, 1], 1.0 - best_c[i, 1])
            if final_r[i] > mr - 1e-9:
                final_r[i] = mr
                changed = True
        if not changed:
            break
            
    final_r = np.maximum(final_r, 0.0)
    final_sum = float(np.sum(final_r))
    
    return best_c, final_r, final_sum
