

<strategy>
The key insight to improve upon the previous joint SLSQP optimization is to **decouple center placement from radius calculation**. The radius subproblem (given fixed centers) is a convex Linear Program. By optimizing centers to maximize the output of this LP, we obtain a smoother and more reliable objective landscape that naturally pushes circles to optimal positions while allowing variable radii to fill gaps.

Strategy:
1. **LP-based Radius Evaluation**: Precompute the constraint matrix structure for `r_i + r_j <= dist(i,j)` and boundary limits. For any given set of centers, solve the LP to get the exact maximum sum of radii. This guarantees optimality for the radius variables.
2. **Diverse Initialization**: Generate multiple starting configurations (hexagonal lattices with varying scales/shifts, square grids, corner-focused layouts, and dense random points) to cover different topological basins.
3. **Derivative-Free Center Optimization**: Use `scipy.optimize.minimize` with the `Powell` method on the center coordinates, minimizing `-LP_sum_radii(centers)`. Powell handles non-smooth objectives well and respects box bounds.
4. **Simulated Annealing**: Run a temperature-cooled perturbation loop on centers to escape local minima. Perturb 1-2 circles at a time, accepting worse moves probabilistically.
5. **Final Refinement & Repair**: Run a final Powell optimization on the best found configuration. Extract radii via LP, then apply a strict numerical repair pass to guarantee the validation constraints are met within tolerance.

This approach avoids the non-convexity pitfalls of joint non-linear constraints, leverages the exactness of LP for radii, and uses robust global/local search hybrids for centers.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
M_PAIRS = N * (N - 1) // 2

# Precompute LP structure for pairwise constraints
A_ub_lp = np.zeros((M_PAIRS, N))
idx = 0
pair_indices = []
for i in range(N):
    for j in range(i + 1, N):
        A_ub_lp[idx, i] = 1.0
        A_ub_lp[idx, j] = 1.0
        pair_indices.append((i, j))
        idx += 1
pair_indices = np.array(pair_indices)
c_obj_lp = -np.ones(N)

def compute_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum(r_i) subject to non-overlap and boundary constraints."""
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-8)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    b_ub = dists[pair_indices[:, 0], pair_indices[:, 1]]
    bounds = [(0.0, u) for u in ub]
    
    try:
        res = linprog(c_obj_lp, A_ub=A_ub_lp, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(N, 1e-7)

def obj_centers(x_flat):
    """Objective for center optimization: minimize negative sum of LP-optimal radii."""
    centers = x_flat.reshape(N, 2)
    radii = compute_lp_radii(centers)
    return -np.sum(radii)

def generate_initial_configs(rng):
    """Generates diverse initial center configurations."""
    configs = []
    
    # 1. Hexagonal lattices with varying densities and shifts
    for r0 in [0.09, 0.10, 0.11, 0.115]:
        for shift in [0.0, 0.02, 0.04]:
            pts = []
            y = r0
            row = 0
            while len(pts) < N:
                x = r0 + shift
                while x + r0 <= 1.0 and len(pts) < N:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3.0) * r0
                row += 1
            configs.append(np.array(pts[:N]))
            
    # 2. Square grids
    for s in [5, 6]:
        xs = np.linspace(0.1, 0.9, s)
        ys = np.linspace(0.1, 0.9, s)
        gx, gy = np.meshgrid(xs, ys)
        grid = np.column_stack((gx.flatten(), gy.flatten()))
        configs.append(grid[:N])
        
    # 3. Corner and edge focused
    corners = np.array([[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]])
    edges = np.array([[0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5]])
    for _ in range(6):
        base = np.vstack([corners, edges])
        rem = rng.uniform(0.2, 0.8, (N - 8, 2))
        configs.append(np.vstack([base, rem]))
        
    # 4. Dense random
    for _ in range(8):
        configs.append(rng.uniform(0.05, 0.95, (N, 2)))
        
    return configs

def run_packing():
    """Main function to pack 26 circles in a unit square maximizing sum of radii."""
    np.random.seed(42)
    rng = np.random.default_rng(42)
    bounds_c = [(0.0, 1.0)] * (2 * N)
    
    best_val = np.inf
    best_centers = None
    
    inits = generate_initial_configs(rng)
    
    # Phase 1: Local optimization on diverse starts using Powell
    for cfg in inits:
        c_jit = cfg + rng.normal(0, 0.002, cfg.shape)
        c_jit = np.clip(c_jit, 0.02, 0.98)
        
        try:
            res = minimize(obj_centers, c_jit.flatten(), method='Powell', 
                           bounds=bounds_c, options={'maxiter': 2000, 'ftol': 1e-13})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass
            
    # Phase 2: Simulated Annealing to escape local minima
    if best_centers is not None:
        curr_c = best_centers
        curr_val = -best_val
        T = 0.008
        
        for step in range(2000):
            # Perturb 1 or 2 circles randomly
            k = rng.integers(1, 3)
            indices = rng.choice(N, k, replace=False)
            new_c = curr_c.copy()
            new_c[indices] += rng.normal(0, T, (k, 2))
            new_c[indices] = np.clip(new_c[indices], 0.02, 0.98)
            
            new_val = -obj_centers(new_c.flatten())
            delta = new_val - curr_val
            
            # Accept better or probabilistically accept worse
            if delta > 0 or rng.random() < np.exp(delta / (T + 1e-9)):
                curr_c = new_c
                curr_val = new_val
                
            T *= 0.998
            if curr_val > -best_val:
                best_val = -curr_val
                best_centers = curr_c.copy()
                
    # Phase 3: Final high-precision local refinement
    try:
        res = minimize(obj_centers, best_centers.flatten(), method='Powell',
                       bounds=bounds_c, options={'maxiter': 3000, 'ftol': 1e-14})
        if -res.fun > -best_val:
            best_centers = res.x.reshape(N, 2)
            best_val = res.fun
    except Exception:
        pass
        
    # Extract optimal radii for the best centers
    radii = compute_lp_radii(best_centers)
    
    # Phase 4: Strict numerical repair to guarantee validation passes
    for _ in range(50):
        changed = False
        # Resolve pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(best_centers[i, 0] - best_centers[j, 0], 
                               best_centers[i, 1] - best_centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
                    
        # Clamp to boundaries
        for i in range(N):
            mr = min(best_centers[i, 0], 1.0 - best_centers[i, 0], 
                     best_centers[i, 1], 1.0 - best_centers[i, 1])
            if radii[i] > mr - 1e-9:
                radii[i] = mr
                changed = True
                
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    sum_r = float(np.sum(radii))
    return best_centers, radii, sum_r
