import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def compute_constraints(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    con = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise constraints: dist_sq - (r_i + r_j)^2 >= 0
    X = x[:, None] - x[None, :]
    Y = y[:, None] - y[None, :]
    R = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    con = np.concatenate([con, (X**2 + Y**2)[mask] - R[mask]**2])
    return con

def objective(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def get_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum(r_i) subject to constraints."""
    n = centers.shape[0]
    m_pairs = n * (n - 1) // 2
    m_bound = 4 * n
    m = m_pairs + m_bound
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b_ub[idx] = d
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-7)

def relax_centers(centers, iterations=200):
    """Force-directed relaxation to spread centers and avoid initial overlaps."""
    c = centers.copy()
    r_est = np.full(len(c), 0.085)
    for _ in range(iterations):
        forces = np.zeros_like(c)
        for i in range(len(c)):
            for j in range(i + 1, len(c)):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = math.hypot(dx, dy)
                min_dist = r_est[i] + r_est[j]
                if dist < min_dist and dist > 1e-6:
                    push = (min_dist - dist) * 2.0 / dist
                    fx = dx * push
                    fy = dy * push
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
            for dim in range(2):
                if c[i, dim] < r_est[i]:
                    forces[i, dim] += (r_est[i] - c[i, dim]) * 5.0
                if c[i, dim] > 1.0 - r_est[i]:
                    forces[i, dim] -= (c[i, dim] - (1.0 - r_est[i])) * 5.0
        c += forces * 0.005
        c = np.clip(c, 0.01, 0.99)
    return c

def generate_inits(num_inits):
    """Generates diverse initial configurations for multi-start optimization."""
    inits = []
    for s in range(num_inits):
        rng = np.random.default_rng(s)
        c = np.zeros((N, 2))
        # Corners
        c[0] = [0.12, 0.12]
        c[1] = [0.88, 0.12]
        c[2] = [0.12, 0.88]
        c[3] = [0.88, 0.88]
        # Edges
        c[4] = [0.5, 0.1]
        c[5] = [0.5, 0.9]
        c[6] = [0.1, 0.5]
        c[7] = [0.9, 0.5]
        # Center hex grid
        idx = 8

        y = 0.4
        row = 0
        while idx < N:

            x = x_start
            while x <= 0.65 and idx < N:
                c[idx] = [x + rng.uniform(-0.02, 0.02), y + rng.uniform(-0.02, 0.02)]
                idx += 1
                x += 2 * r0
            y += np.sqrt(3) * r0
            row += 1

        r_init = get_radii_lp(c)
        v = np.zeros(3 * N)
        v[0::3] = c[:, 0]
        v[1::3] = c[:, 1]
        v[2::3] = r_init
        inits.append(v)
        
        # Add random repulsion starts for diversity

            cr = rng.uniform(0.1, 0.9, (N, 2))

            rr = get_radii_lp(cr)

            vr[0::3] = cr[:, 0]
            vr[1::3] = cr[:, 1]
            vr[2::3] = rr
            inits.append(vr)
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_v = None
    best_sum = -np.inf
    
    # Phase 1: Multi-start Optimization
    inits = generate_inits(15)
    for i, v0 in enumerate(inits):

                    best_sum = s_val

        except Exception:
            pass

            v_trial[1::3] = np.clip(v_trial[1::3], 0.02, 0.98)

                        best_v = res.x.copy()
            except Exception:
                pass
            perturbation_mag *= 0.94

    if best_v is None:
        best_v = inits[0]
        
    # Phase 3: Extract centers and LP-refine radii for guaranteed maximality
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = get_radii_lp(centers)
    
    # Phase 4: Strict Validation Repair
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-9:

                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            if radii[i] > mr - 1e-9:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    sum_r = float(np.sum(radii))
    return centers, radii, sum_r
