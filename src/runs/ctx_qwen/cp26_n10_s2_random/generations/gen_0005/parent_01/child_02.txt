import numpy as np
from scipy.optimize import linprog, minimize
import math

N = 26

def get_radii_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum(r_i) subject to non-overlap and boundary constraints.
    Returns optimal radii array.
    """
    n = centers.shape[0]
    # Distance to boundaries
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    rb = np.maximum(rb, 0.0)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # LP formulation: max sum(r_i) s.t. r_i + r_j <= dists[i,j], r_i <= rb[i], r_i >= 0
    m = n * (n - 1) // 2 + n
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ub[idx] = rb[i]
        idx += 1
        
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    # Fallback to independent maximum radii if LP fails
    rp = 0.5 * np.min(dists, axis=1)
    return np.minimum(rb, rp)

def obj_func(centers_flat):
    """Objective function: minimize negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    # Clip to strictly inside the unit square to avoid boundary singularities
    centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
    radii = get_radii_lp(centers)
    return -np.sum(radii)
