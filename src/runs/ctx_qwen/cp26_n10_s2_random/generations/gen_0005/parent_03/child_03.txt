import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_constraints(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints: (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (dx**2 + dy**2 - dr**2)[mask]])
    return c

def objective_func(v):
    """Objective: minimize negative sum of radii to maximize total radius."""
    return -np.sum(v[2::3])

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def get_safe_radii(centers):
    """Computes strictly feasible initial radii for a given set of centers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    rn = 0.5 * np.min(dists, axis=1)
    return np.minimum(ub, rn) * 0.95

def hex_init(row_counts, r0, seed):
    """Generates initial configuration from a perturbed hexagonal lattice."""
    rng = np.random.default_rng(seed)
    centers = []
    y = r0
    for row_idx, count in enumerate(row_counts):
        offset = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + offset
        for _ in range(count):
            if len(centers) >= N:
                break
            centers.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    centers = np.array(centers[:N])
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def repair_solution(centers, radii):
    """Iteratively shrinks radii to resolve overlaps and clamp to boundaries."""
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
            if radii[i] > mr - 1e-9:
                radii[i] = mr
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    return radii

def refine_with_lp(centers):
    """Given fixed centers, solve LP to exactly maximize sum of radii."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    A_ub = []
    b_ub = []
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c_obj = -np.ones(n)
    
    bounds = [(0, u) for u in ub]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles in a unit square to maximize sum of radii."""
    rng = np.random.default_rng(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_v = None
    best_sum = -np.inf
    
    # Phase 1: Generate diverse initial configurations
    row_patterns = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], 
                    [6, 5, 5, 5, 5], [5, 5, 5, 6, 5], [4, 6, 6, 6, 4]]
    inits = []
    
    for pat in row_patterns:
        for r0 in [0.085, 0.09, 0.095, 0.10]:
            c = hex_init(pat, r0, len(inits))
            v = np.zeros(3 * N)
            v[0::3] = c[:, 0]
            v[1::3] = c[:, 1]
            v[2::3] = get_safe_radii(c)
            inits.append(v)
            
    # Add force-repulsive starts
    for _ in range(15):
        c = rng.random((N, 2)) * 0.8 + 0.1
        for _ in range(150):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if dist < 0.2 and dist > 1e-5:
                        f = (0.2 - dist) * 0.05
                        forces[i] += (d_vec / dist) * f
                        forces[j] -= (d_vec / dist) * f
            c += forces
            c = np.clip(c, 0.05, 0.95)
        v = np.zeros(3 * N)
        v[0::3] = c[:, 0]
        v[1::3] = c[:, 1]
        v[2::3] = get_safe_radii(c)
        inits.append(v)
        
    # Phase 2: Multi-start nonlinear optimization
    for v0 in inits:
        try:
            res = minimize(objective_func, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13})
            s_val = -res.fun
            c_vals = compute_constraints(res.x)
            if np.min(c_vals) >= -1e-6 and s_val > best_sum:
                best_sum = s_val
                best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Perturbation & Swap Search to escape local minima
    if best_v is not None:
        for step in range(80):
            v_trial = best_v.copy()
            c_trial = v_trial[:2*N].reshape(N, 2)
            r_trial = v_trial[2*N:].copy()
            
            noise = 0.002 * (0.95 ** step)
            
            # Random translation of a subset of circles
            idx = rng.choice(N, size=max(1, N//4), replace=False)
            c_trial[idx] += rng.normal(0, noise, (len(idx), 2))
            
            # Random swap of two circles to break symmetry
            if rng.random() < 0.3:
                s_idx = rng.choice(N, 2, replace=False)
                c_trial[s_idx] = c_trial[s_idx[::-1]]
                r_trial[s_idx] = r_trial[s_idx[::-1]]
                
            c_trial = np.clip(c_trial, 0.01, 0.99)
            r_trial *= 0.96
            r_trial = np.clip(r_trial, 0.01, 0.45)
            
            v_trial[:2*N] = c_trial.flatten()
            v_trial[2*N:] = r_trial
            
            try:
                res = minimize(objective_func, v_trial, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13})
                if -res.fun > best_sum:
                    c_vals = compute_constraints(res.x)
                    if np.min(c_vals) >= -1e-6:
                        best_sum = -res.fun
                        best_v = res.x.copy()
            except Exception:
                pass
                
    # Phase 4: Exact LP refinement on best geometry
    centers = best_v[:2*N].reshape(N, 2)
    lp_radii, lp_sum = refine_with_lp(centers)
    
    if lp_radii is not None and lp_sum > best_sum:
        best_radii = lp_radii
        best_sum = lp_sum
    else:
        best_radii = best_v[2*N:].copy()
        
    # Phase 5: Strict numerical repair
    best_radii = repair_solution(centers, best_radii)
    final_sum = float(np.sum(best_radii))
    
    return centers, best_radii, final_sum
