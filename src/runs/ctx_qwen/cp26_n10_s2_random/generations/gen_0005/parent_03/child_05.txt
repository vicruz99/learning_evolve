

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we leverage a two-phase optimization strategy that separates center placement from radius calculation. This avoids the high-dimensional non-convexity of jointly optimizing (x, y, r) for all circles.

1. **LP-based Radius Calculation**: For any fixed set of centers, maximizing the sum of radii subject to non-overlap and boundary constraints is a Linear Programming (LP) problem. Given N=26, solving this LP is extremely fast. This gives us the exact maximum sum of radii for any center configuration.
2. **Derivative-Free Center Optimization**: We optimize the 52 center coordinates using `scipy.optimize.minimize` with the `Powell` method. The objective function is the negative of the LP-computed sum of radii. `Powell` is robust for non-smooth, multi-modal objectives like this.
3. **Diverse Initialization**: We generate multiple starting configurations:
   - Force-directed repulsive points to naturally spread circles and utilize corners/edges.
   - Hexagonal lattice patterns with varying densities.
   - Corner-biased layouts to encourage large boundary circles.
4. **Perturbation Search**: After an initial multi-start optimization, we perform a scheduled perturbation loop on the best configuration to escape local minima and refine the topology.
5. **Strict Repair**: Finally, we re-solve the LP for the optimal centers and run a deterministic repair pass to guarantee strict feasibility within the validator's numerical tolerance (`1e-12`).

This approach efficiently explores the configuration space while guaranteeing mathematically optimal radii for each evaluated layout, consistently pushing past typical gradient-based local minima.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_lp_solution(centers):
    """
    Given fixed centers, solves the LP to maximize sum of radii
    subject to boundary and pairwise non-overlap constraints.
    """
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    
    c_obj = -np.ones(n)
    
    # Preallocate constraints: 4 per circle (boundaries) + N*(N-1)/2 (pairwise)
    n_ineq = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_ineq, n))
    b_ub = np.zeros(n_ineq)
    
    idx = 0
    for i in range(n):
        row_i = np.zeros(n)
        row_i[i] = 1.0
        
        # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        A_ub[idx, :] = row_i; b_ub[idx] = x[i]; idx += 1
        A_ub[idx, :] = row_i; b_ub[idx] = 1.0 - x[i]; idx += 1
        A_ub[idx, :] = row_i; b_ub[idx] = y[i]; idx += 1
        A_ub[idx, :] = row_i; b_ub[idx] = 1.0 - y[i]; idx += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    row_pair = np.zeros(n)
    for i in range(n):
        for j in range(i + 1, n):
            row_pair[:] = 0.0
            row_pair[i] = 1.0
            row_pair[j] = 1.0
            A_ub[idx, :] = row_pair
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.fun != -np.inf:
            return -res.fun, res.x
    except Exception:
        pass
        
    # Fallback analytical relaxation if LP fails
    radii = np.full(n, 0.05)
    for _ in range(50):
        for i in range(n):
            r_max = min(x[i], 1.0-x[i], y[i], 1.0-y[i])
            for j in range(n):
                if i != j:
                    d = np.hypot(x[i]-x[j], y[i]-y[j])
                    r_max = min(r_max, d - radii[j])
            radii[i] = max(0.0, r_max)
    return np.sum(radii), radii

def objective_lp(centers_flat):
    """Objective for center optimization: minimize negative sum of radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 0.0, 1.0)
    s, _ = get_lp_solution(centers)
    return -s

def repulsive_init(rng, n, iters=400):
    """Generates well-spaced initial centers using simple repulsion forces."""
    c = rng.uniform(0.2, 0.8, (n, 2))
    for _ in range(iters):
        forces = np.zeros_like(c)
        for i in range(n):
            for j in range(i+1, n):
                d_vec = c[i] - c[j]
                d = np.linalg.norm(d_vec)
                if d < 0.25 and d > 1e-6:
                    f = (0.25 - d) * 0.1 / d
                    forces[i] += d_vec * f
                    forces[j] -= d_vec * f
        c += forces * 0.1
        c = np.clip(c, 0.01, 0.99)
    return c

def run_packing() -> tuple:
    rng = np.random.default_rng(42)
    
    candidates = []
    
    # 1. Repulsive inits
    for _ in range(4):
        candidates.append(repulsive_init(rng, N))
        
    # 2. Hex grids with varying spacing
    for r0 in [0.095, 0.105]:
        pts = []
        y = r0
        row = 0
        while len(pts) < N:
            x = r0 if row % 2 == 0 else 2 * r0
            while x + r0 <= 1.0 and len(pts) < N:
                pts.append([x, y])
                x += 2 * r0
            y += np.sqrt(3) * r0
            row += 1
        candidates.append(np.array(pts[:N]))
        
    # 3. Corner biased layout
    corners = np.array([[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]])
    rem = rng.uniform(0.25, 0.75, (N-4, 2))
    candidates.append(np.vstack([corners, rem]))
    
    best_sum_radii = 0.0
    best_final_centers = None
    
    # Phase 1: Multi-start optimization
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    for cand in candidates:
        cand = np.clip(cand, 0.001, 0.999)
        cand += rng.normal(0, 0.002, cand.shape)
        cand = np.clip(cand, 0.001, 0.999)
        
        try:
            res = minimize(objective_lp, cand.flatten(), method='Powell', 
                           bounds=bounds_centers,
                           options={'maxiter': 4000, 'ftol': 1e-12})
            curr_sum = -res.fun
            if curr_sum > best_sum_radii:
                best_sum_radii = curr_sum
                best_final_centers = res.x.reshape(N, 2)
        except Exception:
            continue
            
    # Phase 2: Perturbation search to escape local minima
    if best_final_centers is not None:
        current_c = best_final_centers.copy()
        for step in range(30):
            noise = 0.004 * (0.9 ** (step // 5))
            c_pert = current_c + rng.normal(0, noise, current_c.shape)
            c_pert = np.clip(c_pert, 0.001, 0.999)
            
            try:
                res = minimize(objective_lp, c_pert.flatten(), method='Powell',
                               bounds=bounds_centers,
                               options={'maxiter': 3000, 'ftol': 1e-12})
                if -res.fun > best_sum_radii:
                    best_sum_radii = -res.fun
                    current_c = res.x.reshape(N, 2)
            except Exception:
                continue
        best_final_centers = current_c
        
    # Final exact LP to get mathematically optimal radii for the best centers
    _, optimal_radii = get_lp_solution(best_final_centers)
    radii = optimal_radii.copy()
    centers = best_final_centers.copy()
    
    # Strict numerical repair to guarantee validator passes
    for _ in range(20):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-9:
                radii[i] = mr
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
