

<strategy>
The core improvement strategy shifts from simultaneous non-linear optimization of centers and radii to a bilevel approach:
1. **Inner Problem (LP)**: For fixed centers, maximizing the sum of radii is a Linear Programming problem. Constraints are $r_i + r_j \le \|c_i - c_j\|$ and $r_i \le \text{dist}(c_i, \text{boundary})$. This is solved efficiently using `scipy.optimize.linprog` with the `highs` method.
2. **Outer Problem (Derivative-Free Optimization)**: We optimize the centers to maximize the LP's optimal value. Since the LP value is non-smooth (piecewise linear/concave), we use `scipy.optimize.minimize` with the `Powell` method, which handles bounds and derivative-free landscapes well.
3. **Diverse Initialization**: We generate multiple hexagonal lattices with varying row patterns, random spread, and force-repulsion configurations to cover the search space.
4. **Refinement**: After finding the best configuration, we perform a local perturbation search to escape local minima, followed by a final LP solve and a tiny numerical shrink to guarantee validation passes within tolerance.
This approach separates the geometric placement from the radius allocation, allowing the LP to find the exact optimal radii for any placement, while Powell efficiently navigates the center configuration space.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def build_lp_matrix(n):
    """Precomputes the constant constraint matrix for the LP inner problem."""
    m = n * (n - 1) // 2
    A = np.zeros((m, n))
    idx = 0
    pair_indices = []
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            pair_indices.append((i, j))
            idx += 1
    return A, pair_indices

A_LP, pair_indices = build_lp_matrix(N)
c_obj = -np.ones(N)

def solve_lp_fast(centers):
    """Solves the LP to maximize sum of radii for fixed centers."""
    # Distance to boundaries: r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Extract distances for constraint RHS
    b = np.array([dists[i, j] for i, j in pair_indices])
    bounds = [(0, u) for u in ub]
    
    # Maximize sum(r_i) => Minimize -sum(r_i)
    # Constraints: r_i + r_j <= dists[i,j]
    res = linprog(c_obj, A_ub=A_LP, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def objective(centers_flat):
    """Objective for center optimization: negative of the optimal sum of radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    s, _ = solve_lp_fast(centers)
    return -s

def generate_hex_init(row_counts, r0):
    """Generates a hexagonal lattice configuration."""
    centers = []
    y = r0
    for r_idx, cnt in enumerate(row_counts):
        shift = r0 if r_idx % 2 == 1 else 0.0
        x = r0 + shift
        for _ in range(cnt):
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(centers[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles in a unit square maximizing sum of radii."""
    np.random.seed(42)
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = []
    patterns = [
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5],
        [4, 5, 6, 5, 6], [6, 4, 5, 6, 5], [5, 5, 6, 5, 5],
        [5, 5, 5, 6, 5], [6, 6, 4, 5, 5], [5, 7, 5, 4, 5],
        [5, 5, 5, 5, 5, 1], [1, 5, 5, 5, 5, 5], [3, 5, 5, 5, 5, 3]
    ]
    for pat in patterns:
        if sum(pat) < N: continue
        for r0 in [0.08, 0.09, 0.095, 0.10, 0.105]:
            c = generate_hex_init(pat, r0)
            c += np.random.normal(0, 0.005, c.shape)
            c = np.clip(c, 0.02, 0.98)
            inits.append(c)
            
    for _ in range(15):
        c = np.random.rand(N, 2) * 0.8 + 0.1
        inits.append(c)
        
    for _ in range(8):
        c = np.random.rand(N, 2) * 0.6 + 0.2
        for _ in range(400):
            for i in range(N):
                for j in range(i+1, N):
                    d = c[i] - c[j]
                    dist = np.linalg.norm(d)
                    if dist < 0.15 and dist > 1e-5:
                        push = (0.15 - dist) * 0.5
                        c[i] += d / dist * push
                        c[j] -= d / dist * push
        c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    # Phase 1: Multi-start optimization on centers
    for c0 in inits:
        try:
            res = minimize(objective, c0.flatten(), method='Powell', bounds=bounds_centers,
                           options={'maxiter': 1500, 'ftol': 1e-12, 'xtol': 1e-12})
            c_opt = res.x.reshape(N, 2)
            c_opt = np.clip(c_opt, 0.001, 0.999)
            s, r = solve_lp_fast(c_opt)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt
                best_radii = r
        except Exception:
            pass
            
    # Phase 2: Local perturbation search to escape local minima
    if best_centers is not None:
        for step in range(30):
            noise = 0.003 * (0.9 ** (step // 5))
            c_pert = best_centers + np.random.normal(0, noise, best_centers.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            try:
                res = minimize(objective, c_pert.flatten(), method='Powell', bounds=bounds_centers,
                               options={'maxiter': 800, 'ftol': 1e-12})
                c_opt = res.x.reshape(N, 2)
                c_opt = np.clip(c_opt, 0.001, 0.999)
                s, r = solve_lp_fast(c_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = c_opt
                    best_radii = r
            except Exception:
                pass
                
    # Phase 3: Final exact LP solve and numerical repair
    if best_centers is not None:
        best_sum, best_radii = solve_lp_fast(best_centers)
        # Tiny shrink to guarantee strict validity against 1e-12 tolerance
        best_radii *= 0.999999
    else:
        best_centers = np.random.rand(N, 2) * 0.8 + 0.1
        best_sum, best_radii = solve_lp_fast(best_centers)
        
    return best_centers, best_radii, float(best_sum)
```
