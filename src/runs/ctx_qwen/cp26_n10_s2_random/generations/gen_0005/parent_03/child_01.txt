import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(v):
    """Objective: minimize negative sum of radii to maximize total radius."""
    return -np.sum(v[2::3])

def constraints(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints: (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (dx**2 + dy**2 - dr**2)[mask]])
    return c

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-6)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, u) for u in ub], method='highs')
    if res.success:
        return res.x, np.sum(res.x)
    return np.maximum(ub * 0.5, 0.0), 0.0

def force_init(n, seed, iters=400):
    """Generates initial configuration using force-directed repulsion."""
    rng = np.random.default_rng(seed)
    c = rng.uniform(0.1, 0.9, (n, 2))
    r = np.full(n, 0.02)
    corners = [[0.05, 0.05], [0.95, 0.05], [0.05, 0.95], [0.95, 0.95]]
    for i, cor in enumerate(corners):
        c[i] = cor
        
    for _ in range(iters):
        r *= 1.002
        forces = np.zeros_like(c)
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = np.hypot(dx, dy)
                min_d = r[i] + r[j]
                if d < min_d and d > 1e-6:
                    f = (min_d - d) * 10.0 / d
                    fx, fy = dx * f, dy * f
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
            cx, cy = c[i]
            rad = r[i]
            if cx < rad: forces[i, 0] += (rad - cx) * 20.0
            if cx > 1 - rad: forces[i, 0] -= (cx - (1 - rad)) * 20.0
            if cy < rad: forces[i, 1] += (rad - cy) * 20.0
            if cy > 1 - rad: forces[i, 1] -= (cy - (1 - rad)) * 20.0
            
        c += forces * 0.001
        c = np.clip(c, 0.001, 0.999)
        
    # Adjust radii to be strictly feasible
    for _ in range(30):
        min_gap = np.inf
        for i in range(n):
            db = min(c[i,0], 1.0-c[i,0], c[i,1], 1.0-c[i,1])
            min_gap = min(min_gap, db - r[i])
            for j in range(i+1, n):
                d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
                min_gap = min(min_gap, d - r[i] - r[j])
        if min_gap < -1e-6:
            r -= min(0.002, abs(min_gap)/2.0)
        else:
            break
    r = np.maximum(r, 0.01)
    return c, r

def hex_init(row_counts, r0, seed):
    """Generates initial configuration from a perturbed hexagonal lattice."""
    rng = np.random.default_rng(seed)
    centers = []
    y = r0
    for row_idx, count in enumerate(row_counts):
        offset = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + offset
        for _ in range(count):
            centers.append([x + rng.normal(0, 0.003), y + rng.normal(0, 0.003)])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    centers = np.array(centers[:N])
    centers = np.clip(centers, 0.05, 0.95)
    return centers, np.full(N, r0 * 0.9)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles in a unit square."""
    rng = np.random.default_rng(42)
    bnds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -np.inf
    configs = []
    
    # Phase 1: Force repulsion initializations
    for s in range(5):
        c, r = force_init(N, seed=100+s)
        v = np.zeros(3*N)
        v[0::3] = c[:, 0]; v[1::3] = c[:, 1]; v[2::3] = r
        configs.append(v)
        
    # Phase 2: Hexagonal lattice patterns
    patterns = [[5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [5, 6, 5, 5, 5], [5, 5, 6, 5, 5], 
                [5, 5, 5, 6, 5], [6, 5, 6, 4, 5], [5, 4, 6, 5, 6], [6, 6, 4, 5, 5]]
    for pat in patterns:
        for r0 in [0.095, 0.10, 0.105]:
            c, r = hex_init(pat, r0, seed=len(configs))
            # Compute feasible initial radii
            for i in range(N):
                db = min(c[i,0], 1.0-c[i,0], c[i,1], 1.0-c[i,1])
                mask = np.ones(N, dtype=bool); mask[i]=False
                dn = np.min(np.hypot(c[i,0]-c[mask,0], c[i,1]-c[mask,1]))
                r[i] = min(db, dn/2.0) * 0.95
            v = np.zeros(3*N)
            v[0::3] = c[:, 0]; v[1::3] = c[:, 1]; v[2::3] = r
            configs.append(v)

    # Phase 3: Random dense starts
    for _ in range(8):
        c = rng.uniform(0.15, 0.85, (N, 2))
        v = np.zeros(3*N)
        v[0::3] = c[:, 0]; v[1::3] = c[:, 1]; v[2::3] = 0.045
        configs.append(v)
        
    # Phase 4: Multi-start SLSQP Optimization
    for v0 in configs:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13})
            c_val = constraints(res.x)
            if np.min(c_val) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = configs[0]
        
    # Phase 5: LP Refinement on best centers
    centers_best = np.column_stack((best_v[0::3], best_v[1::3]))
    lp_r, lp_sum = lp_radii(centers_best)
    if lp_sum > best_sum:
        best_sum = lp_sum
        best_v[2::3] = lp_r
        
    # Phase 6: Perturbation search to escape local minima
    for step in range(60):
        noise = 0.004 * (0.93 ** step)
        v_trial = best_v + rng.normal(0, noise, 3*N)
        v_trial[0::3] = np.clip(v_trial[0::3], 0.02, 0.98)
        v_trial[1::3] = np.clip(v_trial[1::3], 0.02, 0.98)
        v_trial[2::3] = np.clip(v_trial[2::3], 0.01, 0.45)
        try:
            res = minimize(objective, v_trial, method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-13})
            c_val = constraints(res.x)
            if np.min(c_val) >= -1e-7:
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_v = res.x.copy()
                    # Quick LP refine after improvement
                    c_tmp = np.column_stack((best_v[0::3], best_v[1::3]))
                    lp_r_tmp, lp_s_tmp = lp_radii(c_tmp)
                    if lp_s_tmp > best_sum:
                        best_sum = lp_s_tmp
                        best_v[2::3] = lp_r_tmp
        except Exception:
            pass
            
    # Phase 7: Final SLSQP polish
    try:
        res = minimize(objective, best_v, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': 10000, 'ftol': 1e-13})
        if -res.fun > best_sum:
            best_v = res.x
    except:
        pass
        
    # Phase 8: Strict numerical repair to guarantee validation passes
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3].copy()
    
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-10:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
            if radii[i] > mr - 1e-10:
                radii[i] = mr
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
