import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(v):
    """Objective: minimize negative sum of radii to maximize total radius."""
    return -np.sum(v[2::3])

def constraints(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints: (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (dx**2 + dy**2 - dr**2)[mask]])
    return c

def get_bounds():
    """Returns variable bounds for x, y, r for each circle."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def compute_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii exactly."""
    n = centers.shape[0]
    # Upper bounds from boundaries
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 0.0)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # LP constraints: r_i + r_j <= dists[i,j]
    constraints_list = []
    rhs_list = []
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints_list.append(row)
            rhs_list.append(dists[i, j])
            
    A_ub = np.array(constraints_list)
    b_ub = np.array(rhs_list)
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, u) for u in ub], method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(n), 0.0

def force_init(rng):
    """Initializes centers using force-directed repulsion to spread them out and utilize corners."""
    c = rng.uniform(0.15, 0.85, (N, 2))
    # Force corners to be utilized
    c[0] = [0.12, 0.12]
    c[1] = [0.88, 0.12]
    c[2] = [0.12, 0.88]
    c[3] = [0.88, 0.88]
    
    for _ in range(600):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                d_vec = c[i] - c[j]
                d = np.linalg.norm(d_vec)
                if d < 0.22 and d > 1e-6:
                    f = (0.22 - d) / d
                    forces[i] += d_vec * f
                    forces[j] -= d_vec * f
        c += forces * 0.015
        c = np.clip(c, 0.05, 0.95)
    return c

def hex_init(pat, r0, rng):
    """Generates initial configuration from a perturbed hexagonal lattice."""
    centers = []
    y = r0
    for row_idx, count in enumerate(pat):
        offset = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + offset
        for _ in range(count):
            centers.append([x + rng.normal(0, 0.004), y + rng.normal(0, 0.004)])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    centers = np.array(centers[:N])
    return np.clip(centers, 0.05, 0.95)

def repair_solution(centers, radii):
    """Iteratively shrinks radii to strictly resolve overlaps and clamp to boundaries."""
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
            if radii[i] > mr + 1e-9:
                radii[i] = mr
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles in a unit square."""
    rng = np.random.default_rng(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -np.inf
    inits = []
    
    # Phase 1: Generate diverse initial configurations
    # Hexagonal patterns with varying row structures
    pats = [[5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [6,6,6,4,4], 
            [4,5,6,5,6], [5,7,5,5,4], [6,5,5,6,4], [5,5,6,5,5]]
    for pat in pats:
        for r0 in [0.09, 0.095, 0.10, 0.105, 0.11]:
            c = hex_init(pat, r0, rng)
            v = np.zeros(3 * N)
            v[0::3] = c[:, 0]
            v[1::3] = c[:, 1]
            r, _ = compute_lp_radii(c)
            v[2::3] = r * 0.95  # Start slightly feasible
            inits.append(v)
            
    # Force-directed starts
    for _ in range(20):
        c = force_init(rng)
        v = np.zeros(3 * N)
        v[0::3] = c[:, 0]
        v[1::3] = c[:, 1]
        r, _ = compute_lp_radii(c)
        v[2::3] = r * 0.95
        inits.append(v)
        
    # Random starts
    for _ in range(15):
        c = rng.uniform(0.1, 0.9, (N, 2))
        v = np.zeros(3 * N)
        v[0::3] = c[:, 0]
        v[1::3] = c[:, 1]
        r, _ = compute_lp_radii(c)
        v[2::3] = r * 0.95
        inits.append(v)
        
    # Phase 2: Multi-start constrained optimization
    for v0 in inits:
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
            if -res.fun > best_sum:
                if np.min(constraints(res.x)) >= -1e-7:
                    best_sum = -res.fun
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Local perturbation search to escape local minima
    if best_v is not None:
        for step in range(150):
            noise = 0.005 * (0.96 ** (step // 5))
            v_pert = best_v + rng.normal(0, noise, best_v.shape)
            v_pert[0::3] = np.clip(v_pert[0::3], 0.02, 0.98)
            v_pert[1::3] = np.clip(v_pert[1::3], 0.02, 0.98)
            # Shrink radii to allow centers to move freely
            v_pert[2::3] *= 0.96
            v_pert[2::3] = np.maximum(v_pert[2::3], 0.01)
            
            try:
                res = minimize(objective, v_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 6000, 'ftol': 1e-14})
                if -res.fun > best_sum:
                    if np.min(constraints(res.x)) >= -1e-7:
                        best_sum = -res.fun
                        best_v = res.x.copy()
            except Exception:
                pass
                
    # Phase 4: Exact LP refinement on best centers
    centers = best_v[:2*N].reshape(N, 2)
    radii, _ = compute_lp_radii(centers)
    
    # Strict numerical repair to guarantee validation passes
    radii = repair_solution(centers, radii)
    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
