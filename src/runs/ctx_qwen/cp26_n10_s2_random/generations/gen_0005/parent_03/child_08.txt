import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 0.0)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx_i, idx_j = np.triu_indices(n, 1)
    A_ub = np.zeros((len(idx_i), n))
    A_ub[np.arange(len(idx_i)), idx_i] = 1.0
    A_ub[np.arange(len(idx_i)), idx_j] = 1.0
    b_ub = dists[idx_i, idx_j]
    
    c_obj = -np.ones(n)
    bounds = [(0.0, ub[i]) for i in range(n)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return np.zeros(n)

def objective_centers(centers_flat):
    """Objective for center optimization: maximize sum of LP-optimal radii."""
    centers = centers_flat.reshape(N, 2)
    radii = solve_lp_radii(centers)
    return -np.sum(radii)

def objective_joint(v):
    """Objective for joint optimization: maximize sum of radii."""
    return -np.sum(v[2::3])

def get_constraints_full(v):
    """Boundary and non-overlap constraints for joint optimization."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (dx**2 + dy**2 - dr**2)[mask]])
    return c

def repulsion_init(rng, n):
    """Generate well-spread initial centers using force-directed repulsion."""
    c = rng.uniform(0.15, 0.85, (n, 2))
    c[0] = [0.15, 0.15]
    c[1] = [0.85, 0.15]
    c[2] = [0.15, 0.85]
    c[3] = [0.85, 0.85]
    
    for _ in range(800):
        forces = np.zeros_like(c)
        for i in range(n):
            for j in range(i + 1, n):
                d_vec = c[i] - c[j]
                dist = np.linalg.norm(d_vec)
                if dist < 0.18 and dist > 1e-5:
                    push = (0.18 - dist) * 5.0 / (dist + 1e-5)
                    f = (d_vec / dist) * push
                    forces[i] += f
                    forces[j] -= f
        c += forces * 0.008
        c = np.clip(c, 0.05, 0.95)
    return c

def hex_init(pattern, r0, rng):
    """Generate initial centers from a perturbed hexagonal lattice."""
    centers = []
    y = r0
    for row_idx, count in enumerate(pattern):
        offset = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + offset
        for _ in range(count):
            centers.append([x + rng.normal(0, 0.002), y + rng.normal(0, 0.002)])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    centers = np.array(centers[:N])
    return np.clip(centers, 0.05, 0.95)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize sum of radii."""
    rng = np.random.default_rng(42)
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = []
    
    # 1. Repulsion-based inits
    for _ in range(5):
        inits.append(repulsion_init(rng, N).flatten())
        
    # 2. Hexagonal lattice inits
    patterns = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], [5, 5, 6, 5, 5]]
    for pat in patterns:
        for r0 in [0.09, 0.10, 0.105]:
            inits.append(hex_init(pat, r0, rng).flatten())
            
    # 3. Random inits
    for _ in range(5):
        c = rng.uniform(0.1, 0.9, (N, 2))
        inits.append(c.flatten())
        
    # Phase 1: Optimize centers using LP for radii
    for c0 in inits:
        try:
            res = minimize(objective_centers, c0, method='Powell', bounds=bounds_centers,
                           options={'maxiter': 3000, 'ftol': 1e-12, 'xtol': 1e-12})
            c_opt = res.x.reshape(N, 2)
            r_opt = solve_lp_radii(c_opt)
            s = np.sum(r_opt)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Phase 2: Joint SLSQP refinement
    if best_centers is not None:
        v0 = np.zeros(3 * N)
        v0[0::3] = best_centers[:, 0]
        v0[1::3] = best_centers[:, 1]
        v0[2::3] = best_radii
        
        cons = {'type': 'ineq', 'fun': get_constraints_full}
        bounds_full = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        
        try:
            res = minimize(objective_joint, v0, method='SLSQP', 
                           bounds=bounds_full, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-13})
            if -res.fun > best_sum:
                best_centers = res.x[:2*N].reshape(N, 2)
                best_radii = res.x[2*N:].copy()
                best_sum = -res.fun
        except Exception:
            pass
            
    # Phase 3: Perturbation search to escape local minima
    for _ in range(40):
        c_pert = best_centers + rng.normal(0, 0.002, best_centers.shape)
        c_pert = np.clip(c_pert, 0.05, 0.95)
        
        try:
            res = minimize(objective_centers, c_pert.flatten(), method='Powell', bounds=bounds_centers,
                           options={'maxiter': 1500, 'ftol': 1e-12})
            c_new = res.x.reshape(N, 2)
            r_new = solve_lp_radii(c_new)
            s_new = np.sum(r_new)
            if s_new > best_sum:
                best_sum = s_new
                best_centers = c_new
                best_radii = r_new
        except Exception:
            pass
            
    # Phase 4: Strict numerical repair
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(best_centers[i,0] - best_centers[j,0], best_centers[i,1] - best_centers[j,1])
                if d < best_radii[i] + best_radii[j] - 1e-9:
                    shrink = (best_radii[i] + best_radii[j] - d) / 2.0 + 1e-9
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(best_centers[i,0], 1.0 - best_centers[i,0], best_centers[i,1], 1.0 - best_centers[i,1])
            if best_radii[i] > mr - 1e-9:
                best_radii[i] = mr
                changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
