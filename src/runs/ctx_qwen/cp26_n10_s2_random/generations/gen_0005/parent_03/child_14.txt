import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_lp_radii(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii subject to:
    r_i + r_j <= dist(i, j) for all i < j
    0 <= r_i <= dist_to_boundary(i)
    """
    n = centers.shape[0]
    # Boundary limits
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # LP: Max sum(r) <=> Min -sum(r)
    # s.t. A_ub @ r <= b_ub
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0.0, u) for u in ub], method='highs')
    
    if res.success:
        return res.x, -res.fun
    # Fallback if LP fails (rare)
    return ub * 0.1, np.sum(ub * 0.1)

def objective_centers(centers_flat):
    """Objective for center optimization: minimize negative sum of optimal radii."""
    centers = centers_flat.reshape(N, 2)
    # Keep centers strictly inside to maintain positive bounds for LP
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    _, total_r = compute_lp_radii(centers)
    return -total_r

def force_relax(centers, iterations=800, init_step=0.02):
    """
    Force-directed relaxation to spread circles and allow radii growth.
    """
    n = centers.shape[0]
    radii, _ = compute_lp_radii(centers)
    step = init_step
    
    for it in range(iterations):
        forces = np.zeros_like(centers)
        for i in range(n):
            xi, yi = centers[i]
            ri = radii[i]
            
            # Boundary repulsion
            if xi - ri < 1e-5: forces[i, 0] += (ri - xi) * 10.0
            if 1.0 - xi - ri < 1e-5: forces[i, 0] -= (1.0 - xi - ri) * 10.0
            if yi - ri < 1e-5: forces[i, 1] += (ri - yi) * 10.0
            if 1.0 - yi - ri < 1e-5: forces[i, 1] -= (1.0 - yi - ri) * 10.0
            
            # Pairwise repulsion
            for j in range(i + 1, n):
                dx = xi - centers[j, 0]
                dy = yi - centers[j, 1]
                d = np.hypot(dx, dy)
                min_d = ri + radii[j]
                if d < min_d + 1e-6:
                    overlap = max(0.0, min_d - d)
                    f = overlap * 5.0 / max(d, 1e-9)
                    fx, fy = dx * f, dy * f
                    forces[i] += (fx, fy)
                    forces[j] -= (fx, fy)
                    
        centers += forces * step
        centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
        
        # Periodically update radii and grow slightly to expand packing
        if it % 50 == 0:
            radii, _ = compute_lp_radii(centers)
            radii *= 1.0005 
            
        step *= 0.999
        
    radii, _ = compute_lp_radii(centers)
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_centers = None
    best_sum = -1.0
    
    inits = []
    
    # 1. Hexagonal lattice patterns
    for r0 in [0.08, 0.09, 0.10, 0.105]:
        pts = []
        y = r0
        row = 0
        while len(pts) < N:
            x = r0 if row % 2 == 0 else 2 * r0
            while x + r0 <= 1.0 and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        inits.append(np.array(pts[:N]))
        
    # 2. Random starts with initial force repulsion to spread points
    for _ in range(12):
        c = np.random.rand(N, 2) * 0.8 + 0.1
        for _ in range(300):
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.linalg.norm(c[i] - c[j])
                    if d < 0.25 and d > 1e-6:
                        push = (0.25 - d) * 0.5
                        c[i] += (c[i] - c[j]) / d * push
                        c[j] -= (c[i] - c[j]) / d * push
        inits.append(np.clip(c, 0.05, 0.95))
        
    # 3. Multi-start derivative-free optimization of centers
    for init_c in inits:
        try:
            res = minimize(objective_centers, init_c.flatten(), method='Powell',
                           options={'maxiter': 600, 'ftol': 1e-13, 'xtol': 1e-13})
            curr_centers = res.x.reshape(N, 2)
            _, curr_sum = compute_lp_radii(curr_centers)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = curr_centers.copy()
        except Exception:
            pass
            
    # 4. Force relaxation refinement to escape local jams
    if best_centers is not None:
        best_centers, radii = force_relax(best_centers, iterations=1000, init_step=0.015)
        final_radii, final_sum = compute_lp_radii(best_centers)
        
        # Negligible scaling to strictly satisfy 1e-12 validator tolerance against float errors
        final_radii *= 0.9999999
        final_sum = float(np.sum(final_radii))
        
        return best_centers, final_radii, final_sum
        
    # Fallback (should not be reached)
    return np.random.rand(N, 2), np.full(N, 0.05), 1.3
