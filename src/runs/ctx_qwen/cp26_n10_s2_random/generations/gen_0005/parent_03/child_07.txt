import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective_full(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def compute_constraints_full(v):
    """Computes boundary and pairwise non-overlap constraints (must be >= 0)."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints: (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c = np.concatenate([c, (dx**2 + dy**2 - dr**2)[mask]])
    return c

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-6)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    A_ub = np.zeros((n*(n-1)//2, n))
    b_ub = np.zeros(n*(n-1)//2)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    c_obj = -np.ones(n)
    bounds_lp = [(0, u) for u in ub]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def objective_lp(centers_flat):
    """Objective for center optimization: returns negative max sum of radii via LP."""
    centers = centers_flat.reshape(-1, 2)
    if np.any(centers < 0.001) or np.any(centers > 0.999):
        return 1e6
    _, sum_r = solve_lp_radii(centers)
    return -sum_r

def get_hex_centers(row_counts, r0):
    """Generates initial configuration from a hexagonal lattice."""
    centers = []
    y = r0
    for row_idx, count in enumerate(row_counts):
        offset = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + offset
        for _ in range(count):
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    return np.array(centers[:N])

def repair_solution(centers, radii):
    """Iteratively shrinks radii to resolve overlaps and clamp to boundaries."""
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            mr = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
            if radii[i] > mr - 1e-9:
                radii[i] = mr
                changed = True
        if not changed:
            break
    radii = np.maximum(radii, 0.0)
    return radii

def run_packing() -> tuple:

    np.random.seed(42)
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    row_patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], 
        [4, 6, 6, 6, 4], [5, 7, 5, 6, 3], [6, 6, 4, 6, 4], [5, 5, 6, 5, 5]
    ]
    
    # Phase 1: LP-based center optimization with Powell
    for pat in row_patterns:
        for r0 in [0.09, 0.095, 0.10, 0.105]:
            c0 = get_hex_centers(pat, r0)
            c0 += np.random.normal(0, 0.002, c0.shape)
            c0 = np.clip(c0, 0.05, 0.95)
            
            try:
                res = minimize(objective_lp, c0.flatten(), method='Powell',
                               options={'maxiter': 1500, 'ftol': 1e-11})
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = res.x.reshape(N, 2)
            except Exception:
                pass
                
    # Random force-repulsion starts
    for _ in range(5):
        c_rand = np.random.uniform(0.1, 0.9, (N, 2))
        for _ in range(500):
            forces = np.zeros_like(c_rand)
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c_rand[i] - c_rand[j]
                    d = np.linalg.norm(d_vec)
                    if d < 0.15 and d > 1e-5:
                        f = (0.15 - d) * 0.1
                        forces[i] += f * d_vec / d
                        forces[j] -= f * d_vec / d
            c_rand += forces
            c_rand = np.clip(c_rand, 0.05, 0.95)
            
        try:
            res = minimize(objective_lp, c_rand.flatten(), method='Powell',

    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

    v0 = np.zeros(3 * N)

    v0[1::3] = best_centers[:, 1]
    v0[2::3] = best_radii
    
    try:

            best_sum = -res.fun
            best_centers = res.x[:2*N].reshape(N, 2)

    except Exception:
        pass

        noise = 0.003 * (0.9 ** (step // 5))
        v_trial = np.concatenate([best_centers.flatten(), best_radii])
        v_trial += rng.normal(0, noise, v_trial.shape)
        v_trial[0::3] = np.clip(v_trial[0::3], 0.02, 0.98)
        v_trial[1::3] = np.clip(v_trial[1::3], 0.02, 0.98)
        v_trial[2::3] *= 0.95
        v_trial[2::3] = np.clip(v_trial[2::3], 0.01, 0.45)
        
        try:
            res = minimize(objective_full, v_trial, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-13})
            if -res.fun > best_sum:
                c_vals = compute_constraints_full(res.x)
                if np.min(c_vals) >= -1e-7:
                    best_sum = -res.fun
                    best_centers = res.x[:2*N].reshape(N, 2)
                    best_radii = res.x[2*N:].copy()
        except Exception:
            pass
            
    # Phase 4: Final LP refinement (LP is exact for fixed centers)
    lp_rad, lp_s = solve_lp_radii(best_centers)
    if lp_s > best_sum:
        best_radii = lp_rad
        best_sum = lp_s
        
    # Strict repair to guarantee validation passes
    best_radii = repair_solution(best_centers, best_radii)
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
